def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

def _subgradient_step(h, step_size):
    """One step of subgradient method"""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    r_star = np.argmax(scores)
    G = _grad_matrix(h, dx)
    grad = G[r_star, :]  # gradient of the active score

    grad_proj = grad - np.mean(grad)
    # Take step
    h_new = h - step_size * grad_proj
    # Project onto [0,1]^n ∩ {sum = n/2}

def _slsqp_epigraph(h0, target, dx, top_k, max_iter=200, deadline=None):
    n = len(h0)
    n_lags = 2 * n - 1
    s0 = _scores(h0)
    current_max = float(np.max(s0))
    
    # Select top-k lags
    top_idx = np.argsort(s0)[::-1][:min(top_k, n_lags)]
    
    # Objective: minimize t
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Inequality constraints: t - score_r >= 0 for r in top_idx
    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        s = _scores(h)
        return t - s[top_idx]
    
    def ineq_jac(x):
        h = x[:n]
        G = _grad_matrix(h, dx)
        J = np.zeros((len(top_idx), n + 1))
        J[:, :n] = -G[top_idx]
        J[:, n] = 1.0
        return J
    
    # Equality constraint: sum(h) = target
    def eq_fun(x):
        return np.array([np.sum(x[:n]) - target])
    
    def eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, :n] = 1.0
        return J
    
    # Initial point
    t0_val = current_max + 1e-3
    x0 = np.concatenate([h0.copy(), [t0_val]])
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, current_max + 0.05)]
    
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                      constraints=constraints, method='SLSQP',
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success or res.x is not None:
            h_sol = _project(res.x[:n], target)
            if _exact_c5(h_sol) < current_max - 1e-10:
                return h_sol
    except Exception:
        pass
    return None

def _line_search_quad(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Bounds on alpha for feasibility
    pos = d > 1e-14
    neg = d < -1e-14
    a_lo = 0.0
    a_hi = radius
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, -float(np.max(h[neg] / d[neg])))
    # Actually since we want d to be a valid direction, a_lo should be 0
    # and the step should be forward
    a_lo = 0.0
    a_hi = max(a_hi, 1e-10)
    
    if a_hi - a_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid search
    n_grid = 80
    grid = np.linspace(a_lo, a_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None]**2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_a = grid[best_i]
    best_val = obj_grid[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = a_lo, a_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:

    opt_a = (a+b)/2.0

    n_lags = 2 * n - 1

    c = np.zeros(n + 1)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += (-h[d:])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += (-h[:m])
    G *= dx
    return G

def _lp_dir(h, scores, G, radius):
    n = len(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=-scores, A_eq=A_eq, b_eq=np.zeros(1),
                     bounds=bounds, method='highs')
        if res.success and res.x[n] < float(np.max(scores)) - 1e-13:
            return res.x[:n]
    except Exception:
        pass
    return None
