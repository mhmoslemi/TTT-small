def _project(v, target):
    """Euclidean projection onto {h: 0<=h_i<=1, sum(h)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

    n = len(h)
    dx = 2.0 / n
    d = r - (n - 1)  # lag
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0:
            return g
        g[d:] += dx
        g[d:] -= dx * h[:m]
        g[:m] -= dx * h[d:]
    else:

        m = n - g_abs
        if m <= 0:
            return g
        g[:m] += dx
        g[:m] -= dx * h[g_abs:]
        g[g_abs:] -= dx * h[:m]

        r_indices = range(2 * n - 1)
    G = np.zeros((len(r_indices), n))
    for i, r in enumerate(r_indices):

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')

    t_hi = 0.0

        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _newton_direction(h, r_active, c_reg=10.0):
    """Compute a regularized Newton direction for the active constraint"""
    n = len(h)
    dx = 2.0 / n
    G = _grad_row(h, r_active)
    # Projection onto sum-zero subspace
    G = G - G.mean()  # project gradient onto tangent
    
    # Build Hessian Q for this lag
    # Q is n x n, mostly zero except near the diagonal
    Q = np.zeros((n, n))
    d_lag = r_active - (n - 1)
    if d_lag >= 0:
        m = n - d_lag
        for j in range(m):
            Q[j, j+d_lag] += -dx
            Q[j+d_lag, j] += -dx
    else:
        g = -d_lag
        m = n - g
        for j in range(m):
            Q[j, j+g] += -dx
            Q[j+g, j] += -dx
    
    # Regularized: cI - Q (should be PD if c is large enough)
    A = c_reg * np.eye(n) - Q
    # Project A onto sum-zero: A_proj = (I - ee^T/n) A (I - ee^T/n)
    # But for the solve, we just need to solve A_proj d = -G in the sum-zero space
    # Which is equivalent to solving: A d = -G + λ * 1 (for some λ to enforce sum(d)=0)
    # Using the fact that 1^T A d = 1^T (-G) + λ n
    # And sum(d) = 1^T d = 0
    
    # Simpler: just add a large penalty for sum(d) != 0
    # Or solve the KKT system
    
    # Let me just use numpy lstsq with the constraint
    # Solve: A d = -G, sum(d) = 0
    # KKT: [A, 1; 1^T, 0] [d; λ] = [-G; 0]
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = A
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n+1)
    rhs[:n] = -G
    
    try:
        sol = np.linalg.solve(KKT, rhs)
        d = sol[:n]
    except:
        return None
    
    # Normalize
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return None
    return d / norm

            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

# ===== LP-based direction finding =====

def _lp_direction(h, scores, r_indices, radius):
    """Find LP direction to decrease active constraints"""
    n = len(h)
    G = _grad_matrix(h, r_indices)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(r_indices), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except:
        pass
    return None

# ===== SLSQP refinement =====

def _slsqp_refine(h, n_active=20, maxiter=200, deadline=None):
    n = len(h)
    target = n / 2.0
    c5_cur = _exact_c5(h)
    rows = _corr_rows(h)
    top_idx = np.argsort(-rows)[:n_active].tolist()
    t_init = c5_cur + 1e-4
    
    z0 = np.concatenate([h.copy(), [t_init]])
    
    def obj(z):
        return z[n]
    
    def obj_jac(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    cons = []
    for r in top_idx:
        ri = int(r)
        def ineq(z, ri=ri):
            return z[n] - _corr_rows(z[:n])[ri]
        def ineq_jac(z, ri=ri):
            g = np.zeros(n + 1)
            g[:n] = -_grad_row(z[:n], ri)
            g[n] = 1.0
            return g
        cons.append({'type': 'ineq', 'fun': ineq, 'jac': ineq_jac})
    
    def eq_fun(z):
        return float(z[:n].sum() - target)
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    
    try:
        res = minimize(obj, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                return h_new, c5_new
    except:
        pass
    
    # Try with more active constraints
    if deadline is not None and time.time() < deadline - 5:
        n_active2 = min(2*n_active, 2*n-1)
        top_idx2 = np.argsort(-rows)[:n_active2].tolist()
        t_init2 = c5_cur + 1e-4
        z0_2 = np.concatenate([h.copy(), [t_init2]])
        cons2 = []
        for r in top_idx2:
            ri = int(r)
            def ineq2(z, ri=ri):
                return z[n] - _corr_rows(z[:n])[ri]
            def ineq2_jac(z, ri=ri):
                g = np.zeros(n + 1)
                g[:n] = -_grad_row(z[:n], ri)
                g[n] = 1.0
                return g
            cons2.append({'type': 'ineq', 'fun': ineq2, 'jac': ineq2_jac})
        cons2.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
        try:
            res2 = minimize(obj, z0_2, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons2,
                           options={'maxiter': maxiter//2, 'ftol': 1e-12})
            if res2.success:
                h_new2 = _project(res2.x[:n], target)
                c5_new2 = _exact_c5(h_new2)
                if c5_new2 < c5_cur - 1e-12:
                    return h_new2, c5_new2
        except:
            pass
    
    return h, c5_cur
