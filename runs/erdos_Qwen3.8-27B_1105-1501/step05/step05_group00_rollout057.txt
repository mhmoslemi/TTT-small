def _resample(h, n_new):

def _pair_descent(h, deadline, max_rounds=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    for _r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.005) & (h < 0.995))[0]
        if len(interior) < 2:
            interior = np.arange(n)
        npairs = min(len(interior) * (len(interior) - 1) // 2, 400)
        tried = set()
        for _ in range(npairs):
            if time.time() > deadline:
                break
            i = interior[int(rng.integers(len(interior)))]
            j = interior[int(rng.integers(len(interior)))]
            key = (min(i, j), max(i, j))
            if i == j or key in tried:
                continue
            tried.add(key)
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search(h, d, dx, 0.45)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(1, max(2, n // 8))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.965

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _block_init(n0, rng)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start binary SA (18% time)
    d1 = t0 + budget_s * 0.18
    starts = [h0.copy()]
    for f in [1, 2, 3]:
        for ph in [0.0, 1.0, 2.0]:
            starts.append(_sine_init(n0, f, ph * np.pi / 3.0))
    for _ in range(3):
        starts.append(_block_init(n0, rng))
    for s in starts:
        if time.time() > d1 - 1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.02)
        hb, cb = _fast_binary_sa(s, sub_dl, rng)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()

    # Phase 2: LP refinement (15% time)
    d2 = t0 + budget_s * 0.33
    h2, c5_2 = _lp_refine(best_h, d2, radius_init=0.12, K=5)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Re-round + binary SA (12% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _fast_binary_sa(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3 + budget_s * 0.03:
        h3b, c5_3b = _lp_refine(best_h, min(d3 + budget_s * 0.03, deadline - 10), radius_init=0.08, K=5)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Pair descent (14% time)
    d4 = t0 + budget_s * 0.59
    h4, c5_4 = _pair_descent(best_h, d4, max_rounds=120)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: LP again (10% time)
    d5 = t0 + budget_s * 0.69
    h5, c5_5 = _lp_refine(best_h, d5, radius_init=0.08, K=7)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: More binary SA from continuous best (10% time)
    d6 = t0 + budget_s * 0.79
    h6, c5_6 = _fast_binary_sa(best_h, d6, rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6 + budget_s * 0.02:
        h6b, c5_6b = _lp_refine(best_h, min(d6 + budget_s * 0.02, deadline - 10), radius_init=0.06, K=5)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Multi-resolution (10% time)
    d7 = t0 + budget_s * 0.89
    best_n = len(best_h)
    tried = set()
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12]
    for dn in offsets:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.012)
        hr_b, cb_b = _fast_binary_sa(h_r, sub_dl, rng)
        if time.time() < d7 - 2:
            hr_l, cl = _lp_refine(hr_b, min(d7, time.time() + budget_s * 0.01), radius_init=0.07, K=5)
            if cl < cb_b:
                hr_b, cb_b = hr_l, cl
        if cb_b < best_c5 - 1e-12:
            best_c5 = cb_b
            best_h = hr_b.copy()
            best_n = n_try

    # Phase 8: Final polish (7% time)
    d8 = deadline
    h8, c5_8 = _pair_descent(best_h, min(d8, time.time() + budget_s * 0.025), max_rounds=80)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline - 3:
        h9, c5_9 = _lp_refine(best_h, deadline, radius_init=0.04, K=5)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
