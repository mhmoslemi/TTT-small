                s[idx[rng.integers(len(idx))]] = -1.0
        else:
            for _ in range(neg_cnt - pos_cnt):
                s[jdx[rng.integers(len(jdx))]] = 1.0
    for _ in range(iters):
        corr = np.correlate(s, s, mode='full')
        peak = float(np.max(np.abs(corr[1:2*n-1])))
        peak_lag = n - 1 + int(np.argmax(np.abs(corr))) 
        best_flip = None
        best_p
        best_peak = peak
        for _ in range(20):
            fi = int(rng.integers(0, n))
            s2 = s.copy()
            s2[fi] *= -1.0
            c2 = np.correlate(s2, s2, mode='full')
            pk2 = float(np.max(np.abs(c2[1:2*n-1])))
            if pk2 < best_peak:
                best_peak = pk2
                best_flip = fi
        if best_flip is not None:
            s[best_flip] *= -1.0
    return s

def _cosine_init(n, rng):
    x = (np.arange(n) + 0.5) / n * 2.0
    a1 = rng.uniform(-0.3, 0.3)
    a2 = rng.uniform(-0.2, 0.2)
    a3 = rng.uniform(-0.1, 0.1)
    h = 0.5 + a1 * np.cos(np.pi * x) + a2 * np.cos(np.pi * x * 2) + a3 * np.cos(np.pi * x * 3)
    return _project(h, n / 2.0)

def _combined_local(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    mode = 0
    lr = 0.03
    mom = np.zeros(n)
    radius = 0.12
    stall = 0
    while time.time() < deadline and stall < 60:
        if stall > 20:
            mode = 1 - mode
            stall = 0
            lr = 0.03
            mom = np.zeros(n)
            radius = 0.12
        if mode == 0:
            scores = _scores(h)
            s_max = float(np.max(scores))
            k = min(35, n // 6)
            topk = np.argsort(s_max - scores)[:k]
            d = _grad_rows(h, topk).sum(axis=0)
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn < 1e-12:
                stall += 1
                continue
            d = d / dn
            mom = 0.85 * mom + 0.15 * d
            mn = np.linalg.norm(mom)
            if mn < 1e-12:
                stall += 1
                continue
            mom = mom / mn
            alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.5)
            if new_c5 < cur - 1e-13:
                h = _project(h + alpha * lr * mom, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.40)
            else:
                stall += 1
                lr *= 0.84
        else:
            scores = _scores(h)
            s_max = float(np.max(scores))
            k = min(30, n // 7)
            topk = np.argsort(s_max - scores)[:k]
            G = _grad_rows(h, topk)
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((k, n + 1))
            A_ub[:, :n] = G
            A_ub[:, n] = -1.0
            b_ub = -scores[topk].copy()
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.zeros(1)
            bounds = []
            for j in range(n):
                lo = max(-radius, -h[j])
                hi = min(radius, 1.0 - h[j])
                if lo > hi:
                    lo = hi = 0.0
                bounds.append((lo, hi))
            bounds.append((None, None))
            try:
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs', options={'presolve': True})
                if res.success:
                    delta = res.x[:n]
                    if np.max(np.abs(delta)) > 1e-12:
                        alpha, new_c5 = _line_search(h, delta, dx, radius)
                        if new_c5 < cur - 1e-13:
                            h = _project(h + alpha * delta, target)
                            cur = _exact_c5(h)
                            if cur < best_c5:
                                best_c5 = cur
                                best_h = h.copy()
                            stall = 0
                            radius = min(radius * 1.04, 0.4)
                            continue
            except Exception:
                pass
            stall += 1
            radius *= 0.8
    return best_h, best_c5

def _multi_start_optimize(h_init, n_target, t_alloc, rng, best_c5_so_far):
    n = n_target
    if len(h_init) != n:
        h_init = _resample(h_init, n)
    h = h_init.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    t_start = time.time()
    t_phase1 = t_start + t_alloc * 0.45
    t_phase2 = t_start + t_alloc * 0.80
    h1, c5_1 = _combined_local(h, min(t_phase1, t_start + t_alloc * 0.45), rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    if time.time() < t_phase2:
        h2, c5_2 = _active_lp(best_h, t_phase2, k=min(35, n // 7), radius=0.10)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    if time.time() < t_start + t_alloc - 2:
        h3, c5_3 = _pair_refine(best_h, t_start + t_alloc - 1, rng=rng, n_random=min(300, n * n // 10))
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 0: Quick local at parent resolution
    d0 = t0 + budget_s * 0.06
    h0opt, c5_0 = _combined_local(h, d0, rng)
    if c5_0 < best_c5:
        best_c5 = c5_0
        best_h = h0opt.copy()

    # Phase 1: Multi-resolution exploration with multiple initializations
    resolutions = [192, 256, 320, 400, 512]
    for n_try in resolutions:
        remaining = deadline - time.time()
        if remaining < 80:
            break
        t_alloc = min(remaining * 0.35, 140.0)

        # Init 1: interpolate parent
        h_interp = _resample(best_h, n_try)
        t_sub = min(t_alloc * 0.40, 55.0)
        hA, c5A = _multi_start_optimize(h_interp, n_try, t_sub, rng, best_c5)
        if c5A < best_c5:
            best_c5 = c5A
            best_h = hA.copy()

        # Init 2: low-PEP sequence
        remaining2 = deadline - time.time()
        if remaining2 > 60:
            s_seq = _low_pep_sequence(n_try, rng, iters=80)
            h_seq = _project((s_seq + 1.0) / 2.0, n_try / 2.0)
            t_sub2 = min(remaining2 * 0.35, 50.0)
            hB, c5B = _multi_start_optimize(h_seq, n_try, t_sub2, rng, best_c5)
            if c5B < best_c5:
                best_c5 = c5B
                best_h = hB.copy()

        # Init 3: cosine profile
        remaining3 = deadline - time.time()
        if remaining3 > 40:
            h_cos = _cosine_init(n_try, rng)
            t_sub3 = min(remaining3 * 0.25, 35.0)
            hC, c5C = _multi_start_optimize(h_cos, n_try, t_sub3, rng, best_c5)
            if c5C < best_c5:
                best_c5 = c5C
                best_h = hC.copy()

    # Phase 2: Final intensive refinement at best resolution
    remaining = deadline - time.time()
    if remaining > 40:
        n_best = len(best_h)
        # Aggressive subgradient
        d_s = time.time() + remaining * 0.30
        h_s, c5_s = _subgrad(best_h, d_s, k=min(50, n_best // 5), lr_init=0.02, mom_decay=0.85)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        # Active LP
        if time.time() < deadline - 25:
            d_l = time.time() + remaining * 0.30
            h_l, c5_l = _active_lp(best_h, d_l, k=min(50, n_best // 5), radius=0.08)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()
        # Pair refine
        if time.time() < deadline - 10:
            h_p, c5_p = _pair_refine(best_h, deadline - 2, rng=rng, n_random=min(600, n_best * n_best // 6))
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
