from scipy.optimize import minimize, linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _sig(x):
    """Overflow-safe sigmoid"""
    out = np.zeros_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _q_to_h(q, lam):
    """Map q, lambda to h via logistic"""
    return _sig(q - lam)

def _find_lambda(q, target, n):
    """Find lambda such that sum(sigma(q-lambda)) = target"""
    lo = float(q.min()) - 30.0
    hi = float(q.max()) + 30.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sig(q - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_obj(q, beta, target, n, top_k=0):
    """Objective in q-space: smoothed max of scores"""
    lam = _find_lambda(q, target, n)
    h = _sig(q - lam)
    S = _scores(h)
    maxS = np.max(S)
    if top_k > 0 and top_k < len(S):
        # Focus on top-k lags
        idx = np.argsort(S)[-top_k:]
        Sk = S[idx]
        maxSk = np.max(Sk)
        shifted = beta * (Sk - maxSk)
        shifted = np.clip(shifted, -500, 0)
        return maxSk + np.log(np.mean(np.exp(shifted))) / beta
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500, 0)
    return maxS + np.log(np.mean(np.exp(shifted))) / beta

def _q_grad(q, beta, target, n, top_k=0):
    """Gradient of objective in q-space"""
    lam = _find_lambda(q, target, n)
    h = _sig(q - lam)
    S = _scores(h)
    maxS = np.max(S)
    
    if top_k > 0 and top_k < len(S):
        idx = np.argsort(S)[-top_k:]
        Sk = S[idx]
        maxSk = np.max(Sk)
        shifted = beta * (Sk - maxSk)
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        wsum = np.sum(w)
        if wsum < 1e-300:
            w = np.zeros(len(Sk))
            w[int(np.argmax(Sk))] = 1.0
        else:
            w /= wsum
        # Need gradient of S w.r.t. h, then h w.r.t. q
        # dS/dh: for each lag, the gradient w.r.t. h[i]
        # S[d] = sum_i h[i]*(1-h[i+d]) * dx for valid i
        # dS[d]/dh[j] involves terms where j appears
        # This is complex, so let's use finite differences for the q-gradient
        pass
    
    # Finite difference gradient (simpler and more robust)
    eps = 1e-6
    grad = np.zeros(n)
    for i in range(n):
        q_plus = q.copy()
        q_plus[i] += eps
        q_minus = q.copy()

        gm = _q_obj(q_minus, beta, target, n, top_k)
        grad[i] = (gp - gm) / (2 * eps)
    return grad

def _lbfgs_in_q(q, beta, target, n, top_k=0, max_iter=500, ftol=1e-14):
    """Run L-BFGS-B in q-space"""
    def obj(q_):
        return _q_obj(q_, beta, target, n, top_k)
    
    # Use finite difference gradient (n is small enough)
    res = minimize(obj, q, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10})

        lam = _find_lambda(res.x, target, n)
        h = _sig(res.x - lam)

    return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_matrix(h):
    """Gradient matrix G where dS_r/dh_j = G[r,j] * (2/n)"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

            maxS = np.max(S)

def _lp_direction(h, scores, radius):
    """Find best descent direction via LP"""
    n = len(h)
    n_lags = len(scores)
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _binary_flip_optimize(h, deadline, n_rounds=50):

                if h[j] < 1e-6 or h[j] > 1-1e-6:
                    continue
                # Try moving mass from i to j or j to i
                for direction in [1, -1]:
                    if direction == 1:
                        # move from j to i
                        if h[j] < 1e-8 or h[i] > 1-1e-8:

                    else:

                                            h[j] if direction==1 else 1.0 - h[j])

                    # Quick check: just try moving half
                    mid = hi_a / 2
                    h_try = h.copy()
                    h_try[i] += direction * mid

            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    except Exception:
        pass
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    n_lags = len(scores)
    G = _grad_matrix(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.999
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(35):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _local_optimize_lp(h, deadline, radius_init=0.25, max_iters=5000, stall_limit=150):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _scores(h)
        p, _ = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-14:
            h_new = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h_new)
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.87
            if stall > 50 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _lbfgs_q_obj(q, beta, target, n):
    lam = _find_lambda(q, target, n)
    h = _sig(q - lam)
    S = _scores(h)
    maxS = np.max(S)
    shifted = beta * (S - maxS)
    shifted = np.clip(shifted, -500, 0)
    return maxS + np.log(np.mean(np.exp(shifted))) / beta

def _lbfgs_in_q(q_init, deadline, betas, n, target):
    best_h = None
    best_c5 = float('inf')
    q_cur = q_init.copy()
    for beta in betas:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 8.0)
        rem = sub_dl - time.time()
        if rem < 0.5:
            break
        max_iter = max(30, int(rem * 8))
        try:
            res = minimize(_lbfgs_q_obj, q_cur, args=(beta, target, n),
                          method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iter * 30})
            if np.all(np.isfinite(res.x)):
                lam = _find_lambda(res.x, target, n)
                h_new = _sig(res.x - lam)
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_c = np.clip(h_new, 1e-4, 1 - 1e-4)
                    q_cur = np.log(h_c / (1 - h_c))
                    q_cur -= np.mean(q_cur)
        except Exception:
            pass
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        val = i
        while val > 0:
            cnt += val & 1
            val >>= 1
        h[i] = float(cnt % 2)
    return _project(h, n / 2.0)

def _binary_flip_optimize(h, deadline, max_rounds=30):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 24)
        for i in range(0, n, step):
            if time.time() > deadline:
                break
            if h[i] < 5e-3 or h[i] > 1.0 - 5e-3:
                continue
            for j in range(i + step, n, step):
                if h[j] < 5e-3 or h[j] > 1.0 - 5e-3:
                    continue
                for direction in (1, -1):
                    if direction == 1:
                        a_max = min(1.0 - h[i], h[j])
                    else:
                        a_max = min(h[i], 1.0 - h[j])
                    if a_max < 1e-10:
                        continue
                    lo_a = 0.0
                    hi_a = a_max
                    for _ in range(8):
                        mid = (lo_a + hi_a) / 2.0
                        h_try = h.copy()
                        h_try[i] += direction * mid
                        h_try[j] -= direction * mid
                        h_try = np.clip(h_try, 0.0, 1.0)
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            lo_a = mid
                        else:
                            hi_a = mid
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[i] += direction * a_opt
                    h_try[j] -= direction * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _subgrad_step(h, beta, target, n):
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= grad.mean()
    return grad

def _subgrad_optimize(h, deadline, n_steps=2000, beta_start=10.0, beta_end=80.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        grad = _subgrad_step(h, beta, target, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.08
        improved = False
        for _ in range(40):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.72
            if alpha < 1e-13:
                break
        if not improved and step % 80 == 0:
            h = _project(h - 0.01 * grad, target)
            cur_c5 = _exact_c5(h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Thorough LP optimization from initial
    dl1 = t0 + budget_s * 0.22
    h1, c5_1 = _local_optimize_lp(h_start, dl1, radius_init=0.35, max_iters=12000, stall_limit=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP with many phases
    if time.time() < t0 + budget_s * 0.38:
        dl2 = t0 + budget_s * 0.38
        h2, c5_2 = _slsqp_optimize(best_h, dl2, beta_init=40.0, beta_final=500.0, n_phases=10)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        # LP polish
        if time.time() < dl2:
            h2b, c5_2b = _local_optimize_lp(best_h, dl2, radius_init=0.1, max_iters=3000, stall_limit=150)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Top-k SLSQP focus
    if time.time() < t0 + budget_s * 0.48:
        dl3 = t0 + budget_s * 0.48
        h3, c5_3 = _slsqp_topk(best_h, dl3, beta=800.0, top_k=4)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        h3b, c5_3b = _slsqp_topk(best_h, dl3, beta=1200.0, top_k=3)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: L-BFGS-B in reparameterized space
    if time.time() < t0 + budget_s * 0.62:
        dl4 = t0 + budget_s * 0.62
        n = best_n
        target = n / 2.0
        eps = 1e-3
        h_clipped = np.clip(best_h, eps, 1 - eps)
        q_init = np.log(h_clipped / (1 - h_clipped))
        q_init -= np.mean(q_init)
        betas = [15, 30, 60, 120, 250, 500, 800, 1200]
        hb, cb = _lbfgs_in_q(q_init, dl4, betas, n, target)
        if hb is not None and cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        # LP polish after L-BFGS
        if time.time() < dl4:
            h4, c5_4 = _local_optimize_lp(best_h, dl4, radius_init=0.08, max_iters=2000, stall_limit=100)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

    # Phase 5: Multi-resolution with diverse starts
    if time.time() < t0 + budget_s * 0.78:
        dl5 = t0 + budget_s * 0.78
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
        for n_try in [64, 80, 96, 128, 160, 192]:
            if time.time() > dl5:
                break
            for a in alphas:
                if time.time() > dl5:
                    break
                h_qp = _quasi_periodic_init(n_try, a)
                h_qp = _project(h_qp + rng.normal(0, 0.04, n_try), n_try / 2.0)
                sub_dl = min(dl5, time.time() + budget_s * 0.018)
                ht, ct = _local_optimize_lp(h_qp, sub_dl, radius_init=0.3,
                                            max_iters=3000, stall_limit=150)
                if ct < best_c5:
                    best_c5 = ct
                    best_h = ht.copy()
                    best_n = n_try
                    # SLSQP polish on new best
                    if time.time() < dl5:
                        ht2, ct2 = _slsqp_optimize(ht, min(dl5, time.time() + budget_s * 0.01),
                                                   beta_init=60, beta_final=300, n_phases=4)
                        if ct2 < best_c5:
                            best_c5 = ct2
                            best_h = ht2.copy()
                            best_n = n_try
            # Thue-Morse start
            if time.time() < dl5:
                h_tm = _thue_morse_init(n_try)
                sub_dl2 = min(dl5, time.time() + budget_s * 0.015)
                ht3, ct3 = _local_optimize_lp(h_tm, sub_dl2, radius_init=0.3,
                                              max_iters=2000, stall_limit=120)
                if ct3 < best_c5:
                    best_c5 = ct3
                    best_h = ht3.copy()
                    best_n = n_try

    # Phase 6: Subgradient + LP from best
    if time.time() < t0 + budget_s * 0.85:
        dl6 = t0 + budget_s * 0.85
        h6, c5_6 = _subgrad_optimize(best_h, dl6, n_steps=1500, beta_start=20.0, beta_end=100.0)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        if time.time() < dl6:
            h6b, c5_6b = _local_optimize_lp(best_h, dl6, radius_init=0.12,
                                            max_iters=3000, stall_limit=150)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    # Phase 7: Perturbation restarts
    if time.time() < t0 + budget_s * 0.92:
        dl7 = t0 + budget_s * 0.92
        scale = 0.025
        for attempt in range(400):
            if time.time() > dl7:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl7, time.time() + budget_s * 0.006)
            ht, ct = _local_optimize_lp(h_pert, sub_dl, radius_init=0.06,
                                        max_iters=600, stall_limit=50)
            if ct < best_c5 - 1e-13:
                best_c5 = ct
                best_h = ht.copy()
                scale = max(0.002, scale * 0.82)
            else:
                scale = min(0.08, scale * 1.12)

    # Phase 8: Final SLSQP with high beta
    if time.time() < t0 + budget_s * 0.955:
        dl8 = t0 + budget_s * 0.955
        h8, c5_8 = _slsqp_optimize(best_h, dl8, beta_init=300.0, beta_final=2000.0, n_phases=6)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
        # Top-k final
        if time.time() < dl8:
            h8b, c5_8b = _slsqp_topk(best_h, dl8, beta=2000.0, top_k=3)
            if c5_8b < best_c5:
                best_c5 = c5_8b
                best_h = h8b.copy()

    # Phase 9: Final LP polish
    if time.time() < t0 + budget_s * 0.97:
        h9, c5_9 = _local_optimize_lp(best_h, t0 + budget_s * 0.97,
                                       radius_init=0.03, max_iters=4000, stall_limit=200)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Phase 10: Binary flip polish
    if time.time() < t0 + budget_s * 0.985:
        h10, c5_10 = _binary_flip_optimize(best_h, t0 + budget_s * 0.985, max_rounds=20)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5
    return best_h, final_c5, best_n
