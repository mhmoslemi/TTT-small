```python
# Vectorized delta computation for binary swap a→0, b→1
def _swap_delta(h, a, b, n):
    dx = 2.0 / n
    d = np.zeros(2 * n - 1)
    # For a (1→0):
    # d[n-1+lag] for relevant lags
    lags_a = np.arange(-a, n - a)
    idx_a = n - 1 + lags_a
    # (1-h[a+lag]) for valid a+lag
    valid_plus = (a + lags_a >= 0) & (a + lags_a < n)
    valid_minus = (a - lags_a >= 0) & (a - lags_a < n)
    d[idx_a[valid_plus]] -= dx * (1.0 - h[a + lags_a[valid_plus]])
    d[idx_a[valid_minus]] += dx * h[a - lags_a[valid_minus]]
    # For b (0→1):
    lags_b = np.arange(-b, n - b)
    idx_b = n - 1 + lags_b
    valid_plus_b = (b + lags_b >= 0) & (b + lags_b < n)
    valid_minus_b = (b - lags_b >= 0) & (b - lags_b < n)
    d[idx_b[valid_plus_b]] += dx * (1.0 - h[b + lags_b[valid_plus_b]])
    d[idx_b[valid_minus_b]] -= dx * h[b - lags_b[valid_minus_b]]
    return d
```

This is O(n) with numpy operations, much faster than a Python loop.

dx * n/2 - dx * sum_j h[j]*h[j+k] ≤ t  (using sum h = n/2)
dx * sum_j h[j]*h[j+k] ≥ dx * n/2 - t

So we're maximizing the minimum of {dx * sum_j h[j]*h[j+k]} over k, subject to the constraints.

This is a **maximin** problem, which is dual to a minimax problem. It's also a **quadratic** program (the constraints are quadratic in h).

Specifically: min t s.t. h^T B_k h ≥ c_k - t for all k, 0≤h≤1, 1^T h = n/2
where B_k is a shift matrix and c_k = dx * n/2.

This is a non-convex QP (B_k is not necessarily PSD). So standard QP solvers won't work.

def _exact_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def _scores_of(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1
        else:
            mask = x > 0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_coeff(h):
    n = len(h)
    i_idx = np.arange(n)[:, None].astype(float)
    d_idx = np.arange(1, n)[None, :].astype(float)
    ip = i_idx + d_idx
    im = i_idx - d_idx
    valid_ip = (ip >= 0) & (ip < n)
    valid_im = (im >= 0) & (im < n)
    ip_c = np.clip(ip.astype(int), 0, n - 1)
    im_c = np.clip(im.astype(int), 0, n - 1)
    C_pos = np.where(valid_ip, 1.0 - h[ip_c], 0.0) - np.where(valid_im, h[im_c], 0.0)
    C_neg = np.where(valid_im, 1.0 - h[im_c], 0.0) - np.where(valid_ip, h[ip_c], 0.0)
    C = np.zeros((n, 2 * n - 1))
    C[:, n:] = C_pos
    C[:, :n - 1] = C_neg
    return C

def _to_binary(h, rng):
    n = len(h)
    target = n // 2
    h_pert = h + rng.normal(0, 0.001, n)
    idx = np.argsort(-h_pert)[:target]

def _swap_delta(h, a, b, n):
    dx = 2.0 / n
    d = np.zeros(2 * n - 1)
    lags_a = np.arange(-a, n - a)
    idx_a = n - 1 + lags_a
    ap = a + lags_a
    am = a - lags_a
    vp = (ap >= 0) & (ap < n)
    vm = (am >= 0) & (am < n)
    d[idx_a[vp]] -= dx * (1.0 - h[ap[vp]])
    d[idx_a[vm]] += dx * h[am[vm]]
    lags_b = np.arange(-b, n - b)
    idx_b = n - 1 + lags_b
    bp = b + lags_b
    bm = b - lags_b
    vp2 = (bp >= 0) & (bp < n)
    vm2 = (bm >= 0) & (bm < n)
    d[idx_b[vp2]] += dx * (1.0 - h[bp[vp2]])
    d[idx_b[vm2]] -= dx * h[bm[vm2]]
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
