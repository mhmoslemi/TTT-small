def _project_onto_polytope(v, target):
    # Water-filling projection onto {0 <= x <= 1, sum(x) = target}
    n = len(v)
    # Bisection on threshold
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(0.0, np.minimum(1.0, v - mid)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    x = np.clip(v - t, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_gradient(h, gamma):
    # Compute weighted gradient of soft-max approximation
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[np.argmax(scores)] = 1.0
    else:
        w /= sw
    
    grad = np.zeros(n)
    # Center term (r = n-1, d=0)
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    
    # Positive lags d = 1, ..., n-1
    for d in range(1, n):
        r_pos = n - 1 + d  # index in full correlation
        w_pos = w[r_pos]
        m = n - d
        if m <= 0:
            break
        # corr[r_pos] = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
        # d/dh[j]:
        #   if j < m: from (1-h[j]) term -> -h[j+d]
        #   if j >= d: from h[j] term (where j = i+d, i = j-d < m) -> (1-h[j-d])
        grad[:m] += -w_pos * dx * h[d:]  # contribution from first index i
        grad[d:] += w_pos * dx * (1.0 - h[:m])  # contribution from second index i+d
    
    # Negative lags: r = n-1-d for d = 1, ..., n-1
    for d in range(1, n):
        r_neg = n - 1 - d
        w_neg = w[r_neg]
        m = n - d
        if m <= 0:
            break
        # corr[r_neg] = sum_{i=0}^{m-1} h[i]*(1-h[i+d])
        # d/dh[j]:
        #   if j < m: from h[j] term -> (1-h[j+d])
        #   if j >= d: from (1-h[j]) term (where j = i+d) -> h[j-d]
        grad[:m] += w_neg * dx * (1.0 - h[d:])
        grad[d:] += w_neg * dx * h[:m]
    
    return grad

import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(v, target):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-14:
        if r > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        c = mask.sum()
        if c > 0:
            x[mask] += r / c
            x = np.clip(x, 0.0, 1.0)
    return x

def _wg(h, gamma):
    """Weighted gradient of soft-max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    sc = np.correlate(h, 1.0 - h, mode='full') * dx
    ms = sc.max()
    w = np.exp(gamma * (sc - ms))
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(sc)
        w[sc.argmax()] = 1.0
    else:
        w /= sw
    g = np.zeros(n)
    g += w[n-1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        if m <= 0:
            break
        wp = w[n-1+d]
        wn = w[n-1-d]
        # Positive lag d: corr = sum h[i+d]*(1-h[i])
        g[:m] += -wp * dx * h[d:]
        g[d:] += wp * dx * (1.0 - h[:m])
        # Negative lag d: corr = sum h[i]*(1-h[i+d])
        g[:m] += wn * dx * (1.0 - h[d:])
        g[d:] += wn * dx * h[:m]
    return g

def _pair_opt(h, i, j, n_g=25):

def _grad_step(h, gamma, dl):
    """One gradient step with line search."""
    n = len(h)
    target = n / 2.0
    g = _wg(h, gamma)
    g -= g.mean()  # preserve sum
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, _c5(h)
    g /= gn
    # Line search
    # Find alpha_max
    am = 0.5
    for j in range(n):
        if g[j] > 1e-14:
            am = min(am, (1.0 - h[j]) / g[j])
        elif g[j] < -1e-14:
            am = min(am, h[j] / (-g[j]))
    am = max(am, 1e-12)
    # Golden section
    cur = _c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, am
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(_proj(h + c1 * g, target))
    f2 = _c5(_proj(h + c2 * g, target))
    for _ in range(20):

        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(_proj(h + c1 * g, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2

            f2 = _c5(_proj(h + c2 * g, target))

    c5_new = _c5(h_new)
    if c5_new < cur - 1e-13:
        return h_new, c5_new
    return h, cur

    # Sort by gradient: want to move from high g to low g

    max_attempts = n_pairs

    # Try random pairs

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(xn, xo, h)
    return _proj(h_new, n_new / 2.0)

def _binary_init(n, rng):

    hi = v.max() + 1.0

            x = np.clip(x, 0.0, 1.0)

import numpy as np
import time


def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _proj(v, target):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-14:
        if r > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        c = int(mask.sum())
        if c > 0:
            x[mask] += r / c
            x = np.clip(x, 0.0, 1.0)
    return x


def _wg(h, gamma):
    n = len(h)
    dx = 2.0 / n
    sc = np.correlate(h, 1.0 - h, mode='full') * dx
    ms = sc.max()
    w = np.exp(np.clip(gamma * (sc - ms), -500.0, 500.0))
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(sc)
        w[int(sc.argmax())] = 1.0
    else:
        w /= sw
    g = np.zeros(n)
    g += w[n - 1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        if m <= 0:
            break
        wp = w[n - 1 + d]
        wn = w[n - 1 - d]
        g[:m] += -wp * dx * h[d:]
        g[d:] += wp * dx * (1.0 - h[:m])
        g[:m] += wn * dx * (1.0 - h[d:])
        g[d:] += wn * dx * h[:m]
    return g


def _pair_search(h, i, j):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return 0.0, _c5(h)
    cur = _c5(h)
    best_t = 0.0
    best_c = cur
    h2 = h.copy()
    ng = 20
    for k in range(1, ng):
        t = t_lo + (t_hi - t_lo) * k / ng
        h2[i] = h[i] + t
        h2[j] = h[j] - t
        c = _c5(h2)
        if c < best_c:
            best_c = c
            best_t = t
    if t_lo < best_t < t_hi:
        span = (t_hi - t_lo) / ng
        lo2 = max(t_lo, best_t - span)
        hi2 = min(t_hi, best_t + span)
        for k in range(15):
            t = lo2 + (hi2 - lo2) * k / 14.0
            h2[i] = h[i] + t
            h2[j] = h[j] - t
            c = _c5(h2)
            if c < best_c:
                best_c = c
                best_t = t
    return best_t, best_c


def _pair_optimize(h, dl, max_passes=200, seed=0, n_pairs=0):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = 15.0
    stall = 0
    if n_pairs == 0:
        n_pairs = min(n * (n - 1) // 2, 3000)
    for ps in range(max_passes):
        if time.time() > dl:
            break
        if stall > 50:
            break
        g = _wg(h, gamma)
        order = np.argsort(g)
        improved = False
        npass = min(n_pairs, 2000)
        for _ in range(npass):
            if time.time() > dl:
                break
            quarter = max(1, n // 4)
            hi_idx = int(order[n - 1 - rng.integers(quarter)])
            lo_idx = int(order[rng.integers(quarter)])
            if hi_idx == lo_idx:
                continue
            t, c5_new = _pair_search(h, lo_idx, hi_idx)
            if c5_new < cur_c5 - 1e-13:
                h[lo_idx] += t
                h[hi_idx] -= t
                cur_c5 = c5_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        # Random pairs
        for _ in range(npass // 3):
            if time.time() > dl:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            t, c5_new = _pair_search(h, i, j)
            if c5_new < cur_c5 - 1e-13:
                h[i] += t
                h[j] -= t
                cur_c5 = c5_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if improved:
            stall = 0
            gamma = min(gamma * 1.02, 400.0)
        else:
            stall += 1
            gamma = min(gamma * 1.08, 600.0)
    return best_h, best_c5


def _grad_step(h, gamma, dl):
    n = len(h)
    target = n / 2.0
    g = _wg(h, gamma)
    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, _c5(h)
    g /= gn
    am = 0.5
    for j in range(n):
        if g[j] > 1e-14:
            am = min(am, (1.0 - h[j]) / g[j])
        elif g[j] < -1e-14:
            am = min(am, h[j] / (-g[j]))
    am = max(am, 1e-12)
    cur = _c5(h)
    gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, am
    c1 = hi - gr_phi * (hi - lo)
    c2 = lo + gr_phi * (hi - lo)
    f1 = _c5(_proj(h + c1 * g, target))
    f2 = _c5(_proj(h + c2 * g, target))
    for _ in range(18):
        if time.time() > dl:
            break
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr_phi * (hi - lo)
            f1 = _c5(_proj(h + c1 * g, target))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr_phi * (hi - lo)
            f2 = _c5(_proj(h + c2 * g, target))
    a = (lo + hi) * 0.5
    h_new = _proj(h + a * g, target)
    c5_new = _c5(h_new)
    if c5_new < cur - 1e-13:
        return h_new, c5_new
    return h, cur


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(xn, xo, h)
    return _proj(h_new, n_new / 2.0)


def _local_opt(h, dl, seed=0, max_rounds=300):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = 10.0
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > dl:
            break
        if stall > 30:
            break
        # Pair optimization
        h_new, c_new = _pair_optimize(h, dl, max_passes=10 if n > 160 else 20, seed=seed + rnd)
        if c_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            # Gradient step
            h_new, c_new = _grad_step(h, gamma, dl)
            if c_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                gamma = min(gamma * 1.1, 500.0)
    return best_h, best_c5
