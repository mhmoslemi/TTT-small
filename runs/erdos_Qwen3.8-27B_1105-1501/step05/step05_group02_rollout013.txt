import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = (scores - max_s) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = max_s + tau * np.log(sum_exp)
    w = exp_s / sum_exp
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for r in range(n_lags):
        d = r - (n - 1)
        wr = w[r]
        if wr < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] += -wr * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] += -wr * h[:m]
    return obj, grad * dx

def _smooth_slsqp(h, deadline, tau_steps=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()

    scores_init = _scores(h)
    score_range = np.max(scores_init) - np.min(scores_init)
    if tau_steps is None:
        tau_steps = [max(0.15, 0.5 * score_range), max(0.06, 0.2 * score_range),

                     max(0.004, 0.01 * score_range), max(0.0015, 0.004 * score_range)]
    bounds = [(0.0, 1.0)] * n
    constr = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
               'jac': lambda x: np.ones(n)}]
    for tau in tau_steps:
        if time.time() > deadline:
            break

def _lp_one_step(h, radius):
    n = len(h)
    scores = _scores(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    G *= (2.0 / n)
    topk = min(30, 2 * n - 1)
    idx = np.argsort(scores)[-topk:]
    G_k = G[idx]
    scores_k = scores[idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((topk, n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-13:
                return p
    except Exception:
        pass
    return None

def _line_search_c5(h, p, max_iter=20):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(max_iter):
        if hi - lo < 1e-14:

            c2 = lo + gr * (hi - lo); f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    return opt_a, _exact_c5(h + opt_a * p)

                    best_c5 = c5_qo
                    best_h = h_qo.copy()
                    best_n = n_qp

    # Phase 7: More smooth optimization from best
    if time.time() < t0 + budget_s * 0.88:
        dl7 = t0 + budget_s * 0.92
        # Use smaller tau for finer optimization
        tau_fine = [0.02, 0.008, 0.003, 0.001]
        h_f, c5_f = _smooth_slsqp(best_h, dl7, tau_steps=tau_fine)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Projected gradient fine-tuning
    if time.time() < t0 + budget_s * 0.95:
        dl8 = t0 + budget_s * 0.97
        h_pg, c5_pg = _proj_grad_optimize(best_h, dl8, tau_init=0.01, max_iter=5000)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 9: Final LP polish
    if time.time() < t0 + budget_s * 0.985:
        dl9 = t0 + budget_s * 0.995
        h_fin, c5_fin = _lp_optimize(best_h, dl9, radius_init=0.04, max_it=2000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation and projection
    final_n = len(best_h)
    best_h = _project(best_h, final_n / 2.0)
    final_c5 = _exact_c5(best_h)
    # Ensure feasibility
    if final_c5 > best_c5 + 1e-9:
        best_h = _project(best_h, final_n / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, final_n
