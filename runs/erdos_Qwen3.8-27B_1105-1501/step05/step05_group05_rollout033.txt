def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _golden_search(f, a, b, tol=1e-10):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    while b - a > tol:
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0, min(f((a+b)/2), fc, fd)

def _pair_optimal_t(h, i, j, S0, dx):

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_step(h, i, j, S0, dx, d_buf):
    """Find best t for h[i]+=t, h[j]-=t. Uses S0=preset scores, d_buf for direction."""
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    d_buf[i] = 1.0
    d_buf[j] = -1.0
    T = np.correlate(d_buf, 1.0 - h, mode='full') - np.correlate(h, d_buf, mode='full')
    C = np.correlate(d_buf, d_buf, mode='full')
    d_buf[i] = 0.0
    d_buf[j] = 0.0
    # Golden section: minimize max(S0 + t*T - t^2*C) * dx over [t_lo, t_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c_val = b - gr * (b - a)
    d_val = a + gr * (b - a)
    fc = float(np.max(S0 + c_val * T - c_val * c_val * C)) * dx
    fd = float(np.max(S0 + d_val * T - d_val * d_val * C)) * dx
    for _ in range(18):
        if b - a < 1e-11:
            break
        if fc < fd:
            b, d_val, fd = d_val, c_val, fc
            c_val = b - gr * (b - a)
            fc = float(np.max(S0 + c_val * T - c_val * c_val * C)) * dx
        else:
            a, c_val, fc = c_val, d_val, fd
            d_val = a + gr * (b - a)
            fd = float(np.max(S0 + d_val * T - d_val * d_val * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if fc < opt_val:
        opt_t, opt_val = c_val, fc
    if fd < opt_val:
        opt_t, opt_val = d_val, fd
    return opt_t, opt_val

def _pair_descent_full(h, deadline, max_passes=100):
    """Full-scan pair descent: each pass finds and applies the single best move."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    d_buf = np.zeros(n)
    stall = 0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if stall > 5:
            break
        S0 = np.correlate(h, 1.0 - h, mode='full')
        best_t = 0.0
        best_val = cur_c5
        best_i, best_j = -1, -1
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                t_lo = max(-h[i], h[j] - 1.0)
                t_hi = min(1.0 - h[i], h[j])
                if t_hi - t_lo < 1e-12:
                    continue
                t, val = _pair_step(h, i, j, S0, dx, d_buf)
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t
                    best_i, best_j = i, j
        if best_i < 0:
            break
        h[best_i] += best_t
        h[best_j] -= best_t
        h = _project(h, target)
        new_c5 = _exact_c5(h)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        cur_c5 = new_c5
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _anneal(h, deadline, rng, T_init=0.005, T_final=1e-5, max_steps=200000):
    """Simulated annealing with random pair transfers."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for step in range(max_steps):
        if time.time() > deadline:
            break
        prog = min(step / max_steps, 0.999)
        T = T_init * (1.0 - prog) ** 2.0 + T_final
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        if t_hi - t_lo < 1e-12:
            continue
        # Random magnitude, biased toward small moves at low T
        scale = T / T_init
        t_mag = rng.uniform(0.1, 1.0) * (t_hi - t_lo) * (0.3 + 0.7 * scale)
        t = t_mag if rng.random() < 0.5 else -t_mag
        t = max(t_lo, min(t_hi, t))
        if abs(t) < 1e-10:
            continue
        h[i] += t
        h[j] -= t
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] -= t
            h[j] += t
    return best_h, best_c5
