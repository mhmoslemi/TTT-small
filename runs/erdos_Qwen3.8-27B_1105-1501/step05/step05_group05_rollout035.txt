def exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Water-filling projection to [0,1] with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_constraint(h, r, n):
    """Gradient of corr[r] w.r.t. h (before multiplying by dx)."""
    shift = r - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        # (1-h[i-shift]) for shift <= i <= n-1
        idx1 = range(shift, n)
        g[idx1] += (1.0 - h[np.array(idx1) - shift])

    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0

def slsqp_solve(h, active_idx, n, dx, radius, maxiter=50, ftol=1e-10):
    """SLSQP epigraph optimization with active constraints."""
    target = n / 2.0
    c5_cur = exact_c5(h)
    
    def objective(x):
        return x[-1]
    
    def objective_jac(x):
        j = np.zeros(n + 1)
        j[-1] = 1.0
        return j
    
    K = len(active_idx)
    
    def constraint_fun(x):
        h_sub = x[:n]
        tau = x[-1]
        # Return tau - corr(r)*dx for each active r
        one_minus_h = 1.0 - h_sub
        # Compute all needed correlations efficiently
        # For each active r, we need corr[r] = sum of h[i]*one_minus_h[j] with appropriate indices
        c = np.zeros(K)
        for k, r in enumerate(active_idx):
            shift = r - (n - 1)
            if shift >= 0:
                m = n - shift
                if m > 0:
                    c[k] = np.dot(h_sub[shift:shift+m], one_minus_h[:m])
            else:
                s = -shift
                m = n - s
                if m > 0:
                    c[k] = np.dot(h_sub[:m], one_minus_h[s:s+m])
            c[k] = tau - c[k] * dx
        return c
    
    def constraint_jac(x):
        h_sub = x[:n]
        J = np.zeros((K, n + 1))
        for k, r in enumerate(active_idx):
            J[k, :n] = -dx * grad_corr(h_sub, r, n)
            J[k, n] = 1.0
        return J
    
    constraints = [{'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    # Equality constraint: sum(h) = target
    def eq_fun(x):
        return x[:n].sum() - target
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    # Bounds
    bounds = [(max(0.0, h[i] - radius), min(1.0, h[i] + radius)) for i in range(n)]
    bounds.append((max(0.0, c5_cur - 0.005), c5_cur + 0.001))
    
    x0 = np.concatenate([h, [c5_cur + 1e-4]])
    
    try:
        res = minimize(objective, x0, jac=objective_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or (res.x is not None and np.all(np.isfinite(res.x))):
            h_new = res.x[:n]
            # Project to ensure feasibility
            h_new = project(h_new, target)
            c5_new = exact_c5(h_new)
            if c5_new < c5_cur - 1e-12 and np.all(h_new >= -1e-10) and np.all(h_new <= 1 + 1e-10):
                return h_new, c5_new
    except Exception:
        pass
    return None, None

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

        cnt = int(np.sum(mask))

    shift = r - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            for i in range(shift, n):
                g[i] += (1.0 - h[i - shift])
            m = n - shift
            for i in range(m):
                g[i] += -h[i + shift]
    else:
        s = -shift
        m = n - s
        for i in range(m):
            g[i] += (1.0 - h[i + s])
        for i in range(s, n):
            g[i] += -h[i - s]
    return g

        return x[-1]

    x0[n] = c5_cur + 1e-5

    constraints = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]

    try:
        res = minimize(obj_fun, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project(res.x[:n], target)
            c5_new = exact_c5(h_new)
            if np.isfinite(c5_new) and c5_new < c5_cur - 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return None, None

def slsqp_loop(h, deadline, K_list=(8, 12, 20, 30), radius_init=0.12, maxiter_init=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    ki = 0
    maxiter = maxiter_init
    while time.time() < deadline:
        scores = np.correlate(h, 1.0 - h, mode='full')
        s_max = float(np.max(scores))
        K = K_list[ki % len(K_list)]
        K = min(K, 2 * n - 1)
        active = np.argsort(scores)[-K:]
        active = sorted(active.tolist())
        h_new, c5_new = slsqp_step(h, active, n, radius, maxiter)
        if h_new is not None and c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.45)
            maxiter = min(maxiter + 5, 80)
        else:
            stall += 1
            ki += 1
            radius *= 0.82
            maxiter = max(20, maxiter - 5)
            if radius < 0.015:
                break
            if stall > 25:
                rng = np.random.default_rng((ki * 7919 + 13) & 0x7FFFFFFF)
                for _attempt in range(5):
                    if time.time() > deadline:
                        break
                    scale = rng.uniform(0.03, 0.10)
                    direction = rng.normal(0, 1, n)
                    direction -= direction.mean()
                    h_k = project(h + scale * direction, target)
                    h_new, c5_new = slsqp_step(h_k, active, n, 0.10, 30)
                    if h_new is not None and c5_new < best_c5 - 1e-12:
                        h = h_new
                        cur_c5 = c5_new
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                        radius = 0.12
                        maxiter = 50
                        break
                else:
                    if stall > 50:
                        break
                    radius = max(0.04, radius * 0.5)
                    stall = 0
    return best_h, best_c5

def lp_step(h, scores, n, radius):
    dx = 2.0 / n
    G = grad_corr_vec(h, list(range(2 * n - 1)), n) * dx
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_loop(h, deadline, radius_init=0.20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = lp_step(h, scores, n, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.50)
        else:
            stall += 1
            radius *= 0.75
    return best_h, best_c5

def lse_grad(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s_max = float(np.max(scores))
    v = beta * (scores - s_max)
    v = np.clip(v, -50.0, 0.0)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        g = grad_corr(h, r, n)
        grad += w[r] * g
    grad *= dx
    grad -= np.mean(grad)
    return grad

def smooth_loop(h, deadline, betas=(40, 100, 250, 600), lr_init=0.03):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.82
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 50:
            bi += 1
            if bi > 30:
                break
            stall = 0
            mom = np.zeros(n)
            lr = lr_init
        beta = betas[bi % len(betas)]
        d = lse_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            bi += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = line_search(h, lr * mom, dx, lr * 2.5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * lr * mom
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.78
    return best_h, best_c5

def resample_to(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def pair_refine(h, deadline, n_random=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(12345)
    n_pass = 0
    while time.time() < deadline and n_pass < 100:
        n_pass += 1
        improved = False
        for _ in range(n_random):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if h[i] <= 1e-10 or h[i] >= 1 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1 - 1e-10:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = project(h, target)
                cur = exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def combined_local(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.15
    lr = 0.02
    mom = np.zeros(n)
    stall = 0
    mode = 0
    while time.time() < deadline:
        if stall > 30:
            mode = 1 - mode
            stall = 0
            radius = 0.15
            lr = 0.02
            mom = np.zeros(n)
        if mode == 0:
            scores = np.correlate(h, 1.0 - h, mode='full')
            p = lp_step(h, scores, n, radius)
            if p is not None and np.max(np.abs(p)) > 1e-13:
                alpha, new_c5 = line_search(h, p, 2.0 / n, radius)
                if new_c5 < cur_c5 - 1e-12:
                    h = h + alpha * p
                    h = project(h, target)
                    cur_c5 = exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.06, 0.45)
                    continue
            stall += 1
            radius *= 0.8
        else:
            for beta in (50, 150):
                d = lse_grad(h, 2.0 / n, beta)
                dn = np.linalg.norm(d)
                if dn < 1e-13:
                    continue
                d = d / dn
                mom = 0.82 * mom + 0.18 * d
                mn = np.linalg.norm(mom)
                if mn < 1e-13:
                    continue
                mom = mom / mn
                alpha, new_c5 = line_search(h, lr * mom, 2.0 / n, lr * 2.5)
                if new_c5 < cur_c5 - 1e-12:
                    h = h + alpha * lr * mom
                    h = project(h, target)
                    cur_c5 = exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    lr = min(lr * 1.05, 0.30)
                    break
            else:
                stall += 1
                lr *= 0.8
    return best_h, best_c5

def multi_resolution(h, deadline, n_best_base):
    best_h = h.copy()
    best_c5 = exact_c5(h)
    offsets = sorted(range(-24, 25, 2), key=lambda x: (abs(x), x))
    tried = set()
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n_best_base + dn
        if n_try < 60 or n_try > 250:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample_to(best_h, n_try)
        sub_dl = min(deadline, time.time() + 15.0)
        h_r, c5_r = slsqp_loop(h_r, sub_dl, K_list=(8, 12, 20), radius_init=0.10, maxiter_init=40)
        if c5_r < best_c5 - 1e-12 and time.time() < deadline - 10:
            sub_dl2 = min(deadline, time.time() + 10.0)
            h_r, c5_r = pair_refine(h_r, sub_dl2, n_random=100)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)

    # Phase 1: Combined LP + smooth (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = combined_local(h, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP epigraph (35%)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = slsqp_loop(best_h, d2, K_list=(6, 10, 15, 25), radius_init=0.10, maxiter_init=50)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP refinement (12%)
    d3 = t0 + budget_s * 0.62
    h3, c5_3 = lp_loop(best_h, d3, radius_init=0.15)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Smooth gradient (8%)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = smooth_loop(best_h, d4, betas=(30, 80, 200, 500), lr_init=0.02)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (20%)
    d5 = t0 + budget_s * 0.90
    h5, c5_5 = multi_resolution(best_h, d5, len(best_h))
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final pair refinement (8%)
    d6 = t0 + budget_s - 2.0
    h6, c5_6 = pair_refine(best_h, d6, n_random=300)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final SLSQP polish if time remains
    if time.time() < deadline - 5:
        h7, c5_7 = slsqp_loop(best_h, deadline, K_list=(5, 8, 12), radius_init=0.05, maxiter_init=30)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, len(best_h)
