import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _balance(x, target):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    y = np.clip(x - t, 0.0, 1.0)
    resid = target - y.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = y < 1.0 - 1e-10
        else:
            mask = y > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            y[mask] += resid / cnt
    return np.clip(y, 0.0, 1.0)

def _safe_logit(x, eps=1e-6):
    x = np.clip(x, eps, 1.0 - eps)
    return np.log(x / (1.0 - x))

def _safe_sigma(t):
    # Stable sigmoid
    out = np.zeros_like(t)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def _solve_mu(q, target):
    """Find mu such that sum(sigma(q - mu)) = target."""
    n = len(q)
    def f(mu):
        return _safe_sigma(q - mu).sum() - target
    # f is decreasing in mu
    lo, hi = float(q.min()) - 20.0, float(q.max()) + 20.0
    flo, fhi = f(lo), f(hi)
    if flo * fhi > 0:
        # Expand
        for _ in range(10):
            lo -= 10.0
            hi += 10.0
            flo, fhi = f(lo), f(hi)
            if flo * fhi < 0:
                break
    for _ in range(100):
        mid = (lo + hi) / 2.0
        fm = f(mid)
        if fm > 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _q_to_h(q, target):
    mu = _solve_mu(q, target)
    return _safe_sigma(q - mu)

def _h_to_q(h, mu=0.0):
    return _safe_logit(h) + mu

def _grad_q(h, q, target, beta):
    """Compute gradient of smooth objective w.r.t. q."""
    n = len(h)
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    
    # Compute grad_h
    dx = 2.0 / n
    grad_h = np.zeros(n)
    n_lags = 2 * n - 1
    for j in range(n):
        g = 0.0
        for d in range(-(n-1), n):
            idx = n - 1 + d
            wd = w[idx]
            if wd < 1e-16:
                continue
            if d == 0:
                g += wd * (1.0 - 2.0 * h[j])
            elif d > 0:
                if j + d < n:
                    g += wd * (1.0 - h[j + d])
                if j - d >= 0:
                    g += wd * (-h[j - d])
            else:
                g_ = -d
                if j >= g_:
                    g += wd * (1.0 - h[j - g_])
                if j + g_ < n:
                    g += wd * (-h[j + g_])
        grad_h[j] = g * dx
    
    # Convert to grad_q
    s = h * (1.0 - h)
    s_sum = s.sum()
    if s_sum < 1e-15:
        return np.zeros(n)
    rho = (s * grad_h).sum() / s_sum
    return s * (grad_h - rho)

def _smooth_obj_grad(q, target, beta):
    h = _q_to_h(q, target)
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = beta * (scores - max_s)
    shifted = np.clip(shifted, -500, 0)
    obj = max_s + np.log(np.mean(np.exp(shifted))) / beta
    grad = _grad_q(h, q, target, beta)
    return obj, grad

def _logistic_optimize(h0, deadline, beta_min=None, beta_max=None, n_phases=6):
    """Optimize using L-BFGS-B in logistic parameterization."""
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    n_lags = 2 * n - 1
    if beta_min is None:
        beta_min = max(10.0, 0.15 * n_lags)
    if beta_max is None:
        beta_max = max(beta_min * 5, 3.0 * n_lags)
    
    betas = np.geomspace(beta_min, beta_max, n_phases)
    
    # Initial q
    h_init = np.clip(h0, 1e-6, 1.0 - 1e-6)
    q = _safe_logit(h_init)
    
    B_WALL = 14.0
    bounds = [(-B_WALL, B_WALL)] * n
    
    for phase in range(len(betas)):
        if time.time() > deadline:
            break
        beta = betas[phase]
        
        def objfun(qv):
            hv = _q_to_h(qv, target)
            sv = _scores(hv)
            ms = np.max(sv)
            sh = np.clip(beta * (sv - ms), -500, 0)
            return ms + np.log(np.mean(np.exp(sh))) / beta
        
        def gradfun(qv):
            hv = _q_to_h(qv, target)
            return _grad_q(hv, qv, target, beta)
        
        maxiter = max(100, int(400 * np.sqrt(128.0 / n)))
        
        try:
            res = minimize(objfun, q, jac=gradfun, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
            q = np.clip(res.x, -B_WALL, B_WALL)
        except Exception:
            pass
        
        # Evaluate and update
        h_new = _q_to_h(q, target)
        h_new = _balance(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Update q to match
            h_clipped = np.clip(h_new, 1e-6, 1.0 - 1e-6)
            q = _safe_logit(h_clipped)
    
    # Final exact polish: line search in h-space
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    max_s = np.max(scores)
    shifted = np.clip(beta * (scores - max_s), -500, 0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(2*n-1)
        w[n-1] = 1.0
    else:
        w /= wsum
    
    grad = np.zeros(n)
    # d = 0
    grad += w[n-1] * dx * (1.0 - 2.0*h)
    
    for d in range(1, n):

        wd_neg = w[n-1-d]  # weight for negative lag -d
