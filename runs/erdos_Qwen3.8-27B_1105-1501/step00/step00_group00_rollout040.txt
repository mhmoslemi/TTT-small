from scipy.optimize import minimize
import time

def compute_corr(h):
    """Compute full correlation of h with 1-h."""
    n = len(h)
    corr = np.zeros(2*n - 1)
    for m in range(-(n-1), n):
        # output[n-1+m] = sum_j h[j+m] * (1-h[j]) for valid j
        lo = max(0, -m)
        hi = min(n, n - m)  # j+m < n means j < n-m
        if lo < hi:
            corr[n-1+m] = np.sum(h[lo+m:hi+m] * (1 - h[lo:hi]))
    return corr

    return np.max(compute_corr(h)) * dx

    m = np.max(corr)
