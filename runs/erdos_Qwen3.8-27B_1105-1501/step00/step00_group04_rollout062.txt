def is_feasible(h, tol=1e-8):

def project_feasible(y):
    y = np.asarray(y, dtype=float).ravel()
    if y.size == 0:
        return np.array([0.5, 0.5], dtype=float)
    if not np.all(np.isfinite(y)):

    target = 0.5 * float(n)

    hi = float(np.max(y))
    if hi < lo:

    for _ in range(50):
        mid = 0.5 * (lo + hi)
        z = np.clip(y - mid, 0.0, 1.0)

    h = np.asarray(h, dtype=float).ravel()

def softmax_weights(scores, T):
    e = np.asarray(scores, dtype=float).ravel()
    n = e.size
    if n == 0:
        return np.array([1.0], dtype=float), 0.0
    if not np.all(np.isfinite(e)):
        w = np.zeros(n, dtype=float)
        w[int(np.argmax(e))] = 1.0
        return w, float(np.max(e))

    base = float(np.max(e))
    if T <= 1e-12:
        w = np.zeros(n, dtype=float)
        w[int(np.argmax(e))] = 1.0
        return w, base

    z = (e - base) / T
    ex = np.exp(z)
    S = float(np.sum(ex))
    if not np.isfinite(S) or S <= 0.0:
        w = np.zeros(n, dtype=float)
        w[int(np.argmax(e))] = 1.0
        return w, base

    w = ex / S
    if not np.all(np.isfinite(w)) or float(np.sum(w)) <= 0.0:
        w = np.zeros(n, dtype=float)
        w[int(np.argmax(e))] = 1.0
        return w, base

    return w, base + T * math.log(S)


def lag_weights_to_grad(weights, h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g

    w = np.asarray(weights, dtype=float).ravel()
    sw = float(np.sum(w))
    if not np.all(np.isfinite(w)) or sw <= 1e-15:
        g += (2.0 / float(n)) * (1.0 - 2.0 * h)
        return g

    w = w / sw
    center = n - 1
    scale = 2.0 / float(n)

    if w.size > center:
        w0 = float(w[center])
        if w0 > 0.0:
            g += w0 * scale * (1.0 - 2.0 * h)

    for d in range(1, n):
        L = n - d
        wp = 0.0
        if center + d < w.size:
            wp = float(w[center + d])
        if wp > 0.0:
            g[:L] += wp * (-h[d:])
            g[d:] += wp * (1.0 - h[:L])

        wn = 0.0
        if center - d >= 0:
            wn = float(w[center - d])
        if wn > 0.0:
            g[:L] += wn * (1.0 - h[d:])
            g[d:] += wn * (-h[:L])

    return g

def binarize_feasible(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.array([], dtype=float)

    target = 0.5 * float(n)
    order = np.argsort(h, kind='stable')
    z = np.zeros(n, dtype=float)

    k = int(round(target))
    if abs(target - float(k)) < 1e-12:
        cnt = min(k, n)
        if cnt > 0:
            z[order[-cnt:]] = 1.0
    else:
        low = int(math.floor(target))
        low = min(low, n)
        if low > 0:
            z[order[-low:]] = 1.0
        fill = float(target - low)
        if fill > 1e-14 and low < n:
            if low == 0:
                idx = order[-1]
            else:
                idx = order[-low - 1]
            z[idx] = min(1.0, max(0.0, fill))

    return project_feasible(z)

        return project_feasible(np.full(max(2, new_n), 0.5))

    x_old = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    x_new = (np.arange(new_n, dtype=float) + 0.5) * (2.0 / float(new_n))
    y = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_feasible(y)


def generate_variants(base, rng, n_list, max_variants=12):
    base = np.asarray(base, dtype=float).ravel()
    if base.size < 2:
        base = np.full(70, 0.5, dtype=float)

    cand = []

    cand.append(binarize_feasible(base))

    for s in (0.85, 1.0, 1.15, 1.35, 1.60):
        cand.append(project_feasible(0.5 + s * (base - 0.5)))

    for sig in (0.02, 0.05, 0.10, 0.18):
        cand.append(project_feasible(base + sig * rng.normal(size=base.size)))

    for n in n_list:
        n = int(n)
        if n < 2:
            continue
        cand.append(project_feasible(rng.rand(n)))
        if n != base.size:
            cand.append(resample_feasible(base, n))

    scored = []
    for i, c in enumerate(cand):
        c = project_feasible(c)
        s = score_exact(c)
        if np.isfinite(s):
            scored.append((float(s), i, c))

    scored.sort()

    res = []
    for s, i, c in scored:
        if len(res) >= max_variants:
            break
        dup = False
        for d in res:
            if d.size == c.size and float(np.sum(np.abs(d - c))) < 1e-7:
                dup = True
                break
        if not dup:
            res.append(c)
    return res


def local_refine(h_init, allowed_seconds, rng):
    t0 = time.time()
    budget = float(allowed_seconds)
    margin = min(2.5, max(0.1, budget * 0.02))
    stop = t0 + max(0.0, budget - margin)

    cur = project_feasible(h_init)
    cur_score = score_exact(cur)
    if not np.isfinite(cur_score):
        cur = project_feasible(np.full(70, 0.5, dtype=float))
        cur_score = score_exact(cur)
    if not np.isfinite(cur_score):
        return cur, float('inf')

    best = cur.copy()
    best_score = float(cur_score)

    temps = [0.02, 0.008, 0.003, 0.001, 0.0003, 0.00012]
    base_lens = [0.01, 0.006, 0.003, 0.0015, 0.0008, 0.0004]
    mults = [3.0, 1.0, 0.5, 0.25, 0.12, 0.06]

    for p in range(len(temps)):
        if time.time() > stop - 1.0:
            break

        T = temps[p]
        step_factor = 1.0
        no_improve = 0
        it = 0
        last_grad = np.zeros(cur.size, dtype=float)

        while time.time() < stop - 1.0 and no_improve < 220:
            it += 1

            scores = loss_values(cur)
            if scores.size == 0 or not np.all(np.isfinite(scores)):
                cur = best.copy()
                cur_score = best_score
                no_improve += 1
                continue

            base_score = float(np.max(scores))
            if np.isfinite(base_score) and base_score < cur_score - 1e-13:
                cur_score = float(base_score)

            weights, smooth_cur = softmax_weights(scores, T)
            grad = lag_weights_to_grad(weights, cur)
            if grad.size != cur.size or not np.all(np.isfinite(grad)):
                grad = np.zeros(cur.size, dtype=float)
            last_grad = grad

            gnorm = float(np.sqrt(np.mean(grad * grad))) if grad.size else 0.0
            accepted = False

            if not np.isfinite(gnorm) or gnorm < 1e-9:
                no_improve += 1
                if (no_improve == 1 or no_improve % 50 == 0) and time.time() < stop - 4.0:
                    pz, ps, ok = pair_search(cur, cur_score, grad, 90, 2.0, rng)
                    if ok:
                        pz = project_feasible(pz)
                        ps2 = score_exact(pz)
                        if np.isfinite(ps2) and ps2 < cur_score - 1e-13:
                            cur = pz
                            cur_score = float(ps2)
                            no_improve = 0
                            if cur_score < best_score - 1e-13:
                                best = cur.copy()
                                best_score = cur_score
                continue

            direction = -grad / gnorm
            current_len = base_lens[p] * step_factor
            if current_len < 1e-8:
                current_len = 1e-8

            for m in mults:
                step = current_len * m
                if step <= 1e-10:
                    break
                if time.time() > stop - 1.0:
                    break

                z = project_feasible(cur + step * direction)
                cz = score_exact(z)
                if not np.isfinite(cz):
                    continue

                if cz < cur_score - 1e-13:
                    cur = z
                    cur_score = float(cz)
                    accepted = True
                    no_improve = 0
                    if cur_score < best_score - 1e-13:
                        best = cur.copy()
                        best_score = cur_score
                    break

            if not accepted:
                no_improve += 1

                if no_improve % 30 == 0 and step_factor > 0.12:
                    step_factor *= 0.7

                if no_improve % 120 == 0 and time.time() < stop - 4.0:
                    rz = project_feasible(cur + 0.02 * rng.normal(size=cur.size))
                    rzs = score_exact(rz)
                    if np.isfinite(rzs) and rzs < cur_score - 1e-13:
                        cur = rz
                        cur_score = float(rzs)
                        no_improve = 0
                        if cur_score < best_score - 1e-13:
                            best = cur.copy()
                            best_score = cur_score

                if (no_improve == 20 or no_improve % 60 == 0) and time.time() < stop - 5.0:
                    pz, ps, ok = pair_search(cur, cur_score, grad, 110, 2.5, rng)
                    if ok:
                        pz = project_feasible(pz)
                        ps2 = score_exact(pz)
                        if np.isfinite(ps2) and ps2 < cur_score - 1e-13:
                            cur = pz
                            cur_score = float(ps2)
                            no_improve = 0
                            step_factor = 1.0
                            if cur_score < best_score - 1e-13:
                                best = cur.copy()
                                best_score = cur_score

                if cur_score > best_score + 1e-7:
                    cur = best.copy()
                    cur_score = best_score
                    no_improve = 0

            if no_improve > 180:
                break

        if time.time() < stop - 5.0:
            pz, ps, ok = pair_search(cur, cur_score, last_grad, 150, 3.0, rng)
            if ok:
                pz = project_feasible(pz)
                ps2 = score_exact(pz)
                if np.isfinite(ps2) and ps2 < cur_score - 1e-13:
                    cur = pz
                    cur_score = float(ps2)
                    if cur_score < best_score - 1e-13:
                        best = cur.copy()
                        best_score = cur_score

    rem = stop - time.time()
    if rem > 6.0:
        last_grad = None
        for _ in range(40):
            if time.time() > stop - 1.0:
                break
            if last_grad is None:
                scores = loss_values(best)
                if scores.size == 0 or not np.all(np.isfinite(scores)):
                    break
                weights, _ = softmax_weights(scores, 1e-4)
                last_grad = lag_weights_to_grad(weights, best)

            pz, ps, ok = pair_search(best, best_score, last_grad, 70, 1.2, rng)
            if ok:
                pz = project_feasible(pz)
                ps2 = score_exact(pz)
                if np.isfinite(ps2) and ps2 < best_score - 1e-13:
                    best = pz
                    best_score = float(ps2)
                    scores = loss_values(best)
                    if scores.size == 0 or not np.all(np.isfinite(scores)):
                        break
                    weights, _ = softmax_weights(scores, 1e-4)
                    last_grad = lag_weights_to_grad(weights, best)
                else:
                    break
            else:
                break

    return best.copy(), float(best_score)


def run(seed=42, budget_s=1100, **kwargs):
    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    if budget <= 0.0:
        budget = 30.0

    margin = min(5.0, max(0.5, budget * 0.02))
    t_start = time.time()
    deadline = t_start + max(1.0, budget - margin)

    rng = np.random.RandomState(seed_int % (2**31 - 1))

    best_h = None
    best_c = float('inf')

    parent = globals().get('initial_h_values')
    if parent is not None:
        try:
            hpar = np.asarray(parent, dtype=float).ravel().astype(float, copy=True)
            if hpar.size >= 2 and np.all(np.isfinite(hpar)):
                hpar = project_feasible(hpar)
                cpar = score_exact(hpar)
                if np.isfinite(cpar):
                    best_h = hpar
                    best_c = float(cpar)
        except Exception:
            pass

    if best_h is None:
        for attempt in range(35):
            if attempt == 0:
                h = np.full(70, 0.5, dtype=float)
            elif attempt == 1:
                h = binarize_feasible(np.full(70, 0.5, dtype=float))
            else:
                h = project_feasible(rng.rand(70))
            c = score_exact(h)
            if np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = h

        if best_h is None:
            best_h = project_feasible(np.full(70, 0.5, dtype=float))
            best_c = score_exact(best_h)

    remaining = deadline - time.time()
    if remaining > 20.0:
        alloc = min(remaining * 0.45, 300.0)
        h1, c1 = local_refine(best_h, alloc, rng)
        if np.isfinite(c1) and c1 < best_c - 1e-13:
            best_h = h1
            best_c = float(c1)

    loop_count = 0
    stagnation = 0
    while time.time() < deadline - 40.0 and loop_count < 10:
        loop_count += 1

        n0 = int(best_h.size)
        ns = []
        start_n = max(30, n0 - 14)
        end_n = min(130, n0 + 28)
        for nn in range(start_n, end_n + 1, 2):
            ns.append(nn)

        variants = generate_variants(best_h, rng, ns[:8], 10)
        for z in variants:
            if time.time() > deadline - 30.0:
                break

            if z.size == best_h.size and float(np.sum(np.abs(z - best_h))) < 1e-9:
                continue

            c0 = score_exact(z)
            if not np.isfinite(c0):
                continue

            remaining = deadline - time.time()
            alloc = min(remaining * 0.12, 55.0)
            if alloc < 8.0:
                break

            h2, c2 = local_refine(z, alloc, rng)
            if np.isfinite(c2) and c2 < best_c - 1e-13:
                best_h = h2
                best_c = float(c2)
                stagnation = 0
            else:
                stagnation += 1

        if stagnation >= 8:
            break

    remaining = deadline - time.time()
    if remaining > 10.0:
        alloc = max(5.0, min(remaining * 0.5, 120.0))
        h2, c2 = local_refine(best_h, alloc, rng)
        if np.isfinite(c2) and c2 < best_c - 1e-13:
            best_h = h2
            best_c = float(c2)

    final_h = project_feasible(best_h)
    final_c = score_exact(final_h)

    if not np.isfinite(final_c):
        final_h = project_feasible(np.full(70, 0.5, dtype=float))
        final_c = score_exact(final_h)

    if is_feasible(best_h, 1e-9) and np.isfinite(best_c) and best_c < final_c - 1e-12:
        final_h = best_h
        final_c = best_c
    else:
        final_h = final_h
        final_c = final_c

    final_h = np.clip(final_h, 0.0, 1.0)
    final_c = score_exact(final_h)

    if not np.isfinite(final_c):
        final_h = project_feasible(np.full(70, 0.5, dtype=float))
        final_c = score_exact(final_h)

    return final_h, float(final_c), int(final_h.size)
