import numpy as np
import time
import scipy.optimize as sopt


def project_to_feasible(raw, n):
    target = n / 2.0
    lo = float(raw.min()) - 2.0
    hi = float(raw.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(raw - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(raw - 0.5 * (lo + hi), 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def make_basis(n, n_modes, kind=0, seed=0):
    idx = np.arange(n, dtype=float)
    B = np.zeros((n, n_modes))
    if kind == 0:
        for j in range(n_modes):
            j1 = j + 1.0
            B[:, j] = np.cos(np.pi * (2.0 * idx + 1.0) * j1 / (2.0 * n))
            B[:, j] -= B[:, j].mean()
            nm = np.linalg.norm(B[:, j])
            if nm > 1e-12:
                B[:, j] /= nm
    elif kind == 1:
        rng = np.random.RandomState(seed)
        for j in range(n_modes):
            k = j + 1.0
            ph = rng.uniform(0, 2 * np.pi)
            B[:, j] = np.cos(2.0 * np.pi * k * idx / n + ph)
            B[:, j] -= B[:, j].mean()
            nm = np.linalg.norm(B[:, j])
            if nm > 1e-12:
                B[:, j] /= nm
    elif kind == 2:
        rng = np.random.RandomState(seed)
        for j in range(n_modes):
            vec = np.zeros(n)
            for m in range(3):
                k = m + 1.0
                vec += rng.randn() * np.cos(2.0 * np.pi * k * idx / n + rng.uniform(0, 2 * np.pi))
            B[:, j] = vec
            B[:, j] -= B[:, j].mean()
            nm = np.linalg.norm(B[:, j])
            if nm > 1e-12:
                B[:, j] /= nm
    return B


class ObjWrap:
    def __init__(self, h_base, B, n):
        self.h_base = h_base
        self.B = B
        self.n = n

    def __call__(self, z):
        raw = self.h_base + self.B.dot(z)
        h = project_to_feasible(raw, self.n)
        return compute_c5(h)


def de_phase(h_base, n, seed, budget_s, t0):
    best_h = h_base.copy()
    best_c5 = compute_c5(best_h)
    n_modes = min(10, n - 1)
    for kind in range(3):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.6:
            break
        B = make_basis(n, n_modes, kind=kind, seed=seed + kind * 13)
        wrap = ObjWrap(h_base, B, n)
        bounds = [(-1.0, 1.0)] * n_modes
        try:
            res = sopt.differential_evolution(
                wrap, bounds, seed=seed + kind, maxiter=60, tol=1e-8,
                popsize=10, polish=True, init='latinhypercube')
            if res.fun < best_c5:
                best_c5 = float(res.fun)
                best_h = project_to_feasible(h_base + B.dot(res.x), n)
        except Exception:
            pass
    return best_h, best_c5


def nm_phase(h_base, n, seed, budget_s, t0):

    best_c5 = compute_c5(best_h)
    n_modes = min(10, n - 1)
