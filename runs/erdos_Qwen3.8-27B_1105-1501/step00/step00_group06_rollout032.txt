import time

def compute_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def get_gradient(corr_lag, h, n):
    """Gradient of c_shift with respect to h."""
    shift = corr_lag - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        ip = i + shift
        im = i - shift
        if 0 <= ip < n:
            grad[i] += (1.0 - h[ip])
        if 0 <= im < n:
            grad[i] += -h[im]
    return grad

def feasible_lambda(h, d):

    lam_hi = 0.0

        if di > 1e-14:

            lam_hi = min(lam_hi, (1.0 - h[i]) / di)

    # Golden section search
    gr = (1.0 - np.sqrt(5.0)) / 2.0  # ≈ -0.618, so 1/gr ≈ 1.618
    # Actually golden ratio conjugate: (sqrt(5)-1)/2 ≈ 0.618
    inv_gr = (np.sqrt(5.0) - 1.0) / 2.0  # ≈ 0.618

    c = b - inv_gr * (b - a)
    d_pt = a + inv_gr * (b - a)
    fc = compute_c5(h + c * d)
    fd = compute_c5(h + d_pt * d)
    
    for _ in range(max_iters):
        if fc < fd:
            b = d_pt
            d_pt = c
            fd = fc
            c = b - inv_gr * (b - a)
            fc = compute_c5(h + c * d)
        else:
            a = c
            c = d_pt
            fc = fd
            d_pt = a + inv_gr * (b - a)
            fd = compute_c5(h + d_pt * d)
    
    lam_opt = (a + b) / 2.0
    # Also check endpoints and a few extra points
    candidates = [lam_lo, lam_hi, lam_opt, (lam_lo + lam_opt)/2, (lam_opt + lam_hi)/2]
    best_val = float('inf')
    best_lam = lam_opt
    for lam in candidates:

            val = compute_c5(h + lam * d)
            if val < best_val:
                best_val = val
                best_lam = lam

    h = np.clip(h, 0.0, 1.0)

            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                h[0] += diff2
                h = np.clip(h, 0.0, 1.0)
    return h

where i ranges over [0, n-1] and i+shift ranges over [0, n-1].

def compute_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def all_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_gradient(lag_idx, h):
    n = len(h)
    shift = lag_idx - (n - 1)
    idx = np.arange(n)
    grad = np.zeros(n)
    ip = idx + shift
    valid_p = (ip >= 0) & (ip < n)
    grad[valid_p] += (1.0 - h[ip[valid_p]])
    im = idx - shift
    valid_m = (im >= 0) & (im < n)
    grad[valid_m] += -h[im[valid_m]]
    return grad

def lambda_bounds(h, d):
    n = len(h)
    lo = 0.0
    hi = 0.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        lo = max(lo, float(np.max(-h[pos] / d[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))

        lo = max(lo, float(np.max((1.0 - h[neg]) / d[neg])))
        hi = min(hi, float(np.max(-h[neg] / d[neg])))
    return lo, hi
