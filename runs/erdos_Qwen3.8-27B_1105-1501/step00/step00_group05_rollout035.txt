import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def active_gradient(h, active_shifts):
    n = len(h)
    g = np.zeros(n)
    for s in active_shifts:
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return g

def feasible_delta(h, p, q):
    """Return (delta_min, delta_max) for move h[p]+=d, h[q]-=d"""
    d_min = max(-h[p], h[q] - 1.0)
    d_max = min(h[q], 1.0 - h[p])
    return d_min, d_max

def line_search(h, p, q, c5_current):
    """Find best delta for move h[p]+=d, h[q]-=d"""
    d_min, d_max = feasible_delta(h, p, q)
    if d_min >= d_max - 1e-12:
        return 0, c5_current
    # Try a grid
    n_grid = 20
    deltas = np.linspace(d_min + 1e-9, d_max - 1e-9, n_grid)
    best_d = 0
    best_val = c5_current
    for d in deltas:
        h[p] += d
        h[q] -= d
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_d = d
        h[p] -= d
        h[q] += d
    # Refine around best
    if best_d > 0:
        span = (d_max - d_min) / n_grid
        ref_deltas = np.linspace(max(d_min, best_d - span), min(d_max, best_d + span), 20)
        for d in ref_deltas:
            h[p] += d
            h[q] -= d
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_d = d
            h[p] -= d
            h[q] += d
    return best_d, best_val

def local_optimize(h, time_limit_s, verbose=False):
    n = len(h)
    start = time.time()
    c5 = compute_c5(h)
    
    no_improve_count = 0
    max_no_improve = 5
    
    while time.time() - start < time_limit_s:
        c5_arr = compute_c5_array(h)
        c5_max = c5_arr.max()
        
        if c5_max < c5 - 1e-10:
            c5 = c5_max
            no_improve_count = 0
        
        # Active set
        tol = 1e-4
        active_mask = c5_arr >= c5_max - tol
        active_shifts = np.where(active_mask)[0].astype(float) - (n - 1)
        
        # Gradient
        g = active_gradient(h, active_shifts.astype(int))
        
        # Project to zero-sum subspace
        g = g - np.mean(g)
        
        # Find best pair: minimize g[p] - g[q] (move from q to p)
        # i.e., p has lowest g, q has highest g
        # Try all pairs in order of (g[q] - g[p])
        order_asc = np.argsort(g)
        
        best_improve = 0
        best_move = None
        
        # Try pairs in order of promise
        tried = 0
        for i in range(n):
            p = order_asc[i]  # low gradient
            for j in range(n):
                q = order_asc[n - 1 - j]  # high gradient
                if p == q:
                    continue
                # Skip if not feasible
                d_min, d_max = feasible_delta(h, p, q)
                if d_min >= d_max - 1e-9:
                    continue
                # Quick check: derivative at 0
                deriv = g[p] - g[q]
                if deriv > 1e-6:
                    continue  # not improving direction
                tried += 1
                d, val = line_search(h, p, q, c5_max)
                if d != 0 and c5_max - val > best_improve:
                    best_improve = c5_max - val
                    best_move = (p, q, d)
                if best_improve > 1e-6:
                    break
            if best_move:
                break
        
        if best_move:
            p, q, d = best_move
            h[p] += d
            h[q] -= d
        else:
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                break

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def active_gradient(h, active_shifts):
    n = len(h)
    g = np.zeros(n)
    for s in active_shifts:
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return g

def feasible_delta(h, p, q):
    d_min = max(-h[p], h[q] - 1.0)
    d_max = min(h[q], 1.0 - h[p])
    return d_min, d_max

def line_search_fine(h, p, q, c5_current):
    d_min, d_max = feasible_delta(h, p, q)
    if d_min >= d_max - 1e-12:
        return 0.0, c5_current
    n_coarse = 25
    deltas = np.linspace(d_min + 1e-12, d_max - 1e-12, n_coarse)
    best_d = 0.0
    best_val = c5_current
    for d in deltas:
        h[p] += d
        h[q] -= d
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        if val < best_val:

            best_d = d

            best_h[interior] += diff / interior.sum()
        else:
            best_h += diff / n_final
        best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_final
