def make_feasible(y, target=None):
    """Project y onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.array(y, dtype=np.float64)
    # Handle non-finite
    y[~np.isfinite(y)] = 0.5
    
    # Bisection for lambda
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    
    def phi(lam):
        return float(np.sum(np.clip(y - lam, 0.0, 1.0))) - target
    
    # Ensure bracket
    if phi(lo) < 0:
        lo -= (target - phi(lo))  # should not happen but safety
    if phi(hi) > 0:
        hi += (phi(hi))  # safety
    
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if phi(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    lam = 0.5 * (lo + hi)
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix sum exactly
    diff = target - float(np.sum(z))
    if abs(diff) > 1e-12:
        # Adjust at a free component
        idx = int(np.argmax(z < 0.999)) if diff > 0 else int(np.argmax(z > 0.001))
        z[idx] += diff
        if z[idx] < 0: z[idx] = 0.0
        if z[idx] > 1: z[idx] = 1.0
    
    return z

def score_exact(h):
    """Compute C5 score exactly."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_max_and_grad(h, T):
    """Compute smooth-max surrogate value and gradient."""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    c = np.correlate(h, b, mode='full') * dx
    L = len(c)
    
    base = float(np.max(c))
    ex = np.exp((c - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        # Fallback: uniform over argmax
        weights = np.zeros(L)
        weights[np.isclose(c, base, atol=1e-15)] = 1.0 / np.sum(np.isclose(c, base, atol=1e-15))
    else:
        weights = ex / S
    
    smooth_val = base + T * math.log(max(S, 1e-300))
    
    g = np.zeros(n)
    center = n - 1
    
    # Center contribution
    wc = weights[center] * dx
    g += wc * (1.0 - 2.0 * h)
    
    # Non-zero lags
    for d in range(1, n):
        # Positive lag: p = center + d
        wp = weights[center + d] * dx
        g[:n-d] += wp * (-h[d:])
        g[d:] += wp * (1.0 - h[:n-d])
        
        # Negative lag: p = center - d
        wn = weights[center - d] * dx
        g[:n-d] += wn * (1.0 - h[d:])
        g[d:] += wn * (-h[:n-d])
    
    return smooth_val, g

def local_refine(h0, deadline, rng, T_start=0.1, T_end=1e-4):
    """Local optimization using smooth-max gradient descent + pair line search."""
    n = len(h0)
    cur = h0.copy()
    cur_score = score_exact(cur)
    best = cur.copy()
    best_score = cur_score
    
    n_iter = 0
    step_size = 0.02
    no_improve = 0
    max_no_improve = 50
    
    while time.time() < deadline:
        # Anneal temperature
        frac = min(1.0, n_iter / 200.0)
        T = T_start * (T_end / T_start) ** frac
        
        sv, g = smooth_max_and_grad(cur, T)
        
        # Gradient descent step
        g_norm = float(np.sqrt(np.mean(g**2)))
        if g_norm > 1e-14:
            direction = -g / g_norm
            accepted = False
            for alpha in [step_size, step_size*0.5, step_size*0.25, step_size*2.0]:
                h_trial = make_feasible(cur + alpha * direction)
                s_trial = score_exact(h_trial)
                if s_trial < cur_score - 1e-14:
                    cur = h_trial
                    cur_score = s_trial
                    accepted = True
                    break
            if accepted:

                    best = cur.copy()
                    best_score = cur_score
                no_improve = 0
            else:

                if no_improve > 10:
                    step_size *= 0.7

        if n_iter % 5 == 0 and no_improve > 5:

            for _ in range(8):

                i, j = int(i), int(j)
                lo_a = max(-cur[i], cur[j] - 1.0)

                if hi_a - lo_a < 1e-10:
                    continue
                # Evaluate at a few points

                best_s = cur_score

                    h_trial = cur.copy()
                    h_trial[i] += a
                    h_trial[j] -= a
                    if h_trial[i] < -1e-12 or h_trial[i] > 1+1e-12 or h_trial[j] < -1e-12 or h_trial[j] > 1+1e-12:

                    cur[j] -= best_alpha
                    cur_score = best_s

                        best_score = cur_score

            s_trial = score_exact(h_trial)
            if s_trial < cur_score + 0.002:  # accept slightly worse to escape
                cur = h_trial
                cur_score = s_trial
        
        n_iter += 1
        if no_improve > max_no_improve:
            break
    
    return best, best_score

def random_feasible(n, rng):

    n = len(h)

    idx = np.argsort(h)[::-1][:target]

    x_new = (np.arange(new_n) + 0.5) * dx_new
    y = np.interp(x_new, x_old, h)

import numpy as np

def make_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.array(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - float(np.sum(z))
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(z < 0.9999))
        else:
            idx = int(np.argmax(z > 0.0001))
        z[idx] += diff
        z[idx] = min(1.0, max(0.0, z[idx]))
    return z

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_max_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(c)
    base = float(np.max(c))
    ex = np.exp((c - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        weights = np.zeros(L)
        mask = np.isclose(c, base, atol=1e-14)
        weights[mask] = 1.0 / float(np.sum(mask))
    else:
        weights = ex / S
    smooth_val = base + T * math.log(max(S, 1e-300))
    g = np.zeros(n)
    center = n - 1
    wc = weights[center] * dx
    g += wc * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[center + d] * dx
        g[:n-d] += wp * (-h[d:])
        g[d:] += wp * (1.0 - h[:n-d])
        wn = weights[center - d] * dx
        g[:n-d] += wn * (1.0 - h[d:])
        g[d:] += wn * (-h[:n-d])
    return smooth_val, g

def pair_line_search(cur, i, j):
    hi_a = min(1.0 - cur[i], cur[j])
    lo_a = max(-cur[i], cur[j] - 1.0)
    if hi_a - lo_a < 1e-12:
        return None
    base_score = score_exact(cur)
    best_alpha = 0.0
    best_s = base_score
    alphas = np.linspace(lo_a, hi_a, 17)
    for a in alphas:
        cur[i] += a
        cur[j] -= a
        s = score_exact(cur)
        cur[i] -= a
        cur[j] += a
        if s < best_s - 1e-15:
            best_s = s
            best_alpha = a
    if best_alpha == 0.0:
        return None
    cur[i] += best_alpha
    cur[j] -= best_alpha
    return best_s

def binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    z = np.zeros(n)
    z[idx] = 1.0
    return z

def resample_to_resolution(h, new_n):
    n_old = len(h)
    if new_n == n_old:
        return h.copy()
    dx_old = 2.0 / n_old
    dx_new = 2.0 / new_n
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new
    y = np.interp(x_new, x_old, h)
    return make_feasible(y)
