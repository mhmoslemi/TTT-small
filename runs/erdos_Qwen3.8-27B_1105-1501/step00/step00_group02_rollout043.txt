_ORIENT_CACHE = {}


def is_feasible(h, tol=1e-7):
    if h.ndim != 1:
        return False
    n = h.size
    if n == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(h.min()) < -tol or float(h.max()) > 1.0 + tol:
        return False
    return abs(float(h.sum()) - 0.5 * n) <= tol


def finalize_sum(y, target):
    y = np.clip(y, 0.0, 1.0)
    for _ in range(200):
        res = float(target - y.sum())
        if abs(res) <= 1e-13:
            break
        if res > 0.0:
            idx = np.nonzero(y < 1.0)[0]
            if idx.size == 0:
                break
            caps = 1.0 - y[idx]
            delta = min(abs(res) / float(idx.size), float(np.min(caps)))
            if delta <= 0.0:
                break
            y[idx] += delta
        else:
            idx = np.nonzero(y > 0.0)[0]
            if idx.size == 0:
                break
            caps = y[idx]
            delta = min(abs(res) / float(idx.size), float(np.min(caps)))
            if delta <= 0.0:
                break
            y[idx] -= delta
    res = float(target - y.sum())
    if abs(res) > 1e-14:
        if res > 0.0:
            idx = int(np.argmax(1.0 - y))
            if 1.0 - y[idx] >= 0.0:
                y[idx] += min(res, 1.0 - y[idx])
        else:
            idx = int(np.argmax(y))
            if y[idx] >= 0.0:
                y[idx] += max(res, -y[idx])
    return np.clip(y, 0.0, 1.0)


def project_box_sum(z, target=None):
    z = np.asarray(z, dtype=float).ravel()
    n = z.size
    if n == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.full(n, 0.5)
    if target is None:
        target = 0.5 * n
    target = min(max(float(target), 0.0), float(n))

    zmin = float(np.min(z))
    zmax = float(np.max(z))
    lo = zmin - 2.0
    hi = zmax + 2.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    return finalize_sum(y, target)


def objective(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size

        return np.inf

def _compute_L_formula(h):
    n = h.size
    m = 2 * n - 1
    dx = 2.0 / n
    L = np.zeros((m, n), dtype=float)
    for r in range(m):
        k = r - (n - 1)

        a1 = max(0, k)
        b1 = min(n - 1, n - 1 + k)
        if a1 <= b1:
            L[r, a1:b1 + 1] += 1.0 - h[a1 - k:b1 - k + 1]

        a2 = max(0, -k)
        b2 = min(n - 1, n - 1 - k)
        if a2 <= b2:
            L[r, a2:b2 + 1] -= h[a2 + k:b2 + k + 1]

        L[r] *= dx
    return L


def _orientation_for_n(n):
    if n in _ORIENT_CACHE:
        return _ORIENT_CACHE[n]
    if n <= 1:
        _ORIENT_CACHE[n] = False
        return False

    z = np.linspace(0.25, 0.75, n)
    Lf = _compute_L_formula(z)
    x = n // 2
    e = 1e-6

    zp = z.copy()
    zm = z.copy()
    if z[x] + e <= 1.0:
        zp[x] += e
    else:
        zp[x] = 1.0 - e
    if z[x] - e >= 0.0:
        zm[x] -= e
    else:
        zm[x] = e

    denom = zp[x] - zm[x]
    if denom <= 0.0:
        rev = False
    else:
        dx = 2.0 / n
        bp = np.correlate(zp, 1.0 - zp, "full") * dx
        bm = np.correlate(zm, 1.0 - zm, "full") * dx
        grad = (bp - bm) / denom
        err_direct = float(np.sum((Lf[x] - grad) ** 2))
        err_reversed = float(np.sum((Lf[::-1][x] - grad) ** 2))
        rev = err_reversed < err_direct

    _ORIENT_CACHE[n] = rev
    return rev


def compute_base_L(h):

def pair_interval(h, p, q):
    xp = float(h[p])
    xq = float(h[q])
    lo = max(xp - 1.0, -xq)
    hi = min(xp, 1.0 - xq)
    if hi <= lo + 1e-12:
        return None
    return lo, hi


def evaluate_line_values(base, B, t, n, dx, diff):
    t = np.atleast_1d(np.asarray(t, dtype=float))
    M = base[None, :] + B[None, :] * t[:, None]
    t2 = t * t
    M[:, n - 1] += -2.0 * dx * t2
    if 0 < diff < n:
        i1 = n - 1 + diff
        i2 = n - 1 - diff
        M[:, i1] += dx * t2
        M[:, i2] += dx * t2
    return np.max(M, axis=1)


def add_unique_ts(candidates, tval, tol=1e-10):
    tval = float(tval)
    if not np.isfinite(tval):
        return
    for u in candidates:
        if abs(u - tval) <= tol:
            return
    candidates.append(tval)

    for _ in range(100):
        p = int(rng.randint(n))
        q = int(rng.randint(n))
        if p != q:
            pairs.append((p, q))

    seen = set()
    uniq = []
    for p, q in pairs:
        if p > q:
            p, q = q, p
        key = (p, q)
        if key not in seen:
            seen.add(key)
            uniq.append(key)
    return uniq[:800]


def find_improving_pair(h, base, L, n, dx, c, deadline, rng):
    candidates = []
    threshold = c - 1e-10

    if n <= 100:
        K = 81 if n <= 80 else 57
        for p in range(n - 1):
            if time.time() > deadline - 2.0:
                break
            hp = float(h[p])
            for q in range(p + 1, n):
                xq = float(h[q])
                lo = max(hp - 1.0, -xq)
                hi = min(hp, 1.0 - xq)
                if hi <= lo + 1e-8:
                    continue
                B = L[:, q] - L[:, p]
                diff = q - p
                t = np.linspace(lo, hi, K)
                vals = evaluate_line_values(base, B, t, n, dx, diff)
                im = int(np.argmin(vals))
                vmin = float(vals[im])
                if vmin < threshold:
                    candidates.append((-vmin, p, q, float(t[im])))
                    if len(candidates) > 120:
                        candidates.sort()
                        del candidates[40:]
    else:
        pairs = candidate_pairs_large(h, base, L, c, n, rng)
        K = 57
        for item in pairs:
            p = int(item[0])
            q = int(item[1])
            v, t = quick_line_min(base, L, h, p, q, n, dx, K)
            if v < threshold:
                candidates.append((-v, p, q, t))
                if len(candidates) > 120:
                    candidates.sort()
                    del candidates[40:]

    if not candidates:
        return None

    candidates.sort()
    seen = set()
    limit = 12

    for item in candidates:
        if time.time() > deadline - 2.0:
            break
        p = int(item[1])
        q = int(item[2])
        if -float(item[0]) >= threshold:
            break
        key = (min(p, q), max(p, q))
        if key in seen:
            continue
        seen.add(key)
        res = refine_and_apply(base, L, h, p, q, n, dx, c)
        if res is not None:
            return res
        limit -= 1
        if limit <= 0:
            break

    return None


def try_random_pairs(h, base, L, n, dx, c, deadline, rng, count):
    K = 33
    for _ in range(count):
        if time.time() > deadline - 1.0:
            break
        p = int(rng.randint(n))
        q = int(rng.randint(n))
        if p == q:
            continue
        if p > q:
            p, q = q, p
        v, _ = quick_line_min(base, L, h, p, q, n, dx, K)
        if v < c - 1e-10:
            res = refine_and_apply(base, L, h, p, q, n, dx, c)
            if res is not None:
                return res
    return None


def solution_is_better(c, n, c_best, n_best):
    if not np.isfinite(c):
        return False
    if not np.isfinite(c_best):
        return True
    if c < c_best - 1e-11:
        return True
    if abs(c - c_best) <= 1e-11 and n < n_best:
        return True
    return False


def local_polish(h, deadline, rng):
    h = np.asarray(h, dtype=float).ravel().copy()
    n = h.size
    if n == 0:
        return h, np.inf

    if not is_feasible(h, 1e-6):
        h = project_box_sum(h, 0.5 * n)

    c = objective(h)
    if not np.isfinite(c):
        return h, np.inf

    base, L = compute_base_L(h)
    best_h = h.copy()
    best_c = c
    no_improve = 0

    while time.time() < deadline - 1.0:
        res = find_improving_pair(h, base, L, n, 2.0 / n, c, deadline, rng)
        if res is not None:
            h, c = res
            base, L = compute_base_L(h)
            c2 = float(np.max(base))
            if np.isfinite(c2):
                c = c2
            if solution_is_better(c, n, best_c, best_h.size):
                best_c = c
                best_h = h.copy()
            no_improve = 0
            continue

        if n > 100:
            res = try_random_pairs(h, base, L, n, 2.0 / n, c, deadline, rng, 80)
            if res is not None:
                h, c = res
                base, L = compute_base_L(h)
                c2 = float(np.max(base))
                if np.isfinite(c2):
                    c = c2
                if solution_is_better(c, n, best_c, best_h.size):
                    best_c = c
                    best_h = h.copy()
                no_improve = 0
                continue

        no_improve += 1
        if no_improve >= (1 if n <= 100 else 2):
            break

    return best_h, best_c


def thue_morse_values(n):
    vals = np.empty(n, dtype=float)
    for i in range(n):
        x = i
        parity = 0
        while x:
            parity ^= 1
            x &= x - 1
        vals[i] = 1.0 if parity == 0 else 0.0
    return vals


def random_feasible(n, rng, style=0):
    if n <= 0:
        return np.array([], dtype=float)

    if style == 1:
        z = thue_morse_values(n)
    elif style == 2:
        arr = []
        while len(arr) < n:
            bl = int(rng.randint(1, 4))
            val = float(rng.rand())
            for _ in range(bl):
                arr.append(val)
            if len(arr) > n:
                arr = arr[:n]
        z = np.array(arr, dtype=float)
    else:
        z = np.zeros(n, dtype=float)
        if n % 2 == 0:
            ones = rng.choice(n, size=n // 2, replace=False)
            z[ones] = 1.0
        else:
            ones = rng.choice(n, size=n // 2, replace=False)
            z[ones] = 1.0
            zeros = np.setdiff1d(np.arange(n), ones)
            if zeros.size > 0:
                z[int(rng.choice(zeros))] = 0.5

    return project_box_sum(z, 0.5 * n)


def resample_vector(h, n_new):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0 or n_new < 2:
        return None
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    z = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_box_sum(z, 0.5 * n_new)


def jump_random(h, rng, steps):
    h = h.copy()
    n = h.size
    if n == 0:
        return None

    if steps is None:
        steps = int(rng.randint(2, 6))

    for _ in range(int(steps)):
        ok = False
        for _ in range(60):
            p = int(rng.randint(n))
            q = int(rng.randint(n))
            if p == q:
                continue
            iv = pair_interval(h, p, q)
            if iv is None:
                continue
            lo, hi = iv
            if hi - lo < 1e-6:
                continue
            t = lo + float(rng.rand()) * (hi - lo)
            if abs(t) < 1e-7:
                continue

            h[p] -= t
            h[q] += t

            if h[p] < 0.0 and h[p] > -1e-10:
                h[p] = 0.0
            if h[p] > 1.0 and h[p] < 1.0 + 1e-10:
                h[p] = 1.0
            if h[q] < 0.0 and h[q] > -1e-10:
                h[q] = 0.0
            if h[q] > 1.0 and h[q] < 1.0 + 1e-10:
                h[q] = 1.0

            if h.min() >= -1e-8 and h.max() <= 1.0 + 1e-8:
                ok = True
                break
        if not ok:
            break

    if not is_feasible(h, 1e-6):
        h = project_box_sum(h, 0.5 * n)
    if not is_feasible(h, 1e-6):
        return None
    if not np.isfinite(objective(h)):
        return None
    return h


def try_resolutions(best_h, best_c, deadline, rng):
    n0 = best_h.size
    candidates = []

    for nn in (n0 - 1, n0 + 1, 48, 64, 80):
        if 8 <= nn <= 160 and nn != n0 and nn not in candidates:
            candidates.append(nn)

    for nn in candidates:
        if time.time() > deadline - 60.0:
            break
        h = resample_vector(best_h, nn)
        if h is None:
            continue
        sub_deadline = min(deadline, time.time() + 35.0)
        hd, cd = local_polish(h, sub_deadline, rng)
        if solution_is_better(cd, hd.size, best_c, best_h.size):
            best_h = hd
            best_c = cd

    return best_h, best_c


def _run_impl(seed, budget_s, kwargs):
    t_start = time.time()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1100.0
    deadline = t_start + max(1.0, budget) - 3.0

    rng = np.random.RandomState(int(seed) % (2 ** 32))

    parent = get_initial_parent()
    if parent is not None and is_feasible(parent, 1e-5):
        best_h = parent.copy()
        best_c = objective(best_h)
    else:
        n0 = 48
        best_h = random_feasible(n0, rng, 0)
        best_c = objective(best_h)
        if not np.isfinite(best_c):
            best_h = np.full(48, 0.5)
            best_c = objective(best_h)

    t1 = t_start + 0.5 * budget
    h, c = local_polish(best_h, min(deadline, t1), rng)
    if solution_is_better(c, h.size, best_c, best_h.size):
        best_h, best_c = h, c

    if time.time() < deadline - 60.0:
        for i in range(3):
            if time.time() > deadline - 30.0:
                break
            hd = jump_random(best_h, rng, int(rng.randint(2, 6)) + i)
            if hd is None:
                continue
            sub_deadline = min(deadline, time.time() + 40.0)
            h, c = local_polish(hd, sub_deadline, rng)
            if solution_is_better(c, h.size, best_c, best_h.size):
                best_h, best_c = h, c

    if time.time() < deadline - 60.0:
        for i in range(2):
            if time.time() > deadline - 30.0:
                break
            hd = random_feasible(best_h.size, rng, i + 1)
            sub_deadline = min(deadline, time.time() + 40.0)
            h, c = local_polish(hd, sub_deadline, rng)
            if solution_is_better(c, h.size, best_c, best_h.size):
                best_h, best_c = h, c

    if time.time() < deadline - 120.0:
        best_h, best_c = try_resolutions(best_h, best_c, deadline, rng)

    if time.time() < deadline - 5.0:
        h, c = local_polish(best_h, deadline, rng)
        if solution_is_better(c, h.size, best_c, best_h.size):
            best_h, best_c = h, c

    h_final = best_h.copy()
    if not is_feasible(h_final, 1e-6):
        h_final = project_box_sum(h_final, 0.5 * len(h_final))

    c_final = objective(h_final)
    if not np.isfinite(c_final) or not is_feasible(h_final, 1e-6):
        parent2 = get_initial_parent()
        if parent2 is not None and is_feasible(parent2, 1e-5):
            h_final = parent2.copy()
            c_final = objective(h_final)
        if not np.isfinite(c_final):
            n = len(best_h) if best_h.size > 0 else 48
            h_final = np.full(n, 0.5)
            c_final = objective(h_final)
    else:
        if is_feasible(best_h, 1e-6):
            c_best = objective(best_h)
            if solution_is_better(c_best, best_h.size, c_final, len(h_final)):
                h_final = best_h.copy()
                c_final = c_best

    return h_final, float(c_final), int(h_final.size)


def run(seed=42, budget_s=1100, **kwargs):
    try:
        return _run_impl(seed, budget_s, kwargs)
    except Exception:
        try:
            parent = get_initial_parent()
            if parent is not None and is_feasible(parent, 1e-5):
                return parent.copy(), float(objective(parent)), int(parent.size)
        except Exception:
            pass
        h = np.full(48, 0.5)
        return h, float(objective(h)), 48
