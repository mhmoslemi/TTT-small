import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smoothed_max_c5(h, temperature=0.01):
    """Smoothed version using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return temperature * logsumexp(corr / temperature)

def generate_modes(n, num_modes):
    """Generate zero-mean modes for perturbation"""
    x = np.arange(n)
    modes = []
    # Cosine modes
    for k in range(1, num_modes + 1):
        m = np.cos(2 * np.pi * k * x / n)
        m = m - m.mean()  # zero mean
        modes.append(m)
    # Also add some "block" modes
    for k in range(1, min(4, num_modes // 2 + 1)):
        m = np.zeros(n)
        block_size = n // (2 * k)
        for b in range(2 * k):
            start = b * block_size
            end = min(start + block_size, n)
            m[start:end] = 1 if b % 2 == 0 else -1
        m = m - m.mean()
        modes.append(m)
    
    A = np.array(modes).T  # shape (n, num_modes)
    # Orthogonalize (optional)
    Q, R = np.linalg.qr(A)
    return Q[:, :num_modes]

def run_mode_optimization(base_h, num_modes=20, budget_s=60):
    """Optimize mode coefficients"""
    n = len(base_h)
    n_target_sum = n / 2.0
    
    A = generate_modes(n, num_modes)
    
    def objective(z):
        h = base_h + A @ z
        h = np.clip(h, 0, 1)
        # Enforce sum constraint approximately
        s = h.sum()
        if s != n_target_sum:
            h = h + (n_target_sum - s) / n
            h = np.clip(h, 0, 1)
        return smoothed_max_c5(h, temperature=0.005)
    
    def objective_grad(z):
        # Numerical gradient
        grad = np.zeros(num_modes)
        eps = 1e-5
        f0 = objective(z)
        for i in range(num_modes):
            z_plus = z.copy()
            z_plus[i] += eps
            grad[i] = (objective(z_plus) - f0) / eps
        return grad
    
    # Use Nelder-Mead or Powell for non-smooth

def compute_subgradient_vec(h, active):
    n = len(h)
    sg = np.zeros(n)
    for k in active:
        j = k - (n - 1)
        if j == 0:
            sg += (1.0 - 2.0 * h)
        elif j > 0:
            sg[j:] += (1.0 - h[:n-j])
            sg[:n-j] -= h[j:]
        else:
            j2 = -j
            sg[:n-j2] += (1.0 - h[j2:])
            sg[j2:] -= h[:j2]
    return sg / float(len(active))

def subgradient_optimizer(h, target_sum, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    step = 0.5
    while time.time() - start < time_limit:
        corr = get_corr_vals(h)
        max_val = corr.max()
        
        # Multiple active lags with softmax weighting
        temp = max(0.001, max_val * 0.05)
        w = np.exp((corr - max_val) / temp)
        w_sum = w.sum()
        if w_sum < 1e-10:
            break
        w = w / w_sum
        
        # Only use lags with significant weight
        significant = np.where(w > 0.01)[0]
        if len(significant) == 0:
            significant = np.array([np.argmax(corr)])
        
        sg = compute_subgradient_vec(h, significant)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        sg = sg / sg_norm
        
        # Step with occasional restart
        h_new = h - step * sg
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(step * 1.05, 2.0)
        else:
            step *= 0.95
            if step < 0.001:
                # Try a random perturbation
                pert = rng.randn(n) * 0.05
                h = project_feasible(h + pert, target_sum)
                step = 0.3
        
        h = h_new
    
    return best_h, best_c5

def pair_swap_local_search(h, target_sum, time_limit, seed=42):
    """Local search by swapping pairs of values"""
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    no_improve = 0
    max_no_improve = 500
    
    while time.time() - start < time_limit and no_improve < max_no_improve:
        improved = False
        # Try all pairs
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_new_c5 = best_c5
        
        for i in range(n):
            for j in range(i + 1, n):
                # Transfer from j to i: h[i]+=d, h[j]-=d
                max_d = min(1.0 - h[i], h[j])
                if max_d < 1e-6:
                    continue
                min_d = max(-h[i], -(1.0 - h[j]))  # = max(-h[i], h[j]-1)
                if max_d - min_d < 1e-6:
                    continue
                
                # Try several delta values
                deltas = np.linspace(min_d + 1e-4, max_d - 1e-4, 5)
                for d in deltas:
                    h_test = h.copy()
                    h_test[i] += d
                    h_test[j] -= d
                    c5_test = compute_c5(h_test)
                    if c5_test < best_new_c5:
                        best_new_c5 = c5_test
                        best_delta = d
                        best_i = i
                        best_j = j
        
        if best_i >= 0:
            h[best_i] += best_delta
            h[best_j] -= best_delta
            h = project_feasible(h, target_sum)
            best_c5 = compute_c5(h)
            best_h = h.copy()
            improved = True
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_c5

def mode_based_optimize(base_h, target_sum, time_limit, seed=42):
    """Optimize in a low-dimensional mode space"""
    n = len(base_h)
    num_modes = min(15, n - 1)
    
    # Generate modes
    x = np.arange(n, dtype=float)
    modes = []
    for k in range(1, num_modes + 1):
        m = np.cos(2.0 * np.pi * k * x / n)
        m = m - m.mean()
        modes.append(m)
    # Add a few sine modes
    for k in range(1, max(1, num_modes // 2)):
        m = np.sin(2.0 * np.pi * k * x / n)
        m = m - m.mean()
        modes.append(m)
    
    A = np.array(modes).T  # (n, num_modes)
    # Orthonormalize
    Q, R = np.linalg.qr(A)
    A = Q[:, :num_modes]
    
    start = time.time()
    
    # Use Nelder-Mead on the mode coefficients
    def objective(z):
        h = base_h + A.dot(z)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
        return compute_c5(h)
    
    result = minimize(objective, np.zeros(num_modes), method='Nelder-Mead',
                      options={'maxiter': 5000, 'xatol': 1e-5, 'fatol': 1e-7,
                               'adaptive': True})
    
    best_z = result.x
    h = base_h + A.dot(best_z)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, target_sum)
    c5 = compute_c5(h)
    
    return h, c5

    best_c5 = compute_c5(h)
    best_h = h.copy()

    def remaining_time():
        return budget_s - (time.time() - global_t0) - 20

    # Phase 1: Subgradient descent
    if remaining_time() > 60:
        alloc = min(remaining_time() * 0.30, 300)
        h1, c5_1 = subgradient_optimizer(h, target_sum, alloc, seed=seed)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Pair local search
    if remaining_time() > 60:
        alloc = min(remaining_time() * 0.30, 300)
        h2, c5_2 = pair_local_search(best_h, target_sum, alloc, seed=seed + 100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Mode-based optimization
    if remaining_time() > 30:
        alloc = min(remaining_time() * 0.20, 150)
        h3, c5_3 = mode_based_optimize(best_h, target_sum, alloc, seed=seed + 200)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Binarize + local search
    if remaining_time() > 30:
        h_bin = binarize_h(best_h, target_sum)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        if remaining_time() > 30:
            alloc = min(remaining_time() * 0.15, 120)
            h4, c5_4 = pair_local_search(best_h, target_sum, alloc, seed=seed + 300)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

    # Phase 5: Perturbation restart + subgradient
    if remaining_time() > 30:
        alloc = min(remaining_time() * 0.15, 100)
        h5, c5_5 = perturbation_restart(best_h, target_sum, alloc, seed=seed + 400)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final subgradient polish
    if remaining_time() > 20:
        alloc = min(remaining_time() * 0.3, 150)
        h6, c5_6 = subgradient_optimizer(best_h, target_sum, alloc, seed=seed + 500)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final pair polish
    if remaining_time() > 20:
        alloc = min(remaining_time() * 0.5, 120)
        h7, c5_7 = pair_local_search(best_h, target_sum, alloc, seed=seed + 600)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final validation
    best_h = project_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
