- Use projected gradient descent on f_μ
- Decrease μ over time

import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1-h, 'full')) * dx

def project_feasible(h, n):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

    # And the gradient of the smooth max = sum_m w_m * grad(F_m)

    N = n

        if w[m] < 1e-15:

import numpy as np

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def objective_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def project(h, n):
    """Project onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return result

def smooth_max_gradient(h, n, F, mu):
    """Compute gradient of mu*log(sum(exp(F[m]/mu)))"""
    n2 = 2 * n - 1
    # Softmax weights (truncated)
    F_shifted = F / mu - F.max() / mu
    # Truncate to top terms
    exp_F = np.exp(F_shifted)
    total = exp_F.sum()
    if total < 1e-300:
        w = np.zeros(n2)
        w[int(np.argmax(F))] = 1.0
    else:
        w = exp_F / total
    
    # Only keep terms with significant weight
    thresh = 1e-10
    active = w > thresh
    if not active.any():
        # Fallback: just the max
        w = np.zeros(n2)
        w[int(np.argmax(F))] = 1.0
        active = w > 0
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for m in np.where(active)[0]:
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        # Term 1: i in [max(0,-k), min(n-1, n-1-k)]
        i_lo1 = max(0, -k)

        idx1 = np.arange(i_lo1, i_hi1 + 1)

        i_hi2 = min(n - 1 + k, n - 1)
        idx2 = np.arange(i_lo2, i_hi2 + 1)
        grad[idx2] -= wm * h[idx2 - k]
    
    return grad

def resample(parent, new_n):
    """Resample parent array to new_n points using linear interpolation with area preservation."""
    p = len(parent)
    if p == new_n:
        return parent.copy()
    # Cumulative sum for area-preserving resampling
    x_old = np.linspace(0, 2, p, endpoint=False) + 1.0/p  # midpoints
    # Actually, let's use simple linear interpolation on the midpoints
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    # Interpolate
    # Use np.interp on the function values at midpoints
    result = np.interp(x_new, x_old, parent)
    result = np.clip(result, 0, 1)
    # Re-project to ensure sum constraint
    return project(result, new_n)

def dyadic_repair(h, n):
    """Quantize h to dyadic rationals while preserving sum = n/2 exactly."""
    # Choose scale: S = 2^bits
    if n <= 50:
        bits = 16
    elif n <= 72:
        bits = 14
    else:
        bits = 12
    S = 1 << bits
    if n % 2 != 0:
        S *= 2  # Ensure n*S/2 is integer
    
    target = (n * S) // 2
    # Round to nearest dyadic
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    
    # Water filling to adjust sum
    diff = target - z.sum()
    if diff != 0:
        # Adjust
        if diff > 0:
            # Need to increase total by diff
            # Increase the entries with highest fractional parts first
            fracs = h * S - z  # fractional parts
            order = np.argsort(-fracs)
            for idx in order[:diff]:
                if z[idx] < S:
                    z[idx] += 1
        else:
            # Need to decrease total by -diff
            fracs = h * S - z
            order = np.argsort(fracs)
            for idx in order[:-diff]:
                if z[idx] > 0:
                    z[idx] -= 1
    
    return z.astype(float) / S

            i_hi2 = min(n-1+k, n-1)
            idx2 = np.arange(i_lo2, i_hi2+1)
            grad[idx2] -= h[idx2-k]
        grad /= len(active)
        
        # Best donor: high h, high grad (we want to decrease h where grad is high)
        # Best receiver: low h, low grad (we want to increase h where grad is low)
        # Transfer from i to j changes objective by approximately dx * (grad[j] - grad[i]) * delta
        # We want grad[j] - grad[i] < 0, i.e., grad[i] > grad[j]
        
        # Find best pair
        # Donor: high h AND high grad
        # Receiver: low h AND low grad
        scores = h * 0.5 + 0.5 * (grad - grad.min()) / (grad.max() - grad.min() + 1e-15)
        best_donor = int(np.argmax(scores))
        best_receiver = int(np.argmin(scores))
        
        if best_donor == best_receiver:
            continue
        
        # Try a few transfer amounts
        max_transfer = min(h[best_donor], 1.0 - h[best_receiver]) * 0.5
        for frac in [0.5, 0.3, 0.1, 0.01]:
            delta = max_transfer * frac
            if delta < 1e-8:
                break
            h[best_donor] -= delta
            h[best_receiver] += delta
            h = project(h, n)  # Re-project to maintain sum
            c = exact_objective(h)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                break
            else:
                h = best_h.copy()
        
        if best_c < 0.38080:  # Target achieved (adjust as needed)
            break
    
    return best_h, best_c

import numpy as np
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def objective_vector(h, dx=None):
    n = len(h)
    if dx is None:
        dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def project(h, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gradient_for_component(h, n, m, dx):
    k = m - (n - 1)
    i_lo1 = max(0, -k)
    i_hi1 = min(n - 1, n - 1 - k)
    i_lo2 = max(0, k)
    i_hi2 = min(n - 1 + k, n - 1)
    grad = np.zeros(n)
    if i_lo1 <= i_hi1:
        idx1 = np.arange(i_lo1, i_hi1 + 1)
        grad[idx1] += dx * (1.0 - h[idx1 + k])
    if i_lo2 <= i_hi2:
        idx2 = np.arange(i_lo2, i_hi2 + 1)
        grad[idx2] -= dx * h[idx2 - k]
    return grad

def smooth_max_gradient(h, n, F, mu, max_active=50):
    n2 = 2 * n - 1
    F_norm = F - F.max()
    exp_F = np.exp(F_norm / mu)
    total = exp_F.sum()
    if total < 1e-300:
        w = np.zeros(n2)
        w[int(np.argmax(F))] = 1.0
    else:
        w = exp_F / total
    # Sort by weight, keep top
    if w.sum() > 0:
        sorted_idx = np.argsort(-w)
        top_idx = sorted_idx[:max_active]
        w_trunc = np.zeros(n2)
        w_trunc[top_idx] = w[top_idx]
        w_trunc /= w_trunc.sum()
    else:
        w_trunc = np.zeros(n2)
        w_trunc[int(np.argmax(F))] = 1.0
    dx = 2.0 / n
    grad = np.zeros(n)
    active = w_trunc > 1e-12
    for m in np.where(active)[0]:
        k = m - (n - 1)
        i_lo1 = max(0, -k)
        i_hi1 = min(n - 1, n - 1 - k)
        if i_lo1 <= i_hi1:
            idx1 = np.arange(i_lo1, i_hi1 + 1)
            grad[idx1] += w_trunc[m] * dx * (1.0 - h[idx1 + k])
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1 + k, n - 1)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            grad[idx2] -= w_trunc[m] * dx * h[idx2 - k]
    return grad

def resample(parent, new_n):
    p = len(parent)
    if p == new_n:
        return parent.copy()
    x_old = (np.arange(p) + 0.5) * 2.0 / p
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    result = np.interp(x_new, x_old, parent)
    result = np.clip(result, 0.0, 1.0)
    return project(result, new_n)

def dyadic_repair(h, n):
    if n <= 50:
        bits = 16
    elif n <= 72:
        bits = 14
    else:
        bits = 12
    S = 1 << bits
    if n % 2 != 0:
        S *= 2
    target = (n * S) // 2
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    diff = target - z.sum()
    if diff > 0:
        fracs = h * S - z.astype(float)
        order = np.argsort(-fracs)
        for i in range(min(diff, n)):
            idx = order[i]
            if z[idx] < S:
                z[idx] += 1
        # If still not enough, increment any that can
        remaining = target - z.sum()
        for idx in range(n):
            if remaining <= 0:
                break
            if z[idx] < S:
                z[idx] += 1
                remaining -= 1
    elif diff < 0:
        fracs = h * S - z.astype(float)
        order = np.argsort(fracs)
        for i in range(min(-diff, n)):
            idx = order[i]
            if z[idx] > 0:
                z[idx] -= 1
        remaining = z.sum() - target
        for idx in range(n):
            if remaining <= 0:
                break
            if z[idx] > 0:
                z[idx] -= 1
                remaining -= 1
    return z.astype(float) / S
