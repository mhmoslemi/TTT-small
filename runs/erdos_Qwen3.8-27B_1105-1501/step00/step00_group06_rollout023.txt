def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(r)

def project_box_mass(v, target):
    n = len(v)
    v = np.array(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-13:
        return v
    # Bisection for mu
    lo, hi = -3.0, 3.0
    for _ in range(10):
        if np.sum(np.clip(v + lo, 0, 1)) > target:
            lo -= 1.0
        else:
            break
    for _ in range(10):
        if np.sum(np.clip(v + hi, 0, 1)) < target:
            hi += 1.0
        else:
            break
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    y = np.clip(v + mu, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(y)
    if abs(res) > 1e-12:
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            y[free] += res / nf
            y = np.clip(y, 0.0, 1.0)
            # Second pass
            res2 = target - np.sum(y)
            free2 = (y > 1e-10) & (y < 1.0 - 1e-10)

            if nf2 > 0 and abs(res2) > 1e-14:
                y[free2] += res2 / nf2
                y = np.clip(y, 0.0, 1.0)
    return y

    h_curr = best_h.copy()

    local_budget = 0.55 * budget_s
    betas = [100, 50, 30, 20, 15, 10, 8, 6, 5, 4, 3, 2.5, 2, 1.5, 1, 0.7]
    
    for beta in betas:

            for step in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(r)

def project_box_mass(v, target):
    n = len(v)
    v = np.array(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-13:
        return v
    lo, hi = -3.0, 3.0
    for _ in range(12):
        if np.sum(np.clip(v + lo, 0, 1)) > target:
            lo -= 1.0
        else:
            break
    for _ in range(12):
        if np.sum(np.clip(v + hi, 0, 1)) < target:
            hi += 1.0
        else:
            break
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    y = np.clip(v + mu, 0.0, 1.0)
    res = target - np.sum(y)
    if abs(res) > 1e-12:
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            y[free] += res / nf
            y = np.clip(y, 0.0, 1.0)
            res2 = target - np.sum(y)
            free2 = (y > 1e-10) & (y < 1.0 - 1e-10)
            nf2 = np.sum(free2)
            if nf2 > 0 and abs(res2) > 1e-14:
                y[free2] += res2 / nf2
                y = np.clip(y, 0.0, 1.0)
    return y

def lse_gradient(h, beta):
    n = len(h)
    r = np.correlate(h, 1.0 - h, mode="full")
    r_max = np.max(r)
    exp_r = np.exp(beta * (r - r_max))
    w = exp_r / np.sum(exp_r)
    grad = np.zeros(n)
    j_vals = np.arange(n)
    for i in range(n):
        k_vals_a = i + n - 1 - j_vals
        valid_a = (k_vals_a >= 0) & (k_vals_a < 2 * n - 1)
        if np.any(valid_a):
            grad[i] += np.dot(w[k_vals_a[valid_a]], (1.0 - h[j_vals[valid_a]]))
        k_vals_b = j_vals - i + n - 1
        valid_b = (k_vals_b >= 0) & (k_vals_b < 2 * n - 1)
        if np.any(valid_b):
            grad[i] -= np.dot(w[k_vals_b[valid_b]], h[j_vals[valid_b]])
    return grad

    best_sc = sc_curr

    betas = [80, 50, 30, 20, 15, 10, 8, 6, 5, 4, 3, 2, 1.5, 1, 0.7, 0.5]

def pair_polish(h, target, t0, budget_s, rng):
