import numpy as np
import time

def _score(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _project_box_sum(x, target):
    """Project x onto {z: 0<=z_i<=1, sum(z)=target} using water-filling"""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-15:
        return x.copy()
    
    # Water-filling
    if diff > 0:
        # Need to add mass
        # Find t such that sum(clip(x + t, 0, 1)) = target
        # For x_i < 1, x_i + t contributes; for x_i >= 1, stays at 1
        lo, hi = 0.0, 1.0 - x.min()  # or just 0 to 1
        # Actually use bisection
        for _ in range(60):
            mid = (lo + hi) / 2
            s = np.clip(x + mid, 0.0, 1.0).sum()
            if s < target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2
    else:
        # Need to remove mass
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2
            s = np.clip(x - mid, 0.0, 1.0).sum()
            if s > target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2
    
    x = np.clip(x + t if diff > 0 else x - t, 0.0, 1.0)
    
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-14:
        # Distribute residual to coordinates with room
        if residual > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        idx = np.where(mask)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            x[idx] += per
            x = np.clip(x, 0.0, 1.0)
    
    return x

def _seed_fallback(n, offset=0.0):

    """Resample h to new length using linear interpolation"""
    n = len(h)
    if n == new_n:

def _pair_line_score(h, p, q, delta):
    """Score h with h[p]+=delta, h[q]-=delta applied"""
    n = len(h)
    h_mod = h.copy()
    h_mod[p] += delta
    h_mod[q] -= delta
    dx = 2.0 / n
    c = np.correlate(h_mod, 1.0 - h_mod, mode='full')
    return float(np.max(c) * dx)

def _pair_sweep(h, deadline, max_files=25):

    n = len(h)
    best_score = _score(h)
    best_h = h.copy()
    best_delta = 0.0
    best_pair = None

            # h[q] - delta in [0,1] => delta in [h[q]-1, h[q]]
            delta_lo = max(-h[p], h[q] - 1.0)

            # Sample delta values
            deltas = np.linspace(delta_lo, delta_hi, max_files)

            for d in deltas:
                if d < -1e-12 or d > 1 + 1e-12:
                    continue
                s = _pair_line_score(h, p, q, d)
                file_scores.append((s, d))
            
            if not file_scores:
                continue
            
            file_scores.sort(key=lambda x: x[0])
            s_best, d_best = file_scores[0]
            
            if s_best < best_score - 1e-12:
                best_score = s_best
                best_h = h.copy()
                best_h[p] += d_best
                best_h[q] -= d_best
                best_pair = (p, q)
                best_delta = d_best
        
        if time.perf_counter() > deadline:
            break

    x = np.clip(np.asarray(x, dtype=np.float64), 0.0, 1.0)

def _golden_section(f, a, b, iters=25):

    gr = (np.sqrt(5.0) - 1.0) / 2.0

            d = c
            fd = fc

            c = d
            fc = fd
            d = a + gr * (b - a)

import numpy as np
import time

def _score(h):
    n = len(h)
    if n < 2:
        return 1e10
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _project(x, target):
    x = np.array(x, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-14:
        return x
    if diff > 0:
        lo, hi = 0.0, 2.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            if np.clip(x + mid, 0.0, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2.0
        x = np.clip(x + t, 0.0, 1.0)
    else:
        lo, hi = 0.0, 2.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            if np.clip(x - mid, 0.0, 1.0).sum() > target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2.0
        x = np.clip(x - t, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-8
        else:
            mask = x > 1e-8
        idx = np.where(mask)[0]
        if len(idx) > 0:
            x[idx] += res / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x

def _golden_h(h, p, q, vp, vq, a, b, iters=25):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[p] = vp + c
    h[q] = vq - c
    fc = _score(h)
    h[p] = vp + d
    h[q] = vq - d
    fd = _score(h)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[p] = vp + c
            h[q] = vq - c

            c = d
            fc = fd

            if time.perf_counter() > deadline:

            vp = h[p]

                if time.perf_counter() > deadline:
                    break
                vq = h[q]

                d_hi = min(1.0 - vp, vq)
                if d_hi - d_lo < 1e-14:
                    continue
                ds = np.linspace(d_lo, d_hi, samples)
                best_s = cur_s
                best_d = 0.0
                for d in ds:
                    h[p] = vp + d
                    h[q] = vq - d
                    s = _score(h)

def _resample_to(h, new_n):
    n = len(h)
