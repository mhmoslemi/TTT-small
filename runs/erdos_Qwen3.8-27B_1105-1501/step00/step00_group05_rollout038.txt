def subgrad_descent(h0, N, n_iter=5000, step0=0.05):

    best_h = h.copy()
    best_val = evaluate_exact(h)
    step = step0
    dx = 2.0 / N
    center = N - 1
    stall = 0
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        p = np.argmax(corr)
        k = p - center
        grad = np.zeros(N)
        if k >= 0:
            idx1 = np.arange(k, N)
            grad[idx1] += (1.0 - h[idx1 - k])
            idx2 = np.arange(0, N - k)
            grad[idx2] -= h[idx2 + k]
        else:

            grad[idx1] += (1.0 - h[idx1 + m])
            idx2 = np.arange(m, N)
            grad[idx2] -= h[idx2 - m]

        h_new = project_feasible(h_new, N)
        val = evaluate_exact(h_new)
        if val < best_val - 1e-15:
            best_h = h_new.copy()
            best_val = val
            h = h_new
            stall = 0
        else:

            stall += 1
            if stall > 200:
                step *= 0.9
                stall = 0
            if step < 1e-8:

def eval_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, N):
    T = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_component(h, p, N):
    """Gradient of corr[p] with respect to h."""
    dx = 2.0 / N
    k = p - (N - 1)
    grad = np.zeros(N)
    if k >= 0:
        # corr[p] = sum_{n=0}^{N-1-k} h[n+k]*(1-h[n])
        # d/dh[i]: from h[n+k] term: (1-h[i-k]) for i in [k, N-1]
        #           from (1-h[n]) term: -h[i+k] for i in [0, N-1-k]
        if N - k > 0:
            grad[k:] += (1.0 - h[:N-k])  # (1-h[n]) where n = i-k, i from k to N-1, n from 0 to N-1-k
            grad[:N-k] -= h[k:]  # -h[i+k] where i from 0 to N-1-k
    else:
        m = -k
        # corr[p] = sum_{n=m}^{N-1} h[n-m]*(1-h[n])
        # d/dh[i]: from h[n-m] term: (1-h[i+m]) for i in [0, N-1-m]
        #           from (1-h[n]) term: -h[i-m] for i in [m, N-1]
        if N - m > 0:
            grad[:N-m] += (1.0 - h[m:])  # (1-h[n]) where n = i+m, i from 0 to N-1-m
            grad[m:] -= h[:N-m]  # -h[i-m] where i from m to N-1
    return grad * dx

def slsqp_opt(h0, N, maxiter=300):
    T = N / 2.0
    dx = 2.0 / N
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx
    def con_eq(x):
        return np.sum(x) - T
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * N,
                   constraints=[{'type': 'eq', 'fun': con_eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_feasible(h_new, N)
    return h_new, eval_c5(h_new)

def subgrad_descent(h0, N, n_iter=5000, step0=0.1):
    h = h0.copy()
    best_h = h.copy()
    best_val = eval_c5(h)
    dx = 2.0 / N
    center = N - 1
    for t in range(n_iter):
        step = step0 / np.sqrt(t + 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        p = int(np.argmax(corr))
        g = grad_component(h, p, N)
        h_new = h - step * g
        h_new = project_feasible(h_new, N)
        val = eval_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val

One more idea: what about using LP (linear programming) for the direction finding? At each point, I can:
1. Find the active constraint p* (the one with max overlap)
2. Compute the gradient g = grad_component(h, p*, N)
3. Find the best feasible direction d that minimizes g·d subject to:
   - sum(d) = 0 (area preservation)
   - h + alpha*d stays in [0,1] for some alpha > 0

This is a linear program! The optimal direction would be to move mass from coordinates with high g to coordinates with low g.

Specifically, minimize g·d subject to sum(d) = 0 and d[i] <= (1-h[i])/alpha, d[i] >= -h[i]/alpha for all i.

For a fixed alpha, this is:
minimize g·d
subject to sum(d) = 0
           -h[i]/alpha <= d[i] <= (1-h[i])/alpha

The optimal solution pushes d[i] to the lower bound (-h[i]/alpha) when g[i] is high (we want to decrease h[i]) and to the upper bound ((1-h[i])/alpha) when g[i] is low.

This is a "transportation" or "water-filling" problem. The optimal d is:
- Sort g
- The coordinates with the largest g get d = -h[i]/alpha (decrease as much as possible)
- The coordinates with the smallest g get d = (1-h[i])/alpha (increase as much as possible)
- One coordinate is partially used to satisfy the sum constraint

This is essentially a step function of g, and can be computed efficiently.

But this only considers one active constraint. For multiple active constraints, we'd want to minimize the max of g_p·d over p in active set, which is more complex.

                rc_idx = np.argsort(g)[:max(1, N // 5)]
                i = int(rng.choice(do_idx))
                j = int(rng.choice(rc_idx))
            else:
                i, j = int(rng.randint(N)), int(rng.randint(N))
            if i == j:
                continue
            eps = step
            if not (0.0 <= h[i] - eps <= 1.0 and 0.0 <= h[j] + eps <= 1.0):
                continue
            h[i] -= eps
            h[j] += eps
            v = eval_c5(h)
            if v < best_val:
                best_val = v
                best_h = h.copy()
                found = True
                break
            h[i] += eps
            h[j] -= eps
        if not found:
            step *= 0.85
            if step < 1e-10:
                break
    return best_h, best_val

def make_balanced(N, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(N)
    n_ones = N // 2
    positions = rng.choice(N, size=n_ones, replace=False)
    h[positions] = 1.0
    return h

def make_sturmian(N, alpha, rho, seed=0):
    """Sturmian-like sequence."""
    h = np.zeros(N)
    for i in range(N):
        h[i] = 1.0 if ((i + 1) * alpha + rho) - int((i + 1) * alpha + rho) < alpha else 0.0
    # Adjust to get exactly N/2 ones
    n_ones = int(np.sum(h))
    target = N // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(seed)
        for _ in range(target - n_ones):
            idx = int(rng.choice(zeros))
            h[idx] = 1.0
            zeros = np.where(h == 0)[0]
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(seed)
        for _ in range(n_ones - target):
            idx = int(rng.choice(ones))
            h[idx] = 0.0
            ones = np.where(h == 1)[0]
    return h

def smooth_max_slsqp(h0, N, alpha=50.0, maxiter=300):
    """SLSQP with smooth max approximation."""
    T = N / 2.0
    dx = 2.0 / N
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        c_shifted = c - np.max(c)
        return float(np.log(np.sum(np.exp(alpha * c_shifted))) / alpha + np.max(c))
    def con_eq(x):
        return np.sum(x) - T
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * N,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), N)
        return h_new, eval_c5(h_new)
    except Exception:
        return h0, eval_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    # Initialize from parent if available
    try:
        init = np.array(initial_h_values, dtype=float)
    except Exception:
        init = None

    if init is not None and len(init) > 0:
        best_h = init.copy()
        best_c5 = eval_c5(init)
        best_n = len(init)
    else:
        best_n = 42
        best_h = np.full(best_n, 0.5)
        best_c5 = eval_c5(best_h)

    N0 = best_n

    # Phase 1: Optimize at initial N
    if rem() > 60:
        h, c = slsqp_opt(best_h, best_n, 400)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 50:
        h, c = subgrad_descent(best_h, best_n, 3000, 0.04)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 40:
        h, c = subgrad_composite(best_h, best_n, 2000, 0.03)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 25:
        h, c = directed_pair_search(best_h, best_n, 3000, seed, 0.003)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 20:
        h, c = pair_search(best_h, best_n, 3000, seed + 1, 0.01)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: Smooth max SLSQP
    if rem() > 40:
        h, c = smooth_max_slsqp(best_h, best_n, alpha=80.0, maxiter=300)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 30:
        h, c = smooth_max_slsqp(best_h, best_n, alpha=200.0, maxiter=300)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Try other N values
    n_list = [40, 44, 46, 48, 50, 52, 56, 60, 64, 72, 80]
    for Nt in n_list:
        if rem() < 25:
            break
        xo = np.linspace(0, 2, best_n, endpoint=False)
        xn = np.linspace(0, 2, Nt, endpoint=False)
        hi = project_feasible(np.interp(xn, xo, best_h), Nt)

        ho, co = slsqp_opt(hi, Nt, 250)
        if co < best_c5:
            best_h, best_c5, best_n = ho, co, Nt

        if rem() > 20:
            hs, cs = subgrad_descent(ho, Nt, 2000, 0.03)
            if cs < best_c5:
                best_h, best_c5, best_n = hs, cs, Nt

        if rem() > 15:
            hp, cp = directed_pair_search(best_h, best_n, 2000, seed + Nt, 0.003)
            if cp < best_c5:
                best_h, best_c5 = hp, cp

    # Phase 4: Alternative starting points
    for Nt in [best_n, best_n + 2, best_n + 4, 48, 50]:
        if rem() < 20 or Nt < 10:
            continue
        for s2 in range(3):
            if rem() < 15:
                break
            hbal = make_balanced(Nt, seed + Nt + s2 * 100)
            hbal = project_feasible(hbal, Nt)
            ho, co = slsqp_opt(hbal, Nt, 200)
            if co < best_c5:
                best_h, best_c5, best_n = ho, co, Nt
            if rem() > 10:
                hp, cp = directed_pair_search(ho, Nt, 1500, seed + Nt + s2, 0.003)
                if cp < best_c5:
                    best_h, best_c5, best_n = hp, cp, Nt

    # Phase 5: Final refinement on best
    if rem() > 40:
        h, c = slsqp_opt(best_h, best_n, 500)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 30:
        h, c = subgrad_composite(best_h, best_n, 3000, 0.02, 0.03)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 25:
        h, c = directed_pair_search(best_h, best_n, 5000, seed + 500, 0.002)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 15:
        h, c = pair_search(best_h, best_n, 3000, seed + 800, 0.005)
        if c < best_c5:
            best_h, best_c5 = h, c
    if rem() > 10:
        h, c = slsqp_opt(best_h, best_n, 300)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, best_n
