import numpy as np

def exact_objective(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def objective_vector(h):
    """Compute the full correlation vector."""
    N = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_for_lag(h, lag):
    """Compute gradient of c_lag w.r.t. h.
    lag: integer from -(N-1) to (N-1)
    Output: array of length N
    """
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # c_lag = sum_{m=0}^{N-1-lag} h[m+lag] * (1-h[m])
        # ∂/∂h[r] = (1-h[r-lag]) for r in [lag, N-1]
        #           - h[r+lag] for r in [0, N-1-lag]
        for r in range(lag, N):
            grad[r] += (1.0 - h[r - lag])
        for r in range(0, N - lag):
            grad[r] -= h[r + lag]
    else:
        j = -lag
        # c_{-j} = sum_{m=0}^{N-1-j} h[m] * (1-h[m+j])
        # ∂/∂h[r] = (1-h[r+j]) for r in [0, N-1-j]
        #            - h[r-j] for r in [j, N-1]
        for r in range(0, N - j):
            grad[r] += (1.0 - h[r + j])
        for r in range(j, N):
            grad[r] -= h[r - j]
    return grad

def project_to_constraint(y, N):

def dyadic_repair(h, N, bits=20):
    """Quantize h to dyadic grid and repair sum constraint exactly."""
    S = 1 << bits
    # Make S such that N*S is even
    if (N * S) % 2 != 0:
        S *= 2
    
    z = np.round(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    target = (N * S) // 2
    
    # Water filling to fix sum
    diff = target - np.sum(z)
    if diff > 0:
        # Need to add diff units
        # Sort by how much room each has (S - z[i])
        # Add to positions with most room
        order = np.argsort(-z)  # positions with least room first (highest z)
        # Actually, add to positions with most room (lowest z)
        order = np.argsort(z)
        added = 0
        for i in order:
            room = S - z[i]
            amt = min(room, diff - added)
            z[i] += amt
            added += amt
            if added >= diff:
                break
    elif diff < 0:
        # Need to remove -diff units
        # Remove from positions with most mass (highest z)
        order = np.argsort(-z)
        removed = 0
        to_remove = -diff
        for i in order:
            amt = min(z[i], to_remove - removed)
            z[i] -= amt
            removed += amt
            if removed >= to_remove:
                break
    
    return z.astype(np.float64) / S

def resample(parent_h, new_N):
    """Resample a density from len(parent_h) to new_N points."""
    P = len(parent_h)
    # Use cumulative trapezoidal integration
    # parent is defined on [0, 2] with P points
    # new should be defined on [0, 2] with new_N points
    
    # Treat as piecewise constant: bin edges
    parent_edges = np.linspace(0, 2, P+1)
    new_edges = np.linspace(0, 2, new_N+1)
    
    # For each new bin, compute the average of parent values that fall in it
    new_h = np.zeros(new_N)
    for j in range(new_N):
        lo = new_edges[j]
        hi = new_edges[j+1]
        # Find parent bins overlapping [lo, hi]
        # parent bin i covers [parent_edges[i], parent_edges[i+1]]
        mask = (parent_edges[1:] > lo) & (parent_edges[:-1] < hi)
        if not np.any(mask):
            new_h[j] = 0.5  # default
        else:
            # Weighted average
            weights = np.zeros(P)
            for i in range(P):
                overlap = min(parent_edges[i+1], hi) - max(parent_edges[i], lo)
                if overlap > 0:
                    weights[i] = overlap
            total = np.sum(weights)
            if total > 0:
                new_h[j] = np.sum(weights * parent_h) / total
            else:
                new_h[j] = 0.5
    new_h = np.clip(new_h, 0, 1)
    new_h = project_to_constraint(new_h, new_N)
    return new_h

def exact_objective(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def objective_vector(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_for_lag(h, k):
    """Gradient of correlation value at lag k w.r.t. h.
    k is the lag (negative or positive), not the array index.
    """
    N = len(h)
    grad = np.zeros(N)
    j = abs(k)
    if j == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        # c_k = sum_{i=0}^{N-1-k} h[i+k]*(1-h[i])
        grad[k:N] += (1.0 - h[0:N-k])
        grad[0:N-k] -= h[k:N]
    else:

        grad[0:N-j] += (1.0 - h[j:N])
        grad[j:N] -= h[0:N-j]

def project_to_constraint(y, N):
    """Project y onto {x: 0<=x[i]<=1, sum(x)=N/2} using bisection."""
    target = N / 2.0

        else:

def dyadic_repair(h, N, bits=20):
    """Quantize to dyadic grid and enforce exact sum."""
    S = 1 << bits
    if (N * S) % 2 != 0:
        S *= 2
    z = np.round(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    target = (N * S) // 2
    diff = target - int(np.sum(z))
    if diff > 0:
        # Add diff units to positions with most room
        order = np.argsort(z)
        added = 0
        for i in order:
            if added >= diff:
                break
            room = S - z[i]
            amt = min(room, diff - added)
            z[i] += amt
            added += amt
    elif diff < 0:
        # Remove -diff units from positions with most mass
        order = np.argsort(-z)
        removed = 0
        to_remove = -diff
        for i in order:
            if removed >= to_remove:
                break
            amt = min(z[i], to_remove - removed)
            z[i] -= amt
            removed += amt
    return z.astype(np.float64) / float(S)
