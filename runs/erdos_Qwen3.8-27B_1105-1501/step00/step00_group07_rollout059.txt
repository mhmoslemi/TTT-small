import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(R) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-14:
        order = np.argsort(h) if s < target else np.argsort(-h)
        rem = abs(s - target)
        for i in order:
            if rem < 1e-14:
                break
            if s < target:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
            else:
                r = min(rem, h[i])
                h[i] -= r
                rem -= r
    return h

def resample_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return make_feasible(h_new, n_new)

def stable_sigmoid(x):
    x = np.clip(x, -50, 50)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def solve_lambda(z, n):
    target = n / 2.0
    lam = 0.0
    for _ in range(15):
        s = stable_sigmoid(z + lam)
        S = float(np.sum(s))
        Sp = float(np.sum(s * (1.0 - s)))
        if Sp < 1e-15:
            break
        err = S - target
        if abs(err) < 1e-13:
            break
        lam -= err / Sp
        if lam > 80: lam = 80
        if lam < -80: lam = -80
    return lam

def make_basis(n, pmax):
    x = (np.arange(n) + 0.5) / n
    cols = []
    for k in range(1, 100):
        cols.append(np.cos(2 * np.pi * k * x))
        cols.append(np.sin(2 * np.pi * k * x))
        if len(cols) >= pmax:
            break
    return np.column_stack(cols[:pmax])

def basis_search(h_init, n, pmax, rng, budget_s):
    B = make_basis(n, pmax)
    p = B.shape[1]
    eps = 1e-4
    hc = np.clip(h_init, eps, 1.0 - eps)
    zt = np.log(hc / (1.0 - hc))
    theta, _, _, _ = np.linalg.lstsq(B, zt, rcond=None)
    
    z = B @ theta
    lam = solve_lambda(z, n)
    h = make_feasible(stable_sigmoid(z + lam), n)
    best_score = score_exact(h)
    best_h = h.copy()
    best_theta = theta.copy()
    
    t0 = time.time()
    step = 0.3
    stall = 0
    while time.time() - t0 < budget_s:
        improved = False
        for d in range(p):
            for dirn in (1, -1):
                nt = best_theta.copy()
                nt[d] += dirn * step
                z2 = B @ nt
                lam2 = solve_lambda(z2, n)
                h2 = make_feasible(stable_sigmoid(z2 + lam2), n)
                s2 = score_exact(h2)
                if s2 < best_score - 1e-12:
                    best_score = s2
                    best_h = h2
                    best_theta = nt
                    improved = True
        if not improved:
            step *= 0.5
            stall += 1
            if step < 1e-5 or stall > 15:
                break
        else:
            stall = 0
    return best_h, best_score
