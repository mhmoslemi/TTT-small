def smooth_gradient(h, beta):
    """Compute gradient of log-sum-exp smooth approximation."""
    n = len(h)
    # Compute all f_p values
    corr = np.correlate(h, 1 - h, mode="full")
    f = corr  # f[p] = sum of products for lag p
    
    # Compute weights (softmax)
    beta_f = beta * f
    max_beta_f = np.max(beta_f)
    exp_bf = np.exp(beta_f - max_beta_f)
    weights = exp_bf / exp_bf.sum()
    
    # Compute gradient as weighted sum of individual gradients
    grad = np.zeros(n)
    for p in range(2*n-1):
        w = weights[p]
        if w < 1e-15:
            continue
        k = p - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
