def project_feasible(z, target):
    """Project z onto {h: 0<=h_i<=1, sum(h)=target}"""
    n = len(z)
    v = np.clip(z, 0.0, 1.0)
    excess = np.sum(v) - target
    if excess > 1e-14:
        # Need to reduce
        idx = np.argsort(v)[::-1]
        remaining = excess
        for i in idx:
            d = min(v[i], remaining)
            v[i] -= d
            remaining -= d
            if remaining <= 1e-14:
                break
    elif excess < -1e-14:
        # Need to increase
        idx = np.argsort(v)
        remaining = -excess
        for i in idx:
            d = min(1.0 - v[i], remaining)
            v[i] += d
            remaining -= d
            if remaining <= 1e-14:
                break
    return v

def project_feasible(z, target):
    n = len(z)
    v = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    excess = np.sum(v) - target
    if excess > 1e-13:
        idx = np.argsort(v)[::-1]
        rem = excess
        for i in idx:
            d = min(v[i], rem)
            v[i] -= d
            rem -= d
            if rem <= 1e-13:
                break
    elif excess < -1e-13:
        idx = np.argsort(v)
        rem = -excess
        for i in idx:
            d = min(1.0 - v[i], rem)
            v[i] += d
            rem -= d
            if rem <= 1e-13:
                break

def soft_gradient_raw(h, alpha):
