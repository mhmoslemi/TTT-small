def pairwise_fine(h, n, c_current, time_left_fn):
    order = np.argsort(h)
    for ii in range(n - 1, n - 12, -1):
        if time_left_fn() < 1.0:
            break
        i = order[ii]
        if h[i] < 0.01:
            break
        for jj in range(0, 12):
            if time_left_fn() < 1.0:
                break
            j = order[jj]
            if i == j or h[j] > 0.99:
                continue
            for frac in [0.3, 0.5, 0.7]:
                amt = min(h[i] * frac, (1.0 - h[j]) * frac)
                if amt < 1e-5:
                    continue
                cand = h.copy()
                cand[i] -= amt
                cand[j] += amt
                cand = np.clip(cand, 0, 1)
                c_cand = exact_objective(cand)
                if c_cand < c_current - 1e-9:
                    return cand, c_cand, True
    return h, c_current, False


def resample_to(parent, new_n):
    pn = len(parent)
    xp = np.linspace(0.0, 2.0, pn)
    xn = np.linspace(0.0, 2.0, new_n)
    new_h = np.interp(xn, xp, parent)
    return project_feasible(new_h, new_n)


def make_symmetric_initial(n, seed=0):
    rng = np.random.RandomState(seed)
    t = np.linspace(0, 1, n)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * 2 * t + rng.uniform(0, 2 * np.pi))
    h += 0.1 * rng.randn(n)
    return project_feasible(h, n)


def optimize_dim(h0, n, time_budget):
    t0 = time.time()
    h = h0.copy()
    c_best = exact_objective(h)
    h_best = h.copy()
    stall = 0

    while time.time() - t0 < time_budget:
        c_cur = exact_objective(h)
        if c_cur < c_best - 1e-10:
            c_best = c_cur
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            break
        time_left_fn = lambda: time_budget - (time.time() - t0)
        h, c_new, improved = try_improve(h, n, c_cur, time_left_fn)
        if not improved and stall > 30:
            break

    return h_best, c_best


def run(seed=42, budget_s=1100, **kwargs):
    t_global = time.time()
    time_limit = budget_s - 8.0
    best_h = None
    best_c = 1.0
    best_n = 0

    try:
        init = np.array(initial_h_values, dtype=float)
        n_init = len(init)
    except (NameError, UnboundLocalError, Exception):
        init = None
        n_init = 42

    dims_to_try = []
    if init is not None:
        dims_to_try.append(n_init)
        for d in range(max(24, n_init - 12), min(80, n_init + 16), 2):
            if d != n_init:
                dims_to_try.append(d)
        dims_to_try = sorted(set(d for d in dims_to_try if d >= 16 and d <= 80))
    else:
        dims_to_try = list(range(36, 72, 2))

    # Sort: initial dim first, then by distance from 42
    if init is not None and n_init in dims_to_try:
        dims_to_try = [n_init] + sorted([d for d in dims_to_try if d != n_init],
                                         key=lambda d: abs(d - 42))
    else:
        dims_to_try = sorted(dims_to_try, key=lambda d: abs(d - 42))

    for n in dims_to_try:
        elapsed = time.time() - t_global
        remaining = time_limit - elapsed
        if remaining < 15:
            break
        t_alloc = min(remaining - 10, 180.0)
        if n == n_init and init is not None:
            t_alloc = min(remaining - 10, 300.0)
        if t_alloc < 10:
            break

        if init is not None:
            h_start = init.copy() if n == n_init else resample_to(init, n)
        else:
            h_start = make_symmetric_initial(n, seed=seed + n)

        try:
            h_opt, c_opt = optimize_dim(h_start, n, t_alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = n
        except Exception:
            pass

    # Final polish on best
    if best_h is not None:
        remaining = time_limit - (time.time() - t_global)
        if remaining > 30:
            try:
                h_pol, c_pol = optimize_dim(best_h, best_n, remaining - 10)
                if c_pol < best_c:
                    best_c = c_pol
                    best_h = h_pol
            except Exception:
                pass

    # Emergency fallback
    if best_h is None:
        n = 42
        best_h = np.full(n, 0.5)
        best_n = n
        best_c = exact_objective(best_h)

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    best_c = exact_objective(best_h)

    assert best_h.shape == (best_n,)
    assert np.all(np.isfinite(best_h))
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6

    return (best_h, best_c, best_n)
