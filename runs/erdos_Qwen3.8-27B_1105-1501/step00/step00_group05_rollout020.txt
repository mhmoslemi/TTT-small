    def objective(v):
        return v[n]  # t

        g[n] = 1.0
        return g

            h = v[:n]
            t = v[n]
            corrs = compute_corrs(h) * dx
            return t - corrs[d_idx]

def grad_c_d(h, d, dx):
    """Gradient of C(d)*dx with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        jp = j + d
        if 0 <= jp < n:
            grad[j] += (1.0 - h[jp])
            grad[jp] -= h[j]
    return grad * dx

def subgradient_c5(h, dx):
    """Subgradient of C5 = max_d C(d)*dx"""
    n = len(h)
    corrs = compute_corrs(h) * dx
    max_val = np.max(corrs)
    # Use all active lags with equal weight
    tol = 1e-12 * max(1.0, abs(max_val))
    active = np.where(np.abs(corrs - max_val) <= tol)[0]
    
    grad = np.zeros(n)
    for d_idx in active:
        d = d_idx - (n - 1)
        grad += grad_c_d(h, d, dx)
    grad /= len(active)
    return grad

def smooth_max_val(h, dx, beta):

    corrs = compute_corrs(h) * dx

def smooth_max_grad(h, dx, beta):

    n = len(h)
    corrs = compute_corrs(h) * dx
    m = np.max(corrs)
    weights = np.exp(beta * (corrs - m))
    weights /= np.sum(weights)

        if weights[d_idx] < 1e-15:
            continue
        d = d_idx - (n - 1)
        g = grad_c_d(h, d, dx)
        grad += weights[d_idx] * g
    return grad
