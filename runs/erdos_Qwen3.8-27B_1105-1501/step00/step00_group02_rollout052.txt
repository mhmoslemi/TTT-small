    return np.max(corr) * dx

def project_to_bounds_and_sum(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum} using bisection."""
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:

    """Compute gradient of correlation[idx] with respect to h."""
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        # corr[idx] = sum_{n=0}^{N-1-k} h[n+k]*(1-h[n])
        m = n - k
        grad[k:] += (1.0 - h[:m])
        grad[:m] -= h[k:]
    else:
        j = -k
        m = n - j
        grad[:m] += (1.0 - h[j:])
        grad[j:] -= h[:m]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_set(h, target_sum):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def corr_gradient(h, idx, n):
    """Gradient of np.correlate(h, 1-h, 'full')[idx] w.r.t. h."""
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            grad[k:] += (1.0 - h[:m])
            grad[:m] -= h[k:]
    else:
        j = -k
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] -= h[:m]
    return grad

def projected_gradient(h, dx, target_sum, max_iters=3000, lr_init=0.02):

    n = len(h)
    h = h.copy()
    lr = lr_init
    best = compute_c5(h, dx)
    best_h = h.copy()

        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

        grad = np.zeros(n)
        for i in range(len(corr)):
            if weights[i] > 1e-10:
                grad += weights[i] * corr_gradient(h, i, n) * dx
        return grad
