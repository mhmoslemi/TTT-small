                h_ls2, c5_ls2 = local_search(h_b, min(20, t_avail / 8))
                if c5_ls2 < c5_b:
                    h_b, c5_b = h_ls2, c5_ls2
                if c5_b < c5_best:
                    h_best, c5_best = h_b.copy(), c5_b

    # Phase 6: Final polish
    rem = budget_s - (time.time() - t_start) - 2
    if rem > 5:
        h_final, c5_final = local_search(h_best, max(1, rem))
        if c5_final < c5_best:
            h_best, c5_best = h_final, c5_final

    # Ensure constraint is satisfied
    N_out = len(h_best)
    h_best = project_box_sum(h_best, N_out / 2.0)
    c5_out = eval_c5(h_best)

    return h_best.tolist(), c5_out, N_out
