        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def corr_full(h):
    return np.correlate(h, 1 - h, mode="full")

def grad_at_max(h):
    n = len(h)
    corr = corr_full(h)
    k_star = int(np.argmax(corr))

    n = len(h)
    corr = corr_full(h)

    corr_scaled = alpha * corr
    corr_scaled -= np.max(corr_scaled)  # for numerical stability
    weights = np.exp(corr_scaled)
    weights /= np.sum(weights)
    
    # Weighted sum of gradients
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        offset = k - np.arange(n)
        valid = (offset >= 0) & (offset < n)
        grad[valid] += weights[k] * (1.0 - 2.0 * h[offset[valid]])
    return grad, np.max(corr)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def project_sum(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_grad_smooth(h, alpha):
    """Compute gradient of smooth max of correlation"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    corr_max = np.max(corr)
    corr_shifted = alpha * (corr - corr_max)
    weights = np.exp(corr_shifted)
    wsum = np.sum(weights)
    weights /= wsum
    
    # Compute weighted gradient
    # grad[j] = sum_k weights[k] * (1 - 2*h[k-j]) for valid k-j
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = weights[k]
        if wk < 1e-12:
            continue
        # For this k, the gradient at position j is (1-2*h[k-j]) if 0<=k-j<n
        offset = k - np.arange(n)
        valid = (offset >= 0) & (offset < n)
        grad[valid] += wk * (1.0 - 2.0 * h[offset[valid]])
    
    return grad

            lr = 1e-7
        grad = compute_grad_smooth(h, alpha)
        h = h - lr * grad
        h = project_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000 and it > 100:
            lr *= 0.5
            stall = 0
    return best_h, best_c5

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count("1") % 2 == 0 else 0.0
    return project_sum(h, n / 2.0)

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    if n % 2 == 0:
        h[:n // 2] = 1.0
    else:
        h[:n // 2] = 1.0
        h[n // 2] = 0.5
    rng.shuffle(h)
    return h

def sine_init(n, freq, phase):
    x = np.arange(n) / n
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x + phase)
    return project_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(47, 0.5)

    n = len(h0)
    h0 = project_sum(h0, n / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n

    # Phase 1: Gradient optimization from initial
    t_lim = budget_s - 60
    h_opt, c5_opt = optimize_h(h0, 10000, 0.01, 20, 300, seed, t_lim, start_time)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt
        best_n = len(h_opt)

    # Phase 2: Multiple restarts with diverse initializations
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 40
    n_restarts = max(5, min(25, int(remaining / 20)))

    for trial in range(n_restarts):
        if time.time() - start_time > budget_s - 25:
            break
        strat = trial % 6
        s2 = seed * 1000 + trial * 7
        if strat == 0:
            h_pert = best_h + rng.normal(0, 0.04, best_n)
        elif strat == 1:
            h_pert = random_binary_init(best_n, s2)
        elif strat == 2:
            h_pert = thue_morse_init(best_n) + rng.normal(0, 0.12, best_n)
        elif strat == 3:
            h_pert = sine_init(best_n, rng.uniform(2, 12), rng.uniform(0, 6.28))
        elif strat == 4:
            h_pert = rng.uniform(0, 1, best_n)
            h_pert = np.convolve(h_pert, np.ones(5) / 5.0, mode="same")
        else:
            # Mirror + noise
            half = best_n // 2
            h_pert = best_h[:half].copy()
            h_pert = np.concatenate([h_pert, h_pert[::-1]])[:best_n]
            h_pert += rng.normal(0, 0.06, best_n)

        h_pert = project_sum(h_pert, best_n / 2.0)
        t_per = max(5.0, remaining / (n_restarts - trial + 1) * 0.8)
        lr_val = rng.uniform(0.003, 0.015)
        h_opt, c5_opt = optimize_h(h_pert, 5000, lr_val, 20, 250, s2, t_per, time.time())
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = len(h_opt)

    # Phase 3: Try different n values
    if time.time() - start_time < budget_s - 60:
        for n_try in [31, 41, 51, 61, 71, 81, 101, 121]:
            if time.time() - start_time > budget_s - 30:
                break
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.clip(np.interp(x_new, x_old, best_h), 0, 1)
            h_interp = project_sum(h_interp, n_try / 2.0)
            t_per = max(5.0, (budget_s - (time.time() - start_time) - 20) / 8.0)
            h_opt, c5_opt = optimize_h(h_interp, 4000, 0.005, 20, 250,
                                        seed + n_try, t_per, time.time())
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 4: Final fine-tuning with high alpha
    if time.time() - start_time < budget_s - 10:
        t_rem = budget_s - (time.time() - start_time) - 5
        h_o1, c5_1 = optimize_h(best_h, 6000, 0.002, 100, 500,
                                 seed + 9999, t_rem * 0.6, start_time)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h_o1
        if time.time() - start_time < budget_s - 5:
            t_rem2 = budget_s - (time.time() - start_time) - 2
            h_o2, c5_2 = optimize_h(best_h, 4000, 0.001, 200, 800,
                                     seed + 8888, t_rem2, start_time)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_o2

    # Ensure exact sum
    best_h = project_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
