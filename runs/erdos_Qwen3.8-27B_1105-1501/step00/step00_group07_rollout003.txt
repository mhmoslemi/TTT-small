def feasible_project(z, target_sum):
    n = len(z)
    v = np.clip(np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    e = np.sum(v) - target_sum
    if e > 1e-12:
        order = np.argsort(-v)
        remaining = e
        for idx in order:
            if remaining < 1e-13:
                break
            delta = min(v[idx], remaining)
            v[idx] -= delta
            remaining -= delta
    elif e < -1e-12:
        order = np.argsort(v)
        remaining = -e
        for idx in order:
            if remaining < 1e-13:
                break
            delta = min(1.0 - v[idx], remaining)
            v[idx] += delta
            remaining -= delta
    diff = np.sum(v) - target_sum
    if abs(diff) > 1e-15 and n > 0:
        if diff > 0:
            idx = np.argmax(v)
            v[idx] -= min(diff, v[idx])
        else:
            idx = np.argmin(v)
            v[idx] += min(-diff, 1.0 - v[idx])
    return v

def score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(raw)

def smoothed_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp <= 0 or not np.isfinite(sum_exp):
        return None
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-14:
            continue
        k = i - (n - 1)
        if k >= 0:
            length = n - k
            if length <= 0:
                continue
            grad[k:] += w * (1.0 - h[:length])
            grad[:length] -= w * h[k:]
        else:
            a = -k
            length = n - a
            if length <= 0:
                continue
            grad[:length] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:length]
    grad *= dx
    grad -= np.mean(grad)
    return grad
