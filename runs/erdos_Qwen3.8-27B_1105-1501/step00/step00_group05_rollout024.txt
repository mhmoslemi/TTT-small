    raw = np.asarray(raw, dtype=float)
    lo, hi = float(np.min(raw)) - T - 2.0, float(np.max(raw)) + T + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(raw - mid, 0.0, 1.0)))
        if s > T:
            lo = mid
        else:

def evaluate(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(raw, T):
    raw = np.asarray(raw, dtype=float)
    # Binary search for the right shift
    lo = float(np.min(raw)) - T - 1.0
    hi = float(np.max(raw)) + T + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(raw - mid, 0.0, 1.0)))
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(raw - 0.5*(lo+hi), 0.0, 1.0)

def corr_vector(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def active_lags(h, dx, tol):
    c = corr_vector(h, dx)
    mx = float(c.max())
    return np.where(c >= mx - tol)[0], mx

def grad_sum_active(h, dx, active):
    N = len(h)
    ctr = N - 1
    g = np.zeros(N)
    for m in active:
        d = int(m) - ctr
        lo = max(0, -d)
        hi = min(N, N - d)
        for i in range(lo, hi):
            j = i + d
            if i != j:
                g[i] += dx * (1.0 - h[j])
                g[j] += -dx * h[i]
            else:
                g[i] += dx * (1.0 - 2.0 * h[i])
    return g
