    n = len(v)
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    if not np.all(np.isfinite(v)):

            lo = mid
        else:
            hi = mid

    y = np.clip(v + t, 0.0, 1.0)

        e = target - np.sum(y)
        if abs(e) < 1e-13:
            break
        mov = (y > 1e-12) & (y < 1.0 - 1e-12)

    n = len(h)
    A = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s > 0:
            i1 = np.arange(n - s)
            i2 = i1 + s
            A[k] = np.dot(h[i1], 1.0 - h[i2])
            G[k, i1] += 1.0 - h[i2]
            G[k, i2] -= h[i1]
        elif s == 0:

    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n] - h_c
            if np.max(np.abs(d)) > 1e-14:
                return d
    except Exception:
        pass
    return None
