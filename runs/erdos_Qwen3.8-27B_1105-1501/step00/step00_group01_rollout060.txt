import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    x = np.array(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-15:
            break
        free = (x > 1e-14) & (x < 1.0 - 1e-14)
        if not np.any(free):
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            continue
        x[free] += diff / np.sum(free)
        x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - t0
    
    def try_update(h, c5=None):
        nonlocal best_h, best_c5

            c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:

            mx = np.max(corr)
            return mx + np.log(np.sum(np.exp(beta * (corr - mx)))) / beta

                          bounds=[(0.0, 1.0)] * n0,
                          options={'maxiter': 200, 'ftol': 1e-14})
