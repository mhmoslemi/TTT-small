    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def make_feasible(y):
    """Project y onto {h: 0<=h_i<=1, sum(h)=len(h)/2}"""
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    # Bisection for lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):

        if s > target:
