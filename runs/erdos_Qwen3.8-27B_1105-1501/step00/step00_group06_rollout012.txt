def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return np.max(r) * dx

def project(h):
    """Project h to satisfy 0<=h<=1 and sum(h)=len(h)/2"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Use bisection to find shift
    def F(mu):
        return np.sum(np.clip(h + mu, 0, 1))
    lo, hi = -2.0, 2.0
    # Ensure bracketing
    while F(lo) > target:
        lo -= 1.0
    while F(hi) < target:
        hi += 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if F(mid) < target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2
    y = np.clip(h + mu, 0, 1)
    # Fix residual
    residual = target - np.sum(y)
    if abs(residual) > 1e-10:
        # Distribute over interior points
        interior = (y > 1e-6) & (y < 1 - 1e-6)
        if np.sum(interior) > 0:
            y[interior] += residual / np.sum(interior)
            y = np.clip(y, 0, 1)
    return y

def lse_gradient(h, beta):
    """Compute gradient of log-sum-exp smoothing of correlation"""
    n = len(h)
    r = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    r_shifted = r - np.max(r)
    w = np.exp(beta * r_shifted)
    w /= np.sum(w)
    # Gradient via finite differences of LSE
    eps = 1e-6
    grad = np.zeros(n)
    for i in range(n):

        h_plus[i] += eps
        r_plus = np.correlate(h_plus, 1.0 - h_plus, mode="full")
        r_plus_shifted = r_plus - np.max(r_plus)
        w_plus = np.exp(beta * r_plus_shifted)
        w_plus /= np.sum(w_plus)
        lse_plus = np.log(np.sum(np.exp(beta * r_plus_shifted)) + 1e-300) / beta

        h_minus[i] -= eps
        r_minus = np.correlate(h_minus, 1.0 - h_minus, mode="full")

    n = len(h)
    r = np.correlate(h, 1.0-h, "full")
    # Compute weights
    r_max = np.max(r)
    w = np.exp(beta * (r - r_max))
    w /= np.sum(w)
    # grad[i] = sum_j w[i-j+n-1]*(1-h[j]) - sum_j w[j-i+n-1]*h[j]
    # Use matrix form: M[i,j] = w[i-j+n-1] for valid indices
    # First term: M @ (1-h)
    # Second term: M2 @ h where M2[i,j] = w[j-i+n-1]
    # Note M2 = M.T (check: M[i,j]=w[i-j+n-1], M2[i,j]=w[j-i+n-1]=M[j,i])
    # So second term = M.T @ h

- h[i] - lambda >= 0 → lambda <= h[i]
- h[j] + lambda <= 1 → lambda <= 1 - h[j]
- h[i] - lambda <= 1 (always true if h[i] <= 1)
- h[j] + lambda >= 0 (always true if h[j] >= 0)

So lambda in [0, min(h[i], 1-h[j])] for transfer i→j, or [min(-h[j], h[i]-1), 0] for transfer j→i.

Actually for the direction d = e_j - e_i (move mass from i to j):
h[i] - lambda, h[j] + lambda
Feasible: 0 <= h[i]-lambda <= 1 and 0 <= h[j]+lambda <= 1

    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_target) + 0.5) * 2.0 / n_target
    return np.interp(x_tgt, x_src, h)

def optimize_local(h, v, rng, get_remaining, max_rounds=500):
    n = len(h)
    betas = [3, 7, 15, 40, 100, 300]
    cooldown = 0
    for rnd in range(max_rounds):
        rem = get_remaining()
        if rem < 2.0:
            break
        improved = False
        if cooldown > 0:
            cooldown -= 1
        else:
            # LSE gradient direction
            for beta in betas:
                if get_remaining() < 1.5:
                    break
                grad = lse_grad(h, beta)
                grad -= np.mean(grad)
                gnorm = float(np.max(np.abs(grad)))
                if gnorm < 1e-14:
                    continue
                d = -grad / gnorm
                d -= np.mean(d)
                dnorm = float(np.max(np.abs(d)))
                if dnorm < 1e-14:
                    continue
                d /= dnorm
                h_new, v_new = line_search(h, d, max_step=0.25, n_evals=18)
                if v_new < v - 1e-12:
                    h = h_new
                    v = v_new
                    improved = True
                    break
            # Random direction
            if not improved:
                d = rng.randn(n)
                d -= np.mean(d)
                dnorm = float(np.max(np.abs(d)))
                if dnorm > 1e-14:
                    d /= dnorm
                    h_new, v_new = line_search(h, d, max_step=0.15, n_evals=12)
                    if v_new < v - 1e-12:
                        h = h_new
                        v = v_new
                        improved = True
            # Pair transfers guided by gradient
            if not improved:
                grad = lse_grad(h, 50.0)
                grad -= np.mean(grad)
                # Try top pairs: move from high-grad to low-grad coords
                idx_sorted = np.argsort(grad)
                n_try = min(80, n * (n - 1) // 2)
                tried = 0
                for t in range(n_try):
                    if get_remaining() < 1.0:
                        break
                    if t < n - 1:
                        i = idx_sorted[t]
                        j = idx_sorted[n - 1 - t]
                    else:
                        i = int(rng.randint(n))
                        j = int(rng.randint(n))
                    if i == j:
                        continue
                    d = np.zeros(n)
                    d[j] = 1.0
                    d[i] = -1.0
                    h_new, v_new = line_search(h, d, max_step=0.4, n_evals=12)
                    if v_new < v - 1e-12:
                        h = h_new
                        v = v_new
                        improved = True
                        break
                    tried += 1
            # Full random pair sweep every few rounds
            if not improved and rnd % 10 == 5:
                n_sweep = min(200, n * (n - 1) // 2)
                for _ in range(n_sweep):
                    if get_remaining() < 1.0:
                        break
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    if i == j:
                        continue
                    d = np.zeros(n)
                    d[j] = 1.0
                    d[i] = -1.0
                    h_new, v_new = line_search(h, d, max_step=0.4, n_evals=10)
                    if v_new < v - 1e-12:
                        h = h_new
                        v = v_new
                        improved = True
                        break
        if not improved:
            cooldown = 3
    return h, v

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def get_remaining():
        return budget_s - (time.time() - t_start)

    # Get initial values
    init_vals = globals().get('initial_h_values', None)
    if init_vals is not None and len(init_vals) > 0:
        n_init = len(init_vals)
        h_start = np.array(init_vals, dtype=np.float64).copy()
    else:
        n_init = 70
        h_start = np.zeros(n_init)
        h_start[:n_init // 2] = 1.0

    h_start = project(h_start)
    v_start = evaluate(h_start)

    # Resolutions to try
    resolutions = [n_init]
    for nn in [50, 60, 80, 100, 120, 140, 160]:
        if nn != n_init:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))

    best_h = h_start.copy()
    best_v = v_start

    # Phase 1: Optimize at each resolution
    for n_res in resolutions:
        if get_remaining() < 5.0:
            break

        # Starting points for this resolution
        starts = []
        if n_res == n_init:
            starts.append(h_start.copy())
        else:
            starts.append(project(interpolate_to_n(h_start, n_res)))
        # Blocky starts
        for kb in [4, 5, 7]:
            if n_res % kb == 0:
                bl = n_res // kb
                for pattern in range(min(8, 2 ** kb)):
                    blocks = [(pattern >> b) & 1 for b in range(kb)]
                    total = float(np.sum(blocks))
                    if 0 < total < n_res:
                        h_blk = np.zeros(n_res)
                        for b in range(kb):
                            h_blk[b * bl:(b + 1) * bl] = (n_res / 2.0) / total * blocks[b]
                        starts.append(project(h_blk))
        # Smooth starts
        for freq in [1, 2, 3]:
            x = np.linspace(0, 2, n_res, endpoint=False)
            h_sm = 0.5 + 0.3 * np.sin(np.pi * freq * x) + 0.1 * np.sin(np.pi * 2 * freq * x)
            starts.append(project(h_sm))
        # Random
        for _ in range(2):
            starts.append(project(rng.rand(n_res)))

        # Optimize each starting point
        time_per_start = max(5.0, (get_remaining() - 30.0) / max(1, len(starts)))
        for si, s in enumerate(starts):
            if get_remaining() < 5.0:
                break
            h_s = project(s)
            v_s = evaluate(h_s)
            if v_s > best_v + 0.08:
                continue
            # Limit optimization time per start
            local_deadline = time.time() + min(time_per_start, get_remaining() - 5.0)
            if local_deadline < time.time() + 2.0:
                continue

            def local_remaining():
                return min(get_remaining() - 3.0, local_deadline - time.time())

            h_opt, v_opt = optimize_local(h_s, v_s, rng, local_remaining, max_rounds=300)
            if v_opt < best_v - 1e-10:
                best_v = v_opt
                best_h = h_opt.copy()

    # Phase 2: Intensive polish on best solution
    if get_remaining() > 10.0:
        h_pol = project(best_h)
        v_pol = evaluate(h_pol)
        n_pol = len(h_pol)

        # Repeated LSE + pair optimization
        for beta in [2, 5, 10, 20, 50, 100, 200, 500]:
            if get_remaining() < 3.0:
                break
            for _ in range(5):
                if get_remaining() < 3.0:
                    break
                grad = lse_grad(h_pol, beta)
                grad -= np.mean(grad)
                gnorm = float(np.max(np.abs(grad)))
                if gnorm < 1e-14:
                    break
                d = -grad / gnorm
                d -= np.mean(d)
                h_new, v_new = line_search(h_pol, d, max_step=0.2, n_evals=20)
                if v_new < v_pol - 1e-12:
                    h_pol = h_new
                    v_pol = v_new
                else:
                    break

        # Pair transfer polish: try all pairs
        if get_remaining() > 20.0:
            n_pairs = n_pol * (n_pol - 1) // 2
            # Do multiple passes over all pairs
            max_passes = min(5, int(get_remaining() / 10))
            for p in range(max_passes):
                if get_remaining() < 5.0:
                    break
                for i in range(n_pol):
                    if get_remaining() < 5.0:
                        break
                    for j in range(i + 1, n_pol):
                        if get_remaining() < 5.0:
                            break
                        d = np.zeros(n_pol)
                        d[j] = 1.0
                        d[i] = -1.0
                        h_new, v_new = line_search(h_pol, d, max_step=0.4, n_evals=8)
                        if v_new < v_pol - 1e-12:
                            h_pol = h_new
                            v_pol = v_new

        if v_pol < best_v - 1e-10:
            best_v = v_pol
            best_h = h_pol.copy()

    # Phase 3: Try higher resolution interpolation of best
    if get_remaining() > 15.0:
        for n_try in [len(best_h) * 2, len(best_h) + 30]:
            if get_remaining() < 8.0:
                break
            h_interp = project(interpolate_to_n(best_h, n_try))
            v_interp = evaluate(h_interp)
            if v_interp < best_v + 0.02:
                def hr():
                    return get_remaining() - 5.0
                h_opt, v_opt = optimize_local(h_interp, v_interp, rng, hr, max_rounds=100)
                if v_opt < best_v - 1e-10:
                    best_v = v_opt
                    best_h = h_opt.copy()

    # Final: ensure feasibility and return
    best_h = project(best_h)
    best_v = evaluate(best_h)
    n_final = len(best_h)
    return (best_h, best_v, n_final)
