import numpy as np
from scipy.optimize import minimize
import time

_g_dx = 1.0
_g_target = 21.0
_g_alpha = 100.0

def _obj_fn(h):
    corr = np.correlate(h, 1.0 - h, mode='full') * _g_dx
    a = _g_alpha
    shifted = a * (corr - corr.max())
    return (corr.max() + np.log(np.sum(np.exp(shifted)))) / a

def _eq_fn(h):
    return h.sum() - _g_target

def eval_c5(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    return result

def make_feasible(h, target):
    h = project(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
            h = project(h, target)
    return h

        try:

                         constraints=constraints,
                         options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_cand = make_feasible(res.x, target)
                c5_cand = eval_c5(h_cand)
                if c5_cand < best_c5:

                    best_c5 = c5_cand
