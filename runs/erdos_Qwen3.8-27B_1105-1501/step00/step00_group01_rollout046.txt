    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)

            if 0 <= j + d < n:
                g += (-h[j + d])
            if g != 0.0:

def compute_c5(h, n):
    """Compute C5 = max_k correlate(h, 1-h, full)[k] * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def ls_c5(h, n, t):
    """Log-sum-exp approximation of C5"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    if t > 1e6:
        return m
    return m + (1.0/t) * np.log(np.sum(np.exp(t * (vals - m))))

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    if t > 1e6:
        # Just use the subgradient at the argmax
        k_star = np.argmax(vals)
        d = k_star - (n - 1)
        g = np.zeros(n)
        if d >= 0:
            if n - d > 0:
                g[d:] += (1.0 - h[:n-d])
                g[:n-d] += (-h[d:])
        else:
            dp = -d
            if n - dp > 0:
                g[:n-dp] += (1.0 - h[dp:])
                g[dp:] += (-h[:n-dp])

        else:

    for _ in range(60):  # 60 iterations of bisection gives plenty of precision

        val = np.sum(np.clip(p - mid, 0, 1))
        if val > target:
            lo = mid
        else:
            hi = mid
    
    alpha = (lo + hi) / 2.0
    return np.clip(p - alpha, 0, 1)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def ls_c5(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    if t > 1e6:
        return m
    return m + (1.0/t) * np.log(np.sum(np.exp(t * (vals - m))))

def grad_ls_c5(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    g = np.zeros(n)
    if t > 1e6:
        k_star = int(np.argmax(vals))
        d = k_star - (n - 1)
        if d >= 0:
            if n > d:
                g[d:] += (1.0 - h[:n-d])
                g[:n-d] += (-h[d:])
        else:
            dp = -d
            if n > dp:
                g[:n-dp] += (1.0 - h[dp:])
                g[dp:] += (-h[:n-dp])
        return g * dx
    exp_vals = np.exp(t * (vals - m))
    w = exp_vals / np.sum(exp_vals)

        else:
            dp = -d
