    return np.correlate(h, 1.0 - h, mode="full")

def local_search_gradient(h, time_limit=30):

    N = len(h)

    h = h.astype(float).copy()
    current_c5 = evaluate_c5(h)
    start_time = time.time()

        m_star = int(np.argmax(corr))
        k = m_star - (N - 1)  # lag

                grad[n] += -dx * h[n + k]

        # For a pair transfer (d, a, δ): Δc ≈ δ * (grad[a] - grad[d])

        # Find best pair
        best_improvement = 0
        best_i, best_j = -1, -1
        # d is the source (decrease), a is the target (increase)
        # We want grad[a] - grad[d] to be as negative as possible
        # i.e., we want grad[j] < grad[i] for transfer from i to j
        
        for i in range(N):
            for j in range(N):
                if i == j:
                    continue
                improvement = -(grad[j] - grad[i])  # we want this positive
                if improvement > best_improvement:
                    # Check feasibility
                    d = min(step, h[i], 1.0 - h[j])
                    if d > 1e-12:
                        best_improvement = improvement * d
                        best_i, best_j = i, j
        
        if best_i < 0 or best_improvement < 1e-14:
            no_improve += 1
            step *= 0.5
            if step < 1e-7 or no_improve > 100:
                break
            continue
        
        d = min(step, h[best_i], 1.0 - h[best_j])
        h[best_i] -= d
        h[best_j] += d

        grad = np.zeros(N)
        if k_star >= 0:
            for n in range(N - k_star):
                grad[n + k_star] += dx * (1.0 - h[n])
                grad[n] -= dx * h[n + k_star]

        order = np.arange(N)

    bounds = [(0, 1)] * N + [(0, 1)]

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(raw, N):
    target = N / 2.0
    h = np.clip(raw.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            total = room.sum()
            if total < 1e-15:
                break
            h += diff * room / total
        else:
            mask = h > 0.0
            if not mask.any():
                break
            total = h.sum()
            if total < 1e-15:
                break
            h += diff * h / total
    return np.clip(h, 0.0, 1.0)

def compute_grad_component(h, m, N, dx):
    k = m - (N - 1)
    grad = np.zeros(N)
    if k >= 0:
        for n in range(N - k):
            grad[n + k] += dx * (1.0 - h[n])
            grad[n] -= dx * h[n + k]
    else:
        for n in range(-k, N):
            grad[n + k] += dx * (1.0 - h[n])
            grad[n] -= dx * h[n + k]
    return grad

def local_search(h, time_limit=30, seed=42):
    N = len(h)
    dx = 2.0 / N
    h = h.astype(float).copy()
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start_time = time.time()
    rng = np.random.RandomState(seed)
    step_sizes = [1e-2, 5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5]
    step_idx = 3
    stall = 0

    while time.time() - start_time < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_star = int(np.argmax(corr))
        grad = compute_grad_component(h, m_star, N, dx)

        step = step_sizes[step_idx]
        improved = False

        order = np.arange(N)
        rng.shuffle(order)
        donor_sorted = order[np.argsort(-grad[order])]
        recvr_sorted = order[np.argsort(grad[order])]

        max_pairs = N * 4
        tried = 0
        for di in range(N):
            if time.time() - start_time >= time_limit or improved:
                break
            i = donor_sorted[di % N]
            for ri in range(4):
                j = recvr_sorted[ri % N]
                if j == i:
                    continue
                tried += 1
                d = min(step, h[i], 1.0 - h[j])
                if d < 1e-12:
                    continue
                h[i] -= d
                h[j] += d
                c5 = evaluate_c5(h)
                if c5 < current_c5 - 1e-15:
                    current_c5 = c5
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] += d
                    h[j] -= d
                if tried >= max_pairs:
                    break
            if tried >= max_pairs:
                break

        if improved:
            stall = 0
            if rng.random() < 0.03 and step_idx > 0:
                step_idx -= 1
        else:
            stall += 1
            if step_idx < len(step_sizes) - 1:
                step_idx += 1
            else:
                step_sizes[step_idx] *= 2.5
            if stall > 150:
                break
    return best_h, best_c5

def slsqp_refine(h, time_limit=30, top_k=12):
    N = len(h)
    dx = 2.0 / N
    n_corr = 2 * N - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    active = np.argsort(-corr)[:top_k]
    t_init = float(np.max(corr)) + 0.005

    def obj(x):
        return x[N]

    def obj_grad(x):
        g = np.zeros(N + 1)
        g[N] = 1.0
        return g

    constraints = []
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:N]) - N / 2.0,
        'jac': lambda x: np.concatenate([np.ones(N), [0.0]])
    })

    for m in active:
        k = m - (N - 1)
        if k >= 0:
            pairs = [(n + k, n) for n in range(N - k)]
        else:
            pairs = [(n + k, n) for n in range(-k, N)]

        def cfun(x, pairs=pairs):
            hx = x[:N]
            s = 0.0
            for a, b in pairs:
                s += hx[a] * (1.0 - hx[b])
            return x[N] - dx * s

        def cjac(x, pairs=pairs):
            hx = x[:N]
            g = np.zeros(N + 1)
            g[N] = 1.0
            for a, b in pairs:
                g[a] -= dx * (1.0 - hx[b])
                g[b] += dx * hx[a]
            return g

        constraints.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})

    bounds = [(0.0, 1.0)] * N + [(0.0, 2.0)]
    x0 = np.concatenate([h, [t_init]])
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 150, 'ftol': 1e-14})
        h_new = np.clip(res.x[:N], 0.0, 1.0)
        h_new = project_feasible(h_new, N)
        c5_new = evaluate_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h.copy(), evaluate_c5(h)

def make_block_init(N, rng):
    h = np.zeros(N)
    pos = 0
    while pos < N:
        blk = rng.randint(1, max(2, N // 4))
        val = rng.choice([0.0, 1.0, 0.3, 0.7, 1.0])
        h[pos:pos + blk] = val
        pos += blk
    return project_feasible(h, N)

def make_smooth_init(N, rng):
    freqs = rng.randint(1, 5, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    x = np.linspace(0, 2, N)
    h = np.zeros(N)
    for f, p in zip(freqs, phases):
        h += np.sin(f * np.pi * x + p)
    h = h - h.min()
    if h.max() > 0:
        h = h / h.max()
    return project_feasible(h, N)

def downsample(h, factor):
    N = len(h)
    new_N = N // factor
    if new_N < 4:
        return h.copy()
    h2 = np.zeros(new_N)
    for i in range(new_N):
        h2[i] = h[i * factor:(i + 1) * factor].mean()
    return project_feasible(h2, new_N)

def upsample(h, factor):
    N = len(h)
    new_N = N * factor
    h2 = np.interp(
        np.linspace(0, N - 1, new_N),
        np.arange(N), h
    )
    return project_feasible(h2, new_N)
