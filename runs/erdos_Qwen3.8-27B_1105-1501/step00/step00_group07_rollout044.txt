    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * dx)

def repair_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(5000):
        err = float(np.sum(h)) - target
        if abs(err) < 1e-12:
            break
        if err > 0:
            order = np.argsort(h)[::-1]
            remaining = err
            for i in order:
                if remaining <= 1e-13:
                    break
                if h[i] > 0:
                    d = min(float(h[i]), remaining)
                    h[i] -= d
                    remaining -= d
        else:

                if remaining <= 1e-13:
                    break

    x = (np.arange(n) + 0.5) * 2.0 / n - 1.0

        cols.append(np.cos(k * np.pi * x))
        cols.append(np.sin(k * np.pi * x))
        k += 1

    diag = np.abs(np.diag(R))
    rank = int(np.sum(diag > 1e-10 * diag[0]))
    return Q[:, :rank]

def nm_optimize(f, x0, maxiter=200, tol=1e-7):
