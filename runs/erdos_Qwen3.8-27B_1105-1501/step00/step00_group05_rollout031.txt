import numpy as np
import time

def compute_objective(h):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx"""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Binary search on lambda
    lo = np.min(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        # Spread residual among interior points
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(interior):
            h[interior] += residual / np.sum(interior)
        else:
            h += residual / n
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradients(h):
    """Compute gradient matrix G[i,m] = d(corr[m])/d(h[i])"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((n, 2*n-1))
    for i in range(n):
        for m in range(2*n-1):
            k = m - (n-1)
            g = 0.0
            if 0 <= i + k < n:
                g += (1.0 - h[i + k])
            if 0 <= i - k < n:
                g -= h[i - k]
            G[i, m] = g * dx
    return G

def make_initial(n, seed):
    """Create an initial feasible h"""
    rng = np.random.default_rng(seed)
    # Start with 0.5 everywhere, then add random perturbations
    h = np.full(n, 0.5)

    t = np.linspace(0, 2*np.pi, n, endpoint=False)
    h += 0.3 * np.sin(t) + 0.2 * np.sin(2*t + 1)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h

def resample(parent, new_n):
    """Resample parent density to new_n points"""
    n = len(parent)
    # Use cumulative sum for area-preserving resampling
    # Parent points are at x_i = i * 2/n
    # New points are at y_j = j * 2/new_n
    # h_new[j] = h at the nearest parent point (or interpolated)
    
    # Simple approach: linear interpolation
    x_parent = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    # Actually, let's use the midpoint of each bin
    x_parent = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    
    h_new = np.interp(x_new, x_parent, parent, period=2.0)
    # Actually, the function is defined on [0,2] and we're not doing periodic extension
    # Let me just use simple linear interpolation
    h_new = np.interp(x_new, x_parent, parent)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_feasible(h_new, new_n)
    return h_new

            best_c = c
            stall = 0
        else:
            stall += 1

        G = compute_gradients(h)

        active_mask = corr >= c - 1e-10
        active_idx = np.where(active_mask)[0]

        improved = False
        # To be efficient, shuffle the pair order
