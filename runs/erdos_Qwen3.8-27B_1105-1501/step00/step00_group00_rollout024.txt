            hi = mid
    z = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual mass
    diff = target - z.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - z)[::-1]
        else:
            order = np.argsort(z)[::-1]
        remaining = abs(diff)
        for idx in order:
            if remaining < 1e-14:
                break
            if diff > 0:
                room = 1.0 - z[idx]
                add = min(room, remaining)
                z[idx] += add
                remaining -= add
            else:
                room = z[idx]
                rem = min(room, remaining)
                z[idx] -= rem
                remaining -= rem
    # Final clip (shouldn't be needed but safe)
    z = np.clip(z, 0.0, 1.0)
    return z

    n = len(h)
    y = h + rng.normal(0, sigma, n)
    return project_feasible(y, n)

def transfer_resolution(h_src, n_t):
    """Transfer to different resolution preserving area."""
    n_s = len(h_src)
    dx_s = 2.0 / n_s
    dx_t = 2.0 / n_t
    x_s = np.linspace(0, 2.0, n_s + 1)
    area_s = np.concatenate([[0.0], np.cumsum(h_src * dx_s)])
    x_t = np.linspace(0, 2.0, n_t + 1)
    area_t = np.interp(x_t, x_s, area_s)
    h_t = np.diff(area_t) / dx_t
    return project_feasible(h_t, n_t)

def pair_move_search(h, max_evals=3000, rng=None):
    """Local search with pair moves."""
    n = len(h)
    best = h.copy()
    best_c = exact_c5(best)
    if rng is None:
        rng = np.random.RandomState(12345)
    
    deltas = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    evals_used = 0
    
    for delta in deltas:
        no_improve = 0
        while no_improve < 200 and evals_used < max_evals:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            evals_used += 1
            moved = False
            # Try: decrease h[i] by delta, increase h[j] by delta
            if best[i] >= delta and best[j] + delta <= 1.0:
                best[i] -= delta
                best[j] += delta
                c = exact_c5(best)
                if c < best_c - 1e-12:
                    best_c = c
                    no_improve = 0
                    moved = True
                else:
                    best[i] += delta
                    best[j] -= delta
            
            if not moved:
                # Try opposite direction
                if best[j] >= delta and best[i] + delta <= 1.0:
                    best[j] -= delta
                    best[i] += delta
                    c = exact_c5(best)
                    evals_used += 1
                    if c < best_c - 1e-12:
                        best_c = c
                        no_improve = 0
                        moved = True
                    else:
                        best[j] += delta
                        best[i] -= delta
            
            if not moved:
                no_improve += 1
    
    return best, best_c

    c = _corr_vals(x)
    m = np.max(c)
    shifted = np.clip((c - m) / tau, -50.0, 50.0)
    ev = np.exp(shifted)
    se = np.sum(ev)
    obj = m + tau * np.log(se)
    w = ev / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        L = k - (n - 1)
        if L >= 0:
            ln = n - L
            grad[L:L+ln] += wk * (1.0 - x[:ln])
            grad[:ln] -= wk * x[L:L+ln]
        else:
            p = -L
            ln = n - p
            grad[p:p+ln] += wk * (1.0 - x[:ln])
            grad[:ln] -= wk * x[p:p+ln]
    return obj, grad

    y = np.asarray(y, dtype=np.float64).copy()

        else:
            hi = mid
    z = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - z)[::-1]
        else:
            order = np.argsort(z)[::-1]

        for idx in order:
            if rem < 1e-14:
                break
            if diff > 0:
                a = min(1.0 - z[idx], rem)
                z[idx] += a
                rem -= a
            else:
                a = min(z[idx], rem)
                z[idx] -= a
                rem -= a
    return z

def _transfer(h_src, n_t):
    n_s = len(h_src)
    dxs = 2.0 / n_s
    dxt = 2.0 / n_t
    xs = np.linspace(0.0, 2.0, n_s + 1)
    areas = np.concatenate([[0.0], np.cumsum(h_s * dxs)])
    xt = np.linspace(0.0, 2.0, n_t + 1)
    at = np.interp(xt, xs, areas)
    ht = np.diff(at) / dxt
    return _project(ht, n_t)

def _pair_search(h, max_evals, rng):
    n = len(h)
    best = h.copy()
    best_c = _c5_of(best)
    deltas = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    used = 0
    for delta in deltas:
        stuck = 0
        while stuck < 500 and used < max_evals:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            improved = False
            # Try move i -> j
            if best[i] >= delta and best[j] + delta <= 1.0 + 1e-12:
                best[i] -= delta
                best[j] += delta
                c = _c5_of(best)
                used += 1
                if c < best_c - 1e-12:
                    best_c = c
                    stuck = 0
                    improved = True
                else:
                    best[i] += delta
                    best[j] -= delta
            # Try move j -> i
            if not improved and best[j] >= delta and best[i] + delta <= 1.0 + 1e-12:
                best[j] -= delta
                best[i] += delta
                c = _c5_of(best)
                used += 1
                if c < best_c - 1e-12:
                    best_c = c
                    stuck = 0
                else:
                    best[j] += delta
                    best[i] -= delta
            if not improved:
                stuck += 1
    return best, best_c

def _slsqp_solve(x0, n, target, tau, maxiter=300):
    global _g_n, _g_target, _g_tau
    _g_n = n
    _g_target = target
    _g_tau = tau
    bounds = [(0.0, 1.0)] * n

                       method='SLSQP', bounds=bounds,
                       constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except:
        return x0
