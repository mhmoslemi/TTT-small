def compute_c5(h):
    """Compute the Erdős C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    cors = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cors) * dx)


def compute_lse_grad(h, tau):
    """Compute log-sum-exp value and gradient."""
    n = len(h)
    dx = 2.0 / n
    cors = np.correlate(h, 1.0 - h, mode="full")
    scaled = tau * cors * dx
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    S = np.sum(exps)
    w = exps / S
    val = (max_s + np.log(S)) / tau
    
    grad = np.zeros(n)
    # Center term (lag 0)
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    # Positive lags (k = 1, ..., n-1)
    for k in range(1, n):
        wk = w[n-1+k] * dx
        if wk < 1e-15:
            continue
        grad[k:] += wk * (1.0 - h[:n-k])
        grad[:n-k] -= wk * h[k:]
    # Negative lags (lag = -p, p = 1, ..., n-1)
    for p in range(1, n):
        wp = w[n-1-p] * dx
        if wp < 1e-15:
            continue
        grad[:n-p] += wp * (1.0 - h[p:])
        grad[p:] -= wp * h[:n-p]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cors = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cors) * dx)


def compute_lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    cors = np.correlate(h, 1.0 - h, mode="full")
    scaled = tau * cors * dx
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    S = np.sum(exps)
    w = exps / S
    val = (max_s + np.log(S)) / tau
    
    grad = np.zeros(n)
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n-1+k] * dx
        if wk > 1e-15:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
    for p in range(1, n):
        wp = w[n-1-p] * dx
        if wp > 1e-15:
            grad[:n-p] += wp * (1.0 - h[p:])
            grad[p:] -= wp * h[:n-p]
    
    return val, grad
