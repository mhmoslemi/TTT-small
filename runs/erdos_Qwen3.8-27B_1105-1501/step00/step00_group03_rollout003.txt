For the smooth optimization:

- Objective: Phi_beta(h) (scalar)
- Gradient: dPhi/dh (length n)

- Variable: [h_0, ..., h_{n-1}, theta] (length n+1)
- Objective: theta (scalar)
- Gradient: [0, 0, ..., 0, 1] (length n+1)
- Inequality constraints: theta - dx*F_k >= 0 for each k (2n-1 constraints)
- Equality constraint: sum(h) - n/2 = 0
- Bounds: 0 ≤ h_i ≤ 1, theta ≥ 0 (or some lower bound)

    n = len(h)
    # F_k for k = -(n-1), ..., n-1
    # Return array of length 2n-1, indexed as [k + n - 1]
    result = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        idx = k + n - 1
        # valid i: max(0, -k) <= i <= min(n-1, n-1-k)
        i_min = max(0, -k)
        i_max = min(n, n - k)  # exclusive
        if i_min < i_max:
            result[idx] = np.sum(h[i_min:i_max] * (1.0 - h[i_min+k:i_max+k]))
    return result

def manual_grad(h):

        # Contribution: (1 - h[p+k])
        # As second index (i+k=p, i=p-k): valid if max(0,k) <= p <= min(n-1, n-1+k)
        # Contribution: -h[p-k]
        for p in range(n):
            val = 0.0
            if 0 <= p + k < n:

            pk = p - k
            if 0 <= pk < n and max(0, k) <= p <= min(n-1, n-1+k):
                val += -h[pk]
            grad[idx, p] = val
    return grad

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def manual_values(h):
    n = len(h)
    result = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        i_min = max(0, -k)
        i_max = min(n, n - k)
        if i_min < i_max:
            result[k + n - 1] = np.sum(h[i_min:i_max] * (1.0 - h[i_min + k:i_max + k]))
    return result


def manual_grad(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        idx = k + n - 1
        for p in range(n):
            val = 0.0
            if 0 <= p + k < n:
                val += (1.0 - h[p + k])
            if 0 <= p - k < n and max(0, k) <= p <= min(n - 1, n - 1 + k):
                val += -h[p - k]
            grad[idx, p] = val
    return grad


def make_feasible(z, target_sum):
    n = len(z)
    z = np.asarray(z, dtype=float)
    if not np.all(np.isfinite(z)):
        z = np.zeros(n)
    lo = -1.0 - float(np.max(z)) - 1.0
    hi = 1.0 - float(np.min(z)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z + lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
        else:
            room = h.copy()
        order = np.argsort(-room) if diff > 0 else np.argsort(room)
        remaining = abs(diff)
        for idx in order:
            if remaining <= 1e-14:
                break
            take = min(room[idx], remaining)
            if take > 0:
                h[idx] += take if diff > 0 else -take
                remaining -= take
    return h
