            break

                break
            # Decrease the largest
            i = idx[np.argmax(v[idx])]
            delta = min(v[i], e)

            e -= delta
        else:

            e += delta
    return v

    if B.shape[1] > 1:
        theta[1:] = np.linalg.lstsq(B[:, 1:], q_centered, rcond=None)[0]
    return theta

def surrogate_score(h):

def score_exact(h):

    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(R) * dx)

def repair_feasible(v, target_sum):
    v = np.array(v, dtype=float)
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    e = np.sum(v) - target_sum
    for _ in range(5000):
        if abs(e) < 1e-14:
            break
        if e > 0:
            idx = np.where(v > 1e-14)[0]
            if len(idx) == 0:
                break
            i = idx[np.argmax(v[idx])]
            delta = min(v[i], e)
            v[i] -= delta
            e -= delta
        else:
            idx = np.where(v < 1.0 - 1e-14)[0]
            if len(idx) == 0:
                break
            i = idx[np.argmax(1.0 - v[idx])]
            delta = min(1.0 - v[i], -e)
            v[i] += delta
            e += delta
    return v

def resample_to(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return repair_feasible(h_new, n_new / 2.0)

    x = np.arange(n) / n

        cols.append(np.cos(2.0 * np.pi * k * x))
        cols.append(np.sin(2.0 * np.pi * k * x))
    return np.column_stack(cols)

        h5, c5 = targeted_local_search(best_h, n_steps=200000, rng=rng, time_limit=t_alloc5)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()

    # Final validation and repair
    best_h = repair_feasible(best_h, len(best_h) / 2.0)
    final_c = score_exact(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, best_c, len(best_h)
