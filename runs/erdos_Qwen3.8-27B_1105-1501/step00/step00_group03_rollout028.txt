```python
    m = n // 2

    scaled = dx * corr
    # Log-sum-exp
```

```python
    m = n // 2
    h = np.empty(n)
    h[:m] = u
    h[n-m:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h
```

```python
    return np.correlate(h, 1.0 - h, mode="full")

def grad_rows(h, n):
    G = np.zeros((2*n-1, n))
    for idx in range(2*n-1):
        k = idx - (n-1)
        s = max(0, -k)
        e = min(n, n-k)
        idxs = np.arange(s, e)
        G[idx, idxs + k] += (1.0 - h[idxs])
        G[idx, idxs] += -h[idxs + k]
    return G

def reduced_grad(h, n):
    m = n // 2
    G_full = grad_rows(h, n)

    G_red = G_full[:, :m] - G_full[:, n-1 : n-1-m : -1]

    # So G_red[:, j] = G_full[:, j] - G_full[:, n-1-j] for j=0..m-1. ✓
    return G_red

def lse_objective_and_grad(u, n, beta):

    return log_sum, grad

        return g

    return res.x

def active_subgradient(u, n, n_iter=200, eps=1e-4):
    """Active-set subgradient method with backtracking line search."""
    m = n // 2
    dx = 2.0 / n
    best_u = u.copy()
    best_score = exact_score(h_from_u(u, n))
    
    for it in range(n_iter):
        h = h_from_u(u, n)
        corr = corr_values(h)
        scaled = dx * corr
        max_val = np.max(scaled)
        
        # Active set
        active_mask = scaled >= max_val - eps
        if np.sum(active_mask) == 0:
            active_mask[astype(np.argmax(scaled))] = True
        
        G_red = reduced_grad(h, n)
        # Subgradient = average of active constraint gradients
        g = dx * (G_red[active_mask].sum(axis=0) / np.sum(active_mask))
        
        # Projected gradient step with backtracking
        step = 0.1
        improved = False
        for _ in range(50):
            u_new = np.clip(u - step * g, 0, 1)
            score_new = exact_score(h_from_u(u_new, n))
            if score_new < best_score - 1e-12:
                u = u_new
                best_score = score_new
                best_u = u.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    
    return best_u, best_score

def slsqp_active(u, n, top_k=15, maxiter=300):
    """SLSQP on the top-k most active constraints."""
    m = n // 2
    dx = 2.0 / n
    h = h_from_u(u, n)
    corr = corr_values(h)
    scaled = dx * corr
    
    # Select top-k constraints
    top_idx = np.argsort(scaled)[-top_k:][::-1]
    theta0 = np.max(scaled) + 1e-8
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    cons = []
    for idx in top_idx:
        def make_c(i):
            def c(z):
                return z[-1] - dx * corr_values(h_from_u(z[:m], n))[i]
            def jac(z):

                G_red = reduced_grad(hh, n)
                j = np.zeros(m + 1)
                j[:m] = -dx * G_red[i]
                j[-1] = 1.0
                return j
            return c, jac
        c, j = make_c(idx)
        cons.append({'type': 'ineq', 'fun': c, 'jac': j})
    
    bounds = [(0.0, 1.0)] * m + [(0.0, 3.0)]
    x0 = np.concatenate([u, [theta0]])
    
    try:

            u_new = np.clip(res.x[:m], 0, 1)
            return u_new
    except:
        pass
    return u
```

Let me try a different approach: use the "cutting plane" or "ellipsoid" idea. At each iteration:
1. Evaluate all constraints
2. Find the most violated constraint
3. Add it to a working set
4. Solve a QP on the working set
5. Use the solution as the next iterate

This is essentially what SLSQP does internally. Let me just implement it manually.

Simplified SLSQP:
- Maintain a working set W of active constraints (|W| <= m+1)
- At each iteration:
  a. Compute all constraint values
  b. If most violated constraint v is not in W: add it, remove the least important one (or the one with smallest multiplier)
  c. Solve the QP: minimize d (the increase in theta) subject to the linearized constraints in W
  d. Take a step

```python
import numpy as np
from scipy.optimize import minimize
import time


def h_from_u(u, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = u
    h[n - m:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h


def u_from_h(h, n):
    m = n // 2
    h_sym = 0.5 * (h + (1.0 - h[::-1]))
    return h_sym[:m].copy()


def corr_values(h):
    return np.correlate(h, 1.0 - h, mode="full")


def exact_score(h):
    n = len(h)
    return float(np.max(corr_values(h)) * (2.0 / n))


def grad_rows(h, n):
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        s = max(0, -k)
        e = min(n, n - k)
        idxs = np.arange(s, e)
        G[idx, idxs + k] += (1.0 - h[idxs])
        G[idx, idxs] += -h[idxs + k]
    return G


def reduced_grad(h, n):
    m = n // 2
    G_full = grad_rows(h, n)
    rev_idx = np.arange(n - 1, n - m - 1, -1)
    return G_full[:, :m] - G_full[:, rev_idx]


def lse_val_grad(u, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = h_from_u(u, n)
    corr = corr_values(h)
    scaled = dx * corr
    max_val = np.max(scaled)
    shifted = scaled - max_val
    exps = np.exp(shifted)
    log_sum = max_val + np.log(np.sum(exps))
    weights = exps / np.sum(exps)
    G_red = reduced_grad(h, n)
    grad = dx * (weights @ G_red)
    return log_sum, grad


def optimize_lse(u, n, beta, maxiter=150):
    m = n // 2
    bounds = [(0.0, 1.0)] * m

    def obj(u_curr):
        val, _ = lse_val_grad(u_curr, n, beta)
        return val

    def obj_grad(u_curr):
        _, g = lse_val_grad(u_curr, n, beta)
        return g

    try:
        res = minimize(obj, u.copy(), method='L-BFGS-B', jac=obj_grad,
                       bounds=bounds, maxiter=maxiter, ftol=1e-15, gtol=1e-12)
        return np.clip(res.x, 0, 1)
    except Exception:
        return u


def subgradient_refine(u, n, n_iter=500, eps=1e-5):
    m = n // 2
    dx = 2.0 / n
    best_u = u.copy()
    best_score = exact_score(h_from_u(u, n))
    step_init = 0.05

    for it in range(n_iter):
        h = h_from_u(u, n)
        corr = corr_values(h)
        scaled = dx * corr
        max_val = np.max(scaled)
        active_mask = scaled >= max_val - eps
        if not np.any(active_mask):
            active_mask[np.argmax(scaled)] = True
        G_red = reduced_grad(h, n)
        g = (G_red[active_mask].sum(axis=0) / np.sum(active_mask))

        improved = False
        step = step_init / (1.0 + it * 0.01)
        for _ in range(40):
            u_new = np.clip(u - step * g, 0.0, 1.0)
            s_new = exact_score(h_from_u(u_new, n))
            if s_new < best_score - 1e-14:
                u = u_new
                best_score = s_new
                best_u = u.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            if it > 50:
                break
    return best_u, best_score


def newton_refine(u, n, beta=5000, n_iter=100):
    m = n // 2
    dx = 2.0 / n
    best_u = u.copy()
    best_score = exact_score(h_from_u(u, n))

    for it in range(n_iter):
        h = h_from_u(u, n)
        corr = corr_values(h)
        scaled = dx * corr
        max_val = np.max(scaled)
        shifted = scaled - max_val
        exps = np.exp(shifted)
        p = exps / np.sum(exps)
        G_red = reduced_grad(h, n)
        g = dx * (p @ G_red)
        # Hessian of lse
        H = np.zeros((m, m))
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            diff = dx * (G_red[k] - g)
            H += p[k] * beta * np.outer(diff, diff)
        # regularize
        reg = 1e-6 * np.trace(H) / m + 1e-10
        H += reg * np.eye(m)
        try:
            d = np.linalg.solve(H, -g)
        except np.linalg.LinAlgError:
            d = -g / (np.linalg.norm(g) + 1e-15)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        # line search
        improved = False
        alpha = 1.0
        for _ in range(30):
            u_new = np.clip(u + alpha * d, 0.0, 1.0)
            s_new = exact_score(h_from_u(u_new, n))
            if s_new < best_score - 1e-14:
                u = u_new
                best_score = s_new
                best_u = u.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return best_u, best_score


def random_perturb_and_optimize(u, n, rng, scale=0.05, beta=500, maxiter=60):
    m = len(u)
    u_pert = np.clip(u + scale * rng.randn(m), 0, 1)
    u_opt = optimize_lse(u_pert, n, beta, maxiter)
    return u_opt

    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left(frac=0.92):

    # Get initial values

    best_n = n_init
    best_score = exact_score(h_from_u(u0, n_init))

    # Phase 1: Log-sum-exp with increasing beta from initial
    beta_schedule = [5, 20, 100, 500, 2000, 8000]
    for beta in beta_schedule:
        if not time_left():
            break
        u_curr = optimize_lse(best_u, best_n, beta, maxiter=120)
        s_curr = exact_score(h_from_u(u_curr, best_n))

            best_score = s_curr
            best_u = u_curr.copy()

    # Phase 2: Newton refinement

        cut = int(m * frac)
        u_s = np.zeros(m)
        u_s[:cut] = 1.0
        start_list.append(u_s)
    # Cosine patterns

        u_c = 0.5 + 0.5 * np.cos(np.pi * (jj / m + phase))
        start_list.append(np.clip(u_c, 0, 1))

    for i in range(10):

    for u_s in start_list:
        if not time_left():

            best_u = u_curr.copy()
            if time_left():
```
