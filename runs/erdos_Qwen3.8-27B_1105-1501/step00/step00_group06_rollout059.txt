import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(h, target):
    """Euclidean projection onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    alpha = (lo + hi) / 2.0
    return np.clip(h + alpha, 0.0, 1.0)

def corr_gradient(h, j):
    """Compute d(corr[j])/dh for correlation index j (0 to 2n-2)."""
    n = len(h)
    shift = j - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        # d/dh[m] = (1-h[m-shift]) for shift<=m<n, minus h[m+shift] for 0<=m<n-shift
        for m in range(shift, n):
            grad[m] += (1.0 - h[m - shift])
        for m in range(0, n - shift):
            grad[m] -= h[m + shift]
    else:
        s = -shift
        for m in range(0, n - s):
            grad[m] += (1.0 - h[m + s])
        for m in range(s, n):
            grad[m] -= h[m - s]
    return grad
