The key insight is: at the optimum, the KKT conditions say that for all active lags k*, we have c_{k*} = C5 (the max), and the gradients of the active c_k are linearly dependent with the gradient of the constraint.

So at the optimum, there exist non-negative weights λ_k for active lags such that:
sum_k λ_k * grad(c_k) + μ * 1 = 0 (for interior points)
where μ is the Lagrange multiplier for the sum constraint.

This means the weighted sum of gradients of active c_k's is constant (parallel to the all-ones vector). In other words, the "force" from each active lag balances out.

This gives us a system of equations that we can try to solve directly:
- For each active k: c_k(h) = C5
- sum_k λ_k * grad(c_k, h) = -μ * 1 for some μ
- λ_k >= 0
- sum(h) = S, 0 <= h <= 1

This is a system of (|active| + n + 1) equations with (|active| + n + 1) unknowns (h_1,...,h_n, λ_1,...,λ_|active|, μ). It's nonlinear but might be solvable.

    if f(lo) >= 0:
        return np.clip(h, 0.0, 1.0)
    if f(hi) <= 0:
        return np.clip(h, 0.0, 1.0)
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _smooth_vals(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    s = alpha * (c - cm)
    s -= np.max(s)
    return c, cm, s

def smooth_obj(h, alpha):
    c, cm, s = _smooth_vals(h, alpha)
    return cm + np.log(np.sum(np.exp(s))) / alpha

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c, cm, s = _smooth_vals(h, alpha)
    w = np.exp(s)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk < 1e-14:
            continue
        if k >= 0:
            for i in range(n - k):
                grad[i+k] += wk * (1.0 - h[i])
                grad[i] -= wk * h[i+k]
        else:
            for i in range(-k, n):
                grad[i+k] += wk * (1.0 - h[i])
                grad[i] -= wk * h[i+k]
    return grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    S = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    def elapsed():
        return time.time() - t0
    
    # SLSQP
    for alpha in [20, 50, 100, 300]:
        if elapsed() > budget_s * 0.4:
            break
        try:
            res = minimize(
                smooth_obj, best_h, args=(alpha,),
                jac=smooth_obj_grad,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq',
                              'fun': lambda x: np.sum(x) - S,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 5000, 'ftol': 1e-15}
            )
            ht = project_feasible(res.x, S)
            cv = evaluate_c5(ht)
            if cv < best_c5:
                best_c5 = cv
                best_h = ht
        except:
            pass
    
    # L-BFGS-B with penalty
    if elapsed() < budget_s * 0.6:
        for pen in [1e4, 1e6, 1e8]:
            if elapsed() > budget_s * 0.6:
                break
            def pobj(x):
                return smooth_obj(x, 200) + pen * (np.sum(x) - S)**2
            def pgrad(x):
                g = smooth_obj_grad(x, 200)
                g += 2.0 * pen * (np.sum(x) - S)
                return g
            try:
                res = minimize(
                    pobj, best_h, jac=pgrad,
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n,
                    options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10}
                )
                ht = project_feasible(res.x, S)
                cv = evaluate_c5(ht)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = ht
            except:
                pass
    
    # Projected gradient
    if elapsed() < budget_s * 0.8:
        alpha = 300
        step = 1.0 / alpha
        h_cur = best_h.copy()
        for it in range(10000):
            if elapsed() > budget_s * 0.8:
                break
            g = smooth_obj_grad(h_cur, alpha)
            h_new = project_feasible(h_cur - step * g, S)
            v = evaluate_c5(h_new)
            if v < best_c5 - 1e-15:
                best_c5 = v
                best_h = h_new.copy()
                h_cur = h_new
                step = min(step * 1.01, 10.0 / alpha)
            else:
                step *= 0.9
                if step < 1e-10:
                    break
    
    # Random restarts
    if elapsed() < budget_s * 0.97:
        for _ in range(50):
            if elapsed() > budget_s * 0.97:
                break
            h_p = best_h + np.random.randn(n) * 0.01
            h_p = project_feasible(h_p, S)
            try:
                res = minimize(
                    smooth_obj, h_p, args=(200,),
                    jac=smooth_obj_grad,
                    method='SLSQP',
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq',
                                  'fun': lambda x: np.sum(x) - S,
                                  'jac': lambda x: np.ones(n)}],
                    options={'maxiter': 500, 'ftol': 1e-12}
                )
                ht = project_feasible(res.x, S)
                cv = evaluate_c5(ht)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = ht
            except:
                pass
    
    # Final verification
    best_h = project_feasible(best_h, S)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, n
