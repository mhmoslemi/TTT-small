from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def grad_for_lag(s, h, n, dx):
    """Gradient of overlap at shift s with respect to h."""
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        idx1 = np.arange(s, n)
        g[idx1] += (1.0 - h[idx1 - s])
        idx2 = np.arange(0, n - s)
        g[idx2] += (-h[idx2 + s])
    else:
        d = -s
        idx1 = np.arange(0, n - d)
        g[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        g[idx2] += (-h[idx2 - d])
    return g * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def active_set_lags(h, tol=0.001):
    """Find lags within tol of the maximum."""
    overlaps = get_overlaps(h)
    c5 = np.max(overlaps)
    active = np.where(overlaps >= c5 - tol)[0]
    # Convert to shift values
    n = len(h)
    shifts = active - (n - 1)
    return shifts, c5, overlaps

def slsqp_optimize(h, n, shifts, max_iter=100):
    """Minimize epigraph t s.t. F(s) <= t for active s, 0<=h<=1, sum(h)=n/2."""
    dx = 2.0 / n
    
    def objective(x):
        # x = [h_0, ..., h_{N-1}, t]
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    constraints = []
    for s in shifts:
        def make_constraint(s_val):
            def con(x):
                h_val = x[:n]
                t_val = x[-1]
                return t_val - (get_single_overlap(s_val, h_val, n, dx))
            def con_grad(x):
                h_val = x[:n]
                g = np.zeros(n + 1)
                g[:-1] = -grad_for_lag(s_val, h_val, n, dx)
                g[-1] = 1.0
                return g
            return {'type': 'ineq', 'fun': con, 'jac': con_grad}
        constraints.append(make_constraint(s))
    
    # Sum constraint
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - n/2.0,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    bounds = [(0, 1)] * n + [(0, 1)]
    
    x0 = np.concatenate([h, [evaluate_c5(h) + 0.01]])
    
    result = minimize(
        objective, x0, jac=objective_grad,
        method='SLSQP', bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    
    return result['x'][:n]

def get_single_overlap(s, h, n, dx):
    """Compute overlap at a single shift s."""
    if s == 0:
        return dx * np.sum(h * (1 - h))
    elif s > 0:
        return dx * np.sum(h[s:n] * (1 - h[0:n-s]))
    else:
        d = -s
        return dx * np.sum(h[0:n-d] * (1 - h[d:n]))

        exps = np.exp(alpha * (overlaps - np.max(overlaps)))
        w = exps / np.sum(exps)
        
        g = np.zeros(n)
        for i in range(len(overlaps)):
            if w[i] > 1e-10:
                g += w[i] * grad_overlap(i - center, h_cur, n, dx)
        
        lr = base_lr / (1.0 + 0.01 * step)
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best = h_new.copy()
        h_cur = h_new
    
    return best, best_c5

def targeted_transfer(h, n, num_rounds=100, trials_per_round=500):
    """Targeted pair transfers guided by gradients."""
    dx = 2.0 / n
    best = h.copy()
    best_c5 = evaluate_c5(h)
    rng = np.random.default_rng(123)
    center = n - 1
    
    for round_num in range(num_rounds):
        overlaps = get_overlaps(best)
        # Find most active lags
        top_k = np.argsort(overlaps)[::-1][:5]
        active_shifts = top_k - center
        
        # Compute combined gradient
        g = np.zeros(n)
        for s in active_shifts:
            g += grad_overlap(s, best, n, dx)
        
        # Try moves in the direction of -g (reduce active overlaps)
        step_sizes = [0.01, 0.005, 0.001, 0.0005, 0.0001]
        
        for _ in range(trials_per_round):
            # Choose a pair (i, j) and direction
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            for step in step_sizes:
                for sign in [1, -1]:
                    # Move: h[i] += sign*step, h[j] -= sign*step
                    # Check if this reduces the gradient dot product
                    delta = np.zeros(n)
                    delta[i] = sign * step
                    delta[j] = -sign * step
                    pred_reduction = -np.dot(g, delta)
                    if pred_reduction <= 0:
                        continue
                    
                    h_new = best.copy()
                    h_new[i] += sign * step
                    h_new[j] -= sign * step
                    if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                        continue
                    
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best = h_new.copy()
                        break
    
    return best, best_c5

def make_binary_start(n, rng):
    """Create a random binary-feasible start."""
    ones = n // 2
    positions = rng.choice(n, size=ones, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h
