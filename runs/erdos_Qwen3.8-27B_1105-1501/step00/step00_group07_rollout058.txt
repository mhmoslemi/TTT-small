def repair_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # Distribute diff
    if abs(diff) < 1e-12:
        return h
    n = len(h)
    # Add/subtract uniformly
    per_elem = diff / n
    h_new = h + per_elem
    h_new = np.clip(h_new, 0.0, 1.0)
    # Fix residual
    residual = target_sum - h_new.sum()
    if residual > 0:
        # Add to elements with room
        room = 1.0 - h_new
        idx = np.argsort(-room)
        for i in idx:
            add = min(room[i], residual)
            h_new[i] += add
            residual -= add
            if residual < 1e-14:
                break
    elif residual < 0:
        # Remove from elements with mass
        idx = np.argsort(-h_new)
        for i in idx:
            rem = min(h_new[i], -residual)
            h_new[i] -= rem
            residual += rem
            if residual > -1e-14:
                break

def make_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum=h_target"""
    h = np.array(h, dtype=float)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Fix sum using water-filling
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Add/subtract uniformly to elements with room
    if diff > 0:
        room = 1.0 - h
        room_idx = np.where(room > 1e-10)[0]
        if len(room_idx) == 0:
            return h
        per = diff / len(room_idx)
        h[room_idx] = np.minimum(h[room_idx] + per, 1.0)
        # Fix residual
        resid = target - h.sum()
        if resid > 0:
            for i in np.argsort(-room):
                add = min(1.0 - h[i], resid)
                h[i] += add
                resid -= add
                if resid < 1e-14:
                    break
