def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def build_symmetric(u, n):
    h = np.empty(n)
    m = n // 2
    h[:m] = u
    for i in range(m):
        h[n - 1 - i] = 1.0 - u[i]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def symmetrize(h):
    return np.clip(0.5 * (h + 1.0 - h[::-1]), 0.0, 1.0)

def get_reduced(h_sym):
    return h_sym[:len(h_sym) // 2].copy()

def corr_val(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_grad(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        for i in range(lo, hi + 1):
            j = i + k
            grad[idx, j] += 1.0 - h[i]
            grad[idx, i] -= h[j]
    return grad

    m = n // 2
    rg = np.empty((fg.shape[0], m))
    for j in range(m):
        rg[:, j] = fg[:, j] - fg[:, n - 1 - j]
    return rg

    c0 = corr_val(h0)

    else:
        idx = np.arange(2 * n - 1)

        return z[m]

        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

        J[:, m] = 1.0
        return J

                   constraints=[{'type': 'ineq', 'fun': cfun, 'jac': cjac}],

                   options={'maxiter': maxiter, 'ftol': 1e-14})

    h_opt = build_symmetric(u_opt, n)
    return h_opt, compute_score(h_opt)

def lse_optimize(u0, n, T=200.0, maxiter=500):
    """Optimize using log-sum-exp smooth approximation."""
    m = n // 2
    dx = 2.0 / n
    
    def f_and_grad(u):
        h = build_symmetric(u, n)
        c = corr_val(h)
        scaled = T * dx * c
        max_s = np.max(scaled)
        # Numerical stability
        exps = np.exp(scaled - max_s)
        val = (max_s + np.log(np.sum(exps))) / T
        # Gradient

        rg = reduce_grad(fg, n)
        weights = exps / np.sum(exps)
        grad = dx * (weights @ rg)
        return val, grad
    
    res = minimize(lambda u: f_and_grad(u)[0], u0, method='L-BFGS-B',
                   jac=lambda u: f_and_grad(u)[1],
                   bounds=[(0, 1)] * m,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    
    h_opt = build_symmetric(res.x, n)
    return h_opt, compute_score(h_opt)
