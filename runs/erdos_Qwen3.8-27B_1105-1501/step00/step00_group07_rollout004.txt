import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(R) * dx)

def score_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, n):
    target = n / 2.0
    v = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10000):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.argmax(1.0 - v)
            step = min(1.0 - v[idx], diff)
            v[idx] += step
        else:
            idx = np.argmax(v)
            step = min(v[idx], -diff)
            v[idx] -= step
    if abs(v.sum() - target) > 1e-9:
        v += (target - v.sum()) / n
        v = np.clip(v, 0.0, 1.0)
        for _ in range(100):
            diff = target - v.sum()
            if abs(diff) < 1e-12:
                break
            idx = np.argmax(1.0 - v) if diff > 0 else np.argmax(v)
            if diff > 0:
                v[idx] += min(1.0 - v[idx], diff)
            else:
                v[idx] -= min(v[idx], -diff)
    return v

def compute_gradient(h, n, top_k=8):
    R = score_full(h)
    R_max = R.max()
    order = np.argsort(R)[::-1]
    active = order[:top_k]
    grad = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)
        w = R[k_idx]
        for j in range(n):
            term = 0.0
            if 0 <= j - L < n:
                term += (1.0 - h[j - L])
            if 0 <= j + L < n:
                term -= h[j + L]
            grad[j] += w * term
    return grad, R

def resample_to(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return repair_feasible(h_new, n_new)

def gradient_descent(h, n, t0, t_end, rng):
    C = score_exact(h)
    best_h = h.copy()
    best_C = C
    lr = 0.05
    stall = 0
    while time.time() < t_end:
        if time.time() - t0 > 10:
            break
        grad, R = compute_gradient(h, n, top_k=8)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_try = h - lr * grad / gnorm
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = repair_feasible(h_try, n)
        C_try = score_exact(h_try)
        if C_try < C - 1e-13:
            h = h_try
            C = C_try
            stall = 0
            if C < best_C:
                best_C = C
                best_h = h.copy()
            lr = min(lr * 1.15, 0.5)
        else:
            lr *= 0.7
            stall += 1
            if stall > 80:
                # Perturb to escape
                h = h + rng.normal(0, lr * 0.5, n)
                h = np.clip(h, 0, 1)
                h = repair_feasible(h, n)
                C = score_exact(h)
                stall = 0
    return best_h, best_C

def fourier_basis(n, p):
    x = (np.arange(n) + 0.5) * 2.0 / n
    B = np.zeros((n, p))
    B[:, 0] = 1.0
    col = 1
    for k in range(1, n):
        if col >= p:
            break
        B[:, col] = np.cos(k * np.pi * x)
        col += 1
        if col >= p:
            break
        B[:, col] = np.sin(k * np.pi * x)
        col += 1
    for k in range(p - col):
        B[:, col + k] = rng42_global.rand(n)
    return B

def fourier_optimize(h_init, n, p, t0, t_end, rng):

def simulated_annealing(h, n, t0, t_end, rng):
    C = score_exact(h)
    best_h = h.copy()
    best_C = C
    T = 0.02
    cool = 0.999
    while time.time() < t_end and T > 1e-6:
        # Random pair swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.1, 0.1)
        if h[i] + d < 0 or h[i] + d > 1:
            continue
        if h[j] - d < 0 or h[j] - d > 1:
            continue
        h[i] += d
        h[j] -= d
        C_new = score_exact(h)
        delta = C_new - C
        if delta < 0 or rng.rand() < math.exp(-delta / max(T, 1e-10)):
            C = C_new
            if C < best_C:
                best_C = C
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        T *= cool
    return best_h, best_C
