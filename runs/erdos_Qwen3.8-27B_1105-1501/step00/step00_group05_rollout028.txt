import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_feasibility(h, target_sum):
    """Repair h to satisfy sum constraint and box constraints."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute to points with room
    if diff > 0:
        # Need to increase some values
        room = 1.0 - h
        order = np.argsort(-room)
    else:
        # Need to decrease some values
        room = h
        order = np.argsort(-room)
    
    remaining = abs(diff)
    sign = 1 if diff > 0 else -1
    for idx in order:
        take = min(abs(room[idx]), remaining)
        h[idx] += sign * take
        remaining -= take
        if remaining <= 1e-12:
            break
    return h

def local_search(h, n, target_sum, rng, max_iters, time_limit, T_start, T_end, step_range):
    """Intensive local search with simulated annealing."""
    t0 = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    total_iters = max_iters
    alpha = (T_end / T_start) ** (1.0 / max(total_iters, 1))
    T = T_start
    
    for it in range(total_iters):
        if time.time() - t0 > time_limit:
            break
        T *= alpha
        
        # Move type 1: single pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        s = step_range[0] + rng.rand() * (step_range[1] - step_range[0])
        max_d = min(s, 1.0 - best_h[i], best_h[j])
        if max_d <= 0:
            continue
        delta = max_d * (2.0 * rng.rand() - 1.0)
        
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new <= best_c5 or (T > 1e-8 and rng.rand() < np.exp(-(c5_new - best_c5) / T)):
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5
