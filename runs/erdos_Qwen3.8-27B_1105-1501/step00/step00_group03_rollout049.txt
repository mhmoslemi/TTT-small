    return float(corr.max() * dx)

def make_feasible(z, target=None):
    """Project z onto {0≤x≤1, sum(x)=target} using bisection on shift."""

    # Bisection to find lambda such that sum(clip(z+lambda, 0, 1)) = target
    lam_lo = -1.0 - z.max() + 1e-15
    lam_hi = 1.0 - z.min() - 1e-15
    # G(lambda) = sum(clip(z+lambda, 0, 1))
    for _ in range(60):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        s = np.clip(z + lam_mid, 0.0, 1.0).sum()
        if s < target:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    h = np.clip(z + 0.5 * (lam_lo + lam_hi), 0.0, 1.0)
    # Repair sum to exact target
    resid = target - h.sum()
    if abs(resid) > 1e-10:
        if resid > 0:

            headroom = 1.0 - h
            order = np.argsort(-headroom)
            for idx in order:
                add = min(resid, headroom[idx])
                if add <= 0:

                h[idx] += add

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    """Compute exact C5 = max(np.correlate(h, 1-h, 'full')) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def make_feasible(z, target=None):
    """Project z onto {0<=x<=1, sum(x)=target} via bisection on uniform shift."""
    z = np.asarray(z, dtype=float).ravel()
    n = len(z)
    if target is None:
        target = n / 2.0
    if not np.all(np.isfinite(z)):
        z = np.zeros(n)
    lam_lo = -1.0 - z.max() + 1e-14
    lam_hi = 1.0 - z.min() - 1e-14
    for _ in range(64):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        s = np.clip(z + lam_mid, 0.0, 1.0).sum()
        if s < target:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    h = np.clip(z + 0.5 * (lam_lo + lam_hi), 0.0, 1.0)
    # Repair sum
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        steps = 0
        while abs(resid) > 1e-13 and steps < 2 * n:
            if resid > 0:
                headroom = 1.0 - h
                idx = np.argmax(headroom)
                add = min(resid, headroom[idx])
                if add <= 0:
                    break
                h[idx] += add
            else:
                idx = np.argmax(h)
                rem = min(-resid, h[idx])
                if rem <= 0:
                    break
                h[idx] -= rem
            resid = target - h.sum()
            steps += 1
    h = np.clip(h, 0.0, 1.0)
    return h


def manual_correlations(h):
    """Compute F_s = sum_j h[j+s]*(1-h[j]) for s=-(n-1)..(n-1)."""
    n = len(h)
    num = 2 * n - 1
    F = np.zeros(num)
    for idx in range(num):
        s = idx - (n - 1)
        if s >= 0:
            m = n - s
            F[idx] = np.dot(h[s:], 1.0 - h[:m])
        else:
            t = -s
            m = n - t
            F[idx] = np.dot(h[:m], 1.0 - h[t:])
    return F


def smooth_obj_and_grad(h, beta):
    """Log-sum-exp smooth max of dx*F_s and its gradient."""
    n = len(h)
    dx = 2.0 / n
    F = manual_correlations(h)
    C = dx * F
    Cmax = C.max()
    shifted = beta * (C - Cmax)
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    log_sum = (max_sh + np.log(exp_sh.sum())) / beta
    obj = Cmax + log_sum
    w = exp_sh / exp_sh.sum()
    grad = np.zeros(n)
    num = 2 * n - 1
    for idx in range(num):
        if w[idx] < 1e-15:
            continue
        s = idx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += w[idx] * (1.0 - h[:m])
            grad[:m] += w[idx] * (-h[s:])
        else:
            t = -s
            m = n - t
            grad[:m] += w[idx] * (1.0 - h[t:])
            grad[t:] += w[idx] * (-h[:m])
    return obj, dx * grad


def direct_obj(z):
    """Objective for direct minimax: theta."""
    return z[-1]


def direct_obj_jac(z):
    n = len(z) - 1
    j = np.zeros(len(z))
    j[-1] = 1.0
    return j


def direct_ineq_fun(z):
    """Inequality constraints: theta - dx*F_s >= 0."""
    h = z[:-1]
    n = len(h)
    dx = 2.0 / n
    F = manual_correlations(h)
    return z[-1] - dx * F


def direct_ineq_jac(z):
    h = z[:-1]
    n = len(h)
    dx = 2.0 / n
    F = manual_correlations(h)
    num = 2 * n - 1
    J = np.zeros((num, n + 1))
    for idx in range(num):
        s = idx - (n - 1)
        if s >= 0:
            m = n - s
            J[idx, s:] = (1.0 - h[:m])
            J[idx, :m] += (-h[s:])
        else:
            t = -s
            m = n - t
            J[idx, :m] = (1.0 - h[t:])
            J[idx, t:] += (-h[:m])
        J[idx] *= -dx
        J[idx, -1] = 1.0
    return J


def direct_eq_fun(z):
    return z[:-1].sum() - len(z[:-1]) / 2.0


def direct_eq_jac(z):
    n = len(z) - 1
    j = np.zeros(len(z))
    j[:-1] = 1.0
    return j


def smooth_solve(h0, beta, maxiter=60):
    """Solve smooth surrogate problem."""
    n = len(h0)
    h0 = make_feasible(h0)

    def fun(x):
        v, _ = smooth_obj_and_grad(x, beta)
        return v

    def jac(x):
        _, g = smooth_obj_and_grad(x, beta)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': direct_eq_fun, 'jac': direct_eq_jac}]
    try:
        res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_res = make_feasible(res.x)
            return h_res
    except Exception:
        pass
    return h0


def direct_solve(h0, maxiter=80):
    """Solve direct minimax problem."""
    n = len(h0)
    h0 = make_feasible(h0)
    score0 = official_score(h0)
    theta0 = score0 + 1e-6
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.75)]
    cons = [
        {'type': 'eq', 'fun': direct_eq_fun, 'jac': direct_eq_jac},
        {'type': 'ineq', 'fun': direct_ineq_fun, 'jac': direct_ineq_jac}
    ]
    try:
        res = minimize(direct_obj, z0, method='SLSQP', jac=direct_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_res = make_feasible(res.x[:-1])
            return h_res
    except Exception:
        pass
    return h0


def interpolate_to(h_src, n_target):
    """Interpolate h_src to length n_target using midpoint sampling."""
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_tgt, target=n_target / 2.0)


def antisym_transform(h):
    """Apply anti-symmetric mirror: h_new[i] = 1 - h[n-1-i]."""
    n = len(h)
    h_new = 1.0 - h[::-1].copy()
    return make_feasible(h_new, target=n / 2.0)


def random_perturb(h, sigma, rng):
    """Add random perturbation and re-project."""
    noise = rng.normal(0, sigma, len(h))
    return make_feasible(h + noise)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - max(5.0, 0.05 * budget_s)

    # Get parent
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if pv.size >= 2 and np.all(np.isfinite(pv)) and np.all(pv >= -1e-9) and np.all(pv <= 1 + 1e-9):
            parent = make_feasible(pv, target=pv.size / 2.0)
    except Exception:
        parent = None

    if parent is None:
        n_fallback = 45
        parent = make_feasible(np.linspace(0.3, 0.7, n_fallback), target=n_fallback / 2.0)

    best_h = parent.copy()
    best_score = official_score(parent)
    best_n = len(parent)

    def try_update(h_cand):
        nonlocal best_h, best_score, best_n
        if h_cand is None:
            return
        h_cand = make_feasible(h_cand, target=len(h_cand) / 2.0)
        if not np.all(np.isfinite(h_cand)):
            return
        sc = official_score(h_cand)
        if not np.isfinite(sc):
            return
        tol = 1e-12
        if sc < best_score - tol:
            best_score = sc
            best_h = h_cand.copy()
            best_n = len(h_cand)
        elif abs(sc - best_score) <= tol and len(h_cand) < best_n:
            best_h = h_cand.copy()
            best_score = sc
            best_n = len(h_cand)

    try_update(parent)

    # Try antisymmetric transform
    try_update(antisym_transform(parent))

    # Determine n_steps for budget
    remaining = deadline - time.time()
    if remaining < 30:

        try_update(best_h)
        bound = best_score + 1e-13
        return best_h, bound, best_n

    base_n = best_n
    n_options = []
    n_options.append(base_n)
    # Try a slightly shorter

            break
        remaining = deadline - time.time()

            break

        if n_opt == base_n:

        # Random perturbations
        for sigma in [0.02, 0.05, 0.10, 0.15]:

    if time.time() < deadline - 20:

            if time.time() > deadline:
                break
            h_rand = make_feasible(rng2.uniform(0, 1, best_n), target=best_n / 2.0)
            # Shape it: bias toward a simpler structure
            h_shape = make_feasible(
                np.clip(rng2.normal(0.5, 0.25, best_n), 0, 1), target=best_n / 2.0)
            h_try = 0.5 * h_rand + 0.5 * h_shape
            h_try = make_feasible(h_try, target=best_n / 2.0)
            # Quick smooth polish
            h_try = smooth_solve(h_try, 100, maxiter=30)
            h_try = direct_solve(h_try, maxiter=50)
            try_update(h_try)

    # Final validity check

        final_score = best_score
    # Add tiny safety margin
    bound = final_score + 1e-13

    return best_h, bound, best_n
