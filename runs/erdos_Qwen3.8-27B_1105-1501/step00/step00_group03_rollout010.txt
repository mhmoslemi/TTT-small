def manual_values(h):
    """Compute unscaled correlation values for all lags."""
    n = len(h)
    one_minus_h = 1.0 - h
    # np.correlate(h, one_minus_h, 'full') gives 2n-1 values
    return np.correlate(h, one_minus_h, mode='full')

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = manual_values(h) * dx
    cmax = vals.max()
    # Stable log-sum-exp
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -100, 100)  # avoid overflow
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_gradient(h, beta):

    vals = manual_values(h) * dx

    # For each lag k (indexed by r = k + n - 1 in the full correlation),

def make_feasible(z):
    z = np.asarray(z, dtype=float).ravel().copy()
    n = z.shape[0]
    if n == 0:
        return z
    target = n / 2.0

    def G(lam):
        return np.sum(np.clip(z + lam, 0.0, 1.0))

    lo = -1.0 - float(z.max())
    hi = 1.0 - float(z.min())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if G(mid) < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    lam = 0.5 * (lo + hi)
    h = np.clip(z + lam, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                step = min(diff, 1.0 - h[idx])
                h[idx] += step
            else:
                step = min(-diff, h[idx])
                h[idx] -= step
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(vals.max())
    if beta < 1e-6:
        return cmax
    shifted = np.clip(beta * (vals - cmax), -500.0, 500.0)
    logsum = np.log(np.exp(shifted).sum())
    return cmax + logsum / beta


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(vals.max())
    shifted = np.clip(beta * (vals - cmax), -500.0, 500.0)
    exps = np.exp(shifted)

    if wsum < 1e-300:
        weights = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        weights = exps / wsum

        w = weights[r] * dx
        if w < 1e-18:
            continue
        k = r - (n - 1)

            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n - k])
        else:
            grad[-k:] += w * (1.0 - h[:n + k])
            grad[:n + k] += w * (-h[-k:])
    return grad

                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = cand.copy()
        except Exception:
            pass

    # Final: one more smooth refinement
    if time.time() < deadline - 5:
        n_fin2 = len(best_h)
        for beta in [500.0, 2000.0]:
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(
                    _sm_fun, best_h, args=(beta, n_fin2),
                    method='SLSQP', jac=_sm_jac,
                    bounds=_h_bounds(n_fin2),
                    constraints=[{'type': 'eq', 'fun': _sm_eq, 'jac': _sm_eq_jac}],
                    maxiter=80, ftol=1e-14
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    cand = make_feasible(res.x)
                    sc = official_score(cand)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_h = cand.copy()
            except Exception:
                pass

    # Ensure final feasibility
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    c5_bound = final_score + 1e-13
    return (best_h, c5_bound, len(best_h))
