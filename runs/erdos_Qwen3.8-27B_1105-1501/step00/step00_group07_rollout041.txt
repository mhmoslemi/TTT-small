    n_old = len(h)
    x_old = np.linspace(0, 1, n_old, endpoint=False) + 1.0 / (2.0 * n_old)
    x_new = np.linspace(0, 1, n_new, endpoint=False) + 1.0 / (2.0 * n_new)
    v = np.interp(x_new, x_old, h)
    return project_feas(v, n_new)

def pair_sweep(h, deadline, min_gain):
    n = len(h)
    current_c = exact_score(h)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for a in range(n):
            for b in range(a + 1, n):
                if time.time() > deadline:
                    return h, current_c
                pa, pb = h[a], h[b]
                # Check if there's room to transfer
                t_lo = -min(pa, 1.0 - pb)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw)) * dx


def project_feas(v, n):
    v = np.array(v, dtype=float).copy()
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(v)
    if res > 1e-10:
        idx = np.argsort(1.0 - v)
        for i in idx:
            add = min(res, 1.0 - v[i])
            v[i] += add
            res -= add
            if res < 1e-12:
                break
    elif res < -1e-10:
        idx = np.argsort(-v)
        for i in idx:
            rem = min(-res, v[i])
            v[i] -= rem
            res += rem
            if -res < 1e-12:
                break
    return v


def resample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feas(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    v = np.interp(x_new, x_old, h_old)
    return project_feas(v, n_new)


def score_transient(buf, a, b, t, n):
    dx = 2.0 / n
    if abs(t) < 1e-15:
        raw = np.correlate(buf, 1.0 - buf, mode="full")
        return float(np.max(raw)) * dx
    old_a = buf[a]
    old_b = buf[b]
    buf[a] = old_a + t
    buf[b] = old_b - t
    raw = np.correlate(buf, 1.0 - buf, mode="full")
    c = float(np.max(raw)) * dx
    buf[a] = old_a
    buf[b] = old_b
    return c

def pair_sweep(h_in, deadline, min_gain=1e-11, max_passes=50):
    n = len(h_in)
    buf = h_in.copy()
    cur_c = exact_score(buf)
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for a in range(n):
            if time.time() > deadline:
                break
            pa = buf[a]
            for b in range(a + 1, n):
                pb = buf[b]
                t_lo = max(-pa, pb - 1.0)
                t_hi = min(1.0 - pa, pb)
                if t_hi - t_lo < 1e-10:
                    continue
                nc, t = optimize_pair(buf, a, b, cur_c, n)
                if t != 0.0 and nc < cur_c - min_gain:
                    cur_c = nc
                    improved = True
                    pa = buf[a]  # update for next b iteration
        if not improved:
            break
    return buf, cur_c

def compute_gradient(h):
    """Compute gradient of max_k R(k) w.r.t. h (using subgradient from all near-max lags)."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(raw)
    # Active lags: those within 1e-6 of max
    active = raw >= max_val - 1e-6

import numpy as np
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw)) * dx


def project_feas(v, n):
    v = np.array(v, dtype=float).copy()
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(v)
    if res > 1e-10:
        idx = np.argsort(1.0 - v)
        for i in idx:
            add = min(res, 1.0 - v[i])
            v[i] += add
            res -= add
            if res < 1e-12:
                break
    elif res < -1e-10:
        idx = np.argsort(-v)
        for i in idx:
            rem = min(-res, v[i])
            v[i] -= rem
            res += rem
            if -res < 1e-12:
                break
    return v


def resample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feas(h_old, n_new)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    v = np.interp(x_new, x_old, h_old)
    return project_feas(v, n_new)


def score_transient(buf, a, b, t, n):
    dx = 2.0 / n
    if abs(t) < 1e-15:
        raw = np.correlate(buf, 1.0 - buf, mode="full")
        return float(np.max(raw)) * dx
    oa = buf[a]
    ob = buf[b]
    buf[a] = oa + t
    buf[b] = ob - t
    raw = np.correlate(buf, 1.0 - buf, mode="full")
    c = float(np.max(raw)) * dx
    buf[a] = oa
    buf[b] = ob
    return c


def optimize_pair(buf, a, b, cur_c, n):
    pa = buf[a]
    pb = buf[b]
    t_lo = max(-pa, pb - 1.0)
    t_hi = min(1.0 - pa, pb)
    if t_hi - t_lo < 1e-10:
        return cur_c, 0.0
    best_t = 0.0
    best_c = cur_c
    ns = 13
    for i in range(1, ns):
        t = t_lo + (t_hi - t_lo) * i / ns
        c = score_transient(buf, a, b, t, n)
        if c < best_c:
            best_c = c
            best_t = t
    for t in [t_lo + 1e-9, t_hi - 1e-9]:
        if t <= t_lo or t >= t_hi:
            continue
        c = score_transient(buf, a, b, t, n)
        if c < best_c:
            best_c = c
            best_t = t
    if best_t != 0.0 and best_c < cur_c:
        span = t_hi - t_lo
        w = span / ns
        lo = max(t_lo, best_t - w)
        hi = min(t_hi, best_t + w)
        if hi - lo > 1e-13:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            fc1 = score_transient(buf, a, b, c1, n)
            fc2 = score_transient(buf, a, b, c2, n)
            for _ in range(40):
                if hi - lo < 1e-14:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = score_transient(buf, a, b, c1, n)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = score_transient(buf, a, b, c2, n)
            tm = (lo + hi) / 2.0
            cm = score_transient(buf, a, b, tm, n)
            if cm < best_c:
                best_c = cm
                best_t = tm
    if best_t != 0.0 and best_c < cur_c - 1e-14:
        buf[a] = pa + best_t
        buf[b] = pb - best_t
        return best_c, best_t
    return cur_c, 0.0


def pair_sweep(h_in, deadline, min_gain=1e-11, max_passes=50):
    n = len(h_in)
    buf = h_in.copy()
    cur_c = exact_score(buf)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for a in range(n):
            if time.time() > deadline:
                break
            pa = buf[a]
            for b in range(a + 1, n):
                pb = buf[b]
                t_lo = max(-pa, pb - 1.0)
                t_hi = min(1.0 - pa, pb)
                if t_hi - t_lo < 1e-10:
                    continue
                nc, t = optimize_pair(buf, a, b, cur_c, n)
                if t != 0.0 and nc < cur_c - min_gain:
                    cur_c = nc
                    improved = True
                    pa = buf[a]
        if not improved:
            break
    return buf, cur_c


def make_alternate(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return project_feas(h, n)


def make_smooth(n, freq=3.0):
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / n)
    return project_feas(h, n)


def noise_perturb(h, seed, scale=0.03):
    rng = np.random.RandomState(seed)
    n = len(h)
    noise = rng.uniform(-scale, scale, n)
    return project_feas(h + noise, n)


def compute_gradient(h):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(raw)
    active = raw >= max_val - 1e-4
    g = np.zeros(n)
    for p in range(len(raw)):
        if not active[p]:
            continue
        k = p - (n - 1)
        for j in range(n):
            if j - k >= 0:
                g[j] += (1.0 - h[j - k])
            if j + k < n:
                g[j] += (-h[j + k])
    return g


def gradient_step(h, alpha_list=None):
    if alpha_list is None:
        alpha_list = [0.02, 0.01, 0.005, 0.002]
    n = len(h)
    cur_c = exact_score(h)
    g = compute_gradient(h)
    g_centered = g - np.mean(g)
    g_norm = np.linalg.norm(g_centered)
    if g_norm < 1e-12:
        return h, cur_c
    g_dir = g_centered / g_norm
    best_h = h.copy()
    best_c = cur_c
    for alpha in alpha_list:
        h_new = h - alpha * alpha_list[0] / alpha * g_dir * alpha
        h_new = project_feas(h + (-alpha * g_centered), n)
        c_new = exact_score(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c
