    y = np.asarray(y, dtype=np.float64).copy()
    # Replace non-finite values
    y[~np.isfinite(y)] = 0.5
    # Bisection for lambda
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(x)
    if abs(residual) < 1e-12:
        return x
    # Find coordinate with most slack
    if residual > 0:
        # Need to increase sum
        slacks = 1.0 - x
        idx = np.argmax(slacks)
        x[idx] += min(residual, slacks[idx])
    else:
        # Need to decrease sum
        slacks = x
        idx = np.argmax(slacks)
        x[idx] += max(residual, -slacks[idx])
    # Clamp
    x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    if n == 0:

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(y):
    n = len(y)
    target = n / 2.0
    y = np.array(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    # Bisection
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix mass residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            slacks = 1.0 - x
            idx = int(np.argmax(slacks))
            x[idx] = min(1.0, x[idx] + res)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + res)
    return x

    n = len(x)
