def make_feasible(z, n):
    """Project z to [0,1]^n with sum = n/2 using bisection."""
    n = int(n)
    target = n / 2.0
    z = np.array(z, dtype=float).ravel()
    if z.shape[0] != n:
        # resize
        if z.shape[0] > n:
            z = z[:n]
        else:
            z = np.pad(z, (0, n - z.shape[0]))
    # Bisection for shift
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)

        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0, 1)
    # Fix any residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            caps = 1.0 - h
            idx = np.argmax(caps)
            add = min(residual, caps[idx])
            h[idx] += add

def interpolate_to_n(h, n_new):
    """Interpolate h from its length to n_new using center sampling."""
    n_old = len(h)
    if n_old == n_new:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new)

def smooth_objective(h, beta):
    """Log-sum-exp surrogate for the max."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(vals)
    shifted = vals - max_val
    # Stable log-sum-exp
    max_exp = np.max(shifted * beta)
    log_sum = max_exp + np.log(np.sum(np.exp(shifted * beta - max_exp)))
    return max_val + log_sum / beta

    # Compute all correlation values

    n = len(h0)

    num_lags = 2 * n - 1
    target = n / 2.0

        return z[-1]  # theta

        g = np.zeros(n + 1)

        h = z[:n]
        theta = z[-1]
        # theta - dx * correlation_values >= 0 for all lags

        return theta - corr

        J[:, -1] = 1.0  # d(theta - corr)/d(theta) = 1

        return g

        {'type': 'eq', 'fun': equality, 'jac': equality_jac},

    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.05 * budget_s)
    deadline = t0 + budget_s - reserve

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
        if len(parent) < 4 or not np.all(np.isfinite(parent)):
            raise ValueError
    except (NameError, ValueError, AttributeError):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        best_h = make_feasible(parent, n_parent)
    else:
        n_parent = 45
        best_h = make_feasible(np.full(n_parent, 0.5), n_parent)
    best_n = len(best_h)
    best_score = official_score(best_h)

    rng = np.random.default_rng(seed)

    res_list = []
    for dn in [0, -1, -2, -3, -4, -5, -6, 1, 2, 3, 4, 5, 6, 8, 10]:
        nr = best_n + dn
        if 24 <= nr <= 80:
            res_list.append(nr)
    res_list = sorted(set(res_list))

    def do_smooth(cand, n_t, betas, maxiter):
        h_cur = cand.copy()
        for beta in betas:
            if time.time() > deadline:
                break
            try:
                r = minimize(smooth_obj, h_cur, method='SLSQP', jac=smooth_jac,
                             args=(beta,), bounds=[(0.0, 1.0)] * n_t,
                             options={'maxiter': maxiter, 'ftol': 1e-12})
                if hasattr(r, 'x') and np.all(np.isfinite(r.x)):
                    h_new = make_feasible(r.x, n_t)
                    sc = official_score(h_new)
                    h_cur = h_new
            except Exception:
                pass
        return h_cur

    def do_direct(h0, n_t, maxiter):
        theta0 = official_score(h0) + 1e-6
        x0 = np.append(h0, theta0)
        bds = [(0.0, 1.0)] * n_t + [(0.0, 1.0)]
        cons = [
            {'type': 'ineq', 'fun': dm_ineq, 'jac': dm_ineq_jac, 'args': (n_t,)},
            {'type': 'eq', 'fun': dm_eq, 'jac': dm_eq_jac, 'args': (n_t,)},
        ]
        try:
            r = minimize(dm_obj, x0, method='SLSQP', jac=dm_obj_jac, args=(n_t,),
                         bounds=bds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-12})
            if hasattr(r, 'x') and np.all(np.isfinite(r.x)):
                return make_feasible(r.x[:n_t], n_t)
        except Exception:
            pass
        return h0

    for n_t in res_list:
        if time.time() > deadline:
            break
        base = interp_to_n(best_h, n_t) if n_t != best_n else best_h.copy()
        candidates = [base]
        for sigma in [0.01, 0.03, 0.07, 0.14, 0.25]:
            candidates.append(make_feasible(base + rng.normal(0, sigma, size=n_t), n_t))
        for freq in [1, 2, 3, 5, 7]:
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.12 * np.sin(2 * np.pi * freq * np.arange(n_t) / n_t + phase)
            candidates.append(make_feasible(base + pert, n_t))

        scored = []
        for c in candidates:
            sc = official_score(c)
            scored.append((sc, c))
        scored.sort(key=lambda x: x[0])
        top_cands = [c for _, c in scored[:5]]

        for cand in top_cands:
            if time.time() > deadline:
                break
            h_smooth = do_smooth(cand, n_t, [15, 80, 400], 70)
            sc_s = official_score(h_smooth)
            if sc_s < best_score:
                best_score = sc_s
                best_h = h_smooth
                best_n = n_t
            h_dir = do_direct(h_smooth, n_t, 100)
            sc_d = official_score(h_dir)
            if sc_d < best_score:
                best_score = sc_d
                best_h = h_dir
                best_n = n_t

    # Final polish pass
    if time.time() < deadline - 15:
        h_f = do_direct(best_h, best_n, 200)
        sc_f = official_score(h_f)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

        # Try a second round of smooth at highest beta
        h_s2 = do_smooth(best_h, best_n, [800, 2000], 60)
        sc_s2 = official_score(h_s2)
        if sc_s2 < best_score:
            best_score = sc_s2
            best_h = h_s2
            h_f2 = do_direct(best_h, best_n, 150)
            sc_f2 = official_score(h_f2)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_h = h_f2

    best_h = make_feasible(best_h, best_n)
    final_score = official_score(best_h)
    c5_bound = final_score + 1e-13
    return (best_h, c5_bound, best_n)
