                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub

    s = beta * vals
    s_max = np.max(s)

    sum_exp = np.sum(exp_s)

            continue

                grad[d:n] += w * (1.0 - h[idx])
            # -h[i+d] for i in [0, n-1-d]
            if d < n:

            # (1-h[i+dd]) for i in [0, n-1-dd]

                grad[0:n-dd] += w * (1.0 - h[idx])
            # -h[i-dd] for i in [dd, n-1]
            if dd < n:
                idx2 = np.arange(dd, n) - dd
                grad[dd:n] += w * (-h[idx2])
    return val, grad
