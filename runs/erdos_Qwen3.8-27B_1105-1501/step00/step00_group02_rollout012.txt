        r = p + (n - 1) - i
        if i != p and i != q:
            G[r] += h[i]
        elif i == p and q >= p:
            G[r] += 0.0
    lo = max(0, p + 1)
    hi = min(n, p + 1 + (n - 1))
    for i in range(lo, hi):
        r = p + i - (n - 1)
        if i != p and i != q:
            G[r] -= (1.0 - h[i])
    lo = max(0, q - (n - 1))
    hi = min(q + 1, n)
    for i in range(lo, hi):
        r = q + (n - 1) - i
        if i != p and i != q:
            G[r] -= h[i]
    lo = max(0, q + 1)
    hi = min(n, q + 1 + (n - 1))
    for i in range(lo, hi):
        r = q + i - (n - 1)
        if i != p and i != q:
            G[r] += (1.0 - h[i])
    return dx * G

def _pair_opt(F0, G, Q, tmin, tmax):
    n_g = 41
    ts = np.linspace(tmin, tmax, n_g)
    best_t = 0.0
    best_v = float(np.max(F0))
    for ti in ts:
        v = float(np.max(F0 + G * ti + Q * ti * ti))
        if v < best_v:
            best_v = v
            best_t = ti
    if abs(best_t) > 1e-13 and best_v < best_v:
        pass
    if abs(best_t) > 1e-13:
        span = abs(tmax - tmin) / n_g * 2.0
        lo = max(tmin, best_t - span)
        hi = min(tmax, best_t + span)
        ts2 = np.linspace(lo, hi, 21)
        for ti in ts2:
            v = float(np.max(F0 + G * ti + Q * ti * ti))
            if v < best_v:
                best_v = v
                best_t = ti
    return best_t, best_v

def _local(h, C, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.astype(float).copy()
    best_C = C
    F0 = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cur = float(np.max(F0))
    pairs = [(i, j) for i in range(n) for j in range(n) if i != j]
    stagnation = 0
    while time.time() < deadline and stagnation < 50:
        rng.shuffle(pairs)
        improved = False
        for (p, q) in pairs:
            if time.time() > deadline:
                break
            tmin = max(-1.0, -h[q])
            tmax = min(1.0, 1.0 - h[q])
            if tmax - tmin < 1e-12:
                continue
            G = _get_G(h, p, q)
            Q = np.zeros(2 * n - 1)
            d = q - p
            if abs(d) < n:
                r1 = d + (n - 1)
                Q[r1] = 1.0
                r2 = -d + (n - 1)
                Q[r2] = -1.0
            bt, bv = _pair_opt(F0, G, Q, tmin, tmax)
            if abs(bt) > 1e-10 and bv < cur - 1e-12:
                h[p] -= bt
                h[q] += bt
                h = np.clip(h, 0.0, 1.0)
                F0 = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                cur = float(np.max(F0))
                if cur < best_C - 1e-10:
                    best_C = cur
                improved = True
        if improved:
            stagnation = 0
        else:
            stagnation += 1
    return _repair(h), best_C

def _anneal(h, C, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.astype(float).copy()
    best_C = C
    best_h = h.copy()
    cur_C = C
    T = 5e-4
    total_steps = 50000
    for step in range(total_steps):
        if time.time() > deadline or T < 1e-8:
            break
        T *= 0.999
        src = rng.randint(n)
        snk = rng.randint(n)
        if src == snk:
            continue
        tmin = max(-1.0, -h[snk])
        tmax = min(1.0, 1.0 - h[snk])
        if tmax - tmin < 1e-12:
            continue
        t = rng.uniform(tmin, tmax)
        abs_t = abs(t)
        if abs_t < 1e-12:
            continue
        h[src] -= t
        h[snk] += t
        if h[src] < -1e-12 or h[snk] > 1.0 + 1e-12:
            h = np.clip(h, 0.0, 1.0)
            h = _repair(h)
            cur_C = _eval(h)
            continue
        nc = _eval(h)
        d = nc - cur_C
        if d < 0 or (T > 0 and rng.random() < np.exp(-d / T)):
            cur_C = nc
            if nc < best_C - 1e-10:
                best_C = nc
                best_h = h.copy()
        else:
            h[src] += t
            h[snk] -= t
    return _repair(best_h), best_C

def _resample(h, nnew):
    nold = len(h)
    dxo = 2.0 / nold
    dxn = 2.0 / nnew
    old_c = (np.arange(nold) + 0.5) * dxo
    new_c = (np.arange(nnew) + 0.5) * dxn
    raw = np.interp(new_c, old_c, h, left=h[0], right=h[-1])
    return _repair(raw)

def _try_res(h, C, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h, best_C = h.copy(), C
    candidates = set()
    for f in [0.6, 0.75, 0.85, 1.0, 1.2, 1.45, 1.7]:
        nn = max(16, min(150, int(round(n * f))))
        if nn >= 2 and nn not in candidates and abs(nn - n) > 2:
            candidates.add(nn)
    for nn in sorted(candidates):
        if time.time() > deadline:
            break
        rh = _resample(best_h, nn)
        rc = _eval(rh)
        if np.isfinite(rc) and _feasible(rh):
            rh2, rc2 = _local(rh, rc, deadline, rng.randint(100000))
            if rc2 < best_C - 1e-8:
                best_h = rh2
                best_C = rc2
    return best_h, best_C, len(best_h)

def _finalize(h, current_C):
    h = _repair(np.asarray(h, dtype=float).flatten())
    if not _feasible(h):
        h = _make_bin(48, 0)
        h = _repair(h)
    C = _eval(h)
    if not np.isfinite(C):
        h = _make_bin(48, 0)
        h = _repair(h)
        C = _eval(h)
    return h, float(C), int(len(h))
