                best_h[idx] = max(0.0, best_h[idx] - diff2)
            else:
                idx = np.argmax(best_h)
                best_h[idx] = min(1.0, best_h[idx] - diff2)
    
    final_c = compute_c5(best_h)
    return (best_h, final_c, n_final)
