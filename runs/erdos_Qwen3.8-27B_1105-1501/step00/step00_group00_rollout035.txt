def project_feasible(h, n):

    h = h.copy()
    for _ in range(200):

        diff = target - np.sum(h)

            return h
        if diff > 0:

            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-14
            if not np.any(mask):

            h[mask] += diff / np.sum(mask)
    return h

def smooth_max_grad(h, dx, temp):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    v_max = np.max(vals)
    exponents = (vals - v_max) / temp
    exp_vals = np.exp(exponents - np.max(exponents))
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for p in range(len(vals)):
        if w[p] < 1e-13:
            continue
        wi = w[p] * dx
        k = p - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        idx = np.arange(i_start, i_end)
        grad[idx + k] += wi * (1.0 - h[idx])
        grad[idx] += -wi * h[idx + k]
    
    return grad, float(v_max)

def gradient_descent(h, n, dx, budget_frac, seed):
    """Run gradient descent with momentum and adaptive temperature."""
    np.random.seed(seed)
    start = time.time()
    budget = budget_frac * 1100  # approximate
    
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    t = 0
    temp = 0.1
    temp_min = 0.01
    temp_decay = 0.9999
    
    while time.time() - start < budget:
        t += 1
        grad, curr_max = smooth_max_grad(h, dx, temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = project_feasible(h - update, n)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            # Rejected
            lr = max(0.001, lr * 0.99)
        
        # Decrease temperature
        temp = max(temp_min, temp * temp_decay)
        
        # Random restart every 5000 steps
        if t % 5000 == 0:
            if np.random.random() < 0.3:
                perturbation = np.random.randn(n) * 0.05
                h_test = project_feasible(h + perturbation, n)
                c5_test = compute_c5(h_test, dx)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
                    h = h_test
        
        if t % 10000 == 0:
            lr = min(0.05, lr * 1.05)  # Slowly increase lr
    
    return best_h, best_c5

def simulated_annealing(h, n, dx, budget_s, seed):

    start = time.time()

    best_h = h.copy()

    T_min = 0.001

        curr_c5 = np.max(vals)

            best_c5 = curr_c5
            best_h = h.copy()
        
        # Choose move type
        r = np.random.random()
        if r < 0.4:
            # Smart move: target worst lag
            p_star = int(np.argmax(vals))
            k_star = p_star - (n - 1)
            if k_star == 0:
                j = np.random.randint(n)
                new_h = h.copy()
                new_h[j] += np.random.randn() * step_size
            else:
                i_start = max(0, -k_star)
                i_end = n - abs(k_star)
                i = np.random.randint(i_start, i_end)
                j = i + k_star
                new_h = h.copy()
                if np.random.random() < 0.5:
                    new_h[j] = h[j] + (h[i] - h[j]) * np.random.uniform(0.1, 0.5)
                else:
                    new_h[i] = h[i] + (h[j] - h[i]) * np.random.uniform(0.1, 0.5)
        elif r < 0.7:
            # Random coordinate move
            new_h = h.copy()
            j = np.random.randint(n)
            new_h[j] += np.random.randn() * step_size
        else:
            # Pair move
            new_h = h.copy()
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.randn() * step_size
            new_h[i] += delta
            new_h[j] -= delta  # Preserve sum approximately
        
        new_h = project_feasible(new_h, n)
        new_c5 = compute_c5(new_h, dx)
        
        # Metropolis criterion
        delta_c5 = new_c5 - curr_c5
        if delta_c5 < 0 or np.random.random() < math.exp(-delta_c5 / T):
            h = new_h
        
        # Cool down
        T = max(T_min, T * cooling)
        step_size = max(0.01, step_size * 1.0001)  # Slightly increase step size
    
    return best_h, best_c5
