        start = max(0, -k)
        end = min(n, n - k)
        out[idx] = np.sum(h[start:end] * (1.0 - h[start+k:end+k]))

    n = len(h)
    m = 2*n - 1
    G = np.zeros((m, n))
    for k in range(-(n-1), n):
        idx = k + n - 1
        start = max(0, -k)
        end = min(n, n - k)
        # First element contribution: d/dh[i] of h[i]*(1-h[i+k]) = 1-h[i+k]
        for i in range(start, end):
            G[idx, i] += (1.0 - h[i+k])
        # Second element contribution: d/dh[j] of h[i]*(1-h[j]) where j=i+k => -h[i]
        for i in range(start, end):

    # Bisection on lambda
    z = np.clip(z, -10, 10)  # avoid extreme values

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def manual_values(h):
    n = len(h)
    out = np.empty(2 * n - 1)
    for k in range(-(n - 1), n):
        s = max(0, -k)
        e = min(n, n - k)
        out[k + n - 1] = np.dot(h[s:e], 1.0 - h[s + k:e + k])
    return out

def manual_gradient(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        s = max(0, -k)
        e = min(n, n - k)
        G[k + n - 1, s:e] += (1.0 - h[s + k:e + k])
        G[k + n - 1, s + k:e + k] += (-h[s:e])
    return G

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -3.0 - np.max(z)
    hi = 3.0 - np.min(z)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-14:
        if resid > 0:
            idx_order = np.argsort(1.0 - h)
            for idx in idx_order:
                if h[idx] >= 1.0:
                    break
                add = min(resid, 1.0 - h[idx])
                h[idx] += add
                resid -= add
                if abs(resid) < 1e-14:
                    break
        else:
            idx_order = np.argsort(h)
            for idx in idx_order:
                if h[idx] <= 0.0:
                    break
                sub = min(-resid, h[idx])
                h[idx] -= sub
                resid += sub
                if abs(resid) < 1e-14:
                    break
    return h

def eq_obj(h, n):
    return h.sum() - n / 2.0

def eq_grad(h, n):
    return np.ones(n)

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = manual_values(h) * dx
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -500, 500)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = manual_values(h) * dx
    cmax = vals.max()
    shifted = np.clip(beta * (vals - cmax), -500, 500)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w /= s
    return dx * (w @ manual_gradient(h))

def direct_fun(z):
    return z[-1]

def direct_jac(z):
    n = len(z)
    j = np.zeros(n)
    j[-1] = 1.0
    return j

def ineq_fun(z, n):
    h = z[:n]
    theta = z[n]
    dx = 2.0 / n
    vals = manual_values(h) * dx
    return theta - vals

def ineq_jac(z, n):
    h = z[:n]
    dx = 2.0 / n
    G = manual_gradient(h) * dx
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def eq_fun_z(z, n):
    return z[:n].sum() - n / 2.0

def eq_jac_z(z, n):

def interpolate_vec(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    return np.interp(x_new, x_old, h)

def update_best(candidate, score, best_h, best_score, best_n):
    if not np.all(np.isfinite(candidate)):
        return best_h, best_score, best_n
    if score < best_score - 1e-13:
        return candidate.copy(), score, len(candidate)
    if abs(score - best_score) <= 1e-13 and len(candidate) < best_n:
        return candidate.copy(), score, len(candidate)
    return best_h, best_score, best_n

def run_smooth(h0, n, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_obj, 'jac': eq_grad, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h0, jac=smooth_grad, args=(beta,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x)
    except Exception:
        pass
    return h0

def run_direct(h0, n, theta0, maxiter, remaining):
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    ineq = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac, 'args': (n,)}
    eq = {'type': 'eq', 'fun': eq_fun_z, 'jac': eq_jac_z, 'args': (n,)}
    cons = [ineq, eq]
    try:
        res = minimize(direct_fun, z0, jac=direct_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x[:n])
            return h_new
    except Exception:
        pass
    return h0
