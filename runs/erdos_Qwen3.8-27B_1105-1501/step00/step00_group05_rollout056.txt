from scipy.optimize import linprog
import time

def exact_objective(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def correlation_vector(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_for_lag(h, idx):
    N = len(h)
    dx = 2.0 / N
    s = idx - (N - 1)
    grad = np.zeros(N)
    if s >= 0:
        for n in range(N - s):
            grad[n + s] += (1.0 - h[n])
            grad[n] += -h[n + s]
    else:
        t = -s
        for n in range(t, N):
            grad[n - t] += (1.0 - h[n])
            grad[n] += -h[n - t]
    return grad * dx

def find_active_lags(h, n_active=8):
    corr = correlation_vector(h)
    idx = np.argsort(corr)[::-1][:n_active]
    return idx, corr[idx].max()

def solve_lp_direction(h, active_grads):
    """
    Solve LP to find best direction d that decreases all active lags.
    Returns direction d or None if LP infeasible.
    """
    N = len(h)
    n_active = len(active_grads)
    n_vars = N + 1  # d_0..d_{N-1}, t
    
    # Objective: minimize -t
    c = np.zeros(n_vars)
    c[N] = -1.0
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    A_rows = []
    b_rows = []
    
    # For each active lag: grad_a · d + t <= 0
    for g in active_grads:
        row = np.zeros(n_vars)
        row[:N] = g
        row[N] = 1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # |d[i]| <= 1
    for i in range(N):
        row = np.zeros(n_vars)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1.0)
        
        row = np.zeros(n_vars)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(1.0)
    
    A_ineq = np.array(A_rows)
    b_ineq = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(None, None)] * N + [(None, None)]
    
    result = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:N]
    return None

def max_step(h, d):
    """Find maximum alpha such that h + alpha*d stays in [0,1]."""
    alpha = 1.0
    for i in range(len(h)):
        if d[i] < -1e-12:
            alpha = min(alpha, h[i] / (-d[i]))
        elif d[i] > 1e-12:
            alpha = min(alpha, (1.0 - h[i]) / d[i])
    return max(alpha, 0.0)

def project_feasible(h, N):
    """Project h to satisfy sum(h) = N/2 and 0 <= h <= 1."""
    h = h.copy()
    h = np.clip(h, 0, 1)
    # Adjust sum
    target = N / 2.0
    err = target - np.sum(h)
    if abs(err) > 1e-15:
        # Distribute error
        if err > 0:

            room = 1.0 - h
            total_room = room.sum()
            if total_room > 0:

            else:

        else:
            # Need to subtract

            total_room = room.sum()
            if total_room > 0:
                h += err * room / total_room
            else:
                h += err / N
    h = np.clip(h, 0, 1)

    err = target - np.sum(h)

    # Simple linear interpolation
    new_vals = np.interp(new_x, old_x, parent)
    return new_vals

def exact_obj(h):
    N = len(h)
    return float(np.max(np.correlate(h, 1.0-h, 'full')) * 2.0/N)

def grad_lag(h, idx):
    N = len(h)
    dx = 2.0/N
    s = idx - (N-1)
    g = np.zeros(N)
    if s >= 0:
        for i in range(s, N):
            g[i] += (1.0 - h[i-s])
        for i in range(N-s):
            g[i] -= h[i+s]
    else:
        t = -s
        for i in range(N-t):
            g[i] += (1.0 - h[i+t])
        for i in range(t, N):
            g[i] -= h[i-t]
    return g * dx

def project_sum(h, N):
    """Adjust h to have sum = N/2, staying in [0,1]."""
    h = h.copy()
    target = N/2.0
    err = target - h.sum()
    if abs(err) < 1e-14:
        return h
    if err > 0:
        room = 1.0 - h
        idx = np.where(room > 1e-10)[0]
        if len(idx) == 0:
            return np.clip(h, 0, 1)
        scale = err / room[idx].sum()
        if scale < 1:
            h[idx] += room[idx] * scale
        else:
            h += err / N
    else:
        room = h.copy()
        idx = np.where(room > 1e-10)[0]
        if len(idx) == 0:
            return np.clip(h, 0, 1)
        scale = -err / room[idx].sum()
        if scale < 1:
            h[idx] -= room[idx] * scale
        else:
            h += err / N
    return np.clip(h, 0, 1)

import numpy as np

import time

def exact_obj(h):
    N = len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * 2.0 / N)

def grad_lag(h, idx):
    N = len(h)
    dx = 2.0 / N
    s = idx - (N - 1)
    g = np.zeros(N)
    if s >= 0:
        for i in range(s, N):
            g[i] += 1.0 - h[i - s]
        for i in range(N - s):
            g[i] -= h[i + s]
    else:
        t = -s
        for i in range(N - t):
            g[i] += 1.0 - h[i + t]
        for i in range(t, N):
            g[i] -= h[i - t]
    return g * dx

def project_sum(h, N):
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    err = target - h.sum()
    if abs(err) < 1e-14:
        return h
    if err > 0:
        room = 1.0 - h
        mask = room > 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += err * room[mask] / room[mask].sum()
    else:
        room = h.copy()
        mask = room > 1e-12
        if mask.sum() == 0:
            return h
        h[mask] += err * room[mask] / room[mask].sum()
    return np.clip(h, 0.0, 1.0)

    if pos.any():
