def linear_coefficients(h):
    n=len(h); L=2*n-1
    shifts=(n-1)-np.arange(L)
    coef=np.zeros((n,L),float)
    # For each i, first: valid shifts -i <= d < n-i. idx = i+d.
    for i in range(n):
       valid = (shifts >= -i) & (shifts < n-i)
       idx = shifts[valid] + i
       coef[i, valid] += (1.0 - h[idx])
       valid2 = (shifts <= i) & (shifts >= i - (n-1)) # 0 <= i-d < n
       idx2 = i - shifts[valid2]
       coef[i, valid2] += -h[idx2]

import numpy as np
import time

_SHIFT_CACHE = {}


def get_shifts(n):
    if n not in _SHIFT_CACHE:
        _SHIFT_CACHE[n] = (n - 1) - np.arange(2 * n - 1, dtype=np.int64)
    return _SHIFT_CACHE[n]


def c5(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n <= 0 or not np.all(np.isfinite(h)):
        return float("inf")
    if h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return float("inf")
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_box_sum(z, target):
    z = np.asarray(z, dtype=float).ravel().copy()
    if z.size == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    n = z.size
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    y = np.clip(z - mid, 0.0, 1.0)
    res = target - float(np.sum(y))
    for _ in range(5):
        if abs(res) <= 1e-14:
            break
        if res > 0.0:
            for i in range(n):
                if res <= 1e-14:
                    break
                add = min(res, max(0.0, 1.0 - y[i]))
                if add > 0.0:
                    y[i] += add
                    res -= add
        else:
            for i in range(n):
                if -res <= 1e-14:
                    break
                sub = min(-res, max(0.0, y[i]))
                if sub > 0.0:
                    y[i] -= sub
                    res += sub
    y = np.clip(y, 0.0, 1.0)
    return y


def ensure_feasible(h):
    h = np.asarray(h, dtype=float).ravel().copy()
    if h.size == 0:
        return None
    if not np.all(np.isfinite(h)):
        return None
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * h.size
    res = target - float(np.sum(h))
    if abs(res) > 1e-9:
        return project_box_sum(h, target)
    for _ in range(5):
        if abs(res) <= 1e-14:
            break
        if res > 0.0:
            for i in range(h.size):
                if res <= 1e-14:
                    break
                add = min(res, max(0.0, 1.0 - h[i]))
                if add > 0.0:
                    h[i] += add
                    res -= add
        else:
            for i in range(h.size):
                if -res <= 1e-14:
                    break
                sub = min(-res, max(0.0, h[i]))
                if sub > 0.0:
                    h[i] -= sub
                    res += sub
    h = np.clip(h, 0.0, 1.0)
    return h


def random_balanced(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    if n < 1:
        return h
    if n % 2 == 0:
        k = n // 2
        if k > 0:
            idx = rng.choice(n, k, replace=False)
            h[idx] = 1.0
    else:
        k = (n - 1) // 2
        perm = rng.permutation(n)
        if k > 0:
            h[perm[:k]] = 1.0
        if n > k:
            h[perm[k]] = 0.5
    return h


def continuous_random(n, rng):
    n = int(n)
    z = rng.normal(0.5, 0.25, size=n)
    return project_box_sum(z, 0.5 * n)


def threshold_candidate(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    y = np.zeros(n, dtype=float)
    if n < 1:
        return y
    if n % 2 == 0:
        k = n // 2
        if k > 0:
            order = np.argsort(h)[::-1]
            y[order[:k]] = 1.0
    else:
        k = (n - 1) // 2
        order = np.argsort(h)[::-1]
        if k > 0:
            y[order[:k]] = 1.0
        if n > k:
            y[order[k]] = 0.5
    return y


def resample_to_n(h, n_new):
    n_new = int(n_new)
    old = np.asarray(h, dtype=float).ravel()
    m = old.size
    if n_new < 2:
        return None
    if m == n_new:
        return old.copy()
    if m < 2:
        return project_box_sum(np.full(n_new, old[0]), 0.5 * n_new)
    xo = (np.arange(m, dtype=float) + 0.5) * (2.0 / m)
    xn = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    y = np.interp(xn, xo, old, left=old[0], right=old[-1])
    return project_box_sum(y, 0.5 * n_new)


def kick_from(h, rng, moves):
    h = np.asarray(h, dtype=float).ravel().copy()
    n = h.size
    moves = int(moves)
    for _ in range(moves):
        p = int(rng.integers(n))
        q = int(rng.integers(n))
        if p == q:
            continue
        tmin = max(h[p] - 1.0, -h[q])
        tmax = min(h[p], 1.0 - h[q])
        if tmax - tmin <= 1e-14:
            continue
        t = tmin + float(rng.random()) * (tmax - tmin)
        h[p] -= t
        h[q] += t
    return ensure_feasible(h)


def noise_from(h, rng, sigma):
    h = np.asarray(h, dtype=float).ravel()
    z = h + rng.normal(0.0, float(sigma), size=h.size)
    return project_box_sum(z, 0.5 * h.size)


def make_candidate(typ, base_h, n, rng, param):
    try:
        if typ == "best":
            return base_h.copy()
        n = int(n)
        if n < 2:
            return None
        if typ == "noise":

        if typ == "kick":

                return None
            return kick_from(base_h, rng, param)
        if typ == "random":
            return random_balanced(n, rng)
        if typ == "cont":
            return continuous_random(n, rng)
        if typ == "threshold":
            if base_h is None or base_h.size != n:

        if typ == "resample":
            if base_h is None:
                return None
            return resample_to_n(base_h, n)
    except Exception:
        return None
    return None


def compute_linear_coefficients(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    shifts = get_shifts(n)
    L = 2 * n - 1
    coef = np.zeros((n, L), dtype=float)
    for i in range(n):
        m1 = (shifts >= -i) & (shifts < n - i)
        if np.any(m1):
            idx1 = shifts[m1] + i
            coef[i, m1] = 1.0 - h[idx1]
        m2 = (shifts <= i) & (shifts >= i - n + 1)
        if np.any(m2):
            idx2 = i - shifts[m2]
            coef[i, m2] -= h[idx2]
    return coef


def line_values(corr, D, ts, r0, r1, r2):
    vals = corr[None, :] + D[None, :] * ts[:, None]
    t2 = ts * ts
    vals[:, r0] -= 2.0 * t2
    vals[:, r1] += t2
    if r2 != r1:
        vals[:, r2] += t2
    return np.max(vals, axis=1)


def line_search_pair(h, coef, corr, p, q, raw, grid, refine):
    n = h.size
    xp = float(h[p])
    xq = float(h[q])
    tmin = max(xp - 1.0, -xq)
    tmax = min(xp, 1.0 - xq)
    if tmax - tmin <= 1e-14:
        return None

    D = coef[q] - coef[p]
    r0 = n - 1
    r1 = n - 1 - (q - p)
    r2 = n - 1 - (p - q)

    ts = np.linspace(tmin, tmax, int(grid), dtype=float)
    row = line_values(corr, D, ts, r0, r1, r2)
    if not np.any(np.isfinite(row)):
        return None
    bad = ~np.isfinite(row)
    if np.any(bad):
        row = row.copy()
        if not np.any(np.isfinite(row)):
            return None
        row[bad] = float(np.min(row[~bad])) + 1.0
    idx = int(np.argmin(row))
    if row[idx] >= raw - 1e-11:
        return None

    half = (tmax - tmin) / max(1, int(grid))
    best_t = float(ts[idx])
    best_val = float(row[idx])
    if half > 1e-13:
        lo = max(tmin, best_t - half)
        hi = min(tmax, best_t + half)
        if hi - lo > 1e-13:
            tsr = np.linspace(lo, hi, int(refine), dtype=float)
            rowr = line_values(corr, D, tsr, r0, r1, r2)
            badr = ~np.isfinite(rowr)
            if np.any(badr):
                rowr = rowr.copy()
                rowr[badr] = float(np.min(rowr[~badr]) + 1.0) if np.any(np.isfinite(rowr)) else 1.0
            idxr = int(np.argmin(rowr))
            if rowr[idxr] < best_val - 1e-13:
                best_t = float(tsr[idxr])
                best_val = float(rowr[idxr])

    if raw - best_val > 1e-7:
        span = max(1e-6, 0.25 * half)
        lo = max(tmin, best_t - span)
        hi = min(tmax, best_t + span)
        if hi - lo > 1e-13:
            ts2 = np.linspace(lo, hi, 81, dtype=float)
            row2 = line_values(corr, D, ts2, r0, r1, r2)
            bad2 = ~np.isfinite(row2)
            if np.any(bad2):
                row2 = row2.copy()
                row2[bad2] = float(np.min(row2[~bad2]) + 1.0) if np.any(np.isfinite(row2)) else 1.0
            idx2 = int(np.argmin(row2))
            if row2[idx2] < best_val - 1e-13:
                best_t = float(ts2[idx2])
                best_val = float(row2[idx2])

    cand = [best_t, float(ts[idx]), float(tmin), float(tmax)]
    top = np.argsort(row)[:min(4, row.size)]
    for k in top:
        cand.append(float(ts[int(k)]))

    unique_ts = []
    for tt in cand:
        if not np.isfinite(tt):
            continue
        if tt < tmin - 1e-11 or tt > tmax + 1e-11:
            continue
        dup = False
        for uu in unique_ts:
            if abs(tt - uu) < 1e-10:
                dup = True
                break
        if not dup:
            unique_ts.append(tt)

    best_y = None
    best_dr = raw
    old_sum = float(h.sum())
    for tt in unique_ts:
        y = h.copy()
        yv_p = y[p] - tt
        yv_q = y[q] + tt
        if yv_p < 0.0:
            if yv_p < -1e-9:
                continue
            yv_p = 0.0
        elif yv_p > 1.0:
            if yv_p > 1.0 + 1e-9:
                continue
            yv_p = 1.0
        if yv_q < 0.0:
            if yv_q < -1e-9:
                continue
            yv_q = 0.0
        elif yv_q > 1.0:
            if yv_q > 1.0 + 1e-9:
                continue
            yv_q = 1.0
        y[p] = yv_p
        y[q] = yv_q
        if abs(float(np.sum(y)) - old_sum) > 1e-9:
            y = ensure_feasible(y)
            if y is None:
                continue
        dr = float(np.max(np.correlate(y, 1.0 - y, mode="full")))
        if not np.isfinite(dr):
            continue
        if dr < best_dr - 1e-13:
            best_dr = dr
            best_y = y

    if best_y is not None and best_dr < raw - 1e-10:
        return best_y, float(best_dr)
    return None


def choose_grid(n):
    n = int(n)
    if n <= 42:
        return 449
    if n <= 56:
        return 321
    if n <= 72:
        return 225
    if n <= 96:
        return 161
    return 121


def choose_refine(n):
    n = int(n)
    if n <= 42:
        return 193
    if n <= 56:
        return 161
    if n <= 72:
        return 113
    if n <= 96:
        return 81
    return 65


def polish_pairs(h, deadline, rng=None, grid=None, refine=None):
    h = ensure_feasible(h)
    if h is None:
        h = np.full(32, 0.5, dtype=float)
    n = h.size
    if n < 2:
        h = np.full(32, 0.5, dtype=float)
        n = h.size

    if grid is None:
        grid = choose_grid(n)
    if refine is None:
        refine = choose_refine(n)
    rem = max(0.0, deadline - time.time())
    if rem < 8.0:
        grid = min(int(grid), 97)
        refine = min(int(refine), 49)
    elif rem < 25.0:
        grid = min(int(grid), 193)
        refine = min(int(refine), 97)

    corr = np.correlate(h, 1.0 - h, mode="full")
    raw = float(np.max(corr))
    c = raw * (2.0 / n)
    coef = compute_linear_coefficients(h)

    passes = 0
    while time.time() < deadline and passes < 100000:
        passes += 1
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            if time.time() >= deadline:
                break
            for q in order:
                if time.time() >= deadline:
                    break
                if p >= q:
                    continue
                xp = float(h[p])
                xq = float(h[q])
                tmin = max(xp - 1.0, -xq)
                tmax = min(xp, 1.0 - xq)
                if tmax - tmin <= 1e-12:
                    continue
                res = line_search_pair(h, coef, corr, int(p), int(q), raw, grid, refine)
                if res is not None:
                    h, raw = res
                    c = raw * (2.0 / n)
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    raw = float(np.max(corr))
                    coef = compute_linear_coefficients(h)
                    improved = True
            if time.time() >= deadline:
                break
        if not improved:
            break

    h = ensure_feasible(h)
    if h is not None:
        c = c5(h)
    return h, float(c)

    h = random_balanced(47, rng)

def is_better(c, n, best_c, best_n):
    if np.isfinite(c) and not np.isfinite(best_c):
        return True
    if c < best_c - 1e-12:
        return True
    if abs(c - best_c) <= 1e-12 and int(n) < int(best_n):
        return True
    return False


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    if budget < 1.0:
        budget = 1.0
    deadline = t0 + 0.96 * budget
    rng = np.random.default_rng(seed)

    try:
        best_h, best_c, best_n = initialize_best(rng)
    except Exception:
        best_h = np.full(47, 0.5, dtype=float)
        best_c = c5(best_h)
        best_n = int(best_h.size)

    if best_h is None:
        best_h = np.full(47, 0.5, dtype=float)
        best_c = c5(best_h)
        best_n = int(best_h.size)

    if budget < 30.0:
        h2, c2 = polish_pairs(best_h, deadline, rng, grid=129, refine=65)
        if is_better(c2, h2.size, best_c, best_n):
            best_h, best_c, best_n = h2, c2, h2.size
        best_h = ensure_feasible(best_h)
        return best_h, float(c5(best_h)), int(best_h.size)

    end_primary = min(deadline, t0 + max(15.0, 0.35 * budget))
    h2, c2 = polish_pairs(best_h, end_primary, rng)
    if is_better(c2, h2.size, best_c, best_n):
        best_h, best_c, best_n = h2, float(c2), int(h2.size)

    base_n = int(best_n)
    trials = [
        ("best", base_n, 0),
        ("noise", base_n, 0.05),
        ("noise", base_n, 0.15),
        ("noise", base_n, 0.35),
        ("threshold", base_n, 0),
        ("kick", base_n, 8),
        ("kick", base_n, 25),
        ("resample", max(28, base_n - 4), 0),
        ("resample", min(72, max(28, base_n + 4)), 0),
        ("resample", 64, 0),
        ("random", max(32, base_n - 8), 0),
        ("random", 64, 0),
        ("cont", max(32, base_n - 16), 0),
    ]

    seen = set()
    clean_trials = []
    for typ, nn, param in trials:
        nn = int(nn)
        if nn < 24 or nn > 72:
            continue
        key = (typ, nn)
        if key in seen:
            continue
        seen.add(key)
        clean_trials.append((typ, nn, param))

    remaining = deadline - time.time()
    if remaining > 10.0 and len(clean_trials) > 0:
        slice_t = max(6.0, min(120.0, remaining / float(len(clean_trials))))
        for typ, nn, param in clean_trials:
            if time.time() >= deadline - 4.0:
                break
            if time.time() + 3.0 >= deadline:
                break
            cand = make_candidate(typ, best_h, nn, rng, param)
            cand = ensure_feasible(cand)
            if cand is None:
                continue
            end = min(deadline - 3.0, time.time() + slice_t)
            if end - time.time() < 2.0:
                break
            h2, c2 = polish_pairs(cand, end, rng)
            if is_better(c2, h2.size, best_c, best_n):
                best_h, best_c, best_n = h2, float(c2), int(h2.size)

    if time.time() < deadline - 8.0:
        h2, c2 = polish_pairs(best_h, deadline - 1.0, rng)
        if is_better(c2, h2.size, best_c, best_n):
            best_h, best_c, best_n = h2, float(c2), int(h2.size)

    best_h = ensure_feasible(best_h)
    if best_h is None or not np.all(np.isfinite(best_h)):
        best_h = np.full(47, 0.5, dtype=float)
    best_c = c5(best_h)
    if not np.isfinite(best_c):
        best_h = np.full(47, 0.5, dtype=float)
        best_c = c5(best_h)
    return best_h, float(best_c), int(best_h.size)
