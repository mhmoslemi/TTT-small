    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    """Find [lo, hi] such that h + λd is in [0,1] for all i when λ in [lo,hi]."""
    n = len(h)
    lo = 0.0
    hi = 0.0
    for i in range(n):
        if d[i] > 0:
            lo = max(lo, -h[i] / d[i])
            hi = min(hi, (1.0 - h[i]) / d[i])
        elif d[i] < 0:
            lo = max(lo, (1.0 - h[i]) / d[i])
            hi = min(hi, -h[i] / d[i])
    # Add small margin
    margin = 1e-12
    lo += margin
    hi -= margin
    if lo >= hi:
        lo, hi = 0.0, 0.0
    return lo, hi

    best_h = h.copy()

    """Project h to [0,1]^n with sum = target_sum using bisection on offset."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)

    lo = -max(h.max(), 0) - 1

def project_feasible(h, target_sum):

    hi = 1.0

        if s > target_sum:
            hi = mu
        else:
            lo = mu
    mu = (lo + hi) / 2.0
    result = np.clip(h + mu, 0.0, 1.0)

        mask = (result > 1e-6) & (result < 1.0 - 1e-6)
        cnt = mask.sum()
        if cnt > 0:

    """Compute subgradient of max_k c_k."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Find active shifts (top m)
    top_idx = np.argsort(corr)[-top_m:]
    
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = k_idx - (n - 1)  # actual shift
        for m in range(n):
            g = 0.0
            mk = m - k
            if 0 <= mk < n:

            mk2 = m + k
            if 0 <= mk2 < n:
                g -= h[mk2]
            grad[m] += g
    
    # Project to sum = 0
    grad -= grad.mean()
    return grad

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_feasible(h, target_sum):
    n = len(h)
    if n == 0:
        return h
    h = np.clip(h, 0.0, 1.0).copy()
    if abs(h.sum() - target_sum) < 1e-12:
        return h
    lo = -max(float(h.max()), 0.0) - 0.1
    hi = 1.0 - min(float(h.min()), 1.0) + 0.1
    for _ in range(80):
        mu = (lo + hi) / 2.0
        s = np.clip(h + mu, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mu
        else:
            lo = mu
    mu = (lo + hi) / 2.0
    result = np.clip(h + mu, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-8) & (result < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            result[mask] += diff / cnt
            np.clip(result, 0.0, 1.0, out=result)
    return result

def line_search_general(h, d, n_samples=40):
    n = len(h)
    dx = 2.0 / n
    lo = 0.0
    hi = 0.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        lo = max(lo, float(np.max(-h[pos] / d[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max((1.0 - h[neg]) / d[neg])))
        hi = min(hi, float(np.min(-h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return h.copy(), evaluate_c5(h)
    lams = np.linspace(lo, hi, n_samples)
    best_v = evaluate_c5(h)
    best_lam = 0.0
    for lam in lams:
        h_new = h + lam * d
        v = float(np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx)
        if v < best_v:
            best_v = v
            best_lam = lam
    if best_lam != 0.0:
        return h + best_lam * d, best_v
    return h.copy(), best_v

def compute_subgradient(h, top_m=8):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = int(k_idx - (n - 1))
        for m in range(n):
            g = 0.0
            mk = m - k
            if 0 <= mk < n:
                g += 1.0 - h[mk]
            mk2 = m + k
            if 0 <= mk2 < n:
                g -= h[mk2]
            grad[m] += g
    grad -= grad.mean()
    return grad

def pair_cd(h, n_samples=40, max_passes=100, deadline=None):
    n = len(h)
    dx = 2.0 / n
    val = evaluate_c5(h)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                orig_i, orig_j = h[i], h[j]
                lo = max(-orig_i, orig_j - 1.0)
                hi = min(1.0 - orig_i, orig_j)
                if hi - lo > 1e-14:
                    lams = np.linspace(lo, hi, n_samples)
                    bv = val
                    bl = 0.0
                    for lam in lams:
                        h[i] = orig_i + lam
                        h[j] = orig_j - lam
                        v = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
                        if v < bv:
                            bv = v
                            bl = lam
                    if bl != 0.0:
                        h[i] = orig_i + bl
                        h[j] = orig_j - bl
                        val = bv
                        improved = True
                    else:
                        h[i] = orig_i
                        h[j] = orig_j
                lo = max(-orig_j, orig_i - 1.0)
                hi = min(1.0 - orig_j, orig_i)
                if hi - lo > 1e-14:
                    orig_i, orig_j = h[i], h[j]
                    lams = np.linspace(lo, hi, n_samples)
                    bv = val
                    bl = 0.0
                    for lam in lams:
                        h[i] = orig_i - lam
                        h[j] = orig_j + lam
                        v = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
                        if v < bv:
                            bv = v
                            bl = lam
                    if bl != 0.0:
                        h[i] = orig_i - bl
                        h[j] = orig_j + bl
                        val = bv
                        improved = True
                    else:
                        h[i] = orig_i
                        h[j] = orig_j
        if not improved:
            break
    return h, val
