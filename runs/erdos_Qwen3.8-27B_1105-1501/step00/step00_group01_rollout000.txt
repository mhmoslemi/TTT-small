        if elapsed > budget_s * 0.92:
            break
        lr = lr0 / (1.0 + step * decay)
        grad = _compute_gradient(h, beta)
        h = h - lr * grad
        h = _project_onto_constraint(h)
        if step % 10 == 0:
            c5 = _eval_c5(h)
            if c5 < best_local:
                best_local = c5
                best_local_h = h.copy()
    c5 = _eval_c5(h)
    if c5 < best_local:
        best_local = c5
        best_local_h = h.copy()
    return best_local_h, best_local

def _slsqp_refine(h_init, start_time, budget_s):
    n = len(h_init)
    beta = 40.0
    def obj(v):
        h = np.clip(v, 0.0, 1.0)
        corr = _corr_vector(h)
        m = corr.max()
        s = beta * (corr - m)
        return m + np.log(np.sum(np.exp(s))) / beta
    def jac(v):
        h = np.clip(v, 0.0, 1.0)
        return _compute_gradient(h, beta)
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-10})
        h_new = _project_onto_constraint(res.x)
        c5_new = _eval_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h_init, _eval_c5(h_init)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is not None and len(h0) > 0:
        h0 = _project_onto_constraint(h0)
        c5_0 = _eval_c5(h0)
        best_c5 = c5_0
        best_h = h0.copy()
    else:
        n0 = 63
        best_h = np.full(n0, 0.5)
        best_c5 = _eval_c5(best_h)

    n = len(best_h)

    # Phase 1: PGD with various beta values from initial solution
    beta_list = [8, 15, 25, 40, 60, 100]
    for beta in beta_list:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.55:
            break
        lr0 = 0.05 / np.sqrt(n)
        decay = 0.0005
        steps = min(8000, int((budget_s * 0.08) / max(0.001, (time.time() - start_time) / max(1, len(beta_list)))))
        h_sol, c5_sol = _pgd_solve(best_h, steps, beta, lr0, decay, start_time, budget_s)
        if c5_sol < best_c5:
            best_c5 = c5_sol
            best_h = h_sol.copy()
            n = len(best_h)

    # Phase 2: SLSQP refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.7 and best_h is not None:
        h_new, c5_new = _slsqp_refine(best_h, start_time, budget_s)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            n = len(best_h)

    # Phase 3: Try different n_points
    n_candidates = []
    for nc in [36, 42, 48, 54, 60, 72, 84, 96, 108, 126, 140, 168, 196, 224, 252]:
        if nc != n:
            n_candidates.append(nc)

    for nc in n_candidates:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.82:
            break

        h_init = np.interp(
            np.linspace(0, 2, nc, endpoint=False),
            np.linspace(0, 2, n, endpoint=False),
            best_h
        )
        h_init = _project_onto_constraint(h_init)

        for beta in [20, 50]:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.82:
                break
            lr0 = 0.04 / np.sqrt(nc)
            decay = 0.0008
            steps = 4000
            h_sol, c5_sol = _pgd_solve(h_init, steps, beta, lr0, decay, start_time, budget_s)
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                n = len(best_h)

    # Phase 4: Final SLSQP on best
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.9 and best_h is not None:
        h_new, c5_new = _slsqp_refine(best_h, start_time, budget_s)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            n = len(best_h)

    # Phase 5: Random restarts if time remains
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.88:
        n_r = 63
        for trial in range(5):
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.88:
                break
            h_rand = np.random.dirichlet(np.ones(n_r)) * (n_r / n_r)
            h_rand = h_rand * (n_r / 2.0) / h_rand.sum()
            h_rand = np.clip(h_rand, 0.0, 1.0)
            h_rand = _project_onto_constraint(h_rand)
            for beta in [30, 60]:
                elapsed = time.time() - start_time
                if elapsed > budget_s * 0.88:
                    break
                lr0 = 0.03 / np.sqrt(n_r)
                steps = 3000
                h_sol, c5_sol = _pgd_solve(h_rand, steps, beta, lr0, 0.001, start_time, budget_s)
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
                    n = len(best_h)

    # One final check
    if best_h is not None:
        best_c5 = _eval_c5(best_h)
        n = len(best_h)

    return best_h, best_c5, n
