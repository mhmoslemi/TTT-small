def lag_gradient(d, h):
    """Compute gradient of C_d with respect to h.
    d is the lag (can be negative). Returns gradient array of length n."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        K = n - d
        if K <= 0:
            return grad
        # C_d = sum_{i=0}^{K-1} h[i]*(1-h[i+d])
        # dC_d/dh[j] = (1 - h[j+d]) for j in [0, K-1] (first factor)
        #               - h[j-d] for j in [d, n-1] (second factor, since h[j] = h[i+d] with i=j-d)
        grad[:K] += (1 - h[d:])  # first factor contribution
        grad[d:] += -h[:K]       # second factor contribution
    else:

        K = n - e
        if K <= 0:
            return grad
        # C_d = sum_{i=e}^{n-1} h[i]*(1-h[i-e])
        # dC_d/dh[j] = (1 - h[j-e]) for j in [e, n-1] (first factor)

        grad[e:] += (1 - h[:K])  # first factor: h[j]*(1-h[j-e]), j from e to n-1

        return h

import numpy as np
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        interior = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            per = diff / len(interior)
            h[interior] = np.clip(h[interior] + per, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(0.0, h[idx] + diff2)
    return h

def lag_gradient(d, h):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        K = n - d
        if K <= 0:
            return grad
        grad[:K] += dx * (1.0 - h[d:d+K])
        grad[d:d+K] += -dx * h[:K]
    else:
        e = -d
        K = n - e
        if K <= 0:
            return grad
        grad[e:e+K] += dx * (1.0 - h[:K])
        grad[:K] += -dx * h[e:e+K]
    return grad

def pair_optimize(h, i, j, n_grid=300):
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-12:
        return h, np.inf
    best_val = np.inf
    best_t = h[i]
    for t in np.linspace(lo, hi, n_grid):
        h[i] = t
        h[j] = c - t
        val = evaluate(h)
        if val < best_val:
            best_val = val
            best_t = t
    h[i] = best_t
    h[j] = c - best_t
    return h, best_val

def logsumexp_grad(h, T):
    n = len(h)
    lag_vals = all_lags(h)
    mv = np.max(lag_vals)
    w = np.exp(T * (lag_vals - mv))
    w /= np.sum(w)
    d_list = np.arange(n - 1, -(n - 1), -1)
    g = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-14:
            continue
        d = d_list[p]
        g += w[p] * lag_gradient(d, h)
    g -= np.mean(g)
    return g, w

def lse_descent(h, best_score, n, deadline, T_init=30.0, max_iters=600):
    h = h.copy()
    T = T_init
    lr = 0.003
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        g, w = logsumexp_grad(h, T)
        alpha = lr
        found = False
        for _ in range(50):
            hc = project_feasible(h - alpha * g, n)
            sc = evaluate(hc)
            if sc < best_score:
                h = hc
                best_score = sc
                found = True
                break
            alpha *= 0.5
        if not found:
            T *= 1.3
            lr *= 0.85
            if T > 5000:
                break
    return h, best_score

def make_pairs(n):
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    return pairs

def pair_cd_pass(h, n, n_grid, deadline, max_pairs=None):
    pairs = make_pairs(n)
    np.random.shuffle(pairs)
    if max_pairs and max_pairs < len(pairs):
        pairs = pairs[:max_pairs]
    best_local = evaluate(h)
    for i, j in pairs:
        if time.time() >= deadline:
            break
        h, val = pair_optimize(h, i, j, n_grid)
        if val < best_local:
            best_local = val
    return h, best_local

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
