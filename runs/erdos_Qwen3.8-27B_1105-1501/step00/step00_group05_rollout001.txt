import numpy as np
import time
from scipy.optimize import linprog

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return np.max(compute_overlaps(h))

def gradient_overlap(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.ones(n, dtype=float)
    if lag >= 0:
        g[:n - lag] -= h[lag:]
        if lag > 0:
            g[lag:] -= h[:n - lag]
    else:
        L = -lag
        g[L:] -= h[:n - L]
        g[:n - L] -= h[L:]
    return dx * g

def project_h(h, target_sum):
    h = h.astype(float)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_descent(h, n_steps, lr_init, t0, deadline):
    n = len(h)
    target_sum = n / 2.0
    lr = lr_init
    best_c5 = compute_c5(h)
    best = h.copy()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        h = project_h(h, target_sum)
        overlaps = compute_overlaps(h)
        max_val = np.max(overlaps)
        tol = max(1e-11, max_val * 5e-8)
        active = overlaps >= max_val - tol
        n_act = max(1, int(np.sum(active)))
        grad = np.zeros(n)
        for idx in range(len(overlaps)):
            if active[idx]:
                grad += gradient_overlap(h, idx - (n - 1))
        grad /= n_act
        h = project_h(h - lr * grad, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best = h.copy()
        if step % 3000 == 2999:
            lr *= 0.65
    return best, best_c5

        overlaps = compute_overlaps(h)

        tol = max(1e-13, max_val * 5e-10)
        active = overlaps >= max_val - tol
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            break
        m = len(active_idx)
        A_ub = np.zeros((m, n))
        for i, idx in enumerate(active_idx):
            A_ub[i] = gradient_overlap(h, idx - (n - 1))
        b_ub = np.full(m, -1.0)
        A_eq = np.array([[np.ones(n)]])
        b_eq = np.array([0.0])

        res = linprog(c=np.zeros(n), A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and np.linalg.norm(res.x) > 1e-12:
