        h_work[v] = hv0 - ba
        new_c5 = compute_c5(h_work)

        delta = new_c5 - current_c5
        if delta < -1e-14:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_work.copy()
            stagnation = 0
        elif T > T_floor and delta < 0.05 and rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            stagnation += 1
        else:
            h_work[u] = hu0
            h_work[v] = hv0
            stagnation += 1

        if stagnation >= max_stag:
            h_work = best_h.copy()
            current_c5 = best_c5
            stagnation = 0
            for _ in range(8):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    amt = rng.uniform(0.02, 0.15)
                    if rng.random() < 0.5:
                        amt = min(amt, h_work[i], 1.0 - h_work[j])
                    else:
                        amt = min(amt, h_work[j], 1.0 - h_work[i])
                        i, j = j, i
                    if amt > 0:
                        h_work[i] -= amt
                        h_work[j] += amt
            h_work = make_feasible(h_work, n, rng)
            current_c5 = compute_c5(h_work)
            T = max(T * 2.0, 0.001)

        T = max(T * 0.99992, T_floor)

    # Phase 2: upsample to 2n and optimize
    phase2_end = start_time + budget_s * 0.91
    if time.time() < phase2_end:
        n2 = 2 * n
        h2 = np.zeros(n2)
        h2[0::2] = best_h
        h2[1::2] = best_h
        h2 = make_feasible(h2, n2, rng)

        best_h2 = h2.copy()
        best_c5_2 = compute_c5(best_h2)
        h_work2 = best_h2.copy()
        current_c5_2 = best_c5_2
        T2 = 0.002
        stagnation2 = 0
        n_ternary2 = 32

        while time.time() < phase2_end:
            u = rng.randint(n2)
            v = rng.randint(n2)
            if u == v:
                continue

            a_lo = max(-h_work2[u], h_work2[v] - 1.0)
            a_hi = min(1.0 - h_work2[u], h_work2[v])
            if a_hi - a_lo < 1e-12:
                continue

            hu0 = h_work2[u]
            hv0 = h_work2[v]

            a, b = a_lo, a_hi
            for _ in range(n_ternary2):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h_work2[u] = hu0 + m1
                h_work2[v] = hv0 - m1
                o1 = compute_c5(h_work2)
                h_work2[u] = hu0 + m2
                h_work2[v] = hv0 - m2
                o2 = compute_c5(h_work2)
                if o1 < o2:
                    b = m2
                else:
                    a = m1

            ba = (a + b) / 2.0
            h_work2[u] = hu0 + ba
            h_work2[v] = hv0 - ba
            new_c5 = compute_c5(h_work2)

            delta = new_c5 - current_c5_2
            if delta < -1e-14:
                current_c5_2 = new_c5
                if new_c5 < best_c5_2:
                    best_c5_2 = new_c5
                    best_h2 = h_work2.copy()
                stagnation2 = 0
            elif T2 > 1e-10 and delta < 0.05 and rng.random() < np.exp(-delta / T2):
                current_c5_2 = new_c5
                stagnation2 += 1
            else:
                h_work2[u] = hu0
                h_work2[v] = hv0
                stagnation2 += 1

            if stagnation2 >= 3000:
                h_work2 = best_h2.copy()
                current_c5_2 = best_c5_2
                stagnation2 = 0
                for _ in range(8):
                    i = rng.randint(n2)
                    j = rng.randint(n2)
                    if i != j:
                        amt = rng.uniform(0.02, 0.12)
                        if rng.random() < 0.5:
                            amt = min(amt, h_work2[i], 1.0 - h_work2[j])
                        else:
                            amt = min(amt, h_work2[j], 1.0 - h_work2[i])
                            i, j = j, i
                        if amt > 0:
                            h_work2[i] -= amt
                            h_work2[j] += amt
                h_work2 = make_feasible(h_work2, n2, rng)
                current_c5_2 = compute_c5(h_work2)
                T2 = max(T2 * 2.0, 0.001)

            T2 = max(T2 * 0.99995, 1e-10)

        if best_c5_2 < best_c5:
            best_h = best_h2
            best_c5 = best_c5_2

    # Final: validate and return
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
