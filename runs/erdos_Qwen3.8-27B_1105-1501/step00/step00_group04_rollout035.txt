import math

        G[idx1] += 1.0 - h[idx1 + dd]

def project_feasible(y):

    target = n * 0.5
    y = np.clip(y.astype(float).copy(), 0, 1)
    
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    z = np.clip(y - lam, 0, 1)
    
    # Fix residual
    r = target - np.sum(z)
    if abs(r) > 1e-10:
        if r > 0:
            mask = z < 1.0 - 1e-12
        else:
            mask = z > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            z[mask] += r / cnt
            z = np.clip(z, 0, 1)
        # Final exact fix
        r2 = target - np.sum(z)
        if abs(r2) > 1e-13:
            if r2 > 0:
                order = np.argsort(-z)
            else:
                order = np.argsort(z)
            for i in order:
                if r2 > 0:
                    step = min(1.0 - z[i], r2)
                    z[i] += step
                    r2 -= step
                else:
                    step = min(z[i], -r2)
                    z[i] -= step
                    r2 += step
                if abs(r2) < 1e-13:
                    break

    n = len(h)

    G_list = []
    rhs_list = []
    for p in active_p:

        if np.linalg.norm(G) < 1e-8:

def project_feasible(y):
    """Project onto {0<=x<=1, sum(x)=len(x)/2}."""
    n = len(y)
    target = n * 0.5
    y = np.clip(y.astype(float).copy(), 0, 1)
    
    lo, hi = -2.0, 2.0
    for _ in range(60):

        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) * 0.5
    z = np.clip(y - lam, 0, 1)
    
    r = target - np.sum(z)
    if abs(r) > 1e-10:
        if r > 0:
            mask = z < 1.0 - 1e-12
        else:
            mask = z > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            z[mask] += r / cnt
            z = np.clip(z, 0, 1)
    
    # Exact fix
    r2 = target - np.sum(z)
    if abs(r2) > 1e-13:
        if r2 > 0:
            order = np.argsort(-z)
        else:
            order = np.argsort(z)
        for i in order:
            if r2 > 0:
                step = min(1.0 - z[i], r2)
                z[i] += step
                r2 -= step
            else:
                step = min(z[i], -r2)
                z[i] -= step
                r2 += step
            if abs(r2) < 1e-13:
                break
    return z

    prob = cp.Problem(cp.Minimize(0.5 * cp.sum_squares(u)), constraints_strict)

            u_val = np.array(u.value).flatten()

                nrm = np.linalg.norm(u_val)
                if nrm > rho:
                    u_val *= rho / nrm
                return u_val

        pass

    """Try pair moves to improve."""
    n = len(h)

    # For direction: increase h[i] (where g[i] is negative, meaning increase helps)
    # and decrease h[j] (where g[j] is positive, meaning decrease helps)
    # Actually: if we do h[i] += d, h[j] -= d, the change in Y_p is:
    #   G_p[i] * d + G_p[j] * (-d) = (G_p[i] - G_p[j]) * d
    # We want this negative for all active p, i.e., G_p[i] < G_p[j]
    # So aggregate: we want g_aggregate[i] < g_aggregate[j]
    
    i_cands = np.where(g_aggregate < np.median(g_aggregate))[0]
    j_cands = np.where(g_aggregate > np.median(g_aggregate))[0]
    
    best_c5 = dx * M
    best_h = h.copy()
    improved = False
    
    # Sample pairs
    n_try = min(max_pairs, len(i_cands) * len(j_cands))

        hi = min(1, 1 - h[i])  # h[i] + delta <= 1 => delta <= 1-h[i]

import numpy as np
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode='full'))

def _overlap(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_row(h, p):
    n = len(h)
    c = n - 1
    d = p - c
    G = np.zeros(n)
    if d == 0:
        G = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        G[:m] += -h[d:]
        G[d:] += 1.0 - h[:m]
    else:
        dd = -d
        m = n - dd
        G[:m] += 1.0 - h[dd:]
        G[dd:] += -h[:m]
    return G

def _project(y):
    n = len(y)
    target = n * 0.5
    y = np.clip(y.astype(float).copy(), 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0, 1)
    r = target - np.sum(z)
    if abs(r) > 1e-10:
        mask = (z < 1.0 - 1e-12) if r > 0 else (z > 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            z[mask] += r / cnt
            z = np.clip(z, 0, 1)
    r2 = target - np.sum(z)
    if abs(r2) > 1e-13:
        order = np.argsort(-z) if r2 > 0 else np.argsort(z)
        for i in order:
            if r2 > 0:
                s = min(1.0 - z[i], r2)
                z[i] += s
                r2 -= s
            else:
                s = min(z[i], -r2)
                z[i] -= s
                r2 += s
            if abs(r2) < 1e-13:
                break
    return z

def _solve_qp(h, Y, act_p, M, eta, rho):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    n = len(h)
    Gl, rl = [], []
    for p in act_p:
        G = _grad_row(h, p)
        if np.linalg.norm(G) < 1e-8:
            continue
        Gl.append(G)
        rl.append(M - Y[p] - eta)
    if not Gl:
        return None
    Gmat = np.array(Gl)
    rhs = np.array(rl)
    na = len(Gmat)
    u = cp.Variable(n)
    bc = [u >= -h, u <= 1.0 - h, cp.sum(u) == 0]
    for attempt in range(2):
        try:
            if attempt == 0:
                prob = cp.Problem(cp.Minimize(0.5 * cp.sum_squares(u)),
                                  bc + [Gmat @ u <= rhs])
            else:
                s = cp.Variable(na, nonneg=True)
                prob = cp.Problem(
                    cp.Minimize(0.5 * cp.sum_squares(u) + 40.0 * cp.sum(s)),
                    bc + [Gmat @ u <= rhs + s])
            prob.solve(solver=cp.SCS, verbose=False, max_iters=20000)
            if u.value is not None:
                uv = np.asarray(u.value).flatten().astype(float)
                if np.all(np.isfinite(uv)):
                    uv -= np.sum(uv) / n
                    nm = np.linalg.norm(uv)
                    if nm > 1e-10:
                        if nm > rho:
                            uv *= rho / nm
                        return uv
        except Exception:
            pass
    return None
