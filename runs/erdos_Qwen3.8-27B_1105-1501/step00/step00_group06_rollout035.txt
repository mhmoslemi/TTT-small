from scipy.optimize import linprog
import time

def score_h(h):

def compute_corr_and_grads(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            grads[m, k:] += (1.0 - h[:n-k])
            grads[m, :n-k] -= h[k:]
        else:
            t = -k
            grads[m, :n-t] += (1.0 - h[t:])
            grads[m, t:] -= h[:n-t]
    return corr, grads

def project_box_mass(v):
    n = len(v)
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    # Bisection to find shift that makes sum = target
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v.copy()
    # Use bisection on a shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        v_shifted = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(v_shifted)
        if s < target:
            lo = mid
        else:
            hi = mid
    v_result = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(v_result)
    if abs(residual) > 1e-10:
        # Distribute residual among free variables
        free = (v_result > 1e-10) & (v_result < 1.0 - 1e-10)
        n_free = np.sum(free)
        if n_free > 0:
            adj = residual / n_free
            v_result[free] = np.clip(v_result[free] + adj, 0.0, 1.0)
    return v_result

def feasible_line_range(h, d, eps=1e-12):

    lam_min = 0.0
    lam_max = 0.0

            lam_max = min(lam_max if lam_max != 0 else 1e10, (1-h[i]) / d[i])

            lam_max = min(lam_max if lam_max != 0 else 1e10, -h[i] / d[i])

    return lam_min, lam_max

def solve_lp_trust(h_c, corr_c, grads, radius):
    n = len(h_c)
    # LP: minimize t
    # subject to: corr_c[m] + grads[m] @ (h - h_c) <= t for all m
    #             sum(h) = n/2
    #             h_c[i] - radius <= h[i] <= h_c[i] + radius
    #             0 <= h[i] <= 1
    # Variables: h[0], ..., h[n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: grads[m] @ h - grads[m] @ h_c + corr_c[m] <= t
    # i.e., grads[m] @ h - t <= grads[m] @ h_c - corr_c[m]
    # A_ub @ x <= b_ub
    n_lags = 2*n - 1
    A_ub = np.zeros((n_lags, n+1))
    A_ub[:, :n] = grads  # each row m: grads[m] @ h
    A_ub[:, n] = -1.0    # -t
    b_ub = np.array([np.dot(grads[m], h_c) - corr_c[m] for m in range(n_lags)])
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: max(0, h_c[i]-radius) <= h[i] <= min(1, h_c[i]+radius)
    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_new = res.x[:n]
            # Project to ensure feasibility
            h_new = project_box_mass(h_new)
            return h_new
    except:
        pass
    return None

def make_random_direction(n, rng, h_c):
    d = rng.randn(n)
    d -= np.mean(d)  # zero sum to preserve mass
    if np.sum(d**2) < 1e-20:

    d /= np.sqrt(np.sum(d**2))
    return d

def make_gradient_direction(h_c, grads, rng):

    n = len(h_c)
    corr_c = np.correlate(h_c, 1.0 - h_c, mode='full')

    d -= np.mean(d)
    if np.sum(d**2) < 1e-20:
        return None
    d /= np.sqrt(np.sum(d**2))
    return d

import numpy as np
from scipy.optimize import linprog
import time

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grads(h):
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            grads[m, k:] += (1.0 - h[:n-k])
            if n-k > 0:
                grads[m, :n-k] -= h[k:]
        else:
            t = -k
            if n-t > 0:
                grads[m, :n-t] += (1.0 - h[t:])
                grads[m, t:] -= h[:n-t]
    return grads

def project_box_mass(v):
    n = len(v)
    target = n / 2.0
    v = np.array(v, dtype=float)
    # First clamp
    v = np.clip(v, 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-14:
        return v
    
    # Bisection on shift
    lo, hi = -2.0, 2.0
    # Ensure we bracket
    while np.sum(np.clip(v + hi, 0, 1)) < target and hi < 10:
        hi *= 2
    while np.sum(np.clip(v + lo, 0, 1)) > target and lo > -10:
        lo *= 2
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    v_result = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(v_result)
    if abs(residual) > 1e-12:
        free = (v_result > 1e-9) & (v_result < 1.0 - 1e-9)
        n_free = int(np.sum(free))
        if n_free > 0:
            adj = residual / n_free
            v_result[free] = np.clip(v_result[free] + adj, 0.0, 1.0)
    return v_result

def get_feasible_range(h, d):
    eps = 1e-14
    lam_min = 0.0
    lam_max = 0.0
    pos = d > eps
    neg = d < -eps
    if np.any(pos):
        lam_min = max(lam_min, -np.min(h[pos] / d[pos]))
        lam_max = min(lam_max if lam_max > 0 else 1e10, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        lam_min = max(lam_min, np.min((1.0 - h[neg]) / d[neg]))
        lam_max = min(lam_max if lam_max > 0 else 1e10, -np.min(h[neg] / d[neg]))
    # Shrink slightly
    span = lam_max - lam_min
    lam_min += 1e-10 * max(1.0, span)
    lam_max -= 1e-10 * max(1.0, span)
    if lam_min > lam_max:
        lam_min, lam_max = 0.0, 0.0
    return lam_min, lam_max

def line_search_quad(h_c, d, corr_c, grads, score_c):
    n = len(h_c)
    dx = 2.0 / n
    
    lam_min, lam_max = get_feasible_range(h_c, d)
    if lam_max - lam_min < 1e-12:
        return None, score_c
    
    # Quadratic coefficients
    a = -np.correlate(d, d, mode='full')
    b = grads @ d
    
    # Grid: include both positive and negative lambda
    n_pos = 200
    n_neg = 100
    lams_pos = np.linspace(0, lam_max, n_pos) if lam_max > 0 else np.array([0.0])
    lams_neg = np.linspace(lam_min, 0, n_neg) if lam_min < 0 else np.array([0.0])
    lams = np.concatenate([lams_neg, lams_pos])
    
    # Vectorized evaluation
    A_vals = a[None, :] * (lams[:, None] ** 2) + b[None, :] * lams[:, None] + corr_c[None, :]
    scores = np.max(A_vals, axis=1) * dx
    
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_lam = lams[best_idx]
    
    if best_score >= score_c - 1e-14:
        return None, score_c
    
    # Refine around best
    span = (lam_max - lam_min) / max(1, len(lams))
    lo = max(lam_min, best_lam - 5 * span)
    hi = min(lam_max, best_lam + 5 * span)
    lams_fine = np.linspace(lo, hi, 200)
    A_vals_fine = a[None, :] * (lams_fine[:, None] ** 2) + b[None, :] * lams_fine[:, None] + corr_c[None, :]
    scores_fine = np.max(A_vals_fine, axis=1) * dx
    best_fine_idx = np.argmin(scores_fine)
    best_lam = lams_fine[best_fine_idx]
    best_score = scores_fine[best_fine_idx]

def solve_lp(h_c, corr_c, grads, radius):
    n = len(h_c)
    n_lags = 2 * n - 1
    
    # Variables: h (n) + t (1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequalities: grads @ h - t <= grads @ h_c - corr_c
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_lags)
    for m in range(n_lags):
        b_ub[m] = np.dot(grads[m], h_c) - corr_c[m]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = [(lb[i], ub[i]) for i in range(n)]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def make_random_dir(n, rng):
    d = rng.randn(n)
    d -= np.mean(d)
    norm = np.sqrt(np.sum(d**2))
    if norm < 1e-14:
        return None
    return d / norm

    c_obj[n] = 1.0
    A_ub = np.zeros((nl, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = grads @ h_c - corr_c
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = [(lb[i], ub[i]) for i in range(n)]
    bounds.append((-100.0, 100.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None


def rand_dir(n, rng):
    d = rng.randn(n)
    d -= d.mean()
    nm = np.sqrt(np.sum(d ** 2))
    if nm < 1e-14:
        return None
    return d / nm


def make_harmonic_dir(n, mode, rng):
    t = np.arange(n)
    sign = 1.0 if rng.rand() < 0.5 else -1.0
    d = sign * np.sin(2.0 * np.pi * mode * t / n)
    d -= d.mean()
    nm = np.sqrt(np.sum(d ** 2))
    if nm < 1e-14:
        return None
    return d / nm


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(70, 0.5)

    h = project_box_mass(h)
    sc = score_h(h)
    n = len(h)
    best_h = h.copy()
    best_sc = sc
    radius = 0.15
    nc = 0
    iter_count = 0

    while time.time() - t0 < budget_s - 5:
        iter_count += 1
        corr_c = np.correlate(h, 1.0 - h, mode='full')
        grads = compute_grads(h)
        imp = False

        # Method 1: LP trust region
        if iter_count % 2 == 0 or nc == 0:
            h_lp = solve_lp_dir(h, corr_c, grads, radius)
            if h_lp is not None:
                d = h_lp - h
                dm = np.sqrt(np.sum(d ** 2))
                if dm > 1e-12:
                    d = d * min(1.0, radius / dm)
                    hc, sc2 = ls_quad(h, d, corr_c, grads, sc)
                    if hc is not None:
                        h, sc, imp = hc, sc2, True

        # Method 2: Negative gradient of top active constraints
        if not imp:
            tl = np.argsort(corr_c)[-5:]
            for idx in tl:
                d = -grads[idx].copy()
                d -= d.mean()
                dm = np.sqrt(np.sum(d ** 2))
                if dm < 1e-12:
                    continue
                d = d * radius / dm
                hc, sc2 = ls_quad(h, d, corr_c, grads, sc)
                if hc is not None:
                    h, sc, imp = hc, sc2, True
                    break

        # Method 3: Harmonic directions
        if not imp and nc > 3:
            for mode in [2, 3, 5, 7]:
                d = make_harmonic_dir(n, mode, rng)
                if d is None:
                    continue
                d = d * radius * 0.5
                hc, sc2 = ls_quad(h, d, corr_c, grads, sc)
                if hc is not None:
                    h, sc, imp = hc, sc2, True
                    break

        # Method 4: Random directions
        if not imp:
            n_random = 4
            for _ in range(n_random):
                d = rand_dir(n, rng)
                if d is None:
                    continue
                d = d * radius * 0.7
                hc, sc2 = ls_quad(h, d, corr_c, grads, sc)
                if hc is not None:
                    h, sc, imp = hc, sc2, True
                    break

        if imp:
            nc = 0
            if sc < best_sc:
                best_sc = sc
                best_h = h.copy()
        else:
            nc += 1
            if nc == 5:
                radius = max(0.003, radius * 0.5)
            if nc >= 15:
                # Escape mechanism: try several perturbations
                best_pert_h = h
                best_pert_sc = sc
                for _ in range(8):
                    scale = 0.01 + 0.03 * rng.rand()
                    p = rng.randn(n) * scale
                    p -= p.mean()
                    hp = project_box_mass(h + p)
                    sp = score_h(hp)
                    if sp < best_pert_sc:
                        best_pert_sc = sp
                        best_pert_h = hp
                if best_pert_sc < sc + 0.003:
                    h, sc = best_pert_h, best_pert_sc
                if sc < best_sc:
                    best_sc = sc
                    best_h = h.copy()
                radius = 0.12
                nc = 0

    # Final verification
    if best_sc < sc:
        h = best_h
        sc = best_sc

    sc_final = score_h(h)
    if sc_final > best_sc:
        h = best_h
        sc_final = best_sc

    return (h, sc_final, len(h))
