from scipy.optimize import linprog

def _active_set_lp(h, n, active_indices, max_iter=100):
    """Solve LP to find a step that reduces active overlaps."""
    dx = 2.0 / n
    all_grads = _compute_grad_overlaps(h)  # shape (2n-1, n)
    ov = np.correlate(h, 1-h, mode="full") * dx  # shape (2n-1,)
    
    # Active set: indices where overlap is close to max
    max_ov = np.max(ov)
    active = np.where(ov >= max_ov - 1e-6)[0]
    
    # LP: minimize t
    # s.t. t - sum_j grad[k,j] * h'[j] >= ov[k] for k in active
    #      0 <= h[j] + h'[j] <= 1
    #      sum(h') = 0
    #
    # Variables: t (1) + h' (n) = n+1 variables
    # 
    # In standard form for linprog:
    # minimize c^T x
    # subject to A_ub @ x <= b_ub
    #            A_eq @ x = b_eq
    #            bounds
    
    # c = [1, 0, 0, ..., 0] (minimize t)
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    # Inequality constraints: for each active k:
    # sum_j grad[k,j] * h'[j] - t <= -ov[k]
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for i, k in enumerate(active):
        A_ub[i, 0] = -1.0
        A_ub[i, 1:] = all_grads[k, :]
        b_ub[i] = -ov[k]
    
    # Also: 0 <= h + h' <= 1
    # h' >= -h
    # h' <= 1 - h
    # These are handled by bounds below
    
    # Equality: sum(h') = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: t is free, -h[j] <= h'[j] <= 1-h[j]
    bounds = [(None, None)] + [(-h[j], 1.0 - h[j]) for j in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_step = res.x[1:]
        h_new = h + h_step
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        h_new = _project_feasible(h_new, n)
        return h_new
    return h

For s≠0: f_s = dx * sum_i h[i] - dx * sum_i h[i]h[i+s] = 1 - dx * sum_i h[i]h[i+s]

    n = len(h)
    dx = 2.0 / n
    all_grads = np.zeros((2*n-1, n))
    one_minus_h = 1.0 - h
    for k in range(2*n-1):

        j_start = max(0, -s)
