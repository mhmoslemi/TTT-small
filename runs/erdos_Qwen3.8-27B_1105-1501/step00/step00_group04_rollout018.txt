import time
import random

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def is_feasible(h, tol=1e-9):
    n = len(h)
    if np.any(h < -tol) or np.any(h > 1 + tol):
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def repair_feasible(y):
    n = len(y)
    tau = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    s = np.sum(y)
    if s > tau:

        order = np.argsort(-y)  # largest first

            y[i] -= rem
            excess -= rem

        order = np.argsort(y)  # smallest first
        for i in order:

            add = min(1.0 - y[i], deficit)
            y[i] += add
            deficit -= add

    """Return all C_d * dx values for d = -(n-1), ..., 0, ..., n-1"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def active_gradient(h, scores, score_tol):
    """Compute subgradient of max(scores) using active set."""
    n = len(h)
    max_val = np.max(scores)
    active = scores >= max_val - score_tol
    # scores array has 2n-1 elements, indexed 0 to 2n-2
    # offset d corresponds to index d + n - 1
    g = np.zeros(n)

    for idx in range(2 * n - 1):
        if not active[idx]:
            continue
        d = idx - center  # shift
        if d == 0:
            g += 1.0 - 2.0 * h
        elif d > 0:
            # C_d = sum_{i=0}^{n-1-d} h[i+d](1-h[i])

            for j in range(d, n):

            for j in range(0, n - d):
                g[j] -= h[j + d]

            for j in range(r, n):

This is reminiscent of the problem of constructing sequences with low auto-correlation (like Barker sequences, m-sequences, etc.), but with a different objective (we want to minimize the peak cross-correlation with the complement, not the auto-correlation).

C_d^- = #(p: h[p]=1, h[p+d]=0) for p=0..n-1-d

Note that C_d^+ + C_d^- = #(p: h[p+d] != h[p]) for p=0..n-1-d = number of "transitions" at distance d.

And C_d^+ - C_d^- = #(p: h[p+d]=1) - #(p: h[p]=1) for the relevant range... let me compute:
C_d^+ - C_d^- = sum h[p+d](1-h[p]) - sum h[p](1-h[p+d])
= sum h[p+d] - sum h[p+d]h[p] - sum h[p] + sum h[p]h[p+d]
= sum_{p=0}^{n-1-d} h[p+d] - sum_{p=0}^{n-1-d} h[p]

So C_d^+ - C_d^- = (number of ones in positions d to n-1) - (number of ones in positions 0 to n-1-d)

This is the "running sum" difference. If the ones are uniformly distributed, this is close to 0, and C_d^+ ≈ C_d^- ≈ (transitions at distance d)/2.

So minimizing max(C_d^+, C_d^-) is roughly equivalent to minimizing the maximum number of transitions at any distance, subject to the running sum being balanced.

This is a well-studied problem in coding theory (related to "run length" and "auto-correlation" properties).

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def is_feasible(h, tol=1e-8):
    n = len(h)
    if np.any(np.isnan(h)) or np.any(np.isinf(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1 + tol:
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def repair_feasible(y):
    n = len(y)
    tau = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    diff = tau - np.sum(y)
    if diff > 1e-14:
        # Need to add: add to smallest values first
        order = np.argsort(y)
        for i in order:
            if diff <= 1e-14: break
            add = min(1.0 - y[i], diff)
            y[i] += add
            diff -= add
    elif diff < -1e-14:
        # Need to remove: remove from largest values first
        order = np.argsort(-y)
        for i in order:
            if -diff <= 1e-14: break
            rem = min(y[i], -diff)
            y[i] -= rem
            diff += rem
    return y

def smooth_score_and_grad(h, alpha=50.0):
    """Compute smooth approximation of max score and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    # Log-sum-exp with shift for numerical stability
    max_s = np.max(scores)
    log_sum = max_s + np.log(np.sum(np.exp(alpha * (scores - max_s))))
    f = log_sum / alpha
    # Weights
    weights = np.exp(alpha * (scores - max_s))
    weights /= np.sum(weights)
    # Gradient
    g = np.zeros(n)
    center = n - 1

        if w < 1e-10:

        # Line search
        best_t = 0.0
        best_f = f
        for i in range(1, 21):
            t = i / 20.0 * (2.0 / (1.0 + i / 10.0))  # decreasing steps
            t = min(t, 1.0)
            h_new = h + t * direction
            h_new = repair_feasible(h_new)
            f_new = exact_score(h_new)
            if f_new < best_f - 1e-15:
                best_f = f_new
                best_t = t
                best_h = h_new
        
        if best_t > 0:
            h = best_h
            f = best_f
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
    
    return h, f
