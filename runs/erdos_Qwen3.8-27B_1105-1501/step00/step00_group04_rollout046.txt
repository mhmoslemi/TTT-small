def smooth_max_grad(h, T):
    n = len(h)
    comp = 1.0 - h
    c = np.correlate(h, comp, mode='full')  # length 2n-1
    center = n - 1
    # stable log-sum-exp
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / T)
    S = np.sum(exp_c)
    if S == 0:
        # all underflow, just use argmax
        weights = np.zeros(2*n-1)
        weights[c == cmax] = 1.0 / np.sum(c == cmax)
        smooth_val = cmax
        grad = np.zeros(n)
        # compute gradient with these weights
    else:
        weights = exp_c / S
        smooth_val = cmax + T * np.log(S)
        grad = np.zeros(n)
    
    # center term (d=0)
    w0 = weights[center]
    # c[center] = sum_i h[i]*(1-h[i])
    # d/dh[j] = (1-h[j]) - h[j] = 1-2h[j]
    grad += w0 * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        wp = weights[center + d]
        wn = weights[center - d]

            grad[:n-d] += wp * (-h[d:])  # j in [0, n-d-1], h[j+d] = h[d:n]

    # Fix any residual

def make_feasible(y, n=None):
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=float).flatten()

        return np.sum(np.clip(y - lam, 0.0, 1.0))

            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-13:
        order = np.arange(n)
        rng_local = np.random.default_rng(0)
        # Adjust elements with most slack first
        if diff > 0:
            order = np.where(z < 1.0)[0]
        else:

        for i in order:

                break
            if diff > 0:

                z[i] += add
                diff -= add
            else:
                sub = min(-diff, z[i])
                z[i] -= sub
                diff += sub
    return z

def score_exact(h):
    h = np.asarray(h, dtype=float).flatten()
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def score_vector(h):
    h = np.asarray(h, dtype=float).flatten()
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def smooth_max_and_grad(h, T):
    n = len(h)
    c = score_vector(h)
    L = len(c)
    cmax = c.max()
    scaled = (c - cmax) / T
    # Clip to avoid overflow (though we already subtracted max)
    exp_c = np.exp(np.clip(scaled, -50, 0))
    S = exp_c.sum()
    if S < 1e-300:
        weights = np.zeros(L)
        weights[c == cmax] = 1.0 / np.sum(c == cmax)
        smooth_val = cmax
    else:
        weights = exp_c / S
        smooth_val = cmax + T * np.log(S)
    
    grad = np.zeros(n)
    center = n - 1
    
    # d=0: c[center] = sum h[i]*(1-h[i]), grad = 1-2h
    w0 = weights[center]
    if w0 != 0:
        grad += w0 * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        wp = weights[center + d]
        wn = weights[center - d]
        if wp != 0:
            grad[d:] += wp * (1.0 - h[:n-d])
            grad[:n-d] -= wp * h[d:]
        if wn != 0:
            grad[:n-d] += wn * (1.0 - h[d:])
            grad[d:] -= wn * h[:n-d]
    
    return smooth_val, weights, grad

def resample_vector(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return make_feasible(h_new, new_n)

def pair_optimize(h, deadline, rng, n_pairs=20):
    n = len(h)
    h_cur = h.copy()
    cur_c = score_exact(h_cur)
    
    for _ in range(n_pairs):
        if time.time() >= deadline - 2:
            break
        # Choose a random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # h[i] + alpha, h[j] - alpha
        # Bounds: -h[i] <= alpha <= 1 - h[i] and -(1-h[j]) <= -alpha <= h[j]
        # So alpha in [max(-h[i], h[j]-1), min(1-h[i], h[j])]
        lo = max(-h_cur[i], h_cur[j] - 1.0)
        hi = min(1.0 - h_cur[i], h_cur[j])
        if hi - lo < 1e-10:
            continue
        
        # Grid search
        n_grid = 30
        alphas = np.linspace(lo, hi, n_grid)
        best_alpha = 0.0
        best_c_local = cur_c
        
        for a in alphas:
            h_cur[i] += a
            h_cur[j] -= a
            c_test = score_exact(h_cur)
            h_cur[i] -= a
            h_cur[j] += a
            if c_test < best_c_local:
                best_c_local = c_test
                best_alpha = a
        
        if best_alpha != 0 and best_c_local < cur_c - 1e-12:
            h_cur[i] += best_alpha
            h_cur[j] -= best_alpha
            cur_c = best_c_local
    
    # Refine with smaller grid on best pair
    return h_cur

def score_exact(h):

    c = np.correlate(h, 1.0 - h, mode='full')

def smooth_max_and_grad(h, T):

    c = score_vector(h)
    L = len(c)
    cmax = c.max()

    S = exp_c.sum()
    if S < 1e-300:
        weights = np.zeros(L)
        mask = (c == cmax)
        weights[mask] = 1.0 / mask.sum()
        smooth_val = cmax
    else:
        weights = exp_c / S
        smooth_val = cmax + T * np.log(S)

    grad = np.zeros(n)
    center = n - 1
    w0 = weights[center]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)

        if wn > 1e-15:
            grad[:n - d] += wn * (1.0 - h[d:])
            grad[d:] -= wn * h[:n - d]
    return smooth_val, weights, grad

def resample_vector(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()

    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return make_feasible(h_new, new_n)
