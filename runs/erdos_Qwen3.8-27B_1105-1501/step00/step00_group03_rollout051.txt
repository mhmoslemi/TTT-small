Let me design the implementation:

1. **Feasibility projection**: Given a vector z of length n, find λ such that sum(clip(z+λ, 0, 1)) = n/2 using bisection.

2. **Official score**: `np.correlate(h, 1-h, 'full').max() * dx` where dx = 2/n.

3. **Manual correlation values**: For each lag k from -(n-1) to (n-1), compute sum of h[i]*(1-h[i+k]) over valid indices.

4. **Smooth objective**: Log-sum-exp of scaled correlations.

5. **Smooth gradient**: Weighted sum of correlation gradients.

6. **Direct minimax**: SLSQP with variables (h, θ), minimize θ subject to θ >= dx*F_k(h) for all k, sum(h) = n/2, 0 <= h_i <= 1.

def project_feasible(z, n=None, target=None):
    """Project z onto {h: 0<=h_i<=1, sum(h)=target}"""
    if n is None:
        n = len(z)
    if target is None:
        target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    z = np.clip(z, -10, 10)  # safety
    
    # Bisection on lambda: sum(clip(z + lambda, 0, 1)) = target
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(z + lam, 0.0, 1.0)
    
    # Fix residual
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            # Need to add more; find indices with room
            room = 1.0 - h
            idx = np.argsort(-room)
            add = residual
            for i in idx:
                delta = min(add, room[i])
                h[i] += delta
                add -= delta
                if add <= 1e-12:
                    break
        else:
            # Need to remove
            idx = np.argsort(-h)
            rem = -residual
            for i in idx:
                delta = min(rem, h[i])
                h[i] -= delta
                rem -= delta
                if rem <= 1e-12:
                    break
    
    # Ensure bounds
    h = np.clip(h, 0.0, 1.0)
    
    # Final residual fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # Adjust one element

            h[idx] = min(1.0, h[idx] + residual)

Now for the smooth objective:
Φ_β(h) = C_max + (1/β) * log(sum exp(β(C_r - C_max)))

where C_r = dx * F_r(h)

Gradient: ∇Φ_β = dx * sum_r w_r * ∇F_r

where w_r = exp(β(C_r - C_max)) / sum exp(β(C_r - C_max))

For the SLSQP minimization with the smooth objective:
- Variables: h (length n)
- Objective: Φ_β(h)
- Gradient: as above

After each smooth optimization, project to feasible and evaluate exact score.

- Bounds: 0 <= h_i <= 1, 2e-12 <= θ <= some upper bound
- Equality: sum(h) - n/2 = 0

    z = np.asarray(z, dtype=float).copy()
    n = len(z)

        target = n / 2.0
    z[np.isnan(z)] = 0.5
    z = np.clip(z, -5.0, 5.0)
    lo = -1.0 - z.max() - 1.0

        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(z + lam, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual > 0:
                d = min(residual, 1.0 - h[i])
                h[i] += d
                residual -= d
            else:
                d = min(-residual, h[i])
                h[i] -= d

def manual_correlations(h):
    n = len(h)
    L = 2 * n - 1
    omh = 1.0 - h
    vals = np.empty(L)
    for r in range(L):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            vals[r] = np.dot(h[k:k + m], omh[:m])
        else:
            ka = -k
            m = n - ka
            vals[r] = np.dot(h[:m], omh[ka:ka + m])
    return vals


def smooth_obj(h, beta, dx):
    vals = manual_correlations(h)
    C = vals * dx
    Cmax = C.max()
    shifted = beta * (C - Cmax)
    max_s = shifted.max()
    if max_s < -700:
        return Cmax + np.log(len(C)) / beta
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    if sum_exp < 1e-300:
        return Cmax + np.log(len(C)) / beta
    return Cmax + (max_s + np.log(sum_exp)) / beta


def smooth_jac(h, beta, dx):
    vals = manual_correlations(h)
    C = vals * dx
    Cmax = C.max()
    shifted = beta * (C - Cmax)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    sum_exp = exp_s.sum()
    if sum_exp < 1e-300:
        weights = np.ones(len(C)) / len(C)
    else:
        weights = exp_s / sum_exp
    n = len(h)
    grad = np.zeros(n)
    for r in range(len(C)):
        w = weights[r]
        if w < 1e-15:
            continue
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            grad[k:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[k:k + m])
        else:
            ka = -k
            m = n - ka
            grad[:m] += w * (1.0 - h[ka:ka + m])
            grad[ka:] += w * (-h[:m])

    # Final validation
    best_h = project_feasible(best_h)
    final_score = official_score(best_h)
    if not final_score < best_score + 1e-10:
        final_score = best_score
    best_score = final_score
    c5_bound = float(np.nextafter(best_score, np.inf))

    return (best_h, c5_bound, len(best_h))
