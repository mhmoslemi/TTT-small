def _project(y, target):
    """Projection onto {h: sum(h)=target, 0<=h<=1}."""
    n = len(y)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

def _repair(h):
    """Ensure h is feasible: sum = n/2, 0<=h<=1."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    return _project(h, target)

def _active_gradient(h, shift, n):
    """Gradient of f(shift) w.r.t. h."""
    grad = np.zeros(n)

    elif shift > 0:

        grad[shift:n] -= h[0:n-shift]
    else:
        s = -shift
        grad[s:n] += (1.0 - h[0:n-s])
        grad[0:n-s] -= h[s:n]
    return grad

def _resolution_transfer(h_src, n_tgt):
    """Transfer h from source resolution to target resolution, preserving area."""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    h_tgt = np.zeros(n_tgt)
    src_pos = 0
    
    for k in range(n_tgt):
        t_left = k * dx_tgt
        t_right = (k + 1) * dx_tgt
        
        # Find overlapping source bins
        area = 0.0
        s = src_pos
        while s < n_src:
            s_left = s * dx_src
            s_right = (s + 1) * dx_src

            if s_left >= t_right:

            s += 1
