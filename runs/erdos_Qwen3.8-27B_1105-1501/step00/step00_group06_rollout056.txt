def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")

def project_box_mass(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)

        if s < target:
            mu_lo = mu_mid
        else:
            mu_hi = mu_mid
    mu = 0.5 * (mu_lo + mu_hi)
    y = np.clip(v + mu, 0.0, 1.0)
    residual = target - np.sum(y)
    if abs(residual) > 1e-10:
        interior = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = np.sum(interior)
        if cnt > 0:

            y = np.clip(y, 0.0, 1.0)

def grad_corr_k(h, k_idx):
    n = len(h)
    lag = k_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        L = -lag
        if L < n:
            g[:n - L] += (1.0 - h[L:])
            g[L:] -= h[:n - L]
    return g

def compute_smoothed_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)

    w = exp_c / np.sum(exp_c)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            g += w[k] * grad_corr_k(h, k)
    return dx * g, c

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

    g = np.zeros(n)
    for k_idx in top_k:
        g += grad_corr_k(h, k_idx)
    g /= K
    return dx * g, c
