def compute_grad_overlap(h, m):
    """Compute gradient of overlap component m with respect to h.
    m is the index into np.correlate(h, 1-h, mode='full') output.
    center = N-1, offset = m - center
    """
    N = len(h)
    dx = 2.0 / N
    center = N - 1
    delta = m - center  # offset
    grad = np.zeros(N)
    
    if delta == 0:
        grad = dx * (1.0 - 2.0 * h)
    elif delta > 0:
        # sum over i=0..N-delta-1: h[i] * (1-h[i+delta])
        # d/dh[j]: (1-h[j+delta]) for j < N-delta, minus h[j-delta] for j >= delta
        n_valid = N - delta
        grad[:n_valid] += dx * (1.0 - h[delta:])
        grad[delta:] -= dx * h[:n_valid]

        d = -delta

        grad[:n_valid] -= dx * h[d:]

def dyadic_repair(h, N, bits=20):
    """Repair h to be on a dyadic grid with exact sum."""
    S = 1 << bits
    target_int = N * S // 2  # sum of codes should be N*S/2
    # h values should sum to N/2, so codes should sum to N*S/2
    codes = np.round(h * S).astype(np.int64)
    # Fix sum
    current_sum = np.sum(codes)
    diff = target_int - current_sum
    if diff > 0:
        # Need to add units
        # Add to positions with most room (highest fractional part of h*S - floor)
        fracs = h * S - np.floor(h * S)
        order = np.argsort(-fracs)  # highest fraction first
        added = 0
        for idx in order:
            if added >= diff:
                break
            if codes[idx] < S:
                codes[idx] += 1
                added += 1
    elif diff < 0:
        # Need to remove units
        fracs = h * S - np.floor(h * S)
        order = np.argsort(fracs)  # lowest fraction first
        removed = 0
        for idx in order:
            if removed >= -diff:
                break
            if codes[idx] > 0:
                codes[idx] -= 1
                removed += 1
    return codes.astype(np.float64) / S

    """Project to [0,1]^n with sum = n/2."""
    target = n / 2.0
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = result < 1.0 - 1e-15
            if np.any(mask):
                result[mask] = np.minimum(result[mask] + diff / np.sum(mask), 1.0)
        else:
            mask = result > 1e-15
            if np.any(mask):
                result[mask] = np.maximum(result[mask] + diff / np.sum(mask), 0.0)
    return result

        if parent is not None and parent_n != N:
            x_p = np.linspace(0, 2, parent_n, endpoint=False)
            x_n = np.linspace(0, 2, N, endpoint=False)
            h_interp = np.interp(x_n, x_p, parent)
            candidates.append(project_feasible(np.clip(h_interp, 0, 1), N))
        elif parent is not None and parent_n == N:
            candidates.append(parent.copy())
        
        # Structured
        # Sine wave
        x = np.linspace(0, 2, N, endpoint=False)
        for amp in [0.2, 0.3, 0.35, 0.4]:
            for freq in [1, 2, 3]:
                for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
                    h_sin = np.clip(0.5 + amp * np.sin(np.pi * freq * x + phase), 0, 1)
                    candidates.append(project_feasible(h_sin, N))
        
        # Cosine bump
        for center_frac in [0.25, 0.4, 0.5, 0.6, 0.75]:
            c = center_frac * N
            width = N / 4.0
            h_bump = np.exp(-0.5 * ((np.arange(N) - c) / width) ** 2)
            h_bump = np.clip(h_bump / np.max(h_bump) * 0.8 + 0.1, 0, 1)
            candidates.append(project_feasible(h_bump, N))
        
        # Random
        for _ in range(8):
            h_r = rng.random(N)
            candidates.append(project_feasible(h_r, N))
        
        # Smooth random
        for _ in range(5):
            h_r = rng.random(N)
            # Smooth
            kernel = np.array([1, 2, 3, 2, 1], dtype=float)
            kernel /= kernel.sum()
            h_smooth = np.convolve(h_r, kernel, mode='same')
