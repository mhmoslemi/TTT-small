import numpy as np
import time

def score(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(r) * dx)

def repair(h, n):
    """Ensure h is in [0,1] and sum = n/2"""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Adjust sum
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to reduce: subtract from elements > 0
            order = np.argsort(-h)
            for i in order:
                if h[i] <= 0:
                    break
                adj = min(h[i], diff)
                h[i] -= adj
                diff -= adj
                if abs(diff) < 1e-14:
                    break
        else:
            # Need to increase: add to elements < 1
            order = np.argsort(h)
            for i in order:
                if h[i] >= 1.0:
                    break
                adj = min(1.0 - h[i], -diff)
                h[i] += adj
                diff += adj
                if abs(diff) < 1e-14:
                    break
    return h

def local_search(h, n, dx, max_iter=100000, timeout=None, t0=0):
    """Local search by adjusting pairs of coordinates."""
    best = score(h)
    current = h.copy()
    cur_score = best
    rng = np.random.default_rng(42)
    
    for it in range(max_iter):
        if timeout and time.time() - t0 > timeout:
            break
        # Pick two random indices
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Try to transfer a small amount from i to j or vice versa
        delta = rng.uniform(0.01, 0.1)
        
        # Try i -> j
        if current[i] >= delta:
            new = current.copy()
            new[i] -= delta
            new[j] = min(1.0, new[j] + delta)
            # Check sum is preserved
            if abs(new.sum() - n/2.0) < 1e-10:
                s = score(new)
                if s < cur_score - 1e-12:
                    current = new
                    cur_score = s
                    if s < best:
                        best = s
        # Try j -> i
        if current[j] >= delta:
            new = current.copy()
            new[j] -= delta
            new[i] = min(1.0, new[i] + delta)
            if abs(new.sum() - n/2.0) < 1e-10:
                s = score(new)
                if s < cur_score - 1e-12:
                    current = new
                    cur_score = s
                    if s < best:
                        best = s
    
    return current, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5  # 5s margin
    
    # Get initial values
    if 'initial_h_values' in dir() or True:
        try:
            init = np.array(initial_h_values, dtype=float)
        except:
            init = None
    else:
        init = None
    
    if init is not None:
        n = len(init)
        h = repair(init.copy(), n)
    else:
        n = 50
        h = np.full(n, 0.5)
    
    best_h = h.copy()
    best_c = score(best_h)
    
    # Local search
    h, c = local_search(h, n, 2.0/n, max_iter=200000, timeout=deadline, t0=t0)
    if c < best_c:
        best_h = h.copy()
        best_c = c
    
    # Try different n values
    for n_try in [32, 40, 50, 60, 80, 100, 128, 160, 200]:
        if time.time() > deadline:
            break
        # Resample best_h to new size
        h_new = np.interp(np.linspace(0, 2, n_try, endpoint=False) + 1.0/n_try,
                          np.linspace(0, 2, len(best_h), endpoint=False) + 1.0/len(best_h),
                          best_h)
        h_new = repair(h_new, n_try)
        c_new = score(h_new)
        # Local search on this
        h_ls, c_ls = local_search(h_new, n_try, 2.0/n_try, max_iter=50000, timeout=deadline, t0=t0)
        if c_ls < best_c:
            best_h = h_ls.copy()
            best_c = c_ls
    
    return best_h, best_c, len(best_h)

    target = n / 2.0
    h = h.copy()
    current_score = score_fast(h)
    
    for _pass in range(max_passes):
        improved = False
        for i in range(n):
            old_val = h[i]

            # For each lag k, C_k changes by a_k * delta where delta = new - old
            n_lags = 2*n - 1

                # Positive lag k: C[n-1+k] = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
                # h[i] appears:
                #   as first factor: if i <= n-1-k, contributes (1-h[i+k])
                #   as second factor: if i >= k, contributes -h[i-k]
                if i <= n-1-k:
                    a[n-1+k] += (1.0 - h[i+k])
                if i >= k:
                    a[n-1+k] += (-h[i-k])

                # Equivalent: sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
                # h[i] appears:
                #   as first factor (h[j+k]): if i >= k, j = i-k, contributes (1-h[i-k])
                #   as second factor (1-h[j]): if i <= n-1-k, j = i, contributes -h[i+k]
                if i >= k:

                if i <= n-1-k:
                    a[n-1-k] += (-h[i+k])
