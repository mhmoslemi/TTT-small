def dyadic_snap(h, Q):
    """Snap h to dyadic grid with denominator Q, maintaining exact sum."""
    n = len(h)
    target = n / 2.0
    target_int = round(target * Q)  # Should be exact for Q = 2^b
    
    k = np.round(h * Q).astype(np.int64)
    k = np.clip(k, 0, Q)
    
    diff = target_int - np.sum(k)
    
    if diff > 0:
        # Need to add diff units
        # Sort by fractional part (h_i*Q - k_i) descending - add to those closest to next integer
        frac = h * Q - k
        for _ in range(abs(diff)):
            # Find position where we can add 1 and it's "most deserving"
            mask = k < Q
            if not np.any(mask):
                break
            frac_masked = np.where(mask, frac, -np.inf)
            idx = np.argmax(frac_masked)
            k[idx] += 1
            frac[idx] -= 1.0  # Adjust for bookkeeping
    elif diff < 0:
        # Need to remove |diff| units
        frac = k - h * Q  # How much "below" the current value
        for _ in range(abs(diff)):
            mask = k > 0
            if not np.any(mask):
                break
            frac_masked = np.where(mask, frac, -np.inf)
            idx = np.argmax(frac_masked)
            k[idx] -= 1
            frac[idx] -= 1.0
    
    return k.astype(np.float64) / Q

                    continue

                max_steps = int(max_delta * Q)
                # Coarse search
                step_sizes = [max_steps * f / 10.0 for f in range(1, 11)]

                best_local = best_c

                    h[q] += best_delta / Q
                    best_c = best_local
                    improved = True
        
        if not improved:

    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).flatten()
    except Exception:
        h0 = None
    
    if h0 is not None and len(h0) > 0:
        n = len(h0)
    else:
        n = 47
        h0 = np.full(n, 0.5)
    
    # Ensure feasibility
    h0 = np.clip(h0, 0.0, 1.0)
    Q_init = 2 ** 14
    h0 = dyadic_snap(h0, Q_init)
    
    best_h = h0.copy()
    best_c = exact_c5(best_h)
    
    # Convert to raw space
    z = h_to_raw(best_h, n)
    W_cap = 12.0
    bounds = [(-W_cap, W_cap)] * (n - 1)
    
    # Stage 1: L-BFGS-B with increasing beta
    betas = [3, 8, 20, 50, 150, 400]
    for beta in betas:
        if time.time() > deadline - 40:
            break
        try:
            result = minimize(
                obj_and_grad_z, z,
                args=(n, beta),
                method='L-BFGS-B',
                jac=True,
                bounds=bounds,
                options={'maxiter': 300, 'maxfun': 800, 'gtol': 1e-7}
            )
            if result.fun < 1e10:
                h_new = raw_to_h(result.x, n)
                for Q in [2**12, 2**14, 2**16]:
                    h_s = dyadic_snap(h_new, Q)
                    c = exact_c5(h_s)
                    if c < best_c - 1e-12:
                        best_c = c
                        best_h = h_s.copy()
                        z = h_to_raw(best_h, n)
        except Exception:
            pass
    
    # Stage 2: Random restarts
    rng = np.random.RandomState(seed)
    n_restarts = 8
    if budget_s > 120:
        n_restarts = 15
    for trial in range(n_restarts):
        if time.time() > deadline - 30:
            break
        scale = [0.3, 0.8, 1.5, 3.0, 0.5, 1.0, 2.0, 0.4, 1.2, 2.5, 0.6, 1.8, 0.9, 3.5, 0.7][trial % 15]
        z_pert = z + rng.randn(n - 1) * scale
        z_pert = np.clip(z_pert, -W_cap, W_cap)
        try:
            result = minimize(
                obj_and_grad_z, z_pert,
                args=(n, 15),
                method='L-BFGS-B',
                jac=True,
                bounds=bounds,
                options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-6}
            )
            if result.fun < 1e10:
                h_new = raw_to_h(result.x, n)
                for Q in [2**12, 2**14, 2**16]:
                    h_s = dyadic_snap(h_new, Q)
                    c = exact_c5(h_s)
                    if c < best_c - 1e-12:
                        best_c = c
                        best_h = h_s.copy()
                        z = h_to_raw(best_h, n)
            # Follow-up with higher beta if improved
            if result.fun < 1e10:
                result2 = minimize(
                    obj_and_grad_z, result.x,
                    args=(n, 80),
                    method='L-BFGS-B',
                    jac=True,
                    bounds=bounds,
                    options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-7}
                )
                if result2.fun < 1e10:
                    h_new = raw_to_h(result2.x, n)
                    for Q in [2**12, 2**14, 2**16]:
                        h_s = dyadic_snap(h_new, Q)
                        c = exact_c5(h_s)
                        if c < best_c - 1e-12:
                            best_c = c
                            best_h = h_s.copy()
                            z = h_to_raw(best_h, n)
        except Exception:
            pass
    
    # Stage 3: Pair-line polish
    if time.time() < deadline - 15:
        Q_fine = 2 ** 14
        h_pol, c_pol = pair_line_polish(best_h, Q_fine, n_iter=40, max_pairs=80)
        if c_pol < best_c - 1e-12:
            best_c = c_pol
            best_h = h_pol
    
    # Stage 4: Another round of polish with different Q
    if time.time() < deadline - 10:
        Q_fine2 = 2 ** 16
        h_snap = dyadic_snap(best_h, Q_fine2)
        h_pol2, c_pol2 = pair_line_polish(h_snap, Q_fine2, n_iter=30, max_pairs=50)
        if c_pol2 < best_c - 1e-12:
            best_c = c_pol2
            best_h = h_pol2
    
    # Final verification
    final_c = exact_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    
    # Ensure return value is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c, len(best_h)
