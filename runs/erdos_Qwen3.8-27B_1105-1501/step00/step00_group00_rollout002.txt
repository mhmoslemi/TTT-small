            g[:n - p] += dx * (1.0 - h[p:])
            g[p:] -= dx * h[:n - p]
        grad += w[m] * g
    
    return log_sum_exp, grad

import numpy as np

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(300):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - z.sum()
    if diff > 1e-14:
        order = np.argsort(1.0 - z)
        for idx in order:
            if diff <= 1e-15:
                break
            room = 1.0 - z[idx]
            move = min(room, diff)
            z[idx] += move
            diff -= move
    elif diff < -1e-14:
        order = np.argsort(z)
        for idx in order:
            if diff >= -1e-15:
                break
            room = z[idx]
            move = min(room, -diff)
            z[idx] -= move
            diff += move
    z = np.clip(z, 0.0, 1.0)
    return z

def smooth_obj(x, tau):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    M = np.max(corr)
    return M + tau * np.log(np.sum(np.exp((corr - M) / tau)))

def smooth_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    M = np.max(corr)
    w = np.exp((corr - M) / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += w[m] * dx * (1.0 - x[:n - lag])
            grad[:n - lag] -= w[m] * dx * x[lag:]
        else:
            p = -lag
            grad[:n - p] += w[m] * dx * (1.0 - x[p:])
            grad[p:] -= w[m] * dx * x[:n - p]
    return grad

def mass_fun(x):
    return x.sum() - len(x) / 2.0

def mass_jac(x):
    return np.ones(len(x))

def transfer_resolution(src, new_n):
    n_src = len(src)
    dx_src = 2.0 / n_src
    x_old = np.linspace(0, 2, n_src + 1)
    cum_old = np.zeros(n_src + 1)
    cum_old[1:] = np.cumsum(src * dx_src)
    x_new = np.linspace(0, 2, new_n + 1)
    cum_new = np.interp(x_new, x_old, cum_old)
    new_h = np.diff(cum_new) / (2.0 / new_n)
    return project_feasible(new_h)

def systematic_local_search(h, best_score, max_time, rng, steps):
    n = len(h)
    current = h.copy()
    cur_score = exact_c5(current)
    t0 = time.time()
    
    for step in steps:
        improved = True
        while improved and (time.time() - t0) < max_time:
            improved = False
            for i in range(n):
                if (time.time() - t0) >= max_time:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    for sign in (1, -1):
                        d = sign * step
                        if 0.0 <= current[i] - d + 1e-15 and 0.0 <= current[j] + d + 1e-15:
                            trial = current.copy()
                            trial[i] -= d
                            trial[j] += d
                            trial = np.clip(trial, 0.0, 1.0)
                            # Fix mass
                            dm = current.sum() - trial.sum()
                            if abs(dm) > 1e-14:
                                trial[i] += dm
                                trial = np.clip(trial, 0.0, 1.0)
                            sc = exact_c5(trial)
                            if sc < cur_score - 1e-12:
                                cur_score = sc
                                current = trial
                                improved = True
                                if sc < best_score:
                                    best_score = sc
        # Also try continuous moves
        if (time.time() - t0) >= max_time:
            break
    
    return current, best_score

def random_local_search(h, best_score, max_time, rng, steps):
