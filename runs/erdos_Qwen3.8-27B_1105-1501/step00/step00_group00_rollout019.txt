def compute_all_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # scaled by dx

def compute_gradient_at_shift(h, k):
    """Compute gradient of the k-th correlation term (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    ell = k - (n - 1)
    grad = np.zeros(n)
    
    if ell >= 0:
        # result = sum_{i=ell}^{n-1} h[i]*(1-h[i-ell])
        for j in range(n):
            if j >= ell:
                grad[j] += (1.0 - h[j - ell])
            if j <= n - 1 - ell:
                grad[j] -= h[j + ell]
    else:
        p = -ell
        for j in range(n):
            if j <= n - 1 - p:
                grad[j] += (1.0 - h[j + p])
            if j >= p:
                grad[j] -= h[j - p]

def repair_feasibility(h):
    """Ensure h is in [0,1] and sum(h) = n/2."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        excess = s - target
        order = np.argsort(-h)
        for idx in order:
            reduce = min(excess, h[idx])
            h[idx] -= reduce
            excess -= reduce
            if excess < 1e-14:
                break

        order = np.argsort(h)
        for idx in order:
            add = min(deficit, 1.0 - h[idx])
            h[idx] += add
            deficit -= add
            if deficit < 1e-14:
                break
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Add/subtract to the coordinate with most room
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for idx in order:
            if diff > 0 and h[idx] < 1.0:
                h[idx] += min(diff, 1.0 - h[idx])
                diff -= min(diff, 1.0 - h[idx])
            elif diff < 0 and h[idx] > 0:
                h[idx] += max(diff, -h[idx])
                diff -= max(diff, -h[idx])
            if abs(diff) < 1e-15:

def pair_line_search(h, i, j, n_samples=20):
    """Find best delta for pair (i, j): h[i]+=delta, h[j]-=delta."""
    hi = h[i]
    hj = h[j]

    delta_max = min(1.0 - hi, hj)

    best_delta = 0.0
