1. Start with initial_h_values, make feasible, compute score
2. Run local refinement with smooth-max gradient descent + pair line descent
3. Try multiple starting points (mutations, binarization, different resolutions)
4. Return best found

For the local refinement:

- Apply water-filling projection after each step
- Use exact score for acceptance
- Periodically do pair-line exact descent

For the pair-line descent:
- Pick pairs (i, j) based on gradient differences
- Do exact 1D optimization along the line h_i + alpha, h_j - alpha
- Accept if exact score improves

import numpy as np
import time
import math

def make_feasible(y, target=None):
    """Project y onto {h in [0,1]^n : sum(h) = target}"""
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    # Replace non-finite values
    y = np.where(np.isfinite(y), y, 0.5)
    
    # Bisection for lambda
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix sum exactly
    diff = target - np.sum(z)
    if abs(diff) > 1e-10:
        # Adjust components with slack
        if diff > 0:
            # Need to increase some components
            slack = 1.0 - z  # room to increase
            order = np.argsort(-slack)  # most slack first
            for idx in order:
                adj = min(diff, slack[idx])
                if adj > 0:
                    z[idx] += adj
                    diff -= adj
                    if abs(diff) < 1e-14:
                        break
        else:
            # Need to decrease some components
            slack = z  # room to decrease
            order = np.argsort(-slack)  # most room first
            for idx in order:
                adj = min(-diff, slack[idx])
                if adj > 0:
                    z[idx] -= adj
                    diff += adj
                    if abs(diff) < 1e-14:
                        break
    
    return z

    """Compute all correlation values * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_scores_and_grad(h, T):
    """Compute smooth-max surrogate value and gradient."""
    n = len(h)
    dx = 2.0 / n
    scores = score_scores(h)
    
    base = np.max(scores)
    exp_vals = np.exp((scores - base) / T)
    S = np.sum(exp_vals)
    
    if S <= 0 or not np.isfinite(S):
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = exp_vals / S
    
    smooth_val = base + T * np.log(S)
    
    g = np.zeros(n)
    center = n - 1
    
    # Center: dC/dh[j] = 1 - 2*h[j]
    g += w[center] * dx * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        wp = w[center + d] * dx
        wn = w[center - d] * dx
        
        if 2 * d <= n - 1:
            g[0:d] += wp * (-h[d:2*d])
            g[d:n-d] += wp * ((1.0 - h[0:n-2*d]) - h[2*d:n])
            g[n-d:n] += wp * (1.0 - h[n-2*d:n-d])
            
            g[0:d] += wn * (1.0 - h[d:2*d])
            g[d:n-d] += wn * ((1.0 - h[2*d:n]) - h[0:n-2*d])
            g[n-d:n] += wn * (-h[0:d])
        else:
            g[0:n-d] += wp * (-h[d:n])
            g[d:n] += wp * (1.0 - h[0:n-d])
            
            g[0:n-d] += wn * (1.0 - h[d:n])
            g[d:n] += wn * (-h[0:n-d])
    
    return scores, smooth_val, g

def pair_line_best(h, i, j, n_evals=21):
    """Exact 1D line search along h_i + alpha, h_j - alpha.
    Returns (best_alpha, best_score, best_h) or None if no improvement."""
    cur_h = h
    hi_i = cur_h[i]
    hi_j = cur_h[j]
    
    # Bounds on alpha
    lo = max(-hi_i, hi_j - 1.0)
    hi = min(1.0 - hi_i, hi_j)
    
    if hi - lo < 1e-12:
        return None
    
    cur_score = score_exact(cur_h)
    
    best_alpha = 0.0
    best_score = cur_score
    best_h = cur_h.copy()
    
    # Grid search
    alphas = np.linspace(lo, hi, n_evals)
    h_copy = cur_h.copy()
    
    for a in alphas:
        h_copy[i] = hi_i + a
        h_copy[j] = hi_j - a
        s = score_exact(h_copy)
        if s < best_score - 1e-14:
            best_score = s
            best_alpha = a
            best_h = h_copy.copy()
    
    # Refine if improvement found
    if best_alpha != 0.0:
        # Golden section in neighborhood
        idx = np.argmin(alphas)
        # Actually let's just do a finer grid around the best
        if 0 < idx < len(alphas) - 1:
            lo2 = alphas[idx-1]
            hi2 = alphas[idx+1]
            fine_alphas = np.linspace(lo2, hi2, 15)
            for a in fine_alphas:
                h_copy[i] = hi_i + a
                h_copy[j] = hi_j - a
                s = score_exact(h_copy)
                if s < best_score - 1e-14:
                    best_score = s
                    best_alpha = a
                    best_h = h_copy.copy()
    
    if best_score < cur_score - 1e-12:
        return (best_alpha, best_score, best_h)
    return None

def resample(h, new_n):
    """Resample h to new resolution."""
    n = len(h)
    if n == new_n:
        return h.copy()

    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, target=new_n / 2.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_scores(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = compute_scores(h)
    base = np.max(scores)
    exp_vals = np.exp((scores - base) / T)
    S = np.sum(exp_vals)
    if S <= 0 or not np.isfinite(S):
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = exp_vals / S
    smooth_val = float(base + T * np.log(S))
    g = np.zeros(n)
    center = n - 1
    g += w[center] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[center + d] * dx
        wn = w[center - d] * dx
        if 2 * d <= n - 1:
            g[0:d] += wp * (-h[d:2*d])
            g[d:n-d] += wp * ((1.0 - h[0:n-2*d]) - h[2*d:n])
            g[n-d:n] += wp * (1.0 - h[n-2*d:n-d])
            g[0:d] += wn * (1.0 - h[d:2*d])
            g[d:n-d] += wn * ((1.0 - h[2*d:n]) - h[0:n-2*d])
            g[n-d:n] += wn * (-h[0:d])
        else:
            g[0:n-d] += wp * (-h[d:n])
            g[d:n] += wp * (1.0 - h[0:n-d])
            g[0:n-d] += wn * (1.0 - h[d:n])
            g[d:n] += wn * (-h[0:n-d])
    return scores, smooth_val, g

def pair_line(h, i, j, n_evals=15):

    t = np.zeros(n, dtype=np.float64)

        c = bin(i).count('1')
        t[i] = 1.0 if c % 2 == 1 else 0.0

        t[remove] = 0.0

        add = idx[:target - total_ones]
        t[add] = 1.0

        pair_budget = min(50, int((deadline - time.monotonic()) * 30))
        # Use gradient to find promising pairs
        try:
            _, _, g = smooth_grad(best_h, 0.05)
            g_sorted = np.argsort(g)
            lows = g_sorted[:8]
            highs = g_sorted[-8:]
        except Exception:
            lows = rng.integers(n_b, size=8)
            highs = rng.integers(n_b, size=8)

        for _ in range(pair_budget):
            if time.monotonic() > deadline - 3:
                break
            i = int(lows[rng.integers(len(lows))])
            j = int(highs[rng.integers(len(highs))])
            if i == j:
                continue
            res = pair_line(best_h, i, j, n_evals=13)
            if res is not None:
                ns, nh = res
                update_best(nh)
                best_h = nh if ns < best_c else best_h

    # Also try some random pairs
    if best_h is not None and time.monotonic() < deadline - 5:
        n_b = len(best_h)
        for _ in range(20):
            if time.monotonic() > deadline - 3:
                break
            i = int(rng.integers(n_b))
            j = int(rng.integers(n_b))
            if i == j:
                continue
            res = pair_line(best_h, i, j, n_evals=11)
            if res is not None:
                ns, nh = res
                update_best(nh)
                if ns < best_c:
                    best_h = nh

    # Final validation
    if best_h is None:
        best_h = make_feasible(np.full(70, 0.5), target=35.0)
        best_c = score_exact(best_h)

    best_h = make_feasible(best_h, target=len(best_h) / 2.0)
    best_c = score_exact(best_h)

    return best_h, best_c, len(best_h)
