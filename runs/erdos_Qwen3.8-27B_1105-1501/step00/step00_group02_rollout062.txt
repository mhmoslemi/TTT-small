def local_search_binary(h_bin, max_iters=10000):
    """Greedy local search: swap pairs of 1s and 0s to reduce max correlation."""
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    current_max = compute_c5(h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        improved = False
        best_delta = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                # Try swap
                h[i] = 0
                h[j] = 1
                new_max = compute_c5(h)
                delta = new_max - current_max
                h[i] = 1
                h[j] = 0
                if delta < best_delta - 1e-12:

                    best_pair = (i, j)

            h[i] = 0
            h[j] = 1
            ones = list(ones); ones.remove(i); ones.append(j)
            zeros = list(zeros); zeros.remove(j); zeros.append(i)
            current_max = compute_c5(h)
            improved = True
        if not improved:
            break
    return h, current_max

def project_constraint(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 2.0

        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)
