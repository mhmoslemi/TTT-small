            if j >= p and j <= n - 1:
                if 0 <= j - p <= n - 1 - p:
                    g -= h[j - p]
            grad[j] = dx * g
    return grad

    J[:, n] = 1.0

    return J

    # Use all constraints
    n_lags = 2 * n - 1
    
    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        return t - corr_values(h)
    
    def ineq_jac(x):
        h = x[:n]
        J = np.zeros((n_lags, n + 1))
        J[:, n] = 1.0
        for k_idx in range(n_lags):
            J[k_idx, :n] = -grad_of_c(h, n, k_idx)
        return J
    
    def eq_fun(x):
        return x[:n].sum() - n / 2.0
    
    def eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    res = minimize(
        fun=lambda x: x[n],
        x0=x0,
        jac=lambda x: np.concatenate([np.zeros(n), [1.0]]),
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    
    return feasible_project(res.x[:n].copy(), target)

Actually, I just realized I should think more carefully about the problem structure. 

The key observation is that C_k(h) = dx * sum h[i] * (1-h[i+k]) (for appropriate ranges). This is a bilinear form. The objective max_k C_k is the maximum of 2n-1 bilinear forms.

For the SLSQP formulation, I'm minimizing t subject to C_k(h) <= t for all k. This is correct but might be slow.

An alternative: directly minimize max_k C_k(h) using a subgradient method. At each step:
1. Find k* that achieves the max
2. Compute gradient of C_{k*}
3. Take a step in the negative subgradient direction
4. Project to feasible set

This is simpler and might work well in practice, even though it's not guaranteed to converge to the global optimum.

def feasible_project(x, target_sum):
    """Project x onto {0 <= x <= 1, sum(x) = target_sum}."""
    n = len(x)
    if np.any(~np.isfinite(x)):
        return np.full(n, target_sum / n)
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def c5_exact(h):
