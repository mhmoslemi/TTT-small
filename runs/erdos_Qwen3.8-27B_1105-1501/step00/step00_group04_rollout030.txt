def smooth_max_grad(h, T):
    n = len(h)
    b = 1.0 - h
    dx = 2.0 / n
    c = np.correlate(h, b, 'full') * dx  # length 2n-1
    # compute stable log-sum-exp
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / T)
    S = np.sum(exp_c)
    weights = exp_c / S  # sum to 1
    smooth_val = cmax + T * np.log(S)
    
    # gradient
    grad = np.zeros(n)
    center = n - 1
    
    # center contribution
    grad += weights[center] * dx * (1 - 2*h)
    
    # positive lags d = 1, ..., n-1
    for d in range(1, n):
        k = center + d
        w = weights[k]
        # dc_k/dh_j = (1-h[j-d]) for j >= d, minus h[j+d] for j <= n-1-d
        grad[:n-d] += -w * dx * h[d:]
        grad[d:] += w * dx * (1 - h[:n-d])
    
    # negative lags d = 1, ..., n-1
    for d in range(1, n):
        k = center - d
        w = weights[k]
        # dc_k/dh_j = (1-h[j+d]) for j <= n-1-d, minus h[j-d] for j >= d
        grad[:n-d] += w * dx * (1 - h[d:])
        grad[d:] += -w * dx * h[:n-d]
    
    return smooth_val, grad

import numpy as np
import time
import math

def project_feasible(y, n):
    """Project y onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    # Bisection for lambda
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix any tiny sum error
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # Adjust the element with most slack
        if diff > 0:
            # Need to add mass; find element farthest from 1
            idx = np.argmin(x)
        else:
            idx = np.argmax(x)
        x[idx] += diff
        x[idx] = np.clip(x[idx], 0.0, 1.0)
    return x

def score_exact(h):
    """Compute the exact C5 score."""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    c = np.correlate(h, b, 'full') * dx
    return float(np.max(c))

def smooth_max_and_grad(h, T):
    """Compute smooth-max surrogate and its gradient."""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    c = np.correlate(h, b, 'full') * dx  # length 2n-1
    
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / T)
    S = np.sum(exp_c)
    if S < 1e-300:
        weights = np.zeros(len(c))
        weights[np.argmax(c)] = 1.0
    else:
        weights = exp_c / S
    smooth_val = cmax + T * np.log(max(S, 1e-300))
    
    grad = np.zeros(n)
    center = n - 1
    
    # Center (shift=0)
    grad += weights[center] * dx * (1.0 - 2.0 * h)
    
    # Positive shifts d=1..n-1 (k = center+d)
    for d in range(1, n):
        w = weights[center + d]
        if w < 1e-15:
            continue
        # dc/dh_j = (1-h[j-d])*1[j>=d] - h[j+d]*1[j<=n-1-d]
        grad[d:] += w * dx * (1.0 - h[:n-d])
        if n - 1 - d >= 0:
            grad[:n-d] -= w * dx * h[d:]
    
    # Negative shifts d=1..n-1 (k = center-d)
    for d in range(1, n):
        w = weights[center - d]
        if w < 1e-15:
            continue
        # dc/dh_j = (1-h[j+d])*1[j<=n-1-d] - h[j-d]*1[j>=d]
        if n - 1 - d >= 0:
            grad[:n-d] += w * dx * (1.0 - h[d:])
        grad[d:] -= w * dx * h[:n-d]
    
    return smooth_val, grad

def pair_line_descent(h, i, j, current_score):
    """Try to improve by moving mass between positions i and j."""
    n = len(h)
    # h_i + alpha, h_j - alpha
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return None, current_score
    
    best_alpha = 0.0
    best_score = current_score
    
    # Sample the line
    n_samples = 17
    for s in range(n_samples + 1):
        alpha = lo + (hi - lo) * s / n_samples
        h[i] += alpha
        h[j] -= alpha
        sc = score_exact(h)
        if sc < best_score - 1e-14:
            best_score = sc
            best_alpha = alpha
    
    # Restore
    h[i] -= best_alpha
    h[j] += best_alpha
    
    if best_alpha == 0.0:
        return None, current_score
    
    h[i] = np.clip(h[i] + best_alpha, 0.0, 1.0)
    h[j] = np.clip(h[j] - best_alpha, 0.0, 1.0)
    return best_alpha, best_score

def local_optimize(h0, deadline, rng, n_max_iter=5000):
    """Projected smooth-gradient descent with line search."""
    n = len(h0)
    h = h0.copy()
    cur_score = score_exact(h)
    best_h = h.copy()
    best_score = cur_score
    
    T = 0.10
    step = 0.05
    T_decay = 0.95
    stagnation = 0
    max_stagnation = 50
    
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        
        smooth_val, grad = smooth_max_and_grad(h, T)
        if not np.all(np.isfinite(grad)):
            T *= 0.5
            continue
        
        # Projected gradient step
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            T *= 0.5
            if T < 1e-6:
                break
            continue
        
        direction = -grad / g_norm
        h_trial = project_feasible(h + step * direction, n)
        sc_trial = score_exact(h_trial)
        
        if sc_trial < cur_score - 1e-14:
            h = h_trial
            cur_score = sc_trial
            stagnation = 0
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            stagnation += 1
            step *= 0.7
            if stagnation > max_stagnation:
                step = 0.02
                T *= 0.8
                stagnation = 0
                if T < 5e-5:
                    break
        
        if it % 10 == 0:
            T = max(T * T_decay, 5e-5)
        
        # Periodic pair-line descent
        if it % 20 == 10 and time.time() < deadline - 5:

import time
import math

def project_feasible(y, n):
    target = n * 0.5
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(y - lam, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = compute_scores(h)
    center = n - 1
    
    cmax = float(np.max(scores))
    exps = np.exp((scores - cmax) / T)
    S = float(np.sum(exps))
    if S < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(scores)] = 1.0
    else:
        w = exps / S
    
    grad = np.zeros(n)
    # Center
    grad += w[center] * dx * (1.0 - 2.0*h)
    
    # Positive shifts
    for d in range(1, n):
        wd = w[center + d]
        if wd < 1e-12:
            continue
        grad[d:] += wd * dx * (1.0 - h[:n-d])
        grad[:n-d] -= wd * dx * h[d:]
    
    # Negative shifts
    for d in range(1, n):
        wd = w[center - d]
        if wd < 1e-12:
            continue
        grad[:n-d] += wd * dx * (1.0 - h[d:])
        grad[d:] -= wd * dx * h[:n-d]
    
    return float(cmax + T * math.log(max(S, 1e-300))), grad

def try_step(h, n, step, grad, deadline, cur_score):
    """Try a gradient step with backtracking."""
    best_score = cur_score

def pair_search(h, n, cur_score, rng, n_pairs=10, deadline=None):

        h[v]

import numpy as np

def project_feasible(y, n):
    target = n * 0.5
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(y - lam, 0.0, 1.0)

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = compute_scores(h)
    center = n - 1

    cmax = float(np.max(scores))
    exps = np.exp((scores - cmax) / T)
    S = float(np.sum(exps))
    if S < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(scores))] = 1.0
    else:
        w = exps / S

    grad = np.zeros(n)
    grad += w[center] * dx * (1.0 - 2.0 * h)

    for d in range(1, n):
        wd = w[center + d]
        if wd < 1e-14:
            continue
        grad[d:] += wd * dx * (1.0 - h[:n - d])
        grad[:n - d] -= wd * dx * h[d:]

    for d in range(1, n):
        wd = w[center - d]
        if wd < 1e-14:
            continue
        grad[:n - d] += wd * dx * (1.0 - h[d:])
        grad[d:] -= wd * dx * h[:n - d]

    return grad
