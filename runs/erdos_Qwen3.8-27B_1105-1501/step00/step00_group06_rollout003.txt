import numpy as np
import time

def project_box_mass(v, n):
    target = n / 2.0
    v = np.clip(v.astype(float), 0.0, 1.0)
    s = np.sum(v)
    if abs(s - target) < 1e-10:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) / 2, 0, 1)
    resid = target - np.sum(result)
    if abs(resid) > 1e-10:
        free = np.where((result > 1e-6) & (result < 1 - 1e-6))[0]
        if len(free) > 0:
            result[free] += resid / len(free)
            result = np.clip(result, 0, 1)
    return result

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr * dx

def feasible_range(h, d):
    lo, hi = 0.0, 0.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        hi = np.min((1.0 - h[pos]) / d[pos])
    if np.any(neg):
        lo = np.max(-h[neg] / d[neg])
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    eps = 1e-10 * max(1.0, abs(hi - lo))
    return lo + eps, hi - eps

def line_search(h, d, n_pts=30):
    lo, hi = feasible_range(h, d)
    if hi - lo < 1e-12:
        return h, compute_c5(h)[0]
    cur = compute_c5(h)[0]
    best_v, best_l = cur, 0.0
    lams = np.linspace(lo, hi, n_pts)
    for lam in lams:
        c = h + lam * d
        if np.any(c < -1e-8) or np.any(c > 1 + 1e-8):
            continue
        v = compute_c5(c)[0]
        if v < best_v:
            best_v, best_l = v, lam
    if best_l != 0.0:
        # refine
        w = (hi - lo) / n_pts
        lo2 = max(lo, best_l - w)
        hi2 = min(hi, best_l + w)
        for lam in np.linspace(lo2, hi2, 15):
            c = h + lam * d
            if np.any(c < -1e-8) or np.any(c > 1 + 1e-8):
                continue
            v = compute_c5(c)[0]
            if v < best_v:
                best_v, best_l = v, lam
    return h + best_l * d, best_v

def zero_sum(d):
    d = d.copy()
    d -= d.mean()
    return d

    phase3_end = t0 + budget_s * 0.8
