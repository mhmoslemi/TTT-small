def project_feasible(y, target):
    """Project y onto {g in [0,1]^n : sum(g) = target}"""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    # Fix small discrepancy
    diff = target - z.sum()
    if abs(diff) > 1e-14:
        # Find indices where we can adjust
        if diff > 0:

            idx = np.where(z < 1.0 - 1e-12)[0]
        else:
            idx = np.where(z > 1e-12)[0]

    return float(np.max(score_scores(h)))

    n = len(h)

    h = h.copy()
    n = len(h)
    cur_c = score_exact(h)

import numpy as np

def project_feasible(y, target):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = z < 1.0 - 1e-12
        else:
            mask = z > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            z[mask] += diff / cnt
            z = np.clip(z, 0.0, 1.0)
    return z

def score_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def score_exact(h):
    return float(np.max(score_scores(h)))

def smooth_obj_grad(h, T):
    c = score_scores(h)
    n = len(h)
    dx = 2.0 / n
    base = c.max()
    exp_c = np.exp((c - base) / T)
    S = exp_c.sum()
    if S <= 0 or not np.isfinite(S):
        weights = np.zeros_like(c)
        weights[c.argmax()] = 1.0
    else:
        weights = exp_c / S
    smooth_val = float(base + T * math.log(S))
    g = np.zeros(n)
    center = n - 1
    for k in range(len(c)):
        wk = weights[k]
        if abs(wk) < 1e-14:
            continue
        d = k - center
        if d == 0:
            g += wk * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            g[0:m] += wk * dx * (1.0 - h[d:d+m])
            g[d:n] += wk * dx * (-h[0:m])
        else:
            m_ = -d
            m = n - m_
            g[m_:n] += wk * dx * (1.0 - h[0:m])

    hi_a = min(1.0 - h[i], h[j])

        return h, score_exact(h)
    best_alpha = 0.0

        h[i] = orig_i + alpha
        h[j] = orig_j - alpha

            best_alpha = alpha

def resample_to_n(h, new_n):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)

    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    target = new_n / 2.0
    return project_feasible(h_new, target)
