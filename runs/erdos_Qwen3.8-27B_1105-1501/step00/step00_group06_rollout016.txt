    for _ in range(100):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            break
        if s > target_sum:
            # Need to decrease
            # Decrease values that are > 0, proportional to headroom
            mask = h > 0
            excess = s - target_sum
            # Reduce proportionally
            h[mask] -= excess * h[mask] / h[mask].sum()
            h = np.clip(h, 0, 1)
        else:
            # Need to increase
            mask = h < 1
            deficit = target_sum - s
            # Increase proportionally

            h = np.clip(h, 0, 1)

import numpy as np

def compute_c5_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_c5(h):
    return compute_c5_values(h).max()

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_soft_max(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    c_max = c.max()
    w = np.exp(tau * c - tau * c_max)
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:n] += wk * (1.0 - h[0:m])
                grad[0:m] -= wk * h[d:n]
        else:
            e = -d
            m = n - e
            if m > 0:
                grad[0:m] += wk * (1.0 - h[e:n])
                grad[e:n] -= wk * h[0:m]
    return grad, c, w

def soft_max_gradient_descent(h, budget_s, tau0=8.0, tau1=60.0):
    n = len(h)

    best_h = h.copy()
    start = time.time()
    lr = 0.02
    stuck = 0
    max_stuck = 50
    while time.time() - start < budget_s:
        elapsed_frac = (time.time() - start) / max(budget_s, 1e-9)
        tau = tau0 + (tau1 - tau0) * min(elapsed_frac, 1.0)
        grad, c, w = compute_grad_soft_max(h, tau)

        h_trial = h - lr * grad
