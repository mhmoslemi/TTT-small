import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _obj_soft(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / alpha

def _sum_constraint(h, target_sum):
    return np.sum(h) - target_sum

def _sum_constraint_jac(h, target_sum):
    return np.ones(len(h))

def _project_feasible(h, target_sum):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(mask))
        if nf == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def _do_optimize(h_start, alpha, n, target_sum, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_constraint_jac,
             'args': (target_sum,)}]
    try:
        res = minimize(_obj_soft, h_start, args=(alpha, n),
                     method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            h_new = _project_feasible(res.x, target_sum)
            c5_new = _compute_c5(h_new, n)
            return h_new, c5_new
    except Exception:
        pass
    return None, None

def _do_optimize_lh(h_start, alpha, n, target_sum, maxiter=2000):

        pen = penalty * (np.sum(p) - target_sum) ** 2
        return base + pen

            c5_new = _compute_c5(h_new, n)

        pass
    return None, None
