def evaluate_c5(h):
    """Compute the C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    return np.max(c)

def project_onto_constraint(h, target_sum):
    """Project h onto {0≤x≤1, sum(x)=target_sum} using water filling."""
    h = np.clip(h, 0, 1)

def lp_step(h, n, alpha_active=1e-4, max_step=0.1):
    c_vec = overlap_vec(h)
    c_max = np.max(c_vec)
    active = c_vec >= c_max - alpha_active
    active_indices = np.where(active)[0]
    
    # Build LP: variables are d[0..n-1] and t
    # Minimize t
    # Subject to: grad_f_k · d + c_vec[k] ≤ t  for k in active
    #             -max_step ≤ d_i ≤ max_step
    #             sum(d) = 0
    
    m = n  # number of variables d
    # Number of inequality constraints: len(active)
    # Row for each active k: grad_f_k[0]*d[0] + ... + grad_f_k[n-1]*d[n-1] - t ≤ c_vec[k] - c_vec[k] = 0
    # Wait, let me reformulate:
    # We want: c_vec[k] + grad_f_k · d ≤ t  =>  grad_f_k · d - t ≤ -c_vec[k]
    # In standard form: A_ub @ [d; t] ≤ b_ub
    # A_ub row for active k: [grad_f_k, -1], b_ub = -c_vec[k]
    
    # Equality: sum(d) = 0 => [1,1,...,1, 0] @ [d;t] = 0
    # Bounds: -max_step ≤ d_i ≤ max_step, t ≥ 0 (or t unbounded below)
    
    n_act = len(active_indices)
    A_ub = np.zeros((n_act, n+1))
    b_ub = np.zeros(n_act)
    for i, idx in enumerate(active_indices):
        k = idx - (n-1)
        g = grad_fk(h, k, n)
        A_ub[i, :n] = g
        A_ub[i, n] = -1
        b_ub[i] = -c_vec[idx]
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-max_step, max_step)]*n + [(None, None)]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # minimize t
    
    result = optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq.reshape(-1,1), 
                               b_eq=b_eq.reshape(-1), bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n]
        return d
    return None

def grad_fk(h, k, n):
    """Gradient of f_k with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    j_lo = max(0, -k)
    j_hi = min(n-1, n-1-k)
    if j_lo > j_hi:
        return grad
    j = np.arange(j_lo, j_hi+1)
    grad[j] += (1.0 - h[j+k])
    j2 = j + k
    grad[j2] -= h[j]
    return dx * grad

def full_grad(h, n):
