                    add = min(room, diff)
                    z[idxs[i]] += add
                    diff -= add
                    if abs(diff) < 1e-12:

                    sub = min(avail, diff)
                    z[idxs[i]] -= sub
                    diff -= sub
                    if abs(diff) < 1e-12:
                        break

def pair_line_search(h, i, j, dx, max_evals=30):
    """1D line search along direction e_i - e_j. Returns optimal t and new h."""

    h_orig = h[i]
    j_orig = h[j]
    # t range: h[i] + t in [0,1] => t in [-h[i], 1-h[i]]
    #           h[j] - t in [0,1] => t in [h[j]-1, h[j]]
    t_lo = max(-h_orig, j_orig - 1.0)
    t_hi = min(1.0 - h_orig, j_orig)
    if t_hi - t_lo < 1e-12:
        return h, 0.0
    
    best_t = 0.0
    best_c = score_exact(h)
    
    # Grid search
    ts = np.linspace(t_lo, t_hi, max_evals)
    h_temp = h.copy()
    evals = 0
    for t in ts:
        h_temp[i] = h_orig + t
        h_temp[j] = j_orig - t
        c_val = score_exact(h_temp)
        evals += 1
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_t = t
    
    # Refine around best_t with finer grid
    if best_t != 0.0 and evals < max_evals * 3:
        margin = (t_hi - t_lo) / max_evals * 2
        t_lo2 = max(t_lo, best_t - margin)
        t_hi2 = min(t_hi, best_t + margin)
        ts2 = np.linspace(t_lo2, t_hi2, 20)
        for t in ts2:
            h_temp[i] = h_orig + t
            h_temp[j] = j_orig - t
            c_val = score_exact(h_temp)
            evals += 1
            if c_val < best_c - 1e-12:
                best_c = c_val
                best_t = t
    
    if best_t != 0.0 and best_c < score_exact(h) - 1e-12:
        h_new = h.copy()
        h_new[i] = h_orig + best_t
        h_new[j] = j_orig - best_t
        return h_new, best_c
    return h, best_c

def make_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo+hi)*0.5, 0.0, 1.0)

    if abs(diff) > 1e-12:
