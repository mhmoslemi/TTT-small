            best[j] += d
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
    return best, best_score

def _multi_cycle_search(h, deadline, rng, max_iters=30000):
    n = len(h)
    best = h.copy()
    best_score = _score(best)
    for _ in range(max_iters):
        if time.perf_counter() > deadline:
            break
        k = int(rng.integers(2, min(4, n // 2) + 1))
        idx = rng.permutation(n)[:k]
        u = np.zeros(n)
        for i in range(k):
            u[idx[i]] = 1.0
        for i in range(k):
            u[idx[(i + 1) % k]] = -1.0
        u = u / (np.linalg.norm(u) + 1e-15)
        for sign in [1.0, -1.0]:
            for scale in [0.15, 0.08, 0.03]:
                cand = best + sign * scale * u
                proj = _project_box_sum(cand, n / 2.0)
                s = _score(proj)
                if s < best_score - 1e-11:
                    best = proj
                    best_score = s
                    break
    return best, best_score

def _subgradient_loop(h, deadline, initial_step=0.02):
    n = len(h)
    best = h.copy()
    best_score = _score(best)
    step = initial_step
    for _ in range(50000):
        if time.perf_counter() > deadline:
            break
        c = np.correlate(best, 1.0 - best, mode='full')
        k_active = int(np.argmax(c))
        lag = k_active - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                grad[i] += (1.0 - best[j])
            j2 = i - lag
            if 0 <= j2 < n:
                grad[i] += (-best[j2])
        grad_centered = grad - grad.mean()
        cand = best - step * grad_centered
        proj = _project_box_sum(cand, n / 2.0)
        s = _score(proj)
        if s < best_score - 1e-12:
            best = proj
            best_score = s
        else:
            step *= 0.95
    return best, best_score

def _initial_construction(n, rng):
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    h = np.zeros(n)
    for i in range(n):
        val = (i * phi + rng.uniform(0, 0.1)) % 1.0
        h[i] = 1.0 if val < 0.5 else 0.0
    return _project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    clock = time.perf_counter()
    budget_s = float(budget_s)
    deadline = clock + max(2.0, budget_s - 3.0)
    
    if budget_s < 5.0:
        deadline = clock + max(0.5, budget_s - 0.5)
    
    rng = np.random.default_rng(seed)
    best_h = None
    best_score = float('inf')
    
    try:
        init = np.array(initial_h_values, dtype=np.float64).flatten().copy()
        if len(init) >= 2 and np.all(np.isfinite(init)):
            n_init = len(init)
            init = _project_box_sum(init, n_init / 2.0)
            sc = _score(init)
            if np.isfinite(sc):
                best_h = init
                best_score = sc
    except Exception:
        pass
    
    if best_h is None:
        n = 63
        best_h = _initial_construction(n, rng)
        best_score = _score(best_h)
    
    n_base = len(best_h)
    
    # Build list of lengths to try (prefer smaller)
    length_candidates = []
    for d in range(-5, 6):
        m = n_base + d
        if 8 <= m <= 300:
            length_candidates.append(m)
    for m in [16, 32, 48, 64, 80, 96, 128]:
        if m not in length_candidates and 8 <= m <= 300:
            length_candidates.append(m)
    length_candidates = sorted(set(length_candidates))
    
    # Process lengths in order of starting score (best first), then prefer smaller n
    candidates = []
    for m in length_candidates:
        if time.perf_counter() > deadline:
            break
        h_m = _resample(best_h, m)
        sc_m = _score(h_m)
        candidates.append((sc_m, m, h_m))
    candidates.sort(key=lambda x: (x[0], x[1]))
    
    # Optimize each candidate
    for sc_init, m, h_m in candidates:
        if time.perf_counter() > deadline:
            break
        # Pair sweep
        h_opt, sc_opt = _pair_sweep(h_m, deadline)
        if sc_opt < best_score - 1e-11:
            best_h = h_opt
            best_score = sc_opt
        
        # Random pair search
        if time.perf_counter() < deadline - 5.0:
            h_r, sc_r = _random_pair_search(best_h, min(deadline, clock + 60.0), rng)
            if sc_r < best_score - 1e-11:
                best_score = sc_r
                best_h = h_r
                # Quick follow-up pair sweep
                if time.perf_counter() < deadline - 3.0:
                    best_h, best_score = _pair_sweep(best_h, deadline)
    
    # Multi-cycle search on best
    if time.perf_counter() < deadline - 10.0:
        h_c, sc_c = _multi_cycle_search(best_h, min(deadline, clock + 40.0), rng)
        if sc_c < best_score - 1e-11:
            best_h = h_c
            best_score = sc_c
            if time.perf_counter() < deadline - 5.0:
                best_h, best_score = _pair_sweep(best_h, deadline)
    
    # Subgradient (if time permits)
    if time.perf_counter() < deadline - 10.0:
        h_s, sc_s = _subgradient_loop(best_h, min(deadline, clock + 30.0))
        if sc_s < best_score - 1e-11:
            best_h = h_s
            best_score = sc_s
            if time.perf_counter() < deadline - 3.0:
                best_h, best_score = _pair_sweep(best_h, deadline)
    
    # Final pair sweep
    if time.perf_counter() < deadline - 2.0:
        best_h, best_score = _pair_sweep(best_h, deadline)
    
    # Final validation and cleanup
    final = _project_box_sum(best_h, len(best_h) / 2.0)
    final_score = _score(final)
    
    # Ensure feasibility
    if not (np.all(final >= -1e-10) and np.all(final <= 1.0 + 1e-10)):
        final = np.clip(final, 0.0, 1.0)
        final = _project_box_sum(final, len(final) / 2.0)
        final_score = _score(final)
    
    return final, final_score, len(final)
