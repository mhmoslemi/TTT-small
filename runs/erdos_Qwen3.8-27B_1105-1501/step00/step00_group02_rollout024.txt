def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    cf = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cf) * dx)

def pair_transfer(h, i, j, t):
    """Transfer t from i to j. Returns new h."""
    new_h = h.copy()
    new_h[i] -= t
    new_h[j] += t
    return new_h

def optimize_pair(h, i, j, dx):
    """Find best t for transferring from i to j."""
    best_t = 0.0
    best_c = c5_value(h)
    
    # Feasible range for t:
    # h[i] - t >= 0 -> t <= h[i]
    # h[j] + t <= 1 -> t <= 1 - h[j]
    t_max = min(h[i], 1.0 - h[j])
    t_min = -min(h[j], 1.0 - h[i])  # negative t means transfer from j to i
    
    if t_max <= t_min:
        return h, best_c
    
    # Sample t values
    n_samples = 40
    ts = np.linspace(t_min, t_max, n_samples)
    
    for t in ts:
        if abs(t) < 1e-12:

import numpy as np
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    cf = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cf) * dx)

def optimize_pair(h, i, j, n_samples):
    hi0 = h[i]
    hj0 = h[j]
    t_max = min(hi0, 1.0 - hj0)
    t_min = -min(hj0, 1.0 - hi0)
    if t_max - t_min < 1e-13:
        return
    best_t = 0.0
    best_c = c5_value(h)
    for t in np.linspace(t_min, t_max, n_samples):
        if abs(t) < 1e-14:
            continue
        h[i] = hi0 - t
        h[j] = hj0 + t
        c = c5_value(h)
        if c < best_c:
            best_c = c
            best_t = t
    h[i] = hi0 - best_t
    h[j] = hj0 + best_t

def do_sweep(h, pairs, rng, n_samples):
    order = np.random.permutation(len(pairs))
    for idx in order:
        i, j = pairs[idx]
        optimize_pair(h, i, j, n_samples)

def fix_sum(h, target):
    n = len(h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def quantize_h(h, bits=30):

    Q = 1 << bits

    q = np.round(h * Q).astype(np.int64)
    q = np.clip(q, 0, Q).astype(np.int64)
    diff = int(target - np.sum(q))
    if diff > 0:
        fracs = h * Q - q.astype(np.float64)
        cand = np.where(q < Q)[0]
        if len(cand) >= diff:
            order = np.argsort(fracs[cand])[::-1][:diff]
            q[cand[order]] += 1
        else:
            for idx in cand:
                if diff <= 0:
                    break
                q[idx] += 1
                diff -= 1
    elif diff < 0:
        fracs = h * Q - q.astype(np.float64)

            q[cand[order]] -= 1
        else:
            for idx in cand:

                q[idx] -= 1
                diff += 1
    return q.astype(np.float64) / Q
