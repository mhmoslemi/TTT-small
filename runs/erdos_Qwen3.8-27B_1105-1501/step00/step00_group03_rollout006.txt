    dx = 2.0 / n

def project_feasible(z, n):
    """Project z onto {0<=x<=1, sum(x)=n/2}"""
    h = np.clip(z, 0.0, 1.0)
    target = n / 2.0
    # Bisection on shift
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Exact repair
    e = target - np.sum(h)
    if e > 1e-14:
        # Need to add mass
        avail = 1.0 - h
        idx = np.argsort(-avail)
        for i in idx:
            d = min(e, avail[i])
            h[i] += d
            e -= d
            if e <= 1e-14:
                break
    elif e < -1e-14:

        avail = h
        idx = np.argsort(avail)

            d = min(-e, avail[i])
            h[i] -= d
            e += d

    e = target - np.sum(h)

def slsqp_minimax(h0, max_iter=30, stop_check=None):
    """Direct minimax SLSQP optimization"""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    def objective(x):
        return x[-1]  # theta
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def constraints_func(x):
        h = x[:-1]
        theta = x[-1]
        vals = np.correlate(h, 1.0 - h, mode="full") * dx
        # theta >= each val, i.e., theta - val >= 0
        return theta - vals
    
    def constraints_grad(x):
        h = x[:-1]
        n_lags = 2 * n - 1
        jac = np.zeros((n_lags, n + 1))
        # For each lag k: constraint is theta - dx * F_k >= 0
        # d/dtheta = 1, d/dh = -dx * G_k
        for k_idx in range(n_lags):
            k = k_idx - (n - 1)
            jac[k_idx, -1] = 1.0  # d theta / d theta
            start = max(0, -k)
            end = min(n, n - k)
            for i in range(start, end):
                j = i + k
                # dF_k/dh_i += (1 - h_j), dF_k/dh_j += (-h_i)
                jac[k_idx, i] -= dx * (1.0 - h[j])
                if j != i:
                    jac[k_idx, j] += dx * h[i]
                else:
                    # k=0 case: already added (1-h_i) above, need to add -h_i
                    jac[k_idx, i] += dx * h[i]
        return jac
    
    def eq_constraint(x):
        return np.sum(x[:-1]) - target
    
    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    theta0 = max(official_score(h0) + 1e-6, 0.01)
    x0 = np.concatenate([h0.copy(), [theta0]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    cons = [
        {'type': 'ineq', 'fun': constraints_func, 'jac': constraints_grad},
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_grad},
    ]
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-9})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = project_feasible(res.x[:-1], n)
            return h_out
    except Exception:
        pass
    return h0

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(z, n):
    z = np.asarray(z, dtype=float).ravel()
    if len(z) != n:
        z = np.resize(z, n)
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    e = target - np.sum(h)
    if abs(e) > 1e-13:
        if e > 0:
            avail = 1.0 - h
            idx = np.argsort(-avail)
            for i in idx:
                d = min(e, avail[i])
                h[i] += d
                e -= d
                if abs(e) < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                d = min(-e, h[i])
                h[i] -= d
                e += d
                if abs(e) < 1e-14:
                    break
    e = target - np.sum(h)
    if abs(e) > 1e-13:
        if e > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += e
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_h(h_src, n_target):
    m = len(h_src)
    if m == 0:
        return np.full(n_target, 0.5)
    xs = (np.arange(m) + 0.5) * (2.0 / m)
    xt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_new = np.interp(xt, xs, h_src)
    return project_feasible(h_new, n_target)


def generate_binary_quota(n, rng):
    h = np.zeros(n)
    ones = n // 2
    positions = rng.permutation(n)[:ones]
    h[positions] = 1.0
    remainder = n / 2.0 - ones
    if remainder > 0:
        zpos = np.where(h == 0)[0]
        if len(zpos) > 0:
            h[zpos[0]] = remainder
    return h


def generate_smooth_pattern(n, rng, freq_lo=1.0, freq_hi=5.0):
    t = np.linspace(0, 1, n)
    freq = rng.uniform(freq_lo, freq_hi)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.15, 0.45)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.05, n)
    return project_feasible(h, n)


def compute_grad_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    h_arr = h
    for i in range(start, end):
        j = i + k
        grad[i] += (1.0 - h_arr[j])
        grad[j] += (-h_arr[i])
    return grad


def local_line_descent(h, max_time=30.0, stop_flag=None):
    n = len(h)
    dx = 2.0 / n
    best = official_score(h)
    h_best = h.copy()
    t0 = time.perf_counter()
    rng = np.random.default_rng(12345 + n)

    for iteration in range(100):
        if stop_flag is not None and stop_flag():
            break
        if time.perf_counter() - t0 > max_time:
            break

        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        vals = corr * dx
        max_val = vals.max()
        thresh = max_val - 0.08 * max_val
        active = np.where(vals >= thresh)[0]

        directions = []

        # Direction 1: negative gradient of active set
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            grad -= compute_grad_lag(h_best, k)
        grad -= grad.mean()
        norm = np.max(np.abs(grad))
        if norm > 1e-12:
            directions.append(grad / norm)

        # Direction 2: negative gradient of single worst lag
        worst = int(np.argmax(vals))
        k_worst = worst - (n - 1)
        grad2 = -compute_grad_lag(h_best, k_worst)
        grad2 -= grad2.mean()
        norm2 = np.max(np.abs(grad2))
        if norm2 > 1e-12:
            directions.append(grad2 / norm2)

        # Direction 3: random
        grad3 = rng.normal(0, 1, n)
        grad3 -= grad3.mean()
        norm3 = np.max(np.abs(grad3))
        if norm3 > 1e-12:
            directions.append(grad3 / norm3)

        # Direction 4: pair swap inspired
        if n > 2:
            p = int(np.argmax(h_best))
            q = int(np.argmin(h_best))
            if p != q and h_best[p] - h_best[q] > 0.1:
                d = np.zeros(n)
                d[p] = -1.0
                d[q] = 1.0
                d -= d.mean()
                norm4 = np.max(np.abs(d))
                if norm4 > 1e-12:
                    directions.append(d / norm4)

        improved = False
        for d in directions:
            if improved:
                break
            max_alpha = 0.4 / np.max(np.abs(d))
            alphas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0]
            for a in alphas:
                alpha = a * max_alpha
                for sign in [1, -1]:
                    h_trial = project_feasible(h_best + sign * alpha * d, n)
                    s = official_score(h_trial)
                    if np.isfinite(s) and s < best - 1e-10:
                        best = s
                        h_best = h_trial
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try smaller steps
            for d in directions[:2]:
                max_alpha = 0.1 / np.max(np.abs(d))
                alphas_small = [0.001, 0.003, 0.01, 0.03]
                for a in alphas_small:
                    alpha = a * max_alpha
                    for sign in [1, -1]:
                        h_trial = project_feasible(h_best + sign * alpha * d, n)
                        s = official_score(h_trial)
                        if np.isfinite(s) and s < best - 1e-11:
                            best = s
                            h_best = h_trial
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return h_best, best


def slsqp_minimax(h0, max_iter=35):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    n_lags = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_constraints(x):
        h = x[:-1]
        theta = x[-1]
        vals = np.correlate(h, 1.0 - h, mode="full") * dx
        return theta - vals

    def ineq_jac(x):
        h = x[:-1]
        jac = np.zeros((n_lags, n + 1))
        for k_idx in range(n_lags):
            k = k_idx - (n - 1)
            jac[k_idx, -1] = 1.0
            start = max(0, -k)
            end = min(n, n - k)
            for i in range(start, end):
                j = i + k
                jac[k_idx, i] -= dx * (1.0 - h[j])
                jac[k_idx, j] += dx * h[i]
        return jac

    def eq_constraint(x):
        return np.sum(x[:-1]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    score0 = official_score(h0)
    theta0 = max(score0 + 1e-5, 0.05)
    x0 = np.concatenate([h0.copy(), [theta0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_constraints, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac},
    ]

    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-8})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = project_feasible(res.x[:-1], n)
            s_out = official_score(h_out)
            if np.isfinite(s_out) and s_out < score0 + 0.01:
                return h_out
    except Exception:
        pass
    return h0
