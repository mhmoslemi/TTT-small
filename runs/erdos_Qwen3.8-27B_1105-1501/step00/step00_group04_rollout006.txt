import numpy as np
import time
from scipy.optimize import minimize_scalar

def score_exact(h):
    """Compute exact C5 value."""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(scores))

def score_vector(h):
    """Compute all lag scores (scaled by dx)."""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_grad(scores, T):
    """Compute soft-max value and normalized weights from scores vector."""
    cmax = np.max(scores)
    ex = np.exp((scores - cmax) / T)
    S = np.sum(ex)
    if S < 1e-300:
        # Fallback: uniform over near-max
        mask = scores >= cmax - T
        weights = np.zeros_like(scores)
        mask_indices = np.where(mask)[0]
        weights[mask_indices] = 1.0 / len(mask_indices)
        return cmax, weights
    weights = ex / S
    smooth_val = cmax + T * np.log(S)
    return smooth_val, weights

def projected_grad(h, T):
    """Compute gradient of smooth-max surrogate."""
    n = len(h)
    dx = 2.0 / n
    scores = score_vector(h)
    center = n - 1
    smooth_val, weights = smooth_max_grad(scores, T)
    
    grad = weights[center] * dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = weights[center + k]
        wn = weights[center - k]
        if abs(wp) > 1e-15:
            grad[k:] += wp * dx * (1.0 - h[:n-k])
            grad[:n-k] += wp * dx * (-h[k:])
        if abs(wn) > 1e-15:
            grad[:n-k] += wn * dx * (1.0 - h[k:])
            grad[k:] += wn * dx * (-h[:n-k])
    
    return smooth_val, grad

def project_feasible(y, target):
    """Project y onto {x in [0,1]^n : sum(x) = target}."""
    n = len(y)
    # Bisection on lambda: sum(clip(y - lambda, 0, 1)) = target
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    z = np.clip(y - lam, 0.0, 1.0)
    # Fix tiny sum error
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        # Adjust the element with most slack
        if diff > 0:
            # Need to increase some elements
            slack = 1.0 - z
            order = np.argsort(-slack)
        else:
            # Need to decrease some elements
            slack = z
            order = np.argsort(slack)
        for idx in order:
            if diff > 0:
                adj = min(diff, 1.0 - z[idx])
                z[idx] += adj
                diff -= adj
            else:
                adj = max(diff, -z[idx])
                z[idx] += adj
                diff -= adj
            if abs(diff) < 1e-14:
                break
    return z

def random_feasible(n, rng):
    """Generate random feasible vector."""
    target = n / 2.0
    y = rng.random(n)
    return project_feasible(y, target)

def mutate(h, sigma, rng):
    """Perturb h with Gaussian noise, then project."""
    n = len(h)
    target = n / 2.0
    y = h + sigma * rng.standard_normal(n)
    return project_feasible(y, target)

def binarize(h, rng=None):
    """Create binary vector with sum = n/2 (requires even n)."""
    n = len(h)
    target = n // 2
    if n % 2 != 0:
        # For odd n, target isn't integer; use rounding
        target = round(n / 2.0)
    indices = np.argsort(h, kind='stable')
    h_bin = np.zeros(n)
    top = target
    h_bin[indices[n - top:]] = 1.0
    return h_bin

def resample(h, new_n):
    """Resample h to new resolution."""
    n_old = len(h)
    target = new_n / 2.0
    # Linear interpolation
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    y_new = np.interp(x_new, x_old, h)
    return project_feasible(y_new, target)

def line_search_pair(h, i, j, deadline):
    """Exact line search on pair (i, j). Returns (new_h, new_score) or (h, score)."""
    n = len(h)
    target = n / 2.0
    hi_val = h[j]
    lo_val = h[i]
    
    # h[i] = lo_val + alpha, h[j] = hi_val - alpha
    # Bounds: 0 <= lo_val + alpha <= 1, 0 <= hi_val - alpha <= 1
    a_lo = max(-lo_val, hi_val - 1.0)
    a_hi = min(1.0 - lo_val, hi_val)
    
    if a_hi - a_lo < 1e-12:
        return h, score_exact(h)
    
    # Grid search
    best_alpha = 0.0
    best_score = score_exact(h)
    n_grid = 50
    alphas = np.linspace(a_lo, a_hi, n_grid)
    
    test_h = h.copy()
    for a in alphas:
        test_h[i] = lo_val + a
        test_h[j] = hi_val - a
        if time.time() > deadline:
            break
        s = score_exact(test_h)
        if s < best_score - 1e-14:
            best_score = s
            best_alpha = a
    
    # Refine with golden section if improvement found
    if best_alpha != 0.0 and time.time() < deadline - 2:
        # Narrow search around best_alpha
        step = (a_hi - a_lo) / n_grid
        for _ in range(30):
            a1 = best_alpha - step * 0.618
            a2 = best_alpha + step * 0.618
            a1 = max(a_lo, min(a_hi, a1))
            a2 = max(a_lo, min(a_hi, a2))
            s1 = _eval_pair(h, i, j, lo_val, hi_val, a1)
            s2 = _eval_pair(h, i, j, lo_val, hi_val, a2)
            if s1 < s2:
                a_hi = a2
                best_alpha = min(best_alpha, a1)
            else:
                a_lo = a1
                best_alpha = max(best_alpha, a2)
            step = (a_hi - a_lo)
            if step < 1e-8:
                break
            if time.time() > deadline:
                break
    
    if best_alpha != 0.0:
        new_h = h.copy()
        new_h[i] = lo_val + best_alpha
        new_h[j] = hi_val - best_alpha
        new_score = score_exact(new_h)
        if new_score < score_exact(h) - 1e-14:
            return new_h, new_score
    
    return h, None

def _eval_pair(h, i, j, lo_val, hi_val, alpha):
    """Evaluate score with modified pair."""
    test = h.copy()
    test[i] = lo_val + alpha
    test[j] = hi_val - alpha
    return score_exact(test)

def local_refine(h_init, deadline, rng, T_start=0.1, T_end=1e-4, n_phases=8):
    """Main local optimization loop."""
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = score_exact(best_h)
    cur = h_init.copy()
    
    # Temperature schedule (log-linear)
    log_Ts = np.linspace(np.log(T_start), np.log(T_end), n_phases)
    
    phase_iters = max(20, 8000 // n)  # iterations per phase
    
    for phase in range(n_phases):
        T = np.exp(log_Ts[phase])
        for it in range(phase_iters):
            if time.time() > deadline - 5:
                return best_h, best_score
            if time.time() > deadline:
                return best_h, best_score
            
            # Compute gradient
            try:
                _, grad = projected_grad(cur, T)
            except:
                break
            
            if not np.all(np.isfinite(grad)):
                break
            
            # Multiplier line search
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-14:
                break
            
            g_dir = -grad / g_norm  # descent direction
            step_sizes = np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
            
            improved = False
            for step in step_sizes:
                trial = project_feasible(cur + step * g_dir, target)
                s = score_exact(trial)
                if s < best_score - 1e-12:
                    best_h = trial.copy()
                    best_score = s
                    cur = trial
                    improved = True
                    break
                elif s < score_exact(cur) - 1e-12:
                    # Accept local improvement even if not global best
                    cur = trial
                    improved = True
                    break
            
            if not improved:
                # Try pair line search
                if it % 5 == 0:
                    # Pick pairs based on gradient extremes
                    sorted_idx = np.argsort(grad)
                    n_pairs = min(6, n // 2)
                    for pi in range(n_pairs):
                        if time.time() > deadline - 5:
                            break
                        i = sorted_idx[pi]
                        j = sorted_idx[n - 1 - pi]
                        if i == j:
                            continue
                        new_h, new_s = line_search_pair(cur, i, j, min(deadline - 3, time.time() + 5))
                        if new_s is not None and new_s < best_score - 1e-12:
                            best_h = new_h.copy()
                            best_score = new_s
                            cur = new_h
                else:
                    # Small random perturbation
                    sigma = 0.01 * (1.0 + it / phase_iters)
                    trial = mutate(cur, sigma, rng)
                    s = score_exact(trial)
                    if s < best_score - 1e-12:
                        best_h = trial.copy()
                        best_score = s
                        cur = trial
                    elif s < score_exact(cur) - 1e-12:
                        cur = trial
    
    return best_h, best_score

import numpy as np
import time

def score_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def score_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_weights(scores, T):
    cmax = np.max(scores)
    ex = np.exp((scores - cmax) / T)
    S = np.sum(ex)
    if S < 1e-300:
        mask = scores >= cmax - 1e-10
        idx = np.where(mask)[0]
        w = np.zeros_like(scores)
        w[idx] = 1.0 / len(idx)
        return cmax, w
    w = ex / S
    return cmax + T * np.log(S), w

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    scores = score_vector(h)
    center = n - 1
    _, w = smooth_weights(scores, T)
    grad = w[center] * dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[center + k]
        wn = w[center - k]
        if wp > 1e-14:
            grad[k:] += wp * dx * (1.0 - h[:n-k])
            grad[:n-k] += wp * dx * (-h[k:])
        if wn > 1e-14:
            grad[:n-k] += wn * dx * (1.0 - h[k:])
            grad[k:] += wn * dx * (-h[:n-k])
    return grad

def project_feasible(y, target):
    n = len(y)
    lo = np.min(y) - 1.5
    hi = np.max(y) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - np.sum(z)
    if diff > 0:
        slack = 1.0 - z
        order = np.argsort(-slack)
        for idx in order:
            adj = min(diff, slack[idx])
            if adj <= 0:
                break
            z[idx] += adj
            diff -= adj
            if diff < 1e-14:
                break
    elif diff < 0:
        slack = z.copy()
        order = np.argsort(slack)
        for idx in order:
            adj = max(diff, -slack[idx])
            if adj >= 0:
                break
            z[idx] += adj
            diff -= adj
            if diff > -1e-14:
                break
    return z

def random_feasible(n, rng):
    target = n / 2.0
    return project_feasible(rng.random(n), target)

def mutate(h, sigma, rng):
    target = len(h) / 2.0
    return project_feasible(h + sigma * rng.standard_normal(len(h)), target)

def binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h, kind='stable')
    h_bin = np.zeros(n)
    h_bin[idx[n - target:]] = 1.0
    return h_bin

def resample(h, new_n):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    y = np.interp(x_new, x_old, h)
    return project_feasible(y, new_n / 2.0)

def eval_pair(h, i, j, alpha):
    old_i, old_j = h[i], h[j]
    h[i] = old_i + alpha
    h[j] = old_j - alpha
    s = score_exact(h)
    h[i] = old_i
    h[j] = old_j
    return s

def line_search_pair(h, i, j, deadline):

    a_lo = max(-lo_val, hi_val - 1.0)
    a_hi = min(1.0 - lo_val, hi_val)
    if a_hi - a_lo < 1e-10:
        return None

    best_s = base_s

    for a in alphas:

            break

            best_s = s

        s2 = eval_pair(h, i, j, a2)
        if s1 < s2:

        new_h = h.copy()
        new_h[i] = lo_val + best_a
        new_h[j] = hi_val - best_a
        return new_h
    return None

            try:

                    break
                continue
            gn = np.linalg.norm(grad)

            direction = -grad / gn

            if not improved:

                            cur = new_h

                        cur_s = s
