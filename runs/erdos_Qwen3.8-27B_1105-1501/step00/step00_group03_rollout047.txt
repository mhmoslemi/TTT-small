    h = np.clip(z + (lo+hi)/2, 0, 1)
    # Fix sum precisely
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            # Add to elements that can be increased
            idx = np.argsort(1 - h)[::-1]  # most room first
        else:

                diff -= step

                h[i] -= step
