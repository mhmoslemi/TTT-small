import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def score_counts(h):

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def greedy_improve(h, max_iter=1000, time_limit=60):
    """Greedy: repeatedly find the best improving pair swap."""
    h = h.copy()
    n = len(h)
    score = exact_score(h)
    start = time.time()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Get all positions of 0s and 1s
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        best_score = score
        best_pair = None
        # Try all pairs (i from zeros, j from ones)
        for i in zeros:
            for j in ones:
                if i == j:
                    continue
                h[i], h[j] = 1.0, 0.0  # swap
                s = exact_score(h)
                h[i], h[j] = 0.0, 1.0  # restore
                if s < best_score - 1e-12:
                    best_score = s
                    best_pair = (i, j)
        
        if best_pair is None:
            break  # local optimum
        
        i, j = best_pair
        h[i], h[j] = 1.0, 0.0
        score = best_score
    
    return h, score

- Term where t+k=i (t=i-k, valid if 0≤i-k and i-k≤n-1-k): was h[i-k]*1, now h[i-k]*0. Change: -h[i-k]
- Term t=j (if j≤n-1-k): was 1*(1-h[j+k]), now 0. Change: -(1-h[j+k])
- Term where t+k=j (t=j-k): was h[j-k]*0=0, now h[j-k]*1. Change: +h[j-k]

So the total change to corr_full[index] for each affected index is the sum of applicable deltas.

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h, seed=0):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def random_balanced(n, rng):
    idx = rng.permutation(n)[:n//2]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def transfer(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    old_x = (np.arange(n) + 0.5) * 2.0 / n
    new_x = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    return to_binary(h_new)

def greedy(h, time_limit=10.0):
    h = h.copy().astype(float)
    n = len(h)
    score = exact_score(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        zeros = np.nonzero(h < 0.5)[0]
        ones = np.nonzero(h > 0.5)[0]
        best_s = score
        best_m = None
        for i in zeros:
            for j in ones:

                s = exact_score(h)
                h[i], h[j] = 0.0, 1.0
                if s < best_s - 1e-15:

                    best_m = (i, j)
        if best_m is None:
            break
        i, j = best_m
        h[i], h[j] = 1.0, 0.0
        score = best_s
    return h, score
