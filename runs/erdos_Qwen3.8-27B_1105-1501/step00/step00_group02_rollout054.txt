import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_full(h):
    """Return all correlation values times dx"""
    dx = 2.0 / len(h)
    return np.correlate(h, 1 - h, mode='full') * dx

def gradient_at_shift(h, d):
    """Gradient of C[d] * dx with respect to h, where d is the shift index in full correlation"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        # C[d] = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
        idx = np.arange(n - d)
        grad[idx] += (1 - h[idx + d])
        grad[idx + d] -= h[idx]
    else:
        dp = -d
        # shift is -dp, so we're looking at lag dp in the other direction
        # In full correlation, index k corresponds to lag k-(n-1)
        # If lag is negative, it means we're computing sum h[i+dp]*(1-h[i])
        idx = np.arange(n - dp)
        grad[idx + dp] += (1 - h[idx])
        grad[idx] -= h[idx + dp]
    return grad * dx

    """Project h onto {x: 0<=x_i<=1, sum(x)=c}"""
    # Use bisection on the threshold
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > c:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    return proj

from scipy import correlate
import time

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_all(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_obj(h, alpha):
    C = c5_all(h)
    m = np.max(C)
    if m < 1e-10:
        return m
    s = np.sum(np.exp(alpha * (C - m)))
    return m + np.log(s) / alpha

def smooth_grad(h, alpha):
    C = c5_all(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(C)
    w = alpha * np.exp(alpha * (C - m))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        d = k - (n - 1)
        if d >= 0:
            idx = np.arange(n - d)
            grad[idx] += w[k] * (1 - h[idx + d])
            grad[idx + d] -= w[k] * h[idx]
        else:
            dp = -d
            idx = np.arange(n - dp)
            grad[idx + dp] += w[k] * (1 - h[idx])
            idx2 = np.arange(n - dp)
            grad[idx2] -= w[k] * h[idx2 + dp]
    return grad * dx

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    dx = 2.0 / len(h)
    return np.correlate(h, 1 - h, mode='full') * dx

def grad_at_shift(h, d):
    """Gradient of C[d] w.r.t. h, where d is the lag (can be negative)"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            grad[d:] += (1 - h[:n-d])
            grad[:n-d] -= h[d:]
    else:
        p = -d
        if n - p > 0:
            grad[:n-p] += (1 - h[p:])
            grad[p:] -= h[:n-p]
    return grad * dx

    """Project h onto {x: 0<=x_i<=1, sum(x)=c}"""
    lo, hi = -1.5, 1.5

So the problem is: maximize min_k sum_i h[i]*h[i+k] subject to 0≤h≤1, sum(h)=n/2.

This is equivalent to: find h that maximizes the minimum autocorrelation (at all lags).

This is a well-studied problem in signal processing (optimal autocorrelation). The goal is to make the autocorrelation as "flat" as possible (maximize the minimum).

One classical approach is to use the "Welch bound" or similar, but for our specific problem, numerical optimization is the way to go.

Let me also try a different formulation: since we want to maximize min_k q_k(h) where q_k(h) = h^T B_k h (quadratic form), we can use the following iterative approach:
1. Start with some h
2. Find k* = argmin_k q_k(h) (worst lag)
3. Solve: maximize q_k*(h) subject to constraints → this is a quadratic optimization
4. Repeat

For step 3, maximizing a quadratic form h^T B h subject to box and sum constraints is a non-convex QP. We can use:

- Or use the fact that B is a symmetric 0-1 matrix (adjacency of a path graph shifted)

Actually, B_k for lag k has 1s at positions (i, i+k) and (i+k, i). This is like a banded matrix. The quadratic form h^T B_k h = 2 * sum_{i=0}^{n-1-k} h[i]*h[i+k] (for k>0).
