    R = np.correlate(h, 1.0 - h, mode="full")

    best = current.copy()
    best_score = cur_score

import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(R) * dx)

def correlate_max_raw(h):
    R = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(R))

def make_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        diff = abs(diff)
        sign = 1 if (target - h.sum()) > 0 or diff > 0 else -1
        for i in order:
            if diff <= 1e-12:
                break
            if sign > 0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
            else:
                rem = min(h[i], diff)
                h[i] -= rem
                diff -= rem
    return h

def local_search_binary(h, rng, n_iters, t_start=0.05, t_end=0.001, full_passes=0):
    """Simulated annealing local search for binary sequences."""
    n = len(h)
    current = h.astype(float).copy()
    cur_score = score_exact(current)
    best = current.copy()
    best_score = cur_score
    no_improve = 0
    
    ones = np.where(current >= 0.99)[0]
    zeros = np.where(current <= 0.01)[0]
    
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
        
        if len(ones) == 0 or len(zeros) == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            current[i], current[j] = current[j], current[i]
            new_score = score_exact(current)
            delta = new_score - cur_score
            if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = current.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                current[i], current[j] = current[j], current[i]
            continue
        
        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]
        
        current[i], current[j] = current[j], current[i]

        delta = new_score - cur_score

            cur_score = new_score

            ones = np.where(current >= 0.99)[0]
            zeros = np.where(current <= 0.01)[0]
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            current[i], current[j] = current[j], current[i]
    
    return best, best_score

def targeted_search(h, rng, n_iters):
    """Search that targets the worst lag for improvement."""
    n = len(h)
    current = h.astype(float).copy()
    cur_score = score_exact(current)
    best = current.copy()
    best_score = cur_score
    
    for it in range(n_iters):
        R = np.correlate(current, 1.0 - current, mode="full")
        k_max = int(np.argmax(R))
        lag = k_max - (n - 1)
        
        # Find positions contributing to worst lag
        if lag > 0:
            seg1 = current[:n - lag]
            seg2 = current[lag:]
        elif lag < 0:
            seg2 = current[:n + lag]
            seg1 = current[-lag:]
        else:
            seg1 = current
            seg2 = current
        
        contrib = seg1 * (1.0 - seg2)
        
        if contrib.sum() > 0:
            # Try fixing the worst contributor
            i_rel = int(np.argmax(contrib))
            if lag > 0:
                i_pos = i_rel
                j_pos = i_rel + lag
            elif lag < 0:
                j_pos = i_rel
                i_pos = i_rel - lag
            else:
                i_pos = j_pos = i_rel
            
            if 0 <= i_pos < n and 0 <= j_pos < n and i_pos != j_pos:
                current[i_pos], current[j_pos] = current[j_pos], current[i_pos]
                new_score = score_exact(current)
                if new_score < cur_score:
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best = current.copy()
                else:
                    current[i_pos], current[j_pos] = current[j_pos], current[i_pos]
        
        # Also try random moves
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            current[i], current[j] = current[j], current[i]
            new_score = score_exact(current)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = current.copy()
            else:
                current[i], current[j] = current[j], current[i]
    
    return best, best_score
