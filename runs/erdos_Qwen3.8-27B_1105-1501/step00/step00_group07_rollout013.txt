def correlation_change(h, i, j, delta, n):
    """
    Compute the change in the correlation array when we do:
    h[i] -= delta, h[j] += delta
    
    Returns the change vector (length 2n-1).
    """
    # For position i with change -delta:
    # C[k] changes by: (-delta) * [1_{0<=i-lag<n}*(1-h[i-lag]) - 1_{0<=i+lag<n}*h[i+lag]]
    # where lag = k - (n-1)
    
    change = np.zeros(2*n - 1)
    
    for k in range(2*n - 1):
        lag = k - (n - 1)
        contrib_i = 0.0
        if 0 <= i - lag < n:
            contrib_i += (1.0 - h[i - lag])
        if 0 <= i + lag < n:
            contrib_i -= h[i + lag]
        
        contrib_j = 0.0
        if 0 <= j - lag < n:
            contrib_j += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            contrib_j -= h[j + lag]
        
        change[k] = -delta * contrib_i + delta * contrib_j
    
    return change
