def pair_descent(h, n, S, score, time_limit, rng, max_sweeps, grid_size=17, tol=1e-7):
    best_h=h; best_score=score
    for sweep in range(max_sweeps):
       if time.time() > time_limit: break
       improved=False
       # order pairs: random permutation or active-pair priority
       pairs = list(itertools.combinations(range(n),2))
       rng.shuffle(pairs) # cannot use itertools? Yes standard.
       for a,b in pairs:
           if time.time() > time_limit: break
           sbar = h[a]+h[b]
           L = max(0.0, sbar-1.0); U=min(1.0,sbar)
           if U-L < 1e-12: continue
           cur = h[a]
           # candidate xs: current, endpoints, grid
           xs = np.linspace(L,U,grid_size)
           # include current if not close and maybe current plus small perturbations
           # maybe use np.unique
           best_trial_score = best_score
           best_x = cur
           h_base = h  # read
           for x in xs:
               if abs(x-cur) < 1e-14: continue
               h_t = h_base.copy()
               h_t[a]=x; h_t[b]=sbar-x
               sc = exact_score(h_t)
               if sc < best_trial_score - 1e-12:
                   best_trial_score = sc; best_x = x
           if best_x != cur:
               h[a]=best_x; h[b]=sbar-best_x
               score = best_trial_score
               improved=True
       if not improved: break
       # after each sweep, maybe repair exact sum (should)
       h = repair_exact(h,S)
       score = exact_score(h)
    return h,score

def _repair_exact(h,S):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    n = h.size
    r = S - h.sum()
    if r > 1e-15:
       cap = 1.0 - h
       total = cap.sum()
       if total > 0:
          h += r * cap / total
       else:
          h.fill(1.0)
          r = S - n
    elif r < -1e-15:
       total = h.sum()
       if total > 0:
          h += r * h / total
       else:
          h.fill(0.0)
          r = S
    np.clip(h,0.0,1.0,out=h)
    # residual loop
    for _ in range(n + 3):
       r = S - h.sum()
       if abs(r) <= 1e-13: break
       if r > 0:

          add = min(r, 1.0 - h[idx])
          if add <= 1e-16: break

          idx = int(np.argmin(h))
          dec = min(-r, h[idx])
          if dec <= 1e-16: break

    return h

import numpy as np

_TAU_GLOBAL = 100.0
_S_TARGET_GLOBAL = 0.0
_DE_S_GLOBAL = 0.0


def _corr_c(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _exact_score(h):
    return float(np.max(_corr_c(h)))


def _repair_exact(h, S):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)

    r = S - h.sum()
    if r > 1e-14:
        cap = 1.0 - h
        total = cap.sum()
        if total > 1e-15:
            h += r * cap / total
        else:
            h.fill(1.0)
    elif r < -1e-14:
        total = h.sum()
        if total > 1e-15:
            h += r * h / total
        else:
            h.fill(0.0)

    np.clip(h, 0.0, 1.0, out=h)

    for _ in range(h.size + 5):
        r = S - h.sum()
        if abs(r) <= 1e-13:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            add = min(r, 1.0 - h[idx])
            if add <= 1e-16:
                break
            h[idx] = min(1.0, h[idx] + add)
        else:
            idx = int(np.argmin(h))
            dec = min(-r, h[idx])
            if dec <= 1e-16:
                break
            h[idx] = max(0.0, h[idx] - dec)

    r = S - h.sum()
    if abs(r) > 1e-12:

def _light_repair(h, S):

def _feasible(x, S):

def _detect_orientation(n):
    if n <= 1:
        return 1
    rng = np.random.default_rng(20240528)
    h = rng.uniform(0.1, 0.9, n)
    c = _corr_c(h)
    p = 0
    eps = 1e-5
    g_num = np.zeros(n, dtype=float)
    for i in range(n):
        h2 = h.copy()
        h2[i] += eps
        c2 = _corr_c(h2)
        g_num[i] = (c2[p] - c[p]) / eps

    dx = 2.0 / n
    g_pos = np.zeros(n, dtype=float)
    g_pos[0] = (1.0 - h[-1]) * dx
    g_pos[-1] = (-h[0]) * dx

    g_neg[-1] = (1.0 - h[0]) * dx
    g_neg[0] = (-h[-1]) * dx

    err_pos = float(np.linalg.norm(g_num - g_pos))
    err_neg = float(np.linalg.norm(g_num - g_neg))
    return 1 if err_pos <= err_neg else 0


def _smooth_value_and_weights(h, tau):
    c = _corr_c(h)
    tau = float(tau)
    z = c * tau
    zmax = float(z.max())
    expz = np.exp(z - zmax)
    s = float(expz.sum())
    if s <= 0.0 or not math.isfinite(s):
        return zmax, np.zeros_like(c, dtype=float)
    val = (zmax + math.log(s)) / tau
    w = expz / s
    return val, w


def _smooth_grad_from_weights(h, w):
    n = h.size
    dx = 2.0 / n
    grad = np.zeros(n, dtype=float)

    global _ORIENT_MODE
    if _ORIENT_MODE is None:
        _ORIENT_MODE = _detect_orientation(n)
    pos_mode = (_ORIENT_MODE == 1)

    m = 2 * n - 1
    for idx in range(m):
        wt = float(w[idx])
        if wt <= 1e-16:
            continue
        if pos_mode:
            s = (n - 1) - idx
        else:
            s = idx - (n - 1)

        if s >= 0:
            L = n - s
            if L > 0:

                grad[s:s + L] -= wt * dx * h[:L]

            t = -s
            L = n - t
            if L > 0:
                grad[t:t + L] += wt * dx * (1.0 - h[:L])
                grad[:L] -= wt * dx * h[t:t + L]

    return grad

    return _smooth_grad_from_weights(x, w)


def _eq_fun(x):
    return float(np.sum(x) - _S_TARGET_GLOBAL)

def _slsqp_polish(h, n, S, tau, deadline, maxiter=80):
    global _TAU_GLOBAL, _S_TARGET_GLOBAL

    h = _feasible(h, S)
    base_score = _exact_score(h)

        return h, base_score

    _TAU_GLOBAL = float(tau)
    _S_TARGET_GLOBAL = float(S)

def _lag_gradients(h):
    n = h.size
    c = _corr_c(h)
    m = 2 * n - 1
    G = np.empty((m, n), dtype=float)
    dx = 2.0 / n

    if _ORIENT_MODE is None:
        _ORIENT_MODE = _detect_orientation(n)
    pos_mode = (_ORIENT_MODE == 1)

        g = np.zeros(n, dtype=float)
        if pos_mode:
            s = (n - 1) - idx
        else:
            s = idx - (n - 1)

        if s >= 0:
            L = n - s
            if L > 0:
                g[:L] += dx * (1.0 - h[s:s + L])
                g[s:s + L] -= dx * h[:L]
        else:
            t = -s
            L = n - t
            if L > 0:
                g[t:t + L] += dx * (1.0 - h[:L])
                g[:L] -= dx * h[t:t + L]

        G[idx] = g

    return c, G

    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    max_sweeps = 10 if n <= 80 else 3

    for _ in range(max_sweeps):
        if time.time() >= deadline:
            break
        order = pairs[:]
        if rng is not None:
            rng.shuffle(order)

        improved = 0
        for a, b in order:
            if time.time() >= deadline:
                break
            h, score, did = _optimize_pair(h, n, S, score, a, b, grid, rng)
            if did:
                improved += 1

        if improved == 0:
            break
        h = _light_repair(h, S)
        score = _exact_score(h)

    return h, score


def _resample_sequence(parent, n):
    parent = np.asarray(parent, dtype=float)
    if parent.size == n:
        return _feasible(parent.copy(), n * 0.5)
    old = (np.arange(parent.size, dtype=float) + 0.5) * (2.0 / parent.size)
    new = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    h = np.interp(new, old, parent)
    return _feasible(h, n * 0.5)


def _analytic_starts(n, S, rng, count=3):
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    out = []
    for _ in range(int(count)):
        a = rng.uniform(-0.5, 0.5)
        b = rng.uniform(-0.35, 0.35)
        ph = rng.uniform(0.0, 2.0 * math.pi)
        h1 = 0.5 + a * np.cos(math.pi * (x - 1.0)) + b * np.cos(2.0 * math.pi * (x - 1.0) + ph)
        out.append(_feasible(h1, S))

        q = rng.uniform(0.5, 2.2)
        h2 = 1.0 - q * (x - 1.0) ** 2
        out.append(_feasible(h2, S))
    return out


def _random_starts(n, S, rng, count=2):
    out = []
    for _ in range(int(count)):
        out.append(_feasible(rng.uniform(0.0, 1.0, n), S))

        x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
        h = np.full(n, 0.5, dtype=float)
        for j in range(1, 4):
            amp = rng.uniform(0.02, 0.25)
            ph = rng.uniform(0.0, 2.0 * math.pi)
            h = h + amp * np.cos(j * math.pi * (x - 1.0) + ph)
        out.append(_feasible(h, S))
    return out


def _perturb_candidates(parent, noise_levels, count, rng):
    n = parent.size
    S = n * 0.5
    out = []
    counter = 0
    for noise in noise_levels:
        for _ in range(int(count)):
            h = parent + rng.normal(0.0, float(noise), n)
            hf = _feasible(h, S)
            sc = _exact_score(hf)
            out.append((sc, counter, hf))
            counter += 1
    return out


def _de_objective(x):
    h = np.asarray(x, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h = _repair_exact(h, _DE_S_GLOBAL)
    return _exact_score(h)


def _de_global(n, S, seed, maxiter=60, popsize=15):
    global _DE_S_GLOBAL
    _DE_S_GLOBAL = float(S)
    bounds = [(0.0, 1.0)] * n

    try:
        res = sopt.differential_evolution(
            _de_objective,
            bounds,
            seed=int(seed),
            maxiter=int(maxiter),
            popsize=int(popsize),
            polish=False,
            updating="inplace",
            init="latinhypercube",
            tol=1e-7,
        )
    except TypeError:
        try:
            res = sopt.differential_evolution(
                _de_objective,
                bounds,
                seed=int(seed),
                maxiter=int(maxiter),
                popsize=int(popsize),
                polish=False,
                updating="inplace",
                tol=1e-7,
            )
        except Exception:
            return None, float("inf")
    except Exception:
        return None, float("inf")

    if res is None or getattr(res, "x", None) is None:
        return None, float("inf")

    h = _feasible(res.x, S)
    return h, _exact_score(h)


def _random_binary(n, S, rng):
    if abs(S - round(S)) > 1e-9:
        return None
    target = int(round(S))
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)[:target]
    h[idx] = 1.0
    return h


def _binary_local(h, n, S, score, deadline, rng):
    if h is None:
        return None, float("inf")

    target = int(round(S))
    h = np.array(h, dtype=float, copy=True)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    if len(ones) < target:
        need = target - len(ones)
        if len(zeros) >= need:
            h[zeros[:need]] = 1.0
    elif len(ones) > target:
        extra = len(ones) - target
        ones = np.where(h == 1.0)[0]
        h[ones[:extra]] = 0.0

    score = _exact_score(h)

    for _ in range(12):
        if time.time() >= deadline:
            break
        improved = False
        for _in in range(2 * n + 5):
            if time.time() >= deadline:
                break
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = int(rng.choice(ones))
            b = int(rng.choice(zeros))
            h2 = h.copy()
            h2[a] = 0.0
            h2[b] = 1.0
            sc = _exact_score(h2)
            if sc < score - 1e-13:
                h = h2
                score = sc
                improved = True
        if not improved:
            break

    return h, score


def _optimize_start(h, S, deadline, rng, use_slqp=True):
    n = h.size
    h = _feasible(h, S)
    score = _exact_score(h)

    if time.time() >= deadline:
        return h, score

    rem = deadline - time.time()
    if rem > 8:
        lp_alloc = min(0.25 * rem, 40.0)
        if lp_alloc > 5:
            lp_dead = min(deadline, time.time() + lp_alloc)
            h, score = _lp_polish(h, n, S, score, lp_dead, max_iters=max(5, int(lp_alloc * 0.8) + 5))

    rem = deadline - time.time()
    if rem > 12:
        pair_alloc = min(0.35 * rem, 90.0)
        if pair_alloc > 5:
            grid = 33 if rem > 80 else 21
            pair_dead = min(deadline, time.time() + pair_alloc)
            h, score = _pair_descent(h, n, S, score, pair_dead, rng, grid=grid)

    if use_slqp:
        rem = deadline - time.time()
        if rem > 25:
            alloc = min(25.0, 0.25 * rem)
            if alloc > 8:
                dead = min(deadline, time.time() + alloc)
                cand, sc = _slsqp_polish(h, n, S, 80.0, dead, maxiter=max(30, int(alloc * 3)))
                if sc < score:
                    h, score = cand, sc

        rem = deadline - time.time()
        if rem > 20:
            alloc = min(20.0, 0.25 * rem)
            if alloc > 8:
                dead = min(deadline, time.time() + alloc)
                cand, sc = _slsqp_polish(h, n, S, 180.0, dead, maxiter=max(30, int(alloc * 3)))
                if sc < score:
                    h, score = cand, sc

    rem = deadline - time.time()
    if rem > 10:
        final_alloc = min(0.2 * rem, 45.0)
        if final_alloc > 5:
            final_dead = min(deadline, time.time() + final_alloc)
            h, score = _pair_descent(h, n, S, score, final_dead, rng, grid=25)

    h = _light_repair(h, S)
    score = _exact_score(h)
    return h, score


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(5.0, 0.97 * float(budget_s))
    rng = np.random.default_rng(int(seed))

    best_h = None
    best_score = float("inf")

    parent = None
    try:
        if "initial_h_values" in globals():
            parent = np.array(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is not None and parent.size > 1:
        n0 = int(parent.size)
        S0 = n0 * 0.5
        h0 = _feasible(parent, S0)
        best_h, best_score = _consider_candidate(h0, S0, best_h, best_score)

        parent_alloc = min(350.0, max(30.0, 0.40 * float(budget_s)))
        parent_dead = min(deadline, t0 + parent_alloc)
        hopt, sopt_score = _optimize_start(h0, S0, parent_dead, rng, use_slqp=True)
        best_h, best_score = _consider_candidate(hopt, S0, best_h, best_score)

        if time.time() < deadline - 80:
            cands = _perturb_candidates(
                parent,
                (0.001, 0.003, 0.008, 0.02, 0.05, 0.12),
                4,
                rng,
            )
            cands.sort()

            threshold = (best_score + 0.02) if best_score < 1e9 else 1.0
            chosen = []
            for sc, _idx, h in cands:
                if len(chosen) >= 3:
                    break
                if sc <= threshold:
                    chosen.append(h)

            if len(chosen) == 0 and len(cands) > 0 and (deadline - time.time()) > 160:
                chosen.append(cands[0][2])

            for h in chosen:
                if time.time() > deadline - 40:
                    break
                alloc = min(65.0, deadline - time.time())
                hopt, hscore = _optimize_start(h, S0, min(deadline, time.time() + alloc), rng, use_slqp=True)
                best_h, best_score = _consider_candidate(hopt, S0, best_h, best_score)

        if parent is not None and time.time() < deadline - 80:
            hde, sde = _de_global(n0, S0, int(seed) + 1, maxiter=80, popsize=16)
            if hde is not None and sde < best_score + 0.02:
                alloc = min(100.0, deadline - time.time())
                hopt, hscore = _optimize_start(hde, S0, min(deadline, time.time() + alloc), rng, use_slqp=True)
                best_h, best_score = _consider_candidate(hopt, S0, best_h, best_score)

    for nn in (62, 60, 64, 58, 66):
        if time.time() > deadline - 90:
            break

        Snn = nn * 0.5
        cands = []
        counter = 0

        if parent is not None:
            hres = _resample_sequence(parent, nn)
            cands.append((_exact_score(hres), counter, hres))
            counter += 1

        for h in _analytic_starts(nn, Snn, rng, 2):
            cands.append((_exact_score(h), counter, h))
            counter += 1

        for h in _random_starts(nn, Snn, rng, 1):
            cands.append((_exact_score(h), counter, h))
            counter += 1

        if abs(Snn - round(Snn)) < 1e-9:
            for _rnd in range(4):
                hb = _random_binary(nn, Snn, rng)
                if hb is not None:
                    cands.append((_exact_score(hb), counter, hb))
                    counter += 1

        if time.time() < deadline - 80:
            hde, sde = _de_global(nn, Snn, int(seed) + 100 + nn, maxiter=45, popsize=12)
            if hde is not None:
                cands.append((sde, counter, hde))
                counter += 1

        cands.sort()
        if len(cands) == 0:
            continue

        threshold = (best_score + 0.03) if best_score < 1e9 else 1.0
        if cands[0][0] <= threshold or (best_score < 1e9 and cands[0][0] <= 0.42):
            h = cands[0][2]
            alloc = min(80.0, deadline - time.time())
            use_slqp = (cands[0][0] < best_score + 0.01) or (nn == 62)
            hopt, hscore = _optimize_start(h, Snn, min(deadline, time.time() + alloc), rng, use_slqp=use_slqp)
            best_h, best_score = _consider_candidate(hopt, Snn, best_h, best_score)

    for nb in (62, 60, 64):
        if time.time() > deadline - 35:
            break
        Snb = nb * 0.5
        if abs(Snb - round(Snb)) > 1e-9:
            continue
        for rep in range(3):
            if time.time() > deadline - 20:
                break
            hb = _random_binary(nb, Snb, rng)
            if hb is None:
                continue
            hb, scb = _binary_local(hb, nb, Snb, _exact_score(hb), min(deadline, time.time() + 12.0), rng)
            best_h, best_score = _consider_candidate(hb, Snb, best_h, best_score)

    if best_h is not None and time.time() < deadline - 25:
        nb = int(best_h.size)
        Sbb = nb * 0.5
        hbb, scb = _lp_polish(best_h, nb, Sbb, best_score, min(deadline, time.time() + 20.0), max_iters=25)
        best_h, best_score = _consider_candidate(hbb, Sbb, best_h, best_score)

        hbb, scb = _pair_descent(best_h, nb, Sbb, best_score, min(deadline, time.time() + 30.0), rng, grid=27)
        best_h, best_score = _consider_candidate(hbb, Sbb, best_h, best_score)

    if best_h is None:
        nf = 63
        Sff = nf * 0.5
        best_h = _feasible(np.full(nf, 0.5, dtype=float), Sff)
        best_score = _exact_score(best_h)

    final_S = int(best_h.size) * 0.5
    best_h = _feasible(best_h, final_S)
    final_score = _exact_score(best_h)
    return best_h, float(final_score), int(best_h.size)
