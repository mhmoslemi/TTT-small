import numpy as np
import time


def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    """Compute the exact C5 value."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def gradient_smooth_max(h, temperature=0.05):
    """Gradient of smooth max of (correlation * dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temperature)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-14:
            continue
        a = max(0, s)
        b = min(n - 1, n - 1 + s)
        if a <= b:
            grad[a:b + 1] += w * (1.0 - h[a - s:b - s + 1])
        a2 = max(0, -s)
        b2 = min(n - 1, n - 1 - s)
        if a2 <= b2:
            grad[a2:b2 + 1] -= w * h[a2 + s:b2 + s + 1]
    return grad


def coord_descent_sweep(h, target_sum, n_test=25):
    """One sweep of coordinate descent. Returns (h, num_improved)."""
    n = len(h)
    improved = 0
    for j in range(n):
        sum_others = np.sum(h) - h[j]
        lo = max(0.0, target_sum - sum_others)
        hi = min(1.0, target_sum - sum_others)
        if hi - lo < 1e-10:
            continue
        cur_c5 = compute_c5(h)
        best_val = h[j]
        best_c5_j = cur_c5
        test_vals = np.linspace(lo, hi, n_test)
        for val in test_vals:
            h[j] = val
            c5_j = compute_c5(h)
            if c5_j < best_c5_j:
                best_c5_j = c5_j
                best_val = val
        h[j] = best_val
        if best_c5_j < cur_c5 - 1e-12:
            improved += 1
    return h, improved

    """Create a structured initial pattern."""
    h = np.zeros(n)
    # Use a low-discrepancy style pattern
    phi = 0.6180339887
    positions = np.array([int((i * phi * n) % n) for i in range(n)])
    positions = np.unique(positions)
    if len(positions) >= n // 2:
        h[positions[:n // 2]] = 1.0
    else:
        h[::2] = 1.0
    return project_onto_constraint(h, n / 2.0)
