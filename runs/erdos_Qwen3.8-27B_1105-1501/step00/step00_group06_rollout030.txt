        return 0.0, 0.0
    return lam_lo, lam_hi

        return float(np.max(corr))

    """Find best lam for h[i]+=lam, h[j]-=lam to minimize max correlation."""
    lam_lo = max(-h[i], h[j] - 1.0) + 1e-10
    lam_hi = min(1.0 - h[i], h[j]) - 1e-10
