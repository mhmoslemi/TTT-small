    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def corr_full(h):
    return np.correlate(h, 1 - h, mode='full')

def corr_gradient(h, k_center):
    """Compute gradient of f_k(h) = corr_full(h)[k_center] with respect to h.
    k_center is the index into the full correlation array (0 to 2n-2).
    The actual lag is k_center - (n-1).
    """
    n = len(h)
    lag = k_center - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        # Term 1: j = i + k, so i = j - k
        i = j - lag
        if 0 <= i < n:
            grad[j] += (1 - h[i])
        # Term 2: j = i, so the factor is h[i+k] = h[j+k], derivative of (1-h[i]) is -1
        i_plus_k = j + lag
        if 0 <= i_plus_k < n:
            grad[j] -= h[i_plus_k]
    return grad

    """Project onto {0 <= h <= 1, sum(h) = target_sum} via bisection."""
    n = len(h)
    if h.sum() == target_sum:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid

        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = _project(h_interp, n_try / 2.0)
        try:
            h1, c1 = _slsqp(h_interp, n_try, maxiter=400, ftol=1e-12)
            if c1 < best_c5:
                best_h, best_c5 = h1, c1
                n = n_try
                print(f"n={n_try}: C5={best_c5:.6f}", flush=True)
        except Exception:
            pass

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.97:
        try:
            h1, c1 = _slsqp(best_h, len(best_h), maxiter=500, ftol=1e-13)
            if c1 < best_c5:
                best_h, best_c5 = h1, c1
        except Exception:
            pass

    # Ensure exact constraint satisfaction
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, len(best_h)
