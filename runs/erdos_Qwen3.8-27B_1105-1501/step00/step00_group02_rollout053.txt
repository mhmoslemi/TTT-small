def local_search(h, rng, max_iters=50000, temperature=0.001):
    """Simulated annealing local search."""
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h_local = h.copy()
    best_c5_local = current_c5
    
    for it in range(max_iters):
        # Pick a random coordinate to perturb
        j = rng.randint(n)
        old_val = h[j]
        
        # Perturb
        if rng.random() < 0.5:
            new_val = min(1.0, old_val + rng.uniform(0.01, 0.1))
        else:
            new_val = max(0.0, old_val - rng.uniform(0.01, 0.1))
        
        h[j] = new_val
        
        # Fix sum constraint (project)
        target = n / 2.0
        current_sum = np.sum(h)
        if abs(current_sum - target) > 1e-10:
            # Distribute the correction
            diff = target - current_sum
            # Add to other coordinates proportionally
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            h[mask] += diff / (n - 1)
            h = np.clip(h, 0, 1)
        
        new_c5 = compute_c5(h)
        
        if new_c5 < current_c5 or rng.random() < np.exp((current_c5 - new_c5) / temperature):
            current_c5 = new_c5
            if new_c5 < best_c5_local:
                best_c5_local = new_c5
                best_h_local = h.copy()
        else:
            h[j] = old_val
            # Re-fix sum
            target = n / 2.0
            current_sum = np.sum(h)
            if abs(current_sum - target) > 1e-10:
                diff = target - current_sum
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                h[mask] += diff / (n - 1)
                h = np.clip(h, 0, 1)
        
        # Cool down
        if it % 10000 == 9999:
            temperature *= 0.5
    
    return best_h_local, best_c5_local

    """Project h onto {x : 0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    h = h.copy()
