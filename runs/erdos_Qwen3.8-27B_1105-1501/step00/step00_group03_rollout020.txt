import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

_g_target = 0.0
_g_dx = 1.0
_g_T = 10.0
_g_pen = 100.0

def _objfun(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    c5_approx = logsumexp(_g_T * corr) / _g_T
    s = x.sum()
    return _g_dx * c5_approx + _g_pen * (s - _g_target) ** 2

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            scale = min(1.0, diff / total)
            h += scale * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            scale = min(1.0, -diff / total)
            h -= scale * h
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
            else:
                amt = min(-diff, h[i])
                h[i] -= amt
                diff += amt
            if abs(diff) < 1e-14:
                break
    return h

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def compute_subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    tol = max_val - 1e-10
    active = np.where(corr >= tol)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        for i in range(i_start, i_end):
            j = i + k
            if k == 0:
                grad[i] += (1.0 - 2.0 * h[i])
            else:
                grad[i] += (1.0 - h[j])
                grad[j] += (-h[i])
    grad /= len(active)
    grad -= grad.mean()
    return grad

def optimize_one(n, start_h, rng, t0, budget_s):
    global _g_target, _g_dx, _g_T, _g_pen
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = make_feasible(start_h.copy(), target)
    c5_cur = compute_c5(h_cur)

    # Phase 1: L-BFGS-B with temperature annealing
    temps = [3.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0]
    for T in temps:
        if budget_s - (time.time() - t0) < 8:
            break
        _g_target = target
        _g_dx = dx
        _g_T = T
        _g_pen = max(50.0, float(n))
        res = minimize(
            _objfun, h_cur, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        h_new = make_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-14:
            h_cur = h_new
            c5_cur = c5_new

    # Phase 2: Subgradient with line search
    for _it in range(400):
        if budget_s - (time.time() - t0) < 4:

        grad = compute_subgradient(h_cur)

                c5_cur = c5_try

        if not improved:

                    found = True

    # Phase 3: Pair perturbation (add delta to i, subtract from j)

            if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                continue
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-13:
                h_cur = h_try
                c5_cur = c5_try
                improved_pair = True
                break
        if not improved_pair:
            break

    # Phase 4: Swap moves
    for _it in range(300):
        if budget_s - (time.time() - t0) < 2:
            break
        improved_swap = False
        for _t in range(40):
            i, j = rng.choice(n, 2, replace=False)
            h_try = h_cur.copy()
            h_try[i], h_try[j] = h_cur[j], h_cur[i]
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-13:
                h_cur = h_try
                c5_cur = c5_try
                improved_swap = True
                break
        if not improved_swap:
            break

    # Phase 5: Fine subgradient polish
    for _it in range(200):
        if budget_s - (time.time() - t0) < 2:
            break
        grad = compute_subgradient(h_cur)
        improved = False

                h_cur = h_try
                c5_cur = c5_try
                improved = True
