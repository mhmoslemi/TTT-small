def exact_objective(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_vector(h):
    """Return the full correlation vector (before multiplying by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_subgradient(h, active_lags, N):
    """
    Compute the subgradient of the max objective.
    active_lags: array of lag indices in the correlate output (0 to 2N-2)
    Returns gradient vector of length N.
    """
    grad = np.zeros(N)
    for m in active_lags:
        k = m - (N - 1)  # actual lag
        if k >= 0:
            # overlap(k) = sum_{i=0}^{N-1-k} h[i] * (1-h[i+k])
            # d/dh[j]: 
            #   if j in [0, N-1-k]: contributes (1-h[j+k])
            #   if j in [k, N-1]: contributes -h[j-k]
            for i in range(N - k):
                # h[i] * (1-h[i+k])
                # d/dh[i] = (1-h[i+k])
                grad[i] += (1.0 - h[i + k])
                # d/dh[i+k] = -h[i]
                grad[i + k] += (-h[i])
        else:
            kk = -k  # positive version
            # overlap(-kk) = sum_{i=0}^{N-1-kk} h[i+kk] * (1-h[i])
            # equivalently: sum_{i=kk}^{N-1} h[i] * (1-h[i-kk])

                # d/dh[i] = -h[i+kk]
                grad[i] += (-h[i + kk])

def dyadic_repair(h, N, bits=20):
    """Repair h to satisfy sum(h) = N/2 exactly (within dyadic representation)"""
    S = 1 << bits
    target = N * S // 2  # This works when N*S is even (S is power of 2)
    z = np.round(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    cur = z.sum()
    diff = target - cur
    # Water filling to adjust
    if diff > 0:
        # Need to increase sum by diff
        # Find positions where we can increase (z < S)
        while diff > 0:
            candidates = np.where(z < S)[0]
            if len(candidates) == 0:
                break
            # Prefer positions with smallest z (most room)
            order = candidates[np.argsort(z[candidates])]
            for idx in order:
                if diff <= 0:
                    break
                z[idx] += 1
                diff -= 1
    elif diff < 0:
        # Need to decrease sum by -diff
        while diff < 0:
            candidates = np.where(z > 0)[0]
            if len(candidates) == 0:
                break
            order = candidates[np.argsort(-z[candidates])]
            for idx in order:
                if diff >= 0:
                    break
                z[idx] -= 1
                diff += 1
    return z.astype(np.float64) / S

    """Resample parent (length P) to new length new_N using interpolation"""

    # Piecewise constant: each new bin covers P/N of old bins

    new_idx = np.linspace(0, P - 1, new_N)
    interpolated = np.interp(new_idx, old_idx, parent)
    return interpolated

import numpy as np

def exact_objective(h):

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def objective_values(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_subgradient(h, active_indices, N):
    grad = np.zeros(N)
    for m in active_indices:
        k = abs(m - (N - 1))
        n = N - k
        grad[0:n] += (1.0 - h[k:N])
        grad[k:N] += (-h[0:n])
    return grad / len(active_indices)

def dyadic_repair(h, N, bits=20):
    S = 1 << bits
    target = N * S // 2
    z = np.round(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    diff = target - int(z.sum())
    if diff > 0:
        # Need to add diff units; spread to positions with most room
        room = S - z
        order = np.argsort(-room)
        for i in range(diff):
            z[order[i % N]] += 1
    elif diff < 0:
        # Need to remove -diff units; remove from positions with most mass
        order = np.argsort(-z)
        for i in range(-diff):
            idx = order[i % N]
            if z[idx] > 0:
                z[idx] -= 1
    return z.astype(np.float64) / S

def resample_density(parent, new_N):
    P = len(parent)
    old_mid = (np.arange(P) + 0.5) / P
    new_mid = (np.arange(new_N) + 0.5) / new_N
    # CDF-based resampling
    cum_parent = np.cumsum(parent) / parent.sum()
    # new values are the height of the step function
    interpolated = np.zeros(new_N)
    for i in range(new_N):
        frac = new_mid[i]
        idx = np.searchsorted(cum_parent, frac, side='right')
        if idx >= P:
            interpolated[i] = parent[-1]
        elif idx == 0:
            interpolated[i] = parent[0]
        else:
            prev = cum_parent[idx-1] if idx > 0 else 0.0
            width = cum_parent[idx] - prev
            if width > 1e-15:
                t = (frac - prev) / width
                interpolated[i] = parent[idx-1] * (1-t) + parent[idx] * t
            else:
                interpolated[i] = parent[idx]
    # Rescale to correct sum
    s = interpolated.sum()
    if s > 1e-10:
        interpolated = interpolated * (new_N / 2.0) / s
    interpolated = np.clip(interpolated, 0, 1)
    return dyadic_repair(interpolated, new_N, bits=20)

        dims = sorted([34, 36, 38, 40, 42, 44, 46, 48, 52, 56, 60])
    
    # Phase 1: Quick gradient descent on parent
    if time.time() < deadline - 60:
        tl = min(40, deadline - time.time() - 30)
        h1, C1 = gradient_descent(best_h, best_N, tl)
        if C1 < best_C:
            best_h, best_C, best_N = h1, C1, best_N
    
    # Phase 2: Directed swaps on parent
    if time.time() < deadline - 60:
        tl = min(40, deadline - time.time() - 30)
        h2, C2 = directed_swaps(best_h, best_N, tl, rng)
        if C2 < best_C:
            best_h, best_C = h2, C2
    
    # Phase 3: Sweep other dimensions
    for N in dims:
        if time.time() > deadline - 20:
            break
        if N == best_N and parent is not None:
            pass  # Already tried parent dim
        
        # Create starting point
        if parent is not None:
            if N == N_parent:
                h = best_h.copy()
            else:
                h = resample_density(parent, N)
        else:
            h = make_synthetic(N, seed + N)
        
        # Apply gradient descent
        elapsed = time.time() - t0
        remaining = deadline - time.time()
        if remaining < 10:
            break
        
        alloc = min(remaining * 0.3, 60)
        if alloc > 3:
            h_g, C_g = gradient_descent(h, N, alloc)
            if C_g < best_C:
                best_h, best_C, best_N = h_g, C_g, N
        
        # Directed swaps
        remaining = deadline - time.time()
        if remaining < 10:
            break
        alloc = min(remaining * 0.25, 40)
        if alloc > 3:
            h_d, C_d = directed_swaps(h_g if C_g < exact_objective(h) else h, N, alloc, rng)
            if C_d < best_C:
                best_h, best_C, best_N = h_d, C_d, N
        
        # LP descent
        remaining = deadline - time.time()
        if remaining > 15:
            alloc = min(remaining * 0.15, 25)
            if alloc > 3:
                h_l, C_l = lp_descent(h_d if C_d < exact_objective(h_g) else h_g, N, alloc)
                if C_l < best_C:
                    best_h, best_C, best_N = h_l, C_l, N
        
        # Early exit if target met at small dim
        if best_C <= target and best_N <= 42:
            break
    
    # Phase 4: Final refinement on best
    if time.time() < deadline - 5:
        tl = min(deadline - time.time() - 5, 50)
        h_f, C_f = gradient_descent(best_h, best_N, tl * 0.4, tau_frac=0.005, max_step=0.02)
        if C_f < best_C:
            best_h, best_C = h_f, C_f
        
        tl = min(deadline - time.time() - 5, 40)
        if tl > 5:
            h_f2, C_f2 = directed_swaps(best_h, best_N, tl, rng)
            if C_f2 < best_C:
                best_h, best_C = h_f2, C_f2
    
    # Phase 5: Random hill climbing for diversity
    if time.time() < deadline - 5:
        tl = min(deadline - time.time() - 5, 30)
        if tl > 3:
            h_r, C_r = random_hill_climb(best_h, best_N, tl, rng)
            if C_r < best_C:
                best_h, best_C = h_r, C_r
    
    # Final repair
    best_h = np.clip(best_h, 0, 1)
    best_h = dyadic_repair(best_h, best_N, bits=20)
    final_C = exact_objective(best_h)
    
    # Validate
    if not (np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)):
        # Fallback to parent if available
        if parent is not None:
            best_h = parent.copy()
            best_C = exact_objective(parent)
            best_N = N_parent
        else:
            best_h = make_synthetic(42, seed)
            best_C = exact_objective(best_h)
            best_N = 42
    else:
        if final_C < best_C:
            best_C = final_C
    
    return best_h, best_C, best_N
