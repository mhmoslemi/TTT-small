import numpy as np
from scipy import optimize
import time


def official_score(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_vector(z, n_target):
    h = np.array(z, dtype=float).copy().ravel()[:n_target]
    if len(h) < n_target:
        h = np.pad(h, (0, n_target - len(h)), constant_values=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n_target / 2.0
    for _ in range(500):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            delta = min(residual, room[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            h[idx] += residual
        else:
            idx = np.argmax(h)
            h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_lags(n):
    return np.arange(-(n - 1), n)


def compute_gradient_row(h, k):
    n = len(h)
    p = np.arange(n)
    g = np.zeros(n)
    ok1 = (p - k) >= 0
    ok1 &= (p - k) < n
    g[ok1] += (1.0 - h[p[ok1] - k])
    ok2 = (p + k) >= 0
    ok2 &= (p + k) < n
    g[ok2] += (-h[p[ok2] + k])
    return g


def gen_random_binary(n, rng):
    target = n / 2.0
    m_full = int(target)
    frac = target - m_full
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:m_full]] = 1.0
    if frac > 0:
        h[idx[m_full]] = frac
    return repair_vector(h, n)


def gen_random_continuous(n, rng):
    h = rng.uniform(0, 1, n)
    return repair_vector(h, n)


def gen_step_family(n, rng, num_steps):
    if num_steps >= n:
        h = rng.uniform(0, 1, n)
        return repair_vector(h, n)
    boundaries = sorted(rng.choice(n - 1, size=num_steps - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    vals = rng.uniform(0, 1, num_steps)
    h = np.zeros(n)
    for i in range(num_steps):
        h[boundaries[i]:boundaries[i + 1]] = vals[i]
    return repair_vector(h, n)


def gen_interpolated(parent, n, offset=0):
    m = len(parent)
    if m == 0:
        return repair_vector(np.full(n, 0.5), n)
    xs = (np.arange(m) + 0.5) * 2.0 / m
    xt = ((np.arange(n) + 0.5 + offset) % n) * 2.0 / n
    order = np.argsort(xt)
    xt_sorted = xt[order]
    h = np.interp(xt_sorted, xs, parent, left=parent[0], right=parent[-1])
    h = h[np.argsort(order)]
    return repair_vector(h, n)


def gen_sine(n, rng):
    t = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(2, 8)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * t + phase)
    h += rng.normal(0, 0.05, n)
    return repair_vector(h, n)


def local_descent(h, n, max_iters=60, time_limit=25.0, seed=42):
    rng = np.random.default_rng(seed)
    h = repair_vector(h, n)
    score = official_score(h)
    dx = 2.0 / n
    lags = corr_lags(n)
    start_time = time.perf_counter()

    for it in range(max_iters):
        if time.perf_counter() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = corr * dx
        max_val = np.max(scaled)
        active_mask = scaled >= max_val - 0.005
        active_idx = np.where(active_mask)[0]

        grad = np.zeros(n)
        for ki in active_idx:
            k = lags[ki]
            grad -= dx * compute_gradient_row(h, k)

        grad -= np.mean(grad)

        rnd = rng.normal(0, 0.15, n)
        rnd -= np.mean(rnd)
        direction = grad + rnd

        norm = np.max(np.abs(direction))
        if norm < 1e-12:
            break
        direction /= norm

        best_alpha = 0.0
        best_local = score
        alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
        for sign in [1, -1]:
            for a in alphas:
                alpha = sign * a
                h_new = repair_vector(h + alpha * direction, n)
                s = official_score(h_new)
                if s < best_local - 1e-13:
                    best_local = s
                    best_alpha = alpha

        if best_alpha != 0.0:
            h = repair_vector(h + best_alpha * direction, n)
            score = official_score(h)

        # Try pair exchanges
        improved = False
        for _ in range(30):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.03, 0.05, 0.1, 0.2]:
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                h_new = repair_vector(h_try, n)
                s = official_score(h_new)
                if s < score - 1e-12:
                    h = h_new
                    score = s
                    improved = True
                    break
            if improved:
                break
        if not improved and best_alpha == 0.0:
            break

    return h, official_score(h)


def slsqp_refine(h, n, time_limit=25.0):
    dx = 2.0 / n
    target_sum = n / 2.0
    lags = corr_lags(n)

    def objective(x):
        return x[n]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[:n]
        theta = x[n]
        corr = np.correlate(h_x, 1.0 - h_x, mode="full")
        return theta - dx * corr

    def ineq_jac(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        j = np.zeros((n_lags, n + 1))
        j[:, n] = 1.0
        for ki, k in enumerate(lags):
            grad_row = compute_gradient_row(h_x, k)
            j[ki, :n] = -dx * grad_row
        return j

    def eq_fun(x):
        return np.sum(x[:n]) - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    initial_score = official_score(h)
    theta0 = initial_score + 1e-5

    x0 = np.concatenate([h, [theta0]])

    cons_list = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        result = optimize.minimize(
            objective, x0, jac=objective_grad,
            method='SLSQP', bounds=bounds,
            constraints=cons_list,
            options={'maxiter': 80, 'ftol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x[:n])):
            h_out = repair_vector(result.x[:n], n)
            s = official_score(h_out)
            if s < initial_score - 1e-10:
                return h_out, s
    except Exception:
        pass

    return h, initial_score
