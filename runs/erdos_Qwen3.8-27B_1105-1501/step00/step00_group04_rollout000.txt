def qp_direction(h, active, M, rho, eta_factor):
    """Solve QP to find descent direction."""
    import cvxpy as cp
    
    n = len(h)
    rows = []
    rhs = []
    
    for p in active:
        g = gradient_for_lag_vec(h, p)
        Yp = np.correlate(h, 1.0 - h, mode='full')[p]
        eta = max(2e-4, eta_factor * rho * np.linalg.norm(g))
        eta = min(eta, 0.1 * max(1.0, M))
        rows.append(g)
        rhs.append((M - Yp) - eta)
    
    if not rows:
        return None
    
    G = np.array(rows)
    rhs_arr = np.array(rhs)
    
    u = cp.Variable(n)
    s = cp.Variable(len(rows))
    
    # Objective: minimize ||u||^2 / 2 + slack penalty
    slack_weight = 10.0
    obj = cp.Minimize(0.5 * cp.sum_squares(u) + slack_weight * cp.sum(s))
    
    cons = [
        u >= -h,
        u <= 1.0 - h,
        cp.sum(u) == 0,
    ]
    
    # Linearized constraints: G_p · u + Y_p <= M - eta_p
    # => G_p · u <= M - Y_p - eta_p = rhs_p
    # With slack: G_p · u <= rhs_p + s_p, s_p >= 0
    for i in range(len(rows)):
        cons.append(G[i] @ u <= rhs_arr[i] + s[i])
    cons.append(s >= 0)
    
    prob = cp.Problem(obj, cons)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    if u.value is None:
        return None
    
    u_val = np.array(u.value).flatten()
    if not np.all(np.isfinite(u_val)):
        return None
    
    # Remove mean to ensure sum = 0
    u_val -= np.mean(u_val)
    
    # Check it's not trivially zero
    if np.max(np.abs(u_val)) < 1e-10:
        return None
    
    return u_val

def select_active(Y, M, tol_frac, max_active):
    tol = tol_frac * M + 1e-4
    active = np.where(Y >= M - tol)[0]
    if len(active) > max_active:
        order = np.argsort(Y[active])[::-1][:max_active]
        active = active[order]
    return list(active)

def solve_qp(h, active, Y, M, slack_weight):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    n = len(h)
    rows = []
    rhs = []
    for p in active:
        g = gradient_for_lag(h, p)
        if np.linalg.norm(g) < 1e-12:
            continue
        rows.append(g)
        rhs.append(Y[p] - M)
    if not rows:
        return None
    G = np.array(rows)
    rhs_arr = np.array(rhs)
    k = len(rows)
    u = cp.Variable(n)
    s = cp.Variable(k)
    obj = cp.Minimize(0.5 * cp.sum_squares(u) + slack_weight * cp.sum(s))
    cons = [u >= -h, u <= 1.0 - h, cp.sum(u) == 0]
    for i in range(k):
        cons.append(G[i] @ u <= rhs_arr[i] + s[i])
    cons.append(s >= 0)
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if u.value is None:
        return None
    u_val = np.array(u.value).flatten()
    if not np.all(np.isfinite(u_val)):
        return None
    u_val -= np.mean(u_val)
    if np.max(np.abs(u_val)) < 1e-12:
        return None
    return u_val

def line_search(h, u, dx):
    n = len(h)
    target = n / 2.0
    alpha_max = 1.0
    for i in range(n):
        if u[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / u[i])
        elif u[i] < -1e-14:
            alpha_max = min(alpha_max, -h[i] / u[i])
    if alpha_max < 1e-12:
        return None, None
    alpha_max *= 0.999
    best_h = None
    best_c = None
    for frac in [1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.06, 0.03]:
        alpha = alpha_max * frac
        if alpha < 1e-13:
            continue
        trial = h + alpha * u
        if trial.min() < -1e-10 or trial.max() > 1.0 + 1e-10:
            continue
        if abs(trial.sum() - target) > 1e-8:
            trial = repair_feasible(trial)
        c = dx * np.max(np.correlate(trial, 1.0 - trial, mode='full'))
        if best_c is None or c < best_c:
            best_c = c
            best_h = trial.copy()
    return best_h, best_c

def pair_optimize_step(h, active, Y, M, dx, rng):
    n = len(h)
    g_comp = np.zeros(n)
    for p in active[:min(len(active), 10)]:
        g_comp += gradient_for_lag(h, p)
    src_pool = np.argsort(g_comp)[-8:]
    dst_pool = np.argsort(g_comp)[:8]
    best_c = evaluate_c5(h)
    best_h = h
    for s_idx in src_pool:
        for d_idx in dst_pool:
            if s_idx == d_idx:
                continue
            if h[s_idx] < 1e-10 or h[d_idx] > 1.0 - 1e-10:
                continue
            max_a = min(h[s_idx], 1.0 - h[d_idx]) * 0.5
            for frac in [1.0, 0.6, 0.3, 0.1]:
                alpha = max_a * frac
                if alpha < 1e-12:
                    continue
                trial = h.copy()
                trial[s_idx] -= alpha
                trial[d_idx] += alpha
                if trial[s_idx] < -1e-10 or trial[d_idx] > 1.0 + 1e-10:
                    continue
                c = dx * np.max(np.correlate(trial, 1.0 - trial, mode='full'))
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = trial.copy()
    return best_h, best_c

def local_optimize(h_init, deadline, dx):
    n = len(h_init)
    h = h_init.copy()
    c = evaluate_c5(h)
    best_h = h.copy()
    best_c = c
    rng = np.random.RandomState(123)
    phases = [
        (0.08, 15, 5.0),
        (0.05, 12, 4.0),
        (0.03, 10, 3.0),
        (0.015, 8, 2.0),
        (0.008, 6, 1.5),
    ]
    phase_idx = 0
    fail_count = 0
    max_fails = 20
    
    while time.monotonic() < deadline - 4.0:
        if fail_count > max_fails:
            if phase_idx < len(phases) - 1:
                phase_idx += 1
            else:
                break
            fail_count = 0
        
        tol_frac, max_active, slack_w = phases[phase_idx]
        Y = compute_overlap(h)
        M = np.max(Y)
        c = dx * M
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        active = select_active(Y, M, tol_frac, max_active)
        if not active:
            fail_count += 5
            if phase_idx < len(phases) - 1:
                phase_idx += 1
            else:
                break
            continue
        
        improved = False
        u = solve_qp(h, active, Y, M, slack_w)
        
        if u is not None:
            h_new, c_new = line_search(h, u, dx)
            if h_new is not None and c_new < c - 1e-11:
                h = h_new
                c = c_new
                fail_count = 0
                improved = True
        
        if not improved and time.monotonic() < deadline - 5.0:
            h_pair, c_pair = pair_optimize_step(h, active, Y, M, dx, rng)
            if c_pair < c - 1e-11:
                h = h_pair
                c = c_pair
                fail_count = 0
                improved = True
        
        if not improved:
            fail_count += 1
            if fail_count % 7 == 0:
                n_pert = rng.integers(3, 10)
                idx = rng.choice(n, n_pert, replace=False)
                h_pert = h + rng.normal(0, 0.02, n_pert)[None, :] * np.eye(n)[idx]
                h_pert = repair_feasible(h_pert)
                c_pert = evaluate_c5(h_pert)
                if c_pert < c:
                    h = h_pert
                    c = c_pert
                    fail_count = 0
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0
    
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).flatten()
        if len(h_init) < 2:
            h_init = None
    except (NameError, AttributeError, TypeError):
        h_init = None
    
    if h_init is None:
        n = 70
        h_init = np.full(n, 0.5)
    
    h_init = repair_feasible(h_init)
    n = len(h_init)
    dx = 2.0 / n
    
    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    
    # Main local optimization
    t1 = min(deadline - 30.0, start_time + budget_s * 0.45)
    if time.monotonic() < t1 - 5.0:
        h1, c1 = local_optimize(h_init, t1, dx)
        if c1 < best_c:
            best_h = h1
            best_c = c1
    
    # Try perturbed starts
    t2 = min(deadline - 15.0, start_time + budget_s * 0.75)
    if time.monotonic() < t2 - 10.0:
        rng = np.random.RandomState(seed)
        candidates = []
        
        for a in [0.7, 0.85, 1.15, 1.4]:
            hp = 0.5 + a * (best_h - 0.5)
            candidates.append(repair_feasible(hp))
        
        for nl in [0.04, 0.08, 0.12]:
            hn = best_h + rng.normal(0, nl, n)
            candidates.append(repair_feasible(hn))
        
        for _ in range(4):
            hb = (rng.rand(n) < 0.5).astype(float)
            n_ones = int(hb.sum())
            target_ones = n // 2
            if n_ones < target_ones:
                zeros = np.where(hb == 0)[0]
                fills = rng.choice(zeros, target_ones - n_ones, replace=False)
                hb[fills] = 1.0
            elif n_ones > target_ones:
                ones = np.where(hb == 1)[0]
                zeros_idx = rng.choice(ones, n_ones - target_ones, replace=False)
                hb[zeros_idx] = 0.0
            candidates.append(hb)
        
        # Smoothed/binary hybrid
        for _ in range(2):
            hsm = 0.5 + 0.4 * np.sin(np.linspace(0, 3 * np.pi, n) + rng.uniform(0, 2 * np.pi))
            candidates.append(repair_feasible(hsm))
        
        t_per = max(8.0, (t2 - time.monotonic()) / len(candidates))
        for cand in candidates:
            if time.monotonic() > t2 - 5.0:
                break
            c_eval = evaluate_c5(cand)
            if c_eval < best_c:
                best_h = cand.copy()
                best_c = c_eval
            sub_dl = min(t2, time.monotonic() + t_per)
            if sub_dl - time.monotonic() > 10.0:
                h_opt, c_opt = local_optimize(cand, sub_dl, dx)
                if c_opt < best_c:
                    best_h = h_opt
                    best_c = c_opt
    
    # Final polish
    if time.monotonic() < deadline - 5.0:
        h_fin, c_fin = local_optimize(best_h, deadline, dx)
        if c_fin < best_c:
            best_h = h_fin
            best_c = c_fin
    
    best_h = repair_feasible(best_h)
    final_c = evaluate_c5(best_h)
    
    if best_h.min() < 0:
        best_h = np.clip(best_h, 0, 1)
        best_h = repair_feasible(best_h)
        final_c = evaluate_c5(best_h)
    
    return (best_h, final_c, n)
