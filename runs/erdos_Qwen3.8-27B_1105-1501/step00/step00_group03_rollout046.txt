def smooth_obj_and_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    # C_k = dx * corr[k] for k = 0, ..., 2n-2
    corr = np.correlate(x, 1.0 - x, mode="full")
    C = corr * dx  # shape (2n-1,)
    Cmax = np.max(C)
    # Stable log-sum-exp
    shifted = beta * (C - Cmax)
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    log_sum = max_shifted + np.log(np.sum(exps))
    obj = Cmax + log_sum / beta
    
    # Weights
    w = exps / np.sum(exps)
    
    # Gradient: df/dx[p] = sum_k w[k] * dC_k/dx[p]
    # dC_k/dx[p] = dx * [(1-x[p-k]) * 1_{0<=p-k<n} - x[p+k] * 1_{0<=p+k<n}]
    # where k = idx - (n-1) for corr index idx
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)  # actual lag
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        # Contribution to gradient from this k
        # For each p: dx * [(1-x[p-k]) if 0<=p-k<n else 0 - x[p+k] if 0<=p+k<n else 0]
        for p in range(n):
            val = 0.0
            pk = p - k
            if 0 <= pk < n:
                val += (1.0 - x[pk])
            pk2 = p + k
            if 0 <= pk2 < n:
                val -= x[pk2]
            grad[p] += wk * dx * val
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(z, n):
    z = np.asarray(z, dtype=float).copy()
    if z.size != n:
        z = np.resize(z, n)
    target = n / 2.0
    lo = -2.0 - float(np.max(z))
    hi = 2.0 - float(np.min(z))
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(z + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    shift = (lo + hi) / 2.0
    h = np.clip(z + shift, 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        if res > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if res > 0:
                d = min(res, 1.0 - h[idx])
                h[idx] += d
                res -= d
            else:
                d = min(-res, h[idx])
                h[idx] -= d
                res += d
            if abs(res) < 1e-14:
                break
    return h


def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)


def eq_constraint(x):
    return np.sum(x) - len(x) / 2.0


def eq_jacobian(x):
    return np.ones(len(x))


def smooth_objective(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    C = corr * dx
    Cmax = float(np.max(C))
    shifted = beta * (C - Cmax)
    max_s = float(np.max(shifted))
    exps = np.exp(shifted - max_s)
    s = float(np.sum(exps))
    if s < 1e-300:
        return Cmax
    return Cmax + np.log(s) / beta


def smooth_gradient(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    C = corr * dx
    Cmax = float(np.max(C))
    shifted = beta * (C - Cmax)
    max_s = float(np.max(shifted))
    exps = np.exp(shifted - max_s)
    s = float(np.sum(exps))
    if s < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = exps / s

    grad = np.zeros(n)
    for ii in range(2 * n - 1):
        k = ii - (n - 1)
        wk = w[ii]
        if wk < 1e-14:
            continue
        p_lo = max(0, k)
        p_hi = min(n, k + n)
        if p_hi > p_lo:
            grad[p_lo:p_hi] += wk * dx * (1.0 - x[p_lo - k:p_hi - k])
        p_lo2 = max(0, -k)
        p_hi2 = min(n, n - k)
        if p_hi2 > p_lo2:
            grad[p_lo2:p_hi2] -= wk * dx * x[p_lo2 + k:p_hi2 + k]
    return grad


def run_smooth_solve(h0, n, beta, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian}]
    try:
        res = minimize(
            smooth_objective, h0, args=(beta,),
            jac=smooth_gradient,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            new_h = project_feasible(res.x, n)
            return new_h
    except Exception:
        pass
    return h0


def symmetric_transform(h):
    n = len(h)
    return 1.0 - h[::-1].copy()
