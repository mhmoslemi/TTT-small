        # p is h-factor in entry k when max(0, k-n+1) <= p <= min(n-1, k)

def _pair_move_delta(h, c, n, p, q, max_iter=40):
    """
    Find optimal delta for moving mass from q to p.
    h[p] += delta, h[q] -= delta.
    Returns optimal delta.
    """
    # Compute D[k] = change in c[k] per unit delta
    # c[k] = sum of h[j] * (1-h[m]) over valid pairs
    # When h[p] increases by delta:
    #   For each k where p is h-factor (partner m=p-k+n-1): c[k] increases by delta*(1-h[m])
    #   For each k where p is 1-h-factor (partner j=p+k-n+1): c[k] decreases by delta*h[j]
    # Similarly for h[q] decreasing by delta (i.e., h[q] changes by -delta):
    #   For each k where q is h-factor: c[k] changes by -delta*(1-h[m])
    #   For each k where q is 1-h-factor: c[k] changes by +delta*h[j]
    
    D = np.zeros(2*n - 1)
    
    # Contribution from h[p] += delta
    for k in range(2*n - 1):

        m = p - k + n - 1
