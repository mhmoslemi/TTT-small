def repair_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
            for i in order:
                if diff <= 1e-14: break
                d = min(h[i], diff)
                h[i] -= d
                diff -= d
        else:
            order = np.argsort(h)
            for i in order:
                if diff >= -1e-14: break
                d = min(1.0 - h[i], -diff)
                h[i] += d
                diff += d
    # Final check
    if abs(h.sum() - target) > 1e-9:
        # Force fix
        h = np.clip(h, 0, 1)
        h[np.argmax(h)] -= (h.sum() - target)
        h = np.clip(h, 0, 1)

def repair_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to decrease sum
            order = np.argsort(-h)
            remaining = diff
            for i in order:
                if remaining < 1e-14:
                    break
                d = min(h[i], remaining)
                h[i] -= d
                remaining -= d
        else:
            # Need to increase sum
            order = np.argsort(h)
            remaining = -diff
            for i in order:
                if remaining < 1e-14:
                    break
                d = min(1.0 - h[i], remaining)
                h[i] += d
                remaining -= d
    return h

def make_near_binary(h, rng=None):
    """Convert to near-binary: n/2 mass spread over n positions"""
    n = len(h)
    target = n / 2.0
    # Sort values, set top floor(n/2) to 1, bottom to 0, middle to fill
    h = repair_feasible(h)
    if n % 2 == 0:
        # Exactly n/2 ones and n/2 zeros
        target_ones = n // 2
        idx = np.argsort(-h)[:target_ones]
        b = np.zeros(n)
        b[idx] = 1.0
        return b
    else:
        # n/2 = (n-1)/2 + 0.5, so (n-1)/2 ones, (n-1)/2 zeros, one 0.5
        target_ones = (n - 1) // 2
        idx = np.argsort(-h)
        b = np.zeros(n)
        b[idx[:target_ones]] = 1.0
        b[idx[target_ones]] = 0.5
        return b
