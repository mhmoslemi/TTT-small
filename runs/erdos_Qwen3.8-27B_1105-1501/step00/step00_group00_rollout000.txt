import numpy as np
from scipy.optimize import linprog
import time

def repair_to_feasible(h):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    if n == 0:
        return h
    h[~np.isfinite(h)] = 0.5
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    s = h.sum()
    diff = s - target
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
            remaining = diff
            for idx in order:
                if remaining <= 0:
                    break
                reduce = min(remaining, h[idx])
                h[idx] -= reduce
                remaining -= reduce
            if remaining > 1e-10:
                h[:] = 0.5
                s = h.sum()
                h[0] = target - s + 0.5
                np.clip(h, 0.0, 1.0, out=h)
        else:
            order = np.argsort(h)
            remaining = -diff
            for idx in order:
                if remaining <= 0:
                    break
                increase = min(remaining, 1.0 - h[idx])
                h[idx] += increase
                remaining -= increase
            if remaining > 1e-10:
                h[:] = 0.5
                s = h.sum()
                h[0] = target - s + 0.5
                np.clip(h, 0.0, 1.0, out=h)
    return h

def evaluate_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_gradients(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag >= 0:
            m = lag
            if m > 0:
                idx_first = np.arange(m, n)
                grads[k, idx_first] += dx * (1.0 - h[idx_first - m])
                idx_second = np.arange(0, n - m)
                grads[k, idx_second] -= dx * h[idx_second + m]
            else:
                idx = np.arange(n)
                grads[k, idx] += dx * (1.0 - h[idx])
                grads[k, idx] -= dx * h[idx]
        else:
            p = -lag
            idx_first = np.arange(0, n - p)
            grads[k, idx_first] += dx * (1.0 - h[idx_first + p])
            idx_second = np.arange(p, n)
            grads[k, idx_second] -= dx * h[idx_second - p]
    return grads

def compute_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def choose_active(h, q=None):
    n = len(h)
    if q is None:
        q = min(15, 2 * n - 1)
    values = compute_lag_values(h)
    idx = np.argsort(values)[::-1]
    active = list(idx[:q])
    center = n - 1
    if center not in active:
        active.append(center)
    return active, values

def solve_tangent_lp(h0, active_indices, trust_radius=0.3):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    values = compute_lag_values(h0)
    grads_all = compute_all_gradients(h0)
    
    k = len(active_indices)
    A_ub = np.zeros((k, n + 1))
    b_ub = np.zeros(k)
    
    for ii, m in enumerate(active_indices):
        g = grads_all[m]
        c_m = values[m]
        kappa = c_m - g.dot(h0)
        A_ub[ii, :n] = g
        A_ub[ii, n] = -1.0
        b_ub[ii] = -kappa
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_radius)
        hi = min(1.0, h0[i] + trust_radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success or res.x is None:
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='interior-point')
        except Exception:
            return None
    
    if not res.success or res.x is None:
        return None
    
    h_new = res.x[:n].copy()
    h_new = repair_to_feasible(h_new)
    return h_new

def pair_line_search(h, i, j, n_samples=15):
    n = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h, evaluate_c5(h)
    
    current_best = h.copy()
    current_best_val = evaluate_c5(h)
    
    if n_samples < 3:
        n_samples = 3
    deltas = np.linspace(lo, hi, n_samples)
    
    for delta in deltas:
        h_try = h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        if h_try[i] < -1e-12 or h_try[i] > 1 + 1e-12 or h_try[j] < -1e-12 or h_try[j] > 1 + 1e-12:
            continue
        val = evaluate_c5(h_try)
        if val < current_best_val - 1e-12:
            current_best_val = val
            current_best = h_try.copy()
    
    return current_best, current_best_val

def random_pair_search(h, rng, n_pairs=40, n_samples=11):
    n = len(h)
    best = h.copy()
    best_val = evaluate_c5(h)
    
    for _ in range(n_pairs):
        i, j = rng.randint(0, n, size=2)

            continue
        candidate, val = pair_line_search(h, i, j, n_samples=n_samples)
        if val < best_val - 1e-12:
            best_val = val
            best = candidate
            h = best.copy()
    
    return best, best_val

def gradient_guided_pair_search(h, rng, n_pairs=30, n_samples=11):
    n = len(h)
    active, values = choose_active(h, q=8)
    grads_all = compute_all_gradients(h)
    
    avg_grad = np.zeros(n)

        avg_grad += grads_all[m]
    
    order = np.argsort(avg_grad)
    candidates = []
    for i in order[:min(10, n)]:
        for j in order[-min(10, n):]:
            if i != j:
                candidates.append((i, j))

    best = h.copy()
    best_val = evaluate_c5(h)
    
    for i, j in candidates[:n_pairs]:
        candidate, val = pair_line_search(best, i, j, n_samples=n_samples)

            best = candidate

def transfer_resolution(h_src, n_target):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_target
    
    h_tgt = np.zeros(n_target)
    src_pos = 0.0
    
    for k in range(n_target):
        t_left = k * dx_tgt
        t_right = (k + 1) * dx_tgt
        area = 0.0
        s = src_pos
        while s < t_right:
            idx = int(int(s / dx_src))
            if idx >= n_src:
                break
            s_left = idx * dx_src
            s_right = s_left + dx_src
            overlap = min(s_right, t_right) - max(s_left, t_left)
            if overlap > 0:
                area += h_src[idx] * overlap
            s = s_right
        h_tgt[k] = area / dx_tgt
    
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    h_tgt = repair_to_feasible(h_tgt)
    return h_tgt
