import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def feasible_delta(h, i, j):
    """For move h[i]+=delta, h[j]-=delta, find feasible range of delta."""
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    return lo, hi

def optimize_pairs(h, max_iters=1000, tol=1e-10, n_trials=20):
    """Local optimization via pairwise zero-sum line search."""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                lo, hi = feasible_delta(h, i, j)
                if hi - lo < 1e-12:
                    continue
                # Try a grid of delta values
                deltas = np.linspace(lo, hi, n_trials)
                best_val = best_c5
                best_d = 0
                for d in deltas:
                    h[i] += d
                    h[j] -= d
                    val = compute_c5(h)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_d = d
                # Also try the reverse
                deltas_rev = np.linspace(-hi, -lo, n_trials)
                for d in deltas_rev:
                    h[i] += d
                    h[j] -= d
                    val = compute_c5(h)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_d = d
                h[i] += best_d
                h[j] -= best_d
                if best_d != 0:
                    best_c5 = best_val
                    improved = True
        if not improved:
            break
    
    return h, best_c5

def find_min_of_max_affine(corr, delta_coeff, lo, hi, n_refine=50):
    """Find min of max_p(corr[p] + delta_coeff[p]*d) for d in [lo, hi]."""
    # Ternary search (works for unimodal/convex functions)
    for _ in range(n_refine):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(corr + m1 * delta_coeff)
        f2 = np.max(corr + m2 * delta_coeff)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    d = (lo + hi) / 2
    return d, np.max(corr + d * delta_coeff)

        c2 = np.correlate(h, 1.0 - h, mode='full')
        h[m] += eps
        G[m] = (c1 - c2) / (2.0 * eps)
    return G

def pair_optimize(h, max_passes=500, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    prev_c5 = best_c5

    for p in range(max_passes):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        if abs(prev_c5 - best_c5) < 1e-15:
            break
        prev_c5 = best_c5

        G = compute_grad_matrix(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max_corr = np.max(corr)
        best_improvement = 1e-15
        best_move = None

        for i in range(n):
            for j in range(i + 1, n):
                lo, hi = feasible_delta(h, i, j)
                if hi - lo < 1e-12:
                    continue
                L = G[i] - G[j]
                if np.max(L) <= 0 and np.min(L) >= 0:
                    continue
                d_opt = ternary_search_delta(corr, L, lo, hi, n_iter=50)
                new_max = np.max(corr + d_opt * L)
                improvement = current_max_corr - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_move = (i, j, d_opt)

        if best_move is None:
            break
        i, j, d = best_move
        h[i] += d
        h[j] -= d
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5

    return h, best_c5

def sa_optimize(h, rng, n_steps=50000, T_max=0.002, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    for step in range(n_steps):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        frac = step / max(1, n_steps - 1)
        T = T_max * (1.0 - frac) ** 2
        if T < 1e-7:
            T = 1e-7
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        lo, hi = feasible_delta(h, i, j)
        if hi - lo < 1e-10:
            continue
        max_d = min(hi, lo + 0.05)
        if max_d - lo < 1e-10:
            max_d = hi
        d = rng.uniform(lo, max_d)
        h[i] += d
        h[j] -= d
        new_c5 = compute_c5(h)
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d

    return best_h, best_c5

def upsample_h(h, new_n):
    n = len(h)
    new_pos = (np.arange(new_n) + 0.5) * n / new_n
    orig_pos = np.arange(n) + 0.5
    new_h = np.interp(new_pos, orig_pos, h)
    new_h = np.clip(new_h, 0.0, 1.0)
    target = new_n / 2.0
    diff = target - new_h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = new_h < 1.0
        else:
            mask = new_h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            new_h[mask] += diff / cnt
            new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def fix_sum_exact(h, n):
    h = h.copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.asarray(initial_h_values, dtype=np.float64)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Pair-based local optimization from initial
    time_limit_1 = budget_s * 0.25
    h, c5 = pair_optimize(h, max_passes=500, time_limit=time_limit_1 - (time.time() - start_time), start_time=start_time)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SA + local refinement iterations
    sa_end_time = budget_s * 0.65
    n_sa_rounds = 8
    for r in range(n_sa_rounds):
        elapsed = time.time() - start_time
        if elapsed > sa_end_time:
            break
        remaining = sa_end_time - elapsed
        if remaining < 2:
            break

        # SA round
        n_steps = min(40000, max(3000, int(remaining * 0.4 / 0.00003)))
        h, c5 = sa_optimize(h, rng, n_steps=n_steps, T_max=0.002,
                            time_limit=sa_end_time - (time.time() - start_time), start_time=start_time)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Local refine after SA
        h, c5 = pair_optimize(h, max_passes=200,
                              time_limit=sa_end_time - (time.time() - start_time), start_time=start_time)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Try higher resolution
    if time.time() - start_time < budget_s * 0.80:
        for new_n in [84, 120, 168]:
            if time.time() - start_time > budget_s * 0.92:
                break
            h_up = upsample_h(best_h, new_n)
            h_up = fix_sum_exact(h_up, new_n)
            t_limit = budget_s - (time.time() - start_time) - 5
            if t_limit < 5:
                break
            h_up, c5_up = pair_optimize(h_up, max_passes=300, time_limit=t_limit * 0.6, start_time=start_time)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
            # Quick SA on the higher-res solution
            if time.time() - start_time < budget_s * 0.90:
                t_rem = budget_s * 0.90 - (time.time() - start_time)
                if t_rem > 3:
                    h_up, c5_up = sa_optimize(h_up, rng, n_steps=15000, T_max=0.001,
                                              time_limit=t_rem, start_time=start_time)
                    if c5_up < best_c5:
                        best_c5 = c5_up
                        best_h = h_up.copy()
                    if time.time() - start_time < budget_s * 0.92:
                        t_rem2 = budget_s * 0.92 - (time.time() - start_time)
                        if t_rem2 > 2:
                            h_up, c5_up = pair_optimize(h_up, max_passes=100,
                                                        time_limit=t_rem2, start_time=start_time)
                            if c5_up < best_c5:
                                best_c5 = c5_up
                                best_h = h_up.copy()

    # Final: also try a different random start for diversity
    if time.time() - start_time < budget_s * 0.75:
        n_start = len(h0)
        # Create a randomized version
        h_rand = np.clip(h0 + rng.normal(0, 0.1, n_start), 0.0, 1.0)
        h_rand = fix_sum_exact(h_rand, n_start)
        h_rand, c5_rand = pair_optimize(h_rand, max_passes=300,
                                        time_limit=budget_s * 0.85 - (time.time() - start_time),
                                        start_time=start_time)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()

    # Ensure final solution is feasible
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum_exact(best_h, n_final)
    final_c5 = compute_c5(best_h)

    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
