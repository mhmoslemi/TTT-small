    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        n0 = 48
        h_init = np.zeros(n0)
        h_init[::2] = 1.0

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(best_h)
    tsum = n / 2.0
    print(f"Start: n={n}, C5={best_c5:.6f}")

    # Phase 1: Mode optimization with increasing M
    for M in [4, 6, 8, 10, 12, 16, 20, 24]:
        if time.time() - t0 > time_limit:
            break
        B = build_modes(n, M, seed=seed)

        # DE phase
        scale = max(0.1, 0.4 * (8.0 / M))
        bounds = [(-scale, scale)] * M
        popsize = max(8, min(20, 40 // M + 8))
        try:
            res = differential_evolution(
                mode_obj, bounds, args=(best_h, B, tsum),
                seed=seed, maxiter=200, popsize=popsize,
                tol=1e-8, mutation=(0.3, 1.2), recombination=0.8,
                polish=False
            )
        except Exception:
            res = None

        if res is not None:
            h_c = project_to_box_sum(best_h + B @ res.x, tsum)
            c5_c = evaluate_c5(h_c)
            if c5_c < best_c5:
                best_h = h_c
                best_c5 = c5_c
                print(f"  DE M={M}: C5={best_c5:.6f} t={time.time()-t0:.0f}s")

        # NM polish from DE result
        if time.time() - t0 < time_limit:
            z0 = res.x if res is not None else np.zeros(M)
            try:
                res2 = minimize(
                    mode_obj, z0, args=(best_h, B, tsum),
                    method='Nelder-Mead',
                    options={'maxiter': 30000, 'xatol': 1e-11, 'fatol': 1e-13}
                )
                h_c = project_to_box_sum(best_h + B @ res2.x, tsum)
                c5_c = evaluate_c5(h_c)
                if c5_c < best_c5:
                    best_h = h_c
                    best_c5 = c5_c
                    print(f"  NM M={M}: C5={best_c5:.6f} t={time.time()-t0:.0f}s")
            except Exception:
                pass

        # Simulated annealing
        if time.time() - t0 < time_limit:
            rng = np.random.RandomState(seed + 7777)
            z0 = res.x if res is not None else np.zeros(M)
            z_best, val_best = anneal_mode(
                z0, best_h, B, tsum, rng, n_iter=8000
            )
            h_c = project_to_box_sum(best_h + B @ z_best, tsum)
            c5_c = evaluate_c5(h_c)
            if c5_c < best_c5:
                best_h = h_c
                best_c5 = c5_c
                print(f"  SA M={M}: C5={best_c5:.6f} t={time.time()-t0:.0f}s")

        seed += 1

    # Phase 2: Coordinate local search
    if time.time() - t0 < time_limit:
        best_h, best_c5 = local_coord_search(
            best_h, tsum, max_rounds=40, time_limit=time_limit, t_start=t0
        )
        if best_c5 < 0.381:
            print(f"  CoordSearch: C5={best_c5:.6f}")

    # Phase 3: Try different n values
    n_list = [48, 50, 54, 60, 96, 100, 120]
    for n_new in n_list:
        if time.time() - t0 > time_limit * 0.92:
            break
        if n_new == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_box_sum(h_interp, n_new / 2.0)
        c5_interp = evaluate_c5(h_interp)

        # Mode optimization at new resolution
        B_hi = build_modes(n_new, 8, seed=seed + 5000)
        scale_hi = 0.15
        bounds_hi = [(-scale_hi, scale_hi)] * 8
        try:
            res_hi = differential_evolution(
                mode_obj, bounds_hi, args=(h_interp, B_hi, n_new / 2.0),
                seed=seed, maxiter=100, popsize=12, tol=1e-7, polish=False
            )
            h_c = project_to_box_sum(h_interp + B_hi @ res_hi.x, n_new / 2.0)
            c5_c = evaluate_c5(h_c)
        except Exception:
            h_c = h_interp
            c5_c = c5_interp

        # NM polish
        if time.time() - t0 < time_limit * 0.92:
            z0 = res_hi.x if 'res_hi' in dir() and res_hi is not None else np.zeros(8)
            try:
                res_hi2 = minimize(
                    mode_obj, z0, args=(h_c, B_hi, n_new / 2.0),
                    method='Nelder-Mead',
                    options={'maxiter': 20000, 'xatol': 1e-11, 'fatol': 1e-13}
                )
                h_c = project_to_box_sum(h_c + B_hi @ res_hi2.x, n_new / 2.0)
                c5_c = evaluate_c5(h_c)
            except Exception:
                pass

        if c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c
            n = n_new
            tsum = n_new / 2.0
            print(f"  n={n_new}: C5={best_c5:.6f} t={time.time()-t0:.0f}s")

    # Phase 4: Second round of coord search on best
    if time.time() - t0 < time_limit:
        best_h, best_c5 = local_coord_search(
            best_h, tsum, max_rounds=30, time_limit=time_limit, t_start=t0
        )

    # Final: ensure feasibility
    best_h = project_to_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    print(f"Final: n={len(best_h)}, C5={best_c5:.6f}, time={time.time()-t0:.0f}s")

    return (best_h, best_c5, len(best_h))
