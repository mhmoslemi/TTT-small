                    delta_int = (a[q] - a[p]) / denom
                    if 0 < delta_int < delta_max:
                        test_deltas.append(delta_int)
        
        for dt in test_deltas:
            if dt <= 0 or dt > delta_max:
                continue
            val = np.max(a + dt * d)
            if val < best_val - 1e-14:
                best_val = val
                best_delta = dt
        
        if best_delta > 0 and best_val < best_c - 1e-14:
            # Verify with full correlation
            h_try = h.copy()
            h_try[i] += best_delta
            h_try[j] -= best_delta
            h_try = repair_feasible(h_try, n)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            c_try = float(np.max(corr_try) * dx)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
    
    return best_h, best_c, corr
