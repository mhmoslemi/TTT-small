from scipy.optimize import minimize
import time

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_objective(h, dx, tau):
    corr = compute_corrs(h) * dx
    max_corr = np.max(corr)
    # Numerically stable log-sum-exp
    shifted = tau * (corr - max_corr)
    log_sum = max_corr * tau + np.log(np.sum(np.exp(shifted)))
    return log_sum / tau

def smooth_gradient(h, dx, tau):
    n = len(h)
    corr = compute_corrs(h) * dx
    max_corr = np.max(corr)
    shifted = tau * (corr - max_corr)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-15:
            continue
        offset = k - (n - 1)
        # First term

        j1e = min(n, n - offset)
        if j1e > j1s:

        # Second term
        j2s = max(0, offset)
        j2e = min(n, n + offset)
        if j2e > j2s:
            grad[j2s:j2e] += weights[k] * (-h[j2s-offset:j2e-offset])

                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 2: Try other n values
    n_tried = {best_n} if best_h is not None else set()
    n_candidates = [47, 64, 96, 128, 192, 256]
    for n in n_candidates:
        if remaining() < 40:
            break
        if n in n_tried:
            continue
        n_tried.add(n)
        dx = 2.0 / n
        for s in range(3):
            if remaining() < 20:
                break
            h0 = make_initial_h(n, rng)
            t_alloc = min(remaining() * 0.15, 80)
            h_opt, c5_opt = optimize_h(h0, n, dx, max_time=t_alloc)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
            if remaining() > 15 and c5_opt < best_c5 + 0.005:
                h_cd, c5_cd = coord_descent(h_opt, n, dx, max_sweeps=20,
                                            time_limit=min(30, remaining() * 0.3))
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n

    # Phase 3: Final refinement on best solution
    if best_h is not None and remaining() > 15:
        dx_best = 2.0 / best_n
        h_fd, c5_fd = coord_descent(best_h.copy(), best_n, dx_best,
                                     max_sweeps=50, time_limit=remaining() * 0.6)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_fd.copy()

        if remaining() > 30:
            bounds = [(0.0, 1.0)] * best_n
            M = 200.0
            for tau in [500, 2000, 10000]:
                if remaining() < 10:
                    break
                res = minimize(obj_func, best_h, jac=grad_func,
                             args=(dx_best, tau, M, best_n),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12})
                val = compute_c5(res.x, dx_best)
                if val < best_c5:
                    best_c5 = val
                    best_h = res.x.copy()

    return best_h, best_c5, best_n
