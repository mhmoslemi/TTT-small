```python
def repair_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    y = np.clip(y, 0.0, 1.0)
    s = np.sum(y)
    if abs(s - target) < 1e-10:
        return y
    # Need to adjust sum
    # If s > target, reduce; if s < target, increase
    diff = target - s
    if diff > 0:
        # Need to add mass to elements that can accept it (h[i] < 1)
        slack = 1.0 - y
        # Distribute proportionally
        total_slack = np.sum(slack)
        if total_slack < 1e-15:
            return y  # can't fix
        y = y + diff * slack / total_slack
    else:
        # Need to remove mass from elements that can give it (h[i] > 0)
        avail = y
        total_avail = np.sum(avail)
        if total_avail < 1e-15:
            return y
        y = y + diff * avail / total_avail
    y = np.clip(y, 0.0, 1.0)
    # Fine adjustment
    s = np.sum(y)
    diff = target - s
    if abs(diff) > 1e-12:
        # Distribute to elements not at boundary
        if diff > 0:
            idx = np.where(y < 1.0 - 1e-10)[0]
        else:
            idx = np.where(y > 1e-10)[0]
        if len(idx) > 0:
            y[idx] += diff / len(idx)
            y = np.clip(y, 0.0, 1.0)
    return y

    active_lags: list of offset values (positive or negative) for active lags.
    Returns gradient vector of length n."""
```

```python
import numpy as np
import time

def repair_feasible(z, n):
    """Project z onto {h: 0<=h_i<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    # Binary search for lambda such that sum(clip(z - lambda, 0, 1)) = target
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-12:
            break
    
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjust
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute to non-boundary elements
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = free.sum()
        if nfree > 0:
            h[free] += diff / nfree
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx, c * dx

def smooth_max_grad(h, alpha):
    """Compute gradient of smooth max approximation of the correlation values."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(alpha * c_shifted)
    w /= w.sum()
    
    # Gradient = sum_k w_k * grad(c_k)
    g = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        l = k - center  # lag
        if l == 0:
            g += w[k] * (1.0 - 2.0 * h)
        elif l > 0:
            g[l:] += w[k] * (1.0 - h[:n-l])
            g[:n-l] += w[k] * (-h[l:])
        else:
            r = -l
            g[:n-r] += w[k] * (1.0 - h[r:])
            g[r:] += w[k] * (-h[:n-r])
    return g, np.max(c) * (2.0 / n)
```

```python
import numpy as np
import time

def repair_feasible(z, n):
    """Project z onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Use bisection on lambda
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    """Return (c5_value, scaled_correlation_array)."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c * dx
    return c_scaled.max(), c_scaled

def compute_gradient(h, alpha):
    """Compute gradient of smooth max approximation.
    Returns (gradient_vector, current_smooth_value)."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    c_shift = c - c.max()
    exp_c = np.exp(alpha * c_shift)
    w = exp_c / exp_c.sum()
    
    g = np.zeros(n)
    center = n - 1
    
    # Center lag (d=0)
    if w[center] > 1e-12:
        g += w[center] * (1.0 - 2.0 * h)
    
    # Positive lags
    for d in range(1, n):
        wk = w[center + d]
        if wk > 1e-12:
            g[d:] += wk * (1.0 - h[:n-d])
            g[:n-d] += wk * (-h[d:])
        wk2 = w[center - d]
        if wk2 > 1e-12:
            g[:n-d] += wk2 * (1.0 - h[d:])
            g[d:] += wk2 * (-h[:n-d])
    
    smooth_val = (1.0 / alpha) * np.log(exp_c.sum()) + c.max()
    return g, smooth_val
```

1. **Binarization**: Project to nearest binary point (round and repair). This can sometimes find a good binary solution.

2. **Pair swap**: For a near-binary vector, try swapping h[i] and h[j] (or adjusting them) to reduce the maximum.

Let me also think about the pair swap more carefully. If h[i] = a and h[j] = b (with a > 0 and b < 1), we can transfer mass: h[i] → a-t, h[j] → b+t for t ∈ [0, min(a, 1-b)]. This preserves the sum constraint. We want to choose i, j, t to reduce the maximum.

The change in correlation at lag d due to this swap:
Δc_d = -t * (partial c_d / partial h[i]) + t * (partial c_d / partial h[j])
= t * (partial c_d / partial h[j] - partial c_d / partial h[i])

So the swap helps if for the active lags, the gradient difference favors the swap.

This is essentially the same as a 2-coordinate gradient step. The global approach (smooth max gradient) already captures this, but a focused 2-coordinate search can be more effective near the optimum.

```python
def compute_gradient_active(h, k_top=5):
    """Compute gradient using top-K active lags (uniform weight)."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top K indices
    top_idx = np.argsort(c)[-k_top:]
    w = np.zeros(2*n-1)
    w[top_idx] = 1.0 / k_top
    
    g = np.zeros(n)
    center = n - 1
    
    for k_idx in top_idx:
        d = k_idx - center
        if d == 0:
            g += (1.0 - 2.0 * h) / k_top
        elif d > 0:
            g[d:] += (1.0 - h[:n-d]) / k_top
            g[:n-d] += (-h[d:]) / k_top
        else:
            r = -d
            g[:n-r] += (1.0 - h[r:]) / k_top
            g[r:] += (-h[:n-r]) / k_top
    
    return g
```

import numpy as np
import time

def project_onto_feasible(z, n):
    """Project z onto {h in R^n : 0 <= h_i <= 1, sum(h) = n/2}."""
    target = 0.5 * n
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Fix any residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)
