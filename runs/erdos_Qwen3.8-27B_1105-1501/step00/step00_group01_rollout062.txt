import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp

_SMOOTH_BETA = 10.0


def evaluate_h(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float("inf")
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def project_h(v):
    v = np.asarray(v, dtype=float).ravel().copy()
    n = v.size
    if n == 0:
        return v
    S = 0.5 * float(n)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.5)

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(v - lam, 0.0, 1.0)

    diff = S - float(np.sum(h))
    if diff > 1e-13:
        order = np.argsort(-(1.0 - h))
    elif diff < -1e-13:
        order = np.argsort(-h)
    else:
        order = np.arange(n)

    i = 0
    while abs(diff) > 1e-13 and i < n:
        idx = order[i]
        if diff > 0.0:
            step = min(diff, 1.0 - h[idx])
        else:
            step = min(diff, -h[idx])
        if step != 0.0:
            h[idx] += step
            diff = S - float(np.sum(h))
        i += 1

    h = np.clip(h, 0.0, 1.0)
    return h


def update_best(best_h, best_c, h):
    if h is None:
        return best_h, best_c
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0 or not np.all(np.isfinite(h)):
        return best_h, best_c
    hp = project_h(h)
    c = evaluate_h(hp)
    if c < best_c:
        return hp.copy(), float(c)
    return best_h, best_c


def lag_values_unscaled(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    c = np.empty(2 * n - 1, dtype=float)
    off = n - 1
    c[off] = float(np.dot(h, 1.0 - h))
    for d in range(1, n):
        c[off + d] = float(np.dot(h[d:], 1.0 - h[:n - d]))
        c[off - d] = float(np.dot(h[:n - d], 1.0 - h[d:]))
    return c


def grad_lag(h, k):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if k == 0:
        return 1.0 - 2.0 * h

    g = np.zeros(n, dtype=float)
    if k > 0:
        d = k
        g[d:] += 1.0
        g[:n - d] -= h[d:]
        g[d:] -= h[:n - d]
    else:
        d = -k
        g[:n - d] += 1.0
        g[d:] -= h[:n - d]
        g[:n - d] -= h[d:]
    return g


def smooth_loss_obj(x):
    beta = float(_SMOOTH_BETA)
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return 0.0
    T = lag_values_unscaled(x) * (2.0 / float(n))
    m = float(np.max(T))
    if beta <= 0.0:
        return m
    lse = float(logsumexp(beta * (T - m)))
    return float(m + lse / beta)


def smooth_grad_obj(x):
    beta = float(_SMOOTH_BETA)
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x.copy()
    T = lag_values_unscaled(x) * (2.0 / float(n))
    if beta <= 0.0:
        return np.zeros(n, dtype=float)

    m = float(np.max(T))
    z = beta * (T - m)
    lse = float(logsumexp(z))
    w = np.exp(z - lse)

    g = np.zeros(n, dtype=float)
    del off = n - 1
    for idx in range(2 * n - 1):
        W = float(w[idx])
        if W > 1e-15:
            g += W * grad_lag(x, idx - (n - 1))
    g *= (2.0 / float(n))
    return g


def cons_sum(x):
    return float(np.sum(x) - 0.5 * len(x))


def cons_sum_jac(x):
    return -np.ones(len(x), dtype=float)


def slsqp_optimize(h0, beta, maxiter):
    global _SMOOTH_BETA
    h0 = project_h(h0)
    n = h0.size
    if n < 2 or maxiter <= 0:
        return h0

    _SMOOTH_BETA = float(beta)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
    cons = ({'type': 'eq', 'fun': cons_sum, 'jac': cons_sum_jac},)

    try:
        res = minimize(
            smooth_loss_obj,
            h0,
            jac=smooth_grad_obj,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-12, 'disp': False}
        )
        if res is not None and hasattr(res, 'x') and res.x is not None:
            x = res.x
        else:
            x = h0
    except Exception:
        x = h0

    return project_h(x)


def lp_polish_step(h, delta_max):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n < 2:
        return None, np.inf

    dx = 2.0 / float(n)
    T = lag_values_unscaled(h) * dx
    C = float(evaluate_h(h))
    tm = float(np.max(T))
    if tm > C:
        C = tm

    slacks = C - T
    slacks = np.maximum(slacks, 0.0)

    m = 2 * n - 1
    nv = n + 1
    c = np.zeros(nv, dtype=float)
    c[n] = -1.0

    A = np.empty((m, nv), dtype=float)
    b = np.empty(m, dtype=float)
    for idx in range(m):
        k = idx - (n - 1)
        g = grad_lag(h, k) * dx
        A[idx, :n] = g
        A[idx, n] = 1.0
        b[idx] = float(slacks[idx])

    Aeq = np.zeros((1, nv), dtype=float)
    Aeq[0, :n] = 1.0
    beq = np.zeros(1, dtype=float)

    bounds = []
    for i in range(n):
        lo = max(-float(delta_max), -float(h[i]))
        hi = min(float(delta_max), 1.0 - float(h[i]))
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, max(1e-12, float(C))))

    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq, bounds=bounds)
        except Exception:
            return None, np.inf

    if res is None or not hasattr(res, 'x') or res.x is None:
        return None, np.inf
    x = np.asarray(res.x, dtype=float)
    if not np.all(np.isfinite(x)):
        return None, np.inf

    eps = float(x[n])
    if eps <= 1e-11:
        return None, np.inf

    delta = x[:n]
    for alpha in (0.01, 0.03, 0.1, 0.25, 0.5, 0.75, 1.0):
        h2 = project_h(h + alpha * delta)
        c2 = evaluate_h(h2)
        if c2 < C - 1e-13:
            return h2, c2
    return None, np.inf


def lp_polish(h, max_steps=30, delta_list=(0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002), stop_time=np.inf):
    h = project_h(h)
    best_c = float(evaluate_h(h))

    for dm in delta_list:
        steps = int(max_steps)
        if dm < 0.01:
            steps = max(8, int(0.5 * max_steps))
        for it in range(steps):
            if time.time() >= float(stop_time):
                return h, best_c
            h2, c2 = lp_polish_step(h, float(dm))
            if h2 is None:
                break
            if c2 < best_c - 1e-14:
                h = h2
                best_c = c2
            else:
                break
    return h, best_c


def set_binary_count(vals, target):
    vals = np.asarray(vals, dtype=float).ravel().copy()
    cnt = int(round(float(np.sum(vals))))
    if cnt > target:
        idx = np.flatnonzero(vals > 0.5)[:cnt - target]
        vals[idx] = 0.0
    elif cnt < target:
        idx = np.flatnonzero(vals < 0.5)[:target - cnt]
        vals[idx] = 1.0
    return vals


def fallback_initial(n):
    n = int(n)
    vals = np.empty(n, dtype=float)
    for i in range(n):
        vals[i] = float((int(i).bit_count() & 1))

    if n % 2 == 1:
        idx = n // 2
        vals[idx] = 0.0
        target = (n - 1) // 2
        cnt = int(round(float(np.sum(vals))))
        if cnt > target:
            flip = np.flatnonzero(vals > 0.5)[:cnt - target]
            vals[flip] = 0.0
        elif cnt < target:
            flip = np.flatnonzero(vals < 0.5)[:target - cnt]
            vals[flip] = 1.0
        vals[idx] = 0.5
    else:
        vals = set_binary_count(vals, n // 2)

    return project_h(vals)


def make_binary_near(v, n):
    n = int(n)
    v = np.asarray(v, dtype=float).ravel()
    if v.size == 0:
        v = fallback_initial(n)
    v = np.where(np.isfinite(v), v, 0.5)

    if v.size != n:
        old = np.linspace(0.0, 1.0, v.size)
        new = np.linspace(0.0, 1.0, n)
        v = np.interp(new, old, v)

    order = np.argsort(-v, kind='stable')
    b = np.zeros(n, dtype=np.int32)
    b[order[:n // 2]] = 1
    return b


def c5_binary(h):
    h = np.asarray(h, dtype=np.int32).ravel()
    n = h.size
    if n == 0:
        return float("inf")
    mx = int(np.max(np.correlate(h, 1 - h, mode='full')))
    return float(mx) * (2.0 / float(n))


def adjust_binary(h, rng=None):
    h = np.asarray(h, dtype=np.int32).ravel().copy()
    n = h.size
    target = n // 2
    cnt = int(np.sum(h))

    if cnt > target:
        idx = np.flatnonzero(h == 1)
        if rng is not None and idx.size > 0:
            idx = rng.permutation(idx)
        for k in range(cnt - target):
            h[idx[k]] = 0
    elif cnt < target:
        idx = np.flatnonzero(h == 0)
        if rng is not None and idx.size > 0:
            idx = rng.permutation(idx)
        for k in range(target - cnt):
            h[idx[k]] = 1

    return h


def binary_hill_climb(h0, max_passes=200, stop_time=np.inf, rng=None):
    h = adjust_binary(h0, rng)
    n = h.size
    dx = 2.0 / float(n)
    c = c5_binary(h)
    best = h.copy()
    best_c = c

    for _ in range(int(max_passes)):
        if time.time() >= float(stop_time):
            break
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        best_neighbor = None
        best_neighbor_c = best_c

        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                nc = int(np.max(np.correlate(h, 1 - h, mode='full'))) * dx
                if nc < best_neighbor_c - 1e-15:
                    best_neighbor_c = nc
                    best_neighbor = h.copy()
                h[i] = 1
                h[j] = 0

        if best_neighbor is not None:
            h = best_neighbor
            best = best_neighbor
            best_c = best_neighbor_c
        else:
            break

    return best, best_c


def binary_simulated_annealing(h0, steps=20000, T_start=0.05, T_end=0.001, stop_time=np.inf, rng=None):
    h = adjust_binary(h0, rng)
    n = h.size
    dx = 2.0 / float(n)
    c = c5_binary(h)
    best = h.copy()
    best_c = c

    steps = int(steps)
    if steps <= 0:
        return best, best_c

    for it in range(steps):
        if time.time() >= float(stop_time):
            break
        frac = it / float(steps)
        T = float(T_start) * (float(T_end) / float(T_start)) ** frac

        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break

        oi = int(rng.integers(ones.size))
        oz = int(rng.integers(zeros.size))
        i = int(ones[oi])
        j = int(zeros[oz])

        h[i] = 0
        h[j] = 1
        nc = int(np.max(np.correlate(h, 1 - h, mode='full'))) * dx
        delta = nc - c

        accept = False
        if delta < 0.0:
            accept = True
        elif T > 1e-15:
            try:
                accept = float(rng.random()) < float(np.exp(-delta / T))
            except Exception:
                accept = False

        if accept:
            c = nc
            if c < best_c - 1e-15:
                best = h.copy()
                best_c = c
        else:
            h[i] = 1
            h[j] = 0

    return best, best_c


def binary_search_phase(h0, rng, stop_time):
    best_h = None
    best_c = np.inf
    if h0 is None:
        return best_h, best_c

    bases = []
    bases.append(h0)
    bases.append(h0[::-1].copy())
    bases.append((1.0 - h0).copy())
    bases.append(fallback_initial(64))
    bases.append(fallback_initial(62))

    for n in (64, 62, 60):

            b = make_binary_near(base, n)
            b = adjust_binary(b, rng)

            per_stop = min(float(stop_time), time.time() + 25.0)
            bh, bc = binary_hill_climb(b, max_passes=200, stop_time=per_stop, rng=rng)
            best_h, best_c = update_best(best_h, best_c, bh.astype(float))

            if time.time() < float(stop_time) - 10.0:
                per_stop2 = min(float(stop_time), time.time() + 10.0)

                bh2, bc2 = binary_simulated_annealing(
                    bh2,
                    steps=20000,
                    T_start=0.05,
                    T_end=0.001,
                    stop_time=per_stop2,
                    rng=rng
                )
                best_h, best_c = update_best(best_h, best_c, bh2.astype(float))

        if time.time() >= float(stop_time):
            break

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1100.0
    if budget_s < 5.0:
        budget_s = 5.0

    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(int(seed))

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        h0 = fallback_initial(63)

    if h0.size == 0:
        h0 = fallback_initial(63)
    h0 = project_h(h0)

    best_h, best_c = update_best(None, np.inf, h0)

    variants = []
    variants.append(h0.copy())
    variants.append(h0[::-1].copy())
    variants.append((1.0 - h0).copy())
    variants.append((1.0 - h0[::-1]).copy())

    for v in variants:
        best_h, best_c = update_best(best_h, best_c, v)

    reserve = 150.0 if budget_s > 500.0 else 60.0
    stop = deadline - reserve

    for v in variants:
        if time.time() >= stop:
            break
        h, c = lp_polish(v, max_steps=30, stop_time=stop)
        best_h, best_c = update_best(best_h, best_c, h)

    bases = [best_h]
    for v in variants:
        bases.append(v)

    beta_schedule = (2.0, 5.0, 10.0, 20.0, 50.0, 100.0)
    used = 0
    for base in bases:
        if time.time() >= deadline - 80.0:
            break
        if base is None:
            continue
        used += 1
        if used > 8:
            break

        h = project_h(base)
        c = evaluate_h(h)

        for beta in beta_schedule:
            if time.time() >= deadline - 60.0:
                break
            maxiter = 80 if beta <= 10.0 else 40
            h2 = slsqp_optimize(h, beta, maxiter)
            c2 = evaluate_h(h2)
            best_h, best_c = update_best(best_h, best_c, h2)

            if c2 <= c + 1e-8:
                h = h2
                c = c2
            else:
                h = project_h(h)

        if time.time() < deadline - 30.0:
            h3, c3 = lp_polish(h, max_steps=20, stop_time=deadline - 20.0)
            best_h, best_c = update_best(best_h, best_c, h3)

    if time.time() < deadline - 120.0 and best_h is not None:
        reshape_stop = deadline - 120.0
        for _ in range(6):
            if time.time() >= reshape_stop:
                break
            noise = 0.05 * rng.normal(size=best_h.size)
            h2 = project_h(best_h + noise)
            h2 = slsqp_optimize(h2, 20.0, 40)
            if time.time() >= reshape_stop:
                break
            h3, c3 = lp_polish(h2, max_steps=15, stop_time=reshape_stop)
            best_h, best_c = update_best(best_h, best_c, h3)

    if time.time() < deadline - 120.0:
        bb, bc = binary_search_phase(h0, rng, deadline - 10.0)
        best_h, best_c = update_best(best_h, best_c, bb)

    if time.time() < deadline - 5.0 and best_h is not None:
        small_deltas = (0.1, 0.05, 0.02, 0.01, 0.005, 0.002)
        h4, c4 = lp_polish(best_h, max_steps=20, delta_list=small_deltas, stop_time=deadline - 1.0)
        best_h, best_c = update_best(best_h, best_c, h4)

    if best_h is None:
        best_h = project_h(fallback_initial(63))
        best_c = evaluate_h(best_h)

    best_h = np.asarray(best_h, dtype=float).ravel()
    best_c = float(best_c)
    return best_h, best_c, int(best_h.size)
