```python
    c = np.max(corr)
    shifted = (corr - c) / t
    return c + t * np.log(np.sum(np.exp(shifted)))
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def local_search_binary(h, max_passes=1000, time_limit=None, start_time=None):
    """Local search by swapping 1s and 0s."""
    n = len(h)
    current_c5 = compute_c5(h)
    
    for pass_num in range(max_passes):
        if time_limit and start_time and time.time() - start_time > time_limit:
            break
        
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5(h)
                if new_c5 < current_c5 - 1e-12:
                    current_c5 = new_c5
                    improved = True
                else:
                    h[i], h[j] = h[i], h[j]
        
        if not improved:
            break
    
    return h, current_c5
```

```python
def project(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo = target - np.sum(h) - 1.0  # worst case: need to increase a lot
    hi = target - np.sum(h) + 1.0  # worst case: need to decrease a lot
    
    # Actually, let's think about bounds:
    # sum(clip(h + lambda, 0, 1)) is a continuous, non-decreasing function of lambda
    # At lambda -> -inf: sum -> 0
    # At lambda -> +inf: sum -> n
    # We need sum = n/2
    
    lo = -2.0
    hi = 2.0
    
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h + (lo + hi) / 2, 0, 1)
```
