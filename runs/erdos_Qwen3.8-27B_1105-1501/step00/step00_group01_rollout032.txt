import numpy as np
from scipy.optimize import minimize

def compute_result(h):
    """Compute the correlation results."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def compute_gradient(h, k_weights):
    """Compute weighted sum of gradients of result[k] for each k.
    k_weights has length 2n-1, one weight per k.
    """
    n = len(h)
    grad = np.zeros(n)
    
    for k in range(2*n - 1):
        w = k_weights[k]
        if abs(w) < 1e-12:
            continue
        
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        
        # Contribution from h[m] as h[i]: m in [i_lo, i_hi], grad[m] += w * (1 - h[n-1-k+m])
        # j = n-1-k+i, so when i=m, j = n-1-k+m
        j_lo = n - 1 - k + i_lo
        j_hi = n - 1 - k + i_hi
        # Need 0 <= j < n for valid index
        valid = (j_lo >= 0) & (j_hi < n)
        # Actually j_lo and j_hi should always be valid if i_lo, i_hi are valid
        # Let's check: j_lo = n-1-k+i_lo. Since i_lo >= k-n+1, j_lo >= n-1-k+k-n+1 = 0. ✓
        # j_hi = n-1-k+i_hi. Since i_hi <= k, j_hi <= n-1. ✓
        
        grad[i_lo:i_hi+1] += w * (1.0 - h[j_lo:j_hi+1])
        
        # Contribution from h[m] as h[j]: j=m, i = j - (n-1-k) = m - (n-1-k) = m + k - n + 1
        # Need i_lo <= i <= i_hi, i.e., i_lo <= m+k-n+1 <= i_hi
        # So i_lo - k + n - 1 <= m <= i_hi - k + n - 1
        m_lo = i_lo - k + n - 1
        m_hi = i_hi - k + n - 1
        # Need 0 <= m < n
        m_lo = max(0, m_lo)
        m_hi = min(n - 1, m_hi)
        
        if m_lo <= m_hi:
            i_lo_m = m_lo + k - n + 1  # corresponding i value
            i_hi_m = m_hi + k - n + 1
            grad[m_lo:m_hi+1] += w * (-h[i_lo_m:i_hi_m+1])
    
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    h = initial_h_values.copy().astype(float)
    n_initial = len(h)
    
    def evaluate(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return corr.max()
    
    def project_feasible(h):
        """Project h onto {h : 0 <= h[i] <= 1, sum(h) = n/2}."""
        n = len(h)
        target = n / 2.0
        
        # Find t such that sum(clip(h + t, 0, 1)) = target
        # clip(h+t, 0, 1) is piecewise linear in t, monotone non-decreasing
        h_sorted = np.sort(h)
        
        # Binary search for t
        lo = -h[0] - 1  # h + lo < 0 for all
        hi = 1 - h[-1] + 1  # h + hi > 1 for all
        
        for _ in range(100):
            mid = (lo + hi) / 2
            s = np.sum(np.clip(h + mid, 0, 1))
            if s < target:
                lo = mid
            else:
                hi = mid
        
        t = (lo + hi) / 2
        h_proj = np.clip(h + t, 0, 1)
        
        # Fix any rounding issues
        diff = target - np.sum(h_proj)
        if abs(diff) > 1e-10:
            # Distribute among free coordinates
            free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
            if np.any(free):
                h_proj[free] += diff / np.sum(free)
                h_proj = np.clip(h_proj, 0, 1)
        
        return h_proj
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    # Main optimization loop
    alpha = 50.0  # softmax temperature for subgradient
    lr = 0.01  # learning rate
    
    while time.time() - start_time < budget_s * 0.9:
        # Compute correlation values
        corr = compute_result(h)
        max_val = corr.max()
        
        # Compute softmax weights (concentrate on max terms)
        weights = np.exp(alpha * (corr - max_val))
        weights /= weights.sum()
        
        # Compute gradient
        grad = compute_gradient(h, weights)
        
        # Normalize gradient (optional)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
            
        # Take a step
        # Use backtracking line search
        t = lr
        best_t = 0
        for _ in range(20):
            h_trial = project_feasible(h - t * grad)
            f_trial = evaluate(h_trial)
            if f_trial < max_val:
                best_t = t
                break
            t /= 2
        
        if best_t > 0:
            h = h_trial
            new_val = f_trial
            if new_val < best_c5:
                best_c5 = new_val
                best_h = h.copy()
        else:

            if np.random.random() < 0.1:

                h = project_feasible(h + perturb)
        
        # Gradually increase alpha (sharper subgradient)
        if np.random.random() < 0.01:
            alpha = min(alpha * 1.5, 500.0)
    
    return best_h, best_c5, len(best_h)

    res = linprog(cobj, A_eq=Aeq, b_eq=[0.0], bounds=[(-1.0, 1.0)]*n, method='highs')
    if res.success and res.fun < 0:
        return res.x
    return None

def _interpolate(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)
