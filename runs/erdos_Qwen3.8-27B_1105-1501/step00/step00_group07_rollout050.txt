        k_max = int(np.argmax(R))

        # Lag = k_max - (n-1)
        lag = k_max - (n - 1)
        if lag >= 0:
            # Positive lag: pairs (j, j+lag) for j=0..n-1-lag
            # R[k_max] = sum_j h[j]*(1-h[j+lag]) * dx
            # To reduce: increase h[j+lag] where h[j] is large
            # Or decrease h[j] where h[j+lag] is small
            valid = np.arange(n - lag)
            contrib = h[valid] * (1.0 - h[valid + lag])
            # Find the pair with largest contribution
            idx = int(np.argmax(contrib))
            j = valid[idx]

            d = rng.uniform(0.02, 0.1)

                if h[j] > 0.5:
                    h[j] = min(1.0, h[j] + d)
                else:
                    h[j] = max(0.0, h[j] - d)
            else:
                # Move mass: decrease h[j], increase h[j+lag]
                if h[j] > d and h[j+lag] < 1.0 - d:
                    h[j] -= d
                    h[j+lag] += d
                elif h[j] > d:
                    h[j] -= d
                else:
                    h[j+lag] = min(1.0, h[j+lag] + d)
        else:
            # Negative lag: k_max = n-1+lag where lag < 0
            # Pairs where h[j] high and h[j+lag] low (j+lag > 0, i.e., j > -lag)
            neg_lag = -lag  # positive integer
            valid = np.arange(neg_lag, n)
            contrib = h[valid] * (1.0 - h[valid - neg_lag])
            idx = int(np.argmax(contrib))
            j = valid[idx]
            d = rng.uniform(0.02, 0.1)

                h[j] -= d
                h[j - neg_lag] += d
            elif h[j] > d:
                h[j] -= d

        c_new = score_exact(h)
        if c_new < c:
            c = c_new
        else:
            # Revert (roughly) - just continue
            pass
    
    return h, c
