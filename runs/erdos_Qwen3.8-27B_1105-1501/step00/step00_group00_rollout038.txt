    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    def tleft():
        return budget_s - (time.time() - t0)
    
    # Phase 1: SLSQP with smooth max at moderate beta
    for beta in [5.0, 15.0, 40.0, 100.0]:
        if tleft() < 150:
            break
        try:
            res = slsqp_smooth_max(best_h, beta, n, maxiter=500)
            hc = project_feasible(res.x, n)
            c = compute_c5(hc)
            if c < best_c5:
                best_h = hc
                best_c5 = c
        except Exception:
            pass
    
    # Phase 2: Projected smooth gradient descent (finer control)
    if tleft() > 250:
        hc = best_h.copy()
        for beta in [10.0, 30.0, 80.0, 200.0, 500.0]:
            lr = 8.0
            for it in range(400):
                if tleft() < 250:
                    break
                _, grad = smooth_max_value_grad(hc, beta)
                step = lr
                improved = False
                for _ in range(15):
                    hnew = project_feasible(hc - step * grad, n)
                    c5new = compute_c5(hnew)
                    if c5new < compute_c5(hc) - 1e-13:
                        hc = hnew
                        if c5new < best_c5:
                            best_h = hnew.copy()
                            best_c5 = c5new
                        improved = True
                        break
                    step *= 0.5
                if not improved:
                    lr *= 0.5
                    if lr < 1e-7:
                        break
            if tleft() < 250:
                break
    
    # Phase 3: Epigraph SLSQP from best point
    if tleft() > 150:
        try:
            res = epigraph_solve(best_h, n, maxiter=400)
            hc = project_feasible(res.x[:n], n)
            c = compute_c5(hc)
            if c < best_c5:
                best_h = hc
                best_c5 = c
        except Exception:
            pass
    
    # Phase 4: Random perturbations + local SLSQP
    if tleft() > 200:
        for trial in range(30):
            if tleft() < 120:
                break
            sig = 0.02 + 0.015 * (trial % 5)
            hp = best_h + rng.normal(0, sig, n)
            hp = project_feasible(hp, n)
            cp = compute_c5(hp)
            start_hp = hp if cp < best_c5 else best_h
            if cp < best_c5:
                best_h = hp
                best_c5 = cp
            if tleft() > 80:
                try:
                    res = slsqp_smooth_max(start_hp, 50.0, n, maxiter=300)
                    hc = project_feasible(res.x, n)
                    c = compute_c5(hc)
                    if c < best_c5:
                        best_h = hc
                        best_c5 = c
                except Exception:
                    pass
    
    # Phase 5: Final epigraph polish
    if tleft() > 80:
        try:
            res = epigraph_solve(best_h, n, maxiter=500)
            hc = project_feasible(res.x[:n], n)
            c = compute_c5(hc)
            if c < best_c5:
                best_h = hc
                best_c5 = c
        except Exception:
            pass
    
    # Phase 6: One more smooth SLSQP at very high beta
    if tleft() > 60:
        try:
            res = slsqp_smooth_max(best_h, 500.0, n, maxiter=500)
            hc = project_feasible(res.x, n)
            c = compute_c5(hc)
            if c < best_c5:
                best_h = hc
                best_c5 = c
        except Exception:
            pass
    
    final_h = project_feasible(best_h, n)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n
