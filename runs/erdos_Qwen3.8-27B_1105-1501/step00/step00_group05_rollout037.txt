        h_hi = make_feasible(res_hi.x)
        c5_hi = evaluate(h_hi)
        if c5_hi < best_c5:
            best_h, best_c5, n = h_hi, c5_hi, n2
        if time.time() - t0 < budget_s * 0.88:
            h_hi_cd, c5_hi_cd = coord_descent(best_h, n, n_grid=17, max_rounds=200,
                                              time_limit=budget_s * 0.1)
            if c5_hi_cd < best_c5:
                best_h, best_c5 = h_hi_cd, c5_hi_cd
            h_hi_f, c5_hi_f = fine_coord_descent(best_h, n, max_rounds=100,
                                                 time_limit=budget_s * 0.03)
            h_hi_f = make_feasible(h_hi_f)
            c5_hi_f = evaluate(h_hi_f)
            if c5_hi_f < best_c5:
                best_h, best_c5 = h_hi_f, c5_hi_f

    # Final safety: ensure feasibility
    best_h = make_feasible(best_h)
    best_c5 = evaluate(best_h)

    return best_h, best_c5, len(best_h)
