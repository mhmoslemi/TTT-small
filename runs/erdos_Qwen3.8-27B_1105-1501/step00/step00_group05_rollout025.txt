```python
import numpy as np
from scipy.optimize import minimize
import time

# === Core functions ===

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1 - h, mode="full")

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    vals = dx * c
    vmax = vals.max()
    obj = vmax + np.log(np.sum(np.exp(beta * (vals - vmax)))) / beta
    w = np.exp(beta * (vals - vmax))
    w /= w.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for m in range(len(vals)):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - center
        # First term: (1-h[i-k]) for i in [max(0,k), min(n, n+k))
        s1, e1 = max(0, k), min(n, n + k)
        if e1 > s1:
            grad[s1:e1] += wm * dx * (1 - h[s1-k:e1-k])
        # Second term: -h[i+k] for i in [max(0,-k), min(n, n-k))
        s2, e2 = max(0, -k), min(n, n - k)
        if e2 > s2:
            grad[s2:e2] -= wm * dx * h[s2+k:e2+k]
    return obj, grad

# === Optimization ===

def pg_optimize(h, target_sum, betas, max_iter=3000, tol=1e-14):
    """Projected gradient with backtracking"""
    n = len(h)
    for beta in betas:
        obj, _ = smooth_obj_grad(h, beta)
        for t in range(max_iter):
            obj, grad = smooth_obj_grad(h, beta)
            alpha = 1.0
            improved = False
            for _ in range(40):
                h_new = project_feasible(h - alpha * grad, target_sum)
                obj_new, _ = smooth_obj_grad(h_new, beta)
                if obj_new < obj - 1e-4 * alpha * np.dot(grad, h - h_new):
                    h = h_new
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                break
            if obj - obj_new < tol:
                break
    return h

def pair_transfer(h, target_sum, n_trials=50000, eps_range=(0.001, 0.5)):
    """Random pair transfers with exact evaluation"""
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    rng = np.random.RandomState(42)
    
    for t in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Direction: from j to i or from i to j
        if rng.random() < 0.5:
            di, dj = 1, -1
        else:
            di, dj = -1, 1
        
        max_eps = min(best[i] if di < 0 else 1 - best[i],
                      best[j] if dj < 0 else 1 - best[j])
        if max_eps <= 0:
            continue
        eps = rng.uniform(eps_range[0], max(eps_range[0], max_eps * eps_range[1]))
        if eps > max_eps:
            continue
        
        h_trial = best.copy()
        h_trial[i] += di * eps
        h_trial[j] += dj * eps
        
        val = compute_c5(h_trial)
        if val < best_val:
            best = h_trial
            best_val = val
    
    return best, best_val
```

Actually, let me reconsider the pair transfer phase. Instead of purely random transfers, let me use gradient-guided transfers:

1. Compute the subgradient (average gradient of active components)
2. For each position i, the "benefit" of increasing h[i] is -grad[i] (if negative, increasing helps)
3. For each position j, the "cost" of decreasing h[j] is grad[j] (if positive, decreasing helps)
4. Transfer from j (high grad) to i (low grad)

This is much more efficient than random transfers.

Let me implement a gradient-guided pair transfer:

```python
def gradient_page_transfer(h, n_passes=100):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    
    for p in range(n_passes):
        # Compute subgradient
        vals = corr_vals(best)
        vmax = vals.max()
        active = vals >= vmax - 1e-8
        center = n - 1
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in range(len(vals)):
            if not active[m]:
                continue
            k = m - center
            s1, e1 = max(0, k), min(n, n + k)
            if e1 > s1:
                grad[s1:e1] += dx * (1 - best[s1-k:e1-k])
            s2, e2 = max(0, -k), min(n, n - k)
            if e2 > s2:
                grad[s2:e2] -= dx * best[s2+k:e2+k]
        grad /= max(1, active.sum())
        
        # Find best pair to transfer
        # We want to increase h where grad is most negative, decrease where most positive
        # Try all pairs with gradient guidance
        best_improvement = 0
        best_move = None
        
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                # Transfer from j to i: h[i] += eps, h[j] -= eps
                # First-order change: eps * (grad[i] - grad[j])
                # We want this to be negative (improvement)
                predicted_change = grad[i] - grad[j]
                if predicted_change >= 0:
                    continue
                # Max feasible eps
                max_eps = min(1 - best[i], best[j])
                if max_eps <= 1e-10:
                    continue
                # Try a fraction of max_eps
                eps = 0.1 * max_eps
                h_trial = best.copy()
                h_trial[i] += eps
                h_trial[j] -= eps
                val = compute_c5(h_trial)
                improvement = best_val - val
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_move = (i, j, eps)
        
        if best_move is None:
            break
        i, j, eps = best_move
        best[i] += eps
        best[j] -= eps
        best_val = compute_c5(best)
    
    return best, best_val
```
