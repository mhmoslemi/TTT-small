def score(h):

    n = len(h)

    return dx * Y.max()

def make_feasible(y, n):
    """Project y onto {0 <= x <= 1, sum(x) = n/2} using bisection."""
    target = n / 2.0
    y = np.clip(y, 0.0, 1.0)
    
    def phi(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    
    lo = -1.0
    hi = max(y.max() + 1.0, 0.0)
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if phi(mid) > target:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    r = target - x.sum()
    if abs(r) > 1e-12:
        if r > 0:
            # Need to add mass: add to entries with room
            room = 1.0 - x
            order = np.argsort(-room)
            remaining = r
            for i in order:
                add = min(room[i], remaining)
                if add > 1e-12:
                    x[i] += add
                    remaining -= add
                if abs(remaining) < 1e-12:
                    break
        else:
            # Need to remove mass: remove from entries with mass
            order = np.argsort(x)
            remaining = -r

                rem = min(x[i], remaining)
                if rem > 1e-12:
                    x[i] -= rem
                    remaining -= rem
                if abs(remaining) < 1e-12:

def gradient_row(p, h):
    """Compute gradient of correlation component at output index p."""
    n = len(h)
    center = n - 1
    k = p - center
    g = np.zeros(n)

            g[j] += (1.0 - h[j + m])
        for j in range(m, n):
            g[j] += -h[j - m]

    M = Y.max()
    threshold = M - max(tol_frac * M, tol_abs)
    indices = np.where(Y >= threshold)[0]
    # Sort by value descending
    order = indices[np.argsort(-Y[indices])]
    return order[:max_active]

def solve_qp_direction(h, active_indices, Y, rho=0.1, eta_factor=0.3, slack_weight=10.0):
    """Solve the QP to get a descent direction."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(h)
    M = Y.max()
    
    u = cp.Variable(n)
    s = cp.Variable(len(active_indices))
    
    constraints = [u >= -h, u <= 1.0 - h, cp.sum(u) == 0, s >= 0]
    
    for idx, p in enumerate(active_indices):
        g = gradient_row(p, h)
        # G_i @ u <= (M - Y_p) - eta_i + s_i
        # eta_i = eta_factor * rho * ||g||_2
        eta_i = eta_factor * rho * np.linalg.norm(g)
        eta_i = max(eta_i, 1e-4)
        constraints.append(g @ u <= (M - Y[p]) - eta_i + s[idx])
    
    objective = cp.Minimize(0.5 * cp.sum_squares(u) + slack_weight * cp.sum(s))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None
    
    if u.value is None:
        return None
    
    if not np.all(np.isfinite(u.value)):
        return None
    
    direction = u.value.copy()
    # Remove mean to ensure exact mass preservation
    direction -= direction.mean()
    
    if np.linalg.norm(direction) < 1e-10:
        return None
    
    return direction

                best_c = c
                improved = True
                break

    return h, best_c

        g[k:] += 1.0 - h[:n-k]
    else:
        m = -k
        # Y[p] = sum_{j=0}^{n-m-1} h[j]*(1-h[j+m])
        g[:n-m] += 1.0 - h[m:]
        g[m:] += -h[:n-m]
    
    return g

def active_set(Y, M, frac=0.05, abs_tol=0.01, max_act=25):
    """Get indices of near-max correlation values."""
    threshold = M - max(frac * M, abs_tol)
    idx = np.where(Y >= threshold)[0]
    if len(idx) > max_act:
        order = idx[np.argsort(-Y[idx])]
        idx = order[:max_act]
    return idx

def try_qp(h, active, Y, rho, eta_fac, time_budget=5.0):
    """Solve QP for descent direction. Returns direction or None."""
    t0 = time.time()
    n = len(h)
    M = Y.max()
    
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    u = cp.Variable(n)
    s = cp.Variable(len(active))
    
    constraints = [
        u >= -h,
        u <= 1.0 - h,
        cp.sum(u) == 0,
        s >= 0
    ]
    
    for i, p in enumerate(active):
        g = grad_row(p, h)
        eta = max(eta_fac * rho * np.linalg.norm(g), 2e-4)
        constraints.append(cp.dot(g, u) <= (M - Y[p]) - eta + s[i])
    
    obj = cp.Minimize(0.5 * cp.sum_squares(u) + 50.0/max(rho, 0.01) * cp.sum(s))
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if u.value is not None and np.all(np.isfinite(u.value)):
                d = u.value - u.value.mean()
                if np.linalg.norm(d) > 1e-10:
                    return d
            break
        except Exception:
            continue
    
    return None

def exact_line(h, d, c_ref, alphas=None):

    n = len(h)
    if alphas is None:
        alphas = np.concatenate([

            np.array([0.5, 0.8, 1.2, 1.5]) * 0.01  # very small steps
        ])
    
    best_c = c_ref
    best_h = None
    
    for a in alphas:
        if a <= 0:
            continue
        x = h + a * d
        if x.min() < -1e-9 or x.max() > 1 + 1e-9:
            continue
        x = np.clip(x, 0.0, 1.0)
        if abs(x.sum() - n*0.5) > 1e-7:

        c = score(x)
        if c < best_c - 1e-12:

            best_h = x.copy()
    
    return best_h, best_c

def pair_moves(h, Y, n, rng, max_tries=300):
    """Try random pair mass transfers. Returns (best_h, best_c)."""
    best_h = h.copy()
    best_c = score(h)
    
    order = np.argsort(-h)
    n_half = n // 2
    
    for _ in range(max_tries):
        a = int(order[rng.integers(0, n_half)])
        b = int(order[rng.integers(n_half, n)])
        
        ha, hb = h[a], h[b]
        if ha - hb < 1e-4:
            continue
        
        max_t = min(ha, 1.0 - hb)
        if max_t < 1e-4:
            continue
        
        # Try geometric spread of t values
        for t in [0.1*max_t, 0.3*max_t, 0.5*max_t, 0.8*max_t]:
            if t < 1e-6:
                continue
            x = h.copy()
            x[a] -= t
            x[b] += t
            c = score(x)
            if c < best_c - 1e-12:
                best_c = c
                best_h = x.copy()
                break
    
    return best_h, best_c

def coord_descent(h, n, rng, max_iter=1000, deadline=None):
    """Simple coordinate-style: pick random pair, optimize t exactly."""
    best_h = h.copy()
    best_c = score(h)
    order = np.argsort(-h)
    n_half = n // 2
    
    for it in range(max_iter):
        if deadline and time.time() >= deadline:
            break
        a = int(order[rng.integers(0, n_half)])
        b = int(order[rng.integers(n_half, n)])
        ha, hb = best_h[a], best_h[b]
        if ha - hb < 1e-4:
            continue
        max_t = min(ha, 1.0 - hb)
        
        # Bisection on t to find best
        lo_t, hi_t = 0.0, max_t
        best_t = 0.0
        best_c_pair = best_c
        
        for _ in range(15):
            mid = (lo_t + hi_t) * 0.5
            x = best_h.copy()
            x[a] -= mid
            x[b] += mid
            c = score(x)
            if c < best_c_pair:
                best_c_pair = c
                best_t = mid
                lo_t = mid
            else:
                hi_t = mid
        
        if best_t > 1e-8 and best_c_pair < best_c - 1e-12:
            best_h[a] -= best_t
            best_h[b] += best_t
            best_c = best_c_pair
            order = np.argsort(-best_h)
    
    return best_h, best_c

import numpy as np

def score(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode='full'))

def overlap(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(y, n):
    target = n * 0.5
    y = np.clip(np.asarray(y, dtype=float).copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-12:
        if r > 0:
            order = np.argsort(1.0 - x)
            for i in reversed(order):
                if x[i] < 1.0:
                    add = min(r, 1.0 - x[i])
                    x[i] += add
                    r -= add
                    if abs(r) < 1e-12:
                        break
        else:
            order = np.argsort(x)
            for i in order:
                if x[i] > 0.0:
                    rem = min(-r, x[i])
                    x[i] -= rem
                    r += rem
                    if abs(r) < 1e-12:
                        break
    return np.clip(x, 0.0, 1.0)

def grad_row(p, h):
    n = len(h)
    center = n - 1
    k = p - center
    g = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    elif k > 0:
        g[:n - k] += -h[k:]
        g[k:] += 1.0 - h[:n - k]
    else:
        m = -k
        g[:n - m] += 1.0 - h[m:]
        g[m:] += -h[:n - m]
    return g

def get_active(Y, frac=0.06, abs_tol=0.01, max_act=25):

    idx = np.where(Y >= threshold)[0]
    if len(idx) > max_act:
        order = idx[np.argsort(-Y[idx])]
        idx = order[:max_act]
    elif len(idx) == 0:
        idx = np.array([int(np.argmax(Y))])
    return idx

    try:
        import cvxpy as cp
    except ImportError:
        return None
    u = cp.Variable(n)
    s = cp.Variable(len(active))

    sw = 50.0 / max(rho, 0.01)

        g = grad_row(p, h)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            continue
        eta = max(eta_fac * rho * gnorm, 2e-4)
        constraints.append(cp.dot(g, u) <= (M - Y[p]) - eta + s[i])
    obj = cp.Minimize(0.5 * cp.sum_squares(u) + sw * cp.sum(s))
    prob = cp.Problem(obj, constraints)
    for solver in ['CLARABEL', 'SCS', 'ECOS', 'PDLP']:
        try:
            prob.solve(solver=getattr(cp, solver))
            if u.value is not None and np.all(np.isfinite(u.value)):

                if np.linalg.norm(d) > 1e-10:
                    return d
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return None

def pair_optimize(h, a, b, c_current):
    """Optimize pair transfer (a -> b) using fine grid since f(t) is convex piecewise-linear."""
    n = len(h)
    ha, hb = h[a], h[b]

        return h, c_current

        return h, c_current

def coord_sweep(h, n, deadline=None):
    """Sweep all ordered pairs (i,j) where h[i] >= h[j], optimize each."""
    best_h = h.copy()

    order = np.argsort(-best_h)
    n_half = n // 2

    pairs = []
    for i in range(n_half):
        for j in range(n_half, n):
            a = int(order[i])
            b = int(order[j])
            diff = best_h[a] - best_h[b]
            if diff > 1e-4:
                pairs.append((diff, a, b))
    pairs.sort(reverse=True)
    for diff, a, b in pairs:

            best_c = new_c

    return best_h, best_c

        h_init = make_feasible(h0, n)
        c_init = score(h_init)
        best_h = h_init.copy()
        best_c = c_init

        # Check if raw is already feasible and better
        if h0.min() >= -1e-10 and h0.max() <= 1.0 + 1e-10:
            if abs(h0.sum() - n * 0.5) < 1e-8:
                c_raw = score(h0)
                if c_raw < best_c:
                    best_c = c_raw
                    best_h = h0.copy()
    else:
        # No parent: create a reasonable starting point
        n = 60
        best_h = np.full(n, 0.5)
        # Add some structure: alternating high/low
        for i in range(n):
            best_h[i] = 0.5 + 0.3 * np.cos(2.0 * np.pi * i / (n // 3))
        best_h = make_feasible(best_h, n)
        best_c = score(best_h)

    n = len(best_h)

    # Phase 1: Main local optimization
    phase1_end = min(deadline, t_start + 0.45 * budget_s)
    if phase1_end > t_start + 10:
        h1, c1 = improve(best_h, phase1_end, rng)
        if c1 < best_c:
            best_h = h1
            best_c = c1

    # Phase 2: Try scaled variants and re-optimize
    if time.time() < deadline - 30:
        n = len(best_h)
        variants = []
        for lam in [0.75, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.25]:
            v = make_feasible(0.5 + lam * (best_h - 0.5), n)
            variants.append(v)
        for sig in [0.02, 0.04, 0.06, 0.08, 0.12]:
            for _ in range(2):
                v = make_feasible(best_h + rng.normal(0, sig, n), n)
                variants.append(v)

        for v in variants:
            if time.time() >= deadline - 12:
                break
            cv = score(v)
            if cv < best_c - 1e-13:
                best_h = v.copy()
                best_c = cv
                # Quick improve from here
                h2, c2 = improve(best_h, min(deadline - 5, time.time() + 20), rng)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2

    # Phase 3: Coordinate sweep for final polish
    if time.time() < deadline - 5:
        h3, c3 = pair_polish(best_h, n, rng, max_rounds=50, deadline=deadline)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    # Phase 4: Try different n if time remains
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        for n_try in [50, 56, 60, 80, 100]:
            if time.time() >= deadline - 10:
                break
            if n_try == n_cur:
                continue
            h_try = try_different_n(best_h, n_try, seed)
            c_try = score(h_try)
            if c_try < best_c - 1e-13:
                # Optimize at this n
                h_opt, c_opt = improve(h_try, min(deadline - 3, time.time() + 30), rng)
                if c_opt < best_c:
                    best_h = h_opt
                    best_c = c_opt

    # Final
    n_final = len(best_h)
    c_final = score(best_h)

    # Verify
    dx = 2.0 / n_final
    Y = np.correlate(best_h, 1.0 - best_h, mode='full')
    c_verify = dx * Y.max()
    assert abs(c_final - c_verify) < 1e-12, f"Verification failed: {c_final} vs {c_verify}"
    assert best_h.min() >= -1e-10 and best_h.max() <= 1.0 + 1e-10
    assert abs(best_h.sum() - n_final * 0.5) < 1e-8

    return best_h, c_final, n_final
