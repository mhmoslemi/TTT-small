import numpy as np

def score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(raw)

def project_feasible(z, target):
    n = len(z)
    v = np.clip(z, 0.0, 1.0)
    e = np.sum(v) - target
    if e > 1e-12:
        # Need to reduce sum by e
        # Sort by value descending to reduce from largest first
        order = np.argsort(v)[::-1]
        for i in order:
            delta = min(v[i], e)
            v[i] -= delta
            e -= delta
            if e <= 1e-15:
                break
    elif e < -1e-12:
        # Need to increase sum by -e
        order = np.argsort(1 - v)[::-1]  # most room first
        for i in order:
            delta = min(1.0 - v[i], -e)
            v[i] += delta
            e += delta
            if e >= -1e-15:
                break
    return v

def compute_correlations(h):
    """Compute all 2n-1 correlation values F_k * dx for k = -(n-1), ..., n-1"""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode="full")
    return raw * dx  # These are the y_k values

def soft_gradient(h, alpha):
    """Compute gradient of soft-max objective w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode="full")
    y = raw * dx  # scaled correlations
    L = len(y)
    
    M = np.max(y)
    # Stable softmax weights
    z = np.exp(alpha * (y - M))
    s = np.sum(z)
    if s <= 0 or not np.isfinite(s):
        # Fallback: uniform over max elements
        w = np.zeros(L)
        max_idx = np.where(y >= M - 1e-10)[0]
        w[max_idx] = 1.0 / len(max_idx)
    else:
        w = z / s
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(L):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)  # shift
        if k >= 0:
            length = n - k
            if length <= 0:
                continue
            # ∂F_k/∂h[m] = (1-h[m-k]) for m in [k, n-1] - h[m+k] for m in [0, n-1-k]
            # First term: for m >= k, contribution is (1-h[m-k])
            # This is a vector: for m = k to n-1, h[m-k] = h[0] to h[n-1-k]
            grad[k:] += w[idx] * (1 - h[:length])
            # Second term: for m + k < n, i.e., m < n-k, contribution is -h[m+k]
            # h[m+k] for m = 0 to n-1-k is h[k] to h[n-1]
            grad[:length] -= w[idx] * h[k:]
        else:
            j = -k  # j > 0
            length = n - j  # = n + k
            if length <= 0:
                continue
            # ∂F_{-j}/∂h[m] = (1-h[m+j]) for m in [0, n-j-1] - h[m-j] for m in [j, n-1]
            # First term: for m < n-j, contribution is (1-h[m+j])
            # h[m+j] for m = 0 to n-j-1 is h[j] to h[n-1]
            grad[:length] += w[idx] * (1 - h[j:])
            # Second term: for m >= j, contribution is -h[m-j]
            # h[m-j] for m = j to n-1 is h[0] to h[n-1-j]
            grad[j:] -= w[idx] * h[:length]
    
    grad *= dx
    # Remove mean to preserve sum constraint
    grad -= np.mean(grad)
    return grad

def line_search(h0, C0, direction, min_gain=1e-8):
    """Find best step along direction with projection"""
    n = len(h0)
    target = n / 2.0
    best_C = C0
    best_h = h0.copy()
    
    absmax = np.max(np.abs(direction))
    if absmax < 1e-15:
        return best_h, best_C, False
    
    p = direction / absmax
    
    steps = [0.01, 0.03, 0.06, 0.1, 0.15, 0.25, 0.4, 0.6, 0.9, 1.3, 1.8, 2.5, 3.5, 5.0]
    
    for step in steps:
        x = h0 + step * p
        v = project_feasible(x, target)
        C = score(v)
        if C < best_C - min_gain:
            best_C = C
            best_h = v
    return best_h, best_C, best_C < C0 - min_gain

def random_direction(n, rng):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    return d

def resample(h_old, n_new):
    """Interpolate h from old grid to new grid"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # midpoints
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new  # midpoints
    # For interpolation, use edge values
    x_old_full = np.concatenate([[-1.0/n_old], x_old, [2.0 + 1.0/n_old]])
    h_old_full = np.concatenate([[h_old[0]], h_old, [h_old[-1]]])
    h_new = np.interp(x_new, x_old_full, h_old_full)
    return h_new
