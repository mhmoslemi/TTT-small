import time

def compute_correlations(h):
    """Compute all correlation values."""
    return np.correlate(h, 1-h, mode="full")

def compute_c5(h, dx):
    """Compute C5 value."""
    return np.max(np.correlate(h, 1-h, mode="full")) * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Main optimization loop
    step_size = 0.01
    while remaining > 5:
        # Compute correlations
        c = np.correlate(h, 1-h, mode="full")
        max_val = c.max()
        current_c5 = max_val * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        tolerance = 0.01 * max_val
        active = np.where(c > max_val - tolerance)[0]
        
        # Compute subgradient (average gradient of active lags)
        grad = np.zeros(n)
        for lag_idx in active:
            lag = lag_idx - (n - 1)  # convert to actual lag
            for i in range(n):

                # h[i] appears in (1-h[j]) when j = i (valid: 0<=i+lag<=n-1)

        if new_c5 < current_c5:
            h = h_new
            step_size *= 1.05  # slightly increase step
        else:
            # Try local moves
            # ... (pair transfers, individual adjustments)
            step_size *= 0.95
        
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
    
    return best_h, best_c5, n

import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            if lag == 0:
                g[:] += (1.0 - h)
                g[:] -= h
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def local_search(h, target, dx, max_iters=5000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(max_iters):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if improved:
                continue
            for delta in (0.01, -0.01, 0.02, -0.02, 0.005, -0.005):
                h_new = h.copy()
                h_new[i] += delta
                if h_new[i] < 0 or h_new[i] > 1:
                    continue
                h_new = project_feasible(h_new, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-12:
                    h = h_new
                    best = h.copy()
                    best_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(70, 0.5)

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Subgradient descent with softmax
    step = 0.003
    tau = 20.0
    no_improve = 0
    max_no_improve = 300

    while remaining() > 60:
        c = np.correlate(h, 1.0 - h, mode="full")
        cmax = float(c.max())
        c5 = cmax * dx

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            noise_scale = 0.01 + 0.005 * (no_improve // max_no_improve)
            h = project_feasible(h + rng.normal(0, noise_scale, n), target)
            no_improve = 0
            step = 0.003
            continue

        # Softmax weights for active lags
        shifted = c - cmax
        w = np.exp(tau * shifted)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= wsum

        # Weighted gradient
        grad = np.zeros(n)
        active_idx = np.where(w > 1e-5)[0]
        for k in active_idx:
            lag = k - (n - 1)
            grad += w[k] * grad_lag(h, lag)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            no_improve += 50
            continue

        # Backtracking line search
        accepted = False
        s = step
        for _ in range(8):
            h_new = project_feasible(h - s * grad, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5 - 1e-10:
                h = h_new
                accepted = True
                step = min(s * 1.05, 0.05)
                break
            s *= 0.5

            tau = min(tau * 1.005, 80.0)

                    break

                break

                if c5_try < best_c5:
                    best_h = h_interp
                    best_c5 = c5_try
                    best_n = n_try
            if remaining() < 15:
                break

    return best_h, best_c5, n
