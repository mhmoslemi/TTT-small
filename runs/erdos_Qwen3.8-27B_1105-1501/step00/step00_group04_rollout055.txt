def make_feasible(y, n=None):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    if n is None:
        n = len(y)
    y = np.asarray(y, dtype=np.float64).copy()
    target = n / 2.0
    
    # Bisection on lambda: find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(z)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix any floating point error in sum
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        # Adjust the component closest to 0.5 (most room)
        # Find index with most room
        if diff > 0:
            # Need to increase sum; find index closest to 0 (most room to increase)
            idx = np.argmin(z)
            add = min(diff, 1.0 - z[idx])
            z[idx] += add
            if abs(diff - add) > 1e-12:
                idx2 = np.argmin(z)
                if idx2 != idx:
                    z[idx2] += min(diff - add, 1.0 - z[idx2])
        else:
            # Need to decrease sum
            idx = np.argmax(z)
            sub = min(-diff, z[idx])
            z[idx] -= sub
            if abs(diff + sub) > 1e-12:
                idx2 = np.argmax(z)
                if idx2 != idx:
                    z[idx2] -= min(-diff - sub, z[idx2])
    
    # Final safety clip
    z = np.clip(z, 0.0, 1.0)
    return z

def score_exact(h):

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx

    n = len(h)

def smooth_grad(h, T):
    """Compute smooth-max objective value and gradient"""
    n = len(h)
    dx = 2.0 / n
    scores = score_scores(h)
    
    base = np.max(scores)
    ex = np.exp((scores - base) / T)
    S = np.sum(ex)
    if S < 1e-300:
        # All underflow; use hard max
        w = np.zeros_like(scores)
        w[scores == base] = 1.0 / np.sum(scores == base)
    else:
        w = ex / S
    
    smooth_val = base + T * np.log(S)
    
    # Gradient
    g = np.zeros(n)
    center = n - 1
    
    # d=0 contribution
    g += w[center] * dx * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        wp = w[center + d]
        wn = w[center - d]
        m = n - d
        # g[0:m] gets contributions from both wp and wn
        g[0:m] += (wn * (1.0 - h[d:n]) - wp * h[d:n]) * dx
        g[d:n] += (wp * (1.0 - h[0:m]) - wn * h[0:m]) * dx
    
    return smooth_val, g

def projected_descent(h, deadline, T_start=0.15, T_end=0.001, T_decay=0.9995, rngseed=42):
    """Gradient descent with projected feasibility and temperature annealing"""
    n = len(h)
    h = make_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = score_exact(h)
    cur_c = best_c
    
    T = T_start
    step_size = 0.05
    rng = np.random.default_rng(rngseed)
    
    iters = 0
    while time.time() < deadline:
        smooth_val, g = smooth_grad(h, T)
        
        if not np.all(np.isfinite(g)):
            T *= 0.5
            continue
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            T *= 0.5
            if T < T_end * 0.1:
                break
        
        # Gradient step
        direction = -g / (g_norm + 1e-15)
        
        accepted = False
        for trial_step in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:
            candidate = h + trial_step * direction
            candidate = make_feasible(candidate, n)
            c_new = score_exact(candidate)
            if c_new < cur_c - 1e-13:
                h = candidate
                cur_c = c_new
                accepted = True
                if cur_c < best_c:
                    best_h = h.copy()
                    best_c = cur_c
                break
        
        if not accepted:
            step_size *= 0.9
            if step_size < 1e-8:
                step_size = T * 0.5
        
        # Anneal temperature
        T = max(T_end, T * T_decay)
        iters += 1
    
    return best_h, best_c

def pair_line_search(h, i, j, current_score, n_grid=30):
    """Try to improve by moving mass between indices i and j"""
    n = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    
    if hi - lo < 1e-12:
        return None, current_score
    
    best_alpha = 0.0
    best_score = current_score
    
    for alpha in np.linspace(lo, hi, n_grid):
        h[i] += alpha
        h[j] -= alpha
        s = score_exact(h)
        if s < best_score:
            best_score = s
            best_alpha = alpha
    
    # Golden section refinement around best_alpha
    lo2 = max(lo, best_alpha - (hi-lo)/n_grid * 1.5)
    hi2 = min(hi, best_alpha + (hi-lo)/n_grid * 1.5)
    if hi2 - lo2 > 1e-12:
        gr = (np.sqrt(5) - 1) / 2
        a = lo2 + (1-gr)*(hi2-lo2)
        b = lo2 + gr*(hi2-lo2)
        for _ in range(20):
            fa = 0; fb = 0
            h[i] += a; h[j] -= a
            fa = score_exact(h)
            h[i] -= a; h[j] += a
            h[i] += b; h[j] -= b
            fb = score_exact(h)
            if fa < fb:
                hi2 = b; b = a; a = lo2 + (1-gr)*(hi2-lo2)
            else:
                lo2 = a; a = b; b = lo2 + gr*(hi2-lo2)

        h[i] += best_alpha
        h[j] -= best_alpha
        return h, best_score

            cand = make_feasible(cur + s * direction, n)
            c_new = score_exact(cand)
            if c_new < cur_c - 1e-14:
                cur = cand
                cur_c = c_new
                improved = True
                if cur_c < best_c:
                    best = cur.copy()
                    best_c = cur_c
                break
        if not improved:
            step *= 0.85
            stall += 1
            if stall > 40:
                cur = make_feasible(cur + rng.normal(0, max(0.005, T * 0.3), n), n)
                cur_c = score_exact(cur)
                step = max(T, 0.005)
                stall = 0
        else:
            step = min(step * 1.05, T * 2.0)
            stall = 0
        T = max(T_end, T * T_decay)
    return best, best_c

def pair_search(h, i, j, cur_c, n_pts=40):
    orig_i = h[i]
    orig_j = h[j]
    lo = max(-orig_i, orig_j - 1.0)
    hi = min(1.0 - orig_i, orig_j)
    if hi - lo < 1e-12:
        return None
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(lo, hi, n_pts):
        h[i] = orig_i + a
        h[j] = orig_j - a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    if abs(best_a) > 1e-10 and best_s < cur_c - 1e-14:
        span = (hi - lo) / n_pts
        lo2 = max(lo, best_a - 1.5 * span)
        hi2 = min(hi, best_a + 1.5 * span)
        if hi2 - lo2 > 1e-12:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
            a2 = lo2 + gr * (hi2 - lo2)
            for _ in range(25):
                h[i] = orig_i + a1
                h[j] = orig_j - a1
                s1 = score_exact(h)
                h[i] = orig_i + a2
                h[j] = orig_j - a2
                s2 = score_exact(h)
                if s1 < s2:
                    if s1 < best_s:
                        best_s = s1
                        best_a = a1
                    hi2 = a2
                    a2 = a1
                    a1 = lo2 + (1.0 - gr) * (hi2 - lo2)
                else:
                    if s2 < best_s:
                        best_s = s2
                        best_a = a2
                    lo2 = a1
                    a1 = a2
                    a2 = lo2 + gr * (hi2 - lo2)
            best_a = (lo2 + hi2) * 0.5
            h[i] = orig_i + best_a
            h[j] = orig_j - best_a
            best_s = score_exact(h)
            if best_s >= cur_c:
                best_a = 0.0
    h[i] = orig_i
    h[j] = orig_j
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig_i + best_a
        h[j] = orig_j - best_a
        return float(best_s)
    return None

def coord_search(h, i, cur_c, n_pts=30):
    orig = h[i]
    best_a = 0.0
    best_s = cur_c
    for a in np.linspace(-orig, 1.0 - orig, n_pts):
        h[i] = orig + a
        s = score_exact(h)
        if s < best_s:
            best_s = s
            best_a = a
    h[i] = orig
    if best_s < cur_c - 1e-14 and abs(best_a) > 1e-10:
        h[i] = orig + best_a
        return float(best_s)
    return None

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old or n_new < 10:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return make_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    # Initialize from parent
    try:
        best_h = make_feasible(np.array(initial_h_values, dtype=np.float64))
    except (NameError, Exception):
        best_h = make_feasible(np.full(70, 0.5))
    best_c = score_exact(best_h)
    n = len(best_h)

    # Phase 1: Gradient descent (25% of time)
    p1_end = t0 + budget_s * 0.25
    if time.time() < p1_end:
        h1, c1 = grad_descent(best_h, p1_end, 0.15, 0.001, 0.9995, 0.03, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Aggressive pair + coordinate search (35% of time)
    p2_end = t0 + budget_s * 0.60
    cur = best_h.copy()
    cur_c = best_c
    consecutive_fail = 0
    while time.time() < p2_end:
        mode = rng.random()
        if mode < 0.7:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            res = pair_search(cur, i, j, cur_c, n_pts=35)
        elif mode < 0.85:
            i = int(rng.integers(0, n))
            res = coord_search(cur, i, cur_c, n_pts=30)
        else:
            i = int(rng.integers(0, n - 1))
            res = pair_search(cur, i, i + 1, cur_c, n_pts=40)
        if res is not None:
            cur_c = res
            consecutive_fail = 0
            if cur_c < best_c:
                best_h, best_c = cur.copy(), cur_c
            for _ in range(4):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                res2 = pair_search(cur, i2, j2, cur_c, n_pts=25)
                if res2 is not None:
                    cur_c = res2
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c
                else:
                    break
        else:
            consecutive_fail += 1
            if consecutive_fail > 200:
                cur = make_feasible(best_h + rng.normal(0, 0.01, n), n)
                cur_c = score_exact(cur)
                consecutive_fail = 0

    # Phase 3: Fine gradient descent (15% of time)
    p3_end = t0 + budget_s * 0.78
    if time.time() < p3_end:
        h3, c3 = grad_descent(best_h, p3_end, 0.008, 0.0001, 0.999, 0.003, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3
        n = len(best_h)

    # Phase 4: More pair search on all adjacent pairs systematically (10% of time)
    p4_end = t0 + budget_s * 0.92
    cur = best_h.copy()
    cur_c = best_c
    while time.time() < p4_end:
        improved_soft = False
        for i in range(n - 1):
            if time.time() > p4_end:
                break
            res = pair_search(cur, i, i + 1, cur_c, n_pts=30)
            if res is not None:
                cur_c = res
                improved_soft = True
                if cur_c < best_c:
                    best_h, best_c = cur.copy(), cur_c
        if not improved_soft:
            # Random pairs
            for _ in range(20):
                if time.time() > p4_end:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                res = pair_search(cur, i, j, cur_c, n_pts=30)
                if res is not None:
                    cur_c = res
                    if cur_c < best_c:
                        best_h, best_c = cur.copy(), cur_c

    # Phase 5: Try a few alternative resolutions (5% of time)
    p5_end = deadline
    if time.time() < p5_end:
        n_cur = len(best_h)
        for dn in (-4, -6, 4, 6):
            if time.time() > p5_end - 5:
                break
            nr = n_cur + dn
            if nr < 20 or nr > 200:
                continue
            h_r = resample(best_h, nr)
            if h_r is None:
                continue
            c_r = score_exact(h_r)
            sub_dl = min(p5_end - 2, time.time() + 40)
            if time.time() < sub_dl:
                h_r, c_r = grad_descent(h_r, sub_dl, 0.01, 0.0003, 0.999, 0.005, rng)
            if c_r < best_c:
                best_h, best_c = h_r, c_r

    # Final refinement
    if time.time() < deadline - 2:
        h_f, c_f = grad_descent(best_h, deadline - 2, 0.003, 0.00005, 0.998, 0.002, rng)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Ensure feasibility
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)
    return best_h, best_c, len(best_h)
