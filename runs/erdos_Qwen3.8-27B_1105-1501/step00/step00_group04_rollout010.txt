def repair_feasible(y, n):
    """Project y onto [0,1]^n with sum = n/2"""
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    y = np.clip(y, 0.0, 1.0)
    # Adjust sum using bisection on lambda
    # We need sum(clip(y_i + offset, 0, 1)) = target
    # Or simpler: scale and clip iteratively
    s = np.sum(y)
    if abs(s - target) < 1e-12:
        return y
    # Try scaling
    if s > 0:
        scale = target / s
        y2 = np.clip(y * scale, 0.0, 1.0)
        s2 = np.sum(y2)
        if abs(s2 - target) < 1e-9:
            return y2
    # Bisection on adding constant
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y_m = np.clip(y + mid, 0.0, 1.0)
        s_m = np.sum(y_m)
        if s_m > target:
            hi = mid
        else:
            lo = mid
    y_final = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(y_final)

    n = len(h)
    center = n - 1

    tol = max(1e-6, 0.01 * max_val)

        scored = sorted(active, key=lambda d: -scores[d + center])
        active = scored[:top_k]
    if 0 not in active:
        active.append(0)
    return active

    """Compute sum of gradients over active offsets"""
    n = len(h)
    g = np.zeros(n)
    for d in active_offsets:
        if d == 0:
            g += 1.0 - 2.0 * h
        elif d > 0:
            # C_d = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
            for i in range(n - d):
                g[i] += -h[i + d]
                g[i + d] += (1.0 - h[i])
        else:
            r = -d
            # C_{-r} = sum_{i=0}^{n-1-r} h[i]*(1-h[i+r])
            for i in range(n - r):
                g[i] += (1.0 - h[i + r])
                g[i + r] += -h[i]
    return g

        final_s = exact_score(best_h)
    
    return best_h, final_s, len(best_h)
