import time

def build_full(u, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = u[:m]
    h[n-m:] = 1.0 - u[:m][::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")

def score(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.max(compute_correlation(h))

def reduced_gradients(h, n):
    """Compute reduced gradients for all lags.
    Returns array of shape (2n-1, m) where m = n//2."""
    m = n // 2
    # Full gradients for each lag
    # G_k[i] = sum of contributions to dF_k/dh[i]
    # For each valid pair (i, i+k): G_k[i] += 1-h[i+k], G_k[i+k] += -h[i]

        start = max(0, -k)
        end = min(n, n-k)

            G_full[idx, i+k] += -h[i]

def slsqp_minimax(u0, n, maxiter=200):
    m = n // 2
    dx = 2.0 / n
    h0 = build_full(u0, n)
    theta0 = dx * np.max(compute_correlation(h0)) + 1e-8
    
    z0 = np.concatenate([u0, [theta0]])
    
    def objective(z):
        return z[m]  # theta
    
    def objective_grad(z):
        g = np.zeros(m+1)
        g[m] = 1.0
        return g
    
    def constraints(z):
        u = z[:m]
        theta = z[m]
        h = build_full(u, n)
        corr = compute_correlation(h)  # 2n-1 values
        # Constraint k: theta - dx * corr_k >= 0
        vals = theta - dx * corr
        # For Jacobian, we need d(vals)/dz
        # d/dtheta = 1
        # d/du[j] = -dx * G_red[k, j]
        G_red = reduced_gradients(h, n)
        J = np.zeros((2*n-1, m+1))
        for k in range(2*n-1):
            J[k, :m] = -dx * G_red[k, :]
            J[k, m] = 1.0
        return vals, J
    
    # Bounds
    bounds = [(0, 1)] * m + [(0, 1.5)]
    
    res = minimize(objective, z0, jac=objective_grad, 
                   constraints={'type': 'ineq', 'fun': constraints},
                   bounds=bounds, method='SLSQP', 
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    
    u_opt = res.x[:m]
    u_opt = np.clip(u_opt, 0, 1)
    h_opt = build_full(u_opt, n)
    return h_opt, score(h_opt)

def active_direction(G_red_active, h, n, band_rel=3e-4):
    """Find a direction in u-space that decreases active constraints."""
    m = n // 2
    dx = 2.0 / n
    corr = compute_correlation(h)
    corr_max = np.max(corr)
    
    # Active constraints: those with corr_k >= corr_max * (1 - band_rel)
    threshold = corr_max * (1 - band_rel)
    active_mask = corr >= threshold
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) == 0:
        return None
    
    # Get reduced gradients for active constraints
    G_act = G_red[active_idx, :]  # shape (len(active), m)
    
    # Target: reduce each active constraint value
    # Linearized: corr_k_new ≈ corr_k + dx * G_k @ d
    # We want: corr_k + dx * G_k @ d <= target
    # => G_k @ d <= (target - corr_k) / dx
    # LS: minimize ||G_act @ d - b||^2 where b = (target - corr[active]) / dx
    
    target = corr_max  # or slightly lower
    b = (corr_max - corr[active_idx]) / dx  # this is >= 0, we want G_act @ d <= b
    # Actually we want to minimize the maximum, so let's set target = corr_max
    # and solve G_act @ d ≈ -something (negative to decrease)
    
    # Better formulation: minimize ||G_act @ d + 1||^2 (push all active down equally)
    # Or: minimize ||G_act @ d - (-alpha)||^2 for some alpha > 0
    
    # Actually, the standard approach: minimize ||G_act @ d||^2 subject to G_act @ d <= -1 (or similar)
    # Simpler: just minimize d^T G_act^T G_act d - 2 * sum(b_i) * G_act[i] @ d
    # This is a ridge regression: (G_act^T G_act + lambda*I) d = G_act^T * b
    # where b is the "desired decrease"

    # G_act @ d should be approximately -1 for all active rows
    # LS solution: d = (G_act^T G_act + lambda I)^{-1} G_act^T (-1)
    
    A = G_act.T @ G_act + 1e-8 * np.eye(m)
    rhs = -G_act.T @ np.ones(len(active_idx))
    try:
        d = np.linalg.solve(A, rhs)
    except:
        d = np.linalg.lstsq(A, rhs, rcond=None)[0]
    
    return d

        G_full[idx, i_vals + k] += (-h[i_vals])
    # Reduce
    G_red = G_full[:, :m] - G_full[:, n-m:n][:, ::-1]
    return G_red

import numpy as np
from scipy.optimize import minimize
import time

def build_full(u, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = u[:m]
    h[n - m:] = 1.0 - u[:m][::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def symmetrize(h):
    n = len(h)
    m = n // 2
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return np.clip(h_sym[:m], 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.max(compute_corr(h))

def full_grads(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        idx = k + (n - 1)
        s = max(0, -k)
        e = min(n, n - k)
        if e > s:
            i_arr = np.arange(s, e)
            G[idx, i_arr] += (1.0 - h[i_arr + k])
            G[idx, i_arr + k] += (-h[i_arr])
    return G

def reduced_grads(h, n):
    m = n // 2
    G = full_grads(h, n)
    return G[:, :m] - G[:, n - m:n][:, ::-1]

def slsqp_sym(u0, n, maxiter=300, ftol=1e-14):
    m = n // 2
    dx = 2.0 / n
    h0 = build_full(u0, n)
    theta0 = dx * np.max(compute_corr(h0)) + 1e-6
    z0 = np.concatenate([u0, [theta0]])

    def obj(z):
        return z[m]

    def obj_j(z):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

    def con(z):
        u = np.clip(z[:m], 0, 1)
        theta = z[m]
        h = build_full(u, n)
        corr = compute_corr(h)
        vals = theta - dx * corr
        G_red = reduced_grads(h, n)
        J = np.zeros((2 * n - 1, m + 1))
        J[:, :m] = -dx * G_red
        J[:, m] = 1.0
        return vals, J

    bounds = [(0.0, 1.0)] * m + [(theta0 * 0.3, 1.0)]
    res = minimize(obj, z0, jac=obj_j,
                   constraints={'type': 'ineq', 'fun': con},
                   bounds=bounds, method='SLSQP',
                   options={'maxiter': maxiter, 'ftol': ftol})
    u_opt = np.clip(res.x[:m], 0.0, 1.0)
    h_opt = build_full(u_opt, n)
    return h_opt, score_h(h_opt)

    theta0 = dx * np.max(compute_corr(h0)) + 1e-6
    z0 = np.concatenate([h0, [theta0]])

        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

        vals = theta - dx * corr

        J[:, n] = 1.0
        return vals, J

    def con_wrapped(z):
        return con(z[:n], z[n])

        return np.sum(z[:n]) - n / 2.0

        return g

        active = corr >= (cmax - band)
        G_red = reduced_grads(h, n)
        grad = G_red[active, :].sum(axis=0)
        nrm = np.linalg.norm(grad)
        if nrm < 1e-12:
            break
        d = -grad / nrm
        us = base_step * (0.5 ** (st / max(1, n_steps // 3)))
        for alpha_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = us * alpha_mult
            u_new = np.clip(u0 + alpha * d, 0.0, 1.0)
            h_new = build_full(u_new, n)
            s_new = score_h(h_new)
            if s_new < best_s - 1e-12:
                best_s = s_new
                best_u = u_new.copy()
                h = h_new
                u0 = u_new.copy()
                break
    return best_u, h, best_s

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_score = 1e9

    def update_best(h):
        nonlocal best_h, best_score
        s = score_h(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()

    # Load initial
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = np.clip(h_init, 0.0, 1.0)
    target_sum = n_init / 2.0
    diff = target_sum - np.sum(h_init)
    if abs(diff) > 1e-10:
        mask = (h_init > 0.01) & (h_init < 0.99)
        if np.sum(mask) > 0:
            h_init[mask] += diff / np.sum(mask)
            h_init = np.clip(h_init, 0.0, 1.0)
    update_best(h_init)

    # Phase 1: Symmetric SLSQP on initial
    if time_left() > 60:
        u_init = symmetrize(h_init)
        h_s, s_s = slsqp_sym(u_init, n_init, maxiter=400, ftol=1e-14)
        update_best(h_s)

    # Phase 2: Non-symmetric SLSQP on initial
    if time_left() > 120:
        h_ns, s_ns = slsqp_nonsym(h_init, n_init, maxiter=400, ftol=1e-13)
        update_best(h_ns)

    # Phase 3: Subgradient refinement on best so far
    if time_left() > 80:
        n_b = len(best_h)
        u_b = symmetrize(best_h)
        u_r, h_r, s_r = subgrad_refine(u_b, n_b, n_steps=80)
        update_best(h_r)

    # Phase 4: Try other n values
    n_tries = [30, 35, 40, 50, 55, 60, 70, 80]
    x_init = np.linspace(0, 2, n_init, endpoint=False)

    for n in n_tries:
        if time_left() < 30:
            break
        m = n // 2
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        u_interp = symmetrize(h_interp)

        # Symmetric SLSQP
        h_s, s_s = slsqp_sym(u_interp, n, maxiter=250, ftol=1e-13)
        update_best(h_s)

        if time_left() < 20:
            continue

        # Also try from best solution interpolated
        n_best = len(best_h)
        if n_best != n:
            x_best = np.linspace(0, 2, n_best, endpoint=False)
            h_best_interp = np.interp(x_new, x_best, best_h)
            u_bi = symmetrize(np.clip(h_best_interp, 0.0, 1.0))
            h_s2, s_s2 = slsqp_sym(u_bi, n, maxiter=200, ftol=1e-13)
            update_best(h_s2)

        # Non-symmetric on interpolated
        if time_left() > 40:
            h_ns, s_ns = slsqp_nonsym(h_interp, n, maxiter=200, ftol=1e-13)
            update_best(h_ns)

    # Phase 5: Random perturbations and re-optimize best
    if time_left() > 60:
        n_b = len(best_h)
        u_b = symmetrize(best_h)
        for _ in range(5):
            if time_left() < 15:
                break
            u_pert = np.clip(u_b + rng.normal(0, 0.03, n_b // 2), 0.0, 1.0)
            h_p, s_p = slsqp_sym(u_pert, n_b, maxiter=200, ftol=1e-14)
            update_best(h_p)

    # Phase 6: Final refinement
    if time_left() > 30 and best_h is not None:
        n_b = len(best_h)
        u_b = symmetrize(best_h)
        h_fin, s_fin = slsqp_sym(u_b, n_b, maxiter=500, ftol=1e-15)
        update_best(h_fin)

    # Final feasibility cleanup
    if best_h is not None:
        n_f = len(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        target = n_f / 2.0
        d = target - np.sum(best_h)
        if abs(d) > 1e-12:
            mask = (best_h > 0.005) & (best_h < 0.995)
            if np.sum(mask) > 0:
                best_h[mask] += d / np.sum(mask)
                best_h = np.clip(best_h, 0.0, 1.0)
            else:
                best_h += d / n_f
                best_h = np.clip(best_h, 0.0, 1.0)

    final_score = score_h(best_h)
    return best_h, final_score, len(best_h)
