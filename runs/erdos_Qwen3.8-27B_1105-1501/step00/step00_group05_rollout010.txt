    def slsqp_polish(h0, target_sum, max_seconds=120):
        n = len(h0)
        dx = 2.0 / n
        t_sl = time.time()
        beta = 150.0
        
        def obj(x):
            x = np.clip(x, 0.0, 1.0)
            corrs = get_corrs(x)
            m = np.max(corrs)
            s = corrs - m
            return m + np.log(np.sum(np.exp(beta * s))) / beta
        
        def obj_grad(x):
            x = np.clip(x, 0.0, 1.0)
            corrs = get_corrs(x)
            m = np.max(corrs)
            s = corrs - m
            e = np.exp(beta * s)
            w = e / np.sum(e)
            grad = np.zeros(n)
            for idx in range(len(w)):
                k = idx
                shift = k - (n - 1)
                for i in range(n):
                    g = 0.0
                    j1 = i + shift
                    if 0 <= j1 < n:
                        g += (1.0 - x[j1])
                    j2 = i - shift
                    if 0 <= j2 < n:
                        g -= x[j2]
                    grad[i] += w[idx] * g
            return grad * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                 'jac': lambda x: np.ones(n)}]
        bnds = [(0, 1)] * n
        try:
            res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-15})
            if res.success:
                new_h = project(res.x, target_sum)
                new_c = c5_of(new_h)
                return new_h, new_c
        except Exception:
            pass
        return h0, c5_of(h0)
    
    # === Main ===
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target_sum = n / 2.0
    h = project(h, target_sum)
    
    best_c5 = c5_of(h)
    best_h = h.copy()
    print(f"Initial n={n}, C5={best_c5:.6f}")
    
    # Phase 1: LP active-set on initial
    h, best_c5 = lsqp_optimize(h, target_sum, max_seconds=300)
    best_h = h.copy()
    print(f"After LP: C5={best_c5:.6f}")
    
    # Phase 2: SLSQP polish
    if time.time() - t_start < budget_s - 60:
        h_p, c_p = slsqp_polish(h, target_sum, max_seconds=180)
        if c_p < best_c5:
            best_c5 = c_p
            best_h = h_p.copy()
            print(f"After SLSQP: C5={best_c5:.6f}")
    
    # Phase 3: Another LP pass after SLSQP
    if time.time() - t_start < budget_s - 60:
        h2, c2 = lsqp_optimize(best_h, target_sum, max_seconds=120)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
            print(f"After LP2: C5={best_c5:.6f}")
    
    # Phase 4: Try other n values
    for n_try in [40, 44, 46, 48, 50, 52]:
        if time.time() - t_start > budget_s - 45:
            break
        if n_try == n:
            continue
        n_t = n_try
        ds_t = 2.0 / n_t
        ts_t = n_t / 2.0
        
        # Interpolate from best solution
        x_old = np.linspace(0, 2, n, endpoint=False) + ds_t / 2
        x_new = np.linspace(0, 2, n_t, endpoint=False) + ds_t / 2
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_t = project(h_interp, ts_t)
        
        c_t = c5_of(h_t)
        # Quick LP
        h_t, c_t2 = lsqp_optimize(h_t, ts_t, max_seconds=80)
        if c_t2 < best_c5:
            best_c5 = c_t2
            best_h = h_t.copy()
            n = n_t
            target_sum = ts_t
            print(f"Improved with n={n_t}: C5={best_c5:.6f}")
    
    # Phase 5: Random perturbations from best
    if time.time() - t_start < budget_s - 50:
        for attempt in range(8):
            if time.time() - t_start > budget_s - 30:
                break
            perturb = np.random.randn(n) * 0.008
            h_r = project(best_h + perturb, target_sum)
            c_r = c5_of(h_r)
            if c_r < best_c5:
                h_r, c_r2 = lsqp_optimize(h_r, target_sum, max_seconds=60)
                if c_r2 < best_c5:
                    best_c5 = c_r2
                    best_h = h_r.copy()
                    print(f"Random improve: C5={best_c5:.6f}")
    
    # Final SLSQP
    if time.time() - t_start < budget_s - 20:
        h_f, c_f = slsqp_polish(best_h, target_sum, max_seconds=90)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()
    
    best_h = project(best_h, target_sum)
    final_c5 = c5_of(best_h)
    
    return best_h, final_c5, len(best_h)
