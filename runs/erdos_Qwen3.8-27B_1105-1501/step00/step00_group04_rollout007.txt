    order = np.argsort(-h)

    # Old midpoints: (2*i+1)/(2*old_n) * 2 = (2*i+1)/old_n for i in 0..old_n-1
    # New midpoints: (2*j+1)/(2*new_n) * 2 = (2*j+1)/new_n for j in 0..new_n-1
    old_pts = (np.arange(old_n) + 0.5) / old_n  # in [0,1]

    h_new = np.interp(new_pts, old_pts, h)
    return repair_feasible(h_new, new_n)
