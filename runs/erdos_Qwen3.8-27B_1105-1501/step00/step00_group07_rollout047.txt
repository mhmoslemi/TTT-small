def local_search(h, n, budget_sec, rng, T_init=0.001, T_min=1e-10, delta_max=0.05):
    cur_c5 = score_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    start = time.time()
    stall = 0
    
    while time.time() - start < budget_sec:
        # Adaptive step size
        scale = min(delta_max, 0.01 + 0.04 * (T / T_init))
        i = rng.randint(n)
        j = rng.randint(n)
        attempts = 0
        while i == j and attempts < 10:
            j = rng.randint(n)
            attempts += 1
        if i == j:
            continue
        
        d_pos = min(1.0 - h[i], h[j])
        d_neg = min(h[i], 1.0 - h[j])
        
        if d_pos < 1e-12 and d_neg < 1e-12:
            continue
        
        # Choose direction and magnitude
        if d_pos > 0 and (d_neg <= 0 or rng.random() < 0.5):
            delta = rng.uniform(1e-5, min(scale, d_pos))
        else:
            delta = -rng.uniform(1e-5, min(scale, d_neg))
        
        old_i, old_j = h[i], h[j]
        h[i] = old_i + delta
        h[j] = old_j - delta
        
        new_c5 = score_exact(h)
        d_obj = new_c5 - cur_c5
        
        if d_obj <= 0:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        elif T > T_min and rng.random() < np.exp(-d_obj / T):
            cur_c5 = new_c5
        else:
            h[i], h[j] = old_i, old_j
            stall += 1
        
        T = max(T_min, T * 0.999998)
        
        # Restart if stuck
        if stall > 5000:
            # Large random perturbation
            for _ in range(20):
                ii = rng.randint(n)
                jj = rng.randint(n)
                if ii != jj:
                    dp = min(1.0 - h[ii], h[jj])
                    dn = min(h[ii], 1.0 - h[jj])
                    if rng.random() < 0.5 and dp > 0:
                        d = rng.uniform(0.01, min(0.1, dp))
                        h[ii] += d
                        h[jj] -= d
                    elif dn > 0:
                        d = rng.uniform(0.01, min(0.1, dn))
                        h[ii] -= d
                        h[jj] += d
            cur_c5 = score_exact(h)
            stall = 0
            T = T_init
    
    return best_h, best_c5
