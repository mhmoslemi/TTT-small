```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def make_feasible(z, target=None):
    n = len(z)
    if target is None:
        target = n / 2.0
    z = np.asarray(z, dtype=float)
    # Bisection for lambda such that sum(clip(z + lambda, 0, 1)) = target
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction for tiny residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
            residual -= add
        else:
            idx = np.argmax(h)
            sub = min(-residual, h[idx])
            h[idx] -= sub
            residual += sub
        # If still residual (extreme case), spread among all
        if abs(residual) > 1e-15:
            free = np.abs(h) < 0.999
            if free.any():
                per = residual / free.sum()
                h[free] += per
                h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def corr_values(h):
    """Compute the 2n-1 correlation values (unscaled)."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h):
    """Compute the (2n-1) x n Jacobian of corr_values w.r.t. h."""
    n = len(h)
    R = 2 * n - 1
    J = np.zeros((R, n))
    for r in range(R):
        lag = r - (n - 1)  # lag ranges from -(n-1) to (n-1)
        # c[r] = sum_i h[i] * (1 - h[i+lag]) for valid i
        # dc[r]/dh[p] = (1-h[p+lag]) if p+lag valid, minus h[p-lag] if p-lag valid
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        # First term: p = i, p+lag = i+lag
        for i in range(i_start, i_end):
            J[r, i] += (1.0 - h[i + lag])
            J[r, i + lag] += (-h[i])
    return J

def smooth_obj_and_grad(h, beta):
    """Smooth max surrogate and its gradient."""
    n = len(h)
    dx = 2.0 / n
    vals = corr_values(h)  # length 2n-1
    U = vals.max()
    shifted = beta * (vals - U)
    # Numerical stability
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    obj = dx * (U + np.log(exps.sum()) / beta)
    J = corr_gradient(h)  # (2n-1) x n
    grad = dx * (J.T @ w)
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel()
    n = len(z)
    if target is None:
        target = n / 2.0
    if not np.all(np.isfinite(z)):
        return np.full(n, target / n)
    lo = -1.0 - z.max() - 1e-8
    hi = 1.0 - z.min() + 1e-8
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                if add <= 0:
                    break
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                sub = min(-residual, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h):
    n = len(h)
    R = 2 * n - 1
    J = np.zeros((R, n))
    for r in range(R):
        lag = r - (n - 1)
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        if i_end > i_start:
            ii = np.arange(i_start, i_end)
            jj = ii + lag
            J[r, ii] += (1.0 - h[jj])
            J[r, jj] += (-h[ii])
    return J

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = corr_values(h)
    U = vals.max()
    shifted = np.clip(beta * (vals - U), -50.0, 50.0)
    exps = np.exp(shifted)
    s = exps.sum()
    if s < 1e-300:
        return dx * U
    return dx * (U + np.log(s) / beta)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = corr_values(h)
    U = vals.max()
    shifted = np.clip(beta * (vals - U), -50.0, 50.0)
    exps = np.exp(shifted)
    s = exps.sum()
    if s < 1e-300:
        w = np.ones_like(vals) / len(vals)
    else:
        w = exps / s
    J = corr_gradient(h)
    return dx * (J.T @ w)

def eq_fun(h, n):
    return h.sum() - n / 2.0

def eq_jac(h, n):
    return np.ones(n)

def minimax_obj(z, n):
    return z[n]

def minimax_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def minimax_eq_fun(z, n):
    return z[:n].sum() - n / 2.0

def minimax_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def minimax_ineq_fun(z, n):
    h = z[:n]
    theta = z[n]
    dx = 2.0 / n
    return theta - dx * corr_values(h)

def minimax_ineq_jac(z, n):

    dx = 2.0 / n
    R = 2 * n - 1
    J = corr_gradient(h)

    return G

def interpolate_h(h, new_n):
    n = len(h)
    if n == new_n:

    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)

    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    # Get parent
    parent_h = None
    parent_n = 40
    try:
        raw = np.asarray(initial_h_values, dtype=float).ravel()
        if np.all(np.isfinite(raw)) and len(raw) >= 10:
            parent_n = len(raw)
            parent_h = make_feasible(raw.copy())
    except (NameError, Exception):
        pass
    if parent_h is None:
        parent_h = make_feasible(np.full(parent_n, 0.5))

    best_h = parent_h.copy()
    best_score = official_score(best_h)
    best_n = parent_n

    # Determine resolutions
    resolutions = [best_n]
    for alpha in [0.75, 0.85]:
        nn = max(20, int(round(alpha * best_n)))
        if nn not in resolutions:
            resolutions.append(nn)
    if budget_s > 500:
        for alpha in [1.2, 1.4]:
            nn = int(round(alpha * best_n))
            if 20 <= nn <= 80 and nn not in resolutions:
                resolutions.append(nn)
    resolutions = sorted(set(resolutions))

    def do_smooth_solve(h_init, n, beta, maxiter, ftol):
        bounds = [(0.0, 1.0)] * n
        constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}
        res = minimize(smooth_obj, h_init, args=(beta,),
                       method='SLSQP', jac=smooth_grad,
                       bounds=bounds, constraints=[constraint],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x if res.x is not None else None

    def do_minimax_solve(h_init, n, maxiter, ftol):
        dx = 2.0 / n
        theta0 = official_score(h_init) + 1e-5
        z0 = np.concatenate([h_init, [theta0]])
        eq_c = {'type': 'eq', 'fun': minimax_eq_fun, 'jac': minimax_eq_jac, 'args': (n,)}
        ineq_c = {'type': 'ineq', 'fun': minimax_ineq_fun, 'jac': minimax_ineq_jac, 'args': (n,)}
        bounds_mm = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        res = minimize(minimax_obj, z0, args=(n,),
                       method='SLSQP', jac=minimax_obj_jac,
                       bounds=bounds_mm, constraints=[eq_c, ineq_c],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x if res.x is not None else None

    # Phase 1: Optimize at each resolution
    for n_target in resolutions:
        if time.time() > deadline - 50:
            break

        # Generate starts
        starts = []
        h_int = make_feasible(interpolate_h(parent_h, n_target))
        starts.append(h_int)
        starts.append(make_feasible(antisym_transform(h_int)))
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            starts.append(make_feasible(h_int + sigma * rng.standard_normal(n_target)))
        h_struct = make_feasible(np.sin(np.linspace(0, np.pi, n_target)) * 0.6 + 0.2 + 0.05 * rng.standard_normal(n_target))
        starts.append(h_struct)

        # Score and sort
        scored = []
        for s in starts:
            sc = official_score(s)
            if np.isfinite(sc):
                scored.append((sc, s))
        scored.sort(key=lambda x: x[0])

        # Optimize top candidates
        max_cands = min(len(scored), 6)
        for ci in range(max_cands):
            if time.time() > deadline - 50:
                break
            cur = scored[ci][1].copy()
            cur_score = scored[ci][0]

            # Smooth optimization ladder
            betas = [15, 60, 250, 1000]
            for beta in betas:
                if time.time() > deadline - 50:
                    break
                try:
                    maxiter = min(80, max(30, int(2500 / n_target)))
                    res_x = do_smooth_solve(cur, n_target, beta, maxiter, 1e-10)
                    if res_x is not None and np.all(np.isfinite(res_x)):
                        new_h = make_feasible(res_x)
                        new_sc = official_score(new_h)
                        if new_sc < cur_score:
                            cur = new_h
                            cur_score = new_sc
                except Exception:
                    pass

            # Direct minimax
            if time.time() > deadline - 50:
                break
            try:
                maxiter_mm = min(120, max(50, int(8000 / n_target)))
                res_z = do_minimax_solve(cur, n_target, maxiter_mm, 1e-10)
                if res_z is not None and np.all(np.isfinite(res_z)):
                    new_h = make_feasible(res_z[:n_target])
                    new_sc = official_score(new_h)
                    if new_sc < cur_score:
                        cur = new_h
                        cur_score = new_sc
            except Exception:
                pass

            # Update global
            if cur_score < best_score - 1e-12:
                best_h = cur.copy()
                best_score = cur_score
                best_n = n_target

    # Phase 2: Refine best with direct minimax from multiple angles
    if time.time() < deadline - 60:
        n_ref = best_n
        h_ref = best_h.copy()
        # Try minimax with slightly perturbed starts
        for perturb_sigma in [1e-4, 5e-4, 1e-3]:
            if time.time() > deadline - 60:
                break
            for trial in range(3):
                if time.time() > deadline - 60:
                    break
                try:
                    h_try = make_feasible(h_ref + perturb_sigma * rng.standard_normal(n_ref))
                    maxiter_mm = min(100, max(40, int(6000 / n_ref)))
                    res_z = do_minimax_solve(h_try, n_ref, maxiter_mm, 1e-11)
                    if res_z is not None and np.all(np.isfinite(res_z)):
                        new_h = make_feasible(res_z[:n_ref])
                        new_sc = official_score(new_h)
                        if new_sc < best_score - 1e-12:
                            best_h = new_h
                            best_score = new_sc
                            best_n = n_ref
                except Exception:
                    pass

    # Phase 3: Interpolate best to other resolutions and refine
    if time.time() < deadline - 60:
        for n_alt in resolutions:
            if n_alt == best_n or time.time() > deadline - 60:
                continue
            try:
                h_alt = make_feasible(interpolate_h(best_h, n_alt))
                alt_score = official_score(h_alt)
                if alt_score < best_score:
                    best_h = h_alt
                    best_score = alt_score
                    best_n = n_alt
                    continue
                # Try to optimize at this resolution
                maxiter_mm = min(100, max(40, int(6000 / n_alt)))
                res_z = do_minimax_solve(h_alt, n_alt, maxiter_mm, 1e-10)
                if res_z is not None and np.all(np.isfinite(res_z)):
                    new_h = make_feasible(res_z[:n_alt])
                    new_sc = official_score(new_h)
                    if new_sc < best_score - 1e-12:
                        best_h = new_h
                        best_score = new_sc
                        best_n = n_alt
            except Exception:
                pass

    # Phase 4: Final smooth polish on best
    if time.time() < deadline - 30:
        try:
            for beta in [500, 2000]:
                if time.time() > deadline - 30:
                    break
                maxiter = min(60, max(30, int(2000 / best_n)))
                res_x = do_smooth_solve(best_h, best_n, beta, maxiter, 1e-11)
                if res_x is not None and np.all(np.isfinite(res_x)):
                    new_h = make_feasible(res_x)
                    new_sc = official_score(new_h)
                    if new_sc < best_score - 1e-12:
                        best_h = new_h
                        best_score = new_sc
        except Exception:
            pass

    # Final evaluation
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    safe_bound = final_score + 1e-13
    return (best_h, safe_bound, len(best_h))
```
