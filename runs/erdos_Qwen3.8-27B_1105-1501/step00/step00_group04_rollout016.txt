        h_parent = None

    if h_parent is None or len(h_parent) < 2:
        h_parent = make_feasible(rng.random(70))

    h_parent = make_feasible(h_parent)
    best_h = h_parent.copy()
    best_c = score_exact(h_parent)
    n = len(h_parent)

    # Phase 1: Gradient descent (45% time)
    p1_dead = start_time + budget_s * 0.45
    best_h, best_c = local_refine(best_h, p1_dead, rng)

    # Phase 2: Pair line + small refinement (20% time)
    p2_dead = start_time + budget_s * 0.65
    while time.monotonic() < p2_dead:
        sub_dead = min(time.monotonic() + 20, p2_dead)
        best_h, best_c = pair_line_search(best_h, 30, rng, deadline=sub_dead)
        h_mut = make_feasible(best_h + rng.normal(0, 0.015, n))
        c_mut = score_exact(h_mut)
        if c_mut < best_c - 1e-12:
            best_h = h_mut
            best_c = c_mut
        sub_dead2 = min(time.monotonic() + 15, p2_dead)
        h_ref, c_ref = local_refine(best_h, sub_dead2, rng)
        if c_ref < best_c - 1e-12:
            best_h = h_ref
            best_c = c_ref

    # Phase 3: Binary search (10% time)
    p3_dead = start_time + budget_s * 0.75
    h_bin = binarize(best_h)
    c_bin = score_exact(h_bin)
    if c_bin < best_c - 1e-12:
        best_h = h_bin
        best_c = c_bin
    h_bls, c_bls = binary_local_search(best_h, min(p3_dead, time.monotonic() + budget_s * 0.08), rng)
    if c_bls < best_c - 1e-12:
        best_h = h_bls
        best_c = c_bls
    h_ref, c_ref = local_refine(best_h, p3_dead, rng)
    if c_ref < best_c - 1e-12:
        best_h = h_ref
        best_c = c_ref

    # Phase 4: Multi-start exploration (25% time)
    p4_dead = deadline
    strat_idx = 0
    while time.monotonic() < p4_dead:
        strat = strat_idx % 6
        strat_idx += 1
        if strat == 0:
            h_try = make_feasible(best_h + rng.normal(0, 0.03, n))
        elif strat == 1:
            h_try = make_feasible(best_h + rng.normal(0, 0.2, n))
        elif strat == 2:
            h_try = binarize(best_h)
            h_try = make_feasible(h_try + rng.normal(0, 0.1, n))
        elif strat == 3:
            new_n = int(rng.choice([max(30, n - 10), max(30, n - 20), n + 10, n + 20]))
            new_n = max(24, min(140, new_n))
            h_try = resample_to_resolution(best_h, new_n)
        elif strat == 4:
            h_try = make_feasible(rng.random(n))
        else:
            h_try = make_feasible(best_h + 0.3 * rng.normal(0, 1, n))

        c_try = score_exact(h_try)
        if c_try < best_c - 1e-10:
            best_h = h_try
            best_c = c_try
            sub_dead = min(time.monotonic() + 25, p4_dead)
            h_ref, c_ref = local_refine(best_h, sub_dead, rng)
            if c_ref < best_c - 1e-12:
                best_h = h_ref
                best_c = c_ref
        elif strat == 3 and len(h_try) != n:
            h_resampled = resample_to_resolution(h_try, n)
            c_resampled = score_exact(h_resampled)
            if c_resampled < best_c - 1e-10:
                best_h = h_resampled
                best_c = c_resampled

    # Final validation
    best_h = make_feasible(best_h)
    final_c = score_exact(best_h)
    if final_c <= best_c + 1e-12:
        best_c = final_c

    return best_h, best_c, len(best_h)
