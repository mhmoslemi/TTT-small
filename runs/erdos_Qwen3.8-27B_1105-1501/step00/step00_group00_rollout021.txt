import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def _smooth_obj(h, tau):

    dx = 2.0 / n

    s = tau * dx * corr
    m = np.max(s)
    return (m + np.log(np.sum(np.exp(s - m)))) / tau

def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = tau * dx * corr
    m = np.max(s)
    w = np.exp(s - m)
    w_sum = np.sum(w)
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = tau * dx * w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n - 1, n - 1 - shift)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wk * (1.0 - h[lo1 + shift:hi1 + 1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n - 1, n - 1 + shift)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += wk * (-h[lo2 - shift:hi2 + 1 - shift])
    return grad

                diff -= amt
            else:

                h[i] -= amt
