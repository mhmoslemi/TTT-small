    n = len(y)

    lo = y.min() - 1.0

        else:

    n = len(x)

                grad[neg] -= weights[m] * dx * x[neg + d]

            pos = np.arange(0, n - p)

            neg = np.arange(p, n)
            grad[neg] -= weights[m] * dx * x[neg - p]

import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(y):
    n = len(y)
    if n == 0:
        return y.copy()
    target = n / 2.0
    y = np.array(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    # Bisection for lambda
    lo = float(y.min()) - 1.5
    hi = float(y.max()) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    # Exact mass fix
    residual = target - float(np.sum(x))
    if residual > 1e-14:
        order = np.argsort(1.0 - x)
        for i in order:
            room = 1.0 - x[i]
            if room >= residual:
                x[i] += residual
                break
            x[i] = 1.0
            residual -= room
    elif residual < -1e-14:
        order = np.argsort(x)
        for i in order:
            if x[i] >= -residual:
                x[i] += residual  # residual is negative
                break
            residual += x[i]
            x[i] = 0.0
    return x

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj(x, args):
    tau, dx = args
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = float(np.max(c))
    if tau < 1e-12:
        return M
    shifted = np.clip((c - M) / tau, -400, 400)
    return M + tau * float(np.log(np.sum(np.exp(shifted))))

def smooth_grad(x, args):
    tau, dx = args
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    M = float(np.max(c))
    shifted = np.clip((c - M) / tau, -400, 400)
    exps = np.exp(shifted)
    total = float(np.sum(exps))
    if total < 1e-300:
        # All weights equal
        w = np.full(2*n-1, 1.0/(2*n-1))
    else:
        w = exps / total
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        d = m - (n - 1)
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * x)
        elif d > 0:
            grad[d:] += wm * dx * (1.0 - x[:n-d])
            grad[:n-d] -= wm * dx * x[d:]
        else:
            p = -d
            grad[:n-p] += wm * dx * (1.0 - x[p:])
            grad[p:] -= wm * dx * x[:n-p]
    return grad

    deadline = t0 + budget_s * 0.93

        parent = np.array(initial_h_values, dtype=np.float64).copy()

    best_score = exact_c5(best_h)

        return time.time() - t0

    def remaining():
        return deadline - time.time()

    # Stage 1: Generate diverse starting points
    starts = [best_h.copy()]

    # Binary quantization
    idx_sorted = np.argsort(-best_h)
    binary_h = np.zeros(n)
    binary_h[idx_sorted[:n//2]] = 1.0
    if n % 2 == 1:

    starts.append(project_feasible(binary_h))

    tau_list = [0.05, 0.02, 0.008, 0.003, 0.001]
