            if time.time() - start_time > budget_s * 0.88:
                break
            h_p = project_feasible(pat, n0 * 0.5)
            h_opt, v_opt = optimize_pg(h_p, [20, 50, 100], max_iter=100, alpha_init=0.6)
            if v_opt < best_val:
                best_h, best_val = h_opt, v_opt

    # Phase 6: Final polish
    if time.time() - start_time < budget_s - 15:
        h_opt, v_opt = polish_active(best_h, max_rounds=1000)
        if v_opt < best_val:
            best_h, best_val = h_opt, v_opt

    # Final verification
    final_val = score(best_h)
    if final_val != best_val:
        best_val = final_val

    return best_h, best_val, len(best_h)
