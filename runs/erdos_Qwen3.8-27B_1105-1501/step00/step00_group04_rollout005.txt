def solve_qp_for_direction(h, active_indices, Y, M, rho, eta_factor, strict=False):
    n = len(h)
    m = len(active_indices)
    if m == 0:
        return None
    
    # Build gradient rows
    G = np.array([gradient_row(h, p, n) for p in active_indices])
    # Check for any non-zero gradient
    if np.all(np.abs(G) < 1e-9):
        return None
    
    u = cp.Variable(n)
    s = cp.Variable(m, nonneg=True) if not strict else cp.Constant(0)
    
    # Target reduction per active constraint
    eta = np.maximum(1e-4, eta_factor * rho * np.linalg.norm(G, axis=1))
    # Cap eta
    eta = np.minimum(eta, 0.1 * max(1.0, M))
    
    constraints = [
        u >= -h,
        u <= 1.0 - h,
        cp.sum(u) == 0
    ]
    
    for i in range(m):
        rhs = M - Y[active_indices[i]] - eta[i]
        if strict:
            constraints.append(G[i] @ u <= rhs)
        else:
            constraints.append(G[i] @ u <= rhs + s[i])
    
    if strict:
        obj = cp.Minimize(0.5 * cp.sum_squares(u))

def make_feasible(y, n):
    y = np.asarray(y, dtype=float).flatten()
    target = n / 2.0
    y = np.clip(y, 0.0, 1.0)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - z.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(z < 1.0 - 1e-12)[0]
            for i in idx:
                add = min(r, 1.0 - z[i])
                z[i] += add
                r -= add
                if abs(r) < 1e-12:
                    break
        else:
            idx = np.where(z > 1e-12)[0]
            for i in idx:
                sub = min(-r, z[i])
                z[i] -= sub
                r += sub
                if abs(r) < 1e-12:
                    break
    return z

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(corr))

def gradient_row(h, p, n):
    center = n - 1
    d = center - p
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        idx1 = np.arange(0, n - d)
        g[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        g[idx2] += (-h[idx2 - d])
    else:
        dp = -d
        idx1 = np.arange(dp, n)
        g[idx1] += (1.0 - h[idx1 - dp])
        idx2 = np.arange(0, n - dp)
        g[idx2] += (-h[idx2 + dp])
    return g

def solve_qp_direction(h, active_p, Y, M, rho, eta_factor, strict):
    n = len(h)
    m = len(active_p)
    if m == 0:
        return None
    G = np.array([gradient_row(h, p, n) for p in active_p])
    if np.all(np.abs(G) < 1e-9):
        return None
    u = cp.Variable(n)
    if strict:
        slack = None
    else:
        slack = cp.Variable(m, nonneg=True)
    norms = np.linalg.norm(G, axis=1)
    eta = np.maximum(5e-4, eta_factor * rho * norms)
    eta = np.minimum(eta, 0.12 * max(1.0, M))
    constraints = [u >= -h, u <= 1.0 - h, cp.sum(u) == 0]
    for i in range(m):
        rhs = M - Y[active_p[i]] - eta[i]
        if strict:
            constraints.append(G[i] @ u <= rhs)
        else:
            constraints.append(G[i] @ u <= rhs + slack[i])
    if strict:
        obj = cp.Minimize(0.5 * cp.sum_squares(u))
    else:
        sw = 30.0 / max(rho, 0.02)
        obj = cp.Minimize(0.5 * cp.sum_squares(u) + sw * cp.sum(slack))
    prob = cp.Problem(obj, constraints)
    solvers = [cp.CLARABiQA, cp.SCS, cp.ECOS]
    status = None
    for sv in solvers:
        try:
            prob.solve(solver=sv, verbose=False)
            status = prob.status
            if status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    if status not in ('optimal', 'optimal_inaccurate'):
        return None
    if u.value is None:
        return None
    u_val = np.array(u.value, dtype=float).flatten()
    if not np.all(np.isfinite(u_val)):

    u_val -= np.mean(u_val)
    nrm = np.linalg.norm(u_val)

        return None

    return u_val

def safe_alpha(h, u):
    """Max feasible alpha for h + alpha*u"""
    alpha = 10.0
    pos = u > 1e-14
    neg = u < -1e-14
    if np.any(pos):
        alpha = min(alpha, np.min((1.0 - h[pos]) / u[pos]))
    if np.any(neg):
        alpha = min(alpha, np.min(h[neg] / (-u[neg])))
    return max(alpha, 1e-12)

def line_search(h, u, dx, current_c5):
    """Try several alpha values, return best (alpha, candidate, score)"""
    best_alpha = 0.0
    best_h = None
    best_c = current_c5

    candidates_alpha = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]
    for a in candidates_alpha:
        a = min(a, amax)
        if a <= 0:

            best_alpha = a

def pair_transfer_step(h, rng, dx, current_c5, max_pairs=30):
    """Try random pair transfers, return best improvement"""
    n = len(h)
    best_c = current_c5
    best_h = None
    for _ in range(max_pairs):
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            continue
        # Move mass from b to a: u = e_a - e_b
        # Feasible: h[a]+alpha <= 1, h[b]-alpha >= 0

            trial[b] -= alpha
