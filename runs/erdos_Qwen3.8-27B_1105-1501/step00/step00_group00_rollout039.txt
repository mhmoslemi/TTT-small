def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_box_halfmass(y):
    n = len(y)
    target = n / 2.0
    y = np.array(y, dtype=np.float64).copy()
    # Replace non-finite values
    mask = ~np.isfinite(y)
    if np.any(mask):
        y[mask] = 0.5
    # Bisection on lambda
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    # Fix mass exactly
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        # Find a coordinate to adjust
        if diff > 0:
            # Need to increase sum
            idx = np.argmin(1.0 - z)  # Most room to increase... actually argmax of (1-z)
            idx = np.argmax(1.0 - z)
        else:
            idx = np.argmin(z)
        z[idx] += diff
        z[idx] = np.clip(z[idx], 0.0, 1.0)
        # If clipping caused issue, try another coordinate
        diff2 = target - np.sum(z)
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                candidates = np.where(1.0 - z >= diff2)[0]
            else:
                candidates = np.where(z >= -diff2)[0]
            if len(candidates) > 0:
                z[candidates[0]] += diff2
                z[candidates[0]] = np.clip(z[candidates[0]], 0.0, 1.0)
            else:
                # Distribute across all
                z[:] += diff2 / n
                z = np.clip(z, 0.0, 1.0)
    return z

def grad_c(h, dx, m):
    """Gradient of C_m w.r.t. h, where m is the index in the full correlation array."""
    n = len(h)
    grad = np.zeros(n)
    l = m - (n - 1)
    if l >= 0:
        # C_m = dx * sum_{i=0}^{n-1-l} h[i+l] * (1-h[i])
        # d/dh[j] for j in [l, n-1]: dx*(1-h[j-l])
        # d/dh[j] for j in [0, n-1-l]: -dx*h[j+l]
        for j in range(l, n):
            grad[j] += dx * (1.0 - h[j - l])
        for j in range(0, n - l):
            grad[j] -= dx * h[j + l]
    else:
        p = -l  # p > 0
        # C_m = dx * sum_{i=0}^{n-1-p} h[i] * (1-h[i+p])
        # d/dh[j] for j in [0, n-1-p]: dx*(1-h[j+p])
        # d/dh[j] for j in [p, n-1]: -dx*h[j-p]
        for j in range(0, n - p):
            grad[j] += dx * (1.0 - h[j + p])
        for j in range(p, n):
            grad[j] -= dx * h[j - p]
    return grad

def smoothed_obj_and_grad(x, dx, tau):
    """Compute smoothed objective and its gradient."""
    cvals = c_values(x)
    M = np.max(cvals)
    # For numerical stability
    shifted = (cvals - M) / tau
    shifted = np.clip(shifted, -700, 700)
    exps = np.exp(shifted)
    S = M + tau * np.log(np.sum(exps))
    w = exps / np.sum(exps)
    # Gradient

    grad = np.zeros(n)
    for k in range(len(cvals)):
        if w[k] > 1e-10:
            grad += w[k] * grad_c(x, dx, k)
    return S, grad

def project_box_halfmass(y):
    n = len(y)
    if n == 0:
        return y
    target = n / 2.0

        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break

    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - np.sum(z)

        else:

        diff2 = target - np.sum(z)
        if abs(diff2) > 1e-12:

            else:

                    z[idx2] += diff2
                    break
                elif diff2 < 0 and z[idx2] >= abs(diff2):
                    z[idx2] += diff2
                    break
            else:
                z[:] = np.clip(z + diff2 / n, 0.0, 1.0)

def grad_c_m(h, dx, m):
    n = len(h)
    grad = np.zeros(n)
    l = m - (n - 1)
    if l >= 0:
        grad[l:] += dx * (1.0 - h[:n - l])
        grad[:n - l] -= dx * h[l:]
    else:
        p = -l
        grad[:n - p] += dx * (1.0 - h[p:])
        grad[p:] -= dx * h[:n - p]
    return grad

    cvals = c_values(x)
    M = np.max(cvals)
    shifted = np.clip((cvals - M) / tau, -700.0, 700.0)
    exps = np.exp(shifted)
    S = M + tau * np.log(np.sum(exps) + 1e-300)
    return S

def _smoothed_jac(x):
    dx = _state['dx']
    tau = _state['tau']
    cvals = c_values(x)
    M = np.max(cvals)
    shifted = np.clip((cvals - M) / tau, -700.0, 700.0)

    n = len(x)

    for k in range(len(cvals)):

            grad += w[k] * grad_c_m(x, dx, k)
    return grad

            indices = list(range(n))

                if eval_count >= max_evals:

                hi_val = h[i]
                if hi_val < 1e-12:
