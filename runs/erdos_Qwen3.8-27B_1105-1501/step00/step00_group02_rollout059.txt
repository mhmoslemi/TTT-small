    # So δ ∈ [-hi, 1-hi] ∩ [hj-1, hj]

    delta_hi = min(1 - hi, hj)

        return 0.0

import numpy as np
from scipy.optimize import minimize
import time

_CONST_N = 0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def find_optimal_coord(h, i, n):
    dx = 2.0 / n
    save = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode="full").astype(np.float64) * dx
    h[i]
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode="full").astype(np.float64) * dx
    h[i] = save
    A = corr1 - corr0
    B = corr0
    lo = 0.0
    hi = 1.0
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(A * m1 + B) < np.max(A * m2 + B):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coordinate_descent(h, n, max_rounds, deadline):
    h = enforce_constraints(h, n)
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        for i in range(n):
            h[i] = find_optimal_coord(h, i, n)
        h = enforce_constraints(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if rnd > 50:
            prev_h = h.copy()
            for i in range(n):
                h[i] = find_optimal_coord(h, i, n)
            h = enforce_constraints(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
            if np.max(np.abs(h - prev_h)) < 1e-12:
                break
    return h, best_c5

def find_optimal_pair(h, i, j, n):
    dx = 2.0 / n
    hi0 = h[i]
    hj0 = h[j]
    d_lo = max(-hi0, hj0 - 1.0)
    d_hi = min(1.0 - hi0, hj0)
    if d_hi - d_lo < 1e-14:
        return 0.0
    h[i] = hi0 + d_lo
    h[j] = hj0 - d_lo
    cl = np.correlate(h, 1.0 - h, mode="full").astype(np.float64) * dx
    h[i] = hi0 + d_hi
    h[j] = hj0 - d_hi
    ch = np.correlate(h, 1.0 - h, mode="full").astype(np.float64) * dx
    h[i] = hi0
    h[j] = hj0
    span = d_hi - d_lo
    A = (ch - cl) / span
    B = cl - A * d_lo
    lo = d_lo
    hi = d_hi
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(A * m1 + B) < np.max(A * m2 + B):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def pair_descent(h, n, max_rounds, deadline, rng):
    h = enforce_constraints(h, n)
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(5):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = find_optimal_pair(h, i, j, n)
            if abs(delta) > 1e-12:
                h[i] += delta
                h[j] -= delta
                h = enforce_constraints(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
        if not improved and rnd > 20:
            break
    return h, best_c5

        best_c5 = c5
        best_h = h_opt.copy()

    # Phase 3: Many random restarts
    scales = [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    attempt = 0
    while time.time() < deadline - 20:
        scale = scales[attempt % len(scales)]
        attempt += 1
        noise = rng.randn(n) * scale

        h_pert = enforce_constraints(h_pert, n)
