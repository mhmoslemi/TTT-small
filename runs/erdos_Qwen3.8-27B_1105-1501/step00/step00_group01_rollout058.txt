    def binary_local_search(h, n, max_passes=200):
        """Greedy local search: try all 1-0 swaps, apply best one."""
        dx = 2.0 / n
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        def compute_corr(h):
            return np.correlate(h, 1.0 - h, mode='full')
        
        corr = compute_corr(h)
        current_max = float(np.max(corr))
        improved = True
        passes = 0
        
        while improved and passes < max_passes:
            improved = False
            passes += 1
            best_delta = 0
            best_swap = None
            best_new_max = current_max
            
            for i in ones:
                for j in zeros:
                    # Compute change in corr for swapping i and j
                    # h[i]=1 -> 0, h[j]=0 -> 1
                    change = np.zeros(2 * n - 1)
                    for d in range(-(n-1), n):
                        # d corresponds to index d + n - 1 in corr
                        val = 0.0
                        # (h[i+d]-1) if 0 <= i+d < n
                        ipd = i + d
                        if 0 <= ipd < n:
                            val += (h[ipd] - 1.0)
                        # h[i-d] if 0 <= i-d < n
                        idd = i - d
                        if 0 <= idd < n:
                            val += h[idd]
                        # (1-h[j+d]) if 0 <= j+d < n
                        jpd = j + d
                        if 0 <= jpd < n:
                            val += (1.0 - h[jpd])
                        # -h[j-d] if 0 <= j-d < n
                        jdd = j - d
                        if 0 <= jdd < n:
                            val -= h[jdd]
                        # Correction: +1 if d == j-i or d == i-j
                        if d == j - i or d == i - j:
                            val += 1.0
                        change[d + n - 1] = val
                    
                    new_corr = corr + change
                    new_max = float(np.max(new_corr))
                    if new_max < best_new_max - 1e-10:
                        best_new_max = new_max
                        best_swap = (i, j)
                        best_change = change.copy()
            
            if best_swap is not None:
                i, j = best_swap
                h[i] = 0.0
                h[j] = 1.0
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                corr = corr + best_change
                current_max = best_new_max
                improved = True
        
        return h, current_max * dx
