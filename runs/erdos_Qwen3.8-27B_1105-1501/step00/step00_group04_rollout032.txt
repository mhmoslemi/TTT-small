def score(h):
    """Compute C5 = dx * max(np.correlate(h, 1-h, mode='full'))"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr), corr

def make_feasible(h):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    n = len(h)
    target = n / 2.0
    x = np.clip(h.astype(float), 0.0, 1.0)
    # Bisection for lambda: find lambda such that sum(clip(x - lambda, 0, 1)) = target
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(x - lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(z)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - z)  # most room first
        else:
            idx = np.argsort(z)  # least first
        remaining = residual
        for i in idx:
            if remaining > 0:
                delta = min(remaining, 1.0 - z[i])
                z[i] += delta
                remaining -= delta
            else:
                delta = max(remaining, -z[i])
                z[i] += delta
                remaining -= delta
            if abs(remaining) < 1e-14:
                break
    return z

I can precompute a_p and b_p once per coordinate j, then sweep t efficiently.

But I can do better. Instead of a grid, I can find the minimum analytically:
- The function f(t) = max(yc(t), max_p (a_p*t + b_p)) where yc is concave and the rest are linear
- The minimum occurs at:
  1. The point where yc(t) = max_p(a_p*t + b_p) (if such a point exists in [0,1])
  2. Or at the boundary of the interval where the max switches
  3. Or at t=0 or t=1

Actually, since yc is concave (opens downward), and the max of lines is convex, the max of a concave and convex function can have at most 2 local minima. The global minimum is at one of:
- The minimum of the max of lines (if it's below yc everywhere)
- The intersection points of yc with the max of lines
- t = 0 or t = 1

def score(h):
    """Compute C5 = dx * max(correlate(h, 1-h, 'full'))"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr), corr

def make_feasible(h):
    """Project onto {0<=x<=1, sum(x)=n/2}"""
    n = len(h)
    target = n / 2.0
    x = np.clip(h.astype(float), 0.0, 1.0)
    if abs(np.sum(x) - target) < 1e-10:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(x - lam, 0.0, 1.0)
    # Fix residual mass
    residual = target - np.sum(z)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(-(1.0 - z))
        else:
            order = np.argsort(z)
        rem = residual
        for i in order:
            if rem > 0:
                d = min(rem, 1.0 - z[i])
            else:
                d = max(rem, -z[i])
            z[i] += d
            rem -= d
            if abs(rem) < 1e-14:
                break
    return z

def coord_descent_step(h, j):
    """Optimize coordinate j of h. Returns new h and whether it improved."""
    n = len(h)
    center = n - 1
    L = 2 * n - 1
    
    h_j = h[j]
    
    # Compute S = sum_{k≠j} h[k](1-h[k])
    S = np.sum(h * (1.0 - h)) - h_j * (1.0 - h_j)
    
    # Compute a[p] = coefficient of h[j] in Y[p] for p ≠ center
    a = np.zeros(L)
    for p in range(L):
        d = p - center
        if d == 0:
            continue
        elif d > 0:
            coeff = 0.0
            if d <= j <= n - 1:
                coeff += (1.0 - h[j - d])
            if 0 <= j <= n - 1 - d:
                coeff += (-h[j + d])
            a[p] = coeff
        else:
            d = -d
            coeff = 0.0
            if 0 <= j <= n - 1 - d:
                coeff += (1.0 - h[j + d])
            if d <= j <= n - 1:
                coeff += (-h[j - d])
            a[p] = coeff
    
    # Current correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # b[p] = corr[p] - a[p] * h_j for p ≠ center
    b = np.zeros(L)
    mask = np.ones(L, dtype=bool)
    mask[center] = False
    b[mask] = corr[mask] - a[mask] * h_j
    b[center] = S  # not used directly, center handled separately
    
    # Grid search
    # Focus search around current value
    span = 0.4
    t_lo = max(0.0, h_j - span)
    t_hi = min(1.0, h_j + span)
    grid = np.linspace(t_lo, t_hi, 80)
    # Also include boundaries and current
    grid = np.unique(np.concatenate([grid, [0.0, 1.0, h_j]]))
    
    # Vectorized evaluation
    # yc[:, ] = t*(1-t) + S
    yc = grid * (1.0 - grid) + S  # shape (G,)
    # Y_p[:, ] = a[p] * t + b[p] for p ≠ center
    # Shape: (G, L)
    Y = a[None, :] * grid[:, None] + b[None, :]
    Y[:, center] = yc  # replace center column
    
    f = np.max(Y, axis=1)  # shape (G,)
    best_idx = np.argmin(f)
    best_t = grid[best_idx]
    best_max = f[best_idx]
    
    current_max = corr.max()
    
    if best_max < current_max - 1e-10:
        new_h = h.copy()
        new_h[j] = best_t
        return new_h, True
    return h, False

def pair_move(h, i, j):
    """Try best pair move between i and j. Returns new h and improvement."""
    n = len(h)
    if i == j:
        return h, False
    
    # Transfer from j to i: h[i]+t, h[j]-t
    t_max = min(h[j], 1.0 - h[i])
    if t_max < 1e-8:
        return h, False
    
    base_c5, _ = score(h)
    best_c5 = base_c5
    best_t = 0.0
    
    # Try several t values
    for frac in np.linspace(0.05, 1.0, 10):
        t = frac * t_max
        trial = h.copy()
        trial[i] += t
        trial[j] -= t
        c5, _ = score(trial)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    
    # Also try reverse (transfer from i to j)
    t_max2 = min(h[i], 1.0 - h[j])
    if t_max2 > 1e-8:
        for frac in np.linspace(0.05, 1.0, 10):
            t = frac * t_max2
            trial = h.copy()
            trial[j] += t
            trial[i] -= t
            c5, _ = score(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_t = -t  # negative means reverse
    
    if best_t != 0.0:
        new_h = h.copy()
        new_h[i] += best_t
        new_h[j] -= best_t
        return new_h, True
    return h, False
