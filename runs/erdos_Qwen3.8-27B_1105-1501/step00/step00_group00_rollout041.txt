def repair(h, target_sum):

Actually, one more thought: I should verify that my understanding of the problem is correct. The problem says:

"C₅ = max_k ∫ h(x)(1 - h(x+k)) dx"

In the discrete setting:
- h is defined on [0, 2] with n points
- dx = 2/n
- The integral ≈ sum * dx
- np.correlate(h, 1-h, mode='full') gives the discrete correlation
- C₅ = max(correlation) * dx

This is what I've been computing. Good.

Now, one subtlety: the continuous integral ∫₀² h(x)(1-h(x+k))dx for a shift k that doesn't align with the grid will be different from the discrete sum. But since we're working in the discrete setting as specified, we just use the discrete formula.

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_feasible(h, target_sum):
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-13:
        return h
    n = len(h)
    if diff > 0:
        room = 1.0 - h
        tr = room.sum()
        if tr < 1e-15:
            return h
        scale = min(1.0, diff / tr) if tr > 0 else 0
        h += room * scale
    else:
        vals = h.copy()
        tv = vals.sum()
        if tv < 1e-15:
            return h
        scale = min(1.0, (-diff) / tv) if tv > 0 else 0
        h -= vals * scale
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction on the largest element with room
    s2 = h.sum()
    diff2 = target_sum - s2
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_one(h, m, n):
    dx = 2.0 / n
    ell = m - (n - 1)
    g = np.zeros(n)
    if ell >= 0:
        # c = dx * sum_{p=0}^{n-1-ell} h[p+ell]*(1-h[p])
        # d/dh[j]: for j in [ell, n-1]: +dx*(1-h[j-ell])
        #          for j in [0, n-1-ell]: -dx*h[j+ell]
        for j in range(ell, n):
            g[j] += dx * (1.0 - h[j - ell])
        for j in range(n - ell):
            g[j] += dx * (-h[j + ell])
    else:
        q = -ell
        # c = dx * sum_{p=0}^{n-1-q} h[p]*(1-h[p+q])
        # d/dh[j]: for j in [0, n-1-q]: +dx*(1-h[j+q])
        #          for j in [q, n-1]: -dx*h[j-q]
        for j in range(n - q):
            g[j] += dx * (1.0 - h[j + q])
        for j in range(q, n):
            g[j] += dx * (-h[j - q])
    return g

def lp_step(h, k_active, trust, t_start, t_deadline):
    n = len(h)
    target_sum = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    cv = corr * (2.0 / n)
    idx = np.argsort(cv)[::-1][:k_active]
    grads = np.array([grad_one(h, int(m), n) for m in idx])
    vals = cv[idx]
    r = trust / n
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0
    ka = len(idx)
    A_ub = np.zeros((ka, nv))
    b_ub = np.zeros(ka)
    for a in range(ka):
        A_ub[a, :n] = grads[a]
        A_ub[a, n] = -1.0
        b_ub[a] = float(grads[a] @ h) - float(vals[a])
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(max(0.0, h[i] - r), min(1.0, h[i] + r)) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 200})
    except Exception:
        return None, None
    if not res.success:
        return None, None
    hn = np.clip(res.x[:n], 0.0, 1.0)
    s = hn.sum()
    if abs(s - target_sum) > 1e-9:
        hn += (target_sum - s) / n
        hn = np.clip(hn, 0.0, 1.0)
    return hn, evaluate_c5(hn)

        out[k] = area / dxn
