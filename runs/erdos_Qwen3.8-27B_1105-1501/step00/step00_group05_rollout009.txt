                for amt in [0.01, 0.005, 0.002, 0.001]:
                    for sign in [1, -1]:
                        h_new = h.copy()
                        h_new[i] += sign * amt
                        h_new[j] -= sign * amt
                        if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
                            c_new = exact_objective(h_new)
                            if c_new < c - 1e-13:
                                h, c = h_new, c_new
                                improved = True
                                break
                    if improved:
                        break

        if improved:
            stagnant = 0
            if c < best_in_phase - 1e-8:
                best_in_phase = c
                tol = max(0.001, tol * 0.7)
        else:
            stagnant += 1
            if stagnant > 200:
                tol = max(0.001, tol * 0.5)
                if tol <= 0.001:
                    # Add small random perturbation to escape
                    pert = rng.randn(n) * 0.001
                    pert -= np.mean(pert)
                    h_new = np.clip(h + pert, 0, 1)
                    c_new = exact_objective(h_new)
                    if c_new < c + 0.001:
                        h = h_new
                        c = c_new
                    stagnant = 0

    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = make_synthetic(42, seed)

    best_h = h0.copy()
    best_c = exact_objective(h0)
    best_n = len(h0)

    n0 = len(h0)

    # Build dimension list: prefer smaller, include parent
    dims = []
    for d in range(max(20, n0 - 10), min(72, n0 + 12), 2):
        if d not in dims:
            dims.append(d)
    dims.sort()

    # Order: put parent dim first, then alternate smaller/larger
    parent_idx = dims.index(n0) if n0 in dims else 0
    ordered_dims = [dims[parent_idx]]
    smaller = sorted([d for d in dims if d < n0], key=lambda x: -abs(x - n0))
    larger = sorted([d for d in dims if d > n0], key=lambda x: abs(x - n0))
    ordered_dims.extend(smaller[:4])
    ordered_dims.extend(larger[:4])
    ordered_dims = list(dict.fromkeys(ordered_dims))

    time_per_dim = max(30, (budget_s - 20) / len(ordered_dims))

    for dim_i, n_try in enumerate(ordered_dims):
        if time.time() > deadline - 15:
            break
        remaining = deadline - time.time()
        if remaining < 15:
            break

        if n_try == best_n:
            h_start = best_h.copy()
        else:
            h_start = resample_density(best_h, n_try)

        c_start = exact_objective(h_start)

        # Quick greedy improvement before main search
        h_quick = h_start.copy()
        c_quick = c_start
        for _ in range(200):
            if time.time() > deadline - 15:
                break
            grads, _ = get_active_gradients(h_quick, tol=0.005, max_active=10)
            if len(grads) > 0:
                d = -np.mean(grads, axis=0)
                d -= np.mean(d)
                h_new, c_new = line_search(h_quick, d, c_quick, max_alpha=1.0)
                if h_new is not None:
                    h_quick, c_quick = h_new, c_new

        if c_quick < best_c:
            best_h = h_quick.copy()
            best_c = c_quick
            best_n = len(h_quick)

        # Main local search
        dim_budget = min(time_per_dim, remaining - 10)
        if dim_budget < 10:
            break
        h_opt, c_opt = local_search(h_quick, deadline, t0, seed + dim_i * 1000,
                                     max_iter=15000)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Final dyadic repair on best solution
    if best_c <= 0.3809:
        h_repaired = dyadic_repair(best_h, bits=20)
        c_repaired = exact_objective(h_repaired)
        if c_repaired <= best_c + 0.0001 and abs(np.sum(h_repaired) - best_n / 2.0) < 0.5:
            if c_repaired < best_c:
                best_h = h_repaired
                best_c = c_repaired
        else:
            # Try with fewer bits
            h_repaired = dyadic_repair(best_h, bits=16)
            c_repaired = exact_objective(h_repaired)
            if c_repaired <= best_c + 0.00005:
                best_h = h_repaired
                best_c = c_repaired

    # Final validation
    final_c = exact_objective(best_h)
    if final_c <= 0.5:  # sanity check
        best_c = final_c

    return best_h, best_c, best_n
