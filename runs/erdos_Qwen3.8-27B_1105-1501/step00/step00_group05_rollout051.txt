```python
import numpy as np
import time

def exact_objective(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_values(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def subgradient(h, active_tol=0.005):
    N = len(h)
    dx = 2.0 / N
    corr = compute_corr_values(h)
    max_val = np.max(corr)
    
    if max_val <= 0:
        return np.zeros(N), max_val
    
    active = np.where(corr >= max_val - active_tol)[0]
    
    grad = np.zeros(N)
    for k in active:
        s = k - (N - 1)
        if s >= 0:
            # c[k] = sum_{i=s}^{N-1} h[i] * (1-h[i-s])
            a = h[s:]
            b = h[:N-s]
            # grad[i] += (1-b[i-s]) for i in s..N-1
            # grad[j] -= a[i-s] for j in 0..N-1-s
            grad[s:] += (1.0 - b)
            grad[:N-s] -= a
        else:
            t = -s
            # c[k] = sum_{i=0}^{N-1-t} h[i] * (1-h[i+t])
            a = h[:N-t]
            b = h[t:]
            grad[:N-t] += (1.0 - b)
            grad[t:] -= a
    
    grad *= dx / len(active)
    return grad, max_val

def project_feasible(h, N):
    target = N / 2.0
    # Bisection for lambda
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(proj)
    if abs(diff) > 1e-15:
        # Distribute to elements not at bounds
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            adj = diff / np.sum(mask)
            proj[mask] += adj
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def optimize_subgradient(h0, max_time, seed=42):
    N = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_objective(h)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    step = 0.05
    momentum = np.zeros(N)
    beta = 0.8
    no_improve = 0
    
    while time.time() - t0 < max_time:
        grad, cur_c = subgradient(h, active_tol=0.003)
        
        # Projected gradient direction
        direction = -grad
        direction -= np.mean(direction)  # Ensure zero sum
        
        # Find max step to stay in [0,1]
        pos_mask = direction > 0
        neg_mask = direction < 0
        t_max = 1.0  # normalize
        if np.any(pos_mask):
            t_max = min(t_max, np.min((1.0 - h[pos_mask]) / direction[pos_mask]))
        if np.any(neg_mask):
            t_max = min(t_max, np.min(h[neg_mask] / (-direction[neg_mask])))
        
        # Take a fraction of t_max
        t = min(step, 0.5 * t_max) if t_max > 0 else 0.0
        if t <= 0:

            no_improve += 1

                # Reduce step size or restart
                step *= 0.9
                no_improve = 0
```

```python
from scipy.optimize import linprog

def lp_step(h, N, active_lags):
    dx = 2.0 / N
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_val = np.max(corr)
    
    # Build LP
    # Variables: h[0], ..., h[N-1], t
    # Minimize t
    # Subject to:
    #   sum(h) = N/2
    #   0 <= h <= 1
    #   For each active k: grad_k · (h - h_current) + max_val <= t
    #   i.e., grad_k · h - t <= max_val - grad_k · h_current
    
    # c = [0, 0, ..., 0, 1] (minimize t)
    c_lp = np.concatenate([np.zeros(N), [1.0]])
    
    # Equality constraint: sum(h) = N/2
    A_eq = np.zeros((1, N+1))
    A_eq[0, :N] = 1.0
    b_eq = [N / 2.0]
    
    # Inequality constraints
    # For each active k: grad_k · h - t <= max_val - grad_k · h_current
    A_ub = []
    b_ub = []
    
    for k in active_lags:
        s = k - (N - 1)
        grad_k = np.zeros(N)
        if s >= 0:
            n = N - s
            grad_k[s:N] = 1.0 - h[0:n]
            grad_k[0:n] -= h[s:N]
        else:
            t_shift = -s
            n = N - t_shift
            grad_k[0:n] = 1.0 - h[t_shift:N]
            grad_k[t_shift:N] -= h[0:n]
        grad_k *= dx
        
        row = np.zeros(N+1)
        row[:N] = grad_k
        row[N] = -1.0
        A_ub.append(row)
        b_ub.append(max_val - grad_k @ h)
    
    if not A_ub:
        return h
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, 1)] * N + [(0, None)]
    
    result = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:N]
    return h
```
