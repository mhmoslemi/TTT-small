def feasible_project(z, n=None):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    if n is None:
        n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    # Bisection for lambda
    lam_lo = -1.0 - np.max(z)
    lam_hi = 1.0 - np.min(z)
    for _ in range(60):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        h_tmp = np.clip(z + lam_mid, 0.0, 1.0)
        s = np.sum(h_tmp)
        if s < target:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    lam = 0.5 * (lam_lo + lam_hi)
    h = np.clip(z + lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add more; find indices with most room
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                delta = min(residual, room[i])
                h[i] += delta
                residual -= delta
                if abs(residual) < 1e-12:
                    break
        else:
            # Need to subtract
            avail = h.copy()
            idx = np.argsort(-avail)
            for i in idx:
                delta = min(-residual, avail[i])
                h[i] -= delta
                residual += delta
                if abs(residual) < 1e-12:
                    break

def manual_values(h):
    n = len(h)
    vals = np.zeros(2 * n - 1)
    omh = 1.0 - h
    for j in range(2 * n - 1):
        k = j - n + 1
        lo = max(0, -k)
        hi = min(n, n - k)
        vals[j] = np.dot(h[lo + k:hi + k], omh[lo:hi])
    return vals

def manual_gradient(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for j in range(2 * n - 1):
        k = j - n + 1
        lo = max(0, -k)
        hi = min(n, n - k)
        first = np.arange(lo + k, hi + k)
        second = np.arange(lo, hi)
        np.add.at(grad[j], first, omh[second])
        np.add.at(grad[j], second, -h[first])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def manual_values(h):
    n = len(h)
    vals = np.zeros(2 * n - 1)
    omh = 1.0 - h
    for j in range(2 * n - 1):
        k = j - n + 1
        lo = max(0, -k)
        hi = min(n, n - k)
        vals[j] = np.dot(h[lo + k:hi + k], omh[lo:hi])
    return vals

def manual_gradient(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for j in range(2 * n - 1):
        k = j - n + 1
        lo = max(0, -k)
        hi = min(n, n - k)
        first = np.arange(lo + k, hi + k)
        second = np.arange(lo, hi)
        np.add.at(grad[j], first, omh[second])
        np.add.at(grad[j], second, -h[first])
    return grad

def feasible_project(z, n):
    target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    if z.shape[0] != n:
        z = z[:n].copy() if z.shape[0] > n else np.pad(z, (0, n - z.shape[0]))
    z = np.clip(z, -2.0, 3.0)
    lam_lo = -1.0 - float(np.max(z))
    lam_hi = 1.0 - float(np.min(z))
    for _ in range(60):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        s = float(np.sum(np.clip(z + lam_mid, 0.0, 1.0)))
        if s < target:
            lam_lo = lam_mid
        else:
            lam_hi = lam_mid
    lam = 0.5 * (lam_lo + lam_hi)
    h = np.clip(z + lam, 0.0, 1.0)
    resid = target - float(np.sum(h))
    if abs(resid) > 1e-10:
        if resid > 0:
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                d = min(resid, room[i])
                if d > 0:
                    h[i] += d
                    resid -= d
                if abs(resid) < 1e-14:
                    break
        else:
            idx = np.argsort(-h)
            for i in idx:
                d = min(-resid, h[i])
                if d > 0:
                    h[i] -= d
                    resid += d
                if abs(resid) < 1e-14:
                    break
    return h

def interpolate_to(h_src, n_src, n_tgt):
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return np.interp(x_tgt, x_src, h_src)

def smooth_obj(h, beta, dx):
    vals = manual_values(h) * dx
    cmax = float(np.max(vals))
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -400.0, 400.0)
    ev = np.exp(shifted)
    return cmax + np.log(np.sum(ev)) / beta

def smooth_jac(h, beta, dx):
    vals = manual_values(h) * dx
    cmax = float(np.max(vals))
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -400.0, 400.0)
    ev = np.exp(shifted)
    w = ev / float(np.sum(ev))
    gm = manual_gradient(h)
    return dx * (w @ gm)

def sum_eq_fn(h, n):
    return float(np.sum(h)) - n / 2.0

def sum_eq_jac_fn(h, n):
    return np.ones(n)

def run_smooth_solve(h0, beta, n, maxiter=200):
    dx = 2.0 / n
    h0 = feasible_project(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq_fn, 'jac': sum_eq_jac_fn, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h0, args=(beta, dx), jac=smooth_jac,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return feasible_project(res.x, n)
    except Exception:
        pass
    return None

def antisym_transform(h):
    n = len(h)
    return 1.0 - h[::-1].copy()

def circ_shift(h, s):
    return np.roll(h, s)
