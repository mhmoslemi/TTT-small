def _project(h, target):

        if not mask.any():

    dst = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(dst, src, h)
    return _project(h_new, new_n / 2.0)

    shift = np.zeros((n, 2 * n - 1))

        k_max = min(n - 1, n - 1 - p)

            vals = np.arange(p + k_min, p + k_max + 1)

def _best_pair_move(h):
    """Find the best single pair move for h. Returns (h_new, score) or (None, score)."""
    n = len(h)
    target = n / 2.0
    F = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    current_score = float(np.max(F) * dx)
    
    shift = _compute_shift(h)
    # a for pair (p,q) at lag j: shift[q,j] + shift[q,2(n-1)-j] - shift[p,j] - shift[p,2(n-1)-j]
    
    best_score = current_score
    best_h = None
    
    for p in range(n):
        for q in range(n):
            if p == q:
                continue
            d_lo = max(-h[p], h[q] - 1.0)
            d_hi = min(1.0 - h[p], h[q])
            if d_hi - d_lo < 1e-10:
                continue
            
            sq = shift[q]
            sp = shift[p]
            a = sq + sq[::-1] - sp - sp[::-1]
            
            # Sample and evaluate
            deltas = np.linspace(d_lo, d_hi, 25)
            # C5(δ) = dx * max_k (F[k] + δ * a[k])
            vals = dx * np.maximum.reduce(F[None, :] + deltas[:, None] * a[None, :])
            
            best_idx = np.argmin(vals)
            if vals[best_idx] >= best_score - 1e-13:
                continue
            
            # Golden section refinement
            delta_lo = d_lo
            delta_hi = d_hi
            # We know the best is near deltas[best_idx]
            # Let's do a few bisection steps around it
            center = deltas[best_idx]
            span = (d_hi - d_lo) / 25.0
            delta_lo = max(d_lo, center - 2 * span)
            delta_hi = min(d_hi, center + 2 * span)
            
            gr = (np.sqrt(5) - 1) / 2
            c1 = delta_hi - gr * (delta_hi - delta_lo)
            c2 = delta_lo + gr * (delta_hi - delta_lo)
            v1 = dx * np.max(F + c1 * a)
            v2 = dx * np.max(F + c2 * a)
            for _ in range(20):
                if v1 < v2:
                    delta_hi = c2
                    c2 = c1
                    v2 = v1
                    c1 = delta_hi - gr * (delta_hi - delta_lo)
                    v1 = dx * np.max(F + c1 * a)
                else:
                    delta_lo = c1
                    c1 = c2
                    v1 = v2
                    c2 = delta_lo + gr * (delta_hi - delta_lo)
                    v2 = dx * np.max(F + c2 * a)
            
            delta = (delta_lo + delta_hi) / 2
            v = dx * np.max(F + delta * a)
            if v < best_score - 1e-13:
                h_new = h.copy()
                h_new[p] += delta
                h_new[q] -= delta
                h_new = np.clip(h_new, 0.0, 1.0)
                best_score = v
                best_h = h_new

        hi = min(hi if hi > 0 else 1e10, ((1 - h[pos]) / u[pos]).min())

        hi = min(hi if hi > 0 else 1e10, (-h[neg] / u[neg]).min())

            best_alpha = a

Wait, I realize the random direction approach with projection is different from the pair sweep. In the pair sweep, the sum is automatically preserved (since we move δ from one to another). In the random direction, we need to project after the move, which changes the points non-uniformly. This makes the 1D optimization less clean.

For the random direction, a better approach might be:
1. Generate u with sum = 0
2. Find the maximum step α_max such that h + α*u ∈ [0,1]^n (no projection needed)
3. Sample α in [−α_max, α_max] (or just [0, α_max] for positive direction)
4. Since sum is preserved by u having zero sum, no projection is needed!

Let me fix:

```python
def _random_direction_move(h, rng, n_samples=30):
    n = len(h)
    current_score = _score(h)
    
    u = rng.randn(n)
    u -= u.mean()
    u_norm = np.linalg.norm(u)
    if u_norm < 1e-12:
        return None, current_score
    u /= u_norm  # normalize to unit step
    
    # Find max step
    pos = u > 1e-12
    neg = u < -1e-12
    max_step = 1.0
    if pos.any():
        max_step = min(max_step, ((1 - h[pos]) / u[pos]).min())
    if neg.any():
        max_step = min(max_step, (h[neg] / (-u[neg])).min())
    
    if max_step < 1e-10:
        return None, current_score
    
    alphas = np.linspace(0, max_step, n_samples)
    best_alpha = 0.0
    best_val = current_score
    for a in alphas:
        h_new = h + a * u
        s = _score(h_new)
        if s < best_val:
            best_val = s
            best_alpha = a
    
    if best_val < current_score - 1e-13:
        h_new = h + best_alpha * u
        return h_new, best_val
    return None, current_score
```

Since u has sum 0 and we're not projecting, the sum of h_new is preserved. And if 0 ≤ a ≤ max_step, then 0 ≤ h_new ≤ 1. So no projection needed!

This is much cleaner. The objective is still convex in α (max of linear functions, same argument as pair sweep), so golden section works.
