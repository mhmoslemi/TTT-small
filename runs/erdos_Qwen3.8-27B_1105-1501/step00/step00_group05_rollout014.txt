```python
import numpy as np
import time

def exact_objective(h):
    """Compute exact C5 value."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}."""
    # Bisection on lambda
    # h_proj = clip(h - lambda, 0, 1)
    # Find lambda such that sum(h_proj) = target_sum
    
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any rounding to get exact sum
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-15:
        # Distribute the difference
        n_fix = int(abs(diff))
        if n_fix > 0:
            if diff > 0:
                idx = np.argsort(-h_proj)[:n_fix]
                for i in idx:
                    h_proj[i] = min(1.0, h_proj[i] + 1.0)
            else:
                idx = np.argsort(h_proj)[:n_fix]
                for i in idx:
                    h_proj[i] = max(0.0, h_proj[i] - 1.0)
    return h_proj
```

```python
    z_float = h * S
    z = np.floor(z_float).astype(np.int64)
    z = np.clip(z, 0, S)
    remainders = z_float - z  # fractional parts
    target = int(round(target_sum * S))
    cur = int(np.sum(z))
    diff = target - cur
    if diff > 0:
        # Add diff units to entries with largest remainders (and room)
        order = np.argsort(-remainders)
        added = 0
        for i in order:
            if added >= diff:
                break
            if z[i] < S:
                z[i] += 1
                added += 1
    elif diff < 0:
        # Remove |diff| units from entries with smallest remainders (and z > 0)
        order = np.argsort(remainders)
        removed = 0
        for i in order:
            if removed >= -diff:
                break
            if z[i] > 0:
                z[i] -= 1
                removed += 1
    return z.astype(np.float64) / S
```
