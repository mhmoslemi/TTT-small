import time

def project_feasible(h, target_sum, tol=1e-12):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}."""
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) < tol:
        return h

        else:

def project_feasible(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}."""
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    projected = np.clip(h - lam, 0.0, 1.0)
    gap = target_sum - projected.sum()
    if abs(gap) > 1e-14:
        interior = (projected > 1e-10) & (projected < 1.0 - 1e-10)
        n_int = interior.sum()
        if n_int > 0:
            projected[interior] += gap / n_int
        else:
            # Distribute to points not at bounds
            at_zero = projected == 0.0
            at_one = projected == 1.0
            free = ~(at_zero | at_one)
            if free.any():
                projected[free] += gap / free.sum()
            else:
                projected += gap / n
        projected = np.clip(projected, 0.0, 1.0)
    return projected

def compute_c_vec(h):
    """Compute scaled correlation vector."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h):
    """Get C5 value."""
    return float(np.max(compute_c_vec(h)))

def corr_grad_single(h, k):
    """Gradient of c[k] w.r.t. h, where c[k] is scaled."""
    n = len(h)
    dx = 2.0 / n
    l = k - (n - 1)
    grad = np.zeros(n)
    if l >= 0:
        # c[k] = dx * sum_{i=0}^{n-1-l} h[i] * (1-h[i+l])
        # dc/dh[j] = dx * [(1-h[j+l]) if j < n-l else 0] - dx * [h[j-l] if j >= l else 0]
        for j in range(l, n):
            # j >= l, so second term is active
            grad[j] -= dx * h[j - l]
        for j in range(0, n - l):
            # j < n-l, so first term is active
            grad[j] += dx * (1.0 - h[j + l])
    else:
        p = -l
        # c[k] = dx * sum_{i=0}^{n-1-p} h[i+p] * (1-h[i])
        # dc/dh[j] = dx * [(1-h[j-p]) if j >= p else 0] - dx * [h[j+p] if j < n-p else 0]
        for j in range(p, n):
            grad[j] += dx * (1.0 - h[j - p])
        for j in range(0, n - p):
            grad[j] -= dx * h[j + p]
    return grad

def smooth_obj_grad(h, tau):
    """Smooth max objective and gradient."""
    c = compute_c_vec(h)
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_vals = np.exp(tau * c_shifted)
    obj = c_max + np.log(np.sum(exp_vals)) / tau
    weights = exp_vals / np.sum(exp_vals)
    
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w > 1e-14:
            grad += w * corr_grad_single(h, k)
    return obj, grad
