    A = compute_overlaps(h)
    G = compute_overlap_grads(h)
    
    # Active set: those within eps of max
    max_A = np.max(A)
    active = A > max_A - eps
    
    # Average gradient over active set
    if np.sum(active) == 0:
        active = A > max_A - eps * 10
    
    grad = np.mean(G[active], axis=0)
    
    # We want to decrease the max, so move in -grad direction
    # But we need zero-sum: subtract mean
    d = -grad
    d -= np.mean(d)
    
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    d /= norm
    
    return d

def lp_direction(h, radius=0.05):
    """Solve LP trust region."""
    n = len(h)
    A = compute_overlaps(h)
    G = compute_overlap_grads(h)
    n_lags = 2 * n - 1
    
    # Variables: x = [h_new (n), t (1)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: A_s + G_s . (h_new - h) <= t
    # => G_s . h_new - t <= G_s . h - A_s
    a_ub = np.zeros((n_lags, n + 1))
    a_ub[:, :n] = G
    a_ub[:, n] = -1.0
    b_ub = np.sum(G * h, axis=1) - A
    
    a_eq = np.zeros((1, n + 1))
    a_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lb = max(0.0, h[i] - radius)
        ub = min(1.0, h[i] + radius)
        bounds.append((lb, ub))
    bounds.append((0.0, None))
    
    try:
        result = linprog(c, A_ub=a_ub, b_ub=b_ub, A_eq=a_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        if result.status != 0:
            return None
        d = result.x[:n] - h
        d -= np.mean(d)
        return d
    except:
        return None

def resample_h(h, new_n):
    old_n = len(h)
    x_old = (np.arange(old_n) + 0.5) * 2.0 / old_n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    return project_box_mass(h_new, new_n / 2.0)

            lo = mid
        else:

    return global_best_h, final_score, n_final
