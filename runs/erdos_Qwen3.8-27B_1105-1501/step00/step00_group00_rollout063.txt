    dx = 2.0 / n

            for j in range(m, n):

                grad[j] += wk * (1.0 - h[j + p])
            for j in range(p, n):
                grad[j] += wk * (-h[j - p])

    return obj, grad

        if c5_new < best_obj:

            best_h = h_new.copy()

        if it % 100 == 99:

            T *= 0.98

                grad[j] += wk * (1.0 - h[j + p])
            for j in range(p, n):
                grad[j] += wk * (-h[j - p])
    
    obj = r_max + T * np.log(weights.sum())

def project_feasible(h, target_sum):
    h = h.copy()

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    # Final adjustment
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj > 1e-14
        else:
            mask = proj < 1.0 - 1e-14
        if mask.sum() > 0:
            proj[mask] += -diff / mask.sum()
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_obj = compute_c5(h)
    
    T = 0.05
    lr = 0.05

            h_new = project_feasible(h_new, target_sum)

def make_initial(n, pattern='alternating'):

    h = np.zeros(n)

        if n % 2 == 0:
            h[::2] = 1.0

            h[::2] = 1.0

            h = project_feasible(h, n/2.0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * dx)

def smooth_max_correlation(h, T):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full") * dx
    r_max = float(np.max(r))
    return r_max + T * float(np.log(np.sum(np.exp((r - r_max) / T))))

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full") * dx
    r_max = float(np.max(r))
    weights = np.exp((r - r_max) / T)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        wk = weights[k] * dx
        if m >= 0:
            grad[m:] += wk * (1.0 - h[:n-m])
            grad[:n-m] += wk * (-h[m:])
        else:
            p = -m
            grad[:n-p] += wk * (1.0 - h[p:])
            grad[p:] += wk * (-h[:n-p])
    
    obj = r_max + T * float(np.log(np.sum(weights)))
    return obj, grad

def project_feasible(h, target_sum):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = proj > 1e-14
        else:
            mask = proj < 1.0 - 1e-14
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += -diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def gradient_descent(h, n, max_iter, time_limit, start_time):
    target_sum = n / 2.0
    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_obj = compute_c5(h)
    
    T = 0.04
    lr = 0.08
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        obj, grad = smooth_max_grad(h, T)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_obj:
            best_obj = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
            if no_improve % 150 == 0:
                # Reset and reduce
                h = best_h.copy()
                v = np.zeros(n)
                lr *= 0.7
                T = max(0.003, T * 0.9)
                if lr < 0.003:
                    break
            elif no_improve > 300:
                break
    
    return best_h, best_obj

def local_search_binary(h, n, max_rounds, time_limit, start_time):
    target_sum = n / 2.0
    h = np.round(h)
    h = project_feasible(h, target_sum)
    best_obj = compute_c5(h)
    
    for round_num in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap_obj = best_obj
        best_j_k = None
        
        for i in range(len(zeros)):
            j = zeros[i]
            for k_idx in ones:
                h[j] = 1.0
                h[k_idx] = 0.0
                c5_new = compute_c5(h)
                if c5_new < best_swap_obj - 1e-14:
                    best_swap_obj = c5_new
                    best_j_k = (j, k_idx)
        
        if best_j_k is not None:
            h[best_j_k[0]] = 1.0
            h[best_j_k[1]] = 0.0
            best_obj = best_swap_obj
        else:
            break
    
    return h, best_obj
