        init_h = None

        init_h = np.clip(init_h, 0, 1)
        target = len(init_h) / 2.0
        if abs(np.sum(init_h) - target) > 1e-8:
            init_h += (target - np.sum(init_h)) / len(init_h)
            init_h = np.clip(init_h, 0, 1)
        best_h = init_h.copy()
        best_score = evaluate_h(init_h)
        best_n = len(init_h)
    
    # Phase 1: Symmetric optimization for various n
    n_list = [45, 47, 49, 51, 43, 55, 60, 65, 70]
    alphas_main = [5, 10, 20, 50, 100, 200, 500, 1000]
    
    for n in n_list:
        if time.time() - t_start > budget_s - 120:
            break
        m = n // 2
        starts = []
        
        # Start 1: Interpolated initial
        if init_h is not None:
            h_interp = interpolate_h(init_h, n)
            starts.append(u_from_h(h_interp))
        
        # Start 2: Random
        starts.append(rng.uniform(0, 1, m))
        
        # Start 3: Step function
        x = np.linspace(0, 1, m)
        u_step = np.zeros(m)
        u_step[:m//2] = 1.0
        starts.append(u_step)
        
        # Start 4: Cosine
        starts.append(0.5 + 0.5*np.cos(2*np.pi*x))
        
        # Start 5: Ramp
        starts.append(np.linspace(1.0, 0.0, m))
        
        # Start 6: S-shaped
        starts.append(1.0/(1.0+np.exp(5*(x-0.5))))
        
        for u0 in starts:
            if time.time() - t_start > budget_s - 60:
                break
            u0 = np.clip(u0, 0, 1)
            
            # Smooth optimization
            u_opt = optimize_smooth(u0, n, alphas_main, maxiter_each=60,
                                    time_limit=budget_s - t_start - 45, t_start=t_start)
            
            # Local perturbation
            u_opt, sc_pert = local_perturb(u_opt, n, rng, n_pert=15,
                                           step_sizes=[0.05, 0.02, 0.01])
            
            # SLSQP polish
            if time.time() - t_start < budget_s - 30:
                u_slsqp = slsqp_minimax(u_opt, n, maxiter=80)
                sc_slsqp = score_from_u(u_slsqp, n)
                if sc_slsqp < sc_pert:
                    u_opt = u_slsqp
                    sc_pert = sc_slsqp
            
            sc = score_from_u(u_opt, n)
            if sc < best_score:
                best_score = sc
                best_h = h_from_u(u_opt, n)
                best_n = n
    
    # Phase 2: General (non-symmetric) SLSQP on best solution
    if best_h is not None:
        if time.time() - t_start < budget_s - 30:
            h_gen = general_slsqp(best_h, best_n, maxiter=200)
            sc_gen = evaluate_h(h_gen)
            if sc_gen < best_score:
                best_score = sc_gen
                best_h = h_gen
                best_n = best_n
        
        # Also try general SLSQP on interpolated versions
        for n_try in [45, 50, 55]:
            if time.time() - t_start > budget_s - 20:
                break
            if n_try != best_n:
                h_interp = interpolate_h(best_h, n_try)
                h_gen2 = general_slsqp(h_interp, n_try, maxiter=150)
                sc_gen2 = evaluate_h(h_gen2)
                if sc_gen2 < best_score:
                    best_score = sc_gen2
                    best_h = h_gen2
                    best_n = n_try
    
    # Phase 3: Final local polish on best solution
    if best_h is not None and time.time() - t_start < budget_s - 5:
        u_best = u_from_h(best_h)
        n_best = best_n
        # More aggressive local search
        u_polished, sc_pol = local_perturb(u_best, n_best, rng, n_pert=50,
                                            step_sizes=[0.03, 0.01, 0.005, 0.002])
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_from_u(u_polished, n_best)
        
        # One more smooth optimization round with high alpha
        if time.time() - t_start < budget_s - 10:
            u_final = optimize_smooth(np.clip(u_polished, 0, 1), n_best,
                                      [200, 500, 1000, 2000], maxiter_each=40,
                                      time_limit=budget_s - t_start - 5, t_start=t_start)
            sc_final = score_from_u(u_final, n_best)
            if sc_final < best_score:
                best_score = sc_final
                best_h = h_from_u(u_final, n_best)
    
    # Ensure feasibility
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        target = best_n / 2.0
        diff = target - np.sum(best_h)
        if abs(diff) > 1e-10:
            best_h += diff / best_n
            best_h = np.clip(best_h, 0, 1)
        best_score = evaluate_h(best_h)
    
    if best_h is None:
        # Fallback: simple construction
        n = 45
        h = np.ones(n) * 0.5
        best_h = h
        best_score = evaluate_h(h)
        best_n = n
    
    return best_h, best_score, best_n
