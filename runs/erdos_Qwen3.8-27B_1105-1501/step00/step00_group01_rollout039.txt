Actually, let me reconsider the approach. The key insight is:

For the log-sum-exp approach:
f(h) = (1/α) * log(Σ_k exp(α * c_k * dx))

This is a smooth approximation to max_k c_k * dx. As α → ∞, it approaches the max.

The gradient of f w.r.t. h[j] is:
∂f/∂h[j] = Σ_k (exp(α * c_k * dx) / Σ_m exp(α * c_m * dx)) * ∂c_k/∂h[j] * dx

This is a weighted sum of the individual gradients, where the weights are softmax of the c_k values. Only the large c_k values contribute significantly.

For the equality constraint sum(h) = n/2, I can use a penalty method or handle it by fixing one variable.

Let me use a penalty approach with L-BFGS-B:
f_total(h) = (1/α) * log(Σ_k exp(α * c_k * dx)) + λ * (sum(h) - n/2)^2

This is much faster because L-BFGS-B is efficient for bound-constrained problems.

```python
def slse_objective_and_jac(h, n, alpha, lam):
    """Compute log-sum-exp objective and its gradient, plus penalty."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    log_sum_exp = c_max + np.log(np.sum(np.exp(alpha * (c - c_max))))
    obj = log_sum_exp / alpha
    
    # Softmax weights
    weights = np.exp(alpha * (c - c_max)) / np.sum(np.exp(alpha * (c - c_max)))
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(c)):
        d = k - (n - 1)  # signed lag
        if weights[k] < 1e-15:
            continue
        w = weights[k]
        if d >= 0:
            # ∂c_k/∂h[j] = (1-h[j-d]) * I[j>=d] - h[j+d] * I[j+d<n]
            for j in range(d, n):
                grad[j] += w * (1 - h[j - d])
            for j in range(0, n - d):
                grad[j] -= w * h[j + d]
        else:
            dp = -d
            for j in range(0, n - dp):
                grad[j] += w * (1 - h[j + dp])
            for j in range(dp, n):
                grad[j] -= w * h[j - dp]
    grad *= dx / alpha
    
    # Penalty gradient
    penalty = lam * (np.sum(h) - n/2.0) ** 2
    obj += penalty
    grad += 2 * lam * (np.sum(h) - n/2.0) * np.ones(n)
    
    return obj, grad
```

```python
def project_sum(h, target):
    """Adjust h to have sum = target while keeping in [0,1]."""
    h = np.clip(h, 0, 1).copy()
    for _ in range(200):

    """Log-sum-exp objective and gradient with penalty for sum constraint."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    n_lags = len(c)

            continue

                grad += w * (1 - 2 * h)
            else:

                grad[:n-d] -= w * h[d:]

            grad[:n-dp] += w * (1 - h[dp:])
            grad[dp:] -= w * h[:n-dp]

    sum_h = np.sum(h)
```

```python
            h[mask] += (-diff) / cnt
        h = np.clip(h, 0, 1)
    # Exact fix
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        # Find elements not at boundary
        interior = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] -= diff / np.sum(interior)
            h = np.clip(h, 0, 1)
    return h

def lse_obj_and_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = (c_max + np.log(sum_exp)) / alpha + lam * (np.sum(x) - n/2.0)**2
    
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    n_lags = len(c)
    for k in range(n_lags):
        w = weights[k]
        if w < 1e-14:
            continue
        d = k - (n - 1)
        coeff = w * dx / alpha
        if d >= 0:
            if d == 0:
                grad += coeff * (1 - 2*x)
            else:
                grad[d:] += coeff * (1 - x[:n-d])
                grad[:n-d] -= coeff * x[d:]
        else:
            dp = -d
            grad[:n-dp] += coeff * (1 - x[dp:])
            grad[dp:] -= coeff * x[:n-dp]
    
    grad += 2 * lam * (np.sum(x) - n/2.0)
    return obj, grad

    h = project_sum(h_init.copy(), n/2.0)
    start = time.time()

    for alpha in alphas:
        if time.time() - start > max_time - 5:
            break
        lam = 100.0
        for _ in range(3):
            result = minimize(
                lambda x: lse_obj_and_grad(x, n, alpha, lam)[0],

                method='L-BFGS-B',

                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-10}
            )
            h_new = project_sum(result.x, n/2.0)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                h = h_new
            break  # Don't loop the lam increase for now
    
    return h, compute_c5(h)
```

```python
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)

                break
```

```python
_lse_n = 0
_lse_alpha = 1.0
_lse_lam = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        else:
            mask = h < 1
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += (-diff) / cnt
        h = np.clip(h, 0, 1)
    return h

def lse_full(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c_max = np.max(c)
    sh = alpha * (c - c_max)
    es = np.exp(sh)
    se = np.sum(es)
    obj = (c_max + np.log(se)) / alpha + lam * (np.sum(x) - n * 0.5) ** 2
    w = es / se
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        cf = wk * dx / alpha
        if d >= 0:
            if d == 0:
                grad += cf * (1 - 2 * x)
            else:
                grad[d:] += cf * (1 - x[:n - d])
                grad[:n - d] -= cf * x[d:]
        else:
            dp = -d
            grad[:n - dp] += cf * (1 - x[dp:])
            grad[dp:] -= cf * x[:n - dp]
    grad += 2 * lam * (np.sum(x) - n * 0.5)
    return obj, grad

    h = np.zeros(n)

        cnt = bin(i).count('1')

        idx = np.where(h_bin == 0)[0]

        h_bin[idx[order]] = 1
    return h_bin
```
