    return float(np.max(corr)) * (2.0 / n)

    """Compute subgradient of C5 using top_k active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)

    threshold = max_val - max(1e-6, max_val * 0.05)
    active = np.where(corr >= threshold)[0]
    
    if len(active) > top_k:

        active_vals = corr[active]
        top_indices = active[np.argsort(-active_vals)[:top_k]]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def get_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    center = n - 1
    threshold = max_c * 0.88
    grad = np.zeros(n)
    if corr[center] >= threshold:
        grad += 1.0 - 2.0 * h
    for k in range(1, n):
        m = n - k
        if corr[center + k] >= threshold:
            grad[:m] -= h[k:]
            grad[k:] += 1.0 - h[:m]
        if corr[center - k] >= threshold:
            grad[:m] += 1.0 - h[k:]
            grad[k:] -= h[:m]
    grad *= dx
    return grad, max_c * dx

def fix_sum(h, target):
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            total_slack = np.sum(slack)
            if total_slack > diff:
                h = h + diff * slack / total_slack
        else:
            total_mass = np.sum(h)
            if total_mass > -diff:
                h = h + diff * h / total_mass
    return np.clip(h, 0.0, 1.0)
