def find_optimal_alpha(r, s, alpha_lo, alpha_hi):
    """Find alpha in [alpha_lo, alpha_hi] that minimizes max_k (r[k] + alpha * s[k])."""
    # Breakpoints where two lines intersect
    # r[i] + alpha*s[i] = r[j] + alpha*s[j]
    # alpha = (r[j] - r[i]) / (s[i] - s[j])
    
    n_lags = len(r)
    best_alpha = 0.5 * (alpha_lo + alpha_hi)
    best_val = np.inf
    
    # Evaluate at boundaries
    for a in [alpha_lo, alpha_hi]:
        val = np.max(r + a * s)
        if val < best_val:
            best_val = val
            best_alpha = a
    
    # Evaluate at breakpoints
    # For efficiency, only consider pairs where slopes differ
    for i in range(n_lags):
        for j in range(i+1, n_lags):
            if abs(s[i] - s[j]) < 1e-15:
                continue
            alpha_bp = (r[j] - r[i]) / (s[i] - s[j])
            if alpha_lo <= alpha_bp <= alpha_hi:
                val = np.max(r + alpha_bp * s)
                if val < best_val:
                    best_val = val
                    best_alpha = alpha_bp

def opt_alpha_bisect(r, s, a_lo, a_hi, n_iter=25):
    """Bisection to find alpha minimizing max(r + alpha*s)."""
    if a_lo >= a_hi:
        return (a_lo + a_hi) / 2
    lo, hi = a_lo, a_hi
    for _ in range(n_iter):
        if hi - lo < 1e-14:
            break
        mid = 0.5 * (lo + hi)
        vals = r + mid * s
        mx = vals.max()
        active = vals >= mx - 1e-12
        sa = s[active]
        smax = sa.max()
        smin = sa.min()
        if smin <= 0 and smax >= 0:
            lo = hi = mid
            break
        elif smin > 0:
            hi = mid
        else:  # smax < 0
            lo = mid
    return 0.5 * (lo + hi)

def best_pair_descent(h, S, r, deadline, max_sweeps=5000):

                if imp > best_imp:

                    best_u, best_v, best_a = u, v, a

        h[best_v] -= best_a
        # Clip for safety
        if h[best_u] < 0: h[best_u] = 0
        if h[best_u] > 1: h[best_u] = 1

        # Recompute
        r = get_full_corr(h)
        S = precompute_S(h, n)
