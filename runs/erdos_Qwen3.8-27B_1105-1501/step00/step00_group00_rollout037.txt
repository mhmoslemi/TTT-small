    temperature = 0.01
    weights = np.exp((corr - max_val) / temperature)

        #   m-lag ≥ 0 => m ≥ lag

                h_opt, c5_opt = slsqp_optimize(h_s, n_try, dx_try, 0.005, maxiter=200)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n = n_try
                # Second pass with lower temp
                if c5_opt < best_c5 + 0.01 and _remaining() > 20:
                    h_opt2, c5_opt2 = slsqp_optimize(h_opt, n_try, dx_try, 0.002, maxiter=200)
                    if c5_opt2 < best_c5:
                        best_h = h_opt2.copy()
                        best_c5 = c5_opt2
                        best_n = n_try

    # Phase 4: Final polish on best solution
    if best_h is not None and _remaining() > 20:
        n_f = best_n
        dx_f = 2.0 / n_f
        h_f, c5_f = slsqp_optimize(best_h, n_f, dx_f, 0.001, maxiter=500)
        if c5_f < best_c5:
            best_h = h_f.copy()
            best_c5 = c5_f
            best_n = n_f

    return best_h, best_c5, best_n
