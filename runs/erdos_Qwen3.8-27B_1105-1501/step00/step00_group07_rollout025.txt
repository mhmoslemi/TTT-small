def project_feas(v):
    n = len(v)
    target = n / 2.0
    v = np.array(v, dtype=float)
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-10:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-15:
            break
    v = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            idx = np.argsort(1.0 - v)
        else:
            idx = np.argsort(v)
            diff = -diff
        for i in idx:
            step = min((1.0 - v[i]) if diff > 0 else v[i], diff)
            v[i] += step
            diff -= step
            if diff < 1e-12:
                break
    return v

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw) * dx)

def project_feas(v):
    n = len(v)
    target = n / 2.0
    v = np.array(v, dtype=float)
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-10:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-15:
            break
    v = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            idxs = np.argsort(1.0 - v)
        else:
            diffuse = -diff
            idxs = np.argsort(v)
        for i in idxs:
            if diff > 0:
                step = min(1.0 - v[i], diff)
                v[i] += step
                diff -= step
            else:
                step = min(v[i], diffuse)
                v[i] -= step
                diffuse -= step
            if (diff if diff > 0 else diffuse) < 1e-12:
                break
    return v

def resample_to(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(target_n) + 0.5) / target_n
    v = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feas(v)
