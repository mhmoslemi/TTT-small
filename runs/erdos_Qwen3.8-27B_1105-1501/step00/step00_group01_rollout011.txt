import numpy as np

def _score(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _is_feasible(h, target, tol=1e-10):
    """Check if h is feasible: 0<=h<=1, sum(h)=target"""
    if h.min() < -tol or h.max() > 1.0 + tol:
        return False
    if abs(h.sum() - target) > tol * max(1, abs(target)):
        return False
    return True

def _project_box_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    x = h.astype(float).copy()
    # Binary search for lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    # Fix tiny residual
    residual = target - result.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-10)[0]
        else:
            idx = np.where(result > 1e-10)[0]
        if len(idx) > 0:
            # Distribute residual
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    return np.clip(result, 0.0, 1.0)

def _resample(h, new_len):
    """Resample h from len(h) to new_len using linear interpolation"""
    old_len = len(h)
    if old_len == new_len:
        return h.copy()
    src = np.linspace(0, 1, old_len)
    dst = np.linspace(0, 1, new_len)
    new_h = np.interp(dst, src, h)
    # Project to maintain feasibility
    target = new_len / 2.0
    return _project_box_sum(new_h, target)

                best_h[p] = h[p] + best_ref_d
