    return np.max(corr) * dx

def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    def f(alpha):
        return np.sum(np.clip(h - alpha, 0.0, 1.0)) - target_sum
    lo, hi = -2.0, 2.0
    if f(lo) < 0 or f(hi) > 0:
        h = h + diff / len(h)
        return np.clip(h, 0.0, 1.0)
    alpha = brentq(f, lo, hi, xtol=1e-15)
    return np.clip(h - alpha, 0.0, 1.0)

def pair_pass(h, dx, n_samples=30):
    n = len(h)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            alpha_lo = max(-h[i], h[j] - 1.0)
            alpha_hi = min(1.0 - h[i], h[j])
            if alpha_hi - alpha_lo < 1e-12:
                continue
            h_ai = h[i]
            h_aj = h[j]
            current_max = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
            best_alpha = 0.0
            best_val = current_max
            alphas = np.linspace(alpha_lo, alpha_hi, n_samples)
            for a in alphas:
                h[i] = h_ai + a
                h[j] = h_aj - a
                val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
                if val < best_val:
                    best_val = val
                    best_alpha = a
            h[i] = h_ai
            h[j] = h_aj
            if best_val < current_max - 1e-13:
                h[i] = h_ai + best_alpha
                h[j] = h_aj - best_alpha
                improved = True
    return h, improved

def slsqp_polish(h0, n, maxiter=300, active_tol=5e-3):
    dx = 2.0 / n
    target_sum = n / 2.0
    n_c = 2 * n - 1
    corr = np.correlate(h0, 1.0 - h0, mode="full") * dx
    max_val = corr.max()
    active = np.where(corr >= max_val - active_tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def con_fun(x):
        h = x[:n]
        t = x[n]
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - c[active]

    def con_jac(x):
        h = x[:n]
        m = len(active)
        J = np.zeros((m, n + 1))
        for idx, k in enumerate(active):
            shift = k - (n - 1)
            J[idx, n] = 1.0
            if shift >= 0:
                J[idx, :n - shift] -= dx * (1.0 - h[shift:n])
                J[idx, shift:n] += dx * h[0:n - shift]
            else:
                s = -shift
                J[idx, s:n] -= dx * (1.0 - h[0:n - s])
                J[idx, 0:n - s] += dx * h[s:n]
        return J

    def eq_fun(x):
        return x[:n].sum() - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    cons_ineq = {'type': 'ineq', 'fun': con_fun, 'jac': con_jac}
    cons_eq = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h0, [compute_c5(h0) + 0.01]])

    result = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                      constraints=[cons_ineq, cons_eq],
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
    h_opt = np.clip(result.x[:n], 0.0, 1.0)
    c5_opt = compute_c5(h_opt)
    return h_opt, c5_opt

def subgradient_descent(h, n_iter=3000, initial_step=0.005):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = initial_step
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        c5 = corr[max_idx] * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        shift = max_idx - (n - 1)
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n - shift] += (1.0 - h[shift:n])
            grad[shift:n] -= h[0:n - shift]
        else:
            s = -shift
            grad[s:n] += (1.0 - h[0:n - s])
            grad[0:n - s] -= h[s:n]
        h = h - step * grad
        h = project_feasible(h, target_sum)
        step *= 0.9997
    return best_h, best_c5

def random_perturb(h, magnitude, rng):
    n = len(h)
    target_sum = n / 2.0
    h_pert = h + rng.uniform(-magnitude, magnitude, n)
    return project_feasible(h_pert, target_sum)
