def exact_c5(h):
    """Compute the exact C5 value for a feasible h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_feasible(y):
    """Project y onto {h : 0<=h<=1, sum(h)=n/2}."""
    n = len(y)
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clamp(y - lambda, 0, 1)) = target
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix exact mass
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        # Distribute the difference
        idx = np.argsort(result)
        if diff > 0:
            for i in reversed(idx):
                room = 1.0 - result[i]
                add = min(diff, room)
                result[i] += add
                diff -= add
                if abs(diff) < 1e-12:

                room = result[i]
                sub = min(-diff, room)
                result[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return result
