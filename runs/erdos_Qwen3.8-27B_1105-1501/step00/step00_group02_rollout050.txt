import time

def evaluate_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap), overlap

def compute_softmax_gradient(h, dx, alpha):
    n = len(h)
    n_lags = 2 * n - 1
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    
    # Softmax weights
    a = alpha * overlaps
    a -= np.max(a)
    e = np.exp(a)
    w = e / np.sum(e)
    
    grad = np.zeros(n)
    
    for l in range(n_lags):
        s = l - (n - 1)
        wl = w[l] * dx
        
        # Term 1: for k where 0<=k<n and 0<=k+s<n, contribution is (1-h[k+s])
        k1s = max(0, -s)
        k1e = min(n, n - s)
        if k1s < k1e:
            grad[k1s:k1e] += wl * (1 - h[k1s + s:k1e + s])
        
        # Term 2: for k where 0<=k<n and 0<=k-s<n, contribution is -h[k-s]
        k2s = max(0, s)
        k2e = min(n, n + s)
        if k2s < k2e:
