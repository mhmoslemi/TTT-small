def score(h):
    """Compute C5 score: dx * max(correlate(h, 1-h, 'full'))"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def sym_complement(u, n):
    """Build full h from symmetric complement variable u."""
    m = n // 2
    h = np.zeros(n)
    h[:m] = u
    h[n-1:m-1:-1] = 1.0 - u  # h[n-1-j] = 1 - u[j] for j=0..m-1
    if n % 2 == 1:
        h[m] = 0.5
    return h

def project_to_sym(h, n):
    """Project h onto the symmetric complement subspace, return u."""
    m = n // 2
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    u = h_sym[:m].copy()
    return u

def correlation_values(h):
    """Compute all correlation values (unscaled)."""
    return np.correlate(h, 1.0 - h, mode='full')

def manual_correlation(h):
    """Compute correlation values manually for gradient computation."""
    n = len(h)
    vals = np.zeros(2*n - 1)
    for c in range(2*n - 1):
        k = c - (n - 1)
        start = max(0, -k)
        end = min(n, n - k)
        # sum_{i=start}^{end-1} h[i+k] * (1-h[i])
        vals[c] = np.sum(h[start+k:end+k] * (1 - h[start:end]))
    return vals

def grad_F_k(h, k):
    """Compute gradient of F_k = sum_i h[i+k]*(1-h[i]) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    start = max(0, -k)
    end = min(n, n - k)
    for i in range(start, end):
        # term: h[i+k] * (1-h[i])
        # d/dh[i+k]: (1-h[i])
        # d/dh[i]: -h[i+k]
        g[i+k] += (1.0 - h[i])
        g[i] += (-h[i+k])
    return g

    m = n // 2
    g = grad_F_k(h, k)

    gu = g[:m] - g[n-1:m-1:-1]
    return gu

def active_set_slsqp(u, n, max_iter=10):
    m = n // 2
    dx = 2.0 / n
    best_u = u.copy()
    
    for it in range(max_iter):
        h = sym_complement(best_u, n)
        corr = correlation_values(h)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Find active set (top K by value)
        K = min(20, len(corr))
        active_idx = np.argsort(corr)[-K:]
        
        # SLSQP with only active constraints
        constraints = []
        for c in active_idx:
            constraints.append(make_constraint(c))
        
        theta0 = max_val * dx + 1e-6
        z0 = np.concatenate([best_u, [theta0]])
        
        result = minimize(objective, z0, method='SLSQP', jac=objective_grad,
                        constraints=constraints, bounds=bounds,
                        options={'maxiter': 50, 'ftol': 1e-10})
        
        if result.fun < max_val * dx:
            new_u = result.x[:m]
            h_new = sym_complement(new_u, n)
            s = score(h_new)
            if s < score(h):
                best_u = new_u
            else:
                # Non-active constraint violated, expand active set
                break
        else:
            break

# Global parameter store to avoid closures
_P = {'n': 0, 'm': 0, 'dx': 0, 'beta': 100}

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def build_sym(u, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = u
    if n % 2 == 1:
        h[m] = 0.5
        h[m+1:] = 0.0
        h[n-1:m:-1] = 1.0 - u
    else:
        h[m:] = 0.0
        h[n-1:m-1:-1] = 1.0 - u
    return h

def project_sym(h, n):
    m = n // 2
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return h_sym[:m].copy()

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_Fk_full(h, k, n):
    s = max(0, -k)
    e = min(n, n - k)
    g = np.zeros(n)
    if e > s:
        idx = np.arange(s, e)
        g[idx + k] += (1.0 - h[idx])
        g[idx] -= h[idx + k]
    return g

def grad_Fk_red(h, k, n):
    m = n // 2
    g = grad_Fk_full(h, k, n)
    return g[:m] - g[n-1:m-1:-1]

def all_reduced_grads(h, n):
    m = n // 2
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, m))
    for c in range(n_lags):
        k = c - (n - 1)
        g = grad_Fk_full(h, k, n)
        grads[c] = g[:m] - g[n-1:m-1:-1]
    return grads

    beta = _P['beta']
    dx = _P['dx']

    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return F_max + log_sum / beta

    m = _P['m']

    dx = _P['dx']

    _P['n'] = n
    _P['m'] = m
    _P['dx'] = 2.0 / n
    _P['beta'] = beta
    bounds = [(0.0, 1.0)] * m
    res = minimize(_obj_fun, u_init, jac=_grad_fun, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                           'gtol': 1e-10, 'maxls': 50})
    return res.x

def binary_local_search(u_start, n, max_rounds=20000, t0=0, budget_left=30):
    m = n // 2
    u = u_start.copy()
    h = build_sym(u, n)
    s = compute_score(h)
    for rnd in range(max_rounds):
        if time.time() - t0 > budget_left:
            break
        improved = False
        for j in range(m):
            u_flipped = u.copy()
            u_flipped[j] = 1.0 - u_flipped[j]
            h_try = build_sym(u_flipped, n)
            s_try = compute_score(h_try)
            if s_try < s - 1e-15:
                u = u_flipped
                h = h_try
                s = s_try
                improved = True
                break
        if not improved:
            # Try 2-flip moves
            two_flip_found = False
            for j in range(m):
                if two_flip_found:
                    break
                for k2 in range(j+1, m):
                    if time.time() - t0 > budget_left:
                        return u, s
                    u_try2 = u.copy()
                    u_try2[j] = 1.0 - u_try2[j]
                    u_try2[k2] = 1.0 - u_try2[k2]
                    h_try2 = build_sym(u_try2, n)
                    s_try2 = compute_score(h_try2)
                    if s_try2 < s - 1e-15:
                        u = u_try2
                        s = s_try2
                        improved = True
                        two_flip_found = True
                        break
            if not improved:
                break
    return u, s

def simulated_annealing(u_start, n, rng, t0, budget_left, T_init=0.15, T_min=1e-6, cool=0.9997):
    m = n // 2
    u = u_start.copy()
    h = build_sym(u, n)
    s = compute_score(h)
    T = T_init
    while T > T_min and time.time() - t0 < budget_left:
        j = rng.randint(m)
        u_try = u.copy()
        u_try[j] = 1.0 - u_try[j]
        h_try = build_sym(u_try, n)
        s_try = compute_score(h_try)
        delta = s_try - s
        if delta < 0 or rng.random_sample() < np.exp(-delta / max(T, 1e-12)):
            u = u_try
            s = s_try
        T *= cool
    return u, s

def try_n_value(n, rng, t0, budget_left, u_ref=None):
    m = n // 2
    best_u = None
    best_s = None
    
    # Generate starting points
    starts = []
    if u_ref is not None:
        # Interpolate reference to new n
        n_ref = len(u_ref) * 2 + (1 if (len(u_ref)*2 + 1) == n else 0)
        m_ref = len(u_ref)
        if m_ref > 1:
            x_old = np.linspace(0, 1, m_ref)
            x_new = np.linspace(0, 1, m)
            u_interp = np.interp(x_new, x_old, u_ref)
            starts.append(u_interp)
            starts.append((u_interp > 0.5).astype(float))
    for _ in range(8):
        starts.append((rng.rand(m) > 0.5).astype(float))
    starts.append(np.where(np.arange(m) % 2 == 0, 1.0, 0.0))
    starts.append((np.arange(m) < m // 2).astype(float))
    
    SLACK = 40
    
    for u_start in starts:
        if time.time() - t0 > budget_left - SLACK:
            break
        # Continuous optimization
        try:
            u_opt = optimize_smooth_max(u_start, n, beta=80, maxiter=150)
            h_opt = build_sym(u_opt, n)
            s_opt = compute_score(h_opt)
            if best_s is None or s_opt < best_s:
                best_s = s_opt
                best_u = u_opt.copy()
        except:
            pass
        if time.time() - t0 > budget_left - SLACK:
            break
        # Binary search from the optimized solution
        u_bin = (u_opt > 0.5).astype(float)
        u_bs, s_bs = binary_local_search(u_bin, n, max_rounds=5000, t0=t0, budget_left=budget_left - 10)
        if s_bs < (best_s if best_s is not None else 1.0):
            best_s = s_bs
            best_u = u_bs.copy()
    
    if best_u is not None:
        # Final SLSQP polish
        if time.time() - t0 < budget_left - 5:
            try:
                u_polish = optimize_smooth_max(best_u, n, beta=150, maxiter=200)
                h_polish = build_sym(u_polish, n)
                s_polish = compute_score(h_polish)
                if s_polish < best_s:
                    best_s = s_polish
                    best_u = u_polish
            except:
                pass
        return build_sym(best_u, n), best_s
    return None, None
