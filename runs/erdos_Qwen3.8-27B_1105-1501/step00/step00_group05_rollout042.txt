import numpy as np
from scipy.optimize import minimize
import time


def evaluate(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))


def project_feasible(raw, target_sum):
    N = len(raw)
    lo = float(np.min(raw)) - 2.0
    hi = float(np.max(raw)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(raw - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(raw - 0.5 * (lo + hi), 0.0, 1.0)


def get_all_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_single_corr(h, k_idx):
    N = len(h)
    dx = 2.0 / N
    delta = k_idx - (N - 1)
    if delta >= 0:
        return float(np.sum(h[:N - delta] * (1.0 - h[delta:])) * dx)
    else:
        d = -delta
        return float(np.sum(h[d:] * (1.0 - h[:N - d])) * dx)


def compute_grad_single(h, k_idx):
    N = len(h)
    dx = 2.0 / N
    delta = k_idx - (N - 1)
    grad = np.zeros(N)
    if delta >= 0:
        idx = np.arange(N - delta)
        jdx = idx + delta
        grad[idx] += dx * (1.0 - h[jdx])
        grad[jdx] += -dx * h[idx]
    else:
        d = -delta
        idx = np.arange(N - d)
        jdx = idx + d
        grad[jdx] += dx * (1.0 - h[idx])
        grad[idx] += -dx * h[jdx]
    return grad


class Prob:
    def __init__(self, N, active, target_sum):
        self.N = N
        self.active = active
        self.target_sum = target_sum

    def obj(self, x):
        return x[self.N]

    def ineq_fun(self, x):
        h = x[:self.N]
        t = x[self.N]
        m = len(self.active)
        out = np.zeros(m)
        for i in range(m):
            out[i] = t - compute_single_corr(h, self.active[i])
        return out

    def ineq_jac(self, x):
        h = x[:self.N]
        N = self.N
        m = len(self.active)
        jac = np.zeros((m, N + 1))
        for i in range(m):
            jac[i, :N] = -compute_grad_single(h, self.active[i])
            jac[i, N] = 1.0
        return jac

    def eq_fun(self, x):
        return float(np.sum(x[:self.N]) - self.target_sum)

    def eq_jac(self, x):
        jac = np.zeros(self.N + 1)
        jac[:self.N] = 1.0
        return jac


def slsqp_optimize(h0, active_tol=0.02, max_iter=300):
    N = len(h0)
    target_sum = N / 2.0
    h0 = np.asarray(h0, dtype=float).copy()
    corr = get_all_corr(h0)
    max_val = float(np.max(corr))
    active_mask = corr >= max_val - active_tol
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > 10:
        order = np.argsort(-corr[active_idx])
        active_idx = active_idx[order[:10]]
    if len(active_idx) == 0:
        active_idx = np.argsort(-corr)[-3:]
    prob = Prob(N, active_idx, target_sum)
    bounds = [(0.0, 1.0)] * N + [(0.0, max_val + 0.5)]
    x0 = np.concatenate([h0, [max_val]])
    cons = [
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
    ]
    res = minimize(prob.obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    best_h = h0.copy()
    best_val = max_val
    if res.x[N] < best_val - 1e-10:
        h_cand = np.clip(res.x[:N], 0.0, 1.0)
        h_cand = project_feasible(h_cand, target_sum)
        val = evaluate(h_cand)
        if val < best_val:
            best_h = h_cand
            best_val = val
    return best_h, best_val
