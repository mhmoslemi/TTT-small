```python
def resample(h, N_old, N_new):
    dx_old = 2.0 / N_old
    # Cumulative integral
    cum = np.zeros(N_old + 1)
    cum[1:] = np.cumsum(h * dx_old)
    # New grid points (midpoints)
    x_new = (np.arange(N_new) + 0.5) * 2.0 / N_new
    # Old grid midpoints for interpolation
    x_old = (np.arange(N_old) + 0.5) * 2.0 / N_old
    # Interpolate
    h_new = np.interp(x_new, x_old, h)
    # Clip
    h_new = np.clip(h_new, 0, 1)
    # Project to feasible set
```

```python
def dyadic_repair(h, N, bits=20):
    S = 1 << bits
    target = (N * S) // 2
    
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    
    current_sum = int(z.sum())
    diff = target - current_sum
    
    if diff > 0:
        # Need to add diff units total
        # Use water-filling: find threshold where we add 1
        # Sort indices by (h[i]*S - z[i]) descending (most "deserving" of increase)
        scores = h * S - z  # fractional "desire" to be rounded up
        scores = scores.copy()
        # Indices that can still increase
        can_inc = z < S
        scores[~can_inc] = -1
        # Add 1 to the top 'diff' indices by score
        if diff <= N:
            top_idx = np.argsort(-scores)[:diff]
            z[top_idx] += 1
        # If diff > N (unlikely), need multiple rounds
        elif diff < 0:
        # Need to remove |diff| units
        scores = z - h * S  # fractional "desire" to be rounded down
        can_dec = z > 0
        scores[~can_dec] = -1
        if abs(diff) <= N:
```

```python
def _compute_subgradient(h, active_lags, N):
    dx = 2.0 / N
    grad = np.zeros(N)
    for m in active_lags:
        k = m - (N - 1)
        i_lo = max(0, -k)
        i_hi = min(N - 1, N - 1 - k)
        if i_lo > i_hi:
            continue
        idx1 = np.arange(i_lo, i_hi + 1)
        grad[idx1] += (1.0 - h[idx1 + k])
        p_lo = i_lo + k
        p_hi = i_hi + k
        idx2 = np.arange(p_lo, p_hi + 1)
        grad[idx2] -= h[idx2 - k]
    return dx * grad

def _project_feasible(x, N):
    target = N / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

    # This won't be exact but very close
    return h

def _dyadic_repair(h, N, bits=20):
    S = 1 << bits
    target = (N * S) // 2
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)

        scores[z >= S] = -np.inf

            z[top] += 1

        scores[z <= 0] = -np.inf
        if -diff <= N:

            z[top] -= 1
```

```python
from scipy.optimize import linprog

def _lp_step(h, active_lags, obj_vec, N, delta_max=0.05):
    """Solve LP to find optimal small step."""
    dx = 2.0 / N
    grad = _compute_subgradient(h, active_lags, N)  # shape (len(active_lags), N)
    # Wait, _compute_subgradient returns sum of gradients for all active lags
    # For LP, I need individual gradients for each active lag
    
    # Let me compute individual gradients
    individual_grads = []
    for m in active_lags:
        g = np.zeros(N)
        k = m - (N - 1)
        i_lo = max(0, -k)
        i_hi = min(N - 1, N - 1 - k)
        if i_lo <= i_hi:
            idx1 = np.arange(i_lo, i_hi + 1)
            g[idx1] += (1.0 - h[idx1 + k])
            p_lo = i_lo + k
            p_hi = i_hi + k
            idx2 = np.arange(p_lo, p_hi + 1)
            g[idx2] -= h[idx2 - k]
        individual_grads.append(dx * g)
    
    A_grads = np.array(individual_grads)  # shape (n_active, N)
    f_active = obj_vec[active_lags]  # current values
    
    n_var = N + 1  # N delta + 1 t
    c = np.zeros(n_var)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # For each active m: A_grads[m] @ delta - t <= -f_active[m]
    A_ineq = np.zeros((len(active_lags) + 2*N, n_var))
    b_ineq = np.zeros(len(active_lags) + 2*N)
    
    for i, m_idx in enumerate(range(len(active_lags))):
        A_ineq[i, :N] = A_grads[m_idx]
        A_ineq[i, N] = -1.0
        b_ineq[i] = -f_active[m_idx]
    
    for p in range(N):
        # delta[p] <= min(1-h[p], delta_max)
        A_ineq[len(active_lags)+p, p] = 1.0
        b_ineq[len(active_lags)+p] = min(1-h[p], delta_max)
        # -delta[p] <= min(h[p], delta_max)
        A_ineq[len(active_lags)+N+p, p] = -1.0
        b_ineq[len(active_lags)+N+p] = min(h[p], delta_max)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,

    if res.success:
        delta = res.x[:N]
```

```python
import numpy as np
import time
from scipy.optimize import linprog


def _exact_objective(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def _objective_vector(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _grad_one_lag(h, m, N):
    """Gradient of the m-th correlate entry (times dx) with respect to h."""
    dx = 2.0 / N
    g = np.zeros(N)
    k = m - (N - 1)
    i_lo = max(0, -k)
    i_hi = min(N - 1, N - 1 - k)
    if i_lo > i_hi:
        return g
    idx1 = np.arange(i_lo, i_hi + 1)
    g[idx1] += (1.0 - h[idx1 + k])
    p_lo = i_lo + k
    p_hi = i_hi + k
    idx2 = np.arange(p_lo, p_hi + 1)
    g[idx2] -= h[idx2 - k]
    return dx * g


def _project_feasible(x, N):
    """Project x to {h: 0<=h<=1, sum(h)=N/2} via bisection on shift."""
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)


def _dyadic_repair(h, N, bits=20):
    """Quantize h to dyadic grid with exact sum = N/2."""
    S = 1 << bits
    target = (N * S) // 2
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    cs = int(z.sum())
    diff = target - cs
    if diff > 0:
        scores = (h * S - z).copy()
        scores[z >= S] = -999.0
        order = np.argsort(-scores)
        for j in range(min(diff, N)):
            z[order[j]] += 1
    elif diff < 0:
        scores = (z - h * S).copy()
        scores[z <= 0] = -999.0
        order = np.argsort(-scores)
        for j in range(min(-diff, N)):
            z[order[j]] -= 1
    return z.astype(np.float64) / S


def _resample(h_old, N_old, N_new):
    """Resample h from N_old to N_new grid points."""
    dx_old = 2.0 / N_old
    x_old = (np.arange(N_old) + 0.5) * dx_old
    x_new = (np.arange(N_new) + 0.5) * 2.0 / N_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_feasible(h_new, N_new)
    return h_new


def _lp_step(h, active_lags, obj_vec, N, delta_max):
    """Solve LP for optimal small step reducing active constraints."""
    n_act = len(active_lags)
    if n_act == 0:
        return None
    # Compute gradient for each active lag
    A_grads = np.array([_grad_one_lag(h, m, N) for m in active_lags])
    f_active = obj_vec[active_lags]

    n_var = N + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t

    # Inequalities: A_ineq @ x <= b_ineq
    n_ineq = n_act + 2 * N
    A_ineq = np.zeros((n_ineq, n_var))
    b_ineq = np.zeros(n_ineq)

    for i in range(n_act):
        A_ineq[i, :N] = A_grads[i]
        A_ineq[i, N] = -1.0
        b_ineq[i] = -f_active[i]

    for p in range(N):
        A_ineq[n_act + p, p] = 1.0
        b_ineq[n_act + p] = min(1.0 - h[p], delta_max)
        A_ineq[n_act + N + p, p] = -1.0
        b_ineq[n_act + N + p] = min(h[p], delta_max)

    A_eq = np.zeros((1, n_var))
    A_eq[0, :N] = 1.0
    b_eq = np.array([0.0])

    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            delta = res.x[:N]
            if np.abs(delta).max() < 1e-14:
                return None
            return h + delta
    except Exception:
        pass
    return None


def _subgrad_step(h, active_lags, N, step_size):
    """Subgradient step averaging gradients of active lags."""
    dx = 2.0 / N
    grad = np.zeros(N)
    for m in active_lags:
        grad += _grad_one_lag(h, m, N)
    grad /= len(active_lags)
    grad_centered = grad - grad.mean()
    gnorm = np.linalg.norm(grad_centered)
    if gnorm < 1e-14:
        return None
    alpha = step_size / gnorm
    h_new = h - alpha * grad_centered
    return _project_feasible(h_new, N)


def _optimize(h, N, deadline, rng):
    """Main optimization loop using LP and subgradient steps."""
    best_h = h.copy()
    best_c = _exact_objective(h)
    dx = 2.0 / N
    delta_max = 0.08
    stagnant = 0
    min_delta = 0.0005

    it = 0
    while time.time() < deadline:
        it += 1
        obj_vec = _objective_vector(h)
        current_c = obj_vec.max()

        if current_c < best_c - 1e-10:
            best_c = current_c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 300:
                break

        # Active lags within tolerance
        tol = 0.03 * dx
        active_mask = obj_vec >= current_c - tol
        active_lags = np.where(active_mask)[0]

        # Try LP step
        h_new = _lp_step(h, active_lags, obj_vec, N, delta_max)

        if h_new is not None:
            new_c = _exact_objective(h_new)
            if new_c < current_c - 1e-13:
                h = h_new
                # Adapt step size
                improvement = current_c - new_c
                if improvement > 0.3 * tol:
                    delta_max = min(delta_max * 1.05, 0.2)
                else:
                    delta_max *= 0.97
            else:
                delta_max *= 0.85
                if delta_max < min_delta:
                    # Try subgradient
                    h_new = _subgrad_step(h, active_lags, N, delta_max * 0.5)
                    if h_new is not None:
                        new_c = _exact_objective(h_new)
                        if new_c < best_c:
                            h = h_new
                            best_c = new_c
                            best_h = h.copy()
                            stagnant = 0
                    else:
                        break
        else:
            # LP infeasible or failed, try subgradient
            h_new = _subgrad_step(h, active_lags, N, delta_max)
            if h_new is not None:
                new_c = _exact_objective(h_new)
                if new_c < current_c - 1e-13:
                    h = h_new
                else:
                    delta_max *= 0.8
            if delta_max < min_delta:
                break

        # Periodic random perturbation to escape local optima
        if it % 200 == 0:
            pert = rng.normal(0, 0.005, N)
            pert -= pert.mean()
            h_pert = _project_feasible(h + pert, N)
            c_pert = _exact_objective(h_pert)
            if c_pert < current_c:
                h = h_pert

    return best_h, best_c


def _coordinate_polish(h, N, deadline, rng, max_rounds=5000):
    """Polish using coordinate moves (transfer between pairs)."""
    best_h = h.copy()
    best_c = _exact_objective(h)

    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        # Pick two indices to swap mass
        i = rng.integers(0, N)
        j = rng.integers(0, N)
        if i == j:
            continue
        # Try small transfers in both directions
        for sign in [1, -1]:
            for amt in [0.001, 0.005, 0.01, 0.02]:
                h_new = h.copy()
                h_new[i] += sign * amt
                h_new[j] -= sign * amt
                if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                    continue
                new_c = _exact_objective(h_new)
                if new_c < best_c - 1e-12:
                    h = h_new
                    best_h = h_new.copy()
                    best_c = new_c
                    break

    return best_h, best_c
```
