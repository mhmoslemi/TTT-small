from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    """Project y to {h: 0<=h_i<=1, sum(h)=n/2}"""
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    # Replace non-finite values
    y = np.where(np.isfinite(y), y, 0.5)
    
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(y - lam, 0.0, 1.0)
    
    # Fix residual mass
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            slacks = 1.0 - result
        else:
            slacks = result
        order = np.argsort(-slacks)
        remaining = abs(diff)
        sign = 1.0 if diff > 0 else -1.0
        for idx in order:
            if remaining <= 1e-15:
                break
            move = min(slacks[idx], remaining)
            result[idx] += sign * move
            remaining -= move
    return result

def compute_correlation(h):
    """Return the full correlation vector scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smoothed_value(h, tau):
    c = compute_correlation(h)
    M = np.max(c)
    shifted = np.clip((c - M) / tau, -500, 500)
    return M + tau * np.log(np.sum(np.exp(shifted)))

def smoothed_gradient(h, tau):
    """Gradient of the smoothed max objective."""
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    M = np.max(c)
    shifted = np.clip((c - M) / tau, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k] * dx
        if k == center:
            grad += wk * (1.0 - 2.0 * h)
        elif k > center:
            l = k - center
            if l < n:
                grad[l:] += wk * (1.0 - h[:-l])
                grad[:n-l] -= wk * h[l:]
        else:
            p = center - k
            if p < n:
                grad[:n-p] += wk * (1.0 - h[p:])

def transfer_resolution(h_src, n_target):
    """Transfer a function from n_src resolution to n_target via area preservation."""
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    
    dx_src = 2.0 / n_src
    # Build cumulative area
    cum = np.zeros(n_src + 1)
    cum[1:] = np.cumsum(h_src) * dx_src
    
    # Interpolate to target grid
    x_src = np.linspace(0, 2, n_src + 1)
    x_tgt = np.linspace(0, 2, n_target + 1)
    cum_tgt = np.interp(x_tgt, x_src, cum)
    
    # Convert back to heights
    dx_tgt = 2.0 / n_target
    h_tgt = np.diff(cum_tgt) / dx_tgt
    
    return project_feasible(h_tgt)

def exact_c5(h):
    n = len(h)
    if n == 0:
        return np.inf

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(y - lam, 0.0, 1.0)
    diff = target - float(np.sum(result))
    if abs(diff) > 1e-10:
        if diff > 0:
            slacks = 1.0 - result
        else:
            slacks = result
        order = np.argsort(-slacks)
        remaining = abs(diff)
        sign = 1.0 if diff > 0 else -1.0
        for idx in order:
            if remaining <= 1e-15:
                break
            move = min(slacks[idx], remaining)
            result[idx] += sign * move
            remaining -= move
    return result

        else:

            if p < n:
