import numpy as np
from scipy.optimize import linprog
import time

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_mass(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.array(v, dtype=float).copy()
    np.clip(v, 0, 1, out=v)
    
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) * 0.5, 0, 1)
    
    # Fix residual
    for _ in range(30):
        res = target - np.sum(result)
        if abs(res) < 1e-13:
            break
        movable = (result > 1e-13) & (result < 1 - 1e-13)
        cnt = np.sum(movable)
        if cnt == 0:
            break
        result[movable] = np.clip(result[movable] + res / cnt, 0, 1)
    return result

def compute_grads(h):
    """Compute gradient matrix for all lags. Returns (2n-1, n) array."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k_idx in range(2 * n - 1):
        l = k_idx - (n - 1)
        if l >= 0:
            idx = np.arange(n - l)
            G[k_idx, idx] += (1.0 - h[idx + l])
            G[k_idx, idx + l] -= h[idx]
        elif l < 0:
            t = -l
            idx = np.arange(t, n)
            G[k_idx, idx] += (1.0 - h[idx - t])
            G[k_idx, idx - t] -= h[idx]
        else:
            G[k_idx] = 1.0 - 2.0 * h
    return G

def lp_direction(h_c, trust_radius=0.15):
    """LP-based direction to minimize linearized max overlap."""
    n = len(h_c)
    corr = np.correlate(h_c, 1.0 - h_c, mode="full")
    G = compute_grads(h_c)
    
    r = trust_radius
    lb = np.maximum(0.0, h_c - r)
    ub = np.minimum(1.0, h_c + r)
    
    if np.any(lb > ub + 1e-9):
        return None
    
    num_constraints = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([G, -np.ones((num_constraints, 1))])
    b_ub = G @ h_c - corr
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    t_upper = float(corr.max()) + 5.0
    bounds = list(zip(lb, ub)) + [(0, t_upper)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10})
    except:
        return None
    
    if res.status != 0 or not hasattr(res, 'x'):
        return None
    
    h_new = res.x[:n]
    d = h_new - h_c
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    d = d / norm
    return d

def line_search(h_c, d, current_score, max_iters=300):
    """Exact line search along h_c + t*d maintaining feasibility."""
    n = len(h_c)
    eps = 1e-12
    
    # Ensure zero sum
    d = d.copy()
    d -= np.mean(d)
    
    # Find valid range
    t_min, t_max = 0.0, 2.0
    for i in range(n):
        if d[i] > eps:
            t_min = max(t_min, -h_c[i] / d[i])
            t_max = min(t_max, (1.0 - h_c[i]) / d[i])
        elif d[i] < -eps:
            t_min = max(t_min, (1.0 - h_c[i]) / d[i])
            t_max = min(t_max, -h_c[i] / d[i])
    
    if t_min >= t_max - 1e-10:
        return h_c, current_score
    
    t_min = max(t_min, 0.0)
    
    # Grid search
    best_t = 0.0
    best_score = current_score
    ts = np.linspace(t_min, t_max, max_iters)
    
    for t in ts:
        h_try = h_c + t * d
        if np.min(h_try) < -1e-9 or np.max(h_try) > 1 + 1e-9:
            continue
        h_try = np.clip(h_try, 0, 1)
        if abs(np.sum(h_try) - n / 2.0) > 0.05:

        s = score_h(h_try)
        if s < best_score - 1e-13:
            best_score = s
            best_t = t

        return h_c, current_score
    
    # Refine
    span = (t_max - t_min) / max_iters
    ts_fine = np.linspace(max(t_min, best_t - 3 * span), min(t_max, best_t + 3 * span), 100)
    for t in ts_fine:

            h_try = project_box_mass(h_try)

            best_score = s
            best_t = t

    h_best = project_box_mass(h_best)
    return h_best, score_h(h_best)

def random_direction(n, rng):
    d = rng.randn(n)
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm > 0:
        d /= norm
    return d

def coordinate_pass(h, rng, n_swaps=500):
    """Coordinate descent pass with random 2-point swaps."""
    n = len(h)
    current_score = score_h(h)
    improved = 0
    
    for _ in range(n_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try transferring from i to j
        for delta in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
            if h[i] + delta < 0 or h[i] + delta > 1:
                continue
            if h[j] - delta < 0 or h[j] - delta > 1:
                continue
            h[i] += delta
            h[j] -= delta
            s = score_h(h)
            if s < current_score - 1e-13:
                current_score = s
                improved += 1
                break
            else:
                h[i] -= delta
                h[j] += delta
    
    return h, current_score, improved

def resample(h, target_n):
    """Resample h to target_n points."""
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # centers
    x_new = np.linspace(0, 2, target_n, endpoint=False) + 1.0/target_n
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_box_mass(h_new, target_n / 2.0)
