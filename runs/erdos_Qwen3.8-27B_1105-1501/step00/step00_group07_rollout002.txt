def project_feasible(z, target):
    n = len(z)
    v = np.clip(np.nan_to_num(z), 0.0, 1.0)
    e = np.sum(v) - target
    if e > 1e-12:
        idx = np.argsort(-v)
        for i in idx:
            if e <= 1e-12:
                break
            d = min(v[i], e)
            v[i] -= d
            e -= d
    elif e < -1e-12:
        idx = np.argsort(v)
        for i in idx:
            if e >= -1e-12:
                break
            d = min(1.0 - v[i], -e)
            v[i] += d
            e += d
    return v

def score(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode="full"))

def pair_step(h, i, j, n_grid=40):
    """Optimize pair (i,j) keeping h[i]+h[j] constant."""
    n = len(h)
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-12:
        return h
    
    # Evaluate on grid
    ts = np.linspace(t_lo, t_hi, n_grid)
    best_C = np.inf
    best_t = t_lo
    for t in ts:
        h[i] = t
        h[j] = s - t
        C = score(h)
        if C < best_C:
            best_C = C
            best_t = t
    
    # Golden section refinement
    a, b = max(t_lo, best_t - (ts[1]-ts[0])), min(t_hi, best_t + (ts[1]-ts[0]))
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(30):
        if b - a < 1e-10:
            break
        m1 = a + (1 - gr) * (b - a)
        m2 = a + gr * (b - a)
        h[i] = m1; h[j] = s - m1
        C1 = score(h)
        h[i] = m2; h[j] = s - m2
        C2 = score(h)
        if C1 < C2:
            b = m2
        else:
            a = m1
    
    t_opt = (a + b) / 2
    h[i] = t_opt
    h[j] = s - t_opt
    return h
