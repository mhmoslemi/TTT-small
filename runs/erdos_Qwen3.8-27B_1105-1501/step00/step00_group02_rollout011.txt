    d_last = d[n-1]
    G_last = G[n-1]
    grad_y = G[:n-1] * d[:n-1] - G_last * d_last - ((d[:n-1] - d_last) / D) * S

def dyadic_snap(h, n, qbits=25):
    """Snap h to dyadic grid with exact sum = n/2."""
    Q = 2 ** qbits
    target = n * Q  # sum of integers should be n*Q/2 = n*2^(qbits-1)
    target_int = n * (2 ** (qbits - 1))  # = n * Q / 2
    
    k = np.round(h * Q).astype(np.int64)
    k = np.clip(k, 0, Q)
    
    diff = target_int - np.sum(k)
    
    if diff > 0:
        # Need to add diff units
        # Priority: positions where h is closest to (k+1)/Q from below
        # i.e., where h*Q - k is largest (h is closer to next grid point above)
        frac = h * Q - k
        # Can only add where k < Q
        mask = k < Q
        # Sort by frac descending (most "deserving" of rounding up)
        order = np.argsort(-frac)
        count = 0
        for idx in order:
            if not mask[idx]:
                continue
            if count >= diff:
                break
            k[idx] += 1
            mask[idx] = False
            count += 1
    elif diff < 0:
        # Need to subtract -diff units
        frac = h * Q - k
        mask = k > 0
        order = np.argsort(frac)  # most "deserving" of rounding down (smallest frac)
        count = 0
        for idx in order:
            if not mask[idx]:
                continue
            if count >= -diff:
                break
            k[idx] -= 1
            mask[idx] = False
            count += 1
    
    return k.astype(np.float64) / Q

                if h[p] - h[q] <= 1e-15:

                max_t = min(h[p], 1.0 - h[q])
                if max_t <= 1e-14:
                    continue
                for frac in (0.2, 0.5, 0.8):
                    t = frac * max_t
                    h[p] -= t
                    h[q] += t
                    c = compute_c5(h)
                    if c < best_c - 1e-13:
                        best_c = c
                        improved = True
                        break
                    h[p] += t
                    h[q] -= t
        if not improved:
            break
    return h, best_c


_global_deadline = float('inf')


def run(seed=42, budget_s=1100, **kwargs):
    global _global_deadline
    start_time = time.time()
    seed = int(seed)
    budget_s = float(budget_s)
    _global_deadline = start_time + budget_s * 0.95
    rng = np.random.default_rng(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).flatten()
    except Exception:
        h0 = None

    if h0 is not None and len(h0) > 2:
        if np.all(h0 >= -1e-10) and np.all(h0 <= 1.0 + 1e-10):
            h0 = np.clip(h0, 0.0, 1.0)
        else:
            h0 = None

    if h0 is None:
        n = 47
        h0 = np.full(n, 0.5)

    n = len(h0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    def _remaining():
        return _global_deadline - time.time()

    y0 = h_to_raw(h0, n)

    betas_stage1 = [4, 10, 30]
    for beta in betas_stage1:
        if _remaining() < 15.0:
            break
        n_starts = 4
        for s in range(n_starts):
            if _remaining() < 5.0:
                break
            if s == 0:
                y_start = y0.copy()
            elif s == 1:
                y_start = y0 + rng.normal(0, 0.15, size=n - 1)
            elif s == 2:
                y_start = y0 + rng.normal(0, 0.5, size=n - 1)
            else:
                y_start = y0 + rng.normal(0, 1.5, size=n - 1)

            W = 25.0
            bounds = [(-W, W)] * (n - 1)
            so = SmoothObj(n, beta)
            try:
                res = minimize(so.f, y_start, jac=so.g,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'maxfun': 8000,
                                        'ftol': 1e-14, 'gtol': 1e-9, 'gtol_r': 1e-9})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_cand = raw_to_h(res.x, n)
                    c_cand = compute_c5(h_cand)
                    if c_cand < best_c:
                        best_c = c_cand
                        best_h = h_cand.copy()
                        y0 = res.x.copy()
            except Exception:
                pass

    if _remaining() > 20.0:
        betas_stage2 = [80, 300]
        y_best = h_to_raw(best_h, n)
        for beta in betas_stage2:
            if _remaining() < 10.0:
                break
            W = 25.0
            bounds = [(-W, W)] * (n - 1)
            so = SmoothObj(n, beta)
            try:
                res = minimize(so.f, y_best, jac=so.g,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 600, 'maxfun': 12000,
                                        'ftol': 1e-14, 'gtol': 1e-10})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_cand = raw_to_h(res.x, n)
                    c_cand = compute_c5(h_cand)
                    if c_cand < best_c:
                        best_c = c_cand
                        best_h = h_cand.copy()
                        y_best = res.x.copy()
            except Exception:
                pass

    if _remaining() > 15.0:
        best_h, best_c = pairwise_polish(best_h, n, max_iter=8,
                                         time_limit=min(60.0, _remaining() * 0.4))

    if _remaining() > 10.0:
        best_h, best_c = local_coord_descent(best_h, n, max_iter=4,
                                             time_limit=min(40.0, _remaining() * 0.3))

    if _remaining() > 10.0:
        best_h, best_c = pairwise_polish(best_h, n, max_iter=6,
                                         time_limit=min(40.0, _remaining() * 0.3))

    for qbits in (26, 25, 24, 23):
        h_snap = dyadic_snap(best_h, n, qbits)
        if np.all(h_snap >= -1e-14) and np.all(h_snap <= 1.0 + 1e-14):
            h_snap = np.clip(h_snap, 0.0, 1.0)
            target_sum = n / 2.0
            if abs(np.sum(h_snap) - target_sum) < 1e-7:
                c_snap = compute_c5(h_snap)
                if c_snap <= best_c + 1e-9:
                    best_h = h_snap
                    best_c = c_snap
                    break

    final_c = compute_c5(best_h)
    best_c = min(best_c, final_c)
    return best_h, best_c, n
