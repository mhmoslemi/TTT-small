def official_score(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(z, target=None):

def manual_lags(x):
    n = len(x)
    vals = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        m = idx - (n - 1)
        i_start = max(0, -m)
        i_end = min(n, n - m)
        if i_start >= i_end:
            continue
        i = np.arange(i_start, i_end)
        j = i + m
        vals[idx] = np.sum(x[j] * (1.0 - x[i]))
    return vals


def manual_grad_matrix(x):
    n = len(x)
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        m = idx - (n - 1)
        i_start = max(0, -m)

        if i_start >= i_end:

        j = i + m
        np.add.at(G[idx], i, -x[j])
        np.add.at(G[idx], j, 1.0 - x[i])
    return G


def smooth_objective(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = manual_lags(x) * dx
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    max_s = np.max(shifted)
    logsum = np.log(np.sum(np.exp(shifted - max_s))) + max_s
    return float(cmax + logsum / beta)


def smooth_gradient(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = manual_lags(x) * dx
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    max_s = np.max(shifted)
    weights = np.exp(shifted - max_s)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.ones(len(weights)) / len(weights)
    else:
        weights /= wsum
    G = manual_grad_matrix(x)
    return dx * (G.T @ weights)


def smooth_eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def smooth_eq_jac(x):
    return np.ones(len(x))

    return z[-1]
