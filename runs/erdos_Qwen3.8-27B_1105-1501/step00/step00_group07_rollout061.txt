def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(R) * dx)

def score_and_vec(h):

    dx = 2.0 / n

def repair_sum(h, M):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = np.sum(h) - M
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = int(np.argmax(h))
            h[idx] -= min(h[idx], diff)
        else:
            idx = int(np.argmin(h))
            h[idx] += min(1.0 - h[idx], -diff)
    return h

def make_binary_from_scores(scores, n):
    """Create binary sequence from fractional scores, maintaining n/2 ones."""
    order = np.argsort(-scores)
    h = np.zeros(n)
    h[order[:n//2]] = 1.0
    return h

def resample_to_n(h, n_new):
    """Resample h to new length n_new using interpolation."""
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = repair_sum(h_new, n_new / 2.0)
    return h_new

def local_search(h, M, max_time, rng, verbose=False):
    """Local search on binary sequences via pair swaps."""
    n = len(h)
    h = np.array(h, dtype=float).copy()
    # Ensure binary
    h = make_binary_from_scores(h, n)
    M = n // 2
    
    ones_list = np.where(h == 1)[0]
    zeros_list = np.where(h == 0)[0]
    
    c = score_exact(h)
    best_c = c
    best_h = h.copy()
    
    start = time.time()
    no_improve = 0
    swap_count = 0
    
    while time.time() - start < max_time:
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        # Pick random pair
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        # Evaluate swap
        h[i] = 0.0
        h[j] = 1.0
        c_new = score_exact(h)
        
        if c_new <= best_c - 1e-12:
            # Accept
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
            # Update lists
            idx_i = np.where(ones_list == i)[0][0]
            idx_j = np.where(zeros_list == j)[0][0]
            ones_list[idx_i] = j
            zeros_list[idx_j] = i
        else:
            # Reject (revert)
            h[j] = 0.0
            h[i] = 1.0
            no_improve += 1
        
        swap_count += 1
        
        if no_improve > 50000:
            # Perturbation
            n_pert = rng.randint(5, 20)
            for _ in range(n_pert):
                i2 = ones_list[rng.randint(len(ones_list))]
                j2 = zeros_list[rng.randint(len(zeros_list))]
                h[i2] = 0.0
                h[j2] = 1.0
                idx_i = np.where(ones_list == i2)[0][0]
                idx_j = np.where(zeros_list == j2)[0][0]
                ones_list[idx_i] = j2
                zeros_list[idx_j] = i2
            
            # Re-derive best from perturbed state
            c_pert = score_exact(h)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h.copy()
            no_improve = 0
    
    return best_h, best_c, swap_count

def smooth_optimize(h_init, M, alpha, max_time, rng):
    """Smooth optimization using L-BFGS-B with penalty for sum constraint."""
    n = len(h_init)
    
    def obj(x):
        R = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
        max_R = np.max(R)
        shifted = alpha * (R - max_R)
        shifted = np.clip(shifted, -400, 0)
        logsum = max_R + np.log(np.sum(np.exp(shifted))) / alpha
        penalty = 50.0 * (np.sum(x) - M) ** 2
        return logsum + penalty
    
    bounds = [(0.0, 1.0)] * n

    try:

            obj, h_init, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-10, 'maxfun': 20000}
        )

    h_opt = np.clip(result.x, 0, 1)
    h_opt = repair_sum(h_opt, M)
    c = score_exact(h_opt)
    return h_opt, c

```python
def vectorized_best_sweep(h):
    """One best-improvement sweep using vectorized computation."""
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    R = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
    # Work with lags d = -(n-1), ..., 0, ..., (n-1)
    # R[0] corresponds to lag -(n-1), R[n-1] to lag 0, R[2n-2] to lag (n-1)
    
    # For each position p and lag d, compute S(p, d) = h[p+d] + h[p-d]
    # d ranges from -(n-1) to (n-1), indexed as 0 to 2n-2
    # p ranges from 0 to n-1
    
    # S(p, d) = h[p+d] (if 0 <= p+d < n) + h[p-d] (if 0 <= p-d < n)
    d_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For position p: h[p+d] is valid when 0 <= p+d < n, i.e., -p <= d < n-p
    # h[p-d] is valid when 0 <= p-d < n, i.e., p-n < d <= p
    
    # Build S matrix: S[p, k] where k indexes d_vals[k]
    # S.shape = (n, 2n-1)
    S = np.zeros((n, 2*n-1))
    for p in range(n):
        d = d_vals
        # h[p+d] valid where 0 <= p+d < n
        mask1 = (p + d >= 0) & (p + d < n)
        S[p, mask1] += h[p + d[mask1]]
        # h[p-d] valid where 0 <= p-d < n
        mask2 = (p - d >= 0) & (p - d < n)
        S[p, mask2] += h[p - d[mask2]]
    
    # But at lag d=0 (index n-1), R(0) = 0 always for binary, so ΔR(0) = S[i,0] - S[j,0]
    # but we know R(0) stays 0, so we can just set that column to 0
    S[:, n-1] = 0  # Lag 0 contribution is always 0 for binary
    
    # Now for each (i, j) pair:
    # ΔR(d) = S[i, d] - S[j, d]
    # R_new(d) = R[d] + S[i, d] - S[j, d]
    # new_max = max(R_new)
    
    # Get S for ones and zeros
    S_ones = S[ones]  # shape (n/2, 2n-1)
    S_zeros = S[zeros]  # shape (n/2, 2n-1)
    
    # Compute for all pairs: R_new[i,j,d] = R[d] + S_ones[i,d] - S_zeros[j,d]
    # Shape: (n/2, n/2, 2n-1)
    # This might be too large for memory for bigger n, but for n=95: 48*48*189 ≈ 434K
    # That's fine.
    
    R_new = R[None, None, :] + S_ones[:, None, :] - S_zeros[None, :, :]
    new_maxes = np.max(R_new, axis=2)  # shape (n/2, n/2)
    
    # Find the pair with minimum new_max
    flat_idx = np.argmin(new_maxes)
    i_idx = flat_idx // len(zeros)
    j_idx = flat_idx % len(zeros)
    
    best_i = ones[i_idx]
    best_j = zeros[j_idx]
    best_val = new_maxes[i_idx, j_idx]
    
    return best_i, best_j, best_val, R
```
