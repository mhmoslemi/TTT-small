    m = corr.max()
    shifted = (corr - m) / tau
    exp_shifted = np.exp(shifted)
    ss = exp_shifted.sum()
    val = m + tau * np.log(ss)
    weights = exp_shifted / ss
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if weights[i] < 1e-14:
            continue
        k = i - (n - 1)
        grad += weights[i] * corr_grad_at_lag(h, n, k)
    grad *= dx
    return val, grad

def compute_hard_max(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for i in active:
        k = i - (n - 1)
        grad += corr_grad_at_lag(h, n, k)
    grad /= len(active)
    grad *= dx
    return max_val, grad

def _opt_obj(z):
    v, _ = compute_smooth_max(z, _OPT_N, _OPT_TAU)
    return v

def _opt_obj_grad(z):
    _, g = compute_smooth_max(z, _OPT_N, _OPT_TAU)
    return g

def _eq_fun(z):
    return z.sum() - _OPT_N / 2.0

def _eq_jac(z):
    return np.ones(_OPT_N)

def _get_constraints():
    return [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

def _get_bounds(n):
    return [(0.0, 1.0)] * n

def _run_slsqp(h_curr, n, tau, maxiter, ftol):
    global _OPT_N, _OPT_TAU
    _OPT_N = n
    _OPT_TAU = tau
    cons = _get_constraints()
    bounds = _get_bounds(n)
    try:
        res = minimize(_opt_obj, h_curr, jac=_opt_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_cand = project_feasible(res.x, n)
        return h_cand
    except Exception:
        return h_curr

def _run_slsqp_hard(h_curr, n, maxiter, ftol):
    global _OPT_N
    _OPT_N = n
    dx = 2.0 / n
    def obj_func(z):
        v, _ = compute_hard_max(z, n)
        return v
    def grad_func(z):
        _, g = compute_hard_max(z, n)
        return g
    cons = _get_constraints()
    bounds = _get_bounds(n)
    try:
        res = minimize(obj_func, h_curr, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_cand = project_feasible(res.x, n)
        return h_cand
    except Exception:
        return h_curr

def _gradient_check(h, n, tau):
    dx = 2.0 / n
    val, grad = compute_smooth_max(h, n, tau)
    eps = 1e-7
    for j in range(min(n, 5)):
        h_plus = h.copy()
        h_plus[j] += eps
        h_plus = project_feasible(h_plus, n)
        v_plus, _ = compute_smooth_max(h_plus, n, tau)
        h_minus = h.copy()
        h_minus[j] = max(0.0, h_minus[j] - eps)
        h_minus = project_feasible(h_minus, n)
        v_minus, _ = compute_smooth_max(h_minus, n, tau)
        num_grad = (v_plus - v_minus) / (2.0 * eps)
        if abs(grad[j] - num_grad) > 0.1 * (abs(num_grad) + 1e-10):
            pass  # Gradient check failed for component j
    return True
