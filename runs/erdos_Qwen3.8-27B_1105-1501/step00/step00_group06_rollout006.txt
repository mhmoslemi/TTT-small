                d1 -= np.mean(d1)
                d2 -= np.mean(d2)
                d_combined = d1 + 0.5 * d2
                d_combined /= (np.linalg.norm(d_combined) + 1e-10)
                for step_sz in [0.02, 0.05, 0.10]:
                    h_try = project_box_mass(h - step_sz * d_combined)
                    sc_try = score_h(h_try)
                    if sc_try < cur_score:
                        h = h_try
                        if sc_try < best_score:
                            best_score = sc_try
                            best_h = h_try.copy()
                        improved = True
                        break

        if not improved:
            radius = max(0.06, radius * 0.92)
            if stall % 50 == 0:
                beta_lse = min(beta_lse * 1.5, 2000.0)
            if stall > 200:
                # Larger restart perturbation
                pert = rng.randn(n) * 0.08
                pert -= np.mean(pert)
                h = project_box_mass(h + pert)
                stall = 0
                radius = 0.20

    final_sc = score_h(best_h)
    if final_sc < best_score:
        best_score = final_sc
    return best_h, best_score, n
