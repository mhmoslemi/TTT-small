```python
def _smooth_max(u, n, alpha=100):
    """Smooth approximation of dx * max_k G_k."""
    h = _build_symmetric(u, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr  # unscaled
    # Log-sum-exp for numerical stability
    max_val = np.max(scaled)
    log_sum = max_val + np.log(np.sum(np.exp(alpha * (scaled - max_val))))
    return (1.0 / alpha) * log_sum

def _smooth_max_grad(u, n, alpha=100):
    """Gradient of smooth max."""
    h = _build_symmetric(u, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr
    max_val = np.max(scaled)
    weights = np.exp(alpha * (scaled - max_val))
    weights = weights / np.sum(weights)
    
    # Gradient of G_k with respect to h for each k
    # Then reduce to u
    m = (n - 1) // 2
    dx = 2.0 / n
    grad_u = np.zeros(m)
    
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        w = weights[idx]
        # Gradient of G_k w.r.t. h
        g_h = np.zeros(n)
        for j in range(n):
            if 0 <= j - k < n:
                g_h[j] += (1.0 - h[j - k])
            if 0 <= j + k < n:
                g_h[j] += (-h[j + k])
        # Reduce to u
        for j in range(m):
            grad_u[j] += w * dx * (g_h[j] - g_h[n - 1 - j])
    
    return grad_u
```
