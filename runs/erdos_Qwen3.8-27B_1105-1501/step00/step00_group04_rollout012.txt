def exact_score(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx using np.correlate."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_feasible(y):
    """Project y to [0,1]^n with sum = n/2."""
    n = len(y)
    target = n / 2.0
    h = np.clip(y, 0.0, 1.0)
    # Adjust sum to target
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum; add to elements with room
        room = 1.0 - h
        if room.sum() < diff:
            h = np.clip(y, 0.0, 1.0)  # can't fix, just clip
            return h
        # Distribute diff to elements with most room
        order = np.argsort(-room)
        for i in order:
            add = min(room[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        # Need to decrease sum
        avail = h
        if avail.sum() < -diff:
            return h
        order = np.argsort(-avail)  # reduce largest first
        for i in order:
            rem = min(avail[i], -diff)
            h[i] -= rem
            diff += rem
            if diff >= -1e-14:
                break
    # Final adjustment
    h = np.clip(h, 0.0, 1.0)
    # Exact mass fix
    total = h.sum()
    if abs(total - target) > 1e-10:
        # Small adjustment on a single element
        idx = np.argmax(h) if total > target else np.argmin(h)
        h[idx] += target - total
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def compute_lag_gradient(h, d):
    """Compute gradient of overlap at lag d (positive) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if d >= n:
        return g
    # corr_d = sum_{i=0}^{n-d-1} h[i] * (1-h[i+d])

    #                    - h[j-d] for d<=j<=n-1
    for j in range(n):
        if j <= n - d - 1:
            g[j] += (1.0 - h[j + d])
        if j >= d:
            g[j] -= h[j - d]
    return g

    n = len(h)
    g = np.zeros(n)
    for d in range(1, n):
        g += compute_lag_gradient(h, d)
    # Zero lag: sum h[i]*(1-h[i]), gradient is 1-2*h[i]
    g += (1.0 - 2.0 * h)
    return g

def active_subset(h, n_active=20):

    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    # Center is at n-1
    indices = np.argsort(-corr)[:n_active]
    # Convert correlation indices to lag values
    lags = indices - (n - 1)
    return lags

def projected_subgradient_step(h, lags, step_sizes):
    """Try a subgradient descent step toward a feasible region with lower objective."""
    n = len(h)
    g = np.zeros(n)
    # Include zero lag
    g += (1.0 - 2.0 * h)
    for d in lags:
        if d > 0:
            g += compute_lag_gradient(h, d)
        elif d < 0:

            # Actually for d < 0, use |d| and the formula is the "other direction"
            ad = abs(d)

            # = sum_{i=0}^{n-ad-1} h[i+ad] * (1-h[i])
            # gradient: (1-h[j-ad]) for ad<=j<=n-1, -h[j+ad] for 0<=j<=n-ad-1

- If i appears as k+d (i.e., i≥d and i-d ≤ n-d-1): we gain h[i-d]*(1-h[i]) = h[i-d]*1 = h[i-d]
  Wait, we're CHANGING h[i] from 1 to 0, so the term h[k+d]*(1-h[k]) where k+d=i changes.
  Actually, let me think of it as: the old value at position i was 1, new value is 0.
  And old value at position j was 0, new value is 1.

For corr_d = Σ_{k=0}^{n-d-1} h[k]*(1-h[k+d]):
- Position i contributes to this sum in two ways:
  a) As the left factor h[k] when k=i: contributes h[i]*(1-h[i+d]) = 1*(1-h[i+d]) if i ≤ n-d-1
  b) As the right factor h[k+d] when k+d=i, i.e., k=i-d: contributes h[i-d]*(1-h[i]) = h[i-d]*0 = 0 if i ≥ d

After the swap (h[i] becomes 0):
  a) New contribution as left: 0*(1-h[i+d]) = 0
  b) New contribution as right: h[i-d]*(1-0) = h[i-d] if i ≥ d

So the change from position i in corr_d:
Δ_i = (new - old) = [0 + h[i-d]*1_{i≥d}] - [1*(1-h[i+d])*1_{i≤n-d-1} + 0]
= h[i-d]*1_{i≥d} - (1-h[i+d])*1_{i≤n-d-1}

Similarly for position j (h[j] goes from 0 to 1):
  a) As left factor when k=j: old: 0*(1-h[j+d]) = 0, new: 1*(1-h[j+d])
  b) As right factor when k=j-d: old: h[j-d]*(1-0) = h[j-d], new: h[j-d]*(1-1) = 0

Δ_j = (new - old) = [(1-h[j+d])*1_{j≤n-d-1} + 0] - [0 + h[j-d]*1_{j≥d}]
= (1-h[j+d])*1_{j≤n-d-1} - h[j-d]*1_{j≥d}

Total change in corr_d: Δ_i + Δ_j

For corr_{-d} (which is corr_full[n-1-d] = Σ_{k=d}^{n-1} h[k]*(1-h[k-d])):
This is the same formula with h and (1-h) swapped in a sense. Actually:
corr_{-d} = Σ_{k=0}^{n-d-1} h[k+d]*(1-h[k])

The analysis is similar but with the roles of left and right reversed.

This is getting complex to implement directly. Let me just use the "evaluate change" approach: compute the new score after the swap and accept if it's better.

For efficiency, I'll precompute the correlation and update it incrementally. But for simplicity and correctness, I'll just recompute the full correlation after each candidate swap. For n=70, this is fast (np.correlate is O(n²) ≈ 5000 operations).

```python
import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_feasible(y):
    n = len(y)
    target = n / 2.0
    h = np.clip(y.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            a = min(room[i], diff)
            h[int(i)] += a
            diff -= a
            if diff <= 1e-14:
                break
    else:
        order = np.argsort(-h)
        for i in order:
            a = min(h[int(i)], -diff)
            h[int(i)] -= a
            diff += a
            if diff >= -1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    # Tiny mass fix
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += residual
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def binarize(h):
    """Convert to binary with exact sum = n/2."""
    n = len(h)
    target = n // 2  # n is even
    # Sort by value descending, take top n/2
    order = np.argsort(-h)
    x = np.zeros(n)
    x[order[:target]] = 1.0
    return x

def try_pair_swap(h, i, j):
    """Try swapping h[i] and h[j] (or transferring mass). Return new h if it improves."""
    n = len(h)
    h2 = h.copy()
    h2[i], h2[j] = h[j], h[i]
    h2 = repair_feasible(h2)
    return h2

def pair_swap_optimize(h, max_iters=10000, time_limit=30.0):
    """Iteratively try all pair swaps and accept the best improving one."""
    n = len(h)
    f = exact_score(h)
    start = time.time()
    improved = True
    while improved and elapsed < time_limit:
        improved = False
        best_f = f
        best_h = h
        # Try all pairs
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in ones:
            for j in zeros:
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                f2 = exact_score(h2)
                if f2 < best_f - 1e-12:
                    best_f = f2
                    best_h = h2
            if elapsed > time_limit:
                return best_h, best_f
        if best_f < f - 1e-12:
            h = best_h
            f = best_f
            improved = True
        else:
            improved = False
    return h, f
```

Wait, the all-pairs approach is O(n²) swaps × O(n²) evaluation = O(n⁴) per iteration. For n=70, that's 70⁴ ≈ 24 million operations per iteration, which might be slow. Let me think of a faster way.

Actually, np.correlate for n=70 gives a vector of length 139, and the computation is O(n²) ≈ 5000 multiply-adds. So each evaluation is about 5000 operations. And there are 35*35 = 1225 possible swaps (binary case). So per full sweep: 1225 * 5000 = 6.1 million operations. That's fast, maybe 10-50ms in numpy. So we can do many iterations.

Actually wait, for binary vectors, a "swap" is not swapping h[i] and h[j] (since they're 1 and 0, swapping gives 0 and 1, same as the transfer I described). So the swap is: set h[i]=0, h[j]=1.

Let me also think about a smarter approach. Instead of trying all pairs, I can compute the gradient of the objective with respect to the swap and use that to prioritize. But for simplicity, let me just do all pairs since it's fast enough.

Let me also think about the continuous optimization. The subgradient approach:
1. Compute all lag values: corr = np.correlate(h, 1-h, mode="full")
2. Find active lags: those where corr[k] is close to max(corr)
3. Compute gradient: sum of gradients over active lags
4. Step: h_new = h - step * gradient
5. Project: h_new = repair_feasible(h_new)
6. Accept if score improved

The gradient of corr[k] (where k corresponds to lag d = k - (n-1)):
For d > 0: ∂corr_d/∂h[j] = (1-h[j+d])·[j≤n-d-1] - h[j-d]·[j≥d]
For d < 0 (let a = -d > 0): ∂corr_{-a}/∂h[j] = (1-h[j-a])·[j≥a] - h[j+a]·[j≤n-a-1]
For d = 0: ∂corr_0/∂h[j] = 1 - 2h[j]

Let me verify: corr_0 = Σ h[j](1-h[j]), so ∂/∂h[j] = 1-2h[j]. ✓

For d > 0: corr_d = Σ_{i=0}^{n-d-1} h[i](1-h[i+d])
∂/∂h[j]: term where j=i (j in [0,n-d-1]): (1-h[j+d])
term where j=i+d (j in [d,n-1]): -h[j-d]
So ∂corr_d/∂h[j] = (1-h[j+d])·[0≤j≤n-d-1] + (-h[j-d])·[d≤j≤n-1] ✓

For d < 0, let a = -d: corr_{-a} = Σ_{i=a}^{n-1} h[i](1-h[i-a]) = Σ_{i=0}^{n-a-1} h[i+a](1-h[i])
∂/∂h[j]: term where j=i+a (j in [a,n-1]): (1-h[j-a])
term where j=i (j in [0,n-a-1]): -h[j+a]
So ∂corr_{-a}/∂h[j] = (1-h[j-a])·[a≤j≤n-1] + (-h[j+a])·[0≤j≤n-a-1] ✓

def gradient_at_lag(h, d):
    """Gradient of the overlap at lag d (d>0: corr_full[n-1+d], d<0: corr_full[n-1+d], d=0: center)."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        return 1.0 - 2.0 * h
    if d > 0:
        # corr_d: positions j in [0, n-d-1] get +（1-h[j+d]）
        g[:n-d] += (1.0 - h[d:])
        # positions j in [d, n-1] get -h[j-d]
        g[d:] -= h[:n-d]
    else:
        a = -d
        # corr_{-a}: positions j in [a, n-1] get +（1-h[j-a]）
        g[a:] += (1.0 - h[:n-a])
        # positions j in [0, n-a-1] get -h[j+a]
        g[:n-a] -= h[a:]
    return g
```

```python
import numpy as np
import time
import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_feasible(y):
    n = len(y)
    target = n / 2.0
    h = np.clip(y.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            a = min(room[int(i)], diff)
            h[int(i)] += a
            diff -= a
            if diff <= 1e-14:
                break
    else:
        order = np.argsort(-h)
        for i in order:
            a = min(h[int(i)], -diff)
            h[int(i)] -= a
            diff += a
            if diff >= -1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def binarize(h):
    n = len(h)
    target = n // 2
    order = np.argsort(-h)
    x = np.zeros(n)
    x[order[:target]] = 1.0
    return x

def gradient_at_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        return 1.0 - 2.0 * h
    if d > 0:
        g[:n - d] += (1.0 - h[d:])
        g[d:] -= h[:n - d]
    else:
        a = -d
        g[a:] += (1.0 - h[:n - a])
        g[:n - a] -= h[a:]
    return g

def compute_all_lags(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr  # length 2n-1, center at n-1

def active_lags(h, tol=1e-4):
    corr = compute_all_lags(h)
    mx = np.max(corr)
    # Lags where corr is within tol of max
    active = np.where(corr >= mx - tol)[0]
    # Convert to lag values (center is at n-1)
    n = len(h)
    return active - (n - 1), mx

def subgradient_step(h, step_sizes, n_active=None, tol=1e-4):
    """One subgradient descent step."""
    corr = compute_all_lags(h)
    mx = np.max(corr)
    n = len(h)
    center = n - 1
    if n_active is None:
        n_active = 10
    # Get top n_active lags
    top_idx = np.argsort(-corr)
    active_indices = top_idx[:n_active]
    # Compute subgradient (sum of gradients at active lags)
    g = np.zeros(n)
    for idx in active_indices:
        d = int(idx) - center
        g += gradient_at_lag(h, d)
    # Scale gradient
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
        return h, mx
    g = g / g_norm
    # Try different step sizes
    best_h = h.copy()
    best_f = mx
    for step in step_sizes:
        candidate = h - step * g
        candidate = repair_feasible(candidate)
        f = float(np.max(compute_all_lags(candidate)))
        if f < best_f:
            best_f = f
            best_h = candidate
    return best_h, best_f

def pair_swap_optimize_binary(h, time_limit=30.0):
    """Optimize a binary vector using pair swaps."""
    n = len(h)
    f = exact_score(h)
    start = time.time()
    for iteration in range(100000):
        if time.time() - start > time_limit:
            break
        improved = False
        best_f = f
        best_i, best_j = -1, -1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in ones:
            for j in zeros:
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                f2 = exact_score(h2)
                if f2 < best_f - 1e-12:
                    best_f = f2
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            f = best_f
            improved = True
        if not improved:
            break
    return h, f

def interpolation_to_len(h_old, n_new):
    """Interpolate h_old to length n_new, then repair."""
    n_old = len(h_old)
    # Old midpoints
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    # New midpoints
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    # Interpolate
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return repair_feasible(h_new)

def milp_try(n, target_t, time_limit=30.0):
    """Try to solve MILP for binary vector of length n with max overlap ≤ target_t."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    constraints = [cp.sum(x) == n // 2, t >= 0]
    
    # For each lag d, add constraints pos_d <= t and neg_d <= t
    # pos_d = sum_{i=0}^{n-d-1} x[i] - x[i]*x[i+d]
    # We need auxiliary variables for products
    # Number of products: sum_{d=1}^{n-1} (n-d) = n*(n-1)/2
    # For n=70: 2415, for n=80: 3160
    
    if n > 90:  # Too large for MILP
        return None, None
    
    # Create product variables
    prod_vars = {}
    for d in range(1, n):
        for i in range(n - d):
            prod_vars[(d, i)] = cp.Variable()
            p = prod_vars[(d, i)]
            constraints.append(p <= x[i])
            constraints.append(p <= x[i + d])
            constraints.append(p >= x[i] + x[i + d] - 1)
            constraints.append(p >= 0)
    
    for d in range(1, n):
        # pos_d = sum x[i] for i in [0, n-d-1] - sum prod[d,i]
        pos_d = cp.sum(x[:n-d]) - cp.sum([prod_vars[(d, i)] for i in range(n - d)])
        # neg_d = sum x[i+d] for i in [0, n-d-1] - sum prod[d,i]
        neg_d = cp.sum(x[d:]) - cp.sum([prod_vars[(d, i)] for i in range(n - d)])
        constraints.append(pos_d <= t)
        constraints.append(neg_d <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    start = time.time()
    try:
        # Try different solvers
        for solver in [cp.CBC, cp.GLPK_MI, cp.SCIP]:
            try:
                prob.solve(solver=solver, time_limit=time_limit)
                break
            except Exception:
                continue
    except Exception:
        return None, None
    
    if x.value is not None:
        x_val = np.round(x.value)
        x_val = np.clip(x_val, 0, 1)
        # Check sum
        if abs(x_val.sum() - n // 2) > 0.1:
            # Fix sum
            order = np.argsort(-x_val)
            x_val = np.zeros(n)
            x_val[order[:n//2]] = 1.0
        f = exact_score(x_val)
        return x_val, f
    
    return None, None
