```python
def compute_gradients_numerical(h, n):
    """Compute gradient of each corr[k] with respect to h using central differences.
    Returns array of shape (2n-1, n)."""
    m = 2*n - 1
    grads = np.zeros((m, n))
    eps = 1e-7
    for j in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[j] += eps
        h_minus[j] -= eps
        corr_plus = np.correlate(h_plus, 1-h_plus, 'full')
        corr_minus = np.correlate(h_minus, 1-h_minus, 'full')
        grads[:, j] = (corr_plus - corr_minus) / (2*eps)
    return grads
```

```python
                    if compute_c5(sm, n) < compute_c5(h, n):
                        h = sm
                no_improve = 0
                step_scale = 1.0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=np.float64)
    n = len(h_init)
    target_sum = n / 2.0

    # Ensure feasibility
    h_init = project_feasible(h_init, n, target_sum)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Local optimize from initial (use 40% of time)
    t1 = (budget_s - 20) * 0.4
    h1, c5_1 = local_optimize(h_init, n, target_sum, 200000, t1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    elapsed = time.time() - start_time
    if elapsed > budget_s - 30:
        return best_h, best_c5, n

    # Phase 2: Try perturbed restarts (use 40% of remaining time)
    remaining = budget_s - 20 - elapsed
    n_trials = 5
    trial_time = remaining * 0.4 / n_trials
    for trial in range(n_trials):
        if time.time() - start_time > budget_s - 30:
            break
        scale = 0.08 + 0.04 * trial
        pert = rng.normal(0, scale, size=n)
        pert -= pert.mean()
        h_trial = project_feasible(best_h + pert, n, target_sum)
        actual_time = min(trial_time, budget_s - 30 - (time.time() - start_time))
        if actual_time < 5:
            break
        h_next, c5_next = local_optimize(h_trial, n, target_sum, 50000, actual_time, rng)
        if c5_next < best_c5:
            best_c5 = c5_next
            best_h = h_next.copy()

    # Phase 3: Fine-tune from best (remaining time)
    remaining = budget_s - 15 - (time.time() - start_time)
    if remaining > 5:
        h_final, c5_final = local_optimize(best_h, n, target_sum, 200000, remaining, rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final verification
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
```
