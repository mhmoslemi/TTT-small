    n = len(h)
    target = n / 2.0

    step = start_step
    stagnation = 0
    
    while time.perf_counter() < deadline - 1.0:
        # Generate random direction (zero-sum)
        u = rng.standard_normal(n)
        u -= u.mean()  # make zero-sum

        if u_norm < 1e-10:
            continue
        u = u / u_norm  # unit direction

        found = False
        for trial in range(5):
            s = step * (0.5 ** trial)  # geometric decay
            if s < min_step:

    dst = np.linspace(0, 1, new_n, endpoint=False) + 0.5 / new_n
    new_h = np.interp(dst, src, h)
    return _project(new_h, new_n / 2.0)

def _make_balanced(n):
    h = np.zeros(n)
    for i in range(n):
        x = i
        c = 0
        while x:
            c += x & 1
            x >>= 1
        h[i] = c % 2
    return _project(h, n / 2.0)

def _snap_to_01(h):
    n = len(h)
    h01 = (h > 0.5).astype(np.float64)
    target = n // 2
    diff = target - int(h01.sum())
    if diff > 0:
        # Need more ones
        thresh = np.sort(h)[::-1]
        # Set the next |diff| largest fractional values to 1
        scores = h[h01 == 0]
        idx = np.where(h01 == 0)[0]
        order = np.argsort(-scores)
        for i in range(min(diff, len(order))):
            h01[idx[order[i]]] = 1
    elif diff < 0:
        # Need fewer ones
        scores = h[h01 == 1]
        idx = np.where(h01 == 1)[0]
        order = np.argsort(scores)
        for i in range(min(-diff, len(order))):
            h01[idx[order[i]]] = 0
    return h01

def _pair_sweep(h, deadline, max_passes=5):
    n = len(h)
    best_score = _score(h)
    eps = 1e-11
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    n_s = 12 if n <= 80 else 8
    
    for _pass in range(max_passes):
        if time.perf_counter() > deadline - 2.0:
            break
        any_improved = False
        for p in range(n):
            if time.perf_counter() > deadline - 2.0:
                return h, best_score
            hp = h[p]
            for q in range(p + 1, n):
                hq = h[q]
                lo = max(-hp, hq - 1.0) + 1e-10
                hi = min(1.0 - hp, hq) - 1e-10
                if hi - lo < 1e-8:
                    continue
                deltas = np.linspace(lo, hi, n_s)
