            movable = np.ones(n, dtype=bool)
            cnt = n
        result[movable] += res / cnt
        result = np.clip(result, 0.0, 1.0)
    
    # Final exact fix
    res = target - result.sum()
    if abs(res) > 1e-14:
        idx = np.argsort(np.abs(result - 0.5))
        for i in idx:
            if res > 0 and result[i] < 1.0:
                add = min(res, 1.0 - result[i])
                result[i] += add
                res -= add
            elif res < 0 and result[i] > 0:
                rem = min(-res, result[i])
                result[i] -= rem
                res += rem
            if abs(res) < 1e-14:
                break
    return result

def compute_A(h):
    """Compute all 2n-1 overlap values"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grads(h):
    """Compute gradients of all 2n-1 overlaps w.r.t. h"""
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    
    for s in range(-(n-1), n):
        i_start = max(0, -s)
        i_end = min(n, n - s)
        
        # First factor: d/dh[j] of h[j]*(1-h[j+s]) = (1-h[j+s]) for j in [i_start, i_end)
        G[n-1+s, i_start:i_end] += (1.0 - h[i_start:i_end + s])
        
        # Second factor: d/dh[j] of h[j-s]*(1-h[j]) = -h[j-s] for j where j-s in [i_start, i_end)
        # j in [i_start+s, i_end+s), clipped to [0, n)
        j_start = max(0, i_start + s)
        j_end = min(n, i_end + s)
        if j_start < j_end:
            G[n-1+s, j_start:j_end] -= h[j_start - s : j_end - s]
    
    return G

def feasible_span(h, d):
    """Find [lo, hi] such that h + lam*d in [0,1]^n for lam in [lo,hi]"""
    eps = 1e-14
    lo = -np.inf
    hi = np.inf
    
    pos = d > eps
    if np.any(pos):
        lo = max(lo, float(np.max(-h[pos] / d[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))

        lo = max(lo, float(np.max((h[neg] - 1.0) / d[neg])))
        hi = min(hi, float(np.min(-h[neg] / d[neg])))
    
    return lo, hi

def lp_minimax_direction(h, radius):
    """LP to find direction reducing max overlap (linearized)"""
    n = len(h)
    A = compute_A(h)
    G = compute_grads(h)
    n_lags = len(A)
    
    # Minimize t
    # s.t. A_k + G_k^T (x - h) <= t  =>  G_k @ x - t <= G_k @ h - A_k
    #      sum(x) = n/2
    #      h[i] - r <= x[i] <= h[i] + r
    #      0 <= x[i] <= 1
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - A
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lb = max(0.0, h[i] - radius)
        ub = min(1.0, h[i] + radius)
        bounds.append((lb, ub))
    bounds.append((None, None))  # t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[:n] - h
    d -= d.mean()  # ensure zero sum (should already be close)
    return d

This is much faster! Let me implement this.

For A_k(h + λd) = sum_i (h_i + λd_i)(1 - h_{i+k} - λd_{i+k}):
= sum_i h_i(1-h_{i+k}) + λ sum_i [d_i(1-h_{i+k}) - h_i d_{i+k}] + λ² sum_i (-d_i d_{i+k})

So:
c_k = A_k(h) (the constant term)
b_k = sum_i [d_i(1-h_{i+k}) - h_i d_{i+k}] (the linear coefficient)
a_k = -sum_i d_i d_{i+k} (the quadratic coefficient)

Then A_k(λ) = c_k + b_k λ + a_k λ²

And J(λ) = max_k A_k(λ) * (2/n)

To find the minimum of J(λ), I need to minimize max_k (a_k λ² + b_k λ + c_k) over feasible λ.

This is a 1D problem. I can:
1. Find all pairwise intersection points of the quadratics (where a_i λ² + b_i λ + c_i = a_j λ² + b_j λ + c_j)
2. The minimum of the max must occur at one of these intersection points or at a boundary
3. Evaluate J at all candidate points and pick the minimum

def line_search_fast(h, d):
    """Fast line search using precomputed quadratic coefficients"""
    n = len(h)
    lo, hi = feasible_span(h, d)
    if hi <= lo:
        return h, score_h(h)
    lo += 1e-12
    hi -= 1e-12
    if hi <= lo:
        return h, score_h(h)
    
    # Precompute quadratic coefficients for each lag
    # A_k(λ) = c_k + b_k*λ + a_k*λ²
    n_lags = 2 * n - 1
    A_const = np.zeros(n_lags)
    A_lin = np.zeros(n_lags)
    A_quad = np.zeros(n_lags)
    
    for s in range(-(n-1), n):
        i_start = max(0, -s)
        i_end = min(n, n - s)
        idx = np.arange(i_start, i_end)
        shifted = idx + s
        
        A_const[n-1+s] = np.sum(h[idx] * (1 - h[shifted]))
        A_lin[n-1+s] = np.sum(d[idx] * (1 - h[shifted]) - h[idx] * d[shifted])
        A_quad[n-1+s] = -np.sum(d[idx] * d[shifted])
    
    # Evaluate J(λ) = max_k (A_quad[k]*λ² + A_lin[k]*λ + A_const[k]) * (2/n)
    dx = 2.0 / n
    
    def eval_J(lam):
        vals = A_quad * lam * lam + A_lin * lam + A_const
        return float(np.max(vals) * dx)
    
    # Grid search
    n_iter = 1000
    lams = np.linspace(lo, hi, n_iter)
    vals = A_quad[:, None] * lams[None, :]**2 + A_lin[:, None] * lams[None, :] + A_const[:, None]
    J_vals = np.max(vals, axis=0) * dx
    
    best_idx = np.argmin(J_vals)
    best_lam = lams[best_idx]
    best_J = J_vals[best_idx]
    
    # Refine around best
    span = (hi - lo) / n_iter
    for _ in range(3):
        lams_fine = np.linspace(best_lam - span, best_lam + span, 200)
        lams_fine = lams_fine[(lams_fine >= lo) & (lams_fine <= hi)]
        if len(lams_fine) < 3:
            break
        vals_f = A_quad[:, None] * lams_fine[None, :]**2 + A_lin[:, None] * lams_fine[None, :] + A_const[:, None]
        J_f = np.max(vals_f, axis=0) * dx
        idx_f = np.argmin(J_f)
        best_lam = lams_fine[idx_f]
        best_J = J_f[idx_f]
        span = (lams_fine[1] - lams_fine[0]) if len(lams_fine) > 1 else span * 0.1
    
    h_best = h + best_lam * d
    # Verify with exact computation
    actual_J = score_h(h_best)
    
    return h_best, actual_J

    A_eq[0, :n] = 1.0

        return None

def score_h(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)


def project_feasible(v):
    n = len(v)
    target = n * 0.5
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    lo = -np.max(v) - 0.01
    hi = 1.0 - np.min(v) + 0.01
    for _ in range(80):
        mid = (lo + hi) * 0.5

            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(40):
        res = target - result.sum()
        if abs(res) < 1e-13:
            break
        movable = (result > 1e-13) & (result < 1.0 - 1e-13)
        cnt = max(1, int(movable.sum()))
        result[movable] = np.clip(result[movable] + res / cnt, 0.0, 1.0)
    res = target - result.sum()
    if abs(res) > 1e-15:
        idx = np.argsort(np.abs(result - 0.5))

                break
            if res > 0:

                result[i] += add
                res -= add
            else:
                rem = min(-res, result[i])
                result[i] -= rem
                res += rem
    return result


def feasible_span(h, d):
    eps = 1e-15
    lo = -1e10
    hi = 1e10

        lo = max(lo, float(np.max((h[neg] - 1.0) / d[neg])))

    hi -= 1e-12

        return h.copy(), score_h(h)

    for s in range(-(n - 1), n):

        sh = idx + s

        A_l[n - 1 + s] = np.dot(d[idx], 1.0 - h[sh]) - np.dot(h[idx], d[sh])
        A_q[n - 1 + s] = -np.dot(d[idx], d[sh])

    dx = 2.0 / n
    lams = np.linspace(lo, hi, 1000)
    vals = lams[:, None] ** 2 * A_q[None, :] + lams[:, None] * A_l[None, :] + A_c[None, :]
