    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        step = diff / mask.sum()
        if diff > 0:
            max_s = (1.0 - h[mask]).min()
            step = min(step, max_s * 0.999)
        else:
            min_s = h[mask].min()
            step = max(step, -min_s * 0.999)
        if abs(step) < 1e-16:
            break
        h[mask] += step
        h = np.clip(h, 0, 1)
    return h

def try_update(h):
    n = len(h)
    dx = 2.0 / n
    h = fix_sum(h, n / 2.0)
    c5 = compute_c5(h, dx)
    if c5 < _g_best[1]:
        _g_best[1] = c5
        _g_best[0] = h.copy()
    return c5

def _smooth_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full') * _g_dx
    corr = np.maximum(corr, 0.0)
    return float(np.mean(corr ** _g_p))

def _smooth_grad(h):
    n = len(h)
    omh = 1.0 - h
    corr_raw = np.correlate(h, omh, mode='full')
    corr = corr_raw * _g_dx
    m = 2 * n - 1
    weights = _g_p * np.power(np.maximum(corr, 1e-30), _g_p - 1) / m
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-25:
            continue
        if lag >= 0:
            nv = n - lag
            grad[lag:] += w * omh[:nv]
            grad[:nv] -= w * h[lag:]
        else:
            nl = -lag
            nv = n - nl
            grad[:nv] += w * omh[nl:]
            grad[nl:] -= w * h[:nv]
    return grad * _g_dx

def _eq_fun(h):
    return float(h.sum() - _g_target)

def _eq_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, n, t0, budget_s, maxiter=1000):
    global _g_dx, _g_p, _g_target
    _g_dx = 2.0 / n
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

    best_local = h0.copy()
    best_local_c5 = compute_c5(h0, _g_dx)

    for p in [4, 8, 16, 32, 64, 128, 256]:
        if time.time() - t0 > budget_s - 45:
            break
        _g_p = p
        res = minimize(_smooth_obj, best_local, jac=_smooth_grad,
                      method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = fix_sum(res.x, n / 2.0)
        c5 = compute_c5(h_opt, _g_dx)
        if c5 < best_local_c5 - 1e-14:
            best_local_c5 = c5
            best_local = h_opt.copy()
    return best_local, best_local_c5

def coordinate_refine(h, t0, budget_s, max_rounds=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = fix_sum(h, target)
    best_c5 = compute_c5(h, dx)

    for rd in range(max_rounds):
        if time.time() - t0 > budget_s - 10:
            break
        improved = False
        deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
        for j in range(n):
            if time.time() - t0 > budget_s - 10:
                break
            for d in deltas:
                h_new = h.copy()
                new_val = h[j] + d
                if new_val < 0 or new_val > 1:
                    continue
                h_new[j] = new_val
                h_new = fix_sum(h_new, target)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return h

def project_to_01(h, n, target_ones):
    """Try to project h to a 0-1 valued solution"""
    if n % 2 != 0:
        return h.copy()
    target_ones = n // 2
    # Threshold at the point that gives exactly target_ones values at 1
    sorted_idx = np.argsort(h)[::-1]
    h01 = np.zeros(n)
    h01[sorted_idx[:target_ones]] = 1.0
    return h01

def run(seed=42, budget_s=1100, **kwargs):
    global _g_best, _g_dx, _g_p, _g_target
    np.random.seed(seed)
    t0 = time.time()
    _g_best = [None, float('inf')]

    # Get initial values
    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        try_update(h_init)

    # Phase 1: Optimize from initial construction
    if h_init is not None:
        n = len(h_init)
        h1, _ = slsqp_optimize(h_init, n, t0, budget_s, maxiter=3000)
        try_update(h1)

    # Phase 2: Coordinate refine the best
    if _g_best[0] is not None and time.time() - t0 < budget_s - 60:
        hr = coordinate_refine(_g_best[0], t0, budget_s, max_rounds=30)
        try_update(hr)

    # Phase 3: Try other n values
    n_trials = [40, 44, 46, 50, 60, 80, 100, 120]
    for n in n_trials:
        if time.time() - t0 > budget_s - 80:
            break
        dx = 2.0 / n
        target = n / 2.0

        if _g_best[0] is not None:
            x_old = np.linspace(0, 2, len(_g_best[0]), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, _g_best[0])
        elif h_init is not None:
            x_old = np.linspace(0, 2, len(h_init), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h0 = np.interp(x_new, x_old, h_init)
        else:
            h0 = np.random.rand(n)
        h0 = np.clip(h0, 0, 1)
        h0 = fix_sum(h0, target)

        h_opt, _ = slsqp_optimize(h0, n, t0, budget_s, maxiter=800)
        try_update(h_opt)

        # Try 0-1 projection for even n
        if n % 2 == 0:
            h01 = project_to_01(_g_best[0] if _g_best[0] is not None and len(_g_best[0]) == n else h_opt, n, n // 2)
            try_update(h01)

    # Phase 4: More aggressive optimization on best n
    if _g_best[0] is not None and time.time() - t0 < budget_s - 40:
        n = len(_g_best[0])
        h_opt, _ = slsqp_optimize(_g_best[0], n, t0, budget_s, maxiter=5000)
        try_update(h_opt)

    # Phase 5: Coordinate refinement
    if _g_best[0] is not None and time.time() - t0 < budget_s - 15:
        hr = coordinate_refine(_g_best[0], t0, budget_s, max_rounds=100)
        try_update(hr)

    # Phase 6: Try 0-1 projection if best is even
    if _g_best[0] is not None and len(_g_best[0]) % 2 == 0:
        h01 = project_to_01(_g_best[0], len(_g_best[0]), len(_g_best[0]) // 2)
        try_update(h01)

    return _g_best[0], _g_best[1], len(_g_best[0])
