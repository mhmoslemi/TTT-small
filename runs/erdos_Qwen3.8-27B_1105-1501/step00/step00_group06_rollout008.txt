def project_box_mass(v, n=None):
    """Project v onto {0<=x<=1, sum(x)=n/2}"""
    if n is None:
        n = len(v)
    target = n / 2.0
    v = np.asarray(v, dtype=float)
    # Clip to [0,1] first
    v = np.clip(v, 0.0, 1.0)
    # Adjust to get correct sum

    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    # Bisection
    lo, hi = -2.0, 2.0

def get_gradients(h):
    """Compute gradients of all correlation values with respect to h.
    Returns array G of shape (2n-1, n) where G[k] is gradient of correlation at index k.
    """
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    # For each lag k (index in full correlation output goes from 0 to 2n-2)
    # lag = idx - (n-1)
    for idx in range(2 * n - 1):
        lag = idx - (n - 1)
        if lag == 0:
            G[idx] = 1.0 - 2.0 * h
        elif lag > 0:
            # A = sum_{i=0}^{n-1-lag} h[i] * (1 - h[i+lag])
            # dA/dh[j] = (1-h[j+lag]) for j <= n-1-lag (first factor)
            #          - h[j-lag] for j >= lag (second factor)
            m = n - lag
            G[idx, :m] += (1.0 - h[lag:])
            G[idx, lag:] -= h[:m]
        else:
            # lag < 0, let t = -lag
            t = -lag
            # A = sum_{i=t}^{n-1} h[i] * (1 - h[i-t])
            # = sum_{i=0}^{n-1-t} h[i+t] * (1 - h[i])
            # dA/dh[j] = (1-h[j-t]) for j >= t (first factor)
            #          - h[j+t] for j <= n-1-t (second factor)
            m = n - t
            G[idx, t:] += (1.0 - h[:m])
            G[idx, :m] -= h[t:]
    return G

def solve_lp_direction(h_c, radius=0.1):
    """Solve LP to find a direction that reduces linearized max correlation."""
    n = len(h_c)
    G = get_gradients(h_c)
    corr = np.correlate(h_c, 1.0 - h_c, mode="full")
    
    # Linearize: corr[idx] + G[idx] · (h - h_c) <= t
    # => G[idx] · h <= t - corr[idx] + G[idx] · h_c
    # => G[idx] · h - t <= -corr[idx] + G[idx] · h_c
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G[idx] · h - t <= b[idx]
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -corr + G @ h_c
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h_i <= 1, t free (or bounded)
    bounds = [(0, 1)] * n + [(-1e6, 1e6)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    h_new = res.x[:n]
    d = h_new - h_c
    # Subtract mean to ensure zero-sum direction (for line search)
    d = d - np.mean(d)
    return d

        idx2 = int(rng.integers(0, n))
        if idx2 == idx:
            idx2 = (idx2 + 1) % n
        # h[idx] = v, h[idx2] = h[idx2] + (h_save - v)
        # Need 0 <= v <= 1 and 0 <= h[idx2] + h_save - v <= 1
        lo = max(0.0, h[idx2] + h_save - 1.0)
        hi = min(1.0, h[idx2] + h_save)
        if lo >= hi:
            continue
        # Evaluate at several points
        grid = np.linspace(lo, hi, 30)
        best_v = h_save
        best_sv = s
        for v in grid:
            h[idx] = v
            h[idx2] = h[idx2] + h_save - v
            h[idx2] = np.clip(h[idx2], 0.0, 1.0)
            sv = score_h(h)
            if sv < best_sv:
                best_sv = sv
                best_v = v
        h[idx] = best_v
        h[idx2] = np.clip(h[idx2] + h_save - best_v, 0.0, 1.0)
        s = best_sv
        if abs(h_save - best_v) < tol:
            pass
    return h, s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 20

    try:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_cur = None

    if h_cur is None or len(h_cur) == 0:
        h_cur = np.full(70, 0.5)

    h_cur = project_box_mass(h_cur)
    best_h = h_cur.copy()
    best_s = score_h(best_h)
    best_n = len(best_h)

    base_n = best_n
    n_opts = []
    for dlt in [0, -5, -3, 3, 5, -8, 8, -12, 12, -15, 15, 20, -20]:
        alt = base_n + dlt
        if 30 <= alt <= 200:
            n_opts.append(alt)
    n_opts = sorted(set(n_opts))

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        h_c = best_h.copy()
        n_c = len(h_c)
        s_c = best_s

        # LP direction (most important)
        if time.time() < deadline - 10:
            d = solve_lp_direction(h_c)
            if d is not None and np.linalg.norm(d) > 1e-10:
                h_new, s_new = line_search(h_c, d, s_c)
                if s_new < best_s:
                    best_h, best_s, best_n = h_new, s_new, len(best_h)

        # Second LP with perturbed center
        if time.time() < deadline - 10:
            h_pert = h_c + random_dir(rng, n_c, 0.02)
            h_pert = project_box_mass(h_pert)
            d = solve_lp_direction(h_pert)
            if d is not None and np.linalg.norm(d) > 1e-10:
                h_new, s_new = line_search(h_c, d, s_c)
                if s_new < best_s:
                    best_h, best_s, best_n = h_new, s_new, len(best_h)

        # Random directions
        for sc in [0.3, 0.15, 0.08]:
            if time.time() >= deadline:
                break
            for _ in range(4):
                if time.time() >= deadline:
                    break
                d = random_dir(rng, n_c, sc)
                if d is None:
                    continue
                h_new, s_new = line_search(h_c, d, s_c)
                if s_new < best_s:
                    best_h, best_s, best_n = h_new, s_new, len(best_h)
                    h_c, s_c = best_h.copy(), best_s

        # Harmonic directions
        for mode in [1, 2, 3, 4, 5, 7, 11]:
            if time.time() >= deadline:
                break
            d = harmonic_dir(n_c, mode, rng)
            if d is None:
                continue
            h_new, s_new = line_search(h_c, d, s_c)
            if s_new < best_s:
                best_h, best_s, best_n = h_new, s_new, len(best_h)
                h_c, s_c = best_h.copy(), best_s

        # Two-coordinate transfers
        for _ in range(20):
            if time.time() >= deadline:
                break
            d = two_coord_dir(n_c, rng, 0.12)
            h_new, s_new = line_search(h_c, d, s_c)
            if s_new < best_s:
                best_h, best_s, best_n = h_new, s_new, len(best_h)
                h_c, s_c = best_h.copy(), best_s

        # Coordinate descent refinement
        if iteration % 3 == 0 and time.time() < deadline - 30:
            h_c, s_c = coordinate_descent(h_c.copy(), s_c, rng, n_steps=50)
            if s_c < best_s:
                best_h, best_s, best_n = h_c.copy(), s_c, len(h_c)

        # Multi-resolution exploration
        if iteration % 8 == 0 and time.time() < deadline - 90:
            for alt_n in n_opts:
                if alt_n == best_n or time.time() >= deadline - 30:
                    continue
                h_alt = reshape_to(best_h, alt_n)
                s_alt = score_h(h_alt)
                # Optimize at this resolution
                for _ in range(4):
                    if time.time() >= deadline:
                        break
                    d = solve_lp_direction(h_alt)
                    if d is not None and np.linalg.norm(d) > 1e-10:
                        h_new, s_new = line_search(h_alt, d, s_alt)
                        if s_new < s_alt:
                            h_alt, s_alt = h_new, s_new
                    d = random_dir(rng, alt_n, 0.2)
                    if d is not None:
                        h_new, s_new = line_search(h_alt, d, s_alt)
                        if s_new < s_alt:
                            h_alt, s_alt = h_new, s_new
                if s_alt < best_s:
                    best_h, best_s, best_n = h_alt, s_alt, alt_n

    best_h = project_box_mass(best_h)
    best_s = score_h(best_h)
    return best_h, best_s, len(best_h)
