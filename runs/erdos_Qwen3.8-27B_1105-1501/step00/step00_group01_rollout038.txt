import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target_sum}"""
    # Binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def subgradient_descent(h, dx, n_iter=5000, lr_init=0.05, lr_decay=0.999):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    h = project_to_constraint(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lr = lr_init
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = (corr >= max_val - 1e-10)
        active_lags = np.where(active)[0] - (n - 1)
        
        # Compute subgradient
        g = np.zeros(n)
        for lag in active_lags:
            # (1-h[j+lag]) for valid j+lag
            if lag >= 0:
                # j+lag < n => j < n-lag
                g[:n-lag] += (1.0 - h[lag:])
                # -h[j-lag] for valid j-lag: j >= lag
                g[lag:] += -h[:n-lag]
            else:
                lag_abs = -lag
                # j+lag >= 0 => j >= lag_abs
                g[lag_abs:] += (1.0 - h[:n-lag_abs])
                # -h[j-lag] = -h[j+lag_abs], valid when j+lag_abs < n, i.e., j < n-lag_abs
                g[:n-lag_abs] += -h[lag_abs:]
        
        g *= dx
        
        # Subgradient step
        h = h - lr * g
        h = project_to_constraint(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        lr *= lr_decay
    
    return best_h, best_c5

def pairwise_exchange(h, max_iters=100000, dx=2.0/len(h)):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        # Pick two random indices
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        
        # Try transferring delta from i to j
        # h[j] += delta, h[i] -= delta
        # delta must satisfy 0 <= h[i]-delta and h[j]+delta <= 1
        # delta in (h[i]-1, h[i]) if transferring from i to j
        # Actually delta > 0 means i gives to j
        # h[i] - delta >= 0 => delta <= h[i]
        # h[j] + delta <= 1 => delta <= 1 - h[j]
        
        delta = min(h[i], 1.0 - h[j]) * 0.5
        if delta < 1e-10:
            continue
        
        # Try small perturbations
        h[i] -= delta
        h[j] += delta
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
    
    return best_h, best_c5

    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def optimize_subgradient(h, time_limit=60.0):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_to_constraint(h.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = 0.3
    stall = 0

    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = (corr >= max_val - 1e-9)
        active_lags = np.where(active)[0] - (n - 1)

        g = np.zeros(n)
        for lag in active_lags:
            if lag >= 0:
                if n - lag > 0:
                    g[:n-lag] += (1.0 - h[lag:])
                    g[lag:] += -h[:n-lag]
            else:
                la = -lag
                if n - la > 0:
                    g[la:] += (1.0 - h[:n-la])
                    g[:n-la] += -h[la:]

        h = h - lr * g * dx
        h = project_to_constraint(h, target_sum)

        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                lr *= 0.85
                stall = 0
        lr = max(lr * 0.9997, 0.002)

    return best_h, best_c5

def optimize_smooth(h, time_limit=60.0):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_to_constraint(h.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = 0.05
    n_total = max(1, int(time_limit / 0.005))
    it = 0

    while time.time() - t0 < time_limit and it < n_total:
        progress = it / n_total
        beta = 5.0 + 80.0 * progress

        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_c = np.max(corr)
        shifted = corr - max_c
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-8:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                if n - lag > 0:
                    g[:n-lag] += w[k] * (1.0 - h[lag:])
                    g[lag:] += -w[k] * h[:n-lag]
            else:
                la = -lag
                if n - la > 0:
                    g[la:] += w[k] * (1.0 - h[:n-la])
                    g[:n-la] += -w[k] * h[la:]

        h = h - lr * g * dx
        h = project_to_constraint(h, target_sum)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.9998
        it += 1

    return best_h, best_c5

def optimize_local(h, time_limit=30.0, seed=42):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    t0 = time.time()

    while time.time() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            i = order[ii]
            for jj in range(ii + 1, n):
                j = order[jj]
                dmax = min(h[i], 1.0 - h[j])
                if dmax > 1e-9:
                    d = dmax * 0.5
                    h[i] -= d
                    h[j] += d
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
                dmax2 = min(h[j], 1.0 - h[i])
                if dmax2 > 1e-9:
                    d = dmax2 * 0.5
                    h[j] -= d
                    h[i] += d
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                    else:
                        h[j] += d
                        h[i] -= d
            if time.time() - t0 > time_limit:
                return best_h, best_c5
        if not improved:
            break
        h = best_h.copy()

    return best_h, best_c5

def optimize_local_multi(h, time_limit=30.0, seed=42):

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_constraint(h_new, n_new / 2.0)

def random_valid_h(n, rng):
    h = rng.random(n)
    return project_to_constraint(h, n / 2.0)

def block_pattern(n, n_blocks=4, rng=None):
    h = np.zeros(n)
    seg = n // n_blocks
    for i in range(n_blocks):
        start = i * seg
        end = start + seg if i < n_blocks - 1 else n
        if i % 2 == 0:
            h[start:end] = 1.0
    return project_to_constraint(h, n / 2.0)

def periodic_pattern(n, period_atoms, rng=None):
    h = np.tile(period_atoms, n // len(period_atoms) + 1)[:n].astype(float)
    return project_to_constraint(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    def try_solution(h):
        nonlocal best_h, best_c5, best_n
        if h is None:
            return
        h = np.clip(h, 0.0, 1.0)
        h = project_to_constraint(h, len(h) / 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = len(h)

    # Phase 1: Optimize initial construction
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
        try_solution(h_init)
        n_init = len(h_init)
    except Exception:
        n_init = 64

    if h_init is not None:
        t_lim = min(0.20 * budget_s, 200.0)
        h_opt, c5_opt = optimize_subgradient(h_init, time_limit=t_lim * 0.5)
        try_solution(h_opt)

        h_opt2, c5_opt2 = optimize_smooth(h_opt, time_limit=t_lim * 0.25)
        try_solution(h_opt2)

        h_opt3, c5_opt3 = optimize_local(h_opt2, time_limit=t_lim * 0.25, seed=seed)
        try_solution(h_opt3)

    # Phase 2: Try multiple n values with various initializations
    n_candidates = []
    if h_init is not None:
        n_candidates = [n_init]
    else:
        n_candidates = [64]
    for n in [63, 64, 66, 68, 70, 72, 76, 80, 84, 90, 96, 100, 108, 112, 120]:
        if n not in n_candidates:
            n_candidates.append(n)

    n_candidates.sort()

    # Budget allocation for phase 2
    phase2_budget = 0.50 * budget_s
    n_per_try = max(1.0, phase2_budget / len(n_candidates))

    for n in n_candidates:
        if remaining() < 40:
            break

        t_alloc = min(n_per_try, remaining() - 30)
        if t_alloc < 5:
            continue

        # Multiple starts for this n
        starts = []
        if h_init is not None and n == n_init:
            starts.append(h_init.copy())
        elif h_init is not None:
            starts.append(interpolate_h(h_init, n))
        starts.append(random_valid_h(n, rng))
        if n % 4 == 0:
            starts.append(periodic_pattern(n, [1.0, 1.0, 0.0, 0.0]))
        starts.append(block_pattern(n, 4, rng))

        best_for_n = None
        best_c5_n = float('inf')

        for si, h_start in enumerate(starts):
            if remaining() < 20:
                break
            t_sub = t_alloc / len(starts) * (1.0 if si == 0 else 0.8)

            h_opt, c5_opt = optimize_subgradient(h_start, time_limit=t_sub * 0.5)
            c5_check = compute_c5(h_opt)
            if c5_check < best_c5_n:
                best_c5_n = c5_check
                best_for_n = h_opt.copy()

            if remaining() > 10:
                h_opt2, c5_opt2 = optimize_local(h_opt, time_limit=t_sub * 0.5, seed=seed + n + si * 100)
                r2 = compute_c5(h_opt2)
                if r2 < best_c5_n:
                    best_c5_n = r2
                    best_for_n = h_opt2.copy()

        if best_for_n is not None:
            try_solution(best_for_n)

    # Phase 3: Refine the best solution found
    if remaining() > 20 and best_h is not None:
        h_ref, c5_ref = optimize_smooth(best_h, time_limit=min(30.0, remaining() * 0.3))
        try_solution(h_ref)

    if remaining() > 15 and best_h is not None:
        h_ref, c5_ref = optimize_local_multi(best_h, time_limit=min(40.0, remaining() * 0.4), seed=seed + 777)
        try_solution(h_ref)

    if remaining() > 10 and best_h is not None:
        h_ref, c5_ref = optimize_local(best_h, time_limit=min(30.0, remaining() * 0.3), seed=seed + 999)
        try_solution(h_ref)

    # Phase 4: Final subgradient polish
    if remaining() > 5 and best_h is not None:
        h_ref, c5_ref = optimize_subgradient(best_h, time_limit=max(2.0, remaining() - 3.0))
        try_solution(h_ref)

    # Final check
    if best_h is None:
        h_fallback = random_valid_h(64, rng)
        best_h = h_fallback
        best_c5 = compute_c5(h_fallback)
        best_n = 64

    return best_h, best_c5, best_n
