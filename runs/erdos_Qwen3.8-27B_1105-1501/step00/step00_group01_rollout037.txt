    corr = np.zeros(2*n - 1)
    for lag in range(-(n-1), n):
        if lag >= 0:
            # c = sum_{i=0}^{n-1-lag} h[i+lag] * oh[i]
            corr[n-1+lag] = np.dot(h[lag:], oh[:n-lag])

            corr[n-1+lag] = np.dot(h[:n+lag], oh[-lag:])

def compute_c5(h):
    """Compute C5 = max of correlations * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def project_constraints(h, target_sum):
    """Project h onto {x in [0,1]^n : sum(x) = target_sum}"""
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def corr_and_grad(h, beta):
    """Compute correlation * dx and gradient of LSE-smoothed max."""
    n = len(h)
    dx = 2.0 / n
    oh = 1.0 - h
    
    # Compute correlation values (2n-1 of them)
    corr = np.zeros(2 * n - 1)
    for lag in range(-(n-1), n):
        if lag >= 0:
            corr[n-1+lag] = np.dot(h[lag:], oh[:n-lag])
        else:
            corr[n-1+lag] = np.dot(h[:n+lag], oh[-lag:])
    
    # Softmax weights for LSE
    c_max = corr.max()
    if beta > 1e-10:
        weights = np.exp(beta * (corr - c_max))
        weights /= weights.sum()
    else:

        imax = np.argmax(corr)
        weights = np.zeros(2*n-1)
        weights[imax] = 1.0

    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            # c = sum_{i=0}^{n-1-lag} h[i+lag] * oh[i]
            # dc/dh[j] += (1-h[j-lag]) for j in [lag, n-1]
            if lag < n:

            if al < n:
                grad[al:] -= w * h[:n-al]  # h[0..n-1-al]
    
    grad *= dx
    return corr * dx, grad

    best_h = h.copy()

def corr_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    oh = 1.0 - h
    corr = np.zeros(2 * n - 1)
    for lag in range(-(n-1), n):
        if lag >= 0:
            corr[n-1+lag] = np.dot(h[lag:], oh[:n-lag])
        else:
            corr[n-1+lag] = np.dot(h[:n+lag], oh[-lag:])
    c_max = corr.max()
    exp_vals = np.exp(beta * (corr - c_max))
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += w * oh[:n-lag]
            grad[:n-lag] -= w * h[lag:]
        else:
            al = -lag
            grad[:n-al] += w * oh[al:]
            grad[al:] -= w * h[:n-al]
    grad *= dx
    return corr * dx, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())
