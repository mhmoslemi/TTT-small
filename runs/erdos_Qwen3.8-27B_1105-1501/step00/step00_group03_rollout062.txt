```
def project_to_box(z, n):
    z = np.asarray(z, float).ravel()
    if z.size != n: pad/trunc? For interpolation we call with same length.
    if not finite: fallback target/n
    target = n/2.0
    # waterfilling bisection over lam
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()
    for _ in range(80):
       mid=(lo+hi)/2
       s = np.clip(z+mid,0,1).sum()
       if s < target: lo=mid
       else: hi=mid
    lam=(lo+hi)/2
    h = np.clip(z+lam,0,1)
    err = target - h.sum()
    # repair residual
    for it in range(n*2+10):
       if abs(err)<1e-12: break
       if err>0:
          idx = np.argmax(1-h)
          room = 1-h[idx]
          add = min(err, room)
          if add <=0: break
          h[idx]+=add; err -= add
       else:
          idx = np.argmax(h)
          sub = min(-err, h[idx])
          if sub <=0: break
          h[idx]-=sub; err += sub
    # if residual still, adjust argmax if in bounds
    err = target-h.sum()
    if abs(err)>1e-13:
       idx=np.argmax(1-h) if err>0 else np.argmax(h)
       h[idx]=np.clip(h[idx]+err,0,1) # may not exact if bounds
    return h
```

```
    n = h.size
```

```
def evaluate_candidate(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n <= 0:
        return 1.0
    if not np.all(np.isfinite(h)):
```

```python
    vals = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
    mx = float(np.max(vals))

    return mx


def related_values_scaled(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n <= 0:
        return np.array([0.0], dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def interpolate_to_length(src, m):
    src = np.asarray(src, dtype=float).ravel()
    m = int(m)
    if m <= 0:
        return np.array([], dtype=float)
    n = src.size
    if n == m:
        return src.copy()
    if n <= 1:
        val = float(src[0]) if n > 0 else 0.5
        return np.full(m, val, dtype=float)
    xs = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    xt = (np.arange(m, dtype=float) + 0.5) * (2.0 / float(m))
    return np.interp(xt, xs, src)


def project_to_box(z, n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)

    z = np.asarray(z, dtype=float).ravel()
    if z.size != n:
        z = interpolate_to_length(z, n)
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5, dtype=float)

    target = 0.5 * float(n)
    if (
        np.all(z >= -1e-12)
        and np.all(z <= 1.0 + 1e-12)
        and abs(float(np.sum(z)) - target) < 1e-12
    ):
        return np.clip(z, 0.0, 1.0)

    zc = np.clip(z, -1e8, 1e8)
    lo = -1.0 - float(np.max(zc))
    hi = 1.0 - float(np.min(zc))
    if lo > hi:
        lo, hi = hi, lo

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(zc + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(zc + 0.5 * (lo + hi), 0.0, 1.0)
    err = target - float(h.sum())

    for _ in range(2 * n + 10):
        if err > 0.0:
            if err <= 1e-13:
                break
            cap = 1.0 - h
            idx = int(np.argmax(cap))
            add = min(err, cap[idx])
            if add <= 1e-15:
                break
            h[idx] += add
            err -= add
        else:
            if err >= -1e-13:
                break
            idx = int(np.argmax(h))
            sub = min(-err, h[idx])
            if sub <= 1e-15:
                break
            h[idx] -= sub
            err += sub

    err = target - float(h.sum())
    if abs(err) > 1e-12:
        if err > 0.0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += err
        h = np.clip(h, 0.0, 1.0)

    return h


def finite_diff_jacobian(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    L = max(0, 2 * n - 1)
    if n == 0:
        return np.empty((0, 0), dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros((L, n), dtype=float)

    J = np.empty((L, n), dtype=float)
    eps = 1e-5
    inv_d = 1.0 / (2.0 * eps)

    hp = h.copy()
    hm = h.copy()
    for i in range(n):
        hp[i] += eps
        hm[i] -= eps
        vp = np.correlate(hp, 1.0 - hp, mode="full")
        vm = np.correlate(hm, 1.0 - hm, mode="full")
        J[:, i] = (vp - vm) * inv_d
        hp[i] -= eps
        hm[i] += eps

    J *= 2.0 / float(n)
    if not np.all(np.isfinite(J)):
        J = np.nan_to_num(J, nan=0.0, posinf=0.0, neginf=0.0)
    return J


def smooth_objective(x, beta, n):
    h = np.asarray(x, dtype=float).ravel()
    n = int(n)
    if h.size != n:
        h = interpolate_to_length(h, n)
    vals = related_values_scaled(h)
    if not np.all(np.isfinite(vals)):
        return 1.0

    m = float(np.max(vals))
    if beta <= 0.0:
        return m

    y = beta * (vals - m)
    y = np.clip(y, -700.0, 0.0)
    s = float(np.exp(y).sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return m + float(np.log(len(vals))) / float(beta)
    return m + float(np.log(s)) / float(beta)


def smooth_jacobian(x, beta, n):
    h = np.asarray(x, dtype=float).ravel()
    n = int(n)
    if h.size != n:
        h = interpolate_to_length(h, n)

    vals = related_values_scaled(h)
    if not np.all(np.isfinite(vals)):
        return np.zeros(n, dtype=float)

    m = float(np.max(vals))
    y = beta * (vals - m)
    y = np.clip(y, -700.0, 0.0)
    s = float(np.exp(y).sum())

    if (not np.isfinite(s)) or s <= 0.0:
        w = np.full(len(vals), 1.0 / float(len(vals)), dtype=float)
    else:
        w = np.exp(y) / s

    J = finite_diff_jacobian(h)
    g = J.T.dot(w)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=float)
    return g


def equality_fun(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def equality_jac(x, n):
    return np.ones(np.size(x), dtype=float)


def build_smooth_constraints(n):
    return [
        dict(type="eq", fun=equality_fun, jac=equality_jac, args=(int(n),))
    ]


def make_h_bounds(n):
    return [(0.0, 1.0)] * int(n)


def direct_obj(z):
    z = np.asarray(z, dtype=float).ravel()
    return float(z[-1])


def direct_obj_jac(z):
    g = np.zeros(np.size(z), dtype=float)
    if g.size > 0:
        g[-1] = 1.0
    return g


def direct_ineq(z):
    z = np.asarray(z, dtype=float).ravel()
    h = z[:-1]
    vals = related_values_scaled(h)
    return z[-1] - vals


def direct_ineq_jac(z):
    z = np.asarray(z, dtype=float).ravel()
    n = z.size - 1
    L = max(0, 2 * n - 1)
    out = np.zeros((L, n + 1), dtype=float)
    if n <= 0:
        return out
    J = finite_diff_jacobian(z[:-1])
    out[:, :n] = -J
    return out


def direct_eq_fun(z, n):
    z = np.asarray(z, dtype=float).ravel()
    return float(np.sum(z[:-1]) - 0.5 * float(n))


def direct_eq_jac(z, n):
    z = np.asarray(z, dtype=float).ravel()
    g = np.zeros(z.size, dtype=float)
    if z.size > 0:
        g[:-1] = 1.0
    return g


def build_direct_constraints(n):
    return [
        dict(type="ineq", fun=direct_ineq, jac=direct_ineq_jac),
        dict(type="eq", fun=direct_eq_fun, jac=direct_eq_jac, args=(int(n),)),
    ]


def make_direct_bounds(n):
    return [(0.0, 1.0)] * int(n) + [(0.0, 5.0)]


def smooth_solve(h0, n, beta, maxiter):
    n = int(n)
    h0 = project_to_box(h0, n)
    try:
        res = minimize(
            smooth_objective,
            h0,
            method="SLSQP",
            jac=smooth_jacobian,
            args=(float(beta), n),
            bounds=make_h_bounds(n),
            constraints=build_smooth_constraints(n),
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h0

    if getattr(res, "x", None) is not None and np.all(np.isfinite(res.x)):
        return project_to_box(res.x, n)
    return h0


def direct_solve(h0, n, maxiter):
    n = int(n)
    h0 = project_to_box(h0, n)
    vals = related_values_scaled(h0)
    m = float(np.max(vals)) if vals.size > 0 else 0.0
    theta0 = m + 1e-7
    if theta0 < 1e-9:
        theta0 = 1e-9
    if theta0 > 4.999:
        theta0 = 4.999

    z0 = np.concatenate((h0, np.array([theta0], dtype=float)))
    try:
        res = minimize(
            direct_obj,
            z0,
            method="SLSQP",
            jac=direct_obj_jac,
            bounds=make_direct_bounds(n),
            constraints=build_direct_constraints(n),
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return h0

    if getattr(res, "x", None) is not None and np.all(np.isfinite(res.x)):
        return project_to_box(res.x[:-1], n)
    return h0


def load_parent():
    obj = globals().get("initial_h_values", None)
    if obj is None:
        return None
    arr = np.asarray(obj, dtype=float).ravel()
    if arr.size < 4 or arr.size > 220 or not np.all(np.isfinite(arr)):
        return None
    return project_to_box(arr, arr.size)


def arrays_close(a, b, tol=1e-7):
    a = np.asarray(a, dtype=float).ravel()
    b = np.asarray(b, dtype=float).ravel()
    if a.size != b.size:
        return False
    if a.size == 0:
        return True
    return float(np.max(np.abs(a - b))) < tol


def try_add(cands, raw, n, tol=1e-7):
    n = int(n)
    h = project_to_box(raw, n)
    if h.size != n or not np.all(np.isfinite(h)):
        return
    for idx in range(len(cands)):
        if arrays_close(h, cands[idx][1], tol):
            return
    cands.append((float(evaluate_candidate(h)), h))


def symmetric_average(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return h
    return 0.5 + 0.5 * (h - h[::-1])


def complement_reverse(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return h
    return 1.0 - h[::-1]


def generate_pool(n, parent, rng, max_random, parent_available):
    n = int(n)
    cands = []
    x = np.arange(n, dtype=float)

    try_add(cands, np.full(n, 0.5, dtype=float), n)
    for freq in (1, 2, 3):
        try_add(cands, 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x / float(n)), n)
        try_add(cands, 0.5 + 0.45 * np.cos(2.0 * np.pi * freq * x / float(n)), n)
    try_add(cands, 0.5 + 0.45 * np.sin(np.pi * x / float(n + 1)), n)
    try_add(cands, 0.5 + 0.40 * np.cos(2.0 * np.pi * x / float(n + 2)), n)

    blk = np.empty(n, dtype=float)
    mix = max(1, n // 2)
    blk[:mix] = 0.90
    blk[mix:] = 0.10
    try_add(cands, blk, n)

    blk2 = np.empty(n, dtype=float)
    a = max(1, n // 3)
    b = max(a + 1, 2 * n // 3)
    blk2[:a] = 0.85
    blk2[a:b] = 0.15
    blk2[b:] = 0.70
    try_add(cands, blk2, n)

    if parent is not None:
        p = interpolate_to_length(parent, n)
        try_add(cands, p, n)
        try_add(cands, symmetric_average(p), n)
        try_add(cands, complement_reverse(p), n)
        for sigma in (0.02, 0.05, 0.10, 0.18):
            try_add(cands, p + rng.normal(0.0, sigma, n), n)

    for _ in range(max_random):
        try_add(cands, 0.5 + rng.normal(0.0, 0.25, n), n)

    if not cands:
        return []

    scores = np.array([c[0] for c in cands], dtype=float)
    order = np.argsort(scores)
    return [cands[int(i)] for i in order]


def sort_pairs_by_score(pairs):
    if len(pairs) == 0:
        return []
    scores = np.array([p[0] for p in pairs], dtype=float)
    order = np.argsort(scores)
    return [pairs[int(i)] for i in order]


def select_lengths(base_n, budget_s, max_len=8):
    base = int(base_n) if base_n is not None else 45
    budget = float(budget_s)
    if budget >= 800:
        max_len = max(max_len, 8)
    elif budget >= 400:
        max_len = min(max_len, 6)
    else:
        max_len = min(max_len, 4)

    raw = [
        base, base - 1, base + 1, base - 2, base + 2,
        base - 3, base + 3, base - 5, base + 5,
        base - 8, base + 8, base + 10, base - 12, base + 12,
        base + 15, base - 15, 30, 35, 40, 45, 50, 55, 60,
    ]
    seen = []
    for v in raw:
        v = int(v)
        if 18 <= v <= 90 and v not in seen:
            seen.append(v)

    rest = [v for v in seen if v != base]
    tups = [(abs(v - base), v) for v in rest]
    tups.sort()

    lengths = [base]
    for _, v in tups:
        if len(lengths) >= max_len:
            break
        lengths.append(v)
    return lengths


def choose_maxiter(n, remaining, count, is_direct):
    remaining = float(max(0.0, remaining))
    n = float(max(1, int(n)))

    if remaining < 60.0:
        base = 16
    elif remaining < 180.0:
        base = 30
    elif count < 8:
        base = 75
    elif count < 30:
        base = 50
    else:
        base = 32

    if is_direct:
        base = int(base * 0.95)

    cap = int(7000.0 / max(25.0, n))
    if base > cap:
        base = cap
    return max(10, int(base))


def update_best(best_score, best_h, best_n, candidate_h):
    candidate_h = np.asarray(candidate_h, dtype=float).ravel()
    if candidate_h.size <= 0:
        return best_score, best_h, best_n

    h = project_to_box(candidate_h, candidate_h.size)
    if not np.all(np.isfinite(h)):
        return best_score, best_h, best_n

    sc = float(evaluate_candidate(h))
    if not np.isfinite(sc):
        return best_score, best_h, best_n

    if sc + 1e-13 < best_score:
        return sc, h, int(h.size)
    if abs(sc - best_score) <= 1e-12 and h.size < best_n:
        return sc, h, int(h.size)

    return best_score, best_h, best_n


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    budget = float(budget_s)
    reserve = max(8.0, 0.06 * budget)
    if reserve > 0.25 * budget:
        reserve = 0.12 * budget
    deadline = t_start + budget - reserve

    rng = np.random.default_rng(int(seed))
    parent = load_parent()

    if parent is None:
        n0 = 45
        x = np.arange(n0, dtype=float)
        init_h = 0.5 + 0.45 * np.sin(2.0 * np.pi * x / float(n0 + 1))
        best_h = project_to_box(init_h, n0)
    else:
        best_h = project_to_box(parent, parent.size)

    best_score = float(evaluate_candidate(best_h))
    best_n = int(best_h.size)

    lengths = select_lengths(best_n, budget, max_len=8)
    solve_count = 0

    if budget >= 750:
        top_starts = 3
        betas = [20.0, 120.0, 500.0]
        max_random = 8
    elif budget >= 350:
        top_starts = 2
        betas = [60.0, 300.0]
        max_random = 5
    else:
        top_starts = 1
        betas = [250.0]
        max_random = 2

    for n in lengths:
        n = int(n)
        if time.time() > deadline - 25.0:
            break

        if budget >= 750:
            pool_limit = 24
        elif budget >= 350:
            pool_limit = 16
        else:
            pool_limit = 10

        pool = generate_pool(n, parent, rng, max_random, parent is not None)
        if not pool:
            continue

        if len(pool) > pool_limit:
            pool = pool[:pool_limit]

        polished = []

        for idx in range(min(top_starts, len(pool))):
            if time.time() > deadline - 30.0:
                break

            sc, h = pool[idx]
            best_score, best_h, best_n = update_best(best_score, best_h, best_n, h)

            cur = project_to_box(h, n)
            cur_score = float(evaluate_candidate(cur))
            polished.append((cur_score, cur))

            for beta in betas:
                if time.time() > deadline - 35.0:
                    break

                itmax = choose_maxiter(
                    n,
                    deadline - time.time(),
                    solve_count,
                    False,
                )
                solve_count += 1
                new = smooth_solve(cur, n, beta, itmax)
                new_score = float(evaluate_candidate(new))

                best_score, best_h, best_n = update_best(
                    best_score, best_h, best_n, new
                )
                polished.append((new_score, new))

                if new_score <= cur_score + 1e-9:
                    cur = new
                    cur_score = new_score

            polished.append((cur_score, cur))

        polished = sort_pairs_by_score(polished)

        direct_count = 2 if budget >= 350 else 1
        for j in range(min(direct_count, len(polished))):
            if time.time() > deadline - 50.0:
                break
            _, h = polished[j]
            itmax = choose_maxiter(
                int(best_n if best_n == h.size else h.size),
                deadline - time.time(),
                solve_count,
                True,
            )
            solve_count += 1
            new = direct_solve(h, h.size, itmax)
            best_score, best_h, best_n = update_best(best_score, best_h, best_n, new)

        if best_score <= 0.3808 - 1e-10:
            break

    if time.time() < deadline - 25.0:
        itmax = choose_maxiter(best_n, deadline - time.time(), solve_count, True)
        solve_count += 1
        cand = direct_solve(best_h, best_n, itmax)
        best_score, best_h, best_n = update_best(best_score, best_h, best_n, cand)

    if time.time() < deadline - 25.0:
        itmax = choose_maxiter(best_n, deadline - time.time(), solve_count, False)
        solve_count += 1
        cand = smooth_solve(best_h, best_n, 800.0, itmax)
        best_score, best_h, best_n = update_best(best_score, best_h, best_n, cand)

    if time.time() < deadline - 10.0:
        itmax = choose_maxiter(best_n, deadline - time.time(), solve_count, True)
        solve_count += 1
        cand = direct_solve(best_h, best_n, itmax)
        best_score, best_h, best_n = update_best(best_score, best_h, best_n, cand)

    best_h = project_to_box(best_h, int(best_n))
    final_score = float(evaluate_candidate(best_h))

    if not np.isfinite(final_score):
        final_score = 0.5

    final_bound = float(np.nextafter(final_score, 1.0))
    if not np.isfinite(final_bound) or final_bound < 0.0:
        final_bound = final_score

    return best_h, final_bound, int(best_h.size)
```
