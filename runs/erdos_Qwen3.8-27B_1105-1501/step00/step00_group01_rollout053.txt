            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
            g = active_grad(h, n, 0.99)
            h = pgrad_step(h, n, g, 0.01)

    return best_h, best_c5, best_n
