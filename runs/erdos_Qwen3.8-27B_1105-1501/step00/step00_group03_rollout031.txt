def _proj(y, ts):

    for _ in range(60):

        v = np.clip(y - mid, 0.0, 1.0)
        if v.sum() > ts:
            lo = mid
        else:

    c = np.correlate(h, 1.0 - h, 'full')
    return c.max() * 2.0 / n

    G = np.empty((2*n-1, n))
    for kk in range(2*n-1):
        k = kk - (n-1)
        # For each j, grad = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]

        if lo1 < hi1:

        hi2 = min(n, n - k)

import numpy as np
import time


def _proj(y, ts):
    n = len(y)
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _sc(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / n)


def _grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for kk in range(2 * n - 1):
        k = kk - (n - 1)
        row = np.zeros(n)
        if k >= 0:
            m = n - k
            if m > 0:
                row[k:] += omh[:m]
                row[:m] -= h[k:]
        else:
            m = -k
            if n - m > 0:
                row[:n - m] += omh[m:]
                row[m:] -= h[:n - m]
        G[kk] = row
    return G


def _w(c, b):
    m = c.max()
    e = np.exp(b * (c - m))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return w
    return e / s


def _opt(h, t0, dl, bh, bs, n):
    dx = 2.0 / n
    ts = n * 0.5
    for b in [3, 8, 20, 50, 150, 400, 1200]:
        if time.time() > dl:
            break

                    break

def _coord_refine(h, t0, dl, bh, bs, n):
    ts = n * 0.5
    for _ in range(50):
        if time.time() > dl:
            break
        improved_any = False
        for i in range(n):
            if time.time() > dl:
                break
            best_i = h[i]
            for j in range(n):
                if j == i:
                    continue
                old_i = h[i]
                old_j = h[j]
                for t in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
                    ni = old_i + t
                    nj = old_j - t
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    h[j] = nj
                    s = _sc(h)

                        bs = s
                        bh = h.copy()
                        best_i = ni
                        improved_any = True
                    h[i] = old_i
                    h[j] = old_j
                h[i] = best_i
        if not improved_any:

    return h, bh, bs
