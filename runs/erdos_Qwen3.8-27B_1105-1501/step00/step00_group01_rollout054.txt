    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h.copy(), 0.0, 1.0)

            cnt = np.sum(mask)
            if cnt == 0:

            h[mask] += diff / cnt

    corr = compute_corr(h)
    c_max = np.max(corr)
    scaled = beta * (corr - c_max)
    return np.max(corr) * (2.0 / len(h)) + (2.0 / len(h)) * logsumexp(scaled) / beta

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = beta * (corr - cmax)
    return (2.0 / n) * (cmax + logsumexp(scaled) / beta)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    scaled = beta * (corr - cmax)
    log_sum = logsumexp(scaled)
    w = np.exp(scaled - log_sum)
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        t1 = 0.0
        t2 = 0.0
        for k in range(2 * n - 1):
            s = shifts[k]
            j1 = i - s
            j2 = i + s
            if 0 <= j1 < n:
                t1 += w[k] * (1.0 - h[j1])
            if 0 <= j2 < n:
                t2 += w[k] * h[j2]
        grad[i] = dx * (t1 - t2)
    return grad

def slsqp_optimize(h, n, beta, maxiter=500):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_obj(x, beta),
            h, jac=lambda x: smooth_grad(x, beta),
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return res.x
    except Exception:
        return h

    best = h.copy()
    best_val = smooth_obj(best, beta)
    best_c5 = evaluate_c5(best)
    for _ in range(n_iter):
        g = smooth_grad(best, beta)
        # Step
        h_new = best - lr * g
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_val = c5
            best = h_new.copy()
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-6:
                lr = 0.05
    return best, best_c5

            # Also try a fresh smooth start

            # Optimize both
            for h_start, tag in [(h_try, 'interp'), (h_alt, 'fresh')]:
                if time_left() < 20:
                    break
                for beta in [50, 300, 1000]:
                    if time_left() < 10:
                        break
                    h_new = slsqp_optimize(h_start, n_try, beta, maxiter=300)
                    h_new = project_feasible(h_new, n_try)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        h_start = h_new.copy()
                # Local search
                if time_left() > 30:
                    h_new, c5 = pair_local_search(h_start, n_try, 20000, rng)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()

    # Phase 6: Final polish on best solution
    if time_left() > 15:
        n_final = len(best_h)
        for beta in [5000, 20000, 50000]:
            if time_left() < 10:
                break
            h_new = slsqp_optimize(best_h, n_final, beta, maxiter=300)
            h_new = project_feasible(h_new, n_final)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, len(best_h))
