```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    cf = np.correlate(h, b, mode='full')
    return float(np.max(cf) * dx)

def h_from_z(z):
    """Map z (length n) to h (length n) with sum(h) = n/2.
    h = sigmoid(z + tau) where tau is chosen so sum(h) = n/2."""
    n = len(z)
    target = n / 2.0
    
    # Find tau by bisection
    lo, hi = -50.0, 50.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(expit(z + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return expit(z + tau)

def z_to_h_and_params(z_free):
    """z_free has length n-1. Returns h, g=h*(1-h), G_sum, and full_z."""
    n1 = len(z_free)
    n = n1 + 1
    full_z = np.empty(n)
    full_z[:n1] = z_free
    full_z[n1] = -np.sum(z_free)
    
    # Find tau
    target = n / 2.0
    lo, hi = -50.0, 50.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(expit(full_z + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    
    h = expit(full_z + tau)
    g = h * (1.0 - h)
    G_sum = np.sum(g)
    return h, g, G_sum, full_z

def objective_and_grad(z_free, beta):
    """Compute smoothed objective and gradient wrt z_free."""
    h, g, G_sum, full_z = z_to_h_and_params(z_free)
    n = len(h)
    dx = 2.0 / n
    
    # Compute G_k = cf[k] * dx
    b = 1.0 - h
    cf = np.correlate(h, b, mode='full')
    G = cf * dx
    
    # Softmax weights
    G_max = np.max(G)
    alpha = np.exp(beta * (G - G_max))
    alpha /= np.sum(alpha)
    
    # Smoothed objective
    psi = (G_max + np.log(np.sum(np.exp(beta * (G - G_max)))) / beta)
    
    # Compute J: gradient of cf wrt h, shape (n, 2n-1)
    # J[i,k] = d(cf[k])/dh[i]
    J = np.zeros((n, 2*n-1))
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            # cf[k] = sum_{m=0}^{n-1-s} h[m+s]*(1-h[m])
            for m in range(n - s):
                # h[m+s] contributes +1 to J[m+s, k] with factor (1-h[m])
                J[m+s, k] += (1.0 - h[m])
                # (1-h[m]) contributes -1 to J[m, k] with factor h[m+s]
                J[m, k] += -h[m+s]
        else:
            sp = -s
            # cf[k] = sum_{m=0}^{n-1-sp} h[m]*(1-h[m+sp])
            for m in range(n - sp):
                J[m, k] += (1.0 - h[m+sp])
                J[m+sp, k] += -h[m]
    
    # grad_psi wrt h: dx * J^T @ alpha
    grad_h = dx * (J.T @ alpha)
    
    # Project to constraint: grad_z_full[i] = g[i] * (grad_h[i] - A/G_sum)
    A = np.sum(grad_h * g)
    grad_z_full = g * (grad_h - A / G_sum)
    
    # Gradient wrt z_free: grad_z_free[j] = grad_z_full[j] - grad_z_full[n-1]
    grad_z_free = grad_z_full[:n-1] - grad_z_full[n-1]
    
    return psi, grad_z_free
```

```python
def compute_c5(h):

    dx = 2.0 / n
    cf = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cf) * dx)

def h_from_z_free(z_free, n):
    """z_free: length n-1. Returns h: length n, sum h = n/2."""
    full_z = np.empty(n)
    full_z[:n-1] = z_free
    full_z[n-1] = -np.sum(z_free)
    
    target = n / 2.0
    lo, hi = -30.0, 30.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(expit(full_z + mid)) < target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return expit(full_z + tau)

def obj_and_grad(z_free, n, beta):
    h = h_from_z_free(z_free, n)
    dx = 2.0 / n
    b = 1.0 - h
    cf = np.correlate(h, b, mode='full')
    G = cf * dx  # length 2n-1
    
    # Softmax
    G_max = np.max(G)
    exp_terms = np.exp(beta * (G - G_max))
    alpha = exp_terms / np.sum(exp_terms)
    psi = G_max + np.log(np.sum(exp_terms)) / beta
    
    # Compute J[i,k] = d(cf[k])/dh[i]
    two_n_minus_1 = 2 * n - 1
    J = np.zeros((n, two_n_minus_1))

            J[m + s, k] += (1.0 - h[m])
            J[m, k] += -h[m + s]
        else:

            m = np.arange(n - sp)
            J[m, k] += (1.0 - h[m + sp])
            J[m + sp, k] += -h[m]

    grad_h = dx * (J.T @ alpha)  # length n

    g = h * (1.0 - h)

    grad_z_free = grad_z_full[:n-1] - grad_z_full[n-1]

    n = len(h)
    Q = 2 ** Qbits
    target_int = (n * Q) // 2
    
    h_int = np.round(h * Q).astype(np.int64)
    h_int = np.clip(h_int, 0, Q)
    
    diff = target_int - int(np.sum(h_int))
    if diff != 0:
        if diff > 0:
            room = Q - h_int
            order = np.argsort(-room, kind='stable')
            idx = 0
            remaining = diff
            while remaining > 0:
                h_int[order[idx % n]] += 1
                remaining -= 1
                idx += 1

            remaining = -diff

                remaining -= 1
                idx += 1
```

```python
    elif diff < 0:
        order = np.argsort(h_int, kind='stable')
        for i in range(-diff):
            h_int[order[i % n]] -= 1
    return h_int.astype(float) / Q


def pairwise_polish(h, max_time=30.0):
    n = len(h)
    start = time.time()
    h = h.copy()
    best_c = compute_c5(h)

    while (time.time() - start) < max_time:
        improved = False
        h_desc = np.argsort(-h)
        h_asc = np.argsort(h)
        npairs = 0
        for pi in range(min(15, n)):
            for qi in range(min(15, n)):
                if npairs >= 80:
                    break
                p = int(h_desc[pi])
                q = int(h_asc[qi])
                if p == q:
                    continue
                t_max = min(h[p], 1.0 - h[q])
                if t_max < 1e-10:
                    continue
                ts = np.linspace(0.0, t_max, 20)
                best_t = 0.0
                best_val = best_c
                for t in ts:
                    hp, hq = h[p], h[q]
                    h[p], h[q] = hp - t, hq + t
                    v = compute_c5(h)
                    h[p], h[q] = hp, hq
                    if v < best_val:
                        best_val = v
                        best_t = t
                if best_val < best_c - 1e-12 and best_t > 1e-10:
                    lo_t = max(0.0, best_t - t_max / 20.0)
                    hi_t = min(t_max, best_t + t_max / 20.0)
                    ts2 = np.linspace(lo_t, hi_t, 15)
                    for t in ts2:
                        hp, hq = h[p], h[q]
                        h[p], h[q] = hp - t, hq + t
                        v = compute_c5(h)
                        h[p], h[q] = hp, hq
                        if v < best_val:
                            best_val = v
                            best_t = t
                    if best_val < best_c - 1e-12:
                        h[p] -= best_t
                        h[q] += best_t
                        np.clip(h, 0.0, 1.0, out=h)
                        best_c = best_val
                        improved = True
                        h_desc = np.argsort(-h)
                        h_asc = np.argsort(h)
                npairs += 1
            if npairs >= 80:
                break
        if not improved:
            break
    return h, best_c


def resample_parent(parent, new_n):
    old_n = len(parent)
    old_x = np.linspace(0.0, 2.0, old_n + 2)[1:-1]
    new_x = np.linspace(0.0, 2.0, new_n + 2)[1:-1]
    h_new = np.interp(new_x, old_x, parent)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_int = np.round(h_new * 2**20).astype(np.int64)
    h_int = np.clip(h_int, 0, 2**20)
    target_int = (new_n * 2**20) // 2
    diff = target_int - int(np.sum(h_int))
    if diff > 0:
        room = 2**20 - h_int
        order = np.argsort(-room, kind='stable')
        for i in range(diff):
            h_int[order[i % new_n]] += 1
    elif diff < 0:
        order = np.argsort(h_int, kind='stable')
        for i in range(-diff):
            h_int[order[i % new_n]] -= 1
    return h_int.astype(float) / (2.0**20)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95

    n = 47
    try:
        parent = np.array(initial_h_values, dtype=float)
        n = len(parent)
    except Exception:
        parent = np.full(n, 0.5)

    best_h = parent.copy()
    best_c = compute_c5(best_h)

    eps = 1e-12
    hc = np.clip(parent, eps, 1.0 - eps)
    logit_h = np.log(hc / (1.0 - hc))
    z0 = logit_h - np.mean(logit_h)

    rng = np.random.RandomState(seed)

    betas = [3, 8, 20, 50, 100, 200]
    scales = [0.0, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 3.0, 0.15, 0.4, 0.8, 1.5]

    max_restarts = 30
    for r in range(max_restarts):
        if time.time() > deadline - 180:
            break
        scale = scales[r % len(scales)]
        gen = r // len(scales)
        if gen > 0:
            scale *= 0.5 ** gen

        z_start = z0.copy()
        if scale > 0:
            z_start = z_start + rng.randn(n - 1) * scale

        for beta in betas:
            if time.time() > deadline - 120:
                break
            loss = LossObj(n, beta)
            try:
                res = minimize(loss.func, z_start, jac=loss.grad, method='L-BFGS-B',
                               options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-7})
                h_new = h_from_z_free(res.x, n)
                if np.all(h_new >= -1e-9) and np.all(h_new <= 1.0 + 1e-9):
                    h_new = np.clip(h_new, 0.0, 1.0)
                    # Fix sum
                    s = float(np.sum(h_new))
                    target_s = n / 2.0
                    if abs(s - target_s) > 1e-10:
                        h_new = h_new + (target_s - s) / n
                        h_new = np.clip(h_new, 0.0, 1.0)
                    c_new = compute_c5(h_new)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                        z0 = res.x.copy()
            except Exception:
                pass

    # Try a few different n values
    for trial_n in [n - 7, n + 5, n - 3, n + 10]:
        if trial_n < 10 or trial_n > 200:
            continue
        if time.time() > deadline - 60:
            break
        try:
            h_trial = resample_parent(parent, trial_n)
            c_trial = compute_c5(h_trial)
            if c_trial < best_c:
                best_c = c_trial
                best_h = h_trial
                n = trial_n
                # Re-derive z0
                hc2 = np.clip(h_trial, eps, 1.0 - eps)
                lh2 = np.log(hc2 / (1.0 - hc2))
                z0 = lh2 - np.mean(lh2)
                # Quick optimization on new n
                for beta in [10, 50]:
                    if time.time() > deadline - 30:
                        break
                    loss2 = LossObj(trial_n, beta)
                    res2 = minimize(loss2.func, z0, jac=loss2.grad, method='L-BFGS-B',
                                    options={'maxiter': 100, 'maxfun': 300})
                    h2 = h_from_z_free(res2.x, trial_n)
                    h2 = np.clip(h2, 0.0, 1.0)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()
                        n = trial_n
                        z0 = res2.x.copy()
        except Exception:
            pass

    # Dyadic snap
    h_snap = dyadic_snap(best_h, 22)
    c_snap = compute_c5(h_snap)
    if c_snap < best_c:
        best_c = c_snap
        best_h = h_snap.copy()

    # Pairwise polish
    remain = deadline - time.time()
    if remain > 5:
        polish_time = min(80.0, remain * 0.5)
        best_h, best_c = pairwise_polish(best_h, max_time=polish_time)

    # Final snap
    h_final = dyadic_snap(best_h, 22)
    c_final = compute_c5(h_final)
    if c_final <= best_c + 1e-10:
        best_h = h_final
        best_c = c_final

    return best_h, best_c, n
```
