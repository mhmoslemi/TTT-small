4. Average subgradient: g = (1/|A|) * sum_{m in A} grad_m
5. Step: h_new = h - lr * g

The learning rate lr should be small enough to not overshoot. We can use backtracking:

import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def correlation_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def subgradient(h, active_tol=0.001):
    """Compute subgradient of C5 = max_m corr[m]*dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    
    # Active constraints
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - active_tol / dx)[0]
    
    grad = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        for i in range(n):
            # First term: (1 - h[i-shift]) if max(0,shift) <= i < min(n, n+shift)
            a_lo = max(0, shift)
            a_hi = min(n, n + shift)
            if a_lo <= i < a_hi:
                j = i - shift
                if 0 <= j < n:
                    grad[i] += (1.0 - h[j])
            # Second term: -h[i+shift] if max(0,-shift) <= i < min(n, n+shift)
            b_lo = max(0, -shift)
            b_hi = min(n, n + shift)
            if b_lo <= i < b_hi:
                j = i + shift
                if 0 <= j < n:
                    grad[i] -= h[j]
    
    return c5, grad * dx

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    current_sum = np.sum(h)

        return h

    if current_sum < target:

    else:

        else:
            hi = mid

    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:

        if diff > 0:
            # Need to add diff total
            for i in range(n):

                    h_new[i] += diff
                    break
        else:
            # Need to subtract |diff| total
            for i in range(n):
                if h_new[i] + diff >= 0.0:
                    h_new[i] += diff
                    break

def resample(parent, new_n):
    """Resample a function from len(parent) to new_n points"""
    n_old = len(parent)
    # Create cumulative distribution
    cdf_old = np.cumsum(parent)
    cdf_old = cdf_old / cdf_old[-1]  # normalize to [0,1]
    
    # Sample at new grid points
    new_centers = (np.arange(new_n) + 0.5) / new_n
    # Map to old grid
    old_indices = new_centers * n_old
    i0 = np.floor(old_indices).astype(int)
    i1 = np.minimum(i0 + 1, n_old - 1)
    frac = old_indices - i0
    
    new_h = parent[i0] * (1 - frac) + parent[i1] * frac
    
    # Rescale to correct area
    current_area = np.sum(new_h) * (2.0 / new_n)
    target_area = 1.0
    new_h = new_h * (target_area / current_area)
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = project_feasible(new_h, new_n)
    
    return new_h

def local_pair_swap(h, n, max_pairs=None, eps=0.01, seed=42):
    """Try pair swaps to improve C5"""
    c5 = exact_c5(h)
    rng = np.random.RandomState(seed)
    
    if max_pairs is None:
        max_pairs = n * n
    
    # Priority based on gradient
    c5, grad = subgradient(h, active_tol=0.001)
    
    # Sort by gradient magnitude (most constrained first)
    order = np.argsort(np.abs(grad))
    
    improved = False
    for _ in range(max_pairs):
        # Pick a pair
        i = order[rng.randint(n)]  # low gradient (want to increase h)
        j = order[rng.randint(n)]  # low gradient (want to decrease h)
        
        if i == j:
            continue
        
        # Try moving from higher h to lower h
        h_i, h_j = h[i], h[j]
        
        # Direction: from j to i (if grad[j] > grad[i])
        if grad[j] > grad[i]:
            # Move from j to i
            delta = min(h_j, 1.0 - h_i, eps)
            if delta < 1e-6:
                continue
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            # Move from i to j
            delta = min(h_i, 1.0 - h_j, eps)
            if delta < 1e-6:
                continue
            h_new = h.copy()
            h_new[j] += delta
            h_new[i] -= delta
        
        c5_new = exact_c5(h_new)
        if c5_new < c5 - 1e-12:
            h = h_new
            c5 = c5_new
            improved = True
    
    return h, c5

    n = len(h)
    shift = m - (n - 1)
    g = np.zeros(n)

    a_hi = min(n, n + shift)
