    n = len(v)

    hi = v.max() + target_sum + 1
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
import time

def project(v, target_sum):
    n = len(v)
    lo = v.min() - target_sum - 1.0
    hi = v.max() + target_sum + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def c5_of(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def local_optimize(h, time_limit, t0):
    n = len(h)
    best = c5_of(h)
    # Multi-scale deltas
    scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    
    for scale in scales:
        deltas = [scale, -scale, 2*scale, -2*scale, 0.5*scale, -0.5*scale]
        while time.time() - t0 < time_limit:
            improved = False
            for i in range(n):
                for j in range(i+1, n):
                    for d in deltas:
                        a = h[i] + d
                        b = h[j] - d
                        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
                            h[i] = a; h[j] = b
                            v = c5_of(h)
                            if v < best - 1e-12:
                                best = v; improved = True
                            else:
                                h[i] -= d; h[j] += d
            if not improved:
                break
    
    return h, best

def descent(h, time_limit, t0, lr=0.5, top_k=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = c5_of(h)
    y = h.copy()
    mu = 0.0
    
    for it in range(100000):
        if time.time() - t0 > time_limit:
            break
        # Compute subgradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-top_k:][::-1]
        w = corr[idx]
        w = w / w.sum()
        sg = np.zeros(n)
        for wi, ki in zip(w, idx):
            s = ki - (n - 1)
            if s >= 0 and s < n:
                sg[:n-s] += wi * (1.0 - h[s:])
                sg[s:] += wi * (-h[:n-s])
            elif s < 0:
                t = -s
                if t < n:
                    sg[t:] += wi * (1.0 - h[:n-t])
                    sg[:n-t] += wi * (-h[t:])
        sg *= dx
        
        # Nesterov step
        beta = (1.0 + mu) / (1.5 + mu)
        y = project(h + mu*(y - h), target)
        h = project(y - lr/(1.0+it*0.005) * sg, target)
        mu = beta
        
        if c5_of(h) < best_c5:
            best_c5 = c5_of(h)
            best_h = h.copy()
    return best_h, best_c5

        nn = n + dn
