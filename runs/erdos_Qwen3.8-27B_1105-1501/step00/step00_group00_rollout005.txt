import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    # Find lambda such that sum(clamp(y - lambda, 0, 1)) = target
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix exact mass
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - result)[::-1]
            for i in idx:
                room = 1.0 - result[i]
                if room >= diff + 1e-15:
                    result[i] += diff
                    break
                else:
                    result[i] = 1.0
                    diff -= room
        else:
            diff = -diff
            idx = np.argsort(result)[::-1]
            for i in idx:
                room = result[i]
                if room >= diff - 1e-15:
                    result[i] -= diff
                    break
                else:
                    result[i] = 0.0
                    diff -= room
    return result

def compute_corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def compute_grad_smoothing(h, tau):
    n = len(h)
    c = compute_corr_values(h)
    M = np.max(c)
    # For numerical stability
    exps = np.exp((c - M) / tau)
    w = exps / exps.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    dx = 2.0 / n
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        lag = m - center
        if lag >= 0:
            # c[m] = dx * sum_{j=0}^{n-1-lag} h[j+lag]*(1-h[j])
            # d/dh[i]: (1-h[i-lag]) for lag <= i <= n-1
            #           -h[i+lag] for 0 <= i <= n-1-lag
            if lag == 0:
                grad[:n] += wm * dx * (1.0 - h[:n] - h[:n])  # (1-h[i]) - h[i] = 1-2h[i]
            else:
                # First part: (1-h[i-lag]) for i in [lag, n-1]
                grad[lag:] += wm * dx * (1.0 - h[:n-lag])
                # Second part: -h[i+lag] for i in [0, n-1-lag]
                grad[:n-lag] -= wm * dx * h[lag:]
        else:
            p = -lag
            # c[m] = dx * sum_{j=p}^{n-1} h[j-p]*(1-h[j])
            # d/dh[i]: (1-h[i+p]) for i in [0, n-1-p]
            #           -h[i-p] for i in [p, n-1]
            grad[:n-p] += wm * dx * (1.0 - h[p:])
            grad[p:] -= wm * dx * h[:n-p]
    return grad, M + tau * np.log(exps.sum())

def smooth_solve(h0, tau, n):
    def obj(x):
        _, s = compute_grad_smoothing(x, tau)
        return s
    
    def jac(x):
        g, _ = compute_grad_smoothing(x, tau)
        return g
    
    target = n / 2.0
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 200, 'ftol': 1e-10})
    if res.success or res.fun < obj(h0):
        return project_feasible(res.x)
    return h0

def pair_moves(h, n_trials=1000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    h = h.copy()
    best = exact_c5(h)
    deltas = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for delta in deltas:
        improved = True
        while improved:
            improved = False
            for _ in range(n_trials):
                i, j = rng.choice(n, 2, replace=False)
                # Try both directions
                for sign in [1, -1]:
                    hi = h[i] + sign * delta
                    hj = h[j] - sign * delta
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_old_i, h_old_j = h[i], h[j]
                        h[i], h[j] = hi, hj
                        c = exact_c5(h)
                        if c < best - 1e-12:
                            best = c
                            improved = True
                        else:
                            h[i], h[j] = h_old_i, h_old_j
    return h, best

def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    # Build cumulative area
    areas = h_src * dx_src
    cum_areas = np.cumsum(np.insert(areas, 0, 0.0))
    # Source x-coordinates
    xs_src = np.linspace(0, 2, n_src + 1)
    # Target x-coordinates
    xs_tgt = np.linspace(0, 2, n_tgt + 1)
    # Interpolate cumulative area at target points
    cum_tgt = np.interp(xs_tgt, xs_src, cum_areas)
    # Target areas
    tgt_areas = np.diff(cum_tgt)
    # Convert to heights
    dx_tgt = 2.0 / n_tgt
    h_tgt = tgt_areas / dx_tgt
    return project_feasible(h_tgt)

def diverse_starts(h_parent, n, rng):
    starts = [h_parent.copy()]
    # Complement and reverse
    starts.append(project_feasible(1.0 - h_parent))
    starts.append(project_feasible(h_parent[::-1].copy()))
    starts.append(project_feasible((1.0 - h_parent)[::-1].copy()))
    # Noisy versions
    for sigma in [0.2, 0.1, 0.05]:
        noisy = h_parent + sigma * rng.randn(n)
        starts.append(project_feasible(noisy))
    # Binary quantization
    thresh = np.sort(h_parent)[n // 4]
    binary = (h_parent > thresh).astype(float)
    binary = project_feasible(binary)
    starts.append(binary)
    return starts

import numpy as np
from scipy.optimize import minimize
import time


def exact_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))


def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - result)[::-1]
            for i in order:
                room = 1.0 - result[i]
                if room >= diff - 1e-14:
                    result[i] += diff
                    break
                else:
                    result[i] = 1.0
                    diff -= room
        else:
            diff = -diff
            order = np.argsort(result)[::-1]
            for i in order:
                room = result[i]
                if room >= diff - 1e-14:
                    result[i] -= diff
                    break
                else:
                    result[i] = 0.0
                    diff -= room
    return result


def compute_smoothed(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(c)
    shifted = (c - M) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    exps = np.exp(shifted)
    W = exps.sum()
    w = exps / W
    S = M + tau * np.log(W)
    grad = np.zeros(n)
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        lag = m - center
        if lag == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wm * dx * (1.0 - h[:n - lag])
            grad[:n - lag] -= wm * dx * h[lag:]
        else:
            p = -lag
            grad[:n - p] += wm * dx * (1.0 - h[p:])
            grad[p:] -= wm * dx * h[:n - p]
    return S, grad


def smooth_solve(h0, tau, n, maxiter=150):
    target = n / 2.0

    def obj(x):
        s, _ = compute_smoothed(x, tau)
        return s

    def jac(x):
        _, g = compute_smoothed(x, tau)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x)
    except Exception:
        pass
    return h0


def pairwise_sweep(h, n_trials=2000, rng=None, use_all_pairs=False):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best = exact_c5(h)
    deltas = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for delta in deltas:
        improved = True
        iters = 0
        while improved and iters < 50:
            improved = False
            iters += 1
            if use_all_pairs and n <= 80:
                pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            else:
                pairs = [(int(rng.randint(n)), int(rng.randint(n)))
                         for _ in range(n_trials)]
            for i, j in pairs:
                if not use_all_pairs and i == j:
                    continue
                for sign in (1.0, -1.0):
                    hi = h[i] + sign * delta
                    hj = h[j] - sign * delta
                    if hi < -1e-12 or hi > 1.0 + 1e-12 or hj < -1e-12 or hj > 1.0 + 1e-12:
                        continue
                    hi = np.clip(hi, 0.0, 1.0)
                    hj = np.clip(hj, 0.0, 1.0)
                    old_i, old_j = h[i], h[j]
                    h[i], h[j] = hi, hj
                    c = exact_c5(h)
                    if c < best - 1e-12:
                        best = c
                        improved = True
                    else:
                        h[i], h[j] = old_i, old_j
    return h, best


def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    areas = h_src * dx_src
    cum = np.zeros(n_src + 1)
    cum[1:] = np.cumsum(areas)
    xs_src = np.linspace(0.0, 2.0, n_src + 1)
    xs_tgt = np.linspace(0.0, 2.0, n_tgt + 1)
    cum_tgt = np.interp(xs_tgt, xs_src, cum)
    tgt_areas = np.diff(cum_tgt)
    dx_tgt = 2.0 / n_tgt
    h_tgt = tgt_areas / dx_tgt
    return project_feasible(h_tgt)


def make_random_feasible(n, rng):
    x = rng.rand(n)
    return project_feasible(x)


def make_binary_from(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return project_feasible(b)
