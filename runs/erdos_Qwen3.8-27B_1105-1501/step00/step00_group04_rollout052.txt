    n = len(h)

def make_feasible(y, target=None):

def smooth_objective_and_grad(h, T):
    """Compute the smooth max (log-sum-exp) surrogate and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    L = len(c)  # = 2n-1
    
    # Stable log-sum-exp
    cmax = float(np.max(c))
    ex = np.exp((c - cmax) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        weights = np.zeros(L)
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = ex / S
    smooth_val = cmax + T * math.log(max(S, 1e-300))
    
    # Gradient
    g = np.zeros(n)
    center = n - 1
    
    # d=0 contribution
    wc = weights[center]
    g += wc * dx * (1.0 - 2.0 * h)
    
    # d > 0
    for d in range(1, n):
        wp = weights[center + d]
        wn = weights[center - d]
        m = n - d
        if m <= 0:
            break
        # Positive lag: c = dx * sum h[i+d]*(1-h[i])
        # d(c)/d(h[i]) = -wp*dx*h[i+d] for i < n-d
        # d(c)/d(h[j]) = wp*dx*(1-h[j-d]) for j >= d
        g[:m] += wp * dx * (-h[d:])  # -wp*dx*h[i+d]
        g[d:] += wp * dx * (1.0 - h[:m])  # wp*dx*(1-h[j-d])
        
        # Negative lag: c = dx * sum h[i]*(1-h[i+d])
        # d(c)/d(h[i]) = wn*dx*(1-h[i+d]) for i < n-d
        # d(c)/d(h[j]) = -wn*dx*h[j-d] for j >= d
        g[:m] += wn * dx * (1.0 - h[d:])  # wn*dx*(1-h[i+d])
        g[d:] += wn * dx * (-h[:m])  # -wn*dx*h[j-d]
    
    return smooth_val, g

    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = c.max()
    ex = np.exp((c - cmax) / T)
    S = ex.sum()
    if S < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = ex / S
    val = cmax + T * math.log(S) if S > 1e-300 else cmax
    
    g = np.zeros(n)
    center = n - 1
    g += w[center] * dx * (1 - 2*h)
    for d in range(1, n):
        m = n - d
        if m <= 0: break
        wp = w[center + d]
        wn = w[center - d]
        if wp + wn < 1e-15: continue
        g[:m] += wp * (-h[d:]) * dx + wn * (1 - h[d:]) * dx
        g[d:] += wp * (1 - h[:m]) * dx + wn * (-h[:m]) * dx
    return val, g
