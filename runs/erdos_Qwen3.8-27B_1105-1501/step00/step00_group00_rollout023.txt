def solve_tangent_lp(h0, n, active_lags, trust_radius, target_c5=None):
    """Solve the tangent LP subproblem to get a candidate direction"""
    dx = 2.0 / n
    corr = compute_correlation_values(h0)
    target = n / 2.0
    
    # Build active constraints
    constraints = []
    for m_idx in active_lags:
        ell = m_idx - (n - 1)
        g = compute_gradient(h0, ell)
        c_val = corr[m_idx]
        kappa = c_val - np.dot(g, h0)
        constraints.append((g, kappa, c_val))
    
    if not constraints:
        return None
    
    num_constraints = len(constraints)
    
    # Variables: h (n) + t (1) = n+1 total
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # For each constraint: g·h - t <= -kappa
    # => g·h + 0*... - 1*t <= -kappa
    A_ub = np.zeros((num_constraints, n + 1))
    b_ub = np.zeros(num_constraints)
    for i, (g, kappa, c_val) in enumerate(constraints):
        A_ub[i, 0:n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -kappa
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([target])
    
    # Bounds on h: [max(0, h0-trust), min(1, h0+trust)]
    lower = np.maximum(0.0, h0 - trust_radius)
    upper = np.minimum(1.0, h0 + trust_radius)
    bounds = list(zip(lower, upper)) + [(None, None)]  # t is free

            h_new = repair_feasible(h_new, n)

        pass
    return None

    """Area-preserving resolution transfer"""
    dx_from = 2.0 / n_from
    dx_to = 2.0 / n_to
    
    h_to = np.zeros(n_to)
    # For each target bin, compute overlap with source bins
    for k in range(n_to):
        t_left = k * dx_to
        t_right = (k + 1) * dx_to
        area = 0.0
        # Find overlapping source bins
        s_start = int(np.floor(t_left / dx_from))
        s_end = int(np.ceil(t_right / dx_from))
        s_start = max(0, s_start)
        s_end = min(n_from, s_end)
        for s in range(s_start, s_end):
            s_left = s * dx_from
            s_right = (s + 1) * dx_from
            overlap = min(t_right, s_right) - max(t_left, s_left)
            if overlap > 0:

        h_to[k] = area / dx_to

import numpy as np
import time
from scipy.optimize import linprog

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_feasible(h, n=None):
    if n is None:
        n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-15:
            h += diff * room / tr
        else:
            return h
    else:
        sp = h.copy()
        ts = sp.sum()
        if ts > 1e-15:
            h += diff * sp / ts
        else:
            return h
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        order = np.argsort(1.0 - h) if residual > 0 else np.argsort(h)
        for i in order:
            if residual > 0:
                a = min(residual, 1.0 - h[i])
                h[i] += a
                residual -= a
            else:
                a = min(-residual, h[i])
                h[i] -= a
                residual += a
            if abs(residual) < 1e-14:
                break
    return h

def grad_at_lag(h, ell):
    n = len(h)
    g = np.zeros(n)
    if ell >= 0:
        p = n - ell
        if p > 0:
            g[:p] += (1.0 - h[ell:])
            g[ell:] -= h[:p]
    else:
        p = -ell
        m = n - p
        if m > 0:
            g[p:] += (1.0 - h[:m])
            g[:m] -= h[p:]
    return g

def get_top_lags(h, n, count=18):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top = [center]
    idx = np.argsort(corr)[::-1]
    for i in idx:
        if int(i) != center and len(top) < count:
            top.append(int(i))
    return top

def solve_tangent_lp(h0, n, active_lags, trust):
    corr = np.correlate(h0, 1.0 - h0, mode="full")
    k = len(active_lags)
    A_ub = np.zeros((k, n + 1))
    b_ub = np.zeros(k)
    for idx, m in enumerate(active_lags):
        ell = m - (n - 1)
        g = grad_at_lag(h0, ell)
        c_val = corr[m]
        kappa = c_val - np.dot(g, h0)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -kappa
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h0 - trust)
    hi = np.minimum(1.0, h0 + trust)
    bounds = list(zip(lo, hi)) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return repair_feasible(res.x[:n].copy(), n)
    except Exception:
        pass
    return None

def pair_line_search(h, n, i, j, nsamp=15):

def local_optimize(h_in, n, time_budget, t_start, rng_seed=42):
    cur = h_in.copy()
    cur_c5 = evaluate_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    trust = 0.12
    stag = 0
    rng = np.random.RandomState(rng_seed)
    max_iters = 500
    for it in range(max_iters):
        if time.time() - t_start > time_budget:
            break
        if best_c5 < 0.375:
            break
        active = get_top_lags(cur, n, min(18, n // 2 + 1))
        cand = solve_tangent_lp(cur, n, active, trust)
        if cand is not None:
            c5n = evaluate_c5(cand)
            if c5n < best_c5 - 1e-12:
                best = cand.copy()
                best_c5 = c5n
                cur = cand.copy()
                cur_c5 = c5n
                stag = 0
            elif c5n < cur_c5 - 1e-12:
                cur = cand.copy()
                cur_c5 = c5n
                stag = 0
            elif c5n < cur_c5 + 5e-5:
                cur = cand.copy()
                cur_c5 = c5n
                stag += 1
            else:
                stag += 2
                trust *= 0.6
        else:
            trust *= 0.5
            stag += 1
        if it % 3 == 2 or stag > 4:
            top5 = get_top_lags(cur, n, 5)
            avg_grad = np.zeros(n)
            for m in top5:
                ell = m - (n - 1)

            score = np.outer(np.ones(n), avg_grad) - np.outer(avg_grad, np.ones(n))
            flat = score.ravel()
            top_idx = np.argsort(flat)[::-1][:60]
            pair_found = False
            for pi in top_idx:
                if time.time() - t_start > time_budget * 0.95:
                    break
                i = int(pi // n)
                j = int(pi % n)
                if i == j or score[i, j] < 0.03:
                    break
                new_h, new_c5 = pair_line_search(cur, n, i, j, nsamp=13)
                if new_c5 < best_c5 - 1e-12:
                    best = new_h.copy()
                    best_c5 = new_c5

                    cur = new_h.copy()
                    cur_c5 = new_c5
                    pair_found = True
                    stag = 0
            if not pair_found:
                stag += 1
        if trust < 0.004:

                nh = repair_feasible(cur + pert, n)
                nc5 = evaluate_c5(nh)
                if nc5 < cur_c5 + 5e-4:
                    cur = nh
                    cur_c5 = nc5

            stag = 0

def resolution_transfer(h_from, n_from, n_to):
    dx_f = 2.0 / n_from
    dx_t = 2.0 / n_to
    h_to = np.zeros(n_to)
    for k in range(n_to):
        tl = k * dx_t
        tr = (k + 1) * dx_t
        area = 0.0
        s0 = max(0, int(tl / dx_f))
        s1 = min(n_from, int(np.ceil(tr / dx_f)))
        for s in range(s0, s1):
            sl = s * dx_f
            sr = (s + 1) * dx_f
            ov = min(tr, sr) - max(tl, sl)
            if ov > 0:
                area += h_from[s] * ov
        h_to[k] = area / dx_t
    h_to = np.clip(h_to, 0.0, 1.0)
    h_to = repair_feasible(h_to, n_to)
    return h_to

def run(seed=42, budget_s=1100, **kwargs):
