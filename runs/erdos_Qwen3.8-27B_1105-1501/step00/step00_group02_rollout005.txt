from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def safe_sigmoid(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -500, 500)))

def logit(x):
    x = np.clip(x, 1e-10, 1 - 1e-10)
    return np.log(x / (1 - x))

def raw_to_h(z, target_sum):
    n = len(z)
    lo, hi = -100.0, 100.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(safe_sigmoid(z + mid))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    tau = 0.5 * (lo + hi)
    return safe_sigmoid(z + tau)

def compute_smooth_obj_and_grad(z, target_sum, beta):
    n = len(z)
    h = raw_to_h(z, target_sum)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full") * dx
    n_corr = len(corr)

    bc = beta * corr
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)
    sum_exp = np.sum(exp_bc)
    obj = (max_bc + np.log(sum_exp)) / beta
    
    # Weights
    w = exp_bc / sum_exp
    
    # Gradient of corr wrt h
    # For each m in 0..2n-2, s = m - (n-1)
    # corr[m] = sum over valid j of h[j+s]*b[j] or h[j]*b[j-s] depending on sign of s

    # corr[m] with s = m-(n-1):

    # If s < 0: corr[m] = dx * sum_{j=-s}^{n-1} h[j] * (1-h[j-s])
    #           = dx * sum_{j=0}^{n-1+s} h[j] * (1-h[j-s])  [reindex]

    #   h[i] appears as h[j+s] when j = i-s (if valid: 0 <= i-s <= n-1-s, i.e., s <= i <= n-1)
    #     contribution: dx * (1-h[i-s])
    #   h[i] appears in (1-h[j]) when j = i (if valid: 0 <= i <= n-1-s)

Let me also think about what "pairwise line search" means more concretely for improving C5.

C5 = max_m F_m where F_m = corr[m] * dx

To reduce C5, we want to reduce the maximum F_m. The active set is the m values where F_m is close to the maximum. A pairwise transfer (i, j, t) changes F_m for all m. The key insight is that we want to find transfers that reduce the active F_m values without increasing others too much.

For a simple approach:
1. Identify the top few active m values
2. For pairs (i, j), compute how the transfer affects the active m values
3. Choose the pair and t that most reduces the max

But this is complex. A simpler approach that often works well:
1. For each pair (i, j) with h_i > h_j (or random pairs), do a 1D line search
2. Accept if C5 improves
3. Repeat

For n=47, there are 47*46/2 = 1081 pairs. With a few line search evaluations per pair, this is fast.

    if diff > 0:

        for idx in order:
            if diff <= 0:

                diff -= can_add

        room = k
