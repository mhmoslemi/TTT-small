            if nt % 2 == 0 and nt >= 10:
                n_even_candidates.append(nt)
    if not n_even_candidates:
        n_even_candidates = [94, 96, 100]

    for ne in n_even_candidates:
        if time.time() > t_limit:
            break
        dl = min(time.time() + 120, t_limit)
        h_bin = _binary_init(ne, rng, h_parent)
        h_res, sc_res = _sa_binary(h_bin, rng, dl)
        sc_res = _score(h_res)
        _update(h_res, sc_res)

    # Phase 1b: General SA on parent n (if odd)
    if n_parent > 0 and n_parent % 2 == 1 and h_parent is not None:
        if time.time() < t_limit:
            dl = min(time.time() + 120, t_limit)
            h_res, sc_res = _sa_general(h_parent, rng, dl)
            sc_res = _score(h_res)
            _update(h_res, sc_res)

    # Phase 2: Try random binary starts on a few n values
    n_retry_list = list(set(n_even_candidates + [best_n if best_n % 2 == 0 else best_n + 1]))
    n_retry_list = sorted([x for x in n_retry_list if 10 <= x <= 200 and x % 2 == 0])
    for nr in n_retry_list:
        if time.time() > t_limit - 60:
            break
        dl = min(time.time() + 60, t_limit - 30)
        for restart in range(3):
            if time.time() > dl:
                break
            h_rand = _random_binary(nr, rng)
            h_res, sc_res = _sa_binary(h_rand, rng, min(time.time() + 15, dl))
            sc_res = _score(h_res)
            _update(h_res, sc_res)

    # Phase 3: Try resampling best to nearby n and optimize
    if best_h is not None:
        n_best = best_n
        for dn in [-4, -2, 2, 4, 6, 8]:
            nt = n_best + dn
            if nt < 10 or nt > 200 or time.time() > t_limit - 30:
                continue
            if nt % 2 != 0:
                continue
            dl = min(time.time() + 40, t_limit - 20)
            h_try = _resample(best_h, nt)
            h_bin_try = _binary_init(nt, rng, best_h)
            h_res, sc_res = _sa_binary(h_bin_try, rng, min(time.time() + 20, dl))
            sc_res = _score(h_res)
            _update(h_res, sc_res)
            if time.time() < dl:
                h_res2, sc_res2 = _sa_general(h_try, rng, dl)
                sc_res2 = _score(h_res2)
                _update(h_res2, sc_res2)

    # Phase 4: Greedy polish on best
    if best_h is not None and best_n % 2 == 0:
        h_pol, sc_pol = _greedy_binary(best_h)
        sc_pol = _score(h_pol)
        _update(h_pol, sc_pol)
        # Short SA after greedy
        if time.time() < t_limit - 10:
            h_res, sc_res = _sa_binary(h_pol, rng, min(time.time() + 60, t_limit))
            sc_res = _score(h_res)
            _update(h_res, sc_res)

    # Phase 4b: If best is odd-n, try converting to even
    if best_h is not None and best_n % 2 != 0:
        for ne in [best_n - 1, best_n + 1]:
            if ne < 10 or time.time() > t_limit - 10:
                continue
            h_conv = _binary_init(ne, rng, best_h)
            _update(h_conv, _score(h_conv))
            if time.time() < t_limit - 5:
                h_res, sc_res = _sa_binary(h_conv, rng, min(time.time() + 40, t_limit))
                sc_res = _score(h_res)
                _update(h_res, sc_res)

    # Final: ensure valid output
    if best_h is None:
        n_fb = 100
        best_h = np.zeros(n_fb)
        best_h[rng.choice(n_fb, n_fb // 2, replace=False)] = 1.0
        best_sc = _score(best_h)
        best_n = n_fb
    else:
        best_h = _project_to_sum(np.clip(best_h, 0.0, 1.0), best_n / 2.0)
        best_sc = _score(best_h)

    # Final evaluation with harness function if available
    try:
        eval_sc = evaluate_erdos_solution(best_h)
        if eval_sc < best_sc:
            best_sc = eval_sc
    except Exception:
        pass

    return (best_h, best_sc, best_n)
