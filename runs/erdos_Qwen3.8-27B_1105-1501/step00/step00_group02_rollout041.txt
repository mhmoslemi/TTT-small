def repair_sum(y,T=None):
    y=np.asarray(y,dtype=float).copy()
    n=y.size
    if T is None: T=0.5*n
    if n==0: return y
    y=np.clip(y,0.0,1.0)
    residual=T-y.sum()
    for _ in range(20):
       if abs(residual) < 1e-15: break
       if residual > 0:
          slack=1.0-y
          sl=slack.sum()
          if sl <= 0: break
          add = residual * slack / sl
          # avoid exceeding slack: add = np.minimum(add, slack)
          # but proportional might exceed if rounding
          y = np.clip(y + add, 0.0, 1.0)
       else:
          v=y
          vs=v.sum()
          if vs <= 0: break
          dec = (-residual) * v / vs
          y = np.clip(y - dec, 0.0, 1.0)
       residual=T-y.sum()
    # final single adjustment if small
    residual=T-y.sum()
    if abs(residual) > 0 and abs(residual) < 1e-7:

          add=min(residual, 1.0-y[idx])
          y[idx]+=add

          idx=int(np.argmax(y))
          dec=min(-residual, y[idx])
          y[idx]-=dec
    return y

def subgradient(h, vals):
    n=h.size; dx=2.0/n
    if size<2: return np.zeros(n)
    maxv=float(np.max(vals)); minv=float(np.min(vals))
    spread=maxv-minv
    tau=max(1e-9, 0.08*spread)
    threshold=maxv-tau
    w=np.maximum(vals-threshold,0.0)
    if w.sum() <= 1e-15:
       K=min(12, vals.size)
       idx=np.argpartition(vals, -K)[-K:]
       w=np.zeros_like(vals); w[idx]=vals[idx]
    wsum=float(w.sum())
    if wsum <= 0: return np.zeros(n)
    w = w / wsum
    S=np.zeros(n)
    for r in range(vals.size):
       weight=float(w[r])
       if weight <= 1e-12: continue
       d = r - (n-1)
       if d >=0:
          if d == 0:
             S += weight*dx*(1.0 - 2.0*h)
          else:
             S[d:] += weight*dx*(1.0 - h[:-d])
             S[:-d] -= weight*dx*h[d:]
       else:
          e=-d
          S[e:] += weight*dx*(1.0 - h[:n-e])
          S[:n-e] -= weight*dx*h[e:]
    return S

             if p != q:

                if key not in seen:
                   seen.add(key); pairs.append((p,q))
    # random
    src_all=np.flatnonzero(h > 1e-5)
    sink_all=np.flatnonzero(h < 1.0 - 1e-5)

       for _ in range(n_random):
          p=int(src_all[rng.integers(src_all.size)])
          q=int(sink_all[rng.integers(sink_all.size)])
          if p==q: continue

          if key not in seen:
             seen.add(key); pairs.append((p,q))

def is_feasible(h, tol=1e-5):
    if h.size < 1:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -tol or float(np.max(h)) > 1.0 + tol:
        return False
    return abs(float(np.sum(h)) - 0.5 * h.size) <= tol


def repair_sum(y, T=None):
    y = np.asarray(y, dtype=float).copy().ravel()
    n = y.size
    if n == 0:
        return y
    if T is None:
        T = 0.5 * n
    y = np.clip(y, 0.0, 1.0)
    residual = float(T - y.sum())

    for _ in range(20):
        if abs(residual) < 1e-15:
            break
        if residual > 0.0:
            slack = 1.0 - y
            sl = float(slack.sum())
            if sl <= 0.0:
                break
            y = np.clip(y + residual * slack / sl, 0.0, 1.0)
        else:
            vs = float(y.sum())
            if vs <= 0.0:
                break
            y = np.clip(y - (-residual) * y / vs, 0.0, 1.0)
        residual = float(T - y.sum())

    if abs(residual) > 1e-8:
        if residual > 0.0:
            order = np.argsort(-(1.0 - y))
            rem = residual
            for i in order:
                add = min(rem, 1.0 - y[i])
                if add <= 0.0:
                    continue
                y[i] += add
                rem -= add
                if rem <= 1e-14:
                    break
        else:
            order = np.argsort(-y)
            rem = -residual
            for i in order:
                dec = min(rem, y[i])
                if dec <= 0.0:
                    continue
                y[i] -= dec
                rem -= dec
                if rem <= 1e-14:
                    break

    residual = float(T - y.sum())
    if abs(residual) < 1e-7 and residual != 0.0:
        if residual > 0.0:
            i = int(np.argmax(1.0 - y))
            y[i] = min(1.0, y[i] + residual)
        else:
            i = int(np.argmax(y))
            y[i] = max(0.0, y[i] + residual)

    return y


def project_simplex_box(z, T):
    z = np.asarray(z, dtype=float).copy().ravel()
    n = z.size
    if n == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.5)
    if T < 0.0:
        T = 0.0
    if T > float(n):
        T = float(n)

    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(75):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    return repair_sum(y, T)

def interpolate_h(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float).ravel()
    n0 = h_old.size
    n_new = int(n_new)
    if n_new <= 0:
        return np.zeros(0, dtype=float)
    if n0 == 0:
        return project_simplex_box(np.full(n_new, 0.5), 0.5 * n_new)
    if n_new == n0:

    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    new = np.interp(x1, x0, h_old, left=h_old[0], right=h_old[-1])

def evaluate_ts(h, h_mod, oh_mod, p, q, ts, best_c, best_t, dx):
    hp = h[p]
    hq = h[q]
    for t in ts:
        t = float(t)
        newp = hp - t
        if newp < -1e-10 or newp > 1.0 + 1e-10:
            continue
        if newp < 0.0:
            newp = 0.0
        elif newp > 1.0:
            newp = 1.0
        newq = hq + t
        if newq < -1e-10 or newq > 1.0 + 1e-10:
            continue
        if newq < 0.0:
            newq = 0.0
        elif newq > 1.0:
            newq = 1.0
        h_mod[p] = newp
        oh_mod[p] = 1.0 - newp
        h_mod[q] = newq
        oh_mod[q] = 1.0 - newq
        c = float(np.max(np.correlate(h_mod, oh_mod, mode="full")) * dx)
        if c < best_c - 1e-15:
            best_c = c
            best_t = t
    return best_c, best_t


def line_search_pair(h, p, q, current_C, dx, m_coarse=17, m_fine=9):

    best_t = 0.0

def apply_transfer(h, p, q, t):
    out = h.copy()
    vp = h[p] - t
    vq = h[q] + t
    if vp < 0.0:
        vp = 0.0
    elif vp > 1.0:

    if vq < 0.0:
        vq = 0.0
    elif vq > 1.0:
        vq = 1.0

    return out


def get_all_pairs(n):

    if n not in _PAIR_CACHE:
        lst = []
        for p in range(n - 1):
            for q in range(p + 1, n):

def subgradient(h, vals):
    n = h.size
    if n < 2 or vals.size == 0:
        return np.zeros(n, dtype=float)
    dx = 2.0 / n
    if not np.all(np.isfinite(vals)):
        return np.zeros(n, dtype=float)
    maxv = float(np.max(vals))
    minv = float(np.min(vals))
    spread = maxv - minv
    tau = max(1e-9, 0.08 * spread)
    threshold = maxv - tau
    w = np.maximum(vals - threshold, 0.0)
    if float(w.sum()) <= 1e-15:
        K = min(16, vals.size)
        if K >= vals.size:
            idx = np.arange(vals.size)
        else:
            idx = np.argpartition(vals, -K)[-K:]
        w = np.zeros_like(vals)
        w[idx] = vals[idx]
    ws = float(w.sum())
    if ws <= 0.0:
        return np.zeros(n, dtype=float)
    w = w / ws
    S = np.zeros(n, dtype=float)
    for r in range(vals.size):
        weight = float(w[r])
        if weight <= 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            if d == 0:
                S += weight * dx * (1.0 - 2.0 * h)
            else:
                S[d:] += weight * dx * (1.0 - h[:-d])
                S[:-d] -= weight * dx * h[d:]
        else:
            e = -d
            S[e:] += weight * dx * (1.0 - h[:n - e])
            S[:n - e] -= weight * dx * h[e:]
    return S


def gradient_pairs(h, S, rng, n, top=10, n_random=60):
    pairs = []
    seen = set()
    src = []
    sink = []

    if n > 0:
        order_high = np.argsort(S)[::-1]
        for ii in order_high:
            i = int(ii)
            if h[i] > 1e-5:
                src.append(i)
            if len(src) >= top:
                break

        order_low = np.argsort(S)
        for ii in order_low:
            i = int(ii)
            if 1.0 - h[i] > 1e-5:
                sink.append(i)
            if len(sink) >= top:
                break

        for p in src:
            for q in sink:
                if p == q:
                    continue
                key = (p, q) if p < q else (q, p)
                if key not in seen:
                    seen.add(key)
                    pairs.append((p, q))

    src_all = np.flatnonzero(h > 1e-5)
    sink_all = np.flatnonzero(h < 1.0 - 1e-5)
    if src_all.size > 0 and sink_all.size > 0:
        for _ in range(n_random):
            p = int(src_all[rng.integers(src_all.size)])
            q = int(sink_all[rng.integers(sink_all.size)])
            if p == q:
                continue
            key = (p, q) if p < q else (q, p)
            if key not in seen:
                seen.add(key)
                pairs.append((p, q))
    return pairs

            no_improve = 0
        else:
            no_improve += 1

        if C > best_C + 0.01 and no_improve >= 10 and pass_count % 3 == 0:
            h = best_h.copy()
            C = best_C
            no_improve = 0

    if is_feasible(best_h, 1e-7):
        best_h = repair_sum(best_h)
    else:
        best_h = repair_sum(best_h)
    best_C = objective_raw(best_h)

    if (not is_feasible(best_h, 1e-5)) or (not np.isfinite(best_C)):
        h = repair_sum(h)
        C = objective_raw(h)
        if is_feasible(h, 1e-5) and np.isfinite(C):
            best_h = h
            best_C = C
        else:
            fallback_n = max(2, int(n))
            best_h = random_feasible(fallback_n, rng)
            best_C = objective_raw(best_h)

    return best_h, float(best_C), int(best_h.size)


def better_solution(c1, n1, c2, n2):
    if not np.isfinite(c1):
        return False
    if not np.isfinite(c2):
        return True
    if c1 < c2 - 1e-10:
        return True
    if abs(c1 - c2) <= 1e-10 and n1 < n2:
        return True
    return False


def candidate_ns(base_n):
    base_n = int(base_n)
    limit = max(256, base_n + 40)
    if limit > 400:
        limit = 400

    ns = set()
    for delta in range(-10, 25, 2):
        nn = base_n + delta
        if 16 <= nn <= limit:
            ns.add(nn)

    fixed = [24, 28, 32, 36, 40, 45, 48, 52, 56, 60, 64, 72, 80, 90,
             96, 112, 128, 144, 160, 192, 224]
    for nn in fixed:
        if 16 <= nn <= limit:
            ns.add(nn)

    if base_n in ns:
        ns.remove(base_n)

    scored = []
    for nn in ns:
        scored.append((abs(nn - base_n), nn))
    scored.sort()
    result = []
    for item in scored:
        result.append(int(item[1]))
    return result[:24]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    if budget <= 0.5:
        budget = 1.0
    t_end = t0 + 0.97 * budget
    rng = np.random.default_rng(seed)

    parent = get_parent()
    if parent is None:
        base_n = 47
        best_h = random_feasible(base_n, rng)
    else:
        best_h = parent.copy()
        base_n = int(best_h.size)
    best_C = objective_raw(best_h)

    if (not is_feasible(best_h, 1e-5)) or (not np.isfinite(best_C)):
        base_n = 47
        best_h = random_feasible(base_n, rng)
        best_C = objective_raw(best_h)

    if t_end - time.time() < 3.0:
        best_h = repair_sum(best_h)
        best_C = objective_raw(best_h)
        return best_h, float(best_C), int(best_h.size)

    base_time = min(t_end, t0 + max(20.0, 0.30 * budget))
    if time.time() < base_time:
        h, C, _ = local_polish(best_h, time.time(), base_time, rng)
        if h is not None and better_solution(C, h.size, best_C, best_h.size):
            best_h, best_C = h, C

    if time.time() < t_end - 40.0 and best_h.size <= 128:
        roll_phase_end = min(t_end, time.time() + max(15.0, 0.10 * budget))
        scored = []
        counter = 0
        for shift in [1, -1, 2, -2, 3, -3]:
            arr = np.roll(best_h, shift)
            if not is_feasible(arr, 1e-7):
                arr = repair_sum(arr)
            c0 = objective_raw(arr)
            if np.isfinite(c0):
                scored.append((c0, counter, arr))
                counter += 1
        arr = perturb_h(best_h, 0.03, rng)
        c0 = objective_raw(arr)
        if np.isfinite(c0):
            scored.append((c0, counter, arr))
            counter += 1

        if scored:
            scored.sort()
            max_starts = min(len(scored), 4)
            per = max(8.0, (roll_phase_end - time.time()) / float(max_starts))
            for item in scored[:max_starts]:
                st = time.time()
                if st >= roll_phase_end:
                    break
                arr = item[2]
                en = min(roll_phase_end, st + max(10.0, per))
                h, C, _ = local_polish(arr, st, en, rng)
                if h is not None and better_solution(C, h.size, best_C, best_h.size):
                    best_h, best_C = h, C

    if time.time() < t_end - 20.0:
        final_reserve = max(25.0, 0.10 * budget)
        var_end = t_end - final_reserve
        if var_end > time.time() + 8.0:
            ns = candidate_ns(best_h.size)
            scored = []
            counter = 0
            for nn in ns:
                h0 = interpolate_h(best_h, nn)
                if h0.size != int(nn):
                    continue
                c0 = objective_raw(h0)
                if np.isfinite(c0):
                    scored.append((c0 + 1e-7 * int(nn), counter, int(nn), h0))
                    counter += 1

            if scored:
                scored.sort()
                remaining_count = max(4, int((var_end - time.time()) / 30.0))
                if remaining_count > 16:
                    remaining_count = 16
                remaining_count = min(len(scored), remaining_count)

                for idx in range(remaining_count):
                    st = time.time()
                    if st >= var_end:
                        break
                    left = var_end - st
                    if left < 5.0:
                        break
                    left_count = max(1, remaining_count - idx)
                    alloc = left / float(left_count)
                    if alloc > 120.0:
                        alloc = 120.0
                    en = st + alloc
                    if en > var_end:
                        en = var_end
                    if en - st < 5.0:
                        en = var_end
                    nn, h0 = scored[idx][2], scored[idx][3]
                    h, C, _ = local_polish(h0, st, en, rng)
                    if h is not None and better_solution(C, h.size, best_C, best_h.size):
                        best_h, best_C = h, C

    if time.time() < t_end - 0.5:
        h, C, _ = local_polish(best_h, time.time(), t_end, rng, fine=True)
        if h is not None and better_solution(C, h.size, best_C, best_h.size):
            best_h, best_C = h, C

    best_h = repair_sum(best_h)
    best_C = objective_raw(best_h)

    if (not is_feasible(best_h, 1e-5)) or (not np.isfinite(best_C)):
        if parent is not None and is_feasible(parent, 1e-5):
            best_h = parent.copy()
            best_C = objective_raw(best_h)
        else:
            best_h = random_feasible(max(2, base_n), rng)
            best_C = objective_raw(best_h)

    try:
        ext = evaluate_erdos_solution(best_h)
        if isinstance(ext, tuple) or isinstance(ext, list):
            ext = ext[0]
        ext = float(ext)
        if np.isfinite(ext) and ext > best_C + 1e-12:
            best_C = ext
    except Exception:
        pass

    return best_h, float(best_C), int(best_h.size)
