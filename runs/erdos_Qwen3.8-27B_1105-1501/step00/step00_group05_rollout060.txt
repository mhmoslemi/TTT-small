def gradient_for_lag(h, l):
    """Compute dR(l)/dh for lag l. l ranges from -(N-1) to N-1."""
    n = len(h)
    grad = np.zeros(n)
    # First term: (1-h[j-l]) if 0 <= j-l < n
    # j-l >= 0 → j >= l; j-l < n → j < n+l
    # So j ranges from max(0, ceil(l)) to min(n-1, n+l-1) = min(n-1, n-1+l if l>=0 else n+l-1)
    # For l >= 0: j from l to n-1 (since j < n+l is always true for j < n when l >= 0)
    #   Actually j < n+l means j <= n+l-1. Since j <= n-1 < n+l-1 for l >= 0, this is automatic.
    #   So j from l to n-1.
    #   h[j-l] ranges from h[0] to h[n-1-l]
    if l >= 0:

        grad[idx] += 1.0 - h[idx - l]  # h[j-l] for j in [l, n-1] → h[0] to h[n-1-l]
    else:
        # j - l = j + |l| >= 0 always (j >= 0)
        # j - l < n → j < n + l = n - |l| → j <= n - |l| - 1

        grad[idx] += 1.0 - h[idx - l]  # h[j+l'] for j in [0, n-1-l'] → h[l'] to h[n-1]

def project_onto_constraint(h, n):
    """Project h onto {v: 0<=v<=1, sum(v)=n/2}."""
    target = n / 2.0
    # Binary search for tau such that sum(clamp(h - tau, 0, 1)) = target
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)
