        best_n = len(h_init)
    
    # Phase 1: Optimize from initial with annealing
    if best_h is not None and remaining() > 100:
        max_t = min(remaining() - 80, 300)
        h_opt, c5 = optimize_annealing(best_h, best_n, max_t)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Phase 2: Try SLSQP for further refinement
    if best_h is not None and remaining() > 60:
        h_opt, c5 = optimize_slsqp(best_h, best_n, max_iter=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Phase 3: Try different n values
    n_trials = [best_n]
    for n_try in [41, 51, 63, 75, 83, 91]:
        if n_try not in n_trials:
            n_trials.append(n_try)
    
    for n_try in n_trials:
        if remaining() < 60:
            break
        if n_try == best_n:
            h_start = best_h.copy()
        elif h_init is not None and n_try == len(h_init):
            h_start = h_init.copy()
        elif best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_start = np.interp(x_new, x_old, best_h)
        else:
            h_start = np.full(n_try, 0.5)
        
        target_sum = n_try / 2.0
        h_start = project_to_constraint(h_start, target_sum)
        
        max_t = min(remaining() - 50, 200)
        h_opt, c5 = optimize_annealing(h_start, n_try, max_t)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n_try
    
    # Phase 4: Random restarts
    restart_count = 0
    while remaining() > 40 and restart_count < 15:
        restart_count += 1
        n = best_n
        h_rand = np.random.uniform(0.1, 0.9, n)
        h_rand = project_to_constraint(h_rand, n / 2.0)
        
        max_t = min(remaining() - 30, 60)
        h_opt, c5 = optimize_annealing(h_rand, n, max_t)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Phase 5: Final refinement with multiple methods
    if best_h is not None and remaining() > 20:
        h_opt, c5 = optimize_slsqp(best_h, best_n, max_iter=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        
        if remaining() > 15:
            h_opt, c5 = optimize_annealing(best_h, best_n, min(remaining() - 5, 30))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
    
    # Final: ensure constraint is satisfied
    if best_h is not None:
        best_h = project_to_constraint(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)
    
    if best_h is None:
        best_h = np.full(63, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 63
    
    return (best_h, best_c5, best_n)
