from scipy.optimize import minimize_scalar

def score(h):
    """Compute C5 for a feasible h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def internal_correlations(h):
    """Compute all 2n-1 correlation values (same as np.correlate(h, 1-h, 'full'))."""
    n = len(h)
    L = 2 * n - 1
    F = np.zeros(L)
    g = 1.0 - h
    for idx in range(L):
        k = idx - (n - 1)
        if k >= 0:
            length = n - k
            if length > 0:
                F[idx] = np.dot(h[k:], g[:length])
        else:
            length = n + k
            if length > 0:
                F[idx] = np.dot(h[:length], g[-k:])
    return F

def project_feasible(z, target):
    """Project z onto {0 <= v <= 1, sum(v) = target}."""
    n = len(z)
    v = np.clip(np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    e = np.sum(v) - target
    if e > 0:
        # Need to reduce sum by e
        order = np.argsort(-v)  # largest first
        for i in order:
            if e <= 0:
                break
            delta = min(v[i], e)
            v[i] -= delta
            e -= delta
    elif e < 0:
        # Need to increase sum by -e
        need = -e
        order = np.argsort(v)  # smallest first (most room)
        for i in order:
            if need <= 0:
                break
            room = 1.0 - v[i]
            delta = min(room, need)
            v[i] += delta
            need -= delta
    return v

def soft_max_gradient(h, alpha):
    """Compute smoothed objective value and gradient."""
    n = len(h)
    dx = 2.0 / n
    F = internal_correlations(h)
    L = len(F)
    Y = dx * F  # scaled by dx
    mx = np.max(Y)
    # Stable log-sum-exp
    Z = np.exp(alpha * (Y - mx))
    S = np.sum(Z)
    U = mx + np.log(S) / alpha
    w = Z / S  # weights
    
    # Gradient
    D = np.zeros(n)
    g = 1.0 - h
    for idx in range(L):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        wi = w[idx] * dx
        if k >= 0:
            length = n - k
            if length > 0:
                D[k:] += wi * g[:length]
                D[:length] -= wi * h[k:]
        else:
            length = n + k
            if length > 0:
                D[:length] += wi * g[-k:]
                D[-k:] -= wi * h[:length]
    D -= np.mean(D)  # zero-sum for mass constraint
    return U, D

    p = rng.standard_normal(n)
    p -= np.mean(p)
    if np.max(np.abs(p)) > 0:

def resample_to_dim(h_old, n_new):

import numpy as np
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def _project(z, target):
    v = np.clip(np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    e = float(np.sum(v)) - target
    if e > 1e-13:
        order = np.argsort(-v)
        for i in order:
            if e <= 0: break
            d = min(v[i], e)
            v[i] -= d
            e -= d
    elif e < -1e-13:
        need = -e
        order = np.argsort(v)
        for i in order:
            if need <= 0: break
            room = 1.0 - v[i]
            d = min(room, need)
            v[i] += d
            need -= d
    return v

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    Y = dx * corr
    L = len(Y)
    mx = float(np.max(Y))
    sh = np.clip(alpha * (Y - mx), -700, 700)
    Z = np.exp(sh)
    S = float(np.sum(Z))
    if S > 1e-300:
        w = Z / S
    else:
        w = np.zeros(L)
        w[np.argsort(-Y)[:5]] = 0.2
    D = np.zeros(n)
    g = 1.0 - h
    n1 = n - 1
    for idx in range(L):
        wi = w[idx]
        if wi < 1e-15: continue
        k = idx - n1
        wd = wi * dx
        if k >= 0:
            ln = n - k
            D[k:] += wd * g[:ln]
            D[:ln] -= wd * h[k:]
        else:
            ln = n + k
            D[:ln] += wd * g[-k:]
            D[-k:] -= wd * h[:ln]
    D -= np.mean(D)
    return D

def _rdir(n, rng):
    p = rng.standard_normal(n)
    p -= p.mean()
    m = float(np.max(np.abs(p)))
    if m > 1e-15: p /= m
    return p

def _linesearch(h0, C0, d, target, dgain, dl):
    am = float(np.max(np.abs(d)))
    if am < 1e-15: return h0, C0
    ph = d / am
    best_C = C0
    best_h = h0.copy()
    for sgn in (1.0, -1.0):
        for s in (0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0, 3.0):
            if time.time() > dl: return best_h, best_C
            x = _project(h0 + sgn * s * ph, target)
            C = _score(x)
            if C < best_C - dgain:
                best_C = C
                best_h = x
    return best_h, best_C

def _resample(hOld, nNew):
    nO = len(hOld)
    if nNew == nO: return hOld.copy()
    dO = 2.0 / nO
    dN = 2.0 / nNew
    xO = (np.arange(nO) + 0.5) * dO
    xN = (np.arange(nNew) + 0.5) * dN
    return np.interp(xN, xO, hOld)

def _optimize(h, C, target, dl, niter, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    bh = h.copy()
    bC = C
    alphas = [4, 10, 25, 60, 150, 400, 800, 1500]
    for a in alphas:
        stuck = 0
        for it in range(niter):
            if time.time() > dl: return bh, bC
            grad = _softmax_grad(bh, a)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-14:
                break
            nh, nC = _linesearch(bh, bC, grad, target, 1e-13, dl)
            if nC < bC - 1e-14:
                bh, bC = nh, nC
                stuck = 0
            else:
                stuck += 1
                if stuck > 5: break
            if it % 6 == 5:
                rd = _rdir(n, rng)
                nh, nC = _linesearch(bh, bC, rd, target, 1e-13, dl)
                if nC < bC - 1e-14:
                    bh, bC = nh, nC
                    stuck = 0
    return bh, bC
