def eval_c5(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_gradient(h, n, active_indices):
    """Compute subgradient of max correlation at h."""
    grad = np.zeros(n)
    for idx in active_indices:
        L = idx - (n - 1)
        if L >= 0:
            m = n - L
            if m > 0:
                grad[L:] += (1.0 - h[:m])
                grad[:m] += -h[L:]
        else:
            M = -L
            m = n - M
            if m > 0:
                grad[:m] += (1.0 - h[M:])
                grad[M:] += -h[:m]
    grad /= len(active_indices)
    return grad

    t_min = max(-h[i], -(1.0 - h[j]))

    # Lag L corresponds to index L + (n-1) in corr
    a = np.zeros(2*n - 1)
    b = np.zeros(2*n - 1)

        delta2 = 0.0
