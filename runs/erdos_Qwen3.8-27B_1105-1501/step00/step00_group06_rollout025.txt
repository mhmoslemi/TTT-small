```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_mass(v, target):
    """Project v onto {0<=h<=1, sum(h)=target}"""
    n = len(v)
    v = np.clip(v, 0, 1)
    # Adjust to hit target sum
    diff = target - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    # Distribute the difference
    # Find indices where we can adjust
    if diff > 0:
        # Need to increase sum, find indices that are < 1
        mask = v < 1.0
    else:
        # Need to decrease sum, find indices that are > 0
        mask = v > 0.0
    
    if not np.any(mask):
        # Fallback: uniform
        return np.full(n, target/n)
    
    n_mask = np.sum(mask)
    adj = diff / n_mask
    v[mask] = np.clip(v[mask] + adj, 0, 1)
    
    # Second pass if needed
    diff2 = target - np.sum(v)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            mask2 = v < 1.0
        else:
            mask2 = v > 0.0
        if np.any(mask2):
            v[mask2] = np.clip(v[mask2] + diff2/np.sum(mask2), 0, 1)
    
    # Final exact adjustment
    diff3 = target - np.sum(v)
    if abs(diff3) > 1e-12:
        # Find one element to adjust
        if diff3 > 0:
            idx = np.argmin(1.0 - v)
            v[idx] = min(1.0, v[idx] + diff3)
        else:
            idx = np.argmax(v)
            v[idx] = max(0.0, v[idx] + diff3)
    
    return v

    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 70
        h0 = np.full(n, 0.5)
    
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    # Ensure feasibility
    h0 = project_box_mass(h0, target_sum)
    best_h = h0.copy()
    best_score = score_h(best_h)
    
    # Method 1: SLSQP optimization
    def obj(h):
        return score_h(h)
    
    def obj_grad(h):
        # Subgradient of the max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        active = np.where(np.abs(corr - max_val) < 1e-8)[0]
        
        grad = np.zeros(n)
        for j in active:
            k = j - (n - 1)
            g = np.zeros(n)
            if k == 0:
                g[:] = 1.0 - 2.0 * h
            elif k > 0:
                g[k:] += (1.0 - h[:n-k])
                g[:n-k] -= h[k:]
            else:
                t = -k
                g[:n-t] += (1.0 - h[t:])
                g[t:] -= h[:n-t]
            grad += g
        # Average over active constraints for a smoother gradient
        grad /= len(active)
        return dx * grad
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    # Try SLSQP
    try:
        res = minimize(obj, h0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-12})
        if res.success and res.fun < best_score:
            best_h = res.x
            best_score = res.fun
    except Exception:
        pass
    
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.7:
        return best_h, best_score, n
    
    # Method 2: Subgradient / projected gradient method
    h = best_h.copy()
    lr = 0.01
    for iteration in range(2000):
        if time.time() - start_time > budget_s * 0.85:
            break
        
        sc = score_h(h)
        if sc < best_score:
            best_h = h.copy()
            best_score = sc
        
        g = obj_grad(h)
        # Projected gradient step
        h_new = h - lr * g
        h_new = project_box_mass(h_new, target_sum)
        
        # Check if it improved
        new_sc = score_h(h_new)
        if new_sc < sc:
            h = h_new
        else:
            lr *= 0.9

        if iteration % 100 == 0:
            noise = rng.randn(n) * 0.01
            noise -= np.mean(noise)
            h = project_box_mass(h + noise, target_sum)
    
    return best_h, best_score, n
```

            0 ≤ h_i ≤ 1

This is a linear program with 2n-1 + 1 = 2n constraints and n+1 variables. For n=70, that's 140 constraints and 71 variables. Very tractable!

The LP solution gives a direction d = h_lp - h_c. Then we do a line search from h_c in direction d to find the best point on the line.

```python
def lp_direction(h_c, radius=None):
    """Solve LP to find a direction that reduces the linearized max overlap"""
    n = len(h_c)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    
    # Build the LP
    # Variables: h (n vars) + t (1 var) = n+1 vars
    # Minimize t: c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A_k(h_c) + grad_k^T (h - h_c) <= t
    # => grad_k^T h - t <= -A_k(h_c) + grad_k^T h_c
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for j in range(2*n-1):
        k = j - (n-1)
        g = overlap_gradient(h_c, k)
        A_ub[j, :n] = g
        A_ub[j, n] = -1.0
        b_ub[j] = -corr[j] + g @ h_c
    
    # Equality constraint: sum h_i = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds
    if radius is not None:
        lb = np.maximum(0, h_c - radius)
        ub = np.minimum(1, h_c + radius)
    else:
        lb = np.zeros(n)
        ub = np.ones(n)
    bounds = list(zip(lb, ub)) + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_lp = res.x[:n]
        d = h_lp - h_c
        # Zero out the mean to maintain the sum constraint (numerical)
        d -= np.mean(d)
        return d
    return None
```

```python
    n = len(h)

        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[0:n-k])
        g[0:n-k] -= h[k:]
```
