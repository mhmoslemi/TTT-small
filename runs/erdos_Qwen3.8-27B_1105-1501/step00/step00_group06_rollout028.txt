```python
def project_box_mass(v, n):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = n/2}"""
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    
    s = v.sum()
    if abs(s - target) < 1e-13:
        return v
    
    # Use bisection to find mu such that sum(clip(v + mu, 0, 1)) = target
    def F(mu):
        return np.clip(v + mu, 0.0, 1.0).sum()
    
    lo = -2.0 - v.max()
    hi = 1.0 - v.min() + 2.0
    
    if F(lo) > target:
        v = np.clip(lo - v + 0.5, 0, 1)  # fallback
        # Actually just return a feasible default
        even = n % 2 == 0
        fallback = np.zeros(n)
        if even:
            fallback[:n//2] = 1.0
        else:
            fallback[:(n-1)//2] = 1.0
            fallback[(n-1)//2] = 0.5
        return fallback
    if F(hi) < target:
        fallback = np.zeros(n)
        if n % 2 == 0:
            fallback[:n//2] = 1.0
        else:
            fallback[:(n-1)//2] = 1.0
            fallback[(n-1)//2] = 0.5
        return fallback
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if F(mid) < target:
            lo = mid
        else:
            hi = mid
    
    mu = (lo + hi) / 2.0
    result = np.clip(v + mu, 0.0, 1.0)
    
    # Fix residual
    residual = target - result.sum()
    if abs(residual) > 1e-12:
        # Distribute over interior points
        interior = np.where((result > 1e-8) & (result < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            per_point = residual / len(interior)
            result[interior] += per_point
            result = np.clip(result, 0.0, 1.0)
            # Final fix
            residual2 = target - result.sum()
            if abs(residual2) > 1e-12 and len(interior) > 0:
                result[interior[0]] += residual2
                result = np.clip(result, 0.0, 1.0)
    
    return result

def evaluate_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return np.max(r) * dx

def interpolate_h(h_src, n_dst):
    """Interpolate h from its native resolution to n_dst points"""
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0/n_dst
    h_interp = np.interp(x_dst, x_src, h_src)
    return project_box_mass(h_interp, n_dst)

    # Compute feasible interval for t

    t_min = 0.0
    t_max = 0.0

    best_val = evaluate_c5(h)

        a, b = t_min, t_max
        c = b - phi * (b - a)
        d_pt = a + phi * (b - a)
        f_c = evaluate_c5(project_box_mass(h + c * d, n))

def smoothed_gradient(h, n, beta):
    """Compute gradient of log-sum-exp smoothing using finite differences"""
    eps = 1e-6
    r = np.correlate(h, 1.0 - h, mode="full")
    # Numerical stabilization
    r_shift = r - r.max()
    weights = np.exp(beta * r_shift)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    dx = 2.0 / n
    
    for k in range(len(r)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        # dr_k / dh_i
        for i in range(n):
            # h[i] appears as h[n+lag] when n+lag=i, i.e., n=i-lag
            if 0 <= i - lag < n:
                grad[i] += weights[k] * (1.0 - h[i - lag])
            # h[i] appears as h[n] in (1-h[n]) when n=i
            if 0 <= i + lag < n:
                grad[i] -= weights[k] * h[i + lag]
    
    return grad

def pair_polish(h, n, max_pairs=None, rng=None):
    """Try pairwise transfers: move mass from i to j"""
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    if rng is None:
        rng = np.random.default_rng(0)
    
    if max_pairs is None:
        max_pairs = n * (n - 1) // 2
    
    # Generate pairs
    pairs = []
    for i in range(n):
        for j in range(n):
            if i != j:
                pairs.append((i, j))
    
    if len(pairs) > max_pairs:
        indices = rng.choice(len(pairs), max_pairs, replace=False)
        pairs = [pairs[i] for i in indices]
    
    for i, j in pairs:
        if h[i] < 1e-8 or h[j] > 1.0 - 1e-8:
            continue
        # Transfer eps from i to j
        # Try small transfers
        for eps in [0.01, 0.005, 0.001]:
            d = np.zeros(n)
            d[i] = -eps
            d[j] = eps
            h_try = h + d
            h_try = project_box_mass(h_try, n)
            val = evaluate_c5(h_try)
            if val < best_val - 1e-14:
                best_val = val
                best_h = h_try
                h = h_try
                break
    
    return best_h, best_val

def random_direction_polish(h, n, n_iters=100, rng=None):
    """Try random mass-preserving directions with line search"""
    if rng is None:
        rng = np.random.default_rng(1)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    for _ in range(n_iters):
        # Random mass-preserving direction
        d = rng.standard_normal(n)
        d -= d.mean()
        d /= np.max(np.abs(d)) + 1e-15
        
        h_new, t, val = line_search(best_h, d, n, max_step=0.3)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_new
    
    return best_h, best_val

def smooth_gradient_descent(h, n, betas=None, n_iters=200, rng=None):
    """Iterative smoothed gradient descent with exact line search"""
    if betas is None:
        betas = [5.0, 20.0, 50.0, 100.0, 200.0]
    if rng is None:
        rng = np.random.default_rng(2)
    
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    beta_idx = 0
    for it in range(n_iters):
        beta = betas[min(beta_idx, len(betas) - 1)]
        grad = smoothed_gradient(best_h, n, beta)
        
        # Negate for descent
        d = -grad
        d -= d.mean()  # Ensure mass preservation
        norm = np.max(np.abs(d))
        if norm < 1e-15:
            beta_idx = min(beta_idx + 1, len(betas) - 1)
            continue
        d /= norm
        
        h_new, t, val = line_search(best_h, d, n, max_step=0.3)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_new
        else:
            # Try smaller step or different beta
            beta_idx = min(beta_idx + 1, len(betas) - 1)
    
    return best_h, best_val
```

```python
import numpy as np
import time

def project_box_mass(v, n):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = n/2}"""
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    
    s = v.sum()
    tol = 1e-12
    
    def make_fallback():
        fb = np.zeros(n)
        if n % 2 == 0:
            fb[:n//2] = 1.0
        else:
            fb[:(n-1)//2] = 1.0
            fb[(n-1)//2] = 0.5
        return fb
    
    if abs(s - target) < tol:
        return v
    
    def F(mu):
        return np.clip(v + mu, 0.0, 1.0).sum()
    
    lo = -v.max() - 2.0
    hi = 1.0 - v.min() + 2.0
    
    if F(lo) >= target or F(hi) <= target:
        return make_fallback()
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if F(mid) < target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - result.sum()
    
    if abs(residual) > 1e-14:
        interior = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            result[interior] += residual / len(interior)
            result = np.clip(result, 0.0, 1.0)
            residual2 = target - result.sum()
            if abs(residual2) > 1e-14:
                interior2 = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
                if len(interior2) > 0:
                    result[interior2[0]] += residual2
                    result = np.clip(result, 0.0, 1.0)
    
    return result

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * dx)

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_mass(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_interp = np.interp(x_dst, x_src, h_src)
    return project_box_mass(h_interp, n_dst)

def compute_gradient(h, n, beta):
    """Compute gradient of smoothed objective using exact Jacobian."""
    r = np.correlate(h, 1.0 - h, mode="full")
    r_max = r.max()
    exp_r = np.exp(beta * (r - r_max))
    weights = exp_r / exp_r.sum()
    
    # Build Jacobian J[i, m] = dr[m]/dh[i]
    # and compute grad = J.T @ weights
    grad = np.zeros(n)
    
    for i in range(n):
        gi = 0.0
        for m in range(2 * n - 1):
            w = weights[m]
            if w < 1e-12:
                continue
            lag = m - (n - 1)
            j1 = i - lag
            if 0 <= j1 < n:
                gi += w * (1.0 - h[j1])
            j2 = i + lag
            if 0 <= j2 < n:
                gi -= w * h[j2]
        grad[i] = gi
    
    return grad

def line_search(h, d, n, f_best, max_step=0.4):
    """Exact golden section line search along direction d from h."""
    # Find feasible interval
    eps = 1e-14
    t_lo = -max_step
    t_hi = max_step
    
    d_max = np.max(np.abs(d))
    if d_max < 1e-15:
        return h.copy(), 0.0, f_best
    
    # Tighten bounds based on h and d
    for i in range(n):
        if d[i] > eps:
            t_lo = max(t_lo, -h[i] / d[i])
            t_hi = min(t_hi, (1.0 - h[i]) / d[i])
        elif d[i] < -eps:
            t_lo = max(t_lo, (1.0 - h[i]) / d[i])
            t_hi = min(t_hi, -h[i] / d[i])
    
    t_lo = max(t_lo, -max_step)
    t_hi = min(t_hi, max_step)
    
    if t_hi - t_lo < 1e-12:
        return h.copy(), 0.0, f_best
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    dpt = a + gr * (b - a)
    
    def f(t):
        h_t = h + t * d
        h_t = project_box_mass(h_t, n)
        return evaluate_c5(h_t)
    
    f_c = f(c)
    f_d = f(dpt)
    
    for _ in range(25):
        if f_c < f_d:
            b = dpt
            dpt = c
            f_d = f_c
            c = b - gr * (b - a)
            f_c = f(c)
        else:
            a = c
            c = dpt
            f_c = f_d
            dpt = a + gr * (b - a)
            f_d = f(dpt)
        if b - a < 1e-9:
            break
    
    t_opt = 0.5 * (a + b)
    h_opt = project_box_mass(h + t_opt * d, n)
    v_opt = evaluate_c5(h_opt)
    
    if v_opt < f_best:
        return h_opt, t_opt, v_opt
    return h.copy(), 0.0, f_best

def smooth_gradient_descent(h, n, f_best, budget_s, rng):
    """Main optimization loop with smoothed gradient + line search."""
    h = h.copy()
    start = time.time()
    betas = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    beta_idx = 0
    stall_count = 0
    
    while time.time() - start < budget_s:
        beta = betas[min(beta_idx, len(betas) - 1)]
        grad = compute_gradient(h, n, beta)
        
        # Descent direction
        d = -grad
        d -= d.mean()
        d_norm = np.max(np.abs(d))
        if d_norm < 1e-15:
            beta_idx = min(beta_idx + 1, len(betas) - 1)
            continue
        d = d / d_norm
        
        h_new, t, val_new = line_search(h, d, n, f_best, max_step=0.3)
        
        if val_new < f_best - 1e-13:
            h = h_new
            f_best = val_new
            stall_count = 0
            # After improvement, try increasing beta (sharper focus)
            if rng.random() < 0.3:
                beta_idx = min(beta_idx + 1, len(betas) - 1)
        else:
            stall_count += 1
            if stall_count >= 5:
                # Reset beta to lower value
                beta_idx = max(0, beta_idx - 3)
                # Try a random perturbation
                H = 0.05 * rng.standard_normal(n)
                H -= H.mean()
                h = project_box_mass(h + H, n)
                f_best = evaluate_c5(h)
                stall_count = 0
                beta_idx = 0
    
    return h, f_best

def random_direction_polish(h, n, f_best, n_iters, rng):
    for _ in range(n_iters):
        d = rng.standard_normal(n)
        d -= d.mean()
        dn = np.max(np.abs(d))
        if dn < 1e-15:
            continue
        d /= dn
        h_new, t, val_new = line_search(h, d, n, f_best, max_step=0.2)
        if val_new < f_best:
            h = h_new
            f_best = val_new
    return h, f_best

def pair_polish(h, n, f_best, rng, time_budget=30.0):
    start = time.time()
    improved = True
    while improved and time.time() - start < time_budget:
        improved = False
        # Sample random pairs
        for _ in range(200):
            if time.time() - start > time_budget:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Transfer from i to j (if h[i] > 0) or from j to i (if h[j] > 0)
            for eps in [0.02, 0.01, 0.005]:
                if h[i] > eps:
                    d = np.zeros(n)
                    d[i] = -eps
                    d[j] = eps
                    h_try = project_box_mass(h + d, n)
                    v_try = evaluate_c5(h_try)
                    if v_try < f_best:
                        h = h_try
                        f_best = v_try
                        improved = True
                        break
                if h[j] > eps:
                    d = np.zeros(n)
                    d[j] = -eps
                    d[i] = eps
                    h_try = project_box_mass(h + d, n)
                    v_try = evaluate_c5(h_try)
                    if v_try < f_best:
                        h = h_try
                        f_best = v_try
                        improved = True
                        break
    return h, f_best
```

```python
            j2 = idx2 + lag
```

```python
import numpy as np

def project_box_mass(v, n):
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    tol = 1e-12
    if abs(s - target) < tol:
        return v
    def F(mu):
        return np.clip(v + mu, 0.0, 1.0).sum()
    lo = -v.max() - 2.0
    hi = 1.0 - v.min() + 2.0
    if F(lo) >= target or F(hi) <= target:
        fb = np.zeros(n)
        fb[:n // 2] = 1.0
        if n % 2 == 1:
            fb[n // 2] = 0.5
        return fb
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if F(mid) < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - result.sum()
    if abs(residual) > 1e-14:
        interior = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            result[interior] += residual / len(interior)
            result = np.clip(result, 0.0, 1.0)
            r2 = target - result.sum()
            if abs(r2) > 1e-14:
                in2 = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
                if len(in2) > 0:
                    result[in2[0]] += r2
                    result = np.clip(result, 0.0, 1.0)
    return result

def evaluate_c5(h):
    n = len(h)

    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * dx)

    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_mass(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_interp = np.interp(x_dst, x_src, h_src)
    return project_box_mass(h_interp, n_dst)

def compute_gradient(h, n, beta):
    r = np.correlate(h, 1.0 - h, mode="full")
    r_max = r.max()
    exp_r = np.exp(beta * (r - r_max))
    wsum = exp_r.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_r / wsum
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        lag = m - (n - 1)
        i1s = max(0, lag)
        i1e = min(n, n + lag)
        if i1e > i1s:
            idx = np.arange(i1s, i1e)
            grad[idx] += w * (1.0 - h[idx - lag])
        i2s = max(0, -lag)
        i2e = min(n, n - lag)
        if i2e > i2s:
            idx2 = np.arange(i2s, i2e)
            grad[idx2] -= w * h[idx2 + lag]
    return grad

def line_search(h, d, n, f_best, max_step=0.4):
    d_max = np.max(np.abs(d))
    if d_max < 1e-15:
        return h.copy(), 0.0, f_best

        return h.copy(), 0.0, f_best

def smooth_gradient_descent(h, n, f_best, budget_s, rng):
    h = h.copy()

    betas = [3.0, 8.0, 15.0, 30.0, 60.0, 120.0, 250.0, 500.0, 1000.0]
    beta_idx = 0
    stall = 0
    while time.time() - start < budget_s:
        beta = betas[min(beta_idx, len(betas) - 1)]
        grad = compute_gradient(h, n, beta)
        d = -grad
        d -= d.mean()
        dn = np.max(np.abs(d))
        if dn < 1e-15:
            beta_idx = min(beta_idx + 1, len(betas) - 1)
            continue
        d /= dn
        h_new, t, val = line_search(h, d, n, f_best, max_step=0.3)
        if val < f_best - 1e-13:
            h = h_new
            f_best = val
            stall = 0
            if rng.random() < 0.25:

            if stall >= 4:

                H = 0.08 * rng.standard_normal(n)
                H -= H.mean()
                h = project_box_mass(h + H, n)
                f_best = evaluate_c5(h)
                stall = 0
                beta_idx = 0
    return h, f_best

def random_direction_polish(h, n, f_best, n_iters, rng):
```
