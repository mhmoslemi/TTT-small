    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(vals) * dx)

def make_feasible(z, target=None):
    n = len(z)
    if target is None:
        target = n / 2.0
    z = np.asarray(z, dtype=float).ravel()
    # Bisection on lambda: sum(clip(z + lambda, 0, 1)) = target
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    # Repair residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        # Find coordinates with room
        if residual > 0:
            # Add to coordinates with capacity
            for _ in range(n):
                if abs(residual) < 1e-14:
                    break
                cap = 1.0 - h
                idx = np.argmax(cap)
                delta = min(residual, cap[idx])
                h[idx] += delta
                residual = target - np.sum(h)
        else:
            for _ in range(n):
                if abs(residual) < 1e-14:
                    break
                idx = np.argmax(h)
                delta = min(-residual, h[idx])
                h[idx] -= delta
                residual = target - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    out = np.zeros(2 * n - 1)

    return out

def manual_grad(h):

        # Second factor contribution: p = i, valid when max(0,-k) <= p < min(n, n-k)
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            # h[p] gets -h[p+k]
            G[row, lo2:hi2] -= h[lo2+k:hi2+k]
    return G

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = manual_values(h) * dx
    cmax = np.max(vals)
    exponents = beta * (vals - cmax)
    # Clip for numerical stability
    exponents = np.clip(exponents, -50, 50)
    weights = np.exp(exponents)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        return cmax + np.log(2*n-1) / beta
    return cmax + np.log(wsum) / beta

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = manual_values(h) * dx
    cmax = np.max(vals)
    exponents = beta * (vals - cmax)
    exponents = np.clip(exponents, -50, 50)
    weights = np.exp(exponents)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        return dx * np.ones(n)  # fallback
    w = weights / wsum
    G = manual_grad(h)
    return dx * (w @ G)

    ws = np.sum(np.exp(exponents))

                z0 = np.concatenate([best_h, [s_now + 1e-6]])
                res = minimize(
                    dm_fun, z0, dm_jac, args=(n_best,),
                    method='SLSQP', bounds=[(0, 1)] * n_best + [(0, 1.0)],
                    constraints=[
                        {'type': 'ineq', 'fun': dm_ineq_fun, 'jac': dm_ineq_jac, 'args': (n_best,)},
                        {'type': 'eq', 'fun': dm_eq_fun, 'jac': dm_eq_jac, 'args': (n_best,)}
                    ],
                    options={'maxiter': 300, 'ftol': 1e-14}
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = make_feasible(res.x[:n_best])
                    s_new = official_score(h_new)
                    if s_new < best_score - 1e-14:
                        best_h, best_score = h_new, s_new
            except Exception:
                pass

    # Try a few more resolutions from the final best
    if time.time() < deadline:
        n_best = len(best_h)
        for n_try in [n_best - 3, n_best + 3, n_best - 6, n_best + 6]:
            if n_try < 20 or n_try > 80 or time.time() > deadline:
                continue
            try:
                hi = make_feasible(interpolate_to(best_h, n_try))
                s_hi = official_score(hi)
                if s_hi < best_score - 1e-14:
                    best_h, best_score = hi, s_hi
                res = minimize(
                    smooth_obj, hi, smooth_jac, args=(500,),
                    method='SLSQP', bounds=[(0, 1)] * n_try,
                    constraints=[{'type': 'eq', 'fun': sum_eq_fun,
                                  'jac': sum_eq_jac, 'args': (n_try,)}],
                    options={'maxiter': 100, 'ftol': 1e-13}
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_f = make_feasible(res.x)
                    s_f = official_score(h_f)
                    if s_f < best_score - 1e-14:
                        best_h, best_score = h_f, s_f
                if time.time() < deadline:
                    sf = official_score(h_f)
                    z0f = np.concatenate([h_f, [sf + 1e-5]])
                    res2 = minimize(
                        dm_fun, z0f, dm_jac, args=(n_try,),
                        method='SLSQP', bounds=[(0, 1)] * n_try + [(0, 1.0)],
                        constraints=[
                            {'type': 'ineq', 'fun': dm_ineq_fun, 'jac': dm_ineq_jac, 'args': (n_try,)},
                            {'type': 'eq', 'fun': dm_eq_fun, 'jac': dm_eq_jac, 'args': (n_try,)}
                        ],
                        options={'maxiter': 150, 'ftol': 1e-13}
                    )
                    if res2.x is not None and np.all(np.isfinite(res2.x)):
                        h_f2 = make_feasible(res2.x[:n_try])
                        s_f2 = official_score(h_f2)
                        if s_f2 < best_score - 1e-14:
                            best_h, best_score = h_f2, s_f2
            except Exception:
                pass

    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = float(final_score) + 1e-13
    return final_h, final_bound, len(final_h)
