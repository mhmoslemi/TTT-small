def project_box_mass(v, n=None):

    Solve LP to find a direction that reduces the max of linearized A_s.
    Returns: h_LP (candidate) or None

    num_constr = 2 * n - 1
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: G @ h - t <= A - G @ h_c  for each row

    A_ub = np.column_stack([G, -np.ones(num_constr)])

    return None

            result[free] += residual / nf
            result = np.clip(result, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def compute_A_and_grads(h):
    n = len(h)
    num = 2 * n - 1
    A = np.zeros(num)
    G = np.zeros((num, n))
    for k_idx in range(num):
        s = k_idx - (n - 1)
        if s >= 0:
            valid = n - s
            if valid <= 0:
                continue
            A[k_idx] = np.sum(h[:valid] * (1.0 - h[s:s+valid]))
            G[k_idx, :valid] += (1.0 - h[s:s+valid])
            if s > 0:
                G[k_idx, s:] -= h[:valid]
        else:
            t = -s
            valid = n - t
            if valid <= 0:
                continue
            A[k_idx] = np.sum(h[t:] * (1.0 - h[:valid]))
            G[k_idx, t:] += (1.0 - h[:valid])
            G[k_idx, :valid] -= h[t:]
    return A, G

def solve_lp(h_c, A, G, radius):
    n = len(h_c)
    nc = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    rhs = A - G @ h_c
    A_ub = np.empty((nc, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lower = np.maximum(0.0, h_c - radius)
    upper = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    res = linprog(c, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

    # Refine
    if 0.0 > lam_min + 1e-10 and 0.0 < lam_max - 1e-10:
        # Include lambda=0
        pass
    step = (lam_max - lam_min) / n_evals
    for _ in range(40):

        hi_r = min(lam_max, best_lam + step)

                best_s = s
                best_lam = lam
                best_h = h_t
                improved = True
        if not improved:
            break
        step /= 4.0
    return best_h, best_s
