def repair_feasible(h, tol=1e-9):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * h.size
    s = float(h.sum())
    res = target - s
    if abs(res) <= tol:
        return h
    if s > 1e-12 and abs(res) / s <= 1e-3:
        scale = target / s
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        res = target - s
        if abs(res) <= 1e-7:
            return h
    if res > 0.0:
        rem = res
        slack = 1.0 - h
        order = np.argsort(-slack)
        for idx in order:
            if rem <= 1e-12:
                break
            add = min(float(slack[idx]), float(rem))
            if add > 0.0:
                h[idx] = min(1.0, h[idx] + add)
                rem -= add
    else:
        rem = -res
        order = np.argsort(-h)
        for idx in order:
            if rem <= 1e-12:
                break
            sub = min(float(h[idx]), float(rem))
            if sub > 0.0:
                h[idx] = max(0.0, h[idx] - sub)
                rem -= sub
    # If residual remains due to numerical issue, spread over all with slack or positive
    res = target - float(h.sum())
    if abs(res) > 1e-8:
        if res > 0.0:
            # proportional to slack
            slack = 1.0 - h
            total = float(slack.sum())
            if total > 1e-12:
                h = h + res * (slack / total)
            h = np.clip(h, 0.0, 1.0)
        else:
            total = float(h.sum())
            if total > 1e-12:
                h = h * (1.0 + res / total)  # res negative, subtract proportional
            h = np.clip(h, 0.0, 1.0)
    return h

def get_parent():

def direct_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return math.inf
    if not bool(np.all(np.isfinite(h))):
        return math.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def is_feasible(h, tol=1e-7):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    target = 0.5 * h.size
    return bool(
        np.all(np.isfinite(h))
        and np.all(h >= -tol)
        and np.all(h <= 1.0 + tol)
        and abs(float(h.sum()) - target) <= tol
    )


def repair_feasible(h, tol=1e-9):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * h.size
    s = float(h.sum())
    res = target - s
    if abs(res) <= tol:
        return h

    if s > 1e-12 and abs(res) / s <= 1e-3:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        res = target - float(h.sum())
        if abs(res) <= 1e-7:
            return h

    if res > 0.0:
        rem = res
        slack = 1.0 - h
        order = np.argsort(-slack)
        for idx in order:
            if rem <= 1e-12:
                break
            add = min(float(slack[idx]), float(rem))
            if add > 0.0:
                h[idx] = min(1.0, h[idx] + add)
                rem -= add
    else:
        rem = -res
        order = np.argsort(-h)
        for idx in order:
            if rem <= 1e-12:
                break
            sub = min(float(h[idx]), float(rem))
            if sub > 0.0:
                h[idx] = max(0.0, h[idx] - sub)
                rem -= sub

    res = target - float(h.sum())
    if abs(res) > 1e-8:
        if res > 0.0:
            slack = 1.0 - h
            total = float(slack.sum())
            if total > 1e-12:

        else:
            total = float(h.sum())
            if total > 1e-12:

def project_box_simplex(z, target):
    z = np.asarray(z, dtype=float).copy()
    n = z.size
    if n == 0:
        return z
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    s = float(z.sum())
    if abs(s - target) <= 1e-7:
        return repair_feasible(z)

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0

        else:
            hi = mid

def golden_pair(x, p, q, a, b, max_iter=35):
    if b <= a:
        t = 0.5 * (a + b)
        return t, pair_obj(x, p, q, t)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = pair_obj(x, p, q, c)
    fd = pair_obj(x, p, q, d)
    for _ in range(max_iter):
        if (b - a) <= 1e-10 * (1.0 + abs(a) + abs(b)):
            break
        if fc <= fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = pair_obj(x, p, q, c)

            c = d
            fc = fd

def line_search_pair(x, p, q, base_c, max_iter=35):
    xp = x[p]
    xq = x[q]
    tmin = max(xp - 1.0, -xq)
    tmax = min(xp, 1.0 - xq)
    if tmax < tmin + 1e-11:
        return None

    best_c = base_c
    best_t = 0.0

    for t in (tmin, tmax, 0.0):
        if t < tmin - 1e-12 or t > tmax + 1e-12:
            continue
        c = pair_obj(x, p, q, t)
        if c < best_c - 1e-12:
            best_c = c
            best_t = t

    if tmax - tmin >= 1e-7:
        gt, gc = golden_pair(x, p, q, tmin, tmax, max_iter)
        if gc < best_c - 1e-12:
            best_c = gc
            best_t = gt

        y = apply_pair(x, p, q, best_t)
        c = direct_c5(y)
        if c < base_c - _ACCEPT_TOL:

    return None

def resample_vector(base, n_new, rng, strength=0.0):
    n_new = int(n_new)
    if n_new < 2:
        n_new = 2
    if base is None:
        z = 0.5 + 0.25 * rng.normal(size=n_new)
    else:
        base = np.asarray(base, dtype=float)
        if base.size == 0:
            z = 0.5 + 0.25 * rng.normal(size=n_new)
        elif base.size != n_new:
            old_x = (np.arange(base.size) + 0.5) * (2.0 / base.size)
            new_x = (np.arange(n_new) + 0.5) * (2.0 / n_new)
            z = np.interp(new_x, old_x, base, left=float(base[0]), right=float(base[-1]))
        else:
            z = base.copy()
    if float(strength) > 0.0:
        z = z + float(strength) * rng.normal(size=n_new)
    return project_box_simplex(z, 0.5 * n_new)

    if k < 0:
        k = 0
    if k > 0:
        h[desc[:k]] = 1.0
    if rem > 1e-9 and k < n:
        h[desc[k]] = rem
    return project_box_simplex(h, target)


def make_sinusoidal(n, freq, phase, rng, mode="binary", amp=0.35):
    n = int(n)
    if n < 2:
        n = 2
    x = np.arange(n)
    base = 0.5 + 0.5 * np.cos(2.0 * math.pi * float(freq) * x / float(n) + float(phase))
    if mode == "binary":
        v = base + 0.03 * rng.normal(size=n)
        return binarize_to_target(v, 0.5 * n, rng)
    z = 0.5 + float(amp) * np.cos(2.0 * math.pi * float(freq) * x / float(n) + float(phase))
    z = z + 0.04 * rng.normal(size=n)
    return project_box_simplex(z, 0.5 * n)


def generate_candidate_starts(n, base, rng, count=5):
    n = int(n)
    if n < 2:
        n = 2
    cands = []
    if base is not None:
        cands.append(resample_vector(base, n, rng, 0.0))
        cands.append(resample_vector(base, n, rng, 0.08))

    for f in (1, 2, 3, 5, 8):
        if len(cands) >= count:
            break
        cands.append(make_sinusoidal(n, f, 0.0, rng, "binary"))

    for f in (1, 2, 3):
        if len(cands) >= count:
            break
        cands.append(make_sinusoidal(n, f, math.pi / 2.0, rng, "smooth", 0.25))

    while len(cands) < count:
        cands.append(random_feasible(n, rng))
    return cands


def simulated_annealing(x, c, deadline, max_steps, rng,
                        t0=8e-4, t1=5e-7, polish_interval=30000):
    x = np.asarray(x, dtype=float).copy()
    x = project_box_simplex(x, 0.5 * x.size)
    x = repair_feasible(x)
    current_c = direct_c5(x)
    if not math.isfinite(current_c):
        x = random_feasible(x.size, rng)
        current_c = direct_c5(x)

    best = x.copy()
    best_c = current_c
    n = x.size
    if n < 2:
        return best, best_c

    seg_start = time.time()
    seg_dur = max(1.0, deadline - seg_start)
    step = 0
    while step < max_steps and time.time() < deadline - 0.25:
        frac = (time.time() - seg_start) / seg_dur
        if frac < 0.0:
            frac = 0.0
        if frac > 0.999:
            frac = 0.999
        T = t0 * (t1 / t0) ** frac

        p = int(rng.integers(n))
        q = int(rng.integers(n))
        while p == q:
            q = int(rng.integers(n))

        xp = x[p]
        xq = x[q]
        tmin = max(xp - 1.0, -xq)
        tmax = min(xp, 1.0 - xq)
        if tmax - tmin < 1e-11:
            step += 1
            continue

        u = rng.random()
        if u < 0.25:
            span = tmax - tmin
            if rng.random() < 0.5:
                t = tmin + rng.random() * 0.15 * span
            else:
                t = tmax - rng.random() * 0.15 * span
        else:
            t = tmin + (tmax - tmin) * rng.random()

        y = apply_pair(x, p, q, t)
        newc = direct_c5(y)
        if math.isfinite(newc):
            if newc <= current_c:
                x = y
                current_c = newc
                if current_c < best_c - 1e-12:
                    best = x.copy()
                    best_c = current_c
            else:
                delta = (newc - current_c) / max(T, 1e-12)
                if delta < 700.0 and rng.random() < math.exp(-delta):
                    x = y
                    current_c = newc
                    if current_c < best_c - 1e-12:
                        best = x.copy()
                        best_c = current_c

        step += 1

        if step % 5000 == 0:
            x = repair_feasible(x)
            current_c = direct_c5(x)
            if current_c < best_c - 1e-12:
                best = x.copy()
                best_c = current_c

        if polish_interval and step % polish_interval == 0:
            x2, c2 = polish_feasible(
                x,
                max_rounds=1,
                deadline=min(deadline, time.time() + 5.0),
                pair_max_iter=25,
                rng=rng
            )
            if c2 < current_c - 1e-10:
                x = x2
                current_c = c2
                if c2 < best_c - 1e-10:
                    best = x.copy()
                    best_c = c2

    x = repair_feasible(x)
    current_c = direct_c5(x)
    if current_c < best_c:
        best = x.copy()
        best_c = current_c
    return best, best_c


def unique_sorted_n(candidates, ref):
    uniq = []
    seen = set()
    for k in candidates:
        k = int(k)
        if 20 <= k <= 160 and k not in seen:
            seen.add(k)
            uniq.append(k)
    for i in range(1, len(uniq)):
        item = uniq[i]
        item_key = (abs(item - ref), item)
        j = i - 1
        while j >= 0:
            j_item = uniq[j]
            j_key = (abs(j_item - ref), j_item)
            if j_key > item_key:
                uniq[j + 1] = j_item
                j -= 1
            else:
                break
        uniq[j + 1] = item
    return uniq


def run(seed=42, budget_s=1100, **kwargs):
    try:
        start = time.time()
        try:
            budget = float(budget_s)
        except Exception:
            budget = 1100.0
        if budget <= 0.0:
            budget = 2.0
        deadline = start + 0.98 * budget
        rng = np.random.default_rng(seed)

        if budget < 30.0:
            parent = get_parent()
            if parent is not None:
                if is_feasible(parent, 1e-6):
                    h = parent.copy()
                else:
                    h = project_box_simplex(parent, 0.5 * parent.size)
            else:
                h = make_sinusoidal(48, 3, 0.0, rng, "binary")
            h = repair_feasible(h)
            c = direct_c5(h)
            if time.time() < deadline - 5.0:
                h, c = polish_feasible(h, max_rounds=2, deadline=deadline - 1.0,
                                       pair_max_iter=20, rng=rng)
            return np.asarray(h, dtype=float), float(c), int(h.size)

        parent = get_parent()
        if parent is not None:
            if is_feasible(parent, 1e-6):
                best_h = parent.copy()
            else:
                best_h = project_box_simplex(parent, 0.5 * parent.size)
            best_h = repair_feasible(best_h)
            best_c = direct_c5(best_h)
        else:
            best_h = make_sinusoidal(48, 3, 0.0, rng, "binary")
            best_c = direct_c5(best_h)

        if not math.isfinite(best_c):
            best_h = np.full(48, 0.5)
            best_c = direct_c5(best_h)
        best_n = int(best_h.size)

        if time.time() < deadline - 20.0:
            dead1 = min(deadline, start + 0.30 * budget + 30.0)
            h2, c2 = polish_feasible(best_h, max_rounds=50, deadline=dead1,
                                     pair_max_iter=35, rng=rng)
            if better_solution(c2, h2.size, best_c, best_n):
                best_h, best_c, best_n = h2, c2, int(h2.size)

        if time.time() < deadline - 40.0:
            for sigma in (0.02, 0.05, 0.10, 0.18, 0.35):
                if time.time() >= deadline - 15.0:
                    break
                z = best_h + sigma * rng.normal(size=best_n)
                h = project_box_simplex(z, 0.5 * best_n)
                dead = min(deadline, time.time() + 20.0)
                h2, c2 = polish_feasible(h, max_rounds=5, deadline=dead,
                                         pair_max_iter=30, rng=rng)
                if better_solution(c2, h2.size, best_c, best_n):
                    best_h, best_c, best_n = h2, c2, int(h2.size)

        if time.time() < deadline - 80.0:
            cands = generate_candidate_starts(best_n, best_h, rng, 10)
            for h0 in cands:
                if time.time() >= deadline - 25.0:
                    break
                dead = min(deadline, time.time() + 18.0)
                h2, c2 = polish_feasible(h0, max_rounds=4, deadline=dead,
                                         pair_max_iter=30, rng=rng)
                if better_solution(c2, h2.size, best_c, best_n):
                    best_h, best_c, best_n = h2, c2, int(h2.size)

        n_candidates = [24, 28, 32, 36, 40, 44, 48, 52, 56, 64, 72, 80, 96, 112]
        for factor in (0.80, 0.90, 1.10, 1.25):
            n_candidates.append(int(round(best_n * factor)))
        n_list = unique_sorted_n(n_candidates, best_n)

        for nn in n_list:
            if time.time() >= deadline - 40.0:
                break
            if nn == best_n:
                continue
            count = 5 if nn <= 70 else 3
            cands = generate_candidate_starts(nn, best_h, rng, count)
            for h0 in cands:
                if time.time() >= deadline - 20.0:
                    break
                max_rounds = 4 if nn <= 70 else 2
                time_per = 18.0 if nn <= 70 else 12.0
                dead = min(deadline, time.time() + time_per)
                h2, c2 = polish_feasible(h0, max_rounds=max_rounds, deadline=dead,
                                         pair_max_iter=30, rng=rng)
                if better_solution(c2, h2.size, best_c, best_n):
                    best_h, best_c, best_n = h2, c2, int(h2.size)
            if time.time() < deadline - 60.0:
                h2, c2 = polish_feasible(best_h, max_rounds=4,
                                         deadline=min(deadline, time.time() + 25.0),
                                         pair_max_iter=32, rng=rng)
                if better_solution(c2, h2.size, best_c, best_n):
                    best_h, best_c, best_n = h2, c2, int(h2.size)

        sa_count = 0
        while time.time() < deadline - 30.0 and sa_count < 6:
            seg_deadline = min(deadline, time.time() + 65.0)
            if sa_count == 0:
                sh = best_h.copy()
            else:
                u = rng.random()
                if u < 0.5:
                    sh = best_h + 0.08 * rng.normal(size=best_n)
                    sh = project_box_simplex(sh, 0.5 * best_n)
                elif u < 0.8:
                    freqs = (2, 3, 5, 7, 11)
                    f = freqs[int(rng.integers(len(freqs)))]
                    ph = float(rng.random() * 6.283185307179586)
                    sh = make_sinusoidal(best_n, f, ph, rng, "binary")
                else:
                    sh = random_feasible(best_n, rng)
            shc = direct_c5(sh)
            hsa, csa = simulated_annealing(sh, shc, seg_deadline, 1000000, rng,
                                           t0=8e-4, t1=5e-7, polish_interval=30000)
            if better_solution(csa, hsa.size, best_c, best_n):
                best_h, best_c, best_n = hsa, csa, int(hsa.size)
            if time.time() < deadline - 20.0:
                h2, c2 = polish_feasible(hsa, max_rounds=6,
                                         deadline=min(deadline, time.time() + 30.0),
                                         pair_max_iter=32, rng=rng)
                if better_solution(c2, h2.size, best_c, best_n):
                    best_h, best_c, best_n = h2, c2, int(h2.size)
            sa_count += 1

        if time.time() < deadline - 8.0:
            h2, c2 = polish_feasible(best_h, max_rounds=25,
                                     deadline=deadline - 3.0,
                                     pair_max_iter=35, rng=rng)
            if better_solution(c2, h2.size, best_c, best_n):
                best_h, best_c, best_n = h2, c2, int(h2.size)

        best_h = repair_feasible(best_h)
        if not is_feasible(best_h, 1e-6):
            best_h = project_box_simplex(best_h, 0.5 * best_h.size)
        if not is_feasible(best_h, 1e-5):
            p = get_parent()
            if p is not None:
                if is_feasible(p, 1e-5):
                    best_h = p.copy()
                else:
                    best_h = project_box_simplex(p, 0.5 * p.size)
            else:
                best_h = np.full(48, 0.5)
        best_c = direct_c5(best_h)
        if not math.isfinite(best_c):
            best_h = np.full(48, 0.5)
            best_c = direct_c5(best_h)

        best_h = np.asarray(best_h, dtype=float).copy()
        return best_h, float(best_c), int(best_h.size)

    except Exception:
        try:
            p = get_parent()
            if p is not None:
                if is_feasible(p, 1e-5):
                    h = p.copy()
                else:
                    h = project_box_simplex(p, 0.5 * p.size)
            else:
                h = np.full(48, 0.5)
            h = repair_feasible(h)
            c = direct_c5(h)
            if not math.isfinite(c):
                h = np.full(48, 0.5)
                c = direct_c5(h)
            return np.asarray(h, dtype=float), float(c), int(h.size)
        except Exception:
            h = np.full(48, 0.5)
            c = direct_c5(h)
            return h, float(c), 48
