    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0

        return float(np.clip(y - lam, 0, 1).sum())

        else:
            hi = mid

                best_alpha = alpha
        h[i] -= best_alpha
        h[j] += best_alpha
        if best_line_c < cur_c - 1e-12:
            h[i] += best_alpha
            h[j] -= best_alpha
            cur_c = best_line_c
            improved_any = True
    return h, cur_c

def local_refine(h_init, deadline, rng, T_start=0.12, T_end=8e-5):
    h = make_feasible(h_init.copy())
    best_h = h.copy()
    best_c = score_exact(best_h)
    cur_h = h.copy()
    cur_c = best_c
    step = 0.08
    stagnant = 0
    total_iter = 0
    n_total_target = 200
    while time.time() < deadline:
        frac = min(1.0, total_iter / n_total_target)
        T = T_start * (T_end / T_start) ** frac
        c_vec, smooth_val, w = score_weights(cur_h, T)
        g = compute_grad(cur_h, w)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            stagnant += 1
            step *= 0.6
            total_iter += 1
            continue
        accepted = False
        for scale in [step, step * 0.5, step * 2.0, step * 0.25]:
            h_trial = cur_h - scale * g
            h_trial = make_feasible(h_trial)
            c_trial = score_exact(h_trial)
            if c_trial < cur_c - 1e-13:
                cur_h = h_trial
                cur_c = c_trial
                accepted = True
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = cur_h.copy()
                stagnant = 0
                step = min(step * 1.05, 0.5)
                break
        if not accepted:
            step *= 0.65
            stagnant += 1
            if stagnant >= 4:
                cur_h, cur_c = pair_line_search(cur_h, cur_c, deadline, rng, n_pairs=8)
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = cur_h.copy()
                stagnant = 0
                step = 0.05
        if stagnant > 30:
            cur_h = cur_h + 0.02 * rng.standard_normal(len(cur_h))
            cur_h = make_feasible(cur_h)
            cur_c = score_exact(cur_h)
            stagnant = 0
        total_iter += 1
    return best_h, best_c

def binarize(h, rng):
    n = len(h)
    k = n // 2
    order = np.argsort(h)
    b = np.zeros(n)
    b[order[-k:]] = 1.0
    n_ones = int(b.sum())
    if n_ones != k:
        n_adj = abs(n_ones - k)
        if n_ones > k:
            one_idx = np.where(b == 1)[0]
            flip_idx = rng.choice(one_idx, size=n_adj, replace=False)
            b[flip_idx] = 0.0
        else:
            zero_idx = np.where(b == 0)[0]
            flip_idx = rng.choice(zero_idx, size=n_adj, replace=False)
            b[flip_idx] = 1.0
    return b

def resample(h, new_n):
    old_n = len(h)
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(new_x, old_x, h, left=h[0], right=h[-1])
    return make_feasible(new_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h_parent = None
    n_parent = 70
    try:
        if 'initial_h_values' in globals():
            hv = np.asarray(globals()['initial_h_values'], dtype=np.float64).flatten()
            if len(hv) >= 4 and np.all(np.isfinite(hv)):
                h_parent = hv
                n_parent = len(h)
    except Exception:
        pass

    if h_parent is None:
        h_parent = np.full(n_parent, 0.5)

    h_parent = make_feasible(h_parent)
    c_parent = score_exact(h_parent)

    best_h = h_parent.copy()
    best_c = c_parent

    def try_update(h_cand, c_cand):
        nonlocal best_h, best_c
        if c_cand < best_c - 1e-12:
            best_h = h_cand.copy()
            best_c = c_cand

    # Phase 1: Main refinement of parent
    if time.time() < deadline:
        d1 = min(deadline, t0 + 0.30 * budget_s)
        h_r, c_r = local_refine(h_parent, d1, rng, T_start=0.15, T_end=5e-5)
        try_update(h_r, c_r)

    # Phase 2: Binarized refinement
    if time.time() < deadline:
        h_bin = binarize(best_h, rng)
        c_bin = score_exact(h_bin)
        try_update(h_bin, c_bin)
        if time.time() < deadline:
            d2 = min(deadline, time.time() + 0.12 * budget_s)
            h_r, c_r = local_refine(h_bin, d2, rng, T_start=0.08, T_end=3e-5)
            try_update(h_r, c_r)

    # Phase 3: Resolution exploration
    if time.time() < deadline:
        n_list = []
        for dn in [-10, -6, -4, -2, 2, 4, 6, 10, 16, 24]:
            nn = n_parent + dn
            if 28 <= nn <= 130 and nn % 2 == 0:
                if nn not in n_list:
                    n_list.append(nn)
        n_list.sort()
        for nn in n_list:
            if time.time() > deadline:
                break
            h_new = resample(best_h, nn)
            c_new = score_exact(h_new)
            remaining_for_this = min(50.0, deadline - time.time())
            if remaining_for_this < 15:
                try_update(h_new, c_new)
                continue
            d3 = min(deadline, time.time() + remaining_for_this)
            h_r, c_r = local_refine(h_new, d3, rng, T_start=0.10, T_end=4e-5)
            try_update(h_r, c_r)

    # Phase 4: Mutated restarts
    if time.time() < deadline:
        sigmas = [0.03, 0.06, 0.10, 0.15, 0.25, 0.35]
        for sig in sigmas:
            if time.time() > deadline:
                break
            h_mut = best_h + sig * rng.standard_normal(len(best_h))
            h_mut = make_feasible(h_mut)
            c_mut = score_exact(h_mut)
            d4 = min(deadline, time.time() + 60.0)
            h_r, c_r = local_refine(h_mut, d4, rng, T_start=0.10, T_end=5e-5)
            try_update(h_r, c_r)

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        d5 = deadline
        h_r, c_r = local_refine(best_h, d5, rng, T_start=0.02, T_end=2e-5)
        try_update(h_r, c_r)

    # Final validation
    final_h = make_feasible(best_h)
    final_c = score_exact(final_h)
    if final_c > best_c + 1e-9:
        final_h = best_h
        final_c = best_c

    return final_h, final_c, len(final_h)
