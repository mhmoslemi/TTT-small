    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target_sum}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -max(h), 1 - min(h)  # wide bounds
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

def project_bounded_simplex(v, target_sum):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target_sum} via bisection."""
    n = len(v)
    lo = -5.0
    hi = 5.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-13:
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def c5_exact(h):
    """C5 = dx * max_k C_k(h)."""
    n = len(h)
    dx = 2.0 / n
    return float(dx * np.max(np.correlate(h, 1.0 - h, mode='full')))

def lse_obgrad(h, tau):
    """Log-sum-exp smooth objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    vals = dx * np.correlate(h, 1.0 - h, mode='full')
    vmax = vals.max()
    shifted = np.minimum((vals - vmax) / tau, 0.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    obj = vmax + tau * np.log(s)
    w = exp_vals / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        wk = w[k] * dx
        if lag >= 0:
            m = n - lag
            g[lag:lag + m] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[lag:lag + m]
        else:
            p = -lag
            m = n - p
            g[:m] += wk * (1.0 - h[p:p + m])
            g[p:p + m] -= wk * h[:m]
    return obj, g

def pgd_optimize(h0, tau, deadline, max_iters=10000):
    """Projected gradient descent on LSE-smoothed C5."""
    n = len(h0)
    target = n / 2.0
    h = project_bounded_simplex(h0, target)
    best = c5_exact(h)
    best_h = h.copy()
    obj_cur, _ = lse_obgrad(h, tau)
    stagnant = 0
    lr_scale = 1.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        _, g = lse_obgrad(h, tau)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-11:
            break
        step = lr_scale * 0.1 / gnorm
        found = False
        for _ in range(35):
            h_new = project_bounded_simplex(h - step * g, target)
            obj_new, _ = lse_obgrad(h_new, tau)
            if obj_new < obj_cur - 1e-14:
                found = True
                break
            step *= 0.5
        if not found:
            break
        h = h_new
        obj_cur = obj_new
        lr_scale = min(lr_scale * 1.02, 5.0)
        val = c5_exact(h)
        if val < best - 1e-12:
            best = val
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        tau = max(tau * 0.997, 1e-5)
        if stagnant > 500:
            break
    return best_h, best

def gen_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def gen_sinusoidal(n, rng):
    freq = rng.uniform(0.5, 6.0)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return project_bounded_simplex(np.clip(h, 0, 1), n / 2.0)

def gen_oscillatory(n, rng):
    freqs = rng.uniform(1, 8, 3)
    phases = rng.uniform(0, 2 * np.pi, 3)
    amps = rng.uniform(0.2, 0.45, 3)
    h = 0.5

        h = h + a * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
    return project_bounded_simplex(np.clip(h, 0, 1), n / 2.0)

def gen_blocks(n, rng):
    h = np.zeros(n)
    remaining = n // 2
    i = 0
    while remaining > 0 and i < n:
        block_len = min(rng.randint(1, max(2, n // 4)), n - i, remaining)
        h[i:i + block_len] = 1.0
        remaining -= block_len
        i += block_len
        skip = rng.randint(1, max(2, n // 4))
        i += min(skip, n - i)
    return project_bounded_simplex(h, n / 2.0)
