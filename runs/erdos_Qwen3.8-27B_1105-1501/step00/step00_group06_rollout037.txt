```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * dx)

def project_feasible(v, target):
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(len(v))
    if target >= len(v):
        return np.ones(len(v))
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(y)
    if abs(res) > 1e-14:
        interior = (y > 1e-12) & (y < 1.0 - 1e-12)
        ni = int(np.sum(interior))
        if ni > 0:
            y[interior] += res / ni
            y = np.clip(y, 0.0, 1.0)
    return y

def interpolate_to_resolution(h, new_n):
    n_old = len(h)
    if n_old == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, np.append(h, h[0]))
    target = new_n / 2.0
    return project_feasible(h_new, target)

def compute_lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(r)
    if beta > 0 and beta * m < 500:
        e = np.exp(beta * (r - m))
        s = np.sum(e)
        weights = e / s
    else:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(r)] = 1.0
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-15:
            continue
        lag = p - (n - 1)
        for i in range(n):
            j1 = i - lag
            if 0 <= j1 < n:
                grad[i] += w * (1.0 - h[j1])
            j2 = i + lag
            if 0 <= j2 < n:
                grad[i] -= w * h[j2]
    grad *= dx
    return grad

        e = np.exp(beta * (r - m))
        return (m + np.log(np.sum(e)) / beta) * dx
    return m * dx

    n = len(h)

    for it in range(max_iters):
        if time.time() - time_start > time_limit:
            break

        active = np.where(r >= threshold)[0]

        grad = np.zeros(n)
        for p in active:
            lag = p - (n - 1)
            for i in range(n):

                if 0 <= j1 < n:
                    grad[i] += (1.0 - h[j1])
                j2 = i + lag
                if 0 <= j2 < n:
                    grad[i] -= h[j2]

        grad *= (2.0 / n)

        if gnorm < 1e-14:
            break
        
        # Step with decreasing learning rate
        lr = 0.5 / (1.0 + 0.1 * it)
        h_new = project_feasible(h - lr * grad, target)
        val_new = evaluate_c5(h_new)
        if val_new < best_val - 1e-16:
            h = h_new.copy()
            best = h_new.copy()
            best_val = val_new
```

```python
def compute_lse_grad_vec(h, beta):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(r)
    if beta > 0 and beta * m < 500:
        e = np.exp(beta * (r - m))
        s = np.sum(e)
        weights = e / s
    else:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(r)] = 1.0
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        w = weights[p]
        if w < 1e-15:
            continue
        lag = p - (n - 1)
        if lag >= 0:
            # i-lag valid means i >= lag, so i in [lag, n-1]
            # contribution: (1-h[i-lag]) for i in [lag, n-1]
            grad[lag:] += w * (1.0 - h[:n-lag])
            # i+lag valid means i <= n-1-lag, so i in [0, n-1-lag]
            # contribution: -h[i+lag] for i in [0, n-1-lag]
            grad[:n-lag] -= w * h[lag:]
        else:
            # lag < 0, let l = -lag > 0
            l = -lag
            # i-lag = i+l valid means i <= n-1-l, so i in [0, n-1-l]
            # contribution: (1-h[i+l]) for i in [0, n-1-l]
            grad[:n-l] += w * (1.0 - h[l:])
            # i+lag = i-l valid means i >= l, so i in [l, n-1]
            # contribution: -h[i-l] for i in [l, n-1]
            grad[l:] -= w * h[:n-l]
    
    grad *= dx
    return grad
```

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * dx)

def project_feasible(v, target):
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(y)
    if abs(res) > 1e-14:
        interior = (y > 1e-12) & (y < 1.0 - 1e-12)
        ni = int(np.sum(interior))
        if ni > 0:
            y[interior] += res / ni
            y = np.clip(y, 0.0, 1.0)
    return y

def interpolate_to_resolution(h, new_n):
    n_old = len(h)
    if n_old == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_ext = np.append(h, h[0])
    h_new = np.interp(x_new, np.append(x_old, 2.0), h_ext)
    target = new_n / 2.0
    return project_feasible(h_new, target)

def compute_lse_grad_vec(h, beta):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(r)
    if beta > 0 and beta * m < 500:
        e = np.exp(beta * (r - m))
        s = np.sum(e)
        weights = e / s
    else:

        weights[np.argmax(r)] = 1.0
    grad = np.zeros(n)

        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]

def gradient_descent_lse(h, beta, target, t_end, max_iters=300):
    n = len(h)
    best = h.copy()
    best_val = evaluate_c5(best)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        grad = compute_lse_grad_vec(h, beta)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        alpha = 1.0
        improved = False
        for _ in range(40):
            h_new = project_feasible(h + alpha * grad, target)
            val_new = evaluate_c5(h_new)
            if val_new < best_val - 1e-16:

                best = h_new.copy()
                best_val = val_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return best, best_val

def subgradient_descent(h, target, t_end, max_iters=500):
    n = len(h)
    best = h.copy()
    best_val = evaluate_c5(best)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        r = np.correlate(h, 1.0 - h, mode="full")
        max_r = np.max(r)

        active = np.where(r >= threshold)[0]
        if len(active) == 0:

                            val_new = evaluate_c5(h_new)
                            if val_new < best_val:
                                best_h = h_new
                                best_val = val_new
    
    # Phase 6: Final LSE + subgradient + pair on best
    if best_h is not None and time.time() < t_end - 10:
        target_b = best_n / 2.0
        for beta in [30, 15, 8]:
            if time.time() > t_end - 8:
                break
            h_work, v = gradient_descent_lse(best_h, beta, target_b, t_end - 8, max_iters=80)
            if v < best_val:
                best_h = h_work.copy()
                best_val = v
        if time.time() < t_end - 5:
            h_work, v = subgradient_descent(best_h, target_b, t_end - 5, max_iters=100)
            if v < best_val:
                best_h = h_work.copy()
                best_val = v
        if time.time() < t_end - 3:
            h_work, v = pair_rotation_polish(best_h, target_b, t_end - 2, max_rounds=200, seed_val=seed + 1)
            if v < best_val:
                best_h = h_work.copy()
                best_val = v
    
    if best_h is None:
        n_fallback = 50
        best_h = np.full(n_fallback, 0.5)
        best_val = evaluate_c5(best_h)
        best_n = n_fallback
    
    best_h = project_feasible(best_h, best_n / 2.0)
    best_val = evaluate_c5(best_h)
    return best_h, best_val, best_n
```
