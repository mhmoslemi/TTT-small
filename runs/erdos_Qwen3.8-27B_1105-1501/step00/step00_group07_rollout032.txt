```python
def gradient_step(h, current_c, step_size=0.01):
    """Take a subgradient step to reduce the max correlation."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(raw)
    
    # Compute subgradient of R[k_star] wrt h
    grad = np.zeros(n)
    for j in range(n):
        # Term 1: (1 - h[j + k_star - (n-1)]) if valid
        idx1 = j + k_star - (n - 1)
        if 0 <= idx1 < n:
            grad[j] += (1.0 - h[idx1])
        # Term 2: -h[j - k_star + (n-1)] if valid
        idx2 = j - k_star + (n - 1)
        if 0 <= idx2 < n:
            grad[j] -= h[idx2]
    
    # Project gradient direction onto the sum-zero subspace
    mean_grad = np.mean(grad)
    grad_zero_mean = grad - mean_grad
    
    # Take a step in the negative gradient direction
    h_new = h - step_size * grad_zero_mean
    h_new = project_feas(h_new)
    c_new = exact_score(h_new)
    
    if c_new < current_c:
        return h_new, c_new
    return h, current_c
```

```python
def full_pair_sweep(h, cur_c, deadline, max_sweeps=25):
    n = len(h)
    for _ in range(max_sweeps):
        if time.time() > deadline - 2.0:
            break
        improved = False
        for a in range(n - 1):
            if time.time() > deadline - 1.5:
                break
            for b in range(a + 1, n):
                t, c, ok = pair_search(h, a, b, cur_c)
                if ok:
                    h[a] += t
                    h[b] -= t
                    h = np.clip(h, 0.0, 1.0)
                    cur_c = c
                    improved = True
        if not improved:
            break
    return h, cur_c

def resample_to(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(target_n) + 0.5) / target_n
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feas(h_new)
```
