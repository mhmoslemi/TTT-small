where [P] is 1 if P is true, 0 otherwise. And if lag = 0, the two terms combine to (1-2h[i]).

Great, so the gradient is:
∂output[m]/∂h[i] = (1-h[i+lag]) * I(0≤i≤n-1, 0≤i+lag≤n-1) - h[i-lag] * I(0≤i≤n-1, 0≤i-lag≤n-1)

where lag = m - (n-1).

Now, the objective is C₅(h) = max_m output[m] * dx = max_m output[m] * 2/n.

The subgradient with respect to h[i] is the average of gradients of all active components (those at the maximum):
∂C₅/∂h[i] ≈ (2/n) * (1/|A|) * sum_{m∈A} ∂output[m]/∂h[i]

where A = {m : output[m] = max(output)}.

For the LP approach, we want to find a direction d that decreases all active components simultaneously:
Subject to: sum(d) = 0 (area constraint)
            -bound ≤ d[i] ≤ bound (to stay in [0,1])
            For each active m: sum_i (∂output[m]/∂h[i]) * d[i] < 0

We want to maximize the minimum decrease:
maximize ρ
subject to: for each active m: (grad_m · d) ≤ -ρ
            sum(d) = 0
            |d[i]| ≤ δ for all i (small step)

This is a linear program! Let me code this up.

**Algorithm outline:**

1. Initialize h (from initial_h_values or a synthetic start)
2. Loop until budget exhausted:
   a. Compute output vector, find active set A
   b. Compute gradients for active components
   c. Solve LP to find best descent direction
   d. Apply direction with small step, project to [0,1] and fix sum
   e. Re-evaluate objective
   f. If improved, accept; else reduce step size

```python
import numpy as np
from scipy.optimize import linprog
import time

TOLERANCE = 1e-12

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_active_gradients(h, activeIndices):
    """
    Compute gradient of output[m] with respect to h[i] for m in activeIndices.
    Returns matrix of shape (len(activeIndices), n).
    """
    n = len(h)
    grad = np.zeros((len(activeIndices), n))
    for idx, m in enumerate(activeIndices):
        lag = m - (n - 1)
        for i in range(n):
            g = 0.0
            # Case 1: j = i, need 0 <= i+lag <= n-1
            if 0 <= i + lag <= n - 1:
                g += (1.0 - h[i + lag])
            # Case 2: j+lag = i, need 0 <= i-lag <= n-1
            if 0 <= i - lag <= n - 1:
                g -= h[i - lag]
            grad[idx, i] = g
    return grad

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    diff = h.sum() - n / 2.0
    if abs(diff) > 1e-14:
        # Distribute the correction evenly, respecting bounds
        n_pos = (h < 1.0).sum() if diff > 0 else (h > 0.0).sum()
        step = diff / n_pos
        if diff > 0:
            h[h < 1.0] -= step
        else:
            h[h > 0.0] -= step
        h = np.clip(h, 0, 1)
        # Final fix
        residual = h.sum() - n / 2.0
        if abs(residual) > 1e-14:
            # More careful adjustment
            free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
            if len(free) > 0:
                h[free] -= residual / len(free)
                h = np.clip(h, 0, 1)
                residual = h.sum() - n / 2.0
                if abs(residual) > 1e-14:
                    # Last resort: adjust one element
                    if residual > 0:
                        idx = np.argmin(h)
                        h[idx] = min(1.0, h[idx] - residual)
                    else:
                        idx = np.argmax(h)
                        h[idx] = max(0.0, h[idx] - residual)
                    h = np.clip(h, 0, 1)
    return h

def solve_lp_direction(h, active_grads, n):
    """
    Solve LP to find descent direction.
    Variables: d[0..n-1], rho
    Minimize: -rho
    Subject to:
        active_grads @ d <= -rho  (i.e., grad_m · d + rho <= 0)
        sum(d) = 0
        -bound <= d[i] <= bound
    """
    n_vars = n + 1  # d[0..n-1] and rho
    # Objective: minimize -rho => c = [0,0,...,0, -1]
    c = np.zeros(n_vars)
    c[-1] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # grad_m · d + rho <= 0 for each active m
    # So A_ub row = [grad_m[0], ..., grad_m[n-1], 1], b = 0
    A_ub = np.hstack([active_grads, np.ones((len(active_grads), 1))])
    b_ub = np.zeros(len(active_grads))
    
    # Equality: sum(d) = 0 => [1,1,...,1, 0] @ x = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -bound <= d[i] <= bound, rho >= 0 (or some lower bound)
    bound = 0.05  # max step per coordinate
    bounds = [(-bound, bound)] * n + [(-0.5, 0.5)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        d = res.x[:n]
        # Ensure sum is exactly 0
        d -= d.mean()
        return d
    return None

def local_search(h, n, budget_time, deadline, active_tol=0.005, max_active=20):
    """
    Iterative local search using LP-based subgradient descent.
    """
    start = time.time()
    best_h = h.copy()
    best_c = exact_objective(h)
    
    # Also track the objective vector
    obj_vec = objective_vector(h)
    current_c = obj_vec.max()
    
    it_count = 0
    stagnation = 0
    
    while time.time() < deadline:
        it_count += 1
        
        # Find active set
        threshold = current_c - active_tol
        active_mask = obj_vec >= threshold
        active_idx = np.where(active_mask)[0]
        
        # Limit active set size
        if len(active_idx) > max_active:
            # Keep the top max_active
            top_order = np.argsort(obj_vec[active_idx])[::-1][:max_active]
            active_idx = active_idx[top_order]
        
        # Compute gradients for active components
        active_grads = compute_active_gradients(h, active_idx)
        
        # Solve LP
        d = solve_lp_direction(h, active_grads, n)
        
        if d is None or np.allclose(d, 0, atol=1e-10):
            stagnation += 1
            if stagnation > 50:
                break
            # Try a random perturbation
            rng = np.random.default_rng(it_count)
            d = rng.normal(0, 0.01, n)
            d -= d.mean()
        else:
            stagnation = 0
        
        # Line search
        alpha = 1.0
        improved = False
        for attempt in range(10):
            trial = h + alpha * d
            trial = np.clip(trial, 0, 1)
            # Fix sum
            diff = trial.sum() - n / 2.0
            if abs(diff) > 1e-10:
                free = np.where((trial > 1e-6) & (trial < 1.0 - 1e-6))[0]
                if len(free) > 0:
                    trial[free] -= diff / len(free)
                    trial = np.clip(trial, 0, 1)
            
            trial_c = exact_objective(trial)
            if trial_c < current_c - 1e-12:
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            h = trial.copy()
            current_c = trial_c
            obj_vec = objective_vector(h)
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            stagnation += 1
            if stagnation > 100:
                break
        
        # Occasionally try random restart within small neighborhood
        if it_count % 200 == 0:
            rng = np.random.default_rng(it_count)
            pert = h + rng.normal(0, 0.005, n)
            pert = np.clip(pert, 0, 1)
            diff = pert.sum() - n / 2.0

                best_c = pert_c
                best_h = pert.copy()
                h = pert.copy()
                current_c = pert_c
                obj_vec = objective_vector(h)
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_active_gradients(h, active_indices):
    n = len(h)
    n_active = len(active_indices)
    grad = np.zeros((n_active, n))
    for a in range(n_active):
        m = active_indices[a]
        lag = m - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + lag <= n - 1:
                g += (1.0 - h[i + lag])
            if 0 <= i - lag <= n - 1:
                g -= h[i - lag]
            grad[a, i] = g
    return grad

def fix_sum(h, n):
    """Ensure sum(h) == n/2, 0<=h<=1, minimally."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-15:
        return h
    # Adjust free variables
    for _ in range(5):
        if diff > 0:
            free = np.where(h < 1.0 - 1e-15)[0]
        else:
            free = np.where(h > 1e-15)[0]
        if len(free) == 0:
            break
        step = diff / len(free)
        if abs(step) >= abs(diff):
            # Just adjust one or a few
            if diff > 0:
                idx = free[np.argmin(h[free])]
                h[idx] = max(0.0, h[idx] - diff)
            else:
                idx = free[np.argmax(h[free])]
                h[idx] = min(1.0, h[idx] - diff)
        else:
            h[free] += -step
            h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
    # Final exact fix
    diff = h.sum() - target
    if abs(diff) > 1e-15:
        # Find best single element to adjust
        if diff > 0:
            candidates = np.where(h > 0)[0]
            idx = candidates[np.argmin(h[candidates])]
            h[idx] -= diff
        else:
            candidates = np.where(h < 1)[0]
            idx = candidates[np.argmax(h[candidates])]
            h[idx] -= diff
    return h

def dyadic_repair(h, n, nbits=20):
    """Convert h to dyadic representation with exact sum."""
    S = 1 << nbits
    if n % 2 == 1:
        S *= 2
    target_z = n * S // 2  # Should be integer since S is even when n is odd
    z = np.round(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    # Fix sum
    diff = z.sum() - target_z
    while diff > 0:
        # Decrease some elements
        candidates = np.where(z > 0)[0]
        # Pick the one where decreasing causes least objective increase
        # For simplicity, pick the one with z closest to S (least impact)
        # Or just pick randomly from those with z > 0
        idx = candidates[np.argmin(z[candidates])]  # Decrease smallest positive
        z[idx] -= 1
        diff -= 1
    while diff < 0:
        # Increase some elements
        candidates = np.where(z < S)[0]
        idx = candidates[np.argmax(z[candidates])]  # Increase largest non-full
        z[idx] += 1
        diff += 1
    return z.astype(np.float64) / S

def resample(h_old, new_n):
    """Resample h from old length to new length, preserving integral."""
    old_n = len(h_old)
    # Cumulative integral
    cum = np.concatenate([[0], np.cumsum(h_old)])
    cum_norm = cum / cum[-1]  # Normalize to [0,1]
    # Sample at new points
    new_points = (np.arange(new_n) + 0.5) / new_n
    h_new = np.zeros(new_n)
    for j in range(new_n):
        # Find position in old array
        pos = new_points[j] * old_n
        idx = int(np.floor(pos))
        frac = pos - idx
        if idx >= old_n - 1:
            h_new[j] = cum_norm[old_n] - cum_norm[old_n - 1]
        else:
            h_new[j] = (cum_norm[min(idx + 1, old_n)] - cum_norm[idx]) / (1.0 / old_n) * (1.0 / new_n) * old_n
    # Hmm this is getting complicated. Let me use a simpler approach.
    # Just use linear interpolation of the density
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x_new, x_old, h_old, left=0, right=0)
    # Scale to preserve integral
    if h_new.sum() > 0:
        h_new = h_new * (old_n / new_n) / h_new.sum() * (new_n / 2.0)
    h_new = np.clip(h_new, 0, 1)
    h_new = fix_sum(h_new, new_n)
    return h_new

def make_initial(n, seed=42):

    rng = np.random.default_rng(seed)
    # Start with a smooth function with some structure
    x = np.linspace(0, 1, n)

    h = np.clip(h, 0, 1)

    return h

def local_search_lp(h, n, deadline, seed=42):
    """LP-based local search."""
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c = exact_objective(h)
    current_c = best_c
    obj_vec = objective_vector(h)
    
    stagnation = 0
    it = 0
    active_tol = 0.003
    max_active = 15
    
    while time.time() < deadline:
        it += 1
        
        # Find active set
        threshold = current_c - active_tol
        active_mask = obj_vec >= threshold
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(obj_vec)])
        
        if len(active_idx) > max_active:
            top = np.argsort(obj_vec[active_idx])[::-1][:max_active]
            active_idx = active_idx[top]
        
        # Compute gradients
        grads = compute_active_gradients(h, active_idx)
        
        # Solve LP
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = -1.0
        
        A_ub = np.hstack([grads, np.ones((len(grads), 1))])
        b_ub = np.zeros(len(grads))

        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = [(-step_bound, step_bound)] * n + [(-0.1, 0.1)]

                         bounds=bounds, method='highs', options={'time_limit': 0.5})
        except:

        if res is not None and res.status == 0:
            d = res.x[:n]
            d -= d.mean()  # Ensure zero sum
        else:

            d = -grads.mean(axis=0)
            d -= d.mean()
            if np.linalg.norm(d) < 1e-10:
                d = rng.normal(0, 0.01, n)
                d -= d.mean()

        alpha = 1.0
        found = False
        for _ in range(8):
            trial = h + alpha * d
            trial = np.clip(trial, 0, 1)
            trial = fix_sum(trial, n)
            trial_c = exact_objective(trial)
            if trial_c < current_c - 1e-13:
                found = True
                break
            alpha *= 0.5

            h = trial.copy()
            current_c = trial_c
            obj_vec = objective_vector(h)
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation % 50 == 0 and stagnation > 100:
                # Random restart

                pert = fix_sum(pert, n)

                    best_h = pert.copy()
                    h = pert.copy()
                    current_c = pert_c
                    obj_vec = objective_vector(h)
                    stagnation = 0
            if stagnation > 300:

def pair_sweep(h, n, deadline, seed=42):
    """Try pairwise transfers to improve."""
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c = exact_objective(h)
    current_c = best_c
    
    # Scale for pairwise moves
    scale = 1000  # h values in units of 1/scale
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try transferring from i to j or j to i
        for donor, receiver in [(i, j), (j, i)]:
            # Try different amounts
            for amt in [1, 2, 5, 10, 20, 50]:

                if h[donor] < delta or h[receiver] + delta > 1:

                trial[donor] -= delta
                trial[receiver] += delta
                trial_c = exact_objective(trial)
                if trial_c < current_c - 1e-13:
                    h = trial.copy()
                    current_c = trial_c
                    if current_c < best_c:
                        best_c = current_c
                        best_h = h.copy()
                    break  # Move to next pair
    
    return best_h, best_c
```
