```python
import time
from scipy.optimize import minimize

def official_score(h):
    """Compute the exact C5 score for a feasible h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(z, n):
    """Project z onto {h : 0<=h[i]<=1, sum(h)=n/2} using bisection."""
    target = n / 2.0
    h = z.astype(float).copy()
    # First clip
    h = np.clip(h, 0.0, 1.0)
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_shifted = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_shifted)
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    remaining = target - np.sum(h)
    if remaining > 0:
        # Distribute remaining to elements with most room
        order = np.argsort(1.0 - h)
        for i in order:
            if remaining <= 1e-14:
                break
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
    elif remaining < 0:
        order = np.argsort(h)
        for i in order:
            if -remaining <= 1e-14:
                break
            sub = min(-remaining, h[i])
            h[i] -= sub
            remaining += sub
    return h

def compute_F_values(h):
    """Compute F(k) for all lags k = -(n-1) to n-1."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr  # length 2n-1

defgradient_F(h, k, n):
    """Compute gradient of F(k) wrt h."""
    # dF(k)/dh[p] = (1-h[p+k]) * I[0<=p+k<=n-1] - h[p-k] * I[0<=p-k<=n-1]
    g = np.zeros(n)
    # First term: (1-h[p+k]) for valid p+k
    if k >= 0:
        # p ranges from 0 to n-1-k
        p_range = np.arange(0, n - k)
        g[p_range] += (1.0 - h[p_range + k])
    else:
        # k < 0, p+k = p + k. Need 0 <= p+k <= n-1, so -k <= p <= n-1
        # p ranges from -k to n-1
        p_range = np.arange(-k, n)
        g[p_range] += (1.0 - h[p_range + k])
    # Second term: -h[p-k] for valid p-k
    if k >= 0:
        # p-k ranges from 0 to n-1, so p ranges from k to n-1
        p_range = np.arange(k, n)
        g[p_range] -= h[p_range - k]
    else:

        p_range = np.arange(0, n + k)  # n + k = n - |k|
        g[p_range] -= h[p_range - k]
    return g
```

```python
def grad_corr(h, j, n):
    """Gradient of c[j] = (correlate(h, 1-h, 'full'))[j] wrt h."""
    k = j - (n - 1)
    g = np.zeros(n)
    # Term 1: (1-h[p-k]) for p where 0 <= p-k <= n-1
    # p-k >= 0 → p >= k
    # p-k <= n-1 → p <= n-1+k
    # Also 0 <= p <= n-1
    p_lo = max(0, k)
    p_hi = min(n - 1, n - 1 + k)
    if p_lo <= p_hi:
        g[p_lo:p_hi+1] += (1.0 - h[p_lo - k:p_hi + 1 - k])
    # Term 2: -h[p+k] for p where 0 <= p+k <= n-1
    # p+k >= 0 → p >= -k
    # p+k <= n-1 → p <= n-1-k
    # Also 0 <= p <= n-1
    p_lo2 = max(0, -k)
    p_hi2 = min(n - 1, n - 1 - k)
    if p_lo2 <= p_hi2:
        g[p_lo2:p_hi2+1] -= h[p_lo2 + k:p_hi2 + 1 + k]
    return g
```

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(z, n):
    target = n / 2.0
    h = np.clip(np.asarray(z, dtype=float).ravel(), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    rem = target - np.sum(h)
    if rem > 1e-14:
        for i in np.argsort(1.0 - h):
            if rem <= 1e-14: break
            a = min(rem, 1.0 - h[i]); h[i] += a; rem -= a
    elif rem < -1e-14:
        for i in np.argsort(h):
            if -rem <= 1e-14: break
            a = min(-rem, h[i]); h[i] -= a; rem += a
    return h

def grad_corr(h, j, n):
    k = j - (n - 1)
    g = np.zeros(n)
    p_lo = max(0, k)
    p_hi = min(n - 1, n - 1 + k)

def active_set_descent(h, n, max_steps=200, tol=0.05, max_active=6):
    """Active set descent method."""
    h = h.copy()
    dx = 2.0 / n
    current_score = official_score(h)
    
    for step in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        
        # Find active constraints
        active_mask = corr > max_val - tol * max_val - 1e-6
        active_indices = np.where(active_mask)[0]
        if len(active_indices) > max_active:
            active_indices = active_indices[np.argsort(-corr[active_indices])[:max_active]]
        
        # Compute descent direction
        d = np.zeros(n)
        for j in active_indices:
            d -= grad_corr(h, j, n)
        
        # Center to preserve sum
        d = d - np.mean(d)
        norm = np.max(np.abs(d))
        if norm < 1e-12:
            break
        d /= norm
        
        # Line search
        pos = d > 1e-12
        neg = d < -1e-12
        max_alpha = 1.0
        if np.any(pos):
            max_alpha = min(max_alpha, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            max_alpha = min(max_alpha, np.min(h[neg] / (-d[neg])))
        if not np.isfinite(max_alpha) or max_alpha < 1e-8:
            break
        max_alpha = min(max_alpha, 1.0)
        
        improved = False
        for i in range(20):
            alpha = max_alpha * (0.7 ** i)
            if alpha < 1e-10:
                break
            h_new = project_feasible(h + alpha * d, n)
            s = official_score(h_new)
            if s < current_score - 1e-12:
                h = h_new
                current_score = s
                improved = True
                break
        
        if not improved:
            # Try smaller steps or break
            break
    
    return h, current_score
