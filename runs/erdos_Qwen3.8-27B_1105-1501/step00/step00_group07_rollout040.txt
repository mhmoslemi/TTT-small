import numpy as np
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(F) * dx)

def resample_to_n(h_old, n_new):
    """Resample h from its native resolution to n_new points."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    # Project to [0,1]
    h_new = np.clip(h_new, 0.0, 1.0)
    # Adjust sum to be exactly n_new/2
    target = n_new / 2.0
    current = h_new.sum()
    diff = target - current
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            # Need to add mass; add to positions with most room
            rooms = 1.0 - h_new
            order = np.argsort(-rooms)
            added = 0.0
            for idx in order:
                add = min(rooms[idx], diff - added)
                if add > 0:
                    h_new[idx] += add
                    added += add
                if added >= diff - 1e-12:
                    break
        else:
            # Need to remove mass
            order = np.argsort(-h_new)
            removed = 0.0
            for idx in order:
                rem = min(h_new[idx], -diff - removed)
                if rem > 0:
                    h_new[idx] -= rem
                    removed += rem
                if removed >= -diff - 1e-12:
                    break
    return h_new
