def score_exact(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    if n < 2:

def project_feasible(y, target=None):

def smooth_logsumexp(c, T):
    """Compute smooth max and weights for log-sum-exp with temperature T."""

        w[max_mask] = 1.0 / max_mask.sum()

def resample_h(h, n_new):

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(y, target=None):
    y = np.asarray(y, dtype=np.float64).ravel()
    n = len(y)
    if target is None:
        target = n * 0.5
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - z)
        else:
            idx = np.argmin(z)
        z[idx] = np.clip(z[idx] + diff, 0.0, 1.0)
        diff2 = target - z.sum()
        if abs(diff2) > 1e-14:
            mask = (z > 1e-12) & (z < 1.0 - 1e-12)
            if mask.any():
                z[mask] += diff2 / mask.sum()
                z = np.clip(z, 0.0, 1.0)
    return z

def smooth_logsumexp(c, T):
    c_max = np.max(c)
    ex = np.exp((c - c_max) / T)
    S = np.sum(ex)
    if S < 1e-300:
        w = np.zeros_like(c)
        max_mask = np.isclose(c, c_max, atol=1e-14)
        w[max_mask] = 1.0 / max(max_mask.sum(), 1)
        return c_max, w
    w = ex / S
    smooth_val = c_max + T * np.log(S)
    return smooth_val, w

def smooth_obj_and_grad(h, T):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    dx = 2.0 / n
    c = scores_array(h)
    smooth_val, w = smooth_logsumexp(c, T)
    g = np.zeros(n, dtype=np.float64)
    center = n - 1
    g += w[center] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[center + d]
        if wp > 1e-15:
            g[d:] += wp * dx * (1.0 - h[:n-d])
            g[:n-d] += wp * dx * (-h[d:])
        wn = w[center - d]
        if wn > 1e-15:
            g[:n-d] += wn * dx * (1.0 - h[d:])
            g[d:] += wn * dx * (-h[:n-d])
    return smooth_val, g

def random_feasible(n, rng):
    y = rng.random(n)
    return project_feasible(y, n * 0.5)

def resample_h(h, n_new):
    h = np.asarray(h, dtype=np.float64).ravel()
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(y, n_new * 0.5)

def pair_line_search(h, i, j, deadline):
    h_work = np.array(h, dtype=np.float64)
    base_c = score_exact(h_work)
    lo = max(-h_work[i], h_work[j] - 1.0)
    hi = min(1.0 - h_work[i], h_work[j])
    if hi - lo < 1e-10:
        return h, base_c
    n_samples = 21
    best_alpha = 0.0
    best_c = base_c
    alphas = np.linspace(lo, hi, n_samples)
    for a in alphas:
        if time.time() > deadline:
            break
        h_work[i] = h[i] + a
        h_work[j] = h[j] - a
        c = score_exact(h_work)
        if c < best_c:
            best_c = c
            best_alpha = a
    h_out = np.array(h, dtype=np.float64)
    h_out[i] = h[i] + best_alpha
    h_out[j] = h[j] - best_alpha
    return h_out, best_c

def local_refine(h0, deadline, rng):
    h = np.array(h0, dtype=np.float64)
    n = len(h)
    best_h = h.copy()
    best_c = score_exact(h)
    T = 0.10
    step_size = 0.05
    no_improve = 0
    min_step = 1e-6
    T_floor = 5e-5
    
    while time.time() < deadline:
        skip_pair = (time.time() > deadline - 3)
        if not skip_pair and rng.random() < 0.3:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                h_new, c_new = pair_line_search(h, i, j, deadline - 2)
                if c_new < best_c - 1e-12:
                    best_h = h_new
                    best_c = c_new
                    h = h_new
                    no_improve = 0
                    continue
                else:
                    no_improve += 1
        
        try:
            _, g = smooth_obj_and_grad(h, T)
        except (ValueError, FloatingPointError, OverflowError):
            T = min(0.5, T * 1.5)
            no_improve += 1
            if no_improve > 40:
                break
            continue
        
        if not np.all(np.isfinite(g)):
            T = min(0.5, T * 1.5)
            no_improve += 1
            continue
        
        g_norm = np.sqrt(np.mean(g**2))
        if g_norm < 1e-13:
            T = max(T_floor, T * 0.8)
            if T <= T_floor:
                break
            continue
        
        improved = False
        for trial in range(6):
            sc = step_size * (0.6 ** trial)
            if sc < min_step:
                break
            h_trial = h - sc * g / g_norm
            h_trial = project_feasible(h_trial, n * 0.5)
            c_trial = score_exact(h_trial)
            if c_trial < best_c - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_c = c_trial
                improved = True
                step_size = min(step_size * 1.3, 0.25)
                break
        
        if improved:
            no_improve = 0
            T = max(T_floor, T * 0.85)
        else:
            no_improve += 1
            step_size *= 0.8
            if no_improve >= 20:
                T = max(T_floor, T * 0.85)
                no_improve = 0
                step_size = max(step_size, 0.005)
        
        if T <= T_floor and step_size < min_step:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    deadline = start_time + max(budget_s - 5, 30)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    def _update(h, c, n=None):
        nonlocal best_h, best_c, best_n
        if n is None:
            n = len(h)
        if not np.isfinite(c):
            return
        if c < best_c - 1e-11 or (abs(c - best_c) <= 1e-11 and n < best_n):
            best_h = np.array(h, dtype=np.float64)
            best_c = float(c)
            best_n = int(n)
    
    # Get initial candidate
    n0 = 70
    h_parent = None
    try:
        h_p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(h_p) >= 4:
            n0 = len(h_p)
            h_parent = project_feasible(h_p, n0 * 0.5)
            c_p = score_exact(h_parent)
            _update(h_parent, c_p, n0)
    except (NameError, TypeError, ValueError):
        pass
    
    if best_h is None:
        h_parent = random_feasible(n0, rng)
        _update(h_parent, score_exact(h_parent), n0)
    
    # Multi-resolution seeding
    res_list = sorted(set([n0, max(28, n0 - 14), n0 + 14, n0 + 28, max(20, n0 - 28)]))
    seed_candidates = [h_parent.copy()]
    for n_r in res_list:
        if time.time() > deadline - 15:
            break
        if n_r == n0:
            continue
        h_r = resample_h(h_parent, n_r)
        c_r = score_exact(h_r)
        _update(h_r, c_r, n_r)
        seed_candidates.append(h_r.copy())
    
    # Binarized version
    h_bin = np.zeros(len(h_parent))
    n_ones = int(round(len(h_parent) * 0.5))
    top_idx = np.argsort(h_parent)[-n_ones:]
    h_bin[top_idx] = 1.0
    h_bin = project_feasible(h_bin, len(h_bin) * 0.5)
    _update(h_bin, score_exact(h_bin), len(h_bin))
    seed_candidates.append(h_bin.copy())
    
    # Scaled versions
    for s in [1.3, 1.6, 0.7]:
        h_sc = 0.5 + s * (h_parent - 0.5)
        h_sc = project_feasible(h_sc, len(h_sc) * 0.5)
        _update(h_sc, score_exact(h_sc), len(h_sc))
        seed_candidates.append(h_sc.copy())
    
    # Phase 1: Main local refinement on global best
    if time.time() < deadline - 30:
        t_alloc = min((deadline - time.time()) * 0.35, 180.0)
        h_r, c_r = local_refine(best_h, start_time + t_alloc, rng)
        _update(h_r, c_r, len(h_r))
    
    # Phase 2: Refine other seeds
    for h_s in seed_candidates:
        if time.time() > deadline - 20:
            break
        h_m = h_s + rng.randn(len(h_s)) * 0.015
        h_m = project_feasible(h_m, len(h_m) * 0.5)
        c_m = score_exact(h_m)
        if c_m < best_c + 0.03:
            t_alloc = min((deadline - time.time()) * 0.12, 70.0)
            if t_alloc > 10:
                h_r, c_r = local_refine(h_m, start_time + t_alloc, rng)
                _update(h_r, c_r, len(h_r))
    
    # Phase 3: Fine perturbations around best
    if time.time() < deadline - 15:
        for sigma in [0.004, 0.0015, 0.0005]:
            if time.time() > deadline - 10:
                break
            h_pert = best_h + rng.randn(len(best_h)) * sigma
            h_pert = project_feasible(h_pert, len(h_pert) * 0.5)
            t_alloc = min((deadline - time.time()) * 0.18, 50.0)
            if t_alloc > 8:
                h_r, c_r = local_refine(h_pert, start_time + t_alloc, rng)
                _update(h_r, c_r, len(h_r))
    
    # Phase 4: Final polish
    if time.time() < deadline - 8:
        t_alloc = min((deadline - time.time()) * 0.5, 80.0)
        if t_alloc > 10:
            h_r, c_r = local_refine(best_h, start_time + t_alloc, rng)
            _update(h_r, c_r, len(h_r))
    
    # Final validation
    h_final = np.array(best_h, dtype=np.float64)
    n_final = len(h_final)
    h_final = project_feasible(h_final, n_final * 0.5)
    c_final = score_exact(h_final)
    if c_final < best_c:
        best_c = c_final
    
    return h_final, best_c, n_final
