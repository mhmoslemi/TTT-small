    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Full pair sweeps (main optimization)
    h_work = best_h.copy()
    current_c5 = best_c5
    no_improve_count = 0
    sweep_count = 0
    phase1_end = budget_s * 0.55

    while elapsed() < phase1_end:
        h_work, current_c5, improved = sweep_all_pairs(h_work, rng, current_c5)
        sweep_count += 1
        if current_c5 < best_c5 - 1e-13:
            best_c5 = current_c5
            best_h = h_work.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        if no_improve_count > 8:
            # Perturb and restart
            h_work = best_h + rng.randn(n) * 0.02
            h_work = project_feasible(h_work, target_sum)
            current_c5 = compute_c5(h_work)
            no_improve_count = 0

    # Phase 2: Random pair search for fine-tuning
    if elapsed() < budget_s * 0.75:
        h_work = best_h.copy()
        for _ in range(50000):
            if elapsed() > budget_s * 0.75:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for (a, b) in [(i, j), (j, i)]:
                h_new, c5_new = ternary_optimize_pair(h_work, a, b)
                if c5_new < best_c5 - 1e-13:
                    h_work = h_new
                    best_c5 = c5_new
                    best_h = h_work.copy()
                    break

    # Phase 3: Smooth mode optimization
    if elapsed() < budget_s * 0.88:
        remaining = budget_s * 0.88 - elapsed()
        h_mode, c5_mode = smooth_mode_optimize(
            best_h, 6, target_sum, remaining, seed
        )
        if c5_mode < best_c5:
            best_c5 = c5_mode
            best_h = h_mode

    # Phase 4: Try different n values
    if elapsed() < budget_s * 0.94:
        for n_try in [n - 2, n - 1, n + 1, n + 2, n + 4]:
            if n_try < 20 or n_try > 200:
                continue
            remaining = budget_s * 0.94 - elapsed()
            h_try, c5_try, n_try_actual = try_different_n(
                n_try, seed + n_try, remaining, best_c5
            )
            if c5_try is not None and c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                n = n_try_actual

    # Phase 5: Final polish on best solution
    if elapsed() < budget_s * 0.98:
        n = len(best_h)
        target_sum = n / 2.0
        h_work = best_h.copy()
        for _ in range(30000):
            if elapsed() > budget_s * 0.98:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for (a, b) in [(i, j), (j, i)]:
                h_new, c5_new = ternary_optimize_pair(h_work, a, b)
                if c5_new < best_c5 - 1e-14:
                    h_work = h_new
                    best_c5 = c5_new
                    best_h = h_work.copy()
                    break

    # Final feasibility check
    best_h = project_feasible(best_h, n / 2.0)
    best_c5_final = compute_c5(best_h)
    if best_c5_final > best_c5:
        best_c5 = best_c5_final

    return best_h, best_c5, n
