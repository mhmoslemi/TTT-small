    n = len(h)

    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def compute_score_and_grad(h):
    """Compute score and gradient."""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(F))
    k = k_star - (n - 1)
    G = 1.0 - h
    grad = np.zeros(n)
    # dF/dh[j] = G[j-k] (if valid) - h[j+k] (if valid)
    # G[j-k] valid when 0 <= j-k < n, i.e., k <= j < k+n
    lo = max(0, k)
    hi = min(n, k + n)  # exclusive
    if lo < hi:
        grad[lo:hi] += G[lo-k:hi-k]
    # -h[j+k] valid when 0 <= j+k < n, i.e., -k <= j < n-k
    lo2 = max(0, -k)
    hi2 = min(n, n - k)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2+k:hi2+k]
    return float(F[k_star]) * dx, grad * dx

                              bounds=bounds, constraints=cons,
                              options={'maxiter': 300, 'ftol': 1e-14})
                if res.fun < compute_score(h):
                    h_new = fix_sum(np.clip(res.x, 0, 1), n)
                    if compute_score(h_new) < compute_score(h):
                        h = h_new
            except:
                pass
        
        # SA
        sa_t = t_alloc - slsqp_t - 1
        if sa_t > 1:
            h, _ = sa_optimize(h, n, rng, sa_t, best_score)
        
        s = compute_score(h)
        if s < best_score:
            best_score = s
            best_h = h.copy()
    
    return best_h, best_score, len(best_h)
