For c_k (the k-th component of the correlation, before multiplying by dx):

For k = 0: c_0 = sum_{i=0}^{n-1} h[i] * (1-h[i]) = sum h[i] - sum h[i]^2

For k < 0 (let p = -k > 0): c_{-p} = sum_{i=0}^{n-1-p} h[i] * (1-h[i+p])
dc_{-p}/dh[i] = (1-h[i+p]) * [0 <= i <= n-1-p] - h[i-p] * [p <= i <= n-1]

So for the smooth objective:
smooth = (1/t) * log(sum_k exp(t * c_k * dx))

d(smooth)/dh[i] = dx * sum_k w_k * dc_k/dh[i]

where w_k = exp(t*c_k*dx) / sum_j exp(t*c_j*dx)

```python
def analytic_smooth_grad(h, t):
    """Analytical gradient of the smooth C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx  # shape: (2n-1,)
    # vals[k + n - 1] corresponds to lag k (from -(n-1) to n-1)
    
    # Compute weights
    v_max = vals.max()
    exp_vals = np.exp(t * (vals - v_max))
    weights = exp_vals / exp_vals.sum()  # shape: (2n-1,)
    
    grad = np.zeros(n)
    
    # For each lag k from -(n-1) to (n-1):
    # The index in weights/corr array is k + (n-1)
    for lag in range(-(n-1), n):
        w = weights[lag + (n-1)]
        if w < 1e-15:
            continue
        p = abs(lag)
        if lag >= 0:
            # c_p = sum_{i=0}^{n-1-p} h[i+p]*(1-h[i])
            # dc_p/dh[i] = (1-h[i-p])*[p<=i<=n-1] - h[i+p]*[0<=i<=n-1-p]
            # First term: i from p to n-1
            if p > 0:
                grad[p:] += w * (1.0 - h[:n-p])  # (1-h[i-p]) for i=p..n-1
            # Second term: i from 0 to n-1-p
            grad[:n-p] -= w * h[p:]  # h[i+p] for i=0..n-1-p
        else:
            # c_{-p} = sum_{i=0}^{n-1-p} h[i]*(1-h[i+p])
            # dc_{-p}/dh[i] = (1-h[i+p])*[0<=i<=n-1-p] - h[i-p]*[p<=i<=n-1]
            # First term: i from 0 to n-1-p
            grad[:n-p] += w * (1.0 - h[p:])  # (1-h[i+p]) for i=0..n-1-p
            # Second term: i from p to n-1
            if p > 0:
                grad[p:] -= w * h[:n-p]  # h[i-p] for i=p..n-1
    
    return dx * grad
```

```python
def evaluate_c5(h):
```

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_components(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    v_max = vals.max()
    exp_v = np.exp(t * (vals - v_max))
    w = exp_v / exp_v.sum()
    offset = n - 1
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wi = w[lag + offset]
        if wi < 1e-12:
            continue
        p = abs(lag)
        if lag >= 0:
            if p > 0:
                grad[p:] += wi * (1.0 - h[:n-p])
            grad[:n-p] -= wi * h[p:]
        else:
            grad[:n-p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wi * h[:n-p]
```

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    v_max = vals.max()
    exp_v = np.exp(t * (vals - v_max))
    w = exp_v / exp_v.sum()
    offset = n - 1
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wi = w[lag + offset]
        if wi < 1e-12:
            continue
        p = abs(lag)
        if p == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[p:] += wi * (1.0 - h[:n-p])
            grad[:n-p] -= wi * h[p:]
        else:
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n-p]
    return dx * grad

def project(h, S):
