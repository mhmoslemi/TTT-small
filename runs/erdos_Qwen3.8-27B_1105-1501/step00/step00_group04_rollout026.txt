        prob.solve(solver=cp.ECOS, max_iters=15000, abstol=1e-10, reltol=1e-10)
        if u.value is not None and np.all(np.isfinite(u.value)):
            u_val = np.array(u.value, dtype=np.float64).copy()
            u_val -= np.mean(u_val)
            if np.linalg.norm(u_val) > 1e-10:
                return u_val
    except Exception:
        pass

    try:
        prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-9)
        if u.value is not None and np.all(np.isfinite(u.value)):
            u_val = np.array(u.value, dtype=np.float64).copy()
            u_val -= np.mean(u_val)
            if np.linalg.norm(u_val) > 1e-10:
                return u_val
    except Exception:
        pass

    return None


def _line_search(h, direction, current_c, n, rng):
    best_h = h.copy()
    best_c = current_c

    pos = direction > 1e-14
    neg = direction < -1e-14
    alpha_max = 1.0
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min((-h[neg]) / (-direction[neg]))))
    alpha_max = max(alpha_max, 1e-10)
    alpha_max = min(alpha_max, 1.0)

    alphas = []
    a = alpha_max
    for _ in range(12):
        alphas.append(a)
        a *= 0.72
    alphas += list(rng.uniform(0.005, alpha_max, size=10))
    alphas = sorted(set([max(1e-10, min(x, alpha_max)) for x in alphas]), reverse=True)

    for alpha in alphas:
        trial = h + alpha * direction
        if trial.min() < -1e-9 or trial.max() > 1.0 + 1e-9:
            trial = np.clip(trial, 0.0, 1.0)
        if abs(np.sum(trial) - n / 2.0) > 1e-7:
            trial = _make_feasible(trial, n)
        c = _score(trial)
        if c < best_c - 3e-13:
            best_c = c
            best_h = trial.copy()

    return best_h, best_c


def _pair_polish(h, n, num_trials, seed):
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c = _score(h)

    for _ in range(num_trials):
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        t_lo = max(-best_h[a], best_h[b] - 1.0)
        t_hi = min(1.0 - best_h[a], best_h[b])
        if t_hi - t_lo < 1e-12:
            continue
        t = float(rng.uniform(t_lo + 1e-12, t_hi - 1e-12))
        trial = best_h.copy()
        trial[a] += t
        trial[b] -= t
        c = _score(trial)
        if c < best_c - 3e-13:
            best_c = c
            best_h = trial

    return best_h, best_c


def _perturb_reoptimize(h, n, best_c, sigma, seed, deadline):
    rng = np.random.default_rng(seed)
    noisy = h + sigma * rng.standard_normal(n)
    noisy = _make_feasible(noisy, n)
    local_best_h = noisy.copy()
    local_best_c = _score(noisy)

    if local_best_c >= best_c:
        return h, best_c

    # Quick local optimization from perturbed point
    rho = 0.1
    for _ in range(10):
        if time.monotonic() > deadline - 5:
            break
        M, corr = _score_unscaled(local_best_h)
        tol = 0.04 * M + 0.4
        active = _find_active(corr, M, tol, 20)
        direction = _solve_qp(local_best_h, active, rho, 0.3, 50.0, n)
        if direction is None:
            break
        new_h, new_c = _line_search(local_best_h, direction, local_best_c, n, rng)
        if new_c < local_best_c - 1e-12:
            local_best_h = new_h
            local_best_c = new_c
        else:
            break

    if local_best_c < best_c:
        return local_best_h, local_best_c
    return h, best_c


def _run_qp_phase(h, n, best_c, deadline, seed, rho_start, eta_start, tol_factor):
    rng = np.random.default_rng(seed)
    current_h = h.copy()
    current_c = best_c
    rho = rho_start
    fails = 0

    while time.monotonic() < deadline - 10 and fails < 60:
        M, corr = _score_unscaled(current_h)
        tol = tol_factor * M + 0.3
        max_active = min(50, max(12, n // 2))
        active = _find_active(corr, M, tol, max_active)

        if len(active) < 2:
            active = [int(i) for i in np.argsort(-corr)[:10]]

        direction = _solve_qp(current_h, active, rho, eta_start, 40.0, n)
        if direction is None:
            fails += 1
            rho *= 0.85
            if rho < 0.02:
                rho = rho_start * 0.5
            continue

        new_h, new_c = _line_search(current_h, direction, current_c, n, rng)
        if new_c < current_c - 1e-11:
            current_h = new_h
            current_c = new_c
            fails = 0
            rho = min(rho * 1.05, 0.6)
            if new_c < best_c:
                best_c = new_c
                best_h = current_h.copy()
        else:
            fails += 1
            rho *= 0.85
            if rho < 0.01:
                rho = rho_start

    return current_h, best_c, best_h


def _run_pair_phase(h, n, best_c, deadline, seed):
    num = max(100, int((deadline - time.monotonic()) * 150))
    num = min(num, 20000)
    best_h, best_c = _pair_polish(h, n, num, seed)
    return best_h, best_c


def _try_length_n(n_try, best_c, deadline, seed):
    """Try constructing a solution at a different length."""
    n = n_try
    if n < 6 or n > 200:
        return None

    # Construct initial: smooth step-like function
    x = np.linspace(0, 2, n)
    h_try = np.zeros(n)
    h_try[:n // 2] = 1.0
    h_try = _make_feasible(h_try, n)
    c = _score(h_try)

    if c >= best_c:
        # Try a different initialization
        rng = np.random.default_rng(seed)
        h_try = np.zeros(n)
        indices = rng.choice(n, size=n // 2, replace=False)
        h_try[indices] = 1.0
        h_try = _make_feasible(h_try, n)
        c = _score(h_try)
        if c >= best_c:
            return None

    # Quick optimization
    best_h_local = h_try.copy()
    best_local = c
    rho = 0.1
    for _ in range(8):
        if time.monotonic() > deadline - 5:
            break
        M, corr = _score_unscaled(best_h_local)
        tol = 0.05 * M + 0.5
        active = _find_active(corr, M, tol, 15)
        direction = _solve_qp(best_h_local, active, rho, 0.25, 40.0, n)
        if direction is None:
            break
        rng2 = np.random.default_rng(seed + 7)
        new_h, new_c = _line_search(best_h_local, direction, best_local, n, rng2)
        if new_c < best_local - 1e-12:
            best_h_local = new_h
            best_local = new_c
        else:
            break

    if best_local < best_c:
        return best_h_local, best_local
    return None


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s)

    # Get initial values
    h = None
    try:
        h = np.array(initial_h_values, dtype=np.float64).flatten()
    except Exception:
        h = None

    if h is not None and len(h) >= 4:
        n = len(h)
        h = _make_feasible(h, n)
    else:
        n = 70
        h = np.full(n, 0.5)

    best_h = h.copy()
    best_c = _score(h)

    # Phase 1: Active-set QP with moderate parameters
    time_rem = deadline - time.monotonic()
    if time_rem > 30:
        best_h, best_c, _ = _run_qp_phase(
            h, n, best_c, deadline, seed,
            rho_start=0.15, eta_start=0.3, tol_factor=0.05
        )

    # Phase 2: Pair polish
    time_rem = deadline - time.monotonic()
    if time_rem > 5:
        best_h, best_c = _run_pair_phase(best_h, n, best_c, deadline, seed + 100)

    # Phase 3: Finer QP with smaller tolerance
    time_rem = deadline - time.monotonic()
    if time_rem > 20:
        best_h, best_c, _ = _run_qp_phase(
            best_h, n, best_c, deadline, seed + 200,
            rho_start=0.08, eta_start=0.2, tol_factor=0.02
        )

    # Phase 4: More pair polish
    time_rem = deadline - time.monotonic()
    if time_rem > 3:
        best_h, best_c = _run_pair_phase(best_h, n, best_c, deadline, seed + 300)

    # Phase 5: Perturbation and re-optimization
    time_rem = deadline - time.monotonic()
    if time_rem > 40:
        for sigma in [0.03, 0.02, 0.01, 0.05, 0.015]:
            if time.monotonic() > deadline - 15:
                break
            new_h, new_c = _perturb_reoptimize(
                best_h, n, best_c, sigma, seed + 500 + int(sigma * 1000), deadline
            )
            if new_c < best_c:
                best_h = new_h
                best_c = new_c

    # Phase 6: Final fine QP
    time_rem = deadline - time.monotonic()
    if time_rem > 15:
        best_h, best_c, _ = _run_qp_phase(
            best_h, n, best_c, deadline, seed + 600,
            rho_start=0.05, eta_start=0.15, tol_factor=0.01
        )

    # Phase 7: Final pair polish
    time_rem = deadline - time.monotonic()
    if time_rem > 3:
        best_h, best_c = _run_pair_phase(best_h, n, best_c, deadline, seed + 700)

    # Verify and return
    final_c = _score(best_h)
    if final_c > best_c:
        best_c = final_c
    return best_h, best_c, len(best_h)
