```python
def make_feasible(h, target_sum):
    """Project h to be feasible: 0 <= h <= 1, sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:

            # Need to add mass
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            # Need to remove mass
            h += (-diff) * h / max(np.sum(h), 1e-12)
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

        if residual > 0:

                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add

                rem = min(-residual, h[i])
                h[i] -= rem
                residual += rem
                if abs(residual) < 1e-12:
                    break
    return h
```

```python
def coordinate_descent(h, target_sum, deadline):
    n = len(h)
    best_c5 = c5_value(h)
    
    for _ in range(100):  # max passes
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            # Optimize h[i] while keeping sum fixed
            # If I change h[i] by δ, I need to compensate elsewhere
            # Simplest: change h[i] by δ and adjust h[j] by -δ for some j
            # But this is just a pair swap
            # Alternative: change h[i] by δ and spread -δ evenly over all other elements
            # h_new[i] = h[i] + δ, h_new[j] = h[j] - δ/(n-1) for j ≠ i
            # This preserves the sum
            # Feasibility: h[i]+δ ∈ [0,1] and h[j] - δ/(n-1) ∈ [0,1] for all j≠i
            # So δ ∈ [-h[i], 1-h[i]] and for all j≠i: δ/(n-1) ∈ [h[j]-1, h[j]]
            # The tightest constraint is the minimum h[j] for j≠i
            # δ/(n-1) ≤ h[j] for all j≠i, so δ ≤ (n-1)*min_{j≠i} h[j]
            # And δ/(n-1) ≥ h[j]-1, so δ ≥ (n-1)*(max_{j≠i} h[j] - 1)
            
            min_other = np.min(np.delete(h, i))
            max_other = np.max(np.delete(h, i))
            delta_lo = max(-h[i], (n-1)*(max_other - 1.0))
            delta_hi = min(1.0 - h[i], (n-1) * min_other)
            
            if delta_lo >= delta_hi:
                continue
            
            # Try a few values of δ
            for frac in np.linspace(0, 1, 21):
                delta = delta_lo + frac * (delta_hi - delta_lo)
                h_new = h.copy()
                h_new[i] += delta
                h_new[np.arange(n) != i] -= delta / (n-1)
                if np.min(h_new) < -1e-10 or np.max(h_new) > 1.0 + 1e-10:
                    continue
                c = c5_value(h_new)
                if c < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c
                    improved = True
        if not improved:
            break
    
    return h
```
