import numpy as np
import time

def make_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64).copy()
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(z.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            m = z < 1.0
        else:
            m = z > 0.0
        cnt = int(m.sum())
        if cnt > 0:
            z[m] += diff / cnt
            z = np.clip(z, 0.0, 1.0)
    return z

def score_exact(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * (2.0 / n))

def get_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def smooth_gradient(h, T):
    n = len(h)
    c = get_scores(h)
    base = float(c.max())
    ex = np.exp((c - base) / T)
    S = float(ex.sum())
    if S < 1e-300 or not np.isfinite(S):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / S
    dx = 2.0 / n
    ctr = n - 1
    g = np.zeros(n)
    for d in range(-n + 1, n):
        wk = w[ctr + d] * dx
        if abs(wk) < 1e-18:
            continue
        if d == 0:
            g += wk * (1.0 - 2.0 * h)
        elif d > 0:
            g[d:] += wk * (1.0 - h[:n - d])
            g[:n - d] -= wk * h[d:]
        else:
            p = -d
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g

def pair_optimize(h, deadline, rng, n_trials=50, refine_grid=40):
    n = len(h)
    best_c = score_exact(h)
    g = smooth_gradient(h, 0.003)
    order = np.argsort(g)
    for _ in range(n_trials):
        if time.time() >= deadline:
            break
        r = rng.random()
        if r < 0.35:
            i = int(order[rng.integers(min(12, n))])
            j = int(order[rng.integers(max(0, n - 12), n)])
        elif r < 0.6:
            i = int(order[rng.integers(n // 3)])
            j = int(order[rng.integers(2 * n // 3, n)])
        elif r < 0.8:
            i = int(order[rng.integers(n // 2)])
            j = int(order[rng.integers(n // 2, n)])
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
        if i == j:
            continue
        lo_a = max(-h[i], h[j] - 1.0)
        hi_a = min(1.0 - h[i], h[j])
        if hi_a - lo_a < 1e-12:
            continue
        best_a = 0.0
        best_s = best_c
        alphas = np.linspace(lo_a, hi_a, refine_grid)
        for a in alphas:
            if abs(a) < 1e-14:
                continue
            h[i] += a
            h[j] -= a
            s = score_exact(h)
            h[i] -= a
            h[j] += a
            if s < best_s:
                best_s = s
                best_a = a
        if best_s < best_c - 1e-14:
            h[i] += best_a
            h[j] -= best_a
            best_c = best_s
            if _ % 8 == 0:
                g = smooth_gradient(h, 0.003)
                order = np.argsort(g)
    return h, best_c

def optimize_local(h_init, deadline, rng, T_start=0.08, T_end=0.0015, T_decay=0.82):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c = score_exact(h)
    T = T_start
    step = 0.04
    stall = 0
    pair_counter = 0
    momentum = np.zeros(n)
    beta = 0.7
    while time.time() < deadline:
        g = smooth_gradient(h, T)
        momentum = beta * momentum + (1.0 - beta) * g
        accepted = False
        for alpha in (step, 0.5 * step, 0.25 * step, 0.1 * step, 0.05 * step):
            y = h - alpha * momentum
            h2 = make_feasible(y)
            c2 = score_exact(h2)
            if c2 < best_c - 1e-14:
                h = h2
                best_h = h.copy()
                best_c = c2
                accepted = True
                stall = 0
                step = min(step * 1.08, 0.15)
                break
        if not accepted:
            stall += 1
            step *= 0.7
            if stall > 12:
                T *= T_decay
                step = 0.03
                stall = 0
                momentum = np.zeros(n)
        pair_counter += 1
        if pair_counter % 15 == 0:
            h, c_after = pair_optimize(h, min(deadline, time.time() + 3), rng, n_trials=15, refine_grid=30)
            if c_after < best_c:
                best_c = c_after
                best_h = h.copy()
        if T < T_end:
            h, c_ps = pair_optimize(h, min(deadline, time.time() + 5), rng, n_trials=25, refine_grid=35)
            if c_ps < best_c:
                best_c = c_ps
                best_h = h.copy()
            if time.time() >= deadline:
                break
            if rng.random() < 0.4:
                T = 0.01
                step = 0.015
                stall = 0
                momentum = np.zeros(n)
    if score_exact(h) < best_c:
        best_c = score_exact(h)
        best_h = h.copy()
    return best_h, best_c

def resample_vector(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new)
