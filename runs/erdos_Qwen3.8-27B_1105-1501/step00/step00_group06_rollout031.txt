def make_lp_direction(h_c, radius, rng):

    n = len(h_c)
    dx = 2.0 / n
    
    # Compute A_s and gradients

    G = make_grad_matrix(h_c)
    
    # LP: min t
    # subject to: G @ h - G @ h_c + A_vals <= t
    #             sum(h) = n/2
    #             h_c - radius <= h <= h_c + radius
    #             0 <= h <= 1
    
    # Variables: [h_0, ..., h_{n-1}, t]

    c = np.zeros(n + 1)

    # Inequality constraints: G @ h - G @ h_c + A_vals - t <= 0

    A_ub = np.zeros((2*n-1, n+1))

def make_structured_direction(h, rng, n_modes=None):
    """Create a random feasible direction with low-frequency bias"""
    n = len(h)
    if n_modes is None:
        n_modes = 5
    
    # Random combination of low-frequency sinusoids + random
    d = np.zeros(n)
    for k in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        d += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    
    # Add small random noise
    d += rng.normal(0, 0.1, n)
    
    # Make zero-sum
    d -= np.mean(d)
    
    return d

def make_two_coord_direction(h, rng):
    """Random two-coordinate transfer direction"""
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def project_box_mass(v, target=None):
    n = len(v)

        target = n / 2.0

        return float(np.sum(np.clip(v + t, 0.0, 1.0)))

            lo = mid
        else:

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_mass(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    def sum_clip(t):
        return float(np.sum(np.clip(v + t, 0.0, 1.0)))
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_clip(mid) < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(result))
    for _ in range(100):
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        nfree = int(np.sum(free))
        if nfree == 0 or abs(residual) < 1e-13:
            break
        adj = residual / nfree
        result[free] = np.clip(result[free] + adj, 0.0, 1.0)
        residual = target - float(np.sum(result))
    return result

def make_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    idx_arr = np.arange(n)
    for s in range(-(n - 1), n):
        row = s + (n - 1)
        lo1 = max(0, s)
        hi1 = min(n - 1, n - 1 + s)
        if lo1 <= hi1:
            m = np.arange(lo1, hi1 + 1)
            G[row, m] += (1.0 - h[m - s])
        lo2 = max(0, -s)
        hi2 = min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            m = np.arange(lo2, hi2 + 1)
            G[row, m] -= h[m + s]
    return G

    h = h_bin.copy()
    score = score_h(h)
    ones = np.where(h > 0.5)[0]
