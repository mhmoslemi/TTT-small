import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_feasible(y):
    n = len(y)
    target = float(n) / 2.0
    h = np.clip(np.asarray(y, dtype=np.float64), 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - h
            mask = slack > 1e-12
            if not mask.any():
                break
            add = min(diff, slack[mask].sum())
            h[mask] += (add / slack[mask].sum()) * slack[mask]
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            rem = min(-diff, h[mask].sum())
            h[mask] -= (rem / h[mask].sum()) * h[mask]
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    residual = target - s
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if residual > 0:
                amt = min(residual, 1.0 - h[i])
                h[i] += amt
                residual -= amt
            else:
                amt = min(-residual, h[i])
                h[i] -= amt
                residual += amt
            if abs(residual) < 1e-13:
                break
    return h

def try_upswap(h, i, v, cur_score, threshold):
    delta = v - h[i]
    if delta <= 1e-15:
        return False, cur_score
    if (1.0 - h[i]) * delta > threshold:
        return False, cur_score
    h[i] += delta
    new_s = exact_score(h)
    if new_s < cur_score - 1e-12:
        return True, new_s
    h[i] -= delta
    return False, cur_score

def transfer_to_len(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    xs = (2.0 * np.arange(n_src) + 1.0) / (2.0 * n_src)
    xt = (2.0 * np.arange(n_tgt) + 1.0) / (2.0 * n_tgt)
    h_t = np.interp(xt, xs, h_src, left=h_src[0], right=h_src[-1])
    return repair_feasible(h_t)

def brute_force_small(n, time_limit):
    best_h = None
    best_c5 = 1.0
    half = n // 2
    positions = list(range(n))
    total = 0
    for combo in itertools.combinations(positions, half):
        if time.time() > time_limit:
            break
        x = np.zeros(n, dtype=np.float64)
        x[list(combo)] = 1.0
        c = exact_score(x)
        if c < best_c5:
            best_c5 = c
            best_h = x.copy()
        total += 1
    return best_h, best_c5, total

def milp_solve(n, time_limit):
    try:
        import cvxpy as cp
    except Exception:
        return None, 1.0
    half = n // 2
    x = cp.Variable(n, boolean=True)
    T = cp.Variable()
    z = {}
    constraints = [cp.sum(x) == half, T >= 0]
    for d in range(1, n):
        nvars = n - d
        z_vals = cp.Variable(nvars)
        z[d] = z_vals
        shift = d - 1
        constraints.append(z_vals >= 0)
        constraints.append(z_vals <= x[shift:n])
        constraints.append(z_vals <= x[0:n - d])
        constraints.append(z_vals >= x[0:n - d] + x[shift:n] - 1)
        s_right = cp.sum(x[shift:n])
        s_left = cp.sum(x[0:n - d])
        sz = cp.sum(z_vals)
        constraints.append(s_right - sz <= T)
        constraints.append(s_left - sz <= T)
    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.CBC, time_limit=60)
    except Exception:
        try:
            prob.solve(solver=cp.SCIP, time_limit=60)
        except Exception:
            try:
                prob.solve(solver=cp.GLPK, time_limit=60)
            except Exception:
                return None, 1.0
    try:
        xv = x.value
        if xv is None:
            return None, 1.0
        xv = np.round(np.asarray(xv, dtype=np.float64))
        xv = np.clip(xv, 0.0, 1.0)
        s = int(round(xv.sum()))
