    best_c5 = c5_value(h)

    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        h_sol = slsqp_solve(best_h, n, n_active=n_top, max_iter=200,
                           time_limit=time_limit, start_time=start_time)
        c5_sol = c5_value(h_sol)
        if c5_sol < best_c5:
            best_c5 = c5_sol
            best_h = h_sol.copy()

        # Check if new constraints became active
        c_vals = c_vec(best_h)
        t_val = best_c5
        violated = np.where(c_vals > t_val + 1e-10)[0]
        if len(violated) == 0:
            break
        # Include more active constraints next round
        n_top = min(2 * n + 1, n_top + len(violated) + 5)

    return best_h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(45, 0.5)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    best_c5 = c5_value(h_best)

    target = 0.3808

    # Phase 1: SLSQP from initial point
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        h_sol = slsqp_solve(h_best, n, n_active=25, max_iter=500,
                           time_limit=remaining * 0.8, start_time=start_time)
        c5_sol = c5_value(h_sol)
        if c5_sol < best_c5:
            best_c5 = c5_sol
            h_best = h_sol.copy()

    # Phase 2: Active set iterations from best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        h_sol = active_set_descent(h_best, n, max_rounds=10, n_top=20,
                                   time_limit=remaining * 0.7, start_time=start_time)
        c5_sol = c5_value(h_sol)
        if c5_sol < best_c5:
            best_c5 = c5_sol
            h_best = h_sol.copy()

    # Phase 3: Multiple perturbations + SLSQP
    attempt = 0
    while time.time() - start_time < budget_s * 0.85:
        attempt += 1
        if time.time() - start_time > budget_s * 0.95:
            break

        # Generate perturbation with decreasing sigma
        sigma = max(0.005, 0.04 / np.sqrt(attempt + 1))
        h_pert = h_best + rng.normal(0, sigma, n)
        h_pert = project_feasible(h_pert, n)

        # SLSQP from perturbation
        h_sol = slsqp_solve(h_pert, n, n_active=20, max_iter=200,
                           time_limit=budget_s * 0.9, start_time=start_time)
        c5_sol = c5_value(h_sol)
        if c5_sol < best_c5:
            best_c5 = c5_sol
            h_best = h_sol.copy()
            # Follow up with more aggressive optimization
            if best_c5 < target + 0.001 and time.time() - start_time < budget_s * 0.8:
                h_sol2 = active_set_descent(h_best, n, max_rounds=5, n_top=25,
                                            time_limit=budget_s * 0.85, start_time=start_time)
                c5_sol2 = c5_value(h_sol2)
                if c5_sol2 < best_c5:
                    best_c5 = c5_sol2
                    h_best = h_sol2.copy()

        # Occasionally try smooth descent from a different perturbation
        if attempt % 3 == 0:
            sigma2 = 0.02
            h_pert2 = h_best + rng.normal(0, sigma2, n)
            h_pert2 = project_feasible(h_pert2, n)
            h_smooth = smooth_descent(h_pert2, n, steps=300, tau=0.003,
                                      time_limit=budget_s * 0.9, start_time=start_time)
            c5_smooth = c5_value(h_smooth)
            if c5_smooth < best_c5:
                best_c5 = c5_smooth
                h_best = h_smooth.copy()

    # Phase 4: Final polish with SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5:
        h_sol = slsqp_solve(h_best, n, n_active=30, max_iter=500,
                           time_limit=remaining * 0.9, start_time=start_time)
        c5_sol = c5_value(h_sol)
        if c5_sol < best_c5:
            best_c5 = c5_sol
            h_best = h_sol.copy()

    # Final feasibility check
    h_best = project_feasible(h_best, n)
    best_c5 = c5_value(h_best)

    # Verify mass constraint
    assert abs(np.sum(h_best) - n / 2.0) < 1e-8, f"Mass constraint violated: {np.sum(h_best)}"
    assert np.all(h_best >= -1e-12) and np.all(h_best <= 1 + 1e-12), "Box constraint violated"
    h_best = np.clip(h_best, 0.0, 1.0)
    # Re-project to fix any tiny mass error from clipping
    h_best = project_feasible(h_best, n)
    best_c5 = c5_value(h_best)

    return h_best, best_c5, n
