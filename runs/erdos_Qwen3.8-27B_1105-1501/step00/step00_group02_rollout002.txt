def project_to_feasible(v, target_sum):
    n = len(v)
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(v - t, 0.0, 1.0)

        free = (h > 1e-12) & (h < 1.0 - 1e-12)
