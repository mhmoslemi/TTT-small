            lr = min(lr * 1.1, 2.0)
        else:
            lr *= 0.7
            if lr < 0.01:
                break
    
    return best_h, best_c5, theta

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)
    
    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(parent)
    except Exception:
        n_parent = 64
        parent = np.ones(n_parent) * 0.5
    
    M_parent = n_parent / 2.0
    parent = repair_feasible(parent, M_parent)
    best_h = parent.copy()
    best_c5 = score_exact(parent)
    
    if budget_s <= 5:
        return (best_h, best_c5, n_parent)
    
    # Phase 1: Try resampling to various dimensions
    dim_tries = list(set([n_parent] + [n_parent + d for d in [-4, -2, -1, 1, 2, 4, 8] if n_parent + d >= 16] + [16, 20, 24, 32, 40, 48, 56, 64, 80, 96]))
    dim_tries = [d for d in sorted(dim_tries) if 8 <= d <= 150]
    
    for n_try in dim_tries:
        if time.time() > deadline - 60:
            break
        h_try = resample_to(best_h, n_try)
        c5_try = score_exact(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 2: Local search on best
    if time.time() < deadline - 40:
        n_iter_ls = 8000
        best_h, best_c5 = local_search(best_h, n_iter_ls, rng, min_temp=0.0005, init_temp=0.03, cooling=0.998)
    
    # Phase 3: Parameterized optimization on best dimension
    best_n = len(best_h)
    if time.time() < deadline - 30:
        basis_sizes = [min(6, best_n - 1), min(12, best_n - 1), min(20, best_n - 1), min(32, best_n - 1)]
        basis_sizes = sorted(set(d for d in basis_sizes if d >= 3))
        
        for bdim in basis_sizes:
            if time.time() > deadline - 20:
                break
            B = build_fourier_basis(best_n, bdim)
            if B.shape[1] < 3:
                continue
            
            theta0 = fit_theta(best_h, B)
            
            # Initialize population
            pop_size = 16
            thetas = [theta0.copy()]
            for i in range(1, pop_size):
                sigma = 0.5 if i % 2 == 0 else 1.5
                thetas.append(theta0 + sigma * rng.standard_normal(bdim))
            
            pop_h = [best_h.copy()]
            pop_c5 = [best_c5]
            for i in range(1, pop_size):
                h_c = generate_h(thetas[i], B, best_n)
                c5_c = score_exact(h_c)
                pop_h.append(h_c)
                pop_c5.append(c5_c)
                if c5_c < best_c5 - 1e-12:
                    best_c5 = c5_c
                    best_h = h_c.copy()
            
            # Evolve
            max_gens = 30
            for gen in range(max_gens):
                if time.time() > deadline - 15:
                    break
                
                order = np.argsort(pop_c5)
                
                # Generate offspring
                new_thetas = []
                new_hs = []
                new_c5s = []
                
                for i in range(pop_size):
                    r = rng.random()
                    if r < 0.4:
                        # Mutate good
                        pidx = order[rng.integers(0, max(1, pop_size // 3))]
                        sigma = 0.2 + 0.3 * rng.random()
                        nt = thetas[pidx] + sigma * rng.standard_normal(bdim)
                    elif r < 0.7:
                        # Crossover
                        p1 = order[rng.integers(0, max(1, pop_size // 2))]
                        p2 = order[rng.integers(0, max(1, pop_size // 2))]
                        mask = rng.random(bdim) < 0.5
                        nt = np.where(mask, thetas[p1], thetas[p2])
                        nt = nt + 0.1 * rng.standard_normal(bdim)
                    else:
                        # Exploration
                        nt = theta0 + rng.standard_normal(bdim) * (0.5 + 1.0 * rng.random())
                    
                    hc = generate_h(nt, B, best_n)
                    c5c = score_exact(hc)
                    new_thetas.append(nt)
                    new_hs.append(hc)
                    new_c5s.append(c5c)
                    if c5c < best_c5 - 1e-12:
                        best_c5 = c5c
                        best_h = hc.copy()
                
                # Select: keep top half of old, fill with best of new
                keep = order[:pop_size // 2]
                new_order = np.argsort(new_c5s)
                fill = new_order[:pop_size // 2]
                
                new_pop_thetas = []
                new_pop_hs = []
                new_pop_c5s = []
                for idx in keep:
                    new_pop_thetas.append(thetas[idx])
                    new_pop_hs.append(pop_h[idx])
                    new_pop_c5s.append(pop_c5[idx])
                for idx in fill:
                    new_pop_thetas.append(new_thetas[idx])
                    new_pop_hs.append(new_hs[idx])
                    new_pop_c5s.append(new_c5s[idx])
                
                thetas = new_pop_thetas
                pop_h = new_pop_hs
                pop_c5 = new_pop_c5s
            
            # Gradient polish on best theta
            if time.time() < deadline - 15:
                order = np.argsort(pop_c5)
                best_theta = thetas[order[0]]
                h_polished, c5_polished, _ = param_gradient_descent(best_theta, B, best_n, rng, max_iters=80)
                if c5_polished < best_c5 - 1e-12:
                    best_c5 = c5_polished
                    best_h = h_polished.copy()
    
    # Phase 4: Final local search polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        n_iter = 5000
        best_h, best_c5 = local_search(best_h, n_iter, rng, min_temp=0.0002, init_temp=0.015, cooling=0.998)
    
    # Final verification and repair
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = repair_feasible(best_h, n_final / 2.0)
    best_c5 = score_exact(best_h)
    
    # Ensure sum is correct
    target_sum = n_final / 2.0
    current_sum = np.sum(best_h)
    if abs(current_sum - target_sum) > 1e-10:
        # Redistribute difference
        diff = target_sum - current_sum
        per = diff / n_final
        best_h = np.clip(best_h + per, 0, 1)
        best_h = repair_feasible(best_h, target_sum)
        best_c5 = score_exact(best_h)
    
    return (best_h, best_c5, n_final)
