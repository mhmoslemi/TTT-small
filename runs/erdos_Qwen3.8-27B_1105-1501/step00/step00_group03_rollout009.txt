import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def manual_values(h):
    n = len(h)
    result = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        jdx = k - idx
        result[k] = np.sum(h[idx] * (1.0 - h[jdx]))
    return result

def manual_grad(h):
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        p_start = max(0, k - n + 1)
        p_end = min(n, k + 1)
        p_idx = np.arange(p_start, p_end)
        j_idx = k - p_idx
        J[k, p_idx] = 1.0 - 2.0 * h[j_idx]
    return J

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    # Bisection for lambda
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z + lam, 0.0, 1.0)
    # Fix small sum error
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            # Find coordinates with room to increase
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                step = min(diff, room[idx])
                h[idx] += step
                diff -= step
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for idx in order:
                step = min(-diff, h[idx])
                h[idx] -= step
                diff += step
                if abs(diff) < 1e-14:
                    break
    h = np.clip(h, 0.0, 1.0)
    # Final check
    if not np.all(np.isfinite(h)) or np.sum(h) < target - 1e-10 or np.sum(h) > target + 1e-10:
        h = np.full(n, 0.5)
    return h

def smooth_objective_and_grad(z, beta):
    n = len(z)
    dx = 2.0 / n
    C = manual_values(z) * dx
    C_max = np.max(C)
    shifted = beta * (C - C_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
        obj = C_max + np.log(2 * n - 1) / beta
    else:
        w = exp_vals / total
        obj = C_max + np.log(total) / beta
    # Gradient: dx * sum_k w_k * grad_F_k
    J = manual_grad(z)
    grad = dx * (J.T @ w)
    return obj, grad

def smooth_obj_fn(z, beta):
    return smooth_objective_and_grad(z, beta)[0]

def smooth_grad_fn(z, beta):
    return smooth_objective_and_grad(z, beta)[1]

def direct_minimax_solve(h0, maxiter=120):

    L = 2 * n - 1
    
    def obj(z):
        return z[n]  # theta

        g[n] = 0.0
        return g

        theta = z[n]

        return theta - vals
    
    def con_ineq_jac(z):

        {'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac},
        {'type': 'ineq', 'fun': con_ineq, 'jac': con_ineq_jac}

        h_out = make_feasible(h_out)
        return h_out
    return h0

def interpolate_to_length(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    # Center-based interpolation
    old_x = (np.arange(old_n) + 0.5) * 2.0 / old_n
    new_x = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_interp = np.interp(new_x, old_x, h)
    return make_feasible(h_interp)

    n = len(h)
    h_new = np.zeros(n)
    for i in range(n):
        h_new[i] = 1.0 - h[n - 1 - i]
    return make_feasible(h_new)

    n = len(h)
    result = np.zeros(2 * n - 1)

        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        jdx = k - idx
        result[k] = np.dot(h[idx], 1.0 - h[jdx])
    return result

def manual_grad(h):
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        p_start = max(0, k - n + 1)
        p_end = min(n, k + 1)
        p_idx = np.arange(p_start, p_end)
        j_idx = k - p_idx
        J[k, p_idx] = 1.0 - 2.0 * h[j_idx]
    return J

    target = n / 2.0

            lo = mid
        else:

    lam = (lo + hi) / 2.0
    h = np.clip(z + lam, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                if diff <= 1e-14:
                    break
                step = min(diff, room[idx])

                    diff -= step

        obj = float(C_max + np.log(total) / beta)
    J = manual_grad(z)
    grad = dx * (J.T @ w)
