            if residual > 0:
                delta = min(residual, 1.0 - h[idx[i]])
                h[idx[i]] += delta
                residual -= delta
            else:
                delta = min(-residual, h[idx[i]])
                h[idx[i]] -= delta
                residual += delta
            if abs(residual) < 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    return h


def official_score(h):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def manual_values(h):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    out = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo < hi:
            out[r] = np.dot(h[lo+k:hi+k], 1.0 - h[lo:hi])
    return out


def manual_grad(h):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo < hi:
            js = np.arange(lo, hi)
            grad[r, js + k] += (1.0 - h[js])
            grad[r, js] += -h[js + k]
    return grad


def smooth_obj_grad(h, beta):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    dx = 2.0 / n
    vals = manual_values(h) * dx
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    if s == 0:
        s = 1.0
        exp_s = np.full_like(exp_s, 1.0 / len(exp_s))
    obj = cmax + np.log(s) / beta
    w = exp_s / s
    g = manual_grad(h)
    grad = dx * (w @ g)
    return obj, grad


def smooth_fun(h, beta):
    return smooth_obj_grad(h, beta)[0]


def smooth_jac(h, beta):
    return smooth_obj_grad(h, beta)[1]


def smooth_eq_fun(h):
    return h.sum() - len(h) / 2.0


def smooth_eq_jac(h):
    return np.ones(len(h))


def mm_obj(z):
    return z[-1]


def mm_obj_jac(z):
    j = np.zeros(len(z))
    j[-1] = 1.0
    return j


def mm_ineq_fun(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    theta = z[n]
    vals = manual_values(h) * dx
    return theta - vals


def mm_ineq_jac(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    g = manual_grad(h)
    jac = np.zeros((2 * n - 1, n + 1))
    jac[:, :n] = -dx * g
    jac[:, n] = 1.0
    return jac


def mm_eq_fun(z):
    n = len(z) - 1
    return z[:n].sum() - n / 2.0


def mm_eq_jac(z):
    n = len(z) - 1
    j = np.zeros(len(z))
    j[:n] = 1.0
    return j


def interpolate_to_length(h, new_n):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    if n == new_n:
        return h.copy()
    src_x = (np.arange(n) + 0.5) * 2.0 / n
    dst_x = (np.arange(new_n) + 0.5) * 2.0 / new_n
    new_h = np.interp(dst_x, src_x, h)
    return new_h


def antisymmetric_transform(h):
    h = np.asarray(h, dtype=float).ravel()
    return 1.0 - h[::-1]


def is_feasible(h, tol=1e-8):
    if not np.all(np.isfinite(h)):
        return False
    n = len(h)
    if n == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(h.sum() - n / 2.0) > tol:
        return False
    return True


def update_best(h, best_info):
    if not is_feasible(h):
        h = make_feasible(h)
        if not is_feasible(h):
            return
    score = official_score(h)
    if not np.isfinite(score):
        return
    if best_info is None or score < best_info['score'] - 1e-14:
        best_info['score'] = score
        best_info['h'] = h.copy()
    elif abs(score - best_info['score']) <= 1e-14 and len(h) < len(best_info['h']):
        best_info['score'] = score
        best_info['h'] = h.copy()


def run_smooth(h0, beta, maxiter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac}]
    try:
        res = minimize(
            smooth_fun, h0.copy(),
            args=(beta,),
            jac=smooth_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return res.x
    except Exception:
        return h0.copy()


def run_direct(h0, theta0, maxiter=150):
    n = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = make_feasible(h0)
    z0 = np.append(h0, theta0)
    bounds = [(0.0, 1.0)] * n + [(1e-6, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': mm_ineq_fun, 'jac': mm_ineq_jac},
        {'type': 'eq', 'fun': mm_eq_fun, 'jac': mm_eq_jac}
    ]
    try:
        res = minimize(
            mm_obj, z0,
            jac=mm_obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        h_out = res.x[:n]
        return h_out, res.x[-1]
    except Exception:
        return h0.copy(), theta0


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - max(10.0, 0.05 * budget_s)
    rng = np.random.default_rng(seed)

    # Initialize best
    best_info = None

    # Get parent
    parent = None
    parent_n = 45
    try:
        pv = np.asarray(initial_h_values, dtype=float).ravel()
        if pv.size > 0 and np.all(np.isfinite(pv)):
            parent = pv
            parent_n = len(parent)
    except NameError:
        pass
    except Exception:
        pass

    # If no valid parent, create a default
    if parent is None:
        parent = np.full(parent_n, 0.5)
        parent = make_feasible(parent)

    parent = make_feasible(parent)
    update_best(parent, best_info)

    if best_info is None:
        best_info = {'score': official_score(parent), 'h': parent.copy()}

    # Lengths to try: parent length first, then some variations
    lengths = [parent_n]
    for dn in [-10, -5, 5, 10, 15, 20]:
        ln = parent_n + dn
        if 20 <= ln <= 80 and ln not in lengths:
            lengths.append(ln)
    lengths.sort()

    # Beta schedule for smoothing
    betas = [20, 80, 300, 1000]

    # Process each length
    for n_target in lengths:
        if time.time() > deadline:
            break

        # Build starting candidates for this length
        candidates = []

        # 1. Interpolate best to this length
        h_interp = interpolate_to_length(best_info['h'], n_target)
        h_interp = make_feasible(h_interp)
        candidates.append(h_interp)

        # 2. Antisymmetric transform of interpolated
        h_asym = antisymmetric_transform(h_interp)
        h_asym = make_feasible(h_asym)
        candidates.append(h_asym)

        # 3. Interpolate parent
        if n_target != parent_n:
            h_p = interpolate_to_length(parent, n_target)
            h_p = make_feasible(h_p)
            candidates.append(h_p)

        # 4. Random perturbations of best interpolated
        n_perturb = 6
        for sigma in [0.02, 0.05, 0.1]:
            if time.time() > deadline:
                break
            h_pert = h_interp + rng.normal(0, sigma, n_target)
            h_pert = make_feasible(h_pert)
            candidates.append(h_pert)

        # 5. A few fully random starts
        for _ in range(3):
            h_rnd = rng.uniform(0, 1, n_target)
            h_rnd = make_feasible(h_rnd)
            candidates.append(h_rnd)

        # Sort candidates by score (best first)
        scored = []
        seen = set()
        for c in candidates:
            c = np.clip(c, 0, 1)
            key = np.round(c, 6).tobytes()
            if key in seen:
                continue
            seen.add(key)
            s = official_score(c)
            if np.isfinite(s):
                scored.append((s, c))
        scored.sort(key=lambda x: x[0])

        # Limit candidates
        max_cands = 8
        scored = scored[:max_cands]

        # For each candidate, try smooth optimization then direct
        for cand_score, h_cand in scored:
            if time.time() > deadline:
                break

            # Update best with this candidate
            update_best(h_cand, best_info)

            # Try smooth optimization with increasing beta
            cur_h = h_cand.copy()
            for beta in betas:
                if time.time() > deadline:
                    break
                maxiter = 80
                new_h = run_smooth(cur_h, beta, maxiter=maxiter)
                new_h = make_feasible(new_h)
                new_score = official_score(new_h)
                if np.isfinite(new_score) and new_score < official_score(cur_h) - 1e-14:
                    cur_h = new_h
                update_best(cur_h, best_info)

            # Try direct minimax from current best
            if time.time() > deadline:
                break
            cur_score = official_score(cur_h)
            theta0 = cur_score + 1e-10
            h_dm, theta_dm = run_direct(cur_h, theta0, maxiter=120)
            h_dm = make_feasible(h_dm)
            dm_score = official_score(h_dm)
            if np.isfinite(dm_score):
                update_best(h_dm, best_info)

            # Try direct minimax with slightly perturbed start
            if time.time() > deadline:
                break
            h_pert2 = cur_h + rng.normal(0, 0.03, n_target)
            h_pert2 = make_feasible(h_pert2)
            s2 = official_score(h_pert2)
            theta0_2 = s2 + 1e-10
            h_dm2, _ = run_direct(h_pert2, theta0_2, maxiter=100)
            h_dm2 = make_feasible(h_dm2)
            dm_score2 = official_score(h_dm2)
            if np.isfinite(dm_score2):
                update_best(h_dm2, best_info)

    # Final polish: direct minimax from the overall best at its native length
    if time.time() < deadline:
        h_best = best_info['h']
        n_best = len(h_best)
        cur_score = official_score(h_best)
        theta0 = cur_score + 1e-10
        h_final, _ = run_direct(h_best, theta0, maxiter=200)
        h_final = make_feasible(h_final)
        final_score = official_score(h_final)
        if np.isfinite(final_score):
            update_best(h_final, best_info)

        # One more round with slight perturbation
        if time.time() < deadline:
            h_p = best_info['h'] + rng.normal(0, 0.01, len(best_info['h']))
            h_p = make_feasible(h_p)
            sp = official_score(h_p)
            h_f2, _ = run_direct(h_p, sp + 1e-10, maxiter=150)
            h_f2 = make_feasible(h_f2)
            sf2 = official_score(h_f2)
            if np.isfinite(sf2):
                update_best(h_f2, best_info)

    # Final check and return
    h_out = best_info['h']
    h_out = np.clip(h_out, 0.0, 1.0)
    h_out = make_feasible(h_out)
    c5 = official_score(h_out)
    if not np.isfinite(c5):
        h_out = make_feasible(parent)
        c5 = official_score(h_out)

    # Add tiny safety margin
    c5_bound = float(c5 + 1e-12)
    n_pts = len(h_out)

    return (h_out, c5_bound, n_pts)
