    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')

    max_corr = np.max(corr)
    smooth = max_corr + eps * np.log(np.sum(np.exp((corr - max_corr) / eps)))
    return smooth * dx

def smooth_max_c5_grad(h, eps=0.01):

    n = len(h)

    max_corr = np.max(corr)
    weights = np.exp((corr - max_corr) / eps)

    best_c5 = compute_c5(h)
    
    # Use a Polyak-like step or 1/sqrt(t)
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        c5_val = corr[max_idx] * dx
        
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        
        m = max_idx - (n - 1)
        grad = grad_for_shift(h, m)
        
        # Step size
        step = 0.3 / (1.0 + it * 0.01)
        
        h = h - step * grad
        h = project_onto_constraint(h, target_sum)
    
    return best_h, best_c5

def local_search(h, best_c5, time_limit, start_time, rng):
    """Pairwise swap local search."""
    n = len(h)
    target_sum = n / 2.0
    current_c5 = best_c5
    current_h = h.copy()
    
    # Precompute correlations
    corr_prev = np.correlate(current_h, 1 - current_h, mode='full')
    max_c = np.max(corr_prev)
    
    no_improve = 0
    max_no_improve = 500
    
    while no_improve < max_no_improve and time.time() < time_limit:
        improved = False
        
        # Get current correlations
        corr_prev = np.correlate(current_h, 1 - current_h, mode='full')
        
        # Try random pairs
        for _ in range(200):
            if time.time() > time_limit:
                break
            
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            
            # Try small adjustments
            for delta in [0.1, 0.05, 0.02, 0.01]:
                # Direction 1: decrease h[i], increase h[j]
                if current_h[i] >= delta and current_h[j] <= 1.0 - delta:
                    # Compute change in correlations efficiently
                    # Only correlations involving i or j change
                    # For shift m: g_m changes by:
                    # -delta * (1-h[j+m]) + delta * h[i+m] (for the h[i] part)
                    # ... this gets complex. Let's just recompute.
                    new_h = current_h.copy()
                    new_h[i] -= delta
                    new_h[j] += delta
                    new_c5 = compute_c5(new_h)
                    if new_c5 < current_c5 - 1e-12:
                        current_h = new_h
                        current_c5 = new_c5
                        improved = True
                        no_improve = 0
                        break
                # Direction 2: decrease h[j], increase h[i]
                if current_h[j] >= delta and current_h[i] <= 1.0 - delta:
                    new_h = current_h.copy()
                    new_h[j] -= delta
                    new_h[i] += delta
                    new_c5 = compute_c5(new_h)
                    if new_c5 < current_c5 - 1e-12:
                        current_h = new_h
                        current_c5 = new_c5
                        improved = True
                        no_improve = 0
                        break
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
    
    return current_h, current_c5

Hmm, the local search with random pairs and full correlation recomputation is O(n) per evaluation (for the correlation which is O(n²) for full mode... actually np.correlate in full mode is O(n²) for direct computation).

For n=63, np.correlate is about 63*125 ≈ 8000 operations. With 200 random pairs * 4 deltas = 800 evaluations per "round", that's about 6.4M operations per round. At ~100M ops/sec in Python (optimistic), that's about 0.06s per round. We can afford many rounds within the budget.

But actually, np.correlate might use FFT under the hood for large arrays, but for n=63 it's probably direct. Let me not worry about it.

Let me think about a better local search. Instead of random pairs, I should focus on the pairs that affect the worst shift.

**Improved local search:**
1. Find the shift m* that achieves the maximum
2. For pairs (i, i+m*) that contribute to g_{m*}, try to reduce the product h[i]*(1-h[i+m*])
3. This means: if h[i] is high and h[i+m*] is low, try to decrease h[i] or increase h[i+m*]
4. The constraint sum(h)=n/2 means we need to compensate elsewhere

Let me implement this:
