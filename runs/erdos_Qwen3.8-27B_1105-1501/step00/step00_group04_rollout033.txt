```python
import numpy as np
import time
from scipy.signal import correlate

def evaluate(x):
    """Compute C5 for vector x (already feasible)."""
    n = len(x)
    h = np.asarray(x, dtype=float)
    corr = correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def score_binary(x):
    """Score for binary vector, returns (c5, max_count)."""
    n = len(x)
    h = x.astype(np.float64)
    corr = correlate(h, 1.0 - h, mode='full')
    max_count = int(np.max(corr))
    return max_count * (2.0 / n), max_count

def binarize(h, n_ones=None):
    """Convert h to binary, keeping top n_ones as 1s."""
    n = len(h)
    if n_ones is None:
        n_ones = n // 2
    sorted_idx = np.argsort(-h)
    x = np.zeros(n, dtype=np.int8)
    x[sorted_idx[:n_ones]] = 1
    return x

def repair_feasible(h):
    """Project h to [0,1]^n with sum = n/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h.copy()
    # Adjust
    if s > target:
        excess = s - target
        # Reduce from largest values
        order = np.argsort(-h)
        remaining = excess
        for i in order:
            if remaining <= 1e-15:
                break
            reduce = min(h[i], remaining)
            h[i] -= reduce
            remaining -= reduce
    else:
        deficit = target - s
        order = np.argsort(h)
        remaining = deficit
        for i in order:
            if remaining <= 1e-15:
                break
            add = min(1.0 - h[i], remaining)
            h[i] += add
            remaining -= add
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute to a single element
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def local_search_binary(x, max_time=10.0, seed=42):
    """Greedy local search on binary vector using single swaps."""
    n = len(x)
    x = x.copy()
    _, max_count = score_binary(x)
    best_count = max_count
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    no_improve = 0
    while time.time() - start_time < max_time:
        improved = False
        # Get all (1, 0) pairs
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        
        # Shuffle for diversity
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones:
            if time.time() - start_time > max_time:
                break
            for j in zeros:
                # Try swap
                x[i] = 0
                x[j] = 1
                _, new_count = score_binary(x)
                if new_count < max_count:
                    max_count = new_count
                    best_count = min(best_count, max_count)
                    improved = True
                    no_improve = 0
                    # Update ones and zeros
                    # i is now 0, j is now 1
                    # Reload
                    ones = np.where(x == 1)[0]
                    zeros = np.where(x == 0)[0]
                    break
                else:
                    x[i] = 1
                    x[j] = 0
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
    
    return x, best_count
```
