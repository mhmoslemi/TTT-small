    t_hi = min(h[p], 1.0 - h[q])

    best_val = obj(h)

    hq = h[q]

        h_copy[q] = hq + t

        if v < best_val - 1e-14:

            best_t = t

        hi = min(best_t + span, t_hi)

                best_t = t

    h_new[q] = hq + best_t

    h_new[p] = min(max(h_new[p], 0.0), 1.0)
    h_new[q] = min(max(h_new[q], 0.0), 1.0)
    return h_new, best_val

def gradient(h):
    """Compute subgradient of the max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    # Active lags: within 5% of range from max
    spread = cmax - np.min(corr)
    if spread < 1e-12:
        spread = 1e-12
    threshold = cmax - 0.1 * spread
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        # F_d = dx * sum_{n: valid} h[n+d] * (1 - h[n])
        # valid: 0 <= n < n and 0 <= n+d < n
        if d >= 0:
            n_lo, n_hi = 0, n - d
        else:
            n_lo, n_hi = -d, n
        if n_lo >= n_hi:
            continue
        idx = np.arange(n_lo, n_hi)
        # dF/dh[m] for m = idx + d: += dx * (1 - h[idx])
        grad[idx + d] += dx * (1.0 - h[idx])
        # dF/dh[m] for m = idx: -= dx * h[idx + d]
        grad[idx] -= dx * h[idx + d]
    
    return grad

        # We want to try pairs where this is most negative (best improvement)
        # Actually: dC/dt = -dC/dh[p] + dC/dh[q]
        # If negative, increasing t (h_p decreases, h_q increases) helps
        # So we want -g[p] + g[q] < 0, i.e., g[p] > g[q]
        # Try pairs with large g[p] - g[q]
        
        # Create list of pairs sorted by expected improvement
        # For efficiency, just sample pairs
        n_pairs = min(n * (n - 1) // 2, 200)
        # Sample pairs weighted by gradient difference
        indices = np.arange(n)
        
        # Sort by gradient: high gradient indices are "sources" (decrease h)
        # Low gradient indices are "sinks" (increase h)
        order = np.argsort(g)
        # Sources: high gradient (end of sorted array)
        # Sinks: low gradient (beginning of sorted array)
        sources = order[::-1][:n//2]

            if no_improve_count >= max_no_improve:

import numpy as np
import time

def obj(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, tol=1e-9):
    n = len(h)
    if n < 2:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(h.sum() - 0.5 * n) > tol * max(1.0, n):
        return False
    return True

def project(z, T):
    n = len(z)
    if n == 0:
        return z.copy()
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = T - y.sum()
    if res > 1e-14:
        order = np.argsort(1.0 - y)
        for i in order:
            d = min(res, 1.0 - y[i])
            y[i] += d
            res -= d
            if res < 1e-14:
                break
    elif res < -1e-14:
        order = np.argsort(-y)
        for i in order:
            d = min(-res, y[i])
            y[i] -= d
            res += d
            if -res < 1e-14:
                break
    return y

    hq = h[q]
    t_lo = max(hp - 1.0, -hq)
    t_hi = min(hp, 1.0 - hq)

            best_t = t

def gradient_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    cmin = np.min(corr)
    spread = max(cmax - cmin, 1e-12)
    threshold = cmax - 0.15 * spread
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for r in active:
        d = int(r) - (n - 1)
        if d >= 0:
            n_lo, n_hi = 0, n - d
        else:
            n_lo, n_hi = -d, n
        if n_lo >= n_hi:
            continue
        idx = np.arange(n_lo, n_hi)
        grad[idx + d] += dx * (1.0 - h[idx])
        grad[idx] -= dx * h[idx + d]
    return grad

def pair_descent(h, tlimit, rng, ng=32):

                    return h, C
                i, j = int(indices[ii]), int(indices[jj])
                if h[i] <= 1e-10 and h[j] >= 1.0 - 1e-10:
