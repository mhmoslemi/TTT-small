import math

def deterministic_feasible(n):
    y = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        y[:k] = 1.0
    if n % 2 == 1:
        y[k] = 0.5
    return y


def project_box_mass(v, n):
    target = n / 2.0
    y = np.asarray(v, dtype=float).copy()
    bad = ~np.isfinite(y)
    if bad.any():
        y[bad] = 0.5
    y = np.clip(y, 0.0, 1.0)

    s = float(np.sum(y))
    if abs(s - target) <= 1e-13:
        return y

    lo = -float(np.max(y)) - 1.0
    hi = 1.0 - float(np.min(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        f = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if f < target:
            lo = mid
        else:
            hi = mid

    mu = 0.5 * (lo + hi)
    y = np.clip(y + mu, 0.0, 1.0)

    for _ in range(3):
        res = target - float(np.sum(y))
        if abs(res) <= 1e-13:
            break
        if res > 0:
            free = y < 1.0 - 1e-12
        else:
            free = y > 1e-12
        if free.any():
            y[free] += res / float(np.count_nonzero(free))
            y = np.clip(y, 0.0, 1.0)

    res = target - float(np.sum(y))
    if abs(res) > 1e-9:
        y = deterministic_feasible(n)

    res = target - float(np.sum(y))
    if abs(res) > 1e-14:
        if res > 0:
            idx = int(np.argmax(1.0 - y))
            add = min(res, 1.0 - y[idx])
            y[idx] += add
        else:
            idx = int(np.argmax(y))
            sub = min(-res, y[idx])
            y[idx] -= sub

    if (
        float(np.min(y)) < 0.0
        or float(np.max(y)) > 1.0
        or abs(target - float(np.sum(y))) > 1e-8
    ):
        y = project_box_mass(y, n)

    return y


def fix_mass(h, n, target):
    h = np.asarray(h, dtype=float)
    np.clip(h, 0.0, 1.0, out=h)

    res = target - float(np.sum(h))
    if abs(res) <= 1e-13:
        return h

    if res > 0:
        free = h < 1.0 - 1e-12
    else:
        free = h > 1e-12

    if free.any():
        h[free] += res / float(np.count_nonzero(free))
        np.clip(h, 0.0, 1.0, out=h)

    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        if res > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + res)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + res)

    if (
        float(np.min(h)) < 0.0
        or float(np.max(h)) > 1.0
        or abs(target - float(np.sum(h))) > 1e-9
    ):

    return h


def raw_defined_simple(h):
    n = len(h)
    m = 2 * n - 1
    v = 1.0 - h
    r = np.empty(m, dtype=float)
    for idx, k in enumerate(range(-n + 1, n)):
        if k >= 0:
            a = 0
            b = n - 1 - k
        else:
            a = -k
            b = n - 1
        if a <= b:
            r[idx] = float(np.dot(h[a + k:b + k + 1], v[a:b + 1]))
        else:
            r[idx] = 0.0

def corr_flag(n):

    rdef = raw_defined_simple(h)

    if np.allclose(rnp, rdef[::-1], rtol=1e-12, atol=1e-12):

    return None

    if flag is True:
        r = r[::-1].copy()
    elif flag is None:
        return raw_defined_simple(h)
    return r


def c5_value(h, n):
    return (2.0 / n) * float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def make_structures(n):
    m = 2 * n - 1
    shifts = np.arange(-(n - 1), n, dtype=np.int64)
    start = np.maximum(0, -shifts).astype(np.int64)
    end = np.minimum(n - 1, n - 1 - shifts).astype(np.int64)

    fv = np.zeros((n, m), dtype=np.float64)
    fi = np.zeros((n, m), dtype=np.int64)
    sv = np.zeros((n, m), dtype=np.float64)
    si = np.zeros((n, m), dtype=np.int64)

    for p in range(n):
        sec_mask = (start <= p) & (p <= end)
        if np.any(sec_mask):
            sv[p, sec_mask] = 1.0
            si[p, sec_mask] = p + shifts[sec_mask]

        first_mask = (start <= p - shifts) & (p - shifts <= end)
        if np.any(first_mask):
            fv[p, first_mask] = 1.0
            fi[p, first_mask] = p - shifts[first_mask]

    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))

    flag = corr_flag(n)
    return (n, m, shifts, fv, fi, sv, si, pairs, flag)


def compute_deriv(h, structs):
    n, m, shifts, fv, fi, sv, si, pairs, flag = structs
    D = np.empty((n, m), dtype=np.float64)
    for p in range(n):
        D[p] = fv[p] * (1.0 - h[fi[p]]) - sv[p] * h[si[p]]
    return D


def pair_feasible_bool(v, lo, hi, pos_a, pos_r, neg_a, neg_r,
                       zero_max, plus_a, plus_r, a0, r00, tol=1e-10):
    L = lo
    U = hi

    if zero_max > v + tol:
        return False

    if pos_a:
        um = U
        for i in range(len(pos_a)):
            b = (v - pos_r[i]) / pos_a[i]
            if b < um:
                um = b
        U = um
        if L > U + tol:
            return False

    if neg_a:
        lm = L
        for i in range(len(neg_a)):
            b = (v - neg_r[i]) / neg_a[i]
            if b > lm:
                lm = b
        L = lm
        if L > U + tol:
            return False

    for i in range(len(plus_a)):
        a = plus_a[i]
        rr = plus_r[i]
        disc = a * a + 4.0 * (v - rr)
        if disc < -tol:
            return False
        if disc < 0.0:
            disc = 0.0
        sq = math.sqrt(disc)
        l = 0.5 * (-a - sq)
        u = 0.5 * (-a + sq)
        if l > L:
            L = l
        if u < U:
            U = u
        if L > U + tol:
            return False

    disc = a0 * a0 - 8.0 * (v - r00)
    if disc > tol:
        sq = math.sqrt(disc)
        r1 = 0.25 * (a0 - sq)
        r2 = 0.25 * (a0 + sq)
        if L > r1 + tol and U < r2 - tol:
            return False

    return True


def pair_feasible_intervals(v, lo, hi, pos_a, pos_r, neg_a, neg_r,
                            zero_max, plus_a, plus_r, a0, r00, tol=1e-9):
    L = lo
    U = hi

    if zero_max > v + tol:
        return None

    if pos_a:
        um = U
        for i in range(len(pos_a)):
            b = (v - pos_r[i]) / pos_a[i]
            if b < um:
                um = b
        U = um
        if L > U + tol:
            return None

    if neg_a:
        lm = L
        for i in range(len(neg_a)):
            b = (v - neg_r[i]) / neg_a[i]
            if b > lm:
                lm = b
        L = lm
        if L > U + tol:
            return None

    for i in range(len(plus_a)):
        a = plus_a[i]
        rr = plus_r[i]
        disc = a * a + 4.0 * (v - rr)
        if disc < -tol:
            return None
        if disc < 0.0:
            disc = 0.0
        sq = math.sqrt(disc)
        l = 0.5 * (-a - sq)
        u = 0.5 * (-a + sq)
        if l > L:
            L = l
        if u < U:
            U = u
        if L > U + tol:
            return None

    intervals = []
    disc = a0 * a0 - 8.0 * (v - r00)
    if disc > tol:
        sq = math.sqrt(disc)
        r1 = 0.25 * (a0 - sq)
        r2 = 0.25 * (a0 + sq)

        if L <= r1 + tol:
            u = r1 if r1 < U else U
            if L <= u + tol:
                intervals.append([L, u])

        if U >= r2 - tol:
            l = r2 if r2 > L else L
            if l <= U + tol:
                intervals.append([l, U])
    else:
        intervals.append([L, U])

    merged = []
    for l, u in intervals:
        if l < lo:
            l = lo
        if u > hi:
            u = hi
        if l > u + tol:
            continue
        if merged and l <= merged[-1][1] + tol:
            if u > merged[-1][1]:
                merged[-1][1] = u
        else:
            merged.append([l, u])

    return merged if merged else None


def pair_evaluate_raw(r0, alpha, lin_idx, zero_idx, plus_indices, lam):
    if lin_idx.size:
        val = float(np.max(r0[lin_idx] + alpha[lin_idx] * lam))
    else:
        val = -1e100

    v = r0[zero_idx] + alpha[zero_idx] * lam - 2.0 * lam * lam
    if v > val:
        val = v

    for idx in plus_indices:
        v = r0[idx] + alpha[idx] * lam + lam * lam
        if v > val:
            val = v

    return val


def try_pair_move(h, n, r0, D, structs, p, q, current_raw, min_raw_tol=1e-9):
    n_s, m, shifts, fv, fi, sv, si, pairs, flag = structs
    zero_idx = n - 1

    lo = max(h[q] - 1.0, -h[p])
    hi = min(1.0 - h[p], h[q])
    if hi - lo < 1e-12:
        return None

    lo = lo + 1e-12
    hi = hi - 1e-12
    if hi - lo < 1e-12:
        return None

    alpha = D[p] - D[q]

    plus_indices = []
    for idx in (zero_idx + p - q, zero_idx + q - p):
        if 0 <= idx < m and idx != zero_idx:
            plus_indices.append(idx)
    plus_indices = tuple(plus_indices)

    lin_mask = np.ones(m, dtype=bool)
    lin_mask[zero_idx] = False
    for idx in plus_indices:
        lin_mask[idx] = False

    pos = lin_mask & (alpha > 1e-12)
    neg = lin_mask & (alpha < -1e-12)

    pos_a = alpha[pos].tolist()
    pos_r = r0[pos].tolist()
    neg_a = alpha[neg].tolist()
    neg_r = r0[neg].tolist()

    lin_idx = np.where(lin_mask)[0]
    zero_lin = lin_mask & ~(pos | neg)
    if lin_idx.size and np.any(zero_lin):
        zero_max = float(np.max(r0[zero_lin]))
    else:
        zero_max = -1e100

    a0 = float(alpha[zero_idx])
    r00 = float(r0[zero_idx])
    plus_a = [float(alpha[idx]) for idx in plus_indices]
    plus_r = [float(r0[idx]) for idx in plus_indices]

    high = current_raw + 1e-9
    if not pair_feasible_bool(high, lo, hi, pos_a, pos_r, neg_a, neg_r,
                              zero_max, plus_a, plus_r, a0, r00, 1e-10):
        high = current_raw + 1e-6
        if not pair_feasible_bool(high, lo, hi, pos_a, pos_r, neg_a, neg_r,
                                  zero_max, plus_a, plus_r, a0, r00, 1e-8):
            high = current_raw + 1e-3
            if not pair_feasible_bool(high, lo, hi, pos_a, pos_r, neg_a, neg_r,
                                      zero_max, plus_a, plus_r, a0, r00, 1e-7):
                return None

    low = 0.0
    for _ in range(24):
        if high - low < 1e-10:
            break
        mid = 0.5 * (low + high)
        if pair_feasible_bool(mid, lo, hi, pos_a, pos_r, neg_a, neg_r,
                              zero_max, plus_a, plus_r, a0, r00, 1e-10):
            high = mid
        else:
            low = mid

    if high >= current_raw - min_raw_tol:
        return None

    intervals = pair_feasible_intervals(high + 1e-8, lo, hi, pos_a, pos_r,
                                        neg_a, neg_r, zero_max, plus_a, plus_r,
                                        a0, r00, 1e-8)
    if intervals is None:
        intervals = pair_feasible_intervals(high, lo, hi, pos_a, pos_r,
                                            neg_a, neg_r, zero_max, plus_a, plus_r,
                                            a0, r00, 1e-8)

    cand = []
    if intervals is not None:
        for l, u in intervals:
            if u - l < 1e-11:
                cand.append(0.5 * (l + u))
            else:
                cand.extend(np.linspace(l, u, 17).tolist())
    cand.extend([lo, hi])

    best_lam = 0.0
    best_val = current_raw
    for lam in cand:
        if lam < lo or lam > hi:
            continue
        val = pair_evaluate_raw(r0, alpha, lin_idx, zero_idx, plus_indices, float(lam))
        if val < best_val:
            best_val = val
            best_lam = float(lam)

    span = hi - lo
    wl = max(lo, best_lam - 0.2 * span)
    wh = min(hi, best_lam + 0.2 * span)
    if wh - wl > 1e-13:
        for lam in np.linspace(wl, wh, 21):
            lam = float(lam)
            val = pair_evaluate_raw(r0, alpha, lin_idx, zero_idx, plus_indices, lam)
            if val < best_val:
                best_val = val
                best_lam = lam

    if best_val >= current_raw - min_raw_tol:
        for lam in np.linspace(lo, hi, 51):
            lam = float(lam)
            val = pair_evaluate_raw(r0, alpha, lin_idx, zero_idx, plus_indices, lam)
            if val < best_val:
                best_val = val
                best_lam = lam

    if best_val >= current_raw - min_raw_tol:
        return None

    h2 = h.copy()
    lam = min(max(best_lam, lo), hi)
    h2[p] = min(1.0, max(0.0, h2[p] + lam))
    h2[q] = min(1.0, max(0.0, h2[q] - lam))
    h2 = fix_mass(h2, n, n / 2.0)

    r2 = raw_corr(h2, flag)
    val2 = float(np.max(r2))
    if val2 >= current_raw - min_raw_tol:
        return None

    D2 = compute_deriv(h2, structs)
    return (h2, r2, D2, val2)


def pair_descent_pass(h, n, r0, D, structs, rng, deadline, min_raw_tol=1e-9):
    pairs = structs[7]
    order = list(pairs)
    rng.shuffle(order)

    current_raw = float(np.max(r0))
    improved = False
    count = 0

    for p, q in order:
        count += 1
        if (count & 511) == 0 and time.time() >= deadline:
            break
        if current_raw <= 1e-14:
            break

        res = try_pair_move(h, n, r0, D, structs, p, q, current_raw, min_raw_tol)
        if res is not None:
            h, r0, D, current_raw = res
            improved = True

    return h, r0, D, improved


def dense_evaluate(h, d, lam):
    x = h + lam * d
    np.clip(x, 0.0, 1.0, out=x)
    return float(np.max(np.correlate(x, 1.0 - x, mode="full")))


def dense_line_search(h, n, d, current_raw, deadline, min_tol=1e-8):
    lo = 0.0
    hi = 0.0
    eps = 1e-12

    for i in range(n):
        di = float(d[i])
        hv = float(h[i])
        if di > eps:
            l1 = -hv / di
            u1 = (1.0 - hv) / di
        elif di < -eps:
            l1 = (1.0 - hv) / di
            u1 = -hv / di
        else:
            continue
        if l1 > lo:
            lo = l1
        if u1 < hi:
            hi = u1

    if lo < -1.0:
        lo = -1.0
    if hi > 1.0:
        hi = 1.0
    if lo > 0.0:
        lo = 0.0
    if hi < 0.0:
        hi = 0.0

    if hi - lo < 1e-11:
        return None

    lo_e = lo + 1e-12
    hi_e = hi - 1e-12
    if hi_e - lo_e < 1e-12:
        lo_e, hi_e = lo, hi
        if hi_e - lo_e < 1e-11:
            return None

    best_lam = 0.0
    best_val = current_raw

    for lam in np.linspace(lo_e, hi_e, 33):
        if time.time() >= deadline:
            break
        val = dense_evaluate(h, d, float(lam))
        if val < best_val:
            best_val = val
            best_lam = float(lam)

    span = hi_e - lo_e
    for _ in range(2):
        if time.time() >= deadline:
            break
        wl = max(lo_e, best_lam - 0.333 * span)
        wh = min(hi_e, best_lam + 0.333 * span)
        if wh - wl < 1e-12:
            break
        span = wh - wl
        for lam in np.linspace(wl, wh, 21):
            val = dense_evaluate(h, d, float(lam))
            if val < best_val:
                best_val = val
                best_lam = float(lam)

    if best_val < current_raw - min_tol:
        return best_lam, best_val
    return None


def try_dense_move(h, n, d, current_raw, deadline, min_tol=1e-8):
    res = dense_line_search(h, n, d, current_raw, deadline, min_tol)
    if res is None:
        return None
    lam, _ = res
    h2 = h + lam * d
    np.clip(h2, 0.0, 1.0, out=h2)
    h2 = fix_mass(h2, n, n / 2.0)
    r2 = np.correlate(h2, 1.0 - h2, mode="full")
    val2 = float(np.max(r2))
    if val2 >= current_raw - min_tol:
        return None
    return h2, r2, val2


def active_direction(h, r0, D, beta, rng):
    maxr = float(np.max(r0))
    m = len(r0)

    if beta <= 0.0:
        w = np.full(m, 1.0 / m, dtype=float)
    else:
        s = beta * (r0 - maxr)
        w = np.exp(s)
        sw = float(np.sum(w))
        if not np.isfinite(sw) or sw <= 0.0:
            w = np.zeros(m, dtype=float)
            active = r0 >= maxr - 1e-8
            if not np.any(active):
                active[-1] = True
            w[active] = 1.0
            w /= float(np.sum(w))
        else:
            w /= sw

    grad = D.dot(w)
    d = -grad
    d -= float(np.mean(d))
    nrm = float(np.max(np.abs(d)))
    if nrm < 1e-12:
        return None
    d /= nrm

    if rng.random() < 0.3:
        d += 1e-3 * rng.standard_normal(len(d))
        d -= float(np.mean(d))
        nrm = float(np.max(np.abs(d)))
        if nrm > 0.0:
            d /= nrm

    return d


def random_direction(n, rng):
    d = rng.standard_normal(n)
    d -= float(np.mean(d))
    nrm = float(np.max(np.abs(d)))
    if nrm < 1e-12:
        return None
    return d / nrm


def transfer_parent(parent, n):
    m = len(parent)
    if m == n:
        return project_box_mass(parent.copy(), n)
    old_dx = 2.0 / m
    new_dx = 2.0 / n
    old_x = old_dx * (np.arange(m) + 0.5)
    new_x = new_dx * (np.arange(n) + 0.5)
    h = np.interp(new_x, old_x, parent, left=parent[0], right=parent[-1])
    return project_box_mass(h, n)


def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    if n % 2 == 0:
        idx = np.sort(rng.choice(n, size=n // 2, replace=False))
        h[idx] = 1.0
    else:
        k = (n - 1) // 2
        idx = np.sort(rng.choice(n, size=k, replace=False))
        h[idx] = 1.0
        others = np.setdiff1d(np.arange(n), idx)
        if len(others) > 0:
            h[others[rng.integers(len(others))]] = 0.5
    return h


def random_fourier(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    f1 = float(rng.integers(1, 5))
    f2 = float(rng.integers(1, 6))
    ph1 = float(rng.uniform(0.0, 2.0 * math.pi))
    ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
    y = (
        0.5
        + 0.35 * np.sin(2.0 * math.pi * f1 * x + ph1)
        + 0.25 * np.cos(2.0 * math.pi * f2 * x + ph2)
    )
    return project_box_mass(y, n)


def generate_start(n, rng, parent, mode):
    if mode == 0 and parent is not None:
        return transfer_parent(parent, n)
    if mode == 1:
        return random_binary(n, rng)
    if mode == 2:
        return random_fourier(n, rng)
    return project_box_mass(rng.uniform(size=n), n)


def optimize_local(h, n, deadline, rng):
    h = project_box_mass(h, n)
    if time.time() >= deadline:
        return h, c5_value(h, n)

    structs = make_structures(n)
    flag = structs[8]
    r0 = raw_corr(h, flag)
    D = compute_deriv(h, structs)
    current_raw = float(np.max(r0))

    while time.time() < deadline - 15.0:
        improved = False

        for _ in range(8):
            if time.time() >= deadline - 10.0:
                break
            h, r0, D, imp = pair_descent_pass(
                h, n, r0, D, structs, rng, deadline - 5.0, 1e-9
            )
            if imp:
                improved = True
                current_raw = float(np.max(r0))
                if current_raw <= 1e-14:
                    break
            else:
                break

        if time.time() >= deadline - 15.0:
            break
        if current_raw <= 1e-14:
            break

        for beta in (0.0, 5.0, 20.0, 100.0, 500.0):
            if time.time() >= deadline - 15.0:
                break
            d = active_direction(h, r0, D, beta, rng)
            if d is None:
                continue
            res = try_dense_move(h, n, d, current_raw, deadline - 5.0, 1e-8)
            if res is not None:
                h, r0, current_raw = res
                D = compute_deriv(h, structs)
                improved = True
                if current_raw <= 1e-14:
                    break

        for _ in range(3):
            if time.time() >= deadline - 15.0:
                break
            d = random_direction(n, rng)
            if d is None:
                continue
            res = try_dense_move(h, n, d, current_raw, deadline - 5.0, 1e-8)
            if res is not None:
                h, r0, current_raw = res
                D = compute_deriv(h, structs)
                improved = True

        if not improved:
            break

    h = project_box_mass(h, n)
    return h, c5_value(h, n)


def solution_better(c, n, best_c, best_n):
    if c < best_c - 1e-12:
        return True
    if abs(c - best_c) <= 1e-12 and n < best_n:
        return True
    return False


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(60.0, float(budget_s))
    rng = np.random.default_rng(seed)

    parent = None
    n0 = 70
    try:
        initial_h_values
    except NameError:
        parent = None
    else:
        parent = np.array(initial_h_values, dtype=float)
        n0 = len(parent)

    if parent is None:
        parent = random_binary(n0, rng)

    best_h = project_box_mass(parent.copy(), n0)
    best_c = c5_value(best_h, n0)
    best_n = n0

    phase1_deadline = t0 + min(500.0, 0.5 * budget)
    if phase1_deadline > t0 + 30.0:
        h, c = optimize_local(parent.copy(), n0, phase1_deadline, rng)
        if solution_better(c, n0, best_c, best_n):
            best_h, best_c, best_n = h, c, n0

    final_reserve = 180.0 if budget > 400.0 else 60.0
    attempt_deadline = t0 + budget - final_reserve

    candidate_ns = []
    base = best_n if best_n > 0 else n0
    for n in (
        base,
        max(20, base - 10),
        base + 10,
        35, 40, 45, 50, 55, 60, 65, 75, 80, 90, 100, 30
    ):
        if 20 <= n <= 120 and n not in candidate_ns:
            candidate_ns.append(n)

    modes = [1, 2, 3, 0]
    round_no = 0
    while time.time() < attempt_deadline and round_no < len(modes):
        mode = modes[round_no]
        for n in candidate_ns:
            if time.time() >= attempt_deadline:
                break
            remaining = attempt_deadline - time.time()
            if remaining < 20.0:
                break
            slice_s = min(45.0, 0.5 * remaining)
            if slice_s < 15.0:
                slice_s = max(10.0, 0.8 * remaining)
            deadline = time.time() + slice_s

            h_start = generate_start(n, rng, parent, mode)
            h, c = optimize_local(h_start, n, deadline, rng)
            if solution_better(c, n, best_c, best_n):
                best_h, best_c, best_n = h, c, n

        round_no += 1

    final_deadline = t0 + budget - 5.0
    if time.time() < final_deadline - 10.0:
        h, c = optimize_local(best_h.copy(), best_n, final_deadline, rng)
        if solution_better(c, best_n, best_c, best_n):
            best_h, best_c, best_n = h, c, best_n

    best_h = project_box_mass(best_h, best_n)
    best_c = c5_value(best_h, best_n)

    return best_h.tolist(), float(best_c), int(best_n)
