def grad_lag(h, tau):
    n = len(h)
    grad = np.zeros(n)
    # Vectorized version
    # For each j: grad[j] += (1-h[j+tau]) if 0<=j+tau<n
    #             grad[j] -= h[j-tau] if 0<=j-tau<n
    lo1 = max(0, -tau)
    hi1 = min(n, n - tau)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1+tau:hi1+tau])
    lo2 = max(0, tau)
    hi2 = min(n, n + tau)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2-tau:hi2-tau]
    return grad
