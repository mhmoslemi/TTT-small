                if diff <= 0:
                    break
                add = min(diff, slack[i])
                if add > 0:
                    z[i] += add
                    diff -= add
        else:
            diff = -diff
            idx = np.argsort(z)
            for i in idx:
                if diff <= 0:
                    break
                sub = min(diff, z[i])
                if sub > 0:
                    z[i] -= sub
                    diff -= sub
    return z

def cvals_and_grad(h, dx, tau=None):
    n = len(h)
    center = n - 1
    full = np.correlate(h, 1.0 - h, mode='full') * dx
    if tau is None:
        return full, None
    M = np.max(full)
    shifted = np.clip((full - M) / tau, -500, 500)
    exps = np.exp(shifted)
    tot = np.sum(exps)
    S = M + tau * np.log(tot)
    w = exps / tot
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        l = k - center
        g = np.zeros(n)
        if l >= 0:
            if l == 0:
                g[:] = dx * (1.0 - 2.0 * h)
            else:
                g[l:] += dx * (1.0 - h[:n - l])
                g[:n - l] -= dx * h[l:]
        else:
            p = -l
            g[:n - p] += dx * (1.0 - h[p:])
            g[p:] -= dx * h[:n - p]
        grad += w[k] * g
    return S, grad

def smooth_obj(x, args):
    tau, dx = args
    val, _ = cvals_and_grad(x, dx, tau)
    return val

def smooth_grad(x, args):
    tau, dx = args
    _, g = cvals_and_grad(x, dx, tau)
    return g

def mass_constraint(x, args):
    target = args
    return np.sum(x) - target

def mass_jac(x, args):
    return np.ones(len(x))

def local_pair_moves(h, n_moves, seed_val, best_score):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed_val)
    cur = h.copy()
    cur_score = best_score
    deltas = [0.25, 0.15, 0.1, 0.06, 0.04, 0.025, 0.015, 0.01, 0.006, 0.003]
    attempt = 0
    for delta in deltas:
        while attempt < n_moves:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = delta * (1 if cur[i] >= cur[j] else -1)
            if cur[i] - d < 0 or cur[i] - d > 1:
                continue
            if cur[j] + d < 0 or cur[j] + d > 1:
                continue
            old_i, old_j = cur[i], cur[j]
            cur[i] -= d
            cur[j] += d
            sc = exact_c5(cur)
            if sc < cur_score - 1e-12:
                cur_score = sc
            else:
                cur[i], cur[j] = old_i, old_j
            attempt += 1
    return cur, cur_score

def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    # Build cumulative area of source
    xs_src = np.arange(n_src + 1) * dx_src
    areas = np.concatenate([[0.0], np.cumsum(h_src * dx_src)])
    # Interpolate at target bin boundaries
    xs_tgt = np.arange(n_tgt + 1) * dx_tgt
    lin_areas = np.interp(xs_tgt, xs_src, areas)
    tgt = np.diff(lin_areas) / dx_tgt
    return project_feasible(tgt)

def optimize_smoothed(h0, best_score, time_limit, seed_val):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    x0 = h0.copy()
    constraints = [{'type': 'eq', 'fun': mass_constraint,
                    'jac': mass_jac, 'args': (target,)}]
    bounds = [(0.0, 1.0)] * n
    taus = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    t_start = time.time()
    for tau in taus:
        if time.time() - t_start > time_limit:
            break
        try:
            res = minimize(smooth_obj, x0, jac=smooth_grad,
                           args=(tau, dx), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                x0 = project_feasible(res.x)
                sc = exact_c5(x0)
                if sc < best_score:
                    best_score = sc
        except Exception:
            pass
    return x0, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.93

    # Get parent
    parent = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            p = np.array(initial_h_values, dtype=np.float64).flatten()
            if len(p) >= 10 and np.all(np.isfinite(p)):
                parent = project_feasible(p)
    except Exception:
        pass

    if parent is None:
        n0 = 45
        parent = np.full(n0, 0.5)
        # Add some structure
        rng0 = np.random.RandomState(seed)
        noise = rng0.normal(0, 0.3, n0)
        parent = project_feasible(parent + noise)

    best_h = parent.copy()
    best_score = exact_c5(best_h)
    n0 = len(best_h)
    dx0 = 2.0 / n0

    # Generate diverse starts
    starts = [best_h.copy()]
    # Reverse
    starts.append(parent[::-1].copy())
    # Add noise variants
    rng = np.random.RandomState(seed)
    for sigma in [0.3, 0.15, 0.08, 0.05]:
        noisy = project_feasible(parent + rng.normal(0, sigma, n0))
        starts.append(noisy)
    # Low-discrepancy style
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    ld = np.zeros(n0)
    fracs = np.sort(np.mod(np.arange(n0) * phi, 1.0))
    half_n = n0 // 2
    top_idx = np.argsort(-fracs)[:half_n]
    ld[top_idx] = 1.0
    if n0 % 2 == 1:
        ld[np.argsort(-fracs)[half_n]] = 0.5
    starts.append(project_feasible(ld))

    # Stage 1: Smoothed SLSQP on original resolution
    print(f"Parent C5 = {best_score:.6f}, n={n0}")
    stage_budget = budget_s * 0.4
    for si, h_start in enumerate(starts):
        if time.time() > deadline - stage_budget:
            break
        if time.time() - t0 > stage_budget + 30:
            break
        h_opt, sc = optimize_smoothed(h_start, best_score,
                                      time_limit=stage_budget / max(1, len(starts)),
                                      seed_val=seed + si)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()
            print(f"  Start {si}: improved to {best_score:.6f}")
        elif h_opt is not None:
            h_sc = exact_c5(h_opt)
            if h_sc < best_score:
                best_score = h_sc
                best_h = h_opt.copy()

    # Stage 2: Local pair moves
    if time.time() < deadline - 60:
        moves_budget = min(3000, int((deadline - time.time()) * 50))
        h_p, sc_p = local_pair_moves(best_h, moves_budget, seed + 100, best_score)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p.copy()
            print(f"  Pair moves: {best_score:.6f}")
        # Repeat with different seed
        if time.time() < deadline - 30:
            h_p2, sc_p2 = local_pair_moves(best_h, moves_budget // 2, seed + 200, best_score)
            if sc_p2 < best_score:
                best_score = sc_p2
                best_h = h_p2.copy()

    # Stage 3: Re-optimize best with finer smoothing
    if time.time() < deadline - 30:
        h_opt, sc = optimize_smoothed(best_h, best_score,
                                      time_limit=min(60, deadline - time.time() - 10),
                                      seed_val=seed + 500)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()

    # Stage 4: Resolution transfer
    if time.time() < deadline - 20:
        n_opts = []
        for factor in [0.75, 1.25, 1.5, 2.0]:
            nn = int(round(n0 * factor))
            nn = max(20, min(nn, 120))
            if nn != n0:
                n_opts.append(nn)
        # Also try some absolute values
        for abs_n in [45, 60, 80, 100]:
            if abs_n != n0 and abs_n not in n_opts:
                n_opts.append(abs_n)
        n_opts = sorted(set(n_opts))

        for nn in n_opts:
            if time.time() > deadline - 15:
                break
            h_t = transfer_resolution(best_h, nn)
            sc_t = exact_c5(h_t)
            # Quick smoothing at new resolution
            if time.time() < deadline - 25:
                h_t, sc_t2 = optimize_smoothed(h_t, min(sc_t, best_score),
                                               time_limit=min(20, deadline - time.time() - 15),
                                               seed_val=seed + nn)
                if sc_t2 < sc_t:
                    sc_t = sc_t2
                    h_t_best = h_t
                else:
                    h_t_best = h_t
            else:
                h_t_best = h_t
            sc_final = exact_c5(h_t_best)
            if sc_final < best_score:
                best_score = sc_final
                best_h = h_t_best.copy()
                print(f"  n={nn}: improved to {best_score:.6f}")

    # Stage 5: Final pair moves on best
    if time.time() < deadline - 5:
        remaining = int((deadline - time.time()) * 40)
        remaining = max(500, min(remaining, 5000))
        h_f, sc_f = local_pair_moves(best_h, remaining, seed + 999, best_score)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f.copy()

    # Final validation
    final_h = project_feasible(best_h)
    final_score = exact_c5(final_h)
    n_final = len(final_h)
    assert np.all(final_h >= -1e-15) and np.all(final_h <= 1 + 1e-15)
    assert abs(np.sum(final_h) - n_final / 2.0) < 1e-8

    # Re-check score matches
    expected = float(np.max(np.correlate(final_h, 1.0 - final_h, mode='full')) * (2.0 / n_final))
    final_score = expected

    print(f"Final: C5 = {final_score:.6f}, n = {n_final}")
    return final_h, final_score, int(n_final)
