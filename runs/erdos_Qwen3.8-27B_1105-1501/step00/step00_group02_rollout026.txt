import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj_and_grad(h, n, dx, T):
    """Compute smoothed objective and its gradient."""
    # Compute all correlation values
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # shifts: for output index i, shift k = i - (n-1)
    shifts = np.arange(2*n - 1) - (n - 1)  # from -(n-1) to (n-1)
    
    # Softmax weights
    scaled = T * dx * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / np.sum(exp_scaled)
    
    obj = (max_scaled + np.log(np.sum(exp_scaled))) / T * dx
    
    # Gradient
    grad = np.zeros(n)
    for idx, k in enumerate(shifts):
        if p[idx] < 1e-15:
            continue
        # ∂c(k)/∂h[j] = (1-h[j-k]) * I(valid) - h[j+k] * I(valid)
        for j in range(n):
            term = 0.0
            if 0 <= j - k < n and 0 <= j < n:
                term += (1.0 - h[j - k])
            if 0 <= j < n and 0 <= j + k < n:
                term -= h[j + k]
            grad[j] += p[idx] * term
    grad *= dx
    
    return obj, grad

        # For k >= 0: j in [0, n-1-k], h[j+k]
        # For k < 0: j in [-k, n-1], h[j+k]
        
        if k >= 0:
            # First term: j in [k, n-1], contribution (1-h)[j-k]
            if n - k > 0:
                grad[k:] += p[idx] * complement[:n-k]
            # Second term: j in [0, n-1-k], contribution -h[j+k]
            if n - k > 0:
                grad[:n-k] -= p[idx] * h[k:]
        else:
            kk = -k
            # First term: j in [0, n-1-kk], contribution (1-h)[j+kk]
            if n - kk > 0:
                grad[:n-kk] += p[idx] * complement[kk:]
            # Second term: j in [kk, n-1], contribution -h[j-kk]
            if n - kk > 0:
                grad[kk:] -= p[idx] * h[:n-kk]
    
    grad *= dx
    return obj, grad
```

def smooth_logsumexp(x, T):
    """Compute (1/T)*log(sum(exp(T*x))) stably."""
    m = np.max(x)

```python
import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_lse(x, T):
    m = np.max(x)
    ex = np.exp(T * (x - m))
    return float(m + np.log(np.sum(ex)) / T)

def _smoothed_obj(h, dx, T):
    corr_dx = np.correlate(h, 1.0 - h, mode="full") * dx
    return _smooth_lse(corr_dx, T)

def _smoothed_grad(h, dx, T):
    n = len(h)
    corr_dx = np.correlate(h, 1.0 - h, mode="full") * dx
    scaled = T * corr_dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    p = exp_s / np.sum(exp_s)
    shifts = np.arange(2 * n - 1) - (n - 1)
    comp = 1.0 - h
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = p[idx]
        if wi < 1e-14:
            continue
        k = int(shifts[idx])
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:] += wi * comp[:m]
                grad[:m] -= wi * h[k:]
        else:
            kk = -k
            m = n - kk
            if m > 0:
                grad[:m] += wi * comp[kk:]
                grad[kk:] -= wi * h[:m]
    return grad * dx

def _eq_fun(h, target_sum):
    return np.sum(h) - target_sum

def _eq_jac(h, target_sum):
    return np.ones(len(h))

def _binary_ls(h, dx, time_limit=30.0, start_time=None):
    n = len(h)
    h = h.copy()
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    best_val = _compute_c5(h, dx)
    
    while True:
        if start_time is not None and time.time() - start_time > time_limit:
            break
        improved = False
        best_v = best_val
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                v = _compute_c5(h, dx)
                h[i] = 1.0
                h[j] = 0.0
                if v < best_v - 1e-12:
                    best_v = v
                    best_pair = (i, j)
                    improved = True
        if not improved:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        best_val = best_v
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
    
    return h, best_val

def _periodic_init(n):
    h = np.zeros(n)
    half = n // 2
    # Pattern: 1 for first half of each period-1 block
    for i in range(n):
        h[i] = 1.0 if (i % half) < half // 2 else 0.0
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.any():
                h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

def _random_binary(n, rng):
    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _to_binary(h):
    n = len(h)
    target = n / 2.0
    idx = np.argsort(-h)
    h2 = np.zeros(n)
    h2[idx[:int(target)]] = 1.0
    rem = target - int(target)
    if rem > 0:
        h2[idx[int(target)]] = rem
    return h2

def _fix_sum(h, target):
    h = h.copy()
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            if mask.any():
                h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.any():
                h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)
