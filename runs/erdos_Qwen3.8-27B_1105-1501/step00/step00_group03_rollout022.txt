def project_onto_feasible(vec, target_sum):
    """Project vec onto {h: sum(h) = target_sum, 0 <= h[i] <= 1}"""
    h = np.clip(vec, 0, 1)
    # Now adjust to meet sum constraint
    # Use bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any floating point issues
    diff = target_sum - np.sum(h)

    # Find indices where 0 < h < 1
    free = (h > 1e-10) & (h < 1 - 1e-10)

        h[free] += diff / n_free

    return h

def smooth_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    one_h = 1 - h
    corr = np.correlate(h, one_h, mode="full")
    # Normalize for numerical stability
    corr_shifted = corr - np.max(corr)
    log_sum = np.max(corr) + np.log(np.sum(np.exp(alpha * corr_shifted)))
    return log_sum / alpha

def smooth_gradient(h, alpha):
    n = len(h)

    weights /= np.sum(weights)

import numpy as np
from scipy.optimize import linprog

def project_feasible(h, S):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    residual = S - np.sum(h)
    if abs(residual) > 1e-12:
        free = (h > 1e-9) & (h < 1 - 1e-9)
        nf = np.sum(free)
        if nf > 0:
            h[free] += residual / nf
            h = np.clip(h, 0, 1)
    return h

def compute_grads(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        idx = k + n - 1
        i_start = max(0, -k)
        i_end = min(n, n - k)
        i_range = np.arange(i_start, i_end)
        j_range = i_range + k
        np.add.at(grads[idx], i_range, 1.0 - h[j_range])
        np.add.at(grads[idx], j_range, -h[i_range])
    return grads

def eval_c5_val(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def do_lp(h, S, corr, grads, m):
    n = len(h)
    m = min(m, 2 * n - 1)
    active = np.argsort(corr)[-m:]
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0
    A_ub = np.zeros((m, n_vars))
    b_ub = np.zeros(m)
    for i, idx in enumerate(active):
        g_k = grads[idx]
        F_k = corr[idx]
        A_ub[i, :n] = g_k
        A_ub[i, n] = -1.0
        b_ub[i] = -F_k + g_k @ h
    A_eq = np.array([np.concatenate([np.ones(n), [0.0]])])
    b_eq = np.array([S])
    bounds = [(0, 1)] * n + [(-1, 1)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

        h_t = project_feasible(h + a * direction, S)
        f_t = eval_c5_val(h_t, dx)
        if f_t < best_f:
            best_f = f_t
            best_a = a
    return best_a, best_f
