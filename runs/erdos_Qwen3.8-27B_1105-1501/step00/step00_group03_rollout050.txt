    z = np.asarray(z, dtype=float).ravel()
    n = len(z)
    if target is None:
        target = n / 2.0
    if n == 0:
        return np.array([])
    
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()

        s = np.clip(z + mid, 0.0, 1.0).sum()

            hi = mid

    h = np.clip(z + lam, 0.0, 1.0)
