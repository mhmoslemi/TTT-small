def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def get_c_values(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    return np.correlate(h, b, mode='full') * dx

def get_c_value_k(h, k):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    lag = k - (n - 1)
    if lag >= 0:
        idx = np.arange(n - lag)
        return dx * np.sum(h[idx + lag] * b[idx])
    else:
        p = -lag
        idx = np.arange(n - p)
        return dx * np.sum(h[idx] * b[idx + p])

def grad_c_k(h, k):
    n = len(h)
    dx = 2.0 / n
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += dx * (1.0 - h[:n-lag])
        g[:n-lag] -= dx * h[lag:]
    else:
        p = -lag
        g[:n-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n-p]
    return g

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.array(y, dtype=np.float64).copy()
    # Handle non-finite

        return y
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix mass exactly
    diff = target - np.sum(result)

            else:

    return result

def smooth_obj_func(x, args):
    tau, n = args
    dx = 2.0 / n
    b = 1.0 - x
    c_vals = np.correlate(x, b, mode='full') * dx
    m = np.max(c_vals)
    shifted = (c_vals - m) / tau
    # Handle overflow
    max_shifted = np.max(shifted)
    if max_shifted > 500:
        shifted -= max_shifted
        result = m + max_shifted * tau + tau * np.log(np.sum(np.exp(shifted)))
    else:
        result = m + tau * np.log(np.sum(np.exp(shifted)))
    return result

    tau, n = args

def pair_search(h, rng, max_evals=5000, dx=None):
    """Local search via coordinate pair mass transfers"""
    n = len(h)
    if dx is None:
        dx = 2.0 / n
    current = h.copy()
    current_score = exact_c5(current)
    best_local = current.copy()
    best_local_score = current_score
    target = n / 2.0
    
    deltas = [0.1, 0.05, 0.02, 0.01, 0.005]
    
    for delta in deltas:
        for _ in range(max_evals // len(deltas)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Try moving mass from i to j (and vice versa)
            for direction in [1, -1]:
                move = direction * delta
                new_h = current.copy()
                new_h[i] += move
                new_h[j] -= move
                # Check bounds
                if new_h[i] < -1e-10 or new_h[i] > 1.0 + 1e-10 or new_h[j] < -1e-10 or new_h[j] > 1.0 + 1e-10:
                    continue
                new_h = np.clip(new_h, 0.0, 1.0)
                # Check mass
                if abs(np.sum(new_h) - target) > 1e-10:
                    continue
                sc = exact_c5(new_h)
                if sc < current_score - 1e-12:
                    current = new_h
                    current_score = sc
                    if sc < best_local_score:
                        best_local = current.copy()
                        best_local_score = sc
                    break
    
    return best_local, best_local_score

def transfer_resolution(h_src, n_tgt):
    """Transfer shape from source to target resolution via area interpolation"""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    # Build cumulative area of source
    # Source has bins [j*dx_src, (j+1)*dx_src] with height h_src[j]
    # Cumulative area at point x
    x_src = np.arange(n_src + 1) * dx_src
    cum_src = np.zeros(n_src + 1)
    for j in range(n_src):
        cum_src[j+1] = cum_src[j] + h_src[j] * dx_src
    
    # Target bin centers
    x_tgt = (np.arange(n_tgt) + 0.5) * dx_tgt
    
    # Interpolate cumulative area at target points
    cum_tgt = np.interp(x_tgt, x_src, cum_src)
    
    # Height at target = area in bin / dx_tgt
    # Area in bin = cum at right edge - cum at left edge
    x_tgt_edges = (np.arange(n_tgt + 1)) * dx_tgt
    cum_edges = np.interp(x_tgt_edges, x_src, cum_src)
    heights = np.diff(cum_edges) / dx_tgt
    
    return heights
