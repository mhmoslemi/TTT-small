import numpy as np
import time

def compute_c5_exact(h, n):
    """Compute the exact C5 value."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_onto_feasible(h, n):
    """Project h onto the feasible set: 0<=h<=1, sum(h)=n/2."""
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then adjust to satisfy sum constraint
    target = n / 2
    current_sum = np.sum(h)
    diff = target - current_sum
    if diff > 0:
        # Need to increase sum; add to non-saturated entries
        mask = h < 1
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        # Rescale if needed (shouldn't be needed if we started close)
        h = np.clip(h, 0, 1)
    elif diff < 0:
        mask = h > 0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def soft_max(x, T):
    """Compute soft-max: T * log(sum(exp(x/T)))"""
    x_scaled = x / T
    x_max = np.max(x_scaled)
    return T * x_max + T * np.log(np.sum(np.exp(x_scaled - x_max)))

def compute_gradient(h, n, T):
    """Compute gradient of soft_c5 with respect to h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    # Softmax weights

    scaled -= np.max(scaled)
    weights = np.exp(scaled) / np.sum(np.exp(scaled))

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Force exact
        idx = np.argsort(h)
        if diff > 0:
            for i in range(len(idx)-1, -1, -1):
                room = 1.0 - h[idx[i]]
                add = min(room, diff)
                h[idx[i]] += add
                diff -= add

                sub = min(room, -diff)
