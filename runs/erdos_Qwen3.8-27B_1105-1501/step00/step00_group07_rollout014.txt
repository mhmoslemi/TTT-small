def exact_score(h):
    """Compute C5 = max_k np.correlate(h, 1-h, mode='full') * dx"""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return raw.max() * dx, raw

def project_feas(v, tol=1e-9):
    """Project v to [0,1]^n with sum = n/2"""
    n = len(v)
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    # Replace non-finite
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    # Water-filling to achieve exact sum
    s = v.sum()
    if abs(s - target) <= tol:
        return v
    # Bisection for lambda: sum(clip(v + lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.clip(v + mid, 0.0, 1.0).sum()
        if val > target:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-15 * n:
            break
    lam = (lo + hi) / 2.0
    v2 = np.clip(v + lam, 0.0, 1.0)
    # Fix residual
    resid = v2.sum() - target
    if resid > tol:
        # Need to decrease
        idx = np.argsort(v2)[::-1]  # largest first
        for i in idx:
            dec = min(v2[i], resid)
            v2[i] -= dec
            resid -= dec
            if resid <= tol:
                break
    elif resid < -tol:
        # Need to increase
        idx = np.argsort(v2)  # smallest first (most room)
        for i in idx:
            inc = min(1.0 - v2[i], -resid)
            v2[i] += inc
            resid += inc
            if resid >= -tol:
                break
    return v2

def pair_line_search(h, a, b, current_C, gain_threshold, golden_iters=12):

    # Sample points

        h[b] = qq - t

            gr, gl = t_low, t_high
        else:
            # Narrow bracket around best

            gr = min(t_high, best_t + 2*delta)
            gl = max(t_low, best_t - 2*delta)
        
        # Golden section
        golden = (np.sqrt(5) - 1) / 2
        a_g = gl + (1 - golden) * (gr - gl)

        h[a] = pp + a_g

                b_g = a_g
                Cb = Ca
                a_g = gl + (1 - golden) * (gr - gl)
                h[a] = pp + a_g
                h[b] = qq - a_g
                Ca, _ = exact_score(h)

def resample(h, target_n):

    new_h = np.interp(x_new, x_old, h)

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return raw.max() * dx, raw

def project_feas(v, tol=1e-9):
    n = len(v)
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= tol:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        val = np.clip(v + mid, 0.0, 1.0).sum()
        if val > target:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-14 * n:
            break
    lam = (lo + hi) / 2.0
    v2 = np.clip(v + lam, 0.0, 1.0)
    resid = v2.sum() - target
    if resid > tol:
        idx = np.argsort(v2)[::-1]
        for i in idx:
            dec = min(v2[i], resid)
            v2[i] -= dec
            resid -= dec
            if resid <= tol:
                break
    elif resid < -tol:
        idx = np.argsort(v2)
        for i in idx:
            inc = min(1.0 - v2[i], -resid)
            v2[i] += inc
            resid += inc
            if resid >= -tol:
                break
    return v2

def resample_to_size(h, target_n):

    if n == target_n:
        return h.copy()
    x_old = 2.0 * (np.arange(n) + 0.5) / n
    x_new = 2.0 * (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, h)
    return project_feas(new_h)

def gradient_direction(h, raw, raw_max):
    n = len(h)
    if raw_max < 1e-10:
        return None
    active_margin = 0.1 * raw_max
    active = np.where(raw >= raw_max - active_margin)[0]
    if len(active) == 0:
        active = np.array([np.argmax(raw)])
    G = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        for j in range(n):
            g = 0.0
            jp = j + lag
            jm = j - lag
            if 0 <= jp < n:
                g += (1.0 - h[jp])
            if 0 <= jm < n:
                g -= h[jm]
            G[j] += g
    G -= G.mean()
    norm = np.abs(G).max()
    if norm < 1e-10:
        return None
    return -G / norm

def apply_direction(h, direction, current_C, deadline):
    n = len(h)
    pos = direction > 0
    neg = direction < 0
    max_alpha = 10.0
    if pos.any():
        max_alpha = min(max_alpha, ((1.0 - h[pos]) / direction[pos]).min())
    if neg.any():
        max_alpha = min(max_alpha, (h[neg] / (-direction[neg])).min())
    max_alpha = max(max_alpha, 0.0)
    if max_alpha < 1e-12:
        return h, current_C
    best_C = current_C
    best_h = h
    n_tries = 10
    for i in range(1, n_tries + 1):
        if time.time() > deadline:
            break
        alpha = max_alpha * i / n_tries
        h_new = project_feas(h + alpha * direction)
        C_new, _ = exact_score(h_new)
        if C_new < best_C:
            best_C = C_new
            best_h = h_new
    return best_h, best_C

def random_perturb(h, rng, n_pert=8, mag=0.15):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_pert):
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:

        t = float(rng.uniform(-mag, mag))
        pp = h_new[a]
        qq = h_new[b]
        t = max(-min(pp, 1.0 - qq), min(t, min(1.0 - pp, qq)))
        h_new[a] = pp + t
        h_new[b] = qq - t
    return project_feas(h_new)

            sz_deadline = min(deadline - 3, time.time() + per_size)
            if sz_deadline - time.time() > 3:
                h_sz, C_sz = pair_sweep(h_sz, C_sz, sz_deadline)
            if C_sz < best_C:
                best_C = C_sz
                best_h = h_sz.copy()

    # Phase 3: gradient refinement on best
    if time.time() < deadline - 5:
        grad_deadline = deadline - 5
        for _ in range(30):
            if time.time() > grad_deadline:
                break
            _, raw = exact_score(best_h)
            raw_max = raw.max()
            if raw_max < 1e-10:
                break
            direction = gradient_direction(best_h, raw, raw_max)
            if direction is None:
                break
            best_h, new_C = apply_direction(best_h, direction, best_C, grad_deadline)
            if new_C < best_C:
                best_C = new_C
            else:
                break

    # Phase 4: perturbation + re-optimization to escape local minima
    if time.time() < deadline - 10:
        n_pert = 8
        for p in range(n_pert):
            if time.time() > deadline - 8:
                break
            h_pert = random_perturb(best_h, rng, n_pert=6, mag=0.1)
            C_pert, _ = exact_score(h_pert)
            pert_deadline = min(deadline - 5, time.time() + 20.0)
            h_pert, C_pert = pair_sweep(h_pert, C_pert, pert_deadline)
            if C_pert < best_C:
                best_C = C_pert
                best_h = h_pert.copy()

    # Phase 5: final polish
    if time.time() < deadline - 1:
        best_h, best_C = pair_sweep(best_h.copy(), best_C, deadline - 0.5, min_gain=1e-9)

    best_h = project_feas(best_h)
    final_C, _ = exact_score(best_h)
    if final_C < best_C:
        best_C = final_C

    return best_h, best_C, len(best_h)
