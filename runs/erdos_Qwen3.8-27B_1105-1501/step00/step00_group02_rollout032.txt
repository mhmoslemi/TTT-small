```python
def compute_gradient(h, w, n):
    """Compute gradient of smoothed objective.
    h: array of length n
    w: array of length 2n-1, indexed by m=0,...,2n-2, corresponding to lag m-(n-1)
    """
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(-n+1, n):
            if 0 <= i+k < n:
                s += w[k+n-1] * (1 - h[i+k])
            if 0 <= i-k < n:
                s -= w[k+n-1] * h[i-k]
        grad[i] = s
    return grad
```

```python
def _project(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2}."""
    target = n * 0.5
    # Binary search for water level
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    d = target - h.sum()
    if abs(d) > 1e-14:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.any():
            h[free] += d / free.sum()
            np.clip(h, 0.0, 1.0, out=h)

    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

    """Gradient of smoothed max objective. w is normalized weights over lags."""
    i = np.arange(n)[:, None]
    k = np.arange(-(n - 1), n)[None, :]
    A = i + k  # (n, 2n-1): A[i,k] = i+k

    hA = np.where((A >= 0) & (A < n), h[np.clip(A, 0, n - 1)], 0.0)
```
