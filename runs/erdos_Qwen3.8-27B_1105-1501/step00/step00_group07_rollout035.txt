        else:
            hi = mid

        else:

    """Compute softmax weights for correlation values."""
    m = np.max(F)
    exps = np.exp(alpha * (F - m))
    s = np.sum(exps)
    if s < 1e-300:
        return np.ones(len(F)) / len(F)
    return exps / s
