def _project(z, target, tol=1e-10):

    ts = np.linspace(lo, hi, ng)
    for t in ts:
        if deadline is not None and time.time() > deadline:
            break
        h[p] = hp - t
        h[q] = hq + t
        v = _evaluate(h)
        if v < best_val - 1e-14:
            best_val = v
            best_h = h.copy()
            best_t = t
    h[p] = hp
    h[q] = hq
    
    if best_t == 0.0 and best_val >= _evaluate(h):
        return h, best_val
    
    # Refine around best_t
    span = hi - lo
    lo2 = max(best_t - 0.15 * span, lo)
    hi2 = min(best_t + 0.15 * span, hi)
    if hi2 > lo2:
        ts2 = np.linspace(lo2, hi2, 12)
        for t in ts2:
            if deadline is not None and time.time() > deadline:
                break
            h[p] = hp - t
            h[q] = hq + t
            v = _evaluate(h)
            if v < best_val - 1e-14:
                best_val = v
                best_h = h.copy()
    h[p] = hp
    h[q] = hq
    return best_h, best_val


def _make_binary(n, rng):
    """Create a random binary-like feasible vector"""
    target = n / 2.0
    if target == int(target):
        k = int(target)
        h = np.zeros(n)
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    else:
        k = int(target)
        h = np.zeros(n)
        idx = rng.choice(n, size=k + 1, replace=False)
        h[idx] = 1.0
        # One entry must be 0.5 to get sum = k + 0.5
        h[idx[0]] = 0.5
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.94
    rng = np.random.RandomState(seed)
    
    # Initialize from parent
    parent_h = None
    try:
        parent_h = np.asarray(initial_h_values, dtype=float).flatten().copy()
    except Exception:
        pass
    
    if parent_h is not None and len(parent_h) >= 4:
        n = len(parent_h)
        best_h = _project(parent_h, n / 2.0)
        best_C = _evaluate(best_h)
    else:
        n = 48
        best_h = np.full(n, 0.5)
        best_C = _evaluate(best_h)
    
    best_n = len(best_h)
    
    def _try_improve(hh, cc, dl, n_pairs, ng):
        """Local improvement via pair line search. Returns (h, C, improved)."""
        nn = len(hh)
        improved = False
        for _ in range(n_pairs):
            if time.time() > dl:
                break
            p = int(rng.randint(nn))
            q = int(rng.randint(nn))
            if p == q:
                continue
            h_new, v_new = _pair_search(hh, p, q, ng, dl)
            if v_new < cc - 1e-12:
                hh = h_new
                cc = v_new
                improved = True
        return hh, cc, improved
    
    def _active_pair_search(hh, cc, dl, max_iters):
        """Use active lags to guide pair selection."""
        nn = len(hh)
        for it in range(max_iters):
            if time.time() > dl:
                break
            dx = 2.0 / nn
            corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
            C = np.max(corr)
            
            # Top active lags
            k_active = min(12, 2 * nn - 1)
            top_idx = np.argsort(corr)[-k_active:][::-1]
            
            # Build candidate pairs from active lags
            pairs = []
            for li in top_idx:
                d = int(li) - (nn - 1)
                if d == 0:
                    continue
                i_start = max(0, -d)
                i_end = min(nn, nn - d) if d > 0 else nn
                for i in range(i_start, i_end):
                    j = i + d
                    if 0 <= j < nn and i != j:
                        pairs.append((i, j))
            
            # Add random pairs
            for _ in range(15):
                p = int(rng.randint(nn))
                q = int(rng.randint(nn))
                if p != q:
                    pairs.append((p, q))
            
            if not pairs:
                break
            
            rng.shuffle(pairs)
            local_improved = False
            for p, q in pairs[:80]:
                if time.time() > dl:
                    break
                h_new, v_new = _pair_search(hh, p, q, 24, dl)
                if v_new < cc - 1e-12:
                    hh = h_new
                    cc = v_new
                    local_improved = True
            
            if not local_improved:
                break
        return hh, cc
    
    # Phase 1: Active-lag guided pair search (45% of time)
    dl1 = t0 + budget_s * 0.40
    h_cur = best_h.copy()
    C_cur = best_C
    h_cur, C_cur = _active_pair_search(h_cur, C_cur, dl1, 5000)
    if C_cur < best_C:
        best_h, best_C = h_cur.copy(), C_cur
        best_n = len(best_h)
    
    # Phase 2: Random pair polish (20%)
    dl2 = t0 + budget_s * 0.60
    h_cur = best_h.copy()
    C_cur = best_C
    h_cur, C_cur, _ = _try_improve(h_cur, C_cur, dl2, 5000, 20)
    if C_cur < best_C:
        best_h, best_C = h_cur.copy(), C_cur
        best_n = len(best_h)
    
    # Phase 3: Simulated annealing (20%)
    dl3 = t0 + budget_s * 0.80
    h_cur = best_h.copy()
    C_cur = best_C
    nn = len(h_cur)
    T = max(5e-5, 0.005 * C_cur)
    cool = 0.998
    accept_cnt = 0
    
    while time.time() < dl3:
        p = int(rng.randint(nn))
        q = int(rng.randint(nn))
        while q == p:
            q = int(rng.randint(nn))
        
        hp, hq = h_cur[p], h_cur[q]
        lo = max(hp - 1.0, -hq)
        hi = min(hp, 1.0 - hq)
        if hi - lo < 1e-12:
            continue
        
        t = float(rng.uniform(lo, hi))
        h_cur[p] = hp - t
        h_cur[q] = hq + t
        v_new = _evaluate(h_cur)
        delta = v_new - C_cur
        
        if delta <= 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            C_cur = v_new
            accept_cnt += 1
            if C_cur < best_C:
                best_h = h_cur.copy()
                best_C = C_cur
                best_n = len(best_h)
        else:
            h_cur[p] = hp
            h_cur[q] = hq
        
        T *= cool
    
    # Phase 4: Final polish on best (12%)
    dl4 = t0 + budget_s * 0.92
    h_cur = best_h.copy()
    C_cur = best_C
    
    # Exhaustive-ish pair search
    nn = len(h_cur)
    if nn <= 80:
        all_pairs = [(i, j) for i in range(nn) for j in range(i + 1, nn)]
        rng.shuffle(all_pairs)
        for p, q in all_pairs[:300]:
            if time.time() > dl4:
                break
            h_new, v_new = _pair_search(h_cur, p, q, 20, dl4)
            if v_new < C_cur - 1e-12:
                h_cur = h_new
                C_cur = v_new
    else:
        h_cur, C_cur, _ = _try_improve(h_cur, C_cur, dl4, 2000, 16)
    
    if C_cur < best_C:
        best_h, best_C = h_cur.copy(), C_cur
        best_n = len(best_h)
    
    # Phase 5: Try nearby resolutions (remaining time)
    dl5 = deadline
    remaining = dl5 - time.time()
    if remaining > 40:
        base_n = len(best_h)
        n_list = sorted(set([
            base_n - 2, base_n - 1, base_n + 1, base_n + 2, base_n + 3,
            base_n - 3, base_n + 4,
        ] & {nn for nn in range(24, 160)}))
        # Add some round numbers
        for cand in [40, 44, 48, 50, 52, 56, 60, 64, 72, 80, 88]:
            if 24 <= cand <= 160 and cand != base_n:
                n_list.append(cand)
        n_list = sorted(set(n_list))
        
        per_n_budget = remaining / max(len(n_list), 1)
        
        for nn_new in n_list:
            if time.time() > dl5:
                break
            t_n = min(dl5, time.time() + per_n_budget)
            
            # Resample
            x_old = (np.arange(len(best_h)) + 0.5) * 2.0 / len(best_h)
            x_new = (np.arange(nn_new) + 0.5) * 2.0 / nn_new
            h_r = np.interp(x_new, x_old, best_h, left=best_h[0], right=best_h[-1])
            h_r = _project(h_r, nn_new / 2.0)
            
            # Also try binary construction
            h_b = _make_binary(nn_new, rng)
            
            for h_start in [h_r, h_b]:
                if time.time() > t_n:
                    break
                C_s = _evaluate(h_start)
                if C_s >= best_C:
                    continue
                # Polish
                h_p = h_start.copy()
                C_p = C_s
                h_p, C_p, _ = _try_improve(h_p, C_p, t_n, 300, 16)
                if C_p < best_C:
                    best_h = h_p.copy()
                    best_C = C_p
                    best_n = len(best_h)
    
    # Final validation
    final_h = best_h.copy()
    final_n = len(final_h)
    final_h = np.clip(final_h, 0.0, 1.0)
    res = final_n / 2.0 - final_h.sum()
    if abs(res) > 1e-9:
        if res > 0:
            idx = int(np.argmax(1.0 - final_h))
            final_h[idx] = min(1.0, final_h[idx] + res)
        else:
            idx = int(np.argmax(final_h))
            final_h[idx] = max(0.0, final_h[idx] + res)
    final_h = np.clip(final_h, 0.0, 1.0)
    
    final_C = _evaluate(final_h)
    
    return (final_h, final_C, int(final_n))
