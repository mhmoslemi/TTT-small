def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) < 1e-12:
        return h
    lo, hi = float(h.min()) - 5.0, float(h.max()) + 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - h.sum()
    if resid > 1e-12:
        for i in range(n):
            a = min(resid, 1.0 - h[i])
            if a > 0:
                h[i] += a
                resid -= a
            if resid < 1e-12:
                break
    elif resid < -1e-12:
        for i in range(n):
            r = max(resid, -h[i])
            if r < 0:
                h[i] += r
                resid -= r
            if -resid < 1e-12:
                break
    return h
