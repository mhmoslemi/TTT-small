    n = len(h)

def exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def softmax_objective_and_grad(h, tau):
    n = len(h)
    c_vals = compute_c_values(h)
    m = np.max(c_vals)
    shifted = (c_vals - m) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    obj = m + tau * log_sum
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros(n)
    dx = 2.0 / n
    center = n - 1
    
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        l = k - center
        if l == 0:
            grad += weights[k] * dx * (1.0 - 2.0 * h)
        elif l > 0:
            # (1-h[j+l]) for j in [0, n-1-l]
            grad[:n-l] += weights[k] * dx * (1.0 - h[l:])
            # -h[j-l] for j in [l, n-1]
            grad[l:] -= weights[k] * dx * h[:n-l]
        else:
            p = -l
            # +1 for j in [p, n-1]
            grad[p:] += weights[k] * dx
            # -h[j+p] for j in [0, n-1-p]
            grad[:n-p] -= weights[k] * dx * h[p:]
            # -h[j-p] for j in [p, n-1]
            grad[p:] -= weights[k] * dx * h[:n-p]
    
    return obj, grad

def project_feasible(h):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection to fix the sum
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # If sum too high, need to decrease; if too low, increase
    # Use bisection on a shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        test = np.clip(h + mid, 0.0, 1.0)
        ts = np.sum(test)
        if ts > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # Find an element with slack
        if diff > 0:
            # Need to increase sum
            slacks = 1.0 - result
            order = np.argsort(-slacks)
        else:
            # Need to decrease sum
            slacks = result
            order = np.argsort(-slacks)

                add = min(room, diff)
                result[idx] += add
                diff -= add

                sub = min(room, -diff)

import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def single_c_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    l = k - center
    grad = np.zeros(n)
    if l == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif l > 0:
        grad[:n-l] += (1.0 - h[l:]) * dx
        grad[l:n] -= h[:n-l] * dx
    else:
        p = -l
        grad[p:n] += dx
        grad[:n-p] -= h[p:] * dx
        grad[p:n] -= h[:n-p] * dx
    return grad

def softmax_obj(x, tau):
    c_vals = compute_c_values(x)
    m = np.max(c_vals)
    shifted = np.clip((c_vals - m) / tau, -50.0, 50.0)
    return m + tau * float(np.log(np.sum(np.exp(shifted))))

    n = len(x)
    c_vals = compute_c_values(x)
    m = np.max(c_vals)
    shifted = np.clip((c_vals - m) / tau, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    dx = 2.0 / n
    center = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        l = k - center
        if l == 0:
            grad += w[k] * (1.0 - 2.0 * x) * dx
        elif l > 0:
            grad[:n-l] += w[k] * (1.0 - x[l:]) * dx
            grad[l:n] -= w[k] * x[:n-l] * dx
        else:
            p = -l
            grad[p:n] += w[k] * dx
            grad[:n-p] -= w[k] * x[p:] * dx
            grad[p:n] -= w[k] * x[:n-p] * dx
    return grad

def mass_eq_fun(x):
    return float(np.sum(x) - len(x) / 2.0)

def mass_eq_jac(x):
    return np.ones(len(x))

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-15:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        test = np.clip(h + mid, 0.0, 1.0)
        ts = np.sum(test)
        if ts > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        for _ in range(n):
            if diff > 0:
                idx = int(np.argmax(1.0 - h))
                add = min(1.0 - h[idx], diff)
                if add <= 0:
                    break
                h[idx] += add
                diff -= add
            else:
                idx = int(np.argmax(h))
                sub = min(h[idx], -diff)
                if sub <= 0:
                    break
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-15:
                break
    return h

def epig_obj(x, n):
    return x[n]

def epig_obj_jac(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def epig_ineq_fun(x, active_k, n):
    h = x[:n]
    t = x[n]
    c_vals = compute_c_values(h)
    return t - c_vals[active_k]

def epig_ineq_jac(x, active_k, n):
    h = x[:n]
    jac = np.zeros((len(active_k), n + 1))
    for i, k in enumerate(active_k):
        jac[i, :n] = -single_c_grad(h, k)
        jac[i, n] = 1.0
    return jac

def epig_mass_fun(x):
    n = len(x) - 1
    return float(np.sum(x[:n]) - n / 2.0)

def epig_mass_jac(x):
    n = len(x) - 1
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    x_src = np.linspace(0.0, 2.0, n_src + 1)
    cum = np.zeros(n_src + 1)
    for i in range(n_src):
        cum[i+1] = cum[i] + h_src[i] * dx_src
    x_tgt = np.linspace(0.0, 2.0, n_tgt + 1)
    areas = np.zeros(n_tgt)
    for i in range(n_tgt):
        areas[i] = float(np.interp(x_tgt[i+1], x_src, cum)) - float(np.interp(x_tgt[i], x_src, cum))
    h_tgt = areas / dx_tgt
    return project_feasible(h_tgt)

def targeted_pair_moves(h, max_iters, rng, current_score):
    n = len(h)
    h = h.copy()
    score = current_score
    for _ in range(max_iters):
        c_vals = compute_c_values(h)
        k_max = int(np.argmax(c_vals))
        l = k_max - (n - 1)
        moved = False
        if l > 0:
            best_delta = 0.0
            best_i = -1
            for i in range(n - l):
                if h[i] > h[i+l]:
                    d = min(0.3 * (h[i] - h[i+l]), h[i] - h[i+l])
                    if d > best_delta:
                        best_delta = d
                        best_i = i
            if best_i >= 0 and best_delta > 1e-6:
                delta = best_delta * rng.uniform(0.3, 1.0)
                h[best_i] -= delta
                h[best_i + l] += delta
                new_score = exact_c5(h)
                if new_score < score - 1e-12:
                    score = new_score
                    moved = True
                else:
                    h[best_i] += delta
                    h[best_i + l] -= delta
        elif l < 0:
            p = -l
            best_delta = 0.0
            best_i = -1
            for i in range(n - p):
                if h[i+p] > h[i]:
                    d = min(0.3 * (h[i+p] - h[i]), h[i+p] - h[i])
                    if d > best_delta:
                        best_delta = d
                        best_i = i
            if best_i >= 0 and best_delta > 1e-6:
                delta = best_delta * rng.uniform(0.3, 1.0)
                h[best_i + p] -= delta
                h[best_i] += delta
                new_score = exact_c5(h)
                if new_score < score - 1e-12:
                    score = new_score
                    moved = True
                else:
                    h[best_i + p] += delta
                    h[best_i] -= delta
        elif l == 0:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            if h[i] > 0.3 and h[i] < 0.7:
                direction = 1.0 if h[i] >= 0.5 else -1.0
                delta = 0.05 * rng.uniform(0.5, 1.0) * direction
                h_old = h[i].copy()
                h[i] = min(1.0, max(0.0, h[i] + delta))
                adj = h_old - h[i]
                h[j] = min(1.0, max(0.0, h[j] - adj))
                h = project_feasible(h)
                new_score = exact_c5(h)
                if new_score < score - 1e-12:
                    score = new_score
                    moved = True
                else:
                    h[i] = h_old + adj if h[i] != h_old else h_old
                    h = project_feasible(h)
        if not moved:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta = rng.uniform(0.01, 0.1)
            if h[i] >= h[j]:
                if h[i] - delta < 0 or h[j] + delta > 1:
                    continue
                h[i] -= delta
                h[j] += delta
            else:
                if h[j] - delta < 0 or h[i] + delta > 1:
                    continue
                h[j] -= delta
                h[i] += delta
            new_score = exact_c5(h)
            if new_score < score - 1e-12:
                score = new_score
            else:
                if h[i - 1 + 1] is not None:
                    pass
                h = h  # revert
                # Actually need to revert properly
                # Let me redo this
    return h, score

def targeted_pair_moves_v2(h, max_iters, rng, current_score):
    n = len(h)
    h = h.copy()
    score = current_score
    stuck_count = 0
    for it in range(max_iters):
        c_vals = compute_c_values(h)
        k_max = int(np.argmax(c_vals))
        l = k_max - (n - 1)
        improved = False
        if l > 0:
            for i in range(n - l):
                if h[i] > h[i+l] + 0.05:
                    delta = min(0.2 * (h[i] - h[i+l]), h[i] - h[i+l]) * rng.uniform(0.3, 1.0)
                    old_i, old_j = h[i], h[i+l]
                    h[i] -= delta
                    h[i+l] += delta
                    ns = exact_c5(h)
                    if ns < score - 1e-12:
                        score = ns
                        improved = True
                        break
                    else:
                        h[i], h[i+l] = old_i, old_j
        elif l < 0:
            p = -l
            for i in range(n - p):
                if h[i+p] > h[i] + 0.05:
                    delta = min(0.2 * (h[i+p] - h[i]), h[i+p] - h[i]) * rng.uniform(0.3, 1.0)
                    old_i, old_j = h[i+p], h[i]
                    h[i+p] -= delta
                    h[i] += delta
                    ns = exact_c5(h)
                    if ns < score - 1e-12:
                        score = ns
                        improved = True
                        break
                    else:
                        h[i+p], h[i] = old_i, old_j
        if not improved:
            stuck_count += 1
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j:
                delta = rng.uniform(0.005, 0.08)
                if h[i] >= delta + 1e-6:
                    h[i] -= delta
                    h[j] = min(1.0, h[j] + delta)
                    h = project_feasible(h)
                    ns = exact_c5(h)
                    if ns < score - 1e-12:
                        score = ns
                        stuck_count = 0
                    else:
                        h[i] += delta
                        h[j] -= delta
                        h = project_feasible(h)
                elif h[j] >= delta + 1e-6:
                    h[j] -= delta
                    h[i] = min(1.0, h[i] + delta)
                    h = project_feasible(h)
                    ns = exact_c5(h)
                    if ns < score - 1e-12:
                        score = ns
                        stuck_count = 0
                    else:
                        h[j] += delta
                        h[i] -= delta
                        h = project_feasible(h)
        else:
            stuck_count = 0
        if stuck_count > 200:
            break
    return h, score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s * 0.92

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h0 = np.ones(45) * 0.5

    h0 = project_feasible(h0)
    n = len(h0)
    best_h = h0.copy()
    best_score = exact_c5(best_h)

    bounds = [(0.0, 1.0)] for _ in range(n)
    constraints_smooth = [{'type': 'eq', 'fun': mass_eq_fun, 'jac': mass_eq_jac}]

    # Stage 1: Smoothed SLSQP with decreasing tau
    taus = [0.08, 0.04, 0.02, 0.008, 0.003]
    for tau in taus:
        if time.time() > deadline:
            break
        try:
            res = minimize(softmax_obj, best_h, jac=softmax_grad, args=(tau,),
                          bounds=bounds, constraints=constraints_smooth,
                          method='SLSQP', options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                candidate = project_feasible(res.x)
                sc = exact_c5(candidate)
                if sc < best_score - 1e-11:
                    best_h = candidate
                    best_score = sc
        except Exception:
            pass

    # Stage 2: Active-set epigraph
    for epig_iter in range(6):
        if time.time() > deadline:
            break
        c_vals = compute_c_values(best_h)
        n_active = min(2 * n - 1, max(12, int(0.35 * (2 * n - 1))))
        top_idx = np.argsort(c_vals)[-n_active:]
        x0 = np.append(best_h, best_score + 1e-6)
        epig_bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
        constraints_epig = [
            {'type': 'eq', 'fun': epig_mass_fun, 'jac': epig_mass_jac},
            {'type': 'ineq', 'fun': epig_ineq_fun, 'jac': epig_ineq_jac, 'args': (top_idx, n)}
        ]
        try:
            res = minimize(epig_obj, x0, jac=epig_obj_jac, args=(n,),
                          bounds=epig_bounds, constraints=constraints_epig,
                          method='SLSQP', options={'maxiter': 400, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                candidate = project_feasible(res.x[:n])
                sc = exact_c5(candidate)
                if sc < best_score - 1e-11:
                    best_h = candidate
                    best_score = sc
        except Exception:
            pass

    # Stage 3: Targeted pair moves
    if time.time() < deadline:
        moves_done = 0
        while time.time() < deadline and moves_done < 3000:
            h_new, sc_new = targeted_pair_moves_v2(best_h, 200, rng, best_score)
            moves_done += 200
            if sc_new < best_score - 1e-11:
                best_h = project_feasible(h_new)
                best_score = exact_c5(best_h)
            else:
                break

    # Stage 4: Second pass epigraph (tighter)
    if time.time() < deadline - 30:
        for epig_iter in range(4):
            if time.time() > deadline - 10:
                break
            c_vals = compute_c_values(best_h)
            n_active = min(2 * n - 1, max(15, int(0.5 * (2 * n - 1))))
            top_idx = np.argsort(c_vals)[-n_active:]
            x0 = np.append(best_h, best_score + 1e-7)
            epig_bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
            constraints_epig = [
                {'type': 'eq', 'fun': epig_mass_fun, 'jac': epig_mass_jac},
                {'type': 'ineq', 'fun': epig_ineq_fun, 'jac': epig_ineq_jac, 'args': (top_idx, n)}
            ]
            try:
                res = minimize(epig_obj, x0, jac=epig_obj_jac, args=(n,),
                              bounds=epig_bounds, constraints=constraints_epig,
                              method='SLSQP', options={'maxiter': 600, 'ftol': 1e-15})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    candidate = project_feasible(res.x[:n])
                    sc = exact_c5(candidate)
                    if sc < best_score - 1e-11:
                        best_h = candidate
                        best_score = sc
            except Exception:
                pass

    # Stage 5: Final pair moves polish
    if time.time() < deadline - 5:
        h_new, sc_new = targeted_pair_moves_v2(best_h, 500, rng, best_score)
        if sc_new < best_score - 1e-11:
            best_h = project_feasible(h_new)
            best_score = exact_c5(best_h)

    # Stage 6: Resolution transfer (if time allows and n < 90)
    if time.time() < deadline - 120 and n <= 60:
        n_tgt = min(90, n * 2)
        if n_tgt != n:
            h_transfer = transfer_resolution(best_h, n_tgt)
            sc_transfer = exact_c5(h_transfer)
            if sc_transfer < best_score + 0.01:
                n_t = n_tgt
                bounds_t = [(0.0, 1.0)] * n_t
                cons_t = [{'type': 'eq', 'fun': mass_eq_fun, 'jac': mass_eq_jac}]
                for tau in [0.04, 0.015, 0.005]:
                    if time.time() > deadline - 30:
                        break
                    try:
                        res = minimize(softmax_obj, h_transfer, jac=softmax_grad, args=(tau,),
                                      bounds=bounds_t, constraints=cons_t,
                                      method='SLSQP', options={'maxiter': 300, 'ftol': 1e-14})
