    dx = 2.0 / n

def compute_gradient(h, d):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # For indices j where j+d is valid (0 <= j+d < n): add (1-h[j+d])
    if d >= 0:

        # For indices j where j-d is valid (0 <= j-d < n): subtract h[j-d]

            best_h = h.copy()
            no_improve = 0
            stagnation_level = 0
        else:
            no_improve += 1

        if no_improve > 300:
            stagnation_level += 1
            no_improve = 0
            if stagnation_level >= 4:
                break
            pert_size = 0.02 * (stagnation_level + 1)
            h = _project(h + pert_size * rng.randn(n), target)
            alpha = max(5.0, alpha * 0.7)
            continue

        if no_improve > 100:
            h = _project(h + 0.03 * rng.randn(n), target)
            no_improve = 0
            continue

        nh = n - 1
        improved = False

        # Try LP with various active set sizes
        for k_active in [3, 5, 8, 12, 20]:
            if not time_ok():
                break
            sorted_idx = np.argsort(c_vals)[::-1]
            active = sorted_idx[:k_active]
            direction, t_val = _lp_solve(h, active, n)
            if direction is not None and t_val < -1e-10:
                h_new, c_new = _line_search(h, direction, target)
                if c_new < c_max - 1e-14:
                    h = h_new
                    improved = True
                    break

        if not improved:
            # Smooth gradient approach
            direction = _smooth_direction(h, c_vals, alpha)
            h_new, c_new = _line_search(h, direction, target)
            if c_new < c_max - 1e-14:
                h = h_new
                improved = True

        if not improved:
            # Try negative smooth direction (rarely helps but occasionally)
            direction = _smooth_direction(h, c_vals, alpha)
            h_new, c_new = _line_search(h, -direction * 0.3, target)
            if c_new < c_max - 1e-14:
                h = h_new
                improved = True

        if not improved:
            # Coordinate-ish moves: swap mass between pairs
            i1 = rng.randint(0, n)
            i2 = rng.randint(0, n)
            if i1 != i2:
                delta = 0.05 * rng.choice([-1.0, 1.0])
                h_trial = h.copy()
                h_trial[i1] += delta
                h_trial[i2] -= delta
                if 0 <= h_trial[i1] <= 1 and 0 <= h_trial[i2] <= 1:
                    c_trial = _c5(h_trial)
                    if c_trial < c_max:
                        h = h_trial
                        improved = True

        if not improved:
            h = _project(h + 0.01 * rng.randn(n), target)

    # Polish phase: very fine LP optimization
    if time_ok():
        h = best_h.copy()
        target = len(h) / 2.0
        h = _project(h, target)
        polish_time = min(60.0, max(0.0, budget_s - 5.0 - elapsed()))
        t_polish_end = time.time() + polish_time
        alpha = 80.0
        for _ in range(2000):
            if time.time() > t_polish_end:
                break
            c_vals = _compute_c(h)
            c_max = float(np.max(c_vals))
            if c_max < best_c5 - 1e-14:
                best_c5 = c_max
                best_h = h.copy()
            sorted_idx = np.argsort(c_vals)[::-1]
            for k_a in [2, 3, 4, 6]:
                active = sorted_idx[:k_a]
                direction, t_val = _lp_solve(h, active, n, trust=0.2)
                if direction is not None and t_val < -1e-10:
                    h_new, c_new = _line_search(h, direction, target)
                    if c_new < c_max - 1e-14:
                        h = h_new
                        break
            else:
                direction = _smooth_direction(h, c_vals, alpha)
                h_new, c_new = _line_search(h, direction, target)
                if c_new < c_max - 1e-14:
                    h = h_new
                else:
                    break

    # Try a few different resolutions
    if time_ok():
        for n_try in [n - 3, n + 1, n + 3, n + 5]:
            if not time_ok():
                break
            if n_try < 20:
                continue
            h_try = _resample(best_h, n_try)
            c_try = _c5(h_try)
            if c_try < best_c5:
                # Quick local optimization
                target_t = n_try / 2.0
                sub_end = time.time() + min(30.0, max(0.0, budget_s - 5.0 - elapsed()))
                for _ in range(200):
                    if time.time() > sub_end:
                        break
                    c_vals_t = _compute_c(h_try)
                    c_max_t = float(np.max(c_vals_t))
                    if c_max_t < best_c5:
                        best_c5 = c_max_t
                        best_h = h_try.copy()
                    sorted_idx_t = np.argsort(c_vals_t)[::-1]
                    for k_a in [3, 5, 8]:
                        active_t = sorted_idx_t[:k_a]
                        direction_t, tv = _lp_solve(h_try, active_t, n_try)
                        if direction_t is not None and tv < -1e-10:
                            h_try, c_new_t = _line_search(h_try, direction_t, target_t)
                            if c_new_t < c_max_t:
                                break
                    else:
                        direction_t = _smooth_direction(h_try, c_vals_t, 50.0)
                        h_try, c_new_t = _line_search(h_try, direction_t, target_t)
                c_final = _c5(h_try)
                if c_final < best_c5:
                    best_c5 = c_final
                    best_h = h_try.copy()

    # Final validation
    final_c = _c5(best_h)
    if final_c > best_c5 + 1e-10:
        best_c5 = final_c
    return (best_h, float(best_c5), len(best_h))
