```python
import numpy as np
from scipy.optimize import minimize

def make_feasible(z, target=None):
    """Project z to feasible set: sum=n/2, 0<=h<=1 using bisection on shift."""
    z = np.asarray(z, dtype=float).ravel()
    n = len(z)
    if target is None:
        target = n / 2.0
    
    # Bisection to find lambda such that sum(clip(z+lambda, 0, 1)) = target
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(z + 0.5*(lo+hi), 0.0, 1.0)
    
    # Fix residual
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            # Need to add; find index with most room
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                if room[i] > 1e-12:
                    delta = min(residual, room[i])
                    h[i] += delta
                    residual -= delta
                    if abs(residual) < 1e-12:
                        break
        else:
            # Need to subtract
            idx = np.argsort(-h)
            for i in idx:
                if h[i] > 1e-12:
                    delta = min(-residual, h[i])
                    h[i] -= delta
                    residual += delta
                    if abs(residual) < 1e-12:
                        break
    
    return h

def official_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def manual_values(h):
    """Compute all lag correlations manually. Returns array of shape (2n-1,)"""
    n = len(h)
    vals = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        vals[r] = np.sum(h[i_start:i_end] * (1.0 - h[i_start:i_end - k]))
    return vals

def manual_grad(h):
    """Compute gradient matrix of shape (2n-1, n).
    grad[r] is the gradient of the r-th correlation w.r.t. h."""
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        for i in range(i_start, i_end):
            # term: h[i+k] * (1-h[i])
            # d/dh[i+k] = (1-h[i])
            # d/dh[i] = -h[i+k]
            grad[r, i + k] += (1.0 - h[i])
            grad[r, i] -= h[i + k]
    return grad
```

```python
    grad = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end <= i_start:
            continue

        first_idx = i + k  # indices for h[i+k]
        second_idx = i      # indices for h[i]
        np.add.at(grad[r], first_idx, 1.0 - h[i])
        np.add.at(grad[r], second_idx, -h[i + k])
    return grad
```

```python
def smooth_obj_and_grad(h, beta):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    vals = manual_values(h) * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # Stable log-sum-exp
    max_shifted = shifted.max()
    log_sum = cmax + (max_shifted + np.log(np.exp(shifted - max_shifted).sum())) / beta
    weights = np.exp(shifted - max_shifted)
    weight_sum = weights.sum()
    if weight_sum < 1e-300:
        weights = np.ones(len(vals)) / len(vals)
    else:
        weights = weights / weight_sum
    grad_matrix = manual_grad(h) * dx
    grad = weights @ grad_matrix
    return log_sum, grad
```

Actually, let me reconsider the smooth_fun. The log-sum-exp formulation:

Phi_beta(h) = c_max + (1/beta) * log(sum_k exp(beta * (C_k - c_max)))

where c_max = max_k C_k.

This is always >= c_max, and approaches c_max as beta -> infinity.

The gradient is:
grad Phi = (1/beta) * sum_k exp(beta*(C_k - c_max)) / sum_j exp(beta*(C_j - c_max)) * grad C_k
= sum_k w_k * grad C_k

where w_k = softmax(beta * (C_k - c_max)).

This is correct.

Now, one subtlety: when I pass this to SLSQP, the function value is slightly above the true max (by O(1/beta)), but the gradient points in the right direction. After the SLSQP converges, I evaluate the true max and that's what matters.

```python
import numpy as np
from scipy.optimize import minimize
import time


def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel()
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                if room[i] > 1e-14:
                    d = min(resid, room[i])
                    h[i] += d
                    resid -= d
                    if abs(resid) < 1e-13:
                        break
        else:
            order = np.argsort(-h)
            for i in order:
                if h[i] > 1e-14:
                    d = min(-resid, h[i])
                    h[i] -= d
                    resid += d
                    if abs(resid) < 1e-13:
                        break
    return h


def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def manual_vals(h):
    n = len(h)
    m = 1.0 - h
    vals = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i0 = max(0, -k)
        i1 = min(n, n - k)
        if i1 > i0:
            idx = np.arange(i0, i1)
            vals[r] = np.dot(h[idx + k], m[idx])
    return vals


def manual_grad_mat(h):
    n = len(h)
    m = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i0 = max(0, -k)
        i1 = min(n, n - k)
        if i1 <= i0:
            continue
        idx = np.arange(i0, i1)
        np.add.at(G[r], idx + k, m[idx])
        np.add.at(G[r], idx, -h[idx + k])
    return G


def _smooth_obj(z, beta):
    v = manual_vals(z) * (2.0 / len(z))
    c = v.max()
    s = beta * (v - c)
    ms = s.max()
    return c + (ms + np.log(np.exp(s - ms).sum())) / beta


def _smooth_grad(z, beta):
    n = len(z)
    dx = 2.0 / n
    v = manual_vals(z) * dx
    c = v.max()
    s = beta * (v - c)
    ms = s.max()
    w = np.exp(s - ms)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(v)) / len(v)
    else:
        w /= ws
    return w @ (manual_grad_mat(z) * dx)


def _eq_obj(z):
    return z.sum() - len(z) / 2.0


def _eq_grad(z):
    return np.ones(len(z))


def _direct_obj(z):
    return z[-1]


def _direct_obj_grad(z):
    e = np.zeros(len(z))
    e[-1] = 1.0
    return e


def _direct_ineq(z):
    n = len(z) - 1
    return z[-1] - manual_vals(z[:n]) * (2.0 / n)


def _direct_ineq_grad(z):
    n = len(z) - 1
    rows = 2 * n - 1
    J = np.zeros((rows, n + 1))
    J[:, :n] = -manual_grad_mat(z[:n]) * (2.0 / n)
    J[:, n] = 1.0
    return J


def _direct_eq(z):
    n = len(z) - 1
    return z[:n].sum() - n / 2.0


def _direct_eq_grad(z):
    n = len(z) - 1
    e = np.zeros(n + 1)
    e[:n] = 1.0
    return e


def interp_h(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return make_feasible(h_src.copy())
    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xt = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    return make_feasible(np.interp(xt, xs, h_src))


def anti_h(h):
    return make_feasible(1.0 - h[::-1])


def do_smooth(h0, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_obj, 'jac': _eq_grad}]
    try:
        res = minimize(_smooth_obj, h0, jac=_smooth_grad,
                       args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x)
    except Exception:
        return h0


def do_direct(h0, maxiter):
    n = len(h0)
    c0 = official_score(h0)
    z0 = np.zeros(n + 1)
    z0[:n] = h0
    z0[n] = c0 + 1e-5
    bounds = [(0.0, 1.0)] * n + [(1e-9, c0 + 0.15)]
    cons = [
        {'type': 'ineq', 'fun': _direct_ineq, 'jac': _direct_ineq_grad},
        {'type': 'eq', 'fun': _direct_eq, 'jac': _direct_eq_grad}
    ]
    try:
        res = minimize(_direct_obj, z0, jac=_direct_obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x[:n])
    except Exception:
        pass
    return h0
```
