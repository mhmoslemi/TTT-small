import numpy as np
import time
from scipy.optimize import linprog

def project_box_sum(y, target_sum):
    """Project y onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    
    def phi(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    # phi(lo) >= n (all 1s), phi(hi) <= 0 (all 0s)
    # We want phi(lam) = target_sum
    for _ in range(100):
        mid = (lo + hi) / 2
        if phi(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    x = np.clip(y - lam, 0, 1)
    
    # Fix any residual
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            # Add to coordinates with most room
            room = 1.0 - x
            total_room = room.sum()
            if total_room > 0:
                x += residual * room / total_room
        else:
            # Remove from coordinates with most mass
            mass = x.copy()
            total_mass = mass.sum()
            if total_mass > 0:
                x += residual * mass / total_mass
    
    x = np.clip(x, 0, 1)
    return x

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def gradient_of_lag(h, k, n):
    """
    Compute gradient of f_k with respect to h.
    k is the lag (can be negative).
    f_k = sum_n h[n+k]*(1-h[n]) for k >= 0 (valid n)
    
    Using numpy correlate convention:
    output[n-1+k] = sum_n h[n+k]*(1-h[n])
    """
    grad = np.zeros(n)
    k = int(k)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        # f_k = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
        # ∂f_k/∂h[j] = (1-h[j-k]) [if j >= k] + (-h[j+k]) [if j <= n-1-k]
        if k < n:
            # Part 1: (1-h[j-k]) for j = k, ..., n-1
            grad[k:] += (1.0 - h[:n-k])
            # Part 2: -h[j+k] for j = 0, ..., n-1-k
            grad[:n-k] -= h[k:]
    else:
        m = -k
        # f_{-m} = sum_{n=0}^{n-1-m} h[n]*(1-h[n+m])
        # ∂f_{-m}/∂h[j] = (1-h[j+m]) [if j <= n-1-m] + (-h[j-m]) [if j >= m]
        if m < n:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    
    return grad
