```python
def lp_direction(h, active_indices, n):
    """Find direction d that decreases all active constraints."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    m = len(active_indices)
    
    # Compute the Jacobian rows for active constraints
    J = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        shift = k - (n - 1)
        if shift >= 0:
            for j in range(n - shift):
                J[i, j] += dx * (1 - h[j + shift])
                J[i, j + shift] += -dx * h[j]
        else:
            s = -shift
            for j in range(s, n):
                J[i, j] += dx * (1 - h[j - s])
                J[i, j - s] += -dx * h[j]
    
    # LP: max rho s.t. J @ d <= -rho, sum(d) = 0, 
    #     -(min(h, 1-h)*0.1) <= d <= (min(h, 1-h)*0.1)
    # Variables: d[0..n-1], rho
    # Objective: -rho (minimize)
    # Constraints:
    #   J @ d + rho <= 0  (m constraints)
    #   sum(d) = 0  (1 equality)
    #   Bounds on d
    
    c = np.zeros(n + 1)
    c[-1] = -1  # minimize -rho = maximize rho
    
    # Inequality constraints: J @ d + rho <= 0
    A_ub = np.column_stack([J, np.ones(m)])
    b_ub = np.zeros(m)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = [0]
    
    # Bounds
    bounds = []
    for i in range(n):
        # Allow d[i] to move h[i] within [0,1]
        lo = -h[i]  # h[i] + d[i] >= 0
        hi = 1 - h[i]  # h[i] + d[i] <= 1
        bounds.append((lo, hi))
    bounds.append((-np.inf, np.inf))  # rho unbounded (but will be bounded by constraints)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[-1]
    return None, 0
```

```python
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def objective_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        lo, hi = -3.0, 3.0
        for _ in range(100):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)

def active_set(f_vec, C, tol=1e-6, max_count=8):
    """Find indices where f_vec is within tol of C."""
    active = np.where(f_vec >= C - tol)[0]
    if len(active) > max_count:
        # Take the top max_count
        idx = np.argsort(f_vec)[::-1][:max_count]
        active = idx
    return active

def compute_jacobian_rows(h, active_indices, n):
    """Compute Jacobian rows for active constraints."""
    dx = 2.0 / n
    m = len(active_indices)
    J = np.zeros((m, n))
    for li, k in enumerate(active_indices):
        shift = k - (n - 1)
        if shift >= 0:
            for j in range(n - shift):
                J[li, j] += (1.0 - h[j + shift])
                J[li, j + shift] += -h[j]
        else:
            s = -shift
            for j in range(s, n):
                J[li, j] += (1.0 - h[j - s])
                J[li, j - s] += -h[j]
    J *= dx
    return J

def lp_descent_direction(h, J, n):
    """Solve LP to find direction that decreases all active constraints."""
    m = J.shape[0]
    # Variables: d[0..n-1], rho
    # Minimize -rho
    # s.t. J @ d + rho <= 0
    #      sum(d) = 0
    #      -h[i] <= d[i] <= 1-h[i]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    A_ub = np.column_stack([J, np.ones(m)])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((-np.inf, np.inf))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, 0.0

def make_binary(h, n):
    """Create binary version of h with exactly n/2 ones."""
    b = (h >= 0.5).astype(float)
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        # Need to turn off some ones
        zeros = np.where(b == 1)[0]
        # Turn off the ones with smallest h values
        order = np.argsort(h[zeros])
        b[zeros[order[:n_ones - target]]] = 0
    elif n_ones < target:
        ones = np.where(b == 0)[0]
        order = np.argsort(h[ones])[::-1]
        b[ones[order[:target - n_ones]]] = 1
    return b

def resample(h_old, n_new):
    """Resample h from old size to new size, preserving integral."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
```

```python
def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def objective_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def jacobian_active(h, active, n):
    dx = 2.0 / n
    m = len(active)
    J = np.zeros((m, n))
    for li, k in enumerate(active):
        shift = k - (n - 1)
        if shift >= 0:
            nr = n - shift
            J[li, :nr] += (1.0 - h[shift:shift+nr])
            J[li, shift:] += -h[:nr]
        else:
            s = -shift
            nr = n - s
            J[li, s:] += (1.0 - h[:nr])
            J[li, :nr] += -h[s:s+nr]
    J *= dx
    return J

def lp_direction(h, J, n):
    m = J.shape[0]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.column_stack([J, np.ones(m)])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((-1.0, 1.0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], float(max(res.x[-1], 0))
    except Exception:
        pass
    return None, 0.0
```

```python
def binary_objective_fast(b, n):
    """Compute objective vector for binary b."""
    dx = 2.0 / n
    return np.correlate(b, 1.0 - b, mode='full') * dx

def binary_local_search(b, n, max_rounds=100, time_limit=5.0):
    """Local search for binary sequence."""
    t0 = time.time()
    f_vec = binary_objective_fast(b, n)
    best_c = float(np.max(f_vec))
    
    for round in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        
        # Try all single-bit moves
        best_move = None
        best_delta = 0
        for i in ones:
            for j in zeros:
                # Compute change in max objective
                # Fast: update f_vec for this move
                new_f = f_vec.copy()
                # Apply the move's effect
                for k in range(2*n-1):
                    shift = k - (n-1)
                    delta = 0
                    if 0 <= i + shift < n:
                        delta -= (1 - b[i+shift])
                    if 0 <= i - shift < n:
                        delta += b[i-shift]
                    if 0 <= j + shift < n:
                        delta += (1 - b[j+shift])
                    if 0 <= j - shift < n:
                        delta -= b[j-shift]
                    new_f[k] += delta * (2.0/n)
                
                c_new = float(np.max(new_f))
                delta_c = c_new - best_c
                if delta_c < best_delta - 1e-12:
                    best_delta = delta_c
                    best_move = (i, j, new_f.copy())
        
        if best_move is not None:
            i, j, new_f = best_move
            b[i] = 0
            b[j] = 1
            f_vec = new_f
            best_c = best_c + best_delta
            improved = True
        
        if not improved:
            break
    
    return b, best_c
```

```python
def precompute_flip_effects(b, n):
    """For each position p and shift k, compute the effect of flipping b[p] on f_k.
    Returns a matrix of shape (n, 2n-1) where effect[p, k] is the change in f_k
    (multiplied by dx) when b[p] is flipped from 1 to 0."""

    effect = np.zeros((n, 2*n-1))
    for p in range(n):
        for k in range(2*n-1):
            shift = k - (n-1)
            if shift == 0:

            delta = 0.0
            if 0 <= p + shift < n:
                delta -= (1.0 - b[p+shift])
            if 0 <= p - shift < n:
                delta += b[p-shift]
            effect[p, k] = delta * dx
    return effect
```
