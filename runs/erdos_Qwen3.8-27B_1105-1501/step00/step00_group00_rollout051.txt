from scipy.optimize import minimize
import time

def eval_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, target_sum):
    """Project y onto {0 <= h <= 1, sum(h) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0, 1))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_smooth_obj(h, dx, tau):
    """Compute log-sum-exp smoothed objective."""
    corr = np.correlate(h, 1 - h, mode="full")
    values = corr * dx
    tv = tau * values
    m = np.max(tv)
    return (m + np.log(np.sum(np.exp(tv - m)))) / tau

def compute_smooth_grad(h, dx, tau):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    values = corr * dx
    tv = tau * values
    m = np.max(tv)
    ev = np.exp(tv - m)
    total = np.sum(ev)
    weights = ev / total
    
    grad = np.zeros(n)
    offset = n - 1
    i = np.arange(n)
    for k_idx in range(len(values)):
        w = weights[k_idx] * dx
        if w < 1e-15:
            continue
        lag = k_idx - offset
        if lag == 0:
            grad += w * (1 - 2.0 * h)
        elif lag > 0:
            g = np.zeros(n)
            mask1 = i >= lag
            mask2 = i < (n - lag)
            g[mask1] += 1.0 - h[i[mask1] - lag]
            g[mask2] += -h[i[mask2] + lag]
            grad += w * g
        else:
            mabs = -lag
            g = np.zeros(n)
            mask1 = i < (n - mabs)
            mask2 = i >= mabs
            g[mask1] += 1.0 - h[i[mask1] + mabs]
            g[mask2] += -h[i[mask2] - mabs]

    n = len(h_init)
    bounds = [(0, 1)] * n

    try:

                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x, res.fun
    except Exception:
        return h_init, float('inf')

from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0, 1))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_smooth_obj(h, dx, tau):
    corr = np.correlate(h, 1 - h, mode="full")
    values = corr * dx
    tv = tau * values
    m = np.max(tv)
    return (m + np.log(np.sum(np.exp(tv - m)))) / tau

def compute_smooth_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    values = corr * dx
    tv = tau * values
    m = np.max(tv)
    ev = np.exp(tv - m)
    total = np.sum(ev)
    weights = ev / total
    grad = np.zeros(n)
    offset = n - 1
    i = np.arange(n)
    for k_idx in range(len(values)):
        w = weights[k_idx] * dx
        if w < 1e-15:
            continue
        lag = k_idx - offset
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            g = np.zeros(n)
            mask1 = i >= lag
            mask2 = i < (n - lag)
            g[mask1] += 1.0 - h[i[mask1] - lag]
            g[mask2] += -h[i[mask2] + lag]
            grad += w * g
        else:
            mabs = -lag
            g = np.zeros(n)
            mask1 = i < (n - mabs)
            mask2 = i >= mabs
            g[mask1] += 1.0 - h[i[mask1] + mabs]
            g[mask2] += -h[i[mask2] - mabs]
            grad += w * g
    return grad

def slsqp_obj(h, dx, tau):
    return compute_smooth_obj(h, dx, tau)

def slsqp_grad(h, dx, tau):
    return compute_smooth_grad(h, dx, tau)

def eq_fun(h, target_sum):

def eq_jac(h, target_sum):
    return np.ones(len(h))

def run_slsqp(h_init, dx, target_sum, tau, maxiter=500):
    n = len(h_init)
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'args': (target_sum,), 'jac': eq_jac}]
    try:
        res = minimize(slsqp_obj, h_init, args=(dx, tau), jac=slsqp_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_init

def run_projected_gradient(h_init, dx, target_sum, tau, max_iter=5000, deadline=None):
    current = h_init.copy()
    best = h_init.copy()
    best_val = eval_c5(current)

        if deadline is not None and time.time() > deadline:
            break
        f_curr = compute_smooth_obj(current, dx, tau)
        grad = compute_smooth_grad(current, dx, tau)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        alpha = 1.0

            alpha *= 0.5
        current = candidate
        c = eval_c5(current)
        if c < best_val:
            best_val = c
            best = current.copy()
    return best, best_val
