import numpy as np
import time

def exact_bound(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_feasible(x, N):
    """Project x onto {0 <= h <= 1, sum(h) = N/2}."""
    # Water-filling: find t such that sum(clip(x - t, 0, 1)) = N/2
    target = N / 2.0
    
    # Bisection on t
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(x - t, 0.0, 1.0)
    return h

def compute_subgradient(h, num_active=None):
    """Compute subgradient using finite differences."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = np.max(corr)
    
    if num_active is None:
        num_active = max(1, len(corr) // 4)  # top 25%
    
    # Get top num_active indices
    top_indices = np.argsort(corr)[::-1][:num_active]
    
    # Numerical gradient
    eps = 1e-5
    grad = np.zeros(N)
    for i in range(N):
        old = h[i]
        h[i] = old + eps
        corr_plus = np.correlate(h, 1.0 - h, mode='full')
        h[i] = old
        grad[i] = np.sum((corr_plus[top_indices] - corr[top_indices]) * dx) / (eps * num_active)
    
    return grad

def dyadic_quantize(h, N, bits=20):
    """Quantize h to dyadic values with exact sum."""
    S = 1 << bits
    if N % 2 != 0:
        S *= 2
    
    target = (N * S) // 2
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    
    # Water-fill to achieve exact target sum
    current_sum = np.sum(z)
    diff = target - current_sum
    
    if diff > 0:
        # Need to add diff units
        # Add to positions with most room (z < S)
        room = S - z
        # greedily add to positions with most room
        order = np.argsort(-room)
        for idx in order:
            if diff <= 0:
                break
            add = min(room[idx], diff)
            z[idx] += add
            diff -= add
    elif diff < 0:
        # Need to remove -diff units
        order = np.argsort(z)
        for idx in order:
            if diff >= 0:
                break
            remove = min(z[idx], -diff)
            z[idx] -= remove
            diff += remove
    
    return z.astype(float) / S

    h = np.clip(h, 0, 1)

    """Try pairwise mass transfers for local improvement."""
    best_h = h.copy()
    best_c = exact_bound(best_h)

        # Try moving mass from higher to lower

            c = exact_bound(trial)
            if c < best_c - 1e-10:

                best_c = c

    """Log-sum-exp approximation of max(f_k) * dx."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    # For numerical stability
    shifted = T * (corr - max_val)
    logsum = max_val + np.log(np.sum(np.exp(shifted))) / T
    return logsum

def smooth_gradient(h, T):
    """Gradient of the smoothed objective."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    weights = np.exp(T * (corr - max_val))
    weights /= np.sum(weights)

    # Each f_k = dx * sum_i h[i]*(1-h[i+k]) for valid i
    # grad f_k[i] = dx * [(1-h[i+k])*1_{valid} - h[i-k]*1_{valid}]
    
    grad = np.zeros(N)
    for idx, w in enumerate(weights):
        if w < 1e-10:
            continue
        k = idx - (N - 1)
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:N-k] += w * dx * (1.0 - h[k:])
            grad[k:] -= w * dx * h[:N-k]
        else:
            k2 = -k
            grad[k2:] += w * dx * (1.0 - h[:N-k2])
            grad[:N-k2] -= w * dx * h[k2:]
    return grad
