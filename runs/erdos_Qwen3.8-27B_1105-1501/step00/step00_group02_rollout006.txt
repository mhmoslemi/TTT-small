def snap_dyadic(h, Q_bits):
    """Snap h to dyadic grid with exact sum"""
    n = len(h)
    Q = 1 << Q_bits
    target = n * Q // 2  # target sum in integer units
    
    # Initial rounding
    k = np.round(h * Q).astype(np.int64)
    k = np.clip(k, 0, Q)
    
    diff = target - np.sum(k)
    
    if diff > 0:
        # Need to add diff units
        # Priority: positions where fractional part is closest to next integer
        frac = h * Q - k
        # Sort by frac descending (closest to rounding up)
        order = np.argsort(-frac)
        idx = 0
        while diff > 0 and idx < n:
            j = order[idx]
            if k[j] < Q:
                k[j] += 1
                diff -= 1
            idx += 1
    elif diff < 0:
        # Need to subtract -diff units
        frac = h * Q - k
        # Sort by frac ascending (closest to rounding down)
        order = np.argsort(frac)
        idx = 0
        while diff < 0 and idx < n:
            j = order[idx]
            if k[j] > 0:
                k[j] -= 1
                diff += 1
            idx += 1
    
    return k.astype(np.float64) / Q

    if diff > 0:

        idx = 0
        while diff > 0 and idx < n:
            j = order[idx]
            if k[j] < Q:

                diff -= 1
            idx += 1

        idx = 0

        scale = 0.3 + 0.5 * rng.random()
        z_pert = z_cur + rng.randn(n) * scale
        z_pert = np.clip(z_pert, -W, W)

        for beta in [5, 30, 100]:
            if time.time() > deadline:
                break
            bounds = [(-W, W)] * n
            try:
                res = minimize(
                    smooth_obj_jac, z_pert, jac=True,
                    args=(n, beta, dx, deadline),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 150, 'maxfun': 400,
                             'ftol': 1e-13, 'gtol': 1e-9}
                )
                if res.fun < 1e9:
                    _, _, h_new = compute_loss_grad(res.x, n, beta, dx)
                    c_new = evaluate_c5(h_new)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                        z_cur = res.x.copy()
                        z_pert = z_cur.copy()
            except Exception:
                pass

    # Finer L-BFGS refinement on best
    if time.time() < deadline:
        for beta in [100, 300, 800]:
            if time.time() > deadline:
                break
            bounds = [(-W, W)] * n
            try:
                res = minimize(
                    smooth_obj_jac, z_cur, jac=True,
                    args=(n, beta, dx, deadline),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 500, 'maxfun': 1500,
                             'ftol': 1e-15, 'gtol': 1e-10}
                )
                if res.fun < 1e9:
                    _, _, h_new = compute_loss_grad(res.x, n, beta, dx)
                    c_new = evaluate_c5(h_new)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
            except Exception:
                pass

    # Dyadic snap to get exact constraints
    if time.time() < deadline:
        for Q_bits in [18, 20, 22, 24, 26]:
            h_snap = snap_dyadic(best_h, Q_bits)
            c_snap = evaluate_c5(h_snap)
            if c_snap < best_c:
                best_c = c_snap
                best_h = h_snap

    # Pair-line polish on the snapped (or best) solution
    if time.time() < deadline - 3:
        h_pol, c_pol = pair_line_polish(best_h, n,
                                        max_rounds=20, max_pairs=300,
                                        seed=seed)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    # Final snap
    for Q_bits in [20, 22, 24]:
        h_s = snap_dyadic(best_h, Q_bits)
        c_s = evaluate_c5(h_s)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - target) > 1e-8:
        best_h = snap_dyadic(best_h, 22)

    return best_h, best_c, n
