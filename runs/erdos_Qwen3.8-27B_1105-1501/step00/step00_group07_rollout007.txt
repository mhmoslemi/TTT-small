def score(h):
    """Compute the exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(raw)

def project_feasible(z, target):
    """Project z to the feasible set: 0<=h<=1, sum(h)=target."""
    n = len(z)
    v = np.clip(z, 0.0, 1.0)
    # Handle non-finite
    v[~np.isfinite(v)] = 0.5
    e = np.sum(v) - target
    if e > 1e-12:
        # Need to reduce sum by e
        # Sort by descending value, reduce from largest
        idx = np.argsort(-v)
        for i in idx:
            delta = min(v[i], e)
            v[i] -= delta
            e -= delta
            if e <= 1e-12:
                break
    elif e < -1e-12:
        # Need to increase sum by -e
        # Sort by ascending value, increase from smallest
        idx = np.argsort(v)
        for i in idx:
            delta = min(1.0 - v[i], -e)
            v[i] += delta
            e += delta
            if e >= -1e-12:
                break
    return v

def score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(raw))

def project_feasible(z, target):
    n = len(z)
    v = np.array(z, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    e = np.sum(v) - target
    if e > 1e-12:
        idx = np.argsort(-v)
        for i in idx:
            d = min(v[i], e)
            v[i] -= d
            e -= d
            if e <= 1e-12:
                break
    elif e < -1e-12:
        idx = np.argsort(v)
        for i in idx:
            d = min(1.0 - v[i], -e)
            v[i] += d
            e += d
            if e >= -1e-12:
                break
    return v

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    cor = np.correlate(h, g, mode='full')
    y = dx * cor
    mx = np.max(y)
    shifted = alpha * (y - mx)
    shifted = np.maximum(shifted, -500.0)
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros_like(y)
        w[np.argmax(cor)] = 1.0
        ws = 1.0
    w /= ws
    # Gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        m = r - (n - 1)
        wr = w[r]
        if m >= 0:
            L = n - m
            if L <= 0:
                continue
            grad[m:m+L] += wr * g[:L]
            grad[:L] -= wr * h[m:m+L]
        else:
            j = -m
            L = n - j
            if L <= 0:
                continue
            grad[:L] += wr * g[j:j+L]
            grad[j:j+L] -= wr * h[:L]
    grad *= dx
    grad -= np.mean(grad)
    U = float(mx + np.log(ws) / alpha)
    return U, grad

            best_h = v
    if best_h is not None:

    return None
