import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum"""
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        if diff > 0:
            step = min(step, (1.0 - h[mask]).min())
        else:
            step = max(step, -h[mask].min())
        h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-bound values
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def local_search_binary(h, max_rounds=500, timeout=30.0):
    """Deterministic local search for binary solutions"""
    n = len(h)
    best_c5 = c5_value(h)
    start = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - start > timeout:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_flip = None
        best_new_max = max_val
        
        for i in ones:
            for j in zeros:
                # Flip i: 1->0, j: 0->1
                h[i] = 0
                h[j] = 1
                new_corr = np.correlate(h, 1.0 - h, mode="full")
                new_max = np.max(new_corr)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_flip = (i, j)
                # Undo
                h[i] = 1
                h[j] = 0
        
        if best_flip is None:
            break  # Local optimum
        
        i, j = best_flip
        h[i] = 0
        h[j] = 1
        best_c5 = best_new_max * (2.0 / n)
    
    return h, best_c5

    best_c5 = c5_value(h)
    start = time.time()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

        if time.time() - start > timeout:
            break

        full = np.zeros(full_len)
        d = np.arange(-(n-1), n)

        for dval in range(-(n-1), n):
            if dval >= 0:
                i_min, i_max = 0, n - 1 - dval
            else:
                i_min, i_max = -dval, n - 1
            full[dval + n - 1] = np.sum(h[i_min:i_max+1] * (1 - h[i_min+dval:i_max+1+dval]))
        
        # Actually, let me just use np.correlate
        full = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(full)
        
        # Precompute v_x[d] for all x
        # v_x[d] = h[x+d] + h[x-d] (with h = 0 outside [0,n))
        # d ranges from -(n-1) to (n-1), indexed as 0 to 2n-2
        dvals = np.arange(-(n-1), n)  # length 2n-1
        v = np.zeros((n, full_len))
        for x in range(n):
            for idx, dval in enumerate(dvals):
                val = 0.0
                if 0 <= x + dval < n:
                    val += h[x + dval]
                if 0 <= x - dval < n:
                    val += h[x - dval]
                v[x, idx] = val
        
        # For each pair (p in ones, q in zeros), compute new max
        best_flip = None
        best_new_max = max_val
        
        for p in ones:
            g_p = full + v[p]
            for q in zeros:
                new_vals = g_p - v[q]
                new_max = np.max(new_vals)
                if new_max < best_new_max:

                    best_flip = (p, q)
        
        if best_flip is None:
            break
        
        p, q = best_flip
        h[p] = 0
        h[q] = 1
        ones.remove(p)
        zeros.remove(q)
        ones.append(q)
        zeros.append(p)
        best_c5 = best_new_max * (2.0 / n)

def compute_grad_at_lag(h, d):
    """Compute gradient of corr at lag d with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        # (1-h[k+d]) for k=0..n-1-d
        grad[:n-d] += (1 - h[d:])
        # -h[k-d] for k=d..n-1
        grad[d:] -= h[:n-d]
    else:
        d_pos = -d
        # (1-h[k-d_pos]) for k=d_pos..n-1
        grad[d_pos:] += (1 - h[:n-d_pos])
        # -h[k+d_pos] for k=0..n-1-d_pos
        grad[:n-d_pos] -= h[d_pos:]
    return grad
```

    # v[x, idx] = h[x+d] + h[x-d] where d = dvals[idx]
    # For h[x+d]: valid when 0 <= x+d < n, i.e., -x <= d < n-x
    # For h[x-d]: valid when 0 <= x-d < n, i.e., x-n < d <= x
    
    for x in range(n):
        # h[x+d] part
        valid_d1 = (dvals >= -x) & (dvals < n - x)
        d_vals1 = dvals[valid_d1]
        v[x, valid_d1] += h[x + d_vals1]
        # h[x-d] part
        valid_d2 = (dvals > x - n) & (dvals <= x)
        d_vals2 = dvals[valid_d2]
        v[x, valid_d2] += h[x - d_vals2]
    return v

Sum over all positive lags: sum_{d=1}^{63} count_d = sum_{each 1} (number of 0s after it at various distances) = 32 * 32 = 1024.

Wait, that's not right. For a fixed 1 at position i, it contributes to count_d if there's a 0 at position i+d. The total number of such contributions over all d is the number of 0s that are to the right of position i. But the 0s can be both to the left and right.

Actually: sum_{d=1}^{n-1} count_d = sum_{d=1}^{n-1} #{i: h[i]=1, h[i+d]=0}
= #{(i,j): h[i]=1, h[j]=0, j>i}
= number of (1,0) pairs where the 1 comes first.

This is not 32*32 = 1024 in general. It depends on the arrangement. If all 1s come before all 0s, it's 32*32 = 1024. If they're interleaved, it's less.

Similarly, sum_{d=1}^{n-1} count_{-d} = #{(i,j): h[i]=1, h[j]=0, j<i} = number of (1,0) pairs where the 0 comes first.

And sum_{d=1}^{n-1} (count_d + count_{-d}) = 32*32 = 1024.

So the total over all non-zero lags (both positive and negative) is 1024.
There are 126 non-zero lags (d = -63, ..., -1, 1, ..., 63).
Average = 1024/126 ≈ 8.13.

Plus lag 0 contributes 0 (for binary). So total over all 127 lags is still 1024.
Average per lag = 1024/127 ≈ 8.06.

```python
import numpy as np
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        if diff > 0:
            step = min(step, (1.0 - h[mask]).min())
        else:
            step = max(step, -h[mask].min())
        h[mask] += step
    return np.clip(h, 0.0, 1.0)

def compute_v(h):
    n = len(h)
    full_len = 2 * n - 1
    dvals = np.arange(-(n-1), n)
    v = np.zeros((n, full_len))
    for x in range(n):
        valid1 = (dvals >= -x) & (dvals < n - x)
        d1 = dvals[valid1]
        v[x, valid1] += h[x + d1]
        valid2 = (dvals > x - n) & (dvals <= x)
        d2 = dvals[valid2]
        v[x, valid2] += h[x - d2]
    return v

def local_search_binary_fast(h, max_rounds=5000, timeout=30.0):
    n = len(h)
    full_len = 2 * n - 1
    best_c5 = c5_value(h)
    start = time.time()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for rnd in range(max_rounds):
        if time.time() - start > timeout:
            break
        
        full = np.correlate(h, 1.0 - h, mode="full").astype(float)
        max_val = full.max()
        
        v = compute_v(h)
        
        ones_arr = np.array(ones)
        zeros_arr = np.array(zeros)
        
        g = full[None, None, :] + v[ones_arr][:, None, :]
        new_vals = g - v[zeros_arr][None, :, :]
        new_maxs = new_vals.max(axis=2)
        
        flat_idx = new_maxs.argmin()
        idx_p, idx_q = np.unravel_index(flat_idx, new_maxs.shape)
        best_new_max = new_maxs[idx_p, idx_q]
        
        if best_new_max >= max_val:
            break
        
        p = ones[idx_p]
        q = zeros[idx_q]
        h[p] = 0
        h[q] = 1
        ones[idx_p] = q
        zeros[idx_q] = p
        best_c5 = best_new_max * (2.0 / n)
    
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Get initial solution if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = project_feasible(h0, n0 / 2.0)
        c5_0 = c5_value(h0)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = n0
    
    # Try binary solutions for even n values
    n_candidates = [64, 62, 66, 68, 70, 60, 72]
    rng = np.random.RandomState(seed)
    
    for n in n_candidates:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 30:
            break
        
        n_restarts = min(int(remaining / max(n / 64.0, 1.0) * 10), 30)
        n_restarts = max(n_restarts, 3)
        timeout_per_restart = min(remaining / (n_restarts + 1), 20.0)
        
        for r in range(n_restarts):
            elapsed = time.time() - start_time
            if budget_s - elapsed < timeout_per_restart:
                break
            
            # Random initial binary
            h_bin = np.zeros(n)
            indices = rng.choice(n, size=n//2, replace=False)
            h_bin[indices] = 1
            
            h_opt, c5_opt = local_search_binary_fast(h_bin, max_rounds=3000, timeout=timeout_per_restart)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
    
    # Also try to optimize the initial solution with gradient descent
    if best_n == 63 and best_h is not None:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 30:
            h_grad = gradient_descent_erdos(best_h, timeout=min(remaining/2, 60.0), seed=seed)
            c5_grad = c5_value(h_grad)
            if c5_grad < best_c5:
                best_c5 = c5_grad
                best_h = h_grad
    
    # Final verification
    if best_h is not None:
        final_c5 = c5_value(best_h)
        assert abs(final_c5 - best_c5) < 1e-10
        best_c5 = final_c5
    
    return best_h, best_c5, best_n

def gradient_descent_erdos(h, timeout=60.0, seed=42):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    dvals = np.arange(-(n-1), n)
    full_len = 2 * n - 1
    start = time.time()
    
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = c5_value(h)
    
    step_size = 0.05
    patience = 0
    
    for it in range(50000):
        if time.time() - start > timeout:
            break
        
        full = np.correlate(h, 1.0 - h, mode="full")
        c5 = full.max() * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                step_size *= 0.5
                patience = 0
                if step_size < 0.001:
                    break
        
        # Find active lags
        threshold = full.max() - 0.5
        active = np.where(full >= threshold)[0]
        
        # Compute gradient (average over active lags)
        grad = np.zeros(n)
        for m in active:
            d = dvals[m]
            if d >= 0:
                grad[:n-d] += (1.0 - h[d:])
                grad[d:] -= h[:n-d]
            else:
                d_pos = -d
                grad[d_pos:] += (1.0 - h[:n-d_pos])
                grad[:n-d_pos] -= h[d_pos:]
        grad /= len(active)
        
        # Direction also considers the sum constraint
        # Use simple projection
        h_new = h - step_size * grad * dx
        h_new = project_feasible(h_new, target_sum)
        
        new_c5 = c5_value(h_new)
        if new_c5 > c5:
            step_size *= 0.5
            if step_size < 0.0005:
                break
    
    return best_h

```python
def local_search_binary_fast(h, max_rounds=5000, timeout=30.0, n_perturb=5):
    n = len(h)
    full_len = 2 * n - 1
    best_c5 = c5_value(h)
    start = time.time()
    rng = np.random.RandomState(42)
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for rnd in range(max_rounds):
        if time.time() - start > timeout:
            break
        
        full = np.correlate(h, 1.0 - h, mode="full").astype(float)
        max_val = full.max()
        
        v = compute_v(h)
        ones_arr = np.array(ones)
        zeros_arr = np.array(zeros)
        
        g = full[None, None, :] + v[ones_arr][:, None, :]
        new_vals = g - v[zeros_arr][None, :, :]
        new_maxs = new_vals.max(axis=2)
        
        flat_idx = new_maxs.argmin()
        idx_p, idx_q = np.unravel_index(flat_idx, new_maxs.shape)
        best_new_max = new_maxs[idx_p, idx_q]
        
        if best_new_max >= max_val:
            # Local optimum - perturb
            # Make n_perturb random flips
            for _ in range(n_perturb):
                i = rng.choice(len(ones))
                j = rng.choice(len(zeros))
                p, q = ones[i], zeros[j]
                h[p], h[q] = 0, 1
                ones[i], zeros[j] = q, p
            continue
        
        p = ones[idx_p]
        q = zeros[idx_q]
        h[p] = 0
        h[q] = 1
        ones[idx_p] = q
        zeros[idx_q] = p
        best_c5 = best_new_max
