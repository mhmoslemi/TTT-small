def project_feasible(v, target_sum):
    """Project v onto {h: 0<=h_i<=1, sum(h_i)=target_sum}."""
    n = len(v)
    lo, hi = -2.0, 2.0
    while np.sum(np.clip(v - lo, 0, 1)) < target_sum:
        lo -= 1.0
    while np.sum(np.clip(v - hi, 0, 1)) > target_sum:
        hi += 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    # Fix any tiny sum error
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_values(h):
    n = len(h)

def grad_max_c5(h):
    """Subgradient of max(corr*dx). Returns (grad, max_val)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mv = float(np.max(corr))
    tol = 1e-6
    active = np.where(corr >= mv - tol)[0]
    grad = np.zeros(n)
    for p in active:
        k = p - (n - 1)
        if k >= 0:
            L = n - k
            if L > 0:
                grad[:L] += (1.0 - h[k:])
                grad[k:] -= h[:L]
        else:
            e = -k
            L = n - e
            if L > 0:
                grad[:L] -= h[e:]
                grad[e:] += (1.0 - h[:L])
    grad *= dx / len(active)
    return grad, mv

def smooth_obj_and_grad(h, T, n):
    """Log-sum-exp smoothed objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = corr / T
    m = np.max(scaled)
    obj = T * (m + np.log(np.sum(np.exp(scaled - m))))
    w = np.exp(scaled - m)
    ws = np.sum(w)
    w = w / ws
    grad = np.zeros(n)

            e = -k
            L = n - e
            if L > 0:

def binary_start(n, freq):
    """Create a structured binary-like starting point."""
    t = np.arange(n)
    h = np.zeros(n)
    h[2 * t % n < n // 2] = 1.0
    # Add frequency modulation
    phase = 2.0 * np.pi * freq * t / n
    h = 0.5 + 0.5 * np.sign(np.sin(phase))
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)

def msequence_start(n):
    """Generate an m-sequence-like binary start (for n=2^m-1)."""
    # Use a simple LFSR-based approach for m-sequences
    period = 1 << (n.bit_length() - 1)
    if period - 1 == n:
        # Verify n is 2^m - 1
        lfsr = 1
        seq = []
        for _ in range(n):
            seq.append(lfsr & 1)
            bit = ((lfsr >> 0) ^ (lfsr >> 2) ^ (lfsr >> 4) ^ (lfsr >> 5)) & 1
            lfsr = (lfsr >> 1) | (bit << (n.bit_length() - 2))
        h = np.array(seq, dtype=float)
        if np.sum(h) > n / 2.0:
            h = 1.0 - h
        h = project_feasible(h, n / 2.0)
        return h
    return None

def sublgradient_step(h, grad, alpha, target):
    """One projected subgradient step."""
    n = len(h)
    gp = grad - np.mean(grad)
    h_new = project_feasible(h - alpha * gp, target)
    return h_new

def pairwise_refine(h, target, rng, n_pairs, time_check, time_budget):
    """Refine by optimizing pairs of coordinates."""
    n = len(h)
    h_cur = h.copy()
    c5_cur = c5_value(h_cur)
    for _ in range(n_pairs):
        if time_check() > time_budget:
            break
        j = rng.randint(n)
        jp = rng.randint(n)
        if j == jp:
            continue
        s = h_cur[j] + h_cur[jp]
        t_lo = max(0.0, s - 1.0)
        t_hi = min(1.0, s)
        if t_hi - t_lo < 1e-12:
            continue
        # Evaluate at grid points + refine
        best_t = None
        best_val = c5_cur + 1.0
        grid = np.linspace(t_lo, t_hi, 20)
        for t in grid:
            h_trial = h_cur.copy()
            h_trial[j] = t
            h_trial[jp] = s - t
            v = c5_value(h_trial)
            if v < best_val:
                best_val = v
                best_t = t
        if best_t is not None and best_val < c5_cur - 1e-9:
            h_cur = h_cur.copy()
            h_cur[j] = best_t
            h_cur[jp] = s - best_t
            c5_cur = best_val
    return h_cur, c5_cur
