`np.correlate(a, v, mode='full')` computes:
c[k] = sum_n a[n+k] * conj(v[n])

Actually, the numpy documentation says:
c[a, v, k] = sum_n a[n+k] * conj(v[n])

For mode='full', k ranges from -(len(v)-1) to len(a)-1.
Output has length len(a) + len(v) - 1 = 2n-1.

Index in output: if we let the output be indexed 0 to 2n-2, then:
output[k] corresponds to lag = k - (n-1)

For lag = m - (n-1):
- If lag >= 0: output = sum_{i=0}^{n-lag-1} h[i+lag] * (1-h[i])
- If lag < 0: let p = -lag, output = sum_{i=p}^{n-1} h[i] * (1-h[i-p])

So for lag = ell >= 0:
C(ell) = sum_{i=0}^{n-ell-1} h[i+ell] * (1-h[i])

Gradient wrt h[j]:
- h[j] appears with coefficient (1-h[j-ell]) when j >= ell (from the h[i+ell] term with i+ell=j, so i=j-ell)
- h[j] appears with coefficient (-h[j+ell]) when j < n-ell (from the (1-h[i]) term with i=j)

So:
dC(ell)/dh[j] = (1-h[j-ell]) * 1[j in [ell, n-1]] - h[j+ell] * 1[j in [0, n-ell-1]]

For ell = 0:
C(0) = sum_{i=0}^{n-1} h[i] * (1-h[i])
dC(0)/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]

For lag < 0, let p = -lag:
C(-p) = sum_{i=p}^{n-1} h[i] * (1-h[i-p])

Gradient wrt h[j]:
- h[j] appears with coefficient (1-h[j-p]) when j >= p
- h[j] appears with coefficient (-h[j+p]) when j < n-p

So:
dC(-p)/dh[j] = (1-h[j-p]) * 1[j in [p, n-1]] - h[j+p] * 1[j in [0, n-p-1]]

Now for the LP:
We want to minimize t subject to:
t >= C(ell_m) for each active constraint m
sum(h) = n/2
0 <= h[i] <= 1

Linearization at h0:
C(ell_m; h) ≈ C(ell_m; h0) + G_m · (h - h0) = G_m · h + (C(ell_m; h0) - G_m · h0)

So the constraint becomes:
t >= G_m · h + kappa_m
=> G_m · h - t <= -kappa_m

In linprog form (minimize c^T x subject to A_ub x <= b_ub):
Variable x = [h_0, ..., h_{n-1}, t]
c = [0, ..., 0, 1]
A_ub row for constraint m: [G_m, -1] <= -kappa_m
A_eq: [1, 1, ..., 1, 0] = n/2
bounds: 0 <= h[i] <= 1 (or with trust radius), t free (or t >= 0)

def compute_gradients(h):
    """Compute gradients of all correlation components w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    grads = []
    for ell in range(-(n-1), n):
        g = np.zeros(n)
        if ell >= 0:
            # C = sum_{i=0}^{n-ell-1} h[i+ell] * (1-h[i])
            # dC/dh[j] = (1-h[j-ell]) for j in [ell, n-1]
            #           - h[j+ell] for j in [0, n-ell-1]

            else:
                g[ell:] += (1.0 - h[:n-ell])
                g[:n-ell] -= h[ell:]
        else:
            p = -ell
            # C = sum_{i=p}^{n-1} h[i] * (1-h[i-p])

            #           - h[j+p] for j in [0, n-p-1]
            g[p:] += (1.0 - h[:n-p])
            g[:n-p] -= h[p:]
        grads.append(g * dx)
    return grads

def get_correlation_values(h):
    """Get all correlation values (scaled by dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def select_active(corr_vals, n, q=13):
    """Select active constraint indices"""
    # corr_vals has length 2n-1, index m corresponds to lag m-(n-1)
    # Center is at index n-1
    center = n - 1
    active = [center]
    # Top q by value
    sorted_idx = np.argsort(corr_vals)[::-1]
    for idx in sorted_idx:
        if len(active) >= q:
            break
        if idx not in active:
            active.append(int(idx))
    # Also include some symmetric pairs
    for p in range(1, min(n-1, q)):
        left = center - p
        right = center + p
        if 0 <= left < 2*n-1 and left not in active:
            active.append(left)
        if 0 <= right < 2*n-1 and right not in active:
            active.append(right)
        if len(active) >= 2*q:

def solve_tangent_lp(h0, active_indices, trust_radius=0.15):
    """Solve the tangent linear program at h0"""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    # Compute gradient and value at h0 for each active constraint
    corr_vals = np.correlate(h0, 1.0 - h0, mode='full') * dx

    b_rows = []

        # Compute value

        kappa = val - np.dot(g, h0)

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    # Water filling
    for _ in range(n + 1):
        if diff > 1e-14:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            per = diff / np.sum(mask)
            h[mask] = np.minimum(1.0, h[mask] + per)
        elif diff < -1e-14:
            mask = h > 1e-15
            if not np.any(mask):
                break
            per = -diff / np.sum(mask)
            h[mask] = np.maximum(0.0, h[mask] - per)
        else:
            break
        diff = target - np.sum(h)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmin(1.0 - h)
            room = 1.0 - h[idx]
            if room >= diff:
                h[idx] += diff
        else:
            idx = np.argmax(h)
            if -h[idx] >= diff:  # h[idx] >= -diff
                h[idx] += diff
    return np.clip(h, 0.0, 1.0)

def compute_single_gradient(h, ell, dx):
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    else:
        p = -ell
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g * dx

def select_active(corr_vals, n, q=15):
    center = n - 1
    length = len(corr_vals)
    active = [center]
    # Top values
    sorted_idx = np.argsort(corr_vals)[::-1]
    for idx in sorted_idx[:q]:
        idx = int(idx)
        if idx not in active:
            active.append(idx)
    # Symmetric pairs around center
    for p in range(1, min(n, q + 1)):
        left = center - p
        right = center + p
        if 0 <= left < length and left not in active:
            active.append(left)
        if 0 <= right < length and right not in active:
            active.append(right)
        if len(active) >= 2 * q:
            break
    return list(dict.fromkeys(active))  # dedupe preserving order

def solve_tangent_lp(h0, active_indices, trust_radius=0.2):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    center = n - 1
    corr_vals = np.correlate(h0, 1.0 - h0, mode='full') * dx
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_rows = []
    b_rows = []
    
    for m in active_indices:
        ell = m - center
        g = compute_single_gradient(h0, ell, dx)
        val = corr_vals[m]
        kappa = val - np.dot(g, h0)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-kappa)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_radius)
        hi = min(1.0, h0[i] + trust_radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def pair_line_search(h, pair_i, pair_j, n_samples=17):
    """Try to improve by transferring mass between coordinates i and j"""
    n = len(h)
    best_val = evaluate_c5(h)
    best_h = h.copy()
    
    # Determine feasible range of delta
    delta_min = max(-h[pair_i], h[pair_j] - 1.0)
    delta_max = min(1.0 - h[pair_i], h[pair_j])
    
    if delta_max - delta_min < 1e-12:
        return best_h, best_val
    
    # Sample points including endpoints
    if n_samples <= 2:
        deltas = [delta_min, delta_max]
    else:
        deltas = np.linspace(delta_min, delta_max, n_samples)
    
    for d in deltas:
        h_test = h.copy()
        h_test[pair_i] += d
        h_test[pair_j] -= d
        np.clip(h_test, 0.0, 1.0, out=h_test)
        # Check mass
        if abs(np.sum(h_test) - n/2.0) > 1e-10:
            continue
        val = evaluate_c5(h_test)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_test
    
    return best_h, best_val

def gradient_guided_pairs(h, top_k=20):
    """Select promising coordinate pairs based on gradient"""
    n = len(h)
    dx = 2.0 / n
    corr_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    center = n - 1
    
    # Compute average gradient weighted by correlation values
    # Use top correlations
    active = select_active(corr_vals, n, q=8)
    avg_grad = np.zeros(n)
    for m in active:
        ell = m - center
        g = compute_single_gradient(h, ell, dx)
        avg_grad += g * (corr_vals[m] / max(np.max(corr_vals), 1e-15))
    
    # Pairs with large gradient difference are promising
    # We want to move mass from high-gradient-sum to low-gradient-sum
    # (increase low, decrease high) to reduce the max
    order = np.argsort(avg_grad)
    pairs = []
    for i in range(min(10, n)):
        for j in range(min(10, n)):
            if i != j and order[i] != order[j]:
                a, b = int(order[i]), int(order[j])
                if a > b:
                    a, b = b, a
                pairs.append((a, b))
    # Deduplicate
    seen = set()
    unique_pairs = []
    for p in pairs:
        if p not in seen:
            seen.add(p)
            unique_pairs.append(p)
    return unique_pairs[:top_k]

def resolution_transfer(h_src, n_target):
    """Transfer h from source resolution to target resolution preserving area"""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_target = 2.0 / n_target
    
    h_target = np.zeros(n_target)
    src_pos = 0
    
    for k in range(n_target):
        t_left = k * dx_target
        t_right = (k + 1) * dx_target
        area = 0.0
        s = src_pos
        while s < n_src:
            s_left = s * dx_src
            s_right = (s + 1) * dx_src
            overlap = min(s_right, t_right) - max(s_left, t_left)
            if overlap > 0:
                area += h_src[s] * overlap
            if s_right >= t_right:
                src_pos = s + 1
                break
            s += 1
        h_target[k] = area / dx_target
    
    return repair_feasible(h_target)

def perturb_initial(h, rng, scale=0.1):

    n = len(h)

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_feasible(h):
    n = len(h)
    if n == 0:
        return h
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(n + 2):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + per)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = -diff / cnt
            h[mask] = np.maximum(0.0, h[mask] - per)
        diff = target - np.sum(h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmin(1.0 - h))
            room = 1.0 - h[idx]
            if room >= diff - 1e-15:
                h[idx] += diff
            else:
                h[idx] = 1.0
                diff -= room
                if diff > 1e-15:
                    for i in range(n):
                        if i != idx and h[i] < 1.0 - 1e-15:
                            add = min(diff, 1.0 - h[i])
                            h[i] += add
                            diff -= add
                            if diff < 1e-15:
                                break
        else:
            idx = int(np.argmax(h))
            room = h[idx]
            if room >= -diff - 1e-15:
                h[idx] += diff
            else:
                h[idx] = 0.0
                diff += room
                if diff < -1e-15:
                    for i in range(n):
                        if i != idx and h[i] > 1e-15:
                            sub = min(-diff, h[i])
                            h[i] -= sub
                            diff += sub
                            if diff > -1e-15:
                                break
    return np.clip(h, 0.0, 1.0)

def compute_single_gradient(h, ell, dx):
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    else:
        p = -ell
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g * dx

def select_active(corr_vals, n, q=15):
    center = n - 1
    length = len(corr_vals)
    active = [center]
    sorted_idx = np.argsort(corr_vals)[::-1]
    for idx in sorted_idx[:q]:
        idx = int(idx)
        if idx not in active:
            active.append(idx)
    for p in range(1, min(n, q + 1)):
        left = center - p
        right = center + p
        if 0 <= left < length and left not in active:
            active.append(left)
        if 0 <= right < length and right not in active:
            active.append(right)
        if len(active) >= 2 * q + 2:
            break
    seen = set()
    result = []
    for a in active:
        if a not in seen:
            seen.add(a)
            result.append(a)
    return result

def solve_tangent_lp(h0, active_indices, trust_radius=0.2):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    center = n - 1
    corr_vals = np.correlate(h0, 1.0 - h0, mode='full') * dx

    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0

    A_rows = []
    b_rows = []

    for m in active_indices:
        ell = m - center
        g = compute_single_gradient(h0, ell, dx)
        val = corr_vals[m]
        kappa = val - np.dot(g, h0)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-kappa)

    if not A_rows:
        return None, None

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_radius)
        hi = min(1.0, h0[i] + trust_radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def pair_line_search(h, i, j, n_samples=15):
    n = len(h)
    best_val = evaluate_c5(h)
    best_h = h.copy()

    dmin = max(-h[i], h[j] - 1.0)
    dmax = min(1.0 - h[i], h[j])

    if dmax - dmin < 1e-12:
        return best_h, best_val

    if n_samples <= 2:
        deltas = np.array([dmin, dmax])
    else:
        deltas = np.linspace(dmin, dmax, n_samples)

    for d in deltas:
        h_test = h.copy()
        h_test[i] = h[i] + d
        h_test[j] = h[j] - d
        if h_test[i] < -1e-12 or h_test[i] > 1.0 + 1e-12:
            continue
        if h_test[j] < -1e-12 or h_test[j] > 1.0 + 1e-12:
            continue
        val = evaluate_c5(h_test)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_test

    # Refine around best
    if best_h is not h:
        dmin2 = max(dmin, dmin + (dmax - dmin) * 0.3)
        dmax2 = min(dmax, dmin + (dmax - dmin) * 0.7)
        if dmax2 - dmin2 > 1e-10:
            deltas2 = np.linspace(dmin2, dmax2, 11)
            for d in deltas2:
                h_test = h.copy()
                h_test[i] = h[i] + d
                h_test[j] = h[j] - d
                if h_test[i] < -1e-12 or h_test[i] > 1.0 + 1e-12:
                    continue
                if h_test[j] < -1e-12 or h_test[j] > 1.0 + 1e-12:
                    continue
                val = evaluate_c5(h_test)
                if val < best_val - 1e-14:
                    best_val = val
                    best_h = h_test

    return best_h, best_val

def gradient_guided_pairs(h, rng, top_k=15):
    n = len(h)
    dx = 2.0 / n
    corr_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    center = n - 1

    active = select_active(corr_vals, n, q=8)
    avg_grad = np.zeros(n)
    max_corr = max(float(np.max(corr_vals)), 1e-15)
    for m in active:
        ell = m - center
        g = compute_single_gradient(h, ell, dx)
        avg_grad += g * (corr_vals[m] / max_corr)

    order = np.argsort(avg_grad)
    pairs = []
    seen = set()
    numsel = min(12, n)
    for a_idx in range(numsel):
        for b_idx in range(numsel):
            if a_idx == b_idx:
                continue
            a, b = int(order[a_idx]), int(order[b_idx])
            p = (min(a, b), max(a, b))
            if p not in seen:
                seen.add(p)
                pairs.append(p)
            if len(pairs) >= top_k:
                break
        if len(pairs) >= top_k:
            break

    for _ in range(5):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i != j:
            p = (min(i, j), max(i, j))
            if p not in seen:
                seen.add(p)
                pairs.append(p)
                if len(pairs) >= top_k + 5:
                    break

    return pairs[:top_k + 5]

def resolution_transfer(h_src, n_target):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_target = 2.0 / n_target

    h_target = np.zeros(n_target)
    src_idx = 0

    for k in range(n_target):
        t_left = k * dx_target
        t_right = (k + 1) * dx_target
        area = 0.0
        s = src_idx
        while s < n_src:
            s_left = s * dx_src
            s_right = (s + 1) * dx_src
            overlap = min(s_right, t_right) - max(s_left, t_left)
            if overlap > 1e-15:
                area += h_src[s] * overlap
            if s_right >= t_right - 1e-12:
                src_idx = s + 1
                break
            s += 1
        h_target[k] = area / dx_target

    return repair_feasible(h_target)
