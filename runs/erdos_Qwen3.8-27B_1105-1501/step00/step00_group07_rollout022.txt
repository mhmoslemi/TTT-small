import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw) * dx)

def project_feas(v, n):
    target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    np.clip(v, 0.0, 1.0, out=v)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        sh = np.clip(v + mid, 0.0, 1.0)
        if sh.sum() < target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - v.sum()
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(v)
            for i in order:
                if resid <= 1e-15:
                    break
                amt = min(resid, 1.0 - v[i])
                v[i] += amt
                resid -= amt
        else:
            order = np.argsort(-v)
            for i in order:
                if resid >= -1e-15:
                    break
                amt = min(-resid, v[i])
                v[i] -= amt
                resid += amt
    return v

def resample_to(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = 2.0 * (np.arange(old_n) + 0.5) / old_n
    x_new = 2.0 * (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feas(h_new, new_n)

                    continue
                if hb > 1.0 - 1e-10 and ha > 1.0 - 1e-10:
                    continue
                t_lo = -min(ha, 1.0 - hb)
                t_hi = min(1.0 - ha, hb)
                rng_t = t_hi - t_lo
                if rng_t < 1e-9:
                    continue

                best_t = 0.0
                best_sc = C
                step = rng_t / n_coarse
                one_minus = 1.0 - h
                for i in range(n_coarse + 1):
                    t = t_lo + i * step
                    h[a] = ha + t
                    one_minus[a] = 1.0 - h[a]
                    h[b] = hb - t
                    one_minus[b] = 1.0 - h[b]
                    raw = np.correlate(h, one_minus, mode="full")
                    sc = float(np.max(raw)) * dx
                    if sc < best_sc:
                        best_sc = sc
                        best_t = t
                h[a] = ha
                h[b] = hb
                one_minus[a] = 1.0 - ha
                one_minus[b] = 1.0 - hb
