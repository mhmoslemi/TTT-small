import numpy as np
from scipy.optimize import linprog
from scipy import optimize as spo
import time

def score_h(h):
    """Compute C_5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_A_and_grads(h):
    """
    Compute all A_k values and their gradients.
    Returns: A (array of length 2n-1, in np.correlate order), 
             G (matrix of shape (2n-1, n), where G[k] is grad of A_k)
    """
    n = len(h)
    A = np.correlate(h, 1.0 - h, mode='full')
    # np.correlate(h, v, mode='full') output index: 
    # index k corresponds to lag k - (n-1)
    # So output[0] is lag -(n-1), output[n-1] is lag 0, output[2n-2] is lag (n-1)
    
    # For each lag m (from -(n-1) to n-1), compute gradient
    # A[m] = sum_i h[i] * (1 - h[i+m]) over valid i
    # dA[m]/dh[j] = sum over valid i where j appears
    #   if j = i: contribution is (1 - h[i+m]) = (1 - h[j+m])
    #   if j = i+m (i.e., i = j-m): contribution is -h[i] = -h[j-m]
    
    G = np.zeros((2*n - 1, n))
    for idx in range(2 * n - 1):
        m = idx - (n - 1)  # lag
        if m > 0:
            # Valid i: 0 <= i < n-m
            # dA/dh[i] += (1 - h[i+m]) for i in [0, n-m)
            # dA/dh[i] += -h[i-m] for i in [m, n)
            G[idx, :n-m] += 1.0 - h[m:]
            G[idx, m:] -= h[:n-m]
        elif m < 0:
            # Valid i: -m <= i < n
            # dA/dh[i] += (1 - h[i+m]) for i in [-m, n)
            # dA/dh[i] += -h[i-m] for i in [0, n+m)
            t = -m
            G[idx, t:] += 1.0 - h[:n-t]
            G[idx, :n-t] -= h[t:]
        else:
            # m = 0: A[0] = sum h[i](1-h[i])
            # dA/dh[i] = 1 - 2*h[i]
            G[idx] = 1.0 - 2.0 * h
    
    return A, G

def project_box_mass(v, n):
    """Project v onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    v = np.clip(v, 0, 1)
    # Bisection on offset
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        # Distribute residual to non-boundary elements
        free = (result > 1e-8) & (result < 1 - 1e-8)
        if np.sum(free) > 0:
            per = diff / np.sum(free)
            result[free] = np.clip(result[free] + per, 0, 1)
            # Second pass
            diff2 = target - np.sum(result)
            if abs(diff2) > 1e-12:
                free2 = (result > 1e-10) & (result < 1 - 1e-10)
                if np.sum(free2) > 0:
                    per2 = diff2 / np.sum(free2)
                    result[free2] = np.clip(result[free2] + per2, 0, 1)
    # Final exact fix
    diff_final = target - np.sum(result)
    if abs(diff_final) > 1e-14:
        # Find one element to adjust
        for i in range(n):
            if diff_final > 0 and result[i] < 1.0:
                result[i] = min(result[i] + diff_final, 1.0)
                break
            elif diff_final < 0 and result[i] > 0.0:
                result[i] = max(result[i] + diff_final, 0.0)
                break
    return result

def line_search(h_c, d, n, max_score, max_iter=200):
    """
    Line search along h_c + lambda*d.
    Returns best (h_new, score) found.
    """
    # Find feasible range for lambda
    lam_lo = 0.0
    lam_hi = 1.0
    
    # Constrain to box
    for i in range(n):
        if d[i] > 1e-14:
            lam_hi = min(lam_hi, (1.0 - h_c[i]) / d[i])
        elif d[i] < -1e-14:
            lam_lo = max(lam_lo, -h_c[i] / d[i])
    
    if lam_hi <= lam_lo:
        return None
    
    # Constrain to sum = n/2 (d should already be zero-sum, but fix)
    d_sum = np.sum(d)
    if abs(d_sum) > 1e-10:
        d = d - d_sum / n  # make zero-sum
    
    # Evaluate on a grid
    best_score = max_score
    best_lam = 0.0
    best_h = None
    
    # Coarse grid
    lams = np.linspace(lam_lo + 1e-10, lam_hi - 1e-10, 50)
    for lam in lams:
        h_try = h_c + lam * d
        h_try = np.clip(h_try, 0, 1)
        if abs(np.sum(h_try) - n/2) > 0.01:
            h_try = project_box_mass(h_try, n)
        s = score_h(h_try)
        if s < best_score:
            best_score = s
            best_lam = lam
            best_h = h_try.copy()
    
    # Refine around best
    if best_lam > 0:
        margin = min(lam_hi - lam_lo, 0.1)
        ref_lo = max(lam_lo, best_lam - margin)
        ref_hi = min(lam_hi, best_lam + margin)
        lams = np.linspace(ref_lo, ref_hi, 100)
        for lam in lams:
            h_try = h_c + lam * d
            h_try = np.clip(h_try, 0, 1)
            if abs(np.sum(h_try) - n/2) > 0.01:
                h_try = project_box_mass(h_try, n)
            s = score_h(h_try)
            if s < best_score:
                best_score = s
                best_lam = lam
                best_h = h_try.copy()
    
    if best_h is not None and best_score < max_score:
        best_h = project_box_mass(best_h, n)
        best_score = score_h(best_h)
        return best_h, best_score
    
    return None

def make_random_feasible(n, rng):
    """Generate a random feasible point."""
    # Start with 0.5 everywhere and perturb
    h = np.full(n, 0.5)
    h += rng.standard_normal(n) * 0.2
    h = np.clip(h, 0, 1)
    return project_box_mass(h, n)

        active_idx = sorted_idx[:10]  # top 10
        
        # Solve LP trust region

        # Minimize t

        #             sum(h_new) = n/2
        #             h_best - radius <= h_new <= h_best + radius
        #             0 <= h_new <= 1
        
        m = len(active_idx)
        c_obj = np.zeros(n + 1)

        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)

            A_ub[i, :n] = G[k_idx]
            A_ub[i, -1] = -1.0

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_mass(v, n):
    target = n / 2.0
    v = np.array(v, dtype=float)
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) / 2, 0, 1)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        free = (result > 1e-9) & (result < 1 - 1e-9)
        if np.sum(free) > 0:
            per = diff / np.sum(free)
            result[free] = np.clip(result[free] + per, 0, 1)
            diff2 = target - np.sum(result)
            if abs(diff2) > 1e-12:
                # Final fix
                idx = np.argmax(free)
                result[idx] = np.clip(result[idx] + diff2, 0, 1)
    return result
