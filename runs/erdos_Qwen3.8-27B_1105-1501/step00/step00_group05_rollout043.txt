def project_to_constraints(h, target=None):
    h=np.asarray(h,float).copy()
    n=h.size
    if target is None: target=0.5*n
    h=np.clip(h,0.0,1.0)
    diff=target-h.sum()
    while abs(diff)>1e-12:
        if diff>0:
            cand=np.where(h<1.0)[0]
            if len(cand)==0: break
            room=1.0-h[cand]
            # distribute proportional to room? simple equal but clip
            amt=min(diff, room.sum())
            # adjust each by fraction of room
            add=room*(amt/room.sum())
            h[cand]+=add
        else:
            cand=np.where(h>0.0)[0]
            if len(cand)==0: break
            room=h[cand]
            amt=max(diff, -room.sum()) # negative
            sub=room*(-amt/room.sum())
            h[cand]-=sub
        diff=target-h.sum()
    # exact fix: choose index with max room in direction of residual
    diff=target-h.sum()
    if diff>0: idx=np.argmax(1-h)
    elif diff<0: idx=np.argmin(h)
    else idx=0
    h[idx]=min(1,max(0,h[idx]+diff))
    # if still not exact due clip, adjust another
    diff=target-h.sum()
    if abs(diff)>0:
       # distribute over all with room
       if diff>0:
          idxs=np.argsort(1-h)[::-1]
       else: idxs=np.argsort(h)
       for idx in idxs:
          if diff>0:
             add=min(1-h[idx], diff)
             h[idx]+=add; diff-=add
          else:
             sub=min(h[idx], -diff)
             h[idx]-=sub; diff+=sub
          if abs(diff)<1e-15: break
    return h

def smooth_gradient(h,beta):
    n=h.size; c=np.correlate(h,1-h,'full'); m=c.max(); e=np.exp(beta*(c-m)); p=e/e.sum(); G=np.zeros(n)
    for r in range(len(c)):

       if pr<1e-300: continue
       k=r-(n-1)

          # first u=l, v=l-k, l=k..n-1

def compute_pair_coefficients(h, i, j):
    n = h.size
    A = np.zeros(2*n - 1, dtype=float)
    rev_one = (1.0 - h)[::-1]
    A[i:i+n] = A[i:i+n] + rev_one
    A[j:j+n] = A[j:j+n] - rev_one
    si = n - 1 - i
    A[si:si+n] = A[si:si+n] - h
    sj = n - 1 - j
    A[sj:sj+n] = A[sj:sj+n] + h

    B[n-1] = -2.0
    B[i - j + n - 1] += 1.0
    B[j - i + n - 1] += 1.0
    return A, B

def weighted_gradient(h, p):
    n = h.size
    G = np.zeros(n, dtype=float)
    for r in range(2*n - 1):
        pr = float(p[r])
        if pr == 0.0: continue
        k = r - (n - 1)
        if k >= 0:
            length = n - k
            if length <= 0: continue
            G[k:k+length] += pr * (1.0 - h[0:length])
            G[0:length] -= pr * h[k:k+length]
        else:
            d = -k
            length = n - d
            if length <= 0: continue
            G[0:length] += pr * (1.0 - h[d:d+length])
            G[d:d+length] -= pr * h[0:length]
    return G
```

def project_to_constraints(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0: return h
    n=h.size
    if target is None: target = 0.5*n
    h = np.clip(h,0.0,1.0)
    diff = target - float(np.sum(h))
    for _ in range(50):
       if abs(diff) < 1e-14: break
       if diff > 0:
          idx = np.where(h < 1.0)[0]
          if idx.size == 0: break
          room = 1.0 - h[idx]
          if room.sum() <= 0: break
          frac = diff / room.sum()
          if frac > 1.0: frac = 1.0
          h[idx] += room * frac
       else:
          idx = np.where(h > 0.0)[0]
          if idx.size == 0: break
          room = h[idx]
          if room.sum() <= 0: break
          frac = -diff / room.sum()
          if frac > 1.0: frac = 1.0
          h[idx] -= room * frac
       diff = target - float(np.sum(h))
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-15:
       if diff > 0:
          order = np.argsort(1.0 - h)[::-1]
          for idx in order:
             if diff <= 0: break
             add = min(1.0 - h[idx], diff)
             h[idx] += add
             diff -= add
       else:
          order = np.argsort(h)
          for idx in order:
             if diff >= 0: break
             sub = min(h[idx], -diff)
             h[idx] -= sub
             diff += sub
    return h

import numpy as np
import math
import time
import random
from scipy.optimize import minimize, minimize_scalar


def corr_counts(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def max_corr(h):
    h = np.asarray(h, dtype=float)
    return float(np.max(corr_counts(h)))


def c5_exact(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    return float(np.max(corr_counts(h)) * (2.0 / n))


def project_to_constraints(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    n = h.size
    if target is None:
        target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))

    for _ in range(50):
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            room = 1.0 - h[idx]
            s = float(np.sum(room))
            if s <= 0.0:
                break
            frac = min(1.0, diff / s)
            h[idx] += room * frac
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            room = h[idx]
            s = float(np.sum(room))
            if s <= 0.0:
                break
            frac = min(1.0, -diff / s)
            h[idx] -= room * frac
        diff = target - float(np.sum(h))

    diff = target - float(np.sum(h))
    if abs(diff) > 1e-15:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                if diff <= 0.0:
                    break
                add = min(1.0 - h[idx], diff)
                h[idx] += add
                diff -= add
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= 0.0:
                    break
                sub = min(h[idx], -diff)
                h[idx] -= sub
                diff += sub
    return h


def apply_pair_move(h, i, j, t):
    ht = np.asarray(h, dtype=float).copy()
    target_pair = h[i] + h[j]
    a = h[i] + t
    if a < 0.0:
        a = 0.0
    elif a > 1.0:
        a = 1.0
    b = target_pair - a
    if b < 0.0:
        b = 0.0
        a = target_pair
    elif b > 1.0:
        b = 1.0
        a = target_pair - 1.0
    if a < 0.0:
        a = 0.0
    elif a > 1.0:
        a = 1.0
    if b < 0.0:
        b = 0.0
    elif b > 1.0:
        b = 1.0
    ht[i] = a
    ht[j] = b
    return ht


def compute_pair_coefficients(h, i, j):
    h = np.asarray(h, dtype=float)
    n = h.size
    size = 2 * n - 1
    A = np.zeros(size, dtype=float)
    rev_one = (1.0 - h)[::-1]

    A[i:i + n] += rev_one
    A[j:j + n] -= rev_one

    si = n - 1 - i
    A[si:si + n] -= h
    sj = n - 1 - j
    A[sj:sj + n] += h

    B = np.zeros(size, dtype=float)
    B[n - 1] = -2.0
    B[i - j + n - 1] += 1.0
    B[j - i + n - 1] += 1.0
    return A, B


def fast_line_obj(t, c0, A, B):
    c = c0 + A * t + B * t * t
    return float(np.max(c))


def pair_line_search(h, c0, cmax, i, j, grid_points=101, min_improve=1e-10, deadline=None):
    if deadline is not None and time.time() > deadline:
        return h, c0, cmax, False

    hi = h[i]
    hj = h[j]
    low = max(-hi, hj - 1.0)
    high = min(1.0 - hi, hj)
    if high - low < 1e-12:
        return h, c0, cmax, False

    A, B = compute_pair_coefficients(h, i, j)

    m = max(3, int(grid_points))
    ts = np.linspace(low, high, m)
    tt = ts[:, None]
    vals = np.max(c0[None, :] + A[None, :] * tt + B[None, :] * tt * tt, axis=1)

    best_grid = int(np.argmin(vals))
    best_t = float(ts[best_grid])
    best_val = float(vals[best_grid])

    if best_val >= cmax - min_improve:
        return h, c0, cmax, False

    cand_idx = []
    if vals[0] <= vals[1]:
        cand_idx.append(0)
    if vals[m - 1] <= vals[m - 2]:
        cand_idx.append(m - 1)
    interior = (vals[1:-1] <= vals[:-2]) & (vals[1:-1] <= vals[2:])
    for idx in np.where(interior)[0] + 1:
        cand_idx.append(int(idx))
        if len(cand_idx) >= 10:
            break
    cand_idx.append(best_grid)

    cand_list = []
    for idx in cand_idx:
        if 0 <= idx < m:
            cand_list.append((float(vals[idx]), int(idx)))
    cand_list.sort()

    refs = []
    for _, idx in cand_list:
        if idx not in refs:
            refs.append(idx)
        if len(refs) == 3:
            break

    for idx in refs:
        if idx <= 0:
            left = low
            right = min(high, float(ts[2]) if m > 2 else float(ts[1]))
        elif idx >= m - 1:
            left = max(low, float(ts[m - 3]) if m > 2 else float(ts[m - 2]))
            right = high
        else:
            left = float(ts[idx - 1])
            right = float(ts[idx + 1])
        if right - left < 1e-14:
            continue
        try:
            res = minimize_scalar(
                fast_line_obj,
                bounds=(left, right),
                args=(c0, A, B),
                method="bounded",
                options={"xatol": 1e-9, "maxiter": 60}
            )
            if res.success and res.x is not None:
                t_try = float(res.x)
                v_try = float(res.fun)
            else:
                t_try = 0.5 * (left + right)
                v_try = fast_line_obj(t_try, c0, A, B)
            if v_try < best_val:
                best_val = float(v_try)
                best_t = float(t_try)
        except Exception:
            pass

    if best_val >= cmax - min_improve:
        return h, c0, cmax, False

    h_new = apply_pair_move(h, i, j, best_t)
    c0_new = corr_counts(h_new)
    cnew = float(np.max(c0_new))
    if cnew < cmax - min_improve:
        return h_new, c0_new, cnew, True
    return h, c0, cmax, False


def get_active_pairs(c0, n, tol=1e-5):
    m = float(np.max(c0))
    active = np.where(c0 >= m - tol)[0]
    seen = set()
    pairs = []
    for r in active:
        k = int(r - (n - 1))
        for u in range(n):
            v = u - k
            if 0 <= v < n and u != v:
                a, b = (u, v) if u < v else (v, u)
                p = (a, b)
                if p not in seen:
                    seen.add(p)
                    pairs.append(p)
    return pairs


def local_refine(h, c_counts, deadline, grid_points=101, max_sweeps=500, min_improve=1e-10):
    h = project_to_constraints(np.asarray(h, dtype=float).copy())
    c0 = corr_counts(h)
    cmax = float(np.max(c0))

    best_h = h.copy()
    best_c = cmax
    n = h.size
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    sweeps = 0
    while sweeps < max_sweeps and time.time() < deadline:
        sweeps += 1
        h = project_to_constraints(h)
        c0 = corr_counts(h)
        cmax = float(np.max(c0))

        if cmax < best_c - 1e-12:
            best_c = cmax
            best_h = h.copy()

        active = get_active_pairs(c0, n, tol=1e-5)
        active_set = set(active)
        remaining = [p for p in all_pairs if p not in active_set]
        random.shuffle(remaining)
        order = active
        order.extend(remaining)

        improved = False
        for (i, j) in order:
            if time.time() > deadline:
                break
            h2, c0new, cnew, moved = pair_line_search(
                h, c0, cmax, i, j,
                grid_points=grid_points,
                min_improve=min_improve,
                deadline=deadline
            )
            if moved:
                h = h2
                c0 = c0new
                cmax = cnew
                if cmax < best_c - 1e-12:
                    best_c = cmax
                    best_h = h.copy()
                improved = True

        if not improved:
            break

    best_h = project_to_constraints(best_h)
    cfinal = max_corr(best_h)
    if cfinal < best_c:
        best_h = best_h
        best_c = cfinal
    return best_h, best_c


def weighted_gradient(h, p):
    h = np.asarray(h, dtype=float)
    n = h.size
    G = np.zeros(n, dtype=float)
    for r in range(2 * n - 1):
        pr = float(p[r])
        if pr == 0.0:
            continue
        k = r - (n - 1)
        if k >= 0:
            length = n - k
            if length <= 0:
                continue
            G[k:k + length] += pr * (1.0 - h[0:length])
            G[0:length] -= pr * h[k:k + length]
        else:
            d = -k
            length = n - d
            if length <= 0:
                continue
            G[0:length] += pr * (1.0 - h[d:d + length])
            G[d:d + length] -= pr * h[0:length]
    return G


def smooth_objective(h, beta):
    c = corr_counts(h)
    m = float(np.max(c))
    if beta <= 0.0:
        return m
    e = np.exp(beta * (c - m))
    return float(m + math.log(float(np.sum(e))) / beta)


def smooth_jacobian(h, beta):
    c = corr_counts(h)
    m = float(np.max(c))
    if beta <= 0.0:
        p = np.zeros(c.size, dtype=float)
        p[int(np.argmax(c))] = 1.0
    else:
        e = np.exp(beta * (c - m))
        s = float(np.sum(e))
        p = e / s
    return weighted_gradient(h, p)


def max_objective(h, tol=1e-5):
    c = corr_counts(h)
    return float(np.max(c))


def max_jacobian(h, tol=1e-5):
    c = corr_counts(h)
    m = float(np.max(c))
    active = np.where(c >= m - tol)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))], dtype=int)
    p = np.zeros(c.size, dtype=float)
    p[active] = 1.0 / float(active.size)
    return weighted_gradient(h, p)


def eq_constraint(h):
    return float(np.sum(h) - 0.5 * h.size)


def eq_constraint_jac(h):
    return np.ones(h.size, dtype=float)


def run_slsqp(h, beta=30.0, use_active=False, maxiter=60):
    h0 = project_to_constraints(np.asarray(h, dtype=float).copy())
    current_count = max_corr(h0)
    bounds = [(0.0, 1.0)] * h0.size
    cons = [{"type": "eq", "fun": eq_constraint, "jac": eq_constraint_jac}]
    try:
        if use_active:
            res = minimize(
                max_objective,
                h0,
                jac=max_jacobian,
                args=(1e-5,),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-12}
            )
        else:
            res = minimize(
                smooth_objective,
                h0,
                jac=smooth_jacobian,
                args=(float(beta),),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-12}
            )
        if res is not None and hasattr(res, "x") and res.x is not None:
            hx = project_to_constraints(np.asarray(res.x, dtype=float))
            cx = max_corr(hx)
            if cx < current_count - 1e-12:
                return hx, cx
    except Exception:
        pass
    return h0, current_count


def resample_h(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = h_old.size
    if n_old == n_new:
        return project_to_constraints(h_old, 0.5 * n_new)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=float(h_old[0]), right=float(h_old[-1]))
    return project_to_constraints(h_new, 0.5 * n_new)


def binary_start(n):
    h = np.zeros(n, dtype=float)
    m = n // 2
    if m > 0:
        idx = np.random.choice(n, size=m, replace=False)
        h[idx] = 1.0
    return h


def quantize_binary(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    m = n // 2
    hb = np.zeros(n, dtype=float)
    if m > 0:
        idx = np.argsort(h)[::-1][:m]
        hb[idx] = 1.0
    return hb


def binary_local_search(h, deadline, max_sweeps=20, min_improve=1e-10):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = n // 2
    h = np.where(h > 0.5, 1.0, 0.0)
    s = int(np.sum(h))
    if s > target:
        ones = np.where(h == 1.0)[0]
        for idx in ones[:s - target]:
            h[idx] = 0.0
    elif s < target:
        zeros = np.where(h == 0.0)[0]
        for idx in zeros[:target - s]:
            h[idx] = 1.0

    cmax = max_corr(h)
    best_h = h.copy()
    best_c = cmax

    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        cmax = max_corr(h)
        if cmax < best_c - 1e-12:
            best_c = cmax
            best_h = h.copy()

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_val = cmax

        for a in ones:
            h[a] = 0.0
            for b in zeros:
                if h[b] != 0.0:
                    continue
                h[b] = 1.0
                val = max_corr(h)
                h[b] = 0.0
                if val < best_val - min_improve:
                    best_val = val
                    best_swap = (a, b)
            h[a] = 1.0

        if best_swap is None:
            break
        a, b = best_swap
        h[a] = 0.0
        h[b] = 1.0
        cmax = best_val

    cfinal = max_corr(h)
    if cfinal < best_c:
        best_h = h.copy()
        best_c = cfinal
    return best_h, best_c


def random_pair_move(h, scale):
    h = np.asarray(h, dtype=float)
    n = h.size
    i = int(np.random.randint(n))
    j = int(np.random.randint(n - 1))
    if j >= i:
        j += 1

    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    if high <= low + 1e-12:
        return h

    center = 0.5 * (low + high)
    width = 0.5 * (high - low)
    t = center + np.random.uniform(-scale, scale) * width
    if t < low:
        t = low + float(np.random.random()) * width * 0.1
    elif t > high:
        t = high - float(np.random.random()) * width * 0.1
    return apply_pair_move(h, i, j, float(t))


def simulated_anneal(h, c_counts, deadline):
    h = np.asarray(h, dtype=float).copy()
    c = float(c_counts)
    best_h = h.copy()
    best_c = c

    remaining = deadline - time.time()
    if remaining <= 5.0:
        return best_h, best_c

    max_moves = int(min(30000, max(500, remaining * 20)))
    t0 = 0.05
    t1 = 0.0002

    for it in range(max_moves):
        if time.time() > deadline:
            break
        frac = float(it) / float(max(1, max_moves - 1))
        T = t0 * (t1 / t0) ** frac
        scale = min(0.9, 0.05 + 10.0 * T)

        h_trial = random_pair_move(h, scale)
        c_trial = max_corr(h_trial)
        diff = c_trial - c

        accept = False
        if diff <= 0.0:
            accept = True
        elif diff < 700.0 * T:
            accept = np.random.random() < math.exp(-diff / T)

        if accept:
            h = h_trial
            c = c_trial
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()

    return best_h, best_c


def update_best(best_h, best_c_counts, best_c5, h, c_counts):
    h = np.asarray(h, dtype=float)
    c5 = float(c_counts) * (2.0 / h.size)
    if c5 < best_c5 - 1e-12:
        return h.copy(), float(c_counts), float(c5)
    return best_h, best_c_counts, best_c5


def make_sub_deadline(deadline, fraction, minimum, reserve):
    now = time.time()
    rem = deadline - now
    if rem <= 0.0:
        return now
    if rem <= reserve + 2.0:
        return now + max(0.2, rem)
    d = now + min(rem - reserve, max(minimum, fraction * rem))
    if d < now + 0.4:
        d = now + 0.4
    return d


def default_initial(n=42):
    h = binary_start(n)
    return project_to_constraints(h, 0.5 * n)


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(int(seed))
    np.random.seed(int(seed))

    t_start = time.time()
    total_budget = max(20.0, float(budget_s))
    deadline = t_start + max(20.0, total_budget - 25.0)

    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None or h0.size == 0:
        h0 = default_initial(42)

    h0 = project_to_constraints(h0)
    c0 = max_corr(h0)
    best_h = h0.copy()
    best_c_counts = c0
    best_c5 = c0 * (2.0 / h0.size)

    # Primary local refinement from the parent construction.
    dl1 = min(deadline, t_start + max(30.0, 0.40 * (deadline - t_start)))
    h1, c1 = local_refine(best_h, best_c_counts, dl1, grid_points=101, max_sweeps=300)
    best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h1, c1)

    # SLSQP multi-start refinement around the current best point.
    dl2 = min(deadline, t_start + max(90.0, 0.60 * (deadline - t_start)))
    for beta in (10.0, 30.0, 80.0):
        if time.time() > dl2:
            break
        h2, c2 = run_slsqp(best_h, beta=beta, use_active=False, maxiter=50)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h2, c2)
    if time.time() < dl2:
        h2, c2 = run_slsqp(best_h, use_active=True, maxiter=60)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h2, c2)

    # Second local refinement after SLSQP.
    dl3 = min(deadline, t_start + max(150.0, 0.82 * (deadline - t_start)))
    h3, c3 = local_refine(best_h, best_c_counts, dl3, grid_points=121, max_sweeps=300)
    best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h3, c3)

    # Try quantizing the current continuous point to binary and refilling.
    if best_h.size % 2 == 0 and time.time() < deadline - 90:
        hb = quantize_binary(best_h)
        sub = make_sub_deadline(deadline, 0.08, 10.0, 90.0)
        hb2, cb2 = binary_local_search(hb, sub, max_sweeps=8)
        h4, c4 = local_refine(hb2, cb2, make_sub_deadline(deadline, 0.08, 10.0, 90.0),
                              grid_points=91, max_sweeps=12)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h4, c4)

    # Try a few neighboring discretizations.
    for new_n in (44, 46, 48):
        if new_n % 2 != 0:
            continue
        if time.time() > deadline - 100:
            break
        h_try = resample_h(best_h, new_n)
        c_try = max_corr(h_try)
        sub = make_sub_deadline(deadline, 0.06, 10.0, 100.0)
        h4, c4 = local_refine(h_try, c_try, sub, grid_points=81, max_sweeps=6)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h4, c4)

        if time.time() < deadline - 90:
            hb = binary_start(new_n)
            sub2 = make_sub_deadline(deadline, 0.04, 8.0, 90.0)
            hb2, cb2 = binary_local_search(hb, sub2, max_sweeps=5)
            h5, c5b = local_refine(hb2, cb2, make_sub_deadline(deadline, 0.05, 8.0, 90.0),
                                   grid_points=81, max_sweeps=5)
            best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h5, c5b)

    # Light continuous simulated annealing, followed by local refinement.
    if time.time() < deadline - 130:
        sa_dl = make_sub_deadline(deadline, 0.20, 20.0, 130.0)
        hs, cs = simulated_anneal(best_h, best_c_counts, sa_dl)
        h6, c6 = local_refine(hs, cs, make_sub_deadline(deadline, 0.12, 12.0, 110.0),
                              grid_points=91, max_sweeps=15)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h6, c6)

    # Small-noise restarts around the best point.
    for sigma in (0.05, 0.20):
        if time.time() > deadline - 60:
            break
        h_try = project_to_constraints(best_h + np.random.normal(0.0, sigma, best_h.size))
        c_try = max_corr(h_try)
        sub = make_sub_deadline(deadline, 0.07, 10.0, 60.0)
        h7, c7 = local_refine(h_try, c_try, sub, grid_points=91, max_sweeps=12)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, h7, c7)

    # Final high-resolution polishing.
    if time.time() < deadline:
        hF, cF = local_refine(best_h, best_c_counts, deadline, grid_points=201, max_sweeps=120)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, hF, cF)

    if time.time() < deadline - 15:
        hG, cG = run_slsqp(best_h, use_active=True, maxiter=30)
        best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, hG, cG)
        if time.time() < deadline - 5:
            hH, cH = local_refine(best_h, best_c_counts, deadline, grid_points=201, max_sweeps=20)
            best_h, best_c_counts, best_c5 = update_best(best_h, best_c_counts, best_c5, hH, cH)

    best_h = project_to_constraints(best_h)
    final_c5 = c5_exact(best_h)
    return best_h, float(final_c5), int(best_h.size)
