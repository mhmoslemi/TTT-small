from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def project_feasible(v, target_sum):
    """Project v onto {0 <= x_i <= 1, sum(x) = target_sum} via water filling."""
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, weights, dx):
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    grad = np.zeros(n)
    m = np.arange(n)
    for d in range(-(n - 1), n):
        w = weights[d + n - 1] * dx
        if abs(w) < 1e-15:
            continue
        if d >= 0:
            g = np.zeros(n)
            mask1 = m >= d
            mask2 = (m + d) < n
            g[mask1] += (1.0 - h[m[mask1] - d])
            g[mask2] -= h[m[mask2] + d]
        else:
            d2 = -d
            g = np.zeros(n)
            mask1 = m < (n - d2)
            mask2 = m >= d2
            g[mask1] += (1.0 - h[m[mask1] + d2])
            g[mask2] -= h[m[mask2] - d2]
        grad += w * g
    return grad

    """Smooth gradient descent with annealing beta."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    n_lags = 2 * n - 1
    
    for it in range(n_iters):
        # Anneal beta
        frac = it / max(n_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        c_vals = corr * (2.0 / n)

        c_max = np.max(c_vals)
        shifted = beta * (c_vals - c_max)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -700, 700)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)

        dx = 2.0 / n
        grad = compute_gradient(h, weights, dx)

    return c_max + float(np.log(np.sum(np.exp(shifted)))) / beta

def smooth_grad(h, beta, n):

        mask = rng.random(n) < 0.4
        if np.any(mask):
            h_pert[mask] += rng.normal(0, ps, np.sum(mask))
        h_pert = project_feasible(h_pert, target_sum)
        c5_p = eval_c5(h_pert)

        if c5_p < best_c5:
            h_opt = slsqp_opt(h_pert, n, target_sum, 100000, 150)
            h_opt = project_feasible(h_opt, target_sum)
            c5 = eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                h = h_opt.copy()
        else:
            # Small probability of exploring from perturbed point anyway
            if rng.random() < 0.1 and c5_p < best_c5 + 0.001:
                h_opt = slsqp_opt(h_pert, n, target_sum, 50000, 100)
                h_opt = project_feasible(h_opt, target_sum)
                c5 = eval_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    h = h_opt.copy()

    # Phase 5: Try binary rounding and re-optimize
    if still_going(0.99):
        for thr in [0.01, 0.05, 0.1, 0.2]:
            h_br = binary_round(best_h, n, target_sum, thr)
            c5_br = eval_c5(h_br)
            if c5_br < best_c5:
                h_opt = slsqp_opt(h_br, n, target_sum, 100000, 200)
                h_opt = project_feasible(h_opt, target_sum)
                c5 = eval_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Final verification and cleanup
    best_h = project_feasible(best_h, target_sum)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, n
