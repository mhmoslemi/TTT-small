import numpy as np
from scipy.optimize import minimize
import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def build_full_h(u, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = u
    h[n - m:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def get_u_from_h(h):
    n = len(h)
    m = n // 2
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return h_sym[:m].copy()

def correlation_values(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_soft_max(u, n, beta):
    h = build_full_h(u, n)
    dx = 2.0 / n
    corr = correlation_values(h)
    scaled = beta * dx * corr
    max_s = np.max(scaled)
    shifted = scaled - max_s
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    m = n // 2
    grad = np.zeros(m)
    for r in range(len(corr)):
        if weights[r] < 1e-14:
            continue
        k = r - (n - 1)
        w = weights[r]
        for a in range(m):
            g = 0.0
            j = a + k
            if 0 <= j < n:
                g += (1.0 - h[j])
            j = (n - 1 - a) + k
            if 0 <= j < n:
                g -= (1.0 - h[j])
            j = a - k
            if 0 <= j < n:
                g -= h[j]
            j = (n - 1 - a) - k
            if 0 <= j < n:
                g += h[j]
            grad[a] += w * dx * g
    obj = (np.log(s) + max_s) / beta
    return obj, grad

    m = n // 2
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        h = build_full_h(u_curr, n)
        dx = 2.0 / n
        corr = correlation_values(h)
        max_val = np.max(corr)
        obj = dx * max_val
        if obj < best_score:
            best_score = obj
            best_u = u_curr.copy()
        tol = max(1e-8, 1e-6 * max_val)
        active = np.where(corr >= max_val - tol)[0]
        grad = np.zeros(m)
        for r in active:
            k = r - (n - 1)
            for a in range(m):
                g = 0.0
                j = a + k
                if 0 <= j < n:
                    g += (1.0 - h[j])
                j = (n - 1 - a) + k
                if 0 <= j < n:
                    g -= (1.0 - h[j])
                j = a - k
                if 0 <= j < n:
                    g -= h[j]
                j = (n - 1 - a) - k
                if 0 <= j < n:
                    g += h[j]
                grad[a] += g
        grad *= dx / len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        d = -grad / gnorm
        improved = False
        for alpha in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.8]:
            u_trial = np.clip(u_curr + alpha * d, 0, 1)
            s_trial = compute_score(build_full_h(u_trial, n))
            if s_trial < obj - 1e-14:
                u_curr = u_trial
                obj = s_trial
                improved = True
                break
        if not improved:
            break
    return best_u, best_score

def active_direction_search(u, n, rng, deadline):
    best_u = np.clip(u.copy(), 0, 1)
    best_score = compute_score(build_full_h(best_u, n))
    m = n // 2
    for it in range(20):
        if time.time() > deadline:
            break
        h = build_full_h(best_u, n)
        dx = 2.0 / n
        corr = correlation_values(h)
        max_val = np.max(corr)
        tol = max(1e-6, 1e-4 * max_val)
        active = np.where(corr >= max_val - tol)[0]
        if len(active) == 0:
            break
        # Build active gradient matrix
        G_act = np.zeros((len(active), m))
        for idx, r in enumerate(active):
            k = r - (n - 1)
            for a in range(m):
                g = 0.0
                j = a + k
                if 0 <= j < n:
                    g += (1.0 - h[j])
                j = (n - 1 - a) + k
                if 0 <= j < n:
                    g -= (1.0 - h[j])
                j = a - k
                if 0 <= j < n:
                    g -= h[j]
                j = (n - 1 - a) - k
                if 0 <= j < n:
                    g += h[j]
                G_act[idx, a] = dx * g
        # Solve least squares: minimize ||G_act @ d - target||
        target = -0.02 * (np.max(corr) - max_val) - 0.005 * max_val
        target_vec = np.full(len(active), 1.0) * target
        lam = 1e-6 * np.mean(np.sum(G_act**2, axis=1)) + 1e-10
        A = G_act.T @ G_act + lam * np.eye(m)
        b_vec = G_act.T @ target_vec
        try:
            d = np.linalg.solve(A, b_vec)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(A, b_vec, rcond=None)[0]
        # Also try negative sum of active gradients
        d_neg = -G_act.sum(axis=0)
        dneg_norm = np.linalg.norm(d_neg)
        if dneg_norm > 1e-12:
            d_neg /= dneg_norm
        d_norm = np.linalg.norm(d)
        if d_norm > 1e-12:
            d /= d_norm
        # Line search along both directions
        candidates = [d, d_neg]
        for direction in candidates:
            for alpha in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5, 1.0]:
                u_trial = np.clip(best_u + alpha * direction, 0, 1)
                s_trial = compute_score(build_full_h(u_trial, n))
                if s_trial < best_score - 1e-14:
                    best_score = s_trial
                    best_u = u_trial.copy()
                    break
    return best_u, best_score

def generate_starting_points(n, rng):
    m = n // 2
    points = []
    # Uniform 0.5
    points.append(np.full(m, 0.5))
    # Step function
    u_step = np.zeros(m)
    u_step[:m // 2] = 1.0
    points.append(u_step)
    # Cosine pattern
    x = np.linspace(0, 1, m)
    points.append(0.5 + 0.4 * np.cos(2 * np.pi * x))
    points.append(0.5 + 0.3 * np.sin(2 * np.pi * x))
    # Random
    for _ in range(5):
        points.append(rng.random(m))
    return points

def slsqp_polish(u, n, deadline):
    m = n // 2
    h = build_full_h(np.clip(u, 0, 1), n)
    dx = 2.0 / n
    corr = correlation_values(h)
    max_c = np.max(corr)
    theta0 = dx * max_c + 1e-8
    z0 = np.concatenate([np.clip(u, 0, 1), [theta0]])
    
    def slsqp_fun(z):
        return z[-1]
    
    def slsqp_fun_grad(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    constraints = []
    for r in range(2 * n - 1):
        k = r - (n - 1)
        def make_cf(k):
            def cfun(z):
                u_z = np.clip(z[:m], 0, 1)
                h_z = build_full_h(u_z, n)
                start = max(0, -k)
                end = min(n, n - k)
                F = np.sum(h_z[start:end] * (1.0 - h_z[start + k:end + k]))
                return z[-1] - dx * F
            def cgrad(z):
                u_z = np.clip(z[:m], 0, 1)
                h_z = build_full_h(u_z, n)
                g = np.zeros(m + 1)
                g[-1] = 1.0
                for a in range(m):
                    val = 0.0
                    j = a + k
                    if 0 <= j < n:
                        val += (1.0 - h_z[j])
                    j = (n - 1 - a) + k
                    if 0 <= j < n:
                        val -= (1.0 - h_z[j])
                    j = a - k
                    if 0 <= j < n:
                        val -= h_z[j]
                    j = (n - 1 - a) - k
                    if 0 <= j < n:
                        val += h_z[j]
                    g[a] = -dx * val
                return g
            return {'type': 'ineq', 'fun': cfun, 'jac': cgrad}
        constraints.append(make_cf(k))
    
    res = minimize(slsqp_fun, z0, jac=slsqp_fun_grad, method='SLSQP',
                   bounds=[(0, 1)] * m + [(0, 2)],
                   constraints=constraints,
                   options={'maxiter': 300, 'ftol': 1e-12})
    if res.success and res.fun < theta0:
        u_new = np.clip(res.x[:m], 0, 1)
        s = compute_score(build_full_h(u_new, n))
        return u_new, s
    return np.clip(u, 0, 1), compute_score(build_full_h(np.clip(u, 0, 1), n))

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return np.clip(h_new, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    best_h = np.clip(initial_h_values.copy(), 0, 1)
    best_n = len(best_h)
    # Ensure sum constraint
    target_sum = best_n / 2.0
    cur_sum = np.sum(best_h)
    best_h = best_h + (target_sum - cur_sum) / best_n
    best_h = np.clip(best_h, 0, 1)
    best_score = compute_score(best_h)
    
    n_init = best_n
    u_init = get_u_from_h(best_h)
    
    # Phase 1: Optimize initial in symmetric complement (multiple rounds)
    for round_num in range(4):
        if time.time() > deadline - 120:
            break
        # Soft-max optimization
        u_opt, s_opt = optimize_softmax(u_init, n_init, rng, deadline)
        h_opt = build_full_h(u_opt, n_init)
        real_score = compute_score(h_opt)
        if real_score < best_score:
            best_score = real_score
            best_h = h_opt.copy()
            best_n = n_init
        # Subgradient polish
        u_sg, s_sg = subgradient_line_search(u_opt, n_init, max_iter=300, deadline=deadline)
        h_sg = build_full_h(u_sg, n_init)
        real_sg = compute_score(h_sg)
        if real_sg < best_score:
            best_score = real_sg
            best_h = h_sg.copy()
            best_n = n_init
        u_init = u_sg.copy()
        # Active direction search
        u_ad, s_ad = active_direction_search(u_sg, n_init, rng, deadline)
        h_ad = build_full_h(u_ad, n_init)
        real_ad = compute_score(h_ad)
        if real_ad < best_score:
            best_score = real_ad
            best_h = h_ad.copy()
            best_n = n_init
        u_init = u_ad.copy()
    
    # Phase 2: SLSQP polish on best solution found
    if time.time() < deadline - 60:
        u_best = get_u_from_h(best_h)
        u_slq, s_slq = slsqp_polish(u_best, best_n, deadline)
        h_slq = build_full_h(u_slq, best_n)
        real_slq = compute_score(h_slq)
        if real_slq < best_score:
            best_score = real_slq
            best_h = h_slq.copy()
            best_n = best_n
    
    # Phase 3: Try random restarts in symmetric complement
    for restart in range(8):
        if time.time() > deadline - 60:
            break
        u_rand = rng.random(n_init // 2)
        # Blend with current best
        blend = 0.3 + 0.7 * rng.random()
        u_blend = blend * u_rand + (1 - blend) * get_u_from_h(best_h)
        u_blend = np.clip(u_blend, 0, 1)
        
        u_opt2, s_opt2 = optimize_softmax(u_blend, n_init, rng, deadline)
        h_opt2 = build_full_h(u_opt2, n_init)
        real_opt2 = compute_score(h_opt2)
        if real_opt2 < best_score:
            best_score = real_opt2
            best_h = h_opt2.copy()
            best_n = n_init
        
        u_sg2, _ = subgradient_line_search(u_opt2, n_init, max_iter=100, deadline=deadline)
        h_sg2 = build_full_h(u_sg2, n_init)
        real_sg2 = compute_score(h_sg2)
        if real_sg2 < best_score:
            best_score = real_sg2
            best_h = h_sg2.copy()
            best_n = n_init
    
    # Phase 4: Try different n values (interpolate best solution)
    for n_try in [n_init + 1, n_init + 2, 2 * n_init]:
        if time.time() > deadline - 40:
            break
        if n_try == n_init:
            continue
        h_interp = interpolate_h(best_h, best_n, n_try)
        u_interp = get_u_from_h(h_interp)
        u_opt3, _ = optimize_softmax(u_interp, n_try, rng, deadline)
        h_opt3 = build_full_h(u_opt3, n_try)
        real_opt3 = compute_score(h_opt3)
        if real_opt3 < best_score:
            best_score = real_opt3
            best_h = h_opt3.copy()
            best_n = n_try
            u_best_now = u_opt3.copy()
            u_sg3, _ = subgradient_line_search(u_best_now, n_try, max_iter=100, deadline=deadline)
            h_sg3 = build_full_h(u_sg3, n_try)
            real_sg3 = compute_score(h_sg3)
            if real_sg3 < best_score:
                best_score = real_sg3
                best_h = h_sg3.copy()
                best_n = n_try
    
    # Final: ensure feasibility
    target_sum = best_n / 2.0
    cur_sum = np.sum(best_h)
    if abs(cur_sum - target_sum) > 1e-10:
        best_h = best_h + (target_sum - cur_sum) / best_n
    best_h = np.clip(best_h, 0, 1)
    # Re-apply symmetric complement for exact feasibility
    u_final = get_u_from_h(best_h)
    best_h = build_full_h(u_final, best_n)
    best_score = compute_score(best_h)
    
    return best_h, best_score, best_n
