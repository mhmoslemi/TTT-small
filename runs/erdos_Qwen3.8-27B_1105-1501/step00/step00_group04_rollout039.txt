import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def scores_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_and_grad(h, T):
    """Compute smooth max of scores and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(c)
    center = n - 1
    
    base = float(np.max(c))
    exps = np.exp((c - base) / T)
    S = float(np.sum(exps))
    if S < 1e-300:
        weights = np.zeros(L)
        weights[np.argmax(c)] = 1.0
        smooth_val = base
    else:
        weights = exps / S
        smooth_val = base + T * np.log(S)
    
    # Compute gradient
    grad = np.zeros(n)
    
    # Center (d=0)
    grad += weights[center] * dx * (1.0 - 2.0 * h)
    
    # Non-zero lags
    for d in range(1, n):
        wp = weights[center + d]
        wn = weights[center - d]
        if wp < 1e-15 and wn < 1e-15:
            continue
        a = n - d
        grad[:a] += dx * (-wp * h[d:] + wn * (1.0 - h[d:]))
        grad[d:] += dx * (wp * (1.0 - h[:a]) - wn * h[:a])
    
    return smooth_val, grad

def project_feasible(y, target=None):
    """Project y onto {h: 0<=h_i<=1, sum(h)=target}."""
    n = len(y)
    if target is None:
        target = n / 2.0
    
    # Bisection on lambda
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    
    def phi(lam):
        return float(np.sum(np.clip(y - lam, 0.0, 1.0)))
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if phi(mid) > target:
            lo = mid
        else:
            hi = mid
    
    lam = 0.5 * (lo + hi)
    h = np.clip(y - lam, 0.0, 1.0)
    
    # Exact sum correction
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        # Find elements not at bounds
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            idx = np.where(interior)[0]
            # Distribute diff
            per = diff / len(idx)
            h[idx] += per
            # Re-clip and correct again if needed
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - float(np.sum(h))
            if abs(diff2) > 1e-14:
                interior2 = (h > 1e-12) & (h < 1.0 - 1e-12)
                if np.any(interior2):
                    idx2 = np.where(interior2)[0]
                    h[idx2] += diff2 / len(idx2)
                    h = np.clip(h, 0.0, 1.0)
        else:
            # All at bounds, find one to adjust
            # This shouldn't happen if target is in [0, n]
            pass
    
    return h

def local_refine(h0, deadline, rng, T_start=0.10, T_end=0.001, T_decay=0.85):
    """Local refinement using smooth-max gradient descent."""
    h = h0.copy()
    best_h = h.copy()
    best_c = score_exact(h)
    n = len(h)
    
    T = T_start
    step_size = 0.02
    no_improve_count = 0
    max_no_improve = 50
    
    while time.time() < deadline:
        # Compute smooth max and gradient
        try:
            sm_val, grad = smooth_max_and_grad(h, T)
        except Exception:
            T *= T_decay
            if T < T_end * 0.01:
                break
            continue
        
        if not np.all(np.isfinite(grad)):
            T *= T_decay
            continue
        
        # Gradient descent step
        g_norm = np.sqrt(np.sum(grad**2))
        if g_norm < 1e-14:
            T *= T_decay
            if T < T_end * 0.01:
                break
            continue
        
        # Try a few step sizes
        accepted = False
        for trial in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:
            h_trial = h - trial * grad
            h_trial = project_feasible(h_trial)
            c_trial = score_exact(h_trial)
            if c_trial < best_c - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_c = c_trial
                accepted = True
                no_improve_count = 0
                break
            elif c_trial < score_exact(h) - 1e-12:
                h = h_trial
                accepted = True
                break
        
        if not accepted:
            no_improve_count += 1
            step_size *= 0.7
            if step_size < 1e-6:
                T *= T_decay
                step_size = 0.02
        
        # Also try with slightly larger step occasionally
        if no_improve_count == 20:
            step_size = 0.05
        
        # Update temperature
        if no_improve_count > max_no_improve:
            T *= T_decay
            no_improve_count = 0
            if T < T_end * 0.01:
                break
        
        # Check time
        if time.time() > deadline:
            break
    
    return best_h, best_c

def pair_line_search(h, i, j):
    """Exact line search along direction e_i - e_j."""
    n = len(h)
    # alpha: h[i] += alpha, h[j] -= alpha
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    
    if hi - lo < 1e-14:
        return h.copy(), score_exact(h)
    
    # Sample points
    best_alpha = 0.0
    best_c = score_exact(h)
    
    n_samples = 31
    alphas = np.linspace(lo, hi, n_samples)
    
    for alpha in alphas:
        h[i] += alpha
        h[j] -= alpha
        c = score_exact(h)
        h[i] -= alpha
        h[j] += alpha
        if c < best_c - 1e-12:
            best_c = c
            best_alpha = alpha
    
    if best_alpha != 0.0:

    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    base = float(c.max())
    exps = np.exp((c - base) / T)
    S = float(exps.sum())
    if S < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = exps / S
    ctr = n - 1
    g = np.zeros(n)
    g += w[ctr] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[ctr + d]
        wn = w[ctr - d]
        if wp + wn < 1e-15:
            continue
        a = n - d
        g[:a] += dx * (-wp * h[d:] + wn * (1.0 - h[d:]))
        g[d:] += dx * (wp * (1.0 - h[:a]) - wn * h[:a])
    return g

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    lo = float(y.min()) - 1.5
    hi = float(y.max()) + 0.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(y - mid, 0, 1))) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - float(h.sum())
    if abs(d) > 1e-14:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += d / int(m.sum())
            h = np.clip(h, 0.0, 1.0)
    return h
