def project_feasible(h, n):
    """Project h onto {x : sum(x) = n/2, 0 <= x_i <= 1}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Bisection on the shift
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # We need to add (target - s) total, distributed across elements
    # Use water-filling: find t such that sum(clip(h + t, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0, 1))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
