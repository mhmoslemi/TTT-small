                        best_c5 = c5
            except Exception:
                pass
    
    # Phase 3: Binary/near-binary local search
    if elapsed() < budget_s * 0.95:
        # Create binary starting point
        h_bin = np.round(best_h).clip(0, 1)
        # Fix sum
        diff = target_sum - h_bin.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                zeros = np.where(h_bin < 1.0)[0]
            else:
                zeros = np.where(h_bin > 0.0)[0]
            for idx in zeros:
                adj = min(abs(diff), 1.0 - h_bin[idx]) if diff > 0 else max(diff, -h_bin[idx])
                h_bin[idx] += adj
                diff = target_sum - h_bin.sum()
                if abs(diff) < 1e-10:
                    break
        h_bin = project(h_bin, n)
        
        # Pairs local search (thorough for n=47)
        if elapsed() < budget_s * 0.90:
            h_bin, c5_bin = try_pairs_local_search(h_bin, n, target_sum, max_rounds=3)
            if c5_bin < best_c5:
                best_h = h_bin
                best_c5 = c5_bin
        
        # Random swap search
        remaining_iters = int((budget_s * 0.93 - elapsed()) * 500)
        if remaining_iters > 100:
            h_bin, c5_bin = binary_local_search(h_bin, n, min(remaining_iters, 20000), rng, target_sum)
            if c5_bin < best_c5:
                best_h = h_bin
                best_c5 = c5_bin
    
    # Phase 4: Final SLSQP polish from best solution
    if elapsed() < budget_s * 0.99:
        for temp in [0.001, 0.0001]:
            if elapsed() > budget_s * 0.99:
                break
            try:
                res = slsqp_solve(best_h, n, temp, target_sum, maxiter=2000)
                if res.x is not None:
                    h_new = project(res.x, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new
                        best_c5 = c5
            except Exception:
                pass
    
    # Verify and return
    best_h = project(best_h, n)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
