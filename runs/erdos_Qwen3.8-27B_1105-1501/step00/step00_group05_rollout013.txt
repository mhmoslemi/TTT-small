    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def overlap_vector(h):

    dx = 2.0 / n
    return np.correlate(h, 1-h, mode='full') * dx

def project_feasible(p, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(p + mid, 0, 1)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(p + (lo + hi) / 2, 0, 1)

def gradient_overlap(h, n, active_indices):
    """Compute gradient of max of active overlap values."""
    dx = 2.0 / n
    grad = np.zeros(n)
    offset = n - 1
    for idx in active_indices:
        lag = idx - offset
        if lag == 0:
            grad += (1 - 2 * h)
        elif lag > 0:
            for m in range(n - lag):
                grad[m + lag] += (1 - h[m])
                grad[m] -= h[m + lag]
        else:
            s = -lag
            for m in range(n - s):
                grad[m] += (1 - h[m + s])
                grad[m + s] -= h[m]
    return grad * dx

def subgradient_descent(h, n, iterations=10000, step_init=0.01, tol=1e-8):
    h = h.copy()
    best_h = h.copy()
    best_c = exact_objective(h)

        ov = overlap_vector(h)
        c = ov.max()

        # Check improvement

            h_new = project_feasible(h - alpha * g, n)
            h = h_new
        
        if it > 100 and c >= best_c - tol:
            if it % 1000 == 0:
                break
    
    return best_h, best_c

        improved = False

            for j in range(i+1, n):
                if h[i] < 0 or h[j] > 1:
                    continue

    h = x
    ov = np.correlate(h, 1-h, mode='full')
    c = ov.max()
    dx = 2.0/n

        if lag == 0:
            g += (1 - 2*h)
        elif lag > 0:

                g[m+lag] += (1-h[m])
                g[m] -= h[m+lag]
        else:
            s = -lag
            for m in range(n-s):
                g[m] += (1-h[m+s])
                g[m+s] -= h[m]
    g *= dx / len(active)
    return g
