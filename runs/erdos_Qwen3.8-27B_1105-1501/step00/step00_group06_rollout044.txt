def compute_full_and_grad(h):
    """Compute full correlation array and gradient matrix.
    Returns F (length 2n-1) and G (shape (2n-1, n)) where G[m,:] is grad of F[m] w.r.t. h
    """

        lag = m - (n - 1)

            # Contribution from h[j] as second factor: -h[j-lag] if 0 <= j-lag < n
            im = j - lag
            if 0 <= im < n:
                g -= h[im]
            G[m, j] = g
    return F, G

    Returns direction d (zero-sum, within box) or None."""

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    # G[m,:] @ h_new - t <= -F[m] - G[m,:] @ h_c + G[m,:] @ h_c
    # Wait: F[m] + G[m,:] @ h_new - G[m,:] @ h_c <= t

        A_ub[m, :n] = G[m, :]
        A_ub[m, n] = -1.0
    b_ub = G[:, :n] @ h_c - F  # = G @ h_c - F
    
    # Equality: sum(h_new) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    lo = np.maximum(0.0, h_c - radius)
    hi = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lo, hi)) + [(0, None)]  # t >= 0, unbounded above

    d = h_new - h_c
    # Ensure zero sum (should be by constraint, but fix numerics)
    d -= np.mean(d)
    # Ensure feasibility of direction
    return d

def feasible_range(h_c, d):

    G = np.zeros((2 * n - 1, n))

        jm = idx - lag
        vm = (jm >= 0) & (jm < n)
        if np.any(vm):
            G[m, vm] -= h[jm[vm]]
    return F, G

    n = len(h_c)
    F, G = compute_full_grad(h_c)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h_c - F
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h_c - radius)
    hi = np.minimum(1.0, h_c + radius)
    bounds = [(lo[i], hi[i]) for i in range(n)] + [(-1e6, 1e6)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            d = res.x[:n].copy() - h_c
            d -= np.mean(d)
            return d
    except Exception:
        pass
    return None

    lam_lo = 0.0

def two_coord_search(h, rng, n_pairs=20, n_scan=60):
