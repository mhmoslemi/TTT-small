def exact_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    """Project y to [0,1]^n with sum = n/2"""
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64).copy()
    # Replace non-finite
    bad = ~np.isfinite(y)
    if np.any(bad):
        y[bad] = 0.5
    
    # Find lambda such that sum(clamp(y - lambda, 0, 1)) = target
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix residual mass
    residual = target - np.sum(z)
    if abs(residual) > 1e-12:
        # Find elements with slack
        if residual > 0:
            slack = 1.0 - z
        else:
            slack = z.copy()
        # Sort by slack (prefer elements with most slack for increase, least slack for decrease)
        if residual > 0:
            order = np.argsort(-slack)
        else:
            order = np.argsort(slack)
        remaining = abs(residual)
        for idx in order:
            if remaining <= 1e-14:
                break
            move = min(remaining, slack[idx])
            if residual > 0:
                z[idx] += move
            else:
                z[idx] -= move
            remaining -= move
            if remaining <= 1e-14:
                break
    
    # Final fix
    res = target - np.sum(z)
    if abs(res) > 1e-13:
        # Find best element to adjust
        if res > 0:
            idx = np.argmax(1.0 - z)
            z[idx] += min(res, 1.0 - z[idx])
        else:
            idx = np.argmin(z)
            z[idx] += max(res, -z[idx])
    
    z = np.clip(z, 0.0, 1.0)
    return z

import time

def exact_c5(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64).flatten().copy()
    y[~np.isfinite(y)] = 0.5
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    for _ in range(3):
        resid = target - z.sum()
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            slack = 1.0 - z
            idx = np.argmax(slack)
            move = min(resid, slack[idx])
            z[idx] += move
        else:
            idx = np.argmin(z)
            move = max(resid, -z[idx])
            z[idx] += move
    z = np.clip(z, 0.0, 1.0)
    # Final exact fix
    resid = target - z.sum()
    if abs(resid) > 1e-13:
        if resid > 0:
            idx = np.argmax(1.0 - z)
            z[idx] = min(1.0, z[idx] + resid)
        else:
            idx = np.argmin(z)
            z[idx] = max(0.0, z[idx] + resid)
    return z

def compute_c_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

    c_vals = compute_c_values(x)
    M = np.max(c_vals)
    shifted = np.clip((c_vals - M) / tau, -100, 100)
    return M + tau * np.log(np.sum(np.exp(shifted)))

def smooth_obj_grad(x, dx, tau):
    n = len(x)
    c_vals = compute_c_values(x)
    M = np.max(c_vals)
    shifted = np.clip((c_vals - M) / tau, -100, 100)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        l = m - (n - 1)

            rem = w * dx
            if l == 0:
                grad += rem * (1.0 - 2.0 * x)
            else:
                grad[l:] += rem * (1.0 - x[:n - l])
                grad[:n - l] -= rem * x[l:]
        else:
            p = -l
            rem = w * dx
            grad[:n - p] += rem * (1.0 - x[p:])
            grad[p:] -= rem * x[:n - p]
    return grad

def do_smoothed_solve(h0, tau_list, start_time, budget):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x, t=target: smooth_constraint_eq(x, t),
                    'jac': lambda x, t=target: smooth_constraint_eq_jac(x, t)}]
    
    best_h = h0.copy()
    best_score = exact_c5(h0)
    
    h = h0.copy()
    for tau in tau_list:
        elapsed = time.time() - start_time
        if elapsed > budget * 0.85:
            break
        try:
            res = minimize(
                fun=smooth_obj, x0=h, args=(dx, tau),
                jac=smooth_obj_grad, method='SLSQP',
                bounds=bounds, constraints=constraints,

def pair_move_optimize(h0, n_moves=5000, start_time=None, budget=1100):
    """Try random pair moves to improve"""
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_c5(h)
    n = len(h)
    rng = np.random.RandomState(12345)

        h_new[i] += delta
        h_new[j] -= delta

def transfer_resolution(h_src, n_tgt, start_time=None, budget=1100):
    """Transfer h to different resolution preserving area"""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    # Build cumulative area
    x_src = np.arange(n_src + 1) * dx_src
    area_src = np.concatenate([[0], np.cumsum(h_src * dx_src)])
    
    # Interpolate at target grid
    x_tgt_mid = (np.arange(n_tgt) + 0.5) * dx_tgt
    area_at = np.interp(x_tgt_mid, x_src, area_src)
    h_tgt = area_at / dx_tgt
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    h_tgt = project_feasible(h_tgt)
    
    return h_tgt

def systematic_local_search(h0, start_time, budget, max_iter=3000):
    """Systematic local search with decreasing step sizes"""
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_c5(h)
    n = len(h)
    
    step_sizes = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for ss in step_sizes:
        elapsed = time.time() - start_time
        if elapsed > budget * 0.85:
            break
        for _ in range(max_iter // len(step_sizes)):
            if time.time() - start_time > budget * 0.85:
                break
            # Try moving mass from random i to random j
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            delta = ss * np.random.choice([-1, 1])
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            h_new[j] = np.clip(h_new[j] - delta, 0, 1)
            # Check mass is preserved (approximately)
            if abs(h_new.sum() - n / 2.0) > 1e-6:
                h_new = project_feasible(h_new)
            score = exact_c5(h_new)
            if score < best_score - 1e-12:
                h = h_new
                best_h = h.copy()
                best_score = score
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64).flatten().copy()
    y[~np.isfinite(y)] = 0.5
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(5):
        resid = target - z.sum()
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            slack = 1.0 - z
            idx = int(np.argmax(slack))
            move = min(resid, slack[idx])
            z[idx] += move
        else:
            idx = int(np.argmin(z))
            move = max(resid, -z[idx])
            z[idx] += move
    z = np.clip(z, 0.0, 1.0)
    resid = target - z.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            idx = int(np.argmax(1.0 - z))
            z[idx] = min(1.0, z[idx] + resid)
        else:
            idx = int(np.argmin(z))
            z[idx] = max(0.0, z[idx] + resid)
    return z

def compute_c_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_obj(x, dx, tau):
    c_vals = compute_c_values(x)
    M = np.max(c_vals)
    shifted = np.clip((c_vals - M) / tau, -100.0, 100.0)
    return M + tau * float(np.log(np.sum(np.exp(shifted))))

def smooth_obj_grad(x, dx, tau):
    n = len(x)
    c_vals = compute_c_values(x)
    M = np.max(c_vals)
    shifted = np.clip((c_vals - M) / tau, -100.0, 100.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_vals / total
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        l = m - (n - 1)
        rem = w * dx
        if l >= 0:
            if l == 0:
                grad += rem * (1.0 - 2.0 * x)
            else:
                grad[l:] += rem * (1.0 - x[:n - l])
                grad[:n - l] -= rem * x[l:]
        else:
            p = -l
            grad[:n - p] += rem * (1.0 - x[p:])
            grad[p:] -= rem * x[:n - p]
    return grad

def make_eq_constraint(target):
    def fun(x):
        return x.sum() - target
    def jac(x):
        return np.ones_like(x)
    return fun, jac

def do_smoothed_solve(h0, tau_list, start_time, budget, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_fun, eq_jac = make_eq_constraint(target)
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    best_h = h0.copy()
    best_score = exact_c5(h0)
    h = h0.copy()

    for tau in tau_list:
        elapsed = time.time() - start_time
        if elapsed > budget * 0.88:
            break
        remaining = (budget * 0.92 - elapsed)
        if remaining < 2:
            break
        try:
            res = minimize(
                fun=smooth_obj, x0=h, args=(dx, tau),
                jac=smooth_obj_grad, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_cand = project_feasible(res.x)
                score = exact_c5(h_cand)
                if score < best_score - 1e-11:
                    best_h = h_cand.copy()
                    best_score = score
                h = h_cand.copy() if score < best_score + 0.5 else h
        except Exception:
            pass
    return best_h, best_score

def pair_move_optimize(h0, n_moves=5000, start_time=None, budget=1100, seed=99):
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_c5(h)
    n = len(h)
    rng = np.random.RandomState(seed)
    step_scales = [0.3, 0.15, 0.07, 0.03, 0.01, 0.005, 0.002]
    moves_per_scale = max(50, n_moves // len(step_scales))

    for ss in step_scales:
        if start_time is not None and time.time() - start_time > budget * 0.92:
            break
        for _ in range(moves_per_scale):
            if start_time is not None and time.time() - start_time > budget * 0.92:
                break
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            delta = ss * rng.choice([-1, 1])
            hi_new = h[i] + delta
            hj_new = h[j] - delta
            if hi_new < -1e-9 or hi_new > 1.0 + 1e-9 or hj_new < -1e-9 or hj_new > 1.0 + 1e-9:
                continue
            h[i] = hi_new
            h[j] = hj_new
            score = exact_c5(h)
            if score < best_score - 1e-13:
                best_score = score
                best_h = h.copy()
            else:
                h[i] = hi_new - delta
                h[j] = hj_new + delta

    return best_h, best_score

def active_constraint_solve(h0, start_time, budget, max_rounds=10):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_fun, eq_jac = make_eq_constraint(target)

    best_h = h0.copy()
    best_score = exact_c5(h0)
    h = h0.copy()

    for rnd in range(max_rounds):
        elapsed = time.time() - start_time
        if elapsed > budget * 0.88:
            break
        c_vals = compute_c_values(h)
        max_idx = int(np.argmax(c_vals))
        max_val = c_vals[max_idx]
        l = max_idx - (n - 1)

        def ac_obj(x):
            cv = compute_c_values(x)
            return cv[max_idx]

        def ac_grad(x):
            grad = np.zeros(n)
            if l >= 0:
                if l == 0:
                    grad = dx * (1.0 - 2.0 * x)
                else:
                    grad[l:] = dx * (1.0 - x[:n - l])
                    grad[:n - l] = -dx * x[l:]
            else:
                p = -l
                grad[:n - p] = dx * (1.0 - x[p:])
                grad[p:] = -dx * x[:n - p]
            return grad

        constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        try:
            res = minimize(
                fun=ac_obj, x0=h, jac=ac_grad, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_cand = project_feasible(res.x)
                score = exact_c5(h_cand)
                if score < best_score - 1e-11:
                    best_h = h_cand.copy()
                    best_score = score
                h = h_cand.copy()
        except Exception:
            pass

        # Check if we've converged
        new_c = compute_c_values(h)
        if np.max(new_c) >= max_val - 1e-6:
            break

    return best_h, best_score

def binarize(h):
    n = len(h)
    target = n / 2.0
    # Sort by value, assign 1 to top target values
    if target == int(target):
        k = int(target)
        order = np.argsort(-h)
        b = np.zeros(n)
        b[order[:k]] = 1.0
    else:
        k = int(np.floor(target))
        order = np.argsort(-h)
        b = np.zeros(n)
        b[order[:k]] = 1.0
        b[order[k]] = target - k
    return b

def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    x_src = np.arange(n_src + 1, dtype=np.float64) * dx_src
    area_src = np.concatenate([[0.0], np.cumsum(h_src * dx_src)])
    x_tgt_mid = (np.arange(n_tgt, dtype=np.float64) + 0.5) * dx_tgt
    area_at = np.interp(x_tgt_mid, x_src, area_src)
    h_tgt = area_at / dx_tgt
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    h_tgt = project_feasible(h_tgt)
    return h_tgt

def generate_starts(h0, seed):
    n = len(h0)
    rng = np.random.RandomState(seed)
    starts = [h0.copy()]
    starts.append(project_feasible(h0[::-1].copy()))
    starts.append(project_feasible(1.0 - h0))

    for sigma in [0.2, 0.1, 0.05, 0.02]:
        h_p = h0 + rng.normal(0, sigma, n)
        starts.append(project_feasible(h_p))

    # Binary low-discrepancy
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    fracs = np.array([((i * phi) % 1.0) for i in range(n)])
    k = int(np.floor(n / 2.0))
    h_bd = np.zeros(n)
    idx = np.argsort(fracs)
    h_bd[idx[:k]] = 1.0
    if n % 2 != 0:
        h_bd[idx[k]] = 0.5
    starts.append(h_bd.copy())

    # Alternating binary
    h_alt = np.zeros(n)
    h_alt[0::2] = 1.0
    h_alt = project_feasible(h_alt)
    starts.append(h_alt.copy())

    # Block pattern: first n/2 ones, rest zeros
    h_blk = np.zeros(n)
    h_blk[:n // 2] = 1.0
    if n % 2 != 0:
        h_blk[n // 2] = 0.5
    starts.append(h_blk.copy())

    # Random binary
    h_rb = np.zeros(n)
    perm = rng.permutation(n)
    h_rb[perm[:k]] = 1.0
    if n % 2 != 0:
        h_rb[perm[k]] = 0.5
    starts.append(h_rb.copy())

    # Trimmed sine
    for freq in [2, 3, 5]:
        x = np.linspace(0, 2, n)
        h_sin = 0.5 + 0.4 * np.sin(np.pi * freq * x)
        starts.append(project_feasible(h_sin))

    return starts
