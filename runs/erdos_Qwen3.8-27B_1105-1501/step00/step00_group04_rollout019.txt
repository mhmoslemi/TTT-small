import numpy as np
import time

def make_feasible(y, n=None):
    """Project y onto {h : 0<=h<=1, sum(h)=n/2}"""
    y = np.asarray(y, dtype=np.float64).copy()
    if n is None:
        n = len(y)
    target = n / 2.0
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(70):
        mid = (lo + hi) / 2.0
        val = np.clip(y - mid, 0.0, 1.0)
        s = val.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    # Fix residual
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        # Find elements with slack
        if diff > 0:
            # Need to increase some elements
            idx = np.where(z < 1.0 - 1e-10)[0]
        else:
            # Need to decrease some elements
            idx = np.where(z > 1e-10)[0]
        if len(idx) > 0:
            # Distribute among elements with most slack
            if diff > 0:
                slacks = 1.0 - z[idx]
            else:
                slacks = z[idx]
            # Distribute proportionally
            order = np.argsort(slacks)[::-1]
            remaining = diff
            for i in order:
                actual_i = idx[i]
                if diff > 0:
                    adj = min(remaining, 1.0 - z[actual_i])
                else:
                    adj = max(remaining, -(1e-10))
                    adj = max(-z[actual_i], remaining)
                z[actual_i] += adj
                remaining -= adj
                if abs(remaining) < 1e-12:
                    break
    return z

def score_exact(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    dx = 2.0 / n
