import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(cor)

def smooth_max_weights(cor_values, tau):
    """Return softmax weights for smooth max gradient"""
    m = np.max(cor_values)
    shifted = tau * (cor_values - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    return exp_vals / np.sum(exp_vals)

def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum} using bisection"""
    if np.sum(h) == target_sum and np.all(h >= 0) and np.all(h <= 1):
        return h
    lo = -max(h) - target_sum/len(h) - 1
    hi = 1 - min(h) + target_sum/len(h) + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo+hi)/2, 0, 1)

def gradient_step(h, weights, dx):
    """Compute gradient of smooth max of correlate values"""
    n = len(h)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for shift_idx in range(2*n-1):
        s = shift_idx - (n-1)
        w = weights[shift_idx]
        if abs(w) < 1e-15:
            continue
        # Term 1: w * (1-h[j-s]) for valid j-s in [0,n)
        j_lo = max(0, s)
        j_hi = min(n, s + n)
        if j_lo < j_hi:
            h_idx_lo = j_lo - s
            h_idx_hi = j_hi - s
            term1[j_lo:j_hi] += w * (1.0 - h[h_idx_lo:h_idx_hi])
        # Term 2: w * h[j+s] for valid j+s in [0,n)
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_lo2 < j_hi2:
            h_idx_lo2 = j_lo2 + s
            h_idx_hi2 = j_hi2 + s
            term2[j_lo2:j_hi2] += w * h[h_idx_lo2:h_idx_hi2]
    return dx * (term1 - term2)

def optimize_h(h0, tau_schedule, lr, max_steps, target_sum):
    """Projected gradient descent"""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    h = project_onto_constraint(h, target_sum)
    
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for step in range(max_steps):
        cor = np.correlate(h, 1.0 - h, mode='full') * dx
        tau = tau_schedule(step)
        weights = smooth_max_weights(cor, tau)
        grad = gradient_step(h, weights, dx)
        
        # Adaptive learning rate
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        # Decay learning rate slowly
        lr *= 0.9999
    
    return best_h, best_val

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(cor), cor

def smooth_max_weights(cor_values, tau):
    m = np.max(cor_values)
    shifted = tau * (cor_values - m)
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total == 0:
        return np.ones(len(cor_values)) / len(cor_values)
    return exp_vals / total

def project_onto_constraint(h, target_sum):
    n = len(h)
    # Try to find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10 and np.all(h >= -1e-10) and np.all(h <= 1 + 1e-10):
        return np.clip(h, 0, 1)
    
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    # Extend range to ensure we can reach target_sum
    lo = min(lo, target_sum/n - np.max(h) - 0.1)
    hi = max(hi, target_sum/n - np.min(h) + 0.1)
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi)/2.0, 0.0, 1.0)
    return result

def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    # For each shift s from -(n-1) to n-1:
    # ∂c(s)/∂h[j] = dx * [(1-h[j-s]) * 1_{0≤j-s<n} - h[j+s] * 1_{0≤j+s<n}]
    # weights is indexed by k = s + (n-1), from 0 to 2n-2
    
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        # Term 1: (1-h[j-s]) for j where 0 ≤ j-s < n, i.e., s ≤ j < s+n
        # Also 0 ≤ j < n
        j_lo = max(0, s)
        j_hi = min(n, s + n)
        if j_lo < j_hi:
            idx_lo = j_lo - s
            idx_hi = j_hi - s
            grad[j_lo:j_hi] += w * (1.0 - h[idx_lo:idx_hi])
        # Term 2: -h[j+s] for j where 0 ≤ j+s < n, i.e., -s ≤ j < n-s
        # Also 0 ≤ j < n
        j_lo2 = max(0, -s)
        j_hi2 = min(n, n - s)
        if j_lo2 < j_hi2:
            idx_lo2 = j_lo2 + s
            idx_hi2 = j_hi2 + s
            grad[j_lo2:j_hi2] -= w * h[idx_lo2:idx_hi2]
    
    grad *= dx
    return grad

def projected_gradient_optimize(h0, time_limit, tau_start=5.0, tau_end=50.0, lr_start=0.05):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    h = project_onto_constraint(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    
    start = time.time()
    lr = lr_start
    tau = tau_start
    stuck_count = 0
    
    while time.time() - start < time_limit:
        cor = np.correlate(h, 1.0 - h, mode='full') * dx
        weights = smooth_max_weights(cor, tau)
        grad = compute_gradient(h, weights, dx)
        
        # Try the step
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)[0]
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        h = h_new
        
        # Anneal tau
        tau = min(tau * 1.005, tau_end)
        
        # Adjust learning rate
        if stuck_count > 50:
            lr *= 0.8
            stuck_count = 0
        elif stuck_count == 0 and lr < lr_start:

        # Reset tau periodically to escape local minima
        if stuck_count > 200:
            tau = tau_start
            lr = lr_start
            stuck_count = 0
    
    return best_h, best_c5

def make_binary_start(n, rng):
    """Create a random binary starting point with exactly n/2 ones"""
    if n % 2 != 0:

        n += 1
    positions = rng.permutation(n)

def make_structured_start(n, rng):
    """Create a structured starting point"""
    h = np.zeros(n)
    # Try a pattern based on quadratic residues or similar
    # Simple: block structure
    block_size = max(1, n // 8)

        zero_positions = np.where(h == 0.0)[0]
        add = rng.choice(zero_positions, size=target - n_ones, replace=False)
        h[add] = 1.0
    return h

    """Local search on binary sequences: flip pairs to maintain sum constraint"""
    n = len(h0)

            break

            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_flip_val
            best_h = h.copy()
        else:
            # Try single flip (changes sum by 1, then project)
            # Or just break
            break

    positions = rng.permutation(n)
    h[positions[:n // 2]] = 1.0
    return h

def make_random_soft(n, rng):
    h = rng.rand(n)
    return project_onto_constraint(h, n / 2.0)

def local_binary_search(h0, time_limit, rng):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    target = n // 2
    start = time.time()
    stall = 0

    while time.time() - start < time_limit:
        if stall > 200:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_flip_val = best_c5
        best_flip = None
        n_trials = min(50, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c5 = compute_c5(h)
            if c5 < best_flip_val:
                best_flip_val = c5
                best_flip = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        if best_flip is not None:
            i, j = best_flip
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_flip_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    best_h = None
    best_c5 = 1.0
    best_n = 0

    # Phase 1: Optimize from initial_h_values if available
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if h0 is not None and len(h0) > 0:
            t_limit = min(200, max(30, remaining() * 0.25))
            h_opt, c5_opt = optimize_smooth(h0, t_limit, 5.0, 40.0, 0.03, rng, noise_scale=0.001)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = len(h_opt)
    except Exception:
        pass

    # Phase 2: Try multiple n values with different starts
    n_candidates = [48, 64, 96, 128, 192, 256]

    for n in n_candidates:
        if remaining() < 25:
            break

        n_starts = 3
        t_per_start = max(15, min(50, remaining() / (n_starts * 1.2)))

        for start_idx in range(n_starts):
            if remaining() < 15:
                break

            if start_idx == 0:
                h0 = make_golden_start(n, rng)
            elif start_idx == 1:
                h0 = make_random_binary(n, rng)
            else:
                h0 = make_random_soft(n, rng)

            h_opt, c5_opt = optimize_smooth(h0, t_per_start, 3.0, 25.0, 0.02, rng, noise_scale=0.002)

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

            # Try local binary search on rounded version
            if remaining() > 20 and n % 2 == 0:
                h_rounded = (h_opt > 0.5).astype(float)
                n_ones = int(np.sum(h_rounded))
                target = n // 2
                if n_ones > target:
                    one_pos = np.where(h_rounded == 1.0)[0]
                    rem = rng.choice(one_pos, size=n_ones - target, replace=False)
                    h_rounded[rem] = 0.0
                elif n_ones < target:
                    zero_pos = np.where(h_rounded == 0.0)[0]
                    add = rng.choice(zero_pos, size=target - n_ones, replace=False)
                    h_rounded[add] = 1.0
                h_bin, c5_bin = local_binary_search(h_rounded, min(20, remaining() * 0.3), rng)
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin.copy()
                    best_n = n

    # Phase 3: Final refinement on best solution
    if best_h is not None and remaining() > 40:
        t_final = remaining() * 0.75
        h_final, c5_final = optimize_smooth(best_h, t_final, 8.0, 60.0, 0.015, rng, noise_scale=0.001)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Fallback
    if best_h is None:
        n = 48
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
