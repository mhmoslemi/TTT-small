def smooth_max(R, alpha):
    m = R.max()
    return m + np.log(np.sum(np.exp(alpha * (R - m)))) / alpha

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-np.clip(x, -50, 50)))

def find_lambda(z, target_sum):
    """Find lambda such that sum(sigmoid(z + lambda)) = target_sum"""
    def f(lam):
        return np.sum(sigmoid(z + lam)) - target_sum
    # Find bracket
    lo, hi = -40.0, 40.0
    flo, fhi = f(lo), f(hi)
    if flo * fhi > 0:

        for _ in range(10):
            lo -= 10
            hi += 10
            flo, fhi = f(lo), f(hi)
            if flo * fhi <= 0:

    return 1.0 / (1.0 + np.exp(-np.clip(x, -30, 30)))

def _find_lambda(z, target):
    """Bisection for lambda: sum(sigmoid(z+lambda)) = target"""
    def f(lam):
        return _sigmoid(z + lam).sum() - target
    lo, hi = -40.0, 40.0
    if f(lo) > 0:
        return lo
    if f(hi) < 0:
        return hi
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)

def _h_from_z(z):
    n = len(z)
    lam = _find_lambda(z, n * 0.5)
    return _sigmoid(z + lam)

def _smooth_obj(z, alpha):
    h = _h_from_z(z)
    R = np.correlate(h, 1.0 - h, mode='full')
    mx = R.max()
    return float(mx + np.log(np.sum(np.exp(alpha * (R - mx)))) / alpha)

def _h_from_template(h_template, n_new):
    """Interpolate h_template to n_new points"""
    n_old = len(h_template)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_template)
    return _repair(h_new, n_new * 0.5)

def _optimize_logit(h_init, rng, deadline, alphas=None):
    """Optimize using logit parameterization"""
    n = len(h_init)
    if alphas is None:
        alphas = [30, 60, 120]
    
    h_clip = np.clip(h_init, 1e-6, 1 - 1e-6)
    z = np.log(h_clip / (1.0 - h_clip))
    z = z - z.mean()  # Center
    
    best_z = z.copy()
    best_h = h_init.copy()
    best_c = _score(h_init)
    
    from scipy.optimize import minimize
    
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        
        def obj(z_):
            return _smooth_obj(z_, alpha)
        
        try:
            res = minimize(obj, best_z, method='L-BFGS-B',
                          options={'maxiter': 300, 'maxls': 50, 'ftol': 1e-10, 'gtol': 1e-8})
        except Exception:
            break
        
        h_new = _h_from_z(res.x)
        h_new = _repair(h_new, n * 0.5)
        c_new = _score(h_new)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            best_z = res.x.copy()
    
    return best_h, best_c
