    lo = z.min() - 2.0

def project_feasible(z, n):
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64).copy()
    # Water-filling projection
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    # Exact sum repair
    res = target - h.sum()
    if abs(res) > 1e-14:
        if res > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        rem = abs(res)
        for idx in order:
            if res > 0:
                d = min(rem, 1.0 - h[idx])
                h[idx] += d
            else:
