```python
from scipy.optimize import linprog

def find_descent_direction(h, active_indices, n):
    """Find direction d that decreases all active correlations."""
    # Variables: d[0], ..., d[n-1]
    # Constraints:
    #   grad_k · d <= -1 for all k in active (strict decrease)
    #   sum(d) = 0
    #   |d| <= 1 (normalize)
    #   d is free (unbounded)
    
    # Actually, let's minimize t subject to:
    #   grad_k · d <= t for all k in active
    #   sum(d) = 0
    #   -1 <= d_i <= 1
    # If t < 0, we have a descent direction.
    
    # This is an LP!
    c = np.zeros(n + 1)  # objective: minimize d[n] = t
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # grad_k · d - t <= 0 for each active k
    A_ub = np.zeros((len(active_indices), n + 1))
    b_ub = np.zeros(len(active_indices))
    for i, idx in enumerate(active_indices):
        shift = idx - (n - 1)
        g = compute_single_gradient(h, shift, n)
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
    
    # Box constraints on d: -1 <= d_i <= 1
    #  d_i <= 1 and -d_i <= 1
    A_box = np.zeros((2*n, n + 1))
    b_box = np.zeros(2*n)
    A_box[:n, :n] = np.eye(n)
    b_box[:n] = 1.0
    A_box[n:, :n] = -np.eye(n)
    b_box[n:] = 1.0
    
    A_ub = np.vstack([A_ub, A_box])
    b_ub = np.concatenate([b_ub, b_box])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    
    if res.success and res.fun < -1e-6:
        return res.x[:n]  # descent direction
    else:
        return None
```

import time

def exact_objective(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Then adjust with water-filling
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        if np.sum(projected) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_single_gradient(h, shift, n):
    """Gradient of correlation at given shift w.r.t h."""
    grad = np.zeros(n)
    if shift >= 0:
        # c = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
        # d/dh[j]:
        #   as h[i] term: (1-h[j+shift]) for j = 0..n-1-shift
        #   as h[i+shift] term: -h[j-shift] for j = shift..n-1
        for j in range(n - shift):
            grad[j] += (1.0 - h[j + shift])
        for j in range(shift, n):
            grad[j] -= h[j - shift]
    else:
        s = -shift
        # c = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
        # d/dh[j]:
        #   as h[i] term: (1-h[j-s]) for j = s..n-1
        #   as h[i-s] term: -h[j+s] for j = 0..n-1-s
        for j in range(s, n):
            grad[j] += (1.0 - h[j - s])
        for j in range(n - s):
            grad[j] -= h[j + s]
    return grad

def lp_direction(h, active_indices, n):

    # min t s.t. grad_k . d <= t, |d_i| <= 1, sum(d) = 0

    c_obj[-1] = 1.0
    
    rows_ub = []
    bounds_ub = []

        shift = idx - (n - 1)
        g = compute_single_gradient(h, shift, n)
        row = np.zeros(n + 1)

    # Determine valid range of delta
    delta_min = max(-h[i], -(1.0 - h[j]))
    delta_max = min(1.0 - h[i], h[j])
    
    if delta_max - delta_min < 1e-12:
        return h, exact_objective(h)
    
    res = minimize_scalar(obj_delta, bounds=(delta_min, delta_max), method='bounded',
                          options={'xatol': 1e-10})
    
    if res.success:
        h_new = h.copy()
        h_new[i] += res.x
        h_new[j] -= res.x
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c_new = exact_objective(h_new)
        return h_new, c_new
    return h, exact_objective(h)
