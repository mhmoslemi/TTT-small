- For ell < 0, p = -ell: C_m = dx * sum(h[:n-p] * (1-h[p:]))

  ∂C_m/∂h[j] = dx*(1-h[j-ell]) for j ∈ [ell, n-1]

For ell < 0, p = -ell: C_m = dx * sum_{i=0}^{n-p-1} h[i]*(1-h[i+p])

import numpy as np
import time
import math
from scipy.optimize import linprog

def evaluate_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_feasible(h, target):
    """Repair h to be feasible: 0<=h<=1, sum(h)=target"""
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    s = np.sum(h)
    diff = s - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to reduce sum by diff
        order = np.argsort(h)[::-1]  # largest first
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            red = min(h[idx], remaining)
            h[idx] -= red
            remaining -= red
    else:
        # Need to increase sum by -diff
        order = np.argsort(h)  # smallest first
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            inc = min(1.0 - h[idx], remaining)
            h[idx] += inc
            remaining -= inc
    # Final exact fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if residual > 0:
                chg = min(1.0 - h[idx], residual)
                h[idx] += chg
                residual -= chg
            else:
                chg = min(h[idx], -residual)
                h[idx] -= chg
                residual += chg
            if abs(residual) < 1e-14:
                break
    return h

def compute_all_correlations(h):
    """Return array of all correlation values (unscaled by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_gradient_for_shift(h, ell):
    """Compute gradient of correlation term at shift ell w.r.t. h (before dx scaling)"""
    n = len(h)
    g = np.zeros(n)
    if ell >= 0:
        # C = sum_{i=ell}^{n-1} h[i]*(1-h[i-ell])
        g[ell:] += (1.0 - h[:n-ell])
        g[:n-ell] -= h[ell:]
    else:
        p = -ell
        # C = sum_{i=0}^{n-p-1} h[i]*(1-h[i+p])
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g * (2.0 / n)

def select_active(h, corr_values, max_k):
    """Select top-k active constraints (indices into correlation array)"""
    n = len(h)
    center = n - 1
    indices = list(range(len(corr_values)))
    sorted_idx = np.argsort(corr_values)[::-1]
    active = list(sorted_idx[:max_k])
    if center not in active:
        active.append(center)
    return active

def solve_tangent_lp(h0, active_indices, corr_values, trust_radius):
    """Solve the tangent LP subproblem. Returns candidate h or None."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    k = len(active_indices)
    
    # Build constraint matrix
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: G_m . h - t <= -kappa_m for each active m
    # kappa_m = c_m - G_m . h0
    A_ub = np.zeros((k, n + 1))
    b_ub = np.zeros(k)
    
    for ii, m in enumerate(active_indices):
        G = compute_gradient_for_shift(h0, m - (n - 1))
        c_m = corr_values[m] * dx
        kappa = c_m - np.dot(G, h0)
        A_ub[ii, :n] = G
        A_ub[ii, n] = -1.0
        b_ub[ii] = -kappa
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    lo = np.maximum(0.0, h0 - trust_radius)
    hi = np.minimum(1.0, h0 + trust_radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))  # t is free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        return res.x[:n].copy()
    return None

def pair_line_search(h, i, j, n_samples=17):
    """Exact line search for pair transfer. Returns best delta or None."""
    n = len(h)
    dx = 2.0 / n
    # h[i] += delta, h[j] -= delta
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    
    if delta_max - delta_min < 1e-12:
        return None
    
    current_val = evaluate_c5(h)
    
    # Sample points
    if n_samples < 2:
        n_samples = 2
    deltas = np.linspace(delta_min, delta_max, n_samples)
    
    best_delta = 0.0
    best_val = current_val
    
    for d in deltas:
        h2 = h.copy()
        h2[i] += d
        h2[j] -= d
        h2 = np.clip(h2, 0.0, 1.0)
        val = evaluate_c5(h2)
        if val < best_val - 1e-12:
            best_val = val
            best_delta = d
    
    if best_delta != 0.0:
        h[i] += best_delta
        h[j] -= best_delta
    return best_val

def resolution_transfer(h_src, n_target):
    """Area-preserving transfer from h_src to n_target samples"""
    n_src = len(h_src)
    dx_s = 2.0 / n_src
    dx_t = 2.0 / n_target
    h_t = np.zeros(n_target)
    
    src_pos = 0.0
    src_idx = 0
    
    for k in range(n_target):
        t_left = k * dx_t
        t_right = (k + 1) * dx_t
        # Advance source to t_left
        while src_idx < n_src and (src_idx + 1) * dx_s < t_left:
            src_idx += 1
        # Integrate over overlap
        area = 0.0
        cur = t_left
        si = src_idx
        while cur < t_right and si < n_src:
            s_left = si * dx_s
            s_right = (si + 1) * dx_s
            overlap_left = max(cur, s_left)
            overlap_right = min(t_right, s_right)
            if overlap_right > overlap_left:
                area += h_src[si] * (overlap_right - overlap_left)
            cur = overlap_right
            si += 1
        h_t[k] = area / dx_t
    
    return h_t

def random_perturb(h, scale, seed):
    """Apply small random perturbation preserving feasibility"""
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n / 2.0
    noise = rng.normal(0, scale, n)
    h2 = h + noise
    return repair_feasible(h2, target)

    n = len(h)

    start_time = time.time()
    
    current = h.copy()

    best_val = current_val
    
    # Determine active set size
    if n <= 50:

        candidate = solve_tangent_lp(current, active, scaled_corr, trust_radius)

                current = candidate
                current_val = cand_val

def build_fallback(n=45):
    """Build a feasible fallback vector"""
    target = n / 2.0
    # Alternating pattern
    h = np.zeros(n)
    h[::2] = 1.0
    # Fix sum
    h = repair_feasible(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    total_budget = budget_s

    best_h = None

        pass

        h1, v1 = local_optimize(best_h, phase1_time, seed=seed)

            best_val = v1
        elif v1 <= best_val + 1e-8:
            # Keep best_h (might have different structure)
            pass
    
    # Phase 2: Try different resolutions
    if time.time() - start_time < total_budget * 0.8:
        remaining = total_budget - (time.time() - start_time)
        phase2_budget = remaining * 0.5
        
        # Try a few nearby resolutions
        n_candidates = [n_best]
        for factor in [0.75, 1.0, 1.25, 1.5, 2.0]:
            ni = int(round(n_best * factor))
            ni = max(20, min(ni, 200))
            if ni % 2 == 0:

                best_val = v_final
    
    # Final validation and repair

    final_val = evaluate_c5(best_h)
    
    if not np.isfinite(final_val):
        best_h = build_fallback(45)
        final_val = evaluate_c5(best_h)
    
    return best_h, final_val, len(best_h)
