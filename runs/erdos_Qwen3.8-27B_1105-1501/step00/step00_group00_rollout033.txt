def compute_c5(h):

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(y - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr * dx
    max_s = float(scaled.max())
    exp_s = np.exp(scaled - max_s)
    w = exp_s / float(exp_s.sum())

    hi = float(y.max()) + 2.0

def smooth_obj_grad(x, beta):
    N = len(x)
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode="full")
