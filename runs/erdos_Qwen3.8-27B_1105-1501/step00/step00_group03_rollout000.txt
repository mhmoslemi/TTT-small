Let me analyze this problem carefully:

1. We need to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx
2. The constraint is ∫₀² h(x) dx = 1, which discretizes to sum(h) * dx = 1, i.e., sum(h) = n/2
3. We start from `initial_h_values` (n=45 samples) with current best C₅ ≤ 0.38085
4. Goal: achieve C₅ ≤ 0.3808 (lower is better)

Let me design a clean, working implementation:

1. **Official score**: `max(np.correlate(h, 1-h, mode='full')) * dx`
2. **Manual values**: For each lag k from -(n-1) to (n-1), compute sum over valid i of h[i]*(1-h[i+k])
3. **Manual gradient**: For each lag k, dF_k/dh_p = sum over valid pairs involving p
4. **Smooth objective**: log-sum-exp type surrogate with increasing beta
5. **Direct minimax**: SLSQP with theta variable and all lag constraints
6. **Feasibility projection**: Bisection on lambda to clip(z+lambda) to [0,1] and hit target sum

Key design decisions:
- Use scipy.optimize.minimize with SLSQP
- Smooth max surrogate with increasing beta
- Direct minimax NLP
- Multi-resolution: try parent n, smaller, larger
- Multi-start with random perturbations
- Track best feasible solution throughout

def official_score(h):
    """Compute C5 = max korrelate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(vals) * dx)

def manual_values(h):
    """Compute unscaled F_k for all lags k = -(n-1) to (n-1)"""
    n = len(h)
    result = np.zeros(2*n - 1)
    for lag in range(-(n-1), n):
        r = lag + n - 1
        s = 0.0
        if lag >= 0:
            for i in range(n - lag):
                s += h[i] * (1.0 - h[i + lag])
        else:
            for i in range(n + lag):
                s += h[i] * (1.0 - h[i - lag])
        result[r] = s
    return result

def manual_gradient(h):
    """Compute gradient matrix of shape (2n-1, n) for F_k wrt h_p"""
    n = len(h)

def smooth_grad(h, beta):

    n_large = min(72, int(n_base * 1.3))
    if n_large != n_base and n_large != n_small:
        n_targets.append(n_large)
    # Deduplicate
    n_targets = list(dict.fromkeys(n_targets))

    betas = [20.0, 80.0, 400.0]
    n_rand_starts = max(4, int(30 / max(1, len(n_targets))))

    for n_t in n_targets:
        if time_left() < 40:
            break
        target = n_t / 2.0
        pool = []

        # From best
        h_interp = project_feasible(interpolate_to(best_h, n_t), target)
        pool.append(h_interp)

        # Antisymmetric
        h_anti = 1.0 - interpolate_to(best_h, n_t)[::-1]
        h_anti = project_feasible(h_anti, target)
        pool.append(h_anti)

        # Random perturbations
        for _ in range(n_rand_starts):
            sigma = float(rng.choice(np.array([0.02, 0.05, 0.1, 0.15])))
            h_r = best_h + sigma * rng.standard_normal(len(best_h))
            if len(h_r) != n_t:
                h_r = interpolate_to(h_r, n_t)
            h_r = project_feasible(h_r, target)
            pool.append(h_r)

        # Sort pool by score
        pool_sorted = []
        for p in pool:
            ps = official_score(p)
            pool_sorted.append((ps, p))
        pool_sorted.sort(key=lambda x: x[0])

        max_smooth_it = int(max(15, min(50, 2500.0 / max(20, n_t))))
        max_direct_it = int(max(40, min(140, 5000.0 / max(20, n_t))))

        tried_hashes = set()
        for rank, (ps, p) in enumerate(pool_sorted):
            if time_left() < 25:
                break
            if rank >= min(5, len(pool_sorted)):
                break
            key = tuple(np.round(p, 5))
            if key in tried_hashes:
                continue
            tried_hashes.add(key)

            h_cur = p.copy()
            # Smooth ladder
            for beta in betas:
                if time_left() < 15:
                    break
                it_s = int(max(10, max_smooth_it * (1.0 / (1.0 + 0.3 * betas.index(beta)))))
                h_new = smooth_minimize(h_cur, n_t, beta, it_s)
                s_new = official_score(h_new)
                s_cur = official_score(h_cur)
                if s_new < s_cur - 1e-10:
                    h_cur = h_new
                    best_score = min(best_score, s_new)
                    best_h = h_cur.copy()

            # Direct minimax
            if time_left() > 15:
                h_d = direct_minimax(h_cur, n_t, max_direct_it)
                s_d = official_score(h_d)
                s_cur = official_score(h_cur)
                if s_d < s_cur - 1e-10:
                    h_cur = h_d
                    best_score = min(best_score, s_d)
                    best_h = h_cur.copy()
            else:
                s_cur = official_score(h_cur)
                if s_cur < best_score:
                    best_score = s_cur
                    best_h = h_cur.copy()

    # Final micro-search from best
    for _ in range(6):
        if time_left() < 12:
            break
        n_b = len(best_h)
        target = n_b / 2.0
        sigma = float(rng.choice(np.array([0.01, 0.02, 0.04])))
        h_p = best_h + sigma * rng.standard_normal(n_b)
        h_p = project_feasible(h_p, target)
        h_s = smooth_minimize(h_p, n_b, 100.0, 25)
        s_s = official_score(h_s)
        if s_s < best_score:
            best_score = s_s
            best_h = h_s.copy()
        if time_left() > 12:
            h_d = direct_minimax(h_s, n_b, 60)
            s_d = official_score(h_d)
            if s_d < best_score:
                best_score = s_d
                best_h = h_d.copy()

    # Finalize
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = official_score(final_h)
    c5_bound = float(final_score) + 1e-13

    return final_h, c5_bound, len(final_h)
