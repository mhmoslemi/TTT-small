    return compute_c5_array(h, dx).max()

def grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    js = max(0, lag)
    je = min(n, n + lag)
    if js < je:
        grad[js:je] += (1.0 - h[js-lag:je-lag])
    js = max(0, -lag)
    je = min(n, n - lag)
    if js < je:

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(v, target_sum):
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def compute_c5_array(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def c5_value(h, dx):
    return compute_c5_array(h, dx).max()

def grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    js = max(0, -lag)
    je = min(n, n - lag)
    if js < je:
        grad[js:je] += (1.0 - h[js + lag:je + lag])
    js = max(0, lag)
    je = min(n, n + lag)
    if js < je:
        grad[js:je] -= h[js - lag:je - lag]
    return grad

def logsumexp_grad(h, dx, beta):
    c = compute_c5_array(h, dx)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    n = len(h)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] > 1e-8:
            lag = idx - (n - 1)
            grad += w[idx] * grad_lag(h, lag)
    return grad * dx, c.max()

def active_grad(h, dx, frac=0.99):
    c = compute_c5_array(h, dx)
    cmax = c.max()
    threshold = cmax - (cmax - c.min()) * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    n = len(h)
    grad = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        grad += grad_lag(h, lag)
    grad /= len(active)
    return grad * dx, cmax

    target = n / 2.0

    h_best = h.copy()
    alpha = 0.5 / beta0
    step = 0

        step += 1
        prog = min(1.0, step / 500)
        beta = beta0 + (beta1 - beta0) * prog
        grad, _ = logsumexp_grad(h, dx, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        improved = False
        for bt in range(40):
            trial = project_feasible(h - alpha * grad, target)
            trial_c = c5_value(trial, dx)
            if trial_c < c_best - 1e-14:
                h = trial
                c_best = trial_c
                h_best = h.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            alpha = 0.5 / beta
            if alpha < 1e-12:
                break
    return h_best, c_best
