import numpy as np
import math
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_mass(v, n):
    target = n / 2.0
    v = np.asarray(v, dtype=np.float64)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    
    # Bisection for shift
    def F(mu):
        return np.sum(np.clip(v + mu, 0.0, 1.0))
    
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if F(mid) < target:
            lo = mid
        else:
            hi = mid
    
    mu = (lo + hi) / 2.0
    y = np.clip(v + mu, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(y)
    if abs(residual) > 1e-10:
        interior = (y > 1e-8) & (y < 1.0 - 1e-8)
        if np.any(interior):
            y[interior] += residual / np.sum(interior)
            y = np.clip(y, 0.0, 1.0)
    
    return y

def compute_gradient(h, n, beta):
    """Analytical gradient of log-sum-exp smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_args = beta * (corr - max_c)
    exp_vals = np.exp(exp_args)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    n_full = 2 * n - 1
    
    for k in range(n_full):
        w = dx * weights[k]
        if w < 1e-14:
            continue
        # k corresponds to lag = k - (n-1)
        lag = k - (n - 1)
        
        if lag >= 0:
            # c[lag] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            # d/dh[j] = (1-h[j-lag]) * [lag <= j <= n-1] - h[j+lag] * [0 <= j <= n-1-lag]
            for j in range(lag, n):
                grad[j] += w * (1.0 - h[j - lag])
            for j in range(0, n - lag):
                grad[j] += w * (-h[j + lag])
        else:
            j_abs = -lag  # j > 0
            # c[-j] = sum_{i=0}^{n-1-j} h[i] * (1-h[i+j])
            # d/dh[m] = (1-h[m+j]) * [0 <= m <= n-1-j] - h[m-j] * [j <= m <= n-1]
            for m in range(0, n - j_abs):
                grad[m] += w * (1.0 - h[m + j_abs])
            for m in range(j_abs, n):
                grad[m] += w * (-h[m - j_abs])
    
    return grad

def line_search(h, d, n, best_val, max_iter=50):
    """Golden section line search along direction d from point h."""
    # Find feasible interval
    eps = 1e-12
    lam_lo, lam_hi = 0.0, 0.0
    
    for i in range(n):
        if d[i] > eps:
            lam_lo = max(lam_lo, -h[i] / d[i] - eps)
            lam_hi = min(lam_hi, (1.0 - h[i]) / d[i] - eps)
        elif d[i] < -eps:
            lam_lo = max(lam_lo, -(1.0 - h[i]) / d[i] - eps)  # (h[i]-1)/d[i] = -(1-h[i])/d[i]
            lam_hi = min(lam_hi, h[i] / d[i] - eps)  # -h[i]/d[i]
    
    # Cap the step
    lam_lo = max(lam_lo, -1.0)
    lam_hi = min(lam_hi, 1.0)
    
    if lam_lo >= lam_hi:
        return h, best_val
    
    # Golden section search
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    
    a, b = lam_lo, lam_hi
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    
    def obj(lam):
        h_new = h + lam * d
        h_new = project_box_mass(h_new, n)
        return evaluate(h_new)
    
    f_c = obj(c_)
    f_d = obj(d_)
    
    for _ in range(max_iter):
        if f_c < f_d:
            b = d_
            d_ = c_
            f_d = f_c
            c_ = b - gr * (b - a)
            f_c = obj(c_)
        else:
            a = c_
            c_ = d_
            f_c = f_d
            d_ = a + gr * (b - a)
            f_d = obj(d_)
    
    lam_opt = (a + b) / 2.0
    h_opt = project_box_mass(h + lam_opt * d, n)
    f_opt = evaluate(h_opt)
    
    if f_opt < best_val:
        return h_opt, f_opt
    return h, best_val

    time_per_n = 80
    max_nvals = max(2, int((budget_s - 20) / time_per_n))
    n_candidates = n_candidates[:max_nvals]
    
    for n in n_candidates:
        if elapsed() > budget_s - 30:
            break
        
        # Create initial solution for this n
        if h_parent is not None and n == n_parent:
            h = h_parent.copy()
        elif h_parent is not None:
            x_old = np.linspace(0.0, 2.0, n_parent)
            x_new = np.linspace(0.0, 2.0, n)
            h = np.interp(x_new, x_old, h_parent)
            h = project_box_mass(h, n)
        else:
            h = np.full(n, 0.5)
            q = n // 4
            h[:q] = 1.0
            h[q:2*q] = 0.0
            h[2*q:3*q] = 1.0
            h[3*q:] = 0.0
            h = project_box_mass(h, n)
        
        c5 = update_best(h)
        
        # Time allocation for this n
        t_alloc = min(time_per_n, budget_s - 20 - elapsed())
        if t_alloc < 10:
            continue
        t_deadline = start_time + elapsed() + t_alloc
        
        # Phase 1: LSE gradient optimization
        if t_alloc > 15:
            h, c5 = lse_optimize(h, n, c5, rng, start_time + elapsed() + t_alloc * 0.4)
            update_best(h, c5)
        
        # Phase 2: Pair transfers
        if elapsed() < t_deadline - 5:
            t_alloc2 = min(t_alloc * 0.3, t_deadline - elapsed())
            if t_alloc2 > 5:
                n_pairs = int(2000 * t_alloc2 / 20)
                h, c5 = pair_transfer_optimize(h, n, c5, rng, start_time + elapsed() + t_alloc2, n_pairs)
                update_best(h, c5)
        
        # Phase 3: Binary round trials + re-optimize
        if elapsed() < t_deadline - 10:
            for _ in range(3):
                if elapsed() > t_deadline - 10:
                    break
                h_r, c5_r = binary_round_trial(h, n, c5, rng)
                if c5_r < c5:
                    h = h_r
                    c5 = c5_r
                    # Quick LSE polish
                    h, c5 = lse_optimize(h, n, c5, rng, start_time + elapsed() + 5)
                update_best(h, c5)
        
        # Phase 4: More pair transfers on best
        if elapsed() < t_deadline - 3:
            t_alloc3 = max(3, t_deadline - elapsed() - 3)
            n_pairs = int(1000 * t_alloc3 / 15)
            h, c5 = pair_transfer_optimize(h, n, c5, rng, start_time + elapsed() + t_alloc3, n_pairs)
            update_best(h, c5)
    
    # Final polish on best solution
    if best_h is not None and elapsed() < budget_s - 10:
        n_best = len(best_h)
        c5_best = best_c5
        h_polish = best_h.copy()
        t_final = budget_s - 15
        h_polish, c5_best = lse_optimize(h_polish, n_best, c5_best, rng, start_time + elapsed() + min(30, t_final - elapsed()))
        if c5_best < best_c5:
            update_best(h_polish, c5_best)
        
        if elapsed() < budget_s - 5:
            h_polish, c5_best = pair_transfer_optimize(h_polish, n_best, c5_best, rng, start_time + elapsed() + min(20, budget_s - 10 - elapsed()), 2000)
            if c5_best < best_c5:
                update_best(h_polish, c5_best)
    
    if best_h is None:
        n_fb = 70
        best_h = project_box_mass(np.full(n_fb, 0.5), n_fb)
        best_c5 = evaluate(best_h)
    
    return best_h, best_c5, len(best_h)
