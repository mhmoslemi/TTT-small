import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    """Compute the exact C5 value using the official evaluator."""
    n = len(h)
    dx = 2.0 / n
    val = np.correlate(h, 1.0 - h, mode='full').max()
    return val * dx

def manual_correlate(h):
    """Compute unscaled correlation values for all lags.
    Returns array of length 2n-1, matching np.correlate(h, 1-h, 'full') / 1 (unscaled)."""
    n = len(h)
    result = np.zeros(2 * n - 1)
    for d in range(2 * n - 1):
        l = d - (n - 1)  # lag
        if l >= 0:
            # sum_{j=0}^{n-1-l} h[j+l] * (1-h[j])
            j = np.arange(n - l)
            result[d] = np.sum(h[j + l] * (1.0 - h[j]))
        else:
            # sum_{j=-l}^{n-1} h[j] * (1-h[j+l])
            j = np.arange(-l, n)
            result[d] = np.sum(h[j] * (1.0 - h[j + l]))
    return result

def manual_grad(h):
    """Compute gradient matrix of correlation values.
    Returns array of shape (2n-1, n) where entry [d, p] = ∂F_d/∂h[p]."""
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    for d in range(2 * n - 1):
        l = d - (n - 1)
        for p in range(n):
            g = 0.0
            # h[p] appears as h[j] with j=p, if p+l is valid
            if 0 <= p + l < n:
                g += (1.0 - h[p + l])
            # h[p] appears as h[j+l] with j=p-l, if p-l is valid
            if 0 <= p - l < n:
                g -= h[p - l]
            grad[d, p] = g
    return grad

def make_feasible(z, target=None):
    """Project z to [0,1]^n with sum = target (default: len(z)/2)."""
    n = len(z)
    if target is None:
        target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    # Clip and adjust sum
    # Use bisection on lambda: h = clip(z + lambda, 0, 1)
    def G(lam):
        return np.sum(np.clip(z + lam, 0.0, 1.0))
    
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if G(mid) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            # Add to entries with most room
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                sub = min(-residual, h[idx])
                h[idx] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    
    # Final clip
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(x, beta):
    """Compute smooth approximation to max of correlation values."""
    n = len(x)
    dx = 2.0 / n
    vals = manual_correlate(x) * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    # numerically stable log-sum-exp
    if shifted.max() > 500:
        return cmax + 1.0 / beta * 0.0
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + (1.0 / beta) * logsum

def smooth_gradient(x, beta):
    """Compute gradient of smooth objective."""
    n = len(x)
    dx = 2.0 / n
    vals = manual_correlate(x) * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    if shifted.max() > 500:
        w = np.ones(len(vals)) / len(vals)
    else:
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.ones(len(vals)) / len(vals)
    grads = manual_grad(x)  # shape (2n-1, n)
    g = dx * (w @ grads)
    return g

def smooth_obj_jac(x, beta):
    """Return (objective, gradient) for scipy minimize."""
    return smooth_objective(x, beta), smooth_gradient(x, beta)

def smooth_constraints_eq(x):
    """Equality constraint: sum(x) = len(x)/2."""
    return np.sum(x) - len(x) / 2.0

def smooth_constraints_eq_jac(x):
    """Jacobian of equality constraint."""
    return np.ones(len(x))

def try_smooth_optimize(h0, beta, maxiter=200):
    """Run SLSQP smooth optimization from h0 with given beta."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': smooth_constraints_eq, 'jac': smooth_constraints_eq_jac}]
    
    try:
        res = minimize(
            smooth_objective, h0, args=(beta,),
            jac=smooth_gradient, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if res.success and np.isfinite(res.fun):
            h_new = make_feasible(res.x)
            score = official_score(h_new)
            if np.isfinite(score):
                return h_new, score
        # Try even if not successful
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = make_feasible(res.x)
            score = official_score(h_new)
            if np.isfinite(score):
                return h_new, score
    except Exception:
        pass
    return None, None

def try_direct_minimax(h0, maxiter=150):
    """Direct minimax: minimize theta s.t. theta >= dx*F_l for all l."""
    n = len(h0)
    dx = 2.0 / n
    L = 2 * n - 1  # number of lags
    theta0 = official_score(h0) + 1e-6
    
    def obj(z):
        return z[-1]
    
    def obj_jac(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq_con(z):
        h = z[:n]
        theta = z[-1]
        vals = manual_correlate(h) * dx
        return theta - vals
    
    def ineq_jac(z):
        h = z[:n]
        grads = manual_grad(h)  # shape (L, n)
        J = -dx * grads  # shape (L, n)
        J = np.hstack([J, np.ones((L, 1))])  # add theta column
        return J
    
    def eq_con(z):
        return np.sum(z[:n]) - n / 2.0
    
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_con, 'jac': eq_jac}
    ]
    
    z0 = np.concatenate([h0, [theta0]])
    
    try:
        res = minimize(
            obj, z0, jac=obj_jac, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        h_new = res.x[:n]
        h_new = make_feasible(h_new)
        score = official_score(h_new)
        if np.isfinite(score):
            return h_new, score
    except Exception:
        pass
    return None, None

def interpolate_h(h_src, n_target):
    """Interpolate h from source resolution to target resolution."""
    n_src = len(h_src)
    # Center coordinates
    src_coords = (np.arange(n_src) + 0.5) / n_src * 2.0
    tgt_coords = (np.arange(n_target) + 0.5) / n_target * 2.0
    h_interp = np.interp(tgt_coords, src_coords, h_src)
    return make_feasible(h_interp)

def anti_symmetric(h):

    n = len(h)
    h_rev = h[::-1].copy()
    h_new = 1.0 - h_rev
    return make_feasible(h_new)

And F_{-s} = sum_m h[m]*(1-h[m+s]) = sum_m h[m] - h[m]*h[m+s] = n/2 - sum_m h[m]*h[m+s]

So under anti-symmetry:
F_s = n/2 - sum_j h[j]*h[j-s] (for valid j)
F_{-s} = n/2 - sum_m h[m]*h[m+s] (for valid m)

Note that sum_j h[j]*h[j-s] and sum_m h[m]*h[m+s] are the same sum (just different index variable), so F_s = F_{-s} under anti-symmetry!

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def manual_correlate(h):
    n = len(h)
    result = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            m = np.arange(n - s)
            result[p] = np.dot(h[m + s], 1.0 - h[m])
        else:
            t = -s
            m = np.arange(n - t)
            result[p] = np.dot(h[m], 1.0 - h[m + t])
    return result

def manual_grad(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        s = p - (n - 1)
        if s >= 0:
            for j in range(n):
                g = 0.0
                if s <= j:
                    g += (1.0 - h[j - s])
                if j < n - s:
                    g -= h[j + s]
                grad[p, j] = g
        else:
            t = -s

                g = 0.0
                if j < n - t:
                    g += (1.0 - h[j + t])
                if t <= j:
                    g -= h[j - t]
                grad[p, j] = g
    return grad

    cmax = vals.max()
    shifted = beta * (vals - cmax)
    logsum = np.log(np.sum(np.exp(shifted)))
    return cmax + logsum / beta

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = manual_correlate(x) * dx
    cmax = vals.max()
    shifted = beta * (vals - cmax)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= ws
    grads = manual_grad(x)
    return dx * (w @ grads)
