def project_box_mass(y, target=None):

    lo, hi = -1.0, 1.0

            lo = mid
        else:

import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def is_feasible(h, tol=1e-7):
    if len(h) == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1 + tol):
        return False
    target = len(h) / 2.0
    if abs(np.sum(h) - target) > tol:
        return False
    return True

def project_box_mass(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    if n == 0:
        return y.copy()
    # Handle edge cases
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    y = np.asarray(y, dtype=np.float64).copy()
    # Replace non-finite values
    y[~np.isfinite(y)] = 0.5
    y = np.clip(y, 0, 1)
    # Bisection for lambda
    lo, hi = -0.5, 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(y - lam, 0, 1)
    # Fix mass precisely
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        if diff > 0:
            # Need to add mass; find element with most room
            slack = 1.0 - result
            order = np.argsort(-slack)
        else:
            order = np.argsort(result)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return result

def softmax_obj_grad(h, tau):
    """Compute softmax-smoothed max of correlation values and gradient."""
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    M = np.max(cvals)
    scaled = (cvals - M) / tau
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_c = np.exp(scaled)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    obj = M * dx + tau * np.log(sum_exp)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        l = k - (n - 1)
        if l >= 0:
            grad[l:] += wk * (1.0 - h[:n-l])
            grad[:n-l] -= wk * h[l:]
        else:
            p = -l
            grad[:n-p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:n-p]
    return obj, grad

def epigraph_obj_grad(x, n, active_idx):
    """Epigraph: x = [h (n values), t (1 value)]
    Minimize t, subject to t >= C_k(h) for k in active_idx"""
    h = x[:n]
    t = x[n]
    cvals = np.correlate(h, 1.0 - h, mode='full')
    # Objective: t
    obj = t
    # Gradient of objective
    grad = np.zeros(n + 1)
    grad[n] = 1.0
    # Constraints will be handled separately
    return obj, grad

def mass_constraint(x, n):
    return np.sum(x[:n]) - n / 2.0

def active_constraints(x, n, active_idx):
    """Return constraint values: t - C_k(h) for k in active_idx (should be >= 0)"""
    h = x[:n]
    t = x[n]
    cvals = np.correlate(h, 1.0 - h, mode='full')
    return t - cvals[active_idx]

def transfer_resolution(h_src, n_tgt):
    """Transfer a function from source resolution to target resolution via area interpolation."""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    # Build cumulative area of source
    cum_src = np.zeros(n_src + 1)
    cum_src[1:] = np.cumsum(h_src * dx_src)
    x_src = np.linspace(0, 2.0, n_src + 1)
    
    # Evaluate at target bin edges
    x_tgt = np.linspace(0, 2.0, n_tgt + 1)
    cum_tgt = np.interp(x_tgt, x_src, cum_src)
    
    # Get target values
    h_tgt = np.diff(cum_tgt) / dx_tgt
    # Project to feasibility
    h_tgt = project_box_mass(h_tgt, n_tgt / 2.0)
    return h_tgt

def exact_c5(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_mass(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    if n == 0:
        return y.copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    y = np.asarray(y, dtype=np.float64).copy()
    y[~np.isfinite(y)] = 0.5
    y = np.clip(y, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(y - lam, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - result)
            order = order[::-1]
        else:
            order = np.argsort(result)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
                if add < 1e-15:
                    break
                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                if sub < 1e-15:
                    break
                result[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return result

def softmax_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(cvals)
    scaled = np.clip((cvals - M) / tau, -50.0, 50.0)
    exp_c = np.exp(scaled)
    sum_exp = np.sum(exp_c)
    obj = M * dx + tau * np.log(sum_exp)
    return obj

def softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(cvals)
    scaled = np.clip((cvals - M) / tau, -50.0, 50.0)
    exp_c = np.exp(scaled)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        l = k - (n - 1)
        if l >= 0:
            if l == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[l:] += wk * (1.0 - h[:n - l])
                grad[:n - l] -= wk * h[l:]
        else:
            p = -l
            grad[:n - p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:n - p]
    return grad

def run_smooth_slsqp(h0, tau_sequence, maxiter_per_tau, deadline):

    current = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_fun, 'jac': mass_fun_jac}]

        return softmax_obj(x, tau_current[0])

    def grad_wrapper(x):

    for tau in tau_sequence:
        tau_current[0] = tau
        if time.time() > deadline:

        try:

                jac=grad_wrapper,

    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = exact_c5(best)

        improved = False

                new_h[j] += d
                s = exact_c5(new_h)

                    best_score = s
                    improved = True
                    break
        if not improved:

                    new_h = best.copy()
                    new_h[j] -= d
                    new_h[i] += d
                    s = exact_c5(new_h)
                    if s < best_score - 1e-12:

def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    cum_src = np.zeros(n_src + 1)
    cum_src[1:] = np.cumsum(h_src * dx_src)
    x_src = np.linspace(0.0, 2.0, n_src + 1)
    x_tgt = np.linspace(0.0, 2.0, n_tgt + 1)
    cum_tgt = np.interp(x_tgt, x_src, cum_src)
    h_tgt = np.diff(cum_tgt) / dx_tgt
    h_tgt = project_box_mass(h_tgt, n_tgt / 2.0)
    return h_tgt

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline_final = t_start + budget_s * 0.93

    # Get parent
    parent = None
    if 'initial_h_values' in globals():
        try:
            parent = np.asarray(initial_h_values, dtype=np.float64).copy()
        except Exception:
            parent = None

    if parent is None or len(parent) == 0 or not np.all(np.isfinite(parent)):
        n0 = 45
        parent = np.ones(n0) * 0.5

    n0 = len(parent)
    parent = project_box_mass(parent, n0 / 2.0)
    best_h = parent.copy()
    best_score = exact_c5(best_h)
    rng = np.random.RandomState(seed)

    if best_score < 0.3808:
        return best_h, best_score, n0

    # Generate diverse starts
    starts = [best_h.copy()]
    starts.append(project_box_mass(1.0 - best_h, n0 / 2.0))
    starts.append(project_box_mass(best_h[::-1], n0 / 2.0))

    for sigma in [0.35, 0.2, 0.1, 0.05]:
        perturbed = best_h + sigma * rng.randn(n0)
        starts.append(project_box_mass(perturbed, n0 / 2.0))

    # Binary from sorted
    sorted_idx = np.argsort(best_h)[::-1]
    binary = np.zeros(n0)
    binary[sorted_idx[:n0 // 2]] = 1.0
    if n0 % 2 == 0:
        starts.append(binary.copy())
    else:
        binary[sorted_idx[n0 // 2]] = 0.5
        starts.append(binary.copy())

    # Sinusoidal
    x_grid = np.linspace(0.0, 2.0, n0)
    for freq in [1, 2, 3]:
        for phase in [0.0, np.pi / 2.0, np.pi / 4.0]:
            pat = 0.5 + 0.45 * np.sin(freq * np.pi * x_grid + phase)
            starts.append(project_box_mass(pat, n0 / 2.0))

    # Low-discrepancy
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    fracs = np.sort(np.mod(np.arange(n0, dtype=float) * phi, 1.0))
    thresh = n0 / (2.0 * n0)
    ld = (fracs < 0.5).astype(float)
    ld = project_box_mass(ld, n0 / 2.0)
    starts.append(ld)

    # Deduplicate starts
    unique_starts = [starts[0]]
    for s in starts[1:]:
        is_dup = False
        for u in unique_starts:
            if np.max(np.abs(s - u)) < 1e-6:
                is_dup = True
                break
        if not is_dup:
            unique_starts.append(s)
    starts = unique_starts

    # Stage 1: Smoothed SLSQP from each start
    tau_seq_main = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008]
    maxiter_main = 100

    for si, start in enumerate(starts):
        if time.time() > t_start + budget_s * 0.55:
            break
        cur = start.copy()
        cur_score = exact_c5(cur)
        cur = run_smooth_slsqp(
            cur,
            tau_seq_main if si < 4 else tau_seq_main[2:],
            maxiter_main,
            t_start + budget_s * 0.65
        )
        cur_score = exact_c5(cur)
        if cur_score < best_score - 1e-10:
            best_h = cur
            best_score = cur_score
            if best_score < 0.3808:
                break

    # Stage 2: More refined optimization on best
    if time.time() < t_start + budget_s * 0.80:
        tau_seq_refine = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        best_h = run_smooth_slsqp(best_h, tau_seq_refine, 150, t_start + budget_s * 0.82)
        best_score = exact_c5(best_h)

    # Stage 3: Local pair moves
    if time.time() < t_start + budget_s * 0.90:
        best_h, best_score = local_pair_moves(best_h, 5000, seed + 100, t_start + budget_s * 0.92)

    # Stage 4: Try higher resolution
    if time.time() < t_start + budget_s * 0.75:
        for n_tgt in [int(n0 * 1.5), n0 * 2, int(n0 * 2.5)]:
            if n_tgt < 10 or n_tgt > 250:
                continue
            if time.time() > t_start + budget_s * 0.88:
                break
            h_trans = transfer_resolution(best_h, n_tgt)
            s_trans = exact_c5(h_trans)
            if s_trans < best_score + 0.005:
                h_opt = run_smooth_slsqp(h_trans, [0.03, 0.01, 0.005, 0.002, 0.001], 80, t_start + budget_s * 0.88)
                s_opt = exact_c5(h_opt)
                if s_opt < best_score - 1e-10:
                    best_h = h_opt
                    best_score = s_opt
                if time.time() < t_start + budget_s * 0.90:
                    h_opt, s_opt2 = local_pair_moves(h_opt, 2000, seed + 200, t_start + budget_s * 0.91)
                    if s_opt2 < best_score - 1e-10:
                        best_h = h_opt
                        best_score = s_opt2

    # Stage 5: Try lower resolution (compact representation)
    if time.time() < t_start + budget_s * 0.85 and n0 > 20:
        for n_tgt in [max(10, n0 // 2), max(10, n0 // 3)]:
            if time.time() > t_start + budget_s * 0.90:
                break
            h_trans = transfer_resolution(best_h, n_tgt)
            s_trans = exact_c5(h_trans)
            if s_trans < best_score - 1e-8:
                best_h = h_trans
                best_score = s_trans

    # Final: one more round of pair moves on the best
    if time.time() < deadline_final - 5 and best_score >= 0.3808:
        best_h, best_score = local_pair_moves(best_h, 3000, seed + 500, deadline_final - 2)

    # Final validation
    best_h = project_box_mass(best_h, len(best_h) / 2.0)
    final_score = exact_c5(best_h)

    if not (np.all(best_h >= -1e-9) and np.all(best_h <= 1.0 + 1e-9)):
        best_h = np.ones(n0) * 0.5
        final_score = exact_c5(best_h)

    return best_h, final_score, len(best_h)
