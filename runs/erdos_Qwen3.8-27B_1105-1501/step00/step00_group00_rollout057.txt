def c5_evaluate(h):
    """Evaluate C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_evaluate_all(h):
    """Evaluate all shift values for C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def normalize_h(h, n):
    """Ensure h satisfies constraints: sum(h) = n/2, 0 <= h[i] <= 1."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    current = h.sum()
    if abs(current - target) > 1e-10:
        # Adjust proportionally
        # Add/subtract from elements that have room
        diff = target - current
        if diff > 0:
            # Need to add
            room = 1.0 - h
            if room.sum() > 0:
                h += diff * room / room.sum()
        else:
            # Need to subtract
            room = h
            if room.sum() > 0:
                h += diff * room / room.sum()
    # Fix any remaining clipping
    h = np.clip(h, 0.0, 1.0)
    return h

    corrs = c5_evaluate_all(h)

        # Annealing schedule
        t = t_start * (t_end / t_start) ** (it / max(1, n_iter))

        j = rng.integers(n)

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def fix_sum(h, tol=1e-12):
    """Fix the sum constraint while maintaining 0 <= h <= 1."""
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) <= tol:
        return h
    h = h.copy()
    # Try to adjust using interior points first
    mask = (h > tol) & (h < 1.0 - tol)
    if mask.sum() > 0:
        needed = diff
        for idx in np.where(mask)[0]:
            if needed > 0:
                add = min(needed, 1.0 - h[idx])
                h[idx] += add
                needed -= add
            elif needed < 0:
                sub = min(-needed, h[idx])
                h[idx] -= sub
                needed += sub
            if abs(needed) <= tol:
                break
    if abs(diff - (h.sum() - target)) > tol:
        # Still not exact, force it
        h = np.clip(h, 0, 1)
    return h

def local_search(h, n_iter, seed=42, n_trials=10):
    """Local search with pairwise transfers."""
    rng = np.random.default_rng(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_c5 = c5_eval(h)
    corrs = c5_all(h)

        j = rng.integers(n)
        if i == j:
            continue

        hi = min(1.0 - h[i], h[j])

            continue
        
        best_t = 0.0
        best_val = best_c5

        for t in np.linspace(lo, hi, n_trials + 1)[1:-1]:
            h[i] += t
            h[j] -= t
            val = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

            h[i] -= t

            h[i] += best_t
            h[j] -= best_t

        i = rng.integers(n)
        j = rng.integers(n)

        else:
            h[i] -= dt
            h[j] += dt
    
    return best_h, best_c5

Here's my refined strategy:

- To reduce this, I want to decrease h[i] where h[i+m*] is small, or increase h[i+m*] where h[i] is large
- Transferring mass from i to i+m* (if m* > 0) would directly reduce the overlap at shift m*
- But it might increase the overlap at other shifts

Let me implement a "greedy" strategy:
1. Find worst shift m*
2. Among all valid pairs (i, j), find the one that most reduces C5
3. Make that transfer
4. Repeat

This is O(n²) per step (trying all pairs), which for n=45 is 2025 pairs. With C5 evaluation taking ~5μs, each step takes ~10ms. In 100 seconds, that's ~10000 steps. Not great.

Better: only try a subset of promising pairs. For the worst shift m*, try transfers between (i, i+m*) and (i, i-m*) for various i. That's only O(n) pairs per step.

Let me think about a faster approach. What if I compute the C5 change analytically for a given transfer?

When I transfer t from position j to position i (h[i] += t, h[j] -= t), the change in the correlation at shift m (correlation index n-1+m) is:

Δcorr[n-1+m] = t * (1-h[i+m]) - t*h[i-m] - t*(1-h[j+m]) + t*h[j-m]

Wait, let me be more careful.

The correlation at index k (where k = n-1+m) is:
corr[k] = sum_j h[j - k + n - 1] * (1-h[j])

When h[i] changes by +t and h[j] changes by -t:
- h[i] appears in corr[k] when i = j' - k + n - 1 for some j', i.e., j' = i + k - n + 1 = i + m
  So corr[k] gets +t*(1-h[i+m]) if 0 ≤ i+m ≤ n-1
  And corr[k] gets -t*h[i-m] if... wait, h[i] also appears as (1-h[j']) when j' = i, so the term is h[i-m+k-n+1]... 

This is getting confusing. Let me just use the direct formula.

For the overlap at shift m (the "natural" overlap, not the correlation index):
overlap_m = dx * sum_{i: valid} h[i] * (1-h[i+m])

When h[i0] += t and h[j0] -= t:
- h[i0] appears as h[i] at i=i0: contributes t*(1-h[i0+m])
- h[i0] appears as h[i+m] at i=i0-m: contributes -t*h[i0-m] (since (1-h[i+m]) decreases by t when h[i+m] increases)
- h[j0] appears as h[i] at i=j0: contributes -t*(1-h[j0+m])
- h[j0] appears as h[i+m] at i=j0-m: contributes +t*h[j0-m]

So Δoverlap_m = dx * [t*(1-h[i0+m]) - t*h[i0-m] - t*(1-h[j0+m]) + t*h[j0-m]]
             = dx * t * [(1-h[i0+m]) - h[i0-m] - (1-h[j0+m]) + h[j0-m]]
             = dx * t * [h[j0+m] - h[i0+m] - h[i0-m] + h[j0-m]]

with the convention that h[p] = 0 if p is outside [0, n-1].

So the change in C5 is:
ΔC5 = max_m (overlap_m + Δoverlap_m) - max_m(overlap_m)

This can be computed in O(n) time (for all m) given the positions i0, j0 and transfer t.

But we want to minimize the max, so we want to find t that minimizes:
max_m (overlap_m + dx * t * g_m)

where g_m = h[j0+m] - h[i0+m] - h[i0-m] + h[j0-m] (with boundary handling).

This is a 1D optimization: minimize max_m (a_m + t*b_m) over t in [lo, hi], where a_m = overlap_m and b_m = dx*g_m.

The function max_m (a_m + t*b_m) is a convex piecewise-linear function of t (max of linear functions). Its minimum can be found by:
1. Finding all the "breakpoints" where the argmax changes
2. Checking the value at each breakpoint and at the endpoints

The breakpoints occur where a_m + t*b_m = a_m' + t*b_m' for some m, m', i.e., t = (a_m' - a_m) / (b_m - b_m').

There are O(n²) breakpoints in the worst case, but in practice the number of "active" shifts (near the maximum) is small.

A simpler approach: since the function is convex (as a function of t), we can use binary search on the subgradient. The subgradient at t is the set of b_m for all m that achieve the maximum.

Or even simpler: since n is small (45), I can just evaluate the function at a few points and do a ternary search.

import numpy as np
import time

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def c5_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def fix_sum(h):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0, 1)
    mask = (h > 0) & (h < 1)
    if mask.sum() > 0:
        frac = np.clip(h + diff / mask.sum(), 0, 1)
        h[mask] = frac
        diff2 = target - h.sum()
        if abs(diff2) > 1e-14:
            idxs = np.where(mask)[0]
            for idx in idxs:
                if diff2 > 0:
                    a = min(diff2, 1.0 - h[idx])
                    h[idx] += a
                    diff2 -= a
                else:
                    a = -min(-diff2, h[idx])
                    h[idx] += a
                    diff2 -= a
                if abs(diff2) < 1e-14:
                    break
    else:
        if diff > 0:
            idxs = np.argsort(h)
            for idx in idxs:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if diff < 1e-14:
                    break
        else:
            idxs = np.argsort(-h)
            for idx in idxs:
                a = max(diff, -h[idx])
                h[idx] += a
                diff -= a
                if diff > -1e-14:
                    break
    return np.clip(h, 0, 1)

def local_search(h, n_iter, seed=42, n_trials=8):
    rng = np.random.default_rng(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_c5 = c5_eval(h)
    stall = 0
    stall_limit = 10000
    while n_iter > 0:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            n_iter -= 1
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi - 1e-15:
            n_iter -= 1
            continue
        found = False
        for t in np.linspace(lo, hi, n_trials + 1)[1:-1]:
            h[i] += t
            h[j] -= t
            val = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
            if val < best_c5 - 1e-13:
                best_c5 = val
                found = True
            else:
                h[i] -= t
                h[j] += t
        if found:
            # keep the last accepted move (h already at new state)
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break
        n_iter -= 1
    return h, best_c5

def targeted_search(h, n_iter, seed=42):
    rng = np.random.default_rng(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_c5 = c5_eval(h)
    corrs = c5_all(h)
    stall = 0
    for _ in range(n_iter):
        bad_k = int(np.argmax(corrs))
        m = bad_k - (n - 1)
        cands = []
        if m > 0:
            for ii in range(min(n - m, 15)):
                cands.append((ii, ii + m))
            for ii in range(max(0, n - m - 7), n - m):
                cands.append((ii, ii + m))
        elif m < 0:
            mm = -m
            for ii in range(mm, min(mm + 15, n)):
                cands.append((ii, ii - mm))
            for ii in range(max(mm, n - mm - 7), n):
                cands.append((ii, ii - mm))
        for _ in range(4):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii != jj:
                cands.append((ii, jj))
        found = False
        for (ii, jj) in cands:
            lo = max(-h[ii], h[jj] - 1.0)
            hi = min(1.0 - h[ii], h[jj])
            if lo >= hi - 1e-15:
                continue
            old_i = h[ii]
            old_j = h[jj]
            for t in np.linspace(lo, hi, 5)[1:-1]:
                h[ii] = old_i + t
                h[jj] = old_j - t
                val = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
                if val < best_c5 - 1e-13:
                    best_c5 = val
                    found = True
                    break
            h[ii] = old_i
            h[jj] = old_j
            if found:
                break
        if found:
            h[ii] = old_i + t
            h[jj] = old_j - t
            corrs = c5_all(h)
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
    return h, best_c5

def anneal(h, n_iter, seed=42, t0=0.04, t1=0.0003):
    rng = np.random.default_rng(seed)
    n = len(h)
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = c5_eval(h)
    cur_c5 = best_c5
    for it in range(n_iter):
        temp = t0 * (t1 / t0) ** (it / max(1, n_iter - 1))
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi - 1e-15:
            continue
        dt = float(rng.uniform(lo, hi))
        h[i] += dt
        h[j] -= dt
        nc = c5_eval(h)
        d = nc - cur_c5
        if d < 0 or float(rng.random()) < np.exp(-d / max(temp, 1e-12)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
        else:
            h[i] -= dt
            h[j] += dt
    return best_h, best_c5

def interp_n(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, float(n_old - 1), n_new)
    return np.interp(x_new, x_old, h_old)
