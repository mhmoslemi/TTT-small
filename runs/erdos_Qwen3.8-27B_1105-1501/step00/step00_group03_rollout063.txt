        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-7 or float(np.max(h)) > 1.0 + 1e-7:
        return False
    if abs(float(np.sum(h)) - 0.5 * h.size) > tol:
        return False
    return True


def official_score(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float("inf")
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def manual_values(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)
    rr, ii, jj, _idx_i, _idx_j = _pair_cache(n)
    vals = np.zeros(2 * n - 1, dtype=float)
    if ii.size:
        np.add.at(vals, rr, h[ii] * (1.0 - h[jj]))
    return vals


def manual_grad(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.zeros((0, 0), dtype=float)
    _rr, ii, jj, idx_i, idx_j = _pair_cache(n)
    G = np.zeros((2 * n - 1, n), dtype=float)
    flat = G.ravel()
    if ii.size:
        np.add.at(flat, idx_i, 1.0 - h[jj])
        np.add.at(flat, idx_j, -h[ii])
    return G


def interpolate_to(h_src, n):
    h_src = np.asarray(h_src, dtype=float).ravel()
    m = h_src.size

    if n <= 0:
        return np.empty(0, dtype=float)
    if m == 0:
        return make_feasible(np.full(n, 0.5))
    if m == n:
        return make_feasible(h_src.copy())
    if m == 1:
        return make_feasible(np.full(n, h_src[0]))

    xs = (np.arange(m) + 0.5) * (2.0 / m)
    xt = (np.arange(n) + 0.5) * (2.0 / n)
    vals = np.interp(xt, xs, np.clip(h_src, 0.0, 1.0))
    return make_feasible(vals)


def mirror_transform(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return h
    return make_feasible(1.0 - h[::-1])


def random_start(n, rng, kind=None):
    if kind is None:
        kind = int(rng.integers(0, 6))

    ar = np.arange(n)
    x = (ar + 0.5) / float(n)

    if kind == 0:
        v = 0.5 + 0.10 * rng.normal(size=n)
    elif kind == 1:
        c = float(rng.uniform(0.25, 0.75))
        m = float(rng.uniform(3.0, 12.0))
        v = 1.0 / (1.0 + np.exp(-m * (x - c))) + 0.03 * rng.normal(size=n)
    elif kind == 2:
        p = max(2, int(n // 6) + 1)
        v = 0.5 + 0.35 * ((ar % p) < p // 2).astype(float) + 0.03 * rng.normal(size=n)
    elif kind == 3:
        f = float(rng.uniform(1.0, 5.0))
        a = float(rng.uniform(0.08, 0.35))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        v = 0.5 + a * np.sin(2.0 * np.pi * f * x + ph) + 0.02 * rng.normal(size=n)
    elif kind == 4:
        prob = float(rng.uniform(0.3, 0.7))
        v = (rng.random(n) < prob).astype(float) + 0.02 * rng.normal(size=n)
    else:
        r = rng.random(n)
        threshold = np.sort(r)[n // 2]
        v = (r >= threshold).astype(float) + 0.01 * rng.normal(size=n)

    return make_feasible(v)


def candidate_key(item):
    return item[0]


def add_start(starts, h):
    h = make_feasible(h)
    if is_feasible(h):
        sc = official_score(h)
        if np.isfinite(sc):
            starts.append((float(sc), h))
    return starts


def build_starts(n, seed, base_h, best_h, n_random):
    starts = []

    if base_h is not None and base_h.size > 0:
        hb = interpolate_to(base_h, n)
        add_start(starts, hb)
        add_start(starts, mirror_transform(hb))

    if best_h is not None and best_h.size > 0:
        hbest = interpolate_to(best_h, n)
        add_start(starts, hbest)
        add_start(starts, mirror_transform(hbest))

        rngp = np.random.default_rng(seed + 123456)
        for sigma in (0.005, 0.02, 0.05, 0.12):
            add_start(starts, hbest + sigma * rngp.normal(size=n))

    ar = np.arange(n)
    x = (ar + 0.5) / float(n)
    add_start(starts, (ar % 2 == 0).astype(float))
    add_start(starts, ((ar % 5) < 2).astype(float))
    add_start(starts, 0.5 + 0.20 * np.sin(2.0 * np.pi * 2.0 * x))
    add_start(starts, 0.5 + 0.30 * np.sin(2.0 * np.pi * 3.0 * x + 0.25 * np.pi))

    rng = np.random.default_rng(seed * 1000003 + n * 9973)
    for _ in range(max(0, int(n_random))):
        add_start(starts, random_start(n, rng))

    starts.sort(key=candidate_key)
    return starts


def eq_sum_fun(x):
    return float(np.sum(x) - 0.5 * len(x))


def eq_sum_jac(x):
    return np.ones(len(x), dtype=float)


def smooth_obj(x, beta):
    n = len(x)
    vals = manual_values(x) * (2.0 / n)
    vmax = float(np.max(vals))
    shifted = beta * (vals - vmax)
    wsum = float(np.exp(shifted).sum())
    if not np.isfinite(wsum) or wsum <= 0.0:
        wsum = float(len(vals))
    return vmax + float(np.log(wsum)) / beta


def smooth_jac(x, beta):
    n = len(x)
    vals = manual_values(x) * (2.0 / n)
    vmax = float(np.max(vals))
    shifted = beta * (vals - vmax)
    w = np.exp(shifted)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.ones_like(vals) / float(len(vals))
    else:
        w = w / s
    G = manual_grad(x)
    return (2.0 / n) * (G.T @ w)


def direct_fun(z):
    return float(z[-1])


def direct_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1, dtype=float)
    J[-1] = 1.0
    return J


def direct_eq_fun(z):
    n = len(z) - 1
    return float(np.sum(z[:n]) - 0.5 * n)


def direct_eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1, dtype=float)
    J[:n] = 1.0
    return J


def direct_ineq_fun(z):
    n = len(z) - 1
    x = z[:n]
    return z[-1] - manual_values(x) * (2.0 / n)


def direct_ineq_jac(z):
    n = len(z) - 1
    x = z[:n]
    G = manual_grad(x)
    J = np.zeros((2 * n - 1, n + 1), dtype=float)
    J[:, :n] = -(2.0 / n) * G
    J[:, -1] = 1.0
    return J


def solve_smooth(x0, betas, max_iter, deadline):
    x = make_feasible(x0)
    n = len(x)
    if n == 0 or max_iter <= 0 or time.time() >= deadline:
        return x

    bounds = [(0.0, 1.0)] * n
    constraint = {"type": "eq", "fun": eq_sum_fun, "jac": eq_sum_jac}

    for beta in betas:
        if max_iter <= 0 or time.time() >= deadline:
            break
        try:
            res = minimize(
                smooth_obj,
                x,
                method="SLSQP",
                args=(float(beta),),
                jac=smooth_jac,
                bounds=bounds,
                constraints=constraint,
                options={"maxiter": int(max_iter), "ftol": 1e-10, "disp": False},
            )
        except Exception:
            break

        if res.x is not None and np.all(np.isfinite(res.x)):
            x = make_feasible(res.x)

    return x


def solve_direct(x0, max_iter, deadline):
    x = make_feasible(x0)
    n = len(x)
    if n == 0 or max_iter <= 0 or time.time() >= deadline:
        return x, official_score(x)

    sc = official_score(x)
    manualmax = float(np.max(manual_values(x) * (2.0 / n)))
    theta0 = max(sc, manualmax) + 1e-5

    upper = 1.0 + 1e-7
    if theta0 >= upper:
        theta0 = upper - 1e-8

    z0 = np.empty(n + 1, dtype=float)
    z0[:n] = x
    z0[-1] = theta0

    bounds = [(0.0, 1.0)] * n
    bounds.append((0.0, upper))

    constraints = (
        {"type": "eq", "fun": direct_eq_fun, "jac": direct_eq_jac},
        {"type": "ineq", "fun": direct_ineq_fun, "jac": direct_ineq_jac},
    )

    try:
        res = minimize(
            direct_fun,
            z0,
            method="SLSQP",
            jac=direct_jac,
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(max_iter), "ftol": 1e-10, "disp": False},
        )
    except Exception:
        return x, sc

    best_x = x
    best_sc = sc

    if res.x is not None:
        raw = np.asarray(res.x[:n], dtype=float)
        if np.all(np.isfinite(raw)):
            x1 = make_feasible(raw)
            sc1 = official_score(x1)
            if np.isfinite(sc1) and sc1 < best_sc:
                best_x = x1
                best_sc = sc1

    return best_x, best_sc


def direct_max_iter(n, time_left):
    if time_left < 60.0:
        return 15
    if time_left < 150.0:
        return 40
    if n <= 40:
        return 120
    if n <= 55:
        return 100
    return 80


def smooth_max_iter(n, time_left):
    if time_left < 60.0:
        return 15
    if n <= 50:
        return 35
    return 25


def choose_lengths(parent_n):
    if parent_n is None or not np.isfinite(parent_n) or parent_n < 4 or parent_n > 80:
        parent_n = 45
    parent_n = int(parent_n)

    cand = [parent_n]
    extras = [
        parent_n - 6,
        parent_n - 3,
        36,
        40,
        44,
        parent_n + 4,
        parent_n + 8,
        32,
        24,
        48,
    ]

    seen = {parent_n}
    for n in extras:
        n = int(n)
        if 16 <= n <= 72 and n not in seen:
            seen.add(n)
            cand.append(n)

    return cand


def default_base(n=45):
    x = (np.arange(n) + 0.5) / float(n)
    v = 0.5 + 0.25 * np.sin(2.0 * np.pi * 1.7 * x) + 0.10 * np.cos(4.0 * np.pi * x)
    return make_feasible(v)


def validate_parent(p):
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=float).ravel()
    except Exception:
        return None

    if arr.size < 2 or not np.all(np.isfinite(arr)):
        return None

    arr = np.clip(arr, 0.0, 1.0)
    if arr.size > 80:
        arr = interpolate_to(arr, 45)

    return make_feasible(arr)


def consider_update(best_c, best_h, best_n, cand_c, cand_h, tol=1e-12):
    if cand_h is None or cand_h.size == 0:
        return best_c, best_h, best_n
    if not np.isfinite(cand_c):
        return best_c, best_h, best_n

    if cand_c < best_c - tol:
        return float(cand_c), np.array(cand_h, dtype=float, copy=True), int(cand_h.size)

    if abs(cand_c - best_c) <= tol and cand_h.size < best_n:
        return float(cand_c), np.array(cand_h, dtype=float, copy=True), int(cand_h.size)

    return best_c, best_h, best_n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)

    reserve = max(15.0, 0.07 * budget_s)
    deadline = t0 + max(1.0, budget_s - reserve)

    parent_raw = globals().get("initial_h_values")
    base_h = validate_parent(parent_raw)
    if base_h is None or base_h.size < 2:
        base_h = default_base(45)

    best_h = make_feasible(base_h)
    best_c = official_score(best_h)
    best_n = int(best_h.size)

    if not np.isfinite(best_c) or best_n < 2:
        best_h = default_base(45)
        best_c = official_score(best_h)
        best_n = int(best_h.size)

    lengths = choose_lengths(best_n)
    best_betas = [8.0, 30.0, 120.0, 500.0]
    max_start_gap = 0.06

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break

        n_random = 10 if n == best_n else 5
        max_direct = 4 if n == best_n else 2
        smooth_count = 1
        rounds = 0

        while rounds < 3 and time.time() < deadline - 35.0:
            starts = build_starts(
                n,
                int(seed) + 17 * rounds + 101 * int(n),
                base_h,
                best_h,
                n_random,
            )
            if not starts:
                break

            processed = 0
            for idx in range(len(starts)):
                if processed >= max_direct:
                    break

                sc0, x0 = starts[idx]
                if sc0 > best_c + max_start_gap:
                    break

                time_left = deadline - time.time()
                if time_left < 50.0:
                    break

                x1, sc1 = solve_direct(x0, direct_max_iter(n, time_left), deadline)
                best_c, best_h, best_n = consider_update(best_c, best_h, best_n, sc1, x1)

                time_left = deadline - time.time()
                if idx < smooth_count and time_left > 80.0:
                    xs = solve_smooth(x0, best_betas, smooth_max_iter(n, time_left), deadline)
                    scs = official_score(xs)
                    best_c, best_h, best_n = consider_update(best_c, best_h, best_n, scs, xs)

                    xd, sd = solve_direct(xs, direct_max_iter(n, deadline - time.time()), deadline)
                    best_c, best_h, best_n = consider_update(best_c, best_h, best_n, sd, xd)

                processed += 1

            rounds += 1

    if time.time() < deadline - 60.0:
        starts = build_starts(best_n, int(seed) + 987654, base_h, best_h, 12)
        processed = 0

        for idx in range(len(starts)):
            if processed >= 3:
                break

            sc0, x0 = starts[idx]
            if sc0 > best_c + 0.04:
                break

            time_left = deadline - time.time()
            if time_left < 60.0:
                break

            x1, sc1 = solve_direct(x0, direct_max_iter(best_n, time_left), deadline)
            best_c, best_h, best_n = consider_update(best_c, best_h, best_n, sc1, x1)

            time_left = deadline - time.time()
            if time_left > 80.0:
                xs = solve_smooth(x0, [30.0, 150.0, 600.0], smooth_max_iter(best_n, time_left), deadline)
                scs = official_score(xs)
                best_c, best_h, best_n = consider_update(best_c, best_h, best_n, scs, xs)

                xd, sd = solve_direct(xs, direct_max_iter(best_n, deadline - time.time()), deadline)
                best_c, best_h, best_n = consider_update(best_c, best_h, best_n, sd, xd)

            processed += 1

    time_left = deadline - time.time()
    if time_left > 120.0:
        xd, sd = solve_direct(best_h, 180, deadline)
        best_c, best_h, best_n = consider_update(best_c, best_h, best_n, sd, xd)

    best_h = make_feasible(best_h)
    best_c = official_score(best_h)

    if not np.isfinite(best_c) or best_h.size < 2:
        best_h = default_base(45)
        best_c = official_score(best_h)

    reported = float(best_c + 1e-13)
    if not np.isfinite(reported) or reported < 0.0:
        reported = float(max(0.0, best_c)) + 1e-13

    return best_h, reported, int(best_h.size)
