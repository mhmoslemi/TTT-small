def c5_gradient(h, tau=50.0):
    """Returns (smooth_c5, gradient) for the log-sum-exp smooth approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx  # (2n-1,)
    
    c_max = np.max(c)
    exp_c = np.exp(tau * (c - c_max))
    Z = np.sum(exp_c)
    weights = exp_c / Z
    c5_s = (c_max + np.log(Z)) / tau
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        w = weights[lag + n - 1]
        if w < 1e-12:
            continue
        if lag == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            d = lag
            k1 = np.arange(d, n)
            grad[k1] += w * dx * (1.0 - h[k1 - d])
            k2 = np.arange(0, n - d)
            grad[k2] -= w * dx * h[k2 + d]
        else:

            k1 = np.arange(0, n - d)
            grad[k1] += w * dx * (1.0 - h[k1 + d])
            k2 = np.arange(d, n)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_single_lag(h, lag, dx):
    """Gradient of c_lag with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        # c_lag = dx * sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        k1 = np.arange(lag, n)
        g[k1] += dx * (1.0 - h[k1 - lag])
        k2 = np.arange(0, n - lag)
        g[k2] -= dx * h[k2 + lag]
    else:
        d = -lag
        # c_{-d} = dx * sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
        k1 = np.arange(0, n - d)
        g[k1] += dx * (1.0 - h[k1 + d])
        k2 = np.arange(d, n)
        g[k2] -= dx * h[k2 - d]
    return g

def c5_smooth_and_grad(h, tau):
    """Log-sum-exp smooth approximation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    Z = np.sum(exp_c)
    weights = exp_c / Z
    c5_s = (c_max + np.log(Z)) / tau
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-13:

        lag = idx - (n - 1)
        grad += w * grad_single_lag(h, lag, dx)

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

        found = False

def binary_flip_search(h0, n_iter=100000, time_limit=30.0, start_time=None):

def downscale(h, n_new):
    """Downscale/resize h to n_new points."""
    n = len(h)
    if n == n_new:
        return h.copy()

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_constraint(h_new, n_new / 2.0)
