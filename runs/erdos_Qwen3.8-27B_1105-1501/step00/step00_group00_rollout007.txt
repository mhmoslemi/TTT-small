def evaluate_c5(h):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_feasible(x, n):
    """Repair x to be feasible: 0<=x_i<=1, sum(x)=n/2"""
    h = np.clip(x, 0.0, 1.0).astype(np.float64)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum; add to elements with room
        room = 1.0 - h
        order = np.argsort(-room)  # most room first
        for idx in order:
            add = min(diff, room[idx])
            h[idx] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        # Need to decrease sum; remove from elements with value
        order = np.argsort(h)  # most value first... wait, we want to reduce
        # Actually remove from largest values first to stay in [0,1]
        order = np.argsort(-h)
        for idx in order:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
            if abs(diff) < 1e-12:
                break
    # Final adjustment for floating point
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        # Distribute to first element with room
        if residual > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] = min(1.0, h[i] + residual)
                    break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    h[i] = max(0.0, h[i] + residual)
                    break
    return h

def compute_gradients(h, n, dx):
    """Compute gradients of all correlation components."""
    # Returns array of shape (2n-1, n)
    grads = np.zeros((2*n - 1, n))
    for m in range(2 * n - 1):
        ell = m - (n - 1)
        if ell >= 0:
            # C_m = dx * dot(h[ell:], 1-h[:n-ell])
            size = n - ell
            for j in range(ell, n):
                grads[m, j] += dx * (1.0 - h[j - ell])
            for j in range(n - ell):
                grads[m, j] -= dx * h[j + ell]
        else:
            p = -ell
            size = n - p
            for j in range(size):
                grads[m, j] += dx * (1.0 - h[j + p])
            for j in range(p, n):
                grads[m, j] -= dx * h[j - p]
    return grads
