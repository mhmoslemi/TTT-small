import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def _project_sum(h, target):
    """Project h to sum=target while keeping 0<=h<=1."""
    h = h.copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def _smooth_max_obj(x, tau, target_sum):
    h = np.clip(x, 0, 1)
    corr = c5_vector(h)
    cmax = np.max(corr)
    shifted = (corr - cmax) / tau
    return cmax + tau * np.log(np.sum(np.exp(shifted)))

def _smooth_max_grad(x, tau, target_sum):
    h = np.clip(x, 0, 1)
    n = len(h)
    dx = 2.0 / n
    corr = c5_vector(h)
    cmax = np.max(corr)
    shifted = (corr - cmax) / tau
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    # corr has 2n-1 elements, indexed 0 to 2n-2
    # corr[idx] corresponds to lag = idx - (n-1)
    for idx in range(2 * n - 1):
        if p[idx] < 1e-15:
            continue
        k = idx - (n - 1)
        w = p[idx] * dx
        if k == 0:
            grad += w * (1 - 2 * h)
        elif k > 0:
            # c_k = dx * sum_{i=0}^{n-k-1} h_i*(1-h_{i+k})
            # d/dh_i for i in [0, n-k-1]: dx*(1-h_{i+k})
            # d/dh_i for i in [k, n-1]: dx*(-h_{i-k})
            for i in range(n - k):
                grad[i] += w * (1 - h[i + k])
                grad[i + k] += w * (-h[i])
        else:
            kp = -k
            for i in range(kp, n):
                grad[i] += w * (1 - h[i - kp])
                grad[i - kp] += w * (-h[i])
    return grad

    h_new = np.clip(h_new, 0, 1)
    h_new = _project_sum(h_new, target_sum)

        if abs(diff) < 1e-15:
            break

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def _project_sum(h, target):
    h = h.copy()
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        count = int(np.sum(mask))
        if count == 0:
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, tau):
    corr = c5_vector(x)
    cmax = np.max(corr)
    shifted = (corr - cmax) / tau
    return cmax + tau * np.log(np.sum(np.exp(shifted)))

def _smooth_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    corr = c5_vector(x)
    cmax = np.max(corr)
    shifted = (corr - cmax) / tau
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = p[m]
        if w < 1e-16:
            continue
        k = m - (n - 1)
        w_dx = w * dx
        for j in range(n):
            val = 0.0
            jm = j - k
            if 0 <= jm < n:
                val += 1.0 - x[jm]
            jp = j + k
            if 0 <= jp < n:
                val -= x[jp]
            grad[j] += w_dx * val
    return grad

                   options={'maxiter': maxiter, 'ftol': 1e-14})

                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = hc
            except Exception:
                pass

    # Phase 6: Binary local search on current best if binary-like
    if time_left() > 40:
        n = len(best_h)
        if n % 2 == 0:
            h_bin_like = np.where(best_h > 0.5, 1.0, 0.0)
            if int(np.sum(h_bin_like)) == n // 2:
                hb, cb = _binary_local_search(h_bin_like, n, max_iter=50000, seed=seed + 999)
                if cb < best_c5:
                    best_c5 = cb
                    best_h = hb

    # Final feasibility check
    n = len(best_h)
    best_h = _project_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
