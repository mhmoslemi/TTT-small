    hi = float(np.max(g)) + 2.0

        else:

    e = np.exp((overlaps - m) / tau)
    return e / float(np.sum(e))

        else:

            grad[:n - sp] -= w * h[sp:]
    return grad

def project(g, target_sum):
    lo = float(np.min(g)) - 2.0
    hi = float(np.max(g)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(g - mid, 0.0, 1.0))) > target_sum:
            lo = mid
        else:
            hi = mid

    m = float(np.max(ov))
    e = np.exp((ov - m) / tau)
    return e / float(np.sum(e))

def grad_soft(h, p, dx):
    n = len(h)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        w = p[idx] * dx
        if abs(w) < 1e-15:

            g[s:] -= w * h[:n - s]
