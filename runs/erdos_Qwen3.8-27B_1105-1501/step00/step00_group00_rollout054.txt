    return grad

def mass_eq(x, args):
    n = x.size
    return x.sum() - n / 2.0

def mass_eq_jac(x, args):
    return np.ones(x.size)

def epi_obj(x, args):
    return x[-1]

def epi_obj_jac(x, args):
    g = np.zeros(x.size)
    g[-1] = 1.0
    return g

def epi_eq(x, args):
    n = x.size - 1
    return x[:n].sum() - n / 2.0

def epi_eq_jac(x, args):
    n = x.size - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def epi_ineq(x, args):
    n = x.size - 1
    active = args[0]
    h = x[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    return x[-1] - corr[active] * dx

def epi_ineq_jac(x, args):
    n = x.size - 1
    active = args[0]
    h = x[:n]
    dx = 2.0 / n
    na = len(active)
    J = np.zeros((na, n + 1))
    for i in range(na):
        J[i, :n] = -build_grad_row(h, active[i], dx)
        J[i, n] = 1.0
    return J

def transfer_resolution(h_src, n_tgt):
    n_src = h_src.size
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    cum_src = np.concatenate([[0.0], np.cumsum(h_src) * dx_src])
    x_src_nodes = np.linspace(0.0, 2.0, n_src + 1)
    x_tgt_nodes = np.linspace(0.0, 2.0, n_tgt + 1)
    area_at_tgt = np.interp(x_tgt_nodes, x_src_nodes, cum_src)
    bin_areas = np.diff(area_at_tgt)
    h_tgt = bin_areas / dx_tgt
    return project_feasible(h_tgt)

def local_pair_moves(h, max_evals, rng):
    n = h.size
    best = h.copy()
    best_c = exact_c5(best)
    deltas = [0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002]
    evals = 0
    per_delta = max(1, max_evals // len(deltas))
    for delta in deltas:
        for _ in range(per_delta):
            if evals >= max_evals:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if best[i] + delta > 1.0 + 1e-9 or best[j] - delta < -1e-9:
                continue
            trial = best.copy()
            trial[i] = min(trial[i] + delta, 1.0)
            trial[j] = max(trial[j] - delta, 0.0)
            c = exact_c5(trial)
            evals += 1
            if c < best_c - 1e-12:
                best = trial
                best_c = c
    return best, best_c

def local_clamp_moves(h, max_evals, rng):
    n = h.size
    best = h.copy()
    best_c = exact_c5(best)
    evals = 0
    for _ in range(max_evals):
        if evals >= max_evals:
            break
        i = rng.randint(n)
        if best[i] <= 1e-6 or best[i] >= 1.0 - 1e-6:
            continue
        target_val = 1.0 if best[i] > 0.5 else 0.0
        delta = target_val - best[i]
        j = rng.randint(n)
        if i == j:
            continue
        trial = best.copy()
        trial[i] = target_val
        trial[j] = trial[j] - delta
        if trial[j] < -1e-9 or trial[j] > 1.0 + 1e-9:
            for k in range(n):
                if k == i or k == j:
                    continue
                trial = best.copy()
                trial[i] = target_val
                trial[k] = trial[k] - delta
                if 0.0 <= trial[k] <= 1.0:
                    break
            else:
                continue
        trial = project_feasible(trial)
        c = exact_c5(trial)
        evals += 1
        if c < best_c - 1e-12:
            best = trial
            best_c = c
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s * 0.94

    def elapsed():
        return time.time() - t_start

    def can_continue():
        return elapsed() < deadline

    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).copy().reshape(-1)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            parent = project_feasible(parent)
            c = exact_c5(parent)
            if c < best_c:
                best_c = c
                best_h = parent.copy()
    except Exception:
        pass

    if best_h is None:
        n0 = 45
        best_h = np.full(n0, 0.5)
        best_h = project_feasible(best_h)
        best_c = exact_c5(best_h)

    n = best_h.size
    bounds = [(0.0, 1.0)] * n
    constraints_main = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac, 'args': ()}]

    starts = [best_h.copy()]

    rev = best_h[::-1].copy()
    starts.append(project_feasible(rev))

    for phase in range(3):
        x_vals = np.arange(n) * (1.0 + np.sqrt(5.0)) / 2.0 + phase * 0.37
        fracs = x_vals - np.floor(x_vals)
        sorted_idx = np.argsort(fracs)
        h_bin = np.zeros(n)
        int_target = int(n / 2.0)
        h_bin[sorted_idx[:int_target]] = 1.0
        if n % 2 == 1:
            h_bin[sorted_idx[int_target]] = 0.5
        h_bin = project_feasible(h_bin)
        starts.append(h_bin)

    for freq in [2, 3, 4, 5, 7]:
        x_lin = np.linspace(0.0, 2.0 * freq * np.pi, n)
        h_sin = 0.5 + 0.35 * np.sin(x_lin)
        starts.append(project_feasible(h_sin))

    for sigma in [0.15, 0.08, 0.04]:
        noisy = best_h + sigma * rng.randn(n)
        starts.append(project_feasible(noisy))

    seen_keys = set()
    unique_starts = []
    for s in starts:
        key = tuple(np.round(s, 3))
        if key not in seen_keys:
            seen_keys.add(key)
            unique_starts.append(s)
    starts = unique_starts

    tau_seq = [0.04, 0.018, 0.008, 0.003, 0.0012]

    for si, start in enumerate(starts):
        if not can_continue():
            break
        current = start.copy()
        for tau in tau_seq:
            if not can_continue():
                break
            try:
                res = minimize(
                    smooth_obj, current,
                    args=(tau,),
                    method='SLSQP',
                    jac=smooth_jac,
                    bounds=bounds,
                    constraints=constraints_main,
                    options={'maxiter': 200, 'ftol': 1e-9}
                )
                candidate = project_feasible(res.x)
                c = exact_c5(candidate)
                if c < best_c - 1e-10:
                    best_c = c
                    best_h = candidate.copy()
                current = project_feasible(res.x)
                if abs(tau - tau_seq[-1]) > 1e-12:
                    current = project_feasible(current)
            except Exception:
                pass

    # Epigraph polish on best
    if can_continue():
        n_e = best_h.size
        dx_e = 2.0 / n_e
        corr_vals = np.correlate(best_h, 1.0 - best_h, mode='full') * dx_e
        t_init = np.max(corr_vals)
        active_size = min(12, 2 * n_e - 1)
        active_idx = np.argsort(corr_vals)[-active_size:]
        active_idx = sorted(active_idx.tolist())

        x_epi = np.concatenate([best_h, [t_init]])
        epi_bounds = [(0.0, 1.0)] * n_e + [(0.0, 1.0)]
        epi_cons = [
            {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac, 'args': ()},
            {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac, 'args': (active_idx,)}
        ]

        for iteration in range(3):
            if not can_continue():
                break
            try:
                res = minimize(
                    epi_obj, x_epi,
                    args=(),
                    method='SLSQP',
                    jac=epi_obj_jac,
                    bounds=epi_bounds,
                    constraints=epi_cons,
                    options={'maxiter': 150, 'ftol': 1e-9}
                )
                h_cand = project_feasible(res.x[:n_e])
                c = exact_c5(h_cand)
                if c < best_c - 1e-10:
                    best_c = c
                    best_h = h_cand.copy()
                x_epi = np.concatenate([project_feasible(res.x[:n_e]), [np.max(np.correlate(res.x[:n_e], 1.0 - res.x[:n_e], mode='full') * dx_e)]])
                corr_new = np.correlate(x_epi[:n_e], 1.0 - x_epi[:n_e], mode='full') * dx_e
                active_idx = np.argsort(corr_new)[-active_size:]
                active_idx = sorted(active_idx.tolist())
                epi_cons[1]['args'] = (active_idx,)
            except Exception:
                pass

    # Local pair moves
    if can_continue():
        best_h, best_c = local_pair_moves(best_h, 2000, rng)

    # Local clamp moves
    if can_continue():
        best_h, best_c = local_clamp_moves(best_h, 500, rng)

    # More pair moves with smaller steps
    if can_continue():
        n_m = best_h.size
        best2 = best_h.copy()
        best2_c = best_c
        for delta in [0.003, 0.0015, 0.0008, 0.0004]:
            for _ in range(500):
                if not can_continue():
                    break
                i = rng.randint(n_m)
                j = rng.randint(n_m)
                if i == j:
                    continue
                if best2[i] + delta > 1.0 or best2[j] - delta < 0.0:
                    continue
                trial = best2.copy()
                trial[i] += delta
                trial[j] -= delta
                c = exact_c5(trial)
                if c < best2_c - 1e-12:
                    best2 = trial
                    best2_c = c
        if best2_c < best_c - 1e-12:
            best_c = best2_c
            best_h = best2

    # Resolution transfer
    if can_continue():
        target_ns = []
        for mult in [1.33, 1.5, 2.0]:
            nn = int(round(n * mult))
            if nn <= 200 and nn > n:
                target_ns.append(nn)
        for half in [0.75, 0.5]:
            nn = int(round(n * half))
            if nn >= 20 and nn < n:
                target_ns.append(nn)

        target_ns = sorted(set(target_ns))

        for n_t in target_ns:
            if not can_continue():
                break
            h_t = transfer_resolution(best_h, n_t)
            c_t = exact_c5(h_t)
            if c_t < best_c - 1e-10:
                best_c = c_t
                best_h = h_t.copy()
                n = n_t
                bounds = [(0.0, 1.0)] * n
                constraints_main[0]['args'] = ()

            # Quick SLSQP at new resolution
            if can_continue():
                h_opt = h_t.copy()
                for tau in [0.02, 0.008]:
                    if not can_continue():
                        break
                    try:
                        res = minimize(
                            smooth_obj, h_opt,
                            args=(tau,),
                            method='SLSQP',
                            jac=smooth_jac,
                            bounds=bounds,
                            constraints=constraints_main,
                            options={'maxiter': 100, 'ftol': 1e-8}
                        )
                        h_opt = project_feasible(res.x)
                        c_opt = exact_c5(h_opt)
                        if c_opt < best_c - 1e-10:
                            best_c = c_opt
                            best_h = h_opt.copy()
                    except Exception:
                        pass

                # Quick local moves at new resolution
                if can_continue():
                    h_opt, c_opt = local_pair_moves(h_opt, 500, rng)
                    if c_opt < best_c - 1e-10:
                        best_c = c_opt
                        best_h = h_opt.copy()

    # Final: try a few more diverse starts if time remains
    if can_continue():
        for trial_start in [
            project_feasible(best_h + 0.02 * rng.randn(n)),
            project_feasible(best_h * 0.8 + 0.1),
        ]:
            if not can_continue():
                break
            current = trial_start.copy()
            for tau in [0.02, 0.008, 0.003]:
                if not can_continue():
                    break
                try:
                    res = minimize(
                        smooth_obj, current,
                        args=(tau,),
                        method='SLSQP',
                        jac=smooth_jac,
                        bounds=bounds,
                        constraints=constraints_main,
                        options={'maxiter': 150, 'ftol': 1e-9}
                    )
                    candidate = project_feasible(res.x)
                    c = exact_c5(candidate)
                    if c < best_c - 1e-10:
                        best_c = c
                        best_h = candidate.copy()
                    current = project_feasible(res.x)
                except Exception:
                    pass

    # Final verification and cleanup
    best_h = project_feasible(best_h)
    best_c = exact_c5(best_h)

    return (best_h, best_c, best_h.size)
