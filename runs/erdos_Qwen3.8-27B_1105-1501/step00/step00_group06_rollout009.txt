    corr = np.correlate(h, omh, mode="full")

                    deltas += [d_pos * f for f in (0.05, 0.15, 0.3, 0.5, 0.8, 1.0)]
                if d_neg > 1e-10:
                    deltas += [-d_neg * f for f in (0.05, 0.15, 0.3, 0.5, 0.8, 1.0)]
                if not deltas:
                    continue
                bd = 0.0
                bv = cur_f
                for delta in deltas:
                    h_test = h.copy()
                    h_test[i] -= delta
                    h_test[j] += delta
                    f_t = _eval(h_test)
                    if f_t < bv - 1e-13:
                        bv = f_t
                        bd = delta
                if bd != 0.0:
                    h[i] -= bd
                    h[j] += bd
                    cur_f = bv
                    any_improve = True
                    if cur_f < best_f:
                        best_f = cur_f
                        best_h = h.copy()
        if not any_improve:
            break

    return best_h, best_f


def _shift_h(h, shift):
    """Circular shift with re-projection."""
    n = len(h)
    h_shifted = np.roll(h, shift)
    return _project(h_shifted, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    deadline = t_start + budget_s * 0.95

    # Get parent
    parent = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            parent = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    best_h = None
    best_c5 = np.inf

    def _update(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 1: Multi-resolution gradient optimization
    n_list = [70, 50, 100, 60, 90, 80, 120, 40, 110, 64, 96]
    if parent is not None:
        n_list = [len(parent)] + [n for n in n_list if n != len(parent)]

    for n in n_list:
        if time.time() > deadline:
            break
        time_per_n = max(20.0, min(80.0, (deadline - time.time()) / max(1, len(n_list))))

        candidates = []
        if parent is not None:
            candidates.append(_resample(parent, n))
            # Shifted versions
            for sh in [1, -1, 2, n // 2]:
                candidates.append(_shift_h(_resample(parent, n), sh))
        candidates.append(_rand_binary(n, rng))
        candidates.append(_block_h(n))
        # Smooth random
        h_sm = rng.standard_normal(n)
        h_sm = 0.5 + 0.3 * h_sm / (np.max(np.abs(h_sm)) + 1e-10)
        candidates.append(_project(h_sm, n / 2.0))

        for h0 in candidates:
            if time.time() > deadline:
                break
            tl = min(time_per_n, deadline - time.time())
            if tl < 5.0:
                break
            h_opt, c5 = _grad_opt(h0, tl, rng)
            _update(h_opt, c5)

    # Phase 2: Polish the best solution found
    if best_h is not None and time.time() < deadline:
        n = len(best_h)
        remaining = deadline - time.time()
        # Split time between pair polish and more gradient
        if remaining > 30:
            h_pol, c5_pol = _pair_polish(best_h, remaining * 0.5, rng)
            _update(h_pol, c5_pol)
            if time.time() < deadline:
                h_grad, c5_grad = _grad_opt(best_h, deadline - time.time(), rng)
                _update(h_grad, c5_grad)
        else:
            h_pol, c5_pol = _pair_polish(best_h, remaining, rng)
            _update(h_pol, c5_pol)

    # Phase 3: Final pair polish on absolute best
    if best_h is not None and time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 5:
            h_final, c5_final = _pair_polish(best_h, remaining, rng)
            _update(h_final, c5_final)

    # Safety: if we never found anything, return the parent
    if best_h is None:
        if parent is not None:
            best_h = parent.copy()
            best_c5 = _eval(best_h)
        else:
            best_h = _block_h(70)
            best_c5 = _eval(best_h)

    # Final verification
    final_c5 = _eval(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
