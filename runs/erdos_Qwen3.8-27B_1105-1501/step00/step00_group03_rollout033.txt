import time

    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0-h, mode='full')) * dx

def make_full(u, odd=False):
    m = len(u)
    if odd:
        n = 2*m + 1
        h = np.empty(n)
        h[:m] = u
        h[m] = 0.5
        h[m+1:] = 1.0 - u[::-1]
    else:
        n = 2*m
        h = np.empty(n)
        h[:m] = u
        h[m:] = 1.0 - u[::-1]
    return h

    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return h_sym

def extract_u(h_sym):
    n = len(h_sym)
    m = n // 2
    return h_sym[:m].copy()

def interp_symmetrize(h_src, n_target, odd_target=False):
    """Interpolate h_src to n_target points, then symmetrize and extract u."""
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)
    h_interp = np.interp(x_tgt, x_src, h_src)
    h_sym = symmetrize(h_interp)
    return extract_u(h_sym)

def corr_value(L, h, n):
    """Compute F_L(h) = sum h[i]*(1-h[i+L]) for valid i."""
    lo = max(0, -L)
    hi = min(n, n - L)
    if lo >= hi:
        return 0.0
    return np.sum(h[lo:hi] * (1.0 - h[lo+L:hi+L]))

def corr_grad_u(L, h, n, m):
    """Compute dF_L/du."""
    lo = max(0, -L)
    hi = min(n, n - L)
    grad_h = np.zeros(n)
    for i in range(lo, hi):
        grad_h[i] += (1.0 - h[i+L])
        grad_h[i+L] += (-h[i])
    grad_u = np.zeros(m)
    for j in range(m):
        grad_u[j] = grad_h[j] - grad_h[n-1-j]
    return grad_u

def slsqp_solve(u, active_lags, n, odd, theta_init, maxiter=100):
    """Run SLSQP with given active constraints."""
    m = len(u)
    dx = 2.0 / n
    h = make_full(u, odd)
    
    # Precompute gradients for active lags
    # active_lags is a list of L values (shifts from -(n-1) to n-1)
    
    def objective(z):
        return z[m]  # theta
    
    def objective_grad(z):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g
    
    def constraint_fun(i, z):
        u_vec = z[:m]
        u_vec = np.clip(u_vec, 0, 1)
        h_vec = make_full(u_vec, odd)
        cval = z[m] - dx * corr_value(active_lags[i], h_vec, n)
        return cval
    
    def constraint_jac(i, z):
        u_vec = np.clip(z[:m], 0, 1)
        h_vec = make_full(u_vec, odd)
        g = np.zeros(m + 1)
        g[:m] = -dx * corr_grad_u(active_lags[i], h_vec, n, m)
        g[m] = 1.0
        return g
    
    constraints = []
    for i in range(len(active_lags)):
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: constraint_fun(i, z),
            'jac': lambda z, i=i: constraint_jac(i, z)
        })
    
    bounds = [(0, 1)] * m + [(theta_init * 0.5, theta_init * 2.0)]
    
    z0 = np.append(u, theta_init)
    
    result = minimize(objective, z0, jac=objective_grad,
                      bounds=bounds, constraints=constraints,
                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if result.success:
        u_opt = np.clip(result.x[:m], 0, 1)
        return u_opt
    else:
        return u  # return original if failed

    dx = 2.0 / n
    u = u0.copy()
    best_c5 = score_full(make_full(u, odd))
    
    for round_num in range(max_rounds):
        h = make_full(u, odd)
        corr_vals = np.correlate(h, 1.0-h, mode='full') * dx
        c5 = corr_vals.max()
        
        if c5 < best_c5:

            best_u = u.copy()

        active_lags = top_indices - (n - 1)  # convert to shift L

        h_new = make_full(u_new, odd)
        c5_new = score_full(h_new)
        
        if c5_new < c5 - 1e-10:
            u = u_new
        else:
            # Try expanding active set
            break
    
    return u, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def score_full(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_full(u, odd=False):
    m = len(u)
    if odd:
        n = 2 * m + 1
        h = np.empty(n)
        h[:m] = u
        h[m] = 0.5
        h[m+1:] = 1.0 - u[::-1]
    else:
        n = 2 * m
        h = np.empty(n)
        h[:m] = u
        h[m:] = 1.0 - u[::-1]
    return h

def symmetrize_and_extract(h):
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    n = len(h_sym)
    m = n // 2
    return h_sym[:m].copy()

def interp_to_sym_u(h_src, n_target, odd_target=False):
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)
    h_interp = np.interp(x_tgt, x_src, h_src)
    h_sym = 0.5 * (h_interp + 1.0 - h_interp[::-1])
    m = n_target // 2
    return h_sym[:m].copy()

def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_grad_u_one(k_idx, h, n, m):

    lo = max(0, -L)

        return np.zeros(m)

    m = len(u)
    K = len(active_indices)
    if K == 0:
        return u, False

    def obj(z):
        return z[m]

        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

        uc = np.clip(z[:m], 0.0, 1.0)
        h = make_full(uc, odd)
        return z[m] - corr_array(h)[ki]

    def _con_jac(z, ki):
        uc = np.clip(z[:m], 0.0, 1.0)
        h = make_full(uc, odd)
        g = corr_grad_u_one(ki, h, n, m)

        jac[:m] = -g
        jac[m] = 1.0
