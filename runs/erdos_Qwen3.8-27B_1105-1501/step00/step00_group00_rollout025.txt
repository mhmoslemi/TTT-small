```python
def c5_values(h):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def c5_max(h):
    """Compute the max correlation value."""
    return np.max(c5_values(h))

def c5_gradient(h):
    """Compute gradient of the max correlation (using the active constraint)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    grad = np.zeros(n)

    shifted = tau * (corr - c_max)
    log_sum = np.log(np.sum(np.exp(shifted))) + tau * c_max

    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr_scaled(h):
    """Compute scaled correlation values."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_eval(h):
    """Compute C5 value."""
    return float(np.max(compute_corr_scaled(h)))

def project_feasible(h, target=None):
    """Project h onto {0<=x<=1, sum(x)=target}."""
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lambda_star = (lo + hi) / 2
    h_proj = np.clip(h - lambda_star, 0.0, 1.0)
    # Final correction: adjust one element to make sum exact
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Find an element that can absorb the correction
        if diff > 0:
            idx = np.argmax(h_proj < 1.0)
            h_proj[idx] += min(diff, 1.0 - h_proj[idx])
        else:
            idx = np.argmin(h_proj > 0.0)
            h_proj[idx] += max(diff, -h_proj[idx])
    return h_proj

def corr_gradient(h, k):
    """Compute gradient of the k-th correlation value (scaled)."""
    n = len(h)
    dx = 2.0 / n
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        l = lag
        if l == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[l:n-l] = (1.0 - h[0:n-2*l]) - h[2*l:n]
            grad[n-l:n] = 1.0 - h[n-2*l:n-l]
            grad[0:l] = -h[l:n-l]
    else:
        p = -lag
        if p == 0:
            grad = 1.0 - 2.0 * h
        else:
            grad[p:n-p] = (1.0 - h[2*p:n]) - h[0:n-2*p]
            grad[0:p] = 1.0 - h[p:2*p]
            grad[n-p:n] = -h[n-2*p:n-p]
    return grad * dx

def smooth_objective_and_grad(h, tau):
    """Compute smoothed objective and its gradient."""
    corr = compute_corr_scaled(h)
    c_max = np.max(corr)
    shifted = tau * (corr - c_max)
    exp_s = np.exp(shifted)
    z = np.sum(exp_s)
    f = (np.log(z) + tau * c_max) / tau
    grad = np.zeros(len(h))
    for k in range(len(corr)):
        w = exp_s[k] / z
        if w < 1e-14:
            continue
        grad += w * corr_gradient(h, k)
    return f, grad

def projected_gradient_descent(h, tau_start=100.0, tau_end=500.0, 
                                 n_iters=2000, lr_start=0.01, lr_end=0.001):
    """Run projected gradient descent with annealing tau and learning rate."""
    n = len(h)
    target = n / 2.0
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_c5 = c5_eval(h)
    
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
        
        f, grad = smooth_objective_and_grad(h, tau)
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        c5_new = c5_eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_solve(h0, max_iter=1000):
    """Solve the epigraph formulation using SLSQP."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    def objective(x):
        return x[-1]  # maximize t -> minimize t
    
    def gradient(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    constraints = []
    # Inequality constraints: t - C_k(h) >= 0
    for k in range(2 * n - 1):
        def constr(i=k, x=x):
            hx = x[:n]
            corr_val = np.correlate(hx, 1.0 - hx, mode='full')[k] * dx
            return x[-1] - corr_val
        constraints.append({'type': 'ineq', 'fun': constr})
    
    # Equality constraint
    def eq_constr(x):
        return np.sum(x[:n]) - target
    constraints.append({'type': 'eq', 'fun': eq_constr})
    
    bounds = [(0, 1)] * n + [(0, None)]
    x0 = np.concatenate([h0, [c5_eval(h0) + 0.01]])
    
    res = minimize(objective, x0, method='SLSQP', jac=gradient,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
    
    if res.success and res.x[-1] < c5_eval(h0):
        h_sol = project_feasible(res.x[:n], target)
        c5_sol = c5_eval(h_sol)
        return h_sol, c5_sol
    return None, None
```

But actually, for the purpose of the correlation calculation, what matters is the relative spacing. The correlate function just looks at element-wise products with shifts. So the "physical position" doesn't matter, only the pattern.

However, for the integral interpretation, the dx factor matters: dx = 2/n. So a function h with n samples and sum(h)*dx = 1 has a "mass" of 1 regardless of n.
