def slsqp_polish(h_start, deadline):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr_indices = np.arange(0, 2 * n - 1) - (n - 1)
    
    def _objective(x):
        return x[n]
    
    def _objective_grad(x):
        gx = np.zeros(n + 1)
        gx[n] = 1.0
        return gx
    
    def _ineq_con(x):
        h = x[:n]
        theta = x[n]
        vals = np.zeros(n_corr)
        for idx, k in enumerate(corr_indices):
            if k >= 0:
                i_vals = np.arange(0, n - k)
            else:
                i_vals = np.arange(-k, n)
            f_k = np.sum(h[i_vals] * (1.0 - h[i_vals + k]))
            vals[idx] = theta - dx * f_k
        return vals
    
    def _ineq_jac(x):
        h = x[:n]
        jac = np.zeros((n_corr, n + 1))
        for idx, k in enumerate(corr_indices):
            jac[idx, n] = 1.0
            g_k = compute_gradient_row(h, k, n)
            jac[idx, :n] = -dx * g_k
        return jac
    
    def _eq_con(x):
        return np.sum(x[:n]) - n / 2.0
    
    def _eq_jac(x):
        gx = np.zeros(n + 1)
        gx[:n] = 1.0
        return gx
    
    sc_start = official_score(h_start)
    theta0 = max(sc_start + 0.01, 0.05)
    x0 = np.concatenate([h_start, [theta0]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    cons = [
        {'type': 'ineq', 'fun': _ineq_con, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}
    ]
    
    remaining = deadline - time.perf_counter()
    max_iter = max(15, min(50, int(remaining / 8)))
    
    result = minimize(
        _objective, x0,
        method='SLSQP',
        jac=_objective_grad,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-10}
    )
