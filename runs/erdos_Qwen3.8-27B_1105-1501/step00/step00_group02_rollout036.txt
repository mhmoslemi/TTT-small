def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_h_from_y(y, N):
    z = np.empty(N)
    z[:N-1] = y
    z[N-1] = -np.sum(y)
    target = 0.5 * N
    lo, hi = -40.0, 40.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(1.0 / (1.0 + np.exp(-(z + mid))))
        if s < target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    h = 1.0 / (1.0 + np.exp(-(z + tau)))
    return h


def objective_and_grad(y, beta, N):
    h = compute_h_from_y(y, N)
    d = h * (1.0 - h)
    C = np.correlate(h, 1.0 - h, mode='full')
    bC = beta * C
    maxbC = np.max(bC)
    exp_bC = np.exp(bC - maxbC)
    sum_exp = np.sum(exp_bC)
    psi = (maxbC + np.log(sum_exp)) / beta
    alpha = exp_bC / sum_exp
    # gvec[m] = sum_i alpha[i] * dC[i]/dh[m]
    N1 = N - 1
    gvec = np.zeros(N)
    for m in range(N):
        # First term: dot(1-h, alpha[m:m+N])
        # alpha indices: m, m+1, ..., m+N-1 (all valid since m<=N-1, m+N-1<=2N-2)
        gvec[m] = np.dot(1.0 - h, alpha[m:m+N])
        # Second term: -dot(alpha[0:m+1], h[N-1-m:N])
        if m > 0:
            gvec[m] -= np.dot(alpha[:m+1], h[N1-m:])
        else:
            gvec[m] -= np.dot(alpha[:1], h[N1:])
    # Chain through logistic + mass constraint
    D = np.sum(d)

    return psi, grad

    z = logit_h - np.mean(logit_h)
    return z[:N-1]


def snap_dyadic(h, N, nbits=22):
    Q = 1 << nbits
    target_int = N * Q // 2
    k = np.round(h * Q).astype(np.int64)
    k = np.clip(k, 0, Q)
    diff = target_int - int(np.sum(k))
    if diff > 0:
        # Add units: prefer positions with highest fractional remainder
        frac = h * Q - np.floor(h * Q)
        order = np.argsort(-frac)
        count = 0
        for idx in order:
            if count >= diff:
                break
            if k[idx] < Q:
                k[idx] += 1
                count += 1
        # If not all added, distribute to any position with room
        if count < diff:
            for idx in range(N):
                if count >= diff:
                    break
                if k[idx] < Q:
                    k[idx] += 1
                    count += 1
    elif diff < 0:
        # Remove units: prefer positions with lowest fractional part
        frac = h * Q - np.floor(h * Q)
        order = np.argsort(frac)
        count = 0
        for idx in order:
            if count >= -diff:
                break
            if k[idx] > 0:
                k[idx] -= 1
                count += 1
        if count < -diff:
            for idx in range(N):
                if count >= -diff:
                    break
                if k[idx] > 0:
                    k[idx] -= 1
                    count += 1
    return k.astype(np.float64) / float(Q)


def pair_swap_polish(h, N, nbits=22, max_iter=300, n_pairs_per_iter=300, rng=None):
    if rng is None:
        rng = np.random
    Q = 1 << nbits

    target_int = N * Q // 2

    if diff != 0:
        h = snap_dyadic(h, N, nbits)
        k = np.round(h * Q).astype(np.int64)
        k = np.clip(k, 0, Q)
    
    best_c = c5_exact(k.astype(np.float64) / Q)
    h_best = k.astype(np.float64) / Q
    
    for it in range(max_iter):
        improved = False
        for _ in range(n_pairs_per_iter):
            p = rng.randint(N)
            q = rng.randint(N)
            if p == q:
                continue
            for src, dst in [(p, q), (q, p)]:
                if k[src] <= 0 or k[dst] >= Q:
                    continue
                k[src] -= 1
                k[dst] += 1
                h_new = k.astype(np.float64) / Q
                c_new = c5_exact(h_new)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h_best = h_new.copy()
                    improved = True
                else:
                    k[src] += 1
                    k[dst] -= 1
        if not improved:
            break
    return h_best, best_c
