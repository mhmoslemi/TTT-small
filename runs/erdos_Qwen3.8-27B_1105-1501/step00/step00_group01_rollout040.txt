def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(corr, alpha):
    m = np.max(corr)
    exp_c = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(exp_c)) / alpha

def smooth_max_gradient(h, alpha):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    m = np.max(corr)
    exp_c = np.exp(alpha * (corr - m))
    p = exp_c / np.sum(exp_c)
    
    # p has length 2n-1, p[k] corresponds to shift s = k-(n-1)
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):

            m1 = j - s
            if 0 <= m1 < n:
                grad[j] += p[k] * (1.0 - h[m1])
            m2 = j + s
            if 0 <= m2 < n:
                grad[j] -= p[k] * h[m2]

def project_simplex(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def optimize(h0, alpha_init, lr_init, n_steps, seed=42):
    h = h0.copy()

    best_c5 = compute_c5(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    
    # Gradient computation (O(n^2))
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2*n - 1):
            shift = k - (n-1)
            m1 = j - shift
            if 0 <= m1 < n:
                s += p[k] * (1.0 - h[m1])
            m2 = j + shift
            if 0 <= m2 < n:
                s -= p[k] * h[m2]
        grad[j] = s
    return grad

def project(h, target):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0

        else:
            hi = mid

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    
    T1 = np.convolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2

def project(h, target):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def projected_gradient_step(h, target, lr, alpha):
    grad = smooth_max_grad(h, alpha)
    h_new = h - lr * grad
    return project(h_new, target)

def optimize_smoothed(h0, budget_per_run, seed=42):

    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()

    start = time.time()

    alpha = 5.0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    T1 = np.convolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2

def project_to_simplex(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def optimize_smoothed(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = np.clip(h0.copy(), 0.0, 1.0)
    h = project_to_simplex(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    # Annealed projected gradient
    alpha = 5.0
    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)
    
    step_count = 0
    while time.time() - t0 < time_budget:
        batch_size = 200
        for _ in range(batch_size):
            if time.time() - t0 >= time_budget:
                break
            grad = smooth_max_grad(h, alpha)
            v = momentum * v - lr * grad
            h = project_to_simplex(h + v, target)
            step_count += 1
        
        # Evaluate and track best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Anneal
        alpha *= 1.15
        lr *= 0.95
        momentum = min(momentum * 1.001, 0.99)
    
    return best_h, best_c5

def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = n // 2
    h[:ones] = 1.0
    if n % 2 == 1:
        h[ones] = 0.5
    idx = rng.permutation(n)
    return h[idx]

def coord_descent_refine(h, time_budget):
    n = len(h)
    target = float(n) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    # Try small perturbations

            if h[i] <= 0 or h[i] >= 1:
                continue
            for j in range(n):
                if j == i:
                    continue
                h[j] >= 1 or h[j] <= 0:
                    continue
                # Try moving small delta from i to j
                for delta in [0.01, 0.05, 0.1]:
                    h_test = best_h.copy()
                    h_test[i] -= delta
                    h_test[j] += delta
                    if h_test[i] < -1e-10 or h_test[j] > 1.0 + 1e-10:

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    T1 = np.convolve(1.0 - h, p, mode='full')[n - 1:2 * n - 1]
    T2 = np.correlate(p, h, mode='valid')[::-1]
    return T1 - T2


def project_to_simplex(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def optimize_smoothed(h0, time_budget, seed=42):
    n = len(h0)
    target = float(n) / 2.0
    h = np.clip(h0.astype(float).copy(), 0.0, 1.0)
    h = project_to_simplex(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

    alpha = 5.0
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    stall_count = 0
    stall_limit = 500

    while time.time() - t0 < time_budget:
        batch = 300
        for _ in range(batch):
            if time.time() - t0 >= time_budget:
                break
            grad = smooth_max_grad(h, alpha)
            v = momentum * v - lr * grad
            h_new = project_to_simplex(h + v, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                stall_count = 0
            else:
                stall_count += 1
            h = h_new

        if stall_count > stall_limit:
            # Restart from best with perturbation
            rng = np.random.RandomState(seed + int(stall_count))
            h = best_h + 0.05 * rng.randn(n)
            h = project_to_simplex(h, target)
            v = np.zeros(n)
            stall_count = 0
            lr = max(lr * 0.7, 0.001)

        alpha = alpha * 1.12 + 0.1
        lr = max(lr * 0.97, 0.0005)
        momentum = min(momentum * 1.0005, 0.99)

    return best_h, best_c5


def random_init(n, rng):
    h = np.zeros(n)
    ones = n // 2
    h[:ones] = 1.0
    if n % 2 == 1:
        h[ones] = 0.5
    idx = rng.permutation(n)
    return h[idx]


def structured_init(n, rng):
    # Try periodic-like patterns
    h = np.zeros(n)
    ones = n // 2
    if n % 2 == 1:
        h[n // 2] = 0.5
        ones_left = n // 2
    else:
        ones_left = ones
    # Place ones in a spread pattern
    step = n / max(ones_left, 1)
    positions = [int(round(i * step)) % n for i in range(ones_left)]
    positions = list(set(positions))
    for i, pos in enumerate(positions):
        if i < n - ones // 2 - (1 if n % 2 == 1 else 0):
            h[pos] = 1.0
    # Fill remaining
    if n % 2 == 0:
        need = ones - np.sum(h == 1.0)
        zeros = np.where(h == 0)[0]
        for i in range(min(int(need), len(zeros))):
            h[zeros[i]] = 1.0
    else:
        need1 = n // 2 - np.sum(h == 1.0)
        need05 = int(np.sum(h == 0.5))
        if need05 == 1 and np.sum(h == 1.0) == n // 2:
            pass  # already good
        else:
            zeros = np.where(h == 0)[0]
            for i in range(min(int(need1), len(zeros))):
                h[zeros[i]] = 1.0
    idx = rng.permutation(n)
    return h[idx]


def local_refine(h, time_budget, seed=42):
    n = len(h)
    target = float(n) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    while time.time() - t0 < time_budget:
        improved = False
        # Try random coordinate pairs
        for _ in range(5000):
            if time.time() - t0 >= time_budget:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if best_h[i] <= 1e-8 or best_h[j] >= 1.0 - 1e-8:
                continue
            delta = rng.choice([0.02, 0.05, 0.1, 0.2]) * rng.choice([-1, 1])
            h_test = best_h.copy()
            h_test[i] += delta
            h_test[j] -= delta
            if h_test[i] < -1e-10 or h_test[i] > 1.0 + 1e-10:
                continue
            if h_test[j] < -1e-10 or h_test[j] > 1.0 + 1e-10:
                continue
            h_test = np.clip(h_test, 0.0, 1.0)
            c5 = compute_c5(h_test)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_test.copy()
                improved = True

    return best_h, best_c5
