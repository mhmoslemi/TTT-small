import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_binary_initial(n, seed=0):
    """Create a random binary initial with n//2 ones."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def local_search_binary(h, time_limit=30.0, seed=0):
    """Greedy local search with pair swaps."""
    n = len(h)
    best_c5 = evaluate_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            if time.time() - start > time_limit:
                return h, best_c5
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
            if improved:
                break
        
        if not improved:
            break
    
    return h, best_c5

def simulated_annealing_binary(h, time_limit=30.0, seed=0):
    """Simulated annealing to escape local optima."""
    n = len(h)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    start = time.time()
    T = 0.01  # initial temperature
    
    while time.time() - start < time_limit:
        # Random pair swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = evaluate_c5(cur_h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
        
        T *= 0.999
    
    return best_h, best_c5
