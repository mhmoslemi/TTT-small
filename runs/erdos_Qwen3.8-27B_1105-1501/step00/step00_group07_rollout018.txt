        hj_val = h[j] - sign * t
        if hi_val < -1e-9 or hi_val > 1.0 + 1e-9 or hj_val < -1e-9 or hj_val > 1.0 + 1e-9:
            ops += 1
            continue
        h_try = h.copy()
        h_try[i] = min(max(hi_val, 0.0), 1.0)
        h_try[j] = min(max(hj_val, 0.0), 1.0)
        target = n / 2.0
        h_try = project_feasible(h_try, target)
        C_try = evaluate_C(h_try)
        if C_try < best_C - 1e-14:
            best_C = C_try
            best_h = h_try.copy()
            h = h_try
        ops += 1
    return best_h, best_C


def optimize_state(h, C, deadline, rng):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_C = C
    fail_count = 0
    alphas = [2, 4, 8, 16, 32, 64, 128, 256, 512]
    while time.time() < deadline and fail_count < 80:
        for alpha in alphas:
            if time.time() >= deadline or fail_count >= 80:
                break
            direction = soft_gradient(h, alpha)
            if direction is None or float(np.max(np.abs(direction))) < 1e-14:
                direction = active_subgradient(h, K=min(5, 2 * n - 1))
            if direction is None or float(np.max(np.abs(direction))) < 1e-14:
                direction = random_direction(n, rng)
            gain_t = 1e-12
            new_h, new_C = line_search(h, C, direction, target, gain_t)
            if new_C < best_C - 1e-14:
                best_h = new_h.copy()
                best_C = new_C
                h = new_h
                C = new_C
                fail_count = 0
            else:
                fail_count += 1
                if fail_count % 10 == 0:
                    pert = 0.1 * random_direction(n, rng)
                    z = h + pert
                    h_try = project_feasible(z, target)
                    C_try = evaluate_C(h_try)
                    if C_try < C:
                        h = h_try
                        C = C_try
                        fail_count = 0
                    elif C_try < best_C:
                        best_h = h_try.copy()
                        best_C = C_try
    return best_h, best_C


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    best_h = None
    best_C = float('inf')

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, UnboundLocalError):
        parent = None

    if parent is not None and len(parent) >= 4:
        n_p = len(parent)
        target_p = n_p / 2.0
        h_p = project_feasible(parent, target_p)
        C_p = evaluate_C(h_p)
        best_h = h_p.copy()
        best_C = C_p

        # Build dimension pool
        dims = set()
        dims.add(n_p)
        for d in range(max(32, n_p - 15), min(150, n_p + 15) + 1, 4):
            dims.add(d)
        for d in [48, 64, 80, 96, 112, 128]:
            if 20 <= d <= 180:
                dims.add(d)
        dims = sorted([d for d in dims if 20 <= d <= 180])

        # Limit pool
        max_pool = min(8, len(dims))
        if len(dims) > max_pool:
            keep = {n_p}
            step = max(1, len(dims) // max_pool)
            for i in range(0, len(dims), step):
                keep.add(dims[i])
                if len(keep) >= max_pool:
                    break
            dims = sorted(keep)

        # Phase 1: Gradient descent on multiple dimensions
        time_per_dim = max(20, (deadline - start_time) / len(dims) * 0.6)
        for dim in dims:
            if time.time() >= deadline:
                break
            if dim == n_p:
                h_dim = h_p.copy()
                C_dim = C_p
            else:
                h_dim = resample(h_p, dim)
                C_dim = evaluate_C(h_dim)
            dim_dl = min(deadline, time.time() + time_per_dim)
            h_opt, C_opt = optimize_state(h_dim, C_dim, dim_dl, rng)
            if C_opt < best_C:
                best_h = h_opt.copy()
                best_C = C_opt
            # Pairwise refinement
            pw_dl = min(deadline, time.time() + max(5, time_per_dim * 0.15))
            h_pw, C_pw = refine_pairwise(best_h, best_C, pw_dl, rng, max_ops=1500)
            if C_pw < best_C:
                best_h = h_pw.copy()
                best_C = C_pw

    else:
        # Fallback: no parent
        n_fb = 96
        h_fb = np.zeros(n_fb)
        h_fb[n_fb // 4:3 * n_fb // 4] = 1.0
        h_fb = project_feasible(h_fb, n_fb / 2.0)
        best_h = h_fb
        best_C = evaluate_C(h_fb)
        if time.time() < deadline:
            h_opt, C_opt = optimize_state(best_h, best_C, deadline - 10, rng)
            if C_opt < best_C:
                best_h = h_opt.copy()
                best_C = C_opt

    # Phase 2: Final fine-tuning with pairwise exchanges
    if time.time() < deadline - 5:
        pw_dl = min(deadline, time.time() + max(10, (deadline - time.time()) * 0.3))
        h_pw, C_pw = refine_pairwise(best_h, best_C, pw_dl, rng, max_ops=5000)
        if C_pw < best_C:
            best_h = h_pw.copy()
            best_C = C_pw

    # Phase 3: One more gradient pass if time permits
    if time.time() < deadline - 5:
        g_dl = min(deadline, time.time() + max(5, (deadline - time.time()) * 0.4))
        h_opt, C_opt = optimize_state(best_h, best_C, g_dl, rng)
        if C_opt < best_C:
            best_h = h_opt.copy()
            best_C = C_opt

    # Final validation
    n_final = len(best_h)
    final_h = project_feasible(best_h, n_final / 2.0)
    final_C = evaluate_C(final_h)

    return final_h, final_C, n_final
