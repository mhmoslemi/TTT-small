def project_feasible(z, n):
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    # Bisection for sum constraint
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    e = target - h.sum()
    if abs(e) > 1e-10:
        if e > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:

            if e > 0:

def active_gradient(h, n, num_active=15):
    """Compute negative weighted gradient of active correlation values."""
    corrs = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    scaled = corrs * dx
    # Get top active constraints
    active_idx = np.argsort(scaled)[-num_active:]
    g = np.zeros(n)
    weights = scaled[active_idx] - scaled[active_idx].min() + 0.1
    for k_idx in active_idx:

        w = weights[list(active_idx).index(k_idx)]
        start1 = max(0, k)
        end1 = min(n, n + k)

        start2 = max(0, -k)
        end2 = min(n, n - k)
        for i in range(start2, end2):

    return g

import numpy as np
import time
import scipy.optimize

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(z, n):
    target = n / 2.0
    z = np.clip(np.asarray(z, dtype=float).ravel(), 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    e = target - h.sum()
    if abs(e) > 1e-10:
        if e > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(e) < 1e-13:
                break
            if e > 0:
                d = min(e, 1.0 - h[i])
            else:
                d = max(e, -h[i])
            h[i] += d
            e -= d
    return h

def active_gradient(h, n, num_active):
    corrs = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    scaled = corrs * dx
    active_idx = np.argsort(scaled)[-num_active:]
    g = np.zeros(n)
    for k_idx in active_idx:
        k = k_idx - (n - 1)
        w = scaled[k_idx]
        s1, e1 = max(0, k), min(n, n + k)
        for i in range(s1, e1):
            g[i] -= w * (1.0 - h[i - k])
        s2, e2 = max(0, -k), min(n, n - k)
        for i in range(s2, e2):
            g[i] -= w * h[i + k]
    return g

def make_candidates(n, seed, parent_h):
    rng = np.random.default_rng(seed)
    cands = []
    if parent_h is not None and len(parent_h) > 1:
        m = len(parent_h)
        if m != n:
            xs = (np.arange(m) + 0.5) * 2.0 / m
            xt = (np.arange(n) + 0.5) * 2.0 / n
            cands.append(project_feasible(np.interp(xt, xs, parent_h), n))
        else:
            cands.append(parent_h.copy())
            cands.append(project_feasible(parent_h[::-1], n))
            cands.append(project_feasible(1.0 - parent_h[::-1], n))
    for ns in [2, 3, 4, 5, 6, 8]:
        seg = n // ns
        base = np.zeros(n)
        for i in range(ns):
            a, b = i * seg, min(n, (i + 1) * seg)
            base[a:b] = (i + 1) / (ns + 1)
        perm = rng.permutation(ns)
        v = np.zeros(n)
        for i in range(ns):
            a, b = i * seg, min(n, (i + 1) * seg)
            v[a:b] = (perm[i] + 1) / (ns + 1)
        cands.append(project_feasible(v, n))
    for _ in range(10):
        t = np.linspace(0, 1, n)
        sig = np.zeros(n)
        for _j in range(3):
            f = rng.uniform(1, 6)
            p = rng.uniform(0, 2 * np.pi)
            a = rng.uniform(0.1, 0.35)
            sig += a * np.sin(2 * np.pi * f * t + p)
        cands.append(project_feasible(0.5 + sig, n))
    for _ in range(5):
        pos = set(rng.choice(n, size=n // 2, replace=False).tolist())
        b = np.zeros(n)
        for i in pos:
            b[i] = 1.0
        cands.append(project_feasible(b, n))
    for _ in range(3):
        cands.append(project_feasible(rng.random(n), n))
    return cands

def slsqp_solve(h0, deadline):
    n = len(h0)
    if time.perf_counter() > deadline:
        return h0, official_score(h0)
    theta0 = official_score(h0) + 0.005
    x0 = np.concatenate([h0, [theta0]])
    dx = 2.0 / n
    n_corr = 2 * n - 1

    def objective(x):
        return x[-1]

    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(x):
        hv = np.clip(x[:n], 0.0, 1.0)
        corrs = np.correlate(hv, 1.0 - hv, mode="full") * dx
        return x[-1] - corrs

    def ineq_jac(x):
        hv = np.clip(x[:n], 0.0, 1.0)
        jac = np.zeros((n_corr, n + 1))
        for r in range(n_corr):
            k = r - (n - 1)
            jac[r, -1] = 1.0
            s1, e1 = max(0, k), min(n, n + k)
            for i in range(s1, e1):
                jac[r, i] -= dx * (1.0 - hv[i - k])
            s2, e2 = max(0, -k), min(n, n - k)
            for i in range(s2, e2):
                jac[r, i] -= dx * hv[i + k]
        return jac

    def eq_fun(x):
        return x[:n].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]
    try:
        result = scipy.optimize.minimize(
            objective, x0, jac=objective_grad,
            bounds=bounds, constraints=cons,
            method='SLSQP',
            options={'maxiter': 80, 'ftol': 1e-12, 'disp': False}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = project_feasible(result.x[:n], n)
            s = official_score(h_sol)
            return h_sol, s
    except Exception:
        pass
    return h0, official_score(h0)

def line_search(h, rng, deadline, max_rounds=4):
    n = len(h)
    best_h = h.copy()
    best_s = official_score(best_h)
    for _ in range(max_rounds):
        if time.perf_counter() > deadline:
            break
        improved = False
        directions = []
        for na in [8, 15, 30]:
            g = active_gradient(best_h, n, na)
            g -= g.mean()
            mx = np.max(np.abs(g))
            if mx > 1e-10:
                directions.append(g / mx)
        for _j in range(6):
            d = rng.normal(size=n)
            d -= d.mean()
            mx = np.max(np.abs(d))
            if mx > 1e-10:
                directions.append(d / mx)
        for _j in range(4):
            i, j = rng.choice(n, 2, replace=False)
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            directions.append(d)
        for d in directions:
            if time.perf_counter() > deadline:
                break
            for sign in [1.0, -1.0]:
                sd = sign * d
                pos = sd > 0
                neg = sd < 0
                max_step = 1.0
                if pos.any():
                    max_step = min(max_step, np.min((1.0 - best_h[pos]) / sd[pos]))
                if neg.any():
                    max_step = min(max_step, np.min(-best_h[neg] / sd[neg]))
                if max_step < 1e-6:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0]:
                    a = frac * max_step
                    nh = project_feasible(best_h + a * sd, n)
                    s = official_score(nh)
                    if s < best_s - 1e-12:
                        best_s = s
                        best_h = nh
                        improved = True
        if not improved:
            break
    return best_h, best_s
