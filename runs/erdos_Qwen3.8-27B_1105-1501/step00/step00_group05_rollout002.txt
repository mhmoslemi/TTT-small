import time
import cvxpy as cp

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient_for_lag(h, lag_idx):
    n = len(h)
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] -= h[s:]
    else:
        sp = -s
        grad[:n-sp] += (1.0 - h[sp:])
        grad[sp:] -= h[:n-sp]
    return grad

def compute_subgradient(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in active_lags:
        grad += subgradient_for_lag(h, m)
    grad *= dx / max(len(active_lags), 1)
    return grad

def repair_sum(h, n):
    """Repair h so that sum(h) = n/2, maintaining box constraints."""
    target = n / 2.0
    current = h.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # Distribute the correction proportionally among those with room
    h = h.copy()
    if diff > 0:
        # Need to add mass; prefer bins with room (h < 1)
        room = 1.0 - h
        total_room = room.sum()
        if total_room < diff:
            h[:] = 1.0
            h[:int(n//2)] = 1.0  # fallback
            return h
        h += (diff / total_room) * room
    else:
        # Need to remove mass; prefer bins with excess (h > 0)
        excess = h.copy()
        total_excess = excess.sum()
        if total_excess < -diff:
            h[:] = 0.0
            return h
        h += (diff / total_excess) * excess
    h = np.clip(h, 0, 1)
    # Fine adjustment
    still_diff = target - h.sum()
    if abs(still_diff) > 1e-10:
        # Find a bin to adjust
        if still_diff > 0:
            room = 1.0 - h
            idx = np.argmax(room)
            h[idx] += min(still_diff, room[idx])
        else:
            idx = np.argmax(h)
            h[idx] += max(still_diff, -h[idx])
    return h

def resample(h_old, n_new):
    """Resample h from n_old points to n_new points, preserving integral."""
    n_old = len(h_old)
    # Use cumulative integral approach
    # Old cumulative integral at points linspace(0, 2, n_old+1) with step 2/n_old
    dx_old = 2.0 / n_old
    cum_old = np.zeros(n_old + 1)
    cum_old[1:] = np.cumsum(h_old) * dx_old
    # For new grid, compute integral over each new bin
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for i in range(n_new):
        x_left = i * dx_new
        x_right = (i + 1) * dx_new
        # Find where these fall in the old grid
        # Linear interpolation of the step function integral
        # The old function is constant on [j*dx_old, (j+1)*dx_old)
        integral = 0.0
        for j in range(n_old):
            old_left = j * dx_old
            old_right = (j + 1) * dx_old
            overlap_left = max(x_left, old_left)
            overlap_right = min(x_right, old_right)
            if overlap_right > overlap_left:
                integral += h_old[j] * (overlap_right - overlap_left)
        h_new[i] = integral / dx_new
    h_new = np.clip(h_new, 0, 1)
    return repair_sum(h_new, n_new)

def lp_direction(h, active_lags, n):
    """Find direction d that decreases all active constraints."""
    dx = 2.0 / n
    # Build gradient matrix for active lags
    G = np.zeros((len(active_lags), n))
    for k, m in enumerate(active_lags):
        G[k, :] = subgradient_for_lag(h, m)
    
    # LP: maximize t
    # s.t. G @ d / dx <= -t (each active constraint decreases by at least t*dx)
    #      sum(d) = 0
    #      -h[i]/alpha <= d[i] <= (1-h[i])/alpha for box feasibility
    #      t >= 0
    # We parameterize with step size alpha implicitly by bounding d
    
    # Simpler: minimize max_k (G[k] @ d)
    # s.t. sum(d) = 0, |d[i]| <= step_bound
    
    step_bound = 0.05  # max change per coordinate
    
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -h * step_bound,
        d_var <= (1 - h) * step_bound,
    ]
    for k in range(len(active_lags)):
        constraints.append(G[k] @ d_var <= -t_var * dx)
    
    objective = cp.Maximize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000, abstag=1e-8, reltol=1e-8)
        if d_var.value is not None and t_var.value is not None and t_var.value > 1e-10:
            return d_var.value, t_var.value
    except:
        pass
    return None, 0

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient_for_lag(h, lag_idx):
    n = len(h)
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    else:
        sp = -s
        if n - sp > 0:
            grad[:n - sp] += (1.0 - h[sp:])
            grad[sp:] -= h[:n - sp]
    return grad

def compute_active(h, rel_tau=0.005, abs_tau=1e-6):
    corr = get_corr_array(h)
    max_val = np.max(corr)
    threshold = max_val - max(rel_tau * max_val, abs_tau)

def repair_sum(h, n):

            remaining -= add
