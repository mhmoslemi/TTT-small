            no_improve += 1

        # Periodic reheat: add small random perturbation if stuck
        if no_improve > improve_threshold * 4:
            no_improve = 0
            for _ in range(5):
                ri = rng.randint(n)
                rj = rng.randint(n)
                if ri != rj:
                    ra = rng.random() * 0.05 * min(current_h[ri], 1.0 - current_h[rj])
                    if ra > 1e-7:
                        current_h[ri] -= ra
                        current_h[rj] += ra
                    ra = rng.random() * 0.05 * min(current_h[rj], 1.0 - current_h[ri])
                    if ra > 1e-7:
                        current_h[rj] -= ra
                        current_h[ri] += ra

    return best_h, best_c5

def resample_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    target = n_new / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    best_h_global = None
    best_c5_global = float('inf')
    best_n_global = 0

    def try_solution(h_cand, n_cand):
        nonlocal best_h_global, best_c5_global, best_n_global
        c = eval_c5(h_cand)
        if c < best_c5_global:
            best_c5_global = c
            best_h_global = h_cand
            best_n_global = n_cand

    # Phase 1: Optimize initial_h_values (n=70) - main effort
    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    target0 = n0 / 2.0
    h0_opt = project_feasible(h0, target0)

    rng0 = np.random.RandomState(seed)
    t_deadline1 = t_start + budget_s * 0.45
    h_best0, c5_best0 = sa_optimize(h0_opt, n0, target0, rng0, t_deadline1)
    try_solution(h_best0, n0)

    # Phase 2: Try n=100 from interpolated starting point
    elapsed = time.time() - t_start
    if budget_s - elapsed > 180:
        rng1 = np.random.RandomState(seed + 1)
        h1_init = resample_h(h_best0, 100)
        t_deadline2 = t_start + budget_s * 0.65
        h_best1, c5_best1 = sa_optimize(h1_init, 100, 50.0, rng1, t_deadline2)
        try_solution(h_best1, 100)

    # Phase 3: Try n=60
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        rng2 = np.random.RandomState(seed + 2)
        h2_init = resample_h(h_best0, 60)
        t_deadline3 = t_start + budget_s * 0.80
        h_best2, c5_best2 = sa_optimize(h2_init, 60, 30.0, rng2, t_deadline3)
        try_solution(h_best2, 60)

    # Phase 4: Final polishing on the best solution
    elapsed = time.time() - t_start
    polish_time = budget_s - elapsed - 30
    if polish_time > 20 and best_h_global is not None:
        rng_polish = np.random.RandomState(seed + 100)
        h_final, c5_final = polish(best_h_global, rng_polish, polish_time)
        if c5_final < best_c5_global:
            best_c5_global = c5_final
            best_h_global = h_final

    # Ensure feasibility of final answer
    if best_h_global is not None:
        best_h_global = np.clip(best_h_global, 0.0, 1.0)
        final_sum = best_n_global / 2.0
        residual = final_sum - np.sum(best_h_global)
        if abs(residual) > 1e-6:
            best_h_global = project_feasible(best_h_global, final_sum)
            best_c5_global = eval_c5(best_h_global)

    return best_h_global, best_c5_global, best_n_global
