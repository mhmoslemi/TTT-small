import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_values_and_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    n_out = 2*n - 1
    values = np.zeros(n_out)
    J = np.zeros((n_out, n))
    
    for k in range(-(n-1), n):
        m = n - 1 + k
        if k >= 0:
            values[m] = dx * np.dot(h[k:], b[:n-k])
            # grad: b[j-k] for k<=j<n, -h[j+k] for 0<=j<n-k
            J[m, k:] += dx * b[:n-k]
            J[m, :n-k] -= dx * h[k:]
        else:
            p = -k
            values[m] = dx * np.dot(h[:n-p], b[p:])
            # grad: b[j+p] for 0<=j<n-p, -h[j-p] for p<=j<n
            J[m, :n-p] += dx * b[p:]
            J[m, p:] -= dx * h[:n-p]

                h[idx] -= move

            if abs(delta) < 1e-14:
                break
    return h

def _smooth_pen_fn(x):
    return _sm_params['obj'](x)

def _smooth_pen_gr(x):
    return _sm_params['grad'](x)

_sm_params = {}

def _make_smooth_funcs(tau, mu, target):
    def _obj(x):
        vals = compute_values(x)
        M = np.max(vals)
        exp_v = np.exp((vals - M) / tau)
        s = np.sum(exp_v)
        return M + tau * np.log(s) + mu * (x.sum() - target) ** 2
    def _grad(x):
        vals = compute_values(x)
        M = np.max(vals)
        exp_v = np.exp((vals - M) / tau)
        w = exp_v / np.sum(exp_v)
        J = compute_jacobian(x)
        return J.T @ w + 2.0 * mu * (x.sum() - target) * np.ones(len(x))
    return _obj, _grad

def run_smooth(h0, tau_seq, maxiter, mu=5000.0):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    current_val = exact_c5(h)
    for tau in tau_seq:
        obj_fn, grad_fn = _make_smooth_funcs(tau, mu, target)
        _sm_params['obj'] = obj_fn
        _sm_params['grad'] = grad_fn
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_smooth_pen_fn, h, jac=_smooth_pen_gr,
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                val_new = exact_c5(h_new)
                if val_new <= current_val + 1e-9:
                    h = h_new
                    current_val = val_new
        except Exception:
            pass
    return h

def pair_move_search(h, max_moves, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_val = exact_c5(h)
    deltas = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    moves_done = 0
    for delta in deltas:
        improved = True
        while improved and moves_done < max_moves:
            improved = False
            for _ in range(min(300, max_moves - moves_done)):
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                for direction in (1, -1):
                    di = direction * delta
                    dj = -direction * delta
                    if h[i] + di < -1e-10 or h[i] + di > 1.0 + 1e-10:
                        continue
                    if h[j] + dj < -1e-10 or h[j] + dj > 1.0 + 1e-10:
                        continue
                    h[i] += di
                    h[j] += dj
                    val = exact_c5(h)
                    if val < best_val - 1e-12:
                        best_val = val
                        improved = True
                    else:
                        h[i] -= di
                        h[j] -= dj
                    moves_done += 1
                    if moves_done >= max_moves:
                        break
    return h, best_val

def gradient_guided_mves(h, max_moves, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_val = exact_c5(h)
    moves_done = 0
    deltas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for delta in deltas:
        improved = True
        while improved and moves_done < max_moves:
            improved = False
            for _ in range(min(200, max_moves - moves_done)):
                vals = compute_values(h)
                active = np.where(vals >= np.max(vals) - 1e-6)[0]
                J = compute_jacobian(h)
                grad = J[active].mean(axis=0)
                order = np.argsort(grad)
                i = int(order[0])
                j = int(order[-1])
                if i == j:
                    i = int(rng.randint(n))
                    if i == j:
                        i = (j + 1) % n
                di = delta if grad[i] < grad[j] else -delta
                dj = -di
                if h[i] + di < -1e-10 or h[i] + di > 1.0 + 1e-10:
                    continue
                if h[j] + dj < -1e-10 or h[j] + dj > 1.0 + 1e-10:
                    continue
                h[i] += di
                h[j] += dj
                val = exact_c5(h)
                if val < best_val - 1e-12:
                    best_val = val
                    improved = True
                else:
                    h[i] -= di
                    h[j] -= dj
                moves_done += 1
                if moves_done >= max_moves:
                    break
    return h, best_val

def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    cum_src = np.concatenate([[0.0], np.cumsum(h_src) * dx_src])
    x_src = np.linspace(0, 2.0, n_src + 1)
    x_tgt = np.linspace(0, 2.0, n_tgt + 1)
    cum_tgt = np.interp(x_tgt, x_src, cum_src)
    h_tgt = np.diff(cum_tgt) / dx_tgt
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    return project_feasible(h_tgt, n_tgt)

def generate_starts(parent, n, seed):
    rng = np.random.RandomState(seed)
    starts = [parent.copy()]
    comp = 1.0 - parent
    if abs(comp.sum() - n / 2.0) < 1e-8:
        starts.append(comp.copy())
    starts.append(parent[::-1].copy())
    for sig in (0.25, 0.12, 0.06):
        noisy = parent + sig * rng.randn(n)
        starts.append(project_feasible(noisy, n))
    x = np.linspace(0, 2, n)
    for freq in (1, 2, 3, 5):
        for phase in (0.0, np.pi / 3, 2 * np.pi / 3):
            sig_h = 0.5 + 0.45 * np.sin(np.pi * freq * x + phase)
            starts.append(project_feasible(sig_h, n))
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for trial in range(3):
        vals = np.array([(i * (phi + trial * 0.07)) % 1.0 for i in range(n)])
        order = np.argsort(vals)
        binary = np.zeros(n)
        target = n / 2.0
        cnt = int(target)
        if cnt > 0:
            binary[order[-cnt:]] = 1.0
        frac = target - cnt
        if frac > 1e-12 and cnt > 0:
            binary[order[cnt - 1]] = frac
        elif frac > 1e-12:
            binary[order[-1]] = min(1.0, frac)
        starts.append(project_feasible(binary, n))
    unique_starts = [starts[0]]
    for s in starts[1:]:
        is_new = True
        for u in unique_starts:
            if np.max(np.abs(s - u)) < 0.03:
                is_new = False
                break
        if is_new:
            unique_starts.append(s)
    return unique_starts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s * 0.92
    rng = np.random.RandomState(seed)

    best_h = None
    best_val = np.inf
    best_n = 0

    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).copy()
        if pv.size > 2 and np.all(np.isfinite(pv)):
            parent = project_feasible(pv, pv.size)
    except Exception:
        parent = None

    if parent is None:
        parent = project_feasible(np.full(45, 0.5), 45)

    best_h = parent.copy()
    best_val = exact_c5(best_h)
    best_n = parent.size

    def elapsed():
        return time.time() - t_start

    def remaining():
        return max(0.0, deadline - time.time())

    n0 = best_n
    starts = generate_starts(best_h, n0, seed)

    tau_seqs = [
        [0.05, 0.02, 0.008, 0.003],
        [0.04, 0.015, 0.005],
        [0.06, 0.025, 0.01, 0.004],
    ]

    for si, h_start in enumerate(starts):
        if remaining() < 25:
            break
        n_left = len(starts) - si
        time_alloc = remaining() / (n_left + 0.5)
        maxiter = max(30, min(120, int(time_alloc * 15)))
        tau_seq = tau_seqs[si % len(tau_seqs)]
        h_opt = run_smooth(h_start, tau_seq, maxiter=maxiter, mu=5000.0)
        val = exact_c5(h_opt)
        if val < best_val - 1e-10:
            best_h = h_opt.copy()
            best_val = val
            best_n = n0

    if remaining() > 15:
        h_g, val_g = gradient_guided_mves(best_h, 3000, seed + 1)
        if val_g < best_val - 1e-10:
            best_h = h_g.copy()
            best_val = val_g

    if remaining() > 15:
        h_p, val_p = pair_move_search(best_h, 8000, seed + 2)
        if val_p < best_val - 1e-10:
            best_h = h_p.copy()
            best_val = val_p

    if remaining() > 50:
        for n_try in (32, 36, 40, 44, 48, 52, 56, 60, 64):
            if remaining() < 25:
                break
            if n_try == best_n:
                continue
            h_trans = transfer_resolution(best_h, n_try)
            val_t = exact_c5(h_trans)
            h_opt = run_smooth(h_trans, [0.03, 0.01, 0.004], maxiter=60, mu=5000.0)
            val_t = exact_c5(h_opt)
            h_g2, val_g2 = gradient_guided_mves(h_opt, 2000, seed + 10 + n_try)
            if val_g2 < val_t:
                val_t = val_g2
                h_opt = h_g2
            h_p2, val_p2 = pair_move_search(h_opt, 3000, seed + 20 + n_try)
            if val_p2 < val_t:
                val_t = val_p2
                h_opt = h_p2
            if val_t < best_val - 1e-10:
                best_h = h_opt.copy()
                best_val = val_t
                best_n = n_try

    if remaining() > 20:
        h_g3, val_g3 = gradient_guided_mves(best_h, 10000, seed + 100)
        if val_g3 < best_val - 1e-10:
            best_h = h_g3.copy()
            best_val = val_g3

    if remaining() > 10:
        h_p3, val_p3 = pair_move_search(best_h, 15000, seed + 200)
        if val_p3 < best_val - 1e-10:
            best_h = h_p3.copy()
            best_val = val_p3

    best_h = project_feasible(best_h, len(best_h))
    best_val = exact_c5(best_h)
    best_n = len(best_h)

    return best_h, best_val, best_n
