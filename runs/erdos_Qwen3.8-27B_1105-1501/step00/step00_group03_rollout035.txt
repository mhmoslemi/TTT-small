def build_h(u, n):
    """Build full h from symmetric parameterization u of length m = n//2."""
    h = np.zeros(n)
    m = len(u)
    h[:m] = u
    h[n-m:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def sym_to_u(h):
    """Extract u from symmetric h."""
    n = len(h)
    m = n // 2
    u = 0.5 * (h[:m] + 1.0 - h[n-1-m:n-1:-1])
    return np.clip(u, 0.0, 1.0)

def get_c5(h):
    """Compute C5 for h."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def reduced_grad(u, n, k):
    """Reduced gradient of F_k w.r.t. u."""
    h = build_h(u, n)
    m = len(u)
    grad = np.zeros(m)
    start = max(0, -k)
    end = min(n, n - k)
    for i in range(start, end):
        j1, j2 = i, i + k
        for j, dh in ((j1, 1.0 - h[j2]), (j2, -h[j1])):
            if j < m:
                grad[j] += dh
            elif j != m or n % 2 == 0:
                idx = n - 1 - j
                if idx < m:
                    grad[idx] -= dh
    return grad

def interp_to_n(h_src, n_new):
    """Interpolate h from its original domain to n_new points."""
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_src, h_src)

def gradient_refine(u, n, n_rounds=10):
    """Gradient-based refinement with exact line search."""
    m = len(u)
    dx = 2.0 / n
    u = u.copy()
    
    for round_num in range(n_rounds):
        h = build_h(u, n)
        cv = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(cv)
        
        # Active set
        active_mask = cv >= c_max * 0.999
        active_indices = np.where(active_mask)[0]
        
        # Compute direction
        G = np.zeros((len(active_indices), m))
        for r, idx in enumerate(active_indices):
            k = int(idx) - (n - 1)
            G[r] = reduced_grad(u, n, k)
        
        # Direction: negative sum of active gradients
        d = -np.sum(G, axis=0)
        d_norm = np.max(np.abs(d))
        if d_norm < 1e-15:
            # Try least squares
            try:
                d = np.linalg.lstsq(G, -np.ones(len(active_indices)), rcond=None)[0]
            except:
                break
            d_norm = np.max(np.abs(d))
            if d_norm < 1e-15:
                break
            d = d / d_norm
        
        # Line search
        best_alpha = 0.0
        best_c5 = c_max
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 0.1, -0.01, -0.02, -0.05]:
            u_new = np.clip(u + alpha * d, 0.0, 1.0)
            h_new = build_h(u_new, n)
            c5_new = get_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_alpha = alpha
        
        if best_alpha == 0.0:
            # Try smaller steps
            for alpha in [0.001, 0.002, 0.005, 0.0005]:
                u_new = np.clip(u + alpha * d, 0.0, 1.0)
                h_new = build_h(u_new, n)
                c5_new = get_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        u = np.clip(u + best_alpha * d, 0.0, 1.0)
    
    return u

from scipy.optimize import minimize
import time

def build_h(u, n):
    m = len(u)
    h = np.empty(n)
    h[:m] = u
    h[n-m:] = 1.0 - u[::-1]
    if n % 2 == 1:
        h[m] = 0.5
    return h

def sym_to_u(h):
    n = len(h)
    m = n // 2
    u = 0.5 * (h[:m] + 1.0 - h[n-1-m:n-1:-1])
    return np.clip(u, 0.0, 1.0)

def get_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def reduced_grad_vec(u, n, k):
    h = build_h(u, n)
    m = len(u)
    start = max(0, -k)
    end = min(n, n - k)
    if start >= end:
        return np.zeros(m)
    full_grad = np.zeros(n)
    i_vals = np.arange(start, end)
    full_grad[i_vals] += (1.0 - h[i_vals + k])
    full_grad[i_vals + k] += (-h[i_vals])
    grad = full_grad[:m] - full_grad[n-1-m:n-1:-1]
    return grad

def interp_to_n(h_src, n_new):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_src, h_src)

class MMProb:
    def __init__(self, n, k_list):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.k_list = k_list
    
    def obj(self, z):
        return z[self.m]
    
    def obj_jac(self, z):
        g = np.zeros(self.m + 1)
        g[self.m] = 1.0
        return g
    
    def get_cons(self):
        cons = []
        for k in self.k_list:
            cf = MMConstraint(self.n, self.m, self.dx, k)
            cons.append({'type': 'ineq', 'fun': cf.fun, 'jac': cf.jac})
        return cons

class MMConstraint:
    def __init__(self, n, m, dx, k):
        self.n = n
        self.m = m
        self.dx = dx
        self.k = k
    
    def fun(self, z):
        u = z[:self.m]
        h = build_h(u, self.n)
        start = max(0, -self.k)
        end = min(self.n, self.n - self.k)
        F = np.dot(h[start:end], 1.0 - h[start + self.k:end + self.k])
        return z[self.m] - self.dx * F
    
    def jac(self, z):
        u = z[:self.m]
        g = np.zeros(self.m + 1)
        g[:self.m] = -self.dx * reduced_grad_vec(u, self.n, self.k)
        g[self.m] = 1.0
        return g

def slsqp_solve(u0, n, max_cons=25, max_iter=200):
    m = n // 2
    dx = 2.0 / n
    u = u0.copy()
    h = build_h(u, n)
    cv = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(cv)
    
    # Active set
    thresh = c_max - 0.05 * c_max  # within 5% of max
    active = np.where(cv >= thresh)[0]
    if len(active) > max_cons:
        order = np.argsort(cv[active])[::-1]
        active = active[order[:max_cons]]
    
    k_list = [int(i) - (n - 1) for i in active]
    
    prob = MMProb(n, k_list)
    z0 = np.empty(m + 1)
    z0[:m] = u
    z0[m] = c_max * dx + 1e-10
    
    bounds = [(0.0, 1.0)] * m + [(0.0, 1.0)]
    
    res = minimize(prob.obj, z0, method='SLSQP', jac=prob.obj_jac,
                   bounds=bounds, constraints=prob.get_cons(),
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    
    if res.x is not None and len(res.x) == m + 1:
        u_new = np.clip(res.x[:m], 0.0, 1.0)
        return u_new, get_c5(build_h(u_new, n))
    return u, c_max * dx

def gradient_refine(u, n, n_rounds=15):
    m = len(u)
    dx = 2.0 / n
    u = u.copy()
    
    for _ in range(n_rounds):
        h = build_h(u, n)
        cv = np.correlate(h, 1.0 - h, mode='full')
        c_val = np.max(cv)
        
        # Active set (within 0.1% of max)
        active_mask = cv >= c_val * 0.999
        active_idx = np.where(active_mask)[0]
        
        # Compute reduced gradients for active constraints
        G = np.zeros((len(active_idx), m))
        for r, idx in enumerate(active_idx):
            k = int(idx) - (n - 1)
            G[r] = reduced_grad_vec(u, n, k)
        
        # Direction: LS solution
        # Minimize ||G @ d + c_active||^2 where we want to decrease all
        # Actually: find d that minimizes max_k (c_k + dx * G_k @ d)
        # Linearized: minimize t s.t. c_k + dx*G_k@d <= t
        # Direction of steepest descent for the max: negative of the weighted sum
        d = -np.sum(G, axis=0)
        
        # Also try LS direction
        try:

        except np.linalg.LinAlgError:
            d_ls = d
        
        # Try both directions with line search
        best_c5 = c_val
        best_alpha = 0.0
        best_dir = None

            d_norm = np.max(np.abs(d_try))
            if d_norm < 1e-14:
                continue
            d_try = d_try / d_norm  # unit in max norm
            
            for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, -0.01, -0.05]:
                u_new = np.clip(u + alpha * d_try, 0.0, 1.0)
                c5_new = get_c5(build_h(u_new, n))
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_alpha = alpha
                    best_dir = d_try
        
        if best_alpha == 0.0:

            for dir_name, d_try in [('sum', d), ('ls', d_ls)]:

                d_try = d_try / d_norm

            if best_alpha == 0.0:

Let me also add one more optimization technique: after SLSQP and gradient refinement, I can try a "coordinate descent" approach where I optimize one variable at a time. This is sometimes effective for problems with many active constraints.

Actually, for the coordinate descent, since we're in the symmetric subspace, optimizing one variable u[p] means:
- Fix all other u[p'] and optimize u[p] to minimize C5
- C5 = max_k dx * F_k, and F_k is bilinear in h (and hence in u)
- For a fixed u[p], F_k is linear in u[p] (bilinear overall, linear in one variable)
- So the minimax in one variable is: minimize max_k (a_k * u[p] + b_k) subject to u[p] ∈ [0,1]
- This is a 1D minimax, solvable by checking where the lines intersect

This is actually quite efficient! Let me add this as a refinement step.

For a fixed u with u[p] variable:
F_k = sum_i h[i](1-h[i+k])

The terms involving h[p] (i.e., u[p]) are:
- If p is in the "first" factor (i = p): contribution is h[p] * (1-h[p+k]) = u[p] * (1-h[p+k])
- If p is in the "second" factor (i+k = p, i.e., i = p-k): contribution is h[p-k] * (1-h[p]) = h[p-k] * (1-u[p])
- Similarly for the pair (n-1-p):
  - If n-1-p is in the "first" factor: h[n-1-p] * (1-h[n-1-p+k]) = (1-u[p]) * (1-h[n-1-p+k])
  - If n-1-p is in the "second" factor: h[n-1-p-k] * (1-h[n-1-p]) = h[n-1-p-k] * u[p]

So F_k = a_k * u[p] + b_k for some coefficients a_k, b_k that depend on the other variables.

The minimax is: minimize max_k (a_k * t + b_k) over t ∈ [0,1].

The maximum of a set of lines is a convex piecewise-linear function. Its minimum occurs either at a breakpoint (where two lines cross) or at a boundary (t=0 or t=1).

To find the optimal t:
1. Compute all breakpoints t_ij where a_i*t + b_i = a_j*t + b_j, i.e., t = (b_j - b_i)/(a_i - a_j)
2. Keep only breakpoints in [0,1]
3. Evaluate the max at each breakpoint and at t=0, t=1
4. Return the t that gives the smallest max

This is O(K^2) where K is the number of lags (2n-1). For n=50, K=99, so K^2 ≈ 10000. Very fast.

But actually, I only need to consider the active constraints (those near the max). The number of active constraints is small (typically 5-20). So this is very fast.
