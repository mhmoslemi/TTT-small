                c = np.dot(h[p:], 1.0 - h[:n-p])

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        tau = 0.5 * (lo + hi)
        h = np.clip(y - tau, 0.0, 1.0)
        if h.sum() > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = free.sum()
        if cnt > 0:
            h[free] += residual / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def softmax_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    f = dx * (max_c + log_sum / beta)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k]
        lag = (n - 1) - k
        if lag >= 0:
            grad[:n-lag] += wk * (1.0 - h[lag:])
            grad[lag:] += wk * (-h[:n-lag])
        else:
            p = -lag
            grad[p:] += wk * (1.0 - h[:n-p])
            grad[:n-p] += wk * (-h[p:])
    return f, dx * grad

def run_gradient_descent(h0, n, max_time, beta_start=20.0, beta_end=200.0, lr0=0.02):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    t = 0
    t0 = time.time()
    no_improve = 0
    
    while time.time() - t0 < max_time:
        t += 1
        frac = min(1.0, t / 1500.0)
        beta = beta_start + (beta_end - beta_start) * frac
        _, grad = softmax_obj_and_grad(h, beta)
        gp = grad - grad.mean()
        m = b1*m + (1-b1)*gp
        v = b2*v + (1-b2)*gp*gp
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        step = lr * mh / (np.sqrt(vh) + eps)
        h_new = project_feasible(h - step, n)
        c5_new = c5_value(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.95
                no_improve = 0
                if lr < 1e-5:
                    break
        h = h_new
    return best_h, best_c5

def slsqp_refine(h0, n, maxiter=300, top_k=15):

    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g

        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr[top_indices]
    def ineq_jac(x):
        h = x[:n]
        J = np.zeros((top_k, n+1))
        for ii, k in enumerate(top_indices):
            lag = (n-1) - k
            if lag >= 0:
                J[ii, :n-lag] = 1.0 - h[lag:]
                J[ii, lag:n] += -h[:n-lag]
            else:
                p = -lag
                J[ii, p:n] = 1.0 - h[:n-p]
                J[ii, :n-p] += -h[p:]
            J[ii, -1] = 1.0
        return J

        return x[:n].sum() - n/2.0

        g = np.zeros(n+1)
        g[:n] = 1.0

        pass
    return h0, c5_0
