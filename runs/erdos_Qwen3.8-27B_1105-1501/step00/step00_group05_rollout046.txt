            g[i] -= h[i-m]

import numpy as np
from scipy.optimize import linprog
import time

def exact_obj(h):
    """Compute the exact C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_vector(h):
    """Compute the full correlation vector (not scaled)."""
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, lag, n):
    """Compute gradient of the lag-th correlation value w.r.t. h."""
    g = np.zeros(n)
    if lag >= 0:
        # c_lag = sum_{i=0}^{N-1-lag} h[i+lag] * (1-h[i])
        # d/dh[j] = (1-h[j-lag]) if lag <= j <= N-1  -  h[j+lag] if 0 <= j <= N-1-lag
        for j in range(lag, n):
            g[j] += (1.0 - h[j - lag])
        for j in range(0, n - lag):
            g[j] -= h[j + lag]
    else:
        m = -lag
        # c_{-m} = sum_{i=m}^{N-1} h[i-m] * (1-h[i])
        # d/dh[j] = (1-h[j+m]) if 0 <= j <= N-1-m  -  h[j-m] if m <= j <= N-1
        for j in range(0, n - m):
            g[j] += (1.0 - h[j + m])
        for j in range(m, n):
            g[j] -= h[j - m]
    return g

def get_active_indices(h, tol=1e-9, max_active=30):
    """Get indices of correlation values near the maximum."""
    cv = corr_vector(h)
    n = len(h)
    c_max = np.max(cv)
    # Active: within tol of max
    active = np.where(cv >= c_max - tol * c_max)[0]
    if len(active) > max_active:
        # Take the top max_active
        idx = np.argsort(cv[active])[::-1][:max_active]
        active = active[idx]
    return active, c_max

def lp_direction(h, active_indices, n):
    """Find LP direction to decrease all active constraints."""
    # Build gradient matrix: A[ip, j] = grad for active lag ip, variable j
    A = np.zeros((len(active_indices), n))
    for ip, m in enumerate(active_indices):
        lag = m - (n - 1)
        A[ip] = grad_for_lag(h, lag, n)
    
    # LP: minimize -rho
    # s.t. A @ d <= -rho  (for each active constraint)
    #      sum(d) = 0
    #      -1 <= d_i <= 1
    #
    # Variables: [d_0, ..., d_{n-1}, rho]
    # Objective: minimize -rho => c = [0, ..., 0, -1]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: A @ d + rho <= 0  =>  [A, -1] @ [d; rho] <= 0
    A_ub = np.hstack([A, -np.ones((len(active_indices), 1))])
    b_ub = np.zeros(len(active_indices))
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -1 <= d_i <= 1, rho unbounded below (but we want it positive)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if result.success and result.x[-1] > 1e-10:

    return None, 0.0

    """Resample piecewise constant function from len(h_old) to N_new."""
    N_old = len(h_old)
    # For each new point j, compute the average of h_old over [2j/N_new, 2(j+1)/N_new)
    h_new = np.zeros(N_new)
    for j in range(N_new):
        # New point j covers [2j/N_new, 2(j+1)/N_new)
        start = 2.0 * j / N_new
        end = 2.0 * (j + 1) / N_new
        # Find overlapping old points
        # Old point i covers [2i/N_old, 2(i+1)/N_old)
        total_weight = 0.0
        total_value = 0.0
        for i in range(N_old):
            old_start = 2.0 * i / N_old
            old_end = 2.0 * (i + 1) / N_old
            # Overlap of [start, end) and [old_start, old_end)
            ov_start = max(start, old_start)
            ov_end = min(end, old_end)
            if ov_end > ov_start:
                w = ov_end - ov_start
                total_weight += w
                total_value += w * h_old[i]
        if total_weight > 0:
            h_new[j] = total_value / total_weight
        else:
            h_new[j] = 0.5
    return h_new

                    if di_i == ri_i or h[di_i] <= 1e-10 or h[ri_i] >= 1 - 1e-10:
                        continue
                    eps = min(h[di_i], 1 - h[ri_i]) * 0.1
                    h_trial = h.copy()
                    h_trial[di_i] -= eps
                    h_trial[ri_i] += eps
                    c_trial = exact_obj(h_trial)
                    if c_trial < best_c - 1e-12:
                        h = h_trial
                        best_h = h.copy()
                        best_c = c_trial
                        improvement = True
                        break
                    tried += 1
                    if tried > 100:
                        break
                if improvement:
                    break
            
            if not improvement:
                stagnant += 1
                if stagnant > 50:
                    break
            continue
        
        # Line search along d
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.001]:
            h_trial = h + alpha * d
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = fix_sum(h_trial, n/2.0)
            c_trial = exact_obj(h_trial)
            if c_trial < best_c - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_c = c_trial
                improved = True
                break
        
        if not improved:
            stagnant += 1
            if stagnant > 100:
                break
    
    return best_h, best_c
