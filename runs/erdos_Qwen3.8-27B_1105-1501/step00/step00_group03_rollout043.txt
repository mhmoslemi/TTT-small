def local_optimize(h0, n, max_iter=5000, tol=1e-10):
    h = h0.copy()
    dx = 2.0 / n
    c5 = compute_C5(h)
    
    for it in range(max_iter):
        # Compute correlation values
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        
        # Active set: lags where corr >= max_val - tol_corr
        tol_corr = max(0.01 * max_val, 1e-6)
        active = np.where(corr >= max_val - tol_corr)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient of sum of active corr values
        g = np.zeros(n)
        for r in active:
            k = r - (n-1)
            if k >= 0:
                i_start, i_end = 0, n - k
            else:
                i_start, i_end = -k, n
            for i in range(i_start, i_end):
                g[i] += (1.0 - h[i+k])
                if k != 0:
                    g[i+k] += (-h[i])
                else:
                    g[i] += (-h[i])  # k=0 case, double count at same index
        
        # Project to sum-zero subspace
        g_zero = g - np.mean(g)
        
        # Step direction: negative gradient
        d = -g_zero
        
        # Line search
        best_alpha = None
        alphas = [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]
        for alpha in alphas:
            h_trial = project_to_feasible(h + alpha * d, n)
            c5_trial = compute_C5(h_trial)
            if c5_trial < c5 - 1e-12:
                best_alpha = alpha
                break
        
        if best_alpha is None:
            # Try even smaller steps
            for alpha in [1e-4, 5e-5, 1e-5]:
                h_trial = project_to_feasible(h + alpha * d, n)
                c5_trial = compute_C5(h_trial)

        if best_alpha is None:
            break  # Stuck
        
        h = project_to_feasible(h + best_alpha * d, n)

    return h

    lo = p.min() - 2.0
    hi = p.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(p - tau, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def active_gradient(h, n, tol_frac=0.05):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = float(np.max(corr))
    tol = max(tol_frac * max_val, 1e-8)
    active = np.where(corr >= max_val - tol)[0]
    g = np.zeros(n)
    for r in active:
        k = int(r) - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] += (-h[k:])
        else:
            kp = -k
            g[:n-kp] += (1.0 - h[kp:])
            g[kp:] += (-h[:n-kp])
    return g, active, corr, max_val

def gradient_descent(h0, n, max_iter=5000, time_limit=100.0):
    start = time.time()
    h = h0.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    no_improve = 0
    tol_frac = 0.05
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g, active, corr, max_val = active_gradient(h, n, tol_frac)
        g_zero = g - np.mean(g)
        d = -g_zero
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            break
        d = d / d_norm
        
        best_alpha = -1.0
        best_c5_trial = c5
        # Adaptive step sizes based on problem scale
        base_step = 1.0
        for mult in [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
            alpha = base_step * mult
            h_trial = project_feasible(h + alpha * d, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5_trial - 1e-14:
                best_c5_trial = c5_trial
                best_alpha = alpha
        
        if best_alpha < 0:
            no_improve += 1
            if no_improve > 200:
                break
            if no_improve > 30 and np.random.random() < 0.4:
                scale = 0.01 + 0.03 * min(no_improve / 100.0, 1.0)
                pert = np.random.randn(n) * scale
                h_trial = project_feasible(h + pert, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5:
                    h = h_trial
                    c5 = c5_trial
                    no_improve = 0
            continue
        
        h = project_feasible(h + best_alpha * d, n)
        c5 = compute_c5(h)
        
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
            # Tighten tolerance as we converge
            if best_c5 < 0.381:
                tol_frac = 0.02
        else:
            no_improve += 1
    
    return best_h, best_c5

def simulated_annealing(h0, n, max_iter=20000, time_limit=50.0):
    start = time.time()
    h = h0.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    T = 0.005
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        n_pert = 1 + (np.random.random() < 0.3)
        idx = np.random.choice(n, size=n_pert, replace=False)
        h_trial = h.copy()
        h_trial[idx] += np.random.randn(n_pert) * 0.05
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        
        delta = c5_trial - c5
        if delta < 0 or (T > 1e-8 and np.random.random() < np.exp(-min(delta, 10.0) / T)):
            h = h_trial
            c5 = c5_trial
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        T *= 0.9998
    
    return best_h, best_c5

def pair_sweep(h, n, time_limit=10.0, n_delta=5):
    start = time.time()
    c5 = compute_c5(h)
    pairs = [(i,j) for i in range(n) for j in range(i+1, n)]
    
    deltas = np.linspace(-0.02, 0.02, n_delta)
    
    while time.time() - start < time_limit:

        for i, j in pairs:
            for delta in deltas:
                hi_new = h[i] + delta
                hj_new = h[j] - delta

import numpy as np
import time

def project_feasible(p, n):
    target = n / 2.0
    lo = p.min() - 2.0
    hi = p.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h = np.clip(p - tau, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) > 0:
            h[free] += err / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def active_gradient(h, n, tol_frac):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = float(np.max(corr))
    tol = max(tol_frac * max_val, 1e-8)
    active = np.where(corr >= max_val - tol)[0]
    g = np.zeros(n)
    for r in active:
        k = int(r) - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] += (-h[k:])
        else:
            kp = -k
            g[:n - kp] += (1.0 - h[kp:])
            g[kp:] += (-h[:n - kp])
    return g, active, corr, max_val

def gradient_descent(h0, n, max_iter, time_limit):
    start = time.time()
    h = h0.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    no_improve = 0
    tol_frac = 0.10

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it > 2000:
            tol_frac = 0.05
        if it > 5000:
            tol_frac = 0.02

        g, active, corr, max_val = active_gradient(h, n, tol_frac)
        g_zero = g - np.mean(g)
        d = -g_zero
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm

        best_alpha = -1.0
        best_c5_trial = c5
        steps = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        for alpha in steps:
            h_trial = project_feasible(h + alpha * d, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5_trial - 1e-14:
                best_c5_trial = c5_trial
                best_alpha = alpha

        if best_alpha < 0:
            no_improve += 1
            if no_improve > 300:
                break
            if no_improve > 30 and np.random.random() < 0.5:
                scale = 0.01 + 0.04 * min(no_improve / 200.0, 1.0)
                pert = np.random.randn(n) * scale
                h_trial = project_feasible(h + pert, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5:
                    h = h_trial
                    c5 = c5_trial
                    no_improve = 0
            continue

        h = project_feasible(h + best_alpha * d, n)
        c5 = compute_c5(h)

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

    start = time.time()
    h = h0.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()

    for it in range(max_iter):

            break
        n_pert = 1 + (np.random.random() < 0.25)
