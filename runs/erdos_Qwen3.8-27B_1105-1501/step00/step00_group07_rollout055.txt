            h[j] += delta

import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(R) * dx)

def repair_feasible(h):
    """Ensure 0 <= h <= 1 and sum(h) = len(h)/2."""
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(-h)
            rem = diff
            for idx in order:
                d = min(h[idx], rem)
                h[idx] -= d
                rem -= d
                if rem <= 1e-14:
                    break
        else:
            order = np.argsort(h)
            rem = -diff
            for idx in order:
                d = min(1.0 - h[idx], rem)
                h[idx] += d
                rem -= d
                if rem <= 1e-14:
                    break
    return h

def make_cosine(n, f=1, A=0.5):
    """Generate h[i] = 0.5 + A*cos(2*pi*f*i/n)."""
    i = np.arange(n)
    h = 0.5 + A * np.cos(2.0 * np.pi * f * i / n)
    return repair_feasible(h)

def make_two_tone(n, f1=1, f2=2, A1=0.3, A2=0.2):
    """Generate h[i] = 0.5 + A1*cos(2*pi*f1*i/n) + A2*cos(2*pi*f2*i/n)."""
    i = np.arange(n)
    h = 0.5 + A1 * np.cos(2.0 * np.pi * f1 * i / n) + A2 * np.cos(2.0 * np.pi * f2 * i / n)
    return repair_feasible(h)

def pair_transfer_local_search(h, max_time, seed=42):
    """Local search via random pair transfers with simulated annealing."""
    n = len(h)
    h = h.copy()
    best_c = score_exact(h)
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    T = 0.005
    T_min = 1e-9
    cool_rate = 0.9995
    
    while time.time() - t0 < max_time:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        if rng.random() < 0.5:
            max_delta = min(h[i], 1.0 - h[j])
        else:
            i, j = j, i
            max_delta = min(1.0 - h[i], h[j])
        
        if max_delta <= 1e-12:
            continue
        
        delta = rng.random() * max_delta * 0.5  # use up to half the max
        
        old_hi, old_hj = h[i], h[j]
        h[i] += delta
        h[j] -= delta
        
        new_c = score_exact(h)
        
        if new_c <= best_c:
            best_c = new_c
            best_h = h.copy()
        elif new_c - best_c < -T or rng.random() < np.exp(-(new_c - best_c) / T):
            pass  # accept
        else:
            h[i] = old_hi
            h[j] = old_hj
        
        T = max(T_min, T * cool_rate)
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values

            h = make_two_tone(n, 1, 2, A1, A2)

    # Final validation
    best_h = repair_feasible(best_h)
    best_c = score_exact(best_h)
    
    return (best_h, best_c, len(best_h))

import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(R) * dx)

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
            rem = diff
            for idx in order:
                d = min(h[idx], rem)
                h[idx] -= d
                rem -= d
                if rem <= 1e-14:
                    break
        else:
            order = np.argsort(h)
            rem = -diff
            for idx in order:
                d = min(1.0 - h[idx], rem)
                h[idx] += d
                rem -= d
                if rem <= 1e-14:
                    break
    return h

def make_cosine(n, f=1, A=0.5, phase=0.0):
    i = np.arange(n)
    h = 0.5 + A * np.cos(2.0 * np.pi * f * i / n + phase)
    return repair_feasible(h)

def make_two_tone(n, f1=1, f2=2, A1=0.3, A2=0.2):
    i = np.arange(n)
    h = 0.5 + A1 * np.cos(2.0 * np.pi * f1 * i / n) + A2 * np.cos(2.0 * np.pi * f2 * i / n)
    return repair_feasible(h)

def pair_transfer_sa(h, max_time, seed=42):
    n = len(h)
    h = h.copy()
    best_c = score_exact(h)
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    t0 = time.time()

    T = 0.004
    T_min = 1e-10
    cool = 0.9997

    while time.time() - t0 < max_time:
        batch_end = time.time() + 0.05
        while time.time() < batch_end:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if rng.random() < 0.5:
                max_d = min(h[i], 1.0 - h[j])
            else:
                i, j = j, i
                max_d = min(1.0 - h[i], h[j])
            if max_d <= 1e-13:
                continue
            frac = rng.random()
            if frac < 0.3:
                delta = max_d * 0.1 * rng.random()
            elif frac < 0.7:
                delta = max_d * 0.3 * rng.random()
            else:
                delta = max_d * 0.8 * rng.random()

            h[i] += delta
            h[j] -= delta
            new_c = score_exact(h)

            if new_c <= best_c:
                best_c = new_c
                best_h = h.copy()
            elif new_c - best_c > 0:
                if rng.random() >= np.exp(-(new_c - best_c) / T):
                    h[i] -= delta
                    h[j] += delta
        T = max(T_min, T * cool)

    return best_h, best_c

def greedy_local_search(h, max_time, seed=42):
    n = len(h)
    h = h.copy()
    best_c = score_exact(h)
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    t0 = time.time()

    while time.time() - t0 < max_time:
        improved = False
        R = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(R))
        lag = k_max - (n - 1)

        order = rng.permutation(n)
        for _ in range(200):
            i = int(order[rng.integers(n)])
            if i == lag or (i + lag >= n) or (i - lag < 0):
                j = int(rng.integers(n))
            else:
                j = int(rng.integers(n))
            if i == j:
                continue

            for direction in [1, -1]:
                if direction == 1:
                    max_d = min(h[i], 1.0 - h[j])
                else:
                    max_d = min(h[j], 1.0 - h[i])
                if max_d <= 1e-13:
                    continue
                for frac in [0.1, 0.3, 0.5, 0.8]:
                    delta = max_d * frac
                    if direction == 1:
                        h[i] += delta
                        h[j] -= delta
                    else:
                        h[j] += delta
                        h[i] -= delta
                    new_c = score_exact(h)
                    if new_c < best_c:
                        best_c = new_c
                        best_h = h.copy()
                        improved = True
                    else:
                        if direction == 1:
                            h[i] -= delta
                            h[j] += delta
                        else:
                            h[j] -= delta
                            h[i] += delta
        if not improved:
            T = 0.002
            for _ in range(50):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                max_d = min(h[i], 1.0 - h[j])
                if max_d <= 1e-13:
                    continue
                delta = max_d * rng.random()
                h[i] += delta
                h[j] -= delta
                new_c = score_exact(h)
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                else:
                    h[i] -= delta
                    h[j] += delta
            if best_c == score_exact(h):
                h = best_h.copy()
            break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start = time.time()
    deadline = start + budget_s * 0.94

    candidates = []

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = repair_feasible(np.array(initial_h_values, dtype=float))
        candidates.append(h0)

    for n in [40, 50, 60, 70, 80, 90, 95, 100, 120, 150, 200]:
        for f in [1]:
            for A in [0.45, 0.48, 0.5]:
                candidates.append(make_cosine(n, f, A))

    for n in [95, 100, 120]:
        for A1, A2 in [(0.35, 0.12), (0.3, 0.18), (0.25, 0.22)]:
            candidates.append(make_two_tone(n, 1, 2, A1, A2))

    for n in [95, 100]:
        for phase in [0.0, 0.3, 0.7]:
            candidates.append(make_cosine(n, 1, 0.5, phase))

    best_h = None
    best_c = float('inf')
    for h in candidates:
        c = score_exact(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    elapsed = time.time() - start
    remaining = max(1.0, deadline - time.time())

    # Phase 1: greedy local search
    t1 = remaining * 0.4
    best_h, best_c = greedy_local_search(best_h, max_time=t1, seed=seed)
    elapsed = time.time() - start

    # Phase 2: simulated annealing
    remaining2 = max(1.0, deadline - time.time())
    t2 = remaining2 * 0.5
    h2, c2 = pair_transfer_sa(best_h, max_time=t2, seed=seed + 1)
    if c2 < best_c:
        best_c = c2
        best_h = h2

    # Phase 3: final greedy polish
    remaining3 = max(0.5, deadline - time.time())
    h3, c3 = greedy_local_search(best_h, max_time=remaining3, seed=seed + 2)
    if c3 < best_c:
        best_c = c3
        best_h = h3

    # Final validation and repair
    best_h = repair_feasible(best_h)
    best_c = score_exact(best_h)

    return (best_h, best_c, len(best_h))
