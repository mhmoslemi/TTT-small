def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_feasible(y):

            if abs(diff2) > 1e-14:

def binarize(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[idx[:k]] = 1.0
    return binary

def random_binary(n, rng):

def compute_effects(h):
    n = len(h)
    rem_right = np.zeros((n, n - 1))
    rem_left = np.zeros((n, n - 1))
    for p in range(n):
        d = np.arange(1, n)
        # h[p+d] valid when p+d < n, i.e., d < n-p
        valid_r = d < (n - p)
        rem_right[p, valid_r] = h[p + d[valid_r]]
        valid_l = d <= p
        rem_right[p, valid_l] -= (1.0 - h[p - d[valid_l]])
        vl2 = d < (n - p)
        rem_left[p, vl2] = -(1.0 - h[p + d[vl2]])
        vl3 = d <= p
        rem_left[p, vl3] += h[p - d[vl3]]
    return rem_right, rem_left

def local_search_binary(h, max_passes=300, deadline=None):

    cur = h.copy()
    cur_score = exact_score(cur)
    rem_r, rem_l = compute_effects(cur)
    right = np.zeros(n - 1)
    left = np.zeros(n - 1)
    for d in range(1, n):
        right[d - 1] = np.sum(cur[d:] * (1.0 - cur[:n - d]))
        left[d - 1] = np.sum(cur[:n - d] * (1.0 - cur[d:]))
    cur_max = max(np.max(right), np.max(left))
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break

        best_improve = 1e-14
        best_pair = None
        for i in ones:
            dr_i = rem_r[i]
            dl_i = rem_l[i]
            for j in zeros:
                dr = dr_i - rem_r[j]
                dl = dl_i - rem_l[j]

                nl = left + dl
                new_max = max(float(np.max(nr)), float(np.max(nl)))

                    best_pair = (i, j)

    cur = h.copy()
    cur_score = exact_score(cur)
    alpha = 0.05

                for i in range(n - d):
                    grad[i] += (1.0 - cur[i + d]) / len(active)

                    grad[i] -= cur[i + r] / len(active)
