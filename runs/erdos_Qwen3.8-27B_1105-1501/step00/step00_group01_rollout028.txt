                best_h = h_work.copy()
    if best_score <= target_val:
        return (best_h, best_score, len(best_h))

    # Stage 3: Try nearby n values
    stage3_end = min(deadline, t0 + budget_s * 0.85)
    n_base = len(best_h)
    for n_try in [n_base - 1, n_base + 1, n_base - 2, n_base + 2, n_base - 3, n_base + 3]:
        if time.perf_counter() > stage3_end:
            break
        if n_try < 6 or n_try > 300:
            continue
        h_try = _resample(best_h, n_try)
        s_try = _score(h_try)
        if s_try < best_score:
            # Optimize at this n
            h_try, s_opt = _pair_sweep(h_try, min(deadline, time.perf_counter() + 60), max_rounds=50)
            if s_opt < best_score:
                best_score = s_opt
                best_h = h_try.copy()
        if best_score <= target_val:
            break

    # Stage 4: Random restarts with different seeds
    stage4_end = min(deadline, t0 + budget_s * 0.95)
    for attempt in range(10):
        if time.perf_counter() > stage4_end:
            break
        n_try = max(6, min(300, n_base + rng.randint(-4, 5)))
        h_rand = _random_init(n_try, rng)
        # Quick optimize
        h_rand, s_rand = _pair_sweep(h_rand, min(stage4_end, time.perf_counter() + 20), max_rounds=10)
        if s_rand < best_score:
            best_score = s_rand
            best_h = h_rand.copy()
        if best_score <= target_val:
            break

    # Final: Ensure feasibility and return
    final_h = np.clip(best_h, 0.0, 1.0)
    n_fin = len(final_h)
    final_h = _project(final_h, n_fin / 2.0)
    final_score = _score(final_h)
    return (final_h, final_score, n_fin)
