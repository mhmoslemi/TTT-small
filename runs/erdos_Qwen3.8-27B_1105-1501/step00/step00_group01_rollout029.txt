                kx = n // 2
                hx[:kx] = 1.0
                if n % 2 == 1:
                    hx[kx] = 0.5
                rng.shuffle(hx)
                h_init = hx

            # Local search
            t_ls = time.time() + alloc * 0.5
            hb, cb = swap_ls(h_init, n, t_ls)

            # Guided
            t_g = time.time() + alloc * 0.2
            hg, cg = guided_ls(hb, n, t_g)
            hb = hg if cg < cb else hb
            cb = min(cg, cb)

            # Short SA
            t_sa = time.time() + alloc * 0.3
            hs, cs = sa_opt(hb, n, t_sa, temp0=0.002, cool=0.9998)
            if cs < cb:
                hb, cb = hs, cs

            # Final local search
            if time.time() < deadline - 5:
                hf, cf = swap_ls(hb, n, time.time() + 2)
                if cf < cb:
                    hb, cb = hf, cf

            if cb < best_c5:
                best_c5 = cb
                best_h = hb.copy()

    # Phase 5: final polish
    remaining = deadline - time.time()
    if remaining > 20:
        nb = len(best_h)
        hf, cf = sa_opt(best_h, nb, start_time + budget_s - 3, temp0=0.001, cool=0.9999)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()
        if deadline - time.time() > 5:
            hf2, cf2 = swap_ls(best_h, nb, deadline)
            if cf2 < best_c5:
                best_c5 = cf2
                best_h = hf2.copy()

    c5_eval = evaluate_erdos_solution(best_h)
    if c5_eval < best_c5:
        best_c5 = c5_eval

    return best_h, best_c5, len(best_h)
