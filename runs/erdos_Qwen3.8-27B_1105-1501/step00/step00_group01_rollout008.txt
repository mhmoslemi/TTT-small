    dx = 2.0 / n
    c = compute_all_lags(h)
    cmax = np.max(c)
    active = np.where(c >= cmax - tol)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            grad[0:n-k] += dx * (1.0 - h[k:n])
            grad[k:n] += dx * (-h[0:n-k])
        else:
            kk = -k
            grad[0:n-kk] += dx * (1.0 - h[kk:n])
            grad[kk:n] += dx * (-h[0:n-kk])
    grad /= float(len(active))
    return grad

def slsqp_run(h_init, n, temps, max_iter_each=400):
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    best = h_init.copy()
    best_c = compute_c5(best)
    for temp in temps:
        def f(h):
            c = compute_all_lags(h)
            M = np.max(c)
            return M + temp * np.log(np.sum(np.exp((c - M) / temp)))
        def g(h):
            return grad_smooth(h, temp)
        try:
            res = minimize(f, best, method='SLSQP', jac=g,
                          bounds=bnds, constraints=cons,
                          options={'maxiter': max_iter_each, 'ftol': 1e-14, 'iprint': 0})
        except:
            continue
        if res.x is not None:
            h_res = project_box_sum(res.x, target)
            c_res = compute_c5(h_res)
            if c_res < best_c:
                best = h_res
                best_c = c_res
    return best, best_c

def subgradient_refine(h_init, n, max_steps=20000, tol=1e-9):
    h_cur = h_init.copy()
    c_cur = compute_c5(h_cur)
    target = n / 2.0
    lr = 0.05
    for step in range(max_steps):
        grad = grad_exact_max(h_cur, tol=tol)
        h_new = h_cur - lr * grad
        h_new = project_box_sum(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < c_cur - 1e-14:
            h_cur = h_new
            c_cur = c_new
        if step % 500 == 499:
            lr *= 0.6
            if lr < 1e-6:
                break
    return h_cur, c_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        n = 63
        h0 = np.zeros(n)
        h0[:31] = 1.0
        h0[31] = 0.5

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with annealing temperatures at original n
    n0 = len(best_h)
    temps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h, best_c5 = slsqp_run(best_h, n0, temps, max_iter_each=500)

    if elapsed() > budget_s * 0.35:
        return best_h, best_c5, len(best_h)

    # Phase 2: Subgradient refinement
    best_h, best_c5 = subgradient_refine(best_h, n0, max_steps=15000)

    # Phase 3: SLSQP again after subgradient
    best_h, best_c5 = slsqp_run(best_h, n0, [0.002, 0.001, 0.0005], max_iter_each=400)
    best_h, best_c5 = subgradient_refine(best_h, n0, max_steps=10000)

    if elapsed() > budget_s * 0.55:
        return best_h, best_c5, len(best_h)

    # Phase 4: Try different resolutions
    for n_try in [62, 64, 60, 65, 66, 68, 70, 72, 80, 90, 100, 120]:
        if elapsed() > budget_s * 0.78:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_box_sum(h_new, n_try / 2.0)
        h_opt, c_opt = slsqp_run(h_new, n_try, [0.01, 0.005, 0.002], max_iter_each=300)
        h_opt, c_opt = subgradient_refine(h_opt, n_try, max_steps=5000)
        if c_opt < best_c5:
            best_h = h_opt
            best_c5 = c_opt

    # Phase 5: Random perturbations with local refinement
    n_best = len(best_h)
    for it in range(10000):
        if elapsed() > budget_s * 0.95:
            break
        h_try = best_h.copy()
        n_pert = np.random.randint(1, 5)
        idx = np.random.choice(n_best, size=n_pert, replace=False)
        for i in idx:
            h_try[i] += np.random.uniform(-0.3, 0.3)
        h_try = project_box_sum(h_try, n_best / 2.0)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c_try
            # Quick refinement
            h_opt, c_opt = slsqp_run(best_h, n_best, [0.002, 0.001], max_iter_each=200)
            h_opt, c_opt = subgradient_refine(h_opt, n_best, max_steps=3000)
            if c_opt < best_c5:
                best_h = h_opt
                best_c5 = c_opt

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
