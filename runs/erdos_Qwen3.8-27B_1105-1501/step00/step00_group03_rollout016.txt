    n = len(z)

    tgt_pos = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return np.interp(tgt_pos, src_pos, h_src)

def antisym_transform(h):
    return 1.0 - h[::-1].copy()

def optimize_smooth(h0, deadline, maxiter=120):
    """Optimize using L-BFGS-B with penalty, then SLSQP polish."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cur = h0.copy()
    cur_score = official_score(cur)

    # L-BFGS-B with penalty for sum constraint
    penalties = [50.0, 500.0, 5000.0]
    betas_lbfgs = [30, 200, 1000]

    for i in range(len(betas_lbfgs)):
        if time.time() > deadline:
            break
        beta = betas_lbfgs[i]
        pen = penalties[min(i, len(penalties) - 1)]

        def pen_obj(x, b=beta, p=pen):
            val, _ = smooth_obj_and_grad(x, b)
            s = np.sum(x) - len(x) / 2.0
            return val + p * s * s

        def pen_grad(x, b=beta, p=pen):
            _, g = smooth_obj_and_grad(x, b)
            s = np.sum(x) - len(x) / 2.0
            return g + 2.0 * p * s * np.ones(len(x))

        try:
            res = minimize(pen_obj, cur, jac=pen_grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                new_h = make_feasible(res.x)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-14:
                    cur = new_h
                    cur_score = new_score
        except Exception:
            pass

    # SLSQP polish with exact constraint
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    for beta in [500, 2000, 5000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_func, cur, args=(beta,),
                           jac=smooth_grad_func, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter // 2, 'ftol': 1e-11})
            if res.x is not None and np.all(np.isfinite(res.x)):
                new_h = make_feasible(res.x)
                new_score = official_score(new_h)
                if new_score < cur_score - 1e-14:
                    cur = new_h
                    cur_score = new_score
        except Exception:
            pass

    return cur, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    reserve = max(5.0, 0.05 * budget_s)
    deadline = start_time + budget_s - reserve

    # Get parent
    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
        if not np.all(np.isfinite(parent)) or len(parent) < 5:
            parent = None
    except (NameError, AttributeError):
        parent = None

    # Initialize best
    if parent is not None:
        base_n = len(parent)
        best_h = make_feasible(parent)
    else:
        base_n = 45
        best_h = make_feasible(np.full(base_n, 0.5))

    best_score = official_score(best_h)
    best_n = len(best_h)
    rng = np.random.default_rng(seed)

    # Resolutions to try
    res_list = [base_n]
    if base_n > 24:
        res_list.append(max(18, int(base_n * 0.7)))
    res_list.append(min(72, int(base_n * 1.15)))
    if budget_s > 400:
        res_list.append(min(80, int(base_n * 1.3)))
    # Remove duplicates, sort
    res_list = sorted(set(res_list))

    # Phase 1: Systematic optimization from structured starts
    for n in res_list:
        if time.time() > deadline:
            break

        candidates = []
        if parent is not None:
            candidates.append(make_feasible(interp_to(parent, n)))
        candidates.append(make_feasible(interp_to(best_h, n)))
        candidates.append(make_feasible(antisym_transform(interp_to(best_h, n))))
        # Slightly shifted versions
        shifted = np.roll(best_h, 1)
        candidates.append(make_feasible(interp_to(shifted, n)))

        # Random starts
        remaining = deadline - time.time()
        n_rand = max(4, min(12, int(remaining / 45.0)))
        for _ in range(n_rand):
            sigma = rng.choice([0.1, 0.2, 0.35, 0.5])
            raw = rng.standard_normal(n) * sigma + 0.5
            candidates.append(make_feasible(raw))

        for cand in candidates:
            if time.time() > deadline:
                break
            result_h, result_score = optimize_smooth(cand, deadline)
            if result_score < best_score - 1e-12:
                best_h = result_h
                best_score = result_score
                best_n = n

    # Phase 2: Intensive perturbation around best
    remaining = deadline - time.time()
    if remaining > 30:
        n_perturb = max(5, min(30, int(remaining / 25.0)))
        for _ in range(n_perturb):
            if time.time() > deadline:
                break
            sigma = rng.choice([0.02, 0.04, 0.07, 0.1, 0.15])
            noise = rng.standard_normal(best_n) * sigma
            perturbed = make_feasible(best_h + noise)
            result_h, result_score = optimize_smooth(perturbed, deadline, maxiter=60)
            if result_score < best_score - 1e-12:
                best_h = result_h
                best_score = result_score

    # Phase 3: Try to reduce resolution (smaller n preferred on tie)
    if best_n > 20 and time.time() < deadline:
        for n_small in [best_n - 2, best_n - 5, best_n - 8]:
            if n_small < 16:
                break
            if time.time() > deadline:
                break
            cand = make_feasible(interp_to(best_h, n_small))
            result_h, result_score = optimize_smooth(cand, deadline, maxiter=80)
            if result_score < best_score - 1e-12:
                best_h = result_h
                best_score = result_score
                best_n = n_small
            elif abs(result_score - best_score) < 1e-8 and len(result_h) < best_n:
                best_h = result_h
                best_score = result_score
                best_n = n_small

    # Final
    final_h = make_feasible(best_h)
    final_score = official_score(final_h)
    final_bound = float(final_score) + 1e-12

    return final_h, final_bound, len(final_h)
