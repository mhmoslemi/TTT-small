import numpy as np

def project_feasible(y, target):
    """Project y onto {x in [0,1]^n : sum(x) = target} via bisection on lambda."""
    n = len(y)
    # Find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    
    def phi(lam):
        return float(np.sum(np.clip(y - lam, 0.0, 1.0))) - target
    
    # phi(lo) >= 0, phi(hi) <= 0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if phi(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    lam = 0.5 * (lo + hi)
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix residual
    diff = target - float(np.sum(z))
    if abs(diff) > 1e-12:
        # Adjust elements with most slack
        if diff > 0:
            # Need to increase some elements
            slack = 1.0 - z
            idx = int(np.argmax(slack))
            adj = min(diff, slack[idx])
            z[idx] += adj
            diff -= adj
            if abs(diff) > 1e-14:
                slack[idx] = 0.0
                idx2 = int(np.argmax(slack))
                z[idx2] += diff
        else:
            # Need to decrease some elements
            slack = z
            idx = int(np.argmax(slack))
            adj = min(-diff, slack[idx])
            z[idx] -= adj
            diff += adj
            if abs(diff) > 1e-14:
                slack[idx] = 0.0
                idx2 = int(np.argmax(slack))
                z[idx2] -= diff
    
    return z

def score_exact(h):
    """Compute C_5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def score_and_grad(h, T):
    """Compute smooth-max score and gradient. Returns (smooth_score, grad, c_array)."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cx = c - c.max()
    if T <= 1e-15:
        # Hard max
        weights = np.zeros_like(c)
        weights[c == c.max()] = 1.0 / np.sum(c == c.max())
        smooth = float(c.max())
    else:
        ex = np.exp(cx / T)
        S = float(np.sum(ex))
        if S < 1e-300:
            weights = np.zeros_like(c)
            weights[c == c.max()] = 1.0 / np.sum(c == c.max())
            smooth = float(c.max())
        else:
            weights = ex / S
            smooth = float(c.max() + T * np.log(S))
    
    grad = np.zeros(n)
    center = n - 1
    
    # d = 0
    w0 = weights[center]
    grad += w0 * dx * (1.0 - 2.0 * h)
    
    # d >= 1
    for d in range(1, n):
        wp = weights[center + d]  # positive lag
        wn = weights[center - d]  # negative lag
        m = n - d
        # For indices 0..m-1:
        grad[:m] += (wn * (1.0 - h[d:]) - wp * h[d:]) * dx
        # For indices d..n-1:
        grad[d:] += (wp * (1.0 - h[:m]) - wn * h[:m]) * dx
    
    return smooth, grad, c

    best = cur.copy()
    best_score = cur_score

    for it in range(max_iters):
        if time.time() > deadline:

            if trial_score < cur_score - 1e-14:

                cur_score = trial_score

                    best = cur.copy()
                    best_score = cur_score
                accepted = True

        if not accepted:

        # Occasionally do a pair-line search
        if it % 50 == 49 and step_size > 0.001:

                    cur[j] += a
                    if s < best_s - 1e-14:

                if best_a != 0.0:

                    cur = project_feasible(cur, target)
                    cur_score = score_exact(cur)
                    if cur_score < best_score - 1e-12:
                        best = cur.copy()
                        best_score = cur_score
        
        # Update temperature
        T = max(T * decay, T_end)
    
    return best, best_score

def resample(h_old, n_new):

    n_old = len(h_old)

    h_new = np.interp(x_new, x_old, h_old, left=0.0, right=0.0)
    return project_feasible(h_new, n_new / 2.0)

    it = 0
    stall = 0
    n_stall_thresh = 50
    
    while it < max_iters and time.time() < deadline:
        smooth, grad, c = smooth_score_grad(cur, T)
        if not np.all(np.isfinite(grad)):
            T *= 0.5
            it += 1
            continue
        
        gnorm = float(np.sqrt(np.mean(grad * grad)))
        if gnorm < 1e-15:
            break
        
        accepted = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            ss = step_size * mult
            trial = cur - ss * grad
            trial = project_feasible(trial, target)
            ts = score_exact(trial)
            if ts < cur_score - 1e-13:
                cur = trial
                cur_score = ts
                if cur_score < best_score - 1e-12:
                    best = cur.copy()
                    best_score = cur_score
                accepted = True
                stall = 0
                step_size = min(step_size * 1.05, 0.2)
                break
        
        if not accepted:
            step_size *= 0.75
            stall += 1
            if stall > n_stall_thresh:
                step_size = max(step_size, 0.01)
                stall = 0
        
        # Pair line search every 100 iterations
        if it % 100 == 99 and step_size > 0.0005:
            # Choose pairs based on gradient extremes
            g_sorted = np.argsort(grad)
            lo_idx = g_sorted[:8]
            hi_idx = g_sorted[-8:]
            for attempt in range(6):
                i = int(lo_idx[attempt % 8])
                j = int(hi_idx[attempt % 8])
                if i == j:
                    continue
                lo_a = max(-cur[i], cur[j] - 1.0) + 1e-9
                hi_a = min(1.0 - cur[i], cur[j]) - 1e-9
                if hi_a - lo_a < 1e-8:
                    continue
                alphas = np.linspace(lo_a, hi_a, 13)
                best_a = 0.0
                best_s = cur_score
                for a in alphas:
                    cur[i] += a
                    cur[j] -= a
                    s = score_exact(cur)
                    if s < best_s:
                        best_s = s
                        best_a = a
                    cur[i] -= a
                    cur[j] += a
                if best_a != 0.0 and best_s < cur_score - 1e-13:
                    cur[i] += best_a
                    cur[j] -= best_a
                    cur = project_feasible(cur, target)
                    cur_score = score_exact(cur)
                    if cur_score < best_score - 1e-12:
                        best = cur.copy()
                        best_score = cur_score
        
        # Also try random pairs occasionally
        if it % 200 == 199:
            for _ in range(4):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                lo_a = max(-cur[i], cur[j] - 1.0) + 1e-9
                hi_a = min(1.0 - cur[i], cur[j]) - 1e-9
                if hi_a - lo_a < 1e-8:
                    continue
                alphas = np.linspace(lo_a, hi_a, 9)
                best_a = 0.0
                best_s = cur_score
                for a in alphas:
                    cur[i] += a
                    cur[j] -= a
                    s = score_exact(cur)
                    if s < best_s:
                        best_s = s
                        best_a = a
                    cur[i] -= a
                    cur[j] += a
                if best_a != 0.0 and best_s < cur_score - 1e-13:
                    cur[i] += best_a
                    cur[j] -= best_a
                    cur = project_feasible(cur, target)
                    cur_score = score_exact(cur)
                    if cur_score < best_score - 1e-12:
                        best = cur.copy()
                        best_score = cur_score
        
        # Temperature schedule
        T = max(T_start * (T_end / T_start) ** (it / max(max_iters, 1)), T_end)
        it += 1
    
    return best, best_score

def binarize(h):
    n = len(h)
    target = n / 2.0
    k = int(round(target))
    idx = np.argsort(h)[::-1][:k]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    if abs(target - k) > 1e-10:
        h_bin[idx[-1]] = target - (k - 1)
    return h_bin

def mutate(h, sigma, rng):
    n = len(h)
    noise = rng.normal(0, sigma, n)
    return project_feasible(h + noise, n / 2.0)

def resample(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, left=0.0, right=0.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)
    
    # Get initial vector
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, UnboundLocalError):
        h_init = np.full(70, 0.5)
    
    n0 = len(h_init)
    target0 = n0 / 2.0
    h_init = project_feasible(h_init, target0)
    best_c = score_exact(h_init)
    best_h = h_init.copy()
    
    # Phase 1: Local refine parent (longest phase)
    frac = (deadline - t0)
    d1 = t0 + frac * 0.35
    if time.time() < d1:
        hr, cr = local_refine(h_init, d1, rng, T_start=0.06, T_end=2e-5, max_iters=300000)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
    
    # Phase 2: Try binarized + refine
    if time.time() < deadline - 5:
        h_bin = binarize(best_h)
        c_bin = score_exact(h_bin)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        
        d2 = min(deadline - 5, time.time() + frac * 0.2)
        if time.time() < d2:
            hr2, cr2 = local_refine(h_bin, d2, rng, T_start=0.04, T_end=1e-5, max_iters=200000)
            if cr2 < best_c:
                best_c = cr2
                best_h = hr2.copy()
    
    # Phase 3: Mutations + refine
    if time.time() < deadline - 10:
        for sigma in [0.03, 0.08, 0.15]:
            if time.time() > deadline - 15:
                break
            h_mut = mutate(best_h, sigma, rng)
            d3 = min(deadline - 10, time.time() + frac * 0.08)
            if time.time() < d3:
                hr3, cr3 = local_refine(h_mut, d3, rng, T_start=0.04, T_end=1e-5, max_iters=150000)
                if cr3 < best_c:
                    best_c = cr3
                    best_h = hr3.copy()
    
    # Phase 4: Try different resolutions
    if time.time() < deadline - 20:
        n_base = len(best_h)
        res_list = []
        for dn in [-14, -10, -6, -4, 6, 10, 14, 20, 30, 50]:
            nn = n_base + dn
            if 30 <= nn <= 160 and nn != n_base:
                res_list.append(nn)
        # Also try some absolute values
        for nn in [48, 56, 64, 80, 96, 120, 140]:
            if nn not in res_list:
                res_list.append(nn)
        res_list = sorted(set(res_list))
        
        for nn in res_list:
            if time.time() > deadline - 15:
                break
            h_res = resample(best_h, nn)
            c_res = score_exact(h_res)
            d4 = min(deadline - 12, time.time() + frac * 0.05)
            if time.time() < d4 and c_res < best_c + 0.01:
                hr4, cr4 = local_refine(h_res, d4, rng, T_start=0.04, T_end=1e-5, max_iters=100000)
                if cr4 < best_c:
                    best_c = cr4
                    best_h = hr4.copy()
    
    # Phase 5: Final fine-tune on best
    if time.time() < deadline:
        hr5, cr5 = local_refine(best_h, deadline, rng, T_start=0.02, T_end=1e-6, max_iters=500000)
        if cr5 < best_c:
            best_c = cr5
            best_h = hr5.copy()
    
    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c = score_exact(best_h)
    if final_c < best_c:
        best_c = final_c
    
    return best_h, best_c, n_final
