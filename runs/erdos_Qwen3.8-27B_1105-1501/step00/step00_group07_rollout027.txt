            q = rng.randint(n)
            if p == q:
                continue
            a = 0.05 * rng.rand()
            # Ensure feasibility

                # Try reverse
                if h[q] + a > 1.0 or h[p] - a < 0.0:
                    continue
                p, q = q, p
            a = min(a, 1.0 - h[p], h[q])
            if a < 1e-10:
                continue
        
        # Compute C_new = C + delta
        # delta from the formula
        C_new = C.copy()
        C_new[p:p+n] += a * Gr
        C_new[q:q+n] -= a * Gr
        C_new[n-1-p:n-1-p+n] -= a * h
        C_new[n-1-q:n-1-q+n] += a * h
        C_new[n-1] -= 2.0 * a * a
        idx1 = n - 1 + p - q
        idx2 = n - 1 + q - p
        if 0 <= idx1 < 2*n-1:
            C_new[idx1] += a * a
        if 0 <= idx2 < 2*n-1:
            C_new[idx2] += a * a
        
        new_max = C_new.max()
        delta_obj = (new_max - current_max) * dx
        
        # Acceptance criterion
        if delta_obj <= 0 or rng.rand() < np.exp(-delta_obj / T):
            # Accept
            # Update h
            h[p] += a
            h[q] -= a
            if is_binary:
                h[p] = 1.0
                h[q] = 0.0
            # Update Gr
            Gr[n-1-p] -= a
            Gr[n-1-q] += a
            C = C_new
            current_max = new_max
            num_accepted += 1
            
            if is_binary:
                ones = np.setdiff1d(ones, [q])
                zeros = np.setdiff1d(zeros, [p])
                ones = np.append(ones, p)
                zeros = np.append(zeros, q)
            
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
                last_improve = time.time()
                T = min(T * 1.05, 0.05)  # Maybe reheat slightly
        else:
            num_rejected += 1
        
        # Decay temperature
        T = max(T_min, T * T_decay)
    
    # Final verification
    best_c5 = eval_c5(best_h)
    if best_c5 < best_max * dx:
        best_max = best_c5 / dx
    
    return best_h, best_max * dx
