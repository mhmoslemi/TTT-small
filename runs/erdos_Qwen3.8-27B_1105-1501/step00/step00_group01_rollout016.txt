def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            nc = np.sum(mask)
            if nc == 0:
                break
            amt = diff / nc
            h[mask] = np.minimum(1.0, h[mask] + amt)
        else:
            mask = h > 1e-12
            nc = np.sum(mask)
            if nc == 0:
                break
            amt = diff / nc
            h[mask] = np.maximum(0.0, h[mask] + amt)
    return h

def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

    n = len(h)
    cvals = corr_array(h)
    idx = int(np.argmax(cvals))
    k = idx - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)

            grad[i] += -dx * h[i - j]
    return cvals[idx], grad

    raw = np.correlate(h, 1 - h, mode='full')
    cvals = raw * dx
    maxc = np.max(cvals)

                grad[i] += -p * dx * h[i - j]
    val = maxc + temp * np.log(np.sum(exps))
    return val, grad

        h_new = h - lr * grad

        if new_c5 < best_c5:

            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
    return best_h, best_c5

def binary_local_search(h_binary, n_pts):
    """Local search on binary sequences via pairwise swaps."""
    h = h_binary.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    current_c5 = evaluate_c5(h)
    
    improved = True
    while improved:
        improved = False
        # Try all pairs of swaps
        best_improvement = 0
        best_swap = None
        for i in ones:

                    continue
                h[i] = 0
                h[j] = 1
                new_c5 = evaluate_c5(h)
                h[i] = 1
                h[j] = 0

            h[i] = 0
            h[j] = 1
            current_c5 -= best_improvement

            zeros = np.where(h == 0)[0]

    return h, current_c5, h_cont, c5_cont

def slsqp_opt(h_start, n_pts):
    target_sum = n_pts / 2.0
    h = project_feasible(h_start.copy(), target_sum)
    temp = 0.02
    
    def obj(x):
        return grad_smoothed(np.clip(x, 0, 1), temp)[0]
    
    def jac(x):
        return grad_smoothed(np.clip(x, 0, 1), temp)[1]
    
    bounds = [(0, 1)] * n_pts
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n_pts)}]
    try:

                       options={'maxiter': 300, 'ftol': 1e-9})
        if res.x is not None:
            h_opt = project_feasible(res.x, target_sum)
            return h_opt
    except:
        pass
    return h

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            nc = np.sum(mask)
            if nc == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / nc)
        else:
            mask = h > 1e-12
            nc = np.sum(mask)
            if nc == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / nc)
    return h

def grad_active(h):
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1 - h, mode='full') * dx
    idx = int(np.argmax(cvals))
    k = idx - (n - 1)
    grad = np.zeros(n)

        for i in range(n - k):
            grad[i] -= dx * h[i + k]
            grad[i + k] += dx * (1.0 - h[i])
    else:
        j = -k
        for i in range(j, n):
            grad[i - j] += dx * (1.0 - h[i])
            grad[i] -= dx * h[i - j]
    return float(cvals[idx]), grad

def slsqp_opt(h_start, n_pts, temp=0.02):

    h = project_feasible(h_start.copy(), target_sum)
    
    def obj_and_grad(x):

        n = len(xt)
        dx = 2.0 / n
        raw = np.correlate(xt, 1 - xt, mode='full') * dx
        maxc = np.max(raw)
        exps = np.exp((raw - maxc) / temp)
        probs = exps / np.sum(exps)
        val = maxc + temp * np.log(np.sum(exps))
        grad = np.zeros(n)
        for m in range(len(raw)):

                for ii in range(n - k):
                    grad[ii] -= p * dx * xt[ii + k]
                    grad[ii + k] += p * dx * (1.0 - xt[ii])
            else:
                jj = -k
                for ii in range(jj, n):
                    grad[ii - jj] += p * dx * (1.0 - xt[ii])
                    grad[ii] -= p * dx * xt[ii - jj]
        return val, grad

        pass
