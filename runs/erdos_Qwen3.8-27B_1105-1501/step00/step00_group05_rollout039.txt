def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")

def c5_gradient(h):

        lag = int(k - (n - 1))
        if lag >= 0:
            js1 = np.arange(lag, n)
            grad[js1] += (1.0 - h[js1 - lag])
            js2 = np.arange(0, n - lag)
            grad[js2] -= h[js2 + lag]
        else:

            grad[js1] += (1.0 - h[js1 + L])
            js2 = np.arange(L, n)

    m = len(active)
    if m == 0:
        return None

    c_obj[-1] = 1.0
    A_ub = np.hstack([grads, -np.eye(m)])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
