c_lag = sum_{n: lo≤n≤hi} h[n+lag] * (1-h[n]) where lo = max(0,-lag), hi = min(n-1, n-1-lag)

Now, dc_lag/dh[j]:
h[j] appears in the sum in two ways:
1. As h[n+lag] when n+lag = j, i.e., n = j-lag. Valid if lo ≤ j-lag ≤ hi and 0 ≤ j < n.
   Contribution: (1-h[n]) = (1-h[j-lag])
2. As (1-h[n]) when n = j. Valid if lo ≤ j ≤ hi and 0 ≤ j+lag < n (which is guaranteed by hi ≤ n-1-lag).
   Contribution: -h[n+lag] = -h[j+lag]

So dc_lag/dh[j] = (1-h[j-lag])*1[lo≤j-lag≤hi] - h[j+lag]*1[lo≤j≤hi]

And the full gradient of c5_soft = sum_lag p_lag * c_lag * dx:
grad[j] = dx * sum_lag p_lag * [(1-h[j-lag])*1[lo_lag≤j-lag≤hi_lag] - h[j+lag]*1[lo_lag≤j≤hi_lag]]

This is O(n * (2n-1)) = O(n²) per gradient evaluation, which for n=47 is about 47*93 ≈ 4400 operations. That's fine.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, n):
    """Project h onto the constraint sum(h) = n/2, keeping h in [0,1]."""
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # Re-project (simple iteration)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        h = h + diff / n
        h = np.clip(h, 0, 1)
    return h

def c5_soft_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    bc = beta * corr
    bm = np.max(bc)
    return (bm + np.log(np.sum(np.exp(bc - bm)))) / beta

def c5_soft_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    bc = beta * corr
    bm = np.max(bc)
    exp_bc = np.exp(bc - bm)
    p = exp_bc / np.sum(exp_bc)
    
    grad = np.zeros(n)
    for l_idx in range(len(corr)):
        lag = l_idx - (n - 1)
        if p[l_idx] < 1e-15:
            continue
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        w = p[l_idx] * dx
        
        # Part 1: j in [i_lo, i_hi], contribution = w * (1 - h[j+lag])
        js = max(0, i_lo)
        je = min(n - 1, i_hi)
        if js <= je:
            grad[js:je+1] += w * (1.0 - h[js+lag:je+1+lag])
        
        # Part 2: j in [i_lo+lag, i_hi+lag], contribution = -w * h[j-lag]
        js2 = max(0, i_lo + lag)
        je2 = min(n - 1, i_hi + lag)
        if js2 <= je2:
            grad[js2:je2+1] -= w * h[js2-lag:je2+1-lag]
    
    return grad

def optimize_from_start(h0, beta_schedule, max_iter=5000, tol=1e-10):
    """Optimize starting from h0 using L-BFGS-B with soft max."""
    n = len(h0)
    h = h0.copy()
    
    for beta in beta_schedule:
        bounds = [(0, 1)] * n
        # Use penalty for the sum constraint
        lam = 100.0
        
        def obj(phi):
            val = c5_soft_objective(phi, beta)
            penalty = lam * (np.sum(phi) - n/2.0) ** 2
            return val + penalty
        
        def grad(phi):
            g = c5_soft_gradient(phi, beta)
            g += 2 * lam * (np.sum(phi) - n/2.0)
            return g
        
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8})
        h = res.x
        # Project back
        h = project_to_constraint(h, n)
    
    return h

            b = rng.choice(zeros)

            h[b] = 1
            c_new = compute_c5(h)
            h[a] = 1

            h[b] = 1
            best_c5 = compute_c5(h)
        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target):
    """Project h to have sum ≈ target, staying in [0,1]."""
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def c5_soft_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    bc = beta * corr
    bm = np.max(bc)
    return (bm + np.log(np.sum(np.exp(bc - bm)))) / beta

        lag = l_idx - (n - 1)
        w = p[l_idx] * dx
        
        # Part 1: j in [max(0,lag), min(n-1, n-1+lag)], contributes w*(1-h[j-lag])
        js1 = max(0, lag)
        je1 = min(n-1, n-1+lag)
        if js1 <= je1:
            grad[js1:je1+1] += w * (1.0 - h[js1-lag:je1+1-lag])
        
        # Part 2: j in [max(0,-lag), min(n-1, n-1-lag)], contributes -w*h[j+lag]
        js2 = max(0, -lag)
        je2 = min(n-1, n-1-lag)
        if js2 <= je2:
            grad[js2:je2+1] -= w * h[js2+lag:je2+1+lag]
    
    return grad

def optimize_soft(h0, beta, target, max_iter=5000, lam=50.0):
    """Optimize soft c5 with sum constraint via penalty."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return c5_soft_value(x, beta) + lam * (np.sum(x) - target) ** 2
    
    def grad(x):
        g = c5_soft_grad(x, beta) + 2.0 * lam * (np.sum(x) - target)
        return g
    
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                             'maxls': 50})
    return res.x
