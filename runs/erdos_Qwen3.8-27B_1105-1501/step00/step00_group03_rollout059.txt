def official_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n

def make_feasible(z, target=None):
    """Project z onto {h: 0<=h<=1, sum(h)=target}"""
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    if target is None:
        target = n / 2.0
    
    # Bisection on lambda: find lambda such that sum(clip(z + lambda, 0, 1)) = target
    lo = -1.0 - np.max(z) - 1.0
    hi = 1.0 - np.min(z) + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            # Add to entries with most room
            idx = np.argsort(1.0 - h)[::-1]
            for i in idx:
                add = min(residual, 1.0 - h[i])
                if add <= 0:
                    break
                h[i] += add
                residual -= add
        else:
            idx = np.argsort(h)[::-1]
            for i in idx:
                sub = min(-residual, h[i])
                if sub <= 0:
                    break
                h[i] -= sub
                residual += sub
    
    # Final exact fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        # Find an entry that can absorb it
        if residual > 0:
            room = 1.0 - h
        else:
            room = h
        # Pick the entry with most room
        i = np.argmax(room)
        h[i] += residual
        h = np.clip(h, 0.0, 1.0)
        # Re-fix if clipping broke sum
        residual2 = target - np.sum(h)
        if abs(residual2) > 0:
            # Spread across all with room
            if residual2 > 0:
                room = 1.0 - h
            else:
                room = h
            # proportional
            total_room = np.sum(room)
            if total_room > 0:
                h += residual2 * room / total_room
            h = np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    # Compute all 2n-1 correlation values
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax approximation of max
    max_c = np.max(corr)
    if beta < 1e-6:
        return max_c, np.zeros(n)
    exp_vals = np.exp(beta * (corr - max_c))
    # Handle underflow
    if np.sum(exp_vals) == 0:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = exp_vals / np.sum(exp_vals)
    
    obj = max_c + np.log(np.sum(exp_vals)) / beta
    
    # Gradient: d/dh[j] of sum_s w_s * c_s
    # For each shift s, d c_s / d h[j]
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        # c(s) = sum_{j=max(0,s)}^{min(n, n+s)-1} h[j]*(1-h[j-s])
        w = weights[s + n - 1] * dx
        if w == 0:
            continue
        j_lo = max(0, s)
        j_hi = min(n, n + s)
        for j in range(j_lo, j_hi):
            grad[j] += w * (1.0 - h[j - s])
            if s != 0:
                grad[j - s] += w * (-h[j])
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel().copy()
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -2.0 - np.max(z)
    hi = 2.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
            else:
                add = max(residual, -h[i])
            if abs(add) < 1e-16:
                break
            h[i] += add
            residual -= add
    return np.clip(h, 0.0, 1.0)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(x_dst, x_src, h_src)

def _smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        return max_c
    return max_c + np.log(exp_sum) / beta

def _smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    if exp_sum < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        shift = k - (n - 1)
        j_lo = max(0, shift)
        j_hi = min(n, n + shift)
        for p in range(j_lo, j_hi):
            grad[p] += w * (1.0 - x[p - shift])
        p_lo2 = max(0, j_lo - shift)
        p_hi2 = min(n, j_hi - shift)
        for p in range(p_lo2, p_hi2):
            grad[p] += w * (-x[p + shift])
    return grad

def _eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def _eq_jac(x):
    return np.ones(len(x))

def _box_bounds(n):
    return [(0.0, 1.0)] * n

def do_smooth_opt(h0, beta, maxiter=300):
    n = len(h0)
    try:
        res = minimize(
            _smooth_obj, h0, args=(beta,),
            jac=_smooth_grad,
            method='SLSQP',
            bounds=_box_bounds(n),
            constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res['x'])):
            return make_feasible(res['x'])
    except Exception:
        pass
    return h0

    return z[-1]
