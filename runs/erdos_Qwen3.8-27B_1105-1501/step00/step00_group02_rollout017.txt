import numpy as np
import scipy.optimize as sopt
import time


def project_feasible(v, n):
    """Project v onto {h : 0<=h<=1, sum(h)=n/2} via bisection on threshold."""
    target = n / 2.0
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h):
    """Compute C5 = max_k correlate(h,1-h,'full')*dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def build_basis(n, n_modes):
    """Build centered cosine+sine basis, returns array of shape (2*n_modes, n)."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    parts = []
    for j in range(1, n_modes + 1):
        b = np.cos(j * np.pi * x)
        b -= b.mean()
        parts.append(b)
    for j in range(1, n_modes + 1):
        b = np.sin(j * np.pi * x)
        b -= b.mean()
        parts.append(b)
    return np.array(parts)


def mode_objective(coeffs, h_base, basis, n):
    """C5 for h = project(h_base + coeffs @ basis)."""
    h = h_base + coeffs.dot(basis)
    h = project_feasible(h, n)
    return compute_c5(h)


def optimize_modes(h_start, n, n_modes, deadline, rng=None):
    """Optimize mode coefficients with Nelder-Mead from multiple starts."""
    basis = build_basis(n, n_modes)
    ndim = 2 * n_modes
    x0 = np.zeros(ndim)
    best_f = mode_objective(x0, h_start, basis, n)
    best_x = x0.copy()

    # Main Nelder-Mead from zero
    try:
        res = sopt.minimize(
            mode_objective, x0, args=(h_start, basis, n),
            method='Nelder-Mead',
            options={'maxiter': 8000, 'xatol': 1e-7, 'fatol': 1e-9, 'adaptive': True}
        )
        if res.fun < best_f:
            best_f = res.fun
            best_x = res.x.copy()
    except Exception:
        pass

    # Random restarts
    n_restarts = 5
    for _ in range(n_restarts):
        if time.time() > deadline:
            break
        if rng is not None:
            x0r = rng.standard_normal(ndim) * 0.08
        else:
            x0r = np.random.standard_normal(ndim) * 0.08
        try:
            res = sopt.minimize(
                mode_objective, x0r, args=(h_start, basis, n),
                method='Nelder-Mead',
                options={'maxiter': 4000, 'xatol': 1e-7, 'fatol': 1e-9, 'adaptive': True}
            )
            if res.fun < best_f:
                best_f = res.fun
                best_x = res.x.copy()
        except Exception:
            pass

    h_new = h_start + best_x.dot(basis)
    h_new = project_feasible(h_new, n)
    return h_new


def binary_local_search(n, deadline, rng, threshold):
    """Local search on binary arrangements (n must be even)."""
    if n % 2 != 0:
        return None
    half = n // 2
    best_c5 = float('inf')
    best_h = None

    for restart in range(30):
        if time.time() > deadline:
            break
        h = np.zeros(n)
        ones = rng.choice(n, half, replace=False)
        h[ones] = 1.0
        c5 = compute_c5(h)

        no_improve = 0
        while no_improve < 300 and time.time() < deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] == h[j]:
                no_improve += 1
                continue
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-14:
                c5 = new_c5
                no_improve = 0
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if best_h is not None and best_c5 < threshold:
        return best_h
    return None


def polish(h, n, deadline, rng):
    """Polish via random pair transfers at decreasing scales."""
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for delta in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for _ in range(500):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                # i -> j
                ni = best_h[i] - delta
                nj = best_h[j] + delta
                if 0.0 <= ni and nj <= 1.0:
                    trial = best_h.copy()
                    trial[i] = ni
                    trial[j] = nj
                    c5 = compute_c5(trial)
                    if c5 < best_c5 - 1e-12:
                        best_h = trial
                        best_c5 = c5
                        improved = True
                # j -> i
                ni = best_h[i] + delta
                nj = best_h[j] - delta
                if 0.0 <= ni and nj <= 1.0:
                    trial = best_h.copy()
                    trial[i] = ni
                    trial[j] = nj
                    c5 = compute_c5(trial)
                    if c5 < best_c5 - 1e-12:
                        best_h = trial
                        best_c5 = c5
                        improved = True
    return best_h, best_c5


def soft_max_objective(h, h_base, basis, n, T):
    """Smooth C5 using log-sum-exp for gradient-based methods."""
    h_trial = h_base + h.dot(basis)
    h_trial = project_feasible(h_trial, n)
    dx = 2.0 / n
    corr = np.correlate(h_trial, 1.0 - h_trial, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T
