def active_set_lp(h, n, active_indices, max_iter=100):
    """
    Solve min t s.t. O(k,h) ≈ t for active k, 0≤h≤1, sum(h)=n/2
    using linearization at current h.
    Returns new h.
    """
    from scipy.optimize import linprog
    dx = 2.0 / n
    num_lags = 2*n - 1
    overlaps = np.correlate(h, 1.0-h, mode='full') * dx
    
    # Compute gradients
    grads = np.zeros((num_lags, n))
    for j in range(n):
        # Analytical gradient
        for k in range(-n+1, n):
            # O(k) = sum_i h[i](1-h[i+k])*dx over valid i
            # dO(k)/dh[j] = dx*(1-h[j+k]) - dx*h[j-k] (if valid)
            g = 0.0
            if 0 <= j+k < n:
                g += dx * (1.0 - h[j+k])
            if 0 <= j-k < n:
                g -= dx * h[j-k]
            grads[k+n-1, j] = g
    
    # Active set: use top m overlaps
    m = max(1, min(len(active_indices), n + 5))  # not too many constraints
    active = active_indices[:m]
    
    # LP: min t
    # Variables: h[0], ..., h[n-1], t  (n+1 variables)
    # Constraints:
    #   sum(h) = n/2
    #   0 ≤ h[i] ≤ 1
    #   For each active k: sum_j grads[k,j] * h[j] ≤ t  (linearized)
    #   => sum_j grads[k,j] * h[j] - t ≤ 0
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((m + n, n+1))
    b_ub = np.zeros(m + n)
    
    # Overlap constraints (linearized)
    for i, k_idx in enumerate(active):
        A_ub[i, :n] = grads[k_idx, :]
        A_ub[i, n] = -1.0
        b_ub[i] = 0.0

    A_ub[m:m+n, :n] = np.eye(n)

        else:
