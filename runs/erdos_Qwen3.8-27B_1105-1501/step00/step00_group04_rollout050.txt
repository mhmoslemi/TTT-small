def make_feasible(y, target=None):
    """Project y onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    y[~np.isfinite(y)] = 0.5
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            j = int(idx[0])
            if diff > 0:
                h[j] = min(1.0, h[j] + diff)
            else:
                h[j] = max(0.0, h[j] + diff)
    return h

def score_exact(h):
    """C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def score_scores(h):
    """Full correlation vector scaled by dx"""
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_and_grad(h, T):
    """Smooth max surrogate and gradient."""
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    c = score_scores(h)
    center = n - 1
    cmax = float(np.max(c))
    ex = np.exp((c - cmax) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        smooth_val = cmax
    else:
        w = ex / S
        smooth_val = cmax + T * np.log(S)
    g = np.zeros(n)
    g += w[center] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[center + d]
        wn = w[center - d]
        if wp > 1e-15:
            g[d:] += wp * dx * (1.0 - h[:n - d])
            g[:n - d] += wp * dx * (-h[d:])
        if wn > 1e-15:
            g[:n - d] += wn * dx * (1.0 - h[d:])
            g[d:] += wn * dx * (-h[:n - d])
    return smooth_val, g, w

def local_refine(h_init, deadline, rng, T_start=0.1, T_end=1e-4, step_start=0.05):

    h = h_init.copy()

        pair_count += 1

        except Exception:

            continue

                    else:

            no_improve = 0

def binarize(h):
    """Create binary vector with sum = n/2 from ranking of h."""
    n = len(h)
    h_bin = np.zeros(n)
    target_ones = n // 2
    if n % 2 == 1:
        target_ones = (n - 1) // 2
        h_bin.copy()
        idx = np.argsort(-h, kind='stable')
        h_bin[idx[:target_ones]] = 1.0
        h_bin[idx[target_ones]] = 0.5
    else:
        idx = np.argsort(-h, kind='stable')
        h_bin[idx[:target_ones]] = 1.0
    return h_bin

def resample(h, n_new):
    """Resample to new resolution."""
    n_old = len(h)
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return make_feasible(h_new)

def update_best(state, h_cand):

        state['c'] = c
