import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate

def score_h(h):
    """Compute C5 = max(correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_box_mass(v, target=None):
    """Project v onto {0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.clip(v, 0, 1).astype(float)
    # Bisection on shift
    # phi(t) = sum(clip(v_i + t, 0, 1))
    # We want phi(t) = target
    
    def phi(t):
        return np.sum(np.clip(v + t, 0, 1))
    
    lo, hi = -1.0, 1.0
    # Ensure bracketing
    while phi(lo) > target:
        lo -= 1.0
    while phi(hi) < target:
        hi += 1.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if phi(mid) < target:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2.0
    result = np.clip(v + t, 0, 1)
    
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute residual among non-saturated coordinates
        mask = (result > 1e-8) & (result < 1 - 1e-8)
        if np.any(mask):
            adj = residual / np.sum(mask)
            result[mask] = np.clip(result[mask] + adj, 0, 1)
        # If still residual, add/subtract from boundary elements
        residual = target - np.sum(result)
        if abs(residual) > 1e-10:
            if residual > 0:
                # Need to increase: use coordinates closest to 1
                order = np.argsort(-result)
                for idx in order:
                    add = min(residual, 1.0 - result[idx])
                    result[idx] += add
                    residual -= add
                    if residual <= 1e-12:
                        break
            else:
                # Need to decrease: use coordinates closest to 0
                order = np.argsort(result)
                for idx in order:
                    sub = min(-residual, result[idx])
                    result[idx] -= sub
                    residual += sub
                    if residual >= -1e-12:
                        break
    
    return result

    """Compute all A_k(h) = sum_i h_i * (1-h_{i+k}) for valid i"""
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    return corr  # shape (2n-1,)

def compute_active_gradients(h, num_active=5):
    """Compute gradients of the top num_active A_k values"""
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    # Find top lags
    top_idx = np.argsort(-corr)[:num_active]
    
    grads = np.zeros_like(h)
    for idx in top_idx:
        # idx corresponds to lag k = idx - (n-1)
        k = idx - (n - 1)
        grad = np.zeros_like(h)
        if k >= 0:
            # A_k = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
            # dA_k/dh[i] = (1-h[i+k]) for i in [0, n-1-k) (first factor)
            #             - h[i-k] for i in [k, n-1] (second factor)
            m = n - k
            grad[:m] += (1 - h[k:])  # first factor contribution
            grad[k:] -= h[:m]         # second factor contribution
        else:
            t = -k
            # A_k = sum_{i=t}^{n-1} h[i] * (1-h[i-k]) = sum_{i=t}^{n-1} h[i] * (1-h[i-t])
            m = n - t
            grad[t:] += (1 - h[:m])   # first factor
            grad[:m] -= h[t:]          # second factor
        grads += grad
    
    return grads, corr

def lp_trust_region_step(h_c, radius, gamma_max):
    """Solve LP trust region to find direction"""
    n = len(h_c)
    corr = correlate(h_c, 1 - h_c, mode='full')
    num_lags = 2 * n - 1
    
    # Build constraint matrix: for each lag k, row = gradient of A_k
    # G has shape (num_lags, n)
    G = np.zeros((num_lags, n))
    for idx in range(num_lags):
        k = idx - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            m = n - k
            grad[:m] += (1 - h_c[k:])
            grad[k:] -= h_c[:m]
        else:
            t = -k
            m = n - t
            grad[t:] += (1 - h_c[:m])
            grad[:m] -= h_c[t:]
        G[idx] = grad
    
    # LP: minimize t
    # subject to: A_k + G_k . (h - h_c) <= t  for all k
    #             sum(h) = n/2
    #             h_c_i - r <= h_i <= h_c_i + r
    #             t <= gamma_max (optional upper bound)
    
    # Variables: [h_0, ..., h_{n-1}, t]  (n+1 variables)
    # Objective: c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G_k . h - t <= A_k - G_k . h_c
    # i.e., G_k . h - t <= corr[k] - G_k . h_c
    A_ub = np.hstack([G, -np.ones((num_lags, 1))])
    b_ub = corr - G @ h_c
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lb = np.maximum(0, h_c - radius)
    ub = np.minimum(1, h_c + radius)
    lb_t = -np.inf
    ub_t = gamma_max
    bounds = list(zip(lb, ub)) + [(lb_t, ub_max)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
        elif res.status == 2:
            # Infeasible, try larger radius
            return None
    except:
        return None
    return None

    lam_min = max(lam_min, -1e-10)
    lam_max = min(lam_max, 1e10)

    # Grid search
    best_lam = lam_min
    best_score = current_score
    best_h = None
    
    num_samples = 100

    for lam in lams:

                if s < best_score - 1e-12:
                    best_score = s
                    best_lam = lam
                    best_h = h_try.copy()
            step /= 5
    
    return best_h, best_score

def subgradient_step(h, step_size, num_active=10):
    """Subgradient descent step"""
    n = len(h)
    grads, corr = compute_active_gradients(h, num_active)
    
    # Average gradient of active constraints
    num_active_actual = min(num_active, len(corr))
    # Weight by how close to max each active constraint is
    weights = np.zeros(num_active)
    max_val = corr.max()
    active_idx = np.argsort(-corr)[:num_active_actual]
    for i, idx in enumerate(active_idx):
        weights[i] = max(0, (max_val - corr[idx] + max_val) / max_val)
    
    avg_grad = np.zeros(n)
    for i, idx in enumerate(active_idx):
        k = idx - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            m = n - k
            grad[:m] += (1 - h[k:])
            grad[k:] -= h[:m]
        else:
            t = -k
            m = n - t
            grad[t:] += (1 - h[:m])
            grad[:m] -= h[t:]
        avg_grad += grad / num_active_actual
    
    # Descent direction: -gradient, made zero-sum
    d = -avg_grad
    d -= np.mean(d)
    
    # Scale
    norm = np.linalg.norm(d)
    if norm > 1e-12:
        d = d / norm * step_size
    
    return d

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_mass(v, target=None):

        return float(np.sum(np.clip(v + t, 0.0, 1.0)))

            lo = mid
        else:

def get_gradient_for_lag(h, k):
    """Gradient of A_k with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        grad[:m] += (1.0 - h[k:])
        grad[k:] -= h[:m]
    else:
        t = -k
        m = n - t
        grad[t:] += (1.0 - h[:m])
        grad[:m] -= h[t:]
    return grad

def full_correlation(h):
    """Compute all A_k values"""
    return correlate(h, 1 - h, mode='full')

def subgradient_direction(h, num_active=10):
    """Compute subgradient direction (negative, zero-sum)"""
    n = len(h)
    corr = full_correlation(h)
    
    # Get top lags
    top_idx = np.argsort(-corr)[:num_active]
    
    # Weighted average gradient
    avg_grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        avg_grad += get_gradient_for_lag(h, k)
    avg_grad /= len(top_idx)
    
    # Negative direction, zero-sum
    d = -avg_grad
    d -= np.mean(d)
    
    norm = np.linalg.norm(d)
    if norm > 1e-15:
        d /= norm
    return d

def lp_direction(h_c, radius):
    """LP trust region direction"""
    n = len(h_c)
    corr = full_correlation(h_c)
    num_lags = 2 * n - 1
    
    # Build gradient matrix
    G = np.zeros((num_lags, n))
    for idx in range(num_lags):
        k = idx - (n - 1)
        G[idx] = get_gradient_for_lag(h_c, k)
    
    # LP: min t s.t. G_k . h - t <= corr_k - G_k . h_c, sum(h) = n/2, bounds
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.hstack([G, -np.ones((num_lags, 1))])
    b_ub = corr - G @ h_c
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lb, ub))
    bounds.append((-np.inf, np.inf))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
