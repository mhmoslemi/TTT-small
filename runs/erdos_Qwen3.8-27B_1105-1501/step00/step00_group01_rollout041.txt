def evaluate_c5(h):
    """Compute C5 value for a given h array."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def compute_all_lag_values(h):
    """Compute all lag values (scaled by dx)."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return corr * dx

def compute_lag_gradient(h, lag_index):
    """Compute gradient of a specific lag value with respect to h.
    
    lag_index is the index in the full correlation (0 to 2n-2).
    d = lag_index - (n-1)
    """
    n = len(h)
    dx = 2.0 / n
    d = lag_index - (n - 1)
    grad = np.zeros(n)
    
    if d >= 0:
        L = n - d
        # h[k] in first factor: d <= k <= n-1
        # (1-h[k-d]) * dx
        mask1 = slice(d, n)
        grad[d:] += dx * (1 - h[:L])
        # h[k] in second factor: 0 <= k <= n-d-1
        # -h[k+d] * dx
        mask2 = slice(0, n-d)
        grad[:n-d] -= dx * h[d:]
    else:
        e = -d
        L = n - e
        # h[k] in first factor: 0 <= k <= n-e-1
        # (1-h[k+e]) * dx
        grad[:L] += dx * (1 - h[e:])
        # h[k] in second factor: e <= k <= n-1
        # -h[k-e] * dx
        grad[e:] -= dx * h[:L]
    
    return grad

def project_feasible_direction(h, v, tol=1e-10):

    We want sum(v) = 0 and the direction to be such that we can step in it.
    """
    n = len(h)

    v = v - np.mean(v)

    # We need: for v_i > 0: h_i + t*v_i <= 1 => t <= (1-h_i)/v_i
    #           for v_i < 0: h_i + t*v_i >= 0 => t <= -h_i/v_i = h_i/(-v_i)

    Returns (new_h, new_c5, step_size).

def trust_region_lp(h, delta=0.1):
    """Solve the trust-region LP to find a direction that reduces max lag."""
    n = len(h)
    n_lags = 2 * n - 1
    lag_vals = compute_lag_values(h)
    
    # Variables: [d_0, ..., d_{n-1}, M]
    # Minimize M (index n in the variable vector)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A @ x <= b
    # For each lag p: grad_p · d - M <= -lag_vals[p]
    # i.e., -grad_p · d + M >= lag_vals[p]
    # i.e., grad_p · d - M <= -lag_vals[p]
    
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for p in range(n_lags):
        g = compute_lag_gradient(h, p)
        A_ub[p, :n] = g
        A_ub[p, n] = -1.0
        b_ub[p] = -lag_vals[p]
    
    # Trust region: ||d||_inf <= delta
    # d_i <= delta and -d_i <= delta
    A_trust = np.zeros((2 * n, n + 1))
    b_trust = np.zeros(2 * n)
    for i in range(n):
        A_trust[2*i, i] = 1.0
        A_trust[2*i, n] = 0.0
        b_trust[2*i] = delta
        A_trust[2*i+1, i] = -1.0
        A_trust[2*i+1, n] = 0.0
        b_trust[2*i+1] = delta
    
    # Combine
    A_all = np.vstack([A_ub, A_trust])
    b_all = np.concatenate([b_ub, b_trust])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    
    # Bounds: h_i - 1 <= h_i + d_i <= h_i + 1 (i.e., -1+d_i <= ... just -1 <= d_i - h_i ... )
    # Actually: 0 <= h_i + d_i <= 1 means -h_i <= d_i <= 1-h_i
    lb = -h  # d_i >= -h_i
    ub = 1.0 - h  # d_i <= 1 - h_i
    lb = np.append(lb, -np.inf)  # M unbounded below (in practice, M >= 0)
    ub = np.append(ub, np.inf)
    
    # Solve
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_all, b_ub=b_all, A_eq=A_eq, b_eq=b_eq, 
                     bounds=list(zip(lb, ub)), method='highs')
    
    if result.success:
        d = result.x[:n]
        M_pred = result.x[n]
        return d, M_pred
    
    return None, None

import numpy as np
import time
from scipy.optimize import linprog

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_lag_values(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_all_gradients(h):
    """Compute gradients of all lags. Returns array of shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    for p in range(n_lags):
        d = p - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                grads[p, d:] += dx * (1.0 - h[:L])
                grads[p, :L] -= dx * h[d:]
        else:
            e = -d
            L = n - e
            if L > 0:
                grads[p, :L] += dx * (1.0 - h[e:])
                grads[p, e:] -= dx * h[:L]
    return grads

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def line_search_c5(h, d, c5_max, max_iter=60):
    """Find best step along direction d that keeps C5 < c5_max."""
    n = len(h)
    # Max feasible step
    pos = d > 1e-14
    neg = d < -1e-14
    t_max = 1e10
    if np.any(pos):
        t_max = min(t_max, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_max = min(t_max, np.min(-h[neg] / d[neg]))
    t_max = min(t_max, 1.0)
    if t_max < 1e-14:
        return h, c5_max, 0.0
    
    best_h = h
    best_c5 = c5_max
    best_t = 0.0
    t = t_max
    for _ in range(max_iter):
        if t < 1e-14:
            break
        new_h = np.clip(h + t * d, 0.0, 1.0)
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            best_t = t
        t *= 0.8
    return best_h, best_c5, best_t

def solve_direction_lp(h, delta):
    """Solve trust-region LP for direction."""
    n = len(h)
    n_lags = 2 * n - 1
    lag_vals = compute_lag_values(h)
    grads = compute_all_gradients(h)
    
    # Variables: [d_0..d_{n-1}, M]
    # Minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: grads @ d - M <= -lag_vals  (i.e., lag_vals + grads@d <= M)
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -lag_vals
    
    # Trust region: |d_i| <= delta
    A_t = np.zeros((2 * n, n + 1))
    b_t = np.full(2 * n, delta)
    A_t[0::2, :n] = np.eye(n)
    A_t[1::2, :n] = -np.eye(n)
    
    A_all = np.vstack([A_ub, A_t])
    b_all = np.concatenate([b_ub, b_t])
    
    # Sum constraint
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds
    lb = -h
    ub = 1.0 - h
    lb_full = np.append(lb, -10.0)
    ub_full = np.append(ub, 10.0)
    
    res = linprog(c, A_ub=A_all, b_ub=b_all, A_eq=A_eq, b_eq=[0.0],
                  bounds=list(zip(lb_full, ub_full)), method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

def lp_descent(h, max_iters=100, delta_init=0.05):
    """Trust-region LP-based descent."""
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    delta = delta_init
    stall = 0
    
    for it in range(max_iters):
        d, M_pred = solve_direction_lp(h, delta)
        if d is None:
            # Fallback to simple gradient
            lag_vals = compute_lag_values(h)
            active = np.where(lag_vals >= np.max(lag_vals) - 0.01)[0]
            grads = compute_all_gradients(h)
            grad = grads[active].sum(axis=0)
            d = -grad - np.mean(-grad)
        
        new_h, new_c5, t = line_search_c5(h, d, best_c5)
        
        if t > 1e-10 and new_c5 < best_c5 - 1e-12:
            h = new_h
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
            delta = min(delta * 1.5, 0.5)
        else:
            stall += 1
            delta *= 0.5
            if stall > 20 or delta < 1e-5:
                break
    
    return best_h, best_c5

def perturb_h(h, scale=0.1, n_pairs=3):

import numpy as np

from scipy.optimize import linprog

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_lag_values(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_all_gradients(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    for p in range(n_lags):
        d = p - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                grads[p, d:] += dx * (1.0 - h[:L])
                grads[p, :L] -= dx * h[d:]
        else:
            e = -d
            L = n - e
            if L > 0:
                grads[p, :L] += dx * (1.0 - h[e:])
                grads[p, e:] -= dx * h[:L]
    return grads

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    t_val = 1e10
