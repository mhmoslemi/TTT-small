def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def project_to_box_sum(h, target_sum):
    t_lo = -2.0
    t_hi = 2.0
    for _ in range(80):
        t_mid = (t_lo + t_hi) / 2.0
        s = np.sum(np.clip(h - t_mid, 0.0, 1.0))
        if s > target_sum:
            t_lo = t_mid
        else:
            t_hi = t_mid
    t = (t_lo + t_hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def optimize_coordinate_fast(h, k, dx):

    h_save = h[k]
    h[k] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode="full") * dx
    h[k] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode="full") * dx
    h[k] = h_save

    b = corr0

        vals = a * mid + b
        max_val = np.max(vals)

        g_hi = np.max(a[active])
        if g_lo > 1e-14:
            hi = mid
        elif g_hi < -1e-14:
            lo = mid
        else:

        else:

            h[j] = 0.0
