    n = len(h)

    for r in range(2*n-1):

    """Gradient of F_k(h) = sum_{i valid} h[i]*(1-h[i+k])"""

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def all_correlation_grads(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if k == 0:
            grads[r] = 1.0 - 2.0 * h
        else:
            grads[r, i_start:i_end] += (1.0 - h[i_start + k:i_end + k])
            grads[r, i_start + k:i_end + k] += (-h[i_start:i_end])
    return grads

def smooth_obj_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    my_F = corr[::-1]
    max_c = np.max(my_F)
    shifted = beta * (my_F - max_c)
    max_s = np.max(shifted)
    exps = np.exp(shifted - max_s)
    log_sum = max_s + np.log(np.sum(exps))
    val = max_c + log_sum / beta
    w = exps / np.sum(exps)
    grads = all_correlation_grads(h)
    grad = w @ grads
    return val, grad

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def local_optimize(h0, max_time=60):
    n = len(h0)
    h = h0.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    start = time.time()
    beta = 50.0
    no_improve = 0
    alpha_curr = 0.02

    while time.time() - start < max_time:
        val, grad = smooth_obj_and_grad(h, beta)
        grad_sz = grad - np.mean(grad)

        if np.max(np.abs(grad_sz)) < 1e-14:
            break

        alpha = alpha_curr
        found = False
        for _ in range(60):
            h_new = h - alpha * grad_sz
            h_new = project_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5 - 1e-17:
                h = h_new
                c5 = c5_new
                alpha_curr = alpha
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                found = True
                break
            alpha *= 0.5

        if not found:
            beta = min(beta * 1.15, 500.0)
            no_improve += 1
            alpha_curr = max(alpha, 1e-8)

        if no_improve > 300:
            break

    return best_h, best_c5, n

def polish(h0, max_time=30):
    """Very fine local search with tiny steps."""
    n = len(h0)
    h = h0.copy()
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    start = time.time()

    for _ in range(20000):
        if time.time() - start > max_time:
            break
        beta = 200.0
        val, grad = smooth_obj_and_grad(h, beta)
        grad_sz = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_sz)
        if gnorm < 1e-13:
            break
        alpha = 0.005
        found = False
        for _ in range(40):
            h_new = h - alpha * grad_sz
            h_new = project_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5 - 1e-17:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                found = True
                break
            alpha *= 0.5
        if not found:
            break

    return best_h, best_c5, n

def make_candidate(h_base, n_target, rng):
    """Interpolate or create candidate for a different n."""
    n_src = len(h_base)
    if n_src == n_target:
        return h_base.copy()
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_target, endpoint=False)
    h_ext = np.append(h_base, h_base[0])
    h_interp = np.interp(x_new, x_old, h_ext)
    return project_feasible(h_interp, n_target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)

    best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 2: Polish the best
    h, c5, n = polish(best_h, max_time=60)
    if c5 < best_c5:
        best_h, best_c5, best_n = h.copy(), c5, n

    elapsed = time.time() - t0

    # Phase 3: Perturbed restarts
    remaining = budget_s - elapsed - 20
    if remaining > 40:
        scales = [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]
        n_trials = min(int(remaining / 35), 18)
        for i in range(n_trials):
            if time.time() - t0 > budget_s - 15:
                break
            sc = scales[i % len(scales)]
            pert = rng.randn(len(best_h)) * sc
            h_pert = project_feasible(best_h + pert, len(best_h))
            h, c5, n = local_optimize(h_pert, max_time=25)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n
            h, c5, n = polish(best_h, max_time=10)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 4: Try different n values
    remaining = budget_s - (time.time() - t0) - 10
    if remaining > 30:
        for n_try in [44, 46, 48, 50, 40, 52, 56, 60]:
            if time.time() - t0 > budget_s - 8:
                break
            h_cand = make_candidate(h_init, n_try, rng)
            h, c5, n = local_optimize(h_cand, max_time=25)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n
                h, c5, n = polish(best_h, max_time=10)
                if c5 < best_c5:
                    best_h, best_c5, best_n = h.copy(), c5, n

    # Final: ensure exact sum constraint
    n_final = len(best_h)
    target = n_final / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-15:
        best_h += diff / n_final
        best_h = np.clip(best_h, 0.0, 1.0)

    c5_final = evaluate_c5(best_h)
    return best_h, c5_final, n_final
