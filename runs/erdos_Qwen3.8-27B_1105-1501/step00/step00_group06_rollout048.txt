import time

def score_h(h):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_mass(v):
    """Project v onto {0 <= h <= 1, sum(h) = n/2}"""
    n = len(v)
    target = n / 2.0
    v = np.array(v, dtype=float)
    
    if not np.all(np.isfinite(v)):
        v = np.clip(v, 0, 1)
    
    # Bisection on shift t: sum(clip(v + t, 0, 1)) = target
    lo = -1.0 - max(v.max() if n > 0 else 1.0, 1.0)
    hi = 1.0 - min(v.min() if n > 0 else 0.0, 0.0) + 1.0
    
    # Actually simpler: lo = -max(v) ensures sum=0, hi = 1-min(v) ensures sum=n
    lo = -max(v).item() - 0.1
    hi = 1.0 - min(v).item() + 0.1
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual with iterative adjustment
    for _ in range(50):
        residual = target - np.sum(result)
        if abs(residual) < 1e-12:

        if residual > 0:
            mask = result < 1.0 - 1e-15

        if not np.any(mask):
            break
        adj = residual / np.sum(mask)
        result[mask] = np.clip(result[mask] + adj, 0.0, 1.0)
    
    return result

def compute_A_and_grad(h):
    """Compute all overlap values A_k and their gradients.
    Returns A (array of length 2n-1, indexed as k + n - 1) and G (matrix (2n-1, n)).
    """
    n = len(h)
    nK = 2 * n - 1
    A = np.zeros(nK)
    G = np.zeros((nK, n))
    
    # k ranges from -(n-1) to n-1, stored at index k + n - 1
    for k in range(-(n-1), n):
        idx = k + n - 1
        if k >= 0:
            # c[k] = sum_{i=0}^{n-1-k} h[i+k] * (1 - h[i])
            m = n - k
            A[idx] = np.sum(h[k:] * (1.0 - h[:m]))
            # grad: for j in k..n-1: += (1 - h[j-k]) ; for j in 0..n-1-k: -= h[j+k]
            G[idx, k:] += (1.0 - h[:m])
            G[idx, :m] -= h[k:]
        elif k == 0:
            # c[0] = sum_i h[i] * (1 - h[i])
            A[idx] = np.sum(h * (1.0 - h))
            G[idx, :] = 1.0 - 2.0 * h
        else:
            # k < 0, t = -k
            t = -k
            m = n - t
            # c[k] = sum_{i=0}^{n-1-t} h[i] * (1 - h[i+t])
            A[idx] = np.sum(h[:m] * (1.0 - h[t:]))
            # grad: for j in 0..n-1-t: += (1 - h[j+t]) ; for j in t..n-1: -= h[j-t]
            G[idx, :m] += (1.0 - h[t:])
            G[idx, t:] -= h[:m]
    
    return A, G

def solve_lp_direction(h_c, A, G, radius):
    """Solve LP to find direction that reduces linearized overlaps.
    Minimize t s.t. A_k + G_k @ (h - h_c) <= t, sum(h) = n/2, box constraints.
    Returns h_new or None.
    """
    n = len(h_c)
    
    # Variables: h (n vars) + t (1 var) = n+1 vars
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: G_k @ h + (-1)*t <= A_k - G_k @ h_c
    # i.e., G @ h - t <= A - G @ h_c
    # In linprog form: A_ub @ x <= b_ub
    nK = len(A)
    A_ub = np.hstack([G, -np.ones((nK, 1))])
    b_ub = A - G @ h_c
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: h_c[i] - radius <= h[i] <= h_c[i] + radius, and 0 <= h[i] <= 1
    lower = np.maximum(0.0, h_c - radius)
    upper = np.minimum(1.0, h_c + radius)
    lower_t = 0.0
    upper_t = 1e6  # no upper bound on t really
    
    bounds = list(zip(lower, upper)) + [(lower_t, upper_t)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        # Verify feasibility
        h_new = project_box_mass(h_new)
        return h_new
    return None

def make_random_direction(n, rng):

    d = rng.standard_normal(n)
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm > 1e-12:

    return d

def make_harmonic_direction(n, rng):

    if norm > 1e-12:

    return d

    return np.correlate(h, 1.0 - h, mode='full')

def compute_gradients(h):
    """Compute gradient matrix G where G[k_idx] = dA_k/dh."""
    n = len(h)
    nK = 2 * n - 1
    G = np.zeros((nK, n))
    
    for k in range(-(n-1), n):
        idx = k + n - 1
        if k >= 0:
            m = n - k
            G[idx, k:] += (1.0 - h[:m])
            G[idx, :m] -= h[k:]
        elif k == 0:
            G[idx, :] = 1.0 - 2.0 * h
        else:
            t = -k
            m = n - t
            G[idx, :m] += (1.0 - h[t:])
            G[idx, t:] -= h[:m]
    
    return G

    c_obj[-1] = 1.0

    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = A - G @ h_c
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lower = np.maximum(0.0, h_c - radius)
    upper = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lower, upper)) + [(0.0, 1e6)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except:
        pass
    return None

def line_search(h_c, d, best_score):

    n = len(h_c)

        return h_c, best_score
    
    eps = 1e-10
    pos = d > eps
    neg = d < -eps
    
    # Forward: h_c + lam*d, lam > 0
    lam_f_max = 1e10
    if np.any(pos):

    if np.any(neg):

    best_s = best_score
    
    # Forward search
    if lam_f_max > 1e-8:
        lam_f_max = min(lam_f_max, 5.0)
        n_pts = 50
        lams = np.linspace(0.0, lam_f_max, n_pts)
        best_local_s = best_s
        best_local_lam = 0.0
        prev_s = best_s

            h_try = h_c + lam * d

            if s < best_local_s:

                best_local_lam = lam
            prev_s = s

            best_s = best_local_s
            best_h = project_box_mass(h_c + best_local_lam * d)
            
            # Refine

            hi = lams[min(n_pts-1, idx+2)]

                if s1 < best_s:
                    best_s = s1
                    best_h = h_m1

                    best_s = s2

    if lam_b_max > 1e-8:

            s = score_h(h_try)
            if s < best_s:
                best_s = s
                best_h = h_try

def softmax_smooth_score(h, alpha):
    """Smooth approximation of max overlap."""
    A = np.correlate(h, 1.0 - h, mode='full')
    # Shift for numerical stability
    A_max = np.max(A)
    if alpha * A_max > 500:
        return A_max  # overflow protection
    log_sum = np.log(np.sum(np.exp(alpha * (A - A_max)))) / alpha + A_max
    return log_sum * (2.0 / len(h))

def softmax_smooth_grad(h, alpha):
    """Gradient of smooth score."""
    n = len(h)
    G = compute_gradients(h)
    A = np.correlate(h, 1.0 - h, mode='full')
    A_max = np.max(A)
    if alpha * A_max > 500:
        weights = np.zeros_like(A)
        weights[np.argmax(A)] = 1.0
    else:
        weights = np.exp(alpha * (A - A_max))
        weights /= np.sum(weights)
    grad = weights @ G
    return grad * (2.0 / n)

def make_perturbed_dir(n, rng, h_c):
    d = rng.uniform(-1, 1, n)
    d[h_c > 0.95] *= 0.1
    d[h_c < 0.05] *= 0.1
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm > 1e-14:
        d /= norm
    return d

def smooth_obj_and_grad(h, alpha):
    A = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    A_max = float(np.max(A))
    shifted = alpha * (A - A_max)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    smooth_val = (A_max + log_sum / alpha) * (2.0 / n)
    weights = exp_vals / np.sum(exp_vals)
    G = compute_gradients(h)
    grad = (weights @ G) * (2.0 / n)
    return smooth_val, grad

def smooth_obj_wrapper(x, alpha_holder):
    val, _ = smooth_obj_and_grad(x, alpha_holder[0])
    return val

def smooth_grad_wrapper(x, alpha_holder):
    _, grad = smooth_obj_and_grad(x, alpha_holder[0])
    return grad

def coordinate_polish(h, n_iter, best_score):
    n = len(h)
    for _ in range(n_iter):
        improved = False
        idx = np.random.randint(0, n)
        d = np.zeros(n)
        if h[idx] > 0.01:
            j = np.random.randint(0, n)
            while j == idx:
                j = np.random.randint(0, n)
            d[idx] = 1.0
            d[j] = -1.0
        elif h[idx] < 0.99:
            j = np.random.randint(0, n)
            while j == idx:
                j = np.random.randint(0, n)
            d[idx] = -1.0
            d[j] = 1.0
        else:
            continue
        h_new, s_new = line_search(h, d, best_score)
        if s_new < best_score - 1e-14:
            h = h_new
            best_score = s_new
            improved = True
    return h, best_score

def resample_to_n(h_from, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(x_new, x_old, h_from)
    return project_box_mass(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(70, 0.5)

    h_best = project_box_mass(h_best)
    best_score = score_h(h_best)
    n = len(h_best)

    radius = 0.12
    iter_count = 0
    stall_count = 0
    alpha_holder = [50.0]

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 15:
            break

        # 1. LP direction
        A = compute_A(h_best)
        G = compute_gradients(h_best)
        h_lp = solve_lp_direction(h_best, A, G, radius)
        if h_lp is not None:
            d = h_lp - h_best
            d -= np.mean(d)
            if np.linalg.norm(d) > 1e-10:
                h_new, s_new = line_search(h_best, d, best_score)
                if s_new < best_score - 1e-14:
                    best_score = s_new
                    h_best = h_new
                    radius = min(radius * 1.08, 0.5)
                    stall_count = 0
                else:
                    radius = max(radius * 0.85, 0.01)
                    stall_count += 1
            else:
                radius = min(radius * 1.3, 0.5)

        # 2. Random directions
        n_rand = 4 if elapsed < budget_s * 0.6 else 2
        for _ in range(n_rand):
            elapsed = time.time() - start_time
            if elapsed > budget_s - 15:
                break
            d = make_random_dir(n, rng)
            h_new, s_new = line_search(h_best, d, best_score)
            if s_new < best_score - 1e-14:
                best_score = s_new
                h_best = h_new
                stall_count = 0
            else:
                stall_count += 1

        # 3. Harmonic directions
        for _ in range(2):
            elapsed = time.time() - start_time
            if elapsed > budget_s - 15:
                break
            d = make_harmonic_dir(n, rng)
            h_new, s_new = line_search(h_best, d, best_score)
            if s_new < best_score - 1e-14:
                best_score = s_new
                h_best = h_new
                stall_count = 0

        # 4. Active gradient direction
        A = compute_A(h_best)
        A_max = float(np.max(A))
        active = np.where(A > A_max - 0.1)[0]
        if len(active) > 0:
            G = compute_gradients(h_best)
            grad_avg = np.mean(G[active], axis=0)
            d = -grad_avg
            d -= np.mean(d)
            norm = np.linalg.norm(d)
            if norm > 1e-12:
                d /= norm
                h_new, s_new = line_search(h_best, d, best_score)
                if s_new < best_score - 1e-14:
                    best_score = s_new
                    h_best = h_new
                    stall_count = 0

        # 5. Perturbed direction (biased toward free coordinates)
        elapsed = time.time() - start_time
        if elapsed < budget_s - 15:
            d = make_perturbed_dir(n, rng, h_best)
            h_new, s_new = line_search(h_best, d, best_score)
            if s_new < best_score - 1e-14:
                best_score = s_new
                h_best = h_new
                stall_count = 0

        # 6. Every 20 iterations: smooth L-BFGS-B
        if iter_count % 20 == 10:
            elapsed = time.time() - start_time
            if elapsed < budget_s - 80:
                try:
                    res = minimize(smooth_obj_wrapper, h_best, args=(alpha_holder,),
                                  jac=smooth_grad_wrapper, method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * n,
                                  options={'maxiter': 80, 'ftol': 1e-14, 'gtol': 1e-10})
                    h_try = project_box_mass(res.x)
                    s_try = score_h(h_try)
                    if s_try < best_score - 1e-12:
                        best_score = s_try
                        h_best = h_try
                        stall_count = 0
                except Exception:
                    pass

        # 7. Every 40 iterations: coordinate polish
        if iter_count % 40 == 20:
            elapsed = time.time() - start_time
            if elapsed < budget_s - 40:
                h_best, best_score = coordinate_polish(h_best, 30, best_score)

        # 8. Every 60 iterations: try different n
        if iter_count % 60 == 50:
            elapsed = time.time() - start_time
            if elapsed < budget_s - 120:
                for n_try in [50, 60, 84, 90, 100, 64, 120]:
                    elapsed = time.time() - start_time
                    if elapsed > budget_s - 120:
                        break
                    h_cand = resample_to_n(h_best, n, n_try)
                    s_cand = score_h(h_cand)
                    for _ in range(15):
                        elapsed = time.time() - start_time
                        if elapsed > budget_s - 120:
                            break
                        d = make_random_dir(n_try, rng)
                        h_t, s_t = line_search(h_cand, d, s_cand)
                        if s_t < s_cand - 1e-14:
                            s_cand = s_t
                            h_cand = h_t
                    if s_cand < best_score - 1e-8:
                        best_score = s_cand
                        h_best = h_cand
                        n = n_try
                        radius = 0.12

        # 9. Restart if stuck
        if stall_count > 25:
            d = make_random_dir(n, rng) * 0.05
            h_pert = project_box_mass(h_best + d)
            s_pert = score_h(h_pert)
            if s_pert < best_score + 0.01:
                h_best = h_pert
                best_score = s_pert
                stall_count = 0
                radius = 0.08

        iter_count += 1

    # Final polish
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        h_best, best_score = coordinate_polish(h_best, 50, best_score)

    h_best = project_box_mass(h_best)
    best_score = score_h(h_best)
    return h_best, best_score, len(h_best)
