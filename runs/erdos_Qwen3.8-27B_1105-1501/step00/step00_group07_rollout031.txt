def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return np.max(raw) * dx, raw

def project_feas(v):
    n = len(v)
    target = n / 2.0
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    # Water-filling
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    v = np.clip(v + lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(v)
    if abs(residual) > 1e-10:
        if residual > 0:
            # Need to add mass; increase entries below 1
            avail = 1.0 - v
            order = np.argsort(-avail)
            for i in order:
                add = min(avail[i], residual)
                v[i] += add
                residual -= add
                if residual <= 1e-10:
                    break
        else:
            # Need to remove mass; decrease entries above 0
            order = np.argsort(-v)
            for i in order:
                rem = min(v[i], -residual)
                v[i] -= rem
                residual += rem
                if residual >= -1e-10:
                    break
    v = np.clip(v, 0.0, 1.0)
    return v

def resample(h, target_n):

    v = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feas(v)
