    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def slsqp_optimize(h0, max_iter=500, time_limit=None, start_time=None):
    """Optimize using SLSQP with constrained formulation."""
    n = len(h0)
    
    def obj(x):
        return make_objective(x, n)
    
    def obj_grad(x):
        # Subgradient of max: use the active constraint
        corr = np.correlate(x, 1 - x, mode='full')
        active_idx = np.argmax(corr)
        lag = active_idx - (n - 1)
        
        grad = np.zeros(n)
        if lag >= 0:
            m = n - lag
            grad[:m] += (1 - x[lag:])
            grad[lag:] -= x[:m]
        else:
            m = -lag
            grad[m:] += (1 - x[:n-m])
            grad[:n-m] -= x[m:]
        
        return grad * (2.0 / n)
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2, 'jac': lambda x: np.ones(n)},

    bounds = [(0, 1)] * n

        # Fix mass

from scipy.optimize import linprog, minimize
import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_mass(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, -5, 5)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(v + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) / 2.0, 0, 1)
    res = target - np.sum(y)
    if abs(res) > 1e-10:
        for _ in range(50):
            if abs(res) < 1e-12:
                break
            movable = (y > 1e-9) & (y < 1 - 1e-9)
            if not np.any(movable):
                break
            step = np.clip(res / np.sum(movable), -0.5, 0.5)
            y[movable] = np.clip(y[movable] + step, 0, 1)
            res = target - np.sum(y)
    return np.clip(y, 0, 1)

def compute_gradients(h):
    """Compute matrix G where G[k, i] = dA_k/dh_i."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        lag = idx - (n - 1)
        if lag >= 0:
            m = n - lag
            G[idx, :m] += (1 - h[lag:])
            G[idx, lag:] -= h[:m]
        else:
            m = -lag
            G[idx, m:] += (1 - h[:n - m])
            G[idx, :n - m] -= h[m:]
    return G

def lp_step(h_c, radius):
    """Solve LP trust region. Returns direction or None."""
    n = len(h_c)
    A_c = np.correlate(h_c, 1 - h_c, mode='full')
    G = compute_gradients(h_c)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = A_c - G @ h_c
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lb, ub)) + [(-10, 10)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            d = res.x[:n] - h_c
            d -= np.mean(d)  # make zero-sum
            if np.max(np.abs(d)) > 1e-8:
                return d
    except:
        pass
    return None

def exact_line_search(h_c, d, lam_max=1.0, n_grid=100):
    """Exact line search along h_c + lambda * d."""
    n = len(h_c)
    s_c = score_h(h_c)
    
    # Find feasible range
    eps = 1e-12
    lo, hi = 0.0, lam_max
    for i in range(n):
        if d[i] > eps:
            lo = max(lo, -h_c[i] / d[i])
            hi = min(hi, (1 - h_c[i]) / d[i])
        elif d[i] < -eps:
            lo = max(lo, (1 - h_c[i]) / d[i])
            hi = min(hi, -h_c[i] / d[i])
    
    lo = max(lo, -lam_max) + 1e-10
    hi = min(hi, lam_max) - 1e-10
    if hi <= lo:
        return h_c.copy(), s_c
    
    # Grid search
    lams = np.linspace(lo, hi, n_grid)
    best_h = h_c.copy()
    best_s = s_c
    best_lam = 0.0
    
    for lam in lams:
        h_try = h_c + lam * d
        if np.any(h_try < -1e-6) or np.any(h_try > 1 + 1e-6):
            continue
        h_try = np.clip(h_try, 0, 1)
        s = score_h(h_try)
        if s < best_s:
            best_s = s
            best_h = h_try.copy()
            best_lam = lam
    
    # Refine
    if best_lam != 0.0 and best_s < s_c:
        # Narrow the search around best_lam
        step = (hi - lo) / n_grid
        lo2 = max(lo, best_lam - 3 * step)
        hi2 = min(hi, best_lam + 3 * step)
        lams2 = np.linspace(lo2, hi2, 50)
        for lam in lams2:
            h_try = np.clip(h_c + lam * d, 0, 1)
            s = score_h(h_try)
            if s < best_s:
                best_s = s
                best_h = h_try.copy()
    
    return best_h, best_s

def random_direction(n, rng):
    """Generate a random zero-sum direction."""
    d = rng.randn(n)
    d -= np.mean(d)
    return d / (np.max(np.abs(d)) + 1e-12)

def slsqp_optimize(h0, max_iter=100):

    n = len(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        if lag >= 0:
            m = n - lag
            def fun(x, lag=lag, m=m):
                h = x[:n]
                return float(np.sum(h[:m] * (1 - h[lag:]))) - x[-1]
            def jac(x, lag=lag, m=m):
                h = x[:n]
                g = np.zeros(n + 1)
                g[-1] = -1.0
                g[:m] += (1 - h[lag:])
                g[lag:] -= h[:m]
                return g
        else:
            m = -lag
            def fun(x, lag=lag, m=m):
                h = x[:n]
                return float(np.sum(h[m:] * (1 - h[:n-m]))) - x[-1]
            def jac(x, lag=lag, m=m):
                h = x[:n]
                g = np.zeros(n + 1)
                g[-1] = -1.0
                g[m:] += (1 - h[:n-m])
                g[:n-m] -= h[m:]
                return g
        constraints.append({'type': 'ineq', 'fun': fun, 'jac': jac})
    
    # Equality: sum(h) = n/2

                        'fun': lambda x: np.sum(x[:n]) - n/2,
                        'jac': lambda x: np.concatenate([np.ones(n), [0]])})
    
    bounds = [(0, 1)] * n + [(0, 2)]
    h0_ext = np.concatenate([h0, [score_h(h0) + 0.01]])

        h_res = np.clip(res.x[:n], 0, 1)
        h_res = project_box_mass(h_res)
        s = score_h(h_res)
        if s < score_h(h0):

    except:
        pass
