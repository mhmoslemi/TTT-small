import numpy as np
import time

def project_feasible(v, target_sum=None):
    """Project v onto {x : 0 ≤ x_i ≤ 1, sum(x) = target_sum}"""
    v = np.asarray(v, dtype=float)
    n = len(v)
    if target_sum is None:
        target_sum = n / 2.0
    
    # Clip first
    v = np.clip(v, 0, 1)
    
    # Check if already feasible
    if abs(np.sum(v) - target_sum) < 1e-12:
        return v
    
    # Bisection for the shift
    def S(lam):
        return np.sum(np.clip(v + lam, 0, 1))
    
    lo, hi = -1.0, 1.0
    # Expand if needed
    for _ in range(20):
        if S(lo) > target_sum and S(hi) < target_sum:
            break
        lo *= 2
        hi *= 2
    
    for _ in range(100):
        mid = (lo + hi) / 2
        if S(mid) > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    
    lam = (lo + hi) / 2
    result = np.clip(v + lam, 0, 1)
    
    # Fix any remaining sum error
    err = np.sum(result) - target_sum
    if abs(err) > 1e-12:
        # Adjust
        if err > 0:
            # Need to decrease
            for i in np.argsort(result)[::-1]:  # Start with largest
                dec = min(result[i], err)
                result[i] -= dec
                err -= dec
                if abs(err) < 1e-12:
                    break
        else:
            # Need to increase
            for i in np.argsort(result):  # Start with smallest
                inc = min(1 - result[i], -err)
                result[i] += inc
                err += inc
                if abs(err) < 1e-12:
                    break
    
    return result

def exact_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_pair_coeff(h, a, b):
    """Compute the coefficient vector for pair transfer (a gains t, b loses t).
    Returns coeff[j] such that new_corr[j] = old_corr[j] + t * coeff[j]"""
    n = len(h)
    m = 2 * n - 1  # length of full correlation
    coeff = np.zeros(m)
    
    # Position a in "h" part: j ∈ [a, a+n-1]
    for j in range(a, a + n):
        idx = a - j + n - 1  # = a - (j - n + 1)
        coeff[j] += (1 - h[idx])
    
    # Position a in "(1-h)" part: j ∈ [n-1-a, 2n-2-a]
    for j in range(n - 1 - a, 2 * n - 1 - a):
        idx = a + j - n + 1  # = a + (j - n + 1)
        coeff[j] -= h[idx]
    
    # Position b in "h" part: j ∈ [b, b+n-1]
    for j in range(b, b + n):
        idx = b - j + n - 1
        coeff[j] -= (1 - h[idx])
    
    # Position b in "(1-h)" part: j ∈ [n-1-b, 2n-2-b]
    for j in range(n - 1 - b, 2 * n - 1 - b):
        idx = b + j - n + 1
        coeff[j] += h[idx]
    
    return coeff

def minimize_pair(h, corr, a, b, coeff, t_low, t_high, n_evals=25):
    """Minimize f(t) = max(corr + t*coeff) over t in [t_low, t_high]"""
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~ 0.618
    
    a_gs = t_low
    b_gs = t_high
    c_gs = b_gs - gr * (b_gs - a_gs)
    d_gs = a_gs + gr * (b_gs - a_gs)
    
    fc = np.max(corr + c_gs * coeff)
    fd = np.max(corr + d_gs * coeff)
    
    for _ in range(n_evals):
        if fc < fd:
            b_gs = d_gs
            d_gs = c_gs
            fd = fc
            c_gs = b_gs - gr * (b_gs - a_gs)
            fc = np.max(corr + c_gs * coeff)
        else:
            a_gs = c_gs
            c_gs = d_gs
            fc = fd
            d_gs = a_gs + gr * (b_gs - a_gs)
            fd = np.max(corr + d_gs * coeff)
    
    t_opt = (a_gs + b_gs) / 2
    score = np.max(corr + t_opt * coeff)
    return t_opt, score
