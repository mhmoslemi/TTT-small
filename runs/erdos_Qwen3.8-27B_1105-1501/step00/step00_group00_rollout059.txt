import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_h(h, n):
    """Project h to satisfy 0 <= h <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    if s != target:
        diff = target - s
        # Adjust proportionally
        if diff > 0:
            # Need to increase: add to elements that are < 1
            mask = h < 1.0
            if np.sum(mask) > 0:
                room = 1.0 - h[mask]
                total_room = np.sum(room)
                if total_room > 0:
                    scale = min(diff / total_room, 1.0)
                    h[mask] += room * scale
                # If still not enough, add uniformly
                if np.sum(h) < target:
                    remaining = target - np.sum(h)
                    h += remaining / n
                    h = np.clip(h, 0, 1)
            # Final adjustment
            if np.sum(h) < target:
                # Add to smallest elements
                order = np.argsort(h)
                for idx in order:
                    if h[idx] < 1.0:
                        add = min(1.0 - h[idx], target - np.sum(h))
                        h[idx] += add
                        if np.sum(h) >= target:
                            break
        else:
            # Need to decrease
            mask = h > 0.0
            if np.sum(mask) > 0:
                room = h[mask]
                total_room = np.sum(room)
                if total_room > 0:
                    scale = min(-diff / total_room, 1.0)
                    h[mask] -= room * scale
                if np.sum(h) > target:
                    remaining = np.sum(h) - target
                    h -= remaining / n
                    h = np.clip(h, 0, 1)
            if np.sum(h) > target:
                order = np.argsort(-h)
                for idx in order:
                    if h[idx] > 0.0:
                        sub = min(h[idx], np.sum(h) - target)
                        h[idx] -= sub
                        if np.sum(h) <= target:
                            break
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Add/subtract from a single element
        h[0] += diff
        h = np.clip(h, 0, 1)
    return h

    """Local search with random perturbations."""
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    best_h = h.copy()

            h = new_h.copy()
        # Adaptive step size
        if i % 500 == 0:
            step_size *= 0.95

    """Gradient of soft-max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    exp_val = np.exp(tau * (corr - m))
    weights = exp_val / np.sum(exp_val)
    
    # Gradient of sum_k weights[k] * corr[k]
    # corr[k] = sum_i h[i] * (1-h[i+k-n+1]) * dx (for valid i)
    # Let lag = k - n + 1
    # corr[k] = dx * sum_{i: 0<=i<n, 0<=i+lag<n} h[i] * (1-h[i+lag])
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        lag = k - n + 1
        if lag >= 0:
            # corr[k] = dx * sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            i_max = n - 1 - lag
            for i in range(i_max + 1):
                # d/dh[i]: (1-h[i+lag])*dx
                grad[i] += weights[k] * (1 - h[i+lag]) * dx
                # d/dh[i+lag]: -h[i]*dx
                grad[i+lag] -= weights[k] * h[i] * dx
        elif lag < 0:
            l = -lag
            # corr[k] = dx * sum_{i=l}^{n-1} h[i]*(1-h[i-l])
            for i in range(l, n):
                grad[i] += weights[k] * (1 - h[i-l]) * dx
                grad[i-l] -= weights[k] * h[i] * dx
        else:
            # lag = 0
            for i in range(n):
                grad[i] += weights[k] * (1 - 2*h[i]) * dx
    
    return grad, m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

        mu = (np.sum(h) - target) / n

        diff = target - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Add/subtract uniformly, then clip
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    return h

    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_constraint(h, n):
    """Project h onto {x : 0 <= x <= 1, sum(x) = n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s == target:
        return h
    diff = target - s
    if diff > 0:
        # Need to increase sum
        # Add to elements with most room
        order = np.argsort(1.0 - h)  # most room first
        remaining = diff
        for idx in order:
            add = min(1.0 - h[idx], remaining)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to decrease sum
        order = np.argsort(h)  # least elements first
        remaining = -diff
        for idx in order:
            sub = min(h[idx], remaining)
            h[idx] -= sub
            remaining -= sub
            if remaining <= 1e-12:

                                  base_step=0.1, rng=np.random.RandomState(seed), time_limit=t)
    if c5_opt < c5_best:
        return h_opt, c5_opt
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(45, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    print(f"Initial: C5 = {best_c5:.6f}, n = {n}")

    # Phase 1: Local search from initial
    t1 = min(120, (budget_s - 60) * 0.12)
    h1, c5_1 = local_search(h_init.copy(), n, max_iter=2000000, base_step=0.12,
                            rng=rng, time_limit=t1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    print(f"Phase 1: C5 = {best_c5:.6f} [{time.time()-start_time:.1f}s]")

    # Phase 2: Partial perturbation search
    if time.time() - start_time < budget_s - 100:
        t2 = min(120, (budget_s - 100 - (time.time() - start_time)) * 0.25)
        h2, c5_2 = partial_perturb_search(best_h.copy(), n, max_iter=2000000,
                                           base_step=0.15,
                                           rng=np.random.RandomState(seed + 1), time_limit=t2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
        print(f"Phase 2: C5 = {best_c5:.6f} [{time.time()-start_time:.1f}s]")

    # Phase 3: Simulated annealing
    if time.time() - start_time < budget_s - 100:
        t3 = min(150, (budget_s - 100 - (time.time() - start_time)) * 0.25)
        h3, c5_3 = simulated_annealing(best_h.copy(), n, max_iter=2000000,
                                        t_start=0.05, t_floor=0.002,
                                        rng=np.random.RandomState(seed + 2), time_limit=t3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
        print(f"Phase 3: C5 = {best_c5:.6f} [{time.time()-start_time:.1f}s]")

    # Phase 4: scipy L-BFGS-B
    if time.time() - start_time < budget_s - 80:
        h4 = scipy_optimize(best_h, n, tau=100.0, penalty=10000.0, maxiter=8000)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()
        # Polish
        t4p = min(60, (budget_s - 80 - (time.time() - start_time)) * 0.5)
        if t4p > 5:
            h4p, c5_4p = local_search(best_h.copy(), n, max_iter=1000000, base_step=0.04,
                                       rng=np.random.RandomState(seed + 3), time_limit=t4p)
            if c5_4p < best_c5:
                best_c5, best_h = c5_4p, h4p.copy()
        print(f"Phase 4: C5 = {best_c5:.6f} [{time.time()-start_time:.1f}s]")

    # Phase 5: Multi-start with structured initializations
    if time.time() - start_time < budget_s - 60:
        x = np.linspace(0, 2, n)
        structured_inits = []
        for freq in [1, 2, 3, 4, 5]:
            for amp in [0.15, 0.25, 0.35]:
                structured_inits.append(np.clip(0.5 + amp * np.cos(np.pi * freq * x), 0, 1))
                structured_inits.append(np.clip(0.5 + amp * np.sin(np.pi * freq * x), 0, 1))
        # Add some random inits
        for _ in range(10):
            structured_inits.append(np.clip(rng.uniform(0.2, 0.8, n), 0, 1))

        n_struct = len(structured_inits)
        for si in range(n_struct):
            if time.time() - start_time > budget_s - 20:
                break
            h_try = project_fast(structured_inits[si].copy(), n)
            t_try = min(15, max(3, (budget_s - 20 - (time.time() - start_time)) / max(1, n_struct - si) / 2))
            ht, ct = local_search(h_try, n, max_iter=100000, base_step=0.1,
                                   rng=np.random.RandomState(seed + 100 + si), time_limit=t_try)
            if ct < best_c5:
                best_c5, best_h = ct, ht.copy()
                print(f"  Multi-start {si}: C5 = {best_c5:.6f}")
        print(f"Phase 5: C5 = {best_c5:.6f} [{time.time()-start_time:.1f}s]")

    # Phase 6: Try different n values
    if time.time() - start_time < budget_s - 40:
        for n_try in [30, 35, 40, 50, 55]:
            if time.time() - start_time > budget_s - 20:
                break
            budget_left = budget_s - 20 - (time.time() - start_time)
            if budget_left < 10:
                break
            best_h, best_c5 = try_different_n(best_h, n, best_c5, n_try, budget_left, seed + n_try)
            if best_c5 < 0.3808:
                n = len(best_h)
                print(f"  n={n_try}: C5 = {best_c5:.6f}")
        print(f"Phase 6: C5 = {best_c5:.6f} [{time.time()-start_time:.1f}s]")

    # Phase 7: Final fine refinement
    if time.time() - start_time < budget_s - 5:
        t7 = budget_s - 5 - (time.time() - start_time)
        n = len(best_h)
        h7, c5_7 = local_search(best_h.copy(), n, max_iter=5000000, base_step=0.02,
                                 rng=np.random.RandomState(seed + 999), time_limit=t7)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, h7.copy()
        # Another round of partial perturb
        h7b, c5_7b = partial_perturb_search(best_h.copy(), n, max_iter=2000000, base_step=0.05,
                                             rng=np.random.RandomState(seed + 998), time_limit=min(30, t7/3))
        if c5_7b < best_c5:
            best_c5, best_h = c5_7b, h7b.copy()
        print(f"Phase 7: C5 = {best_c5:.6f} [{time.time()-start_time:.1f}s]")

    # Final projection and evaluation
    n = len(best_h)
    best_h = project_fast(best_h, n)
    best_c5 = compute_c5(best_h)
    elapsed = time.time() - start_time
    print(f"FINAL: C5 = {best_c5:.6f}, n = {n}, time = {elapsed:.1f}s")
    return best_h, best_c5, n
