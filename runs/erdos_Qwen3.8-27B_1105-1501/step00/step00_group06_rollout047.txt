        pass
    return None

def resize_h(h, nt):
    ns = len(h)
    if ns == nt:
        return h.copy()
    if nt > ns:
        xs = (np.arange(ns) + 0.5) / ns
        xt = (np.arange(nt) + 0.5) / nt
        hi = np.interp(xt, xs, h)
    else:
        hi = np.zeros(nt)
        for j in range(nt):
            lo = j * ns // nt
            hi2 = (j + 1) * ns // nt
            hi2 = max(hi2, lo + 1)
            hi[j] = h[lo:min(hi2, ns)].mean()
    return project_box_mass(hi)

def local_opt(h, rng, max_time, tag=""):
    end_t = time.time() + max_time
    n = len(h)
    best = h.copy()
    best_s = score_c5(h)
    K = 5
    sharp = 80.0
    stall = 0
    while time.time() < end_t:
        s = score_c5(h)
        if s < best_s - 1e-14:
            best_s, best, stall = s, h.copy(), 0
        else:
            stall += 1
            if stall > 200:
                h = project_box_mass(best + rng.randn(n) * 0.005)
                stall = 0
                K = max(K - 1, 1)
                if K < 2:
                    K, sharp = 5, 80.0
                continue
        g = subgrad(h, K=K, sharp=sharp)
        h_new, s_new = line_search(h, -g, s, npts=40)
        if h_new is not None and s_new < s - 1e-14:
            h = h_new
        else:
            # Try LP step
            if stall > 50 and rng.rand() < 0.1:
                d = lp_step(h)
                if d is not None:
                    h_lp, s_lp = line_search(h, -d, s, npts=30)
                    if h_lp is not None and s_lp < s - 1e-14:
                        h = h_lp
                    else:
                        h = project_box_mass(h + rng.randn(n) * 0.003)
            else:
                h = project_box_mass(h + rng.randn(n) * 0.004)
    return best, best_s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 70
        h = np.zeros(n_init)
        h[::2] = 1.0
        h = project_box_mass(h)

    n = len(h)
    best_h = h.copy()
    best_s = score_c5(h)

    # Phase 1: Main optimization on original n
    phase1_end = t0 + budget_s * 0.55
    K = 5
    sharp = 80.0
    stall = 0
    consecutive_no_improve = 0

    while time.time() < phase1_end:
        s = score_c5(h)
        if s < best_s - 1e-14:
            best_s, best_h = s, h.copy()
            stall = 0
            consecutive_no_improve = 0
        else:
            stall += 1
            consecutive_no_improve += 1

        if consecutive_no_improve > 400:
            # Restart from best with perturbation
            h = project_box_mass(best_h + rng.randn(n) * 0.01)
            consecutive_no_improve = 0
            stall = 0
            K = rng.choice([1, 2, 3, 5, 8])
            sharp = rng.uniform(30, 200)
            continue

        # Adaptive parameters
        if stall > 100:
            sharp = min(sharp * 1.3, 500.0)
            K = max(K - 1, 1)
        if stall > 150:
            K = 1
            sharp = 500.0

        g = subgrad(h, K=K, sharp=sharp)
        h_new, s_new = line_search(h, -g, s, npts=50)

        if h_new is not None and s_new < s - 1e-14:
            h = h_new
        else:
            # Try LP step
            if stall % 100 == 50:
                d = lp_step(h)
                if d is not None:
                    h_lp, s_lp = line_search(h, -d, s, npts=40)
                    if h_lp is not None and s_lp < s - 1e-14:
                        h = h_lp
                        continue
            # Small random kick
            h = project_box_mass(h + rng.randn(n) * 0.003)

    # Phase 2: Polish best solution
    remaining = budget_s - 40 - (time.time() - t0)
    if remaining > 10:
        best_h, best_s = local_opt(best_h, rng, remaining, "polish")

    # Phase 3: Try other dimensions
    remaining = budget_s - 20 - (time.time() - t0)
    if remaining > 20:
        n_candidates = [40, 48, 50, 56, 60, 64, 72, 80, 84, 90, 96, 100, 112, 120, 140, 160, 180]
        per_n = max(5, remaining // len(n_candidates))
        for nt in n_candidates:
            if time.time() - t0 > budget_s - 15:
                break
            ht = resize_h(best_h, nt)
            ht, st = local_opt(ht, rng, per_n, "dim")
            if st < best_s:
                best_h, best_s = ht, st

    # Final
    best_s = score_c5(best_h)
    return best_h, best_s, len(best_h)
