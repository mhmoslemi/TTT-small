import numpy as np
import time


def _project(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} via bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def _full_corr(h, n):
    """Compute full cross-correlation of h with 1-h, scaled by dx."""
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _weighted_grad(h, p, n):
    """Compute sum_k p[k] * d(c_k)/dh."""
    g = np.zeros(n)
    for k in range(len(p)):
        pk = p[k]
        if pk < 1e-14:
            continue
        l = k - (n - 1)
        if l == 0:
            g += pk * (1.0 - 2.0 * h)
        elif l > 0:
            m = n - l
            if m > 0:
                g[:m] += pk * (1.0 - h[l:])
                g[l:] += pk * (-h[:m])
        else:
            ll = -l
            m = n - ll
            if m > 0:
                g[ll:] += pk * (1.0 - h[:m])
                g[:m] += pk * (-h[ll:])
    return g


def _optimize_mw(h, n, rng, deadline):
    """Multiplicative weights (exponential weights) optimization."""
    ns = n / 2.0
    L = 2 * n - 1
    p = np.ones(L) / L
    best_h = h.copy()
    best_c5 = np.max(_full_corr(h, n))

    eta = 2.0
    alpha = 0.3
    no_improve = 0
    patience = 300

    for it in range(500000):
        if time.time() > deadline:
            break

        c = _full_corr(h, n)
        cm = np.max(c)
        if cm < best_c5 - 1e-12:
            best_c5 = cm
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            h = best_h + 0.08 * rng.randn(n)
            h = _project(h, ns)
            p = np.ones(L) / L
            no_improve = 0
            patience = min(patience + 200, 4000)
            continue

        cmin = np.min(c)
        crange = cm - cmin + 1e-10
        p *= np.exp(eta * (c - cmin) / crange)
        p /= p.sum()

        g = _weighted_grad(h, p, n)
        h = h - alpha * g
        h = _project(h, ns)

    return best_h, best_c5
