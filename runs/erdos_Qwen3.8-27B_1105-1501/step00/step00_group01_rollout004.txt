            t_split = time.time() + min(remaining * 0.4, 300.0)
            h_opt, val_opt = gd_projected(best_h, best_n, t_split, t0)
            try_solution(h_opt)

    # Phase 3: SLSQP refinement on best
    if best_h is not None:
        remaining = deadline - time.time()
        if remaining > 10:
            t_split = time.time() + min(remaining * 0.3, 120.0)
            h_opt, val_opt = slsqp_solve(best_h, best_n, t_split, t0)
            try_solution(h_opt)

    # Phase 4: Gradient descent further
    if best_h is not None:
        remaining = deadline - time.time()
        if remaining > 15:
            t_split = time.time() + min(remaining * 0.3, 200.0)
            h_opt, val_opt = gd_projected(best_h, best_n, t_split, t0)
            try_solution(h_opt)

    # Phase 5: Binary SA on n=62 or n=64
    if time.time() < deadline - 60:
        n_bin = 62 if best_n % 2 == 1 else best_n
        if n_bin % 2 != 0:
            n_bin = 62
        if n_bin in [62, 64]:
            # Create good binary starting point from best continuous
            if best_h is not None and len(best_h) == n_bin:
                bits_init = (best_h > 0.5).astype(int)
            else:
                bits_init = np.zeros(n_bin, dtype=int)
                bits_init[:n_bin // 2] = 1
            # Ensure correct count
            n_ones = np.sum(bits_init)
            target_ones = n_bin // 2
            if n_ones < target_ones:
                zeros = np.where(bits_init == 0)[0]
                bits_init[zeros[:target_ones - n_ones]] = 1
            elif n_ones > target_ones:
                ones = np.where(bits_init == 1)[0]
                bits_init[ones[:n_ones - target_ones]] = 0

            t_split = time.time() + min(deadline - time.time() - 30, 200.0)
            h_opt, val_opt = binary_sa(bits_init, n_bin, t_split, t0)
            try_solution(h_opt)

    # Phase 6: Continuous SA on best
    if best_h is not None and time.time() < deadline - 30:
        t_split = time.time() + min(deadline - time.time() - 15, 150.0)
        h_opt, val_opt = continuous_sa(best_h, best_n, t_split, t0)
        try_solution(h_opt)

    # Phase 7: Final SLSQP polish
    if best_h is not None and time.time() < deadline - 10:
        t_split = time.time() + min(deadline - time.time() - 5, 60.0)
        h_opt, val_opt = slsqp_solve(best_h, best_n, t_split, t0)
        try_solution(h_opt)

    # Phase 8: Final GD polish
    if best_h is not None and time.time() < deadline - 5:
        t_split = time.time() + min(deadline - time.time() - 2, 60.0)
        h_opt, val_opt = gd_projected(best_h, best_n, t_split, t0)
        try_solution(h_opt)

    # Final verification
    if best_h is not None:
        final_val = compute_c5(best_h)
        assert final_val < 1.0, "Solution invalid"
        assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
        assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    return best_h, best_c5, best_n
