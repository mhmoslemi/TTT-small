For np.correlate(h, 1-h, mode="full"):
- Let b = 1 - h
- The output has 2n-1 elements
- For index m (0 to 2n-2), the signed lag is l = m - (n-1)
- If l >= 0: C_m = sum_{i=l}^{n-1} h[i] * b[i-l] = sum_{i=l}^{n-1} h[i] * (1 - h[i-l])
- If l < 0, let p = -l: C_m = sum_{i=0}^{n-1-p} h[i] * b[i+p] = sum_{i=0}^{n-1-p} h[i] * (1 - h[i+p])

Gradient with respect to h[j]:
- Case l >= 0: C_m = sum_{i=l}^{n-1} h[i](1-h[i-l])

  - So: Jacobian[j] = (1-h[j-l]) if j>=l else 0, minus h[j+l] if j<=n-1-l else 0

- Case l < 0 (p = -l > 0): C_m = sum_{i=0}^{n-1-p} h[i](1-h[i+p])

```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_values(h):
    """Compute all correlation values (unscaled)."""
    return np.correlate(h, 1.0 - h, mode="full")

def smoothed_objective_and_grad(h, tau):
    """Compute log-sum-exp smoothed max and its gradient."""
    n = len(h)
    dx = 2.0 / n
    vals = compute_values(h) * dx
    # Shift for numerical stability
    M = np.max(vals)
    log_sum_exp = M + tau * np.log(np.sum(np.exp((vals - M) / tau)))
    weights = np.exp((vals - M) / tau)
    weights /= np.sum(weights)
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-15:
            continue
        l = m - (n - 1)
        g = np.zeros(n)
        if l >= 0:
            # grad[j] = (1-h[j-l]) * [j>=l] - h[j+l] * [j<=N-1-l]
            g[l:] += (1 - h[:n-l])
            g[:n-l] -= h[l:]
        else:
            p = -l
            g[:n-p] += (1 - h[p:])
            g[p:] -= h[:n-p]
        grad += weights[m] * dx * g  # Wait, vals already includes dx
    
    # Actually vals already has dx factor, so the gradient of vals w.r.t. h is dx * (raw gradient)
    # Let me redo: vals[m] = dx * raw_corr[m]
    # d(vals[m])/d(h[j]) = dx * d(raw_corr[m])/d(h[j])
    # So grad = sum_m weights[m] * dx * raw_grad_m[j]
    
    return log_sum_exp, grad

def projection(h):
```

Let me think more carefully about the optimization approach. The key challenge is that the objective is max of bilinear forms, which is non-convex.

For the smoothed approach, I'll use:
- SLSQP with the log-sum-exp smoothed objective
- Mass constraint: sum(h) = n/2 (equality)
- Box constraints: 0 <= h[i] <= 1

For the epigraph approach, I'll use:
- SLSQP or another NLP solver
- Variable: (h, t)
- Minimize t
- Constraints: t >= corr[m] * dx for selected active m
- Mass constraint
- Box constraints

Let me also think about what kind of structure the optimal solution might have. The Erdős overlap problem is related to finding functions with minimal "self-overlap" in a specific sense. 

For the pair moves approach:
- Pick two indices i, j
- Move epsilon from h[i] to h[j] (or vice versa)
- Check if the objective improves
- This preserves the mass constraint automatically

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_raw_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_row(h, m):
    """Compute gradient of corr[m] w.r.t. h."""
    n = len(h)
    l = m - (n - 1)
    g = np.zeros(n)
    if l >= 0:
        g[l:] += (1.0 - h[:n-l])
        g[:n-l] -= h[l:]
    else:
        p = -l
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def smoothed_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    vals = compute_raw_corrs(h) * dx
    M = np.max(vals)
    shifted = (vals - M) / tau
    exp_shifted = np.exp(shifted)
    log_sum_exp = M + tau * np.log(np.sum(exp_shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        grad += weights[m] * dx * grad_row(h, m)
    return log_sum_exp, grad

def projection(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    result = np.clip(h - mid, 0.0, 1.0)
    
    # Exact mass fix
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        if diff > 0:
            slacks = 1.0 - result
        else:
            slacks = result
        order = np.argsort(-slacks) if diff > 0 else np.argsort(slacks)
        remaining = abs(diff)
        for idx in order:
            if remaining <= 1e-15:
                break
            capacity = slacks[idx]
            step = min(capacity, remaining)
            if diff > 0:
                result[idx] += step
            else:
                result[idx] -= step
            remaining -= step
    return result

def smooth_solve(h0, targa, max_iter=50):
    """Solve smoothed max problem with SLSQP."""
    n = len(h0)
    
    def obj(x):
        return smoothed_obj_grad(x, targa)[0]
    
    def grad(x):
        return smoothed_obj_grad(x, targa)[1]
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    res = minimize(
        obj, h0, jac=grad, method='SLSQP',
        bounds=[(0, 1)] * n,
        constraints=[{'type': 'eq', 'fun': con_fun}],
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    if res.success or res.fun < 1e6:
        return projection(res.x)
    return h0

def epigraph_solve(h0, active_indices, max_iter=30):
    """Solve epigraph subproblem for selected active constraints."""
    n = len(h0)
    dx = 2.0 / n
    n_active = len(active_indices)
    
    # Variables: [h(0), ..., h(n-1), t]
    def obj(x):
        return x[n]  # minimize t
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def mass_con(x):
        return np.sum(x[:n]) - n / 2.0
    
    def mass_con_grad(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    constraints = [{'type': 'eq', 'fun': mass_con, 'jac': mass_con_grad}]
    
    h_curr = x  # can't use x directly, need to compute inside
    # Actually SLSQP will call these with different x values
    
    def make_incon(m_idx):
        hc = m_idx
        def fun(x):
            return x[n] - dx * compute_raw_corrs(x[:n])[hc]
        def jac(x):
            raw_grad = grad_row(x[:n], hc)
            g = np.zeros(n + 1)
            g[:n] = -dx * raw_grad
            g[n] = 1.0
            return g
        return {'type': 'ineq', 'fun': fun, 'jac': jac}
    
    for m_idx in active_indices:
        constraints.append(make_incon(m_idx))
    
    # Initial t
    t0 = float(np.max(compute_raw_corrs(h0)) * dx)
    x0 = np.concatenate([h0, [t0]])
    
    bounds = [(0, 1)] * n + [(-1, 2)]
    
    res = minimize(
        obj, x0, jac=obj_grad, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-10}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        return projection(res.x[:n])
    return h0

def pair_moves(h, max_moves=500, time_limit=None, start_time=None):
    """Local pair moves to improve the objective."""
    n = len(h)
    h = h.copy()
    current_score = exact_c5(h)
    rng_local = np.random.RandomState(int(hash(h.tobytes()) % 10000))
    
    deltas = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for delta in deltas:
        improved = True
        while improved and max_moves > 0:
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit:
                    return h, current_score
            improved = False
            for _ in range(200):
                if max_moves <= 0:
                    break
                i, j = rng_local.randint(0, n, 2)
                if i == j:
                    continue
                # Try moving delta from i to j, or from j to i
                for direction in [1, -1]:
                    h_new = h.copy()
                    if direction == 1:
                        # Move from i to j
                        if h[i] < delta or h[j] > 1 - delta:
                            continue
                        h_new[i] -= delta
                        h_new[j] += delta
                    else:
                        if h[j] < delta or h[i] > 1 - delta:
                            continue
                        h_new[j] -= delta
                        h_new[i] += delta
                    max_moves -= 1
                    score = exact_c5(h_new)
                    if score < current_score - 1e-12:
                        h = h_new
                        current_score = score
                        improved = True
                        break
    return h, current_score

def transfer_resolution(h_src, n_tgt):
    """Transfer h from source to target resolution via area interpolation."""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    # Build cumulative area of source
    # Source has bins at [k*dx_src, (k+1)*dx_src] with value h_src[k]
    # Cumulative area at x: sum of h_src[k] * dx_src for k where (k+1)*dx_src <= x
    x_src = np.arange(n_src + 1) * dx_src
    area_src = np.zeros(n_src + 1)
    area_src[1:] = np.cumsum(h_src * dx_src)
    
    # Target bin boundaries
    x_tgt = np.arange(n_tgt + 1) * dx_tgt
    
    # Interpolate
    area_tgt = np.interp(x_tgt, x_src, area_src)
    
    # Target values
    h_tgt = np.diff(area_tgt) / dx_tgt
    h_tgt = np.clip(h_tgt, 0, 1)
    
    return projection(h_tgt)
```

```python
def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_raw_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")


def grad_row(h, m):
    n = len(h)
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g


def smoothed_obj(x, tau):
    n = len(x)
    dx = 2.0 / n
    vals = compute_raw_corrs(x) * dx
    M = np.max(vals)
    shifted = (vals - M) / tau
    return M + tau * np.log(np.sum(np.exp(shifted)))


def smoothed_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    vals = compute_raw_corrs(x) * dx
    M = np.max(vals)
    shifted = (vals - M) / tau
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-14:
            continue
        grad += weights[m] * dx * grad_row(x, m)
    return grad

        return h

    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)

        if s > target:
            lo = mid
        else:
            hi = mid

def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    x_src = np.arange(n_src + 1) * dx_src
    area_src = np.zeros(n_src + 1)
    area_src[1:] = np.cumsum(h_src * dx_src)
    x_tgt = np.arange(n_tgt + 1) * dx_tgt
    area_tgt = np.interp(x_tgt, x_src, area_src)
    h_tgt = np.diff(area_tgt) / dx_tgt
    return projection(h_tgt)

        numeric = np.zeros(n)
        for j in range(n):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            numeric[j] = (compute_raw_corrs(h_plus)[m] - compute_raw_corrs(h_minus)[m]) / (2 * eps)
        if not np.allclose(analytic, numeric, atol=1e-5):
            return False
    return True


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s * 0.92

    def elapsed():
        return time.time() - t_start

    def time_left():
        return deadline - time.time()

    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = None

    if parent is None or len(parent) == 0 or not np.all(np.isfinite(parent)):
        parent = np.full(45, 0.5)

    parent = projection(parent)
    best_h = parent.copy()
    best_score = exact_c5(best_h)
    n0 = len(best_h)

    # Verify gradient correctness once
    verify_gradients(best_h)

    # Build diverse starting points
    starts = [best_h.copy()]
    starts.append(projection(1.0 - best_h))
    starts.append(projection(best_h[::-1].copy()))

    for freq in [1, 2, 3]:
        for phase in [0.0, np.pi / 3, 2 * np.pi / 3]:
            s = projection(make_sinusoidal(n0, freq, phase, amp=0.35))
            starts.append(s)

    starts.append(make_binary_pattern(n0, seed))
    starts.append(projection(make_binary_pattern(n0, seed + 1)[::-1].copy()))

    rng = np.random.RandomState(seed)
    for sig in [0.25, 0.12, 0.06]:
        pert = projection(best_h + rng.randn(n0) * sig)
        starts.append(pert)

    # Stage 1: Smoothed optimization from multiple starts
    taus = [0.05, 0.025, 0.012, 0.006, 0.003]

    for si in range(len(starts)):
        if time_left() < 60:
            break
        h_curr = starts[si].copy()
        curr_score = exact_c5(h_curr)
        for tau in taus:
            if time_left() < 45:
                break
            if elapsed() > budget_s * 0.65:
                break
            h_new = smooth_solve(h_curr, tau, max_iter=45)
            h_new = projection(h_new)
            sc = exact_c5(h_new)
            if sc < curr_score - 1e-10:
                curr_score = sc
                h_curr = h_new
            if sc < best_score - 1e-10:
                best_score = sc
                best_h = h_new.copy()

    # Stage 2: Epigraph polishing (multiple rounds)
    for round_idx in range(4):
        if time_left() < 40:
            break
        vals = compute_raw_corrs(best_h) * (2.0 / len(best_h))
        n_corr = len(vals)
        top_k = min(15, n_corr)
        active = np.argsort(-vals)[:top_k].tolist()

        h_new = epigraph_solve(best_h, active, max_iter=35)
        h_new = projection(h_new)
        sc = exact_c5(h_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h_new.copy()

    # Stage 3: Guided local moves
    if time_left() > 40:
        tb = min(50.0, time_left() * 0.3)
        h_moved, sc_moved = guided_pair_moves(best_h, max_moves=3000,
                                               time_budget=tb, seed=seed)
        if sc_moved < best_score - 1e-10:
            best_score = sc_moved
            best_h = h_moved.copy()

    # Stage 4: Random local moves
    if time_left() > 30:
        tb = min(30.0, time_left() * 0.3)
        h_moved, sc_moved = random_pair_moves(best_h, max_moves=2000,
                                               time_budget=tb, seed=seed + 100)
        if sc_moved < best_score - 1e-10:
            best_score = sc_moved
            best_h = h_moved.copy()

    # Stage 5: Resolution transfer
    if time_left() > 80:
        n_candidates = [30, 36, 40, 45, 50, 54, 60, 72, 80, 90, 100, 120]
        n_candidates = [n for n in n_candidates if n != n0 and n >= 20]
        n_candidates.sort(key=lambda x: abs(x - n0))

        for n_tgt in n_candidates[:5]:
            if time_left() < 60:
                break
            h_trans = transfer_resolution(best_h, n_tgt)
            sc_trans = exact_c5(h_trans)

            if time_left() > 40:
                for tau in [0.02, 0.01]:
                    if time_left() < 30:
                        break
                    h_t2 = smooth_solve(h_trans, tau, max_iter=30)
                    h_t2 = projection(h_t2)
                    sc_t2 = exact_c5(h_t2)
                    if sc_t2 < sc_trans:
                        h_trans = h_t2
                        sc_trans = sc_t2

            if sc_trans < best_score - 1e-10:
                best_score = sc_trans
                best_h = h_trans.copy()

    # Stage 6: Final epigraph + moves on best
    if time_left() > 30:
        vals = compute_raw_corrs(best_h) * (2.0 / len(best_h))
        n_corr = len(vals)
        top_k = min(12, n_corr)
        active = np.argsort(-vals)[:top_k].tolist()
        h_new = epigraph_solve(best_h, active, max_iter=30)
        h_new = projection(h_new)
        sc = exact_c5(h_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h_new.copy()

    if time_left() > 15:
        h_moved, sc_moved = guided_pair_moves(best_h, max_moves=1500,
                                               time_budget=min(15.0, time_left() * 0.5),
                                               seed=seed + 200)
        if sc_moved < best_score - 1e-10:
            best_score = sc_moved
            best_h = h_moved.copy()

    # Final validation
    best_h = projection(best_h)
    best_score = exact_c5(best_h)

    return best_h, best_score, int(len(best_h))
```
