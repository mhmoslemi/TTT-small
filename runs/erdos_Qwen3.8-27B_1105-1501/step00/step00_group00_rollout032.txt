    n = len(h)
    overlaps = np.zeros(n - 1)
    for k in range(1, n):
        overlaps[k-1] = np.sum(h[:n-k] * (1.0 - h[k:])) * dx
    return overlaps

    return np.max(overlaps)

def logsumexp_objective(overlaps, beta):
    """Smooth approximation of max(overlaps)"""
    m = np.max(overlaps)
    if m == 0:
        return 0.0, np.ones(len(overlaps)) / len(overlaps)
    exps = np.exp(beta * (overlaps - m))
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

    weights[k] corresponds to lag k+1 (0-indexed).

import numpy as np
import time

def project_constraint(v, target_sum):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=target_sum}"""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def all_overlaps(h, dx):
    """Compute all 2n-1 overlap values (forward and reverse)."""
    n = len(h)
    # Use np.correlate
    return np.correlate(h, 1.0 - h, mode="full") * dx

def c5_value(h, dx):
    return np.max(all_overlaps(h, dx))

def gradient_full(h, dx, weights):
    """Gradient of weighted sum of all overlaps.
    weights has length 2n-1, corresponding to np.correlate output.
    """
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    # np.correlate(h, 1-h, mode="full")[idx]
    # For idx = n-1+k (k >= 0): sum_i h[i+k]*(1-h[i]), i.e., R_k
    # For idx = n-1-k (k >= 0): sum_i h[i]*(1-h[i+k]), i.e., F_k
    
    # R_k = dx * sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
    # dR_k/dh[j]: 
    #   j appears as h[i+k] when i=j-k (j>=k, j<n): contributes dx*(1-h[j-k])
    #   j appears as (1-h[i]) when i=j (j<n-k): contributes -dx*h[j+k]
    
    # F_k = dx * sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
    # dF_k/dh[j]:
    #   j appears as h[i] when i=j (j<n-k): contributes dx*(1-h[j+k])
    #   j appears as (1-h[i+k]) when i+k=j (j>=k): contributes -dx*h[j-k]
    
    for k in range(1, n):
        # F_k: weight is at index n-1-k in the correlate output
        wf = weights[n-1-k] * dx
        # R_k: weight is at index n-1+k
        wr = weights[n-1+k] * dx
        
        if wf != 0.0 or wr != 0.0:
            # F_k contribution: for j < n-k: +(1-h[j+k])
            if wf != 0.0:
                grad[:n-k] += wf * one_minus_h[k:]
            # F_k contribution: for j >= k: -h[j-k]
            if wf != 0.0:
                grad[k:] -= wf * h[:n-k]
            
            # R_k contribution: for j >= k and j < n: +(1-h[j-k])
            if wr != 0.0:
                grad[k:] += wr * one_minus_h[:n-k]
            # R_k contribution: for j < n-k: -h[j+k]
            if wr != 0.0:
                grad[:n-k] -= wr * h[k:]
    
    # F_0 = R_0: weight at index n-1
    w0 = weights[n-1] * dx
    if w0 != 0.0:
        # dF_0/dh[j] = dx*(1-h[j]) - dx*h[j] = dx*(1-2*h[j])
        grad += w0 * (1.0 - 2.0 * h)
    
    return grad

def optimize(h, dx, target_sum, time_budget, beta_start=5.0):
    """Projected gradient descent with log-sum-exp."""
    n = len(h)
    h = project_constraint(h, target_sum).copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    start = time.time()
    
    beta = beta_start
    lr = 0.05
    stagnation = 0
    
    while time.time() - start < time_budget:
        ov = all_overlaps(h, dx)
        m = np.max(ov)
        exps = np.exp(beta * (ov - m))
        s = np.sum(exps)
        weights = exps / s
        obj_val = m + np.log(s) / beta
        
        grad = gradient_full(h, dx, weights)
        # Center gradient (projection onto affine constraint)
        grad = grad - np.mean(grad)
        
        # Line search
        h_new = h
        alpha = lr
        for _ in range(15):
            h_trial = project_constraint(h - alpha * grad, target_sum)
            ov_t = all_overlaps(h_trial, dx)
            m_t = np.max(ov_t)
            exps_t = np.exp(beta * (ov_t - m_t))
            s_t = np.sum(exps_t)
            obj_t = m_t + np.log(s_t) / beta
            if obj_t < obj_val + 1e-10:
                h_new = h_trial
                break
            alpha *= 0.5
        
        h = h_new
        c5_now = c5_value(h, dx)
        
        if c5_now < best_c5 - 1e-7:
            best_c5 = c5_now
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Adapt parameters
        if stagnation > 30:
            beta = min(beta * 1.5, 50.0)
            lr *= 0.8
            stagnation = 0
        elif stagnation > 100:
            # Restart with perturbation
            h = project_constraint(h + 0.05 * np.random.randn(n), target_sum)
            beta = beta_start
            lr = 0.05
            stagnation = 0
    
    return best_h, best_c5

def project_constraint(v, target_sum):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=target_sum} using bisection on threshold."""
    lo = float(v.min()) - 1.0

        else:

    return np.correlate(h, 1.0 - h, mode="full") * dx

    return float(np.max(all_overlaps(h, dx)))

    """Compute gradient of weighted sum of all overlaps.
    weights: array of length 2n-1 (matches np.correlate output order).
    """
    n = len(h)

            grad[k:] -= wf * h[:n - k]
