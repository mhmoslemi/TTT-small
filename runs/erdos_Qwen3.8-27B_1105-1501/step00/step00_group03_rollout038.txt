from scipy.optimize import minimize
import time

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(z, target=None):
    """Project z onto {h: 0<=h[i]<=1, sum(h)=target} using bisection on shift."""
    n = len(z)
    if target is None:
        target = n / 2.0
    
    def sum_clipped(shift):
        return np.sum(np.clip(z + shift, 0, 1))
    
    # Find shift such that sum_clipped(shift) = target
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    
    for _ in range(80):
        mid = (lo + hi) / 2
        s = sum_clipped(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    
    shift = (lo + hi) / 2
    h = np.clip(z + shift, 0, 1)
    
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1 - h)[::-1]  # most room first
        else:
            order = np.argsort(h)[::-1]  # most value first
        for idx in order:
            if residual > 0:
                delta = min(residual, 1 - h[idx])
                h[idx] += delta
                residual -= delta
            else:
                delta = max(residual, -h[idx])
                h[idx] += delta
                residual -= delta
            if abs(residual) < 1e-12:
                break
    
    return h

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1 - h, mode='full') * dx
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    # Safe log-sum-exp
    shifted = np.minimum(shifted, 500)  # prevent overflow
    logsum = cmax + (np.log(np.sum(np.exp(shifted))) ) / beta
    return logsum

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1 - h, mode='full') * dx
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    shifted = np.minimum(shifted, 500)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.ones_like(vals) / len(vals)
    else:
        weights /= wsum
    
    # Compute gradient
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for r in range(ncorr):
        k = r - (n - 1)
        if abs(weights[r]) < 1e-15:
            continue
        if k >= 0:
            for p in range(k, n):
                grad[p] += weights[r] * (1 - h[p - k])
            for p in range(0, n - k):
                grad[p] += weights[r] * (-h[p + k])
        else:
            m = -k
            for p in range(0, n - m):
                grad[p] += weights[r] * (1 - h[p + m])
            for p in range(m, n):
                grad[p] += weights[r] * (-h[p - m])
    
    return dx * grad

def smooth_eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def smooth_eq_grad(h):
    return np.ones(len(h))

def smooth_bounds(n):
    return [(0.0, 1.0)] * n

def run_smooth(h0, beta, maxiter=100):
    n = len(h0)
    def fun(h):
        return smooth_objective(h, beta)
    def jac(h):
        return smooth_gradient(h, beta)
    
    res = minimize(fun, h0, jac=jac, method='SLSQP',
                   bounds=smooth_bounds(n),
                   constraints=[{'type': 'eq', 'fun': lambda h: smooth_eq_fun(h), 
                                'jac': lambda h: smooth_eq_grad(h)}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(z, target=None):
    n = len(z)
    if target is None:
        target = n / 2.0
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                delta = min(residual, 1.0 - h[idx])
            else:
                delta = max(residual, -h[idx])
            h[idx] += delta
            residual -= delta
    return h

def corr_jac(h):
    n = len(h)
    ncorr = 2 * n - 1
    J = np.zeros((ncorr, n))
    for r in range(ncorr):
        k = r - (n - 1)
        if k >= 0:
            for p in range(k, n):
                J[r, p] += (1.0 - h[p - k])
            for p in range(0, n - k):
                J[r, p] += (-h[p + k])
        else:
            m = -k
            for p in range(0, n - m):
                J[r, p] += (1.0 - h[p + m])
            for p in range(m, n):
                J[r, p] += (-h[p - m])
    return J

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1 - h, mode='full') * dx
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -500, 500)
    return cmax + np.log(np.sum(np.exp(shifted))) / beta

def smooth_jac(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1 - h, mode='full') * dx
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.ones(len(vals)) / len(vals)
    else:
        w /= wsum
    J = corr_jac(h)
    return dx * (J.T @ w)

    # Minimax polish at best resolution
    if time.time() < deadline:
        res = do_minimax(best_h.copy(), 500)
        if res.x is not None and np.all(np.isfinite(res.x)):
            check_and_update(res.x[:len(best_h)])

    # Phase 2: Try other resolutions
    res_list = sorted(set([n_best, n_best - 5, n_best + 5, 30, 35, 40, 45, 50, 55, 60]))
    res_list = [r for r in res_list if 20 <= r <= 70 and r != n_best]

    for n_try in res_list:
        if time.time() > deadline:
            break
        h_interp = project_feasible(interp_to_n(best_h, n_try))
        h_cur = h_interp

        for beta in [50, 500, 5000]:
            if time.time() > deadline:
                break
            res = do_smooth(h_cur, beta, 150)
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x)
                check_and_update(h_new)
                h_cur = h_new

        if time.time() < deadline:
            res = do_minimax(h_cur, 300)
            if res.x is not None and np.all(np.isfinite(res.x)):
                check_and_update(res.x[:n_try])

    # Phase 3: Random restarts
    rng = np.random.default_rng(seed)
    n_r = len(best_h)
    for attempt in range(40):
        if time.time() > deadline:
            break
        sigma = float(rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.4]))
        h_pert = best_h + rng.normal(0.0, sigma, n_r)
        h_pert = project_feasible(h_pert)

        res = do_smooth(h_pert, 500, 100)
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x)
            check_and_update(h_new)

        if time.time() < deadline and attempt % 3 == 0:
            res = do_minimax(best_h.copy(), 200)
            if res.x is not None and np.all(np.isfinite(res.x)):
                check_and_update(res.x[:n_r])

    # Phase 4: Final intensive polish
    if time.time() < deadline:
        n_f = len(best_h)
        res = do_smooth(best_h.copy(), 10000, 500)
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x)
            check_and_update(h_new)

    if time.time() < deadline:
        res = do_minimax(best_h.copy(), 1000)
        if res.x is not None and np.all(np.isfinite(res.x)):
            check_and_update(res.x[:len(best_h)])

    if time.time() < deadline:
        n_f = len(best_h)
        res = do_smooth(best_h.copy(), 50000, 500)
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x)
            check_and_update(h_new)

    if time.time() < deadline:
        res = do_minimax(best_h.copy(), 500)
        if res.x is not None and np.all(np.isfinite(res.x)):
            check_and_update(res.x[:len(best_h)])

    # Return with safe bound
    if best_h is None:
        best_h = np.full(40, 0.5)
        best_c = official_score(best_h)

    final_bound = best_c + 1e-13
    return best_h, final_bound, len(best_h)
