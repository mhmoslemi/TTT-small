def compute_score(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    """Compute all correlation values (unscaled)"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_overlap(h, s):
    """
    Compute gradient of A_s(h) = sum valid h[n]*(1-h[n-s]) with respect to h.
    s is the lag: negative to positive.
    Result index in correlate array: n-1+s
    """
    n = len(h)
    g = np.zeros(n)
    
    if s >= 0:
        # A_s = sum_{n=s}^{n-1} h[n] * (1-h[n-s])
        # For j >= s: g[j] += (1-h[j-s])
        g[s:] += (1.0 - h[:n-s])
        # For j <= n-1-s: g[j] -= h[j+s]
        g[:n-s] -= h[s:]
    else:
        t = -s
        # A_{-t} = sum_{n=0}^{n-1-t} h[n] * (1-h[n+t])
        # For j <= n-1-t: g[j] += (1-h[j+t])
        g[:n-t] += (1.0 - h[t:])
        # For j >= t: g[j] -= h[j-t]
        g[t:] -= h[:n-t]
    
    return g

def project_to_box_sum(vec, target_sum):

    n = len(vec)
    x = vec.copy()

    # x_i = clip(v_i + lambda, 0, 1) where lambda is chosen so sum = target
    lo = vec.min() - 1.0
    hi = vec.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(vec + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    x = np.clip(vec + (lo + hi) / 2.0, 0.0, 1.0)

    if abs(err) > 1e-12:

        if free.sum() > 0:

            x = np.clip(x, 0.0, 1.0)

    """
    Solve the SLP subproblem:
    min t
    s.t. A_k(h) + grad_k(h)^T (h' - h) <= t for all k
         sum(h') = n/2
         |h' - h|_inf <= radius (approximated by box)
    
    Returns proposed h' or None.
    """

    dx = 2.0 / n
    num_lags = 2 * n - 1
    
    # Compute current overlaps and gradients
    corr = compute_all_corr(h)
    max_corr = np.max(corr)

    G = np.zeros((num_lags, n))
    for idx in range(num_lags):

        G[idx] = gradient_overlap(h, s)

    # => G[k,:] dot h' - t <= G[k,:] dot h - corr[k]

    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - corr

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = []

        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
