    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / T)
    w = exp_c / np.sum(exp_c)
    smooth_val = cmax + T * np.log(np.sum(exp_c))
    
    grad = np.zeros(n)
    center = n - 1
    
    # Center (lag 0)
    grad += w[center] * dx * (1.0 - 2.0 * h)
    
    for k in range(1, n):
        # Right shift (position center + k)
        wr = w[center + k]
        if wr > 0:
            # ∂/∂h[j] = dx*(1-h[j-k]) for j>=k, minus dx*h[j+k] for j<=n-1-k

            grad[k:] += wr * dx * (1.0 - h[:-k] if k > 0 else None)
            # Wait, h[j-k] for j=k..n-1 is h[0..n-1-k]

import numpy as np
import time

def score_exact(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx using discrete correlation."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(y, n=None):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    if n is None:
        n = len(y)
    target = n / 2.0
    
    # Bisection for lambda
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix any residual sum error
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        # Adjust the element with most slack
        if diff > 0:
            # Need to add mass; find element with h < 1
            slack = 1.0 - z
            idx = np.argmax(slack)
            adjust = min(diff, slack[idx])
            z[idx] += adjust
            diff -= adjust
            if abs(diff) > 1e-12:
                idx2 = np.argmax(slack)
                if idx2 != idx:
                    z[idx2] += min(diff, slack[idx2])
        else:
            diff = -diff
            # Need to remove mass; find element with h > 0
            room = z.copy()
            idx = np.argmax(room)
            adjust = min(diff, room[idx])
            z[idx] -= adjust
            diff -= adjust
            if abs(diff) > 1e-12:
                idx2 = np.argmax(room)
                if idx2 != idx:
                    z[idx2] -= min(diff, room[idx2])
    
    return z

def smooth_max_grad(h, T):
    """Compute smooth max value and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(c))
    exp_c = np.exp((c - cmax) / T)
    S = float(np.sum(exp_c))
    w = exp_c / S
    smooth_val = cmax + T * np.log(S)
    
    grad = np.zeros(n)
    center = n - 1
    
    # Center (lag 0): c = dx * sum h[i]*(1-h[i])
    # d/dh[j] = dx*(1-2h[j])
    grad += w[center] * dx * (1.0 - 2.0 * h)
    
    for k in range(1, n):
        # Right shift (position center + k)
        wr = w[center + k]
        if wr > 1e-15:
            grad[k:n] += wr * dx * (1.0 - h[0:n-k])
            grad[0:n-k] += wr * dx * (-h[k:n])
        
        # Left shift (position center - k)
        wl = w[center - k]
        if wl > 1e-15:
            grad[0:n-k] += wl * dx * (1.0 - h[k:n])
            grad[k:n] += wl * dx * (-h[0:n-k])
    
    return smooth_val, grad, c

def pair_line_search(h, i, j, n_samples=31):
    """
    Minimize the exact objective along the line h_i+alpha, h_j-alpha.
    Returns (h_new, score_new) or (None, inf) if no improvement.
    """
    n = len(h)
    hi_val = h[i]
    hj_val = h[j]
    
    lo = max(-hi_val, hj_val - 1.0)
    hi = min(1.0 - hi_val, hj_val)
    
    if hi - lo < 1e-10:
        return None, np.inf
    
    alphas = np.linspace(lo, hi, n_samples)
    best_alpha = None
    best_score = np.inf
    
    h_trial = h.copy()
    
    for a in alphas:
        h_trial[i] = hi_val + a
        h_trial[j] = hj_val - a
        s = score_exact(h_trial)
        if s < best_score:
            best_score = s
            best_alpha = a
    
    # Refine around best with golden section
    if best_alpha is not None:
        # Determine local interval
        idx = np.argmin(alphas - best_alpha if False else np.abs(alphas - best_alpha))
        if 0 < idx < n_samples - 1:
            a_lo = alphas[idx-1]
            a_hi = alphas[idx+1]
        else:
            a_lo = lo
            a_hi = hi
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1 = a_hi - gr * (a_hi - a_lo)
        a2 = a_lo + gr * (a_hi - a_lo)
        
        h_trial[i] = hi_val + a1
        h_trial[j] = hj_val - a1
        s1 = score_exact(h_trial)
        h_trial[i] = hi_val + a2
        h_trial[j] = hj_val - a2
        s2 = score_exact(h_trial)
        
        for _ in range(20):
            if s1 < s2:
                a_hi = a2
                a2 = a1
                s2 = s1
                a1 = a_hi - gr * (a_hi - a_lo)
                h_trial[i] = hi_val + a1
                h_trial[j] = hj_val - a1
                s1 = score_exact(h_trial)
            else:
                a_lo = a1
                a1 = a2
                s1 = s2
                a2 = a_lo + gr * (a_hi - a_lo)
                h_trial[i] = hi_val + a2
                h_trial[j] = hj_val - a2
                s2 = score_exact(h_trial)
        
        best_alpha = (a_lo + a_hi) / 2.0
        h_trial[i] = hi_val + best_alpha
        h_trial[j] = hj_val - best_alpha
        best_score = score_exact(h_trial)
        return h_trial, best_score
    
    return None, np.inf

def resample(h_old, n_new):
    """Resample a step function from n_old to n_new points."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    # Midpoints
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    
    # Linear interpolation
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, n_new)
    return h_new

def make_binary(h, n=None):
    """Create a binary (0/1) version of h with correct mass."""
    if n is None:
        n = len(h)

import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(y, n=None):
    if n is None:
        n = len(y)
    target = n / 2.0
    lo = float(np.min(y)) - 1.5
    hi = float(np.max(y)) + 0.5
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-14:
        if diff > 0:
            slack = 1.0 - z
            order = np.argsort(-slack)
            for idx in order:
                adj = min(diff, slack[idx])
                if adj > 0:
                    z[idx] += adj
                    diff -= adj
                if abs(diff) < 1e-14:
                    break
        else:
            diff = -diff
            order = np.argsort(-z)
            for idx in order:
                adj = min(diff, z[idx])
                if adj > 0:
                    z[idx] -= adj
                    diff -= adj
                if abs(diff) < 1e-14:
                    break
    return z

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(c))
    exp_c = np.exp((c - cmax) / T)
    S = float(np.sum(exp_c))
    w = exp_c / S
    smooth_val = cmax + T * np.log(S)
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        wr = w[center + k]
        if wr > 1e-16:
            grad[k:n] += wr * dx * (1.0 - h[0:n-k])
            grad[0:n-k] += wr * dx * (-h[k:n])
        wl = w[center - k]
        if wl > 1e-16:
            grad[0:n-k] += wl * dx * (1.0 - h[k:n])
            grad[k:n] += wl * dx * (-h[0:n-k])
    return smooth_val, grad, c
