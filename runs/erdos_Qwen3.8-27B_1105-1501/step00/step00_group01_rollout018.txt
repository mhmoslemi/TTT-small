import numpy as np
from scipy.optimize import linprog
import time

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def lag_gradient(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        L = n - k
        if k == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        else:
            grad[k:] += dx * (1.0 - h[:L])
            grad[:L] += -dx * h[k:]
    else:
        e = -k
        L = n - e
        grad[:L] += dx * (1.0 - h[e:])
        grad[e:] += -dx * h[:L]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.5
    hi = h.max() + 0.5
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h_proj))
        else:
            idx = int(np.argmin(h_proj))
        h_proj[idx] = np.clip(h_proj[idx] + diff, 0.0, 1.0)
    return h_proj

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c = corr - corr.max()
    exp_c = np.exp(alpha * c)
    weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        grad += w * lag_gradient(h, k)
    return grad

def run_pgd(h, n, alpha, steps, lr, start_time, time_limit):
    best = h.copy()
    best_val = evaluate_h(h)
    current = h.copy()
    for step in range(steps):
        if time.time() - start_time > time_limit:
            break
        grad = smooth_max_grad(current, alpha)
        current = project_feasible(current - lr * grad, n)
        val = evaluate_h(current)
        if val < best_val:
            best = current.copy()
            best_val = val
    return best, best_val

def get_lp_direction(h, n, active_count, trust_r):
    corr = compute_corr(h)
    dx = 2.0 / n
    sorted_idx = np.argsort(-corr)
    active_idx = sorted_idx[:active_count]
    c_obj = np.zeros(2 * n)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for idx in active_idx:
        k = int(idx) - (n - 1)
        gk = lag_gradient(h, k)
        row = np.zeros(2 * n)
        row[:n] = gk
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-dx * corr[int(idx)])
    I = np.eye(n)
    A_rows.append(I)
    b_rows.append(1.0 - h)
    A_rows.append(-I)
    b_rows.append(h)
    if trust_r is not None:
        A_rows.append(I)
        b_rows.append(np.full(n, trust_r))
        A_rows.append(-I)
        b_rows.append(np.full(n, trust_r))
    A_ub = np.vstack(A_rows)
    b_ub = np.concatenate(b_rows)
    A_eq = np.zeros((1, 2 * n))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    d = res.x[:n]
    h_try = np.clip(h + d, 0.0, 1.0)
    h_try = project_feasible(h_try, n)
    return h_try

def line_search_from(h, target, n):
    v = target - h
    best = h.copy()
    best_val = evaluate_h(h)
    alphas = [1.0, 0.5, 0.75, 0.25, 0.1, 0.9, 0.3, 0.6, 0.4, 0.8, 0.15, 0.7]
    for a in alphas:
        h_try = h + a * v
        h_try = project_feasible(h_try, n)
        val = evaluate_h(h_try)
        if val < best_val:
            best_val = val
            best = h_try.copy()
    return best, best_val

def make_start(n, rng, init):
    if init is not None and len(init) == n:
        return project_feasible(init.copy(), n)
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.25 * np.sin(2.0 * np.pi * t) + 0.15 * np.sin(4.0 * np.pi * t + 1.0)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h

def optimize_at_n(h, n, start_time, time_limit):
    current = h.copy()
    current_val = evaluate_h(h)
    # Phase 1: PGD with smooth max
    for alpha in [20, 50, 100, 200, 500]:
        if time.time() - start_time > time_limit:
            break
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005]:
            if time.time() - start_time > time_limit:
                break
            result, val = run_pgd(current, n, alpha, 100, lr, start_time, time_limit)
            if val < current_val:
                current = result
                current_val = val
    # Phase 2: LP-based refinement
    for iteration in range(200):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for active_count in [5, 10, 20, 30, 40]:
            if time.time() - start_time > time_limit:
                break
            for trust_r in [0.1, 0.05, 0.02, 0.5, 1.0]:
                if time.time() - start_time > time_limit:
                    break
                direction = get_lp_direction(current, n, active_count, trust_r)
                if direction is None or np.allclose(direction, current):
                    continue
                new_h, new_val = line_search_from(current, direction, n)
                if new_val < current_val - 1e-12:
                    current = new_h
                    current_val = new_val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    # Phase 3: Random restarts (small perturbations)
    for _ in range(20):
        if time.time() - start_time > time_limit:
            break
        h_pert = current + np.random.randn(n) * 0.1
        h_pert = project_feasible(h_pert, n)
        val = evaluate_h(h_pert)
        if val < current_val:
            current = h_pert
            current_val = val
            for alpha in [100, 300]:
                if time.time() - start_time > time_limit:
                    break
                result, v2 = run_pgd(current, n, alpha, 100, 0.01, start_time, time_limit)
                if v2 < current_val:
                    current = result
                    current_val = v2
    return current, current_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
    n_list = []
    if init is not None:
        n_list.append(len(init))
    for n in [64, 48, 56, 80, 96, 128, 160]:
        if n not in n_list and n % 2 == 0:
            n_list.append(n)
    if init is not None:
        n_init = len(init)
        n_list.sort(key=lambda x: (abs(x - n_init), x))
    else:
        n_list.sort()
    n_trials = len(n_list)
    total_time = budget_s * 0.88
    for trial_idx, n in enumerate(n_list):
        if time.time() - start_time > budget_s * 0.92:
            break
        time_remaining = budget_s * 0.92 - (time.time() - start_time)
        time_per = total_time / n_trials
        time_limit = start_time + min(time_per * 1.5, time_remaining)
        h_start = make_start(n, rng, init)
        val = evaluate_h(h_start)
        if val < best_c5:
            best_c5 = val
            best_h = h_start.copy()
        h_opt, val_opt = optimize_at_n(h_start, n, start_time, time_limit)
        if val_opt < best_c5:
            best_c5 = val_opt
            best_h = h_opt.copy()
    if best_h is None:
        best_h = np.full(10, 0.5)
        best_c5 = evaluate_h(best_h)
    final_val = evaluate_h(best_h)
    best_c5 = min(best_c5, final_val)
    return best_h, best_c5, len(best_h)
