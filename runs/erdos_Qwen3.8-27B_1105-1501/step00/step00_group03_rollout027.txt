from scipy.optimize import linprog

def compute_c5(h):
    """Compute the C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_all_F(h):
    """Compute all F_k(h) values (unscaled, i.e., without dx)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr

def compute_grads_row(h, k):
    """Compute gradient of F_k(h) w.r.t. h. Returns array of shape (n,)."""
    n = len(h)
    grad = np.zeros(n)
    i_start = max(0, -k)
    i_end = min(n, n - k)
    for i in range(i_start, i_end):
        j = i + k
        if k == 0:
            grad[i] += 1 - 2*h[i]
        else:
            grad[i] += 1 - h[j]
            grad[j] += -h[i]
    return grad

def project_box_sum(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}."""
    # This is a quadratic projection
    from scipy.optimize import minimize
    target = n / 2.0

        return np.sum(np.clip(h - tau, 0, 1))

        mid = (lo + hi) / 2
        if sum_at_tau(mid) > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)

def solve_lp(h0, active_lags, n):
    """Solve the linearized LP.
    Minimize t s.t. F_r(h0) + grad_r · (h - h0) <= t for active lags,
    0 <= h <= 1, sum(h) = n/2.
    """
    m = len(active_lags)
    # Variables: h[0..n-1], t => n+1 variables
    # Objective: minimize t => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag r: grad_r · h - t <= F_r(h0) - grad_r · h0
    # i.e., [grad_r, -1] @ [h, t] <= F_r(h0) - grad_r · h0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for idx, k in enumerate(active_lags):
        grad = compute_grads_row(h0, k)
        F_r = h0 @ (1.0 - np.roll(h0, -k))  # Wait, this isn't right for finite arrays
        # Let me compute F_k properly
        i_start = max(0, -k)
        i_end = min(n, n - k)
        F_r = np.sum(h0[i_start:i_end] * (1.0 - h0[i_start+k:i_end+k]))
        A_ub[idx, :n] = grad
        A_ub[idx, n] = -1.0
        b_ub[idx] = F_r - grad @ h0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, t unbounded below (but will be positive)
    bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None
