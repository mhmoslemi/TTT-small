    dx = 2.0 / n

        c0 = interp_to_length(parent, parent_n, n)
        cands.append(c0)
        cands.append(symmetry_transform(c0))
        for sigma in sigmas:
            for _ in range(3):
                cands.append(random_perturb(c0, sigma, rng))

        cands_scored = []
        seen = set()
        for c in cands:
            c = make_feasible(c)
            if not np.all(np.isfinite(c)):
                continue
            key = tuple(np.round(c, 6))
            if key in seen:
                continue
            seen.add(key)
            sc = official_score(c)
            if np.isfinite(sc):
                cands_scored.append((sc, c))
        cands_scored.sort(key=lambda x: x[0])
        cands = [c for _, c in cands_scored[:8]]

        for c in cands:
            try_update(c)

        for ci in range(min(5, len(cands))):
            if time.time() > deadline - 15:
                break
            cur = cands[ci].copy()
            for beta in betas:
                if time.time() > deadline - 10:
                    break
                maxiter = max(50, min(300, int(25000 / max(n, 10))))
                cur = smooth_solve(cur, n, beta, maxiter=maxiter, ftol=1e-10)
                try_update(cur)

        if time.time() > deadline - 20:
            break
        best_for_n = cands[0].copy()
        bfn_score = official_score(best_for_n)
        for c in cands[1:]:
            sc = official_score(c)
            if sc < bfn_score:
                best_for_n = c
                bfn_score = sc
        maxiter_d = max(60, min(200, int(20000 / max(n, 10))))
        dr = direct_solve(best_for_n, n, maxiter=maxiter_d, ftol=1e-10)
        try_update(dr)

    if time.time() < deadline - 30:
        best_n = len(best_h)
        for sigma in [0.01, 0.03, 0.05]:
            if time.time() > deadline - 20:
                break
            for _ in range(4):
                if time.time() > deadline - 15:
                    break
                c = random_perturb(best_h, sigma, rng)
                cur = c
                for beta in [100, 500, 2000]:
                    if time.time() > deadline - 10:
                        break
                    cur = smooth_solve(cur, best_n, beta, maxiter=120, ftol=1e-11)
                try_update(cur)

    if time.time() < deadline - 15:
        best_n = len(best_h)
        dr = direct_solve(best_h, best_n, maxiter=200, ftol=1e-12)
        try_update(dr)

    if time.time() < deadline - 10:
        best_n = len(best_h)
        alt_n = best_n + 4 if best_n + 4 <= 80 else best_n - 4
        if alt_n >= 20:
            alt_h = interp_to_length(best_h, best_n, alt_n)
            for beta in [200, 1000]:
                if time.time() > deadline - 8:
                    break
                alt_h = smooth_solve(alt_h, alt_n, beta, maxiter=150, ftol=1e-11)
            try_update(alt_h)

    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if not np.isfinite(final_score) or final_score < 0:
        if parent is not None:
            best_h = parent.copy()
        else:
            best_h = make_feasible(np.full(parent_n, 0.5))
        final_score = official_score(best_h)

    return best_h, final_score, len(best_h)
