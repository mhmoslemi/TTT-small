    eps = 1e-5
    G = np.zeros((len(raw_vals), n))
    for j in range(n):
        h[j] += eps
        r_plus = np.correlate(h, 1.0 - h, mode="full")
        h[j] -= 2 * eps
        r_minus = np.correlate(h, 1.0 - h, mode="full")
        h[j] += eps
        G[:, j] = (r_plus - r_minus) / (2 * eps)

    # Select active constraints (top correlations)
    n_active = min(15, len(raw_vals))
    top_idx = np.argsort(raw_vals)[-n_active:]

    x = cp.Variable(n)
    q = cp.Variable()
    constraints = [
        cp.sum(x) == 0,
        x >= -h - 0.05,
        x <= 1.0 - h + 0.05,
    ]
    for r in top_idx:
        constraints.append(raw_vals[r] + G[r, :] @ x <= q)

    prob = cp.Problem(cp.Minimize(q), constraints)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve()
        except Exception:
            return h, current_c

    if x.value is None or not np.all(np.isfinite(x.value)):
        return h, current_c

    delta = np.array(x.value, dtype=float)
    if np.max(np.abs(delta)) < 1e-10:
        return h, current_c

    # Line search along delta
    max_step = 1.0
    best_c = current_c
    best_alpha = 0.0
    alphas = [0.1, 0.25, 0.5, 0.75, 1.0]
    for alpha in alphas:
        h_new = h + alpha * max_step * delta
        h_new = project_feas(h_new, n)
        c = score_c5(h_new)
        if c < best_c:
            best_c = c
            best_alpha = alpha

    if best_alpha > 0:
        h_new = h + best_alpha * max_step * delta
        h = project_feas(h_new, n)
        current_c = score_c5(h)

    return h, current_c

def perturb_and_reoptimize(h, current_c, deadline, seed):
    """Perturb h randomly, then re-optimize with pair sweep."""
    n = len(h)
    rng = np.random.default_rng(seed)
    h_pert = h.copy()

    # Random perturbation
    n_pert = max(3, n // 10)
    idx = rng.choice(n, size=n_pert, replace=False)
    h_pert[idx] += rng.normal(0, 0.1, size=n_pert)
    h_pert = project_feas(h_pert, n)
    c_pert = score_c5(h_pert)

    # Re-optimize
    sub_deadline = min(deadline, time.time() + 30)
    h_pert, c_pert = pair_sweep(h_pert, c_pert, sub_deadline, min_gain=1e-11, max_passes=5)

    return h_pert, c_pert

def try_dimension(h, current_c, target_n, deadline):
    """Resample to target_n and optimize."""
    if target_n < 10 or target_n > 200:
        return h, current_c
    h_new = resample_h(h, target_n)
    c_new = score_c5(h_new)
    sub_deadline = min(deadline, time.time() + 20)
    h_new, c_new = pair_sweep(h_new, c_new, sub_deadline, min_gain=1e-11, max_passes=5)
    return h_new, c_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # Get initial values
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h = np.full(95, 0.5)
        h = project_feas(h, 95)

    n0 = len(h)
    h = project_feas(h, n0)
    current_c = score_c5(h)
    best_h = h.copy()
    best_c = current_c

    # Phase 1: Main pair sweep optimization from initial
    d1 = min(deadline, start_time + budget_s * 0.5)
    h, current_c = pair_sweep(h, current_c, d1, min_gain=1e-12, max_passes=30)
    if current_c < best_c:
        best_c = current_c
        best_h = h.copy()

    # Phase 2: Gradient step
    if time.time() < deadline - 60:
        h, current_c = try_gradient_step(h, current_c, deadline)
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()

    # Phase 3: More pair sweep from improved state
    d3 = min(deadline, start_time + budget_s * 0.65)
    h, current_c = pair_sweep(h, current_c, d3, min_gain=1e-13, max_passes=20)
    if current_c < best_c:
        best_c = current_c
        best_h = h.copy()

    # Phase 4: Perturbation + re-optimization cycles
    rng = np.random.default_rng(seed)
    n_pert_cycles = 8
    for i in range(n_pert_cycles):
        if time.time() > deadline - 30:
            break
        h_try, c_try = perturb_and_reoptimize(best_h, best_c, deadline - 30, seed + i * 101)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 5: Dimension exploration
    parent_n = len(best_h)
    dim_options = sorted(set(
        [parent_n - 4, parent_n - 2, parent_n, parent_n + 2, parent_n + 4] +
        [50, 60, 70, 80, 90, 96, 100, 110, 120]
    ))
    for target_n in dim_options:
        if time.time() > deadline - 20:
            break
        if target_n == parent_n:
            continue
        h_try, c_try = try_dimension(best_h, best_c, target_n, deadline - 20)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 6: Final refinement on best
    if time.time() < deadline - 5:
        h_final = best_h.copy()
        h_final, c_final = pair_sweep(h_final, best_c, deadline, min_gain=1e-13, max_passes=10)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    # Final validation
    best_h = project_feas(best_h, len(best_h))
    best_c = score_c5(best_h)

    # Sanity check
    n_final = len(best_h)
    assert np.all(np.isfinite(best_h)), "Non-finite values"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "Out of bounds"
    assert abs(best_h.sum() * 2.0 / n_final - 1.0) < 1e-8, "Sum constraint violated"

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = score_c5(best_h)

    return best_h, best_c, len(best_h)
