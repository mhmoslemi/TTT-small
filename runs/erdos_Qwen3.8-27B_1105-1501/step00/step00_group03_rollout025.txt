import numpy as np
import time
from scipy.optimize import minimize

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(z, n):
    target = n / 2.0
    h = np.clip(z, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    # Bisection
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        trial = np.clip(z + mid, 0.0, 1.0)
        if np.sum(trial) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine fix
    diff = target - np.sum(h)
    if diff > 0:
        mask = h < 1.0 - 1e-15
        if mask.any():
            room = (1.0 - h) * mask
            h[mask] = np.minimum(h[mask] + diff * room[mask] / room.sum(), 1.0)
    elif diff < 0:
        mask = h > 1e-15
        if mask.any():
            avail = h * mask
            h[mask] = np.maximum(h[mask] + diff * avail[mask] / avail.sum(), 0.0)
    return h

def corr_value_and_grad(h, i):
    """Compute correlation value and gradient for output index i."""
    n = len(h)
    k = i - (n - 1)
    if k == 0:
        val = np.sum(h * (1.0 - h))
        grad = 1.0 - 2.0 * h
        return val, grad
    elif k > 0:
        # c = sum_{m=0}^{n-1-k} h[m+k]*(1-h[m])
        m = np.arange(n - k)
        val = np.sum(h[m+k] * (1.0 - h[m]))
        grad = np.zeros(n)
        grad[m+k] += (1.0 - h[m])
        grad[m] += (-h[m+k])
        return val, grad
    else:
        # k < 0, let j = -k
        j = -k
        m = np.arange(n - j)
        val = np.sum(h[m] * (1.0 - h[m+j]))
        grad = np.zeros(n)
        grad[m] += (1.0 - h[m+j])
        grad[m+j] += (-h[m])
        return val, grad

def sol_slsqp(h_init, maxiter=40):
    """SLSQP optimization with minimax reformulation."""
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h_init = project_feasible(h_init, n)
    score_init = official_score(h_init)
    
    def obj(x):
        return x[n]  # theta
    
    def obj_jac(x):
        jac = np.zeros(n + 1)
        jac[n] = 1.0
        return jac
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    # Precompute correlation gradients for all lags
    # For efficiency, we'll compute them on the fly
    
    def ineq_fun(x):
        h = x[:n]
        theta = x[n]
        c = np.correlate(h, 1.0 - h, mode="full")
        return theta - dx * c
    
    def ineq_jac(x):
        h = x[:n]
        theta = x[n]
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        J[:, n] = 1.0  # d(theta - dx*c)/d(theta) = 1
        for i in range(m):
            _, grad_i = corr_value_and_grad(h, i)
            J[i, :n] = -dx * grad_i
        return J
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    bounds = [(0.0, 1.0)] * n + [(max(score_init * 0.5, 0.01), 1.0)]
    x0 = np.concatenate([h_init, [score_init * 1.01]])
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = project_feasible(res.x[:n], n)
            return h_out
    except Exception:
        pass
    return None

def active_gradient_descent(h, n_dirs=8, max_iters=20, tol=1e-10):
    """Gradient descent using active correlation directions."""
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_score = official_score(h)
    
    for it in range(max_iters):
        c = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(c)
        active_thresh = max_c - 0.1 * max_c  # active if within 10% of max
        active_indices = np.where(c >= active_thresh)[0]
        
        # Compute gradient of max (approximated by active set average)
        grad = np.zeros(n)
        for i in active_indices:
            _, g_i = corr_value_and_grad(h, i)
            grad += g_i
        grad *= dx / len(active_indices)
        
        # Negative gradient direction
        d = -grad
        d -= np.mean(d)  # preserve sum
        d_norm = np.max(np.abs(d))
        if d_norm < 1e-14:
            break
        d /= d_norm
        
        # Line search
        best_alpha = 0
        best_local = best_score
        for alpha in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5, 0.8]:
            for sign in [1.0, -1.0]:
                trial = project_feasible(h + sign * alpha * d, n)
                s = official_score(trial)
                if s < best_local - tol:
                    best_local = s
                    best_alpha = sign * alpha
                    best_h = trial
        if best_alpha == 0:
            break
        h = best_h
        best_score = best_local
    
    return h, best_score
