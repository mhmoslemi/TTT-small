        if max_a < 1e-11:
            continue

        F_new = F.copy()
        g_rev = (1.0 - h)[::-1]
        F_new[inc:inc + n] += max_a * g_rev
        F_new[dec:dec + n] -= max_a * g_rev
        F_new[n - 1 - inc:n - 1 - inc + n] -= max_a * h
        F_new[n - 1 - dec:n - 1 - dec + n] += max_a * h
        F_new[n - 1] -= 2.0 * max_a * max_a
        i1 = n - 1 + dec - inc
        if 0 <= i1 < 2 * n - 1:
            F_new[i1] += max_a * max_a
        i2 = n - 1 + inc - dec
        if 0 <= i2 < 2 * n - 1:
            F_new[i2] += max_a * max_a

        new_sc = float(np.max(F_new)) * dx
        if new_sc < cur - 1e-15:
            h[inc] += max_a
            h[dec] -= max_a
            F = F_new
            cur = new_sc
            if cur < best:
                best = cur
                best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 2000:
                break
        if step % 200 == 0:
            F = compute_F(h)
            cur = float(np.max(F)) * dx
            if cur < best:
                best = cur
                best_h = h.copy()

    return best_h, best

def make_random_start(n, rng):
    h = rng.random(n)
    return fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 20

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        if not np.all(np.isfinite(parent_h)) or len(parent_h) < 4:
            parent_h = None
    except Exception:
        parent_h = None

    # Build list of (n, start_h, time_fraction)
    configs = []

    if parent_h is not None:
        pn = len(parent_h)
        parent_h = fix_sum(parent_h, pn)
        configs.append((pn, parent_h, 0.50))

    other_ns = [50, 60, 70, 80, 90, 100, 110, 120, 140]
    available = []
    for cn in other_ns:
        if parent_h is not None and cn == len(parent_h):
            continue
        if parent_h is not None:
            start = fix_sum(interp_to_n(parent_h, cn), cn)
        else:
            start = make_random_start(cn, rng)
        available.append((cn, start))

    n_configs = len(available)
    if n_configs > 0:
        rem_time = 1.0 - sum(c[2] for c in configs)
        per = rem_time / n_configs
        for cn, start in available:
            configs.append((cn, start, per))

    configs.sort(key=lambda x: -x[2])

    for n, h_start, frac in configs:
        if time.time() > deadline - 10:
            break
        t_end_cfg = t_start + frac * (budget_s - 25)
        t_end_cfg = min(t_end_cfg, deadline - 5)
        if t_end_cfg <= time.time() + 3:
            continue

        h_work = h_start.copy()
        h_opt, c5_sa = sa_loop(h_work, n, rng, t_end_cfg)

        # Hill climb polish
        t_he = min(t_start + (frac + 0.02) * (budget_s - 25), deadline - 3)
        if t_he > time.time() + 2:
            h_opt2, c5_hc = hill_climb(h_opt, n, rng, t_he, max_steps=3000000)
            if c5_hc < c5_sa:
                h_opt, c5_sa = h_opt2, c5_hc

        # Verify
        F = compute_F(h_opt)
        c5_verified = float(np.max(F)) * (2.0 / n)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        if c5_verified < best_c5:
            best_c5 = c5_verified
            best_h = h_opt.copy()

    # Try a few random restarts if time remains
    if best_h is not None and time.time() < deadline - 60:
        n_start = len(best_h)
        t_rem = deadline - 60 - time.time()
        n_tries = min(4, int(t_rem // 25))
        for trial in range(n_tries):
            if time.time() > deadline - 30:
                break
            h_rand = make_random_start(n_start, rng)
            h_cv = fix_sum(interp_to_n(best_h, n_start), n_start)
            # Mix random and best
            mix_rng = rng.random(n_start)
            h_mix = 0.5 * h_rand + 0.5 * h_cv
            h_mix = fix_sum(h_mix, n_start)
            t_trial_end = min(time.time() + t_rem / (n_tries - trial), deadline - 5)
            h_opt, c5_opt = sa_loop(h_mix, n_start, rng, t_trial_end, T0=0.005)
            F = compute_F(h_opt)
            c5_v = float(np.max(F)) * (2.0 / n_start)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h_opt.copy()

    if best_h is None:
        best_h = np.zeros(50)
        best_h[::2] = 1.0
        best_h = fix_sum(best_h, 50)
        F = compute_F(best_h)
        best_c5 = float(np.max(F)) * 0.04

    n_final = len(best_h)
    return best_h, best_c5, n_final
