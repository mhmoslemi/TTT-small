```python
import numpy as np
import time
from scipy.optimize import minimize

def safe_logistic(x):
    x = np.asarray(x, dtype=np.float64)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def solve_tau(z, target_sum):
    lo = -300.0
    hi = 300.0
    for _ in range(150):
        mid = 0.5 * (lo + hi)
        s = np.sum(safe_logistic(z + mid))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def raw_to_h(y, n):
    z = np.empty(n, dtype=np.float64)
    z[:n-1] = y
    z[n-1] = -np.sum(y)
    tau = solve_tau(z, n * 0.5)
    h = safe_logistic(z + tau)
    return h

def h_to_raw(h, n):
    h_c = np.clip(h, 1e-12, 1.0 - 1e-12)
    logit = np.log(h_c) - np.log(1.0 - h_c)
    return logit[:n-1].astype(np.float64).copy()

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    cf = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cf) * dx)

def compute_F(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def obj_grad(y, n, beta):
    """Compute smoothed objective and its gradient w.r.t. raw variables y."""
    h = raw_to_h(y, n)
    F = compute_F(h)
    Fmax = np.max(F)
    
    # Softmax weights for the max
    shifted = beta * (F - Fmax)
    max_shifted = np.max(shifted)
    exp_s = np.exp(shifted - max_shifted)
    alpha = exp_s / np.sum(exp_s)
    
    # Gradient of smoothed objective w.r.t. h
    dx = 2.0 / n
    grad_h = np.zeros(n, dtype=np.float64)
    for m in range(2 * n - 1):
        w = alpha[m] * dx
        if m == 0:
            grad_h += w * (1.0 - 2.0 * h)
        elif m < n:
            grad_h[:n-m] += w * (1.0 - h[m:])
            grad_h[m:] += w * (-h[:n-m])
        else:
            s = m - (n - 1)
            grad_h[s:] += w * (1.0 - h[:n-s])
            grad_h[:n-s] += w * (-h[s:])
    
    # Smoothed objective value
    psi = Fmax + (max_shifted + np.log(np.sum(exp_s))) / beta
    
    # Chain rule through logistic + tau mapping
    d = h * (1.0 - h)
    D = np.sum(d)
    if D < 1e-14:
        return psi, np.zeros(n - 1, dtype=np.float64)
    
    A = np.dot(grad_h, d)
    d_last = d[n-1]
    gh_last = grad_h[n-1]
    grad_y = np.empty(n - 1, dtype=np.float64)
    for j in range(n - 1):
        grad_y[j] = (grad_h[j] * d[j] - gh_last * d_last) + (d_last - d[j]) * A / D
    
    return psi, grad_y
```

def dyadic_snap(h, n, bits=20):
    Q = 1 << bits
    target = n * Q // 2
    K = np.round(h * Q).astype(np.int64)
    K = np.clip(K, 0, Q)
    residual = target - int(np.sum(K))
    
    if residual > 0:
        # Need to add `residual` units
        # Prioritize indices with most room (K_i < Q) and closest to target
        order = np.arange(n)[K < Q]
        # Sort by how close h_i is to (K_i + 1)/Q (i.e., adding helps)
        # Or just add to indices with largest h_i
        h_sorted = np.argsort(h)[::-1]
        count = 0
        for idx in h_sorted:
            if K[idx] < Q and count < residual:
                K[idx] += 1
                count += 1
            if count >= residual:
                break
    elif residual < 0:
        # Need to subtract |residual| units
        r = -residual
        h_sorted = np.argsort(h)
        count = 0
        for idx in h_sorted:
            if K[idx] > 0 and count < r:
                K[idx] -= 1
                count += 1
            if count >= r:
                break
    
    h_snap = K.astype(np.float64) / Q
    return h_snap

def pair_line_search(h, n, max_pairs=200, samples=15, max_rounds=3):
    """Try to improve h by moving mass between pairs of positions."""
    best_h = h.copy()
    best_c = eval_c5(best_h)
    
    for rnd in range(max_rounds):
        improved = False
        # Generate candidate pairs
        rng = np.random.default_rng(42 + rnd)
        pairs = set()
        
        # Sort by value
        order_desc = np.argsort(h)[::-1]
        order_asc = np.argsort(h)
        
        # Top-value to low-value pairs
        for i in range(min(10, n)):
            for j in range(min(10, n)):
                p = order_desc[i]
                q = order_asc[j]
                if p != q:
                    pairs.add((int(p), int(q)))
        
        # Random pairs
        for _ in range(min(100, max_pairs)):
            p = rng.integers(0, n)
            q = rng.integers(0, n)
            if p != q:
                pairs.add((int(p), int(q)))
        
        for (p, q) in list(pairs)[:max_pairs]:
            # Try moving mass from p to q (both directions)
            t_low = max(-best_h[q], best_h[p] - 1.0)
            t_high = min(best_h[p], 1.0 - best_h[q])
            
            if t_high <= t_low:
                continue
            
            # Sample t values
            ts = np.linspace(t_low, t_high, samples)
            ts = np.concatenate([ts, [t_low, t_high]])
            
            for t in ts:
                h_new = best_h.copy()
                h_new[p] -= t
                h_new[q] += t
                if np.any(h_new < -1e-10) or np.any(h_new > 1 + 1e-10):
                    continue
                c = eval_c5(h_new)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = h_new.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c

def obj_grad(y, n, beta):
    h = raw_to_h(y, n)
    F = compute_F(h)
    Fmax = np.max(F)
    
    shifted = beta * (F - Fmax)
    max_shifted = np.max(shifted)
    exp_s = np.exp(shifted - max_shifted)
    alpha = exp_s / np.sum(exp_s)
    
    psi = Fmax + (max_shifted + np.log(np.sum(exp_s))) / beta
    
    dx = 2.0 / n
    grad_h = np.zeros(n, dtype=np.float64)
    
    for m in range(2 * n - 1):
        w = alpha[m] * dx
        if w < 1e-18:
            continue
        k = m - (n - 1)
        if k == 0:
            grad_h += w * (1.0 - 2.0 * h)
        elif k > 0:
            s = k
            L = n - s
            grad_h[:L] += w * (1.0 - h[s:s+L])
            grad_h[s:s+L] += w * (-h[:L])
        else:
            s = -k
            L = n - s
            grad_h[s:s+L] += w * (1.0 - h[:L])
            grad_h[:L] += w * (-h[s:s+L])
    
    # Chain rule
    d = h * (1.0 - h)
    D = np.sum(d)
    if D < 1e-14:
        return psi, np.zeros(n-1)
    
    A = np.dot(grad_h, d)
    d_last = d[n-1]
    gh_last = grad_h[n-1]
    grad_y = np.empty(n-1, dtype=np.float64)
    for j in range(n-1):
        grad_y[j] = (grad_h[j]*d[j] - gh_last*d_last) + (d_last - d[j]) * A / D
    
    return psi, grad_y
