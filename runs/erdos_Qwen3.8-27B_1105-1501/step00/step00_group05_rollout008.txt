    n_candidates = [40, 44, 46, 48, 50, 52, 38, 36, 56, 60]
    for n_try in n_candidates:
        if time_left() < 25:
            break
        
        h_interp = interpolate_h(best_h, n0, n_try)
        h_t, _ = greedy_optimize(h_interp.copy(), n_try, max_passes=200, tol=1e-14, num_evals=40)
        C_t = evaluate_c5(h_t)
        if C_t < best_C:
            best_h = h_t
            best_C = C_t
            n0 = n_try  # Update for subsequent interpolations
    
    # Phase 5: Random restarts at n0
    if time_left() > 40:
        for attempt in range(5):
            if time_left() < 20:
                break
            h_rand = random_h(n0, rng)
            # Perturb the best solution instead of purely random
            h_pert = best_h + rng.normal(0, 0.05, n0)
            h_pert = np.clip(h_pert, 0.0, 1.0)
            repair_sum(h_pert, n0)
            
            h_r, _ = greedy_optimize(h_pert.copy(), n0, max_passes=100, tol=1e-13, num_evals=30)
            C_r = evaluate_c5(h_r)
            if C_r < best_C:
                best_h = h_r
                best_C = C_r
    
    # Phase 6: Final fine polish
    if time_left() > 10:
        n_best = len(best_h)
        h_final, _ = greedy_optimize(best_h.copy(), n_best, max_passes=500, tol=1e-15, num_evals=60)
        C_final = evaluate_c5(h_final)
        if C_final < best_C:
            best_h = h_final
            best_C = C_final
    
    # Final: ensure constraint is satisfied exactly
    n_final = len(best_h)
    repair_sum(best_h, n_final)
    final_C = evaluate_c5(best_h)
    
    return best_h, final_C, n_final
