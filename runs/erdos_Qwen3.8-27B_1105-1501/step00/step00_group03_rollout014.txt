                    cand_pool.append((sc_opt, h_opt.copy(), n_c))

        cand_pool.sort(key=lambda t: t[0])
        top2 = [(s, h, n) for s, h, n in cand_pool if n == res_n][:3]

        for (sc, h_c, n_c) in top2:
            if time.time() > deadline:
                break
            maxit = max(60, int(4000 / max(20, n_c)))
            h_opt = try_direct(h_c, n_c, maxit)
            if h_opt is not None:
                sc_opt = official_score(h_opt)
                if sc_opt < best_score - 1e-14:
                    best_score = sc_opt
                    best_h = h_opt.copy()
                    best_n = n_c
                cand_pool.append((sc_opt, h_opt.copy(), n_c))

    # Final polish rounds
    for polish_round in range(3):
        if time.time() > deadline - 15:
            break
        n_best = best_n
        # Smooth polish with high beta
        h_opt = try_smooth(best_h, n_best, 2000, 150)
        if h_opt is not None:
            sc_opt = official_score(h_opt)
            if sc_opt < best_score - 1e-14:
                best_score = sc_opt
                best_h = h_opt.copy()
                best_n = n_best
        # Direct polish
        h_opt = try_direct(best_h, n_best, 200)
        if h_opt is not None:
            sc_opt = official_score(h_opt)
            if sc_opt < best_score - 1e-14:
                best_score = sc_opt
                best_h = h_opt.copy()
                best_n = n_best

    # Coordinate refinement
    if time.time() < deadline - 5:
        n_ref = best_n
        improved = True
        max_rounds = 20
        for _ in range(max_rounds):
            if not improved or time.time() > deadline - 5:
                break
            improved = False
            for _ier in range(5):
                if time.time() > deadline - 5:
                    break
                h_try = best_h.copy()
                idx = rng.integers(0, n_ref)
                for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                    h_try = best_h.copy()
                    h_try[idx] += delta
                    h_try = make_feasible(h_try)
                    sc_try = official_score(h_try)
                    if sc_try < best_score - 1e-14:
                        best_score = sc_try
                        best_h = h_try
                        best_n = n_ref
                        improved = True
                        break

    best_h = np.clip(best_h, 0.0, 1.0)
    c5_bound = official_score(best_h)
    c5_bound = c5_bound + 1e-12
    return best_h, c5_bound, int(best_n)
