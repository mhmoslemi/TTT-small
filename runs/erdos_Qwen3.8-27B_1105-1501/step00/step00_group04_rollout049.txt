- If i is in the range [d, n-1], then h[i] appears as h[i-d] (since i = j+d where j = i-d), so partial/dh[i] = dx * 1 * (1-h[i-d])

So:
- For i in [0, n-1-d]: grad[i] += dx * (-h[i+d]) * weight
- For i in [d, n-1]: grad[i] += dx * (1-h[i-d]) * weight

Similarly for c[center-d] (d > 0):
c[center-d] = dx * sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])

- For i in [0, n-1-d]: h[i] appears directly, grad[i] += dx * (1-h[i+d]) * weight
- For i in [d, n-1]: h[i] appears as h[i-d] in (1-h[i-d]), grad[i] += dx * (-h[i-d]) * weight

For d=0:
c[center] = dx * sum_i h[i]*(1-h[i])
partial/dh[i] = dx * (1 - 2h[i])

Now let me code this up efficiently. The key is to vectorize the gradient computation.

For the smooth-max proxy:
- Compute all scores c_k
- base = max(c)
- weights = exp((c - base) / T) / sum(exp((c-base)/T))
- smooth = base + T * log(sum(exp((c-base)/T)))
- gradient = sum_k weights[k] * grad(c_k)

The gradient computation for a single lag d involves O(n) operations, and we have O(n) lags, so total is O(n²). For n=70, that's about 5000 operations per gradient evaluation, which is fast.

import numpy as np

def make_feasible(y, target=None):
    """Project y onto the feasible set [0,1]^n with sum = target (default n/2)."""
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    # Handle non-finite values
    y = np.where(np.isfinite(y), y, 0.5)
    
    # Bisection for lambda
    lo = np.min(y) - 1.0
    hi = np.max(y) + 0.0  # actually hi should be max(y) since at lambda=max(y), all clips to 0
    # Actually: clip(y - lambda, 0, 1). At lambda = min(y)-1, all are clipped to 1, sum = n.
    # At lambda = max(y), all are 0, sum = 0.
    # We want sum = target.
    
    # More careful:
    # phi(lambda) = sum(clip(y_i - lambda, 0, 1))
    # phi is non-increasing in lambda
    # phi(min(y) - 1) = n (all y_i - lambda >= 1, clip to 1)
    # phi(max(y)) = 0 (all y_i - lambda <= 0, clip to 0)
    
    lo = np.min(y) - 1.0
    hi = np.max(y)
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix small numerical error
    diff = target - np.sum(z)
    if abs(diff) > 1e-12:
        # Find elements with slack
        if diff > 0:
            # Need to increase; find elements < 1
            slack = 1.0 - z
            idx = np.argsort(-slack)[:max(1, int(abs(diff) + 1))]
            for j in idx:
                inc = min(slack[j], diff)
                if inc > 0:
                    z[j] += inc
                    diff -= inc
                    if abs(diff) < 1e-14:
                        break
        else:
            # Need to decrease; find elements > 0
            slack = z
            idx = np.argsort(slack)[:max(1, int(-abs(diff) + 1))]
            for j in idx:
                dec = min(slack[j], -diff)
                if dec > 0:
                    z[j] -= dec
                    diff += dec
                    if abs(diff) < 1e-14:
                        break
    
    return z


def score_exact(h):
    """Compute the exact C5 score."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def score_vector(h):
    """Return the full scaled score vector."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def smooth_and_grad(h, T):
    """Compute smooth-max surrogate value and gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(c)
    center = n - 1
    
    base = np.max(c)
    ex = np.exp((c - base) / T)
    S = np.sum(ex)
    if S < 1e-300:
        weights = np.zeros(L)
        weights[np.argmax(c)] = 1.0
        smooth_val = base
    else:
        weights = ex / S
        smooth_val = base + T * np.log(S)
    
    grad = np.zeros(n)
    
    # d = 0 (center)
    w0 = weights[center]
    grad += w0 * dx * (1.0 - 2.0 * h)
    
    # d = 1, ..., n-1
    for d in range(1, n):
        n_active = n - d
        # Positive lag: c[center + d] = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        wp = weights[center + d]
        if wp > 1e-15:
            grad[0:n_active] += wp * dx * (-h[d:])
            grad[d:n] += wp * dx * (1.0 - h[0:n_active])
        
        # Negative lag: c[center - d] = dx * sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
        wn = weights[center - d]
        if wn > 1e-15:
            grad[0:n_active] += wn * dx * (1.0 - h[d:])
            grad[d:n] += wn * dx * (-h[0:n_active])
    
    return smooth_val, grad

def binarize_top(h):
    """Create a binary vector by setting top n/2 entries to 1."""
    n = len(h)
    target = n // 2
    if n % 2 == 0:
        # Exactly n/2 ones
        idx = np.argsort(-h, kind='stable')[:target]
        z = np.zeros(n)
        z[idx] = 1.0
    else:
        # n/2 isn't integer; use floor and add fractional
        target_floor = n // 2
        idx = np.argsort(-h, kind='stable')[:target_floor]
        z = np.zeros(n)
        z[idx] = 1.0
        # Need to add 0.5 to one element
        rest = np.setdiff1d(np.arange(n), idx)
        if len(rest) > 0:
            z[rest[0]] = 0.5
    return make_feasible(z)


def resample(h, new_n):
    """Resample h from its current resolution to new_n points."""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / new_n
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new
    # Linear interpolation with clamping
    y_new = np.interp(x_new, x_old, h)
    return make_feasible(y_new)


def pair_line_search(h, i, j, max_evals=30):
    """Exact line search along direction e_i - e_j (increase h[i], decrease h[j])."""
    # h_i(alpha) = h[i] + alpha, h_j(alpha) = h[j] - alpha
    # alpha in [lo, hi]
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h, score_exact(h)
    
    best_alpha = 0.0
    best_score = score_exact(h)
    
    # Try a grid of alpha values
    alphas = np.linspace(lo, hi, max_evals)
    for a in alphas:
        h_trial = h.copy()
        h_trial[i] += a
        h_trial[j] -= a
        s = score_exact(h_trial)
        if s < best_score - 1e-14:
            best_score = s
            best_alpha = a
    
    if best_alpha == 0.0:
        return h, best_score
    
    # Refine with golden section around best_alpha
    # Find the interval containing the best point
    idx_best = int(np.argmin(np.abs(alphas - best_alpha)))
    a_lo = alphas[max(0, idx_best-1)]
    a_hi = alphas[min(len(alphas)-1, idx_best+1)]
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(15):
        a1 = a_hi - gr * (a_hi - a_lo)
        a2 = a_lo + gr * (a_hi - a_lo)
        h1 = h.copy(); h1[i] += a1; h1[j] -= a1
        h2 = h.copy(); h2[i] += a2; h2[j] -= a2
        s1 = score_exact(h1)
        s2 = score_exact(h2)
        if s1 < s2:
            a_hi = a2
        else:
            a_lo = a1
    
    a_best = (a_lo + a_hi) / 2
    h_best = h.copy()
    h_best[i] += a_best
    h_best[j] -= a_best
    s_best = score_exact(h_best)
    
    if s_best < best_score - 1e-14:
        return h_best, s_best
    return h, best_score


def local_refine(h_init, deadline, rng):

                break
        
        grad_norm = np.sqrt(np.mean(grad**2))
        if grad_norm < 1e-12:
            # Try pair search
            pass
        else:

            improved = False

                h_trial = make_feasible(h_trial)

                    # Perturb
                    h_noise = h + 0.01 * rng.standard_normal(n)
                    h = make_feasible(h_noise)

        if iter_count % 10 == 5:

import numpy as np

def make_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64).copy()
    y = np.where(np.isfinite(y), y, 0.5)
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - np.sum(z)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - z)
            for j in idx:
                inc = min(1.0 - z[j], diff)
                if inc > 1e-15:
                    z[j] += inc
                    diff -= inc
                    if abs(diff) < 1e-14:
                        break
        else:
            idx = np.argsort(z)
            for j in idx:
                dec = min(z[j], -diff)
                if dec > 1e-15:
                    z[j] -= dec
                    diff += dec
                    if abs(diff) < 1e-14:
                        break
    return z

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def score_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(c)
    center = n - 1
    base = np.max(c)
    ex = np.exp((c - base) / T)
    S = np.sum(ex)
    if S < 1e-300:
        w = np.zeros(L)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / S
    grad = np.zeros(n)
    grad += w[center] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[center + d]
        if wp > 1e-15:
            grad[:m] += wp * dx * (-h[d:])
            grad[d:] += wp * dx * (1.0 - h[:m])
        wn = w[center - d]
        if wn > 1e-15:
            grad[:m] += wn * dx * (1.0 - h[d:])
            grad[d:] += wn * dx * (-h[:m])
    smooth_val = base + T * np.log(S) if S > 0 else base
    return smooth_val, grad

def pair_line(h, i, j, n_grid=20):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h, score_exact(h)
    best_a = 0.0
    best_s = score_exact(h)
    alphas = np.linspace(lo, hi, n_grid)
    for a in alphas:
        h[i] += a
        h[j] -= a
        s = score_exact(h)
        h[i] -= a
        h[j] += a
        if s < best_s:
            best_s = s
            best_a = a
    if best_a == 0.0:
        return h, best_s
    idx_b = int(np.argmin(np.abs(alphas - best_a)))
    a_lo = alphas[max(0, idx_b - 1)]
    a_hi = alphas[min(len(alphas) - 1, idx_b + 1)]
    gr = (math.sqrt(5) - 1) / 2
    for _ in range(12):
        a1 = a_hi - gr * (a_hi - a_lo)
        a2 = a_lo + gr * (a_hi - a_lo)
        h[i] += a1; h[j] -= a1
        s1 = score_exact(h)
        h[i] -= a1; h[j] += a1
        h[i] += a2; h[j] -= a2
        s2 = score_exact(h)
        h[i] -= a2; h[j] += a2
        if s1 < s2:
            a_hi = a2
        else:
            a_lo = a1
    a_best = (a_lo + a_hi) / 2.0
    h[i] += a_best
    h[j] -= a_best
    s_final = score_exact(h)
    if s_final < best_s:
        return h, s_final
    h[i] -= a_best
    h[j] += a_best
    return h, best_s

def full_pair_sweep(h, deadline, top_k=12, n_grid=16):

    cur_s = score_exact(h)
    best_s = cur_s

def binarize_top(h):
    n = len(h)
    target = n // 2
    z = np.zeros(n)
    if n % 2 == 0:
        idx = np.argsort(-h, kind='stable')[:target]
        z[idx] = 1.0
    else:
        idx = np.argsort(-h, kind='stable')[:target]
        z[idx] = 1.0
        rest = np.setdiff1d(np.arange(n), idx)
        if len(rest) > 0:
            z[rest[np.argmax(-h[rest])]] = 1.0 - target / (n / 2.0)
    return make_feasible(z)

def resample(h, new_n):
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / new_n
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new
    return make_feasible(np.interp(x_new, x_old, h))

def make_random_feasible(n, rng):
    return make_feasible(rng.random(n))

def make_block_pattern(n, rng):
    y = np.zeros(n)
    while sum(y) < n * 0.3:
        start = int(rng.integers(0, n))
        blk = int(rng.integers(3, max(4, n // 8)))
        y[start:start+blk] = 1.0
    return make_feasible(y)

def make_sine_pattern(n, phase, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    y = 0.5 + amp * 0.5 * np.sin(2 * np.pi * x / 2.0 + phase)
