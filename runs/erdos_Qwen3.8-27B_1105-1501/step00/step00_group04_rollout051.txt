import numpy as np
import time

def score_exact(h):
    """Compute C5 = max correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def score_array(h):
    """Return the full correlation array scaled by dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(y, target):
    """Project y onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(y)
    # Bisection for lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(y - lam, 0.0, 1.0)
    # Fix tiny numerical error
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Adjust the element with the most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_max_and_grad(h, T):
    """Compute smooth max surrogate and its gradient"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(c)  # 2n-1
    center = n - 1
    
    base = float(c.max())
    exps = np.exp((c - base) / T)
    S = float(exps.sum())
    if S < 1e-300:
        # Underflow: just use argmax
        weights = np.zeros(L)
        weights[c == base] = 1.0 / (L - (center*0))  # Hmm
        idx_max = np.where(c == base)[0]
        weights = np.zeros(L)
        for i in idx_max:
            weights[i] = 1.0 / len(idx_max)
    else:
        weights = exps / S
    
    smooth_val = base + T * np.log(max(S, 1e-300))
    
    # Gradient
    g = np.zeros(n)
    
    # Zero lag
    g += weights[center] * (1.0 - 2.0 * h) * dx
    
    # Positive lags k = 1, ..., n-1
    for k in range(1, n):
        wk = weights[center + k]
        if wk < 1e-15:
            continue
        # h[m] for m in [k, n-1]: contributes (1-h[m-k])
        g[k:] += wk * (1.0 - h[:n-k]) * dx
        # h[m] for m in [0, n-1-k]: contributes -h[m+k]
        g[:n-k] += -wk * h[k:] * dx
    
    # Negative lags k = 1, ..., n-1
    for k in range(1, n):
        wk = weights[center - k]
        if wk < 1e-15:

        # C_{-k} = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
        # h[m] for m in [0, n-1-k]: contributes (1-h[m+k])
        g[:n-k] += wk * (1.0 - h[k:]) * dx
        # h[m] for m in [k, n-1]: contributes -h[m-k]
        g[k:] += -wk * h[:n-k] * dx
    
    return smooth_val, g

def line_search_pair(h, i, j, n_target, max_evals=30):
    """Find best alpha for moving mass from j to i (or vice versa)"""
    # h[i] + alpha, h[j] - alpha
    # Constraints: 0 <= h[i]+alpha <= 1, 0 <= h[j]-alpha <= 1
    # So: max(-h[i], h[j]-1) <= alpha <= min(1-h[i], h[j])
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    
    if hi - lo < 1e-12:
        return h, score_exact(h)
    
    best_h = h.copy()
    best_c = score_exact(h)
    
    # Grid search
    alphas = np.linspace(lo, hi, max_evals)
    for alpha in alphas:
        h[i] += alpha
        h[j] -= alpha
        c = score_exact(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h[i] -= alpha
        h[j] += alpha
    
    return best_h, best_c

def resample(h, new_n):
    """Resample h from its resolution to new_n points"""
    n_old = len(h)
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / new_n
    x_new = (np.arange(new_n) + 0.5) * dx_new
    new_val = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    new_n_target = new_n / 2.0
    return project_feasible(new_val, new_n_target)

import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(y, target):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    center = n - 1
    base = float(c.max())
    exps = np.exp((c - base) / T)
    S = float(exps.sum())
    if S < 1e-300:
        weights = np.zeros(len(c))
        idx_max = np.where(c >= base - 1e-15)[0]
        weights[idx_max] = 1.0 / len(idx_max)
    else:
        weights = exps / S
    g = np.zeros(n)
    g += weights[center] * (1.0 - 2.0 * h) * dx
    for k in range(1, n):
        wk = weights[center + k]
        if wk > 1e-14:
            g[k:] += wk * (1.0 - h[:n-k]) * dx
            g[:n-k] += -wk * h[k:] * dx
        wk = weights[center - k]
        if wk > 1e-14:
            g[:n-k] += wk * (1.0 - h[k:]) * dx
            g[k:] += -wk * h[:n-k] * dx
    return g

def line_search_pair(h, i, j, n_target, max_evals=21):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h, score_exact(h)
    best_h = h.copy()
    best_c = score_exact(h)
    alphas = np.linspace(lo, hi, max_evals)
    for alpha in alphas:
        h[i] += alpha
        h[j] -= alpha
        c = score_exact(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        h[i] -= alpha
        h[j] += alpha
    return best_h, best_c

def resample_vector(h, new_n):
    n_old = len(h)
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / new_n
    x_new = (np.arange(new_n) + 0.5) * dx_new
    new_val = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_feasible(new_val, new_n / 2.0)

def binarize(h, rng=None):
    n = len(h)
    k = n // 2
    idx = np.argsort(h)[::-1][:k]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

            # Quick gradient refinement from perturbed point
            h = trial.copy()
            T = 0.08
            ss = 0.03
            for _ in range(80):
                if time_left() < 5.0:
                    break
                g = smooth_max_grad(h, T)
                g_norm = float(np.linalg.norm(g))
                if g_norm < 1e-14:
                    break
                accepted = False
                for _ in range(5):
                    t2 = project_feasible(h - ss * g / g_norm, n_target)
                    t2c = score_exact(t2)
                    if t2c < trial_c - 1e-12:
                        h = t2
                        trial_c = t2c
                        accepted = True
                        break
                    ss *= 0.5
                if not accepted:
                    break
                T = max(0.003, T * 0.95)
            
            if trial_c < best_c - 1e-12:
                best_c = trial_c
                best_h = h.copy()
    
    # Phase 4: Try binary + smooth
    if time_left() > 30.0:
        h_bin = binarize(best_h, rng)
        h_cur = h_bin.copy()
        T = 0.1
        ss = 0.05
        for _ in range(100):
            if time_left() < 5.0:
                break
            g = smooth_max_grad(h_cur, T)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-14:
                break
            for _ in range(5):
                t2 = project_feasible(h_cur - ss * g / g_norm, n_target)
                t2c = score_exact(t2)
                if t2c < best_c - 1e-12:
                    h_cur = t2
                    best_c = t2c
