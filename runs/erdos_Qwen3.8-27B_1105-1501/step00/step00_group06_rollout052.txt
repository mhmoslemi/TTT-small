    n = len(h)
    r = np.correlate(h, 1.0 - h, mode="full")
    return np.max(r) * 2.0 / n

def compute_subgradient(h):
    """Compute the subgradient of max(r) where r = correlate(h, 1-h, full)."""
    n = len(h)
    r = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(r)
    # Find all lags achieving the max
    active = np.where(r >= max_val - 1e-10)[0]
    g = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        # ∂c[idx]/∂h[j] = (1-h[j-lag])*[0≤j-lag≤n-1] - h[j+lag]*[0≤j+lag≤n-1]
        # For all j:

        # Term 2: -h[j+lag] if 0 ≤ j+lag ≤ n-1
        # Term 1: j ranges over [lag, n-1] if lag >= 0, or [0, n-1+lag] if lag < 0
        #   i.e., j-lag in [0, n-1] means j in [lag, n-1+lag]
        #   intersected with [0, n-1]

    n = len(h)

            r_new_est = np.max(r[None, :] + delta * diff, axis=0)  # shape (n,)

    n = len(h)

import numpy as np
import time

def evaluate_score(h):
    n = len(h)
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * 2.0 / n)

def compute_grad_matrix(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    one_minus_h = 1.0 - h
    for idx in range(2 * n - 1):
        lag = idx - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            grad[idx, lo1:hi1 + 1] += one_minus_h[lo1 - lag:hi1 - lag + 1]
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            grad[idx, lo2:hi2 + 1] -= h[lo2 + lag:hi2 + lag + 1]
    return grad

def project_to_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_descent(h, target_sum, time_limit, start_time, max_iters=5000):
    n = len(h)
    best_h = h.copy()
    best_score = evaluate_score(h)
    current_score = best_score

    for phase in range(6):
        base_step = 0.03 / (phase + 1)
        for it in range(max_iters):
            if time.time() - start_time > time_limit:
                return best_h, best_score
            step = base_step * (0.9997 ** it)
            if step < 1e-5:
                break

            r = np.correlate(h, 1.0 - h, mode="full")
            max_val = np.max(r)
            sorted_idx = np.argsort(r)[::-1]
            n_active = max(1, min(len(sorted_idx), max(1, (2 * n - 1) // 10)))
            active = sorted_idx[:n_active]

            G = compute_grad_matrix(h)
            subgrad = np.mean(G[active, :], axis=0)
            sg_norm = np.linalg.norm(subgrad)
            if sg_norm < 1e-12:
                break

            h_new = h - step * subgrad / sg_norm
            h_new = project_to_constraint(h_new, target_sum)

            new_score = evaluate_score(h_new)
            if new_score < current_score:
                h = h_new
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best_h = h.copy()

    return best_h, best_score

def fast_local_search(h, target_sum, time_limit, start_time, max_sweeps=200):
    n = len(h)
    best_h = h.copy()
    best_score = evaluate_score(h)
    current_score = best_score

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        G = compute_grad_matrix(h)
        r = np.correlate(h, 1.0 - h, mode="full")
        current_max = np.max(r)
        improved = False

        for i in range(n):
            if time.time() - start_time > time_limit:
                break
            if h[i] < 1e-8:
                continue

            for delta in [0.02, 0.05, 0.1, 0.2]:
                if h[i] < delta:
                    continue
                diff = G - G[:, i:i + 1]
                r_new_est = np.max(r[None, :] + delta * diff, axis=0)
                valid = (h + delta <= 1.0 + 1e-8)
                r_new_est[~valid] = np.inf
                r_new_est[i] = np.inf

                best_j = int(np.argmin(r_new_est))
                if r_new_est[best_j] < current_max - 1e-10:
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[best_j] += delta
                    actual = evaluate_score(h_new)
                    if actual < current_score - 1e-12:
                        h = h_new
                        current_score = actual
                        improved = True
                        if current_score < best_score:
                            best_score = current_score
                            best_h = h.copy()
                        break

        if not improved:
            break

    return best_h, best_score

def binary_local_search(h, target_sum, time_limit, start_time, max_sweeps=500):

                break

                        best_h = h.copy()
                    break
        if not improved:

        ones = np.where(h > 0.5)[0]

    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
