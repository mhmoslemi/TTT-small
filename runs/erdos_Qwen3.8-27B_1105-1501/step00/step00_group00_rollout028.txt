import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 value for a feasible h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h to the feasible set {0<=h_i<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Adjust: if we need to add mass, add to elements with slack < 1
    # If we need to remove mass, remove from elements with slack > 0
    if diff > 0:
        # Add mass
        slack = 1.0 - h
        total_slack = np.sum(slack)
        if total_slack > 0:
            h += slack * (diff / total_slack)
        else:
            # All at 1, shouldn't happen if target = n/2
            pass
    else:
        # Remove mass
        total = np.sum(h)
        if total > 0:
            h += h * (diff / total)
    # Clip again
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute remaining diff to elements not at bounds
        idx = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

    g = np.zeros(_GN + 1)
    g[:_GN] = 1.0
    return g

def _ineq(z):

    t = z[_GN]

def _ineq_jac(z):
    h = z[:_GN]
    N = _GN
    dx = 2.0 / N
    nc = 2 * N - 1
    jac = np.zeros((nc, N + 1))
    for k in range(nc):
        lag = k - (N - 1)
        if lag >= 0:
            m = N - lag
            for j in range(N):
                g = 0.0
                if lag <= j:
                    g += 1.0 - h[j - lag]
                if j < m:
                    g -= h[j + lag]
                jac[k, j] = -dx * g
        else:
            p = -lag
            m = N - p
            for j in range(N):
                g = 0.0
                if j < m:
                    g += 1.0 - h[j + p]
                if j >= p:
                    g -= h[j - p]
                jac[k, j] = -dx * g
        jac[k, N] = 1.0
    return jac

def _slsqp_solve(h, n, maxiter=300):

    _GN = n
    c5_cur = compute_c5(h)
    z0 = np.concatenate([h, [c5_cur + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac}
    ]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return h, c5_cur
    if res.x is not None:
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            return h_new, c5_new
    return h, c5_cur

def _proj_grad(h, n, rng, max_iter=5000, tol=1e-10):
    """Projected subgradient descent for min max correlation."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    alpha = 0.01
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find active set
        max_corr = np.max(corr)
        active = np.where(corr >= max_corr - 0.1)[0]
        
        # Compute subgradient (average of active gradients)
        g = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            for j in range(n):
                gg = 0.0
                if lag >= 0:
                    m = n - lag
                    if lag <= j:
                        gg += 1.0 - h[j - lag]
                    if j < m:
                        gg -= h[j + lag]
                else:
                    p = -lag
                    m = n - p
                    if j < m:
                        gg += 1.0 - h[j + p]
                    if j >= p:
                        gg -= h[j - p]
                g[j] += gg
        g /= len(active)
        
        # Step
        h = h - alpha * g
        h = project_feasible(h, n)
        
        # Adaptive step size
        new_c5 = compute_c5(h)
        if new_c5 > best_c5 + 0.01:
            alpha *= 0.5
        elif it % 100 == 0:
            alpha = min(alpha * 1.1, 0.1)
        
        if alpha < 1e-8:
            break
    
    return best_h, best_c5

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
