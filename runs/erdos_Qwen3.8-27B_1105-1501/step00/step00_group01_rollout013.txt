def _project_feasible(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} via water-filling."""
    x = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-10:
        # Fix tiny error
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if len(idx) > 0:
                x[idx[0]] = min(1.0, x[idx[0]] + diff)
        else:
            idx = np.where(x > 0.0)[0]
            if len(idx) > 0:
                idx = idx[np.argmax(x[idx])]
                x[idx[0]] = max(0.0, x[idx[0]] + diff)
        return x
    
    if diff > 0:
        # Add mass to elements < 1
        mask = x < 1.0
        if not mask.any():
            return x
        # Water-filling: find t such that sum(min(t, 1-x[i]) for i in mask) = diff
        caps = 1.0 - x[mask]
        sorted_caps = np.sort(caps)
        n_free = len(sorted_caps)
        # Find how many elements get fully filled
        cumsum = np.cumsum(sorted_caps)
        if cumsum[-1] < diff:
            # Can't fill enough (shouldn't happen if target <= n)
            x[mask] = 1.0
            return x
        # Find threshold
        idx_fill = np.searchsorted(cumsum, diff)
        if idx_fill == n_free:
            # All get partial
            t = diff / n_free
        else:
            remaining = diff - (cumsum[idx_fill-1] if idx_fill > 0 else 0)
            t = remaining / (n_free - idx_fill)
        x[mask] = np.minimum(1.0, x[mask] + t)

        # Remove mass from elements > 0
        mask = x > 0.0
        if not mask.any():
            return x

        vals = x[mask]

import numpy as np
import time

def _score(h):
    """Compute C5 = max_k corr_k * dx where corr = correlate(h, 1-h, full)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(h, target):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target}."""
    x = np.asarray(h, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-13:
        return x
    
    if diff > 0:
        mask = x < 1.0
        if not np.any(mask):
            return x
        lo, hi = 0.0, 1.0 - x[mask].min()
        for _ in range(60):
            mid = (lo + hi) * 0.5
            total = np.minimum(mid, 1.0 - x[mask]).sum()
            if total < diff:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) * 0.5
        x[mask] = np.minimum(1.0, x[mask] + t)
    else:
        mask = x > 0.0
        if not np.any(mask):
            return x
        lo, hi = 0.0, x[mask].max()
        for _ in range(60):
            mid = (lo + hi) * 0.5
            total = np.minimum(mid, x[mask]).sum()
            if total < -diff:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) * 0.5
        x[mask] = np.maximum(0.0, x[mask] - t)
    
    # Final correction
    err = target - x.sum()
    if abs(err) > 1e-14:
        if err > 0:
            i = int(np.argmin(1.0 - x))
            x[i] = min(1.0, x[i] + err)
        else:
            i = int(np.argmax(x))
            x[i] = max(0.0, x[i] + err)
    return x

def _resample(h, new_n):
    """Resample h to new_n points via linear interpolation + projection."""
    n = len(h)
    if new_n == n:
        return h.copy()
    target = new_n / 2.0
    src = np.linspace(0, 1, n, endpoint=False)
    dst = np.linspace(0, 1, new_n, endpoint=False)
    new_h = np.interp(dst, src, h)
    return _project(new_h, target)

def _line_search(h, p, q, target):
    """
    Minimize score along the line h[p] + delta, h[q] - delta.
    Returns (best_delta, best_h, best_score).
    """
    n = len(h)

    d_lo = -hq
    d_hi = 1.0 - hp

        return 0.0, h, _score(h)

        h_new[p] = hp + d
        h_new[q] = hq - d
        return _score(h_new)

    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = eval_d(c), eval_d(d)

            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_d(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_d(d)
    
    d_mid = (a + b) * 0.5
