    n = len(v)

    # h_i = clip(v_i - lambda, 0, 1)

        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(h)
        if diff > 0:
            for i in idx[::-1]:
                adj = min(diff, 1.0 - h[i])
                h[i] += adj
                diff -= adj
                if abs(diff) < 1e-12:
                    break
        else:
            for i in idx:
                adj = min(-diff, h[i])
                h[i] -= adj
                diff += adj
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target_sum):
    """Project v onto {0<=h<=1, sum(h)=target_sum}."""
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_max_val(x, t):
    m = x.max()
    if t < 1:
        return m
    return float(m + np.log(np.sum(np.exp(t * (x - m)))) / t)

def _compute_grad(h, w):
    """Gradient of sum_k w_k * c_k w.r.t. h, where c_k = correlate(h,1-h) at lag k-(n-1)."""
    n = len(h)
    grad = np.zeros(n)
    nw = len(w)
    for ki in range(nw):
        wi = w[ki]
        if wi < 1e-14:
            continue
        lag = ki - (n - 1)
        js1 = max(0, lag)
        je1 = min(n - 1, n - 1 + lag)
        if js1 <= je1:
            grad[js1:je1 + 1] += wi * (1.0 - h[js1 - lag:je1 - lag + 1])
        js2 = max(0, -lag)
        je2 = min(n - 1, n - 1 - lag)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wi * h[js2 + lag:je2 + lag + 1]
    return grad

def _optimize_pgd(h_init, target_sum, t_values, lr_init, time_limit):
    """Projected gradient descent with smooth max. Returns (best_h, best_c5)."""
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5_value(h)
    
    for t in t_values:
        lr = lr_init
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        prev_obj = _smooth_max_val(corr, t)
        stall = 0
        max_iter = 5000
        
        for it in range(max_iter):
            if time.time() > time_limit:
                break
            m = corr.max()
            ex = np.exp(t * (corr - m))
            w = ex / ex.sum()
            grad = _compute_grad(h, w)
            h_new = h - lr * grad
            h_new = _project(h_new, target_sum)
            
            new_c5 = _c5_value(h_new)
            improved = new_c5 < best_c5 - 1e-11
            if improved:
                best_h = h_new.copy()
                best_c5 = new_c5
            
            corr_new = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
            obj_new = _smooth_max_val(corr_new, t)
            
            if obj_new > prev_obj + 1e-7:
                lr *= 0.7
                if lr < 1e-5:
                    break
            else:
                h = h_new
                corr = corr_new
                prev_obj = obj_new
                lr = min(lr * 1.03, lr_init * 3.0)
            
            if not improved and stall > 500:
                break
            stall = 0 if improved else stall + 1
    
    return best_h, best_c5

def _try_slsqp(h_init, target_sum, time_limit):
    """Try SLSQP optimization with smooth max."""
    n = len(h) if (h := h_init) is not None else 0
    n = len(h_init)
    dx = 2.0 / n
    t = 60.0
    
    def obj(y):
        h = _project(y, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return _smooth_max_val(corr, t)
    
    constraints = [{'type': 'eq', 'fun': lambda y: y.sum() - target_sum,
                    'jac': lambda y: np.ones_like(y)}]
    bounds = [(0.0, 1.0)] * n
    
    if time.time() > time_limit:
        return h_init, _c5_value(h_init)
    
    try:
        res = minimize(obj, h_init.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or (hasattr(res, 'fun') and res.fun < _c5_value(h_init)):
            h_out = _project(res.x, target_sum)
            c5 = _c5_value(h_out)
            return h_out, c5
    except:
        pass
    return h_init, _c5_value(h_init)

def _subgradient_polish(h_init, target_sum, time_limit):
    """Polish using subgradient of the true max."""
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5_value(h)
    lr = 0.02
    patience = 0
    
    for it in range(20000):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_val = corr.max()
        active = corr >= max_val - 1e-9
        w = np.zeros_like(corr)
        w[active] = 1.0 / active.sum()
        
        grad = _compute_grad(h, w)
        h_new = h - lr * grad
        h_new = _project(h_new, target_sum)
        
        new_c5 = _c5_value(h_new)
        if new_c5 < best_c5 - 1e-11:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new
            patience = 0
            lr = min(lr * 1.01, 0.05)
        else:
            patience += 1
            lr *= 0.97
            if lr < 1e-7:
                break
            if patience > 500:
                break
    
    return best_h, best_c5
