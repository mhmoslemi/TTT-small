```
def lag_gradient(h,n,s):
    g=np.zeros(n)
    if s==0:
       g = 1.0 - 2.0*h
    elif s>0:
       r = n-s
       # q=s..n-1: +1 - h[q-s]
       # q indices s to n-1, h indices 0..r-1
       g[s:n] += 1.0 - h[0:r]
       # q=0..r-1: -h[q+s]
       g[0:r] -= h[s:n]
    else:
       r=-s
       # q=0..n-r-1: +1 - h[q+r]
       g[0:n-r] += 1.0 - h[r:n]
       # q=r..n-1: -h[q-r]
       g[r:n] -= h[0:n-r]
    return g
```

```python
def time_left(safe_seconds=3.0):
    if RUN_DEADLINE <= 0.0:
        return True
    return (RUN_DEADLINE - time.time()) > safe_seconds


def bound_fast(h, n=None):
    if n is None:
        n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / float(n)))


def explicit_covector(h, n):
    F = np.empty(2 * n - 1, dtype=float)
    F[n - 1] = float(np.dot(h, 1.0 - h))
    for s in range(1, n):
        a = n - s
        F[n - 1 + s] = float(np.dot(h[s:n], 1.0 - h[:a]))
        F[n - 1 - s] = float(np.dot(h[:a], 1.0 - h[s:n]))
    return F


def ensure_orientation(n):
    global ORIENT, ORIENT_VALID
    if ORIENT_VALID:
        return ORIENT
    rng = np.random.default_rng(777 + int(n))
    h = 0.2 + 0.6 * rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    err = float(n) / 2.0 - float(h.sum())
    if abs(err) > 1e-12:
        h += err / float(n)
        h = np.clip(h, 0.0, 1.0)
    F_exp = explicit_covector(h, n)
    F_act = np.correlate(h, 1.0 - h, mode="full")
    d0 = float(np.max(np.abs(F_exp - F_act)))
    d1 = float(np.max(np.abs(F_exp[::-1] - F_act)))
    if d0 <= d1:
        ORIENT = 0
    else:
        ORIENT = 1
    if min(d0, d1) < 1e-7:
        ORIENT_VALID = True
    else:
        ORIENT_VALID = False
    return ORIENT


def lag_gradient(h, n, s):
    g = np.zeros(n, dtype=float)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    if s > 0:
        r = n - s
        g[s:n] += 1.0 - h[0:r]
        g[0:r] -= h[s:n]
    else:
        r = -s
        g[0:n - r] += 1.0 - h[r:n]
        g[r:n] -= h[0:n - r]
    return g


def active_jacobian_fd(h, n, active):
    F = np.correlate(h, 1.0 - h, mode="full")
    G = np.zeros((n, len(active)), dtype=float)
    h2 = h.copy()
    eps = 1e-6
    for i in range(n):
        hi = float(h[i])
        if hi <= 0.0:
            delta = eps
        elif hi >= 1.0:
            delta = -eps
        elif hi < 1.0 - eps:
            delta = eps
        else:
            delta = -eps
        new = hi + delta
        new = min(1.0, max(0.0, new))
        if new == hi:
            delta = 1e-4 if hi < 0.5 else -1e-4
            new = hi + delta
            new = min(1.0, max(0.0, new))
        if new == hi:
            continue
        h2[i] = new
        F2 = np.correlate(h2, 1.0 - h2, mode="full")
        G[i, :] = (F2 - F)[active] / delta
        h2[i] = hi
    return G


def active_jacobian(h, n, active):
    ensure_orientation(n)
    active = list(active)
    if len(active) == 0:
        return np.zeros((n, 0), dtype=float)
    if not ORIENT_VALID:
        return active_jacobian_fd(h, n, active)
    G = np.empty((n, len(active)), dtype=float)
    center = n - 1
    for c, p in enumerate(active):
        s = int(p) - center
        if ORIENT == 1:
            s = -s
        G[:, c] = lag_gradient(h, n, s)
    return G


def soft_gradient(h, n, temperature):
    F = np.correlate(h, 1.0 - h, mode="full")
    Fmax = float(np.max(F))
    dx = 2.0 / float(n)
    temp_u = max(1e-8, float(temperature) / dx)
    w = np.exp((F - Fmax) / temp_u)
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.zeros_like(F)
        w[int(np.argmax(F))] = 1.0
        sw = 1.0
    ensure_orientation(n)
    g = np.zeros(n, dtype=float)
    center = n - 1
    if ORIENT_VALID:
        for p in range(len(F)):
            wp = float(w[p])
            if wp <= sw * 1e-10:
                continue
            coeff = wp / sw
            s = p - center
            if ORIENT == 1:
                s = -s
            g += coeff * lag_gradient(h, n, s)
    else:
        idx = np.where(w > sw * 1e-8)[0]
        if len(idx) == 0:
            idx = np.array([int(np.argmax(F))], dtype=int)
        J = active_jacobian_fd(h, n, idx)
        ws = w[idx] / sw
        g = J.T @ ws
    return g


def dyadic_repair(values, scale_bits=SCALE_BITS):
    values = np.asarray(values, dtype=float)
    values = np.nan_to_num(values, nan=0.0, posinf=1.0, neginf=0.0)
    values = np.clip(values, 0.0, 1.0)
    n = int(len(values))
    S = 1 << int(scale_bits)
    if (n & 1):
        S *= 2
    target = (n * S) // 2
    z = np.rint(values * float(S)).astype(np.int64)
    z = np.clip(z, 0, S, out=z)
    cur = int(z.sum())
    if cur == target:
        return z.astype(float) / float(S)
    if cur > target:
        lo, hi = 0, S
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if int(np.minimum(z, mid).sum()) <= target:
                lo = mid
            else:
                hi = mid - 1
        T = lo
        base = np.minimum(z, T)
        deficit = target - int(base.sum())
        if deficit > 0:
            mask = z > T
            idxs = np.where(mask)[0]
            if len(idxs) >= deficit:
                order = idxs[np.argsort(z[idxs])[:deficit]]
            else:
                order = idxs
            base[order] += 1
            rem = target - int(base.sum())
            while rem > 0:
                cand = int(np.argmax(base < z))
                if base[cand] < z[cand]:
                    base[cand] += 1
                    rem -= 1
                else:
                    break
        z = base
    else:
        lo, hi = 0, S
        while lo < hi:
            mid = (lo + hi) // 2
            if int(np.maximum(z, mid).sum()) >= target:
                hi = mid
            else:
                lo = mid + 1
        T = lo
        base = np.maximum(z, T)
        excess = int(base.sum()) - target
        if excess > 0:
            mask = z < T
            idxs = np.where(mask)[0]
            if len(idxs) >= excess:
                order = idxs[np.argsort(-z[idxs])[:excess]]
            else:
                order = idxs
            base[order] -= 1
            rem = int(base.sum()) - target
            while rem > 0:
                cand = int(np.argmax(z > base))
                if base[cand] > z[cand]:
                    base[cand] -= 1
                    rem -= 1
                else:
                    break
        z = base
    return z.astype(float) / float(S)


def make_synthetic(n, seed):
    rng = np.random.default_rng(seed)
    t = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h = 0.5 + 0.22 * np.sin(math.pi * t) + 0.10 * rng.standard_normal(n)
    h = np.clip(h, 0.0, 1.0)
    return dyadic_repair(h)

def resample_density(parent, n_new):
    n_old = int(len(parent))
    if n_old == n_new:
        return parent.copy()
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    return np.interp(x_new, x_old, parent)


def alpha_safe(h, d, alpha):
    max_a = float(alpha)
    for i in range(len(h)):
        di = float(d[i])
        if di > 0.0:
            room = 1.0 - float(h[i])
            if room < 0.0:
                return 0.0
            max_a = min(max_a, room / di)
        elif di < 0.0:
            room = float(h[i])
            if room < 0.0:
                return 0.0
            max_a = min(max_a, room / (-di))
    return float(max(0.0, min(1.0, max_a)))


def scale_direction_to_bounds(d, h, step_max):
    if d is None:
        return None
    d = np.asarray(d, dtype=float).copy()
    n = len(h)
    if not np.all(np.isfinite(d)):
        return None
    s = float(d.sum())
    if abs(s) > 1e-12:
        d -= s / float(n)
    for i in range(n):
        hi = float(h[i])
        lo = -min(hi, float(step_max))
        up = min(1.0 - hi, float(step_max))
        if d[i] < lo:
            d[i] = lo
        elif d[i] > up:
            d[i] = up
    max_abs = float(np.max(np.abs(d)))
    if max_abs <= 1e-14:
        return None
    if max_abs < float(step_max):
        scale = float(step_max) / max_abs
        for i in range(n):
            hi = float(h[i])
            if d[i] > 1e-15:
                up = min(1.0 - hi, float(step_max))
                if up > 0.0:
                    scale = min(scale, up / d[i])
            elif d[i] < -1e-15:
                lo = -min(hi, float(step_max))
                if lo < 0.0:
                    scale = min(scale, lo / d[i])
        if scale > 1.0:
            d *= min(scale, 2.0)
    for i in range(n):
        hi = float(h[i])
        lo = -min(hi, float(step_max))
        up = min(1.0 - hi, float(step_max))
        if d[i] < lo:
            d[i] = lo
        elif d[i] > up:
            d[i] = up
    s = float(d.sum())
    if abs(s) > 1e-10:
        d -= s / float(n)
    d[np.abs(d) < 1e-15] = 0.0
    return d


def direction_from_gradient(g, h, step_max):
    if g is None or not np.all(np.isfinite(g)):
        return None
    c = np.asarray(g, dtype=float).copy()
    n = len(h)
    bounds = []
    for i in range(n):
        hi = float(h[i])
        lo = -min(hi, float(step_max))
        up = min(1.0 - hi, float(step_max))
        bounds.append((lo, up))
    A_eq = np.ones((1, n), dtype=float)
    b_eq = 0.0
    try:
        res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds)
        except Exception:
            return None
    if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
        return None
    d = res.x.astype(float).copy()
    if float(np.max(np.abs(d))) < 1e-14:
        return None
    return scale_direction_to_bounds(d, h, step_max)


def active_lp_direction(h, n, active, step_max):
    active = list(active)
    m = len(active)
    if m == 0:
        return None
    G = active_jacobian(h, n, active)
    A_ub = np.zeros((m, n + 1), dtype=float)
    b_ub = np.zeros(m, dtype=float)
    for c in range(m):
        A_ub[c, :n] = G[:, c]
        A_ub[c, n] = 1.0
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = -1.0
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = 0.0
    bounds = []
    for i in range(n):
        hi = float(h[i])
        lo = -min(hi, float(step_max))
        up = min(1.0 - hi, float(step_max))
        bounds.append((lo, up))
    bounds.append((0.0, 10.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds)
        except Exception:
            return None
    if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
        return None
    d = res.x[:n].astype(float).copy()
    rho = float(res.x[n])
    if not np.all(np.isfinite(d)) or rho < 1e-10:
        return None
    if float(np.max(np.abs(d))) < 1e-14:
        return None
    return scale_direction_to_bounds(d, h, step_max)


def line_search_direction(h, n, C, d):
    if d is None:
        return None
    d = np.asarray(d, dtype=float).copy()
    n = int(n)
    max_abs = float(np.max(np.abs(d)))
    if max_abs <= 1e-14:
        return None
    results = []
    for a in ALPHAS:
        if not time_left(1.0):
            break
        aa = alpha_safe(h, d, float(a))
        if aa <= 1e-14:
            continue
        cand = h + aa * d
        cand = np.clip(cand, 0.0, 1.0)
        c = bound_fast(cand, n)
        results.append((c, aa))
    if not results:
        return None
    results.sort()
    for c0, a0 in results[:3]:
        if c0 > C + 1e-7:
            continue
        if a0 <= 1e-14:
            continue
        best_a = a0
        best_cu = c0
        lo = max(0.0, a0 * 0.5)
        hi = min(1.0, a0 * 2.0)
        if hi - lo < 1e-5:
            lo = max(0.0, a0 - 1e-3)
            hi = min(1.0, a0 + 1e-3)
        if hi > lo:
            for frac in np.linspace(0.0, 1.0, 9):
                aa = lo + (hi - lo) * float(frac)
                aa = alpha_safe(h, d, aa)
                if aa <= 1e-14:
                    continue
                cand = h + aa * d
                cand = np.clip(cand, 0.0, 1.0)
                cc = bound_fast(cand, n)
                if cc < best_cu - 1e-12:
                    best_cu = cc
                    best_a = aa
        cand = h + best_a * d
        cand = np.clip(cand, 0.0, 1.0)
        cand_rep = dyadic_repair(cand)
        cr = bound_fast(cand_rep, n)
        if cr < C - 1e-12:
            return (cand_rep, cr)
    return None


def random_pair_direction(h, step_max, rng):
    n = len(h)
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while i == j:
        j = int(rng.integers(0, n))
    if h[i] < h[j]:
        i, j = j, i
    if rng.random() < 0.2:
        i, j = j, i
    t = float(step_max) * float(rng.uniform(0.05, 1.0))
    max_t = min(float(h[i]), float(1.0 - h[j]))
    t = min(t, max_t)
    if t <= 0.0:
        return None
    d = np.zeros(n, dtype=float)
    d[i] -= t
    d[j] += t
    return scale_direction_to_bounds(d, h, step_max)


def gradient_pair_direction(g, h, step_max, rng):
    n = len(h)
    if g is None or not np.all(np.isfinite(g)):
        return random_pair_direction(h, step_max, rng)
    cands = []
    for i in range(n):
        gi = float(g[i])
        for j in range(n):
            if i == j:
                continue
            diff = float(g[j]) - gi
            if diff < -1e-12:
                cands.append((diff, i, j))
    if not cands:
        return random_pair_direction(h, step_max, rng)
    cands.sort()
    k = min(8, len(cands))
    chosen = cands[int(rng.integers(0, k))]
    i = int(chosen[1])
    j = int(chosen[2])
    t = float(step_max) * float(rng.uniform(0.2, 1.0))
    max_t = min(float(h[i]), float(1.0 - h[j]))
    t = min(t, max_t)
    if t <= 0.0:
        return None
    d = np.zeros(n, dtype=float)
    d[i] -= t
    d[j] += t
    return scale_direction_to_bounds(d, h, step_max)


def pair_polish(h, n, C, goal, max_evals=50000):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    C = float(C)
    rng = np.random.default_rng(555 + n)
    for _round in range(12):
        if not time_left(20.0):
            break
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
        rng.shuffle(pairs)
        improved = 0
        eval_count = 0
        for i, j in pairs:
            if not time_left(2.0):
                break
            if eval_count >= max_evals:
                break
            hi_i = float(h[i])
            hi_j = float(h[j])
            lo = max(hi_i - 1.0, -hi_j)
            hi = min(hi_i, 1.0 - hi_j)
            if hi - lo < 1e-12:
                continue
            tvals = []
            if hi > 1e-12:
                tvals.append(hi)
                for f in T_FRACTIONS:
                    t = f * hi
                    if 1e-12 < t < hi:
                        tvals.append(t)
                for a in T_ABS:
                    if a < hi:
                        tvals.append(a)
            if lo < -1e-12:
                tvals.append(lo)
                for f in T_FRACTIONS:
                    t = f * lo
                    if lo < t < -1e-12:
                        tvals.append(t)
                for a in T_ABS:
                    if -a > lo:
                        tvals.append(-a)
            best_t = 0.0
            best_c = C
            for t in tvals:
                cand = h.copy()
                cand[i] -= t
                cand[j] += t
                if cand.min() < -1e-9 or cand.max() > 1.0 + 1e-9:
                    cand = np.clip(cand, 0.0, 1.0)
                c2 = bound_fast(cand, n)
                eval_count += 1
                if c2 < best_c - 1e-12:
                    best_c = c2
                    best_t = float(t)
            if best_t != 0.0 and eval_count < max_evals:
                local_best_t = best_t
                local_best_c = best_c
                for f in (0.45, 0.65, 0.82, 0.95, 1.05, 1.15, 1.30, 1.50):
                    t2 = best_t * f
                    cand = h.copy()
                    cand[i] -= t2
                    cand[j] += t2
                    if cand.min() < -1e-9 or cand.max() > 1.0 + 1e-9:
                        continue
                    c2 = bound_fast(cand, n)
                    eval_count += 1
                    if c2 < local_best_c - 1e-12:
                        local_best_c = c2
                        local_best_t = float(t2)
                cand = h.copy()
                cand[i] -= local_best_t
                cand[j] += local_best_t
                cand = np.clip(cand, 0.0, 1.0)
                cand_rep = dyadic_repair(cand)
                cr = bound_fast(cand_rep, n)
                if cr < C - 1e-12:
                    h = cand_rep
                    C = cr
                    improved += 1
        if improved == 0:
            break
    h = dyadic_repair(h)
    C = bound_fast(h, n)
    return h, C


def active_search(h, n, goal, max_iters=20000):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    h = dyadic_repair(h)
    C = bound_fast(h, n)
    best_h = h.copy()
    best_C = C
    temp = 2.0e-4
    step_max = 0.35
    no_improve = 0
    it = 0
    rng = np.random.default_rng(9999 + n)
    ensure_orientation(n)
    while it < max_iters and time_left(10.0):
        F = np.correlate(h, 1.0 - h, mode="full")
        C = float(np.max(F) * (2.0 / float(n)))
        if C < best_C - 1e-12:
            best_C = C
            best_h = h.copy()
        if C <= goal - 1e-8:
            break
        order = np.argsort(F)[::-1]
        active = order[:12]
        d = active_lp_direction(h, n, active, step_max)
        res = line_search_direction(h, n, C, d)
        if res is not None:
            h, C = res
            no_improve = 0
            it += 1
            if C <= goal - 1e-8:
                break
            continue
        g = soft_gradient(h, n, temp)
        d = direction_from_gradient(g, h, step_max)
        res = line_search_direction(h, n, C, d)
        if res is not None:
            h, C = res
            no_improve = 0
            it += 1
            if C <= goal - 1e-8:
                break
            continue
        d = gradient_pair_direction(g, h, step_max, rng)
        res = line_search_direction(h, n, C, d)
        if res is not None:
            h, C = res
            no_improve = 0
            it += 1
            if C <= goal - 1e-8:
                break
            continue
        d = random_pair_direction(h, step_max, rng)
        res = line_search_direction(h, n, C, d)
        if res is not None:
            h, C = res
            no_improve = 0
            it += 1
            if C <= goal - 1e-8:
                break
            continue
        no_improve += 1
        if no_improve > 0 and no_improve % 25 == 0:
            if temp > 1.0e-7:
                temp = max(1.0e-7, temp * 0.75)
            else:
                step_max = max(0.05, step_max * 0.82)
        if no_improve > 80 and no_improve % 50 == 0:
            h, C = pair_polish(h, n, C, goal, max_evals=20000)
            no_improve = 0
            if C < best_C - 1e-12:
                best_C = C
                best_h = h.copy()
            if C <= goal - 1e-8:
                break
        if no_improve > 400:
            break
    h = dyadic_repair(h)
    C = bound_fast(h, n)
    if C <= best_C:
        best_h = h
        best_C = C
    return best_h, best_C


def candidate_dimensions(base_n, max_n=72):
    dims = []
    base_n = int(base_n)
    if base_n not in dims:
        dims.append(base_n)
    for d in range(base_n + 2, max_n + 1, 2):
        if d not in dims:
            dims.append(d)
    lower_start = max(24, base_n - 12)
    for d in range(base_n - 2, lower_start - 1, -2):
        if d not in dims:
            dims.append(d)
    return dims


def update_best(best_C, best_N, best_h, cand_C, cand_N, cand_h):
    cand_C = float(cand_C)
    cand_N = int(cand_N)
    if not np.isfinite(cand_C):
        return best_C, best_N, best_h
    if cand_C < best_C - 1e-12:
        return cand_C, cand_N, np.asarray(cand_h, dtype=float).copy()
    if abs(cand_C - best_C) <= 1e-12 and cand_N < int(best_N):
        return cand_C, cand_N, np.asarray(cand_h, dtype=float).copy()
    return best_C, best_N, best_h


def run(seed=42, budget_s=1100, **kwargs):
    global RUN_DEADLINE
    start = time.time()
    budget = max(30.0, float(budget_s))
    RUN_DEADLINE = start + max(1.0, budget - 12.0)
```
