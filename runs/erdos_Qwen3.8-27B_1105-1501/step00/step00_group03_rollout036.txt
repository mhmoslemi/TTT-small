        h_opt, s_opt = local_optimize(best_h, dl, beta=80.0)
        update(h_opt, s_opt, best_n)
    
    # Ensure final feasibility
    if best_h is not None:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        final_score = compute_score(best_h)
        return (best_h, final_score, n_final)
    
    # Fallback (shouldn't reach here)
    n_fb = 45
    h_fb = project_feasible(np.full(n_fb, 0.5), n_fb / 2.0)
    return (h_fb, compute_score(h_fb), n_fb)
