def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw)) * dx

def project_feas(v):
    n = len(v)
    target = n / 2.0
    v = np.array(v, dtype=float).copy()
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    while np.sum(np.clip(v + lo, 0, 1)) > target:
        lo *= 2
        if lo < -100: break
    while np.sum(np.clip(v + hi, 0, 1)) < target:
        hi *= 2
        if hi > 100: break
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    v = np.clip(v + 0.5*(lo+hi), 0.0, 1.0)
    residual = float(np.sum(v)) - target
    if residual > 1e-10:
        idx = np.argsort(v)[::-1]
        for i in idx:
            dec = min(v[i], residual)
            v[i] -= dec
            residual -= dec
            if residual <= 1e-10: break
    elif residual < -1e-10:
        idx = np.argsort(v)

            inc = min(1.0 - v[i], -residual)
            v[i] += inc
            residual += inc
            if residual >= -1e-10: break
    return v

    """Try a gradient/subgradient step to reduce the max correlation."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(raw))
    m = k_star - (n - 1)  # lag
    
    # Compute gradient of R_{k_star} w.r.t. h
    grad = np.zeros(n)
    if m >= 0:
        # R = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        for j in range(n):
            g = 0.0
            if j <= n - 1 - m:
                g += (1.0 - h[j + m])
            if j >= m:
                g -= h[j - m]
            grad[j] = g
    else:
        # m < 0, lag is |m| to the left
        m_abs = -m
        # R = sum_{i=m_abs}^{n-1} h[i]*(1-h[i-m_abs])
        for j in range(n):
            g = 0.0
            if j >= m_abs:
                g += (1.0 - h[j - m_abs])
            if j <= n - 1 - m_abs:
                g -= h[j + m_abs]
            grad[j] = g
    
    # Project onto sum=0
    grad -= np.mean(grad)
    
    # Normalize
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        return h, current_C
    
    if alpha is None:
        alpha = 0.05
    
    # Try several step sizes
    best_h = h.copy()
    best_C = current_C
    for a in [alpha * 0.1, alpha * 0.3, alpha * 0.5, alpha, alpha * 1.5]:
        h_new = h - a * grad
        h_new = project_feas(h_new)
        C_new = exact_score(h_new)
        if C_new < best_C - 1e-10:
            best_C = C_new
            best_h = h_new
    
    return best_h, best_C

def project_feas(v):
    n = len(v)
    target = n / 2.0
    v = np.array(v, dtype=float).copy()
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(20):
        if np.sum(np.clip(v + lo, 0, 1)) <= target:
            break
        lo *= 2.0
    for _ in range(20):
        if np.sum(np.clip(v + hi, 0, 1)) >= target:
            break
        hi *= 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    v = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    residual = float(np.sum(v)) - target
    if residual > 1e-10:
        idx = np.argsort(v)[::-1]
        for i in idx:
            dec = min(v[i], residual)
            v[i] -= dec
            residual -= dec
            if residual <= 1e-10:
                break
    elif residual < -1e-10:
        idx = np.argsort(v)
        for i in idx:
            inc = min(1.0 - v[i], -residual)
            v[i] += inc
            residual += inc
            if residual >= -1e-10:
                break
    return v

    v = np.interp(x_new, x_old, h)
    return project_feas(v)

            grad[m:] -= h[:n - m]

            grad[mp:] += (1.0 - h[:n - mp])
