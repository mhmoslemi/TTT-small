def project_to_simplex_box(v, n):
    """Project v to {h: 0≤h_i≤1, sum(h)=n/2}"""
    h = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute among non-saturated coordinates
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)

            if abs(diff2) > 1e-12:
                h[free] += diff2 / np.sum(free)
                h = np.clip(h, 0.0, 1.0)
    return h

```python
def line_search_fast(h, d, n):
    """Fast line search using precomputed quadratic coefficients"""
    # Feasible interval
    lam_lo, lam_hi = get_feasible_interval(h, d, n)
    if lam_hi - lam_lo < 1e-12:
        return 0.0, evaluate(h, n)
    
    margin = 1e-10
    lam_lo += margin
    lam_hi -= margin
    if lam_hi <= lam_lo:
        return 0.0, evaluate(h, n)
    
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    a_k = np.correlate(h, one_minus_h, mode='full')
    b_k = np.correlate(d, one_minus_h, mode='full') - np.correlate(h, d, mode='full')
    c_k = -np.correlate(d, d, mode='full')
    
    # Dense sampling
    n_pts = 80
    lams = np.linspace(lam_lo, lam_hi, n_pts)
    # Evaluate J at all points: shape (n_pts, nk)
    vals = a_k[None, :] + b_k[None, :] * lams[:, None] + c_k[None, :] * lams[:, None]**2
    j_vals = vals.max(axis=1) * dx
    
    # Find best region
    best_idx = np.argmin(j_vals)
    best_val = j_vals[best_idx]
    best_lam = lams[best_idx]
    
    # Refine with golden section in a small window
    window = (ham_hi - lam_lo) / n_pts
    a_g = max(lam_lo, best_lam - window)
    b_g = min(lam_hi, best_lam + window)
    
    if b_g - a_g > 1e-14:
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_g = b_g - gr * (b_g - a_g)
        d_g = a_g + gr * (b_g - a_g)
        fc = (a_k + b_k * c_g + c_k * c_g**2).max() * dx
        fd = (a_k + b_k * d_g + c_k * d_g**2).max() * dx
        
        for _ in range(30):
            if fc < fd:
                b_g = d_g
                d_g = c_g
                fd = fc
                c_g = b_g - gr * (b_g - a_g)
                fc = (a_k + b_k * c_g + c_k * c_g**2).max() * dx
            else:
                a_g = c_g
                c_g = d_g
                fc = fd
                d_g = a_g + gr * (b_g - a_g)
                fd = (a_k + b_k * d_g + c_k * d_g**2).max() * dx
        
        final_lam = (a_g + b_g) / 2.0
        final_val = (a_k + b_k * final_lam + c_k * final_lam**2).max() * dx
        if final_val < best_val:
            best_val = final_val
            best_lam = final_lam
    
    return best_lam, best_val
```

import numpy as np

def evaluate_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_h(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / int(np.sum(free))
            h = np.clip(h, 0.0, 1.0)
    return h

def line_search(h, d, n):
    # Get feasible interval
    lam_lo, lam_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        lam_lo = max(lam_lo, float(np.max(-h[pos] / d[pos])))
        lam_hi = min(lam_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lam_lo = max(lam_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        lam_hi = min(lam_hi, float(np.max(-h[neg] / d[neg])))
    
    if lam_hi - lam_lo < 1e-14:
        return 0.0, evaluate_c5(h, n)
    
    margin = 1e-12
    lam_lo += margin
    lam_hi -= margin
    if lam_hi <= lam_lo:
        return 0.0, evaluate_c5(h, n)
    
    dx = 2.0 / n
    omh = 1.0 - h
    a_k = np.correlate(h, omh, mode='full')
    b_k = np.correlate(d, omh, mode='full') - np.correlate(h, d, mode='full')
    c_k = -np.correlate(d, d, mode='full')
    
    # Dense sampling  
    n_pts = 40
    lams = np.linspace(lam_lo, lam_hi, n_pts)
    # Vectorized: shape (n_pts, 2n-1)
    evals = a_k[None, :] + b_k[None, :] * lams[:, None] + c_k[None, :] * (lams[:, None] ** 2)
    j_vals = evals.max(axis=1) * dx
    
    best_idx = int(np.argmin(j_vals))
    best_val = float(j_vals[best_idx])
    best_lam = float(lams[best_idx])
    
    # Golden section refinement
    step = (lam_hi - lam_lo) / n_pts
    a_g = max(lam_lo, best_lam - step)
    b_g = min(lam_hi, best_lam + step)
    
    if b_g - a_g > 1e-15:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_g = b_g - gr * (b_g - a_g)
        d_g = a_g + gr * (b_g - a_g)
        fc = float((a_k + b_k * c_g + c_k * c_g * c_g).max() * dx)
        fd = float((a_k + b_k * d_g + c_k * d_g * d_g).max() * dx)
        
        for _ in range(20):
            if fc < fd:
                b_g = d_g; d_g = c_g; fd = fc
                c_g = b_g - gr * (b_g - a_g)
                fc = float((a_k + b_k * c_g + c_k * c_g * c_g).max() * dx)
            else:
                a_g = c_g; c_g = d_g; fc = fd
                d_g = a_g + gr * (b_g - a_g)
                fd = float((a_k + b_k * d_g + c_k * d_g * d_g).max() * dx)
        
        final_lam = (a_g + b_g) * 0.5
        final_val = float((a_k + b_k * final_lam + c_k * final_lam * final_lam).max() * dx)
        if final_val < best_val:
            best_val = final_val
            best_lam = final_lam
    
    return best_lam, best_val
