                        if 0.0 <= xk <= 1.0:

            candidates = np.unique(np.clip(candidates, 0.0, 1.0))

                    continue

def simulated_annealing(h, target_sum, n_iters=50000, T0=0.01, seed=42):

def logsumexp_optimize(h, target_sum, n_iters=1000, lr=0.01, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    beta = 50.0  # temperature parameter for log-sum-exp
    current_c5 = compute_c5(h)
    
    for it in range(n_iters):
        # Compute gradient of log-sum-exp of correlations
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
        # log-sum-exp approximation to max
        max_corr = np.max(corr)
        weights = np.exp(beta * (corr - max_corr))
        weights /= weights.sum()
        
        # Gradient of log-sum-exp w.r.t. h[i]
        # d/dh[i] log(sum_k exp(beta * c[k])) = beta * sum_k w[k] * dc[k]/dh[i]
        # dc[k]/dh[i] = (2/n) * alpha_k[i] where alpha is from before
        
        # Compute gradient
        grad = np.zeros(n)
        h0 = h.copy()
        h1 = h.copy()
        for i in range(n):
            h0[i] = 0.0
            h1[i] = 1.0
            c0 = np.correlate(h0, 1.0 - h0, mode="full")
            c1 = np.correlate(h1, 1.0 - h1, mode="full")
            alpha = (c1 - c0) * (2.0 / n)
            grad[i] = beta * np.sum(weights * alpha)
        
        # Project gradient onto the constraint sum(h) = target_sum
        grad -= np.mean(grad)
        
        # Take a step
        lr_actual = lr / (it + 10)
        h = h - lr_actual * grad
        h = project_feasible(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new > current_c5 * 1.01:
            # Revert
            h = project_feasible(h + lr_actual * grad, target_sum)
    
    return h, compute_c5(h)
