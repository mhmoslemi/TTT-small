def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def compute_flip_effects(h):
    """Precompute flip_effect[p] for all p.
    flip_effect[p][k] = change in corr[k] when h[p] goes from 0 to 1."""
    n = len(h)
    L = 2 * n - 1
    effects = np.zeros((n, L))
    # For each p, corr[k] = sum_i h[i+k-(n-1)]*(1-h[i])
    # h[p] as first factor: k in [p, p+n-1], change = +(1-h[i]) where i = p+n-1-k
    # h[p] as second factor: k in [n-1-p, n-1+p], change = -h[p+k-(n-1)]
    for p in range(n):

        ks1 = np.arange(p, p + n)
        is1 = p + (n - 1) - ks1  # = i values
        effects[p, ks1] += 1.0 - h[is1]
        # Second factor contribution
        ks2 = np.arange(n - 1 - p, n - 1 + p + 1)
        is2 = p + ks2 - (n - 1)  # = index in h
        effects[p, ks2] -= h[is2]
    return effects

    """Greedy best-improving local search for binary vectors."""
    n = len(h0)
    h = (h0 > 0.5).astype(float)

    effects = compute_flip_effects(h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        current_max = np.max(corr)
        
        best_delta = None
        best_i = -1
        best_j = -1
        best_new_max = current_max
        
        for i in ones:
            for j in zeros:
                delta = effects[j] - effects[i]
                new_corr = corr + delta
                new_max = np.max(new_corr)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
                    best_i = i
                    best_j = j
        
        if best_i < 0:
            break
        
        h[best_i] = 0
        h[best_j] = 1
        corr = corr + best_delta
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        effects = compute_flip_effects(h)
    
    return h, float(np.max(corr) * 2.0 / n)

def binary_sa(h0, max_iters=20000, t_start=2.0, t_end=0.01):
    """Simulated annealing for binary vectors."""
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    best_h = h.copy()
    best_max = current_max
    rng = np.random.default_rng()
    
    for it in range(max_iters):
        temp = t_start * (t_end / t_start) ** (it / max_iters)
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        h[i], h[j] = h[j], h[i]
        
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(new_corr)
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
            current_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # Reject
            h[i], h[j] = h[j], h[i]
    
    return best_h, float(best_max * 2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
        else:
            mask = h > 1e-14
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
    return h

def compute_effects(h):
    n = len(h)
    L = 2 * n - 1
    mid = n - 1
    effects = np.zeros((n, L))
    for p in range(n):
        # First factor: h[p] = h[i+d], i = p-d, k = mid+d = mid+p-i
        # k ranges from p (i=n-1) to mid+p (i=0)
        ks1 = np.arange(p, mid + p + 1)
        i_vals = mid + p - ks1
        effects[p, ks1] += 1.0 - h[i_vals]
        # Second factor: h[p] in (1-h[i]), i=p, term = h[p+d]*(1-h[p])
        # d = k-mid, so p+d in [0,n-1]: k in [mid-p, mid+p]
        ks2 = np.arange(mid - p, mid + p + 1)
        q_vals = p + ks2 - mid
        effects[p, ks2] -= h[q_vals]
        # Fix d=0 double counting
        effects[p, mid] = 0.0
    return effects

def greedy_binary(h, max_iters=5000):
    n = len(h)
    h = (h > 0.5).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full')
    effects = compute_effects(h)
    
    for it in range(max_iters):
        current_max = np.max(corr)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improvement = 0
        best_i, best_j = -1, -1
        best_delta = None
        
        for i in ones:
            for j in zeros:
                delta = effects[j] - effects[i]
                new_max = np.max(corr + delta)
                improvement = current_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
                    best_delta = delta
        
        if best_i < 0:
            break
        
        h[best_i] = 0
        h[best_j] = 1
        corr = corr + best_delta
        effects = compute_effects(h)
    
    return h, float(np.max(corr) * 2.0 / n)

def ills_binary(h, n_kicks=50, kicks_per_round=3, max_iters=2000):
    """Iterated Local Search for binary vectors."""
    n = len(h)
    best_h = (h > 0.5).astype(np.float64)
    best_c5 = c5_exact(best_h)
    rng = np.random.default_rng()
    
    current_h = best_h.copy()
    
    for kick in range(n_kicks):
        # Random perturbation
        for _ in range(kicks_per_round):
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            i = rng.choice(ones)
            j = rng.choice(zeros)
            current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Greedy local search
        h_opt, c5_val = greedy_binary(current_h, max_iters=max_iters)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            current_h = h_opt.copy()
    
    return best_h, best_c5

def sa_binary(h, max_iters=50000, t_start=5.0, t_end=0.01):
    """Simulated annealing for binary vectors."""
    n = len(h)
    h = (h > 0.5).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    best_h = h.copy()
    best_max = current_max
    rng = np.random.default_rng()
    
    for it in range(max_iters):
        temp = t_start * (t_end / t_start) ** (it / max_iters)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        i = ones[i]
        j = zeros[j]
        
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(new_corr)
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            current_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, float(best_max * 2.0 / n)

                best_h = h_ils.copy()
            
            # SA from best ILS result
            if time_left() > 60:
                h_sa, c5_sa = sa_binary(h_ils, max_iters=30000, t_start=3.0, t_end=0.05)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
    
    # Continuous optimization on initial
    if time_left() > 120:

            h_cont = np.array(initial_h_values, dtype=np.float64)
            h_cont = project_feasible(h_cont)
            n_c = len(h_cont)
            target = n_c / 2.0
            
            # SLSQP with smoothed objective
            beta = 15.0
            def smooth_obj(x):
                hc = project_feasible(x)
                corr = np.correlate(hc, 1.0 - hc, mode='full') * beta
                m = np.max(corr)
                return (m + np.log(np.sum(np.exp(corr - m))) / beta) / beta
            
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                           'jac': lambda x: np.ones(n_c)}]
            bounds = [(0.0, 1.0)] * n_c
            
            res = minimize(smooth_obj, h_cont.copy(), method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 300, 'ftol': 1e-12})
            h_opt = project_feasible(res.x)
            c5_opt = c5_exact(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    if time_left() > 20 and best_h is not None:

                    break

                    c5_try = c5_exact(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
                        break
