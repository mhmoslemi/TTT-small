import numpy as np
from scipy.optimize import linprog
import time

def score_h(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_mass(v, target=None):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}."""
    n = len(v)
    if target is None:
        target = n / 2.0
    
    v = np.clip(v, 0.0, 1.0)
    
    # Bisection on the offset
    def phi(t):
        return np.sum(np.clip(v + t, 0.0, 1.0))
    
    lo = -2.0
    hi = 2.0
    # Find bracket
    if phi(lo) > target:
        return np.clip(v + lo, 0.0, 1.0)  # all zeros basically
    if phi(hi) < target:
        return np.clip(v + hi, 0.0, 1.0)
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if phi(mid) < target:
            lo = mid
        else:
            hi = mid
    
    y = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(y)
    if abs(residual) < 1e-10:
        return y
    
    # Distribute residual among non-boundary elements
    for _ in range(50):
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        nfree = np.sum(free)
        if nfree == 0:
            break
        adj = residual / nfree
        y[free] = np.clip(y[free] + adj, 0.0, 1.0)
        residual = target - np.sum(y)
        if abs(residual) < 1e-12:
            break
    
    # Final fix: distribute remaining residual
    if abs(residual) > 1e-12:
        free = (y > 0) & (y < 1)
        if np.sum(free) > 0:
            y[free] += residual / np.sum(free)
            y = np.clip(y, 0.0, 1.0)
    
    return y

def feasible_lambda_range(h, d):
    """Find the range of lambda such that h + lambda*d is in [0,1]^n."""
    n = len(h)
    eps = 1e-12
    lam_min = 0.0
    lam_max = 0.0
    
    for i in range(n):
        if d[i] > eps:
            lam_min = max(lam_min, -h[i] / d[i])
            lam_max = min(lam_max, (1.0 - h[i]) / d[i])
        elif d[i] < -eps:
            lam_min = max(lam_min, (1.0 - h[i]) / d[i])
            lam_max = min(lam_max, -h[i] / d[i])
    
    # Small margin
    margin = 1e-9
    lam_min += margin
    lam_max -= margin
    
    if lam_min >= lam_max:
        return None
    return (lam_min, lam_max)

def line_search(h_c, d, score_c, max_iter=200):
    """Line search along h_c + lambda*d. Returns (best_h, best_score) or None."""
    n = len(h_c)
    dx = 2.0 / n
    
    # Check direction is approximately zero-sum
    d_sum = np.sum(d)
    if abs(d_sum) > 1e-8:
        d = d - d_sum / n  # make it exactly zero-sum
    
    frange = feasible_lambda_range(h_c, d)
    if frange is None:
        return None
    
    lam_lo, lam_hi = frange
    
    # Grid search first
    n_grid = 60
    lams = np.linspace(lam_lo, lam_hi, n_grid)
    best_lam = lam_lo
    best_score = score_c
    
    for lam in lams:
        h_try = h_c + lam * d
        if np.any(h_try < -1e-10) or np.any(h_try > 1.0 + 1e-10):
            continue
        h_try = np.clip(h_try, 0.0, 1.0)
        s = score_h(h_try)
        if s < best_score - 1e-12:
            best_score = s
            best_lam = lam
    
    if best_score >= score_c - 1e-12:
        return None
    
    # Refine around best_lam
    # Find the grid interval containing best_lam
    idx = np.argmin(np.abs(lams - best_lam))
    if idx > 0:
        a = lams[idx-1]
    else:
        a = lam_lo
    if idx < n_grid - 1:
        b = lams[idx+1]
    else:
        b = lam_hi
    
    for _ in range(40):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h1 = np.clip(h_c + m1 * d, 0.0, 1.0)
        h2 = np.clip(h_c + m2 * d, 0.0, 1.0)
        s1 = score_h(h1)
        s2 = score_h(h2)
        if s1 < s2:
            b = m2
        else:
            a = m1
        if abs(b - a) < 1e-10:
            break
    
    best_lam = (a + b) / 2.0
    h_best = np.clip(h_c + best_lam * d, 0.0, 1.0)
    best_score = score_h(h_best)
    
    if best_score < score_c - 1e-12:
        return (h_best, best_score)
    return None

import numpy as np
from scipy.optimize import linprog
import time

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_mass(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    
    # Bisection
    def phi(t):
        return np.sum(np.clip(v + t, 0.0, 1.0))
    
    lo, hi = -1.5, 1.5
    if phi(lo) > target:
        return np.zeros(n)
    if phi(hi) < target:
        return np.ones(n)
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if phi(mid) < target:
            lo = mid
        else:
            hi = mid
    
    y = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(y)
    
    # Fix residual
    for _ in range(100):
        free = (y > 1e-14) & (y < 1.0 - 1e-14)
        nfree = np.sum(free)
        if nfree == 0:
            break
        adj = residual / nfree
        y[free] = np.clip(y[free] + adj, 0.0, 1.0)
        residual = target - np.sum(y)
        if abs(residual) < 1e-12:
            break
    
    return y

def compute_overlaps(h):
    """Compute all A_k values (the correlate array)."""
    return np.correlate(h, 1.0 - h, mode="full")

def compute_grads(h, lags=None):
    """Compute gradients of A_k with respect to h for all k.
    Returns array of shape (2n-1, n) where row i corresponds to lag = i - (n-1).
    """
    n = len(h)
    num_lags = 2 * n - 1
    if lags is None:
        lags = np.arange(-(n-1), n)
    
    grad = np.zeros((num_lags, n))
    
    for idx, k in enumerate(lags):
        if k >= 0:
            # A_k = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
            # grad[j] = (1-h[j+k]) if j+k < n, minus h[j-k] if j-k >= 0
            if k == 0:
                grad[idx] = 1.0 - 2.0 * h
            else:
                # First factor: j from 0 to n-1-k, contribution (1-h[j+k])
                grad[idx, :n-k] += (1.0 - h[k:])
                # Second factor: j from k to n-1, contribution -h[j-k]
                grad[idx, k:] -= h[:n-k]
        else:
            t = -k
            # A_k = sum_{i=t}^{n-1} h[i] * (1 - h[i-t])
            # grad[j] = (1-h[j-t]) if j >= t, minus h[j+t] if j+t < n
            grad[idx, t:] += (1.0 - h[:n-t])
            grad[idx, :n-t] -= h[t:]
    
    return grad

def feasible_range(h, d):
    """Find lambda range for h + lambda*d to stay in [0,1]."""
    eps = 1e-12
    lo, hi = -np.inf, np.inf
    
    pos = d > eps
    neg = d < -eps
    
    if np.any(pos):
        lo = max(lo, np.max(-h[pos] / d[pos]))
        hi = min(hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        lo = max(lo, np.max((1.0 - h[neg]) / d[neg]))
        hi = min(hi, np.min(-h[neg] / d[neg]))
    
    if not np.isfinite(lo) or not np.isfinite(hi):
        # At least one must be bounded
        if not np.isfinite(lo):
            lo = -1.0
        if not np.isfinite(hi):
            hi = 1.0
    
    margin = 1e-9
    lo += margin
    hi -= margin
    if lo >= hi:
        return None
    return (lo, hi)

def line_search(h_c, d, score_c, n_grid=80, refine_iters=50):
    """Exact line search along h_c + lambda*d."""
    n = len(h_c)
    
    # Make d zero-sum
    d = d - np.sum(d) / n
    
    frange = feasible_range(h_c, d)
    if frange is None:
        return None
    
    lo, hi = frange
    
    # Coarse grid
    lams = np.linspace(lo, hi, n_grid)
    best_lam = 0.0
    best_score = score_c
    
    for lam in lams:
        h_try = np.clip(h_c + lam * d, 0.0, 1.0)
        s = score_h(h_try)
        if s < best_score:
            best_score = s
            best_lam = lam
    
    if best_score >= score_c - 1e-14:
        return None
    
    # Golden section refinement
    a, b = lo, hi
    # Narrow to region around best_lam
    idx = np.argmin(np.abs(lams - best_lam))
    if idx > 0:
        a = lams[idx - 1]
    if idx < n_grid - 1:
        b = lams[idx + 1]
    
    for _ in range(refine_iters):
        if b - a < 1e-12:
            break
        m1 = a + 0.382 * (b - a)
        m2 = b - 0.382 * (b - a)
        s1 = score_h(np.clip(h_c + m1 * d, 0.0, 1.0))
        s2 = score_h(np.clip(h_c + m2 * d, 0.0, 1.0))
        if s1 < s2:
            b = m2
        else:
            a = m1
    
    best_lam = (a + b) / 2.0
    h_best = np.clip(h_c + best_lam * d, 0.0, 1.0)
    best_score = score_h(h_best)
    
    if best_score < score_c - 1e-14:
        return (h_best, best_score)
    return None

def lp_direction(h_c, radius=0.15):
    """Solve LP for minimax linearization direction."""
    n = len(h_c)
    num_lags = 2 * n - 1
    
    overlaps = compute_overlaps(h_c)
    grads = compute_grads(h_c)
    
    # Build LP: minimize t
    # s.t. grads[i] · h - grads[i] · h_c + overlaps[i] <= t
    #      sum(h) = n/2
    #      h_c - radius <= h <= min(1, h_c + radius)
    #      0 <= h <= 1
    
    # Variables: h (n), t (1)
    # Objective: c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G @ h - G @ h_c + A <= t
    # => G @ h - t <= G @ h_c - A
    # Matrix shape: (num_lags, n+1)
    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = grads @ h_c - overlaps
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, np.inf))  # t is unbounded below (but will be positive)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:

def subgradient_direction(h_c, n_top=5):
    """Compute subgradient direction from top active lags."""
    n = len(h_c)
    overlaps = compute_overlaps(h_c)
    max_val = np.max(overlaps)
    
    # Find lags that are close to maximum
    threshold = max_val - 0.5
    active = np.where(overlaps >= threshold)[0]
    
    if len(active) == 0:
        active = np.where(overlaps >= max_val - 1e-6)[0]
    
    # Take top n_top
    top_idx = active[np.argsort(overlaps[active])[::-1][:n_top]]
    
    grads = compute_grads(h_c, lags=top_idx - (n - 1))
    
    # Average the gradients (or weighted by overlap value)
    weights = overlaps[top_idx]
    weights = weights / np.sum(weights)
    
    grad_avg = np.sum(weights[:, None] * grads, axis=0)
    
    # Negate for descent
    d = -grad_avg
    # Project to zero-sum
    d = d - np.sum(d) / n
    # Also project out components that would push h out of bounds
    for i in range(n):
        if h_c[i] <= 0 and d[i] < 0:
            d[i] = 0
        if h_c[i] >= 1 and d[i] > 0:
            d[i] = 0
    d = d - np.sum(d) / n
    
    if np.max(np.abs(d)) < 1e-10:
        return None
    return d

def random_direction(n, rng):
    """Generate a random zero-sum direction."""
    d = rng.standard_normal(n)
    d = d - np.sum(d) / n
    return d

def try_pair_swaps(h, score_c, n_swaps_per_iter=200, max_iters=50):

                best_h = h_try
                improved = True
        
        if not improved:
            break
    
    return best_h, best_score

def interpolate_to_length(h_src, n_tgt):
    """Interpolate h from its length to n_tgt."""
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_tgt)
    h_tgt = np.interp(x_tgt, x_src, h_src)
    return project_box_mass(h_tgt, target=n_tgt / 2.0)

def project_box_mass(v, target=None):

def compute_grads_at_lag(h, k):
    """Compute gradient of A_k with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
    else:
        t = -k
        grad[t:] += (1.0 - h[:n-t])
        grad[:n-t] -= h[t:]
    return grad

def lp_direction(h_c, radius=0.15):

    num_lags = 2 * n - 1
    overlaps = np.correlate(h_c, 1.0 - h_c, mode="full")
    
    # Build gradient matrix
    grads = np.zeros((num_lags, n))

        grads[idx] = compute_grads_at_lag(h_c, k)

    A_ub = np.zeros((num_lags, n + 1))

    if len(active) > n_top:
        sorted_active = active[np.argsort(overlaps[active])[::-1][:n_top]]
    else:
        sorted_active = active
    
    if len(sorted_active) == 0:
        return None
    
    # Weighted average of gradients
    weights = overlaps[sorted_active]
    weights = np.maximum(weights, 0.1)  # avoid zero weights
    weights /= np.sum(weights)
    
    grad_sum = np.zeros(n)
    for idx in sorted_active:
        k = idx - (n - 1)
        grad_sum += weights[np.where(sorted_active == idx)[0][0]] * compute_grads_at_lag(h_c, k)
    
    d = -grad_sum
    d -= np.sum(d) / n
    # Project out of bounds
    for i in range(n):
        if h_c[i] <= 1e-10 and d[i] < 0:
            d[i] = 0.0
        if h_c[i] >= 1.0 - 1e-10 and d[i] > 0:
            d[i] = 0.0
    d -= np.sum(d) / n
    
    if np.max(np.abs(d)) < 1e-10:
        return None
    return d

def random_zero_sum(n, rng):

    d -= np.sum(d) / n
    return d

def pair_transfer_optimize(h, score_c, max_rounds=20, n_trial_deltas=5):
    """Optimize by trying pair transfers."""
    n = len(h)
    best_h = h.copy()
    best_score = score_c
    eps = 1e-14
    
    for round_num in range(max_rounds):
        improved = False
        # Sort indices by h value
        order = np.argsort(h)
        
        # Try transferring from high to low
        deltas = np.array([0.02, 0.05, 0.1, 0.2, 0.4])
        
        for idx_high in range(n - 1, -1, -1):
            i = order[idx_high]
            if best_h[i] < 0.05:
                break
            for idx_low in range(0, n // 2):
                j = order[idx_low]
                if best_h[j] > 0.95:
                    break
                diff = best_h[i] - best_h[j]
                if diff <= 0:
                    continue
                for delta in deltas:
                    d = min(delta, diff)
                    h_try = best_h.copy()
                    h_try[i] -= d
                    h_try[j] += d
                    s = score_h(h_try)
                    if s < best_score - eps:
                        best_score = s
                        best_h[i] -= d
                        best_h[j] += d
                        improved = True
                        break
        
        if not improved:
            break
    
    return best_h, best_score

import time


def score_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_mass(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(y)
    for _ in range(100):
        free = (y > 1e-14) & (y < 1.0 - 1e-14)
        nfree = np.sum(free)
        if nfree == 0:
            break
        y[free] += residual / nfree
        np.clip(y, 0.0, 1.0, out=y)
        residual = target - np.sum(y)
        if abs(residual) < 1e-13:
            break
    return y


def compute_grads_at_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
    else:
        t = -k
        grad[t:] += (1.0 - h[:n - t])
        grad[:n - t] -= h[t:]
    return grad


def feasible_range(h, d):
    eps = 1e-12
    lo = -np.inf
    hi = np.inf
    pos = d > eps
    neg = d < -eps
    if np.any(pos):
        lo = max(lo, float(np.max(-h[pos] / d[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max((1.0 - h[neg]) / d[neg])))
        hi = min(hi, float(np.min(-h[neg] / d[neg])))
    if not np.isfinite(lo):
        lo = -1.0
    if not np.isfinite(hi):
        hi = 1.0
    m = 1e-9
    lo += m
    hi -= m
    if lo >= hi:
        return None
    return (lo, hi)

def lp_direction(h_c, radius=0.15):
    n = len(h_c)
    num_lags = 2 * n - 1
    overlaps = np.correlate(h_c, 1.0 - h_c, mode="full")
    grads = np.zeros((num_lags, n))
    for idx in range(num_lags):
        k = idx - (n - 1)
        grads[idx] = compute_grads_at_lag(h_c, k)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = grads @ h_c - overlaps
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.maximum(0.0, h_c - radius)
    ub = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, np.inf))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_bar = res.x[:n]
            d = h_bar - h_c
            d -= np.sum(d) / n
            if np.max(np.abs(d)) > 1e-10:
                return d
    except Exception:
        pass
    return None
