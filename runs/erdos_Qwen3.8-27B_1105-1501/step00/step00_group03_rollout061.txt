```python
import numpy as np
import time
from scipy.optimize import minimize

def make_feasible(z):
    """Project z to feasible set: sum = n/2, 0 <= h <= 1."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float)
    # Bisection on lambda: sum(clip(z + lambda, 0, 1)) = target
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z + lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argmax(1.0 - h)
            delta = min(residual, 1.0 - h[idx])
            h[idx] += delta
            residual -= delta
            while residual > 1e-10:
                idx = np.argmax(1.0 - h)
                if 1.0 - h[idx] < 1e-10:
                    break
                delta = min(residual, 1.0 - h[idx])
                h[idx] += delta
                residual -= delta
        else:
            idx = np.argmax(h)
            delta = min(-residual, h[idx])
            h[idx] -= delta
            residual += delta
            while residual < -1e-10:
                idx = np.argmax(h)
                if h[idx] < 1e-10:
                    break
                delta = min(-residual, h[idx])
                h[idx] -= delta
                residual += delta
    return h

def official_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, 'full')) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, beta):
    """Compute smooth surrogate value and gradient."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    # corr has 2n-1 values
    scaled = corr * dx  # Now in same units as C5
    cmax = np.max(scaled)
    # Stable log-sum-exp
    shifted = scaled - cmax
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 0)
    exp_vals = np.exp(beta * shifted)
    log_sum = np.log(np.sum(exp_vals))
    value = cmax + log_sum / beta
    # Weights
    w = exp_vals / np.sum(exp_vals)
    if not np.isfinite(w).all() or np.sum(w) < 1e-15:
        w = np.ones(len(corr)) / len(corr)
    # Gradient: dx * sum_k w_k * dF_k/dh
    grad_matrix = _corr_grad_matrix(h)
    grad = dx * (w @ grad_matrix)
    return value, grad

def _corr_grad_matrix(h):
    """Compute (2n-1, n) gradient matrix."""
    n = len(h)
    grad = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n - 1)
        if d >= 0:
            if n > d:
                p1 = np.arange(0, n - d)
                grad[k, p1] = (1.0 - h[p1 + d])
            if d > 0:
                p2 = np.arange(d, n)
                grad[k, p2] += (-h[p2 - d])
        else:
            e = -d
            if e < n:
                p1 = np.arange(e, n)
                grad[k, p1] = (1.0 - h[p1 - e])
            if e < n:
                p2 = np.arange(0, n - e)
                grad[k, p2] += (-h[p2 + e])
    return grad
```

```python
from scipy.optimize import minimize

def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(z + lam, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(h)
    if abs(res) > 1e-12:
        if res > 0:
            for _ in range(n):
                idx = np.argmax(1.0 - h)
                if 1.0 - h[idx] < 1e-12:
                    break
                d = min(res, 1.0 - h[idx])
                h[idx] += d
                res -= d
        else:
            for _ in range(n):
                idx = np.argmax(h)
                if h[idx] < 1e-12:
                    break
                d = min(-res, h[idx])
                h[idx] -= d
                res += d
    return h

def _corr_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    grad = np.zeros((m, n))
    for k in range(m):
        d = k - (n - 1)
        if d >= 0:
            if n > d:
                p1 = np.arange(0, n - d)
                grad[k, p1] = 1.0 - h[p1 + d]
            if d > 0:

                grad[k, p2] += -h[p2 - d]
        else:

            if e < n:
                p1 = np.arange(e, n)
                grad[k, p1] = 1.0 - h[p1 - e]
            p2 = np.arange(0, n - e)
            grad[k, p2] += -h[p2 + e]
    return grad

        g = 1.0 - x
        corr = np.correlate(x, g, mode='full')[::-1] * dx
        cmax = np.max(corr)
        shifted = np.clip(corr - cmax, -200.0, 0.0)
        ev = np.exp(beta * shifted)
        return cmax + np.log(np.sum(ev)) / beta

            w = np.ones(m) / m
        else:
            w = ev / s
        gm = _corr_grad_matrix(x)
        return w @ gm

        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n

    result = minimize(obj_func, h0, jac=obj_grad, method='SLSQP',

        return h_new, official_score(h_new)
    return h0, official_score(h0)
```

```python
def make_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float).copy()
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(z + lam, 0.0, 1.0)
    res = target - np.sum(h)
    if abs(res) > 1e-12:
        for _ in range(n):
            if res > 0:
                idx = int(np.argmax(1.0 - h))
                if 1.0 - h[idx] < 1e-12:
                    break
                d = min(res, 1.0 - h[idx])
                h[idx] += d
                res -= d
            else:
                idx = int(np.argmax(h))
                if h[idx] < 1e-12:
                    break
                d = min(-res, h[idx])
                h[idx] -= d
                res += d

        h_base = make_feasible(_interpolate(parent, n))
        candidates.append(h_base)
        candidates.append(make_feasible(_antisym(h_base)))

        for sigma in [0.01, 0.03, 0.08, 0.15]:
            for _ in range(3):
                noise = rng.normal(0, sigma, n)
                candidates.append(make_feasible(h_base + noise))

        # Add a structured candidate: piecewise linear
        x = np.linspace(0, 2, n)
        h_struct = 0.5 + 0.3 * np.sin(np.pi * x)
        candidates.append(make_feasible(h_struct))

        # Score and sort
        scored = []
        for c in candidates:
            s = official_score(c)
            if np.isfinite(s):
                scored.append((s, c))
        scored.sort(key=lambda t: t[0])

        top_k = min(6, len(scored))
        betas = [20, 100, 400, 1500]

        for i in range(top_k):
            if time.time() > deadline:
                break
            h_cur = scored[i][1].copy()
            s_cur = scored[i][0]

            # Smooth SLSQP ladder
            for beta in betas:
                if time.time() > deadline:
                    break
                maxiter = min(200, max(50, int(8000 / (n + 10))))
                h_new, s_new = _run_smooth_slsqp(h_cur, beta, maxiter)
                if s_new < s_cur - 1e-12:
                    h_cur = h_new
                    s_cur = s_new

            # Update global best
            if s_cur < best_score - 1e-12:
                best_score = s_cur
                best_h = h_cur.copy()

            # Direct minimax polish
            if time.time() < deadline - 15:
                maxiter_mm = min(150, max(40, int(6000 / (n + 10))))
                h_mm, s_mm = _run_minimax(h_cur, maxiter_mm)
                if s_mm < best_score - 1e-12:
                    best_score = s_mm
                    best_h = h_mm.copy()

    # Final polish on the best solution found
    if time.time() < deadline - 5:
        for beta in [500, 2000, 5000]:
            if time.time() > deadline:
                break
            maxiter = 200
            h_new, s_new = _run_smooth_slsqp(best_h, beta, maxiter)
            if s_new < best_score - 1e-12:
                best_score = s_new
                best_h = h_new

    if time.time() < deadline - 5:
        h_mm, s_mm = _run_minimax(best_h, 200)
        if s_mm < best_score - 1e-12:
            best_score = s_mm
            best_h = h_mm

    # One more round of perturbation + local search if time allows
    if time.time() < deadline - 20:
        for sigma in [0.005, 0.01, 0.02]:
            if time.time() > deadline:
                break
            for attempt in range(5):
                if time.time() > deadline:
                    break
                h_pert = make_feasible(best_h + rng.normal(0, sigma, len(best_h)))
                s_pert = official_score(h_pert)
                if s_pert < best_score:
                    h_pert, s_pert = _run_smooth_slsqp(h_pert, 200, 150)
                    if s_pert < best_score - 1e-12:
                        best_score = s_pert
                        best_h = h_pert
                elif s_pert < best_score + 0.005:
                    h_try, s_try = _run_smooth_slsqp(h_pert, 200, 100)
                    if s_try < best_score - 1e-12:
                        best_score = s_try
                        best_h = h_try

    # Final feasibility check and return
    best_h = make_feasible(best_h)
    final_score = official_score(best_h)
    if final_score < best_score:
        best_score = final_score

    c5 = max(final_score, 0.0) + 1e-13
    if not np.isfinite(c5) or c5 < 0:
        c5 = 0.39
    return best_h, c5, len(best_h)
```
