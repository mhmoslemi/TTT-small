        if not time_left(90):
            break
        new_h, tau = alt_lp_step(current_h, n, dx)
        if np.isinf(tau):
            break
        c5_new = compute_c5(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        # Damped update
        current_h = 0.5 * new_h + 0.5 * current_h
        current_h = project_h(current_h, n)

    # Phase 2: SLSQP with decreasing temperature
    for temp in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if not time_left(60):
            break
        def objective(h_flat):
            h = np.clip(h_flat, 0.0, 1.0)
            corr = np.correlate(h, 1.0 - h, mode="full") * dx
            return smooth_max_fn(corr, temp)
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
                 'jac': lambda h: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            result = minimize(objective, best_h, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 400, 'ftol': 1e-14})
            if result.x is not None:
                h_opt = project_h(result.x, n)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
        except Exception:
            pass

    # Phase 3: Local search with random pairwise perturbations
    if time_left(150):
        local_h = best_h.copy()
        local_c5 = best_c5
        n_ls = len(local_h)
        dx_ls = 2.0 / n_ls
        for it in range(5000):
            if not time_left(30):
                break
            i = rng.randint(n_ls)
            j = rng.randint(n_ls)
            if i == j:
                continue
            delta = rng.uniform(-0.04, 0.04)
            h_try = local_h.copy()
            old_i = h_try[i]
            h_try[i] = np.clip(old_i + delta, 0.0, 1.0)
            actual_delta = h_try[i] - old_i
            h_try[j] = np.clip(local_h[j] - actual_delta, 0.0, 1.0)
            actual_delta2 = h_try[j] - (local_h[j] - actual_delta)
            if abs(actual_delta2) > 1e-9:
                continue
            c5_try = compute_c5(h_try, dx_ls)
            if c5_try < local_c5 - 1e-12:
                local_h = h_try
                local_c5 = c5_try
        if local_c5 < best_c5:
            best_h = local_h
            best_c5 = local_c5

    # Phase 4: Higher resolution
    for n_new in [126, 200, 252]:
        if not time_left(200):
            break
        dx_new = 2.0 / n_new
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_h(h_interp, n_new)

        # LP steps at new resolution
        current_new = h_interp.copy()
        for it in range(15):
            if not time_left(60):
                break
            new_h, tau = alt_lp_step(current_new, n_new, dx_new)
            if np.isinf(tau):
                break
            c5_new = compute_c5(new_h, dx_new)
            if c5_new < best_c5:
                best_h = new_h.copy()
                best_c5 = c5_new
            current_new = 0.5 * new_h + 0.5 * current_new
            current_new = project_h(current_new, n_new)

        # SLSQP refinement
        for temp in [0.01, 0.005]:
            if not time_left(60):
                break
            def obj_new(h_flat):
                h = np.clip(h_flat, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode="full") * dx_new
                return smooth_max_fn(corr, temp)
            cons_new = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_new / 2.0,
                        'jac': lambda h: np.ones(n_new)}]
            bounds_new = [(0.0, 1.0)] * n_new
            try:
                result = minimize(obj_new, current_new, method='SLSQP',
                                 bounds=bounds_new, constraints=cons_new,
                                 options={'maxiter': 300, 'ftol': 1e-14})
                if result.x is not None:
                    h_opt = project_h(result.x, n_new)
                    c5 = compute_c5(h_opt, dx_new)
                    if c5 < best_c5:
                        best_h = h_opt
                        best_c5 = c5
                        current_new = h_opt.copy()
            except Exception:
                pass

    # Final verification
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
