    D = np.sum(d)
    if D < 1e-10:
        return psi, np.zeros(n-1)
    
    A = np.sum(g_h * d)
    grad_y = np.empty(n - 1)
    for j in range(n - 1):
        dtau_dj = (d[n-1] - d[j]) / D
        grad_y[j] = g_h[j] * d[j] + A * dtau_dj
    
    return psi, grad_y

def dyadic_snap(h, n, Q_bits=25):
    Q = 1 << Q_bits
    target_int = n * Q // 2  # exact since n*Q is even when n is odd (Q is power of 2)
    # Actually: target sum = n/2, so target_int = (n/2) * Q = n*Q/2
    # For n=47: target_int = 47*2^25/2 = 47*2^24
    # n*Q = 47 * 2^25, which is odd * power of 2, so n*Q/2 = 47 * 2^24 (integer) ✓
    
    k = np.round(h * Q).astype(np.int64)
    k = np.clip(k, 0, Q)
    diff = target_int - np.sum(k)
    
    if diff > 0:
        # Need to add diff units
        # Prioritize positions where h is closest to the rounded value from below
        fracs = k.astype(np.float64) - h * Q  # h*Q - k, negative means we rounded up
        # We want to add 1 to positions where k is "most below" h*Q
        # i.e., where h*Q - k is largest (positive)
        order = np.argsort(-(h * Q - k))  # most "below" first
        for i in range(diff):
            k[order[i]] += 1
    elif diff < 0:
        diff = -diff
        # Need to subtract diff units
        order = np.argsort(h * Q - k)  # most "above" first
        for i in range(diff):
            k[order[i]] -= 1
    
    k = np.clip(k, 0, Q)
    # Verify sum (in case clipping changed things)
    actual_sum = np.sum(k)
    if actual_sum != target_int:
        # Re-adjust
        diff2 = target_int - actual_sum
        if diff2 > 0:
            for i in range(n):
                if k[i] < Q and diff2 > 0:
                    k[i] += 1
                    diff2 -= 1
        elif diff2 < 0:
            for i in range(n):
                if k[i] > 0 and diff2 < 0:
                    k[i] -= 1
                    diff2 += 1
    
    return k.astype(np.float64) / Q

Actually, the point is that we're computing the gradient of the objective with respect to the free variables y. The constraint is already built into the mapping, so we just need the chain rule gradient, which is what I computed. There's no requirement that sum_j grad_y[j] = 0.

Let me re-verify the formula dh_i/dy_j = d_i * [delta_{i,j} + dtau/dy_j]:

h_i = sigmoid(z_i + tau)
dh_i/dy_j = sigmoid'(z_i + tau) * (dz_i/dy_j + dtau/dy_j)
= d_i * (dz_i/dy_j + dtau/dy_j)

For j < n-1:
dz_i/dy_j = 1 if i = j, -1 if i = n-1, 0 otherwise.

So for i = j: dh_j/dy_j = d_j * (1 + dtau/dy_j)
For i = n-1: dh_{n-1}/dy_j = d_{n-1} * (-1 + dtau/dy_j)
For i != j, i != n-1: dh_i/dy_j = d_i * (0 + dtau/dy_j) = d_i * dtau/dy_j

Now: grad_y[j] = g_j * d_j * (1 + dtau/dy_j) + g_{n-1} * d_{n-1} * (-1 + dtau/dy_j) + sum_{i!=j, i!=n-1} g_i * d_i * dtau/dy_j

= dtau/dy_j * [sum_{i!=j, i!=n-1} g_i*d_i + g_j*d_j + g_{n-1}*d_{n-1}] + g_j*d_j - g_{n-1}*d_{n-1}
= dtau/dy_j * A + g_j*d_j - g_{n-1}*d_{n-1}
= (d_{n-1}-d_j)/D * A + g_j*d_j - g_{n-1}*d_{n-1}

So: grad_y[j] = g_j * d_j + (d_{n-1}-d_j) * A / D - g_{n-1} * d_{n-1}

Let me factor: = g_j*d_j - g_{n-1}*d_{n-1} + (d_{n-1}-d_j)*A/D

This can be written as:
= (g_j*d_j - A*d_j/D) + A*d_{n-1}/D - g_{n-1}*d_{n-1}
= d_j*(g_j - A/D) + d_{n-1}*(A/D - g_{n-1})
= d_j*(g_j - A/D) - d_{n-1}*(g_{n-1} - A/D)

That's a nice symmetric form! Let me use this.

So: grad_y[j] = d_j * (g_j - A/D) - d_{n-1} * (g_{n-1} - A/D)

where A = sum_i g_i * d_i, D = sum_i d_i, and d_i = h_i*(1-h_i).

This is clean and easy to implement.

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def h_to_y(h):
    n = len(h)
    eps = 1e-12
    hc = np.clip(h, eps, 1.0 - eps)
    lh = np.log(hc / (1.0 - hc))
    tau = np.mean(lh)
    z = lh - tau
    return z[:n-1].copy()

def y_to_h(y):
    n = len(y) + 1
    z = np.empty(n)
    z[:n-1] = y
    z[n-1] = -np.sum(y)
    target = n / 2.0
    lo, hi = -60.0, 60.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.sum(1.0 / (1.0 + np.exp(-(z + tau))))
        if s < target:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    h = 1.0 / (1.0 + np.exp(-(z + tau)))
    return np.clip(h, 0.0, 1.0)

def obj_grad(x, n, beta):
    h = y_to_h(x)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    max_shifted = np.max(shifted)
    psi = (max_c + max_shifted / beta + np.log(np.sum(np.exp(shifted - max_shifted))) / beta)
    w = np.exp(shifted - max_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return psi, np.zeros(n - 1)
    w /= w_sum
    g_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        j_min = 0
        j_max = min(n - 1, 2 * n - 2 - k)
        for j in range(j_min, j_max + 1):
            g_h[j + k] += wk * (1.0 - h[j])
            g_h[j] -= wk * h[j + k]
    d = h * (1.0 - h)
    D = np.sum(d)
    if D < 1e-10:
        return psi, np.zeros(n - 1)
    A = np.sum(g_h * d)
    mean_A = A / D
    grad_y = d[:n-1] * (g_h[:n-1] - mean_A) - d[n-1] * (g_h[n-1] - mean_A)
    return psi, grad_y

def dyadic_snap(h, n, q_bits=25):
    Q = 1 << q_bits
    target_int = (n * Q) // 2
    k = np.round(h * Q).astype(np.int64)
    k = np.clip(k, 0, Q)
    diff = target_int - int(np.sum(k))
    if diff > 0:
        fracs = h * Q - k.astype(np.float64)
        order = np.argsort(-fracs)
        for i in range(min(diff, n)):
            if k[order[i]] < Q:
                k[order[i]] += 1
    elif diff < 0:
        diff = -diff
        fracs = h * Q - k.astype(np.float64)
        order = np.argsort(fracs)
        for i in range(min(diff, n)):
            if k[order[i]] > 0:
                k[order[i]] -= 1
    actual = int(np.sum(k))
    diff2 = target_int - actual
    if diff2 > 0:
        for i in range(n):
            if k[i] < Q and diff2 > 0:
                k[i] += 1
                diff2 -= 1
    elif diff2 < 0:
        for i in range(n):
            if k[i] > 0 and diff2 < 0:
                k[i] -= 1
                diff2 += 1
    return k.astype(np.float64) / float(Q)

def pair_polish(h, n, rng, max_rounds=4):
    dx = 2.0 / n
    current = h.copy()
    best_c = evaluate_c5(h)
    for _ in range(max_rounds):
        improved = False
        c_cur = np.correlate(current, 1.0 - current, mode="full")
        n_try = min(400, n * n)
        for _t in range(n_try):
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            if p == q:
                continue
            for src, dst in [(p, q), (q, p)]:
                t_max = min(current[src], 1.0 - current[dst])
                if t_max < 1e-10:
                    continue
                b = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    val = 0.0
                    if 0 <= src - k < n:
                        val -= (1.0 - current[src - k])
                    if 0 <= src + k < n:
                        val += current[src + k]
                    if 0 <= dst - k < n:
                        val += (1.0 - current[dst - k])
                    if 0 <= dst + k < n:
                        val -= current[dst + k]
                    b[k] = val
                n_samp = 30
                ts = np.linspace(0.0, t_max, n_samp)
                vals = c_cur[None, :] + b[None, :] * ts[:, None]
                maxes = np.max(vals, axis=1) * dx
                bi = np.argmin(maxes)
                if maxes[bi] < best_c - 1e-10:
                    current[src] -= ts[bi]
                    current[dst] += ts[bi]
                    best_c = float(maxes[bi])
                    improved = True
                    c_cur = np.correlate(current, 1.0 - current, mode="full")
        if not improved:
            break
    return current, best_c

def try_diff_n(h_parent, n_parent, n_new, rng):
    if n_new <= 1 or n_new % 1 == 0:
        pass
    x_old = np.linspace(0, 2.0, n_new, endpoint=True)
    x_old_mid = (x_old[:-1] + x_old[1:]) / 2
    x_new = np.linspace(0, 2.0, n_new, endpoint=True)
    x_new_mid = (x_new[:-1] + x_new[1:]) / 2
    h_new = np.interp(x_new_mid, x_old_mid, h_parent)
    h_new = np.clip(h_new, 0.0, 1.0)
    total = np.sum(h_new)
    target = n_new / 2.0
    if total > 0:
        h_new = h_new * target / total
        h_new = np.clip(h_new, 0.0, 1.0)
    # Fix sum with dyadic
    h_new = dyadic_snap(h_new, n_new, 22)
    if np.any(h_new < -1e-9) or np.any(h_new > 1.0 + 1e-9):
        return None
    if abs(np.sum(h_new) - target) > 0.01:
        return None
    c = evaluate_c5(h_new)
    return h_new, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    # Get parent
    n = 47
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            parent = np.array(initial_h_values, dtype=np.float64).flatten()
            n = len(parent)
            if np.all(parent >= -1e-9) and np.all(parent <= 1.0 + 1e-9):
                parent = np.clip(parent, 0.0, 1.0)
            else:
                parent = None
        except Exception:
            parent = None

    if parent is None:
        # Fallback: simple feasible vector
        parent = np.zeros(n)
        parent[:n // 2] = 1.0
        if n % 2 == 0:
            pass
        else:
            parent[n // 2] = 0.5
        parent = dyadic_snap(parent, n, 22)

    target_sum = n / 2.0
    if abs(np.sum(parent) - target_sum) > 0.01:
        parent = dyadic_snap(parent, n, 22)

    best_h = parent.copy()
    best_c = evaluate_c5(best_h)

    # Convert to y-space
    y_current = h_to_y(best_h)

    # Stage 1: L-BFGS-B with increasing beta
    betas = [5, 15, 50, 150, 400]
    W = 18.0
    bounds = [(-W, W)] * (n - 1)

    if time.time() < deadline - 30:
        for beta in betas:
            if time.time() > deadline - 60:
                break
            stage_budget = min(120.0, (deadline - 60 - time.time()) / max(1, len(betas)))
            maxfun = max(50, int(stage_budget * 200))
            maxiter = max(20, int(stage_budget * 50))

            def fun(x):
                v, _ = obj_grad(x, n, beta)
                return v

            def jac(x):
                _, g = obj_grad(x, n, beta)
                return g

            try:
                res = minimize(fun, y_current, jac=jac, method='L-BFGS-B',
                               bounds=bounds, maxiter=maxiter, maxfun=maxfun,
                               options={'gtol': 1e-6, 'ftol': 1e-10})
                h_new = y_to_h(res.x)
                h_new = np.clip(h_new, 0.0, 1.0)
                # Check feasibility
                if abs(np.sum(h_new) - target_sum) > 1e-6:
                    h_new = dyadic_snap(h_new, n, 22)
                c_new = evaluate_c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    y_current = h_to_y(h_new)
            except Exception:
                pass

            # Also try from current best with small perturbation
            if time.time() < deadline - 60:
                for _ in range(3):
                    y_pert = y_current + rng.normal(0, 0.3, n - 1)
                    y_pert = np.clip(y_pert, -W, W)
                    try:
                        res2 = minimize(fun, y_pert, jac=jac, method='L-BFGS-B',
                                       bounds=bounds, maxiter=maxiter // 2,
                                       maxfun=maxfun // 2,
                                       options={'gtol': 1e-5, 'ftol': 1e-9})
                        h2 = y_to_h(res2.x)
                        h2 = np.clip(h2, 0.0, 1.0)
                        if abs(np.sum(h2) - target_sum) > 1e-6:
                            h2 = dyadic_snap(h2, n, 22)
                        c2 = evaluate_c5(h2)
                        if c2 < best_c:
                            best_c = c2
                            best_h = h2.copy()
                            y_current = h_to_y(h2)
                    except Exception:
                        pass

    # Stage 2: More random restarts if time allows
    if time.time() < deadline - 90:
        n_restarts = max(5, int((deadline - 90 - time.time()) / 8.0))
        for _ in range(n_restarts):
            if time.time() > deadline - 45:
                break
            scale = rng.choice([0.1, 0.3, 0.5, 1.0, 2.0])
            y_pert = y_current + rng.normal(0, scale, n - 1)
            y_pert = np.clip(y_pert, -W, W)
            beta_r = rng.choice([10, 30, 100])
            maxfun_r = max(30, int(5.0 * 200))
            maxiter_r = max(10, int(5.0 * 50))

            def fun_r(x):
                v, _ = obj_grad(x, n, beta_r)
                return v

            def jac_r(x):
                _, g = obj_grad(x, n, beta_r)
                return g

            try:
                res = minimize(fun_r, y_pert, jac=jac_r, method='L-BFGS-B',
                               bounds=bounds, maxiter=maxiter_r, maxfun=maxfun_r,
                               options={'gtol': 1e-5, 'ftol': 1e-9})
                h_new = y_to_h(res.x)
                h_new = np.clip(h_new, 0.0, 1.0)
                if abs(np.sum(h_new) - target_sum) > 1e-6:
                    h_new = dyadic_snap(h_new, n, 22)
                c_new = evaluate_c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    y_current = h_to_y(h_new)
            except Exception:
                pass

    # Stage 3: Dyadic snap and evaluate
    best_h_snapped = dyadic_snap(best_h, n, 25)
    c_snapped = evaluate_c5(best_h_snapped)
    if c_snapped < best_c:
        best_h = best_h_snapped.copy()
        best_c = c_snapped

    # Stage 4: Pair polish
    if time.time() < deadline - 10:
        for _ in range(3):
            if time.time() > deadline - 5:
                break
            polished, c_pol = pair_polish(best_h, n, rng, max_rounds=3)
            if c_pol < best_c:
                best_h = polished.copy()
                best_c = c_pol
            # Snap after polish
            best_h_snapped2 = dyadic_snap(best_h, n, 25)
            c_s2 = evaluate_c5(best_h_snapped2)
            if c_s2 < best_c:
                best_h = best_h_snapped2.copy()
                best_c = c_s2

    # Stage 5: Try a few different n values
    if time.time() < deadline - 15:
        for n_try in [n - 5, n + 5, n - 10, n + 10, 40, 50, 30, 60]:
            if n_try < 10 or n_try > 200:
                continue
            if time.time() > deadline - 5:
                break
            result = try_diff_n(best_h, n, n_try, rng)
            if result is not None:
                h_try, c_try = result
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
                    n = n_try

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c_final = evaluate_c5(best_h)
    if best_c_final < best_c:
        best_c = best_c_final

    return (best_h, best_c, len(best_h))
