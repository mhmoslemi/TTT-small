def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def repair_feasible(v, target_sum):
    """Project v to [0,1]^n with sum = target_sum"""
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    diff = target_sum - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    iterations = 0
    while abs(diff) > 1e-10 and iterations < 10000:
        if diff > 0:
            # Need to add mass; add to positions with most room
            room = 1.0 - v
            idx = np.argmax(room)
            delta = min(room[idx], diff)
            v[idx] += delta
            diff -= delta
        else:
            # Need to remove mass; remove from positions with most mass
            idx = np.argmax(v)
            delta = min(v[idx], -diff)
            v[idx] -= delta
            diff += delta
        iterations += 1
    return v

    best_c = c_vec.max()

        # moving mass from k to k+shift helps.
