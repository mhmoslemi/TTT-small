                gm[p, e:e+L] += dx * (1.0 - h[:L])
                gm[p, :L] -= dx * h[e:e+L]

    h = np.asarray(h, dtype=float).copy()
    
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(200):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    
    # Fix rounding
    residual = target - np.sum(x)
    if abs(residual) > 1e-14:
        # Adjust the coordinate with most room
        if residual > 0:
            idx = np.argmax(1.0 - x)
        else:
            idx = np.argmin(x)
        x[idx] += residual
        x[idx] = np.clip(x[idx], 0.0, 1.0)
    
    return x

            if min_delta >= 0 and t_hi <= 0:

This looks reasonable. Let me now think about the resolution change.

To interpolate h from resolution n_old to n_new:
- The old grid is x_i = i * 2/n_old for i = 0, ..., n_old-1
- The new grid is y_j = j * 2/n_new for j = 0, ..., n_new-1
- We want h_new[y_j] ≈ h_old interpolated at y_j

Using linear interpolation:
- For y_j, find the corresponding position in the old grid: pos = y_j * n_old / 2
- Let i = floor(pos), frac = pos - i
- h_new[y_j] = (1-frac) * h_old[i] + frac * h_old[(i+1) % n_old]

After interpolation, the sum won't be exactly n_new/2, so we need to rescale:
- h_new *= (n_new/2) / sum(h_new)
- Then clip to [0,1] and fix the sum

def optimize_logsumexp(h, n, time_limit, start_time, eps_start=0.1, eps_end=0.01):
    """Optimize using log-sum-exp smoothed objective with projected gradient"""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    def ls_c5(h, eps):
        lags = compute_lags(h)
        m = np.max(lags)
        return eps * np.log(np.sum(np.exp((lags - m) / eps))) + m
    
    def ls_grad(h, eps):
        lags = compute_lags(h)
        m = np.max(lags)
        w = np.exp((lags - m) / eps)
        w /= np.sum(w)
        gm = compute_grad_matrix(h)
        return w @ gm  # Weighted sum of gradients
    
    # Annealed log-sum-exp
    n_iter = 200
    for it in range(n_iter):
        if time.time() - start_time > time_limit - 5:
            break
        
        # Interpolate eps
        t = it / max(n_iter - 1, 1)
        eps = eps_start * (eps_end / eps_start) ** t
        
        # Line search
        grad = ls_grad(h, eps)
        cur_val = ls_c5(h, eps)
        
        # Try decreasing step size
        step = 0.5
        found = False
        for _ in range(30):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, n)
            new_val = ls_c5(h_new, eps)
            if new_val < cur_val:
                h = h_new
                found = True
                # Check actual C5
                actual = compute_c5(h)
                if actual < best_c5:
                    best_c5 = actual
                    best_h = h.copy()
                break
            step *= 0.5
        
        if not found:
            # No improvement, try smaller step or break
            if step < 1e-10:
                break
    
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    gm = np.zeros((2 * n - 1, n))
    for p in range(2 * n - 1):
        d = p - (n - 1)
        if d >= 0:
            L = n - d
            if L > 0:
                gm[p, :L] += dx * (1.0 - h[d:d+L])
                gm[p, d:d+L] -= dx * h[:L]
        else:
            e = -d
            L = n - e
            if L > 0:
                gm[p, e:e+L] += dx * (1.0 - h[:L])
                gm[p, :L] -= dx * h[e:e+L]
    return gm

def project_to_feasible(h, n):

    hi = h.max() + 2.0

        if residual > 0:
            free = x < 1.0 - 1e-15
            if np.any(free):
                add = residual / np.sum(free)
                x[free] = np.minimum(x[free] + add, 1.0)
            residual = target - np.sum(x)
            if abs(residual) > 1e-13:

                x = np.clip(x, 0, 1)

                idx = np.argmax(1.0 - x)

    n_old = len(h_old)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    # Treat as periodic
    h_interp = np.interp(x_new, x_old, h_old, period=2.0)
    # Fix sum
    target = n_new / 2.0
    current_sum = np.sum(h_interp)
    if current_sum > 0:
        h_interp *= target / current_sum
    h_interp = np.clip(h_interp, 0.0, 1.0)
    # Final sum fix
    residual = target - np.sum(h_interp)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h_interp)
        else:
            order = np.argsort(h_interp)

                h_interp[idx] -= rem
                residual += rem

Wait, I realize there might be an issue with the `optimize_logsumexp` function. The `cur_ls` calculation is wrong. Let me fix it:

The log-sum-exp value is:
f(h) = eps * log(sum_p exp(C_p(h) / eps))

The gradient is:
grad f = (1/eps) * sum_p exp(C_p/eps) * grad C_p / sum_p exp(C_p/eps)
= sum_p softmax_p * grad C_p

where softmax_p = exp(C_p/eps) / sum_q exp(C_q/eps)

For numerical stability, we should use:
softmax_p = exp((C_p - C_max)/eps) / sum_q exp((C_q - C_max)/eps)

And the log-sum-exp value is:
f(h) = C_max + eps * log(sum_p exp((C_p - C_max)/eps))
