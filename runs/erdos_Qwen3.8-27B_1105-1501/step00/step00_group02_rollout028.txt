import numpy as np
import time

def objective(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c * dx)

def is_feasible(h, tol=1e-9):
    n = len(h)
    T = n / 2.0
    if np.any(h < -tol) or np.any(h > 1+tol):
        return False
    if abs(h.sum() - T) > tol * n:
        return False
    return True

def project_to_feasible(z, T):
    """Project z onto {y: 0<=y_i<=1, sum(y)=T} using bisection on shift."""
    z = np.array(z, dtype=float)
    # g(t) = sum(clip(z_i - t, 0, 1)) is non-increasing in t
    # Find t such that g(t) = T
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        g = np.clip(z - mid, 0.0, 1.0).sum()
        if g > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    residual = T - y.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add more; find indices with y < 1
            mask = y < 1.0 - 1e-15
            if mask.sum() > 0:
                add = residual / mask.sum()
                add = min(add, (1.0 - y[mask]).max())
                y[mask] += add
                # Still might have residual
                residual2 = T - y.sum()
                if abs(residual2) > 1e-12:
                    # Distribute among those with most slack
                    idx = np.where(mask)[0]
                    slack = 1.0 - y[idx]
                    order = np.argsort(-slack)
                    for i in order:
                        d = min(residual2, slack[i]) if residual2 > 0 else min(-residual2, -slack[i])
                        y[idx[i]] += d if residual2 > 0 else -(-d)
                        residual2 -= d
                        if abs(residual2) < 1e-12:
                            break
        else:
            # Need to subtract; find indices with y > 0
            mask = y > 1e-15
            if mask.sum() > 0:
                sub = -residual / mask.sum()
                sub = min(sub, y[mask].max())
                y[mask] -= sub
    y = np.clip(y, 0.0, 1.0)
    # Final exact fix
    residual = T - y.sum()
    if abs(residual) > 1e-14:
        idx = int(np.argmax(y if residual < 0 else 1-y))
        y[idx] += residual
        y = np.clip(y, 0.0, 1.0)
    return y

def compute_gradient(h, active_lags=None):
    """Compute sum of ∂F_r/∂h[m] over active lags for each m."""
    n = len(h)
    dx = 2.0 / n
    # Compute all F_r
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx  # F[r] for r = 0..2n-2
    
    if active_lags is None:
        # Use all lags
        active_lags = range(len(F))
    
    grad = np.zeros(n)
    for r in active_lags:
        d = r - (n - 1)  # shift
        # For each valid i: F_r includes h[i]*(1-h[i+d])
        # ∂F_r/∂h[m]:
        #   If m is first factor (i=m) and m+d valid: dx*(1-h[m+d])
        #   If m is second factor (i+d=m, i=m-d) and m-d valid: -dx*h[m-d]
        lo = max(0, -d)
        hi = min(n, n - d)
        # First factor: h[i]*(1-h[i+d]) for i in [lo, hi-1]
        # ∂/∂h[i] = dx*(1-h[i+d])
        for i in range(lo, hi):
            grad[i] += dx * (1.0 - h[i + d])
            # Second factor: h[i]*(1-h[i+d]), ∂/∂h[i+d] = -dx*h[i]
            grad[i + d] -= dx * h[i]
    return grad

    n = len(h)

def pair_line_search(h, p, q, F):
    """
    For pair (p,q) with transfer t from p to q:
    h_new[p] = h[p] - t, h_new[q] = h[q] + t
    Find t that minimizes max_r F_r(t).
    Returns (best_t, best_value, new_h) or (0, F_max, None) if no improvement.
    """
    n = len(h)
    dx = 2.0 / n
    F_max = np.max(F)
    
    # Feasible range for t
    t_lo = max(h[p] - 1.0, -h[q])
    t_hi = min(h[p], 1.0 - h[q])
    if t_hi - t_lo < 1e-12:
        return 0.0, F_max, None
    
    # Compute G and H for all lags
    G = np.zeros(2*n - 1)
    H = np.zeros(2*n - 1)
    
    for r in range(2*n - 1):
        d = r - (n - 1)
        valid_p_first = 0 <= p + d < n
        valid_q_first = 0 <= q + d < n
        valid_p_second = 0 <= p - d < n
        valid_q_second = 0 <= q - d < n
        
        if p + d == q:
            G[r] = -(1.0 - h[q]) - h[p]
            if valid_q_first:
                G[r] += (1.0 - h[q + d])
            if valid_p_second:
                G[r] += h[p - d]
            H[r] = 1.0
        elif p - d == q:
            G[r] = (1.0 - h[p]) + h[q]
            if valid_p_first:
                G[r] -= (1.0 - h[p + d])
            if valid_q_second:
                G[r] -= h[q - d]
            H[r] = 1.0
        else:

            if valid_p_first:
                g -= (1.0 - h[p + d])
            if valid_q_first:
                g += (1.0 - h[q + d])
            if valid_p_second:
                g += h[p - d]
            if valid_q_second:
                g -= h[q - d]
            G[r] = g
            H[r] = 0.0
    
    G *= dx
    H *= dx
    
    # Now minimize f(t) = max(F + G*t + H*t^2) over t in [t_lo, t_hi]
    # This is a convex function of t.
    # Use grid + refinement.
    n_grid = 32
    ts = np.linspace(t_lo, t_hi, n_grid)

    # F shape (2n-1,), G shape (2n-1,), H shape (2n-1,)
    # For each t: F + G*t + H*t^2
    vals = F[np.newaxis, :] + G[np.newaxis, :] * ts[:, np.newaxis] + H[np.newaxis, :] * ts[:, np.newaxis]**2
    obj_vals = np.max(vals, axis=1)
    
    best_idx = np.argmin(obj_vals)
    best_t = ts[best_idx]
    best_val = obj_vals[best_idx]

    if best_idx == 0:
        lo, hi = t_lo, ts[1]
    elif best_idx == n_grid - 1:
        lo, hi = ts[-2], t_hi
    else:
        lo, hi = ts[best_idx - 1], ts[best_idx + 1]

    vals2 = F[np.newaxis, :] + G[np.newaxis, :] * ts2[:, np.newaxis] + H[np.newaxis, :] * ts2[:, np.newaxis]**2
    obj_vals2 = np.max(vals2, axis=1)
    best_idx2 = np.argmin(obj_vals2)
    best_t = ts2[best_idx2]
    best_val = obj_vals2[best_idx2]
    
    if best_val >= F_max - 1e-12:
        return 0.0, F_max, None
    
    # Create new h
    new_h = h.copy()
    new_h[p] -= best_t
    new_h[q] += best_t
    new_h = np.clip(new_h, 0.0, 1.0)
    # Fix sum
    residual = n/2.0 - new_h.sum()
    if abs(residual) > 1e-12:

            new_h[q] += best_t
            new_h = np.clip(new_h, 0.0, 1.0)
    
    return best_t, best_val, new_h

def full_pass(h, deadline):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0-h, mode='full') * dx
    F_max = F.max()
    
    # Precompute FF and SF
    # FF[m, r] = (1-h[m+d]) if 0<=m+d<n else 0, where d = r-(n-1)
    # SF[m, r] = h[m-d] if 0<=m-d<n else 0
    FF = np.zeros((n, 2*n-1))
    SF = np.zeros((n, 2*n-1))
    for r in range(2*n-1):
        d = r - (n-1)
        # First factor: m+d valid
        m_min = max(0, -d)
        m_max = min(n, n-d)
        for m in range(m_min, m_max):
            FF[m, r] = 1.0 - h[m+d]
        # Second factor: m-d valid
        m_min2 = max(0, d)
        m_max2 = min(n, n+d)
        for m in range(m_min2, m_max2):
            SF[m, r] = h[m-d]
    
    # For each pair (p,q), compute G = -FF[p]+FF[q]+SF[p]-SF[q] (times dx)
    # H is zero except at specific r values
    
    best_delta = 0.0
    best_p, best_q, best_t = -1, -1, 0.0
    
    for p in range(n):
        for q in range(n):
            if p == q:
                continue
            if time.time() > deadline:
                return h, F
            
            t_lo = max(h[p]-1.0, -h[q])
            t_hi = min(h[p], 1.0-h[q])
            if t_hi - t_lo < 1e-12:
                continue
            
            G = (-FF[p] + FF[q] + SF[p] - SF[q]) * dx
            
            # H: non-zero only where p+d==q or p-d==q
            H = np.zeros(2*n-1)
            # p+d==q => d=q-p, r = q-p+n-1
            d1 = q - p
            r1 = d1 + n - 1
            if 0 <= r1 < 2*n-1:
                H[r1] = dx
            d2 = p - q
            r2 = d2 + n - 1
            if 0 <= r2 < 2*n-1:
                H[r2] = dx
            # Note: r1 != r2 since p != q
            
            # Minimize max(F + G*t + H*t^2)
            n_grid = 20
            ts = np.linspace(t_lo, t_hi, n_grid)
            vals = F + G * ts[:, None] + H * ts[:, None]**2
            obj_vals = vals.max(axis=1)
            bi = obj_vals.argmin()
            bv = obj_vals[bi]
            
            if bv < F_max - best_delta - 1e-13:
                # Refine
                lo2 = ts[max(0, bi-1)]
                hi2 = ts[min(n_grid-1, bi+1)]
                ts2 = np.linspace(lo2, hi2, 32)
                vals2 = F + G * ts2[:, None] + H * ts2[:, None]**2
                obj2 = vals2.max(axis=1)
                bi2 = obj2.argmin()
                bv2 = obj2[bi2]
                if bv2 < F_max - best_delta - 1e-13:
                    best_delta = F_max - bv2
                    best_p, best_q, best_t = p, q, ts2[bi2]
    
    if best_p >= 0:
        new_h = h.copy()
        new_h[best_p] -= best_t
        new_h[best_q] += best_t
        new_h = np.clip(new_h, 0.0, 1.0)
        new_F = np.correlate(new_h, 1.0-new_h, mode='full') * dx
        if new_F.max() < F_max - 1e-13:
            return new_h, new_F
    return h, F

def c5_objective(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0-h, mode='full')) * dx

def project_feasible(z, T):
    z = np.asarray(z, dtype=float).copy()
    lo, hi = z.min()-1.0, z.max()+1.0
    for _ in range(80):
        mid = (lo+hi)/2.0
        if np.clip(z-mid, 0, 1).sum() > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(z-(lo+hi)/2.0, 0, 1)
    r = T - y.sum()
    if abs(r) > 1e-10:
        mask = (y < 1-1e-12) if r > 0 else (y > 1e-12)
        if mask.any():
            y[mask] += r / mask.sum()
            y = np.clip(y, 0, 1)
    r = T - y.sum()
    if abs(r) > 1e-15:
        idx = int(np.argmax(y if r < 0 else 1-y))
        y[idx] += r
    return y

def steepest_pass(h, deadline):

    n = len(h)

        for m in range(mlo, mhi):

        mlo2 = max(0, d)
        mhi2 = min(n, n+d)
        for m in range(mlo2, mhi2):

    for p in range(n):
        for q in range(p+1, n):  # unordered pairs
            if time.time() > deadline:
                break
            # Try both directions: p->q and q->p
            for src, dst in [(p, q), (q, p)]:

                t_hi = min(h[src], 1.0-h[dst])

                G = (-FF[src] + FF[dst] + SF[src] - SF[dst]) * dx
                H = np.zeros(2*n-1)

                if bv < F_max - best_delta - 1e-13:
                    # Refine
                    if bi == 0:
                        lo2, hi2 = t_lo, ts[1]
                    elif bi == n_grid-1:
                        lo2, hi2 = ts[-2], t_hi
                    else:
                        lo2, hi2 = ts[bi-1], ts[bi+1]
                    ts2 = np.linspace(lo2, hi2, 40)

        new_h[best_p] -= best_t
        new_h[best_q] += best_t

        # Verify

        if nv < F_max - 1e-13:
            return new_h, nv, True
    return h, F_max, False
