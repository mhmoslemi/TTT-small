from scipy.optimize import linprog
import time

def evaluate_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def correlations(h):
    """Return all correlation values (unscaled)"""
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h):
    """Repair h to satisfy 0<=h_i<=1 and sum(h)=n/2"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > target:
        excess = s - target
        # Reduce from largest values
        idx = np.argsort(h)[::-1]
        for i in idx:
            reduce = min(excess, h[i])
            h[i] -= reduce
            excess -= reduce
            if excess <= 1e-15:
                break
    elif s < target:
        deficit = target - s
        # Increase from smallest values
        idx = np.argsort(h)
        for i in idx:
            increase = min(deficit, 1.0 - h[i])
            h[i] += increase
            deficit -= increase
            if deficit <= 1e-15:
                break
    # Final exact correction
    residual = target - np.sum(h)
    if abs(residual) > 0:
        # Add to a coordinate that can absorb it
        if residual > 0:
            for i in range(n):
                room = 1.0 - h[i]
                if room >= residual:
                    h[i] += residual
                    break
        else:
            for i in range(n):
                if h[i] >= -residual:
                    h[i] += residual
                    break
    return h

def compute_gradient(h, k):

def solve_tangent_lp(h0, corrs, active_indices, trust_radius):
    """Solve tangent LP subproblem"""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    # Variables: h_0..h_{n-1}, t (total n+1 vars)
    # Minimize: t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Constraints: t >= c_m + G_m · (h - h0) for each active m
    # => G_m · h - t <= c_m - G_m · h0
    A_ub = []
    b_ub = []
    for m in active_indices:
        G = compute_gradient(h0, m)
        c_m = corrs[m]  # unscaled correlation value
        # G · h - t <= c_m - G · h0
        row = np.zeros(n + 1)
        row[:n] = G
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(c_m - np.dot(G, h0))
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_radius)
        hi = min(1.0, h0[i] + trust_radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
        # Try interior point
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='interior-point')
        if res.status == 0:
            return res.x[:n]
    except:
        pass
    return None

        while si < n_s - 1 and (si + 1) * dx_s <= t_left + 1e-15:

            si += 1

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    s = np.sum(h)
    if s > target:
        excess = s - target
        idx = np.argsort(h)[::-1]
        for i in idx:
            red = min(excess, h[i])
            h[i] -= red
            excess -= red
            if excess <= 1e-14:
                break
    elif s < target:
        deficit = target - s
        idx = np.argsort(h)
        for i in idx:
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-14:
                break
    res = target - np.sum(h)
    if abs(res) > 1e-15:
        if res > 0:
            for i in range(n):
                if 1.0 - h[i] >= res:
                    h[i] += res
                    break
        else:
            for i in range(n):
                if h[i] >= -res:
                    h[i] += res
                    break
    return h

def compute_gradient(h, k):
    n = len(h)
    ell = k - (n - 1)
    grad = np.zeros(n)
    if ell >= 0:
        p = ell
        if p < n:
            grad[p:] += (1.0 - h[:n - p])
            grad[:n - p] -= h[p:]
    else:
        p = -ell
        if p < n:
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
    return grad

def select_active(corrs, n, q=15):
    center = n - 1
    active = [center]
    top = np.argsort(corrs)[::-1]
    for m in top:
        mi = int(m)
        if mi not in active:
            active.append(mi)
        if len(active) >= q:
            break
    return active

def solve_tangent_lp(h0, corrs, active_indices, trust_radius):
    n = len(h0)
    target = n / 2.0

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub_list = []
    b_ub_list = []
    for m in active_indices:
        G = compute_gradient(h0, m)
        c_m = corrs[m]
        row = np.zeros(n + 1)
        row[:n] = G
        row[n] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(c_m - np.dot(G, h0))

    if not A_ub_list:
        return None

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_radius)
        hi = min(1.0, h0[i] + trust_radius)
        bounds.append((lo, hi))
    bounds.append((None, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='interior-point')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def pair_line_search(h, max_pairs=40, samples=15, seed=0):
    n = len(h)
    best_h = h.copy()
    best_val = evaluate_c5(h)

    corrs = get_correlations(h)
    q = min(12, n // 3)
    top_idx = np.argsort(corrs)[::-1][:q]
    G_avg = np.zeros(n)

        G_avg += compute_gradient(h, m)
    if q > 0:
        G_avg /= q

    rng = np.random.default_rng(seed)
    sorted_g = np.argsort(G_avg)

    high_set = sorted_g[::-1][:max(4, n // 6)]

    pairs = []
    for i in high_set:
        for j in low_set:
            if i != j:
                pairs.append((i, j))
    while len(pairs) < max_pairs:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j:
            pairs.append((i, j))

    seen = set()
    for (i, j) in pairs[:max_pairs]:
        key = (min(i, j), max(i, j))
        if key in seen:
            continue
        seen.add(key)

        d_min = max(-h[i], h[j] - 1.0)
        d_max = min(1.0 - h[i], h[j])
        if d_max - d_min < 1e-10:
            continue

        deltas = np.linspace(d_min, d_max, samples)
        best_d = 0.0
        best_v = best_val

        for d in deltas:
            if abs(d) < 1e-12:
                continue
            h[i] += d
            h[j] -= d
            if h[i] < -1e-12 or h[i] > 1 + 1e-12 or h[j] < -1e-12 or h[j] > 1 + 1e-12:
                h[i] -= d
                h[j] += d
                continue
            v = evaluate_c5(h)
            h[i] -= d
            h[j] += d
            if v < best_v - 1e-10:
                best_v = v
                best_d = d

        if best_d != 0.0:
            best_h[i] += best_d
            best_h[j] -= best_d
            best_h = repair(best_h)
            best_val = evaluate_c5(best_h)

            # Local polish around this point
            h2 = best_h.copy()
            v2 = best_val
            for _ in range(3):
                dd_min = max(-h2[i], h2[j] - 1.0)
                dd_max = min(1.0 - h2[i], h2[j])
                if dd_max - dd_min < 1e-10:
                    break
                ds = np.linspace(dd_min, dd_max, samples)
                bd = 0.0
                bv = v2
                for d in ds:
                    if abs(d) < 1e-12:
                        continue
                    h2[i] += d
                    h2[j] -= d
                    if h2[i] < -1e-12 or h2[i] > 1 + 1e-12:
                        h2[i] -= d
                        h2[j] += d
                        continue
                    v = evaluate_c5(h2)
                    h2[i] -= d
                    h2[j] += d
                    if v < bv - 1e-10:
                        bv = v
                        bd = d
                if bd == 0.0:
                    break
                h2[i] += bd
                h2[j] -= bd
                v2 = evaluate_c5(repair(h2))
                h2 = repair(h2)
            best_h = h2
            best_val = v2

    return best_h, best_val

def transfer_resolution(h_source, n_target):
    n_s = len(h_source)
    dx_s = 2.0 / n_s
    dx_t = 2.0 / n_target
    h_t = np.zeros(n_target)

    for k in range(n_target):
        t_left = k * dx_t
        t_right = (k + 1) * dx_t
        area = 0.0
        si = int(np.floor(t_left / dx_s))
        si = max(0, min(si, n_s - 1))
        while si < n_s:
            s_left = si * dx_s
            s_right = (si + 1) * dx_s
            if s_left >= t_right - 1e-14:
                break
            ov = min(s_right, t_right) - max(s_left, t_left)
            if ov > 1e-15:
                area += h_source[si] * ov
            si += 1
        h_t[k] = area / dx_t

    h_t = np.clip(h_t, 0.0, 1.0)
    h_t = repair(h_t)
    return h_t
