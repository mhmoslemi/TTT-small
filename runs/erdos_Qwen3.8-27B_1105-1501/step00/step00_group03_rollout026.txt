import numpy as np
from scipy.optimize import minimize
import time

_SMOOTH_BETA = 200.0

def official_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel()
    n = len(z)
    if n == 0:
        return z
    if target is None:
        target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - float(np.max(z))
    hi = 1.0 - float(np.min(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                d = min(diff, 1.0 - h[idx])
                h[idx] += d
            else:
                d = min(-diff, h[idx])
                h[idx] -= d
            diff = target - np.sum(h)
            if abs(diff) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def manual_correlation(h):
    n = len(h)
    out = np.zeros(2 * n - 1)
    for lag_idx in range(2 * n - 1):
        k = lag_idx - (n - 1)
        start = max(0, -k)
        end = min(n, n - k)
        if start < end:
            i = np.arange(start, end)
            j = i + k
            out[lag_idx] = np.dot(h[i], 1.0 - h[j])
    return out

def smooth_obj(h):
    n = len(h)
    beta = _SMOOTH_BETA
    dx = 2.0 / n
    vals = manual_correlation(h) * dx
    cmax = np.max(vals)
    scaled = np.clip(beta * (vals - cmax), -500.0, 500.0)
    return cmax + np.log(np.sum(np.exp(scaled))) / beta

def smooth_grad(h):
    n = len(h)
    beta = _SMOOTH_BETA
    dx = 2.0 / n
    vals = manual_correlation(h) * dx
    cmax = np.max(vals)
    scaled = np.clip(beta * (vals - cmax), -500.0, 500.0)
    exps = np.exp(scaled)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        k = lag_idx - (n - 1)
        start = max(0, -k)
        end = min(n, n - k)
        if start < end:
            i = np.arange(start, end)
            j = i + k
            np.add.at(grad, i, (1.0 - h[j]) * w[lag_idx])
            np.add.at(grad, j, (-h[i]) * w[lag_idx])

        h_anti = make_feasible(h_anti, n_opt / 2.0)

            h_r = base + rng.standard_normal(n_opt) * sigma

            candidates.append(h_r)

    final_n = len(final_h)
    final_score = official_score(final_h)
    
    return (final_h, final_score + 1e-13, final_n)

Actually, passing a dict as a parameter is fine - it's not a closure. The function `try_update` takes `best` as an argument.

But wait, the rule says "no closures or lambdas" for helper functions. A function that takes a dict parameter is not a closure. A closure is a function that captures variables from an enclosing scope. So as long as `try_update` doesn't reference `best_h` or `best_score` from the enclosing scope, it's fine.

def make_feasible(z, target=None):
    z = np.asarray(z, dtype=float).ravel()
    n = len(z)
    if n == 0:
        return z
    if target is None:
        target = n / 2.0
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -1.0 - float(np.max(z))
    hi = 1.0 - float(np.min(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                d = min(diff, 1.0 - h[idx])
                h[idx] += d
            else:
                d = min(-diff, h[idx])
                h[idx] -= d
            diff = target - np.sum(h)
            if abs(diff) < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)

def manual_correlation(h):
    n = len(h)
