from scipy import optimize
import time


def project_onto_constraint(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h, n_points):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx"""
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode="full")
    return c.max() * dx


def get_max_lags(h, n_points):
    """Return the lags where the maximum is achieved"""
    c = np.correlate(h, 1.0 - h, mode="full")
    max_val = c.max()
    indices = np.where(c >= max_val - 1e-12)[0]
    return indices - (n_points - 1)  # convert to actual lags


def smooth_max_weights(c, t):
    """Compute softmax weights for smooth max approximation"""
    c_shifted = t * (c - c.max())
    c_shifted -= c_shifted.max()
    w = np.exp(c_shifted)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return w / s

    """Compute gradient of smooth_max(correlate(h, 1-h), t) w.r.t. h"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(c, t)
    grad = np.zeros(n)
    
    one_minus_h = 1.0 - h
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        m = j - (n - 1)  # lag
        # Term 1: (1-h[i-m]) for i in [max(0,m), min(n, n+m)]
        #   valid i: 0 <= i-m < n => max(0,m) <= i < min(n, n+m)
        i_lo1 = max(0, m)
        i_hi1 = min(n, n + m)
        if i_lo1 < i_hi1:
            # h[i-m] for i in [i_lo1, i_hi1) means indices [i_lo1-m, i_hi1-m)
            grad[i_lo1:i_hi1] += wj * one_minus_h[i_lo1 - m:i_hi1 - m]
        
        # Term 2: -h[i+m] for i in [max(0,-m), min(n, n-m)]
        i_lo2 = max(0, -m)
        i_hi2 = min(n, n - m)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wj * h[i_lo2 + m:i_hi2 + m]

def project_onto_constraint(v, target_sum):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode="full")
    return c.max() * dx


def smooth_max_weights(c, t):
    c_shifted = t * (c - c.max())
    c_shifted -= c_shifted.max()
    w = np.exp(c_shifted)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return w / s


def gradient_of_smooth_max(h, t):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(c, t)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-15:
            continue
        m = j - (n - 1)
        
        # Term: +wj * (1-h[i-m]) for i where 0 <= i-m < n
        i_lo = max(0, m)
        i_hi = min(n, n + m)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wj * one_minus_h[i_lo - m:i_hi - m]
        
        # Term: -wj * h[i+m] for i where 0 <= i+m < n
        i_lo2 = max(0, -m)
        i_hi2 = min(n, n - m)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wj * h[i_lo2 + m:i_hi2 + m]
    
    return grad

                        time_budget, start_time, momentum_init=0.5):
    target_sum = n_points / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    v = np.zeros(n_points)
    
    for step in range(n_steps):
        if time.time() - start_time > time_budget:
            break
        
        progress = step / max(n_steps - 1, 1)
        t = t_start * (t_end / t_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        momentum = momentum_init * (1 - progress) + 0.3 * progress
        
        grad = gradient_of_smooth_max(h, t)
        v = momentum * v + (1 - momentum) * grad
        h_new = project_onto_constraint(h - lr * v, target_sum)
        
        c5_new = compute_c5(h_new, n_points)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

            break
        if time_budget is not None and start_time is not None:

                break

        for sw in swap_sizes:
            for i in range(n):
                if best[i] < sw:
                    continue
                for j in range(n):

                    trial[i] -= sw

                        best_c5 = c5

                    break
            if improved or eval_count >= max_evals:
                break

            break

            best_h = h_new.copy()

import numpy as np
from scipy import optimize
import time


def project_onto_constraint(v, target_sum):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode="full")
    return c.max() * dx


def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max_weights(c, t):
    c_shifted = t * (c - c.max())
    c_shifted -= c_shifted.max()
    w = np.exp(c_shifted)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return w / s


def gradient_of_smooth_max(h, t):
    n = len(h)
    c = compute_correlations(h)
    w = smooth_max_weights(c, t)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-15:
            continue
        m = j - (n - 1)
        i_lo = max(0, m)
        i_hi = min(n, n + m)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wj * one_minus_h[i_lo - m:i_hi - m]
        i_lo2 = max(0, -m)
        i_hi2 = min(n, n - m)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wj * h[i_lo2 + m:i_hi2 + m]
    return grad


def smooth_grad_descent(h_init, n_points, n_steps, t_start, t_end, lr_start, lr_end,
