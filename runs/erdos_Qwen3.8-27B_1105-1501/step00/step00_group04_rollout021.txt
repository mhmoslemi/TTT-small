import numpy as np
import time

def score_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    Y = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(Y)


def overlap_vector(h):
    """Compute unscaled overlap vector."""
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h_raw, n):
    """Project h_raw onto {0<=h_i<=1, sum(h)=n/2} using bisection water-filling."""
    target = n * 0.5
    y = np.clip(np.asarray(h_raw, dtype=float), 0.0, 1.0)
    
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    
    # Bisection for lambda
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(z)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix residual
    r = target - np.sum(z)
    if abs(r) > 1e-14:
        # Distribute residual
        if r > 0:
            mask = z < 1.0
            idx = np.where(mask)[0]
            if len(idx) > 0:
                per = r / len(idx)
                for i in idx:
                    add = min(per, 1.0 - z[i])
                    z[i] += add
                    r -= add
        else:
            mask = z > 0.0
            idx = np.where(mask)[0]
            if len(idx) > 0:
                per = -r / len(idx)
                for i in idx:
                    rem = min(per, z[i])
                    z[i] -= rem
                    r += rem
    
    return z


def gradient_row(h, p, n):
    """Compute gradient of overlap component p w.r.t. h."""
    center = n - 1
    g = np.zeros(n)
    
    if p == center:
        g[:] = 1.0 - 2.0 * h
    elif p < center:
        d = center - p
        # dY_p/dh_i = (1-h_{i+d})*[i <= n-1-d] - h_{i-d}*[i >= d]
        for i in range(n - d):
            g[i] += (1.0 - h[i + d])
        for i in range(d, n):
            g[i] -= h[i - d]
    else:
        d = p - center
        # dY_p/dh_i = (1-h_{i-d})*[i >= d] - h_{i+d}*[i <= n-1-d]
        for i in range(d, n):
            g[i] += (1.0 - h[i - d])
        for i in range(n - d):
            g[i] -= h[i + d]
    
    return g


def gradient_row_vectorized(h, p, n):
    """Vectorized gradient computation."""
    center = n - 1
    g = np.zeros(n)
    
    if p == center:
        g[:] = 1.0 - 2.0 * h
    elif p < center:
        d = center - p
        if n - d > 0:
            g[:n-d] += (1.0 - h[d:])
        if d < n:
            g[d:] -= h[:n-d]
    else:
        d = p - center
        if d < n:
            g[d:] += (1.0 - h[:n-d])
        if n - d > 0:
            g[:n-d] -= h[d:]
    
    return g


def get_active_set(h, n, tol_frac=0.05, tol_abs=0.002, max_active=32):
    """Get active overlap indices (near-max)."""
    Y = overlap_vector(h)
    M = np.max(Y)
    threshold = M - (tol_frac * M + tol_abs)
    active = np.where(Y >= threshold)[0]
    # Sort by value descending
    order = np.argsort(-Y[active])
    active = active[order]
    if len(active) > max_active:
        active = active[:max_active]
    return active, Y, M


def try_qp_direction(h, active, Y, M, n, rho, eta_factor, strict=True, slack_weight=10.0):
    """Solve QP subproblem to find descent direction."""
    if not HAS_CVXPY or len(active) == 0:
        return None
    
    # Build gradient rows
    rows = []
    rhs = []
    for p in active:
        g = gradient_row_vectorized(h, p, n)
        norm = np.linalg.norm(g)
        if norm < 1e-12:
            continue
        eta = max(1e-4, eta_factor * rho * norm)
        # Cap eta
        eta = min(eta, 0.15 * max(0.5, M))
        rows.append(g)
        rhs.append(M - Y[p] - eta)
    
    if len(rows) == 0:
        return None
    
    G = np.array(rows)  # shape (k, n)
    rhs = np.array(rhs)
    
    u = cp.Variable(n)
    constraints = [
        u >= -h + 1e-10,
        u <= 1.0 - h - 1e-10,
        cp.sum(u) == 0
    ]
    
    if strict:
        for i in range(len(rows)):
            constraints.append(cp.matmul(rows[i], u) <= rhs[i])
        prob = cp.Problem(cp.Minimize(0.5 * cp.sum_squares(u)), constraints)
    else:
        s_vars = cp.Variable(len(rows), nonneg=True)
        for i in range(len(rows)):
            constraints.append(cp.matmul(rows[i], u) - s_vars[i] <= rhs[i])
        prob = cp.Problem(
            cp.Minimize(0.5 * cp.sum_squares(u) + slack_weight * cp.sum(s_vars)),
            constraints
        )
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000) if prob.constraints else None
        # Try multiple solvers
        try:
            prob.solve(solver=cp.ECOS)
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                prob.solve()
    except:
        return None
    
    if u.value is None:
        return None
    
    u_val = np.asarray(u.value, dtype=float).flatten()
    if not np.all(np.isfinite(u_val)):
        return None
    
    # Remove mean to ensure sum zero
    u_val -= np.mean(u_val)
    
    # Ensure bounds feasibility of step
    alpha_max = 1.0
    pos_mask = u_val > 1e-12
    neg_mask = u_val < -1e-12
    if np.any(pos_mask):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / u_val[pos_mask]))
    if np.any(neg_mask):
        alpha_max = min(alpha_max, np.min(-h[neg_mask] / u_val[neg_mask]))
    
    if alpha_max <= 1e-8:
        return None
    
    return u_val


def line_search(h, direction, n, curr_c5, accept_tol=1e-10):
    """Exact line search along direction. Returns (best_h, best_c5) or (None, curr_c5)."""
    pos_mask = direction > 1e-12
    neg_mask = direction < -1e-12
    alpha_max = 1.0
    if np.any(pos_mask):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / direction[pos_mask]))
    if np.any(neg_mask):
        alpha_max = min(alpha_max, np.min(-h[neg_mask] / direction[neg_mask]))
    
    if alpha_max <= 1e-8:
        return None, curr_c5
    
    alphas = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]
    alphas = [a * alpha_max for a in alphas if a * alpha_max > 1e-10]
    
    best_h = None
    best_c5 = curr_c5
    
    for a in alphas:
        trial = h + a * direction
        # Check feasibility
        if np.any(trial < -1e-10) or np.any(trial > 1.0 + 1e-10):
            continue
        # Check mass
        if abs(np.sum(trial) - n * 0.5) > 1e-8:
            continue
        c5 = score_c5(trial)
        if c5 < best_c5 - accept_tol:
            best_c5 = c5
            best_h = trial.copy()
    
    return best_h, best_c5


def pair_polish(h, n, Y, M, max_pairs=200, accept_tol=1e-10):
    """Try pair mass transfers to improve."""
    center = n - 1
    dx = 2.0 / n
    best_h = h.copy()

    rng = np.random.default_rng(12345)

    agg_grad = np.zeros(n)
    for p in active:
        agg_grad += gradient_row_vectorized(h, p, n)
    
    # For pair (i,j): direction is e_i - e_j (move mass from j to i)
    # Best alpha: minimize over valid range
    # Try pairs ordered by gradient difference
    # Want to increase h_i where agg_grad[i] < 0 and decrease where agg_grad[j] > 0
    
    sorted_idx = np.argsort(agg_grad)

                # Check mass
                c5 = score_c5(trial)

        if abs(float(np.sum(trial)) - 0.5 * n) > 1e-6:
            continue
        c = score_c5(trial)
        if c < best_c5 - 1e-11:
            best_c5 = c
            best_h = trial.copy()
    return best_h, best_c5


def do_pair_polish(h, n, Y, M, max_pairs, seed_val):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = dx * M
    thresh = M - (0.1 * M + 0.005)
    active = np.where(Y >= thresh)[0]
    if len(active) == 0:
        active = np.array([n - 1])
    ag = np.zeros(n)
    for p in active:
        ag += grad_row(h, p, n)
    inc_idx = np.where(ag < -1e-8)[0]
    inc_idx = inc_idx[np.argsort(ag[inc_idx])]
    dec_idx = np.where(ag > 1e-8)[0]
    dec_idx = dec_idx[np.argsort(-ag[dec_idx])]
    n_inc = min(len(inc_idx), 35)
    n_dec = min(len(dec_idx), 35)
    tried = 0
    rng = np.random.default_rng(seed_val)
    shuffle_a = rng.permutation(n_inc)
    shuffle_b = rng.permutation(n_dec)
    for si in shuffle_a:
        if tried >= max_pairs:
            break
        i = inc_idx[si]
        for sj in shuffle_b:
            if tried >= max_pairs:
                break
            j = dec_idx[sj]
            if i == j:
                continue
            tried += 1
            a_max = min(1.0 - best_h[i], best_h[j])
            if a_max <= 1e-10:
                continue
            for frac in [0.5, 0.75, 0.9, 1.0, 0.3, 0.6]:
                a = frac * a_max
                if a < 1e-10:
                    continue
                trial = best_h.copy()
                trial[i] += a
                trial[j] -= a
                c = score_c5(trial)
                if c < best_c5 - 1e-11:
                    best_c5 = c
                    best_h = trial
    return best_h, best_c5


def do_coord_descent(h, n, Y, M, max_moves, seed_val):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = dx * M
    thresh = M - (0.08 * M + 0.004)
    active = np.where(Y >= thresh)[0]
    if len(active) == 0:
        active = np.array([n - 1])
    ag = np.zeros(n)
    for p in active:
        ag += grad_row(h, p, n)
    order = np.argsort(np.abs(ag))[::-1]
    rng = np.random.default_rng(seed_val + 777)
    perm = rng.permutation(len(order))
    moves = 0
    for pi in perm:
        if moves >= max_moves:
            break
        i = order[pi]
        if abs(ag[i]) < 1e-8:
            continue
        moves += 1
        if ag[i] < 0:
            deltas = [0.05, 0.1, 0.2, 0.02, 0.15]
        else:
            deltas = [-0.05, -0.1, -0.2, -0.02, -0.15]
        for dl in deltas:
            trial = best_h.copy()
            trial[i] += dl
            if trial[i] < -1e-9 or trial[i] > 1.0 + 1e-9:
                continue
            if abs(float(np.sum(trial)) - 0.5 * n) > 1e-6:
                continue
            c = score_c5(trial)
            if c < best_c5 - 1e-11:
                best_c5 = c
                best_h = trial
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 5.0

    try:
        h0 = np.asarray(initial_h_values, dtype=float).flatten()
        if len(h0) < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad")
    except Exception:
        h0 = make_feasible(np.random.default_rng(seed).random(70), 70)

    n = len(h0)
    h = make_feasible(h0, n)
    curr_c5 = score_c5(h)

    best_h = h.copy()
    best_c5 = curr_c5
    best_n = n

    # Try a few scaled variants
    for lam in [0.8, 0.9, 1.1, 1.25]:
        hv = make_feasible(h0 * lam, n)
        cv = score_c5(hv)
        if cv < best_c5:
            best_h = hv.copy()
            best_c5 = cv
            best_n = n

    # Also try binary-threshold variants
    for thresh_t in [0.3, 0.4, 0.5, 0.6]:
        hb = np.where(h0 > thresh_t, 1.0, 0.0)
        hb = make_feasible(hb, n)
        cb = score_c5(hb)
        if cb < best_c5:
            best_h = hb.copy()
            best_c5 = cb
            best_n = n

    # Main loop
    no_imp = 0
    rho = 0.12
    eta_f = 0.25
    iter_count = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break
        iter_count += 1

        # Adaptive parameters
        phase = min(iter_count / 50.0, 3.0)
        tol_frac = 0.06 - 0.015 * phase
        tol_abs = 0.004 - 0.001 * phase
        max_a = int(16 + 8 * (1.0 - phase / 3.0))

        active, Y, M = get_active(h, n, max(tol_frac, 0.01), max(tol_abs, 0.001), max(max_a, 8))

        improved = False

        # Try QP direction
        d = qp_solve(h, active, Y, M, n, rho, eta_f, True, 10.0)
        if d is None and rem > 15.0:
            d = qp_solve(h, active, Y, M, n, rho, eta_f, False, 3.0)

        if d is not None:
            new_h, new_c5 = do_line_search(h, d, n, curr_c5)
            if new_h is not None:
                h = new_h
                curr_c5 = new_c5
                if curr_c5 < best_c5 - 1e-12:
                    best_h = h.copy()
                    best_c5 = curr_c5
                    best_n = n
                improved = True
                rho = min(rho * 1.04, 0.3)

        if not improved:
            # Pair polish
            new_h, new_c5 = do_pair_polish(h, n, Y, M, 120, iter_count * 13 + seed)
            if new_c5 < curr_c5 - 1e-11:
                h = new_h
                curr_c5 = new_c5
                if curr_c5 < best_c5 - 1e-12:
                    best_h = h.copy()
                    best_c5 = curr_c5
                    best_n = n
                improved = True

        if not improved:
            # Coordinate descent
            new_h, new_c5 = do_coord_descent(h, n, Y, M, 80, iter_count * 31 + seed)
            if new_c5 < curr_c5 - 1e-11:
                h = new_h
                curr_c5 = new_c5
                if curr_c5 < best_c5 - 1e-12:
                    best_h = h.copy()
                    best_c5 = curr_c5
                    best_n = n
                improved = True

        if not improved:
            no_imp += 1
            rho *= 0.82
            if no_imp >= 8:
                # Perturbation to escape
                scale = 0.03 + 0.02 * (no_imp % 4)
                rng_p = np.random.default_rng(int(time.monotonic() * 100000) % (2**31) + seed)
                h = make_feasible(h + scale * rng_p.standard_normal(n), n)
                curr_c5 = score_c5(h)
                rho = 0.12
                eta_f = max(eta_f * 0.85, 0.1)
                no_imp = 0
                # Re-sync with best if we drifted
                if curr_c5 > best_c5 + 0.002:
                    h = best_h.copy()
                    curr_c5 = best_c5
        else:
            no_imp = 0
            eta_f = max(eta_f * 0.98, 0.08)

    # Final polish on best
    if time.monotonic() < deadline - 2.0:
        Y_b = np.correlate(best_h, 1.0 - best_h, mode='full')
        M_b = float(np.max(Y_b))
        fh, fc = do_pair_polish(best_h, best_n, Y_b, M_b, 300, 99999 + seed)
        if fc < best_c5 - 1e-12:
            best_h = fh
            best_c5 = fc
            best_n = best_n
        if time.monotonic() < deadline - 1.0:
            fh2, fc2 = do_coord_descent(best_h, best_n, Y_b, M_b, 200, 88888 + seed)
            if fc2 < best_c5 - 1e-12:
                best_h = fh2
                best_c5 = fc2

    # Final verification
    final_c5 = score_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, best_c5, best_n
