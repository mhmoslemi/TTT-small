import time

def compute_R(h):
    return np.correlate(h, 1.0 - h, mode="full")

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_R(h)) * dx

def grad_matrix(h):
    """Compute grad[i, k] = dR_k/dh_i for all i, k"""
    n = len(h)
    grad = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for k in range(2 * n - 1):
            d = k - (n - 1)
            gi = 0.0
            if 0 <= i - d < n:
                gi += (1.0 - h[i - d])
            if 0 <= i + d < n:
                gi -= h[i + d]
            grad[i, k] = gi
    return grad

def find_best_move(h, R, grad):
    """Find the best pair move (i, j, delta) that reduces max(R)."""
    n = len(h)
    best_imp = 1e-14
    best_i = best_j = -1
    best_d = 0.0
