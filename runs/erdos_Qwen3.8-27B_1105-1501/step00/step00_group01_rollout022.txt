```python
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _golden_section(f, a, b, tol=1e-10, max_iter=50):
    """Minimize convex function f on [a,b] using golden section."""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / gr
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / gr
            fd = f(d)
    return f((a + b) / 2.0), (a + b) / 2.0
```

```python
        if 0 <= j-lag < n:
            candidates.add(j-lag)
        
        for m in candidates:
            m2 = m + lag
            if not (0 <= m < n and 0 <= m2 < n):
                continue
            # Term h[m] * (1 - h[m2])
            # h[m] changes by: +delta if m==i, -delta if m==j, 0 otherwise
            # h[m2] changes by: +delta if m2==i, -delta if m2==j, 0 otherwise
            dm = (1.0 if m == i else 0.0) - (1.0 if m == j else 0.0)
            dm2 = (1.0 if m2 == i else 0.0) - (1.0 if m2 == j else 0.0)
            # New term: (h[m]+dm*δ) * (1-h[m2]-dm2*δ)
            # = h[m]*(1-h[m2]) + δ*[-h[m]*dm2 + (1-h[m2])*dm] + δ²*(-dm*dm2)
            a += -h[m]*dm2 + (1.0-h[m2])*dm
            b += -dm*dm2
        
        A[idx] = a
        B[idx] = b
    
    return C, A, B
```
