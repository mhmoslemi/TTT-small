def score(h):
    """Exact C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(raw)

def project_feasible(z, target):
    """Project z onto {h: 0<=h_i<=1, sum(h)=target} via binary search on water level."""
    z = np.clip(z, -10, 10)  # handle non-finite
    z = np.where(np.isfinite(z), z, 0.5)
    
    # Binary search for lambda such that sum(clamp(z - lambda, 0, 1)) = target
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        v = np.clip(z - mid, 0.0, 1.0)
        s = np.sum(v)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    v = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix any floating point residual
    residual = target - np.sum(v)
    if abs(residual) > 1e-14:
        # Distribute residual among non-boundary elements
        mask = (v > 1e-10) & (v < 1.0 - 1e-10)
        if np.any(mask):
            v[mask] += residual / np.sum(mask)
            v = np.clip(v, 0.0, 1.0)
        else:
            # Add to elements closest to 0.5
            idx = np.argsort(np.abs(v - 0.5))[:max(1, abs(int(residual)))]
            for i in idx:
                adj = min(max(residual, -v[i], v[i] - 1.0), 1.0 - v[i])
                v[i] += adj
                residual -= adj
    
    return v

import numpy as np
import time

def score(h):
    """Exact C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(raw))

def project_feasible(z, target):
    """Project z onto {h: 0<=h_i<=1, sum(h)=target} via water-filling."""
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, -20.0, 20.0)
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    v = np.clip(z - lam, 0.0, 1.0)
    residual = target - np.sum(v)
    if abs(residual) > 1e-14:
        mask = (v > 1e-12) & (v < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            v[mask] += residual / np.sum(mask)
            v = np.clip(v, 0.0, 1.0)
    return v

def soft_gradient(h, alpha):
    """Gradient of soft-max smoothed min-max objective."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    raw = np.correlate(h, g, mode="full")
    scaled = alpha * dx * raw
    mx = np.max(scaled)
    exp_vals = np.exp(scaled - mx)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-15:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            l = n - lag
            if l <= 0:
                continue
            grad[lag:] += wk * dx * (1.0 - h[0:l])
            grad[0:l] -= wk * dx * h[lag:]
        else:
            p = -lag
            l = n - p
            if l <= 0:
                continue
            grad[0:l] += wk * dx * (1.0 - h[p:])
            grad[p:] -= wk * dx * h[0:l]
    grad -= np.mean(grad)
    return grad

def active_gradient(h, top_k=3):
    """Gradient from top-k active shifts (uniform weight)."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    raw = np.correlate(h, g, mode="full")
    idx_top = np.argsort(raw)[::-1][:top_k]
    w = np.zeros(2 * n - 1)
    for i in idx_top:
        w[i] = 1.0 / len(idx_top)
    grad = np.zeros(n)
    for idx in idx_top:
        wk = w[idx]
        lag = idx - (n - 1)
        if lag == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            l = n - lag
            if l <= 0:
                continue
            grad[lag:] += wk * dx * (1.0 - h[0:l])
            grad[0:l] -= wk * dx * h[lag:]
        else:
            p = -lag
            l = n - p
            if l <= 0:
                continue
            grad[0:l] += wk * dx * (1.0 - h[p:])
            grad[p:] -= wk * dx * h[0:l]
    grad -= np.mean(grad)
    return grad

def line_search(h0, C0, direction, target, n_points=25):
    """Projected line search along direction."""
    n = len(h0)
    abs_max = np.max(np.abs(direction))
    if abs_max < 1e-15:
        return h0, C0
    d = direction / abs_max
    best_h = h0.copy()
    best_C = C0
    # Log-spaced steps + some linear
    steps = np.concatenate([
        np.logspace(-3, 0.3, 12),
        np.linspace(0.3, 3.0, 13)
    ])
    for t in steps:
        z = h0 + t * d
        h_new = project_feasible(z, target)
        if np.max(np.abs(h_new - h0)) < 1e-15:
            continue
        C_new = score(h_new)
        if C_new < best_C - 1e-13:
            best_C = C_new
            best_h = h_new.copy()
    return best_h, best_C

def line_search_bidir(h0, C0, direction, target):
    """Try both +d and -d directions."""
    h1, C1 = line_search(h0, C0, direction, target)
    h2, C2 = line_search(h0, C0, -direction, target)
    if C1 <= C2:
        return h1, C1
    return h2, C2

def optimize_h(h, max_time, alpha_start=5.0, alpha_end=300.0, n_max_iters=8000, seed=42):

    n = len(h)

        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        elif alpha < 100:

            else:
                grad = active_gradient(h, 4)

        direction = momentum

        try bidirectional first for high alpha
        if alpha > 50 and it % 4 == 0:
            h_new, C_new = line_search_bidir(h, C, direction, target)
        else:
            h_new, C_new = line_search(h, C, direction, target)

            h = h_new
            C = C_new
            stall_count = 0

                best_C = C
                best_h = h.copy()
        else:
            stall_count += 1
            # Random perturbation every 30 stalls
            if stall_count % 30 == 0:
                pert = rng.randn(n)
                pert -= np.mean(pert)
                h_new, C_new = line_search(h, C, pert, target, n_points=15)

                    C = C_new

                momentum = np.zeros(n)
                pert = rng.randn(n)
                pert -= np.mean(pert)
                h_new, C_new = line_search_bidir(h, C, pert, target)
                if C_new < C - 1e-12:
                    h = h_new
                    C = C_new
                    stall_count = 0

def resample_h(h_old, n_new):
    """Resample from n_old to n_new via midpoint interpolation."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])

def random_valid_h(n, rng):
    """Generate a random feasible h."""
    h = rng.rand(n)
    return project_feasible(h, n / 2.0)

def sinusoid_h(n, rng):
    """Generate a sinusoidal-shaped feasible h."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    a1, b1 = rng.uniform(-0.3, 0.3), rng.uniform(-0.3, 0.3)
    a2, b2 = rng.uniform(-0.2, 0.2), rng.uniform(-0.2, 0.2)
    z = 0.5 + a1 * np.sin(np.pi * x) + b1 * np.cos(np.pi * x) \
         + a2 * np.sin(2 * np.pi * x) + b2 * np.cos(2 * np.pi * x)
    return project_feasible(z, n / 2.0)
