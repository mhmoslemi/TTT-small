import numpy as np
import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(R) * dx)

def repair_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if diff > 1e-12:
        order = np.argsort(-h)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            reduce = min(h[idx], remaining)
            h[idx] -= reduce
            remaining -= reduce
    elif diff < -1e-12:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            increase = min(1.0 - h[idx], remaining)
            h[idx] += increase
            remaining -= increase
    return h

def compute_soft_gradient(h, alpha):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = np.max(R)
    exp_R = np.exp(alpha * (R - R_max))
    w = exp_R / np.sum(exp_R)
    a_rev = (1.0 - h)[::-1]
    grad = np.correlate(w, a_rev, mode='valid') - h * w[n - 1]
    return grad

def soft_grad_descent(h, target_sum, alpha, lr, n_steps=200):
    """Gradient descent on soft-max objective."""
    n = len(h)
    current = h.copy()
    best = h.copy()
    best_score = score_exact(h)
    
    for step in range(n_steps):
        grad = compute_soft_gradient(current, alpha)
        # Project to sum=0
        grad = grad - np.mean(grad)
        # Step
        new_h = current - lr * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = repair_feasible(new_h, target_sum)
        # Check exact objective
        new_score = score_exact(new_h)
        if new_score < best_score:
            best = new_h.copy()
            best_score = new_score
        # Also accept if same (for exploration)
        current = new_h
    
    return best, best_score

def local_search(h, target_sum, n_pairs=1000, dbeta=0.01, n_iters=100):
    """Local search with pair transfers."""
    n = len(h)
    current = h.copy()
    best = h.copy()
    best_score = score_exact(h)
    rng = np.random.default_rng(42)
    
    for it in range(n_iters):
        improved = False
        for _ in range(n_pairs):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            for direction in [1, -1]:
                delta = direction * dbeta
                if 0 <= current[i] + delta <= 1 and 0 <= current[j] - delta <= 1:
                    current[i] += delta
                    current[j] -= delta
                    s = score_exact(current)
                    if s < best_score:
                        best = current.copy()
                        best_score = s
                        improved = True
                    else:
                        current[i] -= delta
                        current[j] += delta
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try smaller steps
            dbeta *= 0.7
            if dbeta < 1e-5:
                break
    
    return best, best_score

            inc = min(1.0 - h[idx], remaining)
            h[idx] += inc
            remaining -= inc
    return h

def compute_soft_gradient(h, alpha):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = np.max(R)
    exp_R = np.exp(alpha * (R - R_max))
    w = exp_R / np.sum(exp_R)
    a_rev = (1.0 - h)[::-1]
    grad = np.correlate(w, a_rev, mode='valid') - h * w[n - 1]
    return grad

def grad_optimize(h0, target_sum, alpha, lr, n_steps, time_limit, momentum=0.9):
    n = len(h0)
    current = h0.copy()
    best = h0.copy()
    best_score = score_exact(h0)
    vel = np.zeros(n)
    
    for step in range(n_steps):
        if time.time() > time_limit:
            break
        grad = compute_soft_gradient(current, alpha)
        grad -= np.mean(grad)
        # Momentum
        vel = momentum * vel - lr * grad
        new_h = current + vel
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = repair_feasible(new_h, target_sum)
        # Check improvement
        cur_score = score_exact(current)
        new_score = score_exact(new_h)
        if new_score <= cur_score + 1e-12:
            current = new_h
        else:
            # Reject, reduce velocity
            vel *= 0.5
            # Try smaller step
            new_h = current - lr * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = repair_feasible(new_h, target_sum)
            current = new_h
            vel = 0.0
        if score_exact(current) < best_score:
            best = current.copy()
            best_score = score_exact(best)
    
    return best, best_score

def pair_local_search(h0, target_sum, time_limit, rng, init_dbeta=0.05):
    n = len(h0)
    current = h0.copy()
    best = h0.copy()
    best_score = score_exact(h0)
    dbeta = init_dbeta
    no_improve = 0
    
    while time.time() < time_limit:
        improved = False
        n_tries = 200
        for _ in range(n_tries):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for sign in [1, -1]:
                delta = sign * dbeta
                hi = current[i] + delta
                hj = current[j] - delta
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    current[i] = hi
                    current[j] = hj
                    s = score_exact(current)
                    if s < best_score:
                        best = current.copy()
                        best_score = s
                        improved = True
                    else:
                        current[i] -= delta
                        current[j] += delta
                    if improved:
                        break
            if improved:
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 5:
                dbeta *= 0.5
                if dbeta < 1e-5:
                    break
            no_improve = 0 if dbeta != init_dbeta else no_improve
    
    return best, best_score

def resample_to(h0, new_n):
    """Resample h0 from its current size to new_n."""
    old_n = len(h0)
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(x_new, x_old, h0)
    return h_new
