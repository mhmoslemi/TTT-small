        t_trial = min(n_per, budget_s - (time.time() - start_time) - 30)
        if t_trial < 8:
            break
        
        if best_h is not None:
            h_start = interpolate_h(best_h, n_t)
        else:
            h_start = make_random_initial(n_t, rng)
        
        h_opt, C_opt = optimize_sa(h_start, t_trial, rng)
        if C_opt < best_C:
            best_C = C_opt
            best_h = h_opt
    
    # Phase 3: Final polish on best solution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 5
    if best_h is not None and remaining > 10:
        h_pol, C_pol = optimize_sa(best_h, remaining, rng, T0=0.01, T1=0.0002)
        if C_pol < best_C:
            best_C = C_pol
            best_h = h_pol
    
    # Final evaluation
    if best_h is not None:
        h_final = np.clip(best_h, 0.0, 1.0)
        n_final = len(h_final)
        target = n_final / 2.0
        residual = target - h_final.sum()
        if abs(residual) > 1e-10:
            if residual > 0:
                room = 1.0 - h_final
                s = room.sum()
                if s > 1e-12:
                    h_final += residual * room / s
            else:
                s = h_final.sum()
                if s > 1e-12:
                    h_final += residual * h_final / s
            h_final = np.clip(h_final, 0.0, 1.0)
        final_C = evaluate_erdos_solution(h_final)
        if final_C <= best_C + 1e-9:
            best_h = h_final
            best_C = final_C
    
    if best_h is None:
        best_h = make_random_initial(95, rng)
        best_C = evaluate_erdos_solution(best_h)
    
    return best_h, best_C, len(best_h)
