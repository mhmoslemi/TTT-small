def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_c5_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - float(np.sum(result))
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - result))
        else:
            idx = int(np.argmax(result))
        result[idx] += diff
    return np.clip(result, 0.0, 1.0)

    s = np.sum(exp_vals)
    if s < 1e-300:
        return np.ones(len(c5_vals)) / len(c5_vals)
    return exp_vals / s
