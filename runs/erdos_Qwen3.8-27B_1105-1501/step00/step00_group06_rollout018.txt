```python
def compute_gradient(h, beta):
    """Compute gradient of LSE-smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation values c_d for d = -(n-1), ..., n-1
    # c_d = sum_{valid i} h[i+d] * (1-h[i])
    r = np.correlate(h, 1.0 - h, mode="full")
    # r[k] = c_{k-(n-1)} for k = 0, ..., 2n-2
    
    # Softmax weights
    max_r = np.max(r)
    exp_r = np.exp(beta * (r - max_r))
    w = exp_r / np.sum(exp_r)
    
    # Gradient
    grad = np.zeros(n)
    for p in range(n):
        for k in range(2 * n - 1):
            d = k - (n - 1)
            # ∂c_d/∂h[p] = [1 - h[p-d]] * 1_{0<=p-d<n} - h[p+d] * 1_{0<=p+d<n}
            val = 0.0
            if 0 <= p - d < n:
                val += 1.0 - h[p - d]
            if 0 <= p + d < n:
                val -= h[p + d]
            grad[p] += w[k] * val
    grad *= dx
    
    return grad
```

```python
import time

def _evaluate(h):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(r)

def _project_box_mass(v):
    n = len(v)
    target = n / 2.0
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    
    # Find mu via bisection
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    y = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    res = target - np.sum(y)
    if abs(res) > 1e-12:
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            y[free] += res / nf
            y = np.clip(y, 0.0, 1.0)
            # Second correction
            res2 = target - np.sum(y)
            if abs(res2) > 1e-12 and nf > 0:

                y = np.clip(y, 0.0, 1.0)
    
    return y

def _compute_grad(h, beta):
    """Analytical gradient of LSE-smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    max_r = np.max(r)
    exp_r = np.exp(beta * (r - max_r))
    w = exp_r / np.sum(exp_r)
    
    # w is indexed 0..2n-2, corresponding to d = -(n-1), ..., n-1
    # grad[p] = dx * sum_d w[n-1+d] * [(1-h[p-d])*1_{0<=p-d<n} - h[p+d]*1_{0<=p+d<n}]
    
    grad = np.zeros(n)
    for p in range(n):
        s1 = 0.0
        s2 = 0.0
        # First term: sum over d where 0 <= p-d < n, i.e., p-(n-1) <= d <= p
        # d ranges from max(-(n-1), p-(n-1)) to min(n-1, p)
        # Since d >= -(n-1) always, lower = p-(n-1) if p > 0, else -(n-1)
        d_lo = max(-(n-1), p - (n-1))
        d_hi = min(n-1, p)
        for d in range(int(d_lo), int(d_hi) + 1):
            s1 += w[n-1+d] * (1.0 - h[p-d])
        
        # Second term: sum over d where 0 <= p+d < n, i.e., -p <= d <= n-1-p
        d_lo2 = max(-(n-1), -p)
        d_hi2 = min(n-1, n-1-p)
        for d in range(int(d_lo2), int(d_hi2) + 1):
            s2 += w[n-1+d] * h[p+d]
        
        grad[p] = dx * (s1 - s2)
    
    return grad

def _line_search(h, direction, current_val, n_evals=15):
    """Line search in direction 'direction' from h. Returns (best_h, best_val)."""
    n = len(h)
    
    # Compute max feasible step
    max_step = 1.0
    for i in range(n):
        if direction[i] > 1e-14:
            max_step = min(max_step, (1.0 - h[i]) / direction[i])
        elif direction[i] < -1e-14:
            max_step = min(max_step, h[i] / (-direction[i]))
    
    if max_step <= 0:
        return h.copy(), current_val
    
    max_step *= 0.9999
    
    # Try several step sizes
    best_val = current_val
    best_h = h.copy()
    
    # Logarithmic spacing from very small to max_step
    steps = np.geomspace(max_step * 1e-3, max_step, n_evals)
    # Also include 0 and max_step
    steps = np.concatenate([[0.0], steps, [max_step]])
    
    # Sort and unique
    steps = np.unique(steps)
    
    for alpha in steps:
        if alpha == 0:
            continue
        h_try = h + alpha * direction
        h_try = np.clip(h_try, 0.0, 1.0)
        # Re-project to fix mass
        h_try = _project_box_mass(h_try)
        val = _evaluate(h_try)
        if val < best_val - 1e-15:
            best_val = val
            best_h = h_try
    
    return best_h, best_val

def _pair_polish(h, current_val, max_delta_trials=8):
    """Try pair transfers to improve."""
    n = len(h)
    best_h = h.copy()
    best_val = current_val
    improved = True
    
    while improved:
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Direction: increase h[i], decrease h[j], or vice versa
                # Feasible delta range for h[i] += delta, h[j] -= delta:
                # 0 <= h[i]+delta <= 1 and 0 <= h[j]-delta <= 1
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                
                if d_hi - d_lo < 1e-14:
                    continue
                
                # Try several values
                for delta in np.linspace(d_lo, d_hi, max_delta_trials+2)[1:-1]:
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    h_try = np.clip(h_try, 0.0, 1.0)
                    # Mass is preserved by construction
                    val = _evaluate(h_try)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_h = h_try.copy()
                        h = best_h.copy()
                        improved = True
        if improved:
            # Restart the pass
            pass
    
    return best_h, best_val
```
