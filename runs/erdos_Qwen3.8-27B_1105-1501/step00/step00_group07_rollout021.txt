import time

def _score(h):

    if e > 1e-12:

        rem = e
        for idx in order:
            if rem <= 1e-12:

            d = min(v[idx], rem)
            v[idx] -= d
            rem -= d
    elif e < -1e-12:

        rem = -e
        for idx in order:
            if rem <= 1e-12:

            d = min(1.0 - v[idx], rem)
            v[idx] += d
            rem -= d

def _softmax_grad(h, alpha):
    """Gradient of soft-max of correlation values, centered."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    y = raw * dx
    mx = y.max()
    z = np.exp(alpha * (y - mx))
    s = z.sum()
    if s < 1e-300:
        return None
    w = z / s

    D = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx] * dx
        if abs(wk) < 1e-15:
            continue

        if k >= 0:
            length = n - k
            # C_k = dx * sum_{j=0}^{length-1} h[j+k]*(1-h[j])
            # dC_k/dh[i]:
            #   (1-h[i-k]) if k <= i < n  [i = j+k means j = i-k, valid if 0<=i-k<length]
            #   -h[i+k]   if 0 <= i < length  [i = j means j+k = i+k, valid if 0<=i<length]
            for i in range(n):
                c = 0.0
                if k <= i and i - k < length:
                    c += 1.0 - h[i - k]
                if 0 <= i < length:
                    c += -h[i + k]
                if c != 0.0:
                    D[i] += wk * c
        else:
            m = -k
            length = n - m
            # C_{-m} = dx * sum_{j=m}^{n-1} h[j-m]*(1-h[j])
            # dC_{-m}/dh[i]:
            #   (1-h[i+m]) if 0 <= i < length  [i = j-m means j = i+m, valid if m<=i+m<n i.e. 0<=i<n-m]
            #   -h[i-m]   if m <= i < n        [i = j, valid if m<=i<n]
            for i in range(n):
                c = 0.0
                if 0 <= i < length:
                    c += 1.0 - h[i + m]
                if m <= i < n:
                    c += -h[i - m]
                if c != 0.0:
                    D[i] += wk * c

    return D

def _score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(raw))

def _project_feasible(z, target):
    v = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    e = v.sum() - target
    if e > 1e-12:
        order = np.argsort(-v, kind='stable')
        rem = e
        for idx in order:
            if rem <= 1e-14: break
            d = min(v[idx], rem)
            v[idx] -= d
            rem -= d
    elif e < -1e-12:
        order = np.argsort(1.0 - v, kind='stable')
        rem = -e
        for idx in order:
            if rem <= 1e-14: break
            d = min(1.0 - v[idx], rem)
            v[idx] += d
            rem -= d
    return v

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    y = raw * dx
    mx = y.max()
    z = np.exp(alpha * (y - mx))
    s = z.sum()
    if s < 1e-300:
        return None
    w = z / s
    D = np.zeros(n)

        wk = w[idx]
        if wk < 1e-12:
            continue
        k = idx - (n - 1)

            L = n - k
            if L > 0:
