import numpy as np

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _fix_sum(h, target):
    """Fix sum to target while maintaining [0,1] bounds."""
    n = len(h)
    res = target - np.sum(h)
    if abs(res) < 1e-13:
        return h
    if res > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if res > 0:
            d = min(res, 1.0 - h[idx])
            h[idx] += d
            res -= d
        else:
            d = min(-res, h[idx])
            h[idx] -= d
            res += d
        if abs(res) < 1e-14:
            break
    return h

import numpy as np
import time

def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_feasible(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    res = target - np.sum(h)
    if abs(res) < 1e-13:
        return h
    if res > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if res > 0:
            d = min(res, 1.0 - h[idx])
        else:
            d = max(res, -h[idx])
        h[idx] += d
        res -= d
        if abs(res) < 1e-14:
            break
    return h

def _line_search(h, i, j, dx, c5_init, max_iter=22):
    a_lo = max(-h[i], h[j] - 1.0)
    a_hi = min(1.0 - h[i], h[j])
    if a_hi - a_lo < 1e-12:
        return h, c5_init
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = a_lo
    b = a_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h_t = h.copy()
    h_t[i] = h[i] + c
    h_t[j] = h[j] - c
    fc = _c5(h_t, dx)
    h_t[i] = h[i] + d
    h_t[j] = h[j] - d
    fd = _c5(h_t, dx)
    for _ in range(max_iter):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            h_t[i] = h[i] + c; h_t[j] = h[j] - c
            fc = _c5(h_t, dx)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            h_t[i] = h[i] + d; h_t[j] = h[j] - d
            fd = _c5(h_t, dx)
    if fc < fd:
        alpha, f_best = c, fc
    else:
        alpha, f_best = d, fd
    if f_best >= c5_init:
        return h, c5_init
    h_new = h.copy()
    h_new[i] += alpha
    h_new[j] -= alpha
    return h_new, f_best

def _gradient_index(h, n):
    """Compute which coordinate pairs are 'high risk' based on active lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr  # unscaled
    mx = np.max(c_vals)
    active = np.where(c_vals >= mx * 0.97)[0]
    active = active[:30]  # limit
    risk = np.zeros(n)
    for k in active:
        tau = k - (n - 1)
        for i in range(n):
            # d(c[k])/d(h[i])
            di = 0.0
            j1 = i - tau
            if 0 <= j1 < n and 0 <= j1 + tau < n:
                di += (1.0 - h[j1])
            j2 = i
            if 0 <= j2 < n and 0 <= j2 + tau < n:
                di += (-h[j2 + tau])
            risk[i] += abs(di)
    return risk
