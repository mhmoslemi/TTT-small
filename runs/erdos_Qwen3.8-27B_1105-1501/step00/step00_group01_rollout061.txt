import time

def c5_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Binary search on lambda
    lo, hi = h.min() - target_sum/n, h.max() - target_sum/n + 1
    # Actually, lo should be such that clip(h - lo, 0, 1).sum() >= target_sum
    # and hi such that clip(h - hi, 0, 1).sum() <= target_sum
    
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h - (lo + hi) / 2, 0, 1)

def prox_gradient_descent(h, target_sum, n_iter=1000, tau_start=20, tau_end=200):
    """Proximal gradient descent with log-sum-exp smoothing."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for it in range(n_iter):
        # Adaptive tau
        frac = it / max(n_iter - 1, 1)
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Compute gradient
        corr = np.correlate(h, 1 - h, mode='full')
        max_c = np.max(corr)
        exp_c = np.exp(tau * dx * (corr - max_c))
        weights = exp_c / np.sum(exp_c)
        
        grad_h = np.zeros(n)
        for k in range(2*n - 1):
            d = k - (n - 1)
            w = weights[k] * dx
            if w < 1e-15:
                continue
            if d == 0:
                grad_h += w * (1 - 2*h)
            elif d > 0:
                grad_h[:n-d] += w * (1 - h[d:])
                grad_h[d:] += w * (-h[:n-d])
            else:
                dd = -d
                grad_h[dd:] += w * (1 - h[:n-dd])
                grad_h[:n-dd] += w * (-h[dd:])
        
        # Step size (using Lipschitz estimate or fixed)
        # The gradient of the smoothed objective is bounded by roughly 2*dx*sum(weights) = 2*dx
        # But let's use a simple step size
        lr = 0.1 / tau  # smaller step for larger tau
        
        h = h - lr * grad_h
        h = project_to_feasible(h, target_sum)
    
    return h

    return mc + (1.0 / _G_tau) * (msh + np.log(np.sum(np.exp(sh - msh))))

def _sgrad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    sh = _G_tau * dx * (corr - mc)
    msh = np.max(sh)
    ec = np.exp(sh - msh)
    ws = ec.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = ec / ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        if d == 0:
            g += wk * (1.0 - 2.0 * x)
        elif d > 0:
            g[:n-d] += wk * (1.0 - x[d:])
            g[d:] += wk * (-x[:n-d])
        else:
            dd = -d
            g[dd:] += wk * (1.0 - x[:n-dd])
            g[:n-dd] += wk * (-x[dd:])
    return g

def _cfun(x):
    return x.sum() - _G_target

def _cjad(x):
    return np.ones(len(x))

def slsqp_step(h0, tau, maxiter=300):
    global _G_tau, _G_target
    _G_tau = tau
    _G_target = len(h0) / 2.0
    bounds = [(0.0, 1.0)] * len(h0)
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjad}]
    res = minimize(_sobj, h0, method='SLSQP', jac=_sgrad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def local_refine(h, max_rounds=30):
    n = len(h)
    h = h.copy()
    best_s = c5_score(h)
    epsilons = [0.02, 0.05, 0.1, 0.2]
    
    for round_idx in range(max_rounds):
        improved = False
        order = np.random.permutation(n)
        partners = np.random.permutation(n)
        for i in order:
            for j in partners[:5]:
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                for eps in epsilons:
                    # +eps to i, -eps to j
                    if hi + eps <= 1.0 and hj - eps >= 0.0:
                        h[i], h[j] = hi + eps, hj - eps
                        s = c5_score(h)
                        if s < best_s - 1e-10:
                            best_s = s
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                    # -eps to i, +eps to j
                    if hi - eps >= 0.0 and hj + eps <= 1.0:
                        h[i], h[j] = hi - eps, hj + eps
                        s = c5_score(h)
                        if s < best_s - 1e-10:
                            best_s = s
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best_s
