```python
def find_optimal_alpha(corr, m, q, a_lo, a_hi):
    """Find alpha that minimizes max(corr + m*alpha + q*alpha^2) over [a_lo, a_hi]"""
    # It's convex, so ternary search works
    a = a_lo
    b = a_hi
    for _ in range(50):
        m1 = a + (b-a)/3.0
        m2 = b - (b-a)/3.0
        f1 = np.max(corr + m*m1 + q*m1*m1)
        f2 = np.max(corr + m*m2 + q*m2*m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    a_opt = (a+b)/2.0
    
    # Check endpoints
    best_val = np.max(corr + m*a_opt + q*a_opt*a_opt)
    best_a = a_opt
    for ca in [a_lo, a_hi]:
        val = np.max(corr + m*ca + q*ca*ca)
        if val < best_val:
            best_val = val
            best_a = ca
```
