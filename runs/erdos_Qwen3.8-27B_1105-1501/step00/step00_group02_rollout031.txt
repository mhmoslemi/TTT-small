import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_binary(h, target_sum):
    """Convert h to a binary sequence with exactly target_sum ones."""
    n = len(h)
    # Sort indices by h value
    idx = np.argsort(h)
    # The top target_sum indices become 1
    h_bin = np.zeros(n)
    h_bin[idx[-int(target_sum):]] = 1.0
    return h_bin

def local_search(h, max_time=60, seed=42):
    """Local search with simulated annealing for binary sequence."""
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    h = make_binary(h, target_sum).astype(float)
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    
    # Precompute ones and zeros
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    
    start = time.time()
    no_improve_count = 0
    
    while time.time() - start < max_time:
        if not ones_set or not zeros_set:
            break
        
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        
        # Perform swap
        h[i] = 0.0
        h[j] = 1.0
        ones_set.remove(i)
        zeros_set.remove(j)
        ones_set.add(j)
        zeros_set.add(i)
        
        new_c5 = compute_c5(h, dx)
        
        if new_c5 <= curr_c5 + 1e-15:
            curr_c5 = new_c5
            no_improve_count = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject: undo the swap
            h[i] = 1.0
            h[j] = 0.0
            ones_set.add(i)
            zeros_set.add(j)
            ones_set.remove(j)
            zeros_set.remove(i)
            no_improve_count += 1
    
    if best_c5 < curr_c5:
        h = best_h
    return h, best_c5

def make_valid(h, n):
    """Make h valid: h in [0,1], sum(h) = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:

            mask = h > 0.0

                # Perturb
                for _ in range(n // 4):
                    p = rng0.randint(n)
                    if h0[p] > 0.5:
                        h0[p] = 0.0
                    else:
                        h0[p] = 1.0
                # Fix sum
                h0 = to_binary(h0, n)
                # Now make it random again but keep fraction
                h0 = np.where(h0 > 0.5, 1.0, 0.0)
                # Add random noise
                for _ in range(n // 3):
                    p = rng0.randint(n)
                    h0[p] = 1.0 - h0[p]
                h0 = to_binary(h0, n)

            h, c5 = local_search(h0, t_per_candidate, seed * 7 + n * 13 + ci)

            # Polish for small n
            if n <= 80 and c5 < best_c5:
                h, c5 = polish_small(h, max_rounds=5)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

    # If we have time, do extra polish on the best solution
    elapsed = time.time() - start_time
    if best_h is not None and elapsed < budget_s * 0.92 and best_n <= 100:
        h_polished, c5_polished = polish_small(best_h.copy(), max_rounds=3)
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_h = h_polished

    if best_h is None:
        n = 48
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h, 2.0 / n)
        best_n = n

    return best_h, best_c5, best_n
