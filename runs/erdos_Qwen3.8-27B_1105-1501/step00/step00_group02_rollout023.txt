    """Compute change in correlation when swapping h[i] (1→0) and h[j] (0→1)."""
    # h[i] = 1, h[j] = 0
    # Returns array of length 2n-1 with the changes
    delta = np.zeros(2*n-1)
    
    # For each lag k from -(n-1) to n-1 (index k+n-1 in the array):
    for k in range(-(n-1), n):
        idx = k + n - 1

            # After: h'[i+k]*(1-h'[i]) = h'[i+k]*1
            # h'[i+k] = 1 if i+k==j, else h[i+k] (if i+k != i, i.e., k != 0)

            else:

            # h'[j-k] = 0 if j-k==i, else h[j-k] (if j-k != j, i.e., k != 0)
            if j - k == i:
                h_val = 0
            else:
                h_val = h[j-k]
            delta[idx] += (1 - h_val) - 0
    
    return delta

    """Compute C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_initial_binary(n, seed=0):
    """Create a random binary vector with exactly n/2 ones."""
    rng = np.random.RandomState(seed)

    """For odd n, use smooth optimization."""
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def objective(x):
        # x is the parameter vector
        h = np.clip(x, 0, 1)
        # Project to constraint sum(h) = target_sum
        # Soft projection: adjust to satisfy constraint
        s = np.sum(h)
        if s > 0 and s < n:
            h = (h - s/n) * (target_sum / max(n - s, 1e-10)) + target_sum/n
            h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1.0-h, mode='full')
        beta = 50.0
        scaled = beta * corr
        max_s = np.max(scaled)
        return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / beta * dx
    
    # Use SLSQP with bounds and constraint
    bounds = [(0, 1)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    
    start = time.time()

                      constraints=constraints, options={'maxiter': 200, 'ftol': 1e-10})

            h_opt = np.clip(h_opt, 0, 1)
