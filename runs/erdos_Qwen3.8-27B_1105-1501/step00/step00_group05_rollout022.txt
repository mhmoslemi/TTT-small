```python
import time
import sys

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def get_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1 - h, mode="full")

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute to indices not at boundary
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_component(h, offset):
    """Gradient of dx * sum_n h[n+offset]*(1-h[n]) with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    if offset >= 0:
        # Valid n: 0 <= n < n-offset (so that n+offset < n)
        # h[n+offset] * (1-h[n])
        # d/dh[i]:
        #   As first factor: i = n+offset, n = i-offset, valid when 0 <= i-offset and i < n
        #     => i from offset to n-1, contribution: (1 - h[i-offset])
        #   As second factor: i = n, valid when 0 <= i < n-offset
        #     => i from 0 to n-1-offset, contribution: -h[i+offset]
        
        if offset == 0:
            # h[n]*(1-h[n]) for all n
            # d/dh[i] = 1 - 2*h[i]
            grad[:] = 1.0 - 2.0 * h
        else:

            idx1 = np.arange(offset, n)
            grad[idx1] += (1.0 - h[idx1 - offset])
            # Second factor contributions
            idx2 = np.arange(0, n - offset)
            grad[idx2] += -h[idx2 + offset]

            idx2 = np.arange(o, n)
            grad[idx2] += -h[idx2 - o]
```

```python
        A[m] = dx * coeff

        # F_m = dx * sum_{nn valid} h[nn+offset] * (1-h[nn])
        # With h[i] = 0:
        #   If h[i] was a first factor (nn+offset = i), that term becomes 0
        #   If h[i] was a second factor (nn = i), that term becomes h[i+offset] * 1 = h[i+offset]
        
        if offset > 0:
            valid_range = range(0, n - offset)
        else:
            valid_range = range(-offset, n)
        
        b_val = 0.0
        for nn in valid_range:
            first = nn + offset
            second = nn
            # h[first] with h[i]=0 if first==i, else h[first]
            hf = 0.0 if first == i else h[first]
            # (1-h[second]) with h[i]=0 if second==i, else (1-h[second])
            h_s = 1.0 if second == i else (1.0 - h[second])
            b_val += hf * h_s
        B[m] = dx * b_val
    
    return A, B, dx  # dx needed for the quadratic term
```

Then the 1D minimization is:
minimize max( B[center] - dx*x^2, max_{m≠center}(A[m]*x + B[m]) ) over x ∈ [0,1]

This is the max of a concave quadratic and several affine functions. The minimum of this max can be found by:
1. Checking x=0 and x=1
2. Finding where the parabola intersects each line: B_c - dx*x^2 = A[m]*x + B[m] => dx*x^2 + A[m]*x + (B[m]-B_c) = 0
3. Finding where each pair of lines intersect: A[p]*x + B[p] = A[q]*x + B[q] => x = (B[q]-B[p])/(A[p]-A[q])
4. Evaluating the objective at all candidate points

The number of candidates is O(n) from parabola-line, O(n^2) from line-line. Total O(n^2) candidates, each evaluated in O(n) time. So O(n^3) per coordinate, O(n^4) per sweep. For n=42: 42^4 ≈ 3.1M operations. In Python with NumPy vectorization for the evaluation, this should be a few seconds per sweep.

Actually, I can vectorize the evaluation: for a candidate x, compute all F_m(x) in one go.

```python
def coord_descent_efficient(h, n, max_sweeps=200, time_limit=100, start_time=0):
    dx = 2.0 / n
    center = n - 1
    num_lags = 2 * n - 1
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        improved = False
        
        for i in range(n):
            # Compute coefficients
            A = np.zeros(num_lags)
            B = np.zeros(num_lags)
            
            for m in range(num_lags):
                offset = m - center
                if offset == 0:
                    S_others = np.sum(h**2) - h[i]**2
                    B[m] = dx * (n/2 - S_others)
                    continue
                
                coeff = 0.0
                if offset > 0:
                    if 0 <= i - offset and i - offset <= n - 1 - offset:
                        coeff += (1.0 - h[i - offset])
                    if 0 <= i and i <= n - 1 - offset:
                        coeff += -h[i + offset]
                else:
                    p = -offset
                    if p <= i + p and i + p <= n - 1:
                        coeff += (1.0 - h[i + p])
                    if p <= i and i <= n - 1:
                        coeff += -h[i - p]
                A[m] = dx * coeff
                
                # B[m]: F_m with h[i]=0
                if offset > 0:
                    for nn in range(n - offset):
                        first = nn + offset
                        second = nn
                        hf = 0.0 if first == i else h[first]
                        h_s = 1.0 if second == i else (1.0 - h[second])
                        B[m] += hf * h_s
                else:
                    for nn in range(-offset, n):
                        first = nn + offset
                        second = nn
                        hf = 0.0 if first == i else h[first]
                        h_s = 1.0 if second == i else (1.0 - h[second])
                        B[m] += hf * h_s
                B[m] *= dx
            
            # Now minimize max(B[center] - dx*x^2, max_{m≠center}(A[m]*x + B[m]))
            
            # Generate candidates
            candidates = [0.0, 1.0, h[i]]
            
            # Parabola-line intersections
            for m in range(num_lags):
                if m == center:
                    continue
                # dx*x^2 + A[m]*x + (B[m] - B[center]) = 0
                a_q = dx
                b_q = A[m]
                c_q = B[m] - B[center]
                disc = b_q**2 - 4*a_q*c_q
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    x1 = (-b_q + sqrt_disc) / (2*a_q)
                    x2 = (-b_q - sqrt_disc) / (2*a_q)
                    if 0 <= x1 <= 1:
                        candidates.append(x1)
                    if 0 <= x2 <= 1:
                        candidates.append(x2)
            
            # Line-line intersections
            non_center = [m for m in range(num_lags) if m != center]
            for p_idx in range(len(non_center)):
                for q_idx in range(p_idx+1, len(non_center)):
                    p = non_center[p_idx]
                    q = non_center[q_idx]
                    denom = A[p] - A[q]
                    if abs(denom) > 1e-14:
                        x = (B[q] - B[p]) / denom
                        if 0 <= x <= 1:
                            candidates.append(x)
            
            # Evaluate at all candidates
            def obj_at(x):
                val_center = B[center] - dx * x * x
                val_lines = np.max(A * x + B)
                return max(val_center, val_lines)
            
            best_x = h[i]
            best_val = obj_at(h[i])
            for x in candidates:
                x = min(max(x, 0.0), 1.0)
                val = obj_at(x)
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = x
            
            if abs(best_x - h[i]) > 1e-10:
                h[i] = best_x
                improved = True
        
        if not improved:
            break
    
    return h
```
