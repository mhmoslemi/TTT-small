def project_box_mass(v, n):
    """Project v onto {x : 0 <= x_i <= 1, sum(x) = n/2}"""
    target = n / 2.0
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0, 1)
    s = np.sum(v)
    if abs(s - target) < 1e-12:
        return v
    
    # Bisection for the shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(v + mid, 0, 1))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(v + mu, 0, 1)
    
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute over interior points
        interior = (result > 1e-8) & (result < 1 - 1e-8)
        if np.sum(interior) > 0:
            n_int = np.sum(interior)
            adj = residual / n_int
            result[interior] = np.clip(result[interior] + adj, 0, 1)
    
    return result


def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def pair_transfer_search(h, n, i, j, n_trials=20):
    """Try transferring mass between coordinates i and j."""
    hi, hj = h[i], h[j]
    # delta: h[i] += delta, h[j] -= delta
    lo = max(-hi, hj - 1.0)
    hi_bound = min(1.0 - hi, hj)
    
    if lo >= hi_bound:
        return h, evaluate_c5(h, n)
    
    best_h = h.copy()
    best_val = evaluate_c5(h, n)
    
    # Sample points in [lo, hi_bound]
    for trial in range(n_trials):
        t = (trial + 1) / (n_trials + 1)
        delta = lo + t * (hi_bound - lo)
        h_trial = h.copy()
        h_trial[i] += delta
        h_trial[j] -= delta
        val = evaluate_c5(h_trial, n)
        if val < best_val:
            best_val = val
            best_h = h_trial.copy()
    
    return best_h, best_val

```python
def optimal_pair_transfer(r, c, delta_min, delta_max, iterations=60):
    """Find δ in [delta_min, delta_max] that minimizes max(r + δ*c)"""
    if delta_min >= delta_max:
        return 0.0
    a, b = delta_min, delta_max
    for _ in range(iterations):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(r + m1 * c)
        f2 = np.max(r + m2 * c)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0
```

This is very clean and efficient. Each iteration is O(n) (computing max of a vector of length 2n-1), and we do 60 iterations. For n=70, that's 60 * 139 = 8340 operations per pair. With 2450 pairs, total is about 20 million operations. Very fast.
