    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def proj_simplex(y, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

def overlap_gradient(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[:n-s] += dx * (1.0 - h[s:])
        if s > 0:
            g[s:] -= dx * h[:n-s]
    else:
        ss = -s
        if ss < n:
            g[ss:] += dx * (1.0 - h[:n-ss])
        if n - ss > 0:
            g[:n-ss] -= dx * h[ss:]
    return g

def find_descent_direction(h, active_idx, n):
    """LP: find direction d that decreases all active overlaps.
    min t s.t. grad_s · d <= t, sum(d)=0, -h<=d<=1-h
    If optimal t < 0, d is a descent direction.
    """
    n_act = len(active_idx)
    A_rows = []
    b_vals = []
    
    for idx in active_idx:
        s = int(idx) - (n - 1)
        g = overlap_gradient(h, s)
        # g · d - t <= 0
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(0.0)
    
    # sum(d) = 0 → -sum(d) <= 0 and sum(d) <= 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -h_j <= d_j <= 1-h_j, t is free (but bounded for solver)
    bounds = [(max(-h[j], -1.0), min(1.0 - h[j], 1.0)) for j in range(n)]
    bounds.append((-10.0, 10.0))  # t bound
    
    A = np.array(A_rows)
    b = np.array(b_vals)

        res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),
                      A_ub=A, b_ub=b,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs')
        if res.success and res.x[n] < -1e-8:
            d = res.x[:n]
            # Normalize
            d_norm = np.max(np.abs(d))
            if d_norm > 1e-10:
                d /= d_norm
            return d
    except:

    return None

def resample(h, new_n):
    """Resample h from its current length to new_n using linear interpolation"""
    old_n = len(h)
    # Centroid positions

import numpy as np

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def proj_simplex(y, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)


def overlap_gradient(h, s, dx):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[:n - s] += dx * (1.0 - h[s:])
        if s > 0:
            g[s:] -= dx * h[:n - s]
    else:
        ss = -s
        if ss < n:
            g[ss:] += dx * (1.0 - h[:n - ss])
        if n - ss > 0:
            g[:n - ss] -= dx * h[ss:]
    return g


def lp_descent_dir(h, active, n, dx):
    na = len(active)
    A_rows = []
    for idx in active:
        s = int(idx) - (n - 1)
        g = overlap_gradient(h, s, dx)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
    A = np.array(A_rows)
    b = np.zeros(na)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = []

        bounds.append((-min(h[j], 1.0), min(1.0 - h[j], 1.0)))

            norm = np.max(np.abs(d))
