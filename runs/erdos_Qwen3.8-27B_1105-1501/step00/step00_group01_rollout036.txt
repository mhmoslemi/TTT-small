import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def compute_all_lags(h):
    """Compute all lag values c_k * dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_to_feasible(h, n):
    """Project h onto {x: 0 ≤ x_i ≤ 1, sum(x) = n/2} using water-filling"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Water-filling to achieve exact sum
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum: add to elements that are < 1
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room < 1e-12:
            return h
        h = h + diff * room / total_room
    else:
        # Need to decrease sum: subtract from elements that are > 0
        room = h
        total_room = np.sum(room)
        if total_room < 1e-12:
            return h
        h = h + diff * room / total_room
    h = np.clip(h, 0, 1)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        interior = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0, 1)
    return h

def gradient_c5(h, n):
    """Compute gradient of c5 = max_k c_k(h) w.r.t. h.
    Returns gradient of the active (maximizing) constraint."""
    dx = 2.0 / n
    lags = np.correlate(h, 1 - h, mode='full') * dx
    k_star = np.argmax(lags)
    k = k_star - (n - 1)  # Convert to lag index
    
    grad = np.zeros(n)
    # c_k = dx * sum_{i in S_k} h[i+k] * (1-h[i])
    # ∂c_k/∂h[m] = dx * [(1-h[m-k]) * 1_{k≤m<n+k} - h[m+k] * 1_{-k≤m<n-k}]
    
    m = np.arange(n)
    # First term: (1-h[m-k]) where k ≤ m < n+k
    mask1 = (m >= k) & (m < n + k)
    idx1 = m[mask1] - k
    grad[mask1] += (1 - h[idx1])
    
    # Second term: -h[m+k] where -k ≤ m < n-k
    mask2 = (m >= -k) & (m < n - k)
    idx2 = m[mask2] + k
    grad[mask2] -= h[idx2]
    
    grad *= dx
    return grad

def slack_method(h, n, t, alpha=1.0):
    """Compute gradient of smooth max approximation: (1/alpha)log(sum exp(alpha*c_k))"""

    lags = np.correlate(h, 1 - h, mode='full') * dx
    # Soft max
    max_lag = np.max(lags)
    exp_lags = np.exp(alpha * (lags - max_lag))
    weights = exp_lags / np.sum(exp_lags)
    
    # Gradient of sum_k weights[k] * c_k
    k_vals = np.arange(-(n-1), n)
    grad = np.zeros(n)

        mask2 = (m >= -k) & (m < n - k)
        idx2 = m[mask2] + k

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def compute_correlation(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_of_correlation(h, j):
    """Gradient of corr[j] * dx with respect to h.
    j is the index in the full correlation array (0 to 2n-2)."""
    n = len(h)
    dx = 2.0 / n
    lag = j - (n - 1)
    m = np.arange(n)
    grad = np.zeros(n)
    
    # First term: (1-h[m-lag]) where lag ≤ m < n+lag
    mask1 = (m >= lag) & (m < n + lag)
    if np.any(mask1):
        idx = m[mask1] - lag
        grad[mask1] += (1.0 - h[idx])
    
    # Second term: -h[m+lag] where -lag ≤ m < n-lag
    mask2 = (m >= -lag) & (m < n - lag)
    if np.any(mask2):
        idx = m[mask2] + lag
        grad[mask2] -= h[idx]
    
    return grad * dx

def smooth_max_gradient(h, alpha=50.0):
    """Gradient of softmax approximation to max of correlation values."""
    n = len(h)
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros(n)
    for j in range(len(corr)):
        if weights[j] > 1e-12:
            grad += weights[j] * gradient_of_correlation(h, j)
    return grad, max_corr

def project_feasible(h, n):

    target = n / 2.0
    h = np.array(h, dtype=float)

        return np.sum(np.clip(h + shift, 0.0, 1.0))
    
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    
    # Binary search for the shift
    lo, hi = -1.0, 1.0
    if s > target:
        lo, hi = -1.0, 0.0
    else:
        lo, hi = 0.0, 1.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = sum_after_clip(mid)
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    
    # Fine-tune

        if np.any(interior):
            h_new[interior] += diff / np.sum(interior)
            h_new = np.clip(h_new, 0.0, 1.0)

    mk2 = (m >= -lag) & (m < n - lag)
    if mk2.any():
        g[mk2] -= h[m[mk2] + lag]
    return g * dx

def smooth_obj_grad(h, alpha):
    """Smooth max of corr values and its gradient."""
    n = len(h)
    c = compute_corr(h)
    mc = c.max()
    e = np.exp(alpha * (c - mc))
    w = e / e.sum()
    val = mc + np.log(e.sum()) / alpha
    g = np.zeros(n)
    for j in range(len(c)):
        if w[j] > 1e-14:
            g += w[j] * corr_gradient(h, j, n)
    return val, g

def projected_gradient(h, n, alpha, steps, time_limit, lr_init=0.1):
    """Projected gradient descent on smooth max."""
    h = project(h, n).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t0 = time.time()
    for i in range(steps):
        if time.time() - t0 > time_limit:
            break
        val, g = smooth_obj_grad(h, alpha)
        h_new = project(h - lr * g, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
        h = h_new
        if c5 > best_c5 * 1.005:
            lr *= 0.95
        elif i % 100 == 0:
            lr *= 1.01
    return best, best_c5

def slsqp_opt(h, n, time_limit=100):
    """SLSQP optimization with auxiliary t variable."""
    nc = 2*n - 1
    def obj(x):
        return x[n]
    def obj_jac(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    def ineq(x):
        return x[n] - compute_corr(x[:n])
    def ineq_jac(x):
        J = np.zeros((nc, n+1))
        J[:, n] = 1.0
        h = x[:n]
        for j in range(nc):
            J[j, :n] = -corr_gradient(h, j, n)
        return J
    def eq(x):
        return x[:n].sum() - n/2.0
    def eq_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    bounds = [(0,1)]*n + [(0.0, None)]
    x0 = np.append(h, compute_c5(h)+0.01)
    cons = [
        {'type':'ineq','fun':ineq,'jac':ineq_jac},
        {'type':'eq','fun':eq,'jac':eq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter':200,'ftol':1e-12})
        if res.success or res.x is not None:
            return project(res.x[:n], n)
    except:
        pass
    return None

def binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def local_search_binary(h, n, max_iters=20000, time_limit=60):
    """Simulated annealing on binary h (even n only)."""
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    # Ensure exactly target ones
    s = h.sum()
    if s > target:
        ones = np.where(h==1)[0]
        rng = np.random.RandomState(42)
        rng.shuffle(ones)
        h[ones[:int(s-target)]] = 0
    elif s < target:
        zeros = np.where(h==0)[0]
        rng = np.random.RandomState(42)
        rng.shuffle(zeros)
        h[zeros[:int(target-s)]] = 1
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = 0.03
    T_min = 0.003
    t0 = time.time()
    rng = np.random.RandomState(123)
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        # Pick a 1 and a 0 to swap
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0; h[j] = 1
        c5 = compute_c5(h)
        d = c5 - best_c5
        if d < 0 or rng.random() < np.exp(-d/max(T,1e-10)):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
        T = max(T * 0.9995, T_min)
    return best_h, best_c5
