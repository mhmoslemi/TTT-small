                if step < 0.0005:
                    break
                stalled = 0
    return best_h, best_c

def simulated_anneal(h, n, rng, t0, deadline):
    best_c = score_exact(h)
    best_h = h.copy()
    cur = h.copy()
    T = 0.008
    T_min = 1e-7
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(-0.04, 0.04)
        ai, aj = cur[i], cur[j]
        cur[i] += d
        cur[j] -= d
        if cur[i] < 0 or cur[i] > 1 or cur[j] < 0 or cur[j] > 1:
            cur[i] = ai
            cur[j] = aj
            continue
        c = score_exact(cur)
        if c < best_c:
            best_c = c
            best_h = cur.copy()
        elif rng.random() < np.exp(-(c - best_c) / T):
            pass
        else:
            cur[i] = ai
            cur[j] = aj
        T = max(T_min, T * 0.99995)
    return best_h, best_c

def binary_search_nearby(h, n, rng, t0, deadline):
    """Try binary-like solutions near the current one."""
    best_c = score_exact(h)
    best_h = h.copy()
    # Binary projection
    idx = np.argsort(h)
    if n % 2 == 0:
        half = n // 2
        b = np.zeros(n)
        b[idx[half:]] = 1.0
    else:
        b = np.zeros(n)
        b[idx[(n + 1) // 2:]] = 1.0
        b[idx[(n - 1) // 2]] = 0.5
    c_b = score_exact(b)
    if c_b < best_c:
        best_c = c_b
        best_h = b.copy()
    # Now do local search starting from binary solution
    cur = best_h.copy()
    step = 0.02
    stalled = 0
    while time.time() < deadline and stalled < 50000:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(-step, step)
        ai, aj = cur[i], cur[j]
        cur[i] += d
        cur[j] -= d
        if cur[i] < 0 or cur[i] > 1 or cur[j] < 0 or cur[j] > 1:
            cur[i] = ai
            cur[j] = aj
            continue
        c = score_exact(cur)
        if c < best_c - 1e-15:
            best_c = c
            best_h = cur.copy()
            stalled = 0
        else:
            cur[i] = ai
            cur[j] = aj
            stalled += 1
            if stalled > 15000:
                step *= 0.6
                if step < 0.0003:
                    break
                stalled = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, UnboundLocalError):
        h0 = np.full(95, 0.5)

    best_h = h0.copy()
    best_c = score_exact(best_h)
    best_n = len(best_h)
    n = best_n

    # Phase 1: Hill climbing with multiple step sizes (40% budget)
    deadline1 = t0 + budget_s * 0.40
    steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    h_ph1, c_ph1 = hill_climb(best_h, n, rng, t0, deadline1, steps)
    if c_ph1 < best_c:
        best_c = c_ph1
        best_h = h_ph1.copy()

    # Phase 2: Directed search targeting worst lag (20% budget)
    deadline2 = t0 + budget_s * 0.60
    h_ph2, c_ph2 = directed_search(best_h, n, rng, t0, deadline2)
    if c_ph2 < best_c:
        best_c = c_ph2
        best_h = h_ph2.copy()

    # Phase 3: Simulated annealing (20% budget)
    deadline3 = t0 + budget_s * 0.80
    h_ph3, c_ph3 = simulated_anneal(best_h, n, rng, t0, deadline3)
    if c_ph3 < best_c:
        best_c = c_ph3
        best_h = h_ph3.copy()

    # Phase 4: Binary-like search (10% budget)
    deadline4 = t0 + budget_s * 0.90
    h_ph4, c_ph4 = binary_search_nearby(best_h, n, rng, t0, deadline4)
    if c_ph4 < best_c:
        best_c = c_ph4
        best_h = h_ph4.copy()

    # Phase 5: Try nearby n values (10% budget)
    deadline5 = t0 + budget_s * 0.97
    candidates_n = []
    for dn in [-2, -1, 1, 2, 3, 4]:
        nv = best_n + dn
        if 16 <= nv <= 200 and nv not in candidates_n:
            candidates_n.append(nv)
    # Also try round numbers
    for nv in [94, 96, 98, 100, 92]:
        if 16 <= nv <= 200 and nv not in candidates_n:
            candidates_n.append(nv)
    candidates_n.sort()

    for nv in candidates_n:
        if time.time() > deadline5:
            break
        h_try = resample_to(best_h, nv)
        c_try = score_exact(h_try)
        if c_try < best_c:
            # Worth optimizing further
            h_t, c_t = hill_climb(h_try, nv, rng, t0, deadline5, [0.03, 0.01, 0.003])
            if c_t < best_c:
                best_c = c_t
                best_h = h_t.copy()
                best_n = nv
        else:
            # Just check quickly
            if c_try < best_c + 0.001:
                h_t, c_t = hill_climb(h_try, nv, rng, t0, deadline5, [0.01])
                if c_t < best_c:
                    best_c = c_t
                    best_h = h_t.copy()
                    best_n = nv

    # Final fine-tuning with smallest step
    if time.time() < budget_s * 0.99:
        final_deadline = t0 + budget_s * 0.99
        h_f, c_f = hill_climb(best_h, best_n, rng, t0, final_deadline, [0.0005, 0.0002])
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    # Final validation and repair
    best_h = repair(best_h, best_n)
    final_c = score_exact(best_h)

    return (best_h, final_c, best_n)
