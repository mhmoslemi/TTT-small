def _resample(h, new_n):

    if new_n == old_n:

    new_x = (np.arange(new_n) + 0.5) / new_n * 2.0

def _golden_section_min(f, a, b, max_iter=50):
    """Minimize f on [a,b] using golden section. f is assumed unimodal (convex)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)

        if fc < fd:

            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)

            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)

import numpy as np
import time


def _score(h):
    n = len(h)
    if n < 2:
        return 1e10
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def _project(h, target_sum):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        for _ in range(500):
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            t = diff / cnt
            h[mask] = np.minimum(h[mask] + t, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) < 1e-14:
                break
    else:
        for _ in range(500):
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            t = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - t, 0.0)
            diff = target_sum - h.sum()
            if abs(diff) < 1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += residual / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def _resample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = (np.arange(old_n) + 0.5) / old_n * 2.0
    new_x = (np.arange(new_n) + 0.5) / new_n * 2.0
    new_h = np.interp(new_x, old_x, h)
    return _project(new_h, new_n / 2.0)


def _eval_pair(h, p, q, a, b, d):
    work = h.copy()
    work[p] = a + d
    work[q] = b - d
    work = np.clip(work, 0.0, 1.0)
    return _score(work)


def _pair_sweep(h, deadline, max_pairs=None, seed=42):
    n = len(h)
    current_score = _score(h)
    rng = np.random.default_rng(seed)

    all_pairs = [(p, q) for p in range(n) for q in range(n) if p != q]
    if max_pairs is not None and max_pairs < len(all_pairs):
        idxs = rng.choice(len(all_pairs), size=max_pairs, replace=False)
        pairs = [all_pairs[i] for i in idxs]
    else:
        pairs = all_pairs

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for p, q in pairs:
        if time.perf_counter() > deadline:
            break
        a = h[p]
        b = h[q]
        lo = max(-a, b - 1.0)
        hi = min(1.0 - a, b)
        if hi - lo < 1e-12:
            continue

        # Coarse grid: 17 points
        n_grid = 17
        deltas = np.linspace(lo, hi, n_grid)
        vals = np.empty(n_grid)
        for i, d in enumerate(deltas):
            vals[i] = _eval_pair(h, p, q, a, b, d)

        best_idx = int(np.argmin(vals))
        best_val = vals[best_idx]

        if best_val >= current_score - 1e-13:
            continue

        # Golden section refinement
        if 0 < best_idx < n_grid - 1:
            a_g = deltas[best_idx - 1]
            b_g = deltas[best_idx + 1]
        else:
            a_g = lo
            b_g = hi

        c_g = b_g - gr * (b_g - a_g)
        d_g = a_g + gr * (b_g - a_g)
        fc = _eval_pair(h, p, q, a, b, c_g)
        fd = _eval_pair(h, p, q, a, b, d_g)

        for _ in range(35):
            if fc < fd:
                b_g = d_g
                d_g = c_g
                fd = fc
                c_g = b_g - gr * (b_g - a_g)
                fc = _eval_pair(h, p, q, a, b, c_g)
            else:
                a_g = c_g
                c_g = d_g
                fc = fd
                d_g = a_g + gr * (b_g - a_g)
                fd = _eval_pair(h, p, q, a, b, d_g)

        opt_d = (a_g + b_g) / 2.0
        opt_val = _eval_pair(h, p, q, a, b, opt_d)

        if opt_val < best_val:
            best_val = opt_val
            best_d = opt_d
        else:
            best_d = deltas[best_idx]

            h[p] = a + best_d
            h[q] = b - best_d
            h = np.clip(h, 0.0, 1.0)
            current_score = best_val

    h = 1.0 / (1.0 + np.exp(4.0 * t))
    h = _project(h, n / 2.0)
    return h
