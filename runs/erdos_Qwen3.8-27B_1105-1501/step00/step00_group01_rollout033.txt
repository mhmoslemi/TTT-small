def c5_objective(h, n, t):
    """Log-sum-exp approximation of C5"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    # Log-sum-exp
    max_val = np.max(vals)
    if t <= 0:
        return max_val
    return max_val + t * np.log(np.sum(np.exp((vals - max_val) / t)))

def c5_gradient(h, n, t):
    """Gradient of log-sum-exp C5 approximation"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    max_val = np.max(vals)
    if t <= 0:
        # Subgradient: gradient of the active (max) correlation
        active = np.where(vals == max_val)[0]
        grad = np.zeros(n)
        for k in active:
            # d/dh[i] of corr[k] * dx
            # corr[k] = sum_j h[j+d] * (1-h[j]) where d = k - (n-1)
            d = k - (n - 1)
            # For the valid range of j:
            # 0 <= j < n and 0 <= j+d < n
            j_min = max(0, -d)
            j_max = min(n, n - d)
            for j in range(j_min, j_max):
                # term: h[j+d] * (1-h[j])
                # d/dh[j] = -h[j+d]
                # d/dh[j+d] = (1-h[j])
                grad[j] += -h[j+d] * dx
                grad[j+d] += (1-h[j]) * dx
            grad /= len(active)
        return grad
    # Proper gradient
    exp_vals = np.exp((vals - max_val) / t)
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        d = k - (n - 1)
        j_min = max(0, -d)
        j_max = min(n, n - d)
        for j in range(j_min, j_max):
            grad[j] += weights[k] * (-h[j+d]) * dx
            grad[j+d] += weights[k] * (1-h[j]) * dx
    return grad

import numpy as np
from scipy.optimize import minimize
import math
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_to_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt  # diff is negative, so this adds
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-14:
        mask = (h > 0) & (h < 1)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_logsumexp(h, n, t):
    """Compute log-sum-exp approximation of C5 and its gradient"""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    vals = corr * dx
    max_val = np.max(vals)
    
    if t < 1e-10:
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        obj = max_val
    else:
        shifted = (vals - max_val) / t
        m = np.max(shifted)  # should be 0
        exp_shifted = np.exp(shifted - m)
        log_sum = m + np.log(np.sum(exp_shifted))
        obj = max_val + t * log_sum
        p = exp_shifted / np.sum(exp_shifted)
    
    # Gradient computation
    if t < 1e-10:
        weights = np.zeros(2 * n - 1)
        weights[active] = 1.0 / len(active)
    else:
        weights = p
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        j_min = max(0, -d)
        j_max = min(n, n - d)
        if j_min >= j_max:
            continue
        j = np.arange(j_min, j_max)
        # c_k = dx * sum h[j+d] * (1-h[j])
        # d/dh[j] c_k = -dx * h[j+d]
        # d/dh[j+d] c_k = dx * (1-h[j])
        grad[j] += wk * dx * (-h[j + d])
        idx = j + d
        np.add.at(grad, idx, wk * dx * (1.0 - h[j]))
    
    return obj, grad

def run_slsqp(h0, n, t, maxiter=500):
    """Run SLSQP optimization"""
    x0 = h0.copy()
    
    def obj(x):
        x_clipped = np.clip(x, 0, 1)
        val, _ = c5_logsumexp(x_clipped, n, t)
        return val
    
    def grad(x):
        x_clipped = np.clip(x, 0, 1)
        _, g = c5_logsumexp(x_clipped, n, t)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2, 
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    result = minimize(obj, x0, method='SLSQP', jac=grad, 
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
    
    h_opt = np.clip(result.x, 0, 1)
    h_opt = project_to_feasible(h_opt, n)
    return h_opt, compute_c5(h_opt)

        n_init = 63
        h_init = project_feasible(np.full(n_init, 0.5), n_init)
        try_update(h_init, compute_c5(h_init))

    # Phase 1: SLSQP refinement of initial
    h_cur = h_init.copy()
    n_cur = len(h_cur)
    t_list_p1 = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
    for t in t_list_p1:
        if elapsed() > budget_s * 0.25:
            break
        h_new, c5_new = do_slsqp(h_cur, n_cur, t, maxiter=400)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new.copy()

    # Phase 2: Gradient descent fine-tuning
    if elapsed() < budget_s * 0.4:
        h_gd, c5_gd = gradient_descent(best_h, best_n, 0.01, 0.003, 50000, budget_s * 0.15)
        try_update(h_gd, c5_gd)

    # Phase 3: Try different n values via interpolation
    for n_try in [60, 62, 64, 66, 68, 70, 72, 74]:
        if elapsed() > budget_s * 0.6:
            break
        if n_try % 2 != 0:
            continue
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_cur3 = h_interp.copy()
        for t in [0.5, 0.2, 0.1, 0.05, 0.02]:
            if elapsed() > budget_s * 0.6:
                break
            h_new, c5_new = do_slsqp(h_cur3, n_try, t, maxiter=200)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try
                h_cur3 = h_new.copy()

    # Phase 4: Binary sequences with local search
    for n_try in [62, 64, 66, 68, 70, 72]:
        if elapsed() > budget_s * 0.82:
            break
        for attempt in range(2):
            if elapsed() > budget_s * 0.82:
                break
            h_bin = np.zeros(n_try)
            indices = rng.choice(n_try, n_try // 2, replace=False)
            h_bin[indices] = 1.0
            h_sa, c5_sa = binary_local_search(h_bin, n_try, min(3.0, max(0.5, budget_s * 0.82 - elapsed())), rng)
            try_update(h_sa, c5_sa)

    # Phase 5: SLSQP from best (possibly binary) solution
    if elapsed() < budget_s * 0.92:
        h_cur5 = best_h.copy()
        n5 = len(h_cur5)
        for t in [0.3, 0.1, 0.05, 0.03, 0.02, 0.01, 0.008, 0.005, 0.003]:
            if elapsed() > budget_s * 0.92:
                break
            h_new, c5_new = do_slsqp(h_cur5, n5, t, maxiter=300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur5 = h_new.copy()

    # Phase 6: Final gradient descent
    if elapsed() < budget_s * 0.97:
        h_gd2, c5_gd2 = gradient_descent(best_h, best_n, 0.005, 0.001, 30000, budget_s * 0.97 - elapsed())
        try_update(h_gd2, c5_gd2)

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
