def project_feasible(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum} using water-filling."""
    lo, hi = -1.0, 1.0
    for _ in range(60):

        if s > target_sum:

        else:
            hi = mid

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(proj)
        else:
            order = np.argsort(-proj)
        for idx in order:
            if diff > 0:
                adj = min(diff, 1.0 - proj[idx])
                proj[idx] += adj
                diff -= adj
            else:
                adj = max(diff, -proj[idx])
                proj[idx] += adj
                diff -= adj
            if abs(diff) < 1e-14:
                break
    return proj

def subgradient_max_overlap(h, n, dx):
