def exact_objective(h):

def lag_gradient(h, lag_idx):
    N = len(h)
    k = lag_idx - (N - 1)
    grad = np.zeros(N)
    if k >= 0:
        for i in range(N - k):
            grad[i + k] += (1.0 - h[i])
            grad[i] += -h[i + k]
    else:
        j = -k
        for i in range(N - j):
            grad[i] += (1.0 - h[i + j])
            grad[i + j] += -h[i]
    return grad

def dyadic_repair(h, bits=20):

    S = 1 << bits
    if N % 2 == 1:
        S *= 2
    target = (N * S) // 2
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    diff = target - int(np.sum(z))
    if diff > 0:
        order = np.argsort(z)
        remaining = diff
        for idx in order:
            if remaining <= 0:
                break
            add = min(int(S - z[idx]), remaining)
            z[idx] += add
            remaining -= add
    elif diff < 0:
        order = np.argsort(-z)
        remaining = -diff
        for idx in order:
            if remaining <= 0:
                break
            sub = min(int(z[idx]), remaining)
            z[idx] -= sub
            remaining -= sub

    except (NameError, UnboundLocalError):
        parent_arr = make_synthetic_initial(42, seed)

    N_parent = len(parent_arr)
    parent_arr = np.clip(parent_arr, 0, 1)
    parent_arr = project_feasible(parent_arr, N_parent)

    best_h = parent_arr.copy()
    best_C = exact_objective(best_h)

    # Build dimension list: parent first, then nearby
    dims = [N_parent]
    for d in sorted(set(list(range(max(24, N_parent - 8), N_parent, 2)) +
                        list(range(N_parent + 2, min(N_parent + 14, 64), 2)))):
        if d not in dims and d >= 20 and d <= 64:
            dims.append(d)
    dims.sort(key=lambda d: (abs(d - N_parent), d))

    # Phase 1: optimize at parent dimension
    phase1_time = min(120, (deadline - time.time()) * 0.35)
    result_h, result_C = local_search(parent_arr, N_parent, time.time() + phase1_time, rng)
    if result_C < best_C:
        best_h = result_h
        best_C = result_C

    # Phase 2: try other dimensions
    for N in dims:
        if time.time() > deadline - 20:
            break
        if N == N_parent:
            continue

        h_start = resample_density(best_h, N)
        h_start = project_feasible(h_start, N)

        # Quick optimization
        dim_time = min(40, (deadline - time.time()) * 0.25)
        if dim_time < 5:
            break
        result_h, result_C = local_search(h_start, N, time.time() + dim_time, rng)
        if result_C < best_C:
            best_h = result_h
            best_C = result_C

    # Phase 3: refine best at its dimension
    if time.time() < deadline - 15:
        N_best = len(best_h)
        refine_time = deadline - time.time() - 5
        result_h, result_C = local_search(best_h, N_best, time.time() + refine_time, rng, tolerance=0.005)
        if result_C < best_C:
            best_h = result_h
            best_C = result_C

    # Final: dyadic repair and validation
    best_h = dyadic_repair(best_h)
    best_h = np.clip(best_h, 0, 1)
    final_C = exact_objective(best_h)

    return best_h, final_C, len(best_h)
