def slsqp_objective(x, tau, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    M = np.max(c)
    shifted = np.clip((c - M) / tau, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    return M + tau * np.log(Z)

def slsqp_gradient(x, tau, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    M = np.max(c)
    shifted = np.clip((c - M) / tau, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    w = exp_vals / Z
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        l = m - (n - 1)
        if l >= 0:
            grad[l:n] += wm * dx * (1.0 - x[:n-l])
            grad[:n-l] -= wm * dx * x[l:]
        else:
            p = -l
            grad[:n-p] += wm * dx * (1.0 - x[p:])
            grad[p:] -= wm * dx * x[:n-p]
    return grad

def slsqp_constraint_fun(x, n):
    return np.sum(x) - n / 2.0

def slsqp_constraint_jac(x, n):
    return np.ones(n)

def pair_move_search(h, n, target, best_c5, rng, max_evals=5000):
    """Try random pair moves to improve C5."""
    h = h.copy()
    c5 = best_c5
    deltas = [0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    for delta in deltas:
        for _ in range(max_evals // len(deltas)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Try moving mass from j to i
            h_try = h.copy()
            amt = min(delta, h[j], 1.0 - h[i])
            if amt < 1e-10:

            h_try[i] += amt
            h_try[j] -= amt
            c_new = exact_c5(h_try)
            if c_new < c5 - 1e-12:

                c5 = c_new

            c_new2 = exact_c5(h_try2)
            if c_new2 < c5 - 1e-12:

                c5 = c_new2
    return h, c5

def transfer_resolution(h, n_target):
    """Transfer h to a different resolution using area interpolation."""
    n_src = len(h)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_target
    # Build cumulative area
    cum = np.zeros(n_src + 1)
    cum[1:] = np.cumsum(h * dx_src)
    # Interpolate at target bin edges
    x_src = np.linspace(0.0, 2.0, n_src + 1)
    x_tgt = np.linspace(0.0, 2.0, n_target + 1)
    cum_tgt = np.interp(x_tgt, x_src, cum)
    h_tgt = np.diff(cum_tgt) / dx_tgt
    # Project to feasible
    return project_feasible(h_tgt, n_target / 2.0)

Valid when: 0 ≤ j ≤ n-1 and 0 ≤ n-1-m+j ≤ n-1
Second: m-n+1 ≤ j ≤ m
Combined: max(0, m-n+1) ≤ j ≤ min(n-1, m)

For m < n-1 (i.e., m = 0, 1, ..., n-2):
max(0, m-n+1) = m-n+1 (since m < n-1 means m-n+1 < 0, so max is 0)

Wait: m-n+1 < 0 when m < n-1. So max(0, m-n+1) = 0 for m < n-1.
And min(n-1, m) = m for m < n.

So for m = 0, 1, ..., n-2: j ranges from 0 to m.
c[m] = sum_{j=0}^{m} a[j] * v[n-1-m+j]

For m = n-1: j ranges from 0 to n-1.
c[n-1] = sum_{j=0}^{n-1} a[j] * v[j] (since n-1-(n-1)+j = j)

For m > n-1 (m = n, n+1, ..., 2n-2):
max(0, m-n+1) = m-n+1 (positive)
min(n-1, m) = n-1 (since m > n-1)
j ranges from m-n+1 to n-1.
c[m] = sum_{j=m-n+1}^{n-1} a[j] * v[n-1-m+j]
