def optimize_pair(h, i, j, c, grads_diff):
    """
    Given current h, correlation vector c, and grads_diff = grad_i - grad_j,
    find the best δ to transfer from j to i (h[i] += δ, h[j] -= δ).
    Returns the optimal δ and new c5.
    """
    # δ range: h[i] + δ >= 0 and h[j] - δ >= 0
    # Also h[i] + δ <= 1 and h[j] - δ <= 1
    delta_min = max(-h[i], h[j] - 1)
    delta_max = min(1 - h[i], h[j])
    
    if delta_min >= delta_max:
        return 0.0
    
    # c_new[k] = c[k] + delta * grads_diff[k]
    # We want to minimize max(c_new) over delta in [delta_min, delta_max]
    
    # The function f(delta) = max_k (c[k] + delta * grads_diff[k]) is convex and piecewise linear.
    # Its minimum occurs at a breakpoint or at a boundary.

    # delta = (c[k2] - c[k1]) / (gd[k1] - gd[k2])

import numpy as np
import time

def coral(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h):
    return float(np.max(coral(h)) * (2.0 / len(h)))

def project(h, S):
    h = np.clip(h, 0.0, 1.0)
    d = S - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        for i in np.argsort(-r, kind='stable'):
            a = min(d, r[i])
            h[i] += a
            d -= a
            if d < 1e-14:
                break
    else:
        d = -d
        for i in np.argsort(h, kind='stable'):
            a = min(d, h[i])
            h[i] -= a
            d -= a
            if d < 1e-14:
                break
    return h

def grads_mat(h):
    n = len(h)
    G = np.zeros((n, 2 * n - 1))
    for idx in range(n):
        for p in range(2 * n - 1):
            k = p - (n - 1)
            v = 0.0
            m = idx - k
            if 0 <= m < n:
                v += 1.0 - h[m]
            m2 = idx + k
            if 0 <= m2 < n:
                v -= h[m2]
            G[idx, p] = v
    return G

def line_search(c, gd, lo, hi):
    if lo >= hi:
        return (lo + hi) * 0.5
    ds = np.linspace(lo, hi, 50)
    fv = np.max(c[None, :] + ds[:, None] * gd[None, :], axis=1)
    bi = int(np.argmin(fv))
    if 0 < bi < 49:
        ds2 = np.linspace(ds[bi - 1], ds[bi + 1], 50)
        fv2 = np.max(c[None, :] + ds2[:, None] * gd[None, :], axis=1)
        bi2 = int(np.argmin(fv2))
        if 0 < bi2 < 49:
            ds3 = np.linspace(ds2[bi2 - 1], ds2[bi2 + 1], 30)
            fv3 = np.max(c[None, :] + ds3[:, None] * gd[None, :], axis=1)
            bi3 = int(np.argmin(fv3))
            return float(ds3[bi3])
        return float(ds2[bi2])
    return float(ds[bi])

def greedy_pairs(h, S, max_passes=100):
    n = len(h)
    c = coral(h)
    mx = float(c.max())
    for _ in range(max_passes):
        G = grads_mat(h)
        bg = 0.0
        bm = None
        for i in range(n):
            for j in range(i + 1, n):
                gd = G[i] - G[j]
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if lo >= hi - 1e-14:
                    continue
                d = line_search(c, gd, lo, hi)
                nf = float(np.max(c + d * gd))
                g = mx - nf
                if g > bg:
                    bg = g
                    bm = (i, j, d)
        if bm is None:
            break
        i, j, d = bm
        h[i] += d
        h[j] -= d
        h = project(h, S)
        c = coral(h)
        mx = float(c.max())
    return h, c

def stochastic_search(h, S, rng, n_iter, time_limit):
    n = len(h)
    c = coral(h)
    cur_max = float(c.max())
    best_max = cur_max
    best_h = h.copy()
