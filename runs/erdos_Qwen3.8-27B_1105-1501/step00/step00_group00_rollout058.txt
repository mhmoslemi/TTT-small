def smooth_max_grad(c, temperature):
    """Returns (soft_max_value, gradient_weights)"""
    m = np.max(c)
    exp_vals = np.exp((c - m) / temperature)
    weights = exp_vals / np.sum(exp_vals)
    value = m + temperature * np.log(np.sum(exp_vals))
    return value, weights

def grad_smooth_c5(h, dx, temperature):

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    _, weights = smooth_max_grad(c, temperature)

    # c_offset[i] corresponds to lag k = i - (n-1)
    grad = np.zeros(n)
    n_full = 2 * n - 1

        for i in range(n_full):

            # ∂c[k]/∂h[j] = (1-h[j-k])·[0≤j-k<n] - h[j+k]·[0≤j+k<n]

            if 0 <= j + k < n:
                dk -= h[j + k]
            grad[j] += weights[i] * dk

    """Try all systematic pair moves (more thorough but slower)"""
    n = len(h)
    current = h.astype(float).copy()
    current_c5 = compute_c5(current, dx)
    
    for _pass in range(max_passes):
        if time.time() > time_limit:
            break
        best_delta = 1e-11
        best_move = None
        
        for i in range(n):
            if current[i] < 0.05:
                continue
            for j in range(n):
                if j == i:
                    continue
                if current[j] > 0.95:
                    continue
                d = min(current[i], 1.0 - current[j])
                if d < 0.05:
                    continue
                for frac in (1.0, 0.5):
                    dd = d * frac
                    if dd < 0.01:
                        continue
                    new_h = current.copy()
                    new_h[i] = current[i] - dd
                    new_h[j] = current[j] + dd
                    new_h = np.clip(new_h, 0.0, 1.0)
                    if abs(np.sum(new_h) - target_sum) > 1e-9:
                        new_h = project_constraint(new_h, target_sum)
                    c5_new = compute_c5(new_h, dx)
                    delta = current_c5 - c5_new
                    if delta > best_delta:
                        best_delta = delta
                        best_move = (i, j, dd)
        
        if best_move is not None:
            i, j, dd = best_move
            current[i] -= dd
            current[j] += dd
            current = np.clip(current, 0.0, 1.0)
            if abs(np.sum(current) - target_sum) > 1e-9:
                current = project_constraint(current, target_sum)
            current_c5 = compute_c5(current, dx)
        else:
            break
    
    return current, current_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    if h0 is not None:
        n_init = len(h0)
    else:
        n_init = 45
    
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Optimize the initial construction
    n = n_init
    dx = 2.0 / n
    target = n / 2.0
    
    if h0 is not None:
        h = np.clip(h0, 0.0, 1.0)
        h = project_constraint(h, target)
    else:
        h = np.zeros(n)
        h[::2] = 1.0
        h = project_constraint(h, target)
    
    c5_init = compute_c5(h, dx)
    if c5_init < best_c5:
        best_c5 = c5_init
        best_h = h.copy()
    
    # Gradient descent on initial
    t_limit1 = t_start + budget_s * 0.25
    h_gd, c5_gd = gradient_descent(h, dx, target, t_limit1, 50000, 0.05, 0.02)
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd.copy()
    
    # Systematic local search on gradient result
    t_limit2 = t_start + budget_s * 0.4
    h_ls, c5_ls = systematic_local_search(h_gd, dx, target, t_limit2, 500)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
    
    # Also try on the original
    h_ls2, c5_ls2 = systematic_local_search(h, dx, target, t_limit2, 500)
    if c5_ls2 < best_c5:
        best_c5 = c5_ls2
        best_h = h_ls2.copy()
    
    # Phase 2: Random restarts on same n
    t_limit3 = t_start + budget_s * 0.65
    for trial in range(10):
        if time.time() > t_limit3:
            break
        h_rand = (rng.random(n) > 0.5).astype(float)
        h_rand = project_constraint(h_rand, target)
        h_opt, c5_opt = gradient_descent(h_rand, dx, target, t_limit3, 15000, 0.1, 0.05)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            # Local search on the good result
            t_ls = t_start + budget_s * 0.6
            h_opt2, c5_opt2 = systematic_local_search(h_opt, dx, target, t_ls, 200)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
    
    # Phase 3: Try other n values
    t_limit4 = t_start + budget_s * 0.85
    for n_try in [40, 44, 48, 50, 52, 56, 60, 64, 80, 100, 120, 160]:
        if time.time() > t_limit4:
            break
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        
        for trial in range(2):
            if time.time() > t_limit4:
                break
            # Start from scaled version of best or random
            if trial == 0 and best_h is not None:
                # Interpolate best_h to new size
                old_n = len(best_h)
                x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
                x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
                h_interp = np.interp(x_new, x_old, best_h, period=2.0)
                h_t = np.clip(h_interp, 0.0, 1.0)
                h_t = project_constraint(h_t, target_t)
            else:
                h_t = (rng.random(n_try) > 0.5).astype(float)
                h_t = project_constraint(h_t, target_t)
            
            h_opt, c5_opt = gradient_descent(h_t, dx_t, target_t, t_limit4, 10000, 0.08, 0.04)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 4: Final refinement
    t_limit5 = t_start + budget_s * 0.98
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    
    # One more gradient pass
    h_ref, c5_ref = gradient_descent(best_h, dx_final, target_final, t_limit5, 20000, 0.02, 0.01)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()
    
    # Final local search
    h_final, c5_final = systematic_local_search(best_h, dx_final, target_final, t_limit5, 1000)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    
    # Random local search as backup
    h_rls, c5_rls = local_search(best_h, dx_final, target_final, t_limit5, 500, rng)
    if c5_rls < best_c5:
        best_c5 = c5_rls
        best_h = h_rls.copy()
    
    return best_h, best_c5, len(best_h)
