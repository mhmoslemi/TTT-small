def make_feasible(y, n=None):
    """Project y onto {0 <= h <= 1, sum(h) = n/2}"""
    if n is None:
        n = len(y)
    target = n / 2.0
    y = np.array(y, dtype=np.float64).copy()
    # Handle non-finite values
    y[~np.isfinite(y)] = 0.5
    
    # Bisection to find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = y.min() - 1.0
    hi = y.max() + 1.0

        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(y - lam, 0.0, 1.0)
    
    # Exact repair
    diff = target - h.sum()
    if abs(diff) > 1e-12:

        if diff > 0:

            idx = np.where(h < 1.0 - 1e-9)[0]

def make_feasible(y):

    n = len(y)
    target = n / 2.0

    y[~np.isfinite(y)] = 0.5
    lo = y.min() - 1.0

    n = len(h)

        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
        smooth_val = base
    else:

        smooth_val = base + T * np.log(S)
    g = np.zeros(n)
    # d=0

        if wp > 1e-15:

            g[d:n] += wp * (1.0 - h[0:n-d])
        wn = w[center - d]

            g[d:n] += wn * (-h[0:n-d])

    """Resample to new resolution via linear interpolation."""

        accepted = False
        for scale in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1, step_size * 0.05]:
            trial = make_feasible(cur - scale * g_norm)
            trial_score = score_exact(trial)
            if trial_score < cur_score - 1e-13:
                cur = trial
                cur_score = trial_score
                accepted = True
                no_improve = 0
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best = cur.copy()
                break

        if not accepted:
            no_improve += 1
            step_size *= 0.7
            if step_size < 1e-7:
                step_size = 0.03
                cur = make_feasible(cur + 0.015 * rng.standard_normal(n))
                cur_score = score_exact(cur)
                no_improve = 0

        iters += 1

        # Pair line searches periodically
        if iters % pair_interval == 0 and time.time() < deadline - 1.0:
            sorted_idx = np.argsort(g)
            lows = sorted_idx[:10]
            highs = sorted_idx[-10:]
            n_pairs = 3 if n > 50 else 2
            for _ in range(n_pairs):
                i = int(lows[rng.randint(len(lows))])
                j = int(highs[rng.randint(len(highs))])
                if i == j:
                    continue
                h_new, s_new = pair_line_search(cur, i, j)
                if h_new is not None and s_new < cur_score - 1e-13:
                    cur = h_new
                    cur_score = s_new
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best = cur.copy()

        # Occasional random pair
        if iters % 7 == 0 and time.time() < deadline - 1.0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                h_new, s_new = pair_line_search(cur, i, j)
                if h_new is not None and s_new < cur_score - 1e-13:
                    cur = h_new
                    cur_score = s_new
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best = cur.copy()

        if no_improve > max_no_improve:
            no_improve = 0
            step_size = 0.03
            sigma = 0.01 + 0.02 * rng.random()
            cur = make_feasible(cur + sigma * rng.standard_normal(n))
            cur_score = score_exact(cur)

    return best, best_score


def try_candidate(h, deadline, rng, local_time):
    """Run local refine on a candidate, return best found."""
    sub_deadline = min(deadline, time.time() + local_time)
    if time.time() >= sub_deadline:
        s = score_exact(h)
        return h, s
    return local_refine(h, sub_deadline, rng)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 3.0
    rng = np.random.RandomState(seed)

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).flatten().copy()
    except (NameError, TypeError):
        h0 = None

    best_h = None
    best_c = np.inf

    if h0 is not None and len(h0) >= 4:
        h0 = make_feasible(h0)
        c0 = score_exact(h0)
        best_h = h0.copy()
        best_c = c0
    else:
        # Fallback: constant 0.5
        h0 = np.full(70, 0.5)
        best_h = h0.copy()
        best_c = score_exact(h0)

    n0 = len(h0)

    # Phase 1: Long refinement on the initial construction
    phase1_time = min(0.35 * budget_s, 350.0)
    phase1_deadline = start_time + phase1_time
    if time.time() < phase1_deadline:
        h_ref, c_ref = local_refine(best_h, phase1_deadline, rng)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref.copy()

    # Phase 2: Generate diverse candidates and refine each
    candidates = []

    # Binarized version
    b = binarize(best_h)
    candidates.append(b)

    # Mutations of best
    for sigma in [0.02, 0.05, 0.1, 0.2]:
        mu = make_feasible(best_h + sigma * rng.standard_normal(n0))
        candidates.append(mu)

    # Scaled versions
    for s in [0.7, 1.3, 1.6]:
        sc = make_feasible(0.5 + s * (best_h - 0.5))
        candidates.append(sc)

    # Resolution changes
    res_list = []
    for delta in [-10, -6, -4, 0, 4, 8, 12, 16, 20]:
        nn = n0 + delta
        if 20 <= nn <= 160 and nn != n0:
            res_list.append(nn)
    for nn in res_list[:6]:
        try:
            rc = resample(best_h, nn)
            candidates.append(rc)
        except Exception:
            pass

    # Random feasible starts
    for _ in range(4):
        rc = make_feasible(rng.random(n0))
        candidates.append(rc)

    # Structured: sine-based
    for freq in [2, 3, 5]:
        x = np.linspace(0, 2, n0)
        sv = np.clip(0.5 + 0.4 * np.sin(np.pi * freq * x), 0, 1)
        candidates.append(make_feasible(sv))

    # Evaluate and sort candidates by score
    scored_candidates = []
    for cand in candidates:
        if len(cand) != len(cand):
            continue
        s = score_exact(cand)
        if np.isfinite(s):
            scored_candidates.append((s, cand))

    scored_candidates.sort(key=lambda x: x[0])

    # Phase 3: Refine top candidates
    phase3_deadline = deadline
    phase3_time_left = phase3_deadline - time.time()
    n_to_try = min(len(scored_candidates), 12)
    if n0 > 100:
        time_per_cand = max(3.0, phase3_time_left / max(n_to_try, 1) - 2.0)
    else:
        time_per_cand = max(5.0, phase3_time_left / max(n_to_try, 1) - 2.0)
    time_per_cand = min(time_per_cand, 60.0)

    tried = set()
    for rank, (cand_score, cand) in enumerate(scored_candidates[:n_to_try]):
        if time.time() > phase3_deadline - 2.0:
            break
        # Skip if too similar to already tried
        key = tuple(np.round(cand, 2))
        if key in tried:
            continue
        tried.add(key)

        local_time = time_per_cand
        if rank == 0:
            local_time = time_per_cand * 1.5
        hh, cc = try_candidate(cand, phase3_deadline, rng, local_time)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    # Phase 4: Final polish on the best
    remaining = deadline - time.time()
    if remaining > 5.0:
        hh, cc = local_refine(best_h, deadline, rng)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    # Final feasibility check and exact score
    best_h = make_feasible(best_h)
    best_c = score_exact(best_h)

    # Verify constraints
    n_final = len(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(best_h.sum() - n_final / 2.0) < 1e-6

    return (best_h, best_c, n_final)
