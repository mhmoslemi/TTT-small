def dyadic_repair(h, n, bits=20):
    """Snap h to dyadic grid with exact sum."""
    S = 1 << bits
    target = n * S // 2  # since sum should be n/2, and each value is z/S

    z = np.clip(z, 0, S)
    # Water fill to exact target
    cur = z.sum()
    while cur > target:
        # Find element with z > 0, reduce the one that causes least damage
        # Priority: reduce elements that are "redundant" (high value)
        candidates = np.where(z > 0)[0]
        if len(candidates) == 0:
            break
        # Reduce the one with highest z (most "excess")
        i = candidates[np.argmax(z[candidates])]
        z[i] -= 1
        cur -= 1
    while cur < target:
        # Find element with z < S, increase
        candidates = np.where(z < S)[0]
        if len(candidates) == 0:
            break
        i = candidates[np.argmin(z[candidates])]
        z[i] += 1
        cur += 1
    return z.astype(np.float64) / S

        np.random.seed(0)  # deterministic
        order = idx[np.argsort(-np.abs(h - 0.5))]
        for i in order:
            if diff > 0 and h[i] < 1.0:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
            elif diff < 0 and h[i] > 0.0:
                amt = max(diff, -(h[i]))
                h[i] += amt
                diff -= amt
            if abs(diff) < 1e-15:
                break
    return h

def dyadic_repair(h, n, bits=20):
    S = 1 << bits
    if n % 2 == 1 and bits == 0:
        bits = 1
        S = 1 << bits
    target = (n * S) // 2
    z = np.rint(h * S).astype(np.int64)
    z = np.clip(z, 0, S)
    cur = int(z.sum())
    # Water fill
    while cur > target:
        candidates = np.where(z > 0)[0]
        if len(candidates) == 0:
            break
        # Reduce the one with highest z
        i = candidates[np.argmax(z[candidates])]
        z[i] -= 1
        cur -= 1
    while cur < target:
        candidates = np.where(z < S)[0]
        if len(candidates) == 0:
            break
        i = candidates[np.argmin(z[candidates])]
        z[i] += 1
        cur += 1
    return z.astype(np.float64) / S

def resample_h(parent, new_n):
    n_old = len(parent)
    if n_old == new_n:
        return parent.copy()
    idx_float = (np.arange(new_n) + 0.5) * n_old / float(new_n)
    idx_lo = np.floor(idx_float).astype(int)
    idx_hi = np.minimum(idx_lo + 1, n_old - 1)
    frac = idx_float - idx_lo
    new_h = (1.0 - frac) * parent[idx_lo] + frac * parent[idx_hi]
    new_h = np.clip(new_h, 0.0, 1.0)
    target = new_n / 2.0
    diff = target - new_h.sum()
    if abs(diff) > 1e-14:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
        diff = target - new_h.sum()
        if abs(diff) > 1e-15:
            if diff > 0:
                i = np.argmax(1.0 - new_h)
                new_h[i] = min(1.0, new_h[i] + diff)
            else:
                i = np.argmax(new_h)
                new_h[i] = max(0.0, new_h[i] + diff)
    return new_h

    if current_C > best_C_global + 0.01:

            if not improved:

        else:

            ov = overlap_vector(h)

        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += 1.0 - h[lo1 - k:hi1 - k + 1]
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += -h[lo2 + k:hi2 + k + 1]

    with sum(d)=0 and box constraints on d."""
    try:

        return -grad + np.mean(grad)  # fallback: negative subgradient projected

    c = grad
    A_eq = np.array([[np.ones(n)]])
    b_eq = [0.0]
    bounds_lp = [(-bounds, bounds)] * n
    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
    if res.success:
        return res.x
    return None

    dp = dyadic_repair(parent, n_parent, bits=20)
    dp_C = exact_objective(dp)
    if dp_C < best_C:
        best_h = dp.copy()
        best_C = dp_C
    
    # Build dimension list
    dims = [n_parent]
    for d in range(max(16, n_parent - 10), n_parent):
        dims.append(d)
    for d in range(n_parent + 1, min(n_parent + 14, 70) + 1):
        dims.append(d)
    # Sort by distance from parent
    dims = [n_parent] + sorted(set(dims) - {n_parent}, key=lambda d: (abs(d - n_parent), -d))
    
    total_dims = len(dims)
    
    for dim_idx, n_try in enumerate(dims):
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 5
        if remaining < 15:
            break
        
        # Time allocation: proportional to remaining / remaining dims
        rest = total_dims - dim_idx
        dim_budget = remaining / rest
        dim_deadline = time.time() + dim_budget
        
        # Starting point: resample from best
        h_start = resample_h(best_h, n_try)
        h_start_C = exact_objective(h_start)
        if h_start_C < best_C:
            best_h = h_start.copy()
            best_C = h_start_C
            best_n = n_try
        
        # Optimize
        h_opt, C_opt = optimize_local(h_start, n_try, dim_deadline, rng)
        
        # Dyadic snap
        h_dyad = dyadic_repair(h_opt, n_try, bits=20)
        C_dyad = exact_objective(h_dyad)
        if C_dyad < C_opt:
            h_opt = h_dyad
            C_opt = C_dyad
        
        if C_opt < best_C:
            best_h = h_opt.copy()
            best_C = C_opt
            best_n = n_try
    
    # Final: try a few more refinement passes on the best
    remaining = deadline - time.time() - 3
    if remaining > 10:
        h_ref, C_ref = optimize_local(best_h, best_n, time.time() + remaining, rng)
        h_dyad = dyadic_repair(h_ref, best_n, bits=20)
        C_dyad = exact_objective(h_dyad)
        if C_dyad < C_ref:
            h_ref = h_dyad
            C_ref = C_dyad
        if C_ref < best_C:
            best_h = h_ref.copy()
            best_C = C_ref
            best_n = best_n
    
    # Final validation
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    s = best_h.sum()
    target = n_final / 2.0
    if abs(s - target) > 1e-10:
        best_h = dyadic_repair(best_h, n_final, bits=20)
    
    final_C = exact_objective(best_h)
    
    return best_h, final_C, n_final
