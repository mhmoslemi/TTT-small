- Evaluating at the breakpoints (where two of the lines cross)
- Checking the boundaries of the interval

def transfer_descent(h, n, target, max_iter=100):

    h = h.copy()
    best_val = c5_value(h)

    for it in range(max_iter):
        improved = False

        for i in range(n):

                if i == j:
                    continue

                # Then find optimal delta
                # For speed, only try a few pairs
                
                # Compute gradient of each C_k w.r.t. h[i] and h[j]
                # dC_k/dh[i] and dC_k/dh[j]
                diffs = np.zeros(2*n-1)
                for k in range(2*n-1):
                    l = k - (n-1)
                    # dC_k/dh[i]
                    di = 0.0
                    if l >= 0:
                        if i >= l and i <= n-1:
                            di += (1.0 - h[i-l])
                        if i <= n-1-l:

                    else:

                            di += (1.0 - h[i+p])
                        if i >= p:
                            di += (-h[i-p])

                    dj = 0.0
                    if l >= 0:
                        if j >= l and j <= n-1:
                            dj += (1.0 - h[j-l])
                        if j <= n-1-l:
                            dj += (-h[j+l])
                    else:
                        p = -l
                        if j <= n-1-p:
                            dj += (1.0 - h[j+p])
                        if j >= p:
                            dj += (-h[j-p])
                    
                    diffs[k] = dx * (di - dj)

                # subject to max(0, h[i]) - 1 <= delta <= min(1, h[i]) - 0  ... 
                # i.e., -h[i] <= delta <= 1 - h[i] and -(1-h[j]) <= delta <= h[j]
                delta_lo = max(-h[i], -(1-h[j]))
                delta_hi = min(1-h[i], h[j])
                
                if delta_lo >= delta_hi:
                    continue

                    h[j] -= best_d

                    corr = get_corr(h)
                    improved = True
        
        if not improved:
            break
    
    return h, best_val

def find_best_transfer_delta(corr, diffs, delta_lo, delta_hi):
    """Find delta that minimizes max(corr + delta*diffs)."""
    # Compute all pairwise breakpoints
    dc = corr[:, None] - corr[None, :]  # (k, l) matrix
    dd = diffs[None, :] - diffs[:, None]  # (k, l) matrix
    # Where dd != 0: breakpoint = dc / dd
    with np.errstate(divide='ignore', invalid='ignore'):
        d_all = dc / dd
    # Filter valid breakpoints
    valid = (d_all > delta_lo) & (d_all < delta_hi) & np.isfinite(d_all) & (np.abs(dd) > 1e-15)
    candidates = d_all[valid]
    candidates = np.concatenate([candidates, [delta_lo, delta_hi]])
    
    # Evaluate at all candidates
    best_cv = np.inf
    best_d = 0
    for d in candidates:
        cv = np.max(corr + d * diffs)
        if cv < best_cv:
            best_cv = cv
            best_d = d
    
    return best_d, best_cv

        m2 = delta_hi - (delta_hi - delta_lo) / 3

            delta_hi = m2

            delta_lo = m1
