def make_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0

    lo = y.min() - 1.0 - 1e-15
    hi = y.max() + 1e-15

        return np.clip(y - lam, 0.0, 1.0).sum() - target

        else:

def smooth_max_and_grad(h, T):
    """Compute smooth max surrogate value and gradient."""
    c = score_scores(h)
    n = len(h)
    base = np.max(c)
    ex = np.exp((c - base) / T)
    S = ex.sum()
    if S < 1e-300:
        # Fallback: uniform over max elements
        weights = np.zeros_like(c)
        max_mask = c > base - 1e-10
        weights[max_mask] = 1.0 / np.sum(max_mask)
        smooth_val = base
    else:
        weights = ex / S
        smooth_val = base + T * np.log(S)
    
    # Compute gradient
    grad = np.zeros(n, dtype=np.float64)
    center = n - 1
    dx = 2.0 / n
    
    # d = 0 (center)
    w0 = weights[center]
    if w0 > 1e-15:
        grad += w0 * dx * (1.0 - 2.0 * h)
    
    # d = 1, ..., n-1
    for d in range(1, n):
        # c[center + d] = dx * sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        wp = weights[center + d]
        if wp > 1e-15:
            # For j in 0..n-1-d: grad contribution from (1-h[j]) term: wp*dx*(-h[j+d])
            grad[0:n-d] += wp * dx * (-h[d:n])
            # For j in d..n-1: grad contribution from h[j] term: wp*dx*(1-h[j-d])
            grad[d:n] += wp * dx * (1.0 - h[0:n-d])
        
        # c[center - d] = dx * sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        wn = weights[center - d]
        if wn > 1e-15:
            # For j in 0..n-1-d: h[j] is first factor: wn*dx*(1-h[j+d])
            grad[0:n-d] += wn * dx * (1.0 - h[d:n])
            # For j in d..n-1: h[j] in (1-h[i+d]) where i=j-d: wn*dx*(-h[j-d])
            grad[d:n] += wn * dx * (-h[0:n-d])
    
    return smooth_val, grad

def local_refine(h_init, deadline, rng, T_start=0.1, T_end=1e-4, n_phases=8):

    best_h = h.copy()
    best_score = cur_score
    
    # Temperature schedule (log-linear)
    temps = np.geomspace(T_start, T_end, n_phases)
    
    # Phase durations proportional to remaining time
    now = time.time()
    remaining = max(1.0, deadline - now)
    # Use more time for later phases (lower T)

    phase_durations = phase_frac * remaining

        phase_end = time.time() + phase_durations[phase]
        step_size = 0.05 / (1 + 0.3 * phase)
        rejected_count = 0
        max_rejections = 50 + 20 * phase

            if rejected_count > max_rejections:
                break
            
            smooth_val, grad = smooth_max_and_grad(h, T)
            g_norm = np.sqrt(np.mean(grad * grad))
            if g_norm < 1e-14:
                break

                step_size *= 0.8

import numpy as np

def make_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    y = np.where(np.isfinite(y), y, 0.5)
    lo = y.min() - 1.0 - 1e-15
    hi = y.max() + 1e-15
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.clip(y - mid, 0.0, 1.0).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            slack = 1.0 - z
            order = np.argsort(-slack, kind='stable')
            rem = diff
            for idx in order:
                add = min(rem, slack[idx])
                z[idx] += add
                rem -= add
                if rem < 1e-14:
                    break
        else:
            rem = -diff
            order = np.argsort(z, kind='stable')
            for idx in order:
                sub = min(rem, z[idx])
                z[idx] -= sub
                rem -= sub
                if rem < 1e-14:
                    break
    z = np.clip(z, 0.0, 1.0)
    total = z.sum()
    if abs(total - target) > 1e-12:
        idx = np.argmax(1.0 - z) if total < target else np.argmin(z)
        adj = target - total
        z[idx] = np.clip(z[idx] + adj, 0.0, 1.0)
    return z


def score_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n < 2:
        return 0.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def smooth_max_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    base = np.max(c)
    ex = np.exp((c - base) / T)
    S = ex.sum()
    if S < 1e-300:
        w = np.zeros_like(c)
        mask = c >= base - 1e-12
        w[mask] = 1.0 / mask.sum()
    else:
        w = ex / S
    center = n - 1
    grad = np.zeros(n)
    wc = w[center]
    if wc > 1e-14:
        grad += wc * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[center + d]
        if wp > 1e-14:
            grad[d:] += wp * dx * (1.0 - 2.0 * h[n - 1 + d - d:n] [:n - d])
        wn = w[center - d]
        if wn > 1e-14:
            grad[:n - d] += wn * dx * (1.0 - 2.0 * h[n - 1 - d:])
    return base + T * np.log(max(S, 1e-300)), grad


def local_refine(h_init, deadline, rng, T_start=0.10, T_end=1e-4, n_phases=8):
    h = np.array(h_init, dtype=np.float64)
    cur_score = score_exact(h)
    best_h = h.copy()
    best_score = cur_score
    n = len(h)
    temps = np.geomspace(T_start, T_end, n_phases)
    now = time.time()
    remaining = max(1.0, deadline - now)
    wts = np.linspace(0.5, 1.5, n_phases)
    wts = wts / wts.sum()
    phase_durs = wts * remaining
    for phase in range(n_phases):
        T = temps[phase]
        pe = time.time() + phase_durs[phase]
        step = 0.05 / (1.0 + 0.3 * phase)
        rej = 0
        max_rej = 40 + 20 * phase
        while time.time() < pe:
            if rej > max_rej or time.time() > deadline:
                break
            sv, grad = smooth_max_and_grad(h, T)
            gn = np.sqrt(np.mean(grad * grad))
            if gn < 1e-14:
                break
            direction = -grad / gn
            improved = False
            for s in [step, step * 0.5, step * 0.25, step * 0.1]:
                h_t = make_feasible(h + s * direction)
                s_t = score_exact(h_t)
                if s_t < cur_score - 1e-13:
                    h = h_t
                    cur_score = s_t
                    rej = 0
                    improved = True
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = h.copy()
                    break
                elif s_t <= cur_score + 1e-11 and rng.random() < 0.05:
                    h = h_t
                    cur_score = s_t
                    rej = 0
                    improved = True
                    break
            if not improved:
                rej += 1
                step *= 0.85
            # Add small noise occasionally to escape plateaus
            if not improved and rej % 30 == 0:
                noise = rng.randn(n) * 0.02
                h_n = make_feasible(h + noise)
                s_n = score_exact(h_n)
                if s_n <= cur_score:
                    h = h_n
                    cur_score = s_n
    # Pair line search on best
    h = best_h
    cur_s = best_score
    best_hl = h.copy()
    best_sl = cur_s
    pairs_per_iter = min(8, n // 8 + 2)
    it = 0
    while time.time() < deadline - 2:
        it += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        lo_a = max(-h[i], h[j] - 1.0)
        hi_a = min(1.0 - h[i], h[j])
        if hi_a - lo_a < 1e-10:
            continue
        alpha_best = 0.0
        s_best = cur_s
        for _ in range(16):

            h[i] += a
            h[j] -= a
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i] -= a
                h[j] += a
                continue
            s_a = score_exact(h)
            if s_a < s_best:
                s_best = s_a
                alpha_best = a
            h[i] -= a
            h[j] += a
        if alpha_best != 0.0 and s_best < cur_s - 1e-14:
            h[i] += alpha_best
            h[j] -= alpha_best
            h = np.clip(h, 0.0, 1.0)
            cur_s = s_best
            if cur_s < best_sl:
                best_sl = cur_s
                best_hl = h.copy()
    return best_hl, best_sl


def binarize_top(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(h, kind='stable')[-k:]
    z = np.zeros(n)
    z[idx] = 1.0
    return make_feasible(z)


def random_feasible(n, rng):
    y = rng.random(n)
    return make_feasible(y)


def resample_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    dx_o = 2.0 / old_n
    dx_n = 2.0 / new_n
    x_o = (np.arange(old_n) + 0.5) * dx_o
    x_n = (np.arange(new_n) + 0.5) * dx_n
    y = np.interp(x_n, x_o, h)
    return make_feasible(y)
