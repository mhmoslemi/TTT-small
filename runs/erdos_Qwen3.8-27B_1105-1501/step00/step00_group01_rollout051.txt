    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, S):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_val(h, eps):
    corr = compute_corr(h)
    m = np.max(corr)
    return m + eps * np.log(np.sum(np.exp((corr - m) / eps)))

def smooth_max_grad(h, eps):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    exp_c = np.exp((corr - m) / eps)
    w = exp_c / np.sum(exp_c)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        d = j - (n - 1)
        wj = w[j] * dx
        if wj < 1e-14:
            continue
        if d >= 0:
            grad[d:] += wj * (1.0 - h[:n-d])
            grad[:n-d] -= wj * h[d:]
        else:
            p = -d
            grad[:n-p] += wj * (1.0 - h[p:])
            grad[p:] -= wj * h[:n-p]
    return grad

        if gnorm < 1e-12:
