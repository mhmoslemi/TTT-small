def optimize_coordinate(h, i, n):
    """Find optimal value for h[i] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute c(j) as function of h[i] = t
    # c(j) = sum_k h[k]*(1-h[k+j]) for valid k
    # Terms involving h[i]:
    #   k=i: h[i]*(1-h[i+j]) = t*(1-h[i+j])
    #   k=i-j: h[i-j]*(1-h[i]) = h[i-j]*(1-t)
    # So c(j) = t*(1-h[i+j]) + h[i-j]*(1-t) + rest(j)
    #         = t*(1-h[i+j]-h[i-j]) + h[i-j] + rest(j)
    
    # Compute for each lag j:
    # a_j = 1-h[i+j]-h[i-j] (with boundary handling: if index out of range, term is 0)
    # b_j = h[i-j] + rest(j) where rest(j) = sum of h[k]*(1-h[k+j]) for k≠i, k≠i-j
    
    n_corr = 2*n - 1  # length of full correlation
    a = np.zeros(n_corr)
    b = np.zeros(n_corr)
    
    # For each lag index j in [0, 2n-2]:
    # The shift is (j - (n-1)) in "lag" units
    # c(j) = sum_k h[k] * (1 - h[k + j - (n-1)])
    # valid k: 0 ≤ k < n and 0 ≤ k + j - (n-1) < n
    
    shift = j - (n-1)  # this is the actual shift
    # c(j) = sum_{k: 0≤k<n, 0≤k+shift<n} h[k]*(1-h[k+shift])
    
    for j in range(n_corr):
        shift = j - (n-1)
        # Valid k range: max(0, -shift) ≤ k ≤ min(n-1, n-1-shift)
        k_min = max(0, -shift)
        k_max = min(n-1, n-1-shift)
        
        # Compute the sum without h[i], then add back terms with h[i]
        total = 0.0
        for k in range(k_min, k_max+1):
            if k == i:
                continue  # will handle separately
            kp = k + shift
            if kp == i:
                continue  # will handle separately (this is the h[i-j]*(1-h[i]) term)
            total += h[k] * (1 - h[kp])
        
        # Add terms involving h[i] = t
        alpha = 0.0
        beta = 0.0
        
        # Term k=i: h[i]*(1-h[i+shift]) = t*(1-h[i+shift]) if i+shift is valid
        if i >= k_min and i <= k_max:
            kp = i + shift
            if 0 <= kp < n:
                alpha += (1 - h[kp])
        
        # Term where k+shift = i, i.e., k = i-shift: h[i-shift]*(1-h[i]) = h[i-shift]*(1-t)
        k_ks = i - shift
        if k_ks >= k_min and k_ks <= k_max and k_ks != i:
            beta += h[k_ks]
            alpha -= h[k_ks]  # because h[i-shift]*(1-t) = h[i-shift] - t*h[i-shift]
        
        a[j] = alpha
        b[j] = total + beta
    
    # Now find min_{t∈[0,1]} max_j (a[j]*t + b[j])
    # g(t) = max_j (a[j]*t + b[j])
    
    # Evaluate at t=0 and t=1
    g0 = np.max(b)
    g1 = np.max(a + b)
    
    # Find all intersection points
    # For pairs (j1, j2): a[j1]*t + b[j1] = a[j2]*t + b[j2]
    # t = (b[j2] - b[j1]) / (a[j1] - a[j2])
    
    best_val = min(g0, g1)
    best_t = 0 if g0 < g1 else 1
    
    # Only consider pairs where the intersection is in (0,1)
    # For efficiency, consider all pairs
    for j1 in range(n_corr):

            da = a[j1] - a[j2]
            if abs(da) < 1e-12:

            if 0 < t < 1:

                if val < best_val:

                    best_t = t
    
    return best_t

    # a is shape (n_corr,), b is shape (n_corr,)
    
    # For each pair (p, q), intersection at t = (b[q]-b[p])/(a[p]-a[q])
    # Use meshgrid
    A = a[:, None]  # (n_corr, 1)
    B = b[:, None]  # (n_corr, 1)
    A2 = a[None, :]  # (1, n_corr)
    B2 = b[None, :]  # (1, n_corr)
    
    dA = A - A2  # (n_corr, n_corr)
    dB = B2 - B  # (n_corr, n_corr)
    
    # t = dB / dA where dA ≠ 0
    valid = np.abs(dA) > 1e-12
    with np.errstate(divide='ignore', invalid='ignore'):
        t_matrix = np.where(valid, dB / dA, np.nan)
    
    # Filter to (0, 1)
    t_matrix[~valid] = np.nan
    t_matrix[(t_matrix <= 0) | (t_matrix >= 1)] = np.nan
    
    # For each valid t, compute g(t) = max(a*t + b)
    # a*t + b has shape (n_corr, n_corr) for each t
    # Actually, for each valid (p,q) pair, we have a t value.
    # We need to evaluate max_j (a[j]*t + b[j]) for each t.

    valid_t = t_matrix[~np.isnan(t_matrix)]
    
    if len(valid_t) > 0:
        # For each t, compute max(a*t + b)
        vals = np.max(a[None, :] * valid_t[:, None] + b[None, :], axis=1)
        best_idx = np.argmin(vals)
        best_val = vals[best_idx]
        best_t = valid_t[best_idx]

    g0 = np.max(b)

    if g0 < best_val:

        best_t = 0

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_to_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)

def optimize_one_coordinate(h, i, n):
    n_corr = 2 * n - 1
    d_range = np.arange(n_corr) - (n - 1)
    
    h_save = h[i]
    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    
    alpha = np.zeros(n_corr)
    for idx in range(n_corr):
        d = d_range[idx]
        if d == 0:
            alpha[idx] = 1.0
        else:
            hp = h[i + int(d)] if 0 <= i + int(d) < n else 0.0
            hm = h[i - int(d)] if 0 <= i - int(d) < n else 0.0
            alpha[idx] = 1.0 - hp - hm
    
    # Grid search + refinement
    t_grid = np.linspace(0.0, 1.0, 101)
    g_max = np.full(len(t_grid), -np.inf)

            vals = base[j] + t_grid - t_grid * t_grid
        else:
            vals = base[j] + alpha[j] * t_grid
        np.maximum(g_max, vals, out=g_max)
    
    best_idx = int(np.argmin(g_max))
    if best_idx == 0:
        return 0.0
    if best_idx == len(t_grid) - 1:
        return 1.0

    lo = t_grid[best_idx - 1]
    hi = t_grid[best_idx + 1]

            a1 = b1
            fa = fb
