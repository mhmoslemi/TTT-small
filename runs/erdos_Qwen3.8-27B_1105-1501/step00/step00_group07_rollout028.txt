import numpy as np
import time

def exact_score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * 2.0 / n)

def project_feas(v):
    n = len(v)
    target = n / 2.0
    v = np.array(v, dtype=float, copy=True)
    v[~np.isfinite(v)] = 0.5
    np.clip(v, 0.0, 1.0, out=v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    v = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(300):
        res = target - v.sum()
        if abs(res) < 1e-13:
            break
        if res > 0:
            j = int(np.argmax(1.0 - v))
            amt = min(res, 1.0 - v[j])
            v[j] += amt
        else:
            j = int(np.argmin(v))
            amt = min(-res, v[j])
            v[j] -= amt
    return v

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feas(h_new)

def synthetic_initial(n):
    x = np.arange(n)
    h = np.zeros(n)
    # Block construction: alternating blocks
    block = max(1, n // 8)
    for i in range(n):
        phase = (i // block) % 2
        if phase == 0:
            h[i] = 0.9
        else:
            h[i] = 0.1
    return project_feas(h)

def pair_line_search(h, a, b):
    n = len(h)
    p = h[a]
    q = h[b]
    t_low = -min(p, 1.0 - q)
    t_high = min(1.0 - p, q)
    if t_high - t_low < 1e-14:
        return exact_score(h)

    best_t = 0.0
    best_c = exact_score(h)

    nc = 12
    for i in range(1, nc + 1):
        t = t_low + i * (t_high - t_low) / nc
        h[a] = p + t
        h[b] = q - t
        c = exact_score(h)
        if c < best_c:
            best_c = c
            best_t = t
    h[a] = p
    h[b] = q

    if best_t <= t_low + 1e-14 or best_t >= t_high - 1e-14:
        h[a] = p + best_t
        h[b] = q - best_t
        return exact_score(h)

    span = (t_high - t_low) / nc
    lo = max(t_low, best_t - span)
    hi = min(t_high, best_t + span)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = lo + (1.0 - gr) * (hi - lo)
    b_g = lo + gr * (hi - lo)
    h[a] = p + a_g; h[b] = q - a_g
    c_a = exact_score(h)
    h[a] = p + b_g; h[b] = q - b_g
    c_b = exact_score(h)
    for _ in range(22):
        if c_a <= c_b:
            hi = b_g; b_g = a_g; c_b = c_a
            a_g = lo + (1.0 - gr) * (hi - lo)
            h[a] = p + a_g; h[b] = q - a_g
            c_a = exact_score(h)
        else:
            lo = a_g; a_g = b_g; c_a = c_b
            b_g = lo + gr * (hi - lo)
            h[a] = p + b_g; h[b] = q - b_g
            c_b = exact_score(h)
    t_mid = (lo + hi) * 0.5
    h[a] = p + t_mid; h[b] = q - t_mid
    c_mid = exact_score(h)
    if c_mid < best_c:
        best_c = c_mid
        best_t = t_mid

    h[a] = p + best_t
    h[b] = q - best_t
    return exact_score(h)

def pair_sweep(h, deadline, min_gain=1e-12):
    n = len(h)
    total_imp = 0
    sweeps = 0
    while time.time() < deadline:
        improved = False
        for a in range(n):
            if time.time() > deadline:
                break
            for b in range(a + 1, n):
                c_before = exact_score(h)
                c_after = pair_line_search(h, a, b)
                if c_after < c_before - min_gain:
                    improved = True
                    total_imp += 1
        sweeps += 1
        if not improved:
            break
    return h, exact_score(h), total_imp

def random_perturb(h, magnitude, rng):
    n = len(h)
    delta = rng.uniform(-magnitude, magnitude, size=n)
    v = h + delta
    return project_feas(v)
