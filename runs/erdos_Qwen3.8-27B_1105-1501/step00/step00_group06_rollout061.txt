    return d

def make_sinusoidal(n, mode, amplitude, phase=0.0):
    x = np.arange(n)
    h = 0.5 + amplitude * np.cos(2.0 * np.pi * mode * x / n + phase)
    return project_box_mass(h)

def make_random_h(n, rng):
    h = rng.standard_normal(n) * 0.2 + 0.5
    return project_box_mass(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 0:
        h_best = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h_best = make_sinusoidal(70, 1, 0.3)

    c5_best = compute_c5(h_best)
    n_best = len(h_best)

    # Phase 1: Smoothed max optimization on initial n
    if time_left() > 60:
        for beta in [5.0, 10.0, 20.0, 50.0, 100.0]:
            if time_left() < 30:
                break
            h_try = optimize_smoothed(h_best, beta, max_iter=500)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try
                n_best = len(h_best)

    # Phase 2: Subgradient + line search iterations
    if time_left() > 30:
        for iteration in range(500):
            if time_left() < 15:
                break
            d = subgradient_direction(h_best, eps=1e-5)
            if d is None or np.linalg.norm(d) < 1e-14:
                break
            d_norm = np.linalg.norm(d)
            d = d / d_norm * 0.1
            h_try, c5_try = line_search(h_best, d, c5_best, n_coarse=30, n_fine=30, max_span=0.5)
            if c5_try < c5_best - 1e-14:
                c5_best = c5_try
                h_best = h_try
            else:
                # Try smaller step
                d2 = -d * 0.3
                h_try2, c5_try2 = line_search(h_best, d2, c5_best, n_coarse=20, n_fine=20, max_span=0.3)
                if c5_try2 < c5_best - 1e-14:
                    c5_best = c5_try2
                    h_best = h_try2

    # Phase 3: LP-based direction finding
    if time_left() > 30:
        for iteration in range(200):
            if time_left() < 15:
                break
            d = lp_direction(h_best, trust_radius=0.03)
            if d is None:
                break
            d -= np.mean(d)
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-12:
                break
            d = d / d_norm * 0.05
            h_try, c5_try = line_search(h_best, d, c5_best, n_coarse=30, n_fine=30, max_span=0.3)
            if c5_try < c5_best - 1e-14:
                c5_best = c5_try
                h_best = h_try
            else:
                # Try LP with larger trust radius
                d2 = lp_direction(h_best, trust_radius=0.1)
                if d2 is not None:
                    d2 -= np.mean(d2)
                    n2 = np.linalg.norm(d2)
                    if n2 > 1e-12:
                        d2 = d2 / n2 * 0.1
                        h_try2, c5_try2 = line_search(h_best, d2, c5_best, n_coarse=30, n_fine=30, max_span=0.5)
                        if c5_try2 < c5_best - 1e-14:
                            c5_best = c5_try2
                            h_best = h_try2

    # Phase 4: Random feasible directions for diversity
    if time_left() > 30:
        n_cur = len(h_best)
        for iteration in range(200):
            if time_left() < 15:
                break
            for _ in range(5):
                d = random_feasible_direction(n_cur, rng)
                d_norm = np.linalg.norm(d)
                if d_norm < 1e-10:
                    continue
                d = d / d_norm * 0.05
                h_try, c5_try = line_search(h_best, d, c5_best, n_coarse=20, n_fine=20, max_span=0.2)
                if c5_try < c5_best - 1e-14:
                    c5_best = c5_try
                    h_best = h_try
                    break

    # Phase 5: Try different n values
    if time_left() > 60:
        for n_try in [30, 40, 50, 60, 80, 100]:
            if time_left() < 40:
                break
            # Interpolate h_best to n_try
            if n_try == len(h_best):
                h_new = h_best.copy()
            else:
                x_old = np.linspace(0, 2, len(h_best))
                x_new = np.linspace(0, 2, n_try)
                h_new = np.interp(x_new, x_old, h_best)
                h_new = project_box_mass(h_new)
            c5_new = compute_c5(h_new)
            # Optimize h_new
            for beta in [10.0, 30.0]:
                if time_left() < 20:
                    break
                h_opt = optimize_smoothed(h_new, beta, max_iter=300)
                c5_opt = compute_c5(h_opt)
                if c5_opt < c5_new:
                    h_new = h_opt
                    c5_new = c5_opt
            # Subgradient refinement
            for _ in range(50):
                if time_left() < 10:
                    break
                d = subgradient_direction(h_new, eps=1e-5)
                if np.linalg.norm(d) < 1e-14:
                    break
                d = d / np.linalg.norm(d) * 0.05
                h_try2, c5_try2 = line_search(h_new, d, c5_new, n_coarse=20, n_fine=20, max_span=0.3)
                if c5_try2 < c5_new - 1e-14:
                    c5_new = c5_try2
                    h_new = h_try2
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
                n_best = n_try

    # Phase 6: Final fine-tuning with very small steps
    if time_left() > 10:
        n_fin = len(h_best)
        for _ in range(500):
            if time_left() < 5:
                break
            improved = False
            for _ in range(10):
                d = rng.standard_normal(n_fin)
                d -= np.mean(d)
                d = d / (np.linalg.norm(d) + 1e-15) * 0.02
                h_try, c5_try = line_search(h_best, d, c5_best, n_coarse=15, n_fine=15, max_span=0.1)
                if c5_try < c5_best - 1e-14:
                    c5_best = c5_try
                    h_best = h_try
                    improved = True
                    break
            # Also try subgradient
            d = subgradient_direction(h_best, eps=1e-6)
            if np.linalg.norm(d) > 1e-14:
                d = d / np.linalg.norm(d) * 0.02
                h_try, c5_try = line_search(h_best, d, c5_best, n_coarse=15, n_fine=15, max_span=0.1)
                if c5_try < c5_best - 1e-14:
                    c5_best = c5_try
                    h_best = h_try
                    improved = True
            if not improved:
                continue

    # Final verification
    h_best = project_box_mass(h_best)
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    return h_best, c5_best, len(h_best)
