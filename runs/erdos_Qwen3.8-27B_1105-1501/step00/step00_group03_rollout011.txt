def project_box_sum(y, target_sum):
    """Project y onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(y)
    lo, hi = -1.0, 2.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def compute_correlations(h):
    """Compute np.correlate(h, 1-h, 'full') values"""
    return np.correlate(h, 1.0 - h, mode="full")

def c5_score(h):
    """Compute C5 = max_k correlate(h,1-h,'full') * dx"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * 2.0 / n)

def gradient_corr(h, k_offset, n):
    """
    Gradient of c[k] with respect to h, where k_offset is the full-correlate index.
    c[r] corresponds to lag k = r - (n-1)
    ∂c[k]/∂h[j] = (1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)
    """
    k = k_offset - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[j-k]) for valid j-k
    # j-k valid means 0 <= j-k < n, i.e., k <= j < k+n
    j_lo = max(0, k)
    j_hi = min(n, k + n)
    if j_lo < j_hi:
        idx = np.arange(j_lo, j_hi)
        grad[idx] += (1.0 - h[idx - k])
    # Term 2: -h[j+k] for valid j+k
    # j+k valid means 0 <= j+k < n, i.e., -k <= j < n-k
    j_lo2 = max(0, -k)
    j_hi2 = min(n, n - k)
    if j_lo2 < j_hi2:

def logsumexp_gradient(h, alpha):
    """
    Compute gradient of (1/alpha)*log(sum_k exp(alpha*f_k)) where f_k = c[k]*2/n
    Returns (value, gradient)
    """
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    f = c * dx  # shape (2n-1)
    
    # Numerical stability
    f_max = np.max(f)
    exp_f = np.exp(alpha * (f - f_max))
    sum_exp = np.sum(exp_f)
    p = exp_f / sum_exp  # softmax weights, shape (2n-1)
    
    value = f_max + np.log(sum_exp) / alpha
    
    # Gradient: sum_k p[k] * df_k/dh
    # df_k/dh[j] = dx * [(1-h[j-k])*I - h[j+k]*I] where k = r-(n-1)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if p[r] < 1e-15:
            continue
        k = r - (n - 1)
        j_lo = max(0, k)
        j_hi = min(n, k + n)
        if j_lo < j_hi:
            idx = np.arange(j_lo, j_hi)
            grad[idx] += p[r] * dx * (1.0 - h[idx - k])
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            idx = np.arange(j_lo2, j_hi2)
            grad[idx] += p[r] * dx * (-h[idx + k])
    
    return value, grad

def active_set_step(h, top_k=5, target_decrease=0.01):
    """
    Active-set least-squares step.
    Find direction d with 1^T d = 0 that reduces top correlations.
    """
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    # Get top k correlation indices
    top_indices = np.argsort(c)[-top_k:]
    f_max = np.max(c)
    target = f_max * (1.0 - target_decrease)
    
    # Build gradient matrix A (top_k x n)
    A = np.zeros((top_k, n))
    for i, r in enumerate(top_indices):
        A[i] = gradient_corr(h, r, n)
    
    # Target: A @ d = -(c[top_indices] - target)  (we want to decrease by this amount)
    # But this is overdetermined, so solve least squares
    # Also need 1^T d = 0
    # Combine: [A; 1^T] @ d = [-(c[top]-target); 0]
    A_aug = np.vstack([A, np.ones(1, n)])
    b = np.concatenate([-(c[top_indices] - target), [0.0]])
    
    # Solve least squares
    d, _, _, _ = np.linalg.lstsq(A_aug, b, rcond=None)

    d_norm = np.linalg.norm(d)
    if d_norm < 1e-10:
        return None
    d = d / d_norm * 0.1  # scale
    
    return d

        starts.append(gen_rand(n, rng))
        starts.append(gen_rand(n, rng))

        for h0 in starts:
            if time.time() - t0 > budget_s - 40:
                break
            tl1 = min(time.time() + per_n * 0.4, t0 + budget_s - 40)
            h1, c5_1 = optimize_pgd(h0, ts, tl1)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()

            if time.time() - t0 > budget_s - 40:
                break
            tl2 = min(time.time() + per_n * 0.3, t0 + budget_s - 40)
            h2, c5_2 = optimize_active(h1, ts, tl2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

            if time.time() - t0 > budget_s - 40:
                break
            tl3 = min(time.time() + per_n * 0.3, t0 + budget_s - 40)
            h3, c5_3 = optimize_scipy(h2, ts, alpha=60, maxiter=80, t_limit=tl3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

    # Final polish
    if best_h is not None and time.time() - t0 < budget_s - 20:
        nb = len(best_h)
        tsb = nb / 2.0
        # Fine random perturbations
        tl = time.time() + min(30, budget_s - time.time() - 10)
        for _ in range(2000):
            if time.time() > tl:
                break
            delta = rng.randn(nb) * 0.003
            h_try = project_box_sum(best_h + delta, tsb)
            c5_try = c5_score(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

        # Active set polish
        if time.time() - t0 < budget_s - 15:
            tl2 = min(time.time() + 15, t0 + budget_s - 10)
            h_pol, c5_pol = optimize_active(best_h, tsb, tl2)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

        # Scipy with high alpha
        if time.time() - t0 < budget_s - 10:
            h_fin, c5_fin = optimize_scipy(best_h, tsb, alpha=200, maxiter=200,
                                           t_limit=t0 + budget_s - 5)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin

    return (best_h, best_c5, len(best_h))
