```python
import numpy as np
import time

def project_onto_feasible(v, target_sum):
    """Project v onto {h: sum(h) = target_sum, 0 <= h[i] <= 1}."""
    n = len(v)
    # Bisection on t: h = clip(v - t, 0, 1), find t such that sum(h) = target_sum
    lo = v.min() - 1  # will make all h[i] = 1
    hi = v.max()      # will make all h[i] = 0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def subgradient_at_max(h):
    """Compute subgradient of max_k c_k at h, where c = overlaps."""
    n = len(h)
    c = compute_overlaps(h)
    dx = 2.0 / n
    k_star = np.argmax(c)
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    if lag >= 0:
        # c_k = sum_{i=0}^{n-1-lag} h[i] * (1 - h[i+lag])
        for m in range(n):
            g = 0.0
            if 0 <= m <= n - 1 - lag:
                g += (1 - h[m + lag])
            if lag <= m <= n - 1:
                g -= h[m - lag]
            grad[m] = g
    else:
        k_abs = -lag
        for m in range(n):
            g = 0.0
            if k_abs <= m <= n - 1:
                g += (1 - h[m - k_abs])
            if 0 <= m <= n - 1 - k_abs:
                g -= h[m + k_abs]
            grad[m] = g
    
    return grad * dx, c[k_star]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(47, 0.5)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_feasible(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def overlaps(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def c5_value(h):
    return float(np.max(overlaps(h)))

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    M = c.max()
    e = np.exp(tau * (c - M))
    s = e.sum()
    w = e / s
    f = M + np.log(s) / tau
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += wk * dx * (1.0 - h[lag:])
            grad[lag:] -= wk * dx * h[:n-lag]
        else:
            ka = -lag
            grad[ka:] += wk * dx * (1.0 - h[:n-ka])
            grad[:n-ka] -= wk * dx * h[ka:]
    return f, grad

def subgrad_at_max(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    k = int(np.argmax(c))
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] = 1.0 - h[lag:]
        g[lag:] -= h[:n-lag]
    else:
        ka = -lag
        g[ka:] = 1.0 - h[:n-ka]
        g[:n-ka] -= h[ka:]
    return g * dx

def smoothed_subgrad(h, top_k=5):
    """Subgradient using average of top-k overlaps."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # Average of top-k
    top_indices = np.argsort(c)[-top_k:]
    w = 1.0 / top_k
    g = np.zeros(n)
    for k in top_indices:
        lag = k - (n - 1)
        kg = np.zeros(n)
        if lag == 0:
            kg = 1.0 - 2.0 * h
        elif lag > 0:
            kg[:n-lag] = 1.0 - h[lag:]
            kg[lag:] -= h[:n-lag]
        else:
            ka = -lag
            kg[ka:] = 1.0 - h[:n-ka]
            kg[:n-ka] -= h[ka:]
        g += w * kg
    return g * dx
```
