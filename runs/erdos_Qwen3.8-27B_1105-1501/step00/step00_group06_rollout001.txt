def evaluate_c5(h, n):

    dx = 2.0 / n

def project_box_mass(v, n):
    """Project v onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    v = np.array(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    
    # Bisection for shift
    def total_shift(mu):
        return float(np.sum(np.clip(v + mu, 0.0, 1.0)))
    
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if total_shift(mid) < target:
            lo = mid
        else:
            hi = mid
    
    mu = 0.5 * (lo + hi)
    y = np.clip(v + mu, 0.0, 1.0)
    
    # Fine residual correction
    res = target - float(np.sum(y))
    if abs(res) > 1e-10:
        free = np.where((y > 1e-8) & (y < 1.0 - 1e-8))[0]
        if len(free) > 0:
            adj = res / len(free)
            y[free] = np.clip(y[free] + adj, 0.0, 1.0)
    
    y = np.clip(y, 0.0, 1.0)
    return y

import numpy as np

def evaluate_c5(h, n):
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r)) * dx

def project_box_mass(v, n):
    target = n / 2.0
    v = np.array(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v + mid, 0.0, 1.0))) < target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    y = np.clip(v + mu, 0.0, 1.0)
    res = target - float(np.sum(y))
    if abs(res) > 1e-10:
        free = np.where((y > 1e-8) & (y < 1.0 - 1e-8))[0]
        if len(free) > 0:
            y[free] = np.clip(y[free] + res / len(free), 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def pair_sweep(h, n, rng, n_samples=25):
    """One full pass of pair transfers. Returns (h, val, improved)."""
    val = evaluate_c5(h, n)
    improved = False
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    rng.shuffle(pairs)
    for i, j in pairs:
        # Direction j -> i
        ms1 = min(1.0 - h[i], h[j])
        if ms1 > 1e-10:
            alphas = np.linspace(ms1 * 0.05, ms1, n_samples)
            for a in alphas:
                h[i] += a
                h[j] -= a
                v = evaluate_c5(h, n)
                if v < val - 1e-14:
                    val = v
                    improved = True
                else:
                    h[i] -= a
                    h[j] += a
                    break
            else:
                pass  # all improved, keep last

                h[i] = min(1.0, h[i])

def gradient_direction(h, n, beta):
    """Compute smoothed gradient of LSE objective, return descent direction."""
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * r * dx
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    weights = exps / exps.sum()
    
    # Numerical gradient
    eps = 1e-5
    grad = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, hp[i] + eps)
        hp = project_box_mass(hp, n)
        rm = h.copy()
        rm[i] = max(0.0, rm[i] - eps)
        rm = project_box_mass(rm, n)
        rp = np.correlate(hp, 1.0 - hp, mode="full")
        rr = np.correlate(rm, 1.0 - rm, mode="full")
        sp = beta * rp * dx
        sr = beta * rr * dx
        sp -= np.max(sp)
        sr -= np.max(sr)
        LSE_p = np.log(np.exp(sp).sum()) / beta
        LSE_r = np.log(np.exp(sr).sum()) / beta
        grad[i] = (LSE_p - LSE_r) / (2.0 * eps)
    
    # Tangent to mass constraint
    grad -= np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        return None
    return -grad / gnorm
