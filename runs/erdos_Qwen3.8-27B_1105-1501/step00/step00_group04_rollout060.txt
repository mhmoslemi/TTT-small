def score_exact(h):
    """Compute C5 = max(np.correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def feasible_project(y, n):
    """Project y onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = np.min(y) - 1.0
    hi = np.max(y) + 0.0
    
    def g(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    
    # g is non-increasing in lambda
    # g(lo) >= n, g(hi) <= 0 (approximately)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if g(mid) > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    z = np.clip(y - lam, 0.0, 1.0)
    
    # Fix any residual
    diff = target - np.sum(z)
    if abs(diff) > 1e-10:
        # Adjust the component with most slack
        if diff > 0:
            # Need to increase some elements
            slack = 1.0 - z
            idx = np.argmax(slack)
            z[idx] = min(1.0, z[idx] + diff)
        else:
            # Need to decrease some elements
            slack = z
            idx = np.argmax(slack)
            z[idx] = max(0.0, z[idx] + diff)
    
    return z

def smooth_scores_and_grad(h, T):
    """Compute smooth max score and its gradient."""
    n = len(h)

    L = len(c_full)  # 2n-1

    c_max = np.max(c_full)
    if c_max < 0:
        c_max = 0.0  # Actually scores should be non-negative
    exponents = (c_full - c_max) / T
    # Clip to avoid overflow (shouldn't happen since c_full - c_max <= 0)

                # dc/dh[i] for i in [0, n-1]:

                # For i in [0, n-1-d]: -h[i+d]

                    grad[d:n] += w * (1.0 - h[0:n-d])

            lo = mid
        else:
            hi = mid

    diff = target - z.sum()
    if abs(diff) > 1e-12:
        if diff > 0:

            z[idx] = min(1.0, z[idx] + diff)
        else:

def smooth_grad(h, T):

    dx = 2.0 / n
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    L = 2 * n - 1

    weights = np.exp(exponents)
    S = float(weights.sum())
    if S < 1e-300:
        idx = int(np.argmax(c_full))
        weights = np.zeros(L)
        weights[idx] = 1.0
    else:
        weights /= S
    grad = np.zeros(n)

            grad += w * (1.0 - 2.0 * h)

def resample_vec(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return feasible_project(h.copy(), new_n)
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    y = np.interp(x_new, x_old, h)
    return feasible_project(y, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.default_rng(seed)

    try:
        h_p = np.array(initial_h_values, dtype=np.float64).flatten()
    except Exception:
        h_p = None
    if h_p is None or len(h_p) < 4 or not np.all(np.isfinite(h_p)):
        h_p = np.full(60, 0.5)
    n = len(h_p)
    h_p = feasible_project(h_p, n)
    best_h = h_p.copy()
    best_c = score_exact(h_p)

    # Phase 1: Main gradient refinement
    t1 = min(deadline - 150, t0 + 0.42 * budget_s)
    best_h, best_c = local_optimize(best_h, t1, rng, T_start=0.12, T_end=0.0004)

    # Phase 2: Second refinement round with lower start T
    t2 = min(deadline - 130, t0 + 0.58 * budget_s)
    best_h, best_c = local_optimize(best_h, t2, rng, T_start=0.06, T_end=0.0002)

    # Phase 3: Gradient-guided pair search
    t3 = min(deadline - 110, t0 + 0.70 * budget_s)
    best_h, best_c = gradient_pair_search(best_h, t3, n_pairs=120, n_evals=16)

    # Phase 4: Random pair search
    t4 = min(deadline - 90, t0 + 0.80 * budget_s)
    n_b = len(best_h)
    for _ in range(400):
        if time.time() > t4:
            break
        i = int(rng.integers(0, n_b))
        j = int(rng.integers(0, n_b))
        if i == j:
            continue
        h_t = best_h.copy()
        h_t, c_t = pair_search(h_t, i, j, 14)
        if c_t < best_c - 1e-12:
            best_h = h_t
            best_c = c_t

    # Phase 5: Mutated restarts
    t5 = min(deadline - 70, t0 + 0.88 * budget_s)
    for sigma in [0.06, 0.03, 0.015, 0.008]:
        if time.time() > t5:
            break
        h_m = feasible_project(best_h + rng.normal(0, sigma, len(best_h)), len(best_h))
        c_m = score_exact(h_m)
        if c_m < best_c + 0.015:
            t_loc = min(deadline - 45, time.time() + 28)
            h_r, c_r = local_optimize(h_m, t_loc, rng, T_start=0.05, T_end=0.0002)
            if c_r < best_c:
                best_h = h_r
                best_c = c_r
                # Graduated pair polish
                best_h, best_c = gradient_pair_search(best_h, min(deadline - 40, time.time() + 15), n_pairs=30, n_evals=14)

    # Phase 6: Binarize and refine
    if time.time() < deadline - 75:
        h_b = binarize_vec(best_h)
        c_b = score_exact(h_b)
        if c_b < best_c + 0.02:
            t_loc = min(deadline - 45, time.time() + 30)
            h_r, c_r = local_optimize(h_b, t_loc, rng, T_start=0.05, T_end=0.0002)
            if c_r < best_c:
                best_h = h_r
                best_c = c_r

    # Phase 7: Resolution exploration
    n_cur = len(best_h)
    t7_end = min(deadline - 50, t0 + 0.94 * budget_s)
    for dn in [-10, 10, -5, 5, 15, -15, 20, -20]:
        nn = n_cur + dn
        if nn < 24 or nn > 200:
            continue
        if time.time() > t7_end:
            break
        h_r = resample_vec(best_h, nn)
        c_r = score_exact(h_r)
        if c_r < best_c + 0.008:
            t_loc = min(deadline - 40, time.time() + 22)
            h_rf, c_rf = local_optimize(h_r, t_loc, rng, T_start=0.07, T_end=0.0003)
            if c_rf < best_c:
                best_h = h_rf
                best_c = c_rf
                n_cur = len(best_h)

    # Phase 8: Final polishing
    n_pf = len(best_h)
    for _ in range(300):
        if time.time() > deadline - 2.0:
            break
        i = int(rng.integers(0, n_pf))
        j = int(rng.integers(0, n_pf))
        if i == j:
            continue
        h_t = best_h.copy()
        h_t, c_t = pair_search(h_t, i, j, 12)
        if c_t < best_c - 1e-12:
            best_h = h_t
            best_c = c_t

    # Final verification
    n_final = len(best_h)
    best_h = feasible_project(best_h, n_final)
    final_c = score_exact(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, best_c, n_final
