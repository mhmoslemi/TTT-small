        n = len(y)

    s = np.sum(y)
    if abs(s - target) < 1e-12:

            # gradient w.r.t. h[i] is -h[i+d] for i = 0..n-1-d

                g[i] -= h[i + d]

def frank_wolfe_step(h, g):
    """Frank-Wolfe: move towards extreme point minimizing g^T z"""
    n = len(h)
    k = n // 2
    # z = 1 where g is smallest (k of them)
    idx = np.argsort(g)
    z = np.zeros(n)
    z[idx[:k]] = 1.0
    
    # Line search: h_t = (1-t)*h + t*z
    best_h = h.copy()
    best_score = exact_score(h)[0]
    
    for t in np.linspace(0, 1, 21):
        candidate = (1 - t) * h + t * z
        s = exact_score(candidate)[0]
        if s < best_score - 1e-14:
            best_score = s
            best_h = candidate
    
    return best_h, best_score

        for i in range(n - d):
            z = cp.Variable()
            z_vars[(d, i)] = z
            constraints.append(z <= x[i])
            constraints.append(z <= x[i + d])
            constraints.append(z >= x[i] + x[i + d] - 1)
            constraints.append(z >= 0)
    
    # Lag constraints: for each d, both directions must be <= target
    for d in range(1, n):
        # Positive lag: sum h[i+d]*(1-h[i]) = sum(x[i+d] - z[d,i])
        pos_sum = cp.sum([x[i + d] - z_vars[(d, i)] for i in range(n - d)])
        neg_sum = cp.sum([x[i] - z_vars[(d, i)] for i in range(n - d)])
        constraints.append(pos_sum <= target_count)
        constraints.append(neg_sum <= target_count)
    
    # Dummy objective (feasibility problem)
    obj = cp.Minimize(0 * x[0])
    prob = cp.Problem(obj, constraints)
    
    solver_list = ['CBC', 'GLPK', 'SCIP', 'ECOS', 'CVXOPT']
    solved = False
    for solver in solver_list:
        try:
            prob.solve(solver=solver, timeout=30)
            if prob.status in ['optimal', 'optimal_inaccurate', 'feasible']:
                if x.value is not None:
                    sol = np.round(x.value)
                    sol = repair_feasible(sol, n)
                    # Binarize to ensure exact
                    idx = np.argsort(sol)
                    bx = np.zeros(n)
                    bx[idx[n // 2:]] = 1.0
                    s = exact_score(bx)
                    if s <= 0.38085:
                        return bx, s
            solved = True
            break
        except:
            continue
    
    return None, None

def random_binary_search(n, n_trials=100, deadline=None, seed=0):
    rng = np.random.RandomState(seed)
    best_x = None
    best_s = 1.0
    k = n // 2
    
    for t in range(n_trials):
        if deadline and time.time() > deadline:
            break
        idx = rng.choice(n, k, replace=False)
        x = np.zeros(n)
        x[idx] = 1.0
        s = exact_score(x)
        if s < best_s:
            best_s = s
            best_x = x.copy()
    
    if best_x is not None:
        # Quick local improvement
        best_x, best_s = greedy_binary_smart(best_x, max_passes=50, deadline=deadline, seed=seed + 100)
    
    return best_x, best_s

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    def remaining():
        return budget_s - elapsed()
    
    # Get initial values
    h_init = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h_init = np.array(initial_h_values, dtype=float)
            if not np.all(np.isfinite(h_init)) or len(h_init) == 0:
                h_init = None
    except:
        h_init = None
    
    n_init = len(h_init) if h_init is not None else 70
    
    if h_init is None:
        h_init = np.full(70, 0.5)
        n_init = 70
    
    # Ensure initial is feasible
    h_init = repair_feasible(h_init, n_init)
    
    # Track best
    best_h = h_init.copy()
    best_c5 = exact_score(best_h)
    best_n = n_init
    
    def update(h, s):
        nonlocal best_h, best_c5, best_n
        s = exact_score(h)
        n = len(h)
        if s < best_c5 - 1e-13 or (abs(s - best_c5) < 1e-13 and n < best_n):
            best_h = h.copy()
            best_c5 = s
            best_n = n
    
    # Phase 1: Quick evaluations (always)
    s_init = exact_score(h_init)
    update(h_init, s_init)
    
    x_bin = binarize(h_init)
    update(x_bin, x_bin)
    
    # Scaled variants
    for beta in [0.7, 1.3, 1.8]:
        h_scaled = repair_feasible(0.5 + beta * (h_init - 0.5), n_init)
        update(h_scaled, h_scaled)
    
    # Phase 2: Binary greedy on initial
    if remaining() > 15:
        deadline = t0 + min(remaining() * 0.25, 120)
        x_opt, s = greedy_binary_smart(x_bin, max_passes=500, deadline=deadline, seed=seed)
        update(x_opt, x_opt)
    
    # Phase 3: Continuous optimization on initial
    if remaining() > 15:
        deadline = t0 + min(remaining() * 0.25, 120)
        h_opt, s = continuous_active_set(h_init, max_iters=1000, deadline=deadline, seed=seed)
        update(h_opt, h_opt)
    
    # Phase 4: Continuous on binarized
    if remaining() > 15:
        h_from_bin = repair_feasible(x_bin * 1.0 + rng_noise(x_bin, 0.1, seed), n_init)
        deadline = t0 + min(remaining() * 0.15, 60)
        h_opt, s = continuous_active_set(h_from_bin, max_iters=500, deadline=deadline, seed=seed + 50)
        update(h_opt, h_opt)
    
    # Phase 5: Try different sizes
    if remaining() > 30:
        base_h = best_h
        sizes = sorted(set([s for s in range(32, min(120, n_init + 20), 2) if s % 2 == 0 and s != best_n]))
        sizes.sort(key=lambda s: (abs(s - best_n) - s * 0.3))
        
        tried = 0
        for sz in sizes:
            if remaining() < 20 or tried >= 20:
                break
            tried += 1
            
            h_sz = transfer_to_len(base_h, sz)
            update(h_sz, h_sz)
            
            x_sz = binarize(h_sz)
            update(x_sz, x_sz)
            
            if remaining() > 25:
                deadline_min = t0 + min(remaining() * 0.15, 45)
                x_opt, s = greedy_binary_smart(x_sz, max_passes=200, deadline=deadline_min, seed=seed + sz)
                update(x_opt, x_opt)
            
            if remaining() > 25:
                deadline_min = t0 + min(remaining() * 0.15, 45)
                h_opt, s = continuous_active_set(h_sz, max_iters=500, deadline=deadline_min, seed=seed + sz + 1)
                update(h_opt, h_opt)
    
    # Phase 6: MILP for small sizes (bonus)
    if remaining() > 60:
        for sz in [40, 44, 48, 52, 56]:
            if remaining() < 40:
                break
            x_sol, s_sol = milp_attempt(sz, t0 + remaining() - 10)
            if x_sol is not None:
                update(x_sol, x_sol)
    
    # Phase 7: Random restarts with local improvement
    if remaining() > 15:
        rng = np.random.RandomState(seed + 999)
        n_use = best_n
        for r in range(30):
            if remaining() < 8:
                break
            k = n_use // 2
            idx = rng.choice(n_use, k, replace=False)
            x_r = np.zeros(n_use)
            x_r[idx] = 1.0
            
            # Quick greedy
            deadline_r = t0 + min(remaining() * 0.1, 10)
            x_opt, s = greedy_binary_smart(x_r, max_passes=30, deadline=deadline_r, seed=seed + r + 200)
            update(x_opt, x_opt)
            
            # Continuous from random
            if remaining() > 15:
                h_r = repair_feasible(x_r + rng.randn(n_use) * 0.1, n_use)
                deadline_r = t0 + min(remaining() * 0.1, 10)
                h_opt, s = continuous_active_set(h_r, max_iters=200, deadline=deadline_r, seed=seed + r + 300)
                update(h_opt, h_opt)
    
    # Phase 8: Final intensive optimization on best
    if remaining() > 20:
        deadline_final = t0 + min(remaining() - 3, 120)
        h_opt, s = continuous_active_set(best_h, max_iters=2000, deadline=deadline_final, seed=seed + 500)
        update(h_opt, h_opt)
    
    # Final: ensure feasibility
    best_h = repair_feasible(best_h, best_n)
    best_c5 = exact_score(best_h)
    
    # Safety: if anything went wrong, return a valid solution
    if not np.all(np.isfinite(best_h)) or len(best_h) == 0:
        best_h = np.full(best_n if best_n > 0 else 70, 0.5)
        best_n = len(best_h)
        best_c5 = exact_score(best_h)
    
    return best_h, best_c5, best_n

def rng_noise(x, scale, seed):
    rng = np.random.RandomState(seed)
    return rng.randn(len(x)) * scale
