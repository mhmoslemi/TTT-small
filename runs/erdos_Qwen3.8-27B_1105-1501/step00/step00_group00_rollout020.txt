Let me think about the gradient formulas:

For the correlation: `np.correlate(h, 1-h, mode="full")` gives a vector of length 2n-1.

For lag `ell = m - (n-1)` where m is the index in the full correlation output:
- If ell >= 0: C_ell = sum_{i=0}^{n-ell-1} h[i+ell] * (1-h[i])
- If ell < 0, let p = -ell: C_{-p} = sum_{i=0}^{n-p-1} h[i] * (1-h[i+p])

The gradient of C_ell with respect to h[j]:
- For ell >= 0:
  - If j >= ell: h[j] appears as h[i+ell] with i = j-ell, contributing (1-h[j-ell])
  - If j < n-ell: h[j] appears in (1-h[i]) with i=j, contributing -h[j+ell]
  - So: grad[j] = dx * ((1-h[j-ell]) if j>=ell else 0) + dx * (-h[j+ell] if j < n-ell else 0)

- For ell < 0, p = -ell:
  - If j < n-p: h[j] appears as h[i] with i=j, contributing (1-h[j+p])
  - If j >= p: h[j] appears in (1-h[i+p]) with i+p=j, i=j-p, contributing -h[j-p]
  - So: grad[j] = dx * ((1-h[j+p]) if j < n-p else 0) + dx * (-h[j-p] if j >= p else 0)

import numpy as np

def evaluate_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_h(h, target):
    """Project h to feasible set: 0 <= h_i <= 1, sum(h) = target"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # Need to reduce
        excess = s - target
        # Reduce from largest values
        order = np.argsort(-h)
        remaining = excess
        for idx in order:
            if remaining <= 0:
                break
            reduction = min(h[idx], remaining)
            h[idx] -= reduction
            remaining -= reduction
    else:
        # Need to increase
        deficit = target - s
        order = np.argsort(h)
        remaining = deficit
        for idx in order:
            if remaining <= 0:
                break
            increase = min(1.0 - h[idx], remaining)
            h[idx] += increase
            remaining -= increase
    # Final exact adjustment
    s = np.sum(h)
    if abs(s - target) > 1e-14:
        diff = target - s
        # Find a coordinate with room
        if diff > 0:
            for i in range(len(h)):
                if h[i] < 1.0 - 1e-15:
                    h[i] += diff
                    break
        else:
            for i in range(len(h)):
                if h[i] > 1e-15:
                    h[i] += diff
                    break
    return h

def compute_gradients(h):
    """Compute gradients of all correlation components."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        ell = m - (n-1)
        if ell >= 0:
            # C = sum_{i=0}^{n-ell-1} h[i+ell] * (1-h[i])
            if ell < n:
                grads[m, ell:] += dx * (1 - h[:n-ell])
                grads[m, :n-ell] -= dx * h[ell:]
        else:
            p = -ell
            if p < n:
                grads[m, :n-p] += dx * (1 - h[p:])
                grads[m, p:] -= dx * h[:n-p]
    return grads

def solve_lp(h0, active_indices, trust_radius):
    """Solve the tangent LP subproblem."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    # Compute correlation values and gradients at h0
    corr = np.correlate(h0, 1 - h0, mode='full')
    
    # Build LP
    # Variables: h[0..n-1], t
    # Minimize t
    # Constraints: G_m dot h + kappa_m <= t for active m
    #              t >= 0
    #              sum(h) = target
    #              bounds on h
    
    k = len(active_indices)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((k + 1, n + 1))
    b_ub = np.zeros(k + 1)
    
    for i, m in enumerate(active_indices):
        ell = m - (n-1)
        # Gradient of correlation at index m
        g = np.zeros(n)
        if ell >= 0:
            if ell < n:
                g[ell:] += dx * (1 - h0[:n-ell])
                g[:n-ell] -= dx * h0[ell:]
        else:
            p = -ell
            if p < n:
                g[:n-p] += dx * (1 - h0[p:])
                g[p:] -= dx * h0[:n-p]
        
        # L_m(h) = corr[m] + g dot (h - h0) = g dot h + (corr[m] - g dot h0)
        kappa = corr[m] - np.dot(g, h0)
        # Constraint: g dot h + kappa <= t => g dot h - t <= -kappa
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = -kappa
    
    # t >= 0 => -t <= 0
    A_ub[k, -1] = -1.0
    b_ub[k] = 0.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(max(0, h0[i] - trust_radius), min(1, h0[i] + trust_radius)) for i in range(n)]
    bounds.append((0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_candidate = res.x[:n]
        h_candidate = repair_h(h_candidate, target)
        return h_candidate
    return None

    """Transfer h from source resolution to target resolution, preserving area."""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_t = 2.0 / n_target
    
    h_t = np.zeros(n_target)
    src_pos = 0
    src_x = 0.0
    
    for k in range(n_target):
        t_left = k * dx_t
        t_right = (k + 1) * dx_t
        
        # Advance source pointer
        while src_pos < n_src and (src_pos + 1) * dx_src <= t_left:
            src_pos += 1
        
        area = 0.0
        s_start = src_pos
        s_x = s_start * dx_src
        
        while s_x < t_right and src_pos < n_src:
            s_end = (src_pos + 1) * dx_src
            overlap = min(s_end, t_right) - max(s_x, t_left)
            if overlap > 0:
                area += h_src[src_pos] * overlap
            s_x = s_end
            src_pos += 1
        
        h_t[k] = area / dx_t
    
    target = n_target / 2.0
    h_t = repair_h(h_t, target)
    return h_t

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair_h(h, target):
    h = np.asarray(h, dtype=np.float64).copy()
    if len(h) == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if s > target:
        excess = s - target
        order = np.argsort(-h)
        rem = excess
        for idx in order:
            if rem <= 1e-15:
                break
            cut = min(h[idx], rem)
            h[idx] -= cut
            rem -= cut
    elif s < target:
        deficit = target - s
        order = np.argsort(h)
        rem = deficit
        for idx in order:
            if rem <= 1e-15:
                break
            add = min(1.0 - h[idx], rem)
            h[idx] += add
            rem -= add
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14 and len(h) > 0:
        if diff > 0:
            for i in range(len(h)):
                if h[i] < 1.0:
                    h[i] += diff
                    break
        else:
            for i in range(len(h)):
                if h[i] > 0.0:
                    h[i] += diff
                    break
    return h

def get_gradient(h, m, n, dx):
    ell = m - (n - 1)
    g = np.zeros(n)
    if ell >= 0:
        if ell < n:
            g[ell:] += dx * (1.0 - h[:n-ell])
            g[:n-ell] -= dx * h[ell:]
    else:
        p = -ell
        if p < n:
            g[:n-p] += dx * (1.0 - h[p:])
            g[p:] -= dx * h[:n-p]
    return g

def solve_tangent_lp(h0, active, trust_r):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    k = len(active)
    if k == 0:
        return None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((k + 1, n + 1))
    b_ub = np.zeros(k + 1)
    for i, m in enumerate(active):
        g = get_gradient(h0, m, n, dx)
        b_val = float(np.dot(g, h0)) - corr[m]
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = b_val
    A_ub[k, -1] = -1.0
    b_ub[k] = 0.0
    A_eq = np.ones((1, n))
    b_eq = np.array([target])
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_r)
        hi = min(1.0, h0[i] + trust_r)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    except Exception:
        return None
    if res.success:
        h_new = repair_h(res.x[:n], target)
        return h_new
    return None

def pair_search(h, i, j, n_samples=15):

    best_val = evaluate_c5(h)
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    if d_max - d_min < 1e-10:
        return 0.0, best_val
    deltas = np.linspace(d_min, d_max, n_samples)
    for d in deltas:

        h[j] -= d
        v = evaluate_c5(h)
        h[i] -= d
        h[j] += d
        if v < best_val - 1e-12:

            best_delta = d
    if best_delta != 0.0:

        hi = min(d_max, best_delta + span)

            v = evaluate_c5(h)
            h[i] -= d
            h[j] += d
            if v < best_val - 1e-12:
                best_val = v
                best_delta = d
    return best_delta, best_val

def transfer_resolution(h_src, n_target):
    n_src = len(h_src)
    if n_target == n_src:
        return h_src.copy()
    dx_src = 2.0 / n_src
    dx_t = 2.0 / n_target
    h_t = np.zeros(n_target)
    src_pos = 0
    for k in range(n_target):
        t_left = k * dx_t
        t_right = (k + 1) * dx_t
        while src_pos < n_src and (src_pos + 1) * dx_src <= t_left + 1e-15:
            src_pos += 1
        area = 0.0
        sp = src_pos
        while sp < n_src:
            s_left = sp * dx_src
            s_right = (sp + 1) * dx_src
            if s_left >= t_right:
                break
            overlap = min(s_right, t_right) - max(s_left, t_left)
            if overlap > 0:
                area += h_src[sp] * overlap
            sp += 1
        h_t[k] = area / dx_t
    target = n_target / 2.0
    h_t = repair_h(h_t, target)
    return h_t

import numpy as np
from scipy.optimize import linprog
import time


def evaluate_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def repair_h(h, target):
    h = np.asarray(h, dtype=np.float64).copy()
    if len(h) == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if s > target:
        excess = s - target
        order = np.argsort(-h)
        rem = excess
        for idx in order:
            if rem <= 1e-15:
                break
            cut = min(h[idx], rem)
            h[idx] -= cut
            rem -= cut
    elif s < target:
        deficit = target - s
        order = np.argsort(h)
        rem = deficit
        for idx in order:
            if rem <= 1e-15:
                break
            add = min(1.0 - h[idx], rem)
            h[idx] += add
            rem -= add
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14 and len(h) > 0:
        if diff > 0:
            for i in range(len(h)):
                if h[i] < 1.0:
                    h[i] += diff
                    break
        else:
            for i in range(len(h)):
                if h[i] > 0.0:
                    h[i] += diff
                    break
    return h


def get_gradient(h, m, n, dx):
    ell = m - (n - 1)
    g = np.zeros(n)
    if ell >= 0:
        if ell < n:
            g[ell:] += dx * (1.0 - h[:n - ell])
            g[:n - ell] -= dx * h[ell:]
    else:
        p = -ell
        if p < n:
            g[:n - p] += dx * (1.0 - h[p:])
            g[p:] -= dx * h[:n - p]
    return g


def solve_tangent_lp(h0, active, trust_r):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    k = len(active)
    if k == 0:
        return None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((k + 1, n + 1))
    b_ub = np.zeros(k + 1)
    for i, m in enumerate(active):
        g = get_gradient(h0, m, n, dx)
        b_val = float(np.dot(g, h0)) - corr[m]
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = b_val
    A_ub[k, -1] = -1.0
    b_ub[k] = 0.0
    A_eq = np.ones((1, n))
    b_eq = np.array([target])
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_r)
        hi = min(1.0, h0[i] + trust_r)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.success:
        h_new = repair_h(res.x[:n], target)
        return h_new
    return None


def pair_search(h, i, j, n_samples=15):
    best_delta = 0.0
    best_val = evaluate_c5(h)
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    if d_max - d_min < 1e-10:
        return 0.0, best_val
    deltas = np.linspace(d_min, d_max, n_samples)
    for d in deltas:
        h[i] += d
        h[j] -= d
        v = evaluate_c5(h)
        h[i] -= d
        h[j] += d
        if v < best_val - 1e-12:
            best_val = v
            best_delta = d
    if best_delta != 0.0:
        span = (d_max - d_min) / max(n_samples, 2)
        lo = max(d_min, best_delta - span)
        hi = min(d_max, best_delta + span)
        for d in np.linspace(lo, hi, 11):
            h[i] += d
            h[j] -= d
            v = evaluate_c5(h)
            h[i] -= d
            h[j] += d
            if v < best_val - 1e-12:
                best_val = v
                best_delta = d
    return best_delta, best_val


def select_pairs(h, n_top=5, n_pairs=25):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:n_top]
    avg_g = np.zeros(n)
    for m in top_idx:
        avg_g += get_gradient(h, m, n, dx)
    avg_g /= float(len(top_idx))
    i_order = np.argsort(avg_g)
    j_order = np.argsort(-avg_g)
    pairs = []
    seen = set()
    for ii in range(min(n_pairs * 3, n)):
        i = int(i_order[ii])
        for jj in range(min(n_pairs * 3, n)):
            j = int(j_order[jj])
            if i == j:
                continue
            if h[i] >= 1.0 - 1e-10 or h[j] <= 1e-10:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            pairs.append(key)
            if len(pairs) >= n_pairs:
                return pairs
    return pairs


def random_pairs(h, n_pairs, rng):
    n = len(h)
    pairs = []
    seen = set()
    attempts = 0
    while len(pairs) < n_pairs and attempts < n_pairs * 10:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or h[i] >= 1.0 - 1e-10 or h[j] <= 1e-10:
            attempts += 1
            continue
        key = (i, j)
        if key in seen:
            attempts += 1
            continue
        seen.add(key)
        pairs.append(key)
        attempts += 1
    return pairs


def local_optimize(h, max_iters, time_limit, start_time, rng_seed=42):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = evaluate_c5(h)
    trust_r = 0.1
    stagnant = 0
    rng = np.random.default_rng(rng_seed)

    for it in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > time_limit - 2:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        center = n - 1
        n_active = min(15, 2 * n - 2)
        top_idx = np.argsort(-corr)[:n_active]
        active = list(top_idx)
        if center not in active:
            active.append(center)

        # Try LP
        h_new = solve_tangent_lp(h, active, trust_r)
        lp_improved = False
        if h_new is not None:
            c5_new = evaluate_c5(h_new)
            cur_c5 = evaluate_c5(h)
            if c5_new < best_c5 - 1e-10:
                h = h_new
                best_c5 = c5_new
                trust_r = min(0.4, trust_r * 1.3)
                stagnant = 0
                lp_improved = True
            elif c5_new < cur_c5 - 1e-8:
                h = h_new
                best_c5 = min(best_c5, c5_new)
                stagnant = max(0, stagnant - 1)
                lp_improved = True
            else:
                trust_r = max(0.01, trust_r * 0.5)
                stagnant += 2
        else:
            trust_r = max(0.01, trust_r * 0.5)
            stagnant += 2

        # Pair search (skip if LP just improved a lot)
        if not lp_improved or stagnant > 3:
            elapsed = time.time() - start_time
            if elapsed < time_limit - 2:
                pairs = select_pairs(h, n_top=5, n_pairs=20)
                if stagnant > 15:
                    pairs = random_pairs(h, 15, rng)
                improved = False
                for (i, j) in pairs:
                    elapsed = time.time() - start_time
                    if elapsed > time_limit - 2:
                        break
                    delta, val = pair_search(h, i, j, n_samples=13)
                    if val < best_c5 - 1e-12:
                        h[i] += delta
                        h[j] -= delta
                        best_c5 = val
                        improved = True
                if improved:
                    h = repair_h(h, target)
                    best_c5 = evaluate_c5(h)
                    stagnant = 0
                else:
                    stagnant += 1

        # Reheat if very stuck
        if stagnant > 40:
            trust_r = 0.2
            # Small random perturbation
            perturb = rng.normal(0, 0.02, size=n)
            h_pert = h + perturb
            h_pert = repair_h(h_pert, target)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5 + 0.01:
                h = h_pert
            stagnant = 20

        if trust_r < 0.005 and stagnant > 50:
            break

    return h, best_c5


def transfer_resolution(h_src, n_target):
    n_src = len(h_src)
    if n_target == n_src:
        return h_src.copy()
    if n_target < 2:
        return repair_h(h_src[:max(2, n_target)], n_target / 2.0)
    dx_src = 2.0 / n_src
    dx_t = 2.0 / n_target
    h_t = np.zeros(n_target)
    for k in range(n_target):
        t_left = k * dx_t
        t_right = (k + 1) * dx_t
        # Find overlapping source bins
        sp_from = int(t_left / dx_src)
        sp_to = min(n_src, int(t_right / dx_src) + 2)
        area = 0.0
        for sp in range(max(0, sp_from), sp_to):
            s_left = sp * dx_src
            s_right = (sp + 1) * dx_src
            overlap = min(s_right, t_right) - max(s_left, t_left)
            if overlap > 0:
                area += h_src[sp] * overlap
        h_t[k] = area / dx_t
    target = n_target / 2.0
    h_t = repair_h(h_t, target)
    return h_t


def make_initial_h(n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * 1.5 * x) + 0.15 * np.cos(2.0 * np.pi * 3.0 * x)
    h = np.clip(h, 0.0, 1.0)
    return repair_h(h, n / 2.0)
