    exp_c = np.exp((c - c_max) / tau)
    w = exp_c / exp_c.sum()

def project_feasible(h, target_sum):
    """Project h onto {x : sum(x) = target_sum, 0 <= x <= 1}."""
    # Binary search for lambda: sum(clip(h + lambda, 0, 1)) = target_sum
    lo = -np.max(h) - 1.0
    hi = -np.min(h) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def grad_smoothed_max(h, tau):
    """Gradient of tau * logsumexp(correlate(h, 1-h, 'full') * dx / tau)."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')

    c_shift = c - np.max(c)

        else:
            p = -m
