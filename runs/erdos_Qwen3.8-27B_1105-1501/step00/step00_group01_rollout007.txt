        h_new = project_onto_feasible(h - lr * g, target)
        cv = c5_value(h_new)
        if cv < best - 1e-12:
            best = cv
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 2000:
            break
    return best_h, best
