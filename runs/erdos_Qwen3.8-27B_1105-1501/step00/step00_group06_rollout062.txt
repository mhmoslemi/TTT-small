def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def project_feasible(h, target_sum):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to meet sum constraint
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Simple approach: add uniform offset and re-clip, iterate
    for _ in range(1000):
        if diff > 0:
            # Need to add mass; add to elements that can accept
            room = 1.0 - h
            total_room = room.sum()
            if total_room < diff:
                h = np.clip(h + diff / n, 0, 1)
            else:
                h = h + diff / n
                h = np.clip(h, 0, 1)
        else:
            # Need to remove mass
            total_mass = h.sum()
            if -diff >= total_mass:
                h = np.clip(h + diff / n, 0, 1)
            else:
                h = h + diff / n
                h = np.clip(h, 0, 1)
        new_diff = target_sum - h.sum()
        if abs(new_diff) < 1e-12:
            break
        # If uniform adjustment doesn't work, do it element by element
        if abs(new_diff) > abs(diff) * 0.999:
            # Water filling
            if new_diff > 0:
                order = np.argsort(1.0 - h)  # elements with most room first
            else:
                order = np.argsort(h)  # elements with most mass first
            remaining = new_diff
            for idx in order:
                if remaining > 0:
                    add = min(remaining, 1.0 - h[idx])
                    h[idx] += add
                    remaining -= add
                elif remaining < 0:
                    rem = min(-remaining, h[idx])
                    h[idx] -= rem
                    remaining += rem
                else:
                    break
        diff = new_diff
    return h

Key observation: for the binary case (h ∈ {0,1}), the overlap at shift k is:
c_k = #{i: h[i]=1, h[i+k]=0} (for shift k > 0, approximately)

And we want to minimize max_k c_k.

For a binary sequence with exactly n/2 ones, this is related to the autocorrelation of the sequence. Specifically:
c_k = (n/2) - A(k) where A(k) = #{i: h[i]=1, h[i+k]=1} is the autocorrelation.

So minimizing max_k c_k = minimizing (n/2) - min_k A(k) = maximizing min_k A(k).

We want a binary sequence where the autocorrelation (at all non-zero shifts) is as evenly distributed as possible, with the minimum being as high as possible.

This is related to "Golay sequences" or "Barker sequences" or more generally "low autocorrelation sequences". But we want HIGH minimum autocorrelation, which is the opposite of what those are optimized for.

Actually, we want a sequence where the ones are spread out as evenly as possible, so that at every shift, a similar fraction of the ones overlap with other ones.

For a perfectly uniform distribution (impossible for binary), A(k) = (n/2)²/n = n/4 for all k. So min A(k) = n/4, and max c_k = n/2 - n/4 = n/4.
C₅ = (n/4) * (2/n) = 1/2 = 0.5.

But the current best is 0.38, which is much less than 0.5. This suggests the continuous (fractional) values help significantly.

For the continuous case, if h(x) = 1/2 everywhere, then ∫h = 1 (since the interval is [0,2] and h=1/2). Wait, ∫₀² (1/2) dx = 1. Yes! And h(x)(1-h(x+k)) = (1/2)(1/2) = 1/4 for all x, k. So C₅ = ∫ (1/4) dx. But the integral is over the overlap region, which has length 2-|k|. So C₅ = max_k (1/4) * (2-|k|). For k=0, this is (1/4)*2 = 1/2. For k=±1, it's (1/4)*1 = 1/4. So C₅ = 1/2 = 0.5.

That's the same as the uniform binary case. The constant 0.5 is the "trivial" upper bound. To do better, we need a non-uniform h.

The current best of 0.38 is significantly below 0.5, so the optimal h is quite non-uniform.

What structure might give a low C₅? We want h(x)(1-h(x+k)) to be small for all k. This means: where h is high, h shifted by k should also be high (for all k). In other words, h should be "correlated with all its shifts" - which sounds like it should be concentrated in a small region. But if h is concentrated, then at k=0, h(x)(1-h(x)) is small (since h is close to 1 where it's nonzero), but at shifts comparable to the width of h, h(x) and h(x+k) don't overlap, giving h(x)(1-h(x+k)) ≈ h(x) which is large.

So there's a trade-off: concentrated h has low overlap at small shifts but high overlap at medium shifts. Spread-out h has moderate overlap at all shifts.

import time

def _c5(h):
    """Compute C5 score."""
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def _fix(h, S):
    """Repair to satisfy sum=h_target and 0<=h<=1."""
    h = np.clip(h, 0, 1)
    n = len(h)
    for _ in range(100):
        d = S - h.sum()
        if abs(d) < 1e-14:
            return h
        h += d / n
        np.clip(h, 0, 1, out=h)
    return h

    """Local search with pair transfers."""
    n = len(h)
    h = h.copy()

    t0 = time.time()

                        else:
                            h[i] += e; h[j] -= e

            break
