i=2, s=2-2=0: n from 0 to 2: 0 + (-2) + (-6) = -8 ✓
i=3, s=3-2=1: n from 0 to 1: h[1]*(1-h[0]) + h[2]*(1-h[1]) = 0 + (-3) = -3 ✓
i=4, s=4-2=2: n from 0 to 0: h[2]*(1-h[0]) = 0 ✓

So for output index i, shift s = i - (n-1), and:
- If s ≥ 0: sum over n=0 to n-1-s of h[n+s]*(1-h[n])
- If s < 0: let t = -s, sum over n=t to n-1 of h[n-t]*(1-h[n])

Now the gradient:
∂output[i]/∂h[j] where s = i-(n-1):

If s ≥ 0:
output[i] = ∑_{n=0}^{n-1-s} h[n+s]*(1-h[n])
∂/∂h[j]: 
- If j = n+s for some n in [0, n-1-s], i.e., j-s in [0, n-1-s], i.e., s ≤ j ≤ n-1: contributes (1-h[j-s])
- If j = n for some n in [0, n-1-s], i.e., 0 ≤ j ≤ n-1-s: contributes -h[j+s]

So ∂output[i]/∂h[j] = (1-h[j-s])*1_{s≤j≤n-1} - h[j+s]*1_{0≤j≤n-1-s}

If s < 0 (t = -s > 0):
output[i] = ∑_{n=t}^{n-1} h[n-t]*(1-h[n])
∂/∂h[j]:
- If j = n-t for some n in [t, n-1], i.e., j+t in [t, n-1], i.e., 0 ≤ j ≤ n-1-t: contributes (1-h[j+t])
- If j = n for some n in [t, n-1], i.e., t ≤ j ≤ n-1: contributes -h[j-t]

So ∂output[i]/∂h[j] = (1-h[j+t])*1_{0≤j≤n-1-t} - h[j-t]*1_{t≤j≤n-1}

And the final gradient is:
grad[j] = ∑_i w[i] * ∂output[i]/∂h[j] * dx

where w[i] = softmax(t * output[i]) (normalized).

```python
def softmax_obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = float(np.max(c))
    c_shifted = c - c_max
    exp_vals = np.exp(t * c_shifted)
    sum_exp = float(np.sum(exp_vals))
    obj = np.log(sum_exp) / t + c_max
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    for i in range(len(c)):
        wi = w[i]
        if wi < 1e-14:
            continue
        s = i - (n - 1)
        if s >= 0:
            # ∂c[i]/∂h[j] = (1-h[j-s]) for s<=j<=n-1, minus h[j+s] for 0<=j<=n-1-s
            # Total range of j: 0 to n-1
            for j in range(0, n):
                contrib = 0.0
                if s <= j:
                    contrib += (1.0 - h[j - s])
                if j <= n - 1 - s:
                    contrib -= h[j + s]
                grad[j] += wi * contrib * dx
        else:
            t_abs = -s
            for j in range(0, n):
                contrib = 0.0
                if j <= n - 1 - t_abs:
                    contrib += (1.0 - h[j + t_abs])
                if j >= t_abs:
                    contrib -= h[j - t_abs]
                grad[j] += wi * contrib * dx
    return obj, grad
```

import numpy as np
import time

def compute_c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def c5_max_value(h):
    return float(np.max(compute_c5_array(h)))

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = float(np.max(c))
    c_shifted = c - c_max
    exp_vals = np.exp(t * c_shifted)
    sum_exp = float(np.sum(exp_vals))
    obj = np.log(sum_exp) / t + c_max
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    for i in range(len(c)):
        wi = w[i]
        if wi < 1e-14:
            continue
        s = i - (n - 1)
        if s >= 0:
            if s < n:
                pos = np.zeros(n)
                neg = np.zeros(n)
                pos[s:] = 1.0 - h[:n - s]
                neg[:n - s] = h[s:]
                grad += wi * dx * (pos - neg)
        else:
            t_abs = -s
            if t_abs < n:
                pos = np.zeros(n)
                neg = np.zeros(n)
                pos[:n - t_abs] = 1.0 - h[t_abs:]
                neg[t_abs:] = h[:n - t_abs]
                grad += wi * dx * (pos - neg)
    return obj, grad

def pg_optimize(h_init, target_sum, t, max_iters, deadline):
    h_curr = h_init.copy().astype(float)
    alpha = 0.05
    best_local = c5_max_value(h_curr)
    best_local_h = h_curr.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        obj, grad = softmax_obj_and_grad(h_curr, t)
        f_curr = obj
        g_sq = float(np.dot(grad, grad))
        if g_sq < 1e-18:
            break
        a = alpha
        improved = False
        for _ in range(60):
            h_trial = project_feasible(h_curr - a * grad, target_sum)
            f_trial, _ = softmax_obj_and_grad(h_trial, t)
            if f_trial < f_curr - 1e-4 * a * g_sq:
                h_curr = h_trial
                improved = True
                break
            a *= 0.5
        if not improved:
            alpha *= 0.5
            if alpha < 1e-12:
                break
        else:
            alpha = min(alpha * 1.1, 0.5)
        curr_c5 = c5_max_value(h_curr)
        if curr_c5 < best_local:
            best_local = curr_c5
            best_local_h = h_curr.copy()
    return best_local_h, best_local

def coordinate_descent(h_init, target_sum, max_rounds, deadline):
    n = len(h_init)
    h_curr = h_init.copy().astype(float)
    best_c5 = c5_max_value(h_curr)
    best_h = h_curr.copy()
    deltas = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
    for rnd in range(max_rounds):

            break
