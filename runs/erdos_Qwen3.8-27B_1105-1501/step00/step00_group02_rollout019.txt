def line_search_pair(base, p, q, C_curr, n_coarse=32, n_fine=16):
    n = len(base)
    dx = 2.0 / n
    hp, hq = base[p], base[q]
    t_min = max(hp - 1.0, -hq)
    t_max = min(hp, 1.0 - hq)
    if t_max - t_min < 1e-10:
        return None, C_curr
    
    best_t = 0.0
    best_C = C_curr
    
    # Coarse scan
    ts = np.linspace(t_min, t_max, n_coarse)
    for t in ts:
        if abs(t) < 1e-12:
            continue
        y = base.copy()
        y[p] = hp - t
        y[q] = hq + t
        y = np.clip(y, 0.0, 1.0)
        C = float(np.max(np.correlate(y, 1.0 - y, mode="full") * dx))
        if C < best_C - 1e-12:
            best_C = C
            best_t = t
    
    if best_t == 0.0:
        return None, C_curr
    
    # Fine refinement
    span = (t_max - t_min) / n_coarse
    for _ in range(3):
        lo = max(t_min, best_t - span)
        hi = min(t_max, best_t + span)
        ts2 = np.linspace(lo, hi, n_fine)
        for t in ts2:
            y = base.copy()
            y[p] = hp - t
            y[q] = hq + t
            y = np.clip(y, 0.0, 1.0)
            C = float(np.max(np.correlate(y, 1.0 - y, mode="full") * dx))
            if C < best_C - 1e-12:
                best_C = C
                best_t = t
        span *= 0.3
    
    y = base.copy()
    y[p] = hp - best_t
    y[q] = hq + best_t
    y = np.clip(y, 0.0, 1.0)
    T = n / 2.0
    if abs(y.sum() - T) > 1e-10:
        y = project_onto_box_simplex(y, T)
    final_C = compute_C(y)
    return y, final_C

def local_pair_descent(h, C, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    T = n / 2.0
    improved = True
    passes = 0
    
    while improved and passes < max_passes and time.time() < deadline:
        improved = False
        passes += 1
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        F = corr * dx
        C = float(np.max(F))
        
        # Find active lags (top few)
        top_k = min(8, 2 * n - 1)
        top_lags = np.argsort(F)[-top_k:][::-1]
        
        # Generate candidate pairs from active lags
        pairs = set()
        for r in top_lags:
            d = int(r) - (n - 1)
            if d >= 0:
                first = range(0, n - d)
                second = range(d, n)
            else:
                first = range(0, n + d)
                second = range(-d, n)
            
            for p in first:
                if h[p] < 0.05:
                    continue
                for q in second:
                    if q == p or h[q] > 0.95:
                        continue
                    pairs.add((int(p), int(q)))
        
        # Add some random pairs

        for _ in range(n_rand):
            p = np.random.randint(n)
            q = np.random.randint(n)
            if p != q and h[p] > 0.05 and h[q] < 0.95:
                pairs.add((int(p), int(q)))
        
        # Try each pair
        pair_list = list(pairs)[:200]
        np.random.shuffle(pair_list)

            if time.time() > deadline:
                break
            if h[p] < 1e-6 or h[q] > 1.0 - 1e-6:

            if new_h is not None and new_C < C - 1e-10:

def simulated_annealing(h, C, deadline, T_init=0.002, T_min=1e-7, cooling=0.999):
    n = len(h)
    dx = 2.0 / n
    T = n / 2.0
    T_curr = T_init
    best_h = h.copy()
    best_C = C
    cur_h = h.copy()
    cur_C = C
    
    while time.time() < deadline and T_curr > T_min:
        # Choose random pair
        p = np.random.randint(n)
        q = np.random.randint(n)
        if p == q:
            continue
        if cur_h[p] < 1e-8 or cur_h[q] > 1.0 - 1e-8:
            continue
        
        # Random transfer amount
        t_min = max(cur_h[p] - 1.0, -cur_h[q])
        t_max = min(cur_h[p], 1.0 - cur_h[q])
        if t_max - t_min < 1e-8:
            continue
        t = np.random.uniform(t_min, t_max)
        
        y = cur_h.copy()
        y[p] = cur_h[p] - t
        y[q] = cur_h[q] + t
        y = np.clip(y, 0.0, 1.0)
        
        # Fix sum
        res = T - y.sum()
        if abs(res) > 1e-10:
            y = project_onto_box_simplex(y, T)
        
        new_C = compute_C(y)
        delta = new_C - cur_C
        
        if delta < 0 or np.random.random() < np.exp(-delta / T_curr):
            cur_h = y
            cur_C = new_C
            if new_C < best_C:
                best_C = new_C
                best_h = y.copy()
        
        T_curr *= cooling
    
    return best_h, best_C

def compute_C(h):
    n = len(h)
    if n < 2:
        return 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def is_feasible(h, tol=1e-7):
    n = len(h)
    if n < 2:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(h.sum() - n / 2.0) > tol * 10:
        return False
    return True

def project_box_simplex(z, T, max_iter=80):
    n = len(z)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(z - mid, 0.0, 1.0)
        s = float(y.sum())
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - lo, 0.0, 1.0)
    res = T - float(y.sum())
    if abs(res) > 1e-12:
        if res > 0:
            slack = 1.0 - y
            order = np.argsort(-slack)
            rem = res
            for i in order:
                a = min(slack[i], rem)
                y[int(i)] += a
                rem -= a
                if rem <= 1e-12:
                    break
        else:
            order = np.argsort(y)
            rem = -res
            for i in order:
                a = min(y[i], rem)
                y[int(i)] -= a
                rem -= a
                if rem <= 1e-12:
                    break
    return y

def select_pairs(h, F, dx, n_top=6, max_pairs=150):

        y[p] = cur_h[p] - t
        y[q] = cur_h[q] + t

            cur_C = new_C
            if new_C < best_C:
