import numpy as np
from scipy.optimize import linprog

def score_h_normalized(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def project_box_mass(v, target=None):
    """Project v onto [0,1]^n with sum = n/2"""
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.clip(v, 0, 1)
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v + (lo + hi) / 2, 0, 1)
    # Fix residual
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        # Distribute among non-boundary elements
        movable = (v > 1e-10) & (v < 1 - 1e-10)
        count = movable.sum()
        if count > 0:
            v[movable] += residual / count
            v = np.clip(v, 0, 1)
            # Second pass
            residual2 = target - v.sum()
            if abs(residual2) > 1e-12:
                movable2 = (v > 1e-10) & (v < 1 - 1e-10)
                count2 = movable2.sum()
                if count2 > 0:
                    v[movable2] += residual2 / count2
                    v = np.clip(v, 0, 1)
    # Final exact fix
    diff = target - v.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(v < 1 - 1e-10) if diff > 0 else np.argmax(v > 1e-10)
        v[idx] += np.sign(diff) * min(abs(diff), np.inf)
        v = np.clip(v, 0, 1)
    return v

def compute_A_and_grad(h):
    """Compute all overlap values and their gradients.
    Returns A (array of 2n-1 overlap values) and G (matrix of shape (2n-1, n) gradients)
    """
    n = len(h)
    num_lags = 2 * n - 1
    A = np.zeros(num_lags)
    G = np.zeros((num_lags, n))
    
    for m in range(-(n-1), n):
        idx = m + n - 1  # index into A and G
        # A_m = sum_j h[j] * (1-h[j+m]) for valid j
        # Valid j: 0 <= j < n and 0 <= j+m < n
        if m >= 0:
            # j ranges from 0 to n-1-m
            j_range = range(0, n - m)
            for j in j_range:
                A[idx] += h[j] * (1 - h[j+m])
                G[idx, j] += (1 - h[j+m])  # derivative w.r.t. h[j]
                G[idx, j+m] -= h[j]  # derivative w.r.t. h[j+m]
        else:
            # m < 0, let t = -m, j ranges from t to n-1
            t = -m
            j_range = range(t, n)
            for j in j_range:
                A[idx] += h[j] * (1 - h[j+m])  # j+m = j-t
                G[idx, j] += (1 - h[j-t])  # derivative w.r.t. h[j]
                G[idx, j-t] -= h[j]  # derivative w.r.t. h[j-t]
    
    return A, G

def make_lp_direction(h_c, radius, time_limit=1.0):
    """Solve LP to find a direction that reduces the max overlap.
    Returns direction d (zero-sum) or None if LP fails.
    """
    n = len(h_c)
    A_c, G = compute_A_and_grad(h_c)
    num_lags = 2 * n - 1
    
    # LP variables: [h_0, ..., h_{n-1}, t]
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: A_m + G_m · (h - h_c) <= t for all m
    # => G_m · h - t <= A_m - G_m · h_c
    G_ext = np.hstack([G, -np.ones((num_lags, 1))])
    b_ub = A_c - G @ h_c
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: h_c[i] - r <= h[i] <= h_c[i] + r, clipped to [0,1]
    bounds = []
    for i in range(n):
        lo = max(0.0, h_c[i] - radius)
        hi = min(1.0, h_c[i] + radius)
        bounds.append((lo, hi))
    bounds.append((0.0, A_c.max() + radius * n))  # t is free above 0
    
    res = linprog(c_obj, A_ub=G_ext, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    h_lp = res.x[:n]
    d = h_lp - h_c
    # Make it exactly zero-sum
    d -= d.mean()
    
    # Check that the direction is meaningful
    if np.linalg.norm(d) < 1e-10:
        return None
    
    return d

def line_search_exact(h_c, d, score_c, max_iters=200):
    """Find best point on line h_c + lambda*d that is feasible and has lower score.

    lam_min += margin
    lam_max -= margin

        h_try = project_box_mass(h_try)

def make_random_direction(n, rng):
    """Generate a random zero-sum feasible direction."""
    d = rng.standard_normal(n)
    d -= d.mean()
    return d

def make_harmonic_direction(n, rng):
    """Generate a low-frequency harmonic direction."""
    k = rng.integers(1, 4)
    d = np.sin(2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 2*np.pi))
    d -= d.mean()
    return d

def reshape_candidate(h_source, m):
    """Reshape h_source of length n to length m using interpolation/averaging."""
    n = len(h_source)
    if m == n:
        return h_source.copy()
    if m > n:
        # Upsample using linear interpolation
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, m)
        h_new = np.interp(x_new, x_old, h_source)
    else:
        # Downsample by averaging blocks
        h_new = np.zeros(m)
        for j in range(m):
            lo = int(j * n / m)
            hi = int((j + 1) * n / m)
            if hi > lo:
                h_new[j] = h_source[lo:hi].mean()
            else:
                h_new[j] = h_source[lo % n]
    return project_box_mass(h_new)

def make_simple_feasible(n, kind=0):

    elif kind == 3:

        h = np.random.uniform(0, 1, n)
    return project_box_mass(h)

import numpy as np
from scipy.optimize import linprog
import time

def score_h_normalized(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_box_mass(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.asarray(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        movable = (v > 1e-10) & (v < 1.0 - 1e-10)
        cnt = movable.sum()
        if cnt > 0:
            v[movable] += residual / cnt
            v = np.clip(v, 0.0, 1.0)
        residual2 = target - v.sum()
        if abs(residual2) > 1e-12:
            movable2 = (v > 0) & (v < 1.0)
            cnt2 = movable2.sum()
            if cnt2 > 0:
                v[movable2] += residual2 / cnt2
                v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax((1.0 - v) > 1e-10)
        else:
            idx = np.argmax(v > 1e-10)
        v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
    return v

def compute_A_and_grad(h):
    n = len(h)
    num_lags = 2 * n - 1
    A = np.zeros(num_lags)
    G = np.zeros((num_lags, n))
    for m in range(-(n - 1), n):
        idx = m + n - 1
        if m >= 0:
            length = n - m
            A[idx] = np.dot(h[:length], 1.0 - h[m:m + length])
            G[idx, :length] += 1.0 - h[m:m + length]
            G[idx, m:m + length] -= h[:length]
        else:
            t = -m
            length = n - t
            A[idx] = np.dot(h[t:], 1.0 - h[:length])
            G[idx, t:] += 1.0 - h[:length]
            G[idx, :length] -= h[t:]
    return A, G

def lp_direction(h_c, radius):
    n = len(h_c)
    A_c, G = compute_A_and_grad(h_c)
    num_lags = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    G_ext = np.hstack([G, -np.ones((num_lags, 1))])
    b_ub = A_c - G @ h_c
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h_c[i] - radius)
        hi_i = min(1.0, h_c[i] + radius)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, A_c.max() + radius * n + 1.0))
    res = linprog(c_obj, A_ub=G_ext, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0 or res.x is None:
        return None
    h_lp = res.x[:n]
    t_lp = res.x[-1]
    if t_lp >= A_c.max() - 1e-8:
        return None
    d = h_lp - h_c
    d -= d.mean()
    if np.linalg.norm(d) < 1e-10:
        return None
    return d

def line_search_exact(h_c, d, score_c):
    n = len(h_c)
    lam_min = -np.inf
    lam_max = np.inf
    eps = 1e-12
    for i in range(n):
        if d[i] > eps:
            lam_min = max(lam_min, -h_c[i] / d[i])
            lam_max = min(lam_max, (1.0 - h_c[i]) / d[i])
        elif d[i] < -eps:
            lam_min = max(lam_min, (1.0 - h_c[i]) / d[i])
            lam_max = min(lam_max, -h_c[i] / d[i])
    if lam_max <= lam_min + 1e-10:
        return None
    lam_min += 1e-9
    lam_max -= 1e-9
    if lam_max <= lam_min:
        return None
    best_h = None
    best_score = score_c
    best_lam = 0.0
    n_grid = 40
    lams = np.linspace(lam_min, lam_max, n_grid)
    for lam in lams:
        h_try = project_box_mass(h_c + lam * d)
        s = score_h_normalized(h_try)
        if s < best_score:
            best_score = s
            best_h = h_try.copy()
            best_lam = lam
    if best_h is None:
        return None
    for _ in range(3):
        half_w = max((lam_max - lam_min) / (n_grid * 2), 1e-4)
        new_lo = max(lam_min, best_lam - half_w)
        new_hi = min(lam_max, best_lam + half_w)
        if new_hi - new_lo < 1e-6:
            break
        lams2 = np.linspace(new_lo, new_hi, 30)

            h_try = project_box_mass(h_c + lam * d)
            s = score_h_normalized(h_try)
            if s < best_score:
                best_score = s
                best_h = h_try.copy()
                best_lam = lam
        lam_min, lam_max = new_lo, new_hi
    if best_h is not None and best_score < score_c - 1e-13:
        return (best_h, best_score)
    return None

def random_direction(n, rng):
    d = rng.standard_normal(n)
    d -= d.mean()
    return d

def harmonic_direction(n, rng):
    k = int(rng.randint(1, 5))
    phase = rng.uniform(0, 2 * np.pi)
    d = np.sin(2 * np.pi * k * np.arange(n) / n + phase)
    d -= d.mean()
    return d

def reshape_to_m(h_src, m):
    n = len(h_src)
    if m == n:
        return h_src.copy()
    if m > n:
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, m)
        h_new = np.interp(x_new, x_old, h_src)
    else:
        h_new = np.zeros(m)
        for j in range(m):
            lo = int(j * n / m)
            hi = int((j + 1) * n / m)
            hi = max(hi, lo + 1)
            h_new[j] = h_src[lo:hi].mean()
    return project_box_mass(h_new)

def make_simple_n(n, rng, kind):
    if kind == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif kind == 1:
        h = np.zeros(n)
        h[::2] = 1.0
    elif kind == 2:
        h = np.clip(1.0 - 2.0 * np.arange(n) / n, 0, 1)
    elif kind == 3:
        h = rng.uniform(0, 1, n)
    elif kind == 4:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    else:
        h = np.ones(n) * 0.5
    return project_box_mass(h)

def optimize_at_n(h_init, budget_remaining, rng, max_iters=500):
    h_work = h_init.copy()
    score_work = score_h_normalized(h_work)
    best_local = h_work.copy()
    best_local_score = score_work
    radius = 0.12
    no_improve = 0
    it = 0
    while it < max_iters:
        if budget_remaining() < 5:
            break
        d = lp_direction(h_work, radius)
        if d is None:
            if no_improve % 3 == 0:
                d = random_direction(len(h_work), rng)
            elif no_improve % 3 == 1:
                d = harmonic_direction(len(h_work), rng)
            else:
                d = random_direction(len(h_work), rng) * 0.5
        result = line_search_exact(h_work, d, score_work)
        if result is not None:
            h_new, s_new = result
            if s_new < best_local_score - 1e-13:
                best_local = h_new.copy()
                best_local_score = s_new
            h_work = best_local.copy()
            score_work = best_local_score
            no_improve = 0
            radius = min(0.25, radius * 1.03)
        else:
            no_improve += 1
            radius = max(0.03, radius * 0.88)
            if no_improve > 20:
                kick = random_direction(len(h_work), rng) * 0.02
                result = line_search_exact(h_work, kick, score_work)
                if result is not None:
                    h_work = result[0].copy()
                    score_work = result[1]
                    if score_work < best_local_score:
                        best_local = h_work.copy()
                        best_local_score = score_work
                no_improve = 0
                radius = 0.1
        it += 1
    return best_local, best_local_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
