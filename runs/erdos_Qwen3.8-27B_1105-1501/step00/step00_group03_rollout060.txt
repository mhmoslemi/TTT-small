    if n is None:
        n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=float)

    lo = -1.0 - z.max() - 1
    hi = 1.0 - z.min() + 1
    
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(z + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    
    # Fix remaining error
    err = target - np.sum(h)
    if abs(err) > 1e-12:

            # Need to add err
            caps = 1.0 - h
            idx = np.argmax(caps)

    target = n / 2.0
    z = np.array(z, dtype=float)

    n = len(h)
