def _smooth_max(vals, a):
    """Log-sum-exp based smooth max approximation"""
    m = np.max(vals)
    shifted = a * (vals - m)
    return m + np.log(np.sum(np.exp(shifted))) / a

def _smooth_max_weights(vals, a):
    """Softmax weights for smooth max"""
    m = np.max(vals)
    shifted = a * (vals - m)
    return np.exp(shifted) / np.sum(np.exp(shifted))

def _overlap_vector(h, dx):
    """Compute all overlap values (one per shift)"""
    return np.correlate(h, 1-h, mode="full") * dx

def _grad_smooth_max(h, dx, a):
    """Gradient of smooth max of overlaps"""
    n = len(h)
    overlaps = _overlap_vector(h, dx)
    weights = _smooth_max_weights(overlaps, a)
    
    # d(overlap[k])/d(h[m]) where overlap[k] corresponds to shift d = k-(n-1)
    grad = np.zeros(n)
    for m in range(n):
        for k in range(2*n-1):

            term = 0.0
            if 0 <= m+d < n:
                term += (1 - h[m+d])
            if 0 <= m-d < n:
                term -= h[m-d]
            grad[m] += weights[k] * dx * term
    
    return grad

        s = np.sum(proj)
