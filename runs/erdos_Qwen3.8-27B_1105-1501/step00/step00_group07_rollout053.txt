def compute_gradient_matrix(h):
    """Compute ∂c[k]/∂h[i] for all i, k. Shape: (n, 2n-1)"""
    n = len(h)
    # grad[i, k_idx] = ∂c[k]/∂h[i] where k = k_idx - (n-1)
    grad = np.zeros((n, 2*n-1))
    
    # k = 0 (k_idx = n-1)
    grad[:, n-1] = 1.0 - 2.0 * h
    
    # k > 0 (k_idx = k + n - 1, k = 1, ..., n-1)
    for k in range(1, n):
        k_idx = k + n - 1
        # ∂c[k]/∂h[i] = (1-h[i-k]) * [k <= i] - h[i+k] * [i <= n-1-k]
        for i in range(k, n):

        for i in range(0, n-k):
            grad[i, k_idx] -= h[i+k]
    
    # k < 0 (k_idx = k + n - 1, k = -1, ..., -(n-1))
    for k in range(1, n):

        # ∂c[-k]/∂h[i] = (1-h[i+k]) * [i <= n-1-k] - h[i-k] * [k <= i]

            grad[i, k_idx] += (1.0 - h[i+k])
        for i in range(k, n):
            grad[i, k_idx] -= h[i-k]
    
    return grad

def focused_descent(h, R, n, deadline):
    """Focus on the worst lag and try to reduce it."""
    dx = 2.0 / n
    best_c = float(np.max(R) * dx)
    
    while time.time() < deadline:
        max_idx = int(np.argmax(R))
        k = max_idx - (n - 1)
        
        if k >= 0:
            # c[k] = sum_{m=0}^{n-1-k} h[m+k]*(1-h[m])
            m_range = np.arange(0, n - k)
            h_m = h[m_range]
            h_mk = h[m_range + k]
            
            # Try transferring from high h_mk to low h_m
            # Sort by contribution h_mk * (1-h_m) in descending order
            contrib = h_mk * (1.0 - h_m)
            order = np.argsort(-contrib)
            
            for idx in order[:20]:  # Try top 20 contributions
                m = m_range[idx]
                i = m + k  # source (high value)
                j = m      # destination (low value)
                
                step = min(h[i] * 0.3, (1.0 - h[j]) * 0.3, 0.05)
                if step < 1e-10:
                    continue
                
                dR = delta_R_single(h, i, -step) + delta_R_single(h, j, step)
                new_c = float(np.max(R + dR)) * dx
                
                if new_c < best_c - 1e-12:
                    h[i] -= step
                    h[j] += step
                    R = R + dR  # Update R without recomputing
                    best_c = new_c
                    break  # Restart with new worst lag
        else:
            k_abs = -k
            m_range = np.arange(0, n - k_abs)
            h_m = h[m_range]
            h_mk = h[m_range + k_abs]
            # c[-k_abs] = sum h[m]*(1-h[m+k_abs])
            contrib = h_m * (1.0 - h_mk)
            order = np.argsort(-contrib)
            
            for idx in order[:20]:
                m = m_range[idx]
                i = m  # source
                j = m + k_abs  # destination
                
                step = min(h[i] * 0.3, (1.0 - h[j]) * 0.3, 0.05)
                if step < 1e-10:
                    continue
                
                dR = delta_R_single(h, i, -step) + delta_R_single(h, j, step)
                new_c = float(np.max(R + dR)) * dx
                
                if new_c < best_c - 1e-12:
                    h[i] -= step
                    h[j] += step
                    R = R + dR
                    best_c = new_c
                    break
    
    return h, R, best_c
