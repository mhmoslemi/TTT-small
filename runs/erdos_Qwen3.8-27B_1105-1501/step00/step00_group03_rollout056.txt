def official_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def manual_values(h):
    n = len(h)
    vals = np.zeros(2 * n - 1)
    for s in range(-(n-1), n):
        i_min = max(0, -s)
        i_max = min(n, n - s)
        if i_min < i_max:
            i = np.arange(i_min, i_max)
            vals[s + n - 1] = np.dot(h[i], 1.0 - h[i + s])
    return vals

def manual_grad(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    for s in range(-(n-1), n):
        r = s + n - 1
        p1_lo = max(0, -s)
        p1_hi = min(n, n - s)
        if p1_lo < p1_hi:
            p1 = np.arange(p1_lo, p1_hi)
            grad[r, p1] += (1.0 - h[p1 + s])
        p2_lo = max(0, s)
        p2_hi = min(n, n + s)
        if p2_lo < p2_hi:
            p2 = np.arange(p2_lo, p2_hi)
            grad[r, p2] -= h[p2 - s]
    return grad

    n = len(z)
    target = n / 2.0
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(60):

        g = np.sum(np.clip(z + mid, 0.0, 1.0))
        if g < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(interior):
            idx = np.where(interior)[0]
            per = residual / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
        # Second pass if needed
        residual2 = target - np.sum(h)
        if abs(residual2) > 1e-14:
            if residual2 > 0:
                caps = 1.0 - h
                p = np.argmax(caps)
                h[p] = min(1.0, h[p] + residual2)
            else:
                p = np.argmax(h)
                h[p] = max(0.0, h[p] + residual2)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = manual_values(x) * dx
    vmax = np.max(vals)
    shifted = beta * (vals - vmax)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_v = np.exp(shifted)
    logsum = np.log(np.sum(exp_v))
    obj = vmax + logsum / beta
    w = exp_v / np.sum(exp_v)
    gm = manual_grad(x)
    grad = dx * (gm.T @ w)

def smooth_grad(x, beta):

            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
        residual2 = target - np.sum(h)
        if abs(residual2) > 1e-13:
            if residual2 > 0:
                p = np.argmax(1.0 - h)
                h[p] = min(1.0, h[p] + residual2)
            else:
                p = np.argmax(h)
                h[p] = max(0.0, h[p] + residual2)
    return np.clip(h, 0.0, 1.0)


def smooth_obj_and_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    vals = manual_values(x) * dx
    vmax = np.max(vals)
    shifted = beta * (vals - vmax)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_v = np.exp(shifted)
    s = np.sum(exp_v)
    if s <= 0:
        obj = vmax
        grad = np.zeros(n)
        return obj, grad
    logsum = np.log(s)
    obj = vmax + logsum / beta
    w = exp_v / s
    gm = manual_grad(x)
    grad = dx * (gm.T @ w)
    return obj, grad


def smooth_obj(x, beta):
    return smooth_obj_and_grad(x, beta)[0]


def smooth_grad(x, beta):
    return smooth_obj_and_grad(x, beta)[1]


def sum_eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def sum_eq_jac(x):
    return np.ones(len(x))


def antisym(h):
    return 1.0 - h[::-1]


def interpolate_to(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)


def solve_smooth(x0, beta, maxiter=120, ftol=1e-12):
    n = len(x0)
    x0 = feasible_project(x0.copy())
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(smooth_obj, x0, args=(beta,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return feasible_project(res.x)
    except Exception:
        return x0


def direct_obj(z):
    return z[-1]


def direct_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def direct_ineq_fun(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    theta = z[-1]
    vals = manual_values(h) * dx
    return theta - vals


def direct_ineq_jac(z):
    n = len(z) - 1
    dx = 2.0 / n
    L = 2 * n - 1
    h = z[:n]
    J = np.zeros((L, n + 1))
    gm = manual_grad(h)
    J[:, :n] = -dx * gm
    J[:, -1] = 1.0
    return J


def direct_eq_fun(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0


def direct_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def solve_direct(x0, maxiter=150):
    n = len(x0)
    dx = 2.0 / n
    h0 = feasible_project(x0.copy())
    theta0 = official_score(h0) + 1e-5
    z0 = np.concatenate([h0, [theta0]])
    bounds = [(0.0, 1.0)] * n + [(1e-6, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': direct_ineq_fun, 'jac': direct_ineq_jac},
        {'type': 'eq', 'fun': direct_eq_fun, 'jac': direct_eq_jac},
    ]
    try:
        res = minimize(direct_obj, z0, jac=direct_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return feasible_project(res.x[:n])
    except Exception:
        return h0


def try_update(best_c, best_h, new_h):
    if new_h is None:
        return best_c, best_h
    s = official_score(new_h)
    if s < best_c - 1e-14:
        return s, new_h.copy()
    return best_c, best_h


def generate_starts(best_h, rng, count=15):
    n = len(best_h)
    starts = []
    starts.append(best_h.copy())
    starts.append(feasible_project(antisym(best_h)))
    sigmas = [0.02, 0.05, 0.1, 0.15, 0.25]
    for i in range(count):
        sigma = sigmas[i % len(sigmas)]
        p = best_h + rng.normal(0, sigma, n)
        p = feasible_project(p)
        if i % 3 == 0:
            p = feasible_project(antisym(p))
        starts.append(p)
        if i % 4 == 0:
            p2 = best_h + rng.uniform(-sigma, sigma, n)
            p2 = feasible_project(p2)
            starts.append(p2)
    # Non-uniform structured starts
    t = np.linspace(0, 1, n)
    for amp in [0.3, 0.6, 0.9]:
        wave = 0.5 + amp * 0.5 * np.sin(2 * np.pi * 2 * t + rng.uniform(0, 6.28))
        starts.append(feasible_project(wave))
    return starts


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    reserve = max(5.0, 0.05 * budget_s)
    deadline = t_start + budget_s - reserve

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        parent = None

    if parent is not None and len(parent) >= 10 and np.all(np.isfinite(parent)):
        best_h = feasible_project(parent.copy())
        best_c = official_score(best_h)
    else:
        best_h = feasible_project(np.full(40, 0.5))
        best_c = official_score(best_h)

    rng = np.random.default_rng(seed)
    n_best = len(best_h)

    # Phase 1: Smooth optimization with increasing beta on parent
    betas_phase1 = [10, 50, 200, 500, 1500]
    cur = best_h
    for beta in betas_phase1:
        if time.time() > deadline:
            break
        cur = solve_smooth(cur, beta, maxiter=100)
        best_c, best_h = try_update(best_c, best_h, cur)
        if time.time() > deadline:
            break

    # Phase 2: Direct minimax polish
    if time.time() < deadline - 30:
        cur = solve_direct(best_h, maxiter=200)
        best_c, best_h = try_update(best_c, best_h, cur)

    # Phase 3: Second round smooth + direct
    if time.time() < deadline - 60:
        cur = solve_smooth(best_h, 2000, maxiter=80)
        best_c, best_h = try_update(best_c, best_h, cur)
        if time.time() < deadline - 30:
            cur = solve_direct(best_h, maxiter=150)
            best_c, best_h = try_update(best_c, best_h, cur)

    # Phase 4: Multiple restarts
    starts = generate_starts(best_h, rng, count=12)
    max_restarts = 12
    for i in range(max_restarts):
        if time.time() > deadline - 30:
            break
        s = starts[i % len(starts)]
        # Smooth then direct
        cur = solve_smooth(s, 200, maxiter=80)
        best_c, best_h = try_update(best_c, best_h, cur)
        if time.time() < deadline - 30:
            cur = solve_direct(cur, maxiter=100)
            best_c, best_h = try_update(best_c, best_h, cur)
        # Regenerate more starts if we have time
        if i == 5 and time.time() < deadline - 120:
            extra = generate_starts(best_h, rng, count=6)
            starts.extend(extra)

    # Phase 5: Try nearby resolutions
    res_trials = [n_best - 5, n_best + 5, n_best + 10, n_best - 10, n_best + 15]
    for n_try in res_trials:
        if time.time() > deadline - 60:
            break
        if n_try < 20 or n_try > 80:
            continue
        interp = feasible_project(interpolate_to(best_h, n_try))
        cur = solve_smooth(interp, 200, maxiter=80)
        best_c, best_h = try_update(best_c, best_h, cur)
        if time.time() < deadline - 30:
            cur = solve_direct(cur, maxiter=100)
            best_c, best_h = try_update(best_c, best_h, cur)

    # Phase 6: Final intense polish if time remains
    if time.time() < deadline - 30:
        cur = solve_smooth(best_h, 3000, maxiter=60, ftol=1e-14)
        best_c, best_h = try_update(best_c, best_h, cur)
    if time.time() < deadline - 10:
        cur = solve_direct(best_h, maxiter=100)
        best_c, best_h = try_update(best_c, best_h, cur)

    # Final verification
    best_h = feasible_project(best_h)
    best_c = official_score(best_h)
    ret_c = best_c + 1e-12

    return (best_h, ret_c, len(best_h))
