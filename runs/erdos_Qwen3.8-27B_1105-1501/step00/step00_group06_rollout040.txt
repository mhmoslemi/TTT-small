def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_gradient(h, s):
    """Compute gradient of c_s with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # First factor: (1 - h[j+s]) for valid j+s
    # Second factor: -h[j-s] for valid j-s
    for j in range(n):
        if 0 <= j + s < n:
            grad[j] += 1.0 - h[j + s]
        if 0 <= j - s < n:
            grad[j] -= h[j - s]
    return grad

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def line_search_fast(h, d, dx):
    """Fast line search using precomputed quadratic coefficients."""
    n = len(h)
    # Precompute coefficients
    a = np.correlate(d, d, mode='full')
    b = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # Feasible range for λ
    lam_lo, lam_hi = 0.0, 0.0

            lam_hi = min(lam_hi, (1.0 - h[i]) / d[i])

            lam_hi = min(lam_hi, -h[i] / d[i])
    lam_lo = min(lam_lo, 0.0)
    lam_hi = max(lam_hi, 0.0)
    if lam_hi - lam_lo < 1e-12:
        return 0.0, np.max(c) * dx
    
    m = 1e-10
    lam_lo += m
    lam_hi -= m

    best_lam = 0.0

        val = np.max(vals) * dx
        if val < best_val:
            best_val = val
            best_lam = lam
    
    # Refine
    if best_lam != 0.0:
        step = (lam_hi - lam_lo) / n_evals
        for _ in range(5):
            lo = max(lam_lo, best_lam - 2 * step)
            hi = min(lam_hi, best_lam + 2 * step)
            if hi - lo < 1e-14:

            for lam in lams_fine:
                vals = c + lam * b - lam * lam * a
                val = np.max(vals) * dx
                if val < best_val:
                    best_val = val
                    best_lam = lam
            step *= 0.25

def compute_subgradient(h, corrs, n, tol=1e-4):
    """Compute subgradient of max(corrs * dx)."""
    max_val = np.max(corrs)
    active = np.where(corrs >= max_val - tol / (2.0/n))[0]  # tol in C5 units
    n_active = len(active)
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        # ∂c_s/∂h_j = (1-h[j+s])*1[0≤j+s<n] - h[j-s]*1[0≤j-s<n]
        for j in range(n):
            if 0 <= j + s < n:
                grad[j] += (1.0 - h[j + s])
            if 0 <= j - s < n:
                grad[j] -= h[j - s]
    
    return grad / n_active

def compute_subgradient_vec(h, corrs, n, tol=1e-3):
    """Vectorized subgradient computation."""
    max_val = np.max(corrs)
    # tol is in C5 = corrs*dx units, so in corrs units it's tol/dx = tol*n/2
    tol_corr = tol * n / 2.0
    active = np.where(corrs >= max_val - tol_corr)[0]
    n_active = len(active)
    if n_active == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        # First term: (1-h[j+s]) for valid j
        j_valid = np.arange(max(0, -s), min(n, n - s))
        grad[j_valid] += (1.0 - h[j_valid + s])
        # Second term: -h[j-s] for valid j
        j_valid2 = np.arange(max(0, s), min(n, n + s))
        grad[j_valid2] -= h[j_valid2 - s]
    
    return grad / n_active

            lams_f = np.linspace(lo_r, hi_r, n_evals)
            vals_f = c[None, :] + lams_f[:, None] * b[None, :] - (lams_f[:, None] ** 2) * a[None, :]
            obj_f = np.max(vals_f, axis=1) * dx
            idx_f = int(np.argmin(obj_f))
            if obj_f[idx_f] < best_val:
                best_val = float(obj_f[idx_f])
                best_lam = float(lams_f[idx_f])
            step *= 0.2
    return best_lam, best_val

def subgradient(h, corrs, n, tol_c5=5e-4):
    dx = 2.0 / n
    max_val = np.max(corrs)
    tol_corr = tol_c5 / dx
    active = np.where(corrs >= max_val - tol_corr)[0]
    n_act = len(active)
    if n_act == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        j1 = np.arange(max(0, -s), min(n, n - s))
        if len(j1) > 0:
            grad[j1] += (1.0 - h[j1 + s])
        j2 = np.arange(max(0, s), min(n, n + s))
        if len(j2) > 0:
            grad[j2] -= h[j2 - s]
    return grad / n_act

def fix_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(free) == 0:
            if diff > 0:
                z = np.where(h < 1e-6)[0]
                o = np.where(h > 1.0 - 1e-6)[0]
                if len(z) > 0 and len(o) > 0:
                    amt = min(diff / len(z), 1.0 - h[o].max() if len(o) > 0 else 1.0)
                    h[z] += amt
                    h[o] -= amt
            else:
                z = np.where(h < 1e-6)[0]
                o = np.where(h > 1.0 - 1e-6)[0]
                if len(z) > 0 and len(o) > 0:
                    amt = min(-diff / len(o), h[o].min() if len(o) > 0 else 1.0)
                    h[o] += amt
                    h[z] -= amt
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def try_direction(h, d, dx, best_c5):
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return h, best_c5
    lam, val = line_search_fast(h, d, dx)
    if val < best_c5 - 1e-13 and lam != 0.0:
        h_new = h + lam * d
        h_new = fix_feasible(h_new, len(h))
        actual = compute_c5(h_new, dx)
        if actual < best_c5 - 1e-13:
            return h_new, actual
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    timeout = t0 + budget_s * 0.93

    try:
        h = np.array(initial_h_values, dtype=np.float64)
    except (NameError, UnboundLocalError):
        n = 100
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = fix_feasible(h, n)
    else:
        n = len(h)
        h = fix_feasible(h.copy(), n)

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    stag = 0
    tol_sg = 5e-4

    while time.time() < timeout:
        r = rng.random()
        if r < 0.35:
            corrs = np.correlate(h, 1.0 - h, mode='full')
            g = subgradient(h, corrs, n, tol_stg)
            d = -g - g.mean()
        elif r < 0.55:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < 0.72:
            i, j = rng.choice(n, 2, replace=False)
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        elif r < 0.86:
            d = np.zeros(n)
            for _ in range(rng.randint(2, 6)):
                i, j = rng.choice(n, 2, replace=False)
                w = rng.standard_normal()
                d[i] += w
                d[j] -= w
            d -= d.mean()
        else:
            k = rng.randint(1, max(2, n // 5))
            ph = rng.uniform(0, 2.0 * np.pi)
            d = np.sin(2.0 * np.pi * k * np.arange(n) / n + ph)
            d -= d.mean()

        h_new, c5_new = try_direction(h, d, dx, best_c5)
        if c5_new < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            stag = 0
        else:
            stag += 1

        if stag > 200:
            h = best_h + rng.standard_normal(n) * 0.008
            h = fix_feasible(h, n)
            stag = 0

    # Pair polish phase
    remaining = timeout - time.time()
    if remaining > 3.0:
        h_cur = best_h.copy()
        c5_cur = best_c5
        for rnd in range(3):
            if time.time() > timeout:
                break
            improved = False
            order = rng.permutation(n)
            for ii in range(n):
                if time.time() > timeout:
                    break
                i = order[ii]
                if c5_cur >= best_c5:
                    for jj in range(ii + 1, n):
                        if time.time() > timeout:
                            break
                        j = order[jj] if jj < n else (jj - n)
                        d = np.zeros(n)
                        d[i] = 1.0
                        d[j] = -1.0
                        h_cur, c5_cur = try_direction(h_cur, d, dx, best_c5)
                        if c5_cur < best_c5:
                            best_h = h_cur.copy()
                            best_c5 = c5_cur
                            improved = True
                        d[i] = -1.0
                        d[j] = 1.0
                        h_cur, c5_cur = try_direction(h_cur, d, dx, best_c5)
                        if c5_cur < best_c5:
                            best_h = h_cur.copy()
                            best_c5 = c5_cur
                            improved = True
            if not improved:
                break

    best_h = fix_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
