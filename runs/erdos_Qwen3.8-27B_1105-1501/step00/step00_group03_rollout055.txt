    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(vals)) * dx

def project_feasible(z, n):
    """Project z to {0 ≤ h ≤ 1, sum(h) = n/2} using bisection on shift."""
    T = n / 2.0
    # Find lambda such that sum(clip(z + lambda, 0, 1)) = T
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    res = T - np.sum(h)
    if abs(res) > 1e-12:
        if res > 0:
            # Add to coordinates with most room
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(res, 1.0 - h[idx])
                h[idx] += add
                res -= add
                if abs(res) < 1e-12:
                    break
        else:
            res = -res
            order = np.argsort(h)
            for idx in order:
                sub = min(res, h[idx])
                h[idx] -= sub
                res -= sub
                if abs(res) < 1e-12:
                    break
    return h

def manual_correlations(h):
    """Compute F_k for all lags k = -(n-1) to n-1."""
    n = len(h)
    vals = np.empty(2*n - 1)
    for m in range(2*n - 1):

        vals[m] = np.sum(h[i_min+k:i_max+k] * (1.0 - h[i_min:i_max]))

    max_shifted = np.max(shifted)
    logsumexp = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))

    T = n * 0.5
    z = np.asarray(z, dtype=float).copy()
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)
    if hi - lo < 1e-15:

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    res = T - np.sum(h)
    if abs(res) > 1e-10:

def eq_jac(x):
    return np.ones(len(x))


def dm_obj(z):
    return z[-1]


def dm_obj_jac(z):
    out = np.zeros(len(z))
    out[-1] = 1.0
    return out


def dm_ineq(z):
    n = len(z) - 1
    dx = 2.0 / n
    return z[-1] - corr_vals(z[:n]) * dx


def dm_ineq_jac(z):
    n = len(z) - 1
    dx = 2.0 / n
    h = z[:n]
    M = 2 * n - 1
    J = np.zeros((M, n + 1))
    J[:, -1] = 1.0
    for m in range(M):
        k = m - (n - 1)
        if k >= 0:
            idx = np.arange(0, n - k)
        else:
            idx = np.arange(-k, n)
        hk = h[idx + k]
        hi = h[idx]
        np.add.at(J[m], idx + k, -dx * (1.0 - hi))
        np.add.at(J[m], idx, dx * hk)
    return J


def dm_eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n * 0.5


def dm_eq_jac(z):
    n = len(z) - 1
    out = np.zeros(n + 1)
    out[:n] = 1.0
    return out


def interpolate_h(h_src, n_tgt):
    n_src = len(h_src)
    xs = (np.arange(n_src, dtype=float) + 0.5) * (2.0 / n_src)
    xt = (np.arange(n_tgt, dtype=float) + 0.5) * (2.0 / n_tgt)
    return np.interp(xt, xs, h_src, left=0.0, right=0.0)


def antisym_transform(h):
    n = len(h)
    return 1.0 - h[::-1].copy()


def run_smooth(h0, beta, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_sf, h0, jac=True, args=(beta,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return res
    except Exception:
        return None


def run_direct(h0, maxiter=300):
    n = len(h0)
    s0 = official_score(h0)
    z0 = np.concatenate([h0, [s0 + 1e-8]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': dm_ineq, 'jac': dm_ineq_jac},
        {'type': 'eq', 'fun': dm_eq, 'jac': dm_eq_jac}
    ]
    try:
        res = minimize(dm_obj, z0, jac=dm_obj_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res
    except Exception:
        return None


def try_update(best_h, best_s, best_n, h_cand):
    n = len(h_cand)
    h_cand = project_feasible(h_cand, n)
    s = official_score(h_cand)
    if not np.isfinite(s):
        return best_h, best_s, best_n, False
    if s < best_s - 1e-13:
        return h_cand.copy(), s, n, True
    if abs(s - best_s) <= 1e-12 and n < best_n:
        return h_cand.copy(), s, n, True
    return best_h, best_s, best_n, False


def smooth_chain(h0, betas, maxiter_each=200):
    cur = h0.copy()
    for beta in betas:
        res = run_smooth(cur, beta, maxiter=maxiter_each)
        if res is not None and res.x is not None:
            cur = project_feasible(res.x, len(cur))
    return cur


def direct_polish(h0, maxiter=300):
    res = run_direct(h0, maxiter=maxiter)
    if res is not None and res.x is not None:
        return project_feasible(res.x[:-1], len(h0))
    return h0


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - max(5.0, 0.03 * budget_s)
    rng = np.random.default_rng(seed)

    # Get and validate parent
    h_par = np.asarray(initial_h_values, dtype=float).ravel()
    n_par = len(h_par)
    if n_par < 4:
        h_par = np.full(45, 1.0 / 2)
        n_par = 45
    h_par = project_feasible(h_par, n_par)
    best_h = h_par.copy()
    best_s = official_score(h_par)
    best_n = n_par

    # Phase 1: Smooth optimization from parent
    betas = [30, 100, 500, 2000]
    cur = h_par.copy()
    for beta in betas:
        if time.time() > deadline:
            break
        res = run_smooth(cur, beta, maxiter=250)
        if res is not None and res.x is not None:
            cur = project_feasible(res.x, n_par)
            best_h, best_s, best_n, _ = try_update(
                best_h, best_s, best_n, cur)

    # Phase 2: Direct minimax on parent size
    if time.time() < deadline:
        pol = direct_polish(best_h, maxiter=400)
        best_h, best_s, best_n, _ = try_update(best_h, best_s, best_n, pol)

    # Phase 3: Multi-size exploration
    target_ns = sorted(set([
        max(20, int(n_par * 0.7)),
        max(25, int(n_par * 0.85)),
        int(n_par * 0.85),
        int(n_par * 1.1),
        int(n_par * 1.2),
    ]))
    target_ns = [n for n in target_ns if 15 <= n <= 80]

    # Build pool of starting candidates
    pool = []
    # Best so far at various sizes
    for nt in target_ns:
        h_interp = interpolate_h(best_h, nt)
        h_interp = project_feasible(h_interp, nt)
        pool.append(h_interp)

    # Random perturbations of best
    for sigma in [0.02, 0.05, 0.1]:
        h_pert = best_h + sigma * rng.standard_normal(n_par)
        h_pert = project_feasible(h_pert, n_par)
        pool.append(h_pert)

    # Antisymmetric
    h_asym = antisym_transform(best_h)
    h_asym = project_feasible(h_asym, n_par)
    pool.append(h_asym)

    # Deterministic structured starts
    h_sine = 0.5 + 0.4 * np.sin(np.pi * np.arange(n_par) / n_par)
    h_sine = project_feasible(h_sine, n_par)
    pool.append(h_sine)

    # Sort pool by score (best first)
    pool_scored = []
    for h_c in pool:
        sc = official_score(h_c)
        if np.isfinite(sc):
            pool_scored.append((sc, h_c))
    pool_scored.sort(key=lambda x: x[0])

    # Process pool with smooth + direct
    for sc, h_c in pool_scored:
        if time.time() > deadline:
            break
        n = len(h_c)
        # Smooth chain
        h_smooth = smooth_chain(h_c, [100, 500], maxiter_each=150)
        best_h, best_s, best_n, _ = try_update(best_h, best_s, best_n, h_smooth)
        # Direct polish
        if time.time() < deadline:
            h_dir = direct_polish(h_smooth, maxiter=250)
            best_h, best_s, best_n, _ = try_update(
                best_h, best_s, best_n, h_dir)

    # Phase 4: More aggressive search if time remains
    if time.time() < deadline:
        # Try from best with different perturbations
        for _ in range(10):
            if time.time() > deadline:
                break
            sigma = rng.uniform(0.01, 0.15)
            h_try = best_h + sigma * rng.standard_normal(best_n)
            h_try = project_feasible(h_try, best_n)
            h_smooth = smooth_chain(h_try, [200, 1000], maxiter_each=100)
            best_h, best_s, best_n, _ = try_update(
                best_h, best_s, best_n, h_smooth)
            if time.time() < deadline:
                h_dir = direct_polish(h_smooth, maxiter=200)
                best_h, best_s, best_n, _ = try_update(
                    best_h, best_s, best_n, h_dir)

    # Phase 5: Final direct polish on the overall best
    if time.time() < deadline - 15:
        h_final = direct_polish(best_h, maxiter=500)
        best_h, best_s, best_n, _ = try_update(
            best_h, best_s, best_n, h_final)

    # Final: one more smooth with very high beta from best
    if time.time() < deadline - 10:
        h_hs = run_smooth(best_h, 5000, maxiter=200)
        if h_hs is not None and h_hs.x is not None:
            h_hs = project_feasible(h_hs.x, best_n)
            best_h, best_s, best_n, _ = try_update(
                best_h, best_s, best_n, h_hs)

    # Ensure final result is valid
    best_h = project_feasible(best_h, best_n)
    best_s = official_score(best_h)
    if not np.isfinite(best_s):
        best_h = h_par
        best_s = official_score(h_par)
        best_n = n_par

    return best_h, float(best_s), int(best_n)
