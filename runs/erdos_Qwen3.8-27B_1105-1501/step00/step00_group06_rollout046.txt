- If j appears as second index (i+k=j, so i=j-k must be valid): contribution is -h[j-k]

Let me think about the np.correlate convention.
np.correlate(a, v, mode="full") computes:
c[k] = sum_n a[n+k] * conj(v[n]) for valid n

So np.correlate(h, 1-h, mode="full"):

The index k ranges from -(len(v)-1) to (len(a)-1), i.e., from -(n-1) to n-1.
In the output array, index 0 corresponds to k = -(n-1), and index 2n-2 corresponds to k = n-1.

A_k = sum_{n: n >= 0, n < n, n+k >= 0, n+k < n} h[n+k] * (1-h[n])

def exact_A_and_grad(h):
    n = len(h)
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for s in range(-(n-1), n):
        idx = s + (n - 1)  # index in the full correlation array
        if s > 0:
            # A_s = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            m = n - 1 - s
            A[idx] = np.sum(h[:m+1] * (1.0 - h[s:]))
            # Gradient
            G[idx, :m+1] += (1.0 - h[s:m+1+s])  # (1-h[j+s]) for j in 0..m
            G[idx, s:] -= h[:m+1]  # -h[j-s] for j in s..n-1
        elif s < 0:
            t = -s
            # A_s = sum_{i=t}^{n-1} h[i]*(1-h[i-t])
            m = n - t
            A[idx] = np.sum(h[t:] * (1.0 - h[:m]))
            # Gradient
            G[idx, t:] += (1.0 - h[:m])  # (1-h[j-t]) for j in t..n-1
            G[idx, :m] -= h[t:]  # -h[j+t] for j in 0..m-1... 
            # Wait, let me redo this.
            # A_s = sum_{i=t}^{n-1} h[i]*(1-h[i-t])
            # d/dh[j]: 
            #   if j >= t and j-t < n (always true since j < n): 
            #     j appears as first factor: (1-h[j-t])
            #   if j-t >= 0 and j >= t... wait, j appears as second factor when i-t = j, i.e., i = j+t
            #     so if j+t < n: j appears as second factor with coefficient -h[j+t]
            # So:
            #   G[idx, t:] += (1-h[:n-t])  for j in t..n-1, (1-h[j-t])
            #   G[idx, :n-t] -= h[t:]  for j in 0..n-t-1, -h[j+t]
            G[idx, t:t+m] += (1.0 - h[:m])  # j from t to n-1, (1-h[j-t])
            G[idx, 0:m] -= h[t:t+m]  # j from 0 to n-t-1, -h[j+t]
        else:
            # s = 0: A_0 = sum_i h[i]*(1-h[i])
            A[idx] = np.sum(h * (1.0 - h))
            G[idx, :] = 1.0 - 2.0 * h
    return A, G

def make_linprog_direction(h_c, radius, timeout_s=5.0):
    n = len(h_c)
    A_s, G = exact_A_and_grad(h_c)
    
    # LP: minimize t
    # subject to: G @ h - t <= b, where b_s = A_s[s] - G[s] @ h_c
    #              sum(h) = n/2
    #              max(0, h_c - r) <= h <= min(1, h_c + r)
    
    # Decision variables: [h(0), ..., h(n-1), t]
    # Objective: c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G_s @ h - t <= b_s
    # This is: [G_s, -1] @ [h, t] <= b_s
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = A_s - G @ h_c
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lower = np.maximum(0.0, h_c - radius)
    upper = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, A_s.max() + 1.0))  # t is bounded above loosely
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_lp = res.x[:n]
            d = h_lp - h_c
            # Ensure sum(d) = 0
            d -= np.mean(d)
            if np.linalg.norm(d) > 1e-10:
                return d
    except Exception:
        pass
    return None

    for _ in range(100):
        if abs(residual) < 1e-12:
            break
        movable = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        if len(movable) == 0:
            break
        step = residual / len(movable)
        y[movable] = np.clip(y[movable] + step, 0.0, 1.0)
        residual = target - np.sum(y)
    # Exact fix
    if abs(target - np.sum(y)) > 1e-12:
        movable = np.where((y > 1e-12) & (y < 1.0 - 1e-12))[0]
        for idx in movable:
            if abs(target - np.sum(y)) > 1e-12:
                adj = target - np.sum(y)
                y[idx] = np.clip(y[idx] + adj, 0.0, 1.0)
    return y

def exact_A_and_grad(h):
    n = len(h)
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for s in range(-(n-1), n):
        idx = s + (n - 1)
        if s > 0:
            m = n - s  # number of valid pairs
            A[idx] = np.dot(h[:m], 1.0 - h[s:])
            G[idx, :m] += (1.0 - h[s:s+m])
            G[idx, s:] -= h[:m]
        elif s < 0:
            t = -s
            m = n - t
            A[idx] = np.dot(h[t:], 1.0 - h[:m])
            G[idx, t:t+m] += (1.0 - h[:m])
            G[idx, :m] -= h[t:t+m]
        else:
            A[idx] = np.sum(h * (1.0 - h))
            G[idx, :] = 1.0 - 2.0 * h
    return A, G

def solve_lp_direction(h_c, radius):
    n = len(h_c)
    A_s, G = exact_A_and_grad(h_c)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = A_s - G @ h_c
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lower = np.maximum(0.0, h_c - radius)
    upper = np.minimum(1.0, h_c + radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, A_s.max() + 1.0))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 10, 'presolve': True})
        if res.status == 0 and res.x is not None:
            h_lp = res.x[:n]
            d = h_lp - h_c
            d -= d.mean()
            if np.linalg.norm(d) > 1e-10:
                return d
    except Exception:
        pass
    return None

def line_search(h_c, d, current_score, n_grid=60):
    n = len(h_c)
    eps = 1e-9
    max_lam = 0.0
    
    for i in range(n):
        if d[i] > eps:
            max_lam = min(max_lam if max_lam > 0 else 1e10, (1.0 - h_c[i]) / d[i])
        elif d[i] < -eps:
            max_lam = min(max_lam if max_lam > 0 else 1e10, (1.0 - h_c[i]) / d[i])
    
    if max_lam <= 0:
        return h_c.copy(), current_score
    
    max_lam = min(max_lam, 1.0)  # cap
    lams = np.linspace(0.0, max_lam, n_grid)
    scores = np.array([score_h_normalized(h_c + lam * d) for lam in lams])
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_h = h_c + lams[best_idx] * d
    
    if best_idx > 0 and best_idx < n_grid - 1:
        lo = lams[best_idx - 1]
        hi = lams[best_idx + 1]
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            s1 = score_h_normalized(h_c + m1 * d)
            s2 = score_h_normalized(h_c + m2 * d)
            if s1 <= s2:
                hi = m2
            else:
                lo = m1
        lam_best = (lo + hi) / 2.0
        h_ref = h_c + lam_best * d
        s_ref = score_h_normalized(h_ref)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_mass(best_h)
    return best_h, best_score

def reshape_candidate(h_source, m):
    n = len(h_source)
    if m == n:
        return h_source.copy()
    if m > n:
        x_src = np.arange(n, dtype=float)
        x_dst = np.linspace(0, n - 1, m, dtype=float)
        h_m = np.interp(x_dst, x_src, h_source)
        return project_box_mass(h_m, m / 2.0)
    else:
        h_m = np.zeros(m)
        for j in range(m):
            lo = int(round(j * n / m))
            hi = int(round((j + 1) * n / m))
            hi = max(hi, lo + 1)
            h_m[j] = np.mean(h_source[lo:hi])
        return project_box_mass(h_m, m / 2.0)

def make_random_direction(n, rng):
    d = rng.standard_normal(n)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-10:
        return None
    return d / norm

def make_grad_direction(h_c, rng):
    n = len(h_c)
    A_s, G = exact_A_and_grad(h_c)
    max_val = A_s.max()
    # Soft weights: emphasize near-max constraints
    w = np.exp(-(max_val - A_s) / (0.05 + 0.01 * max_val))
    w = w + rng.standard_normal(2*n-1) * 0.1
    w = np.clip(w, 0, None)
    if w.sum() < 1e-10:
        w = np.ones(2*n-1)
    w /= w.sum()
    d = -(G.T @ w)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-10:
        return None
    return d / norm

def pair_swap_optimize(h, rng, n_trials=50):
    """Try random pair swaps and accept if improvement."""
    n = len(h)
    best = h.copy()
    best_s = score_h_normalized(h)
    for _ in range(n_trials):
        i, j = rng.choice(n, size=2, replace=False)
        # Try moving mass from i to j and vice versa
        for delta in [0.01, 0.02, 0.05, 0.1]:
            for sign in [1, -1]:
                h_try = h.copy()
                adj = sign * delta
                if 0 <= h_try[i] + adj <= 1 and 0 <= h_try[j] - adj <= 1:
                    h_try[i] += adj
                    h_try[j] -= adj
                    s = score_h_normalized(h_try)
                    if s < best_s - 1e-12:
                        best = h_try.copy()
                        best_s = s
    return best, best_s

- At point h, the active set is S = {k : A_k(h) = max}

- Take a step in the negative subgradient direction

    best_score = score_h_normalized(h)

    radius = 0.06
    no_improve = 0
    iter_count = 0

    while time.time() - start_time < budget_s - 15:
        iter_count += 1
        current_score = score_h_normalized(h)

        improved = False

        # Phase 1: LP-based direction (most of the time)
        if iter_count % 4 != 3:
            d = solve_lp_direction(h, radius)
        else:
            d = make_grad_direction(h, rng)

        if d is not None:
            new_h, new_score = line_search(h, d, current_score, n_grid=60)
            if new_score < best_score - 1e-12:
                h = project_box_mass(new_h)
                best_score = new_score
                best_h = h.copy()
                no_improve = 0
                improved = True

        # Phase 2: Random direction
        if not improved:
            d = make_random_direction(n, rng)
            if d is not None:
                new_h, new_score = line_search(h, d, current_score, n_grid=40)
                if new_score < best_score - 1e-12:
                    h = project_box_mass(new_h)
                    best_score = new_score
                    best_h = h.copy()
                    no_improve = 0
                    improved = True

        # Phase 3: Smooth optimization
        if not improved and iter_count % 5 == 0:
            h2, s2 = softmax_smooth_optimize(h, n_steps=200, lr=0.001, beta=80.0, rng=rng)
            if s2 < best_score - 1e-12:
                h = project_box_mass(h2)
                best_score = s2
                best_h = h.copy()
                no_improve = 0
                improved = True

        # Phase 4: Pair swaps
        if not improved:
            h3, s3 = pair_swap_optimize(h, rng, n_trials=50)
            if s3 < best_score - 1e-12:
                best_score = s3
                best_h = h3.copy()
                h = h3
                no_improve = 0
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                radius = max(0.01, radius * 0.65)
                no_improve = 0
                pert = rng.standard_normal(n) * radius * 0.3
                pert -= pert.mean()
                h_try = project_box_mass(h + pert)
                s_try = score_h_normalized(h_try)
                if s_try < best_score:
                    h = h_try
                    best_score = s_try
                    best_h = h.copy()
                else:
                    h = best_h.copy()

        # Multi-resolution exploration
        if iter_count % 25 == 0 and time.time() - start_time < budget_s - 50:
            for n_try in [40, 50, 60, 80, 90, 100, 120, 140]:
                if n_try == n:
                    continue
                h_m = reshape_candidate(h, n_try)
                s_m = score_h_normalized(h_m)
                if s_m < best_score - 5e-6:
                    h = h_m
                    n = n_try
                    best_score = s_m
                    best_h = h.copy()
                    no_improve = 0
                    radius = 0.06
                    break

    final_score = score_h_normalized(best_h)
    return best_h, final_score, len(best_h)
