import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def structured_binary(n, seed):
    """Create a structured binary sequence using modular arithmetic."""
    rng = np.random.RandomState(seed)
    # Try several different modular patterns
    # h[i] = 1 if (i * p + q) mod m < threshold
    best_h = None
    best_c5 = float('inf')
    for _ in range(50):
        p = rng.randint(1, n)
        q = rng.randint(0, n)
        m = rng.randint(n//2, n*2)
        h = np.zeros(n)
        for i in range(n):
            if ((i * p + q) % m) < m // 2:
                h[i] = 1
        # Adjust to have exactly n//2 ones
        if h.sum() > n//2:
            ones = np.where(h == 1)[0]
            rng.shuffle(ones)
            h[ones[:int(h.sum()) - n//2]] = 0
        elif h.sum() < n//2:
            zeros = np.where(h == 0)[0]
            rng.shuffle(zeros)
            h[zeros[:n//2 - int(h.sum())]] = 1
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h

def sa_binary(h, time_budget, seed=42):
    """Simulated annealing on binary sequences."""
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    start = time.time()
    
    # Adaptive temperature
    temp = 0.1
    no_improve = 0
    improve_threshold = 5000
    
    while time.time() - start < time_budget:
        # Pick random positions
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        
        # Swap
        h[i], h[j] = h[j], h[i]
        new_c5 = evaluate_c5(h)
        delta = new_c5 - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        
        # Cool down
        temp = max(temp * 0.99999, 1e-8)
        
        # Restart if stuck
        if no_improve > improve_threshold:
            temp = 0.1
            no_improve = 0
    
    return best_h, best_c5

def local_search_binary(h, time_budget, seed=42):
    """Steepest descent local search (only accept improvements)."""
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    
    while time.time() - start < time_budget:
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_c5 = evaluate_c5(h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            h[i], h[j] = h[j], h[i]
    
    return h, best_c5

def nonbinary_polish(h, time_budget, seed=42):
    """Polish a binary solution by allowing small non-binary perturbations."""
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = evaluate_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    
    while time.time() - start < time_budget:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Try small perturbation: h[i] += delta, h[j] -= delta
        best_delta = 0
        best_val = best_c5
        h_i_orig = h[i]
        h_j_orig = h[j]
        
        for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
            if h_i_orig + delta < 0 or h_i_orig + delta > 1:
                continue
            if h_j_orig - delta < 0 or h_j_orig - delta > 1:
                continue
            h[i] = h_i_orig + delta
            h[j] = h_j_orig - delta
            val = evaluate_c5(h)
            if val < best_val:
                best_val = val
                best_delta = delta
        
        h[i] = h_i_orig

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def sa_optimize(h_init, time_budget, seed=42):
    n = len(h_init)
    h = h_init.astype(float).copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    best_h = h.copy()

    start = time.time()
