    if s >= 0:
        for i in range(n - s):
            grad[i] += (1.0 - h[i + s])
            grad[i + s] += (-h[i])

            grad[i] += (1.0 - h[i - d])
            grad[i - d] += (-h[i])
    return grad

import numpy as np
import time
import random

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def gradient_single_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            grad[i] += (1.0 - h[i + s])
            grad[i + s] += (-h[i])
    else:
        d = -s
        for i in range(d, n):
            grad[i] += (1.0 - h[i - d])
            grad[i - d] += (-h[i])
    return grad

def project_h(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def continuous_optimize(h0, dx, time_limit=60):
    start = time.time()
    n = len(h0)
    target_sum = n / 2.0
    h = project_h(h0.copy(), target_sum)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.1
    top_k = 5
    while time.time() - start < time_limit:
        corr_vals = np.correlate(h, 1.0 - h, mode='full')
        sorted_idx = np.argsort(corr_vals)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in sorted_idx:
            s = idx - (n - 1)
            grad += gradient_single_shift(h, s)
        grad /= float(top_k)
        grad *= dx
        h = h - lr * grad
        h = project_h(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.9995
        if lr < 0.005:
            lr = 0.05
    return best_h, best_c5

        rng = random.Random(0)
    start = time.time()

    h = h_bin.copy().astype(np.int8)

    best_c5 = current_c5
    best_h = h.copy()
    ones = [int(i) for i in np.where(h == 1)[0]]
    zeros = [int(i) for i in np.where(h == 0)[0]]
    if not ones or not zeros:
        return h.astype(float), current_c5
    stall = 0
    while time.time() - start < time_limit:
        best_imp = 0.0
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h.astype(float), dx)
                h[i] = 1
                h[j] = 0
                if new_c5 < current_c5 - best_imp:
                    best_imp = current_c5 - new_c5
                    best_move = (i, j)
                    if best_imp > 0.015:
                        break

        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            current_c5 = compute_c5(h.astype(float), dx)

    return best_h.astype(float), best_c5

def random_balanced_binary(n, rng):
    positions = rng.sample(range(n), n // 2)
    h = np.zeros(n, dtype=np.int8)
    h[positions] = 1
    return h

def make_near_binary(h_cont, n):
    target = n // 2
    h = (h_cont > 0.5).astype(np.int8)
    n_ones = int(h.sum())
    if n_ones > target:
        one_pos = [int(i) for i in np.where(h == 1)[0]]
        one_pos.sort(key=lambda i: h_cont[i])
        for i in one_pos[:n_ones - target]:
            h[i] = 0
    elif n_ones < target:
        zero_pos = [int(i) for i in np.where(h == 0)[0]]
        zero_pos.sort(key=lambda i: -h_cont[i])
        for i in zero_pos[:target - n_ones]:
            h[i] = 1
    return h
