import math

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def make_feasible(y):
    n = len(y)
    y = np.array(y, dtype=np.float64).flatten()
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    z = np.clip(y - lam, 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-14:
        mask = (z > 1e-12) & (z < 1.0 - 1e-12)
        if mask.any():
            z[mask] += diff / mask.sum()
    z = np.clip(z, 0.0, 1.0)
    return z

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    center = n - 1
    base = float(np.max(c))
    ex = np.exp((c - base) / T)
    S = float(np.sum(ex))
    if S < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        smooth_val = base
    else:
        weights = ex / S
        smooth_val = base + T * math.log(S)
    grad = np.zeros(n)
    w0 = weights[center]
    grad += w0 * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = weights[center + d]
        wn = weights[center - d]
        grad[:m] += (wp * (-h[d:]) + wn * (1.0 - h[d:])) * dx
        grad[d:] += (wp * (1.0 - h[:m]) + wn * (-h[:m])) * dx
    return smooth_val, grad

def pair_line_search(cur, i, j, best_c):
    n = len(cur)
    i = int(i); j = int(j)
    lo_a = max(-cur[i], cur[j] - 1.0)
    hi_a = min(1.0 - cur[i], cur[j])
    if hi_a - lo_a < 1e-8:
        return cur, best_c
    save_i = cur[i]; save_j = cur[j]
    best_a = 0.0; best_lc = best_c
    alphas = np.linspace(lo_a, hi_a, 15)
    for a in alphas:
        cur[i] = save_i + a
        cur[j] = save_j - a
        if cur[i] < -1e-12 or cur[i] > 1.0 + 1e-12 or cur[j] < -1e-12 or cur[j] > 1.0 + 1e-12:
            cur[i] = save_i; cur[j] = save_j
            continue
        lc = score_exact(cur)
        if lc < best_lc:
            best_lc = lc; best_a = a
    cur[i] = save_i; cur[j] = save_j
    if best_lc < best_c - 1e-13:
