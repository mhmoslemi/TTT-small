def compute_c5(h):
    """Compute C5 exactly as defined."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {sum=h=n/2, 0<=h<=1} using bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_obj(h, beta=None):
    """Gradient of the objective.
    If beta is None: gradient of max(corr)*dx
    If beta given: gradient of (1/beta)*log(sum exp(beta*corr*dx))*dx... 
    Actually let me just do gradient of max for simplicity.

        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
        h_curr = h_trial

            pass

def slsqp_polish(h, n):
    """Polish with SLSQP using max objective."""
    target = n / 2.0
    
    def objective(x):
        h_x = np.clip(x, 0, 1)
        # Penalize constraint violation
        sum_viol = abs(np.sum(h_x) - target)
        corr = np.correlate(h_x, 1 - h_x, mode="full")
        return np.max(corr) * (2.0 / n) + 10 * sum_viol

def solve_sdp(n, initial_h=None):
    target = n / 2.0
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    X = cp.Variable((n+1, n+1), symmetric=True)
    
    constraints = [
        X >> 0,  # PSD
        X[n, n] == 1,  # last entry
    ]
    
    # Link h_var to X: in exact case, X[i,j] = h[i]*h[j], X[i,n] = h[i]*1 = h[i]
    # For relaxation, we use: X[i,n] == h[i] (this is the "exact" linking)
    # But this makes it non-convex in general... 
    # Actually, X[i,n] = h[i] is a linear constraint on X and h, so it's fine.
    # The PSD constraint then implies X[i,j] ≥ h[i]*h[j] (by the fact that
    # the 2×2 minor [X[i,j], h[i]; h[j], 1] ⪰ 0)
    constraints += [X[:n, n] == h_var]
    
    # Sum constraint
    constraints += [cp.sum(h_var) == target]
    
    # Bound constraints
    constraints += [h_var >= 0, h_var <= 1]
    
    # Correlation constraints: sum_i h[i](1-h[i+d]) ≤ t/dx for all d
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            # sum_{i=0}^{n-1-d} h[i] - sum_{i=0}^{n-1-d} h[i]*h[i+d] ≤ t/dx
            # = sum h[i] (for i in range) - sum X[i, i+d] (for i in range)
            idx = range(n - d)
            lhs = cp.sum(h_var[list(idx)]) - cp.sum([X[i, i+d] for i in idx])
        else:
            da = -d
            # sum_{i=0}^{n-1-da} h[i+da] - sum_{i=0}^{n-1-da} h[i+da]*h[i]
            idx = range(n - da)
            lhs = cp.sum(h_var[list(i + da for i in idx)]) - cp.sum([X[i+da, i] for i in idx])
        
        constraints += [lhs <= t_var / dx]
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except:
            return None, None
    
    if X.value is None or h_var.value is None:
        return None, None
    
    h_sol = np.clip(h_var.value, 0, 1)
    h_sol = project_feasible(h_sol, n)
    c5_sol = compute_c5(h_sol)

                grad[i] += w * (1.0 - h[i - da])
