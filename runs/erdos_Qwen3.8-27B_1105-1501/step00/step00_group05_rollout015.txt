    corr = np.correlate(h, 1 - h, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project(v, s):
    """Project v onto {h: 0<=h<=1, sum(h)=s} via bisection on lambda."""
    def f(lam):
        return np.sum(np.clip(v - lam, 0.0, 1.0)) - s
    lo, hi = v.min() - 1.0, v.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj_and_grad(h, dx, beta):
    """Compute smoothed max objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    mx = np.max(corr)
    exps = np.exp(beta * (corr - mx))
    wsum = np.sum(exps)
    weights = exps / wsum
    obj = dx * (mx + np.log(wsum) / beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-16:
            continue
        lag = k - (n - 1)
        if lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        elif lag < 0:
            lp = -lag
            grad[lp:] += w * (1.0 - h[:n - lp])
            grad[:n - lp] -= w * h[lp:]
        else:
            grad += w * (1.0 - 2.0 * h)
    return obj, dx * grad

def local_search_fine(h, s, max_passes, time_limit_fn, deadline):

    n = len(h)
    best_c5 = compute_c5(h)
    deltas_coarse = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
    deltas_fine = [0.001, -0.001, 0.0005, -0.0005, 0.0002, -0.0002, 0.0001, -0.0001]

    for passage in range(max_passes):
        if time_limit_fn() > deadline:
            break
        improved = False
        use_fine = passage > max_passes // 3
        deltas = deltas_fine if use_fine else deltas_coarse

        for i in range(n):
            base = h[i]
            found = False
            for d in deltas:
                t = base + d
                if t <= 0.0 or t >= 1.0:
                    continue
                h_trial = h.copy()
                h_trial[i] = t
                h_trial = project(h_trial, s)
                if np.max(np.abs(h_trial - h)) < 1e-13:
                    continue
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-14:
                    h = h_trial.copy()
                    best_c5 = c5_t
                    improved = True
                    found = True
                    break
            if found:
                continue
            # Try endpoints
            if base > 0.005:
                h_trial = h.copy()
                h_trial[i] = 0.0
                h_trial = project(h_trial, s)
                if np.max(np.abs(h_trial - h)) > 1e-13:
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        h = h_trial.copy()
                        best_c5 = c5_t
                        improved = True
            if base < 0.995:
                h_trial = h.copy()
                h_trial[i] = 1.0
                h_trial = project(h_trial, s)
                if np.max(np.abs(h_trial - h)) > 1e-13:
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        h = h_trial.copy()
                        best_c5 = c5_t
                        improved = True

        # Pair moves every few passes
        if not improved and passage % 5 == 4:
            for _ in range(n):
                if time_limit_fn() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                for d in [0.01, -0.01, 0.02, -0.02]:
                    h_trial = h.copy()
                    h_trial[i] += d
                    h_trial[j] -= d
                    if h_trial[i] < 0 or h_trial[i] > 1 or h_trial[j] < 0 or h_trial[j] > 1:
                        continue
                    h_trial = project(h_trial, s)
                    if np.max(np.abs(h_trial - h)) < 1e-13:
                        continue
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        h = h_trial.copy()
                        best_c5 = c5_t
                        improved = True
                        break

        if not improved and passage > 3:
            # Random restart from current point to escape local optima
            if passage % 15 == 14:
                idx = np.random.randint(n)
                h[idx] = np.clip(h[idx] + np.random.uniform(-0.08, 0.08), 0, 1)
                h = project(h, s)

    return h, best_c5

        if elapsed() > budget_s * 0.92:

        s = n / 2.0
        dx = 2.0 / n

        # Starting point
        if n == best_n and best_h is not None and len(best_h) == n:
            h_cur = best_h.copy()
        elif h0 is not None and n == len(h0):

        # Phase 1: SLSQP with increasing beta
        deadline_1 = t_init_deadline if n == (len(h0) if h0 is not None else 42) else t_global_deadline

                return g

        best_n = 42

    return best_h, best_c5, best_n
