def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_scores(h)) * 2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_L(h, p, q, n):
    """Compute L vector for pair (p,q)."""
    d = np.arange(2*n-1) - (n-1)
    idx1 = p - d
    idx2 = q - d
    idx3 = p + d
    idx4 = q + d
    L = np.zeros(2*n-1)
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    m3 = (idx3 >= 0) & (idx3 < n)
    m4 = (idx4 >= 0) & (idx4 < n)
    L[m1] += (1.0 - h[idx1[m1]])
    L[m2] -= (1.0 - h[idx2[m2]])
    L[m3] -= h[idx3[m3]]
    L[m4] += h[idx4[m4]]
    return L

def _pair_D(p, q, n):
    """Return array of D values (mostly zero)."""
    D = np.zeros(2*n-1)
    D[n-1] = 2.0  # lag 0
    k1 = n - 1 + (p - q)
    D[k1] += -1.0
    k2 = n - 1 + (q - p)
    D[k2] += -1.0
    return D

def _find_best_alpha(scores, L, D, a_max):
    """Minimize max(scores + alpha*L - alpha^2*D) for alpha in [0, a_max]."""
    if a_max < 1e-14:
        return 0.0, float(scores.max())
    cur_max = float(scores.max())
    
    # Grid search
    n_g = 16
    alphas = np.linspace(0.0, a_max, n_g + 1)
    # Compute max for each alpha
    best_a = 0.0
    best_m = cur_max
    for i in range(1, n_g + 1):
        a = alphas[i]
        s = scores + a * L
        # Apply D corrections (D is sparse: only 3 non-zero entries)
        nz = np.nonzero(D)[0]
        if len(nz) > 0:
            s[nz] -= a * a * D[nz]
        m = float(s.max())
        if m < best_m:
            best_m = m
            best_a = a
    
    if best_a <= 0:
        return 0.0, cur_max
    
    # Golden section refine
    lo = max(0.0, best_a - a_max / n_g)
    hi = min(a_max, best_a + a_max / n_g)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def f(a):
        s = scores + a * L
        nz = np.nonzero(D)[0]
        if len(nz) > 0:
            s[nz] -= a * a * D[nz]
        return float(s.max())
    
    f1 = f(c1)

    for _ in range(30):

            f1 = f(c1)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    a_opt = (lo + hi) / 2.0
    f_opt = f(a_opt)
    if f_opt < best_m:
        best_a = a_opt
        best_m = f_opt
    
    return best_a, best_m

def _pair_descent(h, deadline, max_sweeps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        cur_max = float(scores.max())
        improved = False
        best_in_sweep = cur_max
        
        # Try all pairs
        for p in range(n):
            for q in range(p + 1, n):
                if time.time() > deadline:
                    break
                
                # Compute L and D for this pair
                L = _pair_L(h, p, q, n)
                k1 = n - 1 + (p - q)
                k2 = n - 1 + (q - p)
                D_nz_idx = np.array([n-1, k1, k2])
                D_nz_vals = np.array([2.0, -1.0, -1.0])
                
                # Positive direction: h[p] += alpha, h[q] -= alpha
                a_max_pos = min(1.0 - h[p], h[q])
                if a_max_pos > 1e-12:
                    a_opt, m_opt = _find_best_alpha(scores, L, D_nz_idx, D_nz_vals, a_max_pos)
                    if m_opt < best_in_sweep - 1e-13:
                        # Apply
                        scores = scores + a_opt * L
                        scores[D_nz_idx] -= a_opt**2 * D_nz_vals
                        h[p] += a_opt
                        h[q] -= a_opt
                        best_in_sweep = m_opt
                        improved = True
                
                # Negative direction: h[p] -= alpha, h[q] += alpha
                a_max_neg = min(h[p], 1.0 - h[q])
                if a_max_neg > 1e-12:
