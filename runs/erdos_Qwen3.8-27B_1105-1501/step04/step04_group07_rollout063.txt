import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    """Project v onto {x: 0≤x≤1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection on threshold
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_step(h, i, j):
    """Compute optimal alpha for direction h[i]+a, h[j]-a.
    Returns (alpha, new_max_score)."""
    n = len(h)
    # Feasible range
    alpha_lo = max(-h[i], h[j] - 1.0)
    alpha_hi = min(1.0 - h[i], h[j])
    if alpha_hi - alpha_lo < 1e-13:
        return 0.0, None
    
    # Affected lags
    affected = set()
    for d in [0, i-j, j-i]:
        idx = n - 1 + d
        if 0 <= idx < 2*n - 1:
            affected.add(idx)
    
    scores = _scores(h)
    s_all = scores
    
    # Unchanged max
    if len(affected) < 2*n - 1:
        mask = np.ones(2*n-1, dtype=bool)
        mask[list(affected)] = False
        unchanged_max = float(s_all[mask].max())
    else:
        unchanged_max = 0.0
    
    # Coefficients for affected lags
    coeffs = {}
    for d in [0, i-j, j-i]:
        idx = n - 1 + d
        if not (0 <= idx < 2*n - 1):
            continue
        s_cur = float(s_all[idx])
        
        lin = 0.0
        if 0 <= i - d < n:
            lin += (1.0 - h[i - d])
        if 0 <= i + d < n:
            lin -= h[i + d]
        if 0 <= j - d < n:
            lin -= (1.0 - h[j - d])
        if 0 <= j + d < n:
            lin += h[j + d]
        
        quad = 0.0
        if d == j - i:
            quad += 1.0  # k=i, k+d=j: (h[j]-a)(1-h[i]-a) → (-a)(-a)=a²
        if d == i - j:
            quad += 1.0  # k=j, k+d=i: (h[i]+a)(1-h[j]+a) → (a)(a)=a²
        if d == 0:
            quad -= 1.0  # k=i: (h[i]+a)(1-h[i]-a) → (a)(-a)=-a²
            quad -= 1.0  # k=j: (h[j]-a)(1-h[j]+a) → (-a)(a)=-a²
        
        coeffs[idx] = (lin, quad, s_cur)
    
    # Solve 1D problem
    aff_list = list(coeffs.keys())
    candidates = [alpha_lo, alpha_hi]
    
    # Pairwise quadratic intersections
    for p in range(len(aff_list)):
        for q in range(p+1, len(aff_list)):
            lp, qp, sp = coeffs[aff_list[p]]
            lq, qq, sq = coeffs[aff_list[q]]
            A = qp - qq
            B = lp - lq
            C = sp - sq
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    candidates.append(-C / B)
            else:
                disc = B*B - 4*A*C
                if disc >= -1e-14:
                    disc = max(0, disc)
                    sd = np.sqrt(disc)
                    candidates.append((-B + sd) / (2*A))
                    candidates.append((-B - sd) / (2*A))
    
    # Quadratic-constant intersections
    for idx in aff_list:
        l, q, s = coeffs[idx]
        A = q
        B = l
        C = s - unchanged_max
        if abs(A) < 1e-14:
            if abs(B) > 1e-14:
                candidates.append(-C / B)
        else:
            disc = B*B - 4*A*C
            if disc >= -1e-14:
                disc = max(0, disc)
                sd = np.sqrt(disc)
                candidates.append((-B + sd) / (2*A))
                candidates.append((-B - sd) / (2*A))
    
    # Evaluate
    best_alpha = 0.0
    best_val = float('inf')
    for a in candidates:
        if a < alpha_lo - 1e-10 or a > alpha_hi + 1e-10:
            continue
        a = float(np.clip(a, alpha_lo, alpha_hi))
        val = unchanged_max
        for idx in aff_list:
            l, q, s = coeffs[idx]
            v = s + l*a + q*a*a
            if v > val:
                val = v
        if val < best_val:
            best_val = val
            best_alpha = a
    
    # Safety: also check a few grid points
    for a in np.linspace(alpha_lo, alpha_hi, 8):
        val = unchanged_max
        for idx in aff_list:
            l, q, s = coeffs[idx]
            v = s + l*a + q*a*a
            if v > val:
                val = v
        if val < best_val:
            best_val = val
            best_alpha = a
    
    return best_alpha, best_val

def _pair_descent(h, deadline, max_sweeps=500, eps=1e-12):
    """Cyclic pair-optimal coordinate descent."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    dx = 2.0 / n
    stall_count = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improvement = 0.0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                alpha, new_val = _pair_step(h, i, j)
                if new_val is None:
                    continue
                if new_val < cur_c5 - eps:
                    h[i] += alpha
                    h[j] -= alpha
                    h = np.clip(h, 0.0, 1.0)
                    # Re-project to maintain sum
                    h = _project(h, n/2.0)
                    cur_c5 = _exact_c5(h)
                    sweep_improvement += (cur_c5 - new_val)  # might be 0 or slightly negative
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if sweep_improvement < eps:
            stall_count += 1
            if stall_count > 3:
                break
        else:
            stall_count = 0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_coeffs(h, i, j, scores):
    """Compute linear and quadratic coefficients for pair step h[i]+a, h[j]-a."""
    n = len(h)
    affected = []
    for d in [0, i - j, j - i]:
        idx = n - 1 + d
        if 0 <= idx < 2 * n - 1:
            if idx not in affected:
                affected.append(idx)
    coeffs = {}
    for d in [0, i - j, j - i]:
        idx = n - 1 + d
        if not (0 <= idx < 2 * n - 1):
            continue
        s_cur = float(scores[idx])
        lin = 0.0
        if 0 <= i - d < n:
            lin += (1.0 - h[i - d])
        if 0 <= i + d < n:
            lin -= h[i + d]
        if 0 <= j - d < n:
            lin -= (1.0 - h[j - d])
        if 0 <= j + d < n:
            lin += h[j + d]
        quad = 0.0
        if d == j - i:
            quad += 1.0
        if d == i - j:
            quad += 1.0
        if d == 0:
            quad -= 2.0
        coeffs[idx] = (lin, quad, s_cur)
    return affected, coeffs

def _solve_1d_pair(affected, coeffs, unchanged_max, alpha_lo, alpha_hi):
    """Find alpha minimizing max(unchanged_max, max affected quadratics)."""
    aff_list = list(coeffs.keys())
    if not aff_list:
        return 0.0, unchanged_max
    candidates = [alpha_lo, alpha_hi]
    for p in range(len(aff_list)):
        for q in range(p + 1, len(aff_list)):
            lp, qp, sp = coeffs[aff_list[p]]
            lq, qq, sq = coeffs[aff_list[q]]
            A = qp - qq
            B = lp - lq
            C = sp - sq
            if abs(A) < 1e-14:
                if abs(B) > 1e-14:
                    candidates.append(-C / B)
            else:
                disc = B * B - 4 * A * C
                if disc >= -1e-14:
                    sd = np.sqrt(max(0.0, disc))
                    candidates.append((-B + sd) / (2 * A))
                    candidates.append((-B - sd) / (2 * A))
    for idx in aff_list:
        l, q, s = coeffs[idx]
        A = q
        B = l
        C = s - unchanged_max
        if abs(A) < 1e-14:
            if abs(B) > 1e-14:
                candidates.append(-C / B)
        else:
            disc = B * B - 4 * A * C
            if disc >= -1e-14:
                sd = np.sqrt(max(0.0, disc))
                candidates.append((-B + sd) / (2 * A))
                candidates.append((-B - sd) / (2 * A))
    best_alpha = 0.0
    best_val = float('inf')
    for a in candidates:
        if a < alpha_lo - 1e-10 or a > alpha_hi + 1e-10:
            continue
        a = float(np.clip(a, alpha_lo, alpha_hi))
        val = unchanged_max
        for idx in aff_list:
            l, q, s = coeffs[idx]
            v = s + l * a + q * a * a
            if v > val:
                val = v
        if val < best_val:
            best_val = val
            best_alpha = a
    for a in np.linspace(alpha_lo, alpha_hi, 6):
        val = unchanged_max
        for idx in aff_list:
            l, q, s = coeffs[idx]
            v = s + l * a + q * a * a
            if v > val:
                val = v
        if val < best_val:
            best_val = val
            best_alpha = a
    return best_alpha, best_val

def _pair_descent(h, deadline, max_sweeps=3000):
    """Fast incremental pair-optimal coordinate descent."""
    n = len(h)
    h = h.copy()
    scores = _scores(h)
    cur_max = float(scores.max())
    best_h = h.copy()
    best_c5 = cur_max
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                alpha_lo = max(-h[i], h[j] - 1.0)
                alpha_hi = min(1.0 - h[i], h[j])
                if alpha_hi - alpha_lo < 1e-13:
                    continue
                affected, coeffs = _pair_coeffs(h, i, j, scores)
                if len(affected) < 2 * n - 1:
                    mask = np.ones(2 * n - 1, dtype=bool)
                    mask[affected] = False
                    unchanged_max = float(scores[mask].max())
                else:
                    unchanged_max = -np.inf
                best_a, new_val = _solve_1d_pair(affected, coeffs, unchanged_max, alpha_lo, alpha_hi)
                if new_val < cur_max - 1e-13 and abs(best_a) > 1e-14:
                    h[i] += best_a
                    h[j] -= best_a
                    np.clip(h, 0.0, 1.0, out=h)
                    for idx in affected:
                        l, q, s = coeffs[idx]
                        scores[idx] = s + l * best_a + q * best_a * best_a
                    cur_max = max(float(scores.max()), new_val)
                    improved = True
                    if cur_max < best_c5:
                        best_c5 = cur_max
                        best_h = h.copy()
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c5

def _multi_coord_step(h, V, scores, cur_max):
    """Exact 1D optimization along zero-sum direction V."""
    n = len(h)
    alpha_lo, alpha_hi = 0.0, 0.0
    for i in range(n):
        if V[i] > 1e-15:
            alpha_lo = max(alpha_lo, -h[i] / V[i])
            alpha_hi = min(alpha_hi, (1.0 - h[i]) / V[i])
        elif V[i] < -1e-15:
            alpha_lo = max(alpha_lo, (1.0 - h[i]) / V[i])
            alpha_hi = min(alpha_hi, -h[i] / V[i])
    if alpha_hi - alpha_lo < 1e-12:
        return 0.0, None
    a_coeffs = np.correlate(V, 1.0 - h, mode='full') - np.correlate(h, V, mode='full')
    b_coeffs = -np.correlate(V, V, mode='full')
    dx = 2.0 / n
    s0 = scores
    n_grid = 300
    alphas = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = np.zeros(n_grid)
    for k in range(n_grid):
        a = alphas[k]
        vals[k] = float(np.max(s0 + a * a_coeffs + a * a * b_coeffs))
    best_k = int(np.argmin(vals))
    best_a = float(alphas[best_k])
    best_v = float(vals[best_k])
    if best_k > 0 and best_k < n_grid - 1:
        lo = float(alphas[best_k - 1])
        hi = float(alphas[best_k + 1])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(40):
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = float(np.max(s0 + c1 * a_coeffs + c1 * c1 * b_coeffs))
            f2 = float(np.max(s0 + c2 * a_coeffs + c2 * c2 * b_coeffs))
            if f1 < f2:
                hi, f2, c2 = c2, f1, c1
            else:
                lo, f1, c1 = c1, f2, c2
            if hi - lo < 1e-13:
                break
        ma = (lo + hi) / 2.0
        mv = float(np.max(s0 + ma * a_coeffs + ma * ma * b_coeffs))
        if mv < best_v:
            best_a, best_v = ma, mv
    return best_a, best_v * dx

def _multi_coord_descent(h, deadline, rng, n_dirs=5000, dim_range=(2, 6)):
    """Random multi-coordinate exact 1D steps."""
    n = len(h)
    h = h.copy()
    scores = _scores(h)
    cur_max = float(scores.max())
    best_h = h.copy()
    best_c5 = cur_max
    stall = 0
    for it in range(n_dirs):
        if time.time() > deadline:
            break
        dim = int(rng.integers(dim_range[0], dim_range[1] + 1))
        idx = rng.choice(n, size=dim, replace=False)
        V = np.zeros(n)
        idx_sorted = np.sort(idx)
        coeffs_v = rng.normal(size=dim)
        coeffs_v -= coeffs_v.mean()
        for ii, ci in zip(idx_sorted, coeffs_v):
            V[int(ii)] = ci
        gnorm = np.linalg.norm(V)
        if gnorm < 1e-10:
            continue
        V /= gnorm
        a, new_val = _multi_coord_step(h, V, scores, cur_max)
        if new_val is not None and new_val < cur_max - 1e-13:
            h += a * V
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, n / 2.0)
            scores = _scores(h)
            cur_max = float(scores.max())
            stall = 0
            if cur_max < best_c5:
                best_c5 = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > 2000:
                break
    return best_h, best_c5

def _lsm_gradient(h, scores, eps):
    """Compute gradient of log-sum-exp of scores w.r.t. h."""
    n = len(h)
    m = scores.max()
    shifted = (scores - m) / eps
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        for i in range(n):
            g_ri = 0.0
            if 0 <= i - d < n:
                g_ri += (1.0 - h[i - d])
            if 0 <= i + d < n:
                g_ri -= h[i + d]
            grad[i] += w[r] * g_ri * dx / eps
    return grad

def _lsm_descent(h, deadline, n_iters=2000):
    """Log-sum-exp smoothed descent with adaptive step size."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    eps = 0.02
    lr = 0.05
    stall = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        grad = _lsm_gradient(h, scores, eps)
        grad -= np.mean(grad)
        h_new = h - lr * grad
        h_new = _project(h_new, n / 2.0)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = new_c5
            lr = min(lr * 1.05, 0.3)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            lr *= 0.7
            stall += 1
            if stall > 100:
                break
        if it % 80 == 79:
            eps *= 0.88
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_hill_climb(n, rng, deadline, max_iters=50000):
    """Hill-climb a binary sequence."""
    target = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=target, replace=False)
    h[idx] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def _lp_local(h, deadline, top_k=8, radius=0.05, max_iters=2000):
    """Local LP-based refinement."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        n_lags = 2 * n - 1
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
        else:
            idx = np.arange(n_lags)
            sk = scores
        G = np.zeros((len(idx), n))
        for ii, r_idx in enumerate(idx):
            d = r_idx - (n - 1)
            for i in range(n):
                g_ri = 0.0
                if 0 <= i - d < n:
                    g_ri += (1.0 - h[i - d])
                if 0 <= i + d < n:
                    g_ri -= h[i + d]
                G[ii, i] = g_ri
        G *= 2.0 / n
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-12:
            stall += 1
            r *= 0.7
            if r < 1e-6:
                break
            continue
        h_try = h + p
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < cur_c5 - 1e-13:
            h = h_try
            cur_c5 = c5_try
            stall = 0
            r = min(r * 1.05, 0.3)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            r *= 0.8
            if stall > 30 and r < 0.01:
                r = 0.06
    return best_h, best_c5
