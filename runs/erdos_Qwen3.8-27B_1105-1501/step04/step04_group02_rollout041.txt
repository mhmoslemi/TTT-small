def _line_search(h, p, c5_cur, alpha_max):
    """Find best step size along direction p."""
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    
    # Grid search
    n_grid = 40
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    
    # Golden section refinement
    lo_t = ts[max(0, idx - 3)]
    hi_t = ts[min(n_grid - 1, idx + 3)]
    if hi_t <= lo_t:
        return best_t, best_val
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    
    def _eval_h_t(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    
    f1, f2 = _eval_h_t(c1), _eval_h_t(c2)
    for _ in range(40):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _eval_h_t(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _eval_h_t(c2)
    
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _eval_h_t(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    
    return best_t, best_val

                best_h = h.copy()

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, :m] -= h[d:]
            G[r, d:] += (1.0 - h[:m])
        else:
            s = -d
            m = n - s
            G[r, :m] += (1.0 - h[s:s + m])
            G[r, s:] -= h[:m]
    return G

def _lp_direction(h, scores, radius):
    n = len(h)
    G = _grad_matrix(h)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 3)]
    hi_t = ts[min(n_grid - 1, idx + 3)]
    if hi_t - lo_t < 1e-12:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)

    def _eval_at(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)

    f1, f2 = _eval_at(c1), _eval_at(c2)
    for _ in range(45):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _eval_at(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _eval_at(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _eval_at(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_local_search(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline and stall < 120:
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.55
            if radius < 1e-9:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.12, 0.6)
        else:
            stall += 1
            radius *= 0.6
    return best_h, best_c5

class _SLSQPCallbacks:
    def __init__(self, n, target):
        self.n = n
        self.target = target

    def objective(self, x):
        return x[self.n]

    def objective_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def ineq_fun(self, x):
        hvec = x[:self.n]
        T = x[self.n]
        sc = np.correlate(hvec, 1.0 - hvec, mode='full')
        return T - sc

    def ineq_jac(self, x):
        hvec = x[:self.n]
        G = _grad_matrix(hvec)
        jac = np.zeros((2 * self.n - 1, self.n + 1))
        jac[:, :self.n] = -G
        jac[:, self.n] = 1.0
        return jac

    def eq_fun(self, x):
        return np.array([np.sum(x[:self.n]) - self.target])

    def eq_jac(self, x):
        j = np.zeros((1, self.n + 1))
        j[0, :self.n] = 1.0
        return j

def _slsqp_epigraph(h, deadline, maxiter=60):
    n = len(h)
    target = n / 2.0
    if time.time() >= deadline:
        return h, _exact_c5(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    T0 = float(np.max(scores)) + 0.05
    cb = _SLSQPCallbacks(n, target)
    x0 = np.concatenate([h, [T0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'ineq', 'fun': cb.ineq_fun, 'jac': cb.ineq_jac},
        {'type': 'eq', 'fun': cb.eq_fun, 'jac': cb.eq_jac}
    ]
    try:
        res = minimize(cb.objective, x0, method='SLSQP', jac=cb.objective_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(np.clip(res.x[:n], 0.0, 1.0), target)
            c5_new = _exact_c5(h_new)
            if c5_new <= _exact_c5(h) + 1e-12:
                return h_new, c5_new
    except Exception:
        pass
    return h, _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_step(h, w, lr):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] -= wr * h[d:]
            grad[d:] += wr * (1.0 - h[:m])
        else:
            s = -d
            m = n - s
            grad[:m] += wr * (1.0 - h[s:s + m])
            grad[s:] -= wr * h[:m]
    h_new = h - lr * grad * dx
    return _project(h_new, target)

def _grad_descent(h, deadline, tau=20.0, lr=0.03, mom=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    velocity = np.zeros(n)
    stall = 0
    while time.time() < deadline and stall < 60:
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(np.clip(shifted, -500, 0))
        w = exp_s / float(np.sum(exp_s))
        h_new = _grad_step(h, w, lr)
        velocity = mom * velocity + (1 - mom) * (h_new - h)
        h_cand = _project(h - lr * velocity, target)
        c5_cand = _exact_c5(h_cand)
        if c5_cand < cur_c5 - 1e-14:
            h = h_cand
            cur_c5 = c5_cand
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.15)
            tau = min(tau * 1.02, 200.0)
        else:
            stall += 1
            lr *= 0.9
            if stall % 20 == 0:
                tau = max(10.0, tau * 0.7)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(_pass * 7919 + 13)
        n_pairs = min(n * 3, 200)
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                continue
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9 or h[k] <= 1e-9 or h[k] >= 1.0 - 1e-9:
                continue
            for sign in [1, -1]:
                if sign > 0:
                    am = min((1.0 - h[j]), h[k])
                else:
                    am = min(h[j], (1.0 - h[k]))
                if am < 1e-12:
                    continue
                for frac in [0.5, 1.0]:
                    h_t = h.copy()
                    h_t[j] += sign * am * frac
                    h_t[k] -= sign * am * frac
                    if h_t[j] < -1e-12 or h_t[j] > 1.0 + 1e-12 or h_t[k] < -1e-12 or h_t[k] > 1.0 + 1e-12:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _perturb_optimize(h_best, deadline, seed=42):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = 0.06
    attempt = 0
    time_per_attempt = max(1.0, (deadline - time.time()) / 30.0)
    while time.time() < deadline and attempt < 50:
        attempt += 1
        if best_c5 < 0.3805:
            break
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, target)
        sub_dl = min(deadline, time.time() + time_per_attempt)
        sub_dl_lp = min(deadline, time.time() + time_per_attempt * 0.7)
        h_try, c5_try = _lp_local_search(h_p, sub_dl_lp, radius_init=0.08)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.3, scale * 1.3)
    return best_h, best_c5

def _make_smooth_init(n, seed):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / n
    coeffs1 = rng.normal(0, 0.3, 5)
    coeffs2 = rng.normal(0, 0.2, 5)
    h = np.zeros(n)
    for k in range(5):
        h += coeffs1[k] * np.sin((k + 1) * np.pi * 2.0 * x)
        h += coeffs2[k] * np.cos((k + 1) * np.pi * 2.0 * x)
    h = 0.5 + h
    return _project(h, n / 2.0)

def _make_binary_init(n, seed):
    rng = np.random.default_rng(seed)
    n_ones = n // 2
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n_ones]] = 1.0
    h[idx[n_ones:n_ones + 1]] = 0.0
    return _project(h, n / 2.0)

def _multi_resolution(h_best, n_best, deadline, budget_alloc):
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    best_n = n_best
    time_per = budget_alloc / 14.0
    if n_best % 2 == 0:
        offsets = [-10, -8, -6, -4, -2, 2, 4, 6, 8, 10, -12, 12, -14, 14]
    else:
        offsets = [-9, -7, -5, -3, -1, 1, 3, 5, 7, 9, -11, 11]
    tried = set()
    for dn in offsets:
        if time.time() >= deadline:
            break
        n_try = n_best + dn
        if n_try < 36 or n_try > 180 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl1 = min(deadline, time.time() + time_per * 0.4)
        h_r, c5_r = _lp_local_search(h_r, sub_dl1, radius_init=0.12)
        if time.time() < deadline and c5_r < best_c5 + 0.005:
            sub_dl2 = min(deadline, time.time() + time_per * 0.3)
            h_r, c5_r = _slsqp_epigraph(h_r, sub_dl2, maxiter=30)
            h_r, c5_r = _lp_local_search(h_r, min(deadline, time.time() + time_per * 0.3), radius_init=0.08)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    return best_h, best_c5, best_n

def _gradient_warmup(h, deadline):
    n = len(h)
    h = h.copy()
    return _grad_descent(h, deadline, tau=15.0, lr=0.025, mom=0.88)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline_hard = t0 + budget_s * 0.98

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
    else:
        h0 = _make_smooth_init(90, seed)
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Phase 1: Gradient warmup (5%)
    d1 = t0 + budget_s * 0.05
    h, c5 = _gradient_warmup(best_h, d1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: LP local search (15%)
    d2 = t0 + budget_s * 0.20
    h, c5 = _lp_local_search(best_h, d2, radius_init=0.15)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: SLSQP epigraph (10%)
    d3 = t0 + budget_s * 0.30
    h, c5 = _slsqp_epigraph(best_h, d3, maxiter=50)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        # Follow up with LP
        d3b = min(deadline_hard, time.time() + budget_s * 0.05)
        h, c5 = _lp_local_search(best_h, d3b, radius_init=0.1)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Multi-resolution (25%)
    d4 = t0 + budget_s * 0.55
    best_h, best_c5, best_n = _multi_resolution(best_h, best_n, d4, budget_s * 0.25)

    # Phase 5: Intensive LP + SLSQP at best resolution (15%)
    d5 = t0 + budget_s * 0.70
    h, c5 = _lp_local_search(best_h, d5, radius_init=0.12)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d5:
        h, c5 = _slsqp_epigraph(best_h, min(d5, time.time() + budget_s * 0.08), maxiter=40)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            h, c5 = _lp_local_search(best_h, min(d5, time.time() + budget_s * 0.04), radius_init=0.06)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    if time.time() < d5:
        h, c5 = _lp_local_search(best_h, d5, radius_init=0.05)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Perturbation + refine (12%)
    d6 = t0 + budget_s * 0.82
    best_h, best_c5 = _perturb_optimize(best_h, d6, seed=seed + 100)

    # Phase 7: Final polishing (12%)
    d7 = t0 + budget_s * 0.94
    h, c5 = _lp_local_search(best_h, min(d7, time.time() + budget_s * 0.03), radius_init=0.03)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d7:
        h, c5 = _pair_refine(best_h, min(d7, time.time() + budget_s * 0.04))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < d7:
        h, c5 = _slsqp_epigraph(best_h, min(d7, time.time() + budget_s * 0.03), maxiter=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            h, c5 = _lp_local_search(best_h, min(d7, time.time() + budget_s * 0.02), radius_init=0.02)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final LP polish with remaining time
    if time.time() < deadline_hard:
        h, c5 = _lp_local_search(best_h, deadline_hard, radius_init=0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)

    # Validate
    if not np.all(np.isfinite(best_h)) or best_h.min() < -1e-9 or best_h.max() > 1.0 + 1e-9:
        best_h = _project(h0, n0 / 2.0)
        best_n = n0
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
