                tau = max(0.0001, tau * 0.5)
    
    return best_h, best_c5

def _lp_direction_small(h, scores, top_k, radius):
    """LP direction using only top_k active lags."""
    n = len(h)
    # Get top_k lags
    top_idx = np.argsort(scores)[::-1][:top_k]
    
    # Build gradient matrix for these lags only
    G = np.zeros((len(top_idx), n))
    for ii, r in enumerate(top_idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[ii, d:] += 1.0
            G[ii, d:] -= h[:m]
            G[ii, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            G[ii, :m] += 1.0
            G[ii, :m] -= h[g:]
            G[ii, g:] -= h[:m]
    
    # LP: minimize t
    # s.t. G @ p + scores[top_idx] <= t
    #      |p[i]| <= radius
    #      sum(p) = 0
    # Variables: p (n) + t (1) = n+1
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0  # minimize t
    
    A_ub = np.zeros((len(top_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize_small(h, deadline, target, top_k=15, radius_init=0.1):
    """LP-based local search using only top_k lags."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline and stall < 60:
        sc = _scores(h)
        p = _lp_direction_small(h, sc, top_k, radius)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        # Line search: h + t*p, t ∈ [0, t_max]
        t_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                t_max = min(t_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                t_max = min(t_max, h[j] / (-p[j]))
        t_max = max(t_max, 1e-10)
        
        # Grid + golden section
        best_t, best_val = 0.0, cur_c5
        n_grid = 20
        ts = np.linspace(0, t_max, n_grid)
        for t in ts:
            h_t = h + t * p
            if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
                val = _exact_c5(h_t)
                if val < best_val:
                    best_val = val
                    best_t = t
        
        # Refine with golden section
        idx = int(np.argmin([_exact_c5(h + t*p) if h+t*p >= 0 and h+t*p <= 1 else 1.0 for t in ts]))
        # Actually let me just use a simpler approach
        lo = ts[max(0, idx-2)]
        hi = ts[min(n_grid-1, idx+2)]
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        if hi - lo > 1e-10:
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p) if (h + c1*p).min() >= -1e-12 else cur_c5 + 1
            f2 = _exact_c5(h + c2 * p) if (h + c2*p).min() >= -1e-12 else cur_c5 + 1
            for _ in range(30):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _exact_c5(h + c1*p) if (h+c1*p).min() >= -1e-12 else cur_c5+1
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _exact_c5(h + c2*p) if (h+c2*p).min() >= -1e-12 else cur_c5+1
            opt_t = (lo + hi) / 2
            f_opt = _exact_c5(h + opt_t*p) if (h+opt_t*p).min() >= -1e-12 else cur_c5+1
            if f_opt < best_val:
                best_t, best_val = opt_t, f_opt
        
        if best_val < cur_c5 - 1e-12:
            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.3)
        else:
            stall += 1
            radius *= 0.7
    
    return best_h, best_c5
