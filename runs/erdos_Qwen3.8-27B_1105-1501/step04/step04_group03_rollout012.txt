import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        fracs = np.where((x > 1e-10) & (x < 1.0 - 1e-10))[0]
        if len(fracs) > 0:
            per = residual / len(fracs)
            x[fracs] = np.clip(x[fracs] + per, 0.0, 1.0)
    return x

def soft_max_val_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = scores_raw * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            hi_s = js1 + d
            hi_e = je1 + d
            grad[js1:je1 + 1] += wk * (1.0 - h[hi_s:hi_e + 1])
        js2 = max(0, d)
        je2 = min(n - 1, n - 1 + d)
        if js2 <= je2:
            h2_s = js2 - d
            h2_e = je2 - d
            grad[js2:je2 + 1] -= wk * h[h2_s:h2_e + 1]
    return val, dx * grad

def topk_smooth_val_grad(h, k_top, tau):

    n = len(h)

    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = scores_raw * dx

    # Simpler: just use soft-max with small tau (focuses on the max)
    # Actually let's use: tau * log(sum exp(s_i/tau)) but with a "boost" for top k
    # For simplicity, use soft-max (same as above)
    return soft_max_val_grad(h, tau)

def build_G(h):
    """Build gradient matrix for all correlation values: G[k, j] = d(score_k)/d(h_j) / dx"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            hi_s = js1 + d
            hi_e = je1 + d
            G[k, js1:je1 + 1] += (1.0 - h[hi_s:hi_e + 1])
        js2 = max(0, d)
        je2 = min(n - 1, n - 1 + d)
        if js2 <= je2:
            h2_s = js2 - d
            h2_e = je2 - d
            G[k, js2:je2 + 1] -= h[h2_s:h2_e + 1]
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    nfull = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((nfull, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j] - 1e-15), min(radius, 1.0 - h[j] + 1e-15)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search_fine(h, p, c5_cur):
    """Bracketed line search with golden section"""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j] + 1e-12) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, (h[j] - 1e-12) / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    if alpha_max < 1e-14:
        return 0.0, c5_cur
    # Coarse: 20 points
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(20):
        a = alpha_max * (i + 1) / 20.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a < 1e-12:
        return 0.0, c5_cur
    # Golden section refinement
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lp_search(h, deadline, radius_init=0.1, min_radius=1e-5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

        if p is None or np.max(np.abs(p)) < 1e-14:

            radius *= 0.6

            continue
        alpha, new_c5 = line_search_fine(h, p, cur_c5)

            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.6

    return best_h, best_c5

def slsqp_phase(h0, deadline):
    """Multi-temperature SLSQP optimization"""
    n = len(h0)
    target = n / 2.0
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    
    tau_list = [0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01, 0.008, 0.005, 0.003, 0.002]
    iters_list = [80, 80, 80, 70, 70, 60, 60, 50, 50, 50, 40, 40]
    
    for i in range(len(tau_list)):
        if time.time() > deadline:
            break
        tau = tau_list[i]
        mi = iters_list[i]
        dph = time.time()
        rem = max(10, int(deadline - dph))
        if rem < 15:
            break
        
        def obj(x):
            v, _ = soft_max_val_grad(x, tau)
            return v
        def grad(x):
            _, g = soft_max_val_grad(x, tau)
            return g
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-15, 'maxls': 50})
            if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                h_new = project(res.x, target)
                c5_new = exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
                elif c5_new < exact_c5(h_cur):
                    h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_line_search(h, j, k, sign, deadline):
    """Golden section line search for pair (j,k) with given sign"""
    if sign > 0:
        amax = min(h[j] - 1e-15, 1.0 - h[k])
    else:
        amax = min(1.0 - h[j], h[k] - 1e-15)
    if amax < 1e-12:
        return h, exact_c5(h)
    lo_a, hi_a = 0.0, amax
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    
    def f(a):
        ht = h.copy()
        ht[j] += sign * a
        ht[k] -= sign * a
        return exact_c5(ht)
    
    f1, f2 = f(c1), f(c2)
    f0 = f(0.0)
    for _ in range(18):
        if time.time() > deadline:
            break
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = f(c1)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = f(c2)
    opt_a = (lo_a + hi_a) / 2.0
    f_opt = f(opt_a)
    if f_opt < f0 - 1e-14:
        ht = h.copy()
        ht[j] += sign * opt_a
        ht[k] -= sign * opt_a
        return ht, f_opt
    return h, f0

def pair_polish(h, deadline, max_rounds=20):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Randomize pair order for diversity
        pairs = []
        for j in range(n):
            for k in range(j + 1, n):
                pairs.append((j, k))
        np.random.shuffle(pairs)
        for j, k in pairs:
            if time.time() > deadline:
                break
            for sign in [1.0, -1.0]:
                ht, ft = pair_line_search(best_h, j, k, sign, deadline)
                if ft < best_c5 - 1e-14:
                    best_c5 = ft
                    best_h = ht.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def subgradient_descent(h, deadline, steps=2000):
    """Subgradient descent with Polyak step size on max of linear functions"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    z_best = best_c5
    t = 0
    for step in range(steps):
        if time.time() > deadline:
            break
        t += 1
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        k_max = int(np.argmax(scores))
        d = k_max - (n - 1)
        # Subgradient of max at the active point
        g = np.zeros(n)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            hi_s = js1 + d
            hi_e = je1 + d
            g[js1:je1 + 1] = (1.0 - h[hi_s:hi_e + 1])
        js2 = max(0, d)
        je2 = min(n - 1, n - 1 + d)
        if js2 <= je2:
            h2_s = js2 - d
            h2_e = je2 - d
            g[js2:je2 + 1] -= h[h2_s:h2_e + 1]
        g *= dx
        # Polyak step: (f(x) - z*) / ||g||^2, cap it
        gnorm2 = float(np.dot(g, g))
        if gnorm2 < 1e-20:
            break
        step_size = min(2.0 * (cur_c5 - z_best + 0.01) / gnorm2, 0.05)
        h_new = h - step_size * g
        h_new = project(h_new, target)
        c5_new = exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            z_best = best_c5
        h = h_new
        cur_c5 = c5_new
    return best_h, best_c5
