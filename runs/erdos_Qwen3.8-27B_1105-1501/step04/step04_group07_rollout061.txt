            # For d >= 0: p+d < n => p < n-d; p-d >= 0 => p >= d
            # So M[k, 0:n-d] gets (1-h[d:n]) and M[k, d:n] gets -h[0:n-d]
            # But these overlap! For p in [d, n-d-1]: M[k,p] = (1-h[p+d]) - h[p-d]
            # For p in [0, d-1]: M[k,p] = (1-h[p+d]) - 0 = 1-h[p+d] (if p+d < n, which is true since p < d <= n-1)

            # M[k, p] = (1-h[p+d]) * [p+d < n] - h[p-d] * [p-d >= 0]
            
            # Vectorized:
            p_vals = np.arange(n)
            p_plus_d = p_vals + d
            p_minus_d = p_vals - d
            M[k, :] = np.where(p_plus_d < n, 1.0 - h[p_plus_d], 0.0) - np.where(p_minus_d >= 0, h[p_minus_d], 0.0)
        else:

            g = -d
            p_vals = np.arange(n)
            p_plus_d = p_vals + d  # = p - g
            p_minus_d = p_vals - d  # = p + g

    n = len(h)

        term2 = np.where((pm >= 0) & (pm < n), h[pm], 0.0)

def _quadratic_minmax(A, B, C, lo, hi, n_samples=30):
    """Minimize max_k(A[k] + t*B[k] + t^2*C[k]) for t in [lo, hi]"""
    if hi - lo < 1e-14:
        t = lo
        return t, float(np.max(A + t*B + t*t*C))
    
    ts = np.linspace(lo, hi, n_samples)
    vals = np.array([float(np.max(A + t*B + t*t*C)) for t in ts])
    best_i = int(np.argmin(vals))
    best_t = ts[best_i]
    best_v = vals[best_i]
    
    if best_i in (0, n_samples-1):
        return best_t, best_v
    
    # Golden section
    lo2 = ts[best_i-1]
    hi2 = ts[best_i+1]
    gr = (np.sqrt(5.0)-1.0)/2.0
    c1 = hi2 - gr*(hi2-lo2)
    c2 = lo2 + gr*(hi2-lo2)
    f1 = float(np.max(A + c1*B + c1*c1*C))
    f2 = float(np.max(A + c2*B + c2*c2*C))
    for _ in range(30):
        if hi2 - lo2 < 1e-13:
            break
        if f1 < f2:
            hi2 = c2; f2 = f1; c2 = c1
            c1 = hi2 - gr*(hi2-lo2)
            f1 = float(np.max(A + c1*B + c1*c1*C))
        else:
            lo2 = c1; f1 = f2; c1 = c2
            c2 = lo2 + gr*(hi2-lo2)
            f2 = float(np.max(A + c2*B + c2*c2*C))
    
    t_opt = (lo2+hi2)/2.0
    v_opt = float(np.max(A + t_opt*B + t_opt*t_opt*C))
    if v_opt < best_v:

                imp = cur_max - v_opt

        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

        mask1 = (pd >= 0) & (pd < n)
        t1[mask1] = 1.0 - h[pd[mask1]]
        M[k, :] = t1 - t2
    return M

def _quad_minmax(A, B, C, lo, hi, n_samp=25):
    if hi - lo < 1e-14:
        t = lo
        return t, float(np.max(A + t * B + t * t * C))
    ts = np.linspace(lo, hi, n_samp)
    vals = np.empty(n_samp)
    for i in range(n_samp):
        t = ts[i]
        vals[i] = float(np.max(A + t * B + t * t * C))
    best_i = int(np.argmin(vals))
    best_t = float(ts[best_i])
    best_v = float(vals[best_i])
    if best_i in (0, n_samp - 1):
        return best_t, best_v
    lo2 = float(ts[best_i - 1])
    hi2 = float(ts[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi2 - gr * (hi2 - lo2)
    c2 = lo2 + gr * (hi2 - lo2)
    f1 = float(np.max(A + c1 * B + c1 * c1 * C))
    f2 = float(np.max(A + c2 * B + c2 * c2 * C))
    for _ in range(35):
        if hi2 - lo2 < 1e-13:
            break
        if f1 < f2:
            hi2 = c2
            f2 = f1
            c2 = c1
            c1 = hi2 - gr * (hi2 - lo2)
            f1 = float(np.max(A + c1 * B + c1 * c1 * C))
        else:
            lo2 = c1
            f1 = f2
            c1 = c2
            c2 = lo2 + gr * (hi2 - lo2)
            f2 = float(np.max(A + c2 * B + c2 * c2 * C))
    t_opt = (lo2 + hi2) / 2.0
    v_opt = float(np.max(A + t_opt * B + t_opt * t_opt * C))
    if v_opt < best_v:
        best_t, best_v = t_opt, v_opt
    return best_t, best_v

def _subgrad_step(h, tol_frac=0.005):
    n = len(h)
    n_lags = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(scores.max())
    if cur_max < 1e-15:
        return None
    thresh = cur_max * (1.0 - tol_frac)
    active = np.where(scores >= thresh)[0]
    M = _prescore_m(h)
    g = np.zeros(n)
    for k in active:
        g += M[k, :]
    g /= len(active)
    g -= np.mean(g)
    gn = np.linalg.norm(g)
    if gn < 1e-12:
        return None
    d_dir = -g / gn
    B = M @ d_dir
    C = -np.correlate(d_dir, d_dir, mode='full')
    alpha_max = 1.0
    for i in range(n):
        if d_dir[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d_dir[i])
        elif d_dir[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-d_dir[i]))
    alpha_max = max(alpha_max, 0.0)
    alpha_opt, v_opt = _quad_minmax(scores, B, C, 0.0, alpha_max)
    return d_dir, alpha_opt, v_opt, scores, cur_max

def _pair_step(h):
    n = len(h)
    n_lags = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(scores.max())
    M = _prescore_m(h)
    best_imp = 0.0
    best_pq = None
    for p in range(n):
        for q in range(n):
            if p == q:
                continue
            t_max = min(1.0 - h[p], h[q])
            if t_max < 1e-8:
                continue
            B = M[:, q] - M[:, p]
            C = np.zeros(n_lags)
            k_qp = (q - p) + n - 1
            k_pq = (p - q) + n - 1
            if 0 <= k_qp < n_lags:
                C[k_qp] = 1.0
            if 0 <= k_pq < n_lags:
                C[k_pq] = -1.0
            t_opt, v_opt = _quad_minmax(scores, B, C, 0.0, t_max, n_samp=20)
            imp = cur_max - v_opt
            if imp > best_imp:
                best_imp = imp
                best_pq = (p, q, t_opt)
    if best_pq is None or best_imp < 1e-10:
        return None
    return best_pq

def _local_optimize(h, deadline, max_iters=800, use_pair=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        result = _subgrad_step(h, tol_frac=0.003)
        if result is None:
            break
        d_dir, alpha_opt, v_opt, scores, cur_max = result
        if v_opt < cur_max - 1e-12 and alpha_opt > 1e-14:
            h_new = h + alpha_opt * d_dir
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
                if stall < 30:
                    h = h_new
        else:
            if use_pair:
                pq = _pair_step(h)
                if pq is not None:
                    p, q, t = pq
                    h[p] += t
                    h[q] -= t
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, target)
                    c5_new = _exact_c5(h)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    break
            else:
                break
    return best_h, best_c5

def _pair_only_optimize(h, deadline, max_sweeps=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        pq = _pair_step(h)
        if pq is None:
            break
        p, q, t = pq
        h[p] += t
        h[q] -= t
        h = np.clip(h, 0.0, 1.0)
        h = _project(h, target)
        c5_new = _exact_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _random_direction_optimize(h, deadline, n_tries=200):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for tr in range(n_tries):
        if time.time() > deadline:
            break
        d_dir = rng.standard_normal(n)
        d_dir -= np.mean(d_dir)
        gn = np.linalg.norm(d_dir)
        if gn < 1e-12:
            continue
        d_dir /= gn
        scores = np.correlate(h, 1.0 - h, mode='full')
        M = _prescore_m(h)
        B = M @ d_dir
        C = -np.correlate(d_dir, d_dir, mode='full')
        alpha_max = 1.0
        for i in range(n):
            if d_dir[i] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[i]) / d_dir[i])
            elif d_dir[i] < -1e-14:
                alpha_max = min(alpha_max, h[i] / (-d_dir[i]))
        alpha_max = max(alpha_max, 0.0)
        alpha_opt, v_opt = _quad_minmax(scores, B, C, 0.0, alpha_max)
        if alpha_opt > 1e-10 and v_opt < float(np.max(scores)) - 1e-10:
            h_new = h + alpha_opt * d_dir
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new % 2 != 0:
        n_new += 1
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_start(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h

def _alternating_start(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % 4 < 2) else 0.0
    return _project(h, n / 2.0)

def _smooth_start(n, seed):
    rng = np.random.default_rng(seed)
    freqs = rng.uniform(0.5, 5.0, 6)
    phases = rng.uniform(0, 2 * np.pi, 6)
    amps = rng.uniform(0.1, 0.5, 6)
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(6):
        h += amps[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i])
    h = (h - h.min()) / (h.max() - h.min() + 1e-12)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Subgradient optimization (20%)
    dl = t0 + budget_s * 0.20
    h_r, c5_r = _local_optimize(h, dl, max_iters=1000, use_pair=True)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r.copy()

    # Phase 2: Pair-only refinement (10%)
    dl = t0 + budget_s * 0.30
    h_r, c5_r = _pair_only_optimize(best_h, dl, max_sweeps=500)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r.copy()

    # Phase 3: Multi-start from parent (25%)
    dl = t0 + budget_s * 0.55
    rng = np.random.default_rng(seed)
    scale = 0.06
    for attempt in range(200):
        if time.time() > dl:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl, time.time() + budget_s * 0.022)
        h_try, c5_try = _local_optimize(h_pert, sub_dl, max_iters=200, use_pair=True)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.80)
        else:
            scale = min(0.15, scale * 1.10)

    # Phase 4: Subgradient + pair alternating (15%)
    dl = t0 + budget_s * 0.70
    h_r, c5_r = _local_optimize(best_h, dl, max_iters=1500, use_pair=True)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r.copy()
    h_r, c5_r = _pair_only_optimize(best_h, min(dl, t0 + budget_s * 0.75), max_sweeps=300)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r.copy()

    # Phase 5: Random direction exploration (12%)
    dl = t0 + budget_s * 0.87
    h_r, c5_r = _random_direction_optimize(best_h, dl, n_tries=400)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r.copy()

    # Phase 6: Try different starting points (8%)
    dl = t0 + budget_s * 0.95
    starts = []
    starts.append(_alternating_start(best_n))
    starts.append(_smooth_start(best_n, seed + 1))
    starts.append(_smooth_start(best_n, seed + 99))
    starts.append(_binary_start(best_n, seed + 7))
    for s in starts:
        if time.time() > dl:
            break
        c5_s = _exact_c5(s)
        if c5_s >= best_c5:
            continue
        sub_dl = min(dl, time.time() + budget_s * 0.015)
        h_try, c5_try = _local_optimize(s, sub_dl, max_iters=150, use_pair=True)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 7: Resolution sweep (5%)
    dl = t0 + budget_s * 0.99
    for dn in [8, -8, 16, -16, 32]:
        if time.time() > dl:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 192 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl, time.time() + budget_s * 0.012)
        h_r, c5_r = _local_optimize(h_r, sub_dl, max_iters=120, use_pair=True)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Final: project and validate
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
