    ci[v1] += (1.0 - h[i - s[v1]])
    v2 = (i + s >= 0) & (i + s < n)

    B = _corr_full(h) - ci * h[i] - cj * h[j]
    return A, B

    if dlo >= dhi:
        return 0.0, float(np.max(B))

    best_val = float(np.max(A * best_d + B))

        m2 = hi - (hi - lo) / 3.0
        v1 = float(np.max(A * m1 + B))
        v2 = float(np.max(A * m2 + B))
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        mid = 0.5 * (lo + hi)
        vm = float(np.max(A * mid + B))
        if vm < best_val:
            best_val = vm
            best_d = mid

    v_hi = float(np.max(A * dhi + B))
    if v_lo < best_val:
        best_val, best_d = v_lo, dlo
    if v_hi < best_val:
        best_val, best_d = v_hi, dhi
    return best_d, best_val

                    if bv < cur_c5 - 1e-14:
                        h[i] += bd
                        h[j] -= bd
                        cur_c5 = bv
                        corr = _corr_full(h)
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5, best_h = cur_c5, h.copy()
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
        sweep += 1
    return best_h, best_c5


def _sa_search(h, n, deadline, seed=42, T_init=0.005, T_min=0.00005, cool=0.997):
    """Simulated annealing with pair swaps."""
    rng = np.random.default_rng(seed)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    no_improve = 0
    while time.time() < deadline and T > T_min:
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi - 1e-12:
            continue
        d = float(rng.uniform(dlo, dhi))
        h[i] += d
        h[j] -= d
        corr = _corr_full(h)
        new_c5 = float(np.max(corr))
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] -= d
            h[j] += d
            no_improve += 1
        T *= cool
        if no_improve > 5000 and T < T_init * 0.1:
            break
    return best_h, best_c5


def _epi_f(z, n):
    return z[n]


def _epi_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _ineq_f(z, n):
    h = z[:n]
    t = z[n]
    corr = _corr_full(h)
    return t - corr


def _ineq_g(z, n):
    h = z[:n]
    L = 2 * n - 1
    s = np.arange(L) - (n - 1)
    J = np.zeros((L, n + 1))
    j = np.arange(n)
    for r in range(L):
        sr = s[r]
        grad = np.zeros(n)
        v1 = (j - sr >= 0) & (j - sr < n)
        grad[v1] += (1.0 - h[j[v1] - sr])
        v2 = (j + sr >= 0) & (j + sr < n)
        grad[v2] -= h[j[v2] + sr]
        J[r, :n] = -grad
        J[r, n] = 1.0
    return J


def _slsqp_full(h_init, n, maxiter=300, ftol=1e-13):
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr_full(h)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_f, z0, jac=_epi_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)


def _lse_obj(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = _corr_full(h)
    ms = float(np.max(corr))
    vals = alpha * (corr - ms)
    vals = np.clip(vals, -500.0, 500.0)
    exp_v = np.exp(vals)
    s = float(np.sum(exp_v))
    f = ms + (1.0 / alpha) * np.log(s)
    w = exp_v / s
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        sr = r - (n - 1)
        pd = np.zeros(n)
        v1 = (j - sr >= 0) & (j - sr < n)
        pd[v1] += (1.0 - h[j[v1] - sr])
        v2 = (j + sr >= 0) & (j + sr < n)
        pd[v2] -= h[j[v2] + sr]
        grad += w[r] * pd
    grad -= np.mean(grad)
    return f, grad


def _lse_step(h, n, alpha, maxiter=80):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _perturb_start(h, n, rng, scale):
    h_p = h + rng.normal(0, scale, n)
    return _project(h_p, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LSE warm-up (5% budget)
    d1 = t0 + budget_s * 0.05
    for alpha in [10.0, 30.0, 80.0]:
        if time.time() > d1:
            break
        h_lse, c5_lse = _lse_step(best_h, best_n, alpha, maxiter=60)
        if c5_lse < best_c5:
            best_c5, best_h = c5_lse, h_lse.copy()

    # Phase 2: Intensive local search (35% budget)
    d2 = t0 + budget_s * 0.40
    rng_ls = np.random.default_rng(seed + 1)
    best_h, best_c5 = _local_search(best_h, best_n, d2, max_sweeps=10000, rng=rng_ls)

    # Phase 3: SLSQP refinement (15% budget)
    d3 = t0 + budget_s * 0.55
    for attempt in range(8):
        if time.time() > d3:
            break
        start = best_h.copy()
        if attempt > 0:
            start = _perturb_start(best_h, best_n, rng, 0.01 + 0.01 * attempt)
        h_s, c5_s = _slsqp_full(start, best_n, maxiter=250, ftol=1e-14)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
            # Follow-up local search
            sub_d = min(d3, time.time() + budget_s * 0.03)
            h_l, c5_l = _local_search(best_h, best_n, sub_d, max_sweeps=500, rng=rng_ls)
            if c5_l < best_c5:
                best_c5, best_h = c5_l, h_l.copy()

    # Phase 4: Simulated annealing (20% budget)
    d4 = t0 + budget_s * 0.75
    for sa_attempt in range(4):
        if time.time() > d4:
            break
        sub_d = min(d4, time.time() + budget_s * 0.04)
        h_sa, c5_sa = _sa_search(best_h, best_n, sub_d, seed=seed + 100 + sa_attempt,
                                 T_init=0.003 + 0.002 * sa_attempt)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()
        # Local search after SA
        sub_d2 = min(d4, time.time() + budget_s * 0.02)
        h_l, c5_l = _local_search(best_h, best_n, sub_d2, max_sweeps=2000, rng=rng_ls)
        if c5_l < best_c5:
            best_c5, best_h = c5_l, h_l.copy()

    # Phase 5: More SLSQP with structured perturbations (10% budget)
    d5 = t0 + budget_s * 0.85
    scales = [0.005, 0.01, 0.02, 0.03, 0.05]
    for sc in scales:
        if time.time() > d5:
            break
        for _ in range(3):
            if time.time() > d5:
                break
            h_p = _perturb_start(best_h, best_n, rng, sc)
            h_s, c5_s = _slsqp_full(h_p, best_n, maxiter=200, ftol=1e-14)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()

    # Phase 6: Resolution exploration (10% budget)
    d6 = t0 + budget_s * 0.95
    for dn in [-2, 2, -4, 4, -6, 6, 8, -8, 10, -10]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = _local_search(h_r, n_try, sub_d, max_sweeps=300, rng=rng_ls)
        sub_d2 = min(d6, time.time() + budget_s * 0.012)
        h_s, c5_s = _slsqp_full(h_r, n_try, maxiter=150, ftol=1e-13)
        if c5_s < c5_r:
            h_r, c5_r = h_s, c5_s
        sub_d3 = min(d6, time.time() + budget_s * 0.01)
        h_l, c5_l = _local_search(h_r, n_try, sub_d3, max_sweeps=500, rng=rng_ls)
        if c5_l < c5_r:
            h_r, c5_r = h_l, c5_l
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: Final polish (5% budget)
    d7 = t0 + budget_s * 0.99
    h_l, c5_l = _local_search(best_h, best_n, min(d7, time.time() + 20),
                              max_sweeps=2000, rng=rng_ls)
    if c5_l < best_c5:
        best_c5, best_h = c5_l, h_l.copy()
    h_f, c5_f = _slsqp_full(best_h, best_n, maxiter=200, ftol=1e-15)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    h_l2, c5_l2 = _local_search(best_h, best_n, d7, max_sweeps=1000, rng=rng_ls)
    if c5_l2 < best_c5:
        best_c5, best_h = c5_l2, h_l2.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
