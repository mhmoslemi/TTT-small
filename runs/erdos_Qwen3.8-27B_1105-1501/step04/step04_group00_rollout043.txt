def _exact_c5(h):

    return float(np.max(corr) * dx)

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))

        r = n - 1 + d

        G[r, :m] += -dx * h[d:]
    for s in range(1, n):
        r = n - 1 - s
        m = n - s
        G[r, :m] += dx * (1.0 - h[s:])
        G[r, s:] += -dx * h[:m]
    return G

    scores = _scores(h)

        G_k = G

    A_eq = np.zeros((1, n + 1))

        h_cand = _project(h + alpha * p, target)
        actual_c5 = _exact_c5(h_cand)
        if actual_c5 < cur_c5 - 1e-13:
            h = h_cand
            cur_c5 = actual_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.35)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 12 and radius < 0.015:
                radius = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_descent(h, deadline, max_passes=200, verify_thresh=1e-6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores(h)
    G = _build_G(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                if h[j] < 1e-10 and h[k] < 1e-10:
                    continue
                if h[j] > 1 - 1e-10 and h[k] > 1 - 1e-10:
                    continue
                C = G[:, j] - G[:, k]
                a_lo = max(-h[j], h[k] - 1.0)
                a_hi = min(1.0 - h[j], h[k])
                if a_hi - a_lo < 1e-12:
                    continue
                lo_a, hi_a = a_lo, a_hi
                for _ in range(18):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    f1 = np.max(scores + m1 * C)
                    f2 = np.max(scores + m2 * C)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                alpha = (lo_a + hi_a) / 2.0
                pred_c5 = float(np.max(scores + alpha * C))
                if cur_c5 - pred_c5 > verify_thresh and abs(alpha) > 1e-10:
                    h[j] += alpha
                    h[k] -= alpha
                    h[np.clip(h, 0, 1)]
                    actual_c5 = _exact_c5(h)
                    if actual_c5 < cur_c5 - 1e-14:
                        cur_c5 = actual_c5
                        scores = _scores(h)
                        G = _build_G(h)
                        pass_improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                    else:
                        h[j] -= alpha
                        h[k] += alpha
        if not pass_improved:
            break
    return best_h, best_c5

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def _graded_weights(h, scores, top_k=10):
    n_lags = len(scores)
    if top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(n_lags)
    s = scores[idx]
    max_s = s.max()
    beta = 50.0
    w = np.exp(beta * (s - max_s))
    w /= w.sum()
    return w, idx

def _grad_step(h, deadline, n_steps=200, beta_start=20, beta_end=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _build_G(h)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = scores.max()
        w = np.exp(beta * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.06
        for _ in range(25):
            h_new = _project(h - alpha * grad, target) if alpha > 0 else _project(h + 0.06 * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                G = _build_G(h)
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP at original resolution
    dl1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_opt(h_start, dl1, top_k_list=[None, 12, 7, 5],
                          radius_init=0.15, max_iters=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient-based refinement
    if time.time() < t0 + budget_s * 0.15:
        dl2 = t0 + budget_s * 0.17
        h2, c5_2 = _grad_step(best_h, dl2, n_steps=120, beta_start=30, beta_end=300)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Multi-resolution LP search
    dl3 = t0 + budget_s * 0.45
    resolutions = [192, 256, 320, 384, 448, 160]
    alphas_qp = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.555
                 ]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if best_c5 < 0.3805:
            break
        n_starts = 2
        for s in range(n_starts):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas_qp[ri % len(alphas_qp)]
                h_init = _quasi_init(n_try, alpha, rng)
            sub_dl = min(dl3, time.time() + budget_s * 0.04)
            h_try, c5_try = _local_opt(h_init, sub_dl,
                                       top_k_list=[None, 10, 6, 4],
                                       radius_init=0.12, max_iters=2500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.015)
                h_wg, c5_wg = _grad_step(h_try, sub_dl2, n_steps=60,
                                          beta_start=20, beta_end=200)
                if c5_wg < best_c5:
                    best_c5 = c5_wg
                    best_h = h_wg.copy()
                    best_n = n_try

    # Phase 4: Intensive pair descent at current best
    if time.time() < t0 + budget_s * 0.70:
        dl4 = t0 + budget_s * 0.72
        best_h, best_c5 = _pair_descent(best_h, dl4, max_passes=300,
                                         verify_thresh=1e-5)
        # Follow up with LP
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_lp, c5_lp = _local_opt(best_h, sub_dl,
                                     top_k_list=[6, 4, None],
                                     radius_init=0.05, max_iters=1500)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
        # More pair descent
        if time.time() < dl4:
            dl4b = min(dl4, time.time() + budget_s * 0.03)
            best_h, best_c5 = _pair_descent(best_h, dl4b, max_passes=100,
                                             verify_thresh=1e-7)

    # Phase 5: More diverse starts
    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.87
        for n_try in [256, 320, 128]:
            if time.time() > dl5:
                break
            for attempt in range(3):
                if time.time() > dl5:
                    break
                alpha = [0.6180339887, 0.4142135624, 0.7071067812][attempt % 3]
                h_init = _quasi_init(n_try, alpha, rng)
                sub_dl = min(dl5, time.time() + budget_s * 0.03)
                h_try, c5_try = _local_opt(h_init, sub_dl,
                                           top_k_list=[None, 8, 5],
                                           radius_init=0.15, max_iters=2000)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                    # Polish
                    sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
                    h_p, c5_p = _pair_descent(best_h, sub_dl2, max_passes=80,
                                               verify_thresh=1e-6)
                    if c5_p < best_c5:
                        best_c5 = c5_p
                        best_h = h_p.copy()
                else:
                    sub_dl2 = min(dl5, time.time() + budget_s * 0.012)
                    h_g, c5_g = _grad_step(h_try, sub_dl2, n_steps=80,
                                            beta_start=40, beta_end=400)
                    if c5_g < best_c5:
                        best_c5 = c5_g
                        best_h = h_g.copy()
                        best_n = n_try

    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.93
    scale = 0.025
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.008)
        h_try, c5_try = _local_opt(h_pert, sub_dl,
                                   top_k_list=[5, 3, None],
                                   radius_init=0.05, max_iters=400)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Phase 7: Final polish
    if time.time() < hard_deadline:
        dl7 = hard_deadline
        # Pair descent with very tight threshold
        h_pd, c5_pd = _pair_descent(best_h, dl7, max_passes=150,
                                     verify_thresh=1e-8)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
        # Tiny LP steps
        if time.time() < dl7 - 2:
            h_fin, c5_fin = _local_opt(best_h, dl7,
                                       top_k_list=[3, 4, 5],
                                       radius_init=0.01, max_iters=300)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, best_n
