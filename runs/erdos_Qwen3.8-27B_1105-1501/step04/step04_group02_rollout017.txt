```python
import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    """Project v to [0,1]^n with sum = target using bisection."""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    alpha = (lo + hi) / 2
    x = np.clip(v - alpha, 0, 1)
    res = target - x.sum()
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1 - 1e-14
        else:
            mask = x > 1e-14
        cnt = mask.sum()
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    n = len(h)
    scores = np.zeros(2*n - 1)
    for r in range(2*n - 1):
        d = r - (n-1)
        if d >= 0:
            m = n - d
            scores[r] = np.dot(h[d:], 1.0 - h[:m])
        else:
            s = -d
            m = n - s
            scores[r] = np.dot(h[:m], 1.0 - h[s:])
    return scores

def _raw_grad(h, r):
    """Gradient of score r w.r.t. h."""
    n = len(h)
    d = r - (n-1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        g[:m] += -h[d:]
        g[d:] += 1.0 - h[:m]
    else:
        s = -d
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] += -h[:m]
    return g

def _weighted_gradient(h, weights):
    """Weighted sum of gradients for active lags."""
    n = len(h)
    g = np.zeros(n)
    for r in range(len(weights)):
        if weights[r] > 1e-14:
            g += weights[r] * _raw_grad(h, r)
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    x = np.clip(v - alpha, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _raw_scores(h):
    n = len(h)
    scores = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            scores[r] = float(np.dot(h[d:], 1.0 - h[:m]))
        else:
            s = -d
            m = n - s
            scores[r] = float(np.dot(h[:m], 1.0 - h[s:]))
    return scores


def _raw_grad_single(h, r):
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        g[:m] += -h[d:]
        g[d:] += 1.0 - h[:m]
    else:
        s = -d
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] += -h[:m]
    return g


def _weighted_grad(h, weights, top_k=None):
    n = len(h)
    g = np.zeros(n)
    if top_k is not None and top_k < len(weights):
        idx = np.argsort(weights)[::-1][:top_k]
        for r in idx:
            if weights[r] > 1e-14:
                g += weights[r] * _raw_grad_single(h, int(r))
    else:
        for r in range(len(weights)):
            if weights[r] > 1e-14:
                g += weights[r] * _raw_grad_single(h, r)
    return g


class _EpigraphSolver:
    def __init__(self, h0):
        self.n = len(h0)
        self.target = self.n / 2.0
        self.h0 = h0.copy()

    def _obj(self, x):
        return x[self.n]

    def _obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def _eq(self, x):
        return np.array([np.sum(x[:self.n]) - self.target])

    def _eq_jac(self, x):
        J = np.ones(self.n + 1)
        J[self.n] = 0.0
        return J

    def _ineq(self, x):
        h = x[:self.n]
        T = x[self.n]
        scores = _raw_scores(h)
        return T - scores

    def _ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        G = np.zeros((2 * n - 1, n))
        for r in range(2 * n - 1):
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                G[r, :m] += -h[d:]
                G[r, d:] += 1.0 - h[:m]
            else:
                s = -d
                m = n - s
                G[r, :m] += 1.0 - h[s:]
                G[r, s:] += -h[:m]
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    def solve(self, maxiter=50, top_k=None):
        n = self.n
        scores0 = _raw_scores(self.h0)
        T0 = float(np.max(scores0)) * 1.02 + 0.05
        x0 = np.concatenate([self.h0, [T0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
        cons = [
            {'type': 'eq', 'fun': self._eq, 'jac': self._eq_jac},
            {'type': 'ineq', 'fun': self._ineq, 'jac': self._ineq_jac},
        ]
        try:
            res = minimize(self._obj, x0, jac=self._obj_grad,
                           bounds=bounds, constraints=cons, method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            if np.all(np.isfinite(res.x)):
                h_sol = _project(res.x[:n], self.target)
                return h_sol
        except Exception:
            pass
        return self.h0


def _pair_line_search(h, i, j, sign):
    """Line search: h[i] += sign*delta, h[j] -= sign*delta."""
    if sign > 0:
        alpha_max = min(h[i], 1.0 - h[j])
    else:
        alpha_max = min(h[j], 1.0 - h[i])
    if alpha_max < 1e-12:
        return 0.0, _exact_c5(h)

    def _eval(delta):
        h_t = h.copy()
        h_t[i] += sign * delta
        h_t[j] -= sign * delta
        if h_t[i] < -1e-12 or h_t[i] > 1.0 + 1e-12 or h_t[j] < -1e-12 or h_t[j] > 1.0 + 1e-12:
            return 1.0
        return _exact_c5(h_t)

    base = _eval(0.0)
    # Grid search
    n_grid = 20
    deltas = np.linspace(0.0, alpha_max, n_grid)
    vals = np.array([_eval(d) for d in deltas])
    idx = int(np.argmin(vals))
    lo_d = deltas[max(0, idx - 2)]
    hi_d = deltas[min(n_grid - 1, idx + 2)]
    if hi_d <= lo_d:
        lo_d, hi_d = 0.0, alpha_max
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_d - gr * (hi_d - lo_d)
    c2 = lo_d + gr * (hi_d - lo_d)
    f1, f2 = _eval(c1), _eval(c2)
    for _ in range(25):
        if hi_d - lo_d < 1e-12:
            break
        if f1 < f2:
            hi_d, c2, f2 = c2, c1, f1
            c1 = hi_d - gr * (hi_d - lo_d)
            f1 = _eval(c1)
        else:
            lo_d, c1, f1 = c1, c2, f2
            c2 = lo_d + gr * (hi_d - lo_d)
            f2 = _eval(c2)
    opt_d = (lo_d + hi_d) / 2.0
    f_opt = _eval(opt_d)
    best_d = deltas[idx]
    best_f = vals[idx]
    if f_opt < best_f:
        best_d, best_f = opt_d, f_opt
    return best_d, best_f


def _pair_refine(h, deadline, max_passes=50, guidance=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Build candidate pairs
        if guidance and n <= 150:
            scores = _raw_scores(h)
            top_k = min(20, 2 * n - 1)
            top_idx = np.argsort(scores)[::-1][:top_k]
            w = np.zeros(2 * n - 1)
            for r in top_idx:
                w[r] = scores[r]
            w = w / (w.sum() + 1e-15)
            grad = _weighted_grad(h, w, top_k=top_k)
            # Pairs from gradient: increase where grad is most negative, decrease where most positive
            neg_idx = np.argsort(grad)[:15]
            pos_idx = np.argsort(grad)[-15:]
            pairs = set()
            for ni in neg_idx:
                for pi in pos_idx:
                    if ni != pi:
                        pairs.add((int(ni), int(pi)))
            # Add some random pairs
            for _ in range(30):
                a, b = rng.integers(0, n, 2)
                if a != b:
                    pairs.add((int(a), int(b)))
        else:
            pairs = set()
            for _ in range(100):
                a, b = rng.integers(0, n, 2)
                if a != b:
                    pairs.add((int(a), int(b)))

        pair_list = list(pairs)
        rng.shuffle(pair_list)

        for (i, j) in pair_list:
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            for sign in [1.0, -1.0]:
                best_d, new_c5 = _pair_line_search(h, i, j, sign)
                if new_c5 < best_c5 - 1e-12:

                    h[i] += sign * best_d
                    h[j] -= sign * best_d
                    h = _project(h, target)
                    best_c5 = _exact_c5(h)
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _slsqp_rounds(h, deadline, n_rounds=3):
    """Multiple SLSQP rounds with re-evaluation between."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    time_per = (deadline - time.time()) / max(n_rounds, 1)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + time_per)
        solver = _EpigraphSolver(best_h)
        # Adaptive maxiter based on remaining time
        maxiter = max(20, int(time_per * 8))
        maxiter = min(maxiter, 80)
        h_new = solver.solve(maxiter=maxiter)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5


def _grad_polish(h, deadline, tau_init=20.0, lr_init=0.015):
    """Softmax gradient descent for the max objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        scores = _raw_scores(h)
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted - shifted.max())
        w = exp_s / float(np.sum(exp_s))
        grad = _weighted_grad(h, w) / n
        mom = 0.9 * mom + 0.1 * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.04, 0.12)
            tau = min(tau * 1.02, 150.0)
        else:
            stall += 1
            lr *= 0.93
            if stall > 60:
                break
            if stall % 25 == 0:
                tau = max(tau_init, tau * 0.75)
    return best_h, best_c5
```
