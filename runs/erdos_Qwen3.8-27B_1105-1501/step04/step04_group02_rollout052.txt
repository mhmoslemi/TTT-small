        pass
    return h0.copy(), T0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):
    """Local pair refinement."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(hash(_pass) & 0xFFFFFFFF)
        if n > 60:
            idx_j = rng.integers(0, n, size=min(n, 40))
            idx_k = rng.integers(0, n, size=min(n, 40))
            pairs = set()
            for jj in idx_j:
                for kk in idx_k:
                    if jj != kk:
                        pairs.add((min(int(jj), int(kk)), max(int(jj), int(kk))))
            pair_list = list(pairs)
        else:
            pair_list = [(j, k) for j in range(n) for k in range(j + 1, n)]
        rng.shuffle(pair_list)
        
        for (j, k) in pair_list:
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if dj > 0:
                    am = (1.0 - h[j]) / dj
                else:
                    am = h[j] / (-dj)
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-12:
                    continue
                for frac in [0.25, 0.5, 0.8, 1.0]:
                    h_t = h.copy()
                    h_t[j] += dj * am * frac
                    h_t[k] += dk * am * frac
                    if h_t[j] < 0 or h_t[j] > 1 or h_t[k] < 0 or h_t[k] > 1:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _grad_optimize(h, deadline, tau_init=20.0, lr_init=0.015, mom_beta=0.9):
    """Gradient descent with softmax approximation."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    
    while time.time() < deadline:
        dx = 2.0 / n
        scores = _raw_scores(h) * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -50.0, 50.0)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        
        # Weighted gradient
        G = _raw_grad_matrix(h)
        grad = (w @ G) * dx  # shape (n,)
        
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
            tau = min(tau * 1.01, 300.0)
        else:
            stall += 1
            lr *= 0.93
            if stall > 60:
                break
            if stall % 25 == 0:
                tau = max(tau_init, tau * 0.85)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_Ng = 0

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    x = np.clip(v - alpha, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _raw_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            G[r, :cnt] += -h[d:d+cnt]
            G[r, d:d+cnt] += 1.0 - h[:cnt]
        else:
            g = -d
            cnt = n - g
            G[r, :cnt] += 1.0 - h[g:g+cnt]
            G[r, g:g+cnt] += -h[:cnt]
    return G

def _obj_fun(x):
    return x[_Ng]

def _obj_grad(x):
    g = np.zeros(_Ng + 1)
    g[_Ng] = 1.0
    return g

def _ineq_fun(x):
    h = x[:_Ng]
    T = x[_Ng]
    return _raw_scores(h) - T

def _ineq_jac(x):
    h = x[:_Ng]
    n = _Ng
    G = _raw_grad_matrix(h)
    jac = np.zeros((2 * n - 1, n + 1))
    jac[:, :n] = G
    jac[:, n] = -1.0
    return jac

def _eq_fun(x):
    return np.array([np.sum(x[:_Ng]) - _Ng * 0.5])

def _eq_jac(x):
    G = np.zeros((1, _Ng + 1))
    G[0, :_Ng] = 1.0
    return G

def _slsqp(h0, deadline, maxiter=50, ftol=1e-12):
    global _Ng
    n = len(h0)
    _Ng = n
    target = n * 0.5
    
    scores0 = _raw_scores(h0)
    T0 = float(np.max(scores0)) + 0.001
    
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = T0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]
    
    try:
        res = minimize(_obj_fun, x0, method='SLSQP', jac=_obj_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project(res.x[:n], target)
    except Exception:
        pass
    return h0.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _grad_desc(h, deadline, tau=30.0, lr=0.012):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

        max_s = float(np.max(scores))
        shifted = np.clip(tau * (scores - max_s), -50.0, 50.0)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        G = _raw_grad_matrix(h)
        grad = (w @ G) * (2.0 / n)
        mom = 0.88 * mom + 0.12 * grad
        h_new = _project(h - lr * mom, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            if c5_new < _exact_c5(best_h) - 1e-14:
                pass  # already tracked
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.02, 0.08)
            tau = min(tau * 1.01, 200.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.94
            if stall > 50:
                break
            if stall % 20 == 0:
                tau = max(20.0, tau * 0.8)
    return best_h, best_c5
