Actually, let me simplify the LP. Instead of minimizing p (the worst-case increase), I'll just minimize the maximum of the active lag values:

min t s.t. g_k^T p + s_k ≤ t for all active k, ∑p = 0, |p_j| ≤ r

Where g_k is the k-th row of G (the gradient of the k-th correlation with respect to h).

This is a cleaner formulation. The optimal t gives us the best we can do in a small neighborhood.

import numpy as np
from scipy.optimize import linprog
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_mat(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _line_search(h, p, c5_cur, max_alpha=1.0, n_coarse=10):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    alphas = np.linspace(0, alpha_max, n_coarse)
    vals = np.array([_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_coarse - 1):
        return best_a, best_c5
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(h + c1 * p)
    f2 = _c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(h + c2 * p)
    oa = (lo + hi) / 2.0
    oc = _c5(h + oa * p)
    if oc < best_c5:
        return oa, oc
    return best_a, best_c5

def _binary_hill_climb(h_init, deadline, rng):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _c5(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[ones_count - target:]] = 0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - ones_count]] = 1
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    it = 0
    while time.time() < deadline and stall < 20000:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _c5(h)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        it += 1
    return best_h, best_c5

def _binary_2swap(h_init, deadline, rng):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _c5(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline and stall < 5000:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            break
        i1 = int(ones_idx[rng.integers(len(ones_idx))])
        i2 = int(ones_idx[rng.integers(len(ones_idx))])
        j1 = int(zeros_idx[rng.integers(len(zeros_idx))])
        j2 = int(zeros_idx[rng.integers(len(zeros_idx))])
        if i1 == i2 or j1 == j2:
            stall += 1
            continue
        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
        new_c5 = _c5(h)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
            stall += 1
    return best_h, best_c5

def _lp_direction(h, top_k, radius):
    n = len(h)
    n_lags = 2 * n - 1
    scores = _corr_vals(h)
    G = _grad_mat(h)
    if top_k and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(n_lags)
    s_k = scores[idx]
    G_k = G[idx]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], scores, G
        return None, scores, G
    except Exception:
        return None, scores, G

def _lp_optimize(h, deadline, top_k=8, radius=0.15, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        p, scores, G = _lp_direction(h, top_k, r)
        if p is None:
            stall += 1
            r *= 0.7
            if r < 1e-7:
                break
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.85
            if r < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.03, 0.4)
        else:
            stall += 1
            r *= 0.85
            if stall > 50 and r < 0.03:
                r = 0.1
    return best_h, best_c5

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=800):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    cur_c5 = best_c5
    G = _grad_mat(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _corr_vals(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.1, 0.06, 0.03, 0.015, 0.008]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _c5(h)
                G = _grad_mat(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 150:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _equalize(h, deadline, n_steps=400, thresh_frac=0.95):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    cur_c5 = best_c5
    G = _grad_mat(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _corr_vals(h)
        s_max = float(scores.max())
        thresh = s_max * thresh_frac
        active = np.where(scores >= thresh)[0]
        if len(active) < 2:
            thresh_frac = max(0.5, thresh_frac * 0.95)
            continue
        active_scores = scores[active]
        target_s = float(np.mean(active_scores))
        weights = np.zeros(n_lags)
        for idx in active:
            if scores[idx] > target_s:
                weights[idx] = 1.0
        wsum = weights.sum()
        if wsum < 1e-10:
            stall += 1
            if stall > 60:
                break
            continue
        weights /= wsum
        grad = G.T @ weights
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 60:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.08, 0.04, 0.02, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _c5(h)
                G = _grad_mat(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 100:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _msequence_init(n, seed):
    """Generate a good binary initialization using m-sequence-like patterns."""
    rng = np.random.default_rng(seed)
    if n == 32:
        # Use a length-31 m-sequence padded to 32
        # m-sequence from x^5 + x^2 + 1: period 31
        # Generate it
        state = 1
        seq = []
        for _ in range(31):
            seq.append(state & 1)
            new_bit = ((state >> 4) ^ ((state >> 1) & 1)) & 1
            state = ((state << 1) | new_bit) & 0x1F
        # seq is length 31, has 16 ones and 15 zeros (or 15 ones and 16 zeros)
        seq = np.array(seq, dtype=float)
        if seq.sum() == 16.0:
            # Add a 0 to make length 32
            h = np.zeros(32)
            h[:31] = seq
            return h
        elif seq.sum() == 15.0:
            h = np.zeros(32)
            h[:31] = seq
            h[31] = 1.0
            return h
        else:
            h = np.zeros(n)
            idx = rng.choice(n, size=n//2, replace=False)
            h[idx] = 1.0
            return h
    else:
        # For other n, use a structured pattern
        h = np.zeros(n)
        # Place ones in a "spread out" pattern
        positions = np.arange(n // 2) * (2 * n // (n // 2 + 1)) % n
        # If duplicates, fix
        h[positions] = 1.0
        cnt = h.sum()
        if cnt < n // 2:
            avail = np.where(h == 0)[0]
            h[avail[:n//2 - int(cnt)]] = 1.0
        elif cnt > n // 2:
            avail = np.where(h == 1)[0]
            rng.shuffle(avail)
            h[avail[int(cnt) - n//2:]] = 0.0
        return h

def _optimize_one_n(h0, deadline, rng):
    """Run full optimization pipeline for a given n."""
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _c5(h)
    
    # Binary phase: hill climb + 2-swap
    dl_bin = min(deadline, time.time() + (deadline - time.time()) * 0.3)
    if time.time() < dl_bin:
        h_b, c5_b = _binary_hill_climb(h, min(dl_bin, time.time() + (dl_bin - time.time()) * 0.6), rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()
        dl_bin2 = min(dl_bin, time.time() + (dl_bin - time.time()) * 0.5)
        if time.time() < dl_bin2:
            h_b2, c5_b2 = _binary_2swap(h, dl_bin2, rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                h = h_b2.copy()
    
    # Continuous phase: LP
    rem = deadline - time.time()
    if rem > 5:
        dl_lp = min(deadline, time.time() + rem * 0.4)
        tk = max(4, n // 4)
        h_lp, c5_lp = _lp_optimize(h, dl_lp, top_k=tk, radius=0.2, max_iters=8000)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()
    
    # P-norm descent
    rem = deadline - time.time()
    if rem > 5:
        dl_pn = min(deadline, time.time() + rem * 0.4)
        h_pn, c5_pn = _pnorm_descent(h, dl_pn, p_start=6.0, p_end=2048.0, n_steps=600)
        if c5_pn < best_c5:
            best_c5 = c5_pn
            best_h = h_pn.copy()
            h = h_pn.copy()
    
    # Equalization
    rem = deadline - time.time()
    if rem > 5:
        dl_eq = min(deadline, time.time() + rem * 0.5)
        h_eq, c5_eq = _equalize(h, dl_eq, n_steps=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Final LP polish
    rem = deadline - time.time()
    if rem > 3:
        dl_fin = min(deadline, time.time() + rem * 0.9)
        h_fin, c5_fin = _lp_optimize(h, dl_fin, top_k=4, radius=0.05, max_iters=5000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    best_h = _project(best_h, target)
    return best_h, _c5(best_h)
