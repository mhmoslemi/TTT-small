def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corrfull(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_grad(h, corr, dx, tol=0.005):
    """Compute subgradient of C5 using active lags."""
    n = len(h)
    L = 2 * n - 1
    vals = corr * dx
    max_val = np.max(vals)
    active_mask = vals >= max_val - tol
    active_lags = np.where(active_mask)[0]
    
    if len(active_lags) == 0:
        active_lags = np.array([np.argmax(vals)])
    
    grad = np.zeros(n)
    for r in active_lags:
        d = r - (n - 1)
        # For lag d: corr[d] = sum_{i: valid} h[i] * (1-h[i+d])
        # valid: 0 <= i < n and 0 <= i+d < n
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        idx = np.arange(lo, hi)
        # d(corr[d])/dh[idx] = 1 - h[idx+d]
        grad[idx] += (1.0 - h[idx + d])
        # d(corr[d])/dh[idx+d] = -h[idx]
        # So for index k = idx+d: grad[k] += -h[idx]
        k = idx + d
        grad[k] += -h[idx]
    
    grad /= len(active_lags)
    return grad * dx

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corrfull(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(100):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_grad(h, corr, dx, tol):
    n = len(h)
    L = 2 * n - 1
    vals = corr * dx
    max_val = np.max(vals)
    active = np.where(vals >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(vals)])
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        idx = np.arange(lo, hi)
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] += -h[idx]
    grad /= len(active)
    return grad * dx

def _pair_optimize(h, corr, i, j, dx, n_eval=100):

    L = 2 * n - 1
    tau_min = max(-h[i], h[j] - 1.0)
    tau_max = min(1.0 - h[i], h[j])
    if tau_max - tau_min < 1e-12:
        return 0.0, None
    
    # Affected range
    i_lo_r = max(0, n - 1 - i)
    i_hi_r = min(L - 1, 2 * n - 2 - i)
    j_lo_r = max(0, n - 1 - j)
    j_hi_r = min(L - 1, 2 * n - 2 - j)
    aff_lo = min(i_lo_r, j_lo_r)
    aff_hi = max(i_hi_r, j_hi_r)
    if aff_lo > aff_hi:
        return 0.0, None
    
    d_vals = np.arange(aff_lo, aff_hi + 1) - (n - 1)
    m = len(d_vals)
    
    p = np.zeros(m)
    q = np.zeros(m)
    
    i_valid = (i + d_vals >= 0) & (i + d_vals < n)
    j_valid = (j + d_vals >= 0) & (j + d_vals < n)
    i_eq = (i + d_vals == j)
    j_eq = (j + d_vals == i)
    
    mk = i_valid & ~i_eq
    if np.any(mk):
        p[mk] += 1.0 - h[i + d_vals[mk]]
    
    if np.any(i_eq):
        p[i_eq] += 1.0 - h[j] + h[i]
        q[i_eq] += 1.0
    
    mk = j_valid & ~j_eq
    if np.any(mk):
        p[mk] -= (1.0 - h[j + d_vals[mk]])
    
    if np.any(j_eq):
        p[j_eq] -= (1.0 - h[i] + h[j])
        q[j_eq] -= 1.0
    
    aff_base = corr[aff_lo:aff_hi + 1]
    
    # Unaffected max
    unaff_max = -np.inf
    if aff_lo > 0:
        unaff_max = max(unaff_max, np.max(corr[:aff_lo]))
    if aff_hi < L - 1:
        unaff_max = max(unaff_max, np.max(corr[aff_hi + 1:]))
    
    # Evaluate

    # aff_new = aff_base[None,:] + t[:,None]*p[None,:] + t[:,None]**2 * q[None,:]
    tp = t[:, None] * p[None, :]
    tq = t[:, None] ** 2 * q[None, :]
    aff_new = aff_base[None, :] + tp + tq
    vals = np.maximum(aff_new, unaff_max)
    c5_vals = np.max(vals, axis=1) * dx
    
    bi = np.argmin(c5_vals)
    best_c5 = c5_vals[bi]
    best_tau = t[bi]
    
    # Refine if interior

        t2 = np.linspace(lo_t, hi_t, 30)

        tq2 = t2[:, None] ** 2 * q[None, :]
        aff_new2 = aff_base[None, :] + tp2 + tq2
        vals2 = np.maximum(aff_new2, unaff_max)
        c5_vals2 = np.max(vals2, axis=1) * dx
        fi = np.argmin(c5_vals2)
        if c5_vals2[fi] < best_c5:
            best_c5 = c5_vals2[fi]
            best_tau = t2[fi]
    
    return best_tau, best_c5

def _resample(h, n_new):
