def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

    d = r - (n - 1)

        g[d:] += 1.0 - h[:m]
        g[:m] -= h[d:]
    else:
        s = -d
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    return g

    best_val = cur_c5

    for i, t in enumerate(ts):

        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    # Golden section refinement

    gr = (np.sqrt(5.0) - 1.0) / 2.0

        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return cur_c5 + 1.0
        return _exact_c5(h_t)

    for _ in range(30):

            break
        if f1 < f2:

            c1 = hi_t - gr * (hi_t - lo_t)

        else:

            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)

    f_opt = _f(opt_t)
    if f_opt < best_val:

def _line_search_f(h, p, t, cur_c5):
    h_t = h + t * p
    if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:

def _alpha_max(h, p):
    am = 1e10
    for j in range(len(h)):
        if p[j] > 1e-14:
            am = min(am, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            am = min(am, h[j] / (-p[j]))
    return max(am, 1e-12)

def _lp_direction(h, active_r, radius):
    n = len(h)
    A = len(active_r)
    G = np.zeros((A, n))
    for i, r in enumerate(active_r):
        G[i] = _grad_row(h, r)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((A, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lb = max(-radius, -h[j])
        ub = min(radius, 1.0 - h[j])
        if lb > ub:
            lb, ub = 0.0, 0.0
        bounds.append((lb, ub))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    # Area-preserving resampling
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    # Each new cell [j*dx_new, (j+1)*dx_new] overlaps with old cells
    for j in range(n_new):
        x_lo = j * dx_new
        x_hi = (j + 1) * dx_new
        # Old cells that overlap
        i_lo = max(0, int(x_lo / dx_old))
        i_hi = min(n - 1, int((x_hi - 1e-15) / dx_old))
        if i_lo > i_hi:
            i_lo = min(n - 1, int(x_lo / dx_old))
            i_hi = i_lo
        overlap = 0.0
        for i in range(i_lo, i_hi + 1):
            cell_lo = i * dx_old
            cell_hi = (i + 1) * dx_old
            ov = max(0.0, min(x_hi, cell_hi) - max(x_lo, cell_lo))
            overlap += ov * h[i]
        h_new[j] = overlap / dx_new
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project(h_new, n_new / 2.0)

def _gradient_step(h, deadline, target, tau=20.0, lr=0.015, mom=0.9):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    velocity = np.zeros(n)
    stall = 0
    while time.time() < deadline and stall < 60:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(np.clip(shifted, -50, 0))
        w = exp_s / float(np.sum(exp_s))
        # Weighted gradient
        n_lags = 2 * n - 1
        grad = np.zeros(n)
        for r in range(n_lags):
            if w[r] < 1e-12:
                continue
            grad += w[r] * _grad_row(h, r)
        grad *= dx
        velocity = mom * velocity + (1 - mom) * grad
        h_new = h - lr * velocity
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            lr = min(lr * 1.05, 0.1)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            lr *= 0.9
            if stall % 20 == 0:
                tau = max(10.0, tau * 0.7)
    return best_h, best_c5

        # Get top lags
        active_idx = np.argsort(scores)[::-1][:top_k]
        # Try LP with just the worst lag
        p = _lp_direction(h, [active_idx[0]], radius)
        if p is not None and float(np.max(np.abs(p))) > 1e-14:
            am = _alpha_max(h, p)
            am = min(am, radius)
            alpha, new_c5 = _line_search(h, p, cur_c5, am)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                radius = min(radius * 1.15, 0.4)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                continue
        # Try with multiple lags
        if top_k >= 2:
            p = _lp_direction(h, list(active_idx[:2]), radius)
        else:
            p = None
        if p is not None and float(np.max(np.abs(p))) > 1e-14:
            am = _alpha_max(h, p)
            am = min(am, radius)
            alpha, new_c5 = _line_search(h, p, cur_c5, am)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                continue
        # Try null-space: random direction orthogonal to active gradients
        found = False
        for trial in range(5):
            p_rand = np.random.default_rng(trial * 1000 + int(cur_c5 * 10000)).standard_normal(n)
            # Project out active gradients
            for r in active_idx[:top_k]:
                g = _grad_row(h, r)
                proj = float(np.dot(p_rand, g)) / max(float(np.dot(g, g)), 1e-15)
                p_rand -= proj * g
            p_rand -= float(np.mean(p_rand))
            norm = float(np.max(np.abs(p_rand)))
            if norm < 1e-10:
                continue
            p_rand = p_rand / norm * radius
            am = _alpha_max(h, p_rand)
            am = min(am, radius)
            if am < 1e-12:
                continue
            alpha, new_c5 = _line_search(h, p_rand, cur_c5, am)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p_rand
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                found = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not found:
            stall += 1
            radius *= 0.75
            if radius < 0.005:
                break
    return best_h, best_c5

def _binary_search(n, seed):
    """Try random binary patterns and optimize fractional adjustments."""
    rng = np.random.default_rng(seed)
    target = n / 2.0
    # Random placement of exactly target ones
    k = int(target)
    ones = rng.choice(n, size=k, replace=False)
    h = np.zeros(n)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    # Initialize
    try:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
    except (NameError, AttributeError):
        h0 = None
    if h0 is not None and len(h0) > 10 and np.all(np.isfinite(h0)):
        n0 = len(h0)
        h = _project(h0, n0 / 2.0)
        best_h = h.copy()
        best_c5 = _exact_c5(h)
        best_n = n0
    else:
        best_n = 90
        best_h = _binary_search(best_n, seed)
        best_c5 = _exact_c5(best_h)

    # Phase 1: Quick warmup (5%)
    d1 = t0 + budget_s * 0.05
    h, c5 = _gradient_step(best_h, d1, best_n / 2.0, tau=15.0, lr=0.02, mom=0.88)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: LP local search (10%)
    d2 = t0 + budget_s * 0.15
    h, c5 = _lp_local(best_h, d2, radius_init=0.15, top_k=5)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 3: Active targeting (10%)
    d3 = t0 + budget_s * 0.25
    h, c5 = _active_target(best_h, d3, best_n / 2.0, top_k=3)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 4: Multi-resolution (25%)
    d4 = t0 + budget_s * 0.50
    n_current = best_n
    candidates_n = [60, 66, 72, 78, 84, 90, 96, 102, 108, 114, 120]
    candidates_n = [cn for cn in candidates_n if cn >= 36 and cn <= 200 and cn != n_current]
    candidates_n.sort(key=lambda x: abs(x - n_current))
    time_per_res = budget_s * 0.02
    for n_try in candidates_n:
        if time.time() > d4:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + time_per_res)
        h_r, c5_r = _lp_local(h_r, sub_dl, radius_init=0.12, top_k=4)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Intensive optimization at best resolution (20%)
    d5 = t0 + budget_s * 0.70
    target_best = best_n / 2.0
    # LP
    h, c5 = _lp_local(best_h, min(d5, time.time() + budget_s * 0.08), radius_init=0.1, top_k=4)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    # Active target
    if time.time() < d5:
        h, c5 = _active_target(best_h, min(d5, time.time() + budget_s * 0.04), target_best, top_k=2)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    # Pair search
    if time.time() < d5:
        h, c5 = _pair_search(best_h, min(d5, time.time() + budget_s * 0.06), target_best, max_pairs=3000)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    # Coordinate descent
    if time.time() < d5:
        h, c5 = _coord_descent(best_h, min(d5, time.time() + budget_s * 0.02), target_best)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 6: Perturbation + refine (15%)
    d6 = t0 + budget_s * 0.85
    best_h, best_c5 = _perturb_refine(best_h, d6, budget_s, seed=seed + 200, target=best_n / 2.0)

    # Phase 7: Final polishing (10%)
    d7 = t0 + budget_s * 0.96
    h, c5 = _lp_local(best_h, min(d7, time.time() + budget_s * 0.03), radius_init=0.05, top_k=3)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    if time.time() < d7:
        h, c5 = _pair_search(best_h, min(d7, time.time() + budget_s * 0.02), best_n / 2.0, max_pairs=2000)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    if time.time() < d7:
        h, c5 = _active_target(best_h, min(d7, time.time() + budget_s * 0.015), best_n / 2.0, top_k=2)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Projection slightly worsened; use the value before projection
        # Re-project more carefully
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
