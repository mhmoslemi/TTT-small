import numpy as np
import time
from scipy.optimize import minimize_scalar, minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_mean(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_coefficients(mask):
    """Compute quadratic coefficients for two-level C(d) from binary mask."""
    n = len(mask)
    m = mask.astype(np.float64)
    corr_0 = np.correlate(m, 1.0 - m, mode='full')
    corr_1 = corr_0[::-1]
    j = np.arange(2 * n - 1)
    N = np.minimum(j + 1, 2 * n - 1 - j).astype(np.float64)
    corr_05 = 0.25 * N
    
    a = corr_0
    b = -3.0 * corr_0 - corr_1 + 4.0 * corr_05
    c = 2.0 * corr_0 + 2.0 * corr_1 - 4.0 * corr_05
    return a, b, c

def eval_two_level(a, b, c, d, n):
    """Evaluate C5 for two-level parameter d with precomputed coefficients."""
    vals = a + b * d + c * d * d
    return float(np.max(vals) * 2.0 / n)

def best_two_level(a, b, c, n, d_hint=None):
    """Find optimal d in [0, 0.5] minimizing max correlation."""
    if d_hint is not None:
        # Fine grid around hint + coarse global
        ds_coarse = np.linspace(0.0, 0.5, 100)
        ds_fine = np.linspace(max(0.0, d_hint - 0.03), min(0.5, d_hint + 0.03), 100)
        ds = np.concatenate([ds_coarse, ds_fine])
    else:
        ds = np.linspace(0.0, 0.5, 200)
    
    scores = np.array([eval_two_level(a, b, c, d, n) for d in ds])
    best_idx = np.argmin(scores)
    best_d = ds[best_idx]
    best_c = scores[best_idx]
    
    # Local refinement
    lo = max(0.0, best_d - 0.02)
    hi = min(0.5, best_d + 0.02)
    if hi > lo:
        def obj(d):
            return eval_two_level(a, b, c, d, n)
        res = minimize_scalar(obj, bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
        if res.success and res.fun < best_c:
            best_d = res.x
            best_c = res.fun
    
    return best_d, best_c
