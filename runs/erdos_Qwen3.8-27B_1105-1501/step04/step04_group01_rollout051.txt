import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_vector(h):
    """Return the unnormalized correlation scores (before multiplying by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_for_scores(h, scores):
    """Compute gradient of each score component w.r.t. h. Returns (2n-1, n) matrix."""
    n = len(h)
    m_total = 2 * n - 1
    G = np.zeros((m_total, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d+m] += (1.0 - h[d:d+m])
                G[r, 0:m] += (-h[0:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, 0:m] += (1.0 - h[g:g+m])
                G[r, g:g+m] += (-h[0:m])
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _subgradient_direction(h, dx, tau):
    """Compute a weighted subgradient direction using softmax over active scores."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    # Gradient of weighted sum of scores
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[0:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:d+m] += w[r] * (-h[0:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[0:m] += w[r] * (-h[g:g+m])
                grad[g:g+m] += w[r] * (1.0 - h[0:m])
    grad -= np.mean(grad)  # Remove component along constraint direction
    return grad

def _lp_direction_fast(h, scores, G, radius):
    """Find LP direction to reduce max score."""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _local_optimize(h, deadline, use_lp=True, radius_init=0.12):
    """Local optimization loop with LP or subgradient directions."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    tau = 100.0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        d = None
        if use_lp:
            G = _grad_for_scores(h, scores)
            d = _lp_direction_fast(h, scores, G, radius)
        if d is None or float(np.max(np.abs(d))) < 1e-13:
            # Fall back to subgradient
            d = _subgradient_direction(h, dx, tau)
            dn = np.linalg.norm(d)
            if dn < 1e-13:
                stall += 1
                radius *= 0.7
                tau *= 1.3
                if radius < 1e-7:
                    break
                continue
            d = radius * d / dn
        alpha, new_c5 = _line_search(h, d, dx, radius if use_lp else radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
            tau = max(tau / 1.05, 20.0)
        else:
            stall += 1
            radius *= 0.8
            tau = min(tau * 1.15, 2000.0)
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)

            best_h = h_r.copy()

    # Phase 5: Pair refinement (10%)
    d5 = t0 + budget_s * 0.87
    h_p, c5_p = _pair_refine(best_h, d5, max_passes=20)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 6: Final LP polish (10%)
    d6 = t0 + budget_s * 0.98
    h_f, c5_f = _local_opt(best_h, d6, use_lp=True, radius_init=0.03)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
