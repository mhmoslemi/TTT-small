def exact_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(v, target):
    n = len(v)
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0

            lo = mid
        else:
            hi = mid
