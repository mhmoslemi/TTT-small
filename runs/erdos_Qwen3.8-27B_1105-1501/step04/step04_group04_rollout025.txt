def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()

        else:

        cnt = int(np.sum(mask))
        if cnt == 0:
            break

def _gradient_c5(h, n_active=10):
    n = len(h)
    dx = 2.0 / n
    F = _c5_vector(h)
    r0 = n - 1
    F_max = np.max(F)
    active = np.where(F > F_max - 1e-10 * max(1.0, F_max))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(F))])
    if len(active) > n_active:
        active = np.argsort(F)[::-1][:n_active]
    grad = np.zeros(n)
    for r in active:
        d = r - r0
        lo = max(0, -d)
        hi = min(n, n - d)
        for m in range(lo, hi):
            grad[m] += dx * (1.0 - h[m + d])
        for m in range(lo + d, hi + d):
            if 0 <= m < n:
                grad[m] -= dx * h[m - d]
    return grad
