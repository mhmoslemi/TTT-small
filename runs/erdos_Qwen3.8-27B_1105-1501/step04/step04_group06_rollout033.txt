def _project(v, target):

    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:

        else:

                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a

                if diff <= 1e-12:

    return float(np.max(_corr(h)) * 2.0 / n)

    """Compute gradient matrix G where G[r] = grad of correlation r w.r.t. h."""
    G = np.zeros((2 * n - 1, n))

        tgt = j - s
        m_tgt = (tgt >= 0) & (tgt < n)
        G[r, m_tgt] -= h[tgt[m_tgt]]
    return G

    A_eq = np.zeros((1, n + 1))

    bounds.append((0.0, None))

def _project(v, target):
    n = len(v)

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)
            for i in order:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:

            for i in order:
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return float(np.max(_corr_full(h)) * 2.0 / n)


def _grad_rows(h, n, rows):
    """Compute gradient rows for specified correlation indices."""
    j = np.arange(n)
    G = np.zeros((len(rows), n))
    for ki, r in enumerate(rows):
        s = r - (n - 1)
        src = j + s
        m1 = (src >= 0) & (src < n)
        G[ki, m1] += 1.0 - h[src[m1]]
        tgt = j - s
        m2 = (tgt >= 0) & (tgt < n)
        G[ki, m2] -= h[tgt[m2]]
    return G


def _lp_solve(h, corr, G, n, extra_radius=None):
    """Solve the LP minimax. Returns delta, t."""
    K = len(corr)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -corr
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_i = -h[i]
        hi_i = 1.0 - h[i]
        if extra_radius is not None:
            lo_i = max(lo_i, -extra_radius)
            hi_i = min(hi_i, extra_radius)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, None))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return None
    return res.x[:n], res.x[-1]
