import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, :m] += -h[d:d+m]
            G[r, d:] += 1.0 - h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += 1.0 - h[g:g+m]
            G[r, g:] += -h[:m]
    return G

def _slsqp_solve(h, deadline, maxiter=80):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    scores = _raw_scores(h)
    T0 = float(np.max(scores)) + 0.01
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def constr_func(x):
        hv = x[:n]
        T = x[n]
        s = _raw_scores(hv)
        return T - s  # must be >= 0
    
    def constr_jac(x):
        hv = x[:n]
        G = _grad_matrix(hv)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    
    def eq_func(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = T0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    
    constraints = [
        {'type': 'ineq', 'fun': constr_func, 'jac': constr_jac},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac},
    ]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                       constraints=constraints, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_out = _project(res.x[:n], target)
            c5 = _exact_c5(h_out)
            return h_out, c5
    except Exception:
        pass
    return h, _exact_c5(h)

    h_r, c5_r = _slsqp_optimize(best_h, d7, maxiter=40, top_k=30)
    if c5_r < best_c5 - 1e-12:
        best_c5, best_h = c5_r, h_r

    # Final pair polish
    if time.time() < deadline_final:
        h_r, c5_r = _pair_refine(best_h, deadline_final, max_passes=20)
        if c5_r < best_c5 - 1e-12:
            best_c5, best_h = c5_r, h_r

    # Final validation
    best_h = _project(best_h.copy(), best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
