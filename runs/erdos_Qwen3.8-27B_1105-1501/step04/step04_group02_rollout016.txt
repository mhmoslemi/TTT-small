import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _get_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx, corr  # physical, raw

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, r):
    """Gradient of raw correlation score r with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        grad[d:] += 1.0
        grad[d:] -= h[:m]
        grad[:m] -= h[d:]
    else:
        g = -d
        m = n - g
        grad[:m] += 1.0
        grad[:m] -= h[g:]
        grad[g:] -= h[:m]
    return grad

def _grad_all(h):
    """All gradient rows, shape (2n-1, n)."""
    n = len(h)
    rows = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        rows[r] = _grad_row(h, r)
    return rows

def _line_search(h, p, c5_cur, alpha_max, n_grid=40):
    """Line search along direction p."""
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            v = _exact_c5(h_t)
            if v < best_val:
                best_val = v
                best_t = t
    # Refine with golden section
    if best_t > 1e-14:
        idx = int(np.argmin(vals))
        lo_t = ts[max(0, idx - 3)]
        hi_t = ts[min(n_grid - 1, idx + 3)]
        if hi_t > lo_t:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_t - gr * (hi_t - lo_t)
            c2 = lo_t + gr * (hi_t - lo_t)
            f1 = _exact_c5(h + c1 * p) if (h + c1 * p).min() >= -1e-12 and (h + c1 * p).max() <= 1.0 + 1e-12 else c5_cur + 1
            f2 = _exact_c5(h + c2 * p) if (h + c2 * p).min() >= -1e-12 and (h + c2 * p).max() <= 1.0 + 1e-12 else c5_cur + 1
            for _ in range(40):
                if hi_t - lo_t < 1e-14:
                    break
                if f1 < f2:
                    hi_t, c2, f2 = c2, c1, f1
                    c1 = hi_t - gr * (hi_t - lo_t)
                    h_c1 = h + c1 * p
                    f1 = _exact_c5(h_c1) if h_c1.min() >= -1e-12 and h_c1.max() <= 1.0 + 1e-12 else c5_cur + 1
                else:
                    lo_t, c1, f1 = c1, c2, f2
                    c2 = lo_t + gr * (hi_t - lo_t)
                    h_c2 = h + c2 * p
                    f2 = _exact_c5(h_c2) if h_c2.min() >= -1e-12 and h_c2.max() <= 1.0 + 1e-12 else c5_cur + 1
            opt_t = (lo_t + hi_t) / 2.0
            h_opt = h + opt_t * p
            if h_opt.min() >= -1e-12 and h_opt.max() <= 1.0 + 1e-12:
                f_opt = _exact_c5(h_opt)
                if f_opt < best_val:
                    best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_direction(h, active_lags, radius, scores_raw):
    """LP: find direction p that decreases all active scores."""
    n = len(h)
    m = len(active_lags)
    if m == 0:
        return None
    # Build constraint matrix for active lags
    # For each active lag r: grad_row(r) @ p <= -delta (we want to decrease)
    # But actually we want: score_r + grad_r @ p <= max_score - epsilon
    # So: grad_r @ p <= max_score - epsilon - score_r = -epsilon (if score_r = max_score)
    # More generally for active set: grad_r @ p <= -epsilon for all r in active
    
    # Actually, let's minimize t subject to:
    # grad_r @ p <= t - score_r for all r in active
    # |p|_inf <= radius
    # sum(p) = 0
    
    # This is: minimize t
    # s.t. grad_r @ p - t <= -score_r  for all active r
    #      p_j <= radius
    #      p_j >= -radius
    #      sum(p) = 0
    #      (and box: -h_j <= p_j <= 1-h_j)
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, r in enumerate(active_lags):
        A_ub[i, :n] = _grad_row(h, r)
        A_ub[i, n] = -1.0
        b_ub[i] = -scores_raw[r]
    
    # Box constraints for p
    A_box = np.zeros((2 * n, n + 1))
    b_box = np.zeros(2 * n)
    for j in range(n):
        A_box[j, j] = 1.0
        b_box[j] = min(radius, 1.0 - h[j])
        A_box[n + j, j] = -1.0
        b_box[n + j] = min(radius, h[j])
    
    A_all = np.vstack([A_ub, A_box])
    b_all = np.concatenate([b_ub, b_box])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0  # minimize t
    
    try:
        res = linprog(c_vec, A_ub=A_all, b_ub=b_all, A_eq=A_eq, b_eq=b_eq, method='highs')
        if res.success:
            p = res.x[:n]
            if res.x[-1] < 0:  # t < 0 means we're decreasing
                return p
            # Even if t >= 0, the direction might still help if line search finds improvement
            if float(np.max(np.abs(p))) > 1e-12:
                return p
    except Exception:
        pass
    return None

def _active_LP_search(h, deadline, radius_init=0.15, max_top=10):
    """Iterative LP descent on active lags."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline and stall < 100:
        scores_phys, scores_raw = _get_scores(h)
        max_score = float(np.max(scores_raw))
        
        # Find active lags: those within 1% of max
        threshold = max_score * 0.999
        active = np.where(scores_raw >= threshold)[0]
        
        # Try with different subsets
        found = False
        for frac in [1.0, 0.7, 0.5, 0.3]:
            k = max(1, int(len(active) * frac))
            # Take the top k
            top_active = active[np.argsort(scores_raw[active])[::-1][:k]]
            
            p = _lp_direction(h, top_active, radius, scores_raw)
            if p is None:
                continue
            
            # Compute alpha_max
            alpha_max = radius
            for j in range(n):
                if p[j] > 1e-14:
                    alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                elif p[j] < -1e-14:
                    alpha_max = min(alpha_max, h[j] / (-p[j]))
            alpha_max = max(alpha_max, 1e-10)
            
            alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
            if new_c5 < cur_c5 - 1e-13:

                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                found = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                radius = min(radius * 1.1, 0.5)
                break
        if not found:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
    
    return best_h, best_c5

def _threshold_squeeze(h, deadline, n_steps=200, lr_init=0.01):
    """Minimize sum of squared exceedances above a threshold."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    
    while time.time() < deadline and stall < 50:
        scores_phys, scores_raw = _get_scores(h)
        max_raw = float(np.max(scores_raw))
        # Set threshold just below current max
        tau = max_raw * 0.9995
        dx = 2.0 / n
        
        # Compute gradient of P(h) = sum max(score_r - tau, 0)^2
        # P = sum_r max(raw_r - tau, 0)^2
        # dP/dh = sum_r 2*max(raw_r - tau, 0) * grad_raw_row(r)
        exceed = np.maximum(scores_raw - tau, 0.0)
        n_exceed = int(np.sum(exceed > 1e-12))
        if n_exceed == 0:
            break
        
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if exceed[r] > 1e-12:
                grad += 2.0 * exceed[r] * _grad_row(h, r)
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        
        # Project gradient to sum-zero
        grad -= np.mean(grad)
        
        # Try step
        h_new = h - lr * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.05, 0.08)
        else:
            stall += 1
            lr *= 0.85
            if stall > 40:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    # Area-preserving resampling
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_lo = j * dx_new
        x_hi = (j + 1) * dx_new
        total = 0.0
        length = 0.0
        for i in range(n):
            o_lo = i * dx_old
            o_hi = (i + 1) * dx_old
            overlap = min(x_hi, o_hi) - max(x_lo, o_lo)
            if overlap > 0:
                total += h[i] * overlap
                length += overlap
        if length > 0:
            h_new[j] = total / length
    return _project(h_new, n_new / 2.0)
