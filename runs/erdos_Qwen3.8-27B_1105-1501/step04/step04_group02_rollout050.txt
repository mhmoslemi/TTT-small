def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:

        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    # Area-preserving resampling
    # Each old cell i covers [i*dx_old, (i+1)*dx_old]
    # Each new cell j covers [j*dx_new, (j+1)*dx_new]
    # new_h[j] = sum_i h[i] * overlap(i,j) / dx_new
    # Use interpolation for simplicity, then project
    x_old = (np.arange(n) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_perturb(h, scale, mode, n_harmonics=5):
    """Apply structured perturbation and project."""
    n = len(h)
    x = np.arange(n)
    if mode == 0:  # Low-frequency sinusoidal
        pert = np.zeros(n)
        for k in range(1, n_harmonics + 1):
            pert += np.sin(2.0 * np.pi * k * x / n) / k
        pert = pert / np.max(np.abs(pert)) if np.max(np.abs(pert)) > 0 else pert
    elif mode == 1:  # Block perturbation
        block_size = max(3, n // 10)
        pert = np.zeros(n)
        for b in range(0, n, block_size):
            pert[b:b+block_size] = 1.0 if (b // block_size) % 2 == 0 else -1.0
    else:  # Gaussian
        rng = np.random.default_rng(mode * 1000)
        pert = rng.normal(0, 1, n)
        pert -= np.mean(pert)
    
    h_new = h + scale * pert
    return _project(h_new, n / 2.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_solve(h, deadline, maxiter=50):

    target = n / 2.0
    dx = 2.0 / n
    scores_init = np.correlate(h, 1.0 - h, mode='full')
    T0 = float(np.max(scores_init)) + 0.005

    def _obj(x):
        return x[n]

    def _obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def _ineq(x):
        hvec = x[:n]
        s = np.correlate(hvec, 1.0 - hvec, mode='full')
        return x[n] - s

    def _ineq_jac(x):
        hvec = x[:n]
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        for r in range(m):
            d = r - (n - 1)
            if d >= 0:
                cnt = n - d
                J[r, d:d+cnt] += 1.0 - hvec[:cnt]
                J[r, :cnt] -= hvec[d:d+cnt]
            else:
                g = -d
                cnt = n - g
                J[r, :cnt] += 1.0 - hvec[g:g+cnt]
                J[r, g:g+cnt] -= hvec[:cnt]
            J[r, n] = 1.0
        return J

    def _eq(x):
        return np.sum(x[:n]) - target

    def _eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J

    x0 = np.concatenate([h, [T0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
