import numpy as np
from scipy.optimize import minimize

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def smooth_loss(h, tau):
    c = corr_scaled(h)
    cmax = np.max(c)
    if cmax < 1e-15:
        return cmax, np.zeros_like(h)
    vals = np.exp((c - cmax) / tau)
    s = np.sum(vals)
    lse = cmax + tau * np.log(s)
    w = vals / s
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[d:] += wk * (1.0 - h[:n - d])
            grad[:n - d] -= wk * h[d:]
        else:
            dp = -d
            grad[:n - dp] += wk * (1.0 - h[dp:])
            grad[dp:] -= wk * h[:n - dp]
    return lse, grad


def smooth_slsqp(h0, tau, n, target, maxiter=200, ftol=1e-14):
    def obj(x):
        v, _ = smooth_loss(x, tau)
        return v

    def obj_grad(x):
        _, g = smooth_loss(x, tau)
        return g

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n

        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0.copy()


def active_slsqp(h0, n, target, active_indices, maxiter=200, ftol=1e-13):
    def obs(x):

        return c[active_indices].max()

        c = corr_scaled(x)
        mx = c[active_indices].max()
        top = active_indices[np.where(c[active_indices] >= mx - 1e-10)[0]]
        w = np.ones(len(top))
        grad = np.zeros(n)
        for k in top:

def get_active(h, frac=0.4, min_count=8):

    return active


def resample(h, n_new):
    n = len(h)
    if n_new == n:

    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
