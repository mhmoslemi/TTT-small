def _penalty_objective(x, n, mu):
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    violations = np.maximum(0.0, corr - t)
    f = t + mu * np.sum(violations**2)
    # Gradient
    j = np.arange(n)
    grad_h = np.zeros(n)
    active = violations > 0
    for r in range(2*n-1):
        if not active[r]:
            continue
        s = r - (n-1)
        # d corr_r / d h
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        dcorr = np.zeros(n)
        dcorr[m1] = 1.0 - h[gs[m1]]
        dcorr[m2] -= h[ps[m2]]
        grad_h += 2 * mu * violations[r] * dcorr
    grad_t = 1.0 - mu * 2 * np.sum(violations)
    grad = np.concatenate([grad_h, [grad_t]])
    return f, grad

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float).copy()
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)
            for i in order:
                if x[i] >= 1.0: continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12: break
        else:
            order = np.argsort(-x)
            for i in order:
                if x[i] <= 0.0: continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12: break
    return x

    # d(corr_r)/dh_j = (1-h_{j+s}) - h_{j-s} (where appropriate indices exist)

    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    viol = np.maximum(0.0, corr - t)

def _slsqp_eq_f(z, n):
    return np.sum(z[:n]) - 0.5 * n


def _slsqp_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _slsqp_ineq_f(z, n, idx):
    h, t = z[:n], z[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr[idx]


def _slsqp_ineq_g(z, n, idx):
    h = z[:n]
    nc = 2 * n - 1
    j = np.arange(n)
    J = np.zeros((len(idx), n + 1))

        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        J[ii, m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        J[ii, m2] -= h[ps[m2]]
        J[ii, n] = 1.0
        J[ii, :n] *= -1.0
    return J


def _slsqp_solve(h_init, n, idx, maxiter=200, ftol=1e-14):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _lse_f(x, n, alpha):
    h = x[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -400.0, 400.0)
    ev = np.exp(vals)
    s = float(np.sum(ev))
    f = ms + (1.0 / alpha) * np.log(s)
    f += 100.0 * (np.sum(h) - 0.5 * n) ** 2
    return f


def _lse_jac(x, n, alpha):
    h = x[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -400.0, 400.0)
    ev = np.exp(vals)
    s = float(np.sum(ev))
    w = ev / s
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        s_r = r - (n - 1)
        gs = j - s_r
        m1 = (gs >= 0) & (gs < n)
        grad[m1] += w[r] * (1.0 - h[gs[m1]])
        ps = j + s_r
        m2 = (ps >= 0) & (ps < n)
        grad[m2] -= w[r] * h[ps[m2]]
    grad += 200.0 * (np.sum(h) - 0.5 * n)
    return grad


def _lse_step(h, n, alpha, maxiter=100):
