import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _fit_quad(mask):
    m = mask.astype(np.float64)
    cm = 1.0 - m
    n11 = np.correlate(m, m, 'full')
    n10 = np.correlate(m, cm, 'full')
    n01 = np.correlate(cm, m, 'full')
    n00 = np.correlate(cm, cm, 'full')
    A = n01 + n10 - n11 - n00
    B = n11 - 2.0 * n10 + n00
    C = n10
    return A, B, C


def _c5_from_quad(A, B, C, d, mask_n):
    dx = 2.0 / mask_n
    c = A * d * d + B * d + C
    return float(np.max(c) * dx)


def _find_optimal_d(A, B, C, mask_n):
    dx = 2.0 / mask_n
    ds = np.linspace(0.0, 0.5, 300)
    cmat = A[np.newaxis, :] * ds[:, np.newaxis] ** 2 + B[np.newaxis, :] * ds[:, np.newaxis] + C[np.newaxis, :]
    c5s = np.max(cmat, axis=1) * dx
    idx = int(np.argmin(c5s))
    best_d = float(ds[idx])
    best_c5 = float(c5s[idx])
    step = 0.5 / 300.0
    lo = max(0.0, best_d - 2.0 * step)
    hi = min(0.5, best_d + 2.0 * step)
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = _c5_from_quad(A, B, C, m1, mask_n)
        c2 = _c5_from_quad(A, B, C, m2, mask_n)
        if c1 < c2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-12:
            break
    best_d = (lo + hi) / 2.0
    best_c5 = _c5_from_quad(A, B, C, best_d, mask_n)
    return best_d, best_c5


def _binary_local_search(mask, mask_n, deadline, max_passes=200):
    ones = np.where(mask == 1)[0]
    zeros = np.where(mask == 0)[0]
    A, B, C = _fit_quad(mask)
    best_d, best_c5 = _find_optimal_d(A, B, C, mask_n)
    best_mask = mask.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        best_swap_c5 = best_c5
        best_swap_i = -1
        best_swap_j = -1
        for ii in range(len(ones)):
            if time.time() > deadline:
                break
            i = ones[ii]
            for jj in range(len(zeros)):
                j = zeros[jj]
                if i == j:
                    continue
                mask[i] = 0
                mask[j] = 1
                A2, B2, C2 = _fit_quad(mask)
                d2, c52 = _find_optimal_d(A2, B2, C2, mask_n)
                mask[i] = 1
                mask[j] = 0
                if c52 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c52
                    best_swap_i = i
                    best_swap_j = j
        if best_swap_i >= 0:
            best_mask[best_swap_i] = 0
            best_mask[best_swap_j] = 1
            ones = np.where(best_mask == 1)[0]
            zeros = np.where(best_mask == 0)[0]
            A, B, C = _fit_quad(best_mask)
            best_d, best_c5 = _find_optimal_d(A, B, C, mask_n)
            best_mask = best_mask
            improved = True
        if not improved:
            break
    return best_mask, best_d, best_c5


def _gen_periodic_mask(n, period, fill, offset=0):
    mask = np.zeros(n, dtype=np.int32)
    for i in range(n):
        if ((i + offset) % period) < fill:
            mask[i] = 1
    cnt = mask.sum()
    need = n // 2
    if cnt > need:
        excess = cnt - need
        ones = np.where(mask == 1)[0]
        rng = np.random.default_rng(0)
        flip = rng.choice(len(ones), size=int(excess), replace=False)
        mask[ones[flip]] = 0
    elif cnt < need:
        deficit = need - cnt
        zeros = np.where(mask == 0)[0]
        if len(zeros) < deficit:
            return None
        rng = np.random.default_rng(0)
        flip = rng.choice(len(zeros), size=int(deficit), replace=False)
        mask[zeros[flip]] = 1
    return mask


def _gen_random_mask(n, seed):
    rng = np.random.default_rng(seed)
    pos = rng.permutation(n)[:n // 2]
    mask = np.zeros(n, dtype=np.int32)
    mask[pos] = 1
    return mask


def _pair_line_search(h, deadline, max_sweeps=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    for _sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(_sweep * 1000 + 42)
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
        rng.shuffle(pairs)
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            p = h[i] + h[j]
            lo = max(0.0, p - 1.0)
            hi = min(1.0, p)
            if hi - lo < 1e-10:
                continue
            ts = np.linspace(lo, hi, 17)
            best_t = h[i]
            best_c5_local = best_c5
            for t in ts:
                h[i] = t
                h[j] = p - t
                c5_t = _exact_c5(h)
                if c5_t < best_c5_local - 1e-13:
                    best_c5_local = c5_t
                    best_t = t
            h[i] = best_t
            h[j] = p - best_t
            if best_c5_local < best_c5 - 1e-13:
                best_c5 = best_c5_local
                improved = True
        if not improved:
            break
    return h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _slsqp_refine(h, deadline, maxiter=300):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return x[0]

    def g_vec(hh):
        return np.correlate(hh, 1.0 - hh, mode='full')

    g_full = g_vec(h)
    max_g = float(np.max(g_full))
    threshold = max_g * 0.5
    active = np.where(g_full >= threshold)[0]
    if len(active) < 5:
        active = np.argsort(g_full)[-15:]

    def ineq(x):
        hh = x[1:]
        g = g_vec(hh)
        return x[0] - g[active]

    def ineq_jac(x):
        hh = x[1:]
        J = np.zeros((len(active), n + 1))
        J[:, 0] = 1.0
        for idx, j in enumerate(active):
            d = j - (n - 1)
            if d >= 0:
                idx1 = np.arange(d, n)
                J[idx, 1 + idx1 - 0] += (1.0 - hh[idx1 - d])
                if d < n:
                    idx2 = np.arange(0, n - d)
                    J[idx, 1 + idx2] -= hh[idx2 + d]
            else:
                dp = -d
                idx1 = np.arange(0, n - dp)
                J[idx, 1 + idx1] += (1.0 - hh[idx1 + dp])
                idx2 = np.arange(dp, n)
                J[idx, 1 + idx2] -= hh[idx2 - dp]
        return J

    def eq(x):
        return x[1:].sum() - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    x0 = np.zeros(n + 1)
    x0[0] = max_g * 1.01 + 0.01
    x0[1:] = h
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n

    try:
        res = minimize(obj, x0, jac=lambda x: np.concatenate([[1.0], np.zeros(n)]),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
                                    {'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': 0})
        if res.success and np.all(np.isfinite(res.x)):
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h
