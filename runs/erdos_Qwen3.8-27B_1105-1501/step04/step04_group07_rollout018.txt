        return np.zeros(n)

    #             -rho <= delta[i] <= rho
    #             h[i] - rho <= delta[i] <= 1 - h[i]  (box constraints from h+delta in [0,1])

        hi_b = min(rho, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((-2.0, 2.0))  # tau bounds

    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, cur_c5, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-14)
    if alpha_max < 1e-14:
        return 0.0, cur_c5
    n_s = 12
    alphas = np.linspace(0, alpha_max, n_s)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if 0 < best_i < n_s - 1:
        lo = float(alphas[best_i - 1])
        hi = float(alphas[best_i + 1])
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(18):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, f2 = c1, f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo, f1 = c1, f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        oa = (lo + hi) / 2.0
        oc = _exact_c5(h + oa * p)
        if oc < best_c5:
            best_a, best_c5 = oa, oc
    return best_a, best_c5

def _full_lp_step(h, rho):
    n = len(h)
    G = _grad_matrix(h)
    s = _scores(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.empty((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -s
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo_b = max(-rho, -h[j])
        hi_b = min(rho, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((-2.0, 2.0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:

        improved = False
        for j in range(n):
            if time.time() > deadline:

            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:

            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:

                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:

                        m2 = hi - (hi - lo) / 3.0

                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue

                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt

                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()

import numpy as np
from scipy.optimize import linprog
import time
import math

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:])
            G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, cur_c5, max_alpha=1.0):

def _full_lp_step(h, rho):
    n = len(h)
    G = _grad_matrix(h)
    s = _scores(h)
    n_lags = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.empty((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -s
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo_b = max(-rho, -h[j])
        hi_b = min(rho, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((-2.0, 2.0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _optimize(h_init, deadline, rho_init=0.4, min_rho=0.003):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
