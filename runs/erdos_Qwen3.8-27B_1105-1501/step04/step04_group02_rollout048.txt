    """Compute C5 using the official formula."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, d:d+m] += (1.0 - h[0:m])
            G[r, 0:m] += -h[d:d+m]
        else:
            s = -d
            m = n - s
            G[r, 0:m] += (1.0 - h[s:s+m])
            G[r, s:s+m] += -h[0:m]
    return G

    return x[n]

    g[n] = 1.0
    return g

    T = x[n]
    s = _scores(h)
    return T - s
