import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] += dx * (1.0 - h[d:])
            G[r, d:] += -dx * h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += dx * (1.0 - h[g:])
            G[r, g:] += -dx * h[:m]
    return G

def _max_score_at_alpha(s0, B, Q, alpha):
    return float(np.max(s0 + B * alpha - Q * alpha * alpha))

def _line_search(h, V, s0, max_iter=50):
    n = len(h)
    # Find feasible alpha range
    alpha_max = 1.0
    for j in range(n):
        if V[j] > 1e-14:
            a = (1.0 - h[j]) / V[j]
            if a < alpha_max:
                alpha_max = a
        elif V[j] < -1e-14:
            a = h[j] / (-V[j])
            # This gives alpha_min if we go negative, but we're going positive
            # Actually for V[j] < 0, h[j] + alpha*V[j] >= 0 means alpha <= h[j]/(-V[j])
            if a < alpha_max:
                alpha_max = a
    alpha_max = max(alpha_max, 1e-12)
    
    # Compute quadratic coefficients
    dx = 2.0 / n
    B = dx * (np.correlate(V, 1.0 - h, mode='full') - np.correlate(h, V, mode='full'))
    Q = dx * np.correlate(V, V, mode='full')
    
    # Golden section search on [0, alpha_max]
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    lo, hi = 0.0, alpha_max
    f_lo = _max_score_at_alpha(s0, B, Q, lo)
    if f_lo < float(np.max(s0)) - 1e-15:
        return lo, f_lo  # Already at boundary and improving (shouldn't happen)
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _max_score_at_alpha(s0, B, Q, c1)
    f2 = _max_score_at_alpha(s0, B, Q, c2)
    for _ in range(max_iter):
        if hi - lo < 1e-12:
            break
        if f1 <= f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _max_score_at_alpha(s0, B, Q, c1)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _max_score_at_alpha(s0, B, Q, c2)
    best_a = (lo + hi) * 0.5
    best_f = _max_score_at_alpha(s0, B, Q, best_a)
    # Also check endpoints
    f0 = float(np.max(s0))
    fmax = _max_score_at_alpha(s0, B, Q, alpha_max)
    if f0 < best_f:
        best_a, best_f = 0.0, f0
    if fmax < best_f:
        best_a, best_f = alpha_max, fmax
    return best_a, best_f

def _optimize_core(h, deadline, rng, max_iters=100000):
    """Core optimization using subgradient + exact 1D search"""
    n = len(h)
    target = n * 0.5
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    s0 = _scores(h)
    stall = 0
    it = 0
    momentum = np.zeros(n)
    
    while time.time() < deadline and it < max_iters:
        it += 1
        s_max = float(np.max(s0))
        
        # Determine active set
        thresh = s_max - 0.005  # lags within 0.005 of max are "active"
        active = np.where(s0 >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(s0))])
        
        # Compute subgradient direction
        # grad = average of gradients of active lags
        G = _grad_matrix(h)
        w = np.zeros(2 * n - 1)
        # Weight by how close to max (emphasize the most active)
        gaps = s_max - s0[active]  # these are small (<= 0.005)
        weights = 1.0 + 100.0 * (s_max - s0[active])  # emphasize the max lag
        w[active] = weights
        w /= np.sum(w)
        grad = G.T @ w
        grad -= np.mean(grad)  # project onto sum=0
        
        gn = np.linalg.norm(grad)
        
        if gn < 1e-14:
            # Try a random direction
            V = rng.standard_normal(n)
            V -= np.mean(V)
            gn = np.linalg.norm(V)
            if gn < 1e-14:
                break
            V /= gn
        else:
            V = -grad / gn
            # Add momentum
            momentum = 0.8 * momentum + 0.2 * V
            mn = np.linalg.norm(momentum)
            if mn > 1e-14:
                V = momentum / mn
                V -= np.mean(V)
                V /= np.linalg.norm(V)
        
        # Line search
        alpha, new_c5 = _line_search(h, V, s0)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * V
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            s0 = _scores(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            if stall > 200:
                # Try a random perturbation
                h = h + 0.01 * rng.standard_normal(n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                s0 = _scores(h)
                stall = 0
            elif stall % 50 == 0:
                # Try pair direction
                i, j = int(rng.integers(n)), int(rng.integers(n))
                if i == j:
                    j = (j + 1) % n
                V_pair = np.zeros(n)
                V_pair[i] = 1.0
                V_pair[j] = -1.0
                V_pair -= np.mean(V_pair)
                V_pair /= np.linalg.norm(V_pair)

                if new_c5_2 < cur_c5 - 1e-14:

                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    s0 = _scores(h)
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
    
    return best_h, best_c5

def _project(v, target):

        return np.zeros(n)

    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0

            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))

        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] += dx * (1.0 - h[d:])
            G[r, d:] += -dx * h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += dx * (1.0 - h[g:])
            G[r, g:] += -dx * h[:m]
    return G
