def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    """Project v to [0,1]^n with sum = target using water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix rounding
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute gradient matrix G where G[r, i] = d(score_r)/d(h_i) * dx."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        # For shift d >= 0: score = sum_{i=0}^{m-1} h[i]*(1-h[i+d])
        G[r, :m] += (1.0 - h[d:]) * dx
        G[r, d:] += (-h[:m]) * dx
    for d in range(1, n):
        r = n - 1 - d
        m = n - d
        # For shift -d: score = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
        G[r, d:] += (1.0 - h[:m]) * dx
        G[r, :m] += (-h[d:]) * dx
    return G

def _lp_active(h, scores, G, radius, active_set, gap=0.002):
    """Solve LP for direction using active constraint set."""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = -1.0  # maximize t (minimize -t)
    
    A_ub = G[active_set].copy()
    A_ub = np.hstack([A_ub, np.ones((len(active_set), 1))])
    b_ub = np.zeros(len(active_set))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j] + 1e-15)
        hi = min(radius, 1.0 - h[j] - 1e-15)
        if lo >= hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search_quad(h, p, b_coeffs=None):
    """Line search using quadratic model of scores."""
    n = len(h)
    # Compute alpha_max
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    
    # Sample 20 points
    n_samp = 20
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    
    if best_i in (0, n_samp - 1):
        return float(best_a), float(best_c5)
    
    # Golden section refinement
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return float(opt_a), float(opt_c5)
    return float(best_a), float(best_c5)

def _weighted_grad(h, deadline, n_steps=200, gamma_init=10.0):
    """Weighted gradient descent with softmax weights on scores."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    gamma = gamma_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        gamma = gamma_init + 200.0 * frac
        
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(gamma * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        alpha = 0.1
        improved = False
        for _ in range(35):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if not improved and step > 50:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _continuum_anneal(h, deadline, rng, n_iter=500):
    """Continuous simulated annealing: large Gaussian moves + cooling."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    sigma = 0.15
    cool = 0.997
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        h_try = h + sigma * rng.standard_normal(n)
        h_try = _project(h_try, target)
        new_c5 = _exact_c5(h_try)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (sigma * 0.1)):
            h = h_try
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        sigma *= cool
    
    return best_h, best_c5

def _quasi_init(n, alpha):
    """Create quasi-periodic initial function."""
    x = np.arange(n) / n
    h = (x * alpha) % 1.0
    # Smooth it a bit
    h = 0.5 + 0.3 * np.sin(2 * np.pi * h)
    h = np.clip(h, 0, 1)
    h = _project(h, n / 2.0)
    return h

def _fourier_bump_init(n, rng, n_freqs=5):
    """Initialize with random Fourier bumps at incommensurable frequencies."""
    x = np.arange(n) / n

        h += amp * np.sin(2 * np.pi * f * x + phase)
    h = np.clip(h, 0, 1)

    return h

Actually, there's an even simpler way: since f is convex and piecewise linear, we can use a ternary search (or better, find the point where the subgradient changes sign).

The subgradient of f at point t is the set {B_r : r is argmax}. f is minimized where 0 ∈ subgradient, i.e., where there exist r1, r2 with B_{r1} < 0 < B_{r2} (or B_r = 0 for some r on the envelope).

Algorithm:
1. Evaluate f at t = 0 and t = 1
2. If f(0) ≤ f(1): the minimum is in [0, some point] where the subgradient transitions from negative to non-negative
3. Use bisection: at midpoint t, find which constraints are active (on the upper envelope). If all active B_r < 0, decrease t. If all active B_r > 0, increase t.

Actually, the simplest approach for 1D piecewise linear convex minimization:
- Evaluate f at 0 and 1
- The minimum is where the "slope" changes from negative to positive
- Use bisection on the slope: at point t, if the slope of f at t is negative, move right; if positive, move left

The "slope" at point t (when f is differentiable) is B_r where r is the unique active constraint. When f is not differentiable (two constraints are tied), the subgradient is the interval [min active B_r, max active B_r].

So the algorithm is:
- lo = 0, hi = 1
- For 30 iterations:
  - mid = (lo + hi) / 2
  - Compute f(mid) and identify active constraints
  - If all active B_r < 0: lo = mid (slope negative, minimum is to the right)
  - If all active B_r > 0: hi = mid (slope positive, minimum is to the left)
  - Otherwise (mixed or zero): we're at the minimum

This gives O(n) per bisection step (to evaluate f and find active constraints), O(n * 30) per coordinate, O(n² * 30) per sweep. For n=128, that's about 128 * 128 * 30 ≈ 500K per sweep. Very fast!
