def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    n = len(h)
    out = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            out[r] = float(np.dot(h[d:d+m], 1.0 - h[:m]))
        else:
            s = -d
            m = n - s
            out[r] = float(np.dot(h[:m], 1.0 - h[s:s+m]))
    return out

def _grad_rows(h, idx_list):
    n = len(h)
    k = len(idx_list)
    G = np.zeros((k, n))
    for i, r in enumerate(idx_list):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[i, :m] += -h[d:d+m]
            G[i, d:d+m] += 1.0 - h[:m]
        else:
            s = -d
            m = n - s
            G[i, :m] += 1.0 - h[s:s+m]
            G[i, s:s+m] += -h[:m]
    return G

    return x[state[0]]

    h = x[:n]
    T = x[n]
    sc = _raw_scores(h)
    return T - sc[active]

def _slsqp_ineq_jac(x, state):
    n, active, target = state
    h = x[:n]
    G = _grad_rows(h, active)
    jac = np.zeros((len(active), n + 1))
    jac[:, :n] = -G
    jac[:, n] = 1.0
    return jac

def _slsqp_eq_fun(x, state):
    n, active, target = state
    return float(np.sum(x[:n]) - target)

    j = np.zeros((1, n + 1))

def _area_resample(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    old_edges = np.arange(n + 1) * dx_old
    new_edges = np.arange(n_new + 1) * dx_new
    for j in range(n_new):
        js, je = new_edges[j], new_edges[j + 1]
        i_lo = max(0, int(js / dx_old))
        i_hi = min(n - 1, int((je - 1e-12) / dx_old))
        mass = 0.0
        for i in range(i_lo, i_hi + 1):

            if ov > 0:
                mass += h[i] * ov
        h_new[j] = mass / dx_new
    return _project(h_new, n_new / 2.0)
