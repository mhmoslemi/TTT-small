from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_q(h, d):
    """Gradient of correlation at lag d with respect to h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d >= 0:
        # C_d = dx * sum_{i=0}^{n-d-1} h[i+d] * (1 - h[i])
        # d < n required
        if d < n:
            # h[j] as h[i+d]: j = i+d, i = j-d, valid d <= j <= n-1
            # contribution: dx * (1 - h[j-d])
            for j in range(d, n):
                g[j] += dx * (1.0 - h[j - d])
            # h[j] as h[i]: j = i, valid 0 <= j <= n-d-1
            # contribution: -dx * h[j+d]
            for j in range(0, n - d):
                g[j] -= dx * h[j + d]
    else:
        k = -d
        if k < n:
            # C_d = dx * sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
            # h[j] as h[i]: j = i, valid 0 <= j <= n-k-1
            # contribution: dx * (1 - h[j+k])
            for j in range(0, n - k):

            # h[j] as h[i+k]: j = i+k, i = j-k, valid k <= j <= n-1

    target = n // 2
    idx = np.argsort(w)  # ascending
    x = np.zeros(n)
    x[idx[:target]] = 1.0
    return x

def _fw_step(h, eta, max_active):
    """One Frank-Wolfe step. Returns new h or None if no improvement."""
    n = len(h)
    target = n / 2.0
    c = _corr_vec(h)
    max_c = float(np.max(c))
    
    # Select active constraints
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:min(max_active, len(c))]
    
    # Softmax weights
    vals = c[active] - max_c  # for numerical stability
    weights = np.exp(vals / eta)
    weights /= weights.sum()
    
    # Compute subgradient
    g = np.zeros(n)
    for i, q in enumerate(active):
        d = q - (n - 1)
        g += weights[i] * _grad_q(h, d)
    
    # LP vertex
    s = _lp_vertex(g, n)
    
    # Line search
    best_alpha = 0.0
    best_val = max_c
    # Coarse grid
    alphas = np.linspace(0, 1, 21)
    for a in alphas:
        h_t = (1.0 - a) * h + a * s
        c_t = _exact_c5(h_t)
        if c_t < best_val:
            best_val = c_t
            best_alpha = a
    
    if best_alpha <= 0:
        return None, best_val
    
    # Refine around best_alpha
    if best_alpha > 0 and best_alpha < 1:
        lo = max(0.0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        fine_alphas = np.linspace(lo, hi, 11)
        for a in fine_alphas:
            h_t = (1.0 - a) * h + a * s
            c_t = _exact_c5(h_t)
            if c_t < best_val:
                best_val = c_t
                best_alpha = a
    
    if best_alpha > 0:
        h_new = (1.0 - best_alpha) * h + best_alpha * s
        return h_new, best_val
    return None, best_val

def _grad_step_nondiff(h, step_size, n_top=5):
    """Projected subgradient step using top-n correlations."""
    n = len(h)
    target = n / 2.0
    c = _corr_vec(h)
    sorted_idx = np.argsort(c)[::-1]
    top = sorted_idx[:n_top]
    
    # Average gradient of top constraints
    g = np.zeros(n)
    for q in top:
        d = q - (n - 1)
        g += _grad_q(h, d)
    g /= n_top
    
    # Remove mean to preserve sum
    g -= g.mean()
    
    # Line search
    best_alpha = 0.0
    best_val = float(np.max(c))
    for a in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2, step_size * 0.75]:
        h_t = _project(h - a * g, target)
        c_t = _exact_c5(h_t)
        if c_t < best_val:
            best_val = c_t
            best_alpha = a
    
    if best_alpha > 0:
        return _project(h - best_alpha * g, target), best_val
    return h, best_val

def _pair_swap(h, i, j, n_grid=15):
    """Try all values of z for h[i]=z, h[j]=p-z where p=h[i]+h[j]."""
    n = len(h)
    target = n / 2.0
    p = h[i] + h[j]
    lo = max(0.0, p - 1.0)

        return h, _exact_c5(h)
    
    best_z = h[i]
    best_val = _exact_c5(h)
    
    zs = np.linspace(lo, hi, n_grid)
    for z in zs:
        h_t = h.copy()
        h_t[i] = z
        h_t[j] = p - z
        c_t = _exact_c5(h_t)
        if c_t < best_val:
            best_val = c_t
            best_z = z
    
    if best_z != h[i]:
        h_new = h.copy()
        h_new[i] = best_z
        h_new[j] = p - best_z
        return h_new, best_val
    return h, best_val

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_all(h):
    """Compute gradients of all correlation values with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        d = q - (n - 1)
        if d >= 0:
            if d < n:
                grads[q, d:] += dx * (1.0 - h[:n - d])
                grads[q, :n - d] -= dx * h[d:]
        else:
            k = -d
            if k < n:
                grads[q, :n - k] += dx * (1.0 - h[k:])
                grads[q, k:] -= dx * h[:n - k]
    return grads

def _smooth_obj_and_grad(h, tau):
    """Log-sum-exp smoothed objective and gradient."""
    c = _corr_scaled(h)
    m = np.max(c)
    shifted = tau * (c - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    logsum = m + np.log(np.sum(exps)) / tau
    weights = exps / np.sum(exps)
    
    # Gradient: weighted average of all correlation gradients
    # Remove mean to preserve sum constraint in tangent space
    g = np.zeros(len(h))
    n = len(h)
    dx = 2.0 / n
    for q in range(2 * n - 1):
        if weights[q] < 1e-14:
            continue
        d = q - (n - 1)
        if d >= 0:
            if d < n:
                g[d:] += weights[q] * dx * (1.0 - h[:n - d])
                g[:n - d] -= weights[q] * dx * h[d:]
        else:
            k = -d
            if k < n:
                g[:n - k] += weights[q] * dx * (1.0 - h[k:])
                g[k:] -= weights[q] * dx * h[:n - k]
    g -= g.mean()
    return logsum, g

def _bb_step(h, g, n):
    """
    Projected gradient with Barzilai-Borwein step size.
    Returns new h, new c5, success flag.
    """
    target = n / 2.0
    c5_cur = _exact_c5(h)
    
    # Try several step sizes
    best_h = h
    best_val = c5_cur
    
    for lr in [0.05, 0.02, 0.01, 0.005, 0.1, 0.2, 0.03]:
        h_new = _project(h - lr * g, target)
        c_new = _exact_c5(h_new)
        if c_new < best_val - 1e-14:
            best_val = c_new
            best_h = h_new
    
    return best_h, best_val

def _pdg_loop(h, deadline, tau_start=2.0, tau_end=50.0, max_iter=500):
    """Projected gradient with log-sum-exp smoothing and tau annealing."""
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Anneal tau
        frac = min(1.0, it / (max_iter * 0.8))
        tau = tau_start + (tau_end - tau_start) * frac
        if it > max_iter * 0.9:
            tau = tau_end
        
        obj, g = _smooth_obj_and_grad(h_cur, tau)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        
        h_new, c_new = _bb_step(h_cur, g, n)
        
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            best_h = h_new.copy()
            h_cur = h_new
        elif c_new < _exact_c5(h_cur) - 1e-14:
            h_cur = h_new
        else:
            # Try smaller step
            h_tiny = _project(h_cur - 0.001 * g, n / 2.0)
            c_tiny = _exact_c5(h_tiny)
            if c_tiny < best_c5 - 1e-13:
                best_c5 = c_tiny
                best_h = h_tiny.copy()
                h_cur = h_tiny
            else:
                break
    
    return best_h, best_c5
