def _grad_rows_z(h, idxs):
 n=len(h); dx=2/n; base=n-1; K=np.zeros((len(idxs), n))
 for r, idx in enumerate(idxs):
    s=idx-base

       if len(i):
          K[r, i] += (1.0 - h[i+s])
          K[r, i+s] -= h[i]
    else:

       if len(i):
          K[r, i] += (1.0 - h[j])
          K[r, j] -= h[i]
 return K * dx

import time
import numpy as np
from scipy.optimize import minimize


def _force_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    for _ in range(80):
        err = float(target - np.sum(h))
        if abs(err) < 1e-15:
            break
        if err > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            break
        step = err / float(idx.size)
        if err > 0.0:
            slack = float(np.min(1.0 - h[idx]))
            step = min(step, slack * 0.5)
        else:
            slack = float(np.min(h[idx]))
            step = min(step, slack * 0.5)
        if step <= 0.0:
            break
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
    err = float(target - np.sum(h))
    if abs(err) > 1e-15:
        if err > 0.0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                i = int(idx[np.argmax(1.0 - h[idx])])
                add = min(err, 1.0 - h[i])
                if add > 0.0:
                    h[i] += add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                i = int(idx[np.argmax(h)])
                sub = min(-err, h[i])
                if sub > 0.0:
                    h[i] -= sub
        h = np.clip(h, 0.0, 1.0)
    return h


def _project(v, target):
    v = np.asarray(v, dtype=float)
    n = v.size
    if n == 0:
        return v
    if target <= 0.0:
        return _force_sum(np.zeros(n), target)
    if target >= float(n):
        return _force_sum(np.ones(n), target)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    h = np.clip(v - mu, 0.0, 1.0)
    return _force_sum(h, target)


def _fallback_h(n=64, seed=0):
    rng = np.random.default_rng(seed)
    v = 0.5 + 0.4 * rng.normal(size=n)

def _zvec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def _exact_c5(h):
    return float(np.max(_zvec(h)))


def _resample(h, n_new):
    n = len(h)
    n_new = int(n_new)
    if n_new <= 0:
        return _project(np.zeros(1), 0.5)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, float(n_new) / 2.0)


def _bounds_h(n):
    return [(0.0, 1.0)] * n


def _smooth_val(h, tau):
    if tau <= 1e-12:
        tau = 1e-12
    z = _zvec(h)
    m = float(np.max(z))
    a = (z - m) / tau
    return float(m + tau * np.log(np.sum(np.exp(a))))


def _smooth_grad(h, tau):
    if tau <= 1e-12:
        tau = 1e-12
    z = _zvec(h)
    m = float(np.max(z))
    a = (z - m) / tau
    p = np.exp(a)
    p = p / float(np.sum(p))
    n = len(h)
    dx = 2.0 / float(n)
    base = n - 1
    g = np.zeros(n)
    for q in range(len(p)):
        w = float(p[q])
        if w <= 1e-18:
            continue
        t = q - base
        ww = w * dx
        if t >= 0:
            i_len = n - t
            if i_len > 0:
                g[:i_len] -= ww * h[t:t + i_len]
                g[t:t + i_len] += ww * (1.0 - h[:i_len])
        else:
            e = -t
            i_len = n - e
            if i_len > 0:
                g[:i_len] += ww * (1.0 - h[e:e + i_len])
                g[e:e + i_len] -= ww * h[:i_len]
    return g


def _smooth_obj(x, tau):
    return _smooth_val(x, tau)


def _smooth_jac(x, tau):
    return _smooth_grad(x, tau)


def _eq_sum(x, n):
    return float(np.sum(x[:n]) - 0.5 * float(n))

            _smooth_obj,

def _top_idx(z, k):
    m = len(z)
    k = int(max(1, min(k, m)))
    if k >= m:

def _grad_rows_z(h, idxs):
    n = len(h)
    base = n - 1
    dx = 2.0 / float(n)
    idxs = np.asarray(idxs, dtype=int)
    K = np.zeros((len(idxs), n))
    for r in range(len(idxs)):
        t = int(idxs[r]) - base
        row = K[r]
        if t >= 0:
            i_len = n - t
            if i_len > 0:
                row[:i_len] -= h[t:t + i_len]
                row[t:t + i_len] += 1.0 - h[:i_len]
        else:
            e = -t
            i_len = n - e
            if i_len > 0:
                row[:i_len] += 1.0 - h[e:e + i_len]
                row[e:e + i_len] -= h[:i_len]
        K[r] = row * dx
    return K

            c2 = _exact_c5(h2)
            if c2 < best_c - 1e-15:
                best_h = h2.copy()
                best_c = c2
            if c2 < c - 1e-14:
                h = h2
                c = c2
                improved = True
                break
        if improved:
            continue
        if time.time() < deadline - 3:
            h2, c2 = _subgrad_loop(h, min(deadline, time.time() + 4), k=3, lr=1.0, max_iter=10000, rng=None)
            if c2 < c - 1e-14:
                h = h2
                c = c2
                if c2 < best_c - 1e-15:
                    best_h = h.copy()
                    best_c = c2
                continue
        break
    return best_h, best_c


def _optimize_smooth_seed(h, deadline):
    if time.time() >= deadline - 1.0:
        h = h.copy()
        return h, _exact_c5(h)
    h = h.copy()
    cur_h = h.copy()
    cur_c = _exact_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    taus = (0.22, 0.11, 0.055, 0.0275, 0.014, 0.007)
    n = len(cur_h)
    if n <= 80:
        iters = 90
    elif n <= 128:
        iters = 65
    elif n <= 192:
        iters = 45
    else:
        iters = 30
    for tau in taus:
        if time.time() >= deadline - 2.0:
            break
        h2 = _smooth_slsqp(cur_h, tau, maxiter=iters)
        c2 = _exact_c5(h2)
        if c2 < best_c - 1e-15:
            best_h = h2.copy()
            best_c = c2
        if c2 < cur_c + 0.005:
            cur_h = h2
            cur_c = c2
        if time.time() < deadline - 2.0:
            rem = max(0.0, deadline - time.time())
            pg_dead = min(deadline, time.time() + max(2.0, min(6.0, rem * 0.15)))
            h3, c3 = _smooth_pg(cur_h, tau, pg_dead, lr=2.0, max_iter=250)
            if c3 < best_c - 1e-15:
                best_h = h3.copy()
                best_c = c3
            if c3 < cur_c + 0.003:
                cur_h = h3
                cur_c = c3
    if time.time() < deadline - 2.0:
        rem = max(0.0, deadline - time.time())
        sg_dead = min(deadline, time.time() + max(2.0, min(8.0, rem * 0.30)))
        h4, c4 = _subgrad_loop(cur_h, sg_dead, k=3, lr=1.0, max_iter=20000, rng=None)
        if c4 < best_c - 1e-15:
            best_h = h4.copy()
            best_c = c4
    return best_h, best_c


def _seed_add(seeds, h, target):
    h = np.asarray(h, dtype=float)
    h = _project(h, target)
    if not np.all(np.isfinite(h)):
        return seeds
    for sh in seeds:
        if len(sh) == len(h) and float(np.mean(np.abs(h - sh))) < 1e-4:
            return seeds
    if len(seeds) < 30:
        seeds.append(h)
    return seeds


def _make_seeds(base_h, n, rng, count):
    n = int(n)
    target = 0.5 * float(n)
    h0 = _resample(base_h, n)
    seeds = []
    seeds = _seed_add(seeds, h0, target)
    seeds = _seed_add(seeds, 1.0 - h0, target)
    seeds = _seed_add(seeds, h0[::-1], target)
    seeds = _seed_add(seeds, 0.5 * (h0 + h0[::-1]), target)
    seeds = _seed_add(seeds, 0.5 * (h0 + 1.0 - h0[::-1]), target)
    rolls = (3, 7, 13, 21)
    for rr in rolls:
        seeds = _seed_add(seeds, np.roll(h0, rr), target)
    scales = (0.04, 0.10, 0.20, 0.35)
    for sc in scales:
        seeds = _seed_add(seeds, h0 + sc * rng.normal(size=n), target)
    seeds = _seed_add(seeds, 0.5 * h0 + 0.5 * rng.normal(size=n), target)
    if n % 2 == 0:
        hbin = np.zeros(n)
        hbin[np.argsort(h0)[-n // 2:]] = 1.0
        seeds = _seed_add(seeds, hbin, target)
    i = np.arange(n, dtype=float)
    freqs = (0.5, 1.0, 1.7, 2.6, 3.9)
    for ff in freqs:
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        v = 0.5 + 0.45 * np.cos(2.0 * np.pi * ff * i / float(n) + ph)
        seeds = _seed_add(seeds, v, target)
    if len(seeds) > count:
        out = []
        for s in seeds[:count]:
            out.append(s)
        return out
    return seeds


def _seed_count(n):
    if n <= 80:
        return 7
    if n <= 128:
        return 5
    if n <= 176:
        return 3
    return 2


def _append_unique(arr, val):
    val = int(val)
    for v in arr:
        if v == val:
            return
    arr.append(val)


def _better(c, n, bc, bn):
    if c < bc - 1e-12:
        return True
    if abs(c - bc) <= 1e-9 and n < bn:
        return True
    return False


def _add_candidate(pool, h, c, cap=24):
    if h is None or c is None:
        return pool
    if not np.all(np.isfinite(h)) or not np.isfinite(c):
        return pool
    h = _project(h, 0.5 * float(len(h)))
    c = _exact_c5(h)
    n = len(h)
    for item in pool:
        ph, pc, pn = item
        if pn == n and abs(pc - c) <= 1e-8 and float(np.mean(np.abs(h - ph))) < 1e-4:
            return pool
    if len(pool) >= cap:
        worst = 0
        for i in range(1, len(pool)):
            if pool[i][1] > pool[worst][1]:
                worst = i
        del pool[worst]
    pool.append((h.copy(), float(c), n))
    return pool


def _sort_pool(pool):
    for i in range(len(pool)):
        for j in range(i + 1, len(pool)):
            ci, ni = pool[i][1], pool[i][2]
            cj, nj = pool[j][1], pool[j][2]
            if cj < ci - 1e-12 or (abs(cj - ci) <= 1e-12 and nj < ni):
                pool[i], pool[j] = pool[j], pool[i]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(1.0, float(budget_s))
    reserve = min(8.0, max(1.0, 0.01 * total))
    deadline = t0 + total - reserve
    if deadline <= t0:
        deadline = t0 + 0.1
    d_smooth = t0 + total * 0.50
    if d_smooth >= deadline:
        d_smooth = 0.5 * (t0 + deadline)
    d_active = t0 + total * 0.80
    if d_active >= deadline:
        d_active = max(d_smooth + 1.0, 0.5 * (d_smooth + deadline))
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = _fallback_h(64, seed)
    if h0.size == 0 or not np.all(np.isfinite(h0)):
        h0 = _fallback_h(64, seed)
    h0 = _project(h0, 0.5 * float(h0.size))

    best_h = h0.copy()
    best_c = _exact_c5(best_h)
    best_n = len(best_h)
    pool = []
    pool = _add_candidate(pool, best_h, best_c, cap=24)

    base_n = len(h0)
    n_list = []
    _append_unique(n_list, base_n)
    pos_deltas = (32, 64, 96, 128, 160, 192, 224)
    for dd in pos_deltas:
        nn = base_n + dd
        if 16 <= nn <= 256 and nn % 2 == 0:
            _append_unique(n_list, nn)
    neg_deltas = (16, 32, 48, 64)
    for dd in neg_deltas:
        nn = base_n - dd
        if 16 <= nn <= 256 and nn % 2 == 0:
            _append_unique(n_list, nn)

    for nn in n_list:
        if time.time() >= d_smooth - 5.0:
            break
        nn = int(nn)
        if nn % 2 != 0 or nn < 16 or nn > 256:
            continue
        cnt = _seed_count(nn)
        seeds = _make_seeds(best_h, nn, rng, cnt)
        if not seeds:
            continue
        seed_c = []
        for s in seeds:
            seed_c.append(_exact_c5(s))
        order = list(range(len(seeds)))
        for i in range(len(order)):
            for j in range(i + 1, len(order)):
                if seed_c[order[j]] < seed_c[order[i]]:
                    order[i], order[j] = order[j], order[i]
        sorted_seeds = []
        for k in order:
            sorted_seeds.append(seeds[k])
        seeds = sorted_seeds
        for si in range(len(seeds)):
            if time.time() >= d_smooth - 2.0:
                break
            rem = max(0.0, d_smooth - time.time())
            left = float(len(seeds) - si)
            per = rem / max(1.0, left + 1.0)
            per = min(35.0, max(4.0, per))
            subd = time.time() + per
            h2, c2 = _optimize_smooth_seed(seeds[si], subd)
            pool = _add_candidate(pool, h2, c2, cap=24)
            if _better(c2, len(h2), best_c, best_n):
                best_h = h2.copy()
                best_c = c2
                best_n = len(h2)

    active_pool = list(pool)
    _sort_pool(active_pool)
    for item in active_pool:
        if time.time() >= d_active - 5.0:
            break
        hx, cx, nx = item
        if cx > best_c + 0.004:
            break
        if cx - best_c < 1e-9:
            alloc = 30.0
        elif cx - best_c < 0.001:
            alloc = 18.0
        else:
            alloc = 9.0
        subd = min(d_active, time.time() + alloc)
        h2, c2 = _active_polish(hx, subd)
        pool = _add_candidate(pool, h2, c2, cap=24)
        if _better(c2, len(h2), best_c, best_n):
            best_h = h2.copy()
            best_c = c2
            best_n = len(h2)

    loops = 0
    while time.time() < d_active - 5.0 and loops < 12:
        h2, c2 = _active_polish(best_h, min(d_active, time.time() + 20.0))
        if _better(c2, len(h2), best_c, best_n):
            best_h = h2.copy()
            best_c = c2
            best_n = len(h2)
            loops += 1
        else:
            break

    d_final = max(d_active + 1.0, min(deadline - 1.0, t0 + total - max(1.0, 0.005 * total)))
    final_cfgs = (
        (3, 1.0, False),
        (5, 0.7, True),
        (2, 1.4, False),
        (8, 0.4, True),
        (1, 1.8, False),
    )
    for kk, lrv, randomize in final_cfgs:
        if time.time() >= d_final - 4.0:
            break
        rem = max(0.0, d_final - time.time())
        alloc = min(25.0, max(3.0, rem * 0.20))
        rr = rng if randomize else None
        h2, c2 = _subgrad_loop(best_h, time.time() + alloc, k=kk, lr=lrv, max_iter=80000, rng=rr)
        if _better(c2, len(h2), best_c, best_n):
            best_h = h2.copy()
            best_c = c2
            best_n = len(h2)

    if time.time() < d_final - 4.0:
        rem = max(0.0, d_final - time.time())
        alloc = min(15.0, max(3.0, rem * 0.18))
        h2, c2 = _smooth_pg(best_h, 0.004, time.time() + alloc, lr=1.0, max_iter=3000)
        if _better(c2, len(h2), best_c, best_n):
            best_h = h2.copy()
            best_c = c2
            best_n = len(h2)

    if time.time() < deadline - 3.0:
        h2, c2 = _active_polish(best_h, deadline)
        if _better(c2, len(h2), best_c, best_n):
            best_h = h2.copy()
            best_c = c2
            best_n = len(h2)

    best_h = _project(best_h, 0.5 * float(len(best_h)))
    final_c = _exact_c5(best_h)
    if final_c > best_c + 1e-9:
        _sort_pool(pool)
        for item in pool:
            ph, pc, pn = item
            ph = _project(ph, 0.5 * float(pn))
            pc2 = _exact_c5(ph)
            if pc2 < final_c:
                best_h = ph
                final_c = pc2
                break
    best_h = _project(best_h, 0.5 * float(len(best_h)))
    final_c = _exact_c5(best_h)
    return best_h, float(final_c), int(len(best_h))
