def _smooth_max_grad(h, tau):

    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    P = len(scores)

    Z = np.sum(u)
    w = u / Z

        elif d > 0:

        else:
            g = -d
            m = n - g
            grad[:m] += wi * (1.0 - h[g:])
            grad[g:] -= wi * h[:m]

                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        oa = (lo + hi) / 2.0
        oc = _exact_c5(h + oa * p)
        if oc < vals[best_i]:
            best_i = best_i
            vals[best_i] = oc
            alphas[best_i] = oa
    return alphas[best_i], vals[best_i]


def _smooth_anneal(h, deadline, rng, tau_schedule=None, trust_init=0.25, max_it_per_tau=300):
    if tau_schedule is None:
        tau_schedule = [0.04, 0.02, 0.01, 0.005, 0.0025, 0.0012, 0.0007, 0.0004]
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    trust = trust_init
    for tau in tau_schedule:
        if time.time() > deadline:
            break
        stall = 0
        trust = max(trust, 0.05)
        for _it in range(max_it_per_tau):
            if time.time() > deadline:
                break
            _, grad = _smooth_max_grad(h, tau)
            p = -grad
            p -= p.mean()
            pn = float(np.max(np.abs(p)))
            if pn < 1e-14:
                break
            p /= pn
            amax = _max_safe(h, p)
            amax = min(amax, trust)
            if amax < 1e-12:
                stall += 1
                trust *= 0.7
                if stall > 20:
                    break
                continue
            alpha_opt, c5_opt = _line_search(h, p, amax)
            if c5_opt < cur_c5 - 1e-13:
                h = h + alpha_opt * p
                h = np.clip(h, 0.0, 1.0)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                trust = min(trust * 1.04, 0.45)
            else:
                stall += 1
                trust *= 0.72
                if stall > 30:
                    break
    return best_h, best_c5


def _lp_dir_topk(h, top_k=8):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    n_lags = len(scores)
    idx = np.argsort(scores)[-top_k:]
    G = np.zeros((len(idx), n))
    for ii, r in enumerate(idx):
        d = r - (n - 1)
        wi = dx
        if d == 0:
            G[ii] = wi * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            if m > 0:
                G[ii, d:] += wi * (1.0 - h[:m])
                G[ii, :m] -= wi * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[ii, :m] += wi * (1.0 - h[g:])
                G[ii, g:] -= wi * h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    h = np.clip(h, 0.0, 1.0)
    bounds = [(max(-0.3, -h[j]), min(0.3, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_optimize(h, deadline, top_k=8, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 25:
            break
        p = _lp_dir_topk(h, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            continue
        p = np.clip(p, -0.3, 0.3)
        p -= p.mean()
        pn = float(np.max(np.abs(p)))
        if pn < 1e-13:
            stall += 1
            continue
        p /= pn
        amax = min(_max_safe(h, p), 0.3)
        if amax < 1e-12:
            stall += 1
            continue
        alpha_opt, c5_opt = _line_search(h, p, amax)
        if c5_opt < cur_c5 - 1e-13:
            h = h + alpha_opt * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quasi_init(n, alpha, noise_scale=0.0, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0).astype(float)
    h = (h < 0.5).astype(float)
    if noise_scale > 0 and rng is not None:
        h += rng.normal(0, noise_scale, n)
    return _project(h, n / 2.0)


def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng_local = np.random.default_rng(42 + _pass * 1000)
        n_trials = max(50, n * 2)
        for _t in range(n_trials):
            if time.time() > deadline:
                break
            j = int(rng_local.integers(n))
            k = int(rng_local.integers(n))
            if j == k:
                continue
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8 or h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                continue
            for sign in (1, -1):
                if sign > 0:
                    am = min(1.0 - h[j], h[k])
                else:
                    am = min(h[j], 1.0 - h[k])
                if am < 1e-10:
                    continue
                lo_a, hi_a = 0.0, am
                for _ in range(7):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    h_copy = h.copy()
                    h_copy[j] += sign * m1
                    h_copy[k] -= sign * m1
                    f1 = _exact_c5(h_copy)
                    h_copy = h.copy()
                    h_copy[j] += sign * m2
                    h_copy[k] -= sign * m2
                    f2 = _exact_c5(h_copy)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                a_opt = (lo_a + hi_a) / 2.0
                h_try = h.copy()
                h_try[j] += sign * a_opt
                h_try[k] -= sign * a_opt
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def _combined_optimize(h, deadline, rng, phase_budget=0.6):
    """Combined smooth anneal + LP polish."""
    h = h.copy()
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # Part A: Smooth anneal
    dl_a = time.time() + phase_budget * (deadline - time.time())
    tau_sched = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    h_a, c5_a = _smooth_anneal(h, dl_a, rng, tau_schedule=tau_sched, trust_init=0.2, max_it_per_tau=250)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Part B: LP polish
    if time.time() < deadline:
        dl_b = deadline
        h_b, c5_b = _lp_optimize(best_h, dl_b, top_k=10, max_iters=400)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Part C: Very fine smooth anneal
    if time.time() < deadline - 5:
        dl_c = deadline
        tau_fine = [0.001, 0.0005, 0.0003, 0.00015]
        h_c, c5_c = _smooth_anneal(best_h, dl_c, rng, tau_schedule=tau_fine, trust_init=0.05, max_it_per_tau=200)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    hard_deadline = t0 + budget_s * 0.97

    # Phase 1: Optimize parent (n=128)
    if time.time() < hard_deadline:
        dl1 = t0 + budget_s * 0.18
        h1, c5_1 = _combined_optimize(h_start, dl1, rng, phase_budget=0.7)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Multi-resolution
    if time.time() < hard_deadline:
        dl2 = t0 + budget_s * 0.50
        resolutions = [96, 112, 128, 104, 88, 120]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.55]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl2:
                break
            if best_c5 < 0.3805:
                break
            n_starts = 2
            for s in range(n_starts):
                if time.time() > dl2:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    alpha = alphas[ri % len(alphas)]
                    h_init = _quasi_init(n_try, alpha, noise_scale=0.1, rng=rng)
                sub_dl = min(dl2, time.time() + budget_s * 0.04)
                if n_try <= 112:
                    h_try, c5_try = _combined_optimize(h_init, sub_dl, rng, phase_budget=0.6)
                else:
                    h_try, c5_try = _combined_optimize(h_init, sub_dl, rng, phase_budget=0.6)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 3: Diverse quasi-periodic starts
    if time.time() < hard_deadline:
        dl3 = t0 + budget_s * 0.65
        qp_alphas = [0.6180339887, 0.4142135624, 0.3819660113, 0.55, 0.44, 0.35]
        qp_ns = [96, 112, 96, 104, 88, 120]
        for i in range(len(qp_alphas)):
            if time.time() > dl3:
                break
            n_q = qp_ns[i]
            h_q = _quasi_init(n_q, qp_alphas[i], noise_scale=0.15, rng=rng)
            sub_dl = min(dl3, time.time() + budget_s * 0.03)
            h_q, c5_q = _combined_optimize(h_q, sub_dl, rng, phase_budget=0.65)
            if c5_q < best_c5:
                best_c5 = c5_q
                best_h = h_q.copy()
                best_n = n_q

    # Phase 4: Perturbation restarts
    if time.time() < hard_deadline:
        dl4 = t0 + budget_s * 0.82
        scale = 0.04
        n_pert = 60
        for attempt in range(n_pert):
            if time.time() > dl4:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.02)
            h_try, c5_try = _smooth_anneal(h_pert, sub_dl, rng,
                                           tau_schedule=[0.01, 0.005, 0.002, 0.001],
                                           trust_init=0.1, max_it_per_tau=150)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.005, scale * 0.8)
            else:
                scale = min(0.1, scale * 1.15)

    # Phase 5: LP on best
    if time.time() < hard_deadline:
        dl5 = t0 + budget_s * 0.90
        h_lp, c5_lp = _lp_optimize(best_h, dl5, top_k=12, max_iters=600)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 6: Pair polish
    if time.time() < hard_deadline:
        dl6 = t0 + budget_s * 0.95
        hp, c5p = _pair_polish(best_h, dl6, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 7: Final fine refinement
    if time.time() < t0 + budget_s * 0.99:
        h_fin, c5_fin = _smooth_anneal(best_h, t0 + budget_s * 0.99, rng,
                                       tau_schedule=[0.0005, 0.0002, 0.0001],
                                       trust_init=0.03, max_it_per_tau=200)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
