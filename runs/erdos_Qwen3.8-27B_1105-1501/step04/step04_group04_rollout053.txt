    from scipy.optimize import linprog

        G[r, low:high] += dx * omh[low + s:high + s]
        G[r, low + s:high + s] += dx * (-h[low:high])

    c[n] = 1.0
    A_ub = np.zeros((L, n + 1))

    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)

                      options={'maxiter': 20000, 'presolve': True})

    return None

        # Transfer: h[i] increases, h[j] decreases (to reduce gradient)
        d = rng.uniform(0.01, 0.06)
        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-6:
            no_improve += 1
            continue
        d = min(d, max_d)

        h[i] += d
        h[j] -= d
        c5_new = _exact_c5(h)
        if c5_new < cur_c5 - 1e-14:
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
        else:
            h[i] -= d
            h[j] += d
            no_improve += 1

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _fourier_init(n, m, phase=0.0):
    x = np.arange(n) / n
    h = 0.5 + 0.3 * np.cos(2 * np.pi * m * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return _project_box_sum(h, n / 2.0)

def _block_init(n, block_sizes):
    """Create a block-based initialization."""
    h = np.zeros(n)
    pos = 0
    fill = 1
    for bs in block_sizes:
        end = min(pos + bs, n)
        h[pos:end] = fill
        pos = end
        fill = 1 - fill
        if pos >= n:
            break
    return _project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        n0 = 128
        h0 = _fourier_init(n0, 12, 0.0)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP trust region (strong global optimizer)
    dl1 = t0 + budget_s * 0.28
    h1, c5_1 = _lp_loop(h, dl1, radius_start=0.10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Momentum descent with adaptive beta
    dl2 = t0 + budget_s * 0.42
    h2, c5_2 = _momentum_descent(best_h, dl2, beta_start=8.0, beta_end=6000.0,
                                  n_betas=10, momentum=0.82, base_step=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Active-set descent (focus on top K shifts)
    dl3 = t0 + budget_s * 0.52
    h3, c5_3 = _active_set_descent(best_h, dl3, K=12, base_step=0.07)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP polish
    dl4 = t0 + budget_s * 0.62
    h4, c5_4 = _lp_loop(best_h, dl4, radius_start=0.05)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Transfer phase at different K values
    dl5 = t0 + budget_s * 0.72
    for K_try in [10, 6, 20]:
        if time.time() > dl5:
            break
        t_sub = min(dl5, time.time() + budget_s * 0.03)
        h = _active_set_descent(best_h, t_sub, K=K_try, base_step=0.05)
        if h[1] < best_c5:
            best_c5 = h[1]
            best_h = h[0].copy()

    # Phase 6: Gradient-informed pair transfer
    dl6 = t0 + budget_s * 0.80
    h6, c5_6 = _pair_transfer(best_h, dl6, rng, max_iters=150000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Multi-resolution exploration
    dl7 = t0 + budget_s * 0.90
    n_candidates = sorted(set([best_n - 16, best_n - 8, best_n + 8, best_n + 16,
                                64, 80, 96, 112, 144, 160]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 192 and x != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))
    per_time = budget_s * 0.02
    for n_try in n_candidates:
        if time.time() > dl7:
            break
        h_r = _resample(best_h, n_try)
        dl_r = min(dl7, time.time() + per_time)
        h_r, c5_r = _lp_loop(h_r, dl_r, radius_start=0.06)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Try alternative initializations at current best_n
    if time.time() < t0 + budget_s * 0.94:
        dl8 = t0 + budget_s * 0.96
        alt_starts = []
        # Fourier with different frequencies
        for m in [8, 14, 20]:
            alt_starts.append(_fourier_init(best_n, m, rng.uniform(0, 2 * np.pi)))
        # Block patterns
        for vals in [[4, 4, 4, 4], [3, 5, 3, 5], [5, 3, 5, 3]]:
            reps = max(1, best_n // (2 * sum(vals)))
            pattern = []
            for _ in range(reps):
                pattern.extend(vals)
            pattern.extend(vals[:best_n - len(pattern)])
            alt_starts.append(_block_init(best_n, pattern))

        for start_h in alt_starts:
            if time.time() > dl8:
                break
            h_a, c5_a = _lp_loop(start_h, min(dl8, time.time() + 30), radius_start=0.12)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Phase 9: Final polish
    if time.time() < t0 + budget_s * 0.98:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.99, radius_start=0.03)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        h_p, c5_p = _pair_transfer(best_h, t0 + budget_s * 0.995, rng, max_iters=50000)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        # Revert to pre-projection state if projection hurt
        best_h = _project_box_sum(np.clip(best_h, 0, 1), best_n / 2.0)
        best_c5 = _exact_c5(best_h)

    # Safety
    if not np.all(np.isfinite(best_h)):
        best_h = _project_box_sum(h0, n0 / 2.0)
        best_c5 = _exact_c5(best_h)
        best_n = n0

    return best_h, best_c5, best_n
