    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:

        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:

        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)

def _exchange_minimize(h, i, j, n_grid=300):
    """Minimize C5 over h[i]+=tau, h[j]-=tau for tau in feasible range."""
    tau_min = max(-h[i], h[j] - 1.0, -1.0)
    tau_max = min(1.0 - h[i], h[j])
    if tau_max - tau_min < 1e-12:

    # Fine grid
    taus = np.linspace(tau_min, tau_max, n_grid)
    best_tau = 0.0
    best_c5 = _exact_c5(h)
    h_copy = h.copy()
    for t in taus:

        h_copy[j] = h[j] - t
        c = _exact_c5(h_copy)
        if c < best_c5:

            best_tau = t
    # Refine around best
    if best_tau != 0.0:
        r = (tau_max - tau_min) / n_grid
        taus2 = np.linspace(best_tau - 2*r, best_tau + 2*r, 100)

        for t in taus2:

Actually, let me think about the gradient more carefully. The smoothed objective is:
L(β, h) = dx * (1/β) * log(sum_d exp(β * M_d))

The gradient with respect to h[i] is:
∂L/∂h[i] = dx * sum_d w_d * ∂M_d/∂h[i]

where w_d = exp(β*M_d) / sum_k exp(β*M_k) are the softmax weights.

∂M_d/∂h[i] = (1 - h[i+d]) * 1[i in R(d)] - h[i-d] * 1[i-d in R(d)]

(When i+d == i-d, i.e., d=0, this is special: ∂M_0/∂h[i] = 1 - 2*h[i]... wait no, M_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]², so ∂M_0/∂h[i] = 1 - 2*h[i])

OK so the gradient g[i] = ∂L/∂h[i] tells us the "pressure" on each coordinate. A good exchange would be to increase h where g is low (or negative) and decrease h where g is high (or positive).

For a two-point exchange h[i]+=τ, h[j]-=τ, the first-order change in L is:
τ * (g[i] - g[j])

So we want g[i] < g[j] for the exchange to be first-order improving. The best pairs are those with large (g[j] - g[i]).

So the algorithm is:
1. Compute g = gradient of smoothed objective
2. Find pairs (i,j) with large g[j] - g[i] (i.e., move mass from high-g to low-g)
3. For the top K such pairs, do exact 1D minimization
4. Accept the best improving move
5. Repeat

This is essentially a coordinate descent / steepest descent with exact line search, which should converge well.

    """Compute gradient of smoothed objective (log-sum-exp)."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    s = scores - scores.max()
    exp_s = np.exp(beta * s)
    w = exp_s / exp_s.sum()

    # dM_d/dh[i] = (1-h[i+d])*1[i in R(d)] - h[i-d]*1[i-d in R(d)]
    g = np.zeros(n)
    for d in range(-(n-1), n):

            continue

        high = min(n, n - d)

            continue

        idx2 = np.arange(low + d, high + d)

def _exchange_line_fast(h, i, j, base_scores, n_grid=200):
    """Fast 1D minimization for exchange using precomputed scores."""
    n = len(h)
    dx = 2.0 / n
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return 0.0, float(base_scores.max() * dx)
    
    # Precompute coefficients for affected shifts
    # For each d, compute a_d and b_d
    a = np.zeros(2*n - 1)  # linear coeff for each shift
    b = np.zeros(2*n - 1)  # quadratic coeff
    
    hi_val = h[i]
    hj_val = h[j]
    
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        
        # Check if i is in R(d)
        i_in = (i >= low) and (i < high)
        # Check if j is in R(d)
        j_in = (j >= low) and (j < high)
        # Check if i-d is in R(d)
        id_in = (i - d >= low) and (i - d < high)
        # Check if j-d is in R(d)
        jd_in = (j - d >= low) and (j - d < high)
        
        a_d = 0.0
        b_d = 0.0
        
        if i_in:
            if i + d == j:
                # Coupling: h[i]*h[i+d] = h[i]*h[j]
                a_d += (hj_val - hi_val)  # from (h[i]+τ)(h[j]-τ) expansion
                b_d += -1.0
            else:
                a_d += (1.0 - h[i + d])
        if j_in:
            if j + d == i:
                a_d += (hj_val - hi_val)
                b_d += -1.0
            else:
                a_d += -(1.0 - h[j + d])
        if id_in:
            if i - d == j:
                pass  # already counted in coupling
            else:
                a_d += -h[i - d]
        if jd_in:
            if j - d == i:
                pass  # already counted in coupling
            else:
                a_d += h[j - d]
        
        if a_d != 0.0 or b_d != 0.0:
            a[r] = a_d
            b[r] = b_d
    
    # Now find min over tau of max_r (base_scores[r] + a[r]*tau + b[r]*tau^2) * dx
    hi_save = h[i]
    hj_save = h[j]
    best_tau = 0.0
    best_val = float(base_scores.max())

        val_d = base_scores + a * mid + b * mid**2
        f_val = val_d.max()

        grads = a[active] + 2*b[active]*mid

        g_max = grads.max()

            a_right = mid
        elif g_max < -1e-10:
            # f is decreasing at mid, minimum is to the right
            a_left = mid
        else:

            break

    h_save_j = h[j]
    h[i] = h_save_i + best_tau

    return best_tau, val
