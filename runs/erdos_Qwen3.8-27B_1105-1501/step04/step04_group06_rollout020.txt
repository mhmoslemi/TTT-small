def _grad_row(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        sp = -s
        g[:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g

    n = len(h)
    G = np.zeros((2*n-1, n))

    w = e / e.sum()
    G = _grad_matrix(x)
