            grad[g:] -= wk * h[:n-g]
    return val, grad

The gradient is: (1/p) * (∑ s_k^p)^(1/p - 1) * p * s_k^(p-1) * ds_k/dh = (∑ s_k^p)^(1/p - 1) * s_k^(p-1) * ds_k/dh

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += residual / cnt
    return np.clip(x, 0.0, 1.0)

def lse_val_grad(h, tau):
    """Log-sum-exp smooth max of scores with gradient."""
    n = len(h)
    dx = 2.0 / n
    s = np.correlate(h, 1.0 - h, mode='full') * dx
    m = len(s)
    max_s = float(np.max(s))
    exp_s = np.exp((s - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    
    grad = np.zeros(n)
    for k in range(m):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        wk = w[k] * dx
        if d >= 0:
            if d < n:
                grad[d:] += wk * (1.0 - h[:n - d])
                grad[:n - d] -= wk * h[d:]
        else:
            g = -d
            if g < n:
                grad[:n - g] += wk * (1.0 - h[g:])
                grad[g:] -= wk * h[:n - g]
    return val, grad

def build_corr_grad(h):
    """Build gradient matrix G where G[k, i] = d(score_k)/d(h_i)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

        elif p[j] < -1e-14:

            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)

def block_move_polish(h, deadline, rng):
    """Try moving a block of mass from one location to another."""
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    target = n / 2.0
    
    while time.time() < deadline:
        improved = False
        # Try different block sizes
        for bsize in [1, 2, 3, 4, 5, 8, 10]:
            if bsize >= n // 2:
                break
            # Random positions
            for _ in range(50):
                if time.time() > deadline:
                    break
                i = rng.integers(0, n - bsize)
                j = rng.integers(0, n - bsize)
                if abs(i - j) < bsize:
                    continue
                # Move mass from block [j, j+bsize) to block [i, i+bsize)
                # or vice versa
                for direction in [1, -1]:
                    h_try = best_h.copy()
                    amt = 0.1 * direction  # move 0.1 units
                    h_try[i:i+bsize] += amt
                    h_try[j:j+bsize] -= amt

                    c5_try = exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(12):
        if hi_a - lo_a < 1e-13:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        s = scores_of(h)
        G = build_corr_grad(h)
        p = lp_direction(h, s, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = line_search_c5(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def dual_ascent(h0, deadline, tau_init=0.1):
    """Alternate: update dual weights (softmax), update primal (projected gradient)."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    tau = tau_init
    step = 0.004
    count = 0
    while time.time() < deadline:
        s = scores_of(h)
        max_s = np.max(s)
        q = np.exp((s - max_s) / tau)
        q /= np.sum(q)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if q[k] < 1e-14:
                continue
            d = k - (n - 1)
            wk = q[k] * (2.0 / n)
            if d >= 0:
                if d < n:
                    grad[d:] += wk * (1.0 - h[:n - d])
                    grad[:n - d] -= wk * h[d:]
            else:
                g = -d
                if g < n:
                    grad[:n - g] += wk * (1.0 - h[g:])
                    grad[g:] -= wk * h[:n - g]
        h = project(h - step * grad, target)
        count += 1
        if count % 40 == 0:
            c5_now = exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            tau *= 0.995
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def block_polish(h, deadline, rng):
    """Try block moves to improve C5."""
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    target = n / 2.0
    stall = 0
    while time.time() < deadline and stall < 8:
        improved = False
        for bsize in [1, 2, 3, 5, 8, 12, 16]:
            if bsize >= n // 3:
                break
            for _ in range(80):
                if time.time() > deadline:
                    break
                i = int(rng.integers(0, n - bsize))
                j = int(rng.integers(0, n - bsize))
                if abs(i - j) < bsize:
                    continue
                for direction in [1.0, -1.0]:
                    for frac in [0.3, 0.6, 1.0]:
                        amt = 0.08 * frac * direction
                        h_try = best_h.copy()
                        h_try[i:i + bsize] += amt
                        h_try[j:j + bsize] -= amt
                        h_try = project(h_try, target)
                        c5_try = exact_c5(h_try)
                        if c5_try < best_c5 - 1e-12:
                            best_h = h_try
                            best_c5 = c5_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def pair_polish(h, deadline, max_passes=20):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    target = n / 2.0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        a = amax * frac
                        h_t = best_h.copy()
                        h_t[j] += sign * a
                        h_t[k] -= sign * a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_h = h_t.copy()
                            best_c5 = c5_t
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, _ = lse_val_grad(x, tau)
        return v

    def grad(x):
        _, g = lse_val_grad(x, tau)
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.955

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = project(h0, n0 / 2.0)
    best_c5 = exact_c5(best_h)
    best_n = n0

    # Phase 1: LP local search
    d1 = t0 + budget_s * 0.12
    h_lp, c5_lp = lp_loop(best_h, d1, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: Nesterov momentum (new method)
    d2 = t0 + budget_s * 0.32
    for tau in [0.6, 0.35, 0.18, 0.10, 0.05, 0.025]:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.035)
        h_nes, c5_nes = nesterov_opt(best_h, tau, sub_dl)
        if c5_nes < best_c5:
            best_c5 = c5_nes
            best_h = h_nes.copy()

    # Phase 3: LP refinement
    d3 = t0 + budget_s * 0.42
    h_lp2, c5_lp2 = lp_loop(best_h, d3, radius_init=0.08)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 4: Dual ascent (new method)
    d4 = t0 + budget_s * 0.52
    h_dual, c5_dual = dual_ascent(best_h, d4, tau_init=0.1)
    if c5_dual < best_c5:
        best_c5 = c5_dual
        best_h = h_dual.copy()

    # Phase 5: diversify with structured perturbations
    d5 = t0 + budget_s * 0.68
    scales = [0.03, 0.06, 0.1, 0.15, 0.22, 0.3]
    for ns in scales:
        if time.time() > d5:
            break
        # Gaussian perturbation
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        # Quick Nesterov
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_try, c5_try = nesterov_opt(h_pert, 0.12, sub_dl)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Quick LP
        if time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.012)
            h_try2, c5_try2 = lp_loop(h_pert, sub_dl2, radius_init=0.06)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase 6: Multi-resolution search (more aggressive)
    d6 = t0 + budget_s * 0.82
    tried_n = set()
    for dn in [-10, -20, -30, -40, -50, -60, -80, -100, 10, 20, 30, 40, -120, -150]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        if time.time() < d6:
            h_r, c5_r = nesterov_opt(h_r, 0.1, sub_dl)
        else:
            c5_r = exact_c5(h_r)
        if c5_r < best_c5:
            # Refine with LP at this n
            ref_dl = min(d6, time.time() + budget_s * 0.01)
            h_r2, c5_r2 = lp_loop(h_r, ref_dl, radius_init=0.06)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = n_try
            elif c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Block polish
    d7 = t0 + budget_s * 0.90
    h_bp, c5_bp = block_polish(best_h, d7, rng)
    if c5_bp < best_c5:
        best_c5 = c5_bp
        best_h = h_bp.copy()

    # Phase 8: Pair polish
    d8 = t0 + budget_s * 0.94
    h_pp, c5_pp = pair_polish(best_h, d8, max_passes=15)
    if c5_pp < best_c5:
        best_c5 = c5_pp
        best_h = h_pp.copy()

    # Phase 9: Final LP micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.015)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
