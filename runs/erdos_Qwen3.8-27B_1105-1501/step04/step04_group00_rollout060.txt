def _project(v, target):
    # Water-filling projection to {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Binary search for the water level
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any rounding error
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):

        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

    bounds.append((None, None))

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]

        pass
    return None, None

def _nesterov_smoothed(h, deadline, target_mut=0.005, max_iters=2000, rng=None):
    """Nesterov-accelerated gradient descent on smoothed max."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    y = h.copy()  # Nesterov momentum variable
    mu = target_mut
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _scores(y)
        # Softmax weights
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / mu)
        w = exp_s / np.sum(exp_s)
        G = _grad_matrix(y)
        grad = G.T @ w
        # Remove mean for sum constraint
        grad -= grad.mean()
        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = (t_k - 1.0) / (t_k + 1.0)
        step = 2.0 / (it + 2.0)
        h_new = _project(y - step * grad, target)
        y = h_new + beta * (h_new - h)
        y = _project(y, target)
        h_new = _project(h_new + beta * (h_new - h), target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        # Gradually increase mu (less smoothing)
        if it % 50 == 0:
            mu *= 1.1
    return best_h, best_c5

def _resample(h, n_new):

def _sturmian_init(n, slope):
    """Generate a Sturmian (mechanical) word pattern."""
    h = np.zeros(n)
    x = np.arange(n)
    # Mechanical word: h[i] = 1 if {i*slope} < slope (or some threshold)
    # For the Erdos problem, we want the set to have measure 1/2
    # So use threshold = 0.5
    frac = (x * slope) % 1.0
    h = (frac < 0.5).astype(float)
    # Project to exact sum
    h = _project(h, n / 2.0)
    return h

def _beatty_init(n, alpha):
    """Beatty sequence based initialization."""
    h = np.zeros(n)
    # Beatty sequence: floor(k*alpha) for k=1,...,n/2
    n_ones = n // 2
    seq = np.floor(np.arange(1, n_ones + 1) * alpha).astype(int)
    seq = seq[seq < n]
    if len(seq) > 0:
        h[seq] = 1.0
    # If we don't have exactly n/2 ones, adjust
    n_ones_actual = int(np.sum(h == 1.0))
    if n_ones_actual < n_ones:
        # Fill remaining from highest frequency positions
        fill = np.argsort(-np.abs((np.arange(n) * alpha) % 1.0 - 0.5))
        remaining = n_ones - n_ones_actual
        for idx in fill:
            if h[idx] == 0:
                h[idx] = 1.0
                remaining -= 1
                if remaining <= 0:
                    break
    elif n_ones_actual > n_ones:
        # Remove excess
        excess = n_ones_actual - n_ones
        remove = np.argsort(np.abs((np.arange(n) * alpha) % 1.0 - 0.5))
        for idx in remove:
            if h[idx] == 1.0:
                h[idx] = 0.0
                excess -= 1
                if excess <= 0:
                    break
    h = _project(h, n / 2.0)
    return h
