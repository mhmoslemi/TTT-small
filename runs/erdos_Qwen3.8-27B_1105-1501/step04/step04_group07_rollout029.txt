- j appears as n: n = j, valid when 0 ≤ j ≤ n-1-d
  Contribution: dx * (-h[j+d])

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m > 0:
            if d >= 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
            else:
                g = -d
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, max_alpha=1.0):
    n = len(h)
    for j in range(n):
        if p[j] > 1e-14:
            max_alpha = min(max_alpha, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            max_alpha = min(max_alpha, h[j] / (-p[j]))
    max_alpha = max(max_alpha, 1e-12)

    best_a = float(alphas[best_i])
    if best_i in (0, n_s - 1):
        return best_a, best_c5

    hi = float(alphas[best_i + 1])

    c2 = lo + gr * (hi - lo)

    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):

            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    oa = (lo + hi) / 2.0
    oc = _exact_c5(h + oa * p)
    if oc < best_c5:

def _logistic_descent(h, deadline, beta_start=30.0, beta_end=500.0, n_steps=500, momentum=0.7):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    vel = np.zeros(n)
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        # Softmax weights
        s_shifted = scores - scores.max()
        w = np.exp(beta * s_shifted)
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 50:
                break
            continue
        d = -grad / gn
        vel = momentum * vel + (1 - momentum) * d
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            stall += 1
            if stall > 50:

            continue

                stall = 0
                found = True
                break
        if not found:
            vel = 0.5 * vel
            stall += 1
            if stall > 60:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1

    stall = 0
    for step in range(n_steps):

        frac = step / max(1, n_steps - 1)

        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:

        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0

            else:

        found = False

def _lp_optimize(h, deadline, top_k=None, radius=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.5)
        else:
            stall += 1
            r *= 0.85
            if stall > 40 and r < 0.01:
                r = 0.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_perturb(h, rng, n):
    """Generate structured perturbations (sinusoidal patterns)."""
    x = np.arange(n) / n
    freq = int(rng.integers(1, 8))
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.08)
    pert = amp * np.sin(2 * np.pi * freq * x + phase)
    return h + pert

                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:

                        else:

                    h_try[j] += sign * a_opt

                    if c5_try < best_c5 - 1e-13:

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        m = n - abs(d)
        if m > 0:
            if d >= 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
            else:
                g = -d
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, max_alpha=1.0):
    for j in range(len(h)):
        if p[j] > 1e-14:
            max_alpha = min(max_alpha, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            max_alpha = min(max_alpha, h[j] / (-p[j]))
    max_alpha = max(max_alpha, 1e-12)
    n_s = 25
    alphas = np.linspace(0, max_alpha, n_s)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_s - 1):
        return best_a, best_c5
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(28):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    oa = (lo + hi) / 2.0
    oc = _exact_c5(h + oa * p)
    if oc < best_c5:
        return oa, oc
    return best_a, best_c5

def _smooth_descent(h, deadline, mode='logistic', p_start=10.0, p_end=8192.0,
                    n_steps=1000, momentum=0.3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    vel = np.zeros(n)
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        scores = _scores(h)
        if mode == 'logistic':
            beta = p_start * (p_end / p_start) ** frac
            s_shifted = scores - scores.max()
            w = np.exp(beta * s_shifted)
            sw = float(w.sum())
            if sw < 1e-300:
                w = np.zeros(n_lags)
                w[int(np.argmax(scores))] = 1.0
            else:
                w /= sw
        else:
            p = p_start * (p_end / p_start) ** frac
            s_max = float(scores.max())
            if s_max < 1e-15:
                break
            ratio = scores / s_max
            w = ratio ** p
            sw = float(w.sum())
            if sw < 1e-300:
                w = np.zeros(n_lags)
                w[int(np.argmax(scores))] = 1.0
            else:
                w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 60:
                break
            continue
        d = -grad / gn
        if momentum > 0:
            vel = momentum * vel + (1.0 - momentum) * d
            vn = np.linalg.norm(vel)
            if vn < 1e-15:
                d = d
            else:
                d = vel / vn
        found = False
        for scale in [0.12, 0.08, 0.05, 0.025, 0.012, 0.006]:
            alpha, new_c5 = _line_search(h, d * scale, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            vel *= 0.5
            stall += 1
            if stall > 80:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_optimize(h, deadline, top_k=None, radius=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        # Adaptive active set
        if top_k is None:
            active_thresh = scores.max() * 0.95
            idx = np.where(scores >= active_thresh)[0]
            if len(idx) < 3:
                idx = np.argsort(scores)[-3:]
        elif top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
        else:
            idx = np.arange(n_lags)
        sk = scores[idx]
        Gk = G[idx]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.04, 0.4)
        else:
            stall += 1
            r *= 0.82
            if stall > 30 and r < 0.015:
                r = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_perturb(h, rng, n):
    x = np.arange(n) / n
    k1 = int(rng.integers(1, 6))
    k2 = int(rng.integers(1, 6))
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a1 = rng.uniform(0.02, 0.07)
    a2 = rng.uniform(0.01, 0.05)
    pert = a1 * np.sin(2 * np.pi * k1 * x + p1) + a2 * np.sin(2 * np.pi * k2 * x + p2)
    return h + pert

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(10):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _triple_polish(h, deadline, max_passes=2):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if time.time() > deadline:
                    break
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for l in range(k + 1, n):
                    if h[l] < 1e-8 or h[l] > 1.0 - 1e-8:
                        continue
                    # Try moving from j to k and l simultaneously
                    a_max = min(1.0 - h[k], 1.0 - h[l], h[j] / 1.0)
                    if a_max < 1e-10:
                        continue
                    best_delta = 0.0
                    best_try = None
                    for frac_j in [0.5, 0.3, 0.7]:
                        aj = a_max
                        ak = aj * frac_j
                        al = aj * (1.0 - frac_j)
                        if ak + al > a_max:
                            continue
                        h_try = h.copy()
                        h_try[j] -= aj
                        h_try[k] += ak
                        h_try[l] += al
                        if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:
                            continue
                        h_try = np.clip(h_try, 0.0, 1.0)
                        c5_try = _exact_c5(h_try)

                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h = h_try.copy()
                            improved = True
                            best_delta = aj
                            break
                    if best_delta > 0:
                        break
        if not improved:
            break
    return best_h, best_c5
