def _coordinate_optimize(h, n, j):
    """Find optimal h[j] given all other coordinates fixed."""
    s_vals = np.arange(2*n-1) - (n-1)  # shifts for each correlation lag
    
    # For each lag r, compute slope a_r and intercept b_r of c[r] as function of h[j]
    # c[r] = sum_n h[n+s]*(1-h[n])
    # Terms involving h[j]:
    #   +h[j] if 0 <= j-s < n (from h[n+s] with n = j-s)
    #   -h[j]*h[j-s] if 0 <= j-s < n (from product term)
    #   -h[j+s]*h[j] if 0 <= j+s < n (from product term with n=j)
    
    a = np.zeros(2*n-1)  # slope
    b = np.zeros(2*n-1)  # intercept
    
    for r in range(2*n-1):
        s = s_vals[r]
        # Compute c[r] without h[j] terms
        # c[r] = sum_n h[n+s]*(1-h[n]) for valid n, excluding n where j is involved
        val = 0.0
        for nn in range(n):
            ns = nn + s
            if 0 <= ns < n:
                if nn == j:
                    # term is h[ns]*(1-h[j]) = h[ns] - h[ns]*h[j]
                    # slope: -h[ns], intercept: h[ns]
                    a[r] += -h[ns]
                    b[r] += h[ns]
                elif ns == j:
                    # term is h[j]*(1-h[nn]) = h[j] - h[nn]*h[j]
                    # slope: 1 - h[nn], intercept: 0
                    a[r] += (1 - h[nn])
                    b[r] += 0
                else:
                    val += h[ns] * (1 - h[nn])
            # Also need to handle the case where nn == j and ns == j (s=0)
        b[r] += val
    
    # Now minimize max_r (a[r]*t + b[r]) over t in [0,1]
    # This is a 1D convex problem
    # Use bisection on the subgradient
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        # Subgradient: for the max, it's the sum of a[r] for all r that achieve the max
        vals = a * mid + b
        mx = vals.max()
        active = vals >= mx - 1e-12
        sg = a[active].sum()
        if sg > 0:
            hi = mid  # function is increasing, minimum is to the left
        else:
            lo = mid  # function is decreasing or flat, minimum is to the right
    return (lo + hi) / 2

    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    vals = alpha * (c - mc)
    exp_v = np.exp(np.clip(vals, -500.0, 500.0))
    s = exp_v.sum()
    f = mc + np.log(s) / alpha
    v = x.sum() - target
    return f + 1e4 * v * v

    grad = J.T @ w
    v = x.sum() - target
