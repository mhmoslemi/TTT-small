```python
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        else:
            hi = mid
```

```python
def _fast_coord_descent(h, deadline, max_sweeps=500, shuffle_order=False, rng=None):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scores = _scores_array(h).copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        # Determine variable order
        if shuffle_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for j in order:
            if time.time() > deadline:
                break
            old_val = h[j]
            
            # Compute coefficients a[d] for each lag
            # score_d(w) = a[d]*w + b[d] where w = h[j]
            # a[d] = dx * sum of contributions
            a = np.zeros(2 * n - 1)
            for d_idx in range(2 * n - 1):
                shift = d_idx - (n - 1)
                coeff = 0.0
                jp = j + shift
                if 0 <= jp < n:
                    coeff += (1.0 - h[jp])
                jm = j - shift
                if 0 <= jm < n:
                    coeff += (-h[jm])
                a[d_idx] = coeff * dx
            
            b = scores - a * old_val
            
            # Minimize max(a*w + b) over w in [0,1] using ternary search
            lo_t, hi_t = 0.0, 1.0
            # First check endpoints
            f_lo = float(np.max(a * 0.0 + b))
            f_hi = float(np.max(a * 1.0 + b))
            if f_lo <= f_hi:
                # Minimum might be near 0
                hi_t = 0.5
            else:
                lo_t = 0.5
            
            for _ in range(20):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = float(np.max(a * m1 + b))
                f2 = float(np.max(a * m2 + b))
                if f1 <= f2:
                    hi_t = m2
                else:
                    lo_t = m1
            new_val = (lo_t + hi_t) / 2.0
            
            # Also check original value
            f_orig = float(np.max(a * old_val + b))
            f_new = float(np.max(a * new_val + b))
            
            if f_new < f_orig - 1e-14:
                delta = new_val - old_val
                h[j] = new_val
                scores += a * delta
                cur_c5 = float(np.max(scores))
                if cur_c5 < best_c5 - 1e-13:
                    best_c5 = cur_c5
                    best_h = h.copy()
                sweep_improved = True
        # End for j
        
        # Project to maintain sum
        h = _project(h, target)
        scores = _scores_array(h)
        cur_c5 = float(np.max(scores))
        if cur_c5 < best_c5 - 1e-10:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        elif sweep_improved:
            no_improve += 1
        else:
            no_improve += 2
        if no_improve > 30:
            break
    
    return best_h, best_c5
```

```python
def _compute_a_vector(h, j):
    n = len(h)
    dx = 2.0 / n
    shifts = np.arange(2*n-1) - (n-1)  # range from -(n-1) to (n-1)

    valid_p = (jp >= 0) & (jp < n)

    jm = j - shifts
    valid_m = (jm >= 0) & (jm < n)
    
    a = np.zeros(2*n-1)
    a[valid_p] += (1.0 - h[jp[valid_p]])
    a[valid_m] += (-h[jm[valid_m]])
    a *= dx
    return a
```

```python
        sweep_impr = False
```

```python
        for _p in range(n_pairs_per_sweep):
            if time.time() > deadline:
                break
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j == k:
                continue
            old_j, old_k = h[j], h[k]
            if old_j < 1e-10 and old_k < 1e-10:
                continue
            if old_j > 1 - 1e-10 and old_k > 1 - 1e-10:
                continue

            a_j = np.zeros(2 * n - 1)
            a_k = np.zeros(2 * n - 1)

            jm = j - shifts
            vm = (jm >= 0) & (jm < n)
            a_j[vp] += (1.0 - h[jp[vp]])
            a_j[vm] += (-h[jm[vm]])
            
            kp = k + shifts
            vp2 = (kp >= 0) & (kp < n)
            km = k - shifts
            vm2 = (km >= 0) & (km < n)
            a_k[vp2] += (1.0 - h[kp[vp2]])
            a_k[vm2] += (-h[km[vm2]])

            a_k *= dx
```

```python
def _weighted_grad_descent(h, deadline, n_steps=500, gamma_start=10, gamma_end=200, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        
        scores = _scores_array(h)
        max_s = scores.max()
        w = np.exp(gamma * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(2*n-1)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for j in range(n):
            jp = j + shifts
            vp = (jp >= 0) & (jp < n)
            jm = j - shifts
            vm = (jm >= 0) & (jm < n)
            a = np.zeros(2*n-1)
            a[vp] = (1.0 - h[jp[vp]])
            a[vm] += (-h[jm[vm]])
            a *= dx
            grad[j] = w @ a
        
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        # Step
        alpha = 0.05 / (1.0 + frac)
        for _ in range(20):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
    
    return best_h, best_c5
```

Actually, np.correlate(a, v, 'full') computes sum_k a[k] * v[k + ia] for all valid ia.

For a = h (length n), v = 1-h (length n):

The 'full' output has length 2n-1, with index 0 corresponding to ia = -(n-1), index n-1 to ia = 0, index 2n-2 to ia = n-1.

So result[r] where r = n-1+ia:
result[r] = sum_k h[k] * (1-h[k+ia]) for 0 ≤ k < n and 0 ≤ k+ia < n

For ia = d (shift):
result[n-1+d] = sum_k h[k] * (1-h[k+d]) for 0 ≤ k, k+d < n

Now, the gradient of result[n-1+d] w.r.t. h[j]:
- h[j] appears as h[k] when k = j: term is h[j]*(1-h[j+d]), requires j+d < n
  Derivative: (1-h[j+d])
- h[j] appears as (1-h[k+d]) when k+d = j, i.e., k = j-d: term is h[j-d]*(1-h[j])
  Derivative: -h[j-d]

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _precompute_shifts(n):
    return np.arange(2 * n - 1) - (n - 1)

def _min_max_lines(a, b, lo_t, hi_t, n_iter=20):
    for _ in range(n_iter):
        m1 = lo_t + (hi_t - lo_t) / 3.0
        m2 = hi_t - (hi_t - lo_t) / 3.0
        f1 = float(np.max(a * m1 + b))
        f2 = float(np.max(a * m2 + b))
        if f1 <= f2:
            hi_t = m2
        else:
            lo_t = m1
    return (lo_t + hi_t) / 2.0

def _coord_descent(h, deadline, max_sweeps=2000, shuffle=False, rng=None, n_iter=20):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scores = _scores_array(h).copy()
    shifts = _precompute_shifts(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if shuffle and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for j in order:
            if time.time() > deadline:
                break
            old_val = h[j]
            a = np.zeros(2 * n - 1)
            jp = j + shifts
            valid_p = (jp >= 0) & (jp < n)
            jm = j - shifts
            valid_m = (jm >= 0) & (jm < n)
            a[valid_p] += (1.0 - h[jp[valid_p]])
            a[valid_m] += (-h[jm[valid_m]])
            a *= dx

            b = scores - a * old_val
            new_val = _min_max_lines(a, b, 0.0, 1.0, n_iter=n_iter)

            f_new = float(np.max(a * new_val + b))
            f_old = float(np.max(a * old_val + b))

            if f_new < f_old - 1e-14:
                delta = new_val - old_val
                h[j] = new_val
                scores += a * delta

        h = _project(h, target)
        scores = _scores_array(h)
        cur_c5 = float(np.max(scores))
        if cur_c5 < best_c5 - 1e-11:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_h, best_c5

def _pair_coord_descent(h, deadline, n_pairs_per_sweep=384, rng=None, max_sweeps=300):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scores = _scores_array(h).copy()
    shifts = _precompute_shifts(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_impr = False

        for _p in range(n_pairs_per_sweep):
            if time.time() > deadline:
                break
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                continue
            old_j, old_k = h[j], h[k]
            if old_j < 1e-10 or old_k < 1e-10 or old_j > 1-1e-10 or old_k > 1-1e-10:
                continue

            a_j = np.zeros(2 * n - 1)
            a_k = np.zeros(2 * n - 1)
            jp = j + shifts
            vp = (jp >= 0) & (jp < n)
            jm = j - shifts
            vm = (jm >= 0) & (jm < n)
            a_j[vp] += (1.0 - h[jp[vp]])
            a_j[vm] += (-h[jm[vm]])
            kp = k + shifts
            vp2 = (kp >= 0) & (kp < n)
            km = k - shifts
            vm2 = (km >= 0) & (km < n)
            a_k[vp2] += (1.0 - h[kp[vp2]])
            a_k[vm2] += (-h[km[vm2]])
            a_j *= dx
            a_k *= dx

            c = a_j - a_k
            t_lo = max(0.0, -old_k)
            t_hi = min(1.0 - old_j, old_k)
            if t_hi - t_lo < 1e-12:
                continue

            t_opt = _min_max_lines(c, scores, t_lo, t_hi, n_iter=16)
            f_old = float(np.max(scores))
            f_new = float(np.max(c * t_opt + scores))

            if f_new < f_old - 1e-14:
                h[j] = old_j + t_opt
                h[k] = old_k - t_opt
                scores += c * t_opt
                c5_new = float(np.max(scores))
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h.copy()
                sweep_impr = True

        if not sweep_impr:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_h, best_c5

def _weighted_grad(h, deadline, n_steps=300, gamma_start=5.0, gamma_end=100.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    shifts = _precompute_shifts(n)

    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        scores = _scores_array(h)
        max_s = scores.max()
        w = np.exp(gamma * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw

        grad = np.zeros(n)
        for j in range(n):
            jp = j + shifts
            vp = (jp >= 0) & (jp < n)
            jm = j - shifts
            vm = (jm >= 0) & (jm < n)
            a = np.zeros(2 * n - 1)
            a[vp] = (1.0 - h[jp[vp]])
            a[vm] += (-h[jm[vm]])
            a *= dx
            grad[j] = w @ a

        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn

        alpha = 0.06 / (1.0 + 2.0 * frac)
        for _ in range(25):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            alpha *= 0.65
            if alpha < 1e-10:
                break

    return best_h, best_c5

def _binary_sa(h_start, n, deadline, rng, n_iters=30000):

    h = np.zeros(n)

    zeros = np.where(h == 0.0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5

            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5
```
