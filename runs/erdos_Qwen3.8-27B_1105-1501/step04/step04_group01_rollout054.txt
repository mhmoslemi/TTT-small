def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else:

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

    if np.any(neg):

    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0

def _active_set_lp(h, scores, radius, epsilon):
    n = len(h)
    max_s = float(np.max(scores))
    active_idx = np.where(scores > max_s - epsilon)[0]
    n_active = len(active_idx)
    if n_active == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for idx, r in enumerate(active_idx):
        g = _compute_grad_row(h, r, n)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -scores[r]
    A_eq = np.ones((1, n))
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _full_lp(h, scores, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        g = _compute_grad_row(h, r, n)
        A_ub[r, :n] = g

                      bounds=bounds, method='highs')

            return res.x[:n]
    except Exception:
        pass
    return None
