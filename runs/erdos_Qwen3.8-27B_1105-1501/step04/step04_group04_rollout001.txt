    return G

def _lp_minimax(h, active_idx, radius):
    """Solve LP: min t s.t. G[active]@delta - t <= -F[active], sum(delta)=0, bounds."""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = _grad_matrix(h)
    
    ka = len(active_idx)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((ka, n + 1))
    A_ub[:, :n] = G[active_idx]
    A_ub[:, n] = -1.0
    b_ub = -F[active_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    lo = np.maximum(-radius, -h + 1e-10)
    hi = np.minimum(radius, 1.0 - h - 1e-10)
    # Ensure lo < hi
    bad = lo >= hi
    if bad.any():
        lo[bad] = (lo[bad] + hi[bad]) / 2
        hi[bad] = lo[bad] + 1e-10
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'maxiter': 20000, 'presolve': True})
    if res.success and res.x is not None:
        return res.x[:n]
    return None

def _lp_loop(h, deadline, top_k=40, radius_init=0.12):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline and stall < 200:
        F = np.correlate(cur_h, 1.0 - cur_h, mode='full') * dx
        # Select top k active lags
        top_idx = np.argsort(F)[::-1][:top_k]
        # Also include any lag within 1% of max
        thresh = F.max() * 0.99
        active = np.unique(np.concatenate([top_idx, np.where(F >= thresh)[0]]))
        
        delta = _lp_minimax(cur_h, active, radius)
        if delta is None:
            stall += 1
            radius *= 0.8
            if radius < 1e-8:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_trial = cur_c5
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]:
            trial = _proj(cur_h + alpha * delta, n / 2.0)
            tc = _c5(trial)
            if tc < best_trial - 1e-14:
                best_trial = tc
                best_alpha = alpha
        
        if best_alpha > 0:
            cur_h = _proj(cur_h + best_alpha * delta, n / 2.0)
            cur_c5 = _c5(cur_h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.3)
        else:
            stall += 1
            radius *= 0.85
    
    return best_h, best_c5

def _binary_snap(h, threshold=0.15):
    """Push values toward 0 or 1, then re-project."""
    n = len(h)
    h2 = h.copy()
    h2[h2 < threshold] = 0.0
    h2[h2 > 1.0 - threshold] = 1.0
    h2 = _proj(h2, n / 2.0)
    return h2

def _combinatorial_search(h, deadline, rng, max_iters=500000):
    """Try discrete moves: flip pairs, or adjust free coordinates."""
    n = len(h)
    cur_h = h.copy()
    cur_c5 = _c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    no_improve = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 10000:
            break
        
        # Choose move type
        r = rng.random()
        if r < 0.4:
            # Swap: pick one 1-ish and one 0-ish, swap their values
            ones = np.where(cur_h > 0.7)[0]
            zeros = np.where(cur_h < 0.3)[0]
            if len(ones) == 0 or len(zeros) == 0:
                no_improve += 1
                continue
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            c = _c5(cur_h)
            if c < cur_c5 - 1e-14:
                cur_c5 = c
                if c < best_c5:
                    best_c5 = c
                    best_h = cur_h.copy()
                no_improve = 0
            else:
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
                no_improve += 1
        elif r < 0.7:
            # Adjust a free coordinate
            free = np.where((cur_h > 0.15) & (cur_h < 0.85))[0]
            if len(free) == 0:
                no_improve += 1
                continue
            i = free[rng.integers(len(free))]
            old = cur_h[i]
            cur_h[i] = np.clip(cur_h[i] + rng.normal(0, 0.1), 0, 1)
            cur_h = _proj(cur_h, n / 2.0)
            c = _c5(cur_h)
            if c < cur_c5 - 1e-14:
                cur_c5 = c
                if c < best_c5:
                    best_c5 = c
                    best_h = cur_h.copy()
                no_improve = 0
            else:
                cur_h = _proj(np.maximum(0, np.minimum(1, cur_h)), n / 2.0)
                no_improve += 1
        else:
            # Pair adjustment (continuous)
            free = np.where((cur_h > 0.1) & (cur_h < 0.9))[0]
            if len(free) < 2:
                no_improve += 1
                continue
            idx = rng.choice(len(free), size=2, replace=False)
            i, j = free[idx[0]], free[idx[1]]
            d = rng.uniform(-0.08, 0.08)
            ni = cur_h[i] + d
            nj = cur_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                cur_h[i], cur_h[j] = ni, nj
                c = _c5(cur_h)
                if c < cur_c5 - 1e-14:
                    cur_c5 = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = cur_h.copy()
                    no_improve = 0
                else:
                    cur_h[i], cur_h[j] = ni - d, nj + d
                    no_improve += 1
            else:
                no_improve += 1
    
    return best_h, best_c5
