def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = trust
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    t_lo = 0.0
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:

    obj = np.max(vals, axis=1) * dx

    best_t = grid[best_i]
    best_val = float(obj[best_i])
    if best_i > 0 and best_i < n_grid - 1:

    else:

    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C) * dx)

def _lp_direction(h, scores, radius, K):
    n = len(h)
    nr = 2 * n - 1
    idx = np.argsort(scores)[-K:]
    active_scores = scores[idx]
    G_active = _grad_rows(h, idx)
    dx = 2.0 / n
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = dx * G_active
    A_ub[:, n] = -1.0
    b_ub = -active_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if float(np.max(np.abs(d))) > 1e-10:
                return d
    except Exception:
        pass
    return None

def _subgrad_direction(h, scores, tau):
    n = len(h)
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    nz = np.where(w > 1e-14)[0]
    if len(nz) == 0:
        return np.zeros(n)
    G_rows = _grad_rows(h, nz)
    grad = (G_rows.T @ (w[nz] * (2.0 / n)))
    grad -= np.mean(grad)
    return grad

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_gaussian(h, rng, scale):
    n = len(h)
    return _project(h + rng.normal(0, scale, n), n / 2.0)

def _perturb_fourier(h, rng, n_modes=4):
    n = len(h)
    d = np.zeros(n)
    for _ in range(n_modes):
        k = rng.integers(1, max(2, n // 3))
        amp = rng.uniform(0.3, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        d += amp * np.sin(2 * np.pi * k * np.arange(n) / n + phase)
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn < 1e-12:
        return h.copy()
    d = d / dn * 0.1
    return _project(h + d, n / 2.0)

def _perturb_bump(h, rng):
    n = len(h)
    center = rng.uniform(0, n)
    width = rng.uniform(2, max(3, n * 0.15))
    x = np.arange(n)
    bump = np.exp(-0.5 * ((x - center) / width) ** 2)
    amp = rng.choice([-1, 1]) * rng.uniform(0.05, 0.2)
    d = amp * bump
    d -= np.mean(d)
    return _project(h + d, n / 2.0)

def _coord_descent(h, deadline, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                s_ij = h[i] + h[j]
                t_lo = max(0.0, s_ij - 1.0)
                t_hi = min(1.0, s_ij)
                if t_hi - t_lo < 1e-12:
                    continue
                best_t = h[i]
                best_val = best_c5
                for t_try in np.linspace(t_lo, t_hi, 12):
                    if abs(t_try - h[i]) < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[i] = t_try
                    h_try[j] = s_ij - t_try
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_val:
                        best_val = c5_try
                        best_t = t_try
                if best_val < best_c5 - 1e-14:
                    h[i] = best_t
                    h[j] = s_ij - best_t
                    best_c5 = best_val
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _pair_refine_fast(h, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for _pass in range(60):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(n):
                if i == j:
                    continue
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                h_test = h + alpha * d
                h_test = _project(h_test, target)
                actual = _exact_c5(h_test)
                if actual < best_c5 - 1e-14:
                    h = h_test
                    best_c5 = actual
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Phase 1: Subgradient warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _subgrad_loop(h, d1, tau_list=(80, 200, 500), lr_init=0.015)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 2: Main LP optimization (40%)
    d2 = t0 + budget_s * 0.48
    h2, c5_2 = _lp_loop(best_h, d2, K=35, radius_init=0.18)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2
    
    # Phase 3: Subgradient + LP alternating (15%)
    d3 = t0 + budget_s * 0.63
    while time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h3, c5_3 = _subgrad_loop(best_h, sub_dl, tau_list=(100, 300, 800), lr_init=0.01)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
        if time.time() > d3:
            break
        sub_dl2 = min(d3, time.time() + budget_s * 0.04)
        h3b, c5_3b = _lp_loop(best_h, sub_dl2, K=40, radius_init=0.12)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b
    
    # Phase 4: Multi-resolution search (18%)
    d4 = t0 + budget_s * 0.81
    best_n = len(best_h)
    offsets = [-3, 3, -4, 4, -6, 6, -5, 5, -8, 8, -7, 7, -10, 10, -2, 2,
               -12, 12, -14, 14, -9, 9, -16, 16]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 220:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.016)
        h_r, c5_r = _lp_loop(h_r, sub_dl, K=30, radius_init=0.12)
        if c5_r < best_c5 - 1e-10:
            # Further optimize
            rem = d4 - time.time()
            if rem > 5:
                sub_dl2 = min(d4, time.time() + min(rem * 0.5, budget_s * 0.01))
                h_r, c5_r = _lp_loop(h_r, sub_dl2, K=40, radius_init=0.08)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    
    # Phase 5: Perturbation + re-optimize (10%)
    d5 = t0 + budget_s * 0.91
    scale = 0.04
    for attempt in range(200):
        if time.time() > d5:
            break
        if best_c5 < 0.3806:
            break
        method = attempt % 3
        if method == 0:
            h_pert = _perturb_gaussian(best_h, rng, scale)
        elif method == 1:
            h_pert = _perturb_fourier(best_h, rng)
        else:
            h_pert = _perturb_bump(best_h, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, K=30, radius_init=0.1)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.82)
        else:
            scale = min(0.15, scale * 1.1)
    
    # Phase 6: Pair refinement (5%)
    d6 = t0 + budget_s * 0.96
    h_p, c5_p = _pair_refine_fast(best_h, d6)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p
    
    # Phase 7: Final LP polish
    d7 = t0 + budget_s * 0.985
    h_f, c5_f = _lp_loop(best_h, d7, K=50, radius_init=0.05)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5):
        best_h = _project(h0, n0 / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
