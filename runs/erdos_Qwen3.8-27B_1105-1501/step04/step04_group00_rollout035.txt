    cur_c5 = best_c5
    trust = trust_start
    tau_hi = 0.05
    tau_lo = 0.0008
    log_ratio = np.log(tau_lo / tau_hi)
    taues = [tau_hi * np.exp(log_ratio * i / max(1, n_taus - 1)) for i in range(n_taus)]
    mom = np.zeros(n)
    for tau_idx, tau in enumerate(taues):
        if time.time() > deadline:
            break
        frac = tau_idx / max(1, n_taus - 1)
        drift = 0.0015 * (1.0 - frac) + 0.00005
        n_stall = 0
        for step in range(steps_per_tau):
            if time.time() > deadline:
                break
            if n_stall > 25:
                break
            F, grad = _smooth_grad(h, tau)
            p = -grad
            p -= p.mean()
            p = p + momentum * mom
            p -= p.mean()
            pn = float(np.max(np.abs(p)))
            if pn < 1e-14:
                n_stall += 1
                if n_stall > 10:
                    break
                continue
            p /= pn
            mom = 0.7 * mom + 0.3 * p
            alpha_max = _max_alpha(h, p)
            alpha_max = min(alpha_max, trust)
            if alpha_max < 1e-12:
                trust *= 0.7
                n_stall += 1
                continue
            a, c5_new = _line_search(h, p, alpha_max, n_scan=13)
            if a < 1e-12:
                n_stall += 1
                trust *= 0.8
                continue
            h_new = h + a * p
            h_new = _project(h_new, target)
            c5_proj = _exact_c5(h_new)
            if c5_proj < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_proj
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                n_stall = 0
                trust = min(trust * 1.04, 0.5)
            elif c5_proj < cur_c5 + drift:
                F_new, _ = _smooth_grad(h_new, tau)
                if F_new < F - 1e-12:
                    h = h_new
                    cur_c5 = c5_proj
                    n_stall = 0
                    trust = min(trust * 1.02, 0.5)
                else:
                    n_stall += 1
                    trust *= 0.8
            else:
                n_stall += 1
                trust *= 0.72
    return best_h, best_c5

def _lp_optimize(h, deadline, top_k_list=None, radius_init=0.15, max_iters=2000):
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    radius = radius_init
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        n_lags = len(scores)
        if tk is not None and tk < n_lags:
            idx = np.argsort(scores)[-tk:]
            scores_k = scores[idx]
            G_k = G[idx]
        else:
            scores_k = scores
            G_k = G
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(scores_k), n + 1))
        A_ub[:, :n] = G_k
        A_ub[:, n] = -1.0
        b_ub = -scores_k
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                ti += 1
                continue
        except Exception:
            stall += 1
            radius *= 0.7
            ti += 1
            continue
        p = res.x[:n]
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        a, new_c5 = _line_search(h, p, _max_alpha(h, p), n_scan=13)
        if a < 1e-12:
            stall += 1
            radius *= 0.75
            ti += 1
            continue
        h_new = h + a * p
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.7
            ti += 1
            if stall > 10 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            step = max(1, n // 32)
            for k in range(0, n, step):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0 < 0.5).astype(float)
    return _project(h, n / 2.0)

def _weighted_grad_step(h, deadline, n_steps=150):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 3.0 + 120.0 * frac
        scores = _scores(h)
        max_s = float(np.max(scores))
        w = np.exp(beta * (scores - max_s))
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.08
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0.copy(), n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Smooth gradient annealing on parent size
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _anneal_smooth(h_start, dl1, n_taus=8, steps_per_tau=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: LP polish on best
    if time.time() < t0 + budget_s * 0.32:
        dl2 = t0 + budget_s * 0.35
        h2, c5_2 = _lp_optimize(best_h, dl2, top_k_list=[None, 20, 10, 5],
                                 radius_init=0.15, max_iters=2000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Weighted gradient
    if time.time() < t0 + budget_s * 0.42:
        dl3 = t0 + budget_s * 0.45
        h3, c5_3 = _weighted_grad_step(best_h, dl3, n_steps=120)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-resolution search
    dl4 = t0 + budget_s * 0.68
    resolutions = [96, 112, 80, 128, 144]
    alphas = [0.6180339887, 0.41421356, 0.7071067812, 0.3819660113, 0.5556]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl4:
            break
        if best_c5 < 0.3804:
            break
        for s in range(2):
            if time.time() > dl4:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_init(n_try, alpha)
                h_init = _project(h_init + rng.normal(0, 0.06, n_try), n_try / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.035)
            h_try, c5_try = _anneal_smooth(h_init, sub_dl, n_taus=6, steps_per_tau=60)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                sub_dl2 = min(dl4, time.time() + budget_s * 0.018)
                h_lp, c5_lp = _lp_optimize(h_try, sub_dl2,
                                            top_k_list=[15, 8, 5],
                                            radius_init=0.12, max_iters=800)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = len(h_lp)

    # Phase 5: Smooth anneal from best at different tau ranges
    if time.time() < t0 + budget_s * 0.78:
        dl5 = t0 + budget_s * 0.80
        h5, c5_5 = _anneal_smooth(best_h, dl5, n_taus=6, steps_per_tau=80,
                                   trust_start=0.15, momentum=0.1)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.90
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        h_try, c5_try = _anneal_smooth(h_pert, sub_dl, n_taus=4, steps_per_tau=30,
                                        trust_start=0.08)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.1)

    # Phase 7: LP final polish
    if time.time() < t0 + budget_s * 0.95:
        dl7 = t0 + budget_s * 0.96
        h7, c5_7 = _lp_optimize(best_h, dl7, top_k_list=[5, 3, 4],
                                 radius_init=0.05, max_iters=800)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.98:
        dl8 = t0 + budget_s * 0.985
        hp, c5p = _pair_polish(best_h, dl8, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: ensure feasibility and compute exact value
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
