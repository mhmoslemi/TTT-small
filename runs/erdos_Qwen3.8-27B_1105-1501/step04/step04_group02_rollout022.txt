def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    """Gradient of the r-th correlation score w.r.t. h."""
    n = len(h)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        p = np.zeros(n)
        p[d:] += (1.0 - h[:m])
        p[:m] -= h[d:]
    else:
        g = -d
        m = n - g
        p = np.zeros(n)
        p[:m] += (1.0 - h[g:])
        p[g:] -= h[:m]
    return p

def _line_search(h, p, c5_cur, alpha_max, n_grid=40):
    """Line search: minimize exact C5 along h + t*p for t in [0, alpha_max]."""
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i in range(n_grid):
        t = ts[i]
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    # Golden section refinement
    lo_t = ts[max(0, idx - 3)]
    hi_t = ts[min(n_grid - 1, idx + 3)]
    if hi_t <= lo_t:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    
    f1, f2 = _f(c1), _f(c2)
    for _ in range(40):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _biased_pair_move(h, scores, n, rng, tau=0.1):
    """Generate a biased pair move."""
    # Find tight shifts
    s_max = np.max(scores)
    tight_mask = scores >= s_max - 0.01
    tight_shifts = np.where(tight_mask)[0]
    
    # Pick a random tight shift
    r = rng.choice(tight_shifts)
    d = r - (n - 1)
    
    if d >= 0:
        m = n - d
        # Contributions: h[k+d] * (1-h[k]) for k=0..m-1
        contribs = h[d:d+m] * (1 - h[:m])
        # Pick a position with large contribution
        # Weight by contribution
        w = contribs / (contribs.sum() + 1e-15)
        k = rng.choice(m, p=w)
        i = k + d    # high value (could decrease)
        j = k        # low value (could increase)
    else:
        g = -d
        m = n - g
        contribs = h[:m] * (1 - h[g:g+m])
        w = contribs / (contribs.sum() + 1e-15)
        k = rng.choice(m, p=w)
        i = k        # high value
        j = k + g    # low value
    
    # Transfer mass from i to j (decrease i, increase j)
    # This reduces the contribution at this position
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta < 1e-10:
        return None
    delta = rng.uniform(0.01, max_delta)
    
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(h, r):
    n = len(h)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        p = np.zeros(n)
        p[d:] += (1.0 - h[:m])
        p[:m] -= h[d:]
    else:
        g = -d
        m = n - g
        p = np.zeros(n)
        p[:m] += (1.0 - h[g:])
        p[g:] -= h[:m]
    return p

def _line_search(h, p, c5_cur, alpha_max, n_grid=50):
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i in range(n_grid):
        h_t = h + ts[i] * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 3)]
    hi_t = ts[min(n_grid - 1, idx + 3)]
    if hi_t - lo_t < 1e-12:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    f1, f2 = _f(c1), _f(c2)
    for _ in range(45):
        if hi_t - lo_t < 1e-15:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, d:] += 1.0
            G[r, d:] -= h[:m]
            G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += 1.0
            G[r, :m] -= h[g:]
            G[r, g:] -= h[:m]
    return G

def _lp_direction(h, scores, radius):
    n = len(h)
    G = _build_G(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_subgrad(h, deadline, tau_init=20.0, lr_init=0.015, mom_beta=0.9, max_stall=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    lr = lr_init
    stall = 0
    while time.time() < deadline and stall < max_stall:
        scores = _scores(h)
        s_max = float(np.max(scores))
        shifted = tau * (scores - s_max)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            wr = w[r]
            if wr < 1e-14:
                continue
            grad += wr * _row_grad(h, r)
        dx = 2.0 / n
        grad *= dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)

def _resample_exact(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    ratio = n_new / n_old
    if abs(ratio - round(ratio)) < 1e-10:
        r = int(round(ratio))
        h_new = np.tile(h, r)
        return h_new
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng_local = np.random.default_rng(_pass * 7919 + 13)
        n_try = min(n, 50)
        idx_j = rng_local.integers(0, n, size=n_try)
        idx_k = rng_local.integers(0, n, size=n_try)
        tried = set()
        for jj in idx_j:
            for kk in idx_k:
                if jj == kk:
                    continue
                key = (min(jj, kk), max(jj, kk))
                if key in tried:
                    continue
                tried.add(key)
                j, k = key
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                    if dj > 0:
                        am = (1.0 - h[j]) / dj
                    else:
                        am = h[j] / (-dj)
                    if dk > 0:
                        am = min(am, (1.0 - h[k]) / dk)
                    else:
                        am = min(am, h[k] / (-dk))
                    if am < 1e-12:
                        continue
                    for frac in [1.0, 0.5, 0.25]:
                        h_t = h.copy()
                        h_t[j] += dj * am * frac
                        h_t[k] += dk * am * frac
                        if h_t[j] < -1e-12 or h_t[j] > 1 + 1e-12 or h_t[k] < -1e-12 or h_t[k] > 1 + 1e-12:
                            continue
                        c5_t = _exact_c5(h_t)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            h = h_t.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

    # Phase 7: Perturbation + refine (8% budget)
    d7 = t0 + budget_s * 0.88
    n_curr = len(best_h)
    scale = 0.03
    attempt = 0
    while time.time() < d7 and attempt < 100:
        attempt += 1
        h_pert = best_h + rng.normal(0, scale, n_curr)
        h_pert = _project(h_pert, n_curr / 2.0)
        sub_dl = min(d7, time.time() + 8.0)
        h_try, c5_try = _lp_local(h_pert, sub_dl, radius_init=0.08, max_stall=20)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.2, scale * 1.3)

    # Phase 8: Final intensive LP + pair refine (7% budget)
    d8 = t0 + budget_s * 0.95
    h8, c5_8 = _lp_local(best_h, d8, radius_init=0.05, max_stall=100)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8:
        h8b, c5_8b = _pair_refine(best_h, min(d8, time.time() + budget_s * 0.02))
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()

    # Phase 9: Final LP polish (5% budget)
    d9 = t0 + budget_s * 0.995
    h9, c5_9 = _lp_local(best_h, d9, radius_init=0.02, max_stall=200)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        # projection slightly changed things, use the pre-projection if it was better
        # but we need feasibility, so use projected version
        pass
    return best_h, final_c5, len(best_h)
