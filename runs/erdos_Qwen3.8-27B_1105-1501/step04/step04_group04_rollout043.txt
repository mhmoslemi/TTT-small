def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _compute_c(h, i, j):
    """Compute c_d for the 2-coordinate move h_i += tau, h_j -= tau."""
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    
    # h_i contribution (+tau):
    # +dx*(1-h[i+d]) for d in [-i, n-1-i]
    d_lo = -i
    d_hi = n - 1 - i
    if d_lo <= d_hi:

        idx = (i + d_vals) % n  # i+d is always in [0,n) given the range
        # Actually i+d_lo = i-i = 0, i+d_hi = i+n-1-i = n-1, so idx = [0, n-1]
        c[n - 1 + d_vals] += dx * (1.0 - h[idx])

    d_lo = i - n + 1
    d_hi = i

        idx = i - d_vals  # ranges from i-(i-n+1)=n-1 to i-i=0
        c[n - 1 + d_vals] -= dx * h[idx]
    
    # h_j contribution (-tau):
    # -dx*(1-h[j+d]) for d in [-j, n-1-j]
    d_lo = -j
    d_hi = n - 1 - j

def _pair_minimax_tau(M, c, h_i, h_j):
    """
    Find tau minimizing max_k(M[k] + tau*c[k]) subject to tau in [tau_min, tau_max].
    Returns (tau_star, min_max_value).
    """
    tau_min = max(-h_i, h_j - 1.0)
    tau_max = min(1.0 - h_i, h_j)
    
    if tau_max - tau_min < 1e-15:
        return 0.0, float(np.max(M))
    
    lo = tau_min
    hi = tau_max
    
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = M + mid * c
        max_val = float(np.max(vals))
        # Find active set (within tolerance)
        tol = 1e-12
        active = np.where(vals >= max_val - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(vals)])
        c_min = float(np.min(c[active]))
        c_max = float(np.max(c[active]))
        if c_min > 1e-12:
            hi = mid
        elif c_max < -1e-12:
            lo = mid
        else:
            break
    
    tau_star = (lo + hi) / 2.0
    
    # Evaluate at critical points
    candidates = np.array([tau_min, tau_max, tau_star])
    best_val = float('inf')
    best_tau = 0.0
    for t in candidates:
        v = float(np.max(M + t * c))
        if v < best_val:
            best_val = v
            best_tau = t
    return best_tau, best_val

def _pair_sweep(h, rng, deadline, n_active_trials=6, max_passes=200):
    """
    Multiple passes of 2-coordinate minimax moves.
    Focus on coordinates where h is interior.
    """
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    
    for pat in range(max_passes):
        if time.time() > deadline:
            break
        if no_improve > 50:
            break
        
        # Get active indices (interior values)
        active_mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        active_idx = np.where(active_mask)[0]
        if active_idx.size == 0:
            break
        
        M = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        
        pass_improved = False
        # Process active indices in random order
        order = rng.permutation(active_idx.size)
        
        for k in order:
            if time.time() > deadline:
                break
            i = active_idx[k]
            if h[i] <= 1e-6 or h[i] >= 1.0 - 1e-6:
                continue
            
            # Try several j values
            found = False
            for _ in range(n_active_trials):
                j = rng.integers(n)
                if j == i:
                    continue
                c = _compute_c(h, i, j)
                tau, val_at_tau = _pair_minimax_tau(M, c, h[i], h[j])
                if abs(tau) < 1e-12:
                    continue
                # Check if this actually improves
                if val_at_tau < cur_c5 - 1e-13:
                    # Apply the move
                    h[i] += tau
                    h[j] -= tau
                    h = np.clip(h, 0.0, 1.0)
                    # The sum should be preserved, but let's verify
                    if abs(h.sum() - n / 2.0) > 1e-10:
                        h = _project_box_sum(h, n / 2.0)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    no_improve = 0
                    pass_improved = True
                    # Recompute M for next iteration
                    M = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                    found = True
                    break
            
            if not found:
                no_improve += 1
                if no_improve > 50:
                    break
    
    return best_h, best_c5

def _project_box_sum(v, target):

        b_ub = G @ h - F
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        lo_b = np.maximum(0.0, h - radius)
        hi_b = np.minimum(1.0, h + radius)
        bounds = list(zip(lo_b, hi_b))
        bounds.append((None, None))
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 10000, 'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_start=0.10, max_iters=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    radius = radius_start
    stall = 0
    factors = [1.0, 0.7, 0.5, 0.3, 0.15]
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        x_new = _lp_trust_step(h, radius)
        if x_new is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        x_new = _project_box_sum(x_new, n / 2.0)
        best_step = 0.0
        best_tc5 = cur_c5
        for fac in factors:
            trial = _project_box_sum(h + fac * (x_new - h), n / 2.0)
            tc5 = _exact_c5(trial)
            if tc5 < best_tc5 - 1e-13:
                best_tc5 = tc5
                best_step = fac
        if best_step > 0:
            h = _project_box_sum(h + best_step * (x_new - h), n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.72
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _quantize_pass(h, deadline):
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for i in range(n):
        if time.time() > deadline:
            break
        if h[i] <= 1e-6 or h[i] >= 1.0 - 1e-6:
            continue
        for target in [0.0, 1.0]:
            h_new = h.copy()
            h_new[i] = target
            h_new = _project_box_sum(h_new, n / 2.0)
            c = _exact_c5(h_new)
            if c < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
                break
    return best_h, best_c5

def _multi_n_explore(h_best, c5_best, n_best, rng, deadline):
    n_list = [n_best - 8, n_best - 4, n_best + 4, n_best + 8,
              n_best + 16, n_best - 16, 64, 80, 96, 112, 128, 144, 160]
    n_list = sorted(set(max(32, x) for x in n_list if 32 <= x <= 200))
    n_list = sorted(n_list, key=lambda x: abs(x - n_best))
    per_n = max(5.0, (deadline - time.time()) / max(1, len(n_list)) * 0.8)
    for n_try in n_list:
        if time.time() > deadline:
            break
        if n_try == n_best:
            continue
        dl_n = min(deadline, time.time() + per_n)
        if dl_n - time.time() < 2.0:
            break
        h_r = _resample(h_best, n_try)
        # Quick pair sweeps
        h_r, c5_r = _pair_sweeps(h_r, rng, dl_n, max_passes=100, trials_per_i=3)
        if c5_r < c5_best - 1e-10:
            h_best = h_r.copy()
            c5_best = c5_r
            n_best = n_try
    return h_best, c5_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        n0 = 128
        h0 = np.zeros(n0)
        h0[np.arange(0, n0, 2)] = 1.0
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Pair minimax sweeps (main driver) - 35% time
    dl1 = t0 + budget_s * 0.35
    h1, c5_1 = _pair_sweeps(h, rng, dl1, max_passes=300, trials_per_i=5)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP trust region - 15% time
    dl2 = t0 + budget_s * 0.50
    h2, c5_2 = _lp_loop(best_h, dl2, radius_start=0.08)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More pair sweeps on improved point - 25% time
    dl3 = t0 + budget_s * 0.75
    h3, c5_3 = _pair_sweeps(best_h, rng, dl3, max_passes=400, trials_per_i=6)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Quantization pass
    if time.time() < t0 + budget_s * 0.80:
        h4, c5_4 = _quantize_pass(best_h, t0 + budget_s * 0.80)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
            # Follow-up pair sweeps
            dl4 = t0 + budget_s * 0.87
            h4b, c5_4b = _pair_sweeps(best_h, rng, dl4, max_passes=200, trials_per_i=4)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: Multi-n exploration - 10% time
    dl5 = t0 + budget_s * 0.95
    best_h, best_c5, best_n = _multi_n_explore(best_h, best_c5, best_n, rng, dl5)

    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.97:
        h6, c5_6 = _pair_sweeps(best_h, rng, t0 + budget_s * 0.99, max_passes=200, trials_per_i=4)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _exact_c5(best_h)

    return best_h, best_c5, best_n
