import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _compute_G(h):
    """Gradient matrix G[(2n-1), n]"""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        if d >= 0:
            m = n - d
            G[r, d:] = (1.0 - h[:m])
            G[r, :m] = -h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] = (1.0 - h[g:])
            G[r, g:] = -h[:m]
    return G

def _active_A(h, scores, G):
    """Weighted gradient from active lags"""
    max_s = float(np.max(scores))
    thresh = max_s - 0.003
    active = scores >= thresh
    if not np.any(active):
        active = scores >= max_s - 0.01
    A = G[active, :].sum(axis=0)
    return A

def _project(v, target):
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    x = np.clip(v - t, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return x

def _stochastic_search(h_init, deadline, seed=42, T_start=0.004):
    """Stochastic annealing with gradient-biased proposals."""
    n = len(h_init)
    rng = np.random.default_rng(seed)
    
    h = h_init.copy()
    cur_scores = _scores_of(h)
    cur_c5 = float(np.max(cur_scores))
    best_h = h.copy()
    best_c5 = cur_c5
    
    phase_start = time.time()
    phase_dur = max(deadline - phase_start, 1.0)
    
    G = _compute_G(h)
    A = _active_A(h, cur_scores, G)
    
    no_improve_count = 0
    n_iter = 0
    
    while time.time() < deadline:
        frac = min(1.0, (time.time() - phase_start) / phase_dur)
        T = max(1e-7, T_start * (1.0 - frac) ** 2.5)
        
        # Recompute gradient periodically
        if n_iter % 200 == 0:
            G = _compute_G(h)
            A = _active_A(h, cur_scores, G)
        
        # Build proposal weights
        can_inc = h < 1.0 - 1e-8
        can_dec = h > 1e-8
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        A_scale = max(0.005, float(np.std(A)))
        w_inc = np.exp(-A / A_scale).copy()
        w_dec = np.exp(A / A_scale).copy()
        w_inc += 0.2  # exploration floor
        w_dec += 0.2
        w_inc[~can_inc] = 0.0
        w_dec[~can_dec] = 0.0
        wis = float(w_inc.sum())
        wds = float(w_dec.sum())
        if wis < 1e-10 or wds < 1e-10:
            break
        
        i = int(rng.choice(n, p=w_inc / wis))
        j = int(rng.choice(n, p=w_dec / wds))
        if i == j:
            n_iter += 1
            continue
        
        max_delta = min(1.0 - h[i], h[j])
        if max_delta < 1e-10:
            n_iter += 1
            continue
        
        # Step size: larger early, smaller late
        step = max(0.003, 0.2 * T / T_start)
        delta = min(max_delta, step * rng.uniform(0.5, 2.0))
        if rng.random() < 0.3:
            delta = -delta  # sometimes reverse direction
        
        if abs(delta) < 1e-10:
            n_iter += 1
            continue
        
        # Apply move
        h[i] += delta
        h[j] -= delta
        
        # Check bounds (should be fine by construction but safety)
        if h[i] < -1e-12 or h[i] > 1.0 + 1e-12 or h[j] < -1e-12 or h[j] > 1.0 + 1e-12:
            h[i] -= delta
            h[j] += delta
            n_iter += 1
            continue
        
        # Evaluate
        new_scores = _scores_of(h)
        new_c5 = float(np.max(new_scores))
        
        # Metropolis
        dC = new_c5 - cur_c5
        accept = False
        if dC <= 0:
            accept = True
        else:
            # Cap: don't accept moves that increase C5 by more than 3*T
            if dC < 3.0 * T:
                if rng.random() < np.exp(-dC / T):
                    accept = True
        
        if accept:
            cur_scores = new_scores
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            h[i] -= delta
            h[j] += delta
            no_improve_count += 1
        
        # Reheat if stuck
        if no_improve_count > 5000:
            T = max(T, T_start * 0.5)
            no_improve_count = 0
            # Jump to best with perturbation
            noise = rng.normal(0, 0.02, n)
            h_pert = best_h + noise
            h_pert = _project(h_pert, n / 2.0)
            h = h_pert
            cur_scores = _scores_of(h)
            cur_c5 = float(np.max(cur_scores))
            G = _compute_G(h)
            A = _active_A(h, cur_scores, G)

def _line_search(h, p, c5_cur, alpha_max, n_grid=20, n_golden=25):
    """Exact line search along direction p."""
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    
    # Grid search
    ts = np.linspace(0.0, alpha_max, n_grid)
    best_grid_t = 0.0
    best_grid_val = c5_cur
    for t in ts:
        h_t = h + t * p
        c5_t = _exact_c5(h_t)
        if c5_t < best_grid_val:
            best_grid_val = c5_t
            best_grid_t = t
    
    # Golden section refinement
    lo = ts[max(0, int(np.argmin([_exact_c5(h + t * p) for t in ts])) - 2)]
    # Actually let me just do it properly
    idx_best = int(np.argmin([_exact_c5(h + t * p) for t in ts]))
    lo = ts[max(0, idx_best - 1)]
    hi = ts[min(n_grid - 1, idx_best + 1)]
    if hi <= lo:
        lo, hi = 0.0, alpha_max
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def _f(t):
        return _exact_c5(h + t * p)
    
    f1 = _f(c1)
    f2 = _f(c2)
    for _ in range(n_golden):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _f(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _f(c2)
    
    opt_t = 0.5 * (lo + hi)
    opt_val = _f(opt_t)
    if opt_val < best_grid_val:
        best_grid_val = opt_val
        best_grid_t = opt_t
    
    return best_grid_t, best_grid_val
