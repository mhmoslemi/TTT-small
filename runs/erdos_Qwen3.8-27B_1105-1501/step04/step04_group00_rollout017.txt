def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:d+m] += (1.0 - h[:m])
                G[idx, :m] += -h[d:d+m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:g+m] += -h[:m]
    return G * dx

def _lp_step(h, scores, active_idx, radius):

    G = _grad_rows(h, lags)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.ones((1, n + 1))
    A_eq[0, -1] = 0.0
    b_eq = np.array([0.0])
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_fp, c5_fp = _lp_local(best_h, sub_dl, max_iters=2000, init_radius=0.08)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()
    
    # Phase 5: Subgradient + LP alternating on best
    if time.time() < t0 + budget_s * 0.78:
        dl5 = t0 + budget_s * 0.82
        for _round in range(4):
            if time.time() > dl5:
                break
            h_sg, c5_sg = _subgradient_step(best_h, min(dl5, time.time() + budget_s * 0.02), n_steps=100)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
            h_l, c5_l = _lp_local(best_h, min(dl5, time.time() + budget_s * 0.02), max_iters=1500, init_radius=0.06)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()
    
    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.91
    scale = 0.025
    n_no_improve = 0
    while time.time() < dl6 and n_no_improve < 60:
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        h_try, c5_try = _lp_local(h_pert, sub_dl, max_iters=600, init_radius=0.06)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
            scale = max(0.003, scale * 0.82)
            n_no_improve = 0
        else:
            scale = min(0.10, scale * 1.08)
            n_no_improve += 1
    
    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.95:
        dl7 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, dl7, max_passes=25)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
    
    # Phase 8: Final LP
    if time.time() < t0 + budget_s * 0.985:
        h_fin, c5_fin = _lp_local(best_h, dl_final, max_iters=2000, init_radius=0.015)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
