def _epi_obj(z, n):
    return z[n]

def _epi_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq_fun(z, n):
    return np.sum(z[:n]) - n * 0.5

def _eq_jac(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g

def _active_ineq_fun(z, n, top_idx):
    h = z[:n]
    t = z[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr[top_idx]

def _active_ineq_jac(z, n, top_idx):
    h = z[:n]
    _, G = _corr_and_grad(h, n)
    J = np.zeros((len(top_idx), n + 1))
    J[:, :n] = -G[top_idx, :]
    J[:, n] = 1.0
    return J

def _slsqp_active(h_init, n, K=10, maxiter=200, ftol=1e-12):
    """SLSQP with only top-K correlation constraints."""
    h = np.asarray(h_init, dtype=float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:K]
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, n)]
    
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _active_ineq_fun, 'jac': _active_ineq_jac, 'args': (n, top_idx)}
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_jac, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)

def _lse_smooth(h, n, alpha, maxiter=100):
    """Log-sum-exp smoothing step."""
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(corr))
        vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
        exp_v = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_v)))
        w = exp_v / float(np.sum(exp_v))
        # Gradient
        _, G = _corr_and_grad(x, n)
        grad = G.T @ w
        grad -= np.mean(grad)  # Remove component along constraint
        return f, grad
    
    try:
        res = minimize(obj_and_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 50})
        x = _project(res.x, target)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)

def _coord_descent(h, n, max_iters=1000, deadline=None):
    """Coordinate descent: try moving each coordinate to reduce max correlation."""
    h = h.copy()
    target = n * 0.5
    best_c5 = _exact_c5(h)

        if deadline is not None and time.time() > deadline:

                if c5_new < best_c5 - 1e-10:

                    break
            if improved:
                break

def _pair_swap(h, n, max_iters=500, deadline=None):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

    h_new = np.interp(x_new, x_old, h)

        m2 = (ps >= 0) & (ps < n)
        G[r, m2] -= h[ps[m2]]
    return corr, G

def _verify_grad(n, h):
    corr, G = _corr_grad(h, n)
    eps = 1e-6
    max_err = 0.0
    for r in [0, n // 2, n - 1, 2 * n - 2]:
        for j in [0, n // 4, n // 2, 3 * n // 4, n - 1]:
            hp = h.copy(); hp[j] += eps
            hm = h.copy(); hm[j] -= eps
            fp = np.correlate(hp, 1.0 - hp, mode='full')[r]
            fm = np.correlate(hm, 1.0 - hm, mode='full')[r]
            fd = (fp - fm) / (2 * eps)
            max_err = max(max_err, abs(fd - G[r, j]))
    return max_err < 1e-3

def _epi_obj(z, n):
    return z[n]

def _epi_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5

def _eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g

def _act_f(z, n, idx):
    h = z[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return z[n] - corr[idx]

def _act_g(z, n, idx):
    h = z[:n]
    _, G = _corr_grad(h, n)
    J = np.zeros((len(idx), n + 1))
    J[:, :n] = -G[list(idx), :]
    J[:, n] = 1.0
    return J

def _slsqp_act(h init, n, K, maxiter, ftol):
    h = np.asarray(h_init, dtype=float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(-corr)[:K]
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, n)]
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _act_f, 'jac': _act_g, 'args': (n, top)}
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_jac, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)

def _lse_step(h, n, alpha, maxiter, lam):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        ms = float(np.max(corr))
        vals = np.clip(alpha * (corr - ms), -400.0, 400.0)
        ev = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(ev)))
        f += lam * (np.sum(x) - target) ** 2
        w = ev / float(np.sum(ev))
        _, G = _corr_grad(x, n)
        grad = G.T @ w
        grad += 2.0 * lam * (np.sum(x) - target) * np.ones(n)
        return f, grad

    try:
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 80})
        x = _project(res.x, target)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)

def _subgrad_loop(h, n, deadline, max_iters=500):
    h = h.copy()
    target = n * 0.5
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    step = 0.5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        r_max = int(np.argmax(corr))
        s = r_max - (n - 1)
        j = np.arange(n)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        pd -= np.mean(pd)
        nm = np.linalg.norm(pd)
        if nm < 1e-14:
            break
        pd = pd / nm
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_new = _project(h - frac * step * pd, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                step = min(step * 1.1, 2.0)
                break
        else:
            step *= 0.7
            if step < 0.01:
                break
    return best_h, best_c5

def _pair_climb(h, n, deadline, max_iters=1000):
    h = h.copy()
    target = n * 0.5
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        r_max = int(np.argmax(corr))
        s = r_max - (n - 1)
        j = np.arange(n)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        # Find best pair: move mass from low-pd to high-pd (reduces corr at r_max)
        # Actually we want to DECREASE corr[r_max], so move in -pd direction
        # h_new[i] += d, h_new[j] -= d reduces corr by d*(pd[i] - pd[j])
        # We want pd[i] - pd[j] > 0, i.e., pd[i] > pd[j]
        # So move mass FROM j (low pd) TO i (high pd)
        # Wait: corr change = d * pd[i] + (-d) * pd[j] = d*(pd[i]-pd[j])
        # We want this negative: pd[i] - pd[j] < 0, so pd[i] < pd[j]
        # Move from j (high pd) to i (low pd): h[i]+=d, h[j]-=d
        # Change = d*(pd[i]-pd[j]) < 0 when pd[i] < pd[j]
        improved = False
        order = np.argsort(pd)
        n_try = min(8, n // 4)
        for a in range(n_try):
            for b in range(n - n_try, n):
                i, jj = order[a], order[b]
                if i == jj:
                    continue
                for d in [0.05, 0.1, 0.15, 0.03, -0.03, -0.05]:
                    h_new = h.copy()
                    h_new[i] += d
                    h_new[jj] -= d
                    h_new = _project(h_new, target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < best_c5 - 1e-12:
                        h = h_new
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try random pairs
            found = False
            for _ in range(10):
                i, jj = np.random.randint(0, n, 2)
                if i == jj:
                    continue
                for d in [0.1, -0.1, 0.2, -0.2]:
                    h_new = h.copy()
                    h_new[i] += d
                    h_new[jj] -= d
                    h_new = _project(h_new, target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < best_c5 - 1e-12:
                        h = h_new
                        best_c5 = c5_new
                        best_h = h.copy()
                        found = True
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _diverse_starts(n, seed, count=8):
    rng = np.random.default_rng(seed)
    target = n * 0.5
    starts = []
    # Random binary
    h = (rng.random(n) > 0.5).astype(float)
    starts.append(_project(h, target))
    # Sinusoidal patterns
    x = np.linspace(0, 2, n)
    for f in [1, 2, 3, 4]:
        h = 0.5 + 0.39 * np.sin(np.pi * f * x)
        starts.append(_project(h, target))
    # Random smooth
    h = 0.5 + rng.normal(0, 0.3, n)
    starts.append(_project(h, target))
    # Block pattern
    blk = max(2, n // 10)
    h = np.zeros(n)
    h[::2 * blk:blk] = 1.0
    starts.append(_project(h, target))
    return starts[:count]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Verify gradient correctness
    _ = _verify_grad(n0, h)

    # Phase 1: LSE warmup (10%)
    d1 = t0 + budget_s * 0.10
    for alpha in [5.0, 10.0, 25.0, 60.0, 150.0]:
        if time.time() > d1:
            break
        h, c5 = _lse_step(h, n0, alpha, 80, 100.0)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Active-set SLSQP (20%)
    d2 = t0 + budget_s * 0.30
    for K in [5, 8, 10, 15]:
        if time.time() > d2:
            break
        h_s, c5_s = _slsqp_act(best_h, n0, K, 400, 1e-14)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    # Phase 3: Subgradient + pair climb (15%)
    d3 = t0 + budget_s * 0.45
    h_sg, c5_sg = _subgrad_loop(best_h, n0, d3, 300)
    if c5_sg < best_c5:
        best_c5, best_h = c5_sg, h_sg.copy()
    h_pc, c5_pc = _pair_climb(best_h, n0, d3, 800)
    if c5_pc < best_c5:
        best_c5, best_h = c5_pc, h_pc.copy()

    # Phase 4: Multistart with perturbation + SLSQP (25%)
    d4 = t0 + budget_s * 0.70
    scale = 0.04
    for attempt in range(300):
        if time.time() > d4:
            break
        if best_c5 <= 0.3806:
            break
        h_p = _project(best_h + rng.normal(0, scale, n0), target)
        h_s, c5_s = _slsqp_act(h_p, n0, 10, 200, 1e-13)
        # Quick pair climb
        h_pc, c5_pc = _pair_climb(h_s, n0, min(d4, time.time() + budget_s * 0.005), 100)
        if c5_pc < c5_s:
            c5_s, h_s = c5_pc, h_pc
        if c5_s < best_c5 - 1e-10:
            best_c5, best_h = c5_s, h_s.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 5: Resolution search (12%)
    d5 = t0 + budget_s * 0.82
    for dn in [16, 32, -16, 48, -32, 64]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 256 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d5, time.time() + budget_s * 0.018)
        h_r, c5_r = _lse_step(h_r, n_try, 30.0, 50, 100.0)
        h_s, c5_s = _slsqp_act(h_r, n_try, 10, 200, 1e-13)
        if c5_s < c5_r:
            h_r, c5_r = h_s, c5_s
        sub_d2 = min(d5, time.time() + budget_s * 0.012)
        h_pc, c5_pc = _pair_climb(h_r, n_try, sub_d2, 300)
        if c5_pc < c5_r:
            h_r, c5_r = h_pc, c5_pc
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Diverse starts (4%)
    d6 = t0 + budget_s * 0.86
    for h_start in _diverse_starts(best_n, seed + 100, 5):
        if time.time() > d6:
            break
        c5_st = _exact_c5(h_start)
        if c5_st < best_c5:
            best_c5, best_h = c5_st, h_start.copy()
        h_s, c5_s = _lse_step(h_start, best_n, 30.0, 40, 100.0)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        h_s2, c5_s2 = _slsqp_act(h_s, best_n, 8, 150, 1e-13)
        if c5_s2 < best_c5:
            best_c5, best_h = c5_s2, h_s2.copy()

    # Phase 7: Final polish (9%)
    d7 = deadline - 8.0
    h_f, c5_f = _slsqp_act(best_h, best_n, 10, 400, 1e-15)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    h_sg, c5_sg = _subgrad_loop(best_h, best_n, d7, 200)
    if c5_sg < best_c5:
        best_c5, best_h = c5_sg, h_sg.copy()
    h_pc, c5_pc = _pair_climb(best_h, best_n, d7, 500)
    if c5_pc < best_c5:
        best_c5, best_h = c5_pc, h_pc.copy()
    if time.time() < d7:
        h_f2, c5_f2 = _slsqp_act(best_h, best_n, 5, 200, 1e-15)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
