    if target >= n:
        return np.ones(n)

def _trust_region_loop(h, n, deadline, delta_init=0.05, max_iters=5000):

    h = h.copy()
    cur_f = _exact_c5(h)
    best_h, best_f = h.copy(), cur_f
    delta = delta_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        # Solve LP
        p, predicted_t, M = _lp_minimax(h, n, delta)
        
        if p is None:
            stall += 5
            delta *= 0.7
            if delta < 1e-6:
                break
            continue
        
        # Check if LP says no improvement possible
        if predicted_t >= M - 1e-10:
            # No predicted improvement, try smaller delta or break
            stall += 1
            delta *= 0.7
            if delta < 1e-6:
                break
            continue
        
        predicted_redu = M - predicted_t  # This is t* (the LP optimum)
        # Actually, the LP minimizes t such that G*r <= t - corr
        # So at optimum, t = max_r (corr_r + G_r*p)
        # The predicted value is t, and current is M = max(corr)
        # So predicted reduction = M - t
        
        # Line search
        alpha, new_f = _line_search(h, p, n, cur_f)
        
        if alpha > 0 and new_f < cur_f - 1e-12:
            actual_redu = cur_f - new_f
            ratio = actual_redu / max(predicted_redu, 1e-10)
            
            h = _project(h + alpha * p, n * 0.5)
            cur_f = new_f
            
            if cur_f < best_f:
                best_f, best_h = cur_f, h.copy()
            
            # Trust region update
            if ratio > 0.7:
                delta = min(delta * 1.5, 0.5)
            elif ratio < 0.25:
                delta = max(delta * 0.5, 1e-6)
            # else: keep delta
            
            stall = 0
        else:
            stall += 1
            delta *= 0.6
            if delta < 1e-6:
                break
    
    return best_h, best_f

def _lse_solve(h, n, alpha, maxiter=200):
    """LSE-based optimization with given alpha."""
    j = np.arange(n)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1.0 - h, mode='full')
        ms = float(np.max(corr))
        vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
        exp_v = np.exp(vals)
        f = ms + (1.0/alpha) * np.log(float(np.sum(exp_v)))
        
        # Gradient
        w = exp_v / float(np.sum(exp_v))
        grad = np.zeros(n)
        for r in range(2*n-1):
            s = r - (n-1)
            gs = j + s
            m1 = (gs >= 0) & (gs < n)
            if m1.any():
                grad[m1] += w[r] * (1.0 - h[gs[m1]])
            ps = j - s
            m2 = (ps >= 0) & (ps < n)
            if m2.any():
                grad[m2] -= w[r] * h[ps[m2]]
        grad -= np.mean(grad)  # Project onto sum=0
        return f, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter,
                       'ftol': 1e-15, 'maxls': 100, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

import numpy as np

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0: continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12: break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0: continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12: break
    return x

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def _grad_matrix(h, n):
    G = np.zeros((2*n-1, n))
    j = np.arange(n)
    for r in range(2*n-1):
        s = r - (n-1)
        idx1 = j + s
        m1 = (idx1 >= 0) & (idx1 < n)
        G[r, m1] += 1.0 - h[idx1[m1]]
        idx2 = j - s
        m2 = (idx2 >= 0) & (idx2 < n)
        G[r, m2] -= h[idx2[m2]]
    return G

def _solve_lp(h, delta, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = float(np.max(corr))
    G = _grad_matrix(h, n)
    m = 2*n - 1
    c = np.zeros(n+1); c[-1] = 1.0
    A_ub = np.zeros((m, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corr
    A_eq = np.ones((1, n)); 
    A_eq_full = np.zeros((1, n+1)); A_eq_full[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for jj in range(n):
        lo_b = max(-delta, -h[jj])
        hi_b = min(delta, 1.0 - h[jj])
        if lo_b > hi_b: lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq_full, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], float(res.x[-1]), M
    return None, None, M

    for jj in range(n):
        if p[jj] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[jj]) / p[jj])
        elif p[jj] < -1e-12:

    alpha_max = max(alpha_max, 1e-10)
    best_a, best_f = 0.0, f_cur
    # Coarse search
