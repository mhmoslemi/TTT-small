def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(300):

        diff = target - s
        if abs(diff) < 1e-15:

        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):

        low = max(0, -d)

        i = np.arange(low, high)
        j = i + d
        G[r, i] += dx * (1.0 - h[j])
        G[r, j] += -dx * h[i]
    return G

def _F_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _gradient_max_h(h, top_k=5):
    """Gradient of the top-k average of F values."""
    F = _F_values(h)
    # Get top k indices
    k = min(top_k, len(F))
    idx = np.argsort(F)[-k:]
    w = np.zeros(len(F))
    w[idx] = 1.0 / k
    G = _build_G(h)
    return G.T @ w

    F = _F_values(h)
    Fm = F.max()
    e = np.exp(beta * (F - Fm))
    w = e / e.sum()
    G = _build_G(h)

    n = len(h)
    target = int(n / 2)
    # Sort by value, keep top n/2 as 1
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0

    """Binary swap search: swap 1s and 0s to reduce max overlap."""
    n = len(h)

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

            break
        if stall > 20000:
            break

        j = zeros[rng.integers(len(zeros))]

        h[j] = 1.0

                best_h = h.copy()
            stall = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    
    return best_h, best_c5
