import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_target(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(v - mu, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr(h, d):
    """Gradient of corr[q] (unscaled) w.r.t. h, where d = abs(q - (n-1))."""
    n = len(h)
    if d >= n:
        return np.zeros(n)
    m = n - d
    g = np.zeros(n)
    g[0:m] += (1.0 - h[d:n])
    g[d:n] += -h[0:m]
    return g  # unscaled; multiply by dx = 2/n when needed

def weighted_grad(h, top_dists, weights):
    """Compute weighted sum of gradients for top lags."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for d, w in zip(top_dists, weights):
        g += w * grad_corr(h, d)
    return g * dx

def lp_vertex_binary(w, n):
    """For target = n/2 (integer), find the binary vertex minimizing w·s.
    Sets the n/2 smallest w entries to 1, rest to 0."""
    idx = np.argsort(w)
    s = np.zeros(n)
    s[idx[:n//2]] = 1.0
    return s

def fw_iteration(h, eta, max_active):
    """One Frank-Wolfe iteration. Returns (new_h, improved)."""
    n = len(h)
    target = n / 2.0
    corr = corr_full(h)
    lam = float(np.max(corr))
    
    # Softmax weights over top correlations
    # Find the indices of the top correlations
    top_idx = np.argsort(corr)[-max_active:]
    top_vals = corr[top_idx]
    
    # Softmax
    exp_vals = np.exp((top_vals - top_vals.max()) / eta)
    weights = exp_vals / exp_vals.sum()
    
    # Compute weighted gradient
    dx = 2.0 / n
    g = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        d = abs(idx - (n - 1))
        g += w * grad_corr(h, d)
    g *= dx
    
    # LP vertex
    s = lp_vertex_binary(g, n)
    
    # Line search
    best_alpha = 0.0
    best_val = exact_c5(h)
    for alpha in np.linspace(0.1, 1.0, 10):
        h_t = (1.0 - alpha) * h + alpha * s
        # Check feasibility (should be exact since both are feasible)
        if abs(h_t.sum() - target) > 1e-10:
            h_t = project_to_target(h_t, target)
        val = exact_c5(h_t)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0:
        new_h = (1.0 - best_alpha) * h + best_alpha * s
        if abs(new_h.sum() - target) > 1e-10:
            new_h = project_to_target(new_h, target)
        return new_h, True
    return h, False
