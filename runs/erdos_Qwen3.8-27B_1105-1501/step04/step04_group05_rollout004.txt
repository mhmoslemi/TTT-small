```python
import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(v, target):
    n = len(v)
    t = float(target)
    if t <= 0: return np.zeros(n)
    if t >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > t:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def lp_vertex(w, target):
    order = np.argsort(w, kind='mergesort')
    n = len(w)
    x = np.zeros(n)
    k = int(target)
    x[order[:k]] = 1.0
    if k < n and abs(target - k) > 1e-12:
        x[order[k]] = target - k
    return x

def fw_line(h, s):
    """Line search h -> s, return best (score, h_new)."""
    best_sc, best_h, best_a = exact_c5(h), h.copy(), 0.0
    # Level 1: coarse
    for a in np.linspace(0.05, 1.0, 19):
        h_t = (1-a)*h + a*s
        sc = exact_c5(h_t)
        if sc < best_sc:
            best_sc, best_h, best_a = sc, h_t, a
    if best_a == 0: return best_sc, best_h
    # Level 2: medium
    lo, hi = max(0, best_a-0.1), min(1, best_a+0.1)
    for a in np.linspace(lo, hi, 21):
        h_t = (1-a)*h + a*s
        sc = exact_c5(h_t)
        if sc < best_sc:
            best_sc, best_h, best_a = sc, h_t, a
    # Level 3: fine
    lo, hi = max(0, best_a-0.02), min(1, best_a+0.02)
    for a in np.linspace(lo, hi, 21):
        h_t = (1-a)*h + a*s
        sc = exact_c5(h_t)
        if sc < best_sc:
            best_sc, best_h, best_a = sc, h_t, a
    return best_sc, best_h

def fw_iter(h, target, dx, lambdas):
    """One FW iteration trying multiple active set temperatures.
    lambdas: list of temperature fractions (as fraction of max corr)"""
    n = len(h)
    nc = 2*n - 1
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    lam = float(corr.max())
    best_sc, best_h = lam, h.copy()
    
    for frac in lambdas:
        tau = max(lam * frac, 1e-6)
        wts = np.exp((corr - lam) / tau)
        wts /= wts.sum()
        
        # Weighted gradient
        g = np.zeros(n)
        for q in range(nc):
            if wts[q] < 1e-12: continue
            d = q - (n-1)
            m = n - abs(d)
            if m <= 0: continue
            c = wts[q] * dx
            if d >= 0:
                A = np.arange(d, n); B = np.arange(0, m)
                g[A] += c * (1-h[B]); g[B] -= c * h[A]
            else:
                A = np.arange(0, m); B = np.arange(-d, n)
                g[A] += c * (1-h[B]); g[B] -= c * h[A]
        
        s = lp_vertex(g, target)
        sc, h_new = fw_line(h, s)
        if sc < best_sc:
            best_sc, best_h = sc, h_new
    
    return best_sc, best_h
```
