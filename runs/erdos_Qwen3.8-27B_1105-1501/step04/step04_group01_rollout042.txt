- The line search is the same (exact, using the quadratic formula)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _waterfill(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, rho):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -rho)
    t_hi = min(t_hi, rho)
    
    if t_hi - t_lo < 1e-13:
        # Try the other sign
        return _line_search_neg(h, -d, dx, rho)
    
    # Coarse search
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_val = obj_grid[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(30):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    
    best_t = opt_t if opt_val < best_val else grid[best_i]
    best_val = min(opt_val, best_val)
    
    return best_t, best_val

def _line_search_neg(h, d, dx, rho):
    # Try negative direction
    return _line_search(h, d, dx, rho)  # Same function, just called with -d

def _weighted_grad(h, w):
    n = len(h)
    conv = np.convolve(w, 1.0 - h, mode='full')
    corr = np.correlate(w, h, mode='full')
    g = conv[n-1:2*n-1] - corr[n-1::-1]
    return g

def _local_opt(h, deadline, beta_init=50.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_init
    lr = 0.05
    stall = 0
    
    while time.time() < deadline - 2:
        if stall > 80:
            break
        
        R = np.correlate(h, 1.0 - h, mode='full')
        max_R = float(np.max(R))
        # Softmax weights
        shifted = beta * (R - max_R)
        w = np.exp(shifted)
        w = w / float(np.sum(w))
        
        # Gradient
        g = _weighted_grad(h, w)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            beta *= 1.5
            stall += 5
            continue
        g = g / gn
        
        # Try both signs
        for sign in [-1, 1]:
            d = sign * g
            t, new_c5_ls = _line_search(h, d * lr, dx, lr)
            if new_c5_ls < cur_c5 - 1e-13:
                h_new = h + t * d * lr
                h_new = _waterfill(h_new, target)
                new_c5 = _exact_c5(h_new)
                if new_c5 < cur_c5 - 1e-13:
                    h = h_new

def _waterfill(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:

def _line_search(h, d, dx, rho):
    n = len(h)
    d = d - d.mean()
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):

    t_lo = max(t_lo, -rho)
    t_hi = min(t_hi, rho)

        return 0.0, float(np.max(S0) * dx)
    
    # Coarse grid
    n_grid = 32
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_val = float(obj_grid[best_i])
    
    # Golden section
    if best_i > 0 and best_i < n_grid - 1:
        a, b = float(grid[best_i - 1]), float(grid[best_i + 1])

    gr = (np.sqrt(5.0) - 1.0) / 2.0

            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx

        best_i_t = opt_t
    else:

def _weighted_grad(h, w):

    conv = np.convolve(w, 1.0 - h, mode='full')
    corr = np.correlate(w, h, mode='full')
    g = conv[n-1:2*n-1] - corr[n-1::-1]

def _worst_shift_grad(h, w_idx):
    """Gradient of the overlap at a specific lag index."""
    n = len(h)
    d = w_idx - (n - 1)  # actual shift
    g = np.zeros(n)
    for j in range(n):

        # First term: [0 <= j-d < n] * (1-h[j-d])
        idx1 = j - d
        if 0 <= idx1 < n:
            val += (1.0 - h[idx1])

        idx2 = j + d
        if 0 <= idx2 < n:
            val -= h[idx2]
        g[j] = val
    return g

T' = correlate(-d, 1-h) - correlate(h, -d) = -correlate(d, 1-h) + correlate(h, d) = -T

C' = correlate(-d, -d) = correlate(d, d) = C
