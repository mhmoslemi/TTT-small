    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    n_lags = 2 * n - 1
    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_s = float(np.max(scores))
        G = _grad_matrix_lags(h, np.arange(n_lags))
        p = _lp_direction_full(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.015:
                radius = 0.06
    return best_h, best_c5

def _scored_pair_refine(h, deadline, max_top=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    n_lags = 2 * n - 1
    n_top = 10
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(scores)[-n_top:]
        G = _grad_matrix_lags(h, top_lags)
        pair_scores = np.full((n, n), 1e9)
        for k in range(n_top):
            diff = G[k, :, None] - G[k, None, :]
            pair_scores = np.minimum(pair_scores, diff)
        np.fill_diagonal(pair_scores, 0.0)
        flat = pair_scores.flatten()
        order = np.argsort(flat)[::-1]
        improved = False
        for idx in order[:max_top]:
            if time.time() > deadline:
                break
            i = idx // n
            j = idx % n
            if i == j:
                continue
            d_val = flat[idx]
            if d_val <= 1e-12:
                break
            if h[i] <= 1e-12 or h[i] >= 1.0 - 1e-12:
                continue
            if h[j] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-15:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0
    return best_h, best_c5

def _equalize_top(h, deadline, K=5):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    n_lags = 2 * n - 1
    stall = 0
    while time.time() < deadline and stall < 100:
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(scores)[-K:]
        top_vals = scores[top_idx]
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-10:
            break
        max_s = float(np.max(scores))
        top_lags = np.argsort(scores)[-K:]
        G = _grad_matrix_lags(h, top_lags)
        grad_avg = np.mean(G, axis=0)
        grad_avg -= np.mean(grad_avg)
        gn = np.linalg.norm(grad_avg)
        if gn < 1e-14:
            break
        d = -grad_avg / gn
        alpha, new_c5 = _line_search(h, d, dx, 0.15)
        if new_c5 < best_c5 - 1e-15:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            d2 = 0.5 * d
            alpha2, new_c52 = _line_search(h, d2, dx, 0.07)
            if new_c52 < best_c5 - 1e-15:
                h = h + alpha2 * d2
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _softmax_direction(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    for r in range(n_lags):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _grad_descent(h, deadline, tau_list=(50, 120, 300), lr_init=0.03, mom_decay=0.85):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    ti = 0
    while time.time() < deadline:
        if stall > 80:
            break
        tau = tau_list[ti % len(tau_list)]
        d = _softmax_direction(h, dx, tau)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            ti += 1
            continue
        d = d / dn
        mom = mom_decay * mom + (1.0 - mom_decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.5)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.07, 0.4)
        else:
            stall += 1
            lr *= 0.82
            if stall % 30 == 0:
                ti += 1
                mom = np.zeros(n)
    return best_h, best_c5

def _structured_search(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    trust = 0.1
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        directions = []
        for tau in [30, 80, 300, 800]:
            d = _softmax_direction(h, dx, tau)
            dn = np.linalg.norm(d)
            if dn > 1e-13:
                directions.append(d / dn)
        for _ in range(8):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                directions.append(d)
        for _ in range(4):
            freq = rng.uniform(0.2, 5.0)
            phase = rng.uniform(0, 6.28)
            d = np.sin(np.arange(n) * freq + phase)
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn > 1e-13:
                directions.append(d / dn)
        improved = False
        for d in directions:
            if time.time() > deadline:
                break
            alpha, new_c5 = _line_search(h, trust * d, dx, trust)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * trust * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if improved:
            trust = min(trust * 1.05, 0.22)
            stall = 0
        else:
            stall += 1
            trust *= 0.82
            if trust < 0.005:
                trust = 0.04
    return best_h, best_c5

def _full_pair_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        if stall > 15:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Gradient warmup (4%)
    d1 = t0 + budget_s * 0.04
    h1, c5_1 = _grad_descent(h, d1, tau_list=(60, 150, 400), lr_init=0.025, mom_decay=0.84)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP local search (20%)
    d2 = t0 + budget_s * 0.24
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Scored pair refinement (12%)
    d3 = t0 + budget_s * 0.36
    h3, c5_3 = _scored_pair_refine(best_h, d3, max_top=100)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Equalize top lags (8%)
    d4 = t0 + budget_s * 0.44
    h4, c5_4 = _equalize_top(best_h, d4, K=5)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Structured direction search (10%)
    d5 = t0 + budget_s * 0.54
    h5, c5_5 = _structured_search(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (18%)
    d6 = t0 + budget_s * 0.72
    best_n = len(best_h)
    offsets = list(range(-20, 21, 2))
    tried = set()
    rng_mr = np.random.default_rng(seed + 1)
    rng_mr.shuffle(offsets)
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 180:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + max(2, budget_s * 0.012))
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1)
        rem = d6 - time.time()
        if rem > 4 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d6, time.time() + max(2, budget_s * 0.008))
            h_r, c5_r = _scored_pair_refine(h_r, sub_dl2, max_top=60)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Perturbation + re-optimization (10%)
    d7 = t0 + budget_s * 0.82
    scale = 0.04
    for attempt in range(200):
        if time.time() > d7:
            break
        if best_c5 < 0.3804:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + max(2, budget_s * 0.008))
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08)
        rem = d7 - time.time()
        if rem > 3 and c5_t < best_c5 + 0.001:
            h_t, c5_t = _scored_pair_refine(h_t, min(d7, time.time() + 3), max_top=50)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.15, scale * 1.1)

    # Phase 8: Full pair refinement (7%)
    d8 = t0 + budget_s * 0.89
    h_p, c5_p = _full_pair_refine(best_h, d8)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 9: Final scored pair + equalize (6%)
    d9 = t0 + budget_s * 0.95
    h9, c5_9 = _scored_pair_refine(best_h, d9, max_top=120)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()
    d10 = t0 + budget_s * 0.97
    h10, c5_10 = _equalize_top(best_h, d10, K=4)
    if c5_10 < best_c5:
        best_c5 = c5_10
        best_h = h10.copy()

    # Final LP polish
    d11 = t0 + budget_s * 0.995
    h_f, c5_f = _lp_loop(best_h, d11, radius_init=0.03)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
