```python
import numpy as np
from scipy.optimize import minimize
import time


def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _projectboxed(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def _weighted_corr_grad(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k] * dx
        if d == 0:
            g += wk * (1.0 - 2.0 * h)
        elif d > 0:
            g[d:n] += wk * (1.0 - h[0:n - d])
            g[0:n - d] += -wk * h[d:n]
        else:
            dp = -d
            g[0:n - dp] += wk * (1.0 - h[dp:n])
            g[dp:n] += -wk * h[0:n - dp]
    return g


def _lse_func(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = (c - m) / tau
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    val = m + tau * np.log(Z)
    w = exp_s / Z
    grad = _weighted_corr_grad(h, w, dx)
    return val, grad


def _slsqp_lse(h0, tau, maxiter=200, ftol=1e-12):
    n = len(h0)
    target = n / 2.0

    def obj(h):
        h_clip = np.clip(h, 0.0, 1.0)
        v, g = _lse_func(h_clip, tau)
        return v

    def jac(h):
        h_clip = np.clip(h, 0.0, 1.0)
        _, g = _lse_func(h_clip, tau)
        return g

    def eq_fun(h):
        return h.sum() - target

    def eq_jac(h):
        return np.ones(len(h))

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _projectboxed(res.x, target)
    except Exception:
        pass
    return h0


def _pg_step(h, tau, lr):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = (c - m) / tau
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    grad = _weighted_corr_grad(h, w, dx)
    grad -= grad.mean()
    return _projectboxed(h - lr * grad, target)


def _pg_anneal(h, deadline, tau_start=0.08, tau_end=0.004, lr_start=0.02, n_iter=2000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    lr = lr_start
    stall = 0
    n_actual = min(n_iter, 5000)
    for i in range(n_actual):
        if time.time() > deadline:
            break
        frac = i / max(n_actual - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        h_new = _pg_step(h, tau, lr)
        c5_new = _c5_exact(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.03, 0.06)
        elif c5_new < _c5_exact(h) - 1e-14:
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 0.06)
        else:
            stall += 1
            lr = max(lr * 0.85, 0.001)
            if stall > 50:
                break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _projectboxed(h_new, n_new / 2.0)


def _gen_starts(n, seed, count):
    rng = np.random.default_rng(seed)
    starts = []
    target = n / 2.0
    # Binary random
    for _ in range(count // 3):

        h = (v > 0.5).astype(float)
        h = _projectboxed(h, target)
        starts.append(h)
    # Smooth sinusoidal

        phase = rng.uniform(0, 2 * np.pi)
        v = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        v += rng.normal(0, 0.05, n)
        h = _projectboxed(v, target)
        starts.append(h)
    # Random continuous
    for _ in range(count - 2 * (count // 3)):
        v = rng.random(n) * 0.5 + 0.25
        h = _projectboxed(v, target)
        starts.append(h)
    return starts


def _active_slsqp(h, active_frac=0.3, min_count=10, maxiter=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()
    threshold = max_c * active_frac
    active = np.where(corr >= threshold)[0]
    if len(active) < min_count:
        k = min(min_count, len(corr))
        active = np.argsort(corr)[-k:]

    def obj(h):
        return h[0]

    def jac(h):
        g = np.zeros(len(h))
        g[0] = 1.0
        return g

    ineq_list = []
    for idx in active:
        d = int(idx) - (n - 1)

        def make_ineq(dd):
            def f(h):
                hh = h[1:]
                c = np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / len(hh))
                return h[0] - c[dd]
            def fdj(h):
                hh = h[1:]
                g = np.zeros(len(h))
                g[0] = 1.0
                dx_local = 2.0 / len(hh)
                if dd == 0:
                    g[1:] -= dx_local * (1.0 - 2.0 * hh)
                elif dd > 0:
                    g[1 + dd:] -= dx_local * (1.0 - hh[:len(hh) - dd])
                    g[1:1 + len(hh) - dd] += dx_local * hh[dd:]
                else:
                    dp = -dd
                    g[1:1 + len(hh) - dp] -= dx_local * (1.0 - hh[dp:])
                    g[1 + dp:] += dx_local * hh[:len(hh) - dp]
                return g
            return f, fdj

        f, fdj = make_ineq(d)
        ineq_list.append({'type': 'ineq', 'fun': f, 'jac': fdj})

    def eq_fun(h):
        return h[1:].sum() - target

    def eq_jac(h):
        g = np.zeros(len(h))
        g[1:] = 1.0
        return g

    max_c = float(corr.max())
    x0 = np.zeros(n + 1)
    x0[0] = max_c * 1.001 + 0.01
    x0[1:] = h

    constraints = ineq_list + [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(obj, x0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _projectboxed(res.x[1:], target)
            return h_new
    except Exception:
        pass
    return h

        hg, c5g = _pg_anneal(best_h, d2, tau_start=0.06, tau_end=0.003,
                             lr_start=0.015, n_iter=1500)
        if c5g < best_c5:
            best_c5 = c5g
            best_h = hg.copy()

    # Phase 3: Multi-start with diverse initializations (25% budget)
    d3 = t0 + budget_s * 0.65
    starts = _gen_starts(n, seed, 8)
    # Also add perturbed best
    for sc in [0.03, 0.06, 0.10]:
        h_p = best_h + rng.normal(0, sc, n)
        starts.append(_projectboxed(h_p, target))
    # Mirror
    starts.append(_projectboxed(best_h[::-1], target))

    for s_h in starts:
        if time.time() > d3:
            break
        h_try = s_h.copy()
        for tau in [0.05, 0.03, 0.015, 0.008]:
            if time.time() > d3:
                break
            h_try = _slsqp_lse(h_try, tau, maxiter=80, ftol=1e-10)
        c5_try = _c5_exact(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Active-set SLSQP refinement (15% budget)
    d4 = t0 + budget_s * 0.80
    for frac in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
        if time.time() > d4:
            break
        h_new = _active_slsqp(best_h, active_frac=frac, min_count=8, maxiter=200)
        c5_new = _c5_exact(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Resolution search (10% budget)
    d5 = t0 + budget_s * 0.90
    for dn in [2, -2, 4, -4, 6, -6, 8, 16, -8]:
        if time.time() > d5:
            break
        n_try = n + dn
        if n_try < 32 or n_try > 256:
            continue
        h_r = _resample(best_h, n_try)
        # Quick LSE at new resolution
        for tau in [0.04, 0.02, 0.01]:
            if time.time() > d5:
                break
            h_r = _slsqp_lse(h_r, tau, maxiter=60, ftol=1e-10)
        c5_r = _c5_exact(h_r)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            n = n_try

    # Phase 6: Final polish with tight LSE + PG (10% budget)
    d6 = deadline - 3.0
    if time.time() < d6:
        for tau in [0.005, 0.003, 0.002]:
            if time.time() > d6:
                break
            h_f = _slsqp_lse(best_h, tau, maxiter=300, ftol=1e-13)
            c5_f = _c5_exact(h_f)
            if c5_f < best_c5 - 1e-14:
                best_c5 = c5_f
                best_h = h_f.copy()
        if time.time() < d6:
            hg2, c5g2 = _pg_anneal(best_h, d6, tau_start=0.004, tau_end=0.001,
                                   lr_start=0.005, n_iter=800)
            if c5g2 < best_c5:
                best_c5 = c5g2
                best_h = hg2.copy()

    # Final validation
    best_h = _projectboxed(best_h, len(best_h) / 2.0)
    final_c5 = _c5_exact(best_h)
    return best_h, final_c5, len(best_h)
```
