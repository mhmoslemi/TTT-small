def _project_box_sum(v, target):

    if d_lo2 <= d_hi2:

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (v < 1.0) if diff > 0 else (v > 0.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _slope_vec(k, h):
    n = len(h)
    L = 2 * n - 1
    s = np.zeros(L)
    d_lo = max(-(n - 1), -k)
    d_hi = min(n - 1, n - 1 - k)
    if d_lo <= d_hi:
        dr = np.arange(d_lo, d_hi + 1)
        s[n - 1 + dr] += (1.0 - h[k + dr])
    d_lo2 = max(-(n - 1), k - (n - 1))
    d_hi2 = min(n - 1, k)
    if d_lo2 <= d_hi2:
        dr2 = np.arange(d_lo2, d_hi2 + 1)
        s[n - 1 + dr2] -= h[k - dr2]
    return s

def _line_minimize(M, S, t_lo, t_hi):
    if t_hi - t_lo < 1e-15:
        return 0.0, float(np.max(M))
    lo, hi = t_lo, t_hi
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = M + S * mid
        f_mid = float(np.max(vals))
        active = vals >= f_mid - 1e-13
        if not np.any(active):
            active = vals >= f_mid - 1e-10
        b_act = S[active]
        if np.min(b_act) <= 0.0 and np.max(b_act) >= 0.0:
            return mid, f_mid
        elif np.min(b_act) > 0.0:
            hi = mid
        else:
            lo = mid
    mid = 0.5 * (lo + hi)
    return mid, float(np.max(M + S * mid))

def _pair_step(h, i, j, M):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-14:
        return None, None
    S = _slope_vec(i, h) - _slope_vec(j, h)
    t_star, f_star = _line_minimize(M, S, t_lo, t_hi)
    curr_max = float(np.max(M))
    if f_star < curr_max - 1e-12:
        new_h = h.copy()
        new_h[i] = min(max(h[i] + t_star, 0.0), 1.0)
        new_h[j] = min(max(h[j] - t_star, 0.0), 1.0)
        return new_h, f_star
    return None, None

def _random_pair_loop(h, M, rng, deadline, max_iters=2000000):
    n = len(h)
    cur_max = float(np.max(M))
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 8000:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        new_h, f_star = _pair_step(h, i, j, M)
        if new_h is not None:
            h = new_h
            M = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(np.max(M))
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_max, M

def _smart_pair_loop(h, M, rng, deadline, max_iters=1000000):
    n = len(h)
    cur_max = float(np.max(M))
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    n_1 = n - 1

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 6000:
            break

        use_random = rng.random() < 0.15
        if use_random:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
        else:
            active = M >= cur_max - 1e-10
            active_rows = np.where(active)[0]
            if len(active_rows) == 0:
                break
            if len(active_rows) > 16:
                active_rows = rng.choice(active_rows, size=16, replace=False)

            best_pair = None
            best_dd = 0.0
            n_cand = min(12, n)
            candidates = rng.choice(n, size=n_cand, replace=False)
            for ic in range(n_cand):
                i = int(candidates[ic])
                Si = _slope_vec(i, h)
                Si_act = Si[active_rows]
                for jc in range(ic + 1, n_cand):
                    j = int(candidates[jc])
                    Sj = _slope_vec(j, h)
                    Sj_act = Sj[active_rows]
                    S_diff_act = Si_act - Sj_act
                    dd = float(np.max(S_diff_act))
                    if dd < best_dd:
                        best_dd = dd
                        best_pair = (i, j)
                    S_diff_act2 = Sj_act - Si_act
                    dd2 = float(np.max(S_diff_act2))
                    if dd2 < best_dd:
                        best_dd = dd2
                        best_pair = (j, i)

            if best_pair is not None:
                i, j = best_pair
            else:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    no_improve += 1
                    continue

        new_h, f_star = _pair_step(h, i, j, M)
        if new_h is not None:
            h = new_h
            M = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(np.max(M))
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_max, M

def _three_point_step(h, i, j, k, M, rng):
    n = len(h)
    # h[i]+=a, h[j]+=b, h[k]-=a-b (sum preserved)
    # Feasible: 0<=h[i]+a<=1, 0<=h[j]+b<=1, 0<=h[k]-a+b<=1
    # This is a 2D convex minimization - use random 1D cuts
    best_h = None
    best_val = float(np.max(M))
    for _ in range(5):
        # Random direction in (a,b) space
        a_dir = rng.uniform(-1, 1)
        b_dir = rng.uniform(-1, 1)
        norm = np.sqrt(a_dir*a_dir + b_dir*b_dir)
        if norm < 1e-6:
            continue
        a_dir /= norm
        b_dir /= norm
        # Find feasible range of t: a = t*a_dir, b = t*b_dir
        # Constraints:
        # 0 <= h[i] + t*a_dir <= 1
        # 0 <= h[j] + t*b_dir <= 1
        # 0 <= h[k] - t*(a_dir - b_dir) <= 1
        t_lo, t_hi = 0.0, 0.0
        if a_dir > 0:
            t_hi = min(t_hi + 1e30, (1.0 - h[i]) / a_dir)
        elif a_dir < 0:
            t_lo = max(t_lo, -h[i] / a_dir)
        if b_dir > 0:
            t_hi = min(t_hi, (1.0 - h[j]) / b_dir)
        elif b_dir < 0:
            t_lo = max(t_lo, -h[j] / b_dir)
        c_dir = -(a_dir - b_dir)
        if c_dir > 0:
            t_hi = min(t_hi, (1.0 - h[k]) / c_dir)
        elif c_dir < 0:
            t_lo = max(t_lo, -h[k] / c_dir)
        if t_hi - t_lo < 1e-14:
            continue
        # Compute slope vector for this direction
        S = a_dir * _slope_vec(i, h) + b_dir * _slope_vec(j, h) + c_dir * _slope_vec(k, h)
        t_star, f_star = _line_minimize(M, S, t_lo, t_hi)
        if f_star < best_val - 1e-12:
            best_val = f_star
            new_h = h.copy()
            new_h[i] = min(max(h[i] + t_star * a_dir, 0.0), 1.0)
            new_h[j] = min(max(h[j] + t_star * b_dir, 0.0), 1.0)
            new_h[k] = min(max(h[k] + t_star * c_dir, 0.0), 1.0)
            best_h = new_h
    return best_h, best_val

def _three_point_loop(h, M, rng, deadline, max_iters=300000):
    n = len(h)
    cur_max = float(np.max(M))
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    idx = np.arange(n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 4000:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        k = int(rng.integers(n))
        if len(set([i, j, k])) < 3:
            continue
        new_h, f_star = _three_point_step(h, i, j, k, M, rng)
        if new_h is not None:
            h = new_h
            M = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(np.max(M))
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_max, M

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _thue_morse(n):
    h = np.zeros(n)
    for k in range(n):
        h[k] = 1.0 if bin(k).count('1') % 2 == 0 else 0.0
    return h

def _anneal_pair_loop(h, M, rng, deadline, max_iters=500000, t_start=0.05, t_end=0.001):
    n = len(h)
    cur_max = float(np.max(M))
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 10000:
            break
        temperature = t_start * (t_end / t_start) ** (it / max_iters)
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        new_h, f_star = _pair_step(h, i, j, M)
        if new_h is not None:
            h = new_h
            M = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(np.max(M))
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_h = h.copy()
            no_improve = 0
        else:
            # Simulated annealing: accept worse moves with probability
            # Pick a random small perturbation
            if rng.random() < 0.1:
                delta = rng.uniform(-0.05, 0.05)
                if 0 < h[i] + delta < 1 and 0 < h[j] - delta < 1:
                    h[i] += delta
                    h[j] -= delta
                    M = np.correlate(h, 1.0 - h, mode='full')
                    new_max = float(np.max(M))
                    if new_max - cur_max < temperature:
                        cur_max = new_max
                    else:
                        h[i] -= delta
                        h[j] += delta
                    no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
    return best_h, best_max, M

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        n0 = 128
        h0 = _thue_morse(n0)

    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    n = len(h)
    M = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: Random pair exchange (30% time)
    dl1 = t0 + budget_s * 0.30
    h1, max1, M1 = _random_pair_loop(h, M, rng, dl1)
    c5_1 = max1 * (2.0 / n) if n == len(h1) else _exact_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)
    h = h1 if c5_1 < _exact_c5(best_h) else best_h
    M = np.correlate(h, 1.0 - h, mode='full')

    # Phase 2: Smart pair exchange (25% time)
    dl2 = t0 + budget_s * 0.55
    h2, max2, M2 = _smart_pair_loop(h, M, rng, dl2)
    c5_2 = max2 * (2.0 / len(h2))
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)
    h = h2 if c5_2 < _exact_c5(best_h) else best_h
    M = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)

    # Phase 3: Three-point exchanges (15% time)
    dl3 = t0 + budget_s * 0.70
    h3, max3, M3 = _three_point_loop(h, M, rng, dl3)
    c5_3 = max3 * (2.0 / len(h3))
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)
    h = h3 if c5_3 < _exact_c5(best_h) else best_h
    M = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)

    # Phase 4: Annealing (10% time)
    dl4 = t0 + budget_s * 0.80
    h4, max4, M4 = _anneal_pair_loop(h, M, rng, dl4)
    c5_4 = max4 * (2.0 / len(h4))
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        best_n = len(h4)

    # Phase 5: Multi-resolution (12% time)
    dl5 = t0 + budget_s * 0.92
    n_tries = [48, 64, 80, 96, best_n - 8, best_n - 4, best_n + 4, best_n + 8, 144, 160, 192]
    n_tries = sorted(set([x for x in n_tries if 32 <= x <= 256]))
    n_tries.sort(key=lambda x: abs(x - best_n))
    per_time = budget_s * 0.04
    for n_try in n_tries:
        if time.time() > dl5:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        M_r = np.correlate(h_r, 1.0 - h_r, mode='full')
        dl_r = min(dl5, time.time() + per_time)
        h_r, max_r, M_r = _random_pair_loop(h_r, M_r, rng, dl_r, max_iters=200000)
        c5_r = max_r * (2.0 / n_try)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final polish (8% time)
    dl6 = t0 + budget_s * 0.98
    h = best_h.copy()
    n = len(h)
    M = np.correlate(h, 1.0 - h, mode='full')
    h6, max6, M6 = _smart_pair_loop(h, M, rng, dl6, max_iters=500000)
    c5_6 = max6 * (2.0 / len(h6))
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)

    # Final validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    return best_h, best_c5, best_n
