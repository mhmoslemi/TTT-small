def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_exact(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_rows(h, indices):
    """Build gradient rows for specified correlation output positions."""
    n = len(h)
    dx = 2.0 / n
    m = len(indices)
    G = np.zeros((m, n))
    for k in range(m):
        r = indices[k]
        d = r - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                G[k, d:d+m2] += dx * (1.0 - h[:m2])
                G[k, :m2] -= dx * h[d:d+m2]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                G[k, :m2] += dx * (1.0 - h[g:g+m2])
                G[k, g:g+m2] -= dx * h[:m2]
    return G

def active_qp_direction(h, scores, active_idx, trust, ridge=1e-3, rough=0.0):
    """Find direction using active constraint QP."""
    n = len(h)
    m = len(active_idx)
    if m == 0:
        return None
    
    G = grad_rows(h, active_idx)
    s_active = scores[active_idx]
    target_level = np.min(s_active) - 0.001 * (np.max(s_active) - np.min(s_active))
    residual = target_level - s_active  # should be <= 0 for all
    
    # Weights: prioritize higher-scoring active constraints
    gaps = np.max(s_active) - s_active
    max_gap = np.max(gaps)
    if max_gap < 1e-15:
        q = np.ones(m)
    else:
        q = 1.0 + 50.0 * (gaps / max_gap)**2
        q = q / np.sum(q) * m  # normalize
    
    # Build H = G^T diag(q) G + ridge*I + roughness
    H = (G * q[:, None]).T @ G
    H += ridge * np.eye(n)
    
    if rough > 0:
        for i in range(n):
            H[i, i] += 2.0 * rough
            if i > 0:
                H[i, i-1] -= rough
                H[i-1, i] -= rough
            if i < n-1:
                H[i, i+1] -= rough
                H[i+1, i] -= rough
    
    # RHS: G^T q * residual
    rhs = G.T @ (q * residual)
    
    # KKT system with equality constraint sum(delta) = 0
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs_full = np.append(rhs, 0.0)
    
    try:
        sol = np.linalg.solve(KKT, rhs_full)
        delta = sol[:n]
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(KKT, rhs_full, rcond=None)[0]
            delta = sol[:n]
        except:
            return None
    
    if not np.all(np.isfinite(delta)):
        return None
    
    max_abs = np.max(np.abs(delta))
    if max_abs < 1e-14:
        return None
    
    # Normalize and trust-region
    delta = delta / max_abs
    # Now max abs is 1, trust limits the step
    return delta

def line_search(h, p, trust=0.3):
    """Exact line search along direction p from h."""
    n = len(h)
    # Compute max step before hitting bounds
    alpha_max = trust
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def obj(a):
        h_try = h + a * p
        if np.min(h_try) < -1e-10 or np.max(h_try) > 1.0 + 1e-10:
            return float('inf')
        return exact_c5(h_try)
    
    f1 = obj(c1)
    f2 = obj(c2)
    
    for _ in range(25):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = obj(opt_a)
    
    # Also check 0 and alpha_max
    f0 = exact_c5(h)
    fmax = obj(alpha_max)
    
    best_a = 0.0
    best_c5 = f0
    if opt_c5 < best_c5:
        best_a = opt_a
        best_c5 = opt_c5
    if fmax < best_c5:
        best_a = alpha_max
        best_c5 = fmax
    
    return best_a, best_c5

def weighted_subgradient(h, deadline, n_steps=300, beta_init=5.0, beta_max=200.0):
    """Weighted subgradient with adaptive beta."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps)
        beta = beta_init + (beta_max - beta_init) * frac
        
        scores = scores_exact(h)
        s_max = np.max(scores)
        # Soft-max weights
        w = np.exp(beta * (scores - s_max))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        
        G = grad_rows(h, np.arange(len(scores)))  # full gradient
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        # Line search
        alpha = 0.1

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def binary_sa(n, deadline, rng, n_iters=30000):
    """Binary simulated annealing."""
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.05
    T_min = 0.001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    
    return best_h, best_c5

def pair_polish(h, deadline, max_passes=8):
    """Polish by pairwise swaps."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 24)
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(0, n, step):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Ternary search
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(7):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0, 1)
                        f1 = exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0, 1)
                        f2 = exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5

def smooth_random_init(n, rng, scale=0.1):
    # Low-frequency smooth random
    x = np.arange(n) / n
    h = np.zeros(n)
    for k in range(1, 6):

        phase = rng.uniform(0, 2*np.pi)

    return project(h, n / 2.0)

def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_exact(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_rows(h, indices):
    n = len(h)
    dx = 2.0 / n
    m = len(indices)
    G = np.zeros((m, n))
    for k in range(m):
        r = indices[k]
        d = r - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                G[k, d:d+m2] += dx * (1.0 - h[:m2])
                G[k, :m2] -= dx * h[d:d+m2]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                G[k, :m2] += dx * (1.0 - h[g:g+m2])
                G[k, g:g+m2] -= dx * h[:m2]
    return G

    n = len(h)
    m = len(active_idx)
    if m == 0:
        return None
    G = grad_rows(h, active_idx)
    s_a = scores[active_idx]
    c_min_a = np.min(s_a)
    spread = np.max(s_a) - c_min_a
    target_level = c_min_a - 0.001 * max(spread, 1e-8)

    gaps = np.max(s_a) - s_a

    if max_gap < 1e-15:

    else:
        q = 1.0 + 80.0 * (gaps / max_gap) ** 2

    H = (G * q[:, None]).T @ G
    H += ridge * np.eye(n)
    if rough > 0:
        for i in range(n):
            H[i, i] += 2.0 * rough
            if i > 0:
                H[i, i-1] -= rough
                H[i-1, i] -= rough
            if i < n - 1:
                H[i, i+1] -= rough
                H[i+1, i] -= rough

    KKT = np.zeros((n + 1, n + 1))

    KKT[:n, n] = 1.0

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def binary_sa(n, deadline, rng, n_iters):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.04
    T_min = 0.0008
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(7):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0, 1)
                        f1 = exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0, 1)
                        f2 = exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = exact_c5(h_start)
    best_n = n0
    deadline_all = t0 + budget_s * 0.99

    def still_going():
        return time.time() < deadline_all

    # Phase 1: Active QP variant 0 on parent
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = local_optimize(h_start, dl1, 16, 0.22, 3000, 0)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1.copy(), len(h1)

    # Phase 2: LP on parent
    if still_going():
        dl2 = t0 + budget_s * 0.20
        h2, c5_2 = lp_local(best_h, dl2, 12, 0.18, 1500)
        if c5_2 < best_c5:
            best_c5, best_h, best_n = c5_2, h2.copy(), len(h2)

    # Phase 3: Weighted subgradient
    if still_going():
        dl3 = t0 + budget_s * 0.27
        h3, c5_3 = weighted_subgrad(best_h, dl3, 150, 5.0, 180.0)
        if c5_3 < best_c5:
            best_c5, best_h, best_n = c5_3, h3.copy(), len(h3)

    # Phase 4: Active QP variant 1 (tolerance-based active set)
    if still_going():
        dl4 = t0 + budget_s * 0.35
        h4, c5_4 = local_optimize(best_h, dl4, 20, 0.15, 2000, 1)
        if c5_4 < best_c5:
            best_c5, best_h, best_n = c5_4, h4.copy(), len(h4)

    # Phase 5: Multi-resolution diverse search
    dl5 = t0 + budget_s * 0.62
    resolutions = [n0, n0 - 16, n0 + 16, 96, 112, 80, 128, 144, 100, 120]
    resolutions = sorted(set([r for r in resolutions if 48 <= r <= 192]))
    alphas_qp = [0.618034, 0.414214, 0.707107, 0.381966]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl5:
            break
        h_init = resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + budget_s * 0.035)
        h_try, c5_try = local_optimize(h_init, sub_dl, 14, 0.18, 1800, ri % 3)
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_try.copy(), n_try
        if time.time() < dl5:
            alpha = alphas_qp[ri % 4]
            h_qp = quasi_periodic_init(n_try, alpha)
            h_qp = project(h_qp + rng.normal(0, 0.06, n_try), n_try / 2.0)
            sub_dl2 = min(dl5, time.time() + budget_s * 0.025)
            h_qp2, c5_qp2 = local_optimize(h_qp, sub_dl2, 12, 0.15, 1200, (ri + 1) % 3)
            if c5_qp2 < best_c5:
                best_c5, best_h, best_n = c5_qp2, h_qp2.copy(), n_try
                sub_dl3 = min(dl5, time.time() + budget_s * 0.015)
                h_q3, c5_q3 = lp_local(h_qp2, sub_dl3, 8, 0.1, 600)
                if c5_q3 < best_c5:
                    best_c5, best_h, best_n = c5_q3, h_q3.copy(), n_try

    # Phase 6: Binary SA
    if time.time() - t0 < budget_s * 0.72:
        dl6 = t0 + budget_s * 0.79
        for n_bin in [96, 128, 112]:
            if time.time() > dl6:
                break
            sub_dl = min(dl6, time.time() + budget_s * 0.03)
            h_b, c5_b = binary_sa(n_bin, sub_dl, rng, 25000)
            if c5_b < best_c5:
                best_c5, best_h, best_n = c5_b, h_b.copy(), n_bin
                sub_dl2 = min(dl6, time.time() + budget_s * 0.02)
                h_br, c5_br = local_optimize(h_b, sub_dl2, 10, 0.1, 700, 0)
                if c5_br < best_c5:
                    best_c5, best_h, best_n = c5_br, h_br.copy(), n_bin

    # Phase 7: Active QP variant 2 (tight, low ridge)
    if time.time() - t0 < budget_s * 0.84:
        dl7 = t0 + budget_s * 0.89
        h7, c5_7 = local_optimize(best_h, dl7, 10, 0.10, 2000, 2)
        if c5_7 < best_c5:
            best_c5, best_h, best_n = c5_7, h7.copy(), len(h7)

    # Phase 8: Perturbation restarts
    if time.time() - t0 < budget_s * 0.93:
        dl8 = t0 + budget_s * 0.95
        scale = 0.025
        for attempt in range(250):
            if time.time() > dl8:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = project(h_pert, best_n / 2.0)
            sub_dl = min(dl8, time.time() + budget_s * 0.01)
            h_try, c5_try = local_optimize(h_pert, sub_dl, 10, 0.06, 150, attempt % 3)
            if c5_try < best_c5 - 1e-12:
                best_c5, best_h, best_n = c5_try, h_try.copy(), best_n
                scale = max(0.004, scale * 0.88)
            else:
                scale = min(0.07, scale * 1.08)

    # Phase 9: Pair polish
    if time.time() - t0 < budget_s * 0.97:
        dl9 = t0 + budget_s * 0.985
        hp, c5p = pair_polish(best_h, dl9, 6)
        if c5p < best_c5:
            best_c5, best_h, best_n = c5p, hp.copy(), len(hp)

    # Final: tiny LP refinement
    if time.time() - t0 < budget_s * 0.995:
        h_fin, c5_fin = lp_local(best_h, t0 + budget_s * 0.998, 5, 0.02, 300)
        if c5_fin < best_c5:
            best_c5, best_h, best_n = c5_fin, h_fin.copy(), len(h_fin)

    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
