def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum exactly
    residual = target - np.sum(p)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - p)[:int(np.ceil(abs(residual)))]
            for i in idx:
                add = min(residual, 1.0 - p[i])
                p[i] += add
                residual -= add
        else:
            idx = np.argsort(p)[:int(np.ceil(abs(residual)))]
            for i in idx:
                sub = min(-residual, p[i])
                p[i] -= sub
                residual += sub
    return p

def _resample(h, n_new):
