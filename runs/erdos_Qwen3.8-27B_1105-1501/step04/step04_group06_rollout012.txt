    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _epi_eq_f, 'jac': _epi_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _epi_ineq_f, 'jac': _epi_ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_grad, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)

def _binarize(h, n):
    """Push h towards the nearest binary point with the right sum."""
    h = h.copy()
    # Sort by value, set top n/2 to 1, rest to 0
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin
