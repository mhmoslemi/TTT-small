        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

    n = len(h)
    scores = np.zeros(2*n - 1)
    for r in range(2*n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            scores[r] = float(np.dot(h[d:], 1.0 - h[:m]))
        else:
            s = -d
            m = n - s
            scores[r] = float(np.dot(h[:m], 1.0 - h[s:]))
    return scores

def _raw_jacobian(h):
    n = len(h)
    m = 2*n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            G[r, d:] += 1.0
            G[r, d:] -= h[:cnt]
            G[r, :cnt] -= h[d:]
        else:
            s = -d
            cnt = n - s
            G[r, :cnt] += 1.0
            G[r, :cnt] -= h[s:s+cnt]
            G[r, s:] -= h[:cnt]
    return G

_slsqp_target = 0.0

    return x[_slsqp_n]  # minimize T

def _slsqp_obj_grad(x):
    g = np.zeros(_slsqp_n + 1)
    g[_slsqp_n] = 1.0
    return g

    return np.array([np.sum(x[:_slsqp_n]) - _slsqp_target])

    G = np.zeros((1, _slsqp_n + 1))
    G[0, :_slsqp_n] = 1.0
    return G

    h = x[:_slsqp_n]
    T = x[_slsqp_n]
    scores = _raw_scores(h)
    return T - scores  # we want scores <= T, so T - scores >= 0

    G = _raw_jacobian(h)
    m = 2*_slsqp_n - 1
    jac = np.zeros((m, _slsqp_n + 1))
    jac[:, :_slsqp_n] = -G  # d/dh of (T - S_r) = -G_r
    jac[:, _slsqp_n] = 1.0  # d/dT of (T - S_r) = 1

    T0 = cur_raw + 1e-4 + 1e-6
    x0 = np.concatenate([h0, [T0]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    
    if maxiter is None:
        if n <= 70:
            maxiter = 60
        elif n <= 120:
            maxiter = 40
        else:
            maxiter = 25
    
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac},
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac}
    ]
    
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], _slsqp_target)
            c5 = _exact_c5(h_new)
            return h_new, c5
    except Exception:
        pass
    return h0, _exact_c5(h0)

def _resample_area(h_old, n_new):
    n = len(h_old)
    if n_new == n:
        return h_old.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    # For each new cell, compute the weighted average of old cells that overlap
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        # Find overlapping old cells
        i_start = int(np.floor(x_start / dx_old))
        i_end = int(np.ceil(x_end / dx_old))
        i_start = max(0, i_start)
        i_end = min(n - 1, i_end)
        mass = 0.0

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0.0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _raw_scores(h):
    n = len(h)
    m = 2 * n - 1
    scores = np.zeros(m)
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            scores[r] = float(np.dot(h[d:], 1.0 - h[:cnt]))
        else:
            s = -d
            cnt = n - s
            scores[r] = float(np.dot(h[:cnt], 1.0 - h[s:]))
    return scores

def _raw_jac(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            G[r, d:] += 1.0
            G[r, d:] -= h[:cnt]
            G[r, :cnt] -= h[d:]
        else:
            s = -d
            cnt = n - s
            G[r, :cnt] += 1.0
            G[r, :cnt] -= h[s:s + cnt]
            G[r, s:] -= h[:cnt]
    return G

_g_n = 0
_g_target = 0.0

def _epi_obj(x):
    return x[_g_n]

def _epi_obj_grad(x):
    g = np.zeros(_g_n + 1)
    g[_g_n] = 1.0
    return g

def _epi_eq(x):
    return np.array([np.sum(x[:_g_n]) - _g_target])

def _epi_eq_jac(x):
    J = np.zeros((1, _g_n + 1))
    J[0, :_g_n] = 1.0
    return J

def _epi_ineq(x):
    h = x[:_g_n]
    T = x[_g_n]
    scores = _raw_scores(h)
    return T - scores

def _epi_ineq_jac(x):
    h = x[:_g_n]
    G = _raw_jac(h)
    m = 2 * _g_n - 1
    J = np.zeros((m, _g_n + 1))
    J[:, :_g_n] = -G
    J[:, _g_n] = 1.0
    return J

def _slsqp_epigraph(h0, deadline, maxiter=None):
    global _g_n, _g_target
    n = len(h0)
    if n > 220:
        return h0, _exact_c5(h0)
    if time.time() >= deadline:
        return h0, _exact_c5(h0)
    _g_n = n
    _g_target = n * 0.5
    h0 = _project(h0.copy(), _g_target)
    cur_raw = float(np.max(_raw_scores(h0)))
    T0 = cur_raw + 1e-4
    x0 = np.concatenate([h0, [T0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    if maxiter is None:
        if n <= 60:
            maxiter = 100
        elif n <= 100:
            maxiter = 60
        elif n <= 150:
            maxiter = 40
        else:
            maxiter = 25
    cons = [
        {'type': 'eq', 'fun': _epi_eq, 'jac': _epi_eq_jac},
        {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac}
    ]
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    try:
        res = minimize(_epi_obj, x0, method='SLSQP', jac=_epi_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], _g_target)
            c5 = _exact_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
    except Exception:
        pass
    return best_h, best_c5

def _resample_area(h_old, n_new):
    n = len(h_old)
    if n_new == n:
        return h_old.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        i_start = max(0, int(np.floor(x_start / dx_old)))
        i_end = min(n - 1, int(np.ceil(x_end / dx_old)))
        mass = 0.0
        for i in range(i_start, i_end + 1):
            old_start = i * dx_old
            old_end = (i + 1) * dx_old
            overlap = min(x_end, old_end) - max(x_start, old_start)
            if overlap > 0:
                mass += h_old[i] * overlap
        h_new[j] = mass / dx_new
    return _project(h_new, n_new * 0.5)

def _grad_descent(h, deadline, tau=50.0, lr=0.015, beta=0.88):
    n = len(h)
    target = n * 0.5
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    mom = np.zeros(n)
    dx = 2.0 / n
    while time.time() < deadline:
        scores = _raw_scores(h)
        max_s = float(np.max(scores))
        shifted = np.clip(tau * (scores - max_s), -50.0, 0.0)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        G = _raw_jac(h)
        grad = (G.T @ w) * dx
        mom = beta * mom + (1.0 - beta) * grad
        h_new = _project(h - lr * mom, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            lr = min(lr * 1.04, 0.08)
            tau = min(tau * 1.015, 400.0)
        else:
            lr *= 0.92
            if lr < 5e-5:
                break
    return best_h, best_c5

def _active_lp_direction(h, scores, K, radius):
    n = len(h)
    top_K = np.argsort(scores)[::-1][:K]
    G = _raw_jac(h)
    G_top = G[top_K]
    s_top = scores[top_K]
    m = len(top_K)
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G_top
    A_ub[:, n] = -1.0
    b_ub = -s_top
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    from scipy.optimize import linprog
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max):
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    n_grid = 25
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 2)]
    hi_t = ts[min(n_grid - 1, idx + 2)]
    if hi_t <= lo_t:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = _exact_c5(h + c1 * p) if (h + c1 * p).min() >= -1e-12 else c5_cur + 1.0
    f2 = _exact_c5(h + c2 * p) if (h + c2 * p).min() >= -1e-12 else c5_cur + 1.0
    for _ in range(30):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _exact_c5(h + c1 * p) if (h + c1 * p).min() >= -1e-12 else c5_cur + 1.0
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _exact_c5(h + c2 * p) if (h + c2 * p).min() >= -1e-12 else c5_cur + 1.0
    opt = (lo_t + hi_t) * 0.5
    f_opt = _exact_c5(h + opt * p) if (h + opt * p).min() >= -1e-12 else c5_cur + 1.0
    if f_opt < best_val:
        best_t, best_val = opt, f_opt
    return best_t, best_val

def _lp_search(h, deadline, radius=0.15, K=25):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = _raw_scores(h)
        p = _active_lp_direction(h, scores, min(K, 2 * n - 1), radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.65
            if radius < 1e-7:
                break
            continue
        alpha_max = radius
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        alpha, new_c5 = _line_search(h, p, best_c5, alpha_max)
        if new_c5 < best_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            best_c5 = _exact_c5(h)
            best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=80):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() >= deadline:
            break
        improved = False
        rng = np.random.default_rng(_pass * 7919 + 13)
        if n > 50:
            idx_a = rng.integers(0, n, size=min(n, 35))
            idx_b = rng.integers(0, n, size=min(n, 35))
            pairs = set()
            for a in idx_a:
                for b in idx_b:
                    if a != b:
                        pairs.add((int(min(a, b)), int(max(a, b))))
            pair_list = list(pairs)
            rng.shuffle(pair_list)
        else:
            pair_list = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for (i, j) in pair_list:
            if time.time() >= deadline:
                break
            hi = h[i]
            hj = h[j]
            if hi <= 1e-9 or hi >= 1.0 - 1e-9 or hj <= 1e-9 or hj >= 1.0 - 1e-9:
                continue
            for di, dj in [(1.0, -1.0), (-1.0, 1.0)]:
                amax = 1e18
                if di > 0:
                    amax = min(amax, (1.0 - hi) / di)
                else:
                    amax = min(amax, hi / (-di))
                if dj > 0:
                    amax = min(amax, (1.0 - hj) / dj)
                else:
                    amax = min(amax, hj / (-dj))
                if amax < 1e-12:
                    continue
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    alpha = amax * frac
                    h_t = h.copy()
                    h_t[i] += di * alpha
                    h_t[j] += dj * alpha
                    if h_t[i] < -1e-9 or h_t[i] > 1.0 + 1e-9 or h_t[j] < -1e-9 or h_t[j] > 1.0 + 1e-9:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _make_transformed(h0, transform, seed=0):
    n = len(h0)
    rng = np.random.default_rng(seed)
    if transform == 'identity':
        return h0.copy()
    elif transform == 'reversed':
        return h0[::-1].copy()
    elif transform == 'relaxed':
        q = 0.3
        return _project(h0 - q * (h0 - 0.5), n * 0.5)
    elif transform == 'sharpened':
        v = np.where(h0 >= 0.5, h0 + 0.3 * (1.0 - h0), h0 - 0.3 * h0)
        return _project(v, n * 0.5)
    elif transform == 'binarized':
        v = np.where(h0 >= 0.5, 1.0, 0.0)
        return _project(v, n * 0.5)
    elif transform == 'noise':
        return _project(h0 + 0.05 * rng.normal(0, 1, n), n * 0.5)
    elif transform == 'roll':
        k = n // 4
        return _project(np.roll(h0, k), n * 0.5)
    else:
        return h0.copy()

    t0 = time.time()
    deadline = t0 + budget_s * 0.96

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: SLSQP at native resolution
    d1 = min(deadline, t0 + budget_s * 0.06)
    hh, cc = _slsqp_epigraph(best_h, d1, maxiter=60)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 2: Gradient descent polish
    d2 = min(deadline, t0 + budget_s * 0.14)
    hh, cc = _grad_descent(best_h, d2, tau=40.0, lr=0.012, beta=0.87)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 3: LP local search
    d3 = min(deadline, t0 + budget_s * 0.24)
    hh, cc = _lp_search(best_h, d3, radius=0.15, K=20)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 4: Multi-resolution search
    d4 = min(deadline, t0 + budget_s * 0.62)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, 16, -16]
    tried = set()
    time_per_res = budget_s * 0.025
    for dn in offsets:
        if time.time() >= d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 180:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample_area(best_h, n_try)
        sub_dl = min(d4, time.time() + time_per_res)
        h_r, c5_r = _slsqp_epigraph(h_r, sub_dl, maxiter=30)
        if time.time() < sub_dl:
            h_r, c5_r = _grad_descent(h_r, sub_dl, tau=30.0, lr=0.01, beta=0.85)
        if time.time() < d4 and (d4 - time.time()) > 1.0:
            sub_dl2 = min(d4, time.time() + time_per_res * 0.4)
            h_r, c5_r = _lp_search(h_r, sub_dl2, radius=0.1, K=15)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Intensive optimization at best resolution
    d5 = min(deadline, t0 + budget_s * 0.78)
    hh, cc = _slsqp_epigraph(best_h, d5, maxiter=50)
    if cc < best_c5:
        best_c5, best_h = cc, hh
    if time.time() < d5:
        hh, cc = _lp_search(best_h, d5, radius=0.08, K=18)
        if cc < best_c5:
            best_c5, best_h = cc, hh
    if time.time() < d5:
        hh, cc = _grad_descent(best_h, d5, tau=80.0, lr=0.006, beta=0.92)
        if cc < best_c5:
            best_c5, best_h = cc, hh

    # Phase 6: Multi-start perturbation
    d6 = min(deadline, t0 + budget_s * 0.88)
    transforms = ['noise', 'sharpened', 'relaxed', 'reversed', 'roll']
    for ti, tr in enumerate(transforms):
        if time.time() >= d6:
            break
        if best_c5 < 0.3806:
            break
        h_t = _make_transformed(best_h, tr, seed=seed + ti * 100)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        hh, cc = _slsqp_epigraph(h_t, sub_dl, maxiter=25)
        if cc < best_c5:
            best_c5, best_h = cc, hh
        if time.time() < sub_dl:
            hh, cc = _lp_search(hh, sub_dl, radius=0.08, K=12)
            if cc < best_c5:
                best_c5, best_h = cc, hh

    # Phase 7: Pair polishing
    d7 = min(deadline, t0 + budget_s * 0.94)
    hh, cc = _pair_polish(best_h, d7, max_passes=100)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 8: Final SLSQP + LP polish
    if time.time() < deadline:
        hh, cc = _slsqp_epigraph(best_h, deadline, maxiter=30)
        if cc < best_c5:
            best_c5, best_h = cc, hh
    if time.time() < deadline:
        hh, cc = _lp_search(best_h, deadline, radius=0.03, K=10)
        if cc < best_c5:
            best_c5, best_h = cc, hh
    if time.time() < deadline:
        hh, cc = _pair_polish(best_h, deadline, max_passes=40)
        if cc < best_c5:
            best_c5, best_h = cc, hh

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    # Ensure we return the best
    out_h = best_h.copy()
    out_c5 = final_c5
    out_n = best_n
    return out_h, out_c5, out_n
