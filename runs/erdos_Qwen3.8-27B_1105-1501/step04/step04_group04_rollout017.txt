```python
def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
```

To minimize a piecewise linear convex function of one variable, we can:
1. Evaluate at τ_lo and τ_hi
2. The minimum is where the subgradient contains 0, i.e., where some slope is negative and some is positive
3. We can find this by bisection on the subgradient

The subgradient of f at τ is {slope_d : M_d(0) + slope_d * τ = f(τ)} (the set of slopes of active functions).

For bisection:
- At τ, compute f(τ) and the set of active d's
- If all active slopes are positive, decrease τ
- If all active slopes are negative, increase τ
- If there's a mix, we're at the minimum (or very close)

This is much faster than golden section! We only need O(log(precision/range)) iterations, and each iteration is O(n) to compute the max and check slopes.

Actually, even better: since f(τ) is piecewise linear convex, the minimum occurs at a "kink" where two lines intersect, or at a boundary. The kinks occur where M_d1(τ) = M_d2(τ) for some pair of shifts.

```python
def _pair_min_fast(h, i, j, tau_lo, tau_hi, n_bisect=40):
    """Minimize max_d M_d(τ) for pair swap using bisection on subgradient.
    Returns (best_c5, best_tau)."""
    n = len(h)
    dx = 2.0 / n
    oi, oj = h[i], h[j]
    
    # Current scores
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(scores)  # 2n-1
    center = n - 1
    
    # Compute slope_d for each shift d
    # slope_d = dx * [(1-h_{i+d}) * 1_{valid} - h_{i-d} * 1_{valid} - (1-h_{j+d}) * 1_{valid} + h_{j-d} * 1_{valid}]
    
    slopes = np.zeros(L)
    
    # For shift d (corresponding to index d + center in scores):
    # Term from h_i at position a=i: valid if 0 ≤ i < n and 0 ≤ i+d < n
    #   → index in scores: center + d, and we need 0 ≤ i < n and 0 ≤ i+d < n
    #   → contributes (1-h[i+d]) to slope
    # Term from h_i at position a=i-d (where a+d=i): valid if 0 ≤ i-d < n
    #   → contributes -h[i-d] to slope
    # Similarly for h_j.
    
    d = np.arange(-(n-1), n)  # shift values
    # For h_i:
    #   (1-h[i+d]) valid when 0≤i<n and 0≤i+d<n → i+d in [0,n) → d in [-i, n-1-i]
    #   -h[i-d] valid when 0≤i-d<n → d in [i-n+1, i]
    
    # Let's compute for each d:
    valid_id = (d >= -i) & (d <= n-1-i)  # i+d valid
    valid_id2 = (d >= i-n+1) & (d <= i)   # i-d valid (i.e., h[i-d] exists)
    
    # (1-h[i+d]) value:
    h_i_plus_d = np.where(valid_id, 1.0 - h[np.clip(i+d, 0, n-1)], 0.0)
    # -h[i-d] value:
    h_i_minus_d = np.where(valid_id2, -h[np.clip(i-d, 0, n-1)], 0.0)
    
    # For h_j (with -δ change):
    valid_jd = (d >= -j) & (d <= n-1-j)
    valid_jd2 = (d >= j-n+1) & (d <= j)
    
    h_j_plus_d = np.where(valid_jd, -(1.0 - h[np.clip(j+d, 0, n-1)]), 0.0)
    h_j_minus_d = np.where(valid_jd2, h[np.clip(j-d, 0, n-1)], 0.0)
    
    slopes = dx * (h_i_plus_d + h_i_minus_d + h_j_plus_d + h_j_minus_d)
    
    # Now minimize f(τ) = max_d (scores[d] + slopes[d] * τ) over τ ∈ [tau_lo, tau_hi]
    # Using bisection on subgradient
    
    a, b = tau_lo, tau_hi
    
    for _ in range(n_bisect):
        mid = 0.5 * (a + b)
        vals = scores + slopes * mid
        f_mid = np.max(vals)
        # Subgradient: slopes of active functions
        active = np.where(np.abs(vals - f_mid) < 1e-12)[0]
        # For safety, use a tolerance
        active = np.where(vals > f_mid - 1e-10)[0]
        if len(active) == 0:
            active = np.where(vals == np.max(vals))[0]
        
        min_slope = np.min(slopes[active])
        max_slope = np.max(slopes[active])
        
        if min_slope >= 0:
            # f is increasing at mid, go left
            b = mid
        elif max_slope <= 0:
            # f is decreasing at mid, go right
            a = mid
        else:
            # 0 is in the subgradient, we're at (or very near) the minimum
            break
    
    best_tau = 0.5 * (a + b)
    # Apply
    h[i] = oi + best_tau
    h[j] = oj - best_tau
    return _exact_c5(h), best_tau
```

```python
def _pair_swap_optimal(h, i, j, n_bisect=35):
    """Find optimal τ for h_i += τ, h_j -= τ. Returns (c5_new, τ)."""
    n = len(h)
    dx = 2.0 / n
    oi, oj = h[i], h[j]
    
    tau_lo = max(-oi, oj - 1.0)
    tau_hi = min(1.0 - oi, oj)
    if tau_hi - tau_lo < 1e-12:
        return _exact_c5(h), 0.0
    
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    d = np.arange(-(n-1), n)
    
    valid_id = (d >= -i) & (d <= n - 1 - i)
    valid_id2 = (d >= i - n + 1) & (d <= i)
    valid_jd = (d >= -j) & (d <= n - 1 - j)
    valid_jd2 = (d >= j - n + 1) & (d <= j)
    
    idx_id = np.clip(i + d, 0, n-1)
    idx_id2 = np.clip(i - d, 0, n-1)
    idx_jd = np.clip(j + d, 0, n-1)
    idx_jd2 = np.clip(j - d, 0, n-1)
    
    slopes = dx * (
        np.where(valid_id, 1.0 - h[idx_id], 0.0) +
        np.where(valid_id2, -h[idx_id2], 0.0) +
        np.where(valid_jd, -(1.0 - h[idx_jd]), 0.0) +
        np.where(valid_jd2, h[idx_jd2], 0.0)
    )
    
    # Bisection on subgradient
    a, b = tau_lo, tau_hi
    for _ in range(n_bisect):
        mid = 0.5 * (a + b)
        vals = scores + slopes * mid
        f_mid = float(np.max(vals))
        # Find active shifts (within tolerance of max)
        tol = 1e-10
        active = np.where(vals >= f_mid - tol)[0]
        if len(active) == 0:
            active = np.argmax(vals)
            active = np.array([active])
        
        s_min = float(np.min(slopes[active]))
        s_max = float(np.max(slopes[active]))
        
        if s_min >= 0:
            b = mid
        elif s_max <= 0:
            a = mid
        else:
            break
    
    best_tau = 0.5 * (a + b)
    h[i] = oi + best_tau
    h[j] = oj - best_tau
    c5_new = _exact_c5(h)
    return c5_new, best_tau
```

This is O(n) per bisection step (just vector operations) + O(n) for the initial correlate + O(n) for the final evaluation. Total: O(n * n_bisect) ≈ O(35n) per pair swap. For n=128, that's about 4500 operations, which is very fast.
