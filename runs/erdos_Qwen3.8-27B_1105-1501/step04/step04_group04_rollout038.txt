def _c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def _project(v, target):

    for _ in range(100):
        d = target - v.sum()
        if abs(d) < 1e-13: break

        cnt = m.sum()
        if cnt == 0: break

        v = np.clip(v, 0.0, 1.0)

    """Try transferring mass from j to i (or i to j). Return (new_tau, new_c5) if improved."""
    hi, hj = h[i], h[j]
    t_lo = max(-hi, hj - 1.0)
    t_hi = min(1.0 - hi, hj)
    if t_hi - t_lo < 1e-12:
        return None
    # Scan
    cur_c5 = _c5(h)
    best_t = 0.0
    best_val = cur_c5
    for t in np.linspace(t_lo, t_hi, n_pts):

            best_t = t
        h[i] = hi
        h[j] = hj
    if best_t == 0.0:
        return None
    # Refine with a few more points around best_t
    idx = (best_t - t_lo) / (t_hi - t_lo) * (n_pts - 1)

    """Convert h to bang-bang with exactly n/2 ones."""
    # Sort by value, threshold
    sorted_h = np.sort(h)[::-1]
    # We want n/2 ones, so threshold is between sorted_h[n/2-1] and sorted_h[n/2]
    # Just use 0.5 threshold first, then adjust
    bb = (h >= 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(bb.sum())
    if n_ones > n // 2:
        # Need to flip some 1s to 0s
        ones = np.where(bb == 1.0)[0]
        # Flip the ones with smallest h values
        order = np.argsort(h[ones])[::-1]  # smallest h among ones first
        for idx in order[:n_ones - n//2]:
            bb[ones[idx]] = 0.0
    elif n_ones < n // 2:
        zeros = np.where(bb == 0.0)[0]
        order = np.argsort(h[zeros])  # largest h among zeros first
        for idx in order[:n//2 - n_ones]:
            bb[zeros[idx]] = 1.0
    return bb
