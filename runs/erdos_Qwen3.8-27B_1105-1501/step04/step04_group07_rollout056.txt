def project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):

        s = np.sum(np.clip(v - mid, 0, 1))

        else: hi = mid

        x[m] += r / m.sum()
    elif r < 0:

def pair_linear_coeff(h, i, j):
    """Compute L[r] such that new_score[r] ≈ old_score[r] + alpha * L[r]"""
    n = len(h)
    d = np.arange(2*n-1) - (n-1)

    #   As first factor: alpha * (1-h[i-d]) if 0 <= i-d < n
    #   As in (1-h[k-d]): -alpha * h[i+d] if 0 <= i+d < n

    m2 = (i+d >= 0) & (i+d < n)
    a[m2] += -h[i+d[m2]]

    b = np.zeros(2*n-1)
    m3 = (j-d >= 0) & (j-d < n)
    b[m3] -= (1 - h[j-d[m3]])
    m4 = (j+d >= 0) & (j+d < n)
    b[m4] += h[j+d[m4]]
    
    return a + b

        h_try[i] += alpha
        h_try[j] -= alpha
        corr = np.correlate(h_try, 1 - h_try, mode='full')

    s0 = float(np.max(scores))

    lo, hi = a_min, a_max
    c = hi - gr * (hi - lo)
    d_ = lo + gr * (hi - lo)

            c = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = obj(d_)
