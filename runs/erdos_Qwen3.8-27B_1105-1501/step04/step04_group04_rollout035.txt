def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _grad_active(h, beta=0):
    """Compute gradient of the smoothed max or the active term gradient."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = np.max(scores)
    
    if beta < 1e-6:
        # Use only the active (max) term
        active = scores == max_s
        if not np.any(active):
            active = np.argmax(scores)
            active = np.zeros(L, dtype=bool)
            active[np.argmax(scores)] = True
    else:
        # Smoothed weights
        w = np.exp(beta * (scores - max_s))
        w /= np.sum(w)
        active = w
    
    # Compute gradient
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        if beta < 1e-6:
            r = n - 1 + d
            w_r = 1.0 if active[r] else 0.0
        else:
            w_r = active[n - 1 + d]
        if w_r < 1e-15:
            continue
        low = max(0, -d)
        high = min(n, n - d)
        i = np.arange(low, high)
        # d/dh_j of dx * sum h_i * (1 - h_{i+d})
        # = dx * [1_{j in [low,high-1]} - h_{j+d} * 1_{j in [low,high-1], j+d valid} - h_{j-d} * 1_{j-d in [low,high-1], j-d valid}]
        for j in i:

                # So jm = j - d must be in [low, high-1]

    return grad

def _grad_single_lag(h, d, dx):
    n = len(h)
    low = max(0, -d)
    high = min(n, n - d)
    grad = np.zeros(n)
    
    # Linear part: dx in positions [low, high)
    grad[low:high] += dx
    
    # Bilinear part 1: for i in [low, high), the term h_i * h_{i+d}
    # d/dh_i = h_{i+d}
    i = np.arange(low, high)
    ip = i + d
    valid_p = (ip >= 0) & (ip < n)
    grad[i[valid_p]] -= dx * h[ip[valid_p]]
    
    # Bilinear part 2: for i in [low, high), the term h_i * h_{i+d}
    # d/dh_{i+d} = h_i
    # So for j = i+d, grad[j] -= dx * h[j-d] = dx * h[i]
    # j = i+d, valid when 0 <= j < n
    valid_p2 = (ip >= 0) & (ip < n)
    grad[ip[valid_p2]] -= dx * h[i[valid_p2]]
    
    return grad

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - v.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-15
        else:
            mask = v > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        v[mask] += diff / cnt
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _grad_lse(h, beta):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    ms = scores.max()
    if beta < 1e-6:
        w = np.zeros(L)
        w[int(np.argmax(scores))] = 1.0
    else:
        w = np.exp(beta * (scores - ms))
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(L)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= ws
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wr = w[n - 1 + d]
        if wr < 1e-14:
            continue
        low = max(0, -d)
        high = min(n, n - d)
        ii = np.arange(low, high)
        jj = ii + d
        va = (jj >= 0) & (jj < n)
        grad[jj[va]] += wr * dx * (1.0 - h[ii[va]])
        vb = (jj >= 0) & (jj < n)
        grad[ii[vb]] -= wr * dx * h[jj[vb]]
    return grad

def _mom_descent(h, deadline, beta_lo=15, beta_hi=3000, n_beta=10, mu0=0.6):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)
    mu = mu0
    lr = 0.04
    betas = np.geomspace(beta_lo, beta_hi, n_beta)
    for beta in betas:
        if time.time() > deadline:
            break
        stall = 0
        for it in range(80000):
            if time.time() > deadline:
                break
            if stall > 300:
                break
            g = _grad_lse(h, beta)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g /= gn
            v = mu * v - lr * g
            v -= v.mean()
            vn = np.linalg.norm(v)
            if vn < 1e-15:
                break
            best_step = 0.0
            best_tc = cur_c5
            for f in [1.0, 0.66, 0.44, 0.22, 0.11, 0.05, 0.025]:
                tr = _project(h + f * v, tgt)
                tc = _exact_c5(tr)
                if tc < best_tc - 1e-14:
                    best_tc = tc
                    best_step = f
                else:
                    break
            if best_step > 0:
                h = _project(h + best_step * v, tgt)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.25)
                mu = min(mu * 1.01, 0.85)
            else:
                stall += 1
                lr *= 0.82
                mu = max(mu * 0.97, 0.3)
    return best_h, best_c5

def _local_single(h, deadline, rng, max_iter=500000):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hop(h, deadline, rng, n_hops=20, hop_size=0.1):
    n = len(h)
    tgt = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for k in range(n_hops):
        if time.time() > deadline:
            break
        pert = rng.normal(0, hop_size, n)
        pert -= pert.mean()
        h_p = _project(h + pert, tgt)
        h_r, c5_r = _mom_descent(h_p, min(deadline, time.time() + 15),
                                  beta_lo=50, beta_hi=500, n_beta=5, mu0=0.5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            h = h_r.copy()
    return best_h, best_c5

def _systematic_local(h, deadline, max_rounds=20):
    """Try all single-coordinate moves systematically, sorted by potential."""
    n = len(h)
    tgt = n / 2.0
    cur_c5 = _exact_c5(h)
