```python
        g[self.n] = 1.0
        return g
    
    def ineq(self, x):
        h = x[:self.n]
        T = x[self.n]
        return T - _manual_scores(h)
    
    def ineq_jac(self, x):
        h = x[:self.n]
        G = _manual_grad_matrix(h)
        n = self.n
        jac = np.empty((2 * n - 1, n + 1))
        jac[:, :n] = -G
        jac[:, n] = 1.0
        return jac
    
    def eq(self, x):
        return float(np.sum(x[:self.n])) - self.target
    
    def eq_jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j


def _epigraph_solve(h0, deadline, maxiter=80):
    n = len(h0)
    target = n / 2.0
    prob = _EpiProb(n)
    scores = _manual_scores(h0)
    T0 = float(np.max(scores)) + 1e-5
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = T0
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n) + 1.0)]
    constraints = [
        {'type': 'ineq', 'fun': prob.ineq, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq, 'jac': prob.eq_jac}
    ]
    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            result = minimize(
                fun=prob.obj, x0=x0, jac=prob.obj_grad,
                bounds=bounds, constraints=constraints,
                method='SLSQP',
                options={'maxiter': maxiter, 'ftol': 1e-13}
            )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = np.clip(result.x[:n], 0.0, 1.0)
            h_sol = _project(h_sol, target)
            return h_sol
    except Exception:
        pass
    return h0
```

```python
def _line_search_c5(h, p, alpha_max):
    n = len(h)
    best_t = 0.0
    best_val = _exact_c5(h)
    # Grid search
    n_grid = 20
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, best_val + 1.0)
    for i, t in enumerate(ts):
        ht = h + t * p
        if ht.min() >= -1e-12 and ht.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(ht)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    # Golden section
    lo_t = ts[max(0, idx - 1)]
    hi_t = ts[min(n_grid - 1, idx + 1)]
    if hi_t - lo_t < 1e-12:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = _exact_c5(h + c1 * p) if (h + c1 * p).min() >= -1e-12 and (h + c1 * p).max() <= 1.0 + 1e-12 else best_val + 1
    f2 = _exact_c5(h + c2 * p) if (h + c2 * p).min() >= -1e-12 and (h + c2 * p).max() <= 1.0 + 1e-12 else best_val + 1
    for _ in range(30):
        if hi_t - lo_t < 1e-13:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            ht = h + c1 * p
            f1 = _exact_c5(ht) if ht.min() >= -1e-12 and ht.max() <= 1.0 + 1e-12 else best_val + 1
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            ht = h + c2 * p
            f2 = _exact_c5(ht) if ht.min() >= -1e-12 and ht.max() <= 1.0 + 1e-12 else best_val + 1
    opt_t = 0.5 * (lo_t + hi_t)
    ht = h + opt_t * p
    if ht.min() >= -1e-12 and ht.max() <= 1.0 + 1e-12:
        f_opt = _exact_c5(ht)
        if f_opt < best_val:
            best_t, best_val = opt_t, f_opt
    return best_t, best_val
```

```python
def _lp_direction(h, scores, radius):
    n = len(h)
    G = _manual_grad_matrix(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_local(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = _manual_scores(h)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        alpha, new_c5 = _line_search_c5(h, p, alpha_max)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5
```

```python
def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if n <= 60:
            pairs = [(j, k) for j in range(n) for k in range(j + 1, n)]
            rng.shuffle(pairs)
        else:
            n_pairs = min(50 * n, n * (n - 1) // 2)
            pairs = []
            for _ in range(n_pairs):
                j = int(rng.integers(0, n))
                k = int(rng.integers(0, n))
                if j != k:
                    pairs.append((min(j, k), max(j, k)))
        for (j, k) in pairs:
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                am = 1.0
                if dj > 0:
                    am = min(am, (1.0 - h[j]) / dj)
                else:
                    am = min(am, h[j] / (-dj))
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-12:
                    continue
                for frac in [0.5, 0.25, 0.75, 0.1]:
                    ht = h.copy()
                    ht[j] += dj * am * frac
                    ht[k] += dk * am * frac
                    if ht[j] < -1e-12 or ht[j] > 1.0 + 1e-12 or ht[k] < -1e-12 or ht[k] > 1.0 + 1e-12:
                        continue
                    c5_t = _exact_c5(ht)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = ht.copy()
                        h = ht.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5
```

```python
def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
```

```python
def _weighted_grad(h, w):
    n = len(h)
    G = _manual_grad_matrix(h)
    return G.T @ w


def _grad_optimize(h, deadline, tau_init=15.0, lr_init=0.02, mom_beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        dx = 2.0 / n
        scores = _manual_scores(h) * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = _weighted_grad(h, w) * dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.15)
            tau = min(tau * 1.02, 200.0)
        else:
            stall += 1
            lr *= 0.92
            if stall > 60:
                break
            if stall % 20 == 0:
                tau = max(tau_init, tau * 0.8)
    return best_h, best_c5
```
