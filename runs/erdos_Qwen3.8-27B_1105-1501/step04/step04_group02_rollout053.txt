def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):

        if d >= 0:
            k = n - d
            G[r, :k] -= h[d:]
            G[r, d:] += 1.0 - h[:k]
        else:
            s = -d
            k = n - s
            G[r, :k] += 1.0 - h[s:]
            G[r, s:] -= h[:k]

def _nesterov(h, deadline, tau=40.0, max_iter=500):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    h_prev = h.copy()
    cur_c5 = _exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    beta = 0.9
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Compute Nesterov extrapolation
        beta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        h_ext = h_cur + (beta_new / beta - 1.0) * (h_cur - h_prev)
        beta = beta_new
        
        # Compute gradient at extrapolated point
        scores = _raw_scores(h_ext)
        ms = float(np.max(scores))
        sh = tau * (scores - ms)
        exp_s = np.exp(sh)
        w = exp_s / float(np.sum(exp_s))
        G = _grad_matrix(h_ext)
        grad = G.T @ w
        
        # Project gradient onto null space of equality constraint
        grad -= np.mean(grad)
        
        # Line search
        best_alpha = 0.0
        best_val = cur_c5
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            h_try = h_cur - alpha * grad
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = alpha
        
        if best_alpha > 0:
            h_new = h_cur - best_alpha * grad
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h_prev = h_cur
                h_cur = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_cur.copy()
            else:
                # If line search didn't improve, try smaller steps
                improved = False
                for alpha in [0.001, 0.002, 0.005]:
                    h_try = h_cur - alpha * grad
                    h_try = _project(h_try, target)
                    c5_try = _exact_c5(h_try)
                    if c5_try < cur_c5 - 1e-14:
                        h_prev = h_cur
                        h_cur = h_try
                        cur_c5 = c5_try
                        improved = True
                        if cur_c5 < best_c5:

    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    best_t = 0.0
    for t in ts:
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            v = _exact_c5(h_t)
            if v < best_val:

    return best_t, best_val
