    If j = i-d_r: term is h[j]*(1-h[i]). ∂/∂h[i] gives -h[j], then ∂/∂h[j] gives -1.
    But wait, j = i-d_r means d_r = i-j. And we're computing ∂²/∂h[i]∂h[j].
    
  Let me just enumerate: the mixed partial is -1 if the pair (i,j) appears in some product h[k]*h[k+d_r] (with appropriate indexing). Specifically:
  - If i - j = d_r: term k=j gives h[j]*h[j+d_r] = h[j]*h[i], mixed partial = -1 (from the -h[k]*h[k+d] part of h[k](1-h[k+d]))

  - If i - j = -d_r, i.e., j - i = d_r: same as second case
  - If j - i = -d_r, i.e., i - j = d_r: same as first case

- If i - j = d: contributes -1
- If j - i = d: contributes -1
So: -1 if |i-j| = |d|, else 0. But we also need the indices to be valid (0 <= i, j < n and 0 <= i+d, j+d < n for the corresponding terms to exist).

Actually, I think it's simpler to just say: the cross term exists if and only if i and j differ by exactly |d_r| AND both are in the valid range for that lag. For the purposes of the pair swap optimization, I can just compute:
- linear coefficient: L_r = grad_r[i] - grad_r[j]
- quadratic coefficient: Q_r = -1 if |i-j| == |d_r| (and valid), else 0

Then S_r(a) ≈ S_r(0) + L_r * a + Q_r * a²

Since the max of (quadratic or linear) functions is convex, we can minimize by:
1. Checking endpoints

    return np.correlate(h, 1.0 - h, mode='full') * dx

def proj(v, target):

    v = np.asarray(v, float)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    r = target - x.sum()
    if abs(r) > 1e-12:
        if r > 0:
            m = (x < 1 - 1e-10)
        else:
            m = (x > 1e-10)
        c = m.sum()
        if c > 0:
            x[m] += r / c
    x = np.clip(x, 0, 1)
    r2 = target - x.sum()
    if abs(r2) > 1e-13:
        idx = int(np.argmax(1 - x)) if r2 > 0 else int(np.argmin(x))
        x[idx] += r2
        x = np.clip(x, 0, 1)
    return x

def grad_lag(h, d):
    """Gradient of raw correlation at lag offset d (d = r - (n-1))."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0: return g
        g[d:] += (1.0 - h[:m])
        g[:m] += -h[d:]
    else:
        q = -d
        m = n - q
        if m <= 0: return g
        g[:m] += (1.0 - h[q:])
        g[q:] += -h[:m]
    return g

def subgrad(h, scores, topk):
    """Weighted subgradient using top-k active lags."""
    n = len(h)
    c = n - 1
    # Find top-k lags
    idx = np.argsort(scores)[::-1][:topk]
    # Weight by softmax of (scores - max) / (small temp)
    max_s = scores[idx[0]]
    w = np.exp((scores[idx] - max_s) * 20.0)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        r = idx[ii]
        d = r - c
        g += w[ii] * grad_lag(h, d)
    return g

def grad_descent(h, deadline, topk=15):

    n = len(h)

    best_h = h.copy()
    scores = scores_of(h)

    eta = 0.05
    patience = 0
    
    while time.time() < deadline:
        g = subgrad(h, scores, topk)
        # Project gradient to sum-zero direction
        p = g - g.mean()
        pn = p / (np.max(np.abs(p)) + 1e-15)
        
        # Line search
        best_a = 0.0
        best_f = best_c5
        for trial in range(12):
            a = eta * (trial + 1) * 0.5
            ht = proj(h + a * pn, target)
            ct = exact_c5(ht)
            if ct < best_f - 1e-13:
                best_f = ct
                best_a = a
            elif ct > best_f + 0.01:

            best_c5 = exact_c5(h)
            if best_c5 < exact_c5(best_h):

        else:
            patience += 1
            eta *= 0.7
            if patience > 30:
                break

def pair_line_search(h, i, j, target):
    """Find optimal swap amount a for h[i]+=a, h[j]-=a."""
    # Feasible range
    a_lo = max(-h[i], h[j] - 1.0)
    a_hi = min(1.0 - h[i], h[j])
    if a_hi - a_lo < 1e-12:
        return 0.0, exact_c5(h)
    
    best_a = 0.0
    best_c5 = exact_c5(h)
    
    # Coarse grid
    for k in range(1, 10):
        a = a_lo + (a_hi - a_lo) * k / 10.0
        ht = h.copy()
        ht[i] += a
        ht[j] -= a
        c = exact_c5(ht)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_a = a
    # Also check negative side
    for k in range(1, 5):
        a = a_lo + (a_hi - a_lo) * (-k) / 10.0
        if a < a_lo: break
        ht = h.copy()
        ht[i] += a
        ht[j] -= a
        c = exact_c5(ht)
        if c < best_c5 - 1e-13:
            best_c5 = c
            best_a = a
    
    if best_a == 0.0:
        return 0.0, best_c5
    
    # Golden section
    lo = a_lo if best_a < (a_lo + a_hi) / 2 else best_a - (a_hi - a_lo) * 0.1
    hi = a_hi if best_a > (a_lo + a_hi) / 2 else best_a + (a_hi - a_lo) * 0.1
    lo = max(lo, a_lo)
    hi = min(hi, a_hi)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def eval_a(a):
        ht = h.copy()
        ht[i] += a
        ht[j] -= a
        return exact_c5(ht)
    
    f1 = eval_a(c1)
    f2 = eval_a(c2)
    for _ in range(20):
        if hi - lo < 1e-12: break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = eval_a(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = eval_a(c2)
    
    a_opt = (lo + hi) / 2.0
    c_opt = eval_a(a_opt)
    if c_opt < best_c5:
        return a_opt, c_opt
    return best_a, best_c5

    r2 = target - x.sum()
    if abs(r2) > 1e-13:
        idx = int(np.argmax(1 - x)) if r2 > 0 else int(np.argmin(x))
        x[idx] += r2
        x = np.clip(x, 0, 1)
    return x

def grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0: return g
        g[d:] += (1.0 - h[:m])
        g[:m] += -h[d:]
    else:
        q = -d
        m = n - q
        if m <= 0: return g
        g[:m] += (1.0 - h[q:])
        g[q:] += -h[:m]
    return g

def compute_sg(h, topk=12):
    """Compute weighted subgradient from top-k active lags."""
    scores = scores_of(h)
    n = len(h)
    c = n - 1
    idx = np.argsort(scores)[::-1][:topk]
    max_s = scores[idx[0]]
    w = np.exp((scores[idx] - max_s) * 15.0)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        d = int(idx[ii]) - c
        g += w[ii] * grad_lag(h, d)
    return g, scores
