```python
def _build_G(h, n):
    """Build the full gradient matrix G where G[r] = gradient of corr_r wrt h."""
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        s = r - (n-1)
        if s >= 0:
            # corr = sum_{j=s}^{n-1} h[j]*(1-h[j-s])
            # d/dh[m] = (1-h[m-s]) for s<=m<n, minus h[m+s] for 0<=m<=n-1-s
            m_range1 = np.arange(s, n)  # where first term applies
            G[r, m_range1] += (1.0 - h[m_range1 - s])
            m_range2 = np.arange(0, n-s)  # where second term applies
            if s > 0:
                G[r, m_range2] -= h[m_range2 + s]
            # For s=0: both ranges are 0..n-1, so G[r,m] = 1-h[m] - h[m] = 1-2h[m]
        else:
            t = -s
            # corr = sum_{j=0}^{n-1-t} h[j]*(1-h[j+t])
            # d/dh[m] = (1-h[m+t]) for 0<=m<=n-1-t, minus h[m-t] for t<=m<n
            m_range1 = np.arange(0, n-t)
            G[r, m_range1] += (1.0 - h[m_range1 + t])
            m_range2 = np.arange(t, n)
            G[r, m_range2] -= h[m_range2 - t]
    return G
```

```python
def _optimal_pair_step(h, n, i, j):
    """Find the optimal epsilon for swapping between coordinates i and j."""
    corr = _corr(h)
    m = 2*n - 1
    # For shift s = r - (n-1), G_r[i] and G_r[j]:
    # delta_corr_r = eps * (G_r[i] - G_r[j])
    # We want to minimize max_r (corr_r + eps * d_r) where d_r = G_r[i] - G_r[j]
    
    # Compute d_r for all r
    s_vals = np.arange(-(n-1), n)
    d = np.zeros(m)
    for idx, s in enumerate(s_vals):
        # G_r[i] for shift s:
        # (1-h[i-s]) if s<=i<n, minus h[i+s] if 0<=i<=n-1-s (for s>0)
        # Let me just compute it directly
        gi = 0.0
        gj = 0.0
        if s >= 0:
            # i in [s, n-1]: contributes (1-h[i-s])
            if s <= i < n:
                gi += (1.0 - h[i - s])
            # i in [0, n-1-s]: contributes -h[i+s]
            if s > 0 and 0 <= i <= n-1-s:  # for s=0, both terms combine to 1-2h[i]
                gi -= h[i + s]
            elif s == 0:
                gi = 1.0 - 2.0 * h[i]  # override
            # Same for j
            if s <= j < n:
                gj += (1.0 - h[j - s])
            if s > 0 and 0 <= j <= n-1-s:
                gj -= h[j + s]
            elif s == 0:
                gj = 1.0 - 2.0 * h[j]
        else:

            if 0 <= i <= n-1-t:
                gi += (1.0 - h[i + t])
            if t <= i < n:
                gi -= h[i - t]
            if 0 <= j <= n-1-t:
                gj += (1.0 - h[j + t])
            if t <= j < n:
                gj -= h[j - t]

    # Intersection of r1 and r2: corr_r1 + eps*d_r1 = corr_r2 + eps*d_r2
    # eps = (corr_r2 - corr_r1) / (d_r1 - d_r2)
    
    # But there are O(m^2) intersection points. For m=127, that's ~8000 points.
    # We can be smarter: only consider lines that could be active at the optimum.
    
    # Simple approach: ternary search or golden section on the convex function
    # Since it's 1D convex, golden section works well
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = eps_lo, eps_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(corr + c1 * d))
    f2 = float(np.max(corr + c2 * d))
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(corr + c1 * d))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(corr + c2 * d))
    
    eps_opt = (lo + hi) / 2.0
    return eps_opt, float(np.max(corr + eps_opt * d))
```

This is O(m) per evaluation of f(eps), and we do ~50 evaluations, so O(50m) = O(50*127) ≈ 6350 operations per pair step. Very fast!

But we need to choose which pair (i,j) to use. Options:
1. Random pairs
2. Pairs that give the best improvement (expensive to find)
3. Pairs related to the active correlations

I'll use a combination: for the active correlations, compute the gradient, and choose pairs that align with the gradient direction.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_single(h, s, i):
    """Gradient of corr at shift s with respect to h[i]."""
    n = len(h)
    if s >= 0:
        g = 0.0
        if s <= i < n:
            g += (1.0 - h[i - s])
        if s == 0:
            g = 1.0 - 2.0 * h[i]
        elif 0 <= i <= n - 1 - s:
            g -= h[i + s]
        return g
    else:
        t = -s
        g = 0.0
        if 0 <= i <= n - 1 - t:
            g += (1.0 - h[i + t])
        if t <= i < n:
            g -= h[i - t]
        return g


def _pair_delta(h, n, i, j):
    """Compute d_r = G_r[i] - G_r[j] for all shifts."""
    m = 2 * n - 1
    d = np.zeros(m)
    for idx in range(m):
        s = idx - (n - 1)
        d[idx] = _grad_single(h, s, i) - _grad_single(h, s, j)
    return d


def _pair_optimal_eps(corr, d, eps_lo, eps_hi):
    """Find eps minimizing max_r(corr_r + eps*d_r) via golden section."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = eps_lo, eps_hi
    if hi - lo < 1e-14:
        return (lo + hi) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(corr + c1 * d))
    f2 = float(np.max(corr + c2 * d))
    for _ in range(60):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(corr + c1 * d))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(corr + c2 * d))
    return (lo + hi) / 2.0


def _best_pair_swap(h, n, corr, max_pairs=100):
    """Try best pairs and return the most improving one."""
    # Get active correlations (top few)
    active = np.argsort(-corr)[:5]
    # Compute gradient of active max
    g = np.zeros(n)
    for r in active:
        s = r - (n - 1)
        for i in range(n):
            g[i] += _grad_single(h, s, i)
    g /= len(active)
    
    # Candidate pairs: (i,j) where g[i] < g[j] (transfer from j to i reduces active corr)
    # Sort by g value
    order = np.argsort(g)
    # Try pairs from the extreme ends
    pairs_tried = 0
    best_imp = 0.0
    best_i, best_j, best_eps = -1, -1, 0.0
    best_new_c5 = _exact_c5(h)
    
    for ii in range(min(8, n)):
        for jj in range(max(0, n - 8 - ii), n):
            i = order[ii]
            j = order[jj]
            if i == j:
                continue
            if g[i] >= g[j]:
                continue
            d = _pair_delta(h, n, i, j)
            eps_lo = -h[j]
            eps_hi = 1.0 - h[i]
            if eps_hi - eps_lo < 1e-12:
                continue
            eps_opt = _pair_optimal_eps(corr, d, eps_lo, eps_hi)
            new_max = float(np.max(corr + eps_opt * d))
            improvement = best_new_c5 - new_max
            if improvement > best_imp:
                best_imp = improvement
                best_i, best_j, best_eps = i, j, eps_opt
            pairs_tried += 1
            if pairs_tried >= max_pairs:
                break
        if pairs_tried >= max_pairs:
            break
    
    if best_i < 0:
        return h.copy(), best_new_c5
    
    h_new = h.copy()
    h_new[best_i] += best_eps
    h_new[best_j] -= best_eps
    h_new = _project(h_new, n * 0.5)
    new_c5 = _exact_c5(h_new)
    return h_new, new_c5


def _fw_pair_loop(h, n, deadline, max_iter=20000):
    """Combined FW and pair-swap local optimization."""
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    corr = _corr(h)
    stale = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stale > 100:
            break
        
        # Every 10 iterations do a FW step, otherwise pair swap
        if it % 10 == 0:
            # FW: find vertex minimizing subgradient
            active = np.argsort(-corr)[:5]
            g = np.zeros(n)
            for r in active:
                s = r - (n - 1)
                for i in range(n):
                    g[i] += _grad_single(h, s, i)
            g /= len(active)
            
            # Best vertex
            k = n // 2
            v_idx = np.argpartition(g, k)[:k]
            v = np.zeros(n)
            v[v_idx] = 1.0
            gap = float(np.dot(v - h, g))
            
            if gap < -1e-12:
                # Take step
                gamma = 2.0 / (it + 2)
                h_new = _project(h + gamma * (v - h), n * 0.5)
                new_c5 = _exact_c5(h_new)
                if new_c5 < cur_c5:
                    h = h_new
                    cur_c5 = new_c5
                    corr = _corr(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stale = 0
                else:
                    stale += 1
            else:
                stale += 1
        else:
            # Pair swap
            h_new, new_c5 = _best_pair_swap(h, n, corr, max_pairs=50)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                corr = _corr(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stale = 0
            else:
                stale += 1
    
    return best_h, best_c5


def _build_G_matrix(h, n):
    """Build full gradient matrix."""
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        s = r - (n - 1)
        for i in range(n):
            G[r, i] = _grad_single(h, s, i)
    return G


def _build_G_fast(h, n):
    """Build gradient matrix using vectorized operations."""
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        s = r - (n - 1)
        if s == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif s > 0:
            G[r, s:] += (1.0 - h[:n - s])
            G[r, :n - s] -= h[s:]
        else:
            t = -s
            G[r, :n - t] += (1.0 - h[t:])
            G[r, t:] -= h[:n - t]
    return G


def _lp_trust_region(h, n, deadline, radius_init=0.05, max_iters=3000):
    """LP trust-region polishing."""
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stale = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stale > 80:
            break
        
        corr = _corr(h)
        G = _build_G_fast(h, n)
        m = 2 * n - 1
        
        # LP: min t s.t. corr_r + G_r·p <= t, box on p
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr
        
        bounds = []
        for i in range(n):
            lo_b = max(-radius, -h[i])
            hi_b = min(radius, 1.0 - h[i])
            if lo_b > hi_b - 1e-12:
                lo_b = hi_b = h[i] if 0 <= h[i] <= 1 else 0.0
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success or res.x is None:
            stale += 1
            radius *= 0.7
            continue
        
        p = res.x[:n]
        if np.max(np.abs(p)) < 1e-13:
            stale += 1
            radius *= 0.7
            if radius < 1e-9:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_c5_trial = cur_c5
        for alpha_frac in [0.1, 0.25, 0.5, 0.75, 1.0, 1.5]:
            alpha = alpha_frac * radius
            h_try = _project(h + alpha * p / (radius + 1e-15), n * 0.5)
            if np.max(np.abs(h_try - h)) < 1e-14:
                continue
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_trial:
                best_c5_trial = c5_try
                best_alpha = alpha
        
        # Also try the LP suggested step directly
        h_try = _project(h + p, n * 0.5)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5_trial:
            best_c5_trial = c5_try
            best_alpha = 1.0  # flag: use p directly
        
        if best_alpha > 0 and best_c5_trial < cur_c5 - 1e-14:
            if best_alpha == 1.0:
                h = _project(h + p, n * 0.5)
            else:
                h = _project(h + best_alpha * p / (radius + 1e-15), n * 0.5)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stale = 0
            radius = min(radius * 1.05, 0.15)
        else:
            stale += 1
            radius *= 0.6
            if stale % 20 == 0 and radius < 0.01:
                radius = 0.03
    
    return best_h, best_c5


def _slsqp_epigraph(h_init, n, maxiter=300, ftol=1e-14):
    """SLSQP epigraph formulation."""
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    
    def _obj(z, nn):
        return z[nn]
    
    def _obj_jac(z, nn):
        jj = np.zeros(nn + 1)
        jj[nn] = 1.0
        return jj
    
    def _eq_fun(z, nn):
        return np.sum(z[:nn]) - nn * 0.5
    
    def _eq_jac(z, nn):
        jj = np.ones(nn + 1)
        jj[nn] = 0.0
        return jj
    
    def _ineq_fun(z, nn):
        hh = z[:nn]
        cc = _corr(hh)
        return z[nn] - cc
    
    def _ineq_jac(z, nn):
        hh = z[:nn]
        G = _build_G_fast(hh, nn)
        J = np.zeros((2 * nn - 1, nn + 1))
        J[:, :nn] = -G
        J[:, nn] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac, 'args': (n,)}
    ]
    
    try:
        res = minimize(_obj, z0, jac=_obj_jac, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)
```
