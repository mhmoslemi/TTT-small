    """Find exact minimum of C5(h + alpha*v) over feasible alpha range."""
    n = len(h)
    dx = 2.0 / n
    
    # Find feasible alpha range
    alpha_min = 0.0
    alpha_max = max_alpha
    for i in range(n):
        if v[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-15:
            alpha_min = max(alpha_min, -h[i] / v[i])
    
    if alpha_max < alpha_min:
        return 0.0, _exact_c5(h)
    
    # Compute score as quadratic in alpha for each lag
    g = 1.0 - h
    n_lags = 2 * n - 1
    
    # score_r(alpha) = a_r + b_r * alpha + c_r * alpha^2
    # a_r = sum_i h[i] * g[i+d]
    # b_r = sum_i (v[i]*g[i+d] - h[i]*v[i+d])
    # c_r = -sum_i v[i]*v[i+d]
    
    # Use correlate for efficiency
    a = np.correlate(h, g, mode='full')  # a_r for r=0..2n-2
    b = np.correlate(v, g, mode='full') - np.correlate(h, v, mode='full')
    # For c: -autocorrelation of v
    c = -np.correlate(v, v, mode='full')
    
    # Now find min of max(a + b*al + c*al^2) over al in [alpha_min, alpha_max]
    # This is the max of 127 quadratics
    # Find all critical points: where two quadratics are equal, or where one is at its vertex
    
    # Collect all candidate alpha values
    candidates = np.array([alpha_min, alpha_max])
    
    # Vertex of each quadratic
    for r in range(n_lags):
        if c[r] != 0:
            vertex = -b[r] / (2.0 * c[r])
            if alpha_min <= vertex <= alpha_max:
                candidates = np.append(candidates, vertex)
    
    # Intersections of pairs of quadratics
    # (a_i + b_i*al + c_i*al^2) = (a_j + b_j*al + c_j*al^2)
    # (c_i - c_j)*al^2 + (b_i - b_j)*al + (a_i - a_j) = 0
    for i in range(n_lags):
        for j in range(i+1, n_lags):
            dc = c[i] - c[j]
            db = b[i] - b[j]
            da = a[i] - a[j]
            if abs(dc) < 1e-15:
                # Linear equation: db*al + da = 0
                if abs(db) > 1e-15:
                    al = -da / db
                    if alpha_min <= al <= alpha_max:
                        candidates = np.append(candidates, al)
            else:
                disc = db*db - 4.0*dc*da
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    al1 = (-db + sqrt_disc) / (2.0 * dc)
                    al2 = (-db - sqrt_disc) / (2.0 * dc)
                    if alpha_min <= al1 <= alpha_max:
                        candidates = np.append(candidates, al1)
                    if alpha_min <= al2 <= alpha_max:
                        candidates = np.append(candidates, al2)
    
    # Evaluate C5 at all candidates
    best_c5 = float('inf')
    best_alpha = alpha_min
    for al in candidates:
        h_try = h + al * v
        c5_val = _exact_c5(h_try)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_alpha = al
    
    return best_alpha, best_c5

def _resample(h, n_new):

import numpy as np

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    nl = 2 * n - 1
    G = np.zeros((nl, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        for j in range(n):
            val = 0.0
            if 0 <= j + d < n:
                val += (1.0 - h[j + d])
            if 0 <= j - d < n:
                val -= h[j - d]
            G[r, j] = val
    return G * dx

def _quad_line_search(h, v, max_alpha=1.0, top_m=None):
    n = len(h)
    if float(np.max(np.abs(v))) < 1e-15:
        return 0.0, _exact_c5(h)
    dx = 2.0 / n
    a_min = 0.0
    a_max = max_alpha
    pm = v > 1e-14
    nm = v < -1e-14
    if np.any(pm):
        a_max = min(a_max, float(np.min((1.0 - h[pm]) / v[pm])))
    if np.any(nm):
        a_min = max(a_min, float(np.min(-h[nm] / v[nm])))
    if a_max <= a_min + 1e-14:
        return 0.0, _exact_c5(h)
    g = 1.0 - h
    A = np.correlate(h, g, mode='full') * dx
    B = (np.correlate(v, g, mode='full') - np.correlate(h, v, mode='full')) * dx
    C = (-np.correlate(v, v, mode='full')) * dx
    nl = len(A)
    if top_m is not None and top_m < nl:
        idx = np.argsort(A)[-top_m:]
        A, B, C = A[idx].copy(), B[idx].copy(), C[idx].copy()
        nl = len(A)
    candidates = [a_min, a_max]
    for i in range(nl):
        for j in range(i + 1, nl):
            dc = C[i] - C[j]
            db = B[i] - B[j]
            da = A[i] - A[j]
            if abs(dc) < 1e-13:
                if abs(db) > 1e-13:
                    al = -da / db
                    if a_min - 1e-10 <= al <= a_max + 1e-10:
                        candidates.append(min(max(al, a_min), a_max))
            else:
                disc = db * db - 4.0 * dc * da
                if disc >= -1e-13:
                    disc = max(disc, 0.0)
                    sd = np.sqrt(disc)
                    al1 = (-db + sd) / (2.0 * dc)
                    al2 = (-db - sd) / (2.0 * dc)
                    if a_min - 1e-10 <= al1 <= a_max + 1e-10:
                        candidates.append(min(max(al1, a_min), a_max))
                    if a_min - 1e-10 <= al2 <= a_max + 1e-10:
                        candidates.append(min(max(al2, a_min), a_max))
    alphas = np.array(candidates)
    if len(alphas) == 0:
        return 0.0, _exact_c5(h)
    vals = A[None, :] + B[None, :] * alphas[:, None] + C[None, :] * (alphas[:, None] ** 2)
    max_vals = np.max(vals, axis=1)
    best_i = int(np.argmin(max_vals))
    return float(alphas[best_i]), float(max_vals[best_i])

        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(nl)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            stall += 1
            if stall > 120:
                break
            continue
        d = -grad / gn
        found = False

                break
        if cur_c5 < best_c5:

def _pair_polish(h, deadline, max_passes=8, top_m=25):

    h = h.copy()
    best_h = h.copy()

            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):

            break

            break

            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            r *= 0.85
            if r < 1e-7:

            continue

        if alpha > 1e-12 and new_c5 < cur_c5 - 1e-12:
            h_new = h + alpha * p
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()

                r = min(r * 1.05, 0.5)
            else:
                stall += 1
                r *= 0.85
        else:
            stall += 1

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        best_h = hr.copy()

    # Phase 5: Descent with sharper weighting (10%)
    dl = t0 + budget_s * 0.56
    hr, cr = _descend(best_h, dl, top_k=8, n_steps=2000, p_base=32.0, p_growth=200.0)
    if cr < best_c5:
        best_c5 = cr
        best_h = hr.copy()

    # Phase 6: Pair polish again (6%)
    dl = t0 + budget_s * 0.62
    hr, cr = _pair_polish(best_h, dl, max_passes=8, top_m=20)
    if cr < best_c5:
        best_c5 = cr
        best_h = hr.copy()

    # Phase 7: Multi-resolution (14%)
    dl = t0 + budget_s * 0.76
    for dn in [16, -16, 8, -8, 32, -32, 48, 4]:
        if time.time() > dl:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl, time.time() + budget_s * 0.018)
        h_r, c5_r = _descend(h_r, sub_dl, top_k=12, n_steps=400, p_base=8.0, p_growth=30.0)
        if c5_r < best_c5 - 5e-5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            hr2, cr2 = _pair_polish(best_h, min(dl, time.time() + budget_s * 0.01), max_passes=3, top_m=25)
            if cr2 < best_c5:
                best_c5 = cr2
                best_h = hr2.copy()

    # Phase 8: Perturbation + re-optimize (12%)
    dl = t0 + budget_s * 0.88
    scale = 0.035
    for attempt in range(300):
        if time.time() > dl:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl, time.time() + budget_s * 0.01)
        h_try, c5_try = _lp_step(h_pert, sub_dl, top_k=8, radius=0.09, max_iters=200)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.82)
        else:
            scale = min(0.15, scale * 1.08)

    # Phase 9: Final LP ultra-fine (5%)
    dl = t0 + budget_s * 0.93
    hr, cr = _lp_step(best_h, dl, top_k=5, radius=0.015, max_iters=5000)
    if cr < best_c5:
        best_c5 = cr
        best_h = hr.copy()

    # Phase 10: Final pair polish (5%)
    dl = t0 + budget_s * 0.98
    hr, cr = _pair_polish(best_h, dl, max_passes=10, top_m=15)
    if cr < best_c5:
        best_c5 = cr
        best_h = hr.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
