    n = len(h)
    target = n / 2.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    n_evals = 0
    
    while time.time() < deadline:
        # Adaptive temperature
        elapsed_frac = 0.5  # placeholder
        temp = start_temp * (end_temp / start_temp) ** elapsed_frac
        # Simpler: decay based on step count
        temp = start_temp * (0.99 ** n_evals)
        temp = max(temp, end_temp)
        
        # Choose pair
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        sign = 1.0 if rng.random() < 0.5 else -1.0
        # h[i] += sign*t, h[j] -= sign*t
        if sign > 0:
            t_max = min(1.0 - h[i], h[j])
        else:
            t_max = min(1.0 - h[j], h[i])
        if t_max < 1e-10:
            continue
        
        t = t_max * rng.random()
        if t < 1e-12:
            continue
        
        if sign > 0:
            h[i] += t
            h[j] -= t
        else:
            h[j] += t
            h[i] -= t
        
        h = project_equal(h, target)
        new_c5 = exact_c5(h)
        n_evals += 1
        
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

            if sign > 0:
                h[i] -= t
                h[j] += t
            else:
                h[j] -= t
                h[i] += t
            h = project_equal(h, target)
    
    return best_h, best_c5

def resample_h(h_old, n_new):

            row = G[a]
            diffs = row[:, None] - row[None, :]
            f = np.maximum(f, diffs)
        np.fill_diagonal(f, np.inf)
        flat_f = f.ravel()
        n_cand = min(top_k, n * (n - 1))
        if n_cand > len(flat_f):
            n_cand = len(flat_f)
        top_idx = np.argpartition(flat_f, n_cand)[:n_cand]
        top_idx = top_idx[np.argsort(flat_f[top_idx])]
        improved = False
        for idx in top_idx:
            if time.time() > deadline:
                break
            i = int(idx // n)
            j = int(idx % n)
            if i == j:
                continue
            best_t, best_local = line_search_pair(h, i, j, cur_c5)
            if best_local < cur_c5 - 1e-13 and best_t > 1e-14:
                h[i] += best_t
                h[j] -= best_t
                h = project_equal(h, target)
                cur_c5 = exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            stall += 1
            if stall > 40:
                break
    return best_h, best_c5

def anneal_pairs(h, deadline, rng, start_temp=0.0015, decay=0.998):
    n = len(h)
    target = n / 2.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    temp = start_temp
    min_temp = 0.00005
    while time.time() < deadline:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if rng.random() < 0.5:
            sig = 1.0
            t_max = min(1.0 - h[i], h[j])
        else:
            sig = -1.0
            t_max = min(1.0 - h[j], h[i])
        if t_max < 1e-10:
            continue
        t = t_max * rng.random()
        if t < 1e-12:
            continue
        if sig > 0:
            h[i] += t
            h[j] -= t
        else:
            h[j] += t
            h[i] -= t
        h = project_equal(h, target)
        new_c5 = exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-9)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            if sig > 0:
                h[i] -= t
                h[j] += t
            else:
                h[j] -= t
                h[i] += t
            h = project_equal(h, target)
        temp = max(temp * decay, min_temp)
    return best_h, best_c5

def guided_anneal(h, deadline, rng, n_active=10):
    n = len(h)
    target = n / 2.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    temp = 0.0008
    min_temp = 0.00003
    while time.time() < deadline:
        corr = compute_scores(h)
        max_c = float(np.max(corr))
        threshold = max_c - max_c * 0.08 - 1e-12
        active_mask = corr >= threshold
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(corr))])
        if len(active_lags) > n_active:
            order = np.argsort(corr[active_lags])[::-1]
            active_lags = active_lags[order[:n_active]]
        G = compute_grad_lags(h, active_lags)
        f = np.full((n, n), -np.inf)
        for a in range(len(active_lags)):
            row = G[a]
            diffs = row[:, None] - row[None, :]
            f = np.maximum(f, diffs)
        np.fill_diagonal(f, np.inf)
        # Sample pair proportional to "promisingness" (more negative f = better)
        # Use softmax-like weighting on -f
        neg_f = -f
        neg_f += 0.5 * np.max(neg_f)  # shift for numerical stability
        weights = np.exp(neg_f / 0.01)
        weights[np.diag_indices_from(weights)] = 0.0
        total_w = np.sum(weights)
        if total_w < 1e-10:
            break
        probs = weights / total_w
        flat_idx = int(rng.choice(len(probs.ravel()), p=probs.ravel()))
        i = int(flat_idx // n)
        j = int(flat_idx % n)
        if i == j:
            continue
        t_max = min(1.0 - h[i], h[j])
        if t_max < 1e-10:
            continue
        t = t_max * (0.3 + 0.7 * rng.random())
        h[i] += t
        h[j] -= t
        h = project_equal(h, target)
        new_c5 = exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-9)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] -= t
            h[j] += t
            h = project_equal(h, target)
        temp = max(temp * 0.999, min_temp)
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    if n_new < 20:
        n_new = 20
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_equal(h_new, n_new / 2.0)

def multi_block_descent(h, deadline, n_active=10):
    """Try moving two pairs simultaneously for larger steps."""
    n = len(h)
    target = n / 2.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng_multi = np.random.default_rng(123)
    while time.time() < deadline:
        corr = compute_scores(h)
        max_c = float(np.max(corr))
        threshold = max_c - max_c * 0.06 - 1e-12
        active_lags = np.where(corr >= threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(corr))])
        if len(active_lags) > n_active:
            order = np.argsort(corr[active_lags])[::-1]
            active_lags = active_lags[order[:n_active]]
        G = compute_grad_lags(h, active_lags)
        f = np.full((n, n), -np.inf)
        for a in range(len(active_lags)):
            row = G[a]
            diffs = row[:, None] - row[None, :]
            f = np.maximum(f, diffs)
        np.fill_diagonal(f, np.inf)
        # Find best single pair
        flat_f = f.ravel()
        idx1 = int(np.argmin(flat_f))
        i1 = int(idx1 // n)
        j1 = int(idx1 % n)
        if f[i1, j1] > -1e-12:
            break
        # Try a second random pair combined
        i2 = int(rng_multi.integers(n))
        j2 = int(rng_multi.integers(n))
        if i2 == j2 or i2 == i1 or j2 == j1:
            # Just do single pair
            best_t, best_local = line_search_pair(h, i1, j1, cur_c5)
            if best_local < cur_c5 - 1e-13 and best_t > 1e-14:
                h[i1] += best_t
                h[j1] -= best_t
                h = project_equal(h, target)
                cur_c5 = exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            continue
        # Block move: h[i1]+t, h[j1]-t, h[i2]+t2, h[j2]-t2
        # Try small t, t2
        t1_max = min(1.0 - h[i1], h[j1])
        t2_max = min(1.0 - h[i2], h[j2])
        if t1_max < 1e-12 or t2_max < 1e-12:
            best_t, best_local = line_search_pair(h, i1, j1, cur_c5)
            if best_local < cur_c5 - 1e-13 and best_t > 1e-14:
                h[i1] += best_t
                h[j1] -= best_t
                h = project_equal(h, target)
                cur_c5 = exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            continue
        improved = False
        for frac1 in [0.3, 0.5, 0.7]:
            for frac2 in [0.3, 0.5, 0.7]:
                t1 = t1_max * frac1
                t2 = t2_max * frac2
                h[i1] += t1
                h[j1] -= t1
                h[i2] += t2
                h[j2] -= t2
                c5_t = exact_c5(h)
                if c5_t < cur_c5 - 1e-13:
                    h = project_equal(h, target)
                    cur_c5 = exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i1] -= t1
                    h[j1] += t1
                    h[i2] -= t2
                    h[j2] += t2
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = project_equal(h0, n0 / 2.0).copy()
    best_c5 = exact_c5(best_h)
    best_n = n0

    # Phase 1: Fast greedy pair descent
    d1 = t0 + budget_s * 0.20
    h1, c1 = greedy_pair_descent(best_h, d1, n_active=12, top_k=8)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Guided annealing
    d2 = t0 + budget_s * 0.38
    h2, c2 = guided_anneal(best_h, d2, rng, n_active=10)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()

    # Phase 3: More greedy descent
    d3 = t0 + budget_s * 0.52
    h3, c3 = greedy_pair_descent(best_h, d3, n_active=15, top_k=5)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3.copy()

    # Phase 4: Multi-block descent
    d4 = t0 + budget_s * 0.62
    h4, c4 = multi_block_descent(best_h, d4, n_active=10)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4.copy()

    # Phase 5: Multi-resolution exploration
    d5 = t0 + budget_s * 0.80
    tried_n = {best_n}
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -32, 32, -40, 40, -48, 48]
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c_r = greedy_pair_descent(h_r, sub_dl, n_active=10, top_k=4)
        if c_r < best_c5 - 1e-13:
            best_c5 = c_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final guided anneal
    d6 = t0 + budget_s * 0.90
    h6, c6 = guided_anneal(best_h, d6, rng, n_active=8)
    if c6 < best_c5:
        best_c5 = c6
        best_h = h6.copy()

    # Phase 7: Final micro-polish
    if time.time() < hard_deadline:
        h7, c7 = greedy_pair_descent(best_h, hard_deadline, n_active=6, top_k=3)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7.copy()

    # Final validation
    best_h = project_equal(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project_equal(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
