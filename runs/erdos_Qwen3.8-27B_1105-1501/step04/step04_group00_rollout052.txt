import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_minimax(h, scores, G, radius, top_k=None):
    """LP: min t s.t. s_k + G_k @ p <= t, sum(p)=0, |p|<=radius, 0<=h+p<=1"""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    m = len(scores_k)
    # Variables: p (n vars) + t (1 var), total n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    A_ub = np.zeros((m + n, n + 1))
    b_ub = np.zeros(m + n)
    # Constraints: s_k + G_k @ p - t <= 0, i.e., G_k @ p - t <= -s_k
    A_ub[:m, :n] = G_k
    A_ub[:m, n] = -1.0
    b_ub[:m] = -scores_k
    # Box constraints on p: h_j + p_j <= 1 => p_j <= 1 - h_j
    # h_j + p_j >= 0 => -p_j <= h_j
    A_ub[m:m+n, :n] = np.eye(n)
    b_ub[m:m+n] = np.minimum(radius, 1.0 - h)
    A_ub[m+n:m+2*n, :n] = -np.eye(n)
    b_ub[m+n:m+2*n] = np.minimum(radius, h)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * (n + 1), method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_c5(h, p, alpha_max=None):
    n = len(h)
    if alpha_max is None:
        alpha_max = 0.5
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
    f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    opt_a = (lo + hi) / 2.0
    return opt_a, _exact_c5(np.clip(h + opt_a * p, 0, 1))

def _subgradient_step(h, gamma, deadline=None):
    """One subgradient step with softmax weights."""
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    max_s = float(np.max(scores))
    w = np.exp(gamma * (scores - max_s))
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= grad.mean()
    return grad, w

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _sine_init(n, freq, phase=0.0):
    x = np.arange(n) / n
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
    return _project(h, n / 2.0)

def _binary_sa(n, deadline, rng, n_iters=30000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or h[i] == h[j]:
            T *= cool
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cool
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_stride = max(1, n // 40)
    for _pass in range(max_passes):

        improved = False
        for j in range(0, n, step_stride):
            if time.time() > deadline:

            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step_stride, min(n, j + 4 * step_stride), step_stride):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Ternary search for optimal alpha
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(6):
                        m1 = lo_a + (hi_a - lo_a) * 0.333
                        m2 = hi_a - (hi_a - lo_a) * 0.333
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt

                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _accelerated_subgrad(h, deadline, gamma0=20.0, n_steps=300, lam0=0.08):
    """Nesterov-style accelerated subgradient for minimax."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    y = h.copy()  # extrapolated point
    t_k = 1.0
    for k in range(n_steps):
        if time.time() > deadline:
            break
        gamma = gamma0 * (1.0 + 0.02 * k)
        grad, _ = _subgradient_step(y, gamma)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        lam_k = lam0 / (1.0 + 0.05 * k)
        candidate = _project(y - lam_k * grad, target)
        c5_cand = _exact_c5(candidate)
        if c5_cand < cur_c5:
            h = candidate
            cur_c5 = c5_cand
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = _project(h + (t_k - 1.0) / t_new * (h - y), target)
        t_k = t_new
    return best_h, best_c5

            best_c5 = c5_2
            best_h = h2.copy()
        sub_dl = min(dl2, time.time() + budget_s * 0.02)
        h2b, c5_2b = _local_opt(best_h, sub_dl, max_iters=500)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Multi-resolution diverse search (30%)
    dl3 = t0 + budget_s * 0.54
    resolutions = [64, 80, 96, 112, 128, 96, 80, 64, 128]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.25, 0.75]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        n_starts = 2
        for s in range(n_starts):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_init(n_try, alpha)
                noise = rng.normal(0, 0.04, n_try)
                h_init = _project(h_init + noise, n_try / 2.0)
            sub_dl = min(dl3, time.time() + budget_s * 0.038)
            h_try, c5_try = _local_opt(h_init, sub_dl, max_iters=1200)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.012)
                h_da, c5_da = _dual_ascent_opt(h_try, sub_dl2, eta=0.6, n_steps=60)
                if c5_da < best_c5:
                    best_c5 = c5_da
                    best_h = h_da.copy()
                    best_n = n_try

    # Phase 4: LP from best at multiple nearby resolutions (12%)
    if time.time() < t0 + budget_s * 0.68:
        dl4 = t0 + budget_s * 0.68
        for dn in [-8, 8, -16, 16]:
            if time.time() > dl4:
                break
            n_try = best_n + dn
            if n_try < 48 or n_try > 200:
                continue
            h_r = _resample(best_h, n_try)
            sub_dl = min(dl4, time.time() + budget_s * 0.03)
            h_t, c5_t = _local_opt(h_r, sub_dl, max_iters=1000)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try
            sub_dl2 = min(dl4, time.time() + budget_s * 0.01)
            h_da2, c5_da2 = _dual_ascent_opt(h_t, sub_dl2, eta=0.5, n_steps=50)
            if c5_da2 < best_c5:
                best_c5 = c5_da2
                best_h = h_da2.copy()

    # Phase 5: Binary SA (10%)
    if time.time() < t0 + budget_s * 0.78:
        dl5 = t0 + budget_s * 0.78
        for n_bin in [64, 96, 80]:
            if time.time() > dl5:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=18000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
                h_br, c5_br = _local_opt(h_b, sub_dl2, max_iters=600)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()
            # Quasi-periodic + LP
            if time.time() < dl5:
                h_qp = _quasi_init(n_bin, 0.6180339887)
                h_qp = _project(h_qp + rng.normal(0, 0.03, n_bin), n_bin / 2.0)
                sub_dl3 = min(dl5, time.time() + budget_s * 0.02)
                h_qp, c5_qp = _local_opt(h_qp, sub_dl3, max_iters=800)
                if c5_qp < best_c5:
                    best_c5 = c5_qp
                    best_h = h_qp.copy()
                    best_n = n_bin

    # Phase 6: Perturbation + local (12%)
    dl6 = t0 + budget_s * 0.92
    scale = 0.025
    for attempt in range(250):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.011)
        h_try, c5_try = _local_opt(h_pert, sub_dl, max_iters=200)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.10, scale * 1.18)

    # Phase 7: Pair polish (5%)
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, dl7, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 8: Final LP refinement (3%)
    if time.time() < t0 + budget_s * 0.998:
        h_fin, c5_fin = _local_opt(best_h, t0 + budget_s * 0.998, max_iters=400)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
