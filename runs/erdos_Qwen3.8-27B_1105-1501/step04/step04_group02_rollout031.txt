        dec_idx = np.where(dec_mask)[0]
        inc_w = np.exp(-g[inc_idx] / T_bias)
        dec_w = np.exp(g[dec_idx] / T_bias)
        s_inc = inc_w.sum()
        s_dec = dec_w.sum()
        if s_inc < 1e-30 or s_dec < 1e-30:
            inc_w = np.ones(len(inc_idx))
            dec_w = np.ones(len(dec_idx))
            s_inc = len(inc_idx)
            s_dec = len(dec_idx)
        else:
            inc_w /= s_inc
            dec_w /= s_dec
        i = inc_idx[rng.choice(len(inc_idx), p=inc_w)]
        j = dec_idx[rng.choice(len(dec_idx), p=dec_w)]
        if i == j:
            continue
        q[i] += 1
        q[j] -= 1
        h_new = q.astype(float) / m
        c5_new = _exact_c5(h_new)
        if c5_new <= cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / T):
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            q[i] -= 1
            q[j] += 1
    return best_h, best_c5

def _pair_refine(h, deadline, seed=42):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    iters = 0
    while time.time() < deadline:
        iters += 1
        if iters % 500 == 0 and best_c5 < 0.3807:
            break
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        for sign in [1, -1]:
            delta = sign * rng.uniform(0.005, 0.04)
            if h[j] + delta < -1e-10 or h[j] + delta > 1.0 + 1e-10:
                continue
            if h[k] - delta < -1e-10 or h[k] - delta > 1.0 + 1e-10:
                continue
            ht = h.copy()
            ht[j] += delta
            ht[k] -= delta
            c5_t = _exact_c5(ht)
            if c5_t < cur_c5 - 1e-14:
                h = ht
                cur_c5 = c5_t
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
    return best_h, best_c5

def _lp_local(h, deadline, radius=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline and stall < 100:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        G = np.zeros((2 * n - 1, n))
        for r in range(2 * n - 1):
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                G[r, d:] += 1.0
                G[r, d:] -= h[:m]
                G[r, :m] -= h[d:]
            else:
                g2 = -d
                m = n - g2
                if m <= 0:
                    continue
                G[r, :m] += 1.0
                G[r, :m] -= h[g2:]
                G[r, g2:] -= h[:m]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                if np.max(np.abs(p)) > 1e-14:
                    alpha_max = 1.0
                    for j in range(n):
                        if p[j] > 1e-14:
                            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                        elif p[j] < -1e-14:
                            alpha_max = min(alpha_max, h[j] / (-p[j]))
                    alpha_max = max(alpha_max, 1e-10) * 0.95
                    t, new_c5 = _line_search(h, p, alpha_max, n_grid=30, n_golden=25)
                    if new_c5 < cur_c5 - 1e-14:
                        h = h + t * p
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        stall = 0
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        continue
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
        except Exception:
            stall += 10
    return best_h, best_c5

def _gradient_descent(h, deadline, tau=20.0, lr=0.02, beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline and stall < 100:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        g = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-14:
                continue
            d = r - (n - 1)
            g += w[r] * _lag_grad(h, d)
        g = g * dx
        mom = beta * mom + (1.0 - beta) * g
        h_new = _project(h - lr * mom, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            lr = min(lr * 1.02, 0.1)
            tau = min(tau * 1.01, 100.0)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            lr *= 0.9
            if stall % 40 == 0:
                tau = max(5.0, tau * 0.7)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(best_h)
    best_n = n0
    
    # Phase 1: Quick gradient descent to confirm local minimum (5% budget)
    d1 = t0 + budget_s * 0.05
    h1, c51 = _gradient_descent(best_h, d1, tau=30.0, lr=0.02, beta=0.9)
    if c51 < best_c5:
        best_c5, best_h = c51, h1
    
    # Phase 2: Active-set descent (10% budget)
    d2 = t0 + budget_s * 0.15
    h2, c52 = _active_descent(best_h, d2, top_k=6, seed=seed)
    if c52 < best_c5:
        best_c5, best_h = c52, h2
    
    # Phase 3: Stochastic annealing at parent resolution (20% budget)
    d3 = t0 + budget_s * 0.35
    h3, c53 = _anneal_exchange(best_h, d3, seed=seed, m=20)
    if c53 < best_c5:
        best_c5, best_h = c53, h3
    # Dequantize the annealed result
    h3c = _project(h3, best_h.size / 2.0)
    if _exact_c5(h3c) < best_c5:
        best_c5 = _exact_c5(h3c)
        best_h = h3c
    
    # Phase 4: LP local search (15% budget)
    d4 = t0 + budget_s * 0.50
    h4, c54 = _lp_local(best_h, d4, radius=0.12)
    if c54 < best_c5:
        best_c5, best_h = c54, h4
    
    # Phase 5: Multi-resolution exploration (20% budget)
    d5 = t0 + budget_s * 0.70
    n_candidates = [n0 - 2, n0 - 4, n0 + 2, n0 + 4, n0 - 6, n0 + 6, 84, 96, 100, 80, 104, 110, 92, 88]
    n_candidates = sorted(set(x for x in n_candidates if 60 <= x <= 160 and x != n0))
    time_per_res = max(0.5, (d5 - time.time()) / max(len(n_candidates), 1))
    for nc in n_candidates:
        if time.time() > d5:
            break
        hr = _resample(best_h, nc)
        sub_dl = min(d5, time.time() + time_per_res)
        hr, cr = _active_descent(hr, sub_dl, top_k=5, seed=seed + nc)
        if cr < best_c5:
            best_c5, best_h, best_n = cr, hr, nc
    
    # Phase 6: Intensive refinement at best resolution (15% budget)
    d6 = t0 + budget_s * 0.85
    h6, c56 = _lp_local(best_h, d6, radius=0.08)
    if c56 < best_c5:
        best_c5, best_h = c56, h6
    if time.time() < d6:
        sub_dl = min(d6, time.time() + budget_s * 0.04)
        h6b, c56b = _gradient_descent(best_h, sub_dl, tau=50.0, lr=0.01, beta=0.92)
        if c56b < best_c5:
            best_c5, best_h = c56b, h6b
    
    # Phase 7: Stochastic pair refinement (10% budget)
    d7 = t0 + budget_s * 0.95
    h7, c57 = _pair_refine(best_h, d7, seed=seed + 200)
    if c57 < best_c5:
        best_c5, best_h = c57, h7
    
    # Final: short LP polish
    if time.time() < t0 + budget_s * 0.99:
        h8, c58 = _lp_local(best_h, t0 + budget_s * 0.995, radius=0.03)
        if c58 < best_c5:
            best_c5, best_h = c58, h8
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
