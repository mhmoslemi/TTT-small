def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_target(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def corr_all(h):
    """Return all correlation values (unscaled)."""
    return np.correlate(h, 1.0 - h, mode='full')

    """Gradient of the q-th correlation value (unscaled) with respect to h."""
    n = len(h)
    d = q - (n - 1)
    grad = np.zeros(n)
    if d >= 0:

            # Second term: j in [d, n-1]

        # Second term: j in [0, n-1-k]

    """Solve min w·s s.t. 0<=s<=1, sum(s)=target. Closed form."""
    n = len(w)
    target = float(target)
    order = np.argsort(w, kind='stable')
    s = np.zeros(n)
    # Set the |target| smallest-weight coordinates to 1
    n_full = int(np.floor(target))
    frac = target - n_full
    s[order[:n_full]] = 1.0
    if frac > 1e-15:
        s[order[n_full]] = frac
    return s

def fw_step(h, n_lags, eta, alpha_grid=None):
    """One Frank-Wolfe step. Returns new h."""
    n = len(h)
    target = n / 2.0
    corr = corr_all(h)
    dx = 2.0 / n
    max_c = float(np.max(corr))
    
    # Select active lags
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:n_lags]
    
    # Softmax weights
    vals = corr[active]
    weights = np.exp((vals - vals[0]) / eta)
    weights /= weights.sum()
    
    # Combined gradient (scaled by dx)
    grad = np.zeros(n)
    for w, q in zip(weights, active):
        grad += w * corr_grad_q(h, q)
    grad *= dx  # scale to match objective
    
    # LP oracle: minimize grad · s
    s = lp_vertex(grad, target)
    
    # Line search
    if alpha_grid is None:
        alphas = np.linspace(0, 1, 33)
    else:
        alphas = alpha_grid
    
    best_alpha = 0.0
    best_val = exact_c5(h)
    h_orig_c5 = best_val
    
    for a in alphas:
        if a == 0:
            continue
        h_try = (1 - a) * h + a * s
        # Check feasibility
        if abs(float(h_try.sum()) - target) > 1e-10:
            h_try = project_to_target(h_try, target)
        v = exact_c5(h_try)
        if v < best_val - 1e-14:
            best_val = v
            best_alpha = a
            best_h_try = h_try.copy()
    
    if best_alpha > 0:
        return best_h_try, best_val
    return h, h_orig_c5

                    best_h = h_new3.copy()
                    h_cur = h_new3

import numpy as np
from scipy.optimize import minimize
import time


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_to_target(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def corr_all(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad_q(h, q):
    n = len(h)
    d = q - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            m1 = np.arange(d, n)
            grad[m1] += (1.0 - h[m1 - d])
            m2 = np.arange(0, n - d)
            grad[m2] -= h[m2 + d]
    else:
        j = -d
        if j < n:
            m1 = np.arange(0, n - j)
            grad[m1] += (1.0 - h[m1 + j])
            m2 = np.arange(j, n)
            grad[m2] -= h[m2 - j]
    return grad


def lp_vertex(w, target):
    n = len(w)
    target = float(target)
    order = np.argsort(w, kind='stable')
    s = np.zeros(n)
    n_full = int(np.floor(target))
    frac = target - n_full
    s[order[:n_full]] = 1.0
    if frac > 1e-15:
        s[order[n_full]] = frac
    return s


def fw_single_step(h, n_lags, eta, n_alpha=33):
    n = len(h)
    target = n / 2.0
    corr = corr_all(h)
    max_c = float(np.max(corr))
    if max_c <= 0:
        return h.copy(), exact_c5(h)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:min(n_lags, 2 * n - 1)]
    vals = corr[active]
    weights = np.exp((vals - vals[0]) / max(eta, 1e-10))
    weights /= weights.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    for w, q in zip(weights, active):
        grad += w * corr_grad_q(h, q)
    grad *= dx
    s = lp_vertex(grad, target)
    alphas = np.linspace(0, 1, n_alpha)
    best_val = exact_c5(h)
    best_h = h.copy()
    for a in alphas:
        if a < 1e-15:
            continue
        h_try = (1.0 - a) * h + a * s
        if abs(float(h_try.sum()) - target) > 1e-9:
            h_try = project_to_target(h_try, target)
        v = exact_c5(h_try)
        if v < best_val - 1e-14:
            best_val = v
            best_h = h_try.copy()
    return best_h, best_val


def fw_softmax_all_step(h, eta, n_alpha=33):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = corr_all(h)
    max_c = float(np.max(corr))
    if max_c <= 0:
        return h.copy(), exact_c5(h)
    weights = np.exp((corr - max_c) / max(eta, 1e-10))
    sw = weights.sum()
    if sw < 1e-30:
        s = lp_vertex(np.zeros(n), target)
    else:
        weights /= sw
        grad = np.zeros(n)
        for q in range(len(corr)):
            if weights[q] > 1e-10:
                grad += weights[q] * corr_grad_q(h, q)
        grad *= dx
        s = lp_vertex(grad, target)
    alphas = np.linspace(0, 1, n_alpha)
    best_val = exact_c5(h)
    best_h = h.copy()
    for a in alphas:
        if a < 1e-15:
            continue
        h_try = (1.0 - a) * h + a * s
        if abs(float(h_try.sum()) - target) > 1e-9:
            h_try = project_to_target(h_try, target)
        v = exact_c5(h_try)
        if v < best_val - 1e-14:
            best_val = v
            best_h = h_try.copy()
    return best_h, best_val


def fw_loop(h, deadline, max_iter=300):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if no_improve > 40:
            break
        frac_t = float(it) / max(max_iter, 1)
        lambda_curr = float(np.max(corr_all(h_cur)))
        if lambda_curr <= 0:
            break
        eta_top = max(0.04 - 0.03 * frac_t) * lambda_curr
        n_lags_top = 3 + int(4 * frac_t)
        h_new, c5_new = fw_single_step(h_cur, n_lags_top, eta_top, 25)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            no_improve = 0
            continue
        eta_all = max(0.02 - 0.01 * frac_t) * lambda_curr
        h_new2, c5_new2 = fw_softmax_all_step(h_cur, eta_all, 25)
        if c5_new2 < best_c5 - 1e-13:
            best_c5 = c5_new2
            best_h = h_new2.copy()
            h_cur = h_new2
            no_improve = 0
            continue
        if c5_new < exact_c5(h_cur) - 1e-13:
            h_cur = h_new
            no_improve = 0
        elif c5_new2 < exact_c5(h_cur) - 1e-13:
            h_cur = h_new2
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5


def penalty_grad_step(h, target_level, n_active, scales):
    n = len(h)
    t = n / 2.0
    dx = 2.0 / n
    corr = corr_all(h)
    curr_c5 = float(np.max(corr)) * dx
    if curr_c5 <= target_level:
        return h.copy(), curr_c5
    threshold = target_level / dx
    active = np.where(corr > threshold)[0]
    if len(active) == 0:
        active = np.argsort(corr)[::-1][:min(n_active, 2 * n - 1)]
    else:
        active = np.argsort(corr)[::-1][:min(len(active) + n_active, 2 * n - 1)]
    grad = np.zeros(n)
    for q in active:
        grad += corr_grad_q(h, q)
    grad *= dx
    grad /= max(len(active), 1)
    grad -= np.mean(grad)
    best_h = h.copy()
    best_val = curr_c5
    for sc in scales:
        h_try = project_to_target(h - sc * grad, t)
        v = exact_c5(h_try)
        if v < best_val - 1e-14:
            best_val = v
            best_h = h_try.copy()
    return best_h, best_val


def penalty_loop(h, deadline, n_active=8):
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.04]

            break

def pair_refine(h, deadline, max_passes=200):
    n = len(h)
    t = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        corr = corr_all(h)
        top_q = np.argsort(corr)[::-1][:6]
        grad_sum = np.zeros(n)
        for q in top_q:
            grad_sum += corr_grad_q(h, q)
        coords = np.argsort(np.abs(grad_sum))[::-1][:min(40, n)]
        for i in range(len(coords)):
            if time.time() > deadline:
                break
            ci = int(coords[i])
            hi_val = h[ci]
            if hi_val < 1e-6 or hi_val > 1.0 - 1e-6:
                continue
            for j_idx in range(i + 1, len(coords)):
                cj = int(coords[j_idx])
                hj_val = h[cj]
                if hj_val < 1e-6 or hj_val > 1.0 - 1e-6:
                    continue
                p = hi_val + hj_val
                lo = max(0.0, p - 1.0)
                hi = min(1.0, p)
                if hi - lo < 1e-10:
                    continue
                grid = np.linspace(lo, hi, 15)
                grid = np.clip(grid, 0.0, 1.0)
                for z in grid:
                    zj = p - z
                    if zj < -1e-12 or zj > 1.0 + 1e-12:
                        continue
                    z = max(0.0, min(1.0, z))
                    zj = max(0.0, min(1.0, zj))
                    h[ci] = z
                    h[cj] = zj
                    v = exact_c5(h)
                    if v < best_c5 - 1e-13:
                        best_c5 = v
                        best_h = h.copy()
                        improved = True
                    else:
                        h[ci] = hi_val
                        h[cj] = hj_val
                if improved:
                    hi_val = h[ci]
                    hj_val = h[cj]
                    grad_sum = np.zeros(n)
                    corr = corr_all(h)
                    for q in top_q:
                        grad_sum += corr_grad_q(h, q)
                    coords = np.argsort(np.abs(grad_sum))[::-1][:min(40, n)]
        if not improved:
            break
    return best_h, best_c5


def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def make_structured_h(n, period, phase=0.5, rng=None):
    x = np.arange(n) / n
    val = 0.5 + 0.45 * np.sin(2 * np.pi * period * x + phase)
    val = np.clip(val, 0.0, 1.0)
    return project_to_target(val, n / 2.0)


def make_block_h(n, period, duty=0.5, rng=None):
    x = np.arange(n)
    pattern = np.where((x % period) < int(period * duty), 0.9, 0.1)
    return project_to_target(pattern.astype(float), n / 2.0)


def get_active_set(g_full, frac=0.5, min_count=8):
    max_g = float(np.max(g_full))
    if max_g <= 0:
        return np.arange(len(g_full))
    threshold = max_g * frac
    active = np.where(g_full >= threshold)[0]
    if len(active) < min_count:
        k = min(min_count, len(g_full))
        active = np.argsort(g_full)[-k:]
    return active


def slsqp_solve(h, active_indices, target, maxiter=200, ftol=1e-12):
    n = len(h)

    def slp_obj(x):
        return x[0]

    def slp_obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def slp_ineq(x):
        h_local = x[1:]
        g = corr_all(h_local)
        return x[0] / (2.0 / n) - g[active_indices]

    def slp_ineq_jac(x):
        h_local = x[1:]
        J = np.zeros((len(active_indices), n + 1))
        J[:, 0] = 1.0 / (2.0 / n)
        dx = 2.0 / n
        for row, q in enumerate(active_indices):
            J[row, 1:] = -dx * corr_grad_q(h_local, q)
        return J

    def slp_eq(x):
        return x[1:].sum() - target

    def slp_eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    dx = 2.0 / n
    x0 = np.zeros(n + 1)
    x0[0] = float(np.max(corr_all(h))) * dx * 1.01 + 0.02
    x0[1:] = h
    constraints = [
        {'type': 'ineq', 'fun': slp_ineq, 'jac': slp_ineq_jac},
        {'type': 'eq', 'fun': slp_eq, 'jac': slp_eq_jac}
    ]
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(slp_obj, x0, jac=slp_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = project_to_target(h_new, target)
            return h_new
    except Exception:
        pass
    return h


def run(seed=42, budget_s=1100, **kwargs):

            if best_c5 < 0.3805:

    n_cur = len(best_h)
    if time.time() < d6:
        for dn in [2, -2, 4, -4, 6, -6, 8, -8]:
            if time.time() > d6:
                break
            n_try = n_cur + dn
            if n_try < 20 or n_try > 120:
                continue
            h_r = resample(best_h, n_try)
            c5_r = exact_c5(h_r)
            # Quick refinement at new resolution
            sub_dl = min(d6, time.time() + budget_s * 0.015)
            h_r, c5_r2 = fw_loop(h_r, sub_dl, max_iter=30)
            c5_r = min(c5_r, c5_r2)
            if c5_r < best_c5 - 1e-9:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_cur = n_try

    # Phase 7: Final polish (3% budget)
    d7 = deadline
    if time.time() < d7:
        n_final = len(best_h)
        t_final = n_final / 2.0
        h_f, c5_f = pair_refine(best_h, d7, max_passes=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # Final penalty step
        if time.time() < d7 - 1:
            h_pf, c5_pf = penalty_loop(best_h, d7 - 0.5, n_active=6)
            if c5_pf < best_c5:
                best_c5 = c5_pf
                best_h = h_pf.copy()

    # Final validation
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # recovered from projection, use pre-projection value if valid
        pass
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, len(best_h)
