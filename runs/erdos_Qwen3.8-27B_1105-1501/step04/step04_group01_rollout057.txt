from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 100
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += -h[d:]
                G[r, d:] += (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += -h[:m]
                G[r, :m] += (1.0 - h[g:])
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 2
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.82
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.08
    return best_h, best_c5

def _softmax_dir(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(corr))
    exp_s = np.exp(tau * (corr - max_s))
    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _grad_opt(h, deadline, tau_list=(40, 100, 250, 600), lr_init=0.018):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.83
    stall = 0
    ti = 0
    while time.time() < deadline:
        if stall > 50:
            break
        tau = tau_list[ti % len(tau_list)]
        d = _softmax_dir(h, tau)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            ti += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.0)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.04, 0.25)
        else:
            stall += 1
            lr *= 0.84
            if stall % 20 == 0:
                ti += 1
                mom[:] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = 2 if _pass > 20 else 1
        for i in range(0, n, step):
            if time.time() > deadline:
                break
            hi_val = h[i]
            if hi_val <= 1e-10 or hi_val >= 1.0 - 1e-10:
                continue
            for j in range(i + step, n, step):
                hj_val = h[j]
                if hj_val <= 1e-10 or hj_val >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved and _pass > 5:
            break
    return best_h, best_c5

def _perturb_optimize(h_best, deadline, rng, scale_init=0.03):
    n = len(h_best)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = scale_init
    attempts = 0
    while time.time() < deadline and attempts < 200:
        attempts += 1
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, n / 2.0)
        c5_p = _exact_c5(h_p)
        sub_dl = min(deadline, time.time() + 8.0)
        h_t, c5_t = _lp_optimize(h_p, sub_dl, radius_init=0.06)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.12, scale * 1.15)
    return best_h, best_c5
