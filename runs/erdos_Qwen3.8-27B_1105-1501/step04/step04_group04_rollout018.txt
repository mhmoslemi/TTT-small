                            h[tgt]+=c2; h[src]-=c2; f2=_c5(h); h[tgt]-=c2; h[src]+=c2

                    h[tgt]+=best_t; h[src]-=best_t
                    ft = _c5(h)
                    h[tgt]-=best_t; h[src]+=best_t
                    if ft < best_tc:
                        best_tc = ft
                if best_tc < cur_c5 - 1e-13 and abs(best_t) > 1e-8:

                    h[src] -= best_t

                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    no_imp = 0
                    imp_found = True
                    break

                no_imp += 1
