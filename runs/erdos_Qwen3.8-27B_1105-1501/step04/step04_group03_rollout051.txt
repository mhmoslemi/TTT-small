    target = n / 2.0
    
    def obj_fn(x):
        v, _ = soft_value_grad(x, tau)
        return v

        _, g = soft_value_grad(x, tau)
        return g

        return np.sum(x) - target

        return np.ones(n)

def soft_value_grad(h, tau):
    n = len(h)
    scores = all_scores(h)
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    G = build_corr_grad(h)
    grad = G.T @ w
    return val, grad


def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0

    def obj_fn(x):
        v, _ = soft_value_grad(x, tau)
        return v

    def grad_fn(x):
        _, g = soft_value_grad(x, tau)
        return g

    def constr_fun(x):
        return np.sum(x) - target

    def constr_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

                    h_t[j] += sign * amax
                    h_t[k] -= sign * amax
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if improved:
            continue
        # Fractional moves with binary search
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    # Try a few fractions
                    for frac in [0.5, 0.25, 0.75, 0.125, 0.875, 0.375, 0.625]:
                        a = amax * frac
                        h_t = best_h.copy()
                        h_t[j] += sign * a
                        h_t[k] -= sign * a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def coordinate_sweep(h, deadline, n_rounds=5):
    """Try small perturbations of individual coordinates."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    deltas = [0.01, 0.02, 0.05, 0.1, 0.2]
    for _r in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for d in deltas:
                for sign in [1.0, -1.0]:
                    new_val = best_h[i] + sign * d
                    if new_val < -1e-10 or new_val > 1.0 + 1e-10:
                        continue
                    # Need to compensate elsewhere to maintain sum
                    # Find a coordinate with room
                    comp = 0
                    for j in range(n):
                        if j == i:
                            continue
                        if sign > 0 and best_h[j] > 1e-10:
                            comp = j
                            break
                        elif sign < 0 and best_h[j] < 1.0 - 1e-10:
                            comp = j
                            break
                    if comp == 0 and i == 0:
                        continue
                    h_t = best_h.copy()
                    h_t[i] = new_val
                    h_t[comp] -= sign * d
                    if h_t[comp] < -1e-10 or h_t[comp] > 1.0 + 1e-10:
                        continue
                    h_t = np.clip(h_t, 0.0, 1.0)
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Momentum-based gradient descent (fast, new)
    d1 = t0 + budget_s * 0.12
    h_gd, c5_gd = momentum_descent(h, d1, lr_init=0.03, mom=0.95, top_k=4)
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd.copy()

    # Phase 2: LP local search
    d2 = t0 + budget_s * 0.28
    h_lp, c5_lp = lp_loop(best_h, d2, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: SLSQP with decreasing tau
    d3 = t0 + budget_s * 0.42
    tau_list = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    h_slsqp = best_h.copy()
    for tau in tau_list:
        if time.time() > d3:
            break
        h_slsqp = slsqp_smooth(h_slsqp, tau, 70)
        h_slsqp = project(h_slsqp, best_n / 2.0)
        c5_s = exact_c5(h_slsqp)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_slsqp.copy()

    # Phase 4: Second LP pass
    d4 = t0 + budget_s * 0.52
    h_lp2, c5_lp2 = lp_loop(best_h, d4, radius_init=0.08)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 5: Diversified restarts with mixed strategies
    d5 = t0 + budget_s * 0.70
    scales = [0.02, 0.04, 0.06, 0.09, 0.13, 0.18, 0.25, 0.35, 0.50, 0.70, 1.0]
    for ns in scales:
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        # Strategy A: momentum
        if time.time() < d5:
            sub_dl = min(d5, time.time() + budget_s * 0.012)
            h_try, c5_try = momentum_descent(h_pert, sub_dl, lr_init=0.02, mom=0.93, top_k=3)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
        # Strategy B: LP
        if time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.012)
            h_try2, c5_try2 = lp_loop(h_pert, sub_dl2, radius_init=0.06)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()
        # Strategy C: SLSQP quick
        if time.time() < d5:
            h_try3 = slsqp_smooth(h_pert, 0.1, 40)
            h_try3 = project(h_try3, best_n / 2.0)
            c5_try3 = exact_c5(h_try3)
            if c5_try3 < best_c5:
                best_c5 = c5_try3
                best_h = h_try3.copy()

    # Phase 6: Multi-resolution search
    d6 = t0 + budget_s * 0.82
    tried_n = set([best_n])
    for dn in [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -32, 32, -40, 40, -48, 48, -56, 64]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.007)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.05)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Pair polish (improved with fractional steps)
    d7 = t0 + budget_s * 0.91
    h_pol, c5_pol = pair_polish_v2(best_h, d7, max_passes=100)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 8: Coordinate sweep
    if time.time() < hard_deadline:
        d8 = min(hard_deadline, time.time() + budget_s * 0.02)
        h_cs, c5_cs = coordinate_sweep(best_h, d8, n_rounds=10)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs.copy()

    # Phase 9: Final LP micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.02)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final pair micro-polish
    if time.time() < hard_deadline:
        h_fp, c5_fp = pair_polish_v2(best_h, hard_deadline, max_passes=30)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
