import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _g_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    gi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + gi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            gi = mid
    mu = (lo + gi) / 2.0
    p = np.clip(v - mu, 0.0, 1.0)
    return p


def _compute_C(h):
    """Precompute coefficient matrix C[k,j] = dg_k/dh_j"""
    n = len(h)
    dx = 2.0 / n
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j + d < n:
                val += (1.0 - h[j + d])
            if 0 <= j - d < n:
                val -= h[j - d]
            C[k, j] = dx * val
    return C


def _smooth_max_and_grad(h, alpha):
    """Smooth max of g_k(h) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    g = _g_raw(h) * dx
    gm = np.max(g)
    eg = np.exp(alpha * (g - gm))
    w = eg / np.sum(eg)
    obj = gm + np.log(np.sum(eg)) / alpha
    # Gradient via coefficient matrix
    C = _compute_C(h)
    grad = C.T @ w
    return obj, grad


def _pair_search(h, i, j, g, C, n_grid=80):
    """Optimize along pair (i,j) keeping sum constant. Returns best t for h[i]."""
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h[i]
    delta_C = C[:, i] - C[:, j]
    base = h[i]
    ds = np.linspace(0, hi - base, n_grid)
    best_t = base
    best_val = np.max(g)
    for dd in ds:
        t = base + dd
        g_new = g + dd * delta_C
        val = np.max(g_new)
        if val < best_val - 1e-14:
            best_val = val
            best_t = t
    # Refine with finer grid around best
    margin = (hi - lo) / n_grid
    lo2 = max(lo, best_t - 2 * margin)
    hi2 = min(hi, best_t + 2 * margin)
    if hi2 - lo2 > 1e-12:
        ds2 = np.linspace(lo2 - base, hi2 - base, 40)
        for dd in ds2:
            t = base + dd
            g_new = g + dd * delta_C
            val = np.max(g_new)
            if val < best_val - 1e-14:
                best_val = val
                best_t = t
    return best_t


def _coord_descent(h, deadline, max_passes=50, pair_subset=None):
    """Coordinate descent with exact pair line searches."""
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    g = _g_raw(h)
    C = _compute_C(h)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Get pairs to try
        if pair_subset is not None:
            pairs = [(i, j) for i, j in pair_subset]
        else:
            # Focus on pairs involving high-contribution indices
            top_lags = np.argsort(g)[-min(8, 2 * n - 1) :]
            active_idx = set()
            for k in top_lags:
                d = k - (n - 1)
                if d >= 0:
                    for i in range(n - d):
                        if h[i] * (1 - h[i + d]) > 0.05:
                            active_idx.add(i)
                            active_idx.add(i + d)
                else:

                    for i in range(dp, n):
                        if h[i] * (1 - h[i - dp]) > 0.05:
                            active_idx.add(i)
                            active_idx.add(i - dp)
            idx_list = sorted(active_idx)
            if len(idx_list) < 4:
                idx_list = list(range(n))
            # Generate pairs from active indices + some random
            np.random.seed(p * 7919 + 3)
            pairs = []
            for ii in range(len(idx_list)):
                for jj in range(ii + 1, len(idx_list)):
                    pairs.append((idx_list[ii], idx_list[jj]))
            # Add some random pairs
            for _ in range(min(50, n)):
                a, b = np.random.choice(n, 2, replace=False)
                pairs.append((int(a), int(b)))
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            t_new = _pair_search(h, i, j, g, C, n_grid=64)
            if abs(t_new - h[i]) > 1e-10:

                h[j] = h[i] + h[j] - t_new
                h = np.clip(h, 0.0, 1.0)
                # Re-sum fix

                if abs(residual) > 1e-12:
                    h = _project(h, n / 2.0)
                    t_new_check = h[i]
                    h[j] = h[i] + h[j] - (t_new_check - (h[i] + h[j]))
                    h = _project(h, n / 2.0)

                c5 = _exact_c5(h)

def _pg_optimize(h, deadline, alpha_start=20.0, alpha_end=200.0, n_alphas=8):
    """Projected gradient with smooth max continuation."""
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    
    alphas = np.linspace(alpha_start, alpha_end, n_alphas)
    lr = 0.01
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        for it in range(200):
            if time.time() > deadline:
                break
            obj, grad = _smooth_max_and_grad(h, alpha)
            # Project gradient: remove mean to preserve sum
            grad_centered = grad - np.mean(grad)
            # Adaptive step size
            h_new = _project(h - lr * grad_centered, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.15, 0.05)
            elif c5_new < _exact_c5(h) + 1e-10:
                h = h_new
                lr = max(lr * 0.85, 1e-5)
            else:
                lr = max(lr * 0.7, 1e-6)
                if lr < 1e-5:
                    break
    
    return best_h, best_c5


def _diverse_starts(n, rng):
    """Generate diverse initial candidates of length n."""
    candidates = []
    target = n / 2.0
    
    # 1. Uniform
    candidates.append(np.full(n, 0.5))
    
    # 2. Alternating binary
    alt = np.zeros(n)
    alt[::2] = 1.0
    if np.sum(alt) != target:
        alt[1::2][:int(abs(target - np.sum(alt)))] = 1.0 if np.sum(alt) < target else 0.0
    if abs(np.sum(alt) - target) < 1e-10:
        candidates.append(alt.copy())
    
    # 3. Low-discrepancy (golden ratio)
    phi = (1 + np.sqrt(5)) / 2
    positions = np.unique(np.floor(np.arange(int(target)) * n / phi % n).astype(int))
    if len(positions) == int(target):
        mask = np.zeros(n)
        mask[positions] = 1.0
        candidates.append(mask.copy())
    
    # 4. Block structure
    for nblocks in [2, 3, 4, 5]:
        if n % nblocks == 0:
            block_size = n // nblocks
            blk = np.zeros(n)
            for b in range(nblocks):
                if b % 2 == 0:
                    blk[b * block_size:(b + 1) * block_size] = 1.0
            if abs(np.sum(blk) - target) < 1e-10:
                candidates.append(blk.copy())
    
    # 5. Sine structured
    for k in [2, 3, 5, 7, 11]:
        x = np.linspace(0, 1, n, endpoint=False)
        s = 0.5 + 0.3 * np.sin(2 * np.pi * k * x)
        s = _project(s, target)
        candidates.append(s.copy())
    
    # 6. Random balanced
    for _ in range(5):
        mask = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        mask[idx] = 1.0
        # Blur slightly
        blurred = mask * 0.7 + 0.15
        blurred = _project(blurred, target)
        candidates.append(blurred.copy())
        candidates.append(mask.copy())
    
    # 7. Two-level from random mask
    for _ in range(3):
        mask = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        mask[idx] = 1.0
        d = rng.uniform(0.1, 0.4)
        two_level = np.where(mask == 1, 1 - d, d)
        if abs(np.sum(two_level) - target) < 1e-10:
            candidates.append(two_level.copy())
    
    # Project all
    result = []
    for c in candidates:
        c = _project(c, target)
        if np.all(np.isfinite(c)) and c.min() >= -1e-12 and c.max() <= 1 + 1e-12:
            c = np.clip(c, 0, 1)
            c = _project(c, target)
            result.append(c)
    return result


def _two_level_sweep(mask, n_grid=200):
    """Sweep the two-level family for a given binary mask."""
    n = len(mask)
    target = n / 2.0
    m = int(np.sum(mask))
    if m != target:
        return None, None
    best_d = 0.5
    best_c5 = float('inf')
    ds = np.linspace(0.001, 0.499, n_grid)
    for d in ds:
        h = np.where(mask == 1, 1.0 - d, d)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_d = d
    # Refine
    margin = 1.0 / n_grid
    lo = max(0.0, best_d - 2 * margin)
    hi = min(0.5, best_d + 2 * margin)
    for d in np.linspace(lo, hi, 50):
        h = np.where(mask == 1, 1.0 - d, d)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_d = d
    h_best = np.where(mask == 1, 1.0 - best_d, best_d)
    h_best = _project(h_best, target)
    return h_best, best_c5


def _optimize_at_resolution(h_start, deadline):
    """Full optimization pipeline for a single resolution."""
    n = len(h_start)
    h = _project(h_start.copy(), n / 2.0)
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    
    # Phase 1: Projected gradient with smooth max
    d1 = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    h1, c5_1 = _pg_optimize(h, d1, alpha_start=10, alpha_end=150, n_alphas=6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Coordinate descent
    d2 = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    h2, c5_2 = _coord_descent(best_h, d2, max_passes=30)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Another PG pass
    d3 = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    if time.time() < d3:
        h3, c5_3 = _pg_optimize(best_h, d3, alpha_start=50, alpha_end=300, n_alphas=5)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    
    # Phase 4: Final coordinate descent
    if time.time() < deadline:
        h4, c5_4 = _coord_descent(best_h, deadline, max_passes=20)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    return best_h, best_c5


def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 64
    
    # Load parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
        h_parent = _project(h_parent, n_parent / 2.0)
        best_h = h_parent.copy()
        best_c5 = _exact_c5(h_parent)
        best_n = n_parent
    except Exception:
        n_parent = 64
        h_parent = _project(np.full(n_parent, 0.5), n_parent / 2.0)
        best_h = h_parent.copy()
        best_c5 = _exact_c5(h_parent)
        best_n = n_parent
    
    # Phase 1: Optimize at parent resolution (40% budget)
    d1 = t0 + budget_s * 0.40
    h_opt, c5_opt = _optimize_at_resolution(h_parent, d1)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
    
    # Phase 2: Diverse starts at parent resolution (15% budget)
    d2 = t0 + budget_s * 0.55
    if time.time() < d2:
        starts = _diverse_starts(best_n, rng)
        # Also try two-level from parent threshold
        mask_p = (h_parent >= np.median(h_parent))
        # Ensure exactly n/2 ones
        if np.sum(mask_p) > best_n / 2:
            order = np.argsort(-h_parent)
            mask_p = np.zeros(best_n)
            mask_p[order[:best_n // 2]] = 1
        elif np.sum(mask_p) < best_n / 2:
            order = np.argsort(-h_parent)
            mask_p = np.zeros(best_n)
            mask_p[order[:best_n // 2]] = 1
        starts.append(np.where(mask_p == 1, 0.8, 0.2))
        
        for s in starts[:12]:
            if time.time() > d2:
                break
            s = _project(s, best_n / 2.0)
            sub_deadline = min(d2, time.time() + budget_s * 0.015)
            h_try, c5_try = _optimize_at_resolution(s, sub_deadline)
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()
        
        # Two-level sweep on best binary structure
        if time.time() < d2:
            mask_best = np.zeros(best_n)
            mask_best[np.argsort(-best_h)[:best_n // 2]] = 1
            h_tl, c5_tl = _two_level_sweep(mask_best, n_grid=300)
            if h_tl is not None and c5_tl < best_c5:
                best_c5 = c5_tl
                best_h = h_tl.copy()
                # Optimize from there
                sub_dl = min(d2, time.time() + budget_s * 0.02)
                h_tl2, c5_tl2 = _optimize_at_resolution(h_tl, sub_dl)
                if c5_tl2 < best_c5:
                    best_c5 = c5_tl2
                    best_h = h_tl2.copy()
    
    # Phase 3: Resolution search (35% budget)
    d3 = t0 + budget_s * 0.90
    if time.time() < d3:
        resolutions = []
        for dn in [-16, -8, -12, -4, 4, 8, 12, 16, -24, 24]:
            nr = best_n + dn
            if 24 <= nr <= 192 and nr % 2 == 0:
                resolutions.append(nr)
        # Deduplicate and sort by distance from best_n
        resolutions = sorted(set(resolutions), key=lambda x: abs(x - best_n))
        
        for nr in resolutions[:8]:
            if time.time() > d3:
                break
            h_r = _resample(best_h, nr)
            sub_deadline = min(d3, time.time() + budget_s * 0.03)
            h_r2, c5_r2 = _optimize_at_resolution(h_r, sub_deadline)
            if c5_r2 < best_c5 - 1e-10:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = nr
                # If we found improvement, give more time to this resolution
                if time.time() < d3:
                    sub_dl2 = min(d3, time.time() + budget_s * 0.03)
                    h_r3, c5_r3 = _optimize_at_resolution(h_r2, sub_dl2)
                    if c5_r3 < best_c5:
                        best_c5 = c5_r3
                        best_h = h_r3.copy()
    
    # Phase 4: Multi-start refinement on best (10% budget)
    d4 = deadline
    if time.time() < d4:
        scale = 0.03
        for attempt in range(30):
            if time.time() > d4:
                break
            h_p = best_h + rng.normal(0, scale, best_n)
            h_p = _project(h_p, best_n / 2.0)
            sub_dl = min(d4, time.time() + budget_s * 0.012)
            h_try, c5_try = _optimize_at_resolution(h_p, sub_dl)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.005, scale * 0.7)
            else:
                scale = min(0.12, scale * 1.3)
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    # Ensure feasibility
    assert np.all(best_h >= -1e-10), "h has negative values"
    assert np.all(best_h <= 1 + 1e-10), "h exceeds 1"
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6, "sum constraint violated"
    
    return best_h, final_c5, best_n
