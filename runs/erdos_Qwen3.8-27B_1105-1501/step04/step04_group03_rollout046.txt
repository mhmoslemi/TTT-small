def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def grad_of_shift(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
    else:
        g2 = -d
        m = n - g2
        if m > 0:
            g[g2:] += (1.0 - h[:m])
            g[:m] -= h[g2:]
    return g

def topk_lp_direction(h, scores, top_k, radius):
    n = len(h)
    top_indices = np.argsort(scores)[::-1][:top_k]
    m = len(top_indices)
    G = np.zeros((m, n))
    for i, idx in enumerate(top_indices):
        d = int(idx) - (n - 1)
        if d >= 0:
            mm = n - d
            if mm > 0:
                G[i, :mm] += (1.0 - h[d:])
                G[i, d:] += -h[:mm]
        else:
            g2 = -d
            mm = n - g2
            if mm > 0:
                G[i, g2:] += (1.0 - h[:mm])
                G[i, :mm] += -h[g2:]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -scores[top_indices]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def subgrad_loop(h, deadline, top_k=4, base_alpha=0.06, momentum=0.9):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = base_alpha
    v = np.zeros(n)
    
    while time.time() < deadline:
        scores = compute_scores(h) * dx
        max_s = np.max(scores)
        top_indices = np.argsort(scores)[::-1][:top_k]
        top_scores = scores[top_indices]
        tau = 0.015
        weights = np.exp((top_scores - top_scores.max()) / tau)
        weights /= weights.sum()
        g = np.zeros(n)
        for idx, w in zip(top_indices, weights):
            d = int(idx) - (n - 1)
            g += w * grad_of_shift(h, d)
        v = momentum * v + (1 - momentum) * g
        h_new = project(h - alpha * dx * v, target)
        new_c5 = exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            alpha = min(alpha * 1.03, 0.6)
        else:
            alpha *= 0.95
        if alpha < 1e-9:
            break
    return best_h, best_c5

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        dx = 2.0 / n
        scores = np.correlate(x, 1.0 - x, mode='full') * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        return max_s + tau * np.log(np.sum(exp_s))
    
    def grad(x):
        dx = 2.0 / n
        scores = np.correlate(x, 1.0 - x, mode='full') * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        Z = np.sum(exp_s)
        w = exp_s / Z
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            d = k - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[:m] += w[k] * (1.0 - x[d:])
                    g[d:] -= w[k] * x[:m]
            else:
                g2 = -d
                m = n - g2
                if m > 0:
                    g[g2:] += w[k] * (1.0 - x[:m])
                    g[:m] -= w[k] * x[g2:]
        return dx * g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def block_perturb(h, rng, scale):
    n = len(h)
    target = n / 2.0
    h_new = h.copy()
    n_blocks = rng.integers(2, 6)

    best_n = n0
    
    # Phase 1: Adaptive top-K LP
    d1 = t0 + budget_s * 0.16
    h1, c5_1 = lp_loop_adaptive(h, d1, k_start=8, k_end=50, radius_init=0.18)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Subgradient with momentum
    d2 = t0 + budget_s * 0.28
    h2, c5_2 = subgrad_loop(best_h, d2, top_k=4, base_alpha=0.08)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: LP refinement
    d3 = t0 + budget_s * 0.42
    h3, c5_3 = lp_loop_adaptive(best_h, d3, k_start=15, k_end=60, radius_init=0.12)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: SLSQP smooth
    d4 = t0 + budget_s * 0.58
    tau_list = [0.35, 0.18, 0.09, 0.05, 0.025, 0.012, 0.006]
    iter_list = [90, 85, 75, 65, 55, 45, 35]
    h_s = best_h.copy()
    for i in range(len(tau_list)):
        if time.time() > d4:
            break
        h_s = slsqp_smooth(h_s, tau_list[i], iter_list[i])
        h_s = project(h_s, best_n / 2.0)
        c5_s = exact_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 5: Final LP
    d5 = t0 + budget_s * 0.70
    h5, c5_5 = lp_loop_adaptive(best_h, d5, k_start=20, k_end=80, radius_init=0.1)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Diversified restarts with block perturbations
    d6 = t0 + budget_s * 0.83
    for round_i in range(12):
        if time.time() > d6:
            break
        for method in ['gauss', 'block']:
            if time.time() > d6:
                break
            ns = 0.03 + 0.02 * round_i
            if method == 'gauss':
                h_pert = best_h + rng.normal(0, ns, best_n)
            else:
                h_pert = block_perturb(best_h, rng, ns * 0.8)
            h_pert = project(h_pert, best_n / 2.0)
            # Quick LP
            sub_dl = min(d6, time.time() + budget_s * 0.01)
            h_try, c5_try = lp_loop_adaptive(h_pert, sub_dl, k_start=10, k_end=40, radius_init=0.1)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Phase 7: Multi-resolution
    d7 = t0 + budget_s * 0.91
    for dn in [-8, 8, -16, 16, -24, 24, -40, 40, -60, 60]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 250:
            continue
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.012)
        h_r, c5_r = lp_loop_adaptive(h_r, sub_dl, k_start=10, k_end=40, radius_init=0.1)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 8: Pair polish
    d8 = t0 + budget_s * 0.955
    h_pol, c5_pol = pair_polish(best_h, d8, max_passes=120)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    
    # Final micro-LP if time remains
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop_adaptive(best_h, hard_deadline, k_start=30, k_end=80, radius_init=0.04)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    
    # Safety: never worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0
    
    return best_h, final_c5, best_n
