import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    G *= dx
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search_c5(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:

    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur

        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section
    if best_a < alpha_max * 0.05:
        lo_a, hi_a = 0.0, min(alpha_max, best_a * 3.0)
    elif best_a > alpha_max * 0.95:
        lo_a, hi_a = max(0.0, best_a * 0.7), alpha_max
    else:
        lo_a = max(0.0, best_a * 0.7)
        hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)

    return best_a, best_c5

def subgradient_search(h, deadline, n_iter=5000):
    """Nesterov-style subgradient method with averaging."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    y = h.copy()  # Nesterov momentum point
    best_h = h.copy()
    best_c5 = exact_c5(h)
    beta = 1.0  # momentum
    
    for t in range(1, n_iter + 1):
        if time.time() > deadline:
            break
        step = 2.0 / (1.0 + t)  # diminishing step size (scaled)
        # Compute scores at y
        scores = compute_scores(y)
        k_max = int(np.argmax(scores))
        d = k_max - (n - 1)
        # Gradient of f_k at y
        g = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += (1.0 - y[:m])
                g[:m] -= y[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += (1.0 - y[gg:])
                g[gg:] -= y[:m]
        # Update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = (beta * y - step * g) / beta_new
        y = project(y, target)
        h = (t * h + y) / (t + 1.0)  # averaging
        h = project(h, target)
        c5_h = exact_c5(h)
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h.copy()
        beta = beta_new
    
    return best_h, best_c5

def binary_local_search(h, deadline, max_moves=20000):
    """Combinatorial local search: swap pairs to improve C5."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = exact_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    # Make it truly binary first
    h_bin = np.zeros(n)
    h_bin[ones] = 1.0
    h = h_bin
    if np.sum(h) != target:
        # Need to adjust
        diff = target - np.sum(h)
        if diff > 0:
            # Need more ones: flip some zeros to ones
            for _ in range(int(diff)):
                if len(zeros) == 0:
                    break
                idx = np.random.choice(zeros)
                h[idx] = 1.0
                zeros = zeros[zeros != idx]
        else:
            # Need fewer ones: flip some ones to zeros
            for _ in range(int(-diff)):
                if len(ones) == 0:
                    break
                idx = np.random.choice(ones)
                h[idx] = 0.0
                ones = ones[ones != idx]
    
    cur_c5 = exact_c5(h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    else:
        best_h = h.copy()
    
    no_improve = 0
    moves = 0
    while time.time() < deadline and moves < max_moves:
        if no_improve > 500:
            break
        moves += 1
        # Random pair (one from ones, one from zeros)
        if len(ones) == 0 or len(zeros) == 0:
            break
        j = np.random.choice(ones)
        k = np.random.choice(zeros)
        h[j] = 0.0
        h[k] = 1.0
        ones = ones[ones != j]
        zeros = zeros[zeros != k]
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[j] = 1.0
            h[k] = 0.0
            ones = np.append(ones, j)
            zeros = np.append(zeros, k)
            no_improve += 1
    
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                amax_pos = min(best_h[j], 1.0 - best_h[k])
                amax_neg = min(1.0 - best_h[j], best_h[k])
                # Try moving mass from j to k
                if amax_pos > 1e-10:
                    h_t = best_h.copy()
                    h_t[j] -= amax_pos
                    h_t[k] += amax_pos
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t
                        improved = True
                # Try moving mass from k to j
                if amax_neg > 1e-10 and best_h is not h_t:
                    h_t = best_h.copy()
                    h_t[k] -= amax_neg
                    h_t[j] += amax_neg
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
