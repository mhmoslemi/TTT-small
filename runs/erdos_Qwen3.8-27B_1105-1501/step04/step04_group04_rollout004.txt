def exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0).copy()
    for _ in range(100):

def grad_F(h, s):
    """
    Compute gradient of F_s(h) = dx * sum_i h_{i+s} * (1-h_i)
    with respect to h.
    
    For shift s (in range [-(n-1), n-1]):
    F_s = dx * sum_{i=min}^{max-1} h_{i+s} * (1-h_i)
    
    dF_s/dh_a = dx * sum over terms where a appears as i+s (coefficient: (1-h_i))
              - dx * sum over terms where a appears as i (coefficient: h_{i+s})
    """
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    
    i_min = max(0, -s)
    i_max = min(n, n - s)
    
    if i_min >= i_max:
        return g
    
    # a = i + s, so i = a - s
    # a ranges from i_min + s to i_max - 1 + s
    a_low = i_min + s
    a_high = i_max - 1 + s
    
    # First term: sum over i where a = i+s, coefficient (1-h_i)
    # i = a - s, so for a in [a_low, a_high-1], coefficient is (1 - h[a-s])
    # But we need i in [i_min, i_max-1], which means a-s in [i_min, i_max-1]
    # a in [i_min+s, i_max-1+s] = [a_low, a_high-1]
    
    # Second term: sum over i where a = i, coefficient h_{i+s}
    # a = i, so i = a, need i in [i_min, i_max-1], so a in [i_min, i_max-1]
    
    # First term contribution to g[a]: dx * (1 - h[a-s]) for a where i=a-s is valid
    # Second term contribution to g[a]: -dx * h[a+s] for a where a is valid as i
    
    # For first term: a-s must be in [i_min, i_max-1] AND a must be in [0, n-1]
    # For second term: a must be in [i_min, i_max-1] AND a+s must be in [0, n-1]
    
    # Let me be more careful.
    # F_s = dx * sum_{i=i_min}^{i_max-1} h[i+s] * (1-h[i])
    
    # dF_s/dh[a] = dx * [sum over i in range where i+s==a of (1-h[i]) 
    #                    - sum over i in range where i==a of h[i+s]]
    
    # First: i+s==a => i=a-s. Valid if i in [i_min, i_max-1] and a in [0,n-1]
    #   => a-s in [i_min, i_max-1] => a in [i_min+s, i_max-1+s]
    #   AND a in [0, n-1]
    # Second: i==a. Valid if a in [i_min, i_max-1] and a+s in [0,n-1]
    
    # First term
    a1_lo = max(0, i_min + s)
    a1_hi = min(n, i_max + s)  # exclusive
    if a1_lo < a1_hi:
        idx1 = np.arange(a1_lo, a1_hi)
        g[idx1] += dx * (1.0 - h[idx1 - s])
    
    # Second term
    a2_lo = max(0, i_min)
    a2_hi = min(n, i_max)  # exclusive
    if a2_lo < a2_hi:
        idx2 = np.arange(a2_lo, a2_hi)
        g[idx2] -= dx * h[idx2 + s]
    
    return g

def active_weighted_grad(h, beta, K):
    """Compute weighted gradient of top-K active constraints."""
    scores = raw_scores(h)
    n = len(h)
    L = 2 * n - 1
    max_s = scores.max()
    
    # Soft weights
    shifted = scores - max_s
    w = np.exp(beta * shifted)
    w = w / w.sum()
    
    # Focus on top K
    top_k_idx = np.argsort(scores)[-K:]
    w_mod = np.zeros(L)
    w_mod[top_k_idx] = w[top_k_idx]

import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def raw_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0).copy()
    for _ in range(100):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def grad_F(h, s):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    i_min = max(0, -s)
    i_max = min(n, n - s)
    if i_min >= i_max:
        return g
    a1_lo = max(0, i_min + s)
    a1_hi = min(n, i_max + s)
    if a1_lo < a1_hi:
        idx = np.arange(a1_lo, a1_hi)
        g[idx] += dx * (1.0 - h[idx - s])
    a2_lo = max(0, i_min)
    a2_hi = min(n, i_max)
    if a2_lo < a2_hi:
        idx = np.arange(a2_lo, a2_hi)
        g[idx] -= dx * h[idx + s]
    return g

def line_search(h, v, n_samples=15, refine=3):
    n = len(h)
    v = v - np.mean(v)
    if np.max(np.abs(v)) < 1e-14:
        return 0.0, exact_c5(h), h.copy()
    v = v / np.max(np.abs(v))
    pos = v > 0
    neg = v < 0
    t_min = -1e6
    t_max = 1e6
    if np.any(pos):
        t_min = -np.min(h[pos] / v[pos])
    if np.any(neg):
        t_max = -np.min(h[neg] / v[neg])
    t_min = max(t_min, -2.0)
    t_max = min(t_max, 2.0)
    t_min = min(t_min, -1e-10)
    t_max = max(t_max, 1e-10)
    if t_max - t_min < 1e-12:
        return 0.0, exact_c5(h), h.copy()
    t_min += 1e-13
    t_max -= 1e-13
    if t_min >= t_max:
        return 0.0, exact_c5(h), h.copy()
    cur_c5 = exact_c5(h)
    best_t = 0.0
    best_c5 = cur_c5
    ts = np.concatenate([np.linspace(t_min, t_max, n_samples), [0.0]])
    ts = np.unique(ts)
    best_idx = 0
    best_s = cur_c5
    for k, t in enumerate(ts):
        trial = h + t * v
        if np.any(trial < -0.01) or np.any(trial > 1.01):
            continue
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_box_sum(trial, n / 2.0)
        s_val = exact_c5(trial)
        if s_val < best_s:
            best_s = s_val
            best_idx = k
            best_t = t
    best_c5 = best_s
    for _ in range(refine):
        lo = max(0, best_idx - 2)
        hi = min(len(ts), best_idx + 3)
        if hi - lo < 3:
            lo = max(0, lo - 2)
            hi = min(len(ts), hi + 3)
        t_lo = ts[lo]
        t_hi = ts[hi - 1]
        if t_hi - t_lo < 1e-11:
            break
        ts_f = np.linspace(t_lo, t_hi, 13)
        fi = 0
        fs = best_c5
        for k2, t in enumerate(ts_f):
            trial = h + t * v
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_box_sum(trial, n / 2.0)
            s_val = exact_c5(trial)
            if s_val < fs:
                fs = s_val
                fi = k2
                best_t = t
        best_c5 = fs
        lo_f = max(0, fi - 2)
        hi_f = min(len(ts_f), fi + 3)
        t_lo = ts_f[lo_f]
        t_hi = ts_f[hi_f - 1]
        if t_hi - t_lo < 1e-11:
            break
    trial = h + best_t * v
    trial = np.clip(trial, 0.0, 1.0)
    trial = project_box_sum(trial, n / 2.0)
    fc5 = exact_c5(trial)
    if fc5 < best_c5:
        best_c5 = fc5
    if best_c5 < cur_c5 - 1e-14:
        return best_t, best_c5, trial
    return 0.0, cur_c5, h.copy()

def weighted_grad(h, beta, K):
    scores = raw_scores(h)
    n = len(h)
    L = 2 * n - 1
    max_s = scores.max()
    top_k = min(K, L)
    top_idx = np.argsort(scores)[-top_k:]
    w = np.zeros(L)
    sw = scores[top_idx] - max_s
    ew = np.exp(beta * sw)
    ew /= ew.sum()
    w[top_idx] = ew
    g = np.zeros(n)
    for r in top_idx:
        if w[r] < 1e-12:
            continue
        s = r - (n - 1)
        g += w[r] * grad_F(h, s)
    return g

def polish(h, deadline, rng, max_cycles=2000):
    n = len(h)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    cyc = 0
    while time.monotonic() < deadline and cyc < max_cycles:
        if stall > 80:
            break
        cyc += 1
        scores = raw_scores(h)
        max_sc = scores.max()
        L = len(scores)
        thresh = max_sc * (1.0 - 3e-4)
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        K = min(L, max(8, len(active)))
        improved = False
        beta_list = [10.0, 25.0, 50.0, 100.0]
        beta = beta_list[cyc % len(beta_list)]
        g = weighted_grad(h, beta, K)
        g -= np.mean(g)
        dirs = []
        if np.max(np.abs(g)) > 1e-13:
            dirs.append(g)
        top_active = active[np.argsort(scores[active])[-min(5, len(active)):]]
        for r in top_active:
            s = r - (n - 1)
            gr = grad_F(h, s)
            gr -= np.mean(gr)
            if np.max(np.abs(gr)) > 1e-13:
                dirs.append(gr)
        for _ in range(min(6, n // 10)):
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                v = np.zeros(n)
                v[i] = 1.0
                v[j] = -1.0
                dirs.append(v)
        if cyc % 5 == 0:
            v = rng.standard_normal(n)
            v -= np.mean(v)
            dirs.append(v)
        if cyc % 10 == 0:
            freq = 1.0 + (cyc // 10) * 0.3
            x = (np.arange(n) + 0.5) / n
            v = np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
            v -= np.mean(v)

            if tc5 < cur_c5 - 1e-14:
                h = trial
                cur_c5 = tc5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def pair_sweep(h, deadline, rng, max_rounds=50):

                    if tc5 < cur_c5 - 1e-14:

def perturb_retry(h, cur_c5, deadline, rng, max_tries=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(max_tries):
        if time.monotonic() >= deadline:
            break
        scale = 0.02 * (1 + rng.random())
        pert = rng.standard_normal(n) * scale
        pert -= np.mean(pert)
        h_p = np.clip(h + pert, 0.0, 1.0)
        h_p = project_box_sum(h_p, n / 2.0)
        c5_p = exact_c5(h_p)
        dl_short = min(deadline, time.monotonic() + max(2.0, (deadline - time.monotonic()) * 0.1))
        h_q, c5_q = polish(h_p, dl_short, rng, max_cycles=80)
        if c5_q < best_c5:
            best_c5 = c5_q
            best_h = h_q.copy()
        elif c5_q < cur_c5 and rng.random() < 0.3:
            best_c5 = c5_q
            best_h = h_q.copy()
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)
