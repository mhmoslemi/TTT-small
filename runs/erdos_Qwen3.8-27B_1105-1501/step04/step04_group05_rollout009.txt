import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    # Water-filling projection to sum exactly target while staying in [0,1]
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

def binary_score(mask):
    # mask is a 0/1 array
    n = len(mask)
    dx = 2.0 / n
    corr = np.correlate(mask.astype(float), 1.0 - mask.astype(float), mode='full')
    return float(np.max(corr) * dx)

def two_level_score(mask, d):
    # mask is 0/1, d in [0, 0.5]
    h = np.where(mask == 1, 1.0 - d, d)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def optimize_d(mask, n_samples=200):
    # Find best d in [0, 0.5] for given mask
    d_values = np.linspace(0, 0.5, n_samples)
    scores = np.array([two_level_score(mask, d) for d in d_values])
    best_idx = np.argmin(scores)
    # Refine around best
    d_lo = max(0, d_values[best_idx] - 0.01)
    d_hi = min(0.5, d_values[best_idx] + 0.01)
    d_fine = np.linspace(d_lo, d_hi, 100)
    scores_fine = np.array([two_level_score(mask, d) for d in d_fine])
    best_fine = d_fine[np.argmin(scores_fine)]
    return best_fine, float(np.min(scores_fine))

def binary_descent(mask, deadline, max_iters=10000):
    # Local search on binary mask
    n = len(mask)
    mask = mask.copy()
    current_score = binary_score(mask)
    for _iter in range(max_iters):
        if time.time() > deadline: break
        ones = np.where(mask == 1)[0]
        zeros = np.where(mask == 0)[0]
        best_gain = 0
        best_move = None
        for i in ones:
            for j in zeros:
                mask[i] = 0; mask[j] = 1
                new_score = binary_score(mask)
                mask[j] = 0; mask[i] = 1
                gain = current_score - new_score
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            mask[i] = 0; mask[j] = 1
            current_score -= best_gain
        else:
            break
    return mask, current_score

def _c5_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _binary_c5(mask):
    """C5 score for a binary mask (0/1 array)."""
    n = len(mask)
