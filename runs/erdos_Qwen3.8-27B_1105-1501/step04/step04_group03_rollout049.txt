    top_idx = np.argsort(scores)[::-1][:K]

        # score_k = sum_i h_{i+d}*(1-h_i) * dx for appropriate ranges
        # d(score_k)/d(h_j) = dx * [(1-h_{j+d}) if j in range for first term - h_{j-d} if j in range for second term]

        # h(x)(1-h(x+k)) is summed over valid x
        # If d >= 0: sum_{x=0}^{n-1-d} h[x] * (1-h[x+d])
        #   d/dh[j] = (1-h[j+d]) for j < n-d (from first term) - h[j-d] for j >= d (from second term)
        # If d < 0: let g = -d, sum_{x=g}^{n-1} h[x] * (1-h[x-g])
        #   = sum_{x=0}^{n-1-g} h[x+g] * (1-h[x])
        #   d/dh[j] = (1-h[j]) for j < n-g (from (1-h[x]) where x=j) - h[j+g] for j+g < n (from h[x+g] where x+g=j)

            for j in range(d, n):
                A_ub[row, j] -= dx * h[j - d]
        else:

            m = n - g
            for j in range(m):
                A_ub[row, j] -= dx * h[j + g]
            for j in range(g, n):
                A_ub[row, j] += dx * (1.0 - h[j - g])
        
        A_ub[row, n] = -1.0
        b_ub[row] = -scores[k]

    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds.append((None, None))
    
    try:

            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:

        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)

    for _ in range(12):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = np.max(scores)
    exp_s = np.exp((scores - max_s) / tau)
    Z = np.sum(exp_s)
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += w[k] * (1.0 - h[d:])
            grad[d:] -= w[k] * h[:m]
        else:
            g = -d
            m = n - g
            grad[:m] -= w[k] * h[g:]
            grad[g:] += w[k] * (1.0 - h[:m])
    return val, dx * grad

def nesterov_subgrad(h0, deadline, tau_init=0.3, tau_min=0.01):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    v = np.zeros(n)  # momentum
    beta = 1.0
    alpha = 1.0  # step size
    
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    
    t = 0
    while time.time() < deadline:
        t += 1
        tau = max(tau_min, tau_init * (tau_min / tau_init) ** (t / 500.0))
        # or exponential decay: tau = tau_init * exp(-t/200)
        
        val, grad = soft_grad(h, tau)
        
        # Nesterov step
        beta = (1 + np.sqrt(1 + 4 * beta * beta)) / 2
        # Actually standard Nesterov: beta_t = (1 + sqrt(1 + 4*t^2)) / 2
        # Let me use the standard form
        t_beta = (1 + np.sqrt(1 + 4 * t * t)) / 2
        t_beta_prev = (1 + np.sqrt(1 + 4 * (t-1) * (t-1))) / 2 if t > 1 else 1.0

            best_h = h_new.copy()
        
        h = h_new
        # Decay step size
        alpha = alpha_init / (1 + t * 0.01)
    
    return best_h, best_c5

def soft_grad(h, tau):

    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx

    Z = np.sum(exp_s)
    w = exp_s / Z
    val = max_s + tau * np.log(Z)

            m = n - g
            grad[:m] -= wk * dx * h[g:]
            grad[g:] += wk * dx * (1.0 - h[:m])

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=200):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    a = amax
                    h_t = best_h.copy()
                    h_t[j] += sign * a
                    h_t[k] -= sign * a
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def random_pair_polish(h, deadline, seed=0, n_trials=50000):
    n = len(h)
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    improved = 0
    for i in range(n_trials):
        if time.time() > deadline:
            break
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        sign = 1.0 if rng.random() < 0.5 else -1.0
        if sign > 0:
            amax = min(best_h[j], 1.0 - best_h[k])
        else:
            amax = min(1.0 - best_h[j], best_h[k])
        if amax < 1e-10:
            continue
        a = amax
        h_t = best_h.copy()
        h_t[j] += sign * a
        h_t[k] -= sign * a
        c5_t = exact_c5(h_t)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            improved += 1
    return best_h, best_c5

def binarize_and_optimize(h, deadline):
    n = len(h)
    target = n / 2.0
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    if s > target:
        idx = np.argsort(h_bin - h)[:int(s - target)]
        h_bin[idx] = 0.0
    elif s < target:
        idx = np.argsort(h - h_bin)[:int(target - s)]
        h_bin[idx] = 1.0
    h_bin = project(h_bin, target)
    c5_bin = exact_c5(h_bin)
    sub_dl = min(deadline, time.time() + 30)
    h_opt, c5_opt = lp_loop_active(h_bin, sub_dl, K=25, radius_init=0.3)
    if c5_opt < c5_bin:
        return h_opt, c5_opt
    return h_bin, c5_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Active LP (main workhorse) - 25% budget
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = lp_loop_active(h, d1, K=30, radius_init=0.2)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Nesterov subgradient for exploration - 12% budget
    d2 = t0 + budget_s * 0.37
    h2, c5_2 = nesterov_opt(best_h, d2, tau_init=0.35)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP refinement after subgradient - 15% budget
    d3 = t0 + budget_s * 0.52
    h3, c5_3 = lp_loop_active(best_h, d3, K=45, radius_init=0.1)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binarization attempt - 8% budget
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = binarize_and_optimize(best_h, d4)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Diversified restarts - 15% budget
    d5 = t0 + budget_s * 0.75
    scales = [0.03, 0.06, 0.1, 0.15, 0.2, 0.3]
    for ns in scales:
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_try, c5_try = lp_loop_active(h_pert, sub_dl, K=25, radius_init=0.15)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 6: Multi-resolution - 12% budget
    d6 = t0 + budget_s * 0.87
    tried_n = set([best_n])
    for dn in [0, -6, 6, -12, 12, -18, 18, -24, 24, -32, 32, -40, 40, -50, 50]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = lp_loop_active(h_r, sub_dl, K=25, radius_init=0.12)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Pair polish (deterministic) - 5% budget
    d7 = t0 + budget_s * 0.92
    h7, c5_7 = pair_polish(best_h, d7, max_passes=300)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Random pair polish - 3% budget
    d8 = min(hard_deadline, t0 + budget_s * 0.95)
    h8, c5_8 = random_pair_polish(best_h, d8, seed=seed, n_trials=80000)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final: one more LP micro-pass if time allows
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop_active(best_h, hard_deadline, K=30, radius_init=0.05)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
