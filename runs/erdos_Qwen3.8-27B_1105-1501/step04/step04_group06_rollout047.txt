        ps = j + s_off
        m2 = (ps >= 0) & (ps < n)
        grad[m2] -= w[r] * h[ps[m2]]

    target = n*0.5

        h = z[:n]
        corr = np.correlate(h, 1.0-h, mode='full')
        ms = np.max(corr)
        return float(ms + (1.0/alpha)*np.log(np.sum(np.exp(alpha*(corr-ms)))))

        h = z[:n]
        corr = np.correlate(h, 1.0-h, mode='full')
        ms = np.max(corr)

            ps=j+s; m2=(ps>=0)&(ps<n)
            g[m2] -= w[r]*h[ps[m2]]
