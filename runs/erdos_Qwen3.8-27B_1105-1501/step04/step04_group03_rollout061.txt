```python
def pair_cd_pass(h, target):
    n = len(h)
    improved = True
    while improved:
        improved = False
        for j in range(n):
            for k in range(j+1, n):
                s = target - (np.sum(h) - h[j] - h[k])
                # Actually s = target - sum(h[others]) = target - (sum(h) - h[j] - h[k])
                # But sum(h) = target always (we maintain the constraint)
                # So s = target - (target - h[j] - h[k]) = h[j] + h[k]
                s = h[j] + h[k]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                # Golden section search for x in [lo, hi]
                # f(x) = exact_c5 with h[j]=x, h[k]=s-x
                best_x = x_cur = h[j]
                best_f = exact_c5_with_pair(h, j, k, x_cur, s)
                # Golden section
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c = b - gr*(b-a)
                d = a + gr*(b-a)
                fc = exact_c5_with_pair(h, j, k, c, s)
                fd = exact_c5_with_pair(h, j, k, d, s)
                for _ in range(50):
                    if b - a < 1e-10:
                        break
                    if fc < fd:
                        b = d; d = c; fd = fc
                        c = b - gr*(b-a)
                        fc = exact_c5_with_pair(h, j, k, c, s)
                    else:
                        a = c; c = d; fc = fd
                        d = a + gr*(b-a)
                        fd = exact_c5_with_pair(h, j, k, d, s)
                x_opt = (a+b)/2.0
                f_opt = exact_c5_with_pair(h, j, k, x_opt, s)
                if f_opt < best_f - 1e-13:
                    h[j] = x_opt
                    h[k] = s - x_opt
                    improved = True
    return h
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    x = np.clip(v - tau, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def golden_pair(h, j, k, max_iter=30):
    s = h[j] + h[k]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[j] = c; h[k] = s - c
    fc = exact_c5(h)
    h[j] = d; h[k] = s - d
    fd = exact_c5(h)
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            h[j] = c; h[k] = s - c
            fc = exact_c5(h)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            h[j] = d; h[k] = s - d
            fd = exact_c5(h)
    x_opt = (a + b) / 2.0
    h[j] = max(0.0, min(1.0, x_opt))
    h[k] = max(0.0, min(1.0, s - x_opt))

    for (j, k) in pairs:
        old_j, old_k = h[j], h[k]
        golden_pair(h, j, k)
        if abs(h[j] - old_j) > 1e-11 or abs(h[k] - old_k) > 1e-11:
            n_changed += 1
    return n_changed

def build_pairs(n):
    return [(j, k) for j in range(n) for k in range(j + 1, n)]

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def build_corr_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
```
