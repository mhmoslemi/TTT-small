def exact_pair_optimize(h, j, k):
    """Find optimal t for h[j]+t, h[k]-t that minimizes max correlation."""
    n = len(h)
    dx = 2.0 / n
    # Current scores
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Delta for each shift
    # delta[d] = ds_d/dh[j] - ds_d/dh[k] (times dx)
    delta = np.zeros(2*n-1)
    for shift_k in range(2*n-1):
        d = shift_k - (n-1)
        # ds_d/dh[j]
        dj = 0.0
        if 0 <= j + d < n:
            dj += (1.0 - h[j + d])
        if 0 <= j - d < n:
            dj -= h[j - d]
        # ds_d/dh[k]
        dk = 0.0
        if 0 <= k + d < n:
            dk += (1.0 - h[k + d])
        if 0 <= k - d < n:
            dk -= h[k - d]
        delta[shift_k] = (dj - dk) * dx
    
    # New scores: scores + t * delta
    # Find t that minimizes max(scores + t*delta)
    # subject to 0 <= h[j]+t <= 1 and 0 <= h[k]-t <= 1
    
    t_min = max(-h[j], -(1-h[k]))  # h[j]+t >= 0, h[k]-t >= 0
    t_max = min(1-h[j], h[k])       # h[j]+t <= 1, h[k]-t <= 1
    
    if t_min >= t_max:
        return 0.0
    
    # The function f(t) = max_k (scores[k] + t*delta[k]) is convex piecewise linear
    # Find minimum: where subgradient contains 0
    # Sort by breakpoints
    # Breakpoints: t where scores[a] + t*delta[a] = scores[b] + t*delta[b]
    # t = (scores[b] - scores[a]) / (delta[a] - delta[b])
    
    # Simpler approach: ternary search or golden section on the 1D convex function
    # Since it's piecewise linear with at most O(n) pieces, we can use:
    # - Evaluate at t_min and t_max
    # - Find where the "slope" changes from negative to positive
    
    # The slope of f at t (away from breakpoints) is:
    # delta[argmax_k (scores[k] + t*delta[k])]
    # i.e., the delta of the maximizing shift
    
    # So f is decreasing where the maximizing shift has delta < 0
    # and increasing where it has delta > 0
    
    # At the optimum, either:
    # - t is at a boundary, or
    # - the maximizing shift has delta = 0 (unlikely for generic data), or
    # - two shifts are tied at the max and one has delta < 0, other has delta > 0
    
    # Practical approach: golden section search (works for unimodal functions)
    lo, hi = t_min, t_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = np.max(scores + c1 * delta)
    f2 = np.max(scores + c2 * delta)
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = np.max(scores + c1 * delta)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = np.max(scores + c2 * delta)
    return (lo + hi) / 2.0

def project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = np.sum(np.clip(v - mid, 0.0, 1.0))

        else:

    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10

        if cnt > 0:

        else:
            idx = int(np.argmin(x))

Let me also add a **block pair CD**: instead of pairs, try blocks of 3-4 coordinates. For a block of size m, we solve a small optimization problem (max of m+1 linear functions in m-1 variables). For m=3, this is a 2D problem (max of lines in 2D), which can be solved by checking all line intersections. For m=4, it's 3D (max of planes), which is more complex.

For m=3: Let coordinates be a, b, c with constraint a+b+c = const. We optimize over (a,b) with c = const - a - b. The objective is max_s (scores[s] + a*da[s] + b*db[s]) where da[s] = G[s,a] - G[s,c] and db[s] = G[s,b] - G[s,c].

The minimum of a max of linear functions in 2D is at a vertex of the arrangement, i.e., at the intersection of two lines. There are O(n²) intersection points. We can check all of them.

For n=92 and 2n-1=183 shifts, there are 183*182/2 ≈ 16.7K intersection points per triple. With C(92,3) ≈ 125K triples, that's too many.

So block CD of size 3 is too expensive. Let me stick with pairs.

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for j in range(n):
        for shift in range(2*n-1):
            d = shift - (n-1)
            val = 0.0
            jd = j + d
            jd2 = j - d
            if 0 <= jd < n:
                val += 1.0 - h[jd]
            if 0 <= jd2 < n:
                val -= h[jd2]
            G[shift, j] = val
    return G

def pair_cd(h, deadline, max_passes=200, tol=1e-14):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(best_h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for pas in range(max_passes):
        if time.time() > deadline:
            break
        G = build_grad_matrix(best_h)
        scores = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0/n)
        max_s = float(np.max(scores))
        active = np.where(scores >= max_s - 0.01)[0]
        improved = False
        
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j+1, n):
                t_min = max(-best_h[j], -(1.0-best_h[k]))
                t_max = min(1.0-best_h[j], best_h[k])
                if t_min >= t_max - 1e-12:
                    continue
                # Quick filter
                delta_a = G[active, j] - G[active, k]
                if np.all(delta_a >= 0):
                    continue
                delta = G[:, j] - G[:, k]
                lo_t, hi_t = t_min, t_max
                c1 = hi_t - gr * (hi_t - lo_t)
                c2 = lo_t + gr * (hi_t - lo_t)
                f1 = float(np.max(scores + c1 * delta))
                f2 = float(np.max(scores + c2 * delta))
                for _ in range(18):
                    if hi_t - lo_t < 1e-11:
                        break
                    if f1 < f2:
                        hi_t, c2, f2 = c2, c1, f1
                        c1 = hi_t - gr * (hi_t - lo_t)
                        f1 = float(np.max(scores + c1 * delta))
                    else:
                        lo_t, c1, f1 = c1, c2, f2
                        c2 = lo_t + gr * (hi_t - lo_t)
                        f2 = float(np.max(scores + c2 * delta))
                opt_t = (lo_t + hi_t) / 2.0
                if abs(opt_t) > 1e-9:
                    best_h[j] = np.clip(best_h[j] + opt_t, 0.0, 1.0)
                    best_h[k] = np.clip(best_h[k] - opt_t, 0.0, 1.0)
                    new_c5 = exact_c5(best_h)
                    if new_c5 < best_c5 - tol:
                        best_c5 = new_c5
                        scores = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0/n)
                        G = build_grad_matrix(best_h)
                        max_s = float(np.max(scores))
                        active = np.where(scores >= max_s - 0.01)[0]
                        improved = True
        
        if not improved:
            break
    return best_h, best_c5
