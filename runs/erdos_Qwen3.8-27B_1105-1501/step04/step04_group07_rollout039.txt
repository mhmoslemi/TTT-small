def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target} using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    threshold_idx = n // 2
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:threshold_idx]] = 1.0
    return b

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _binarize(h, n):
    threshold_idx = n // 2
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:threshold_idx]] = 1.0
    return b

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, c5_cur, max_alpha=0.5):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_s = 30
    alphas = np.linspace(0, alpha_max, n_s)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_s - 1):
        return best_a, best_c5
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    oa = (lo + hi) / 2.0
    oc = _exact_c5(h + oa * p)
    if oc < best_c5:
        return oa, oc
    return best_a, best_c5

def _sa_binary(h_init, deadline, seed, n_iters=300000):
    n = len(h_init)
    h = _binarize(h_init, n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T_start = 0.04
    T_end = 0.00005
    factor = (T_end / T_start) ** (1.0 / max(1, n_iters))
    T = T_start
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if T < T_end:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_pos = ones[rng.integers(len(ones))]
        j_pos = zeros[rng.integers(len(zeros))]
        h[i_pos] = 0.0
        h[j_pos] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i_pos] = 1.0
            h[j_pos] = 0.0
        T *= factor
    return best_h, best_c5

def _greedy_binary(h_init, deadline, max_iter=500000, seed=42):
    n = len(h_init)
    h = _binarize(h_init, n)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 30000:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_pos = ones[rng.integers(len(ones))]
        j_pos = zeros[rng.integers(len(zeros))]
        h[i_pos] = 0.0
        h[j_pos] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i_pos] = 1.0
            h[j_pos] = 0.0
            stall += 1
    return best_h, best_c5

def _lp_refine(h, deadline, top_k=None, radius=0.15, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            Gk = G[idx]
        else:
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((Gk.shape[0], n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -scores[:Gk.shape[0]] if top_k is None else -scores[np.argsort(scores)[-top_k:]]
        if top_k is None:
            b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.03, 0.4)
        else:
            stall += 1
            r *= 0.85
            if stall > 40 and r < 0.03:
                r = 0.10
    return best_h, best_c5

def _subgradient_descent(h, deadline, n_steps=5000, alpha0=0.08):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        scores = _scores(h)
        s_max = scores.max()
        if s_max < 1e-15:
            break
        # Use top 20% of lags with equal weight
        thresh = s_max * 0.92
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        w = np.zeros(len(scores))
        w[active] = 1.0 / len(active)
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        alpha = alpha0 / (1.0 + 0.01 * step)
        for scale in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            a, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + a * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _msequence(n_bits):
    """Generate an m-sequence of length 2^n_bits - 1."""
    if n_bits == 8:
        poly = 0b100100011  # x^8 + x^4 + x^3 + x^2 + 1
    elif n_bits == 6:
        poly = 0b1100011  # x^6 + x^5 + 1
    else:
        poly = 0b100100011
    length = (1 << n_bits) - 1
    reg = [1] * n_bits
    seq = []
    for _ in range(length):
        seq.append(reg[-1])
        feedback = 0
        bits = bin(poly)[2:]
        for i in range(n_bits):
            if i + 1 < len(bits) and bits[i + 1] == '1':
                feedback ^= reg[i]
        reg = reg[1:] + [feedback]
    return np.array(seq, dtype=float)

def _construct_candidates(n, parent_h):
    """Generate diverse starting candidates for given n."""
    candidates = []
    # 1. Binarized parent
    if parent_h is not None and len(parent_h) > 0:
        h_res = _resample(parent_h, n)
        candidates.append(h_res)
        candidates.append(_binarize(h_res, n))
    
    # 2. m-sequence based
    if n >= 62 and n <= 66:
        mseq = _msequence(6)  # length 63
        if n == 64:
            h_m = np.zeros(64)
            h_m[:63] = mseq
            # Need 32 ones; mseq has 32 ones already
            if h_m.sum() > n // 2:
                h_m = _binarize(h_m, n)
            candidates.append(h_m)
    
    # 3. Low-harmonic constructions
    u = (np.arange(n) + 0.5) / n
    for k1 in range(2, 6):
        for k2 in range(k1 + 1, 8):
            for phase in [0.0, 0.25, 0.5, 0.75]:
                raw = 0.5 + 0.3 * np.cos(2 * np.pi * k1 * (u + phase)) + \
                      0.2 * np.cos(2 * np.pi * k2 * (u + phase * 0.7)) + \
                      0.15 * np.sin(2 * np.pi * k1 * (u + phase * 0.3))
                candidates.append(_project(raw, n / 2.0))
                candidates.append(_binarize(raw, n))
    
    # 4. Periodic block patterns
    for period in [4, 5, 6, 7, 8, 10, 12, 14, 16, 20]:
        for duty in [0.3, 0.4, 0.5]:
            h_block = np.zeros(n)
            h_block[:int(duty * period)] = 1.0
            tiled = np.tile(h_block, n // period + 1)[:n]
            if tiled.sum() > n // 2:
                tiled = _binarize(tiled, n)
            elif tiled.sum() < n // 2:
                tiled = _binarize(-tiled + 1, n) if tiled.sum() == 0 else _project(tiled, n/2)
            candidates.append(tiled)
    
    # 5. Quadratic residue construction (for n close to prime)
    for p in [31, 37, 43, 47, 53, 59, 61, 67, 71, 73, 79, 83]:
        if abs(p - n) <= 4:
            h_qr = np.zeros(n)
            for i in range(p):
                if i * i % p == 1 or (i > 0 and pow(i, (p-1)//2, p) == 1):
                    h_qr[i % n] = 1.0
            if h_qr.sum() > 0 and h_qr.sum() < n:
                h_qr = _project(h_qr, n / 2.0)
                candidates.append(h_qr)
                candidates.append(_binarize(h_qr, n))
    
    return candidates
