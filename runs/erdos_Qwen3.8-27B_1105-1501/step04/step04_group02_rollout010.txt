import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _build_grad_row(h, r):
    n = len(h)
    d = r - (n - 1)
    row = np.zeros(n)
    if d >= 0:
        m = n - d
        row[d:] += 1.0
        row[d:] -= h[:m]
        row[:m] -= h[d:]
    else:
        g = -d
        m = n - g
        row[:m] += 1.0
        row[:m] -= h[g:]
        row[g:] -= h[:m]
    return row

def _line_search(h, p, c5_cur, alpha_max, deadline):
    best_t = 0.0
    best_val = c5_cur
    n_grid = 24
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    if best_val >= c5_cur - 1e-14:
        return best_t, best_val
    lo_t = ts[max(0, idx - 2)]
    hi_t = ts[min(n_grid - 1, idx + 2)]
    if hi_t <= lo_t:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    f1, f2 = _f(c1), _f(c2)
    for _ in range(30):
        if hi_t - lo_t < 1e-13:
            break
        if time.time() > deadline:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_direction(h, scores, radius, k_top=10):
    n = len(h)
    num_sc = len(scores)
    max_sc = float(np.max(scores))
    idx_sorted = np.argsort(scores)[::-1]
    k_use = min(k_top, num_sc)
    active = idx_sorted[:k_use]
    G = np.zeros((k_use, n))
    for i in range(k_use):
        G[i] = _build_grad_row(h, active[i])
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((k_use, n + 1))
    A_ub[:, :n] = -G
    A_ub[:, n] = 1.0
    b_ub = scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((0, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_direction_full(h, scores, radius, margin_frac=0.05):

    num_sc = len(scores)
    max_sc = float(np.max(scores))
    min_sc = float(np.min(scores))
    margin = margin_frac * (max_sc - min_sc)
    active = np.where(scores >= max_sc - margin)[0]

    A_eq = np.zeros((1, n + 1))

    except Exception:
        pass
    return None

def _softmax_grad(h, scores, tau):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(-(scores - max_s) / tau)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        return None
    w = w / w_sum
    grad = np.zeros(n)
    for r in range(len(scores)):
        if w[r] < 1e-12:
            continue
        grad += w[r] * _build_grad_row(h, r)
    grad -= np.mean(grad)
    norm = float(np.max(np.abs(grad)))
    if norm < 1e-14:
        return None
    return -grad / norm

def _local_opt(h, deadline, k_top=10, radius_init=0.12, use_full_lp=False, margin_frac=0.05):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    consec_improve = 0
    while time.time() < deadline:

        improved = False
        p = None
        if use_full_lp:
            p = _lp_direction_full(h, scores, radius, margin_frac)
        else:
            p = _lp_direction(h, scores, radius, k_top)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            p = _softmax_grad(h, scores, 8.0)
            if p is not None:
                p = p * radius
        if p is not None:
            alpha_max = radius
            for j in range(n):
                if p[j] > 1e-14:
                    alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                elif p[j] < -1e-14:
                    alpha_max = min(alpha_max, h[j] / (-p[j]))
            alpha_max = max(alpha_max, 1e-12)
            if alpha_max > 1e-12:
                alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max, time.time() + 3.0)
                if new_c5 < cur_c5 - 1e-14:
                    h = h + alpha * p
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    improved = True
                    consec_improve += 1
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    radius = min(radius * 1.15, 0.35)
        if not improved:
            stall += 1
            consec_improve = 0
            radius *= 0.65
            if radius < 1e-7:
                break
            if stall > 100:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binarize(h):
    n = len(h)
    target = n / 2.0
    sorted_idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    whole = int(target)
    frac = target - whole
    for i in range(whole):
        h_b[sorted_idx[i]] = 1.0
    if frac > 1e-10:
        h_b[sorted_idx[whole]] = frac
    return h_b

def _shift_h(h, shift):
    n = len(h)
    return np.roll(h, shift)
