def _lp_loop(h, deadline, radius_init=0.15, active_frac=1.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    iter_count = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        if iter_count % 20 == 0:
            p = _lp_direction(h, scores, G, radius, active_frac=0.3)
            if p is None:
                p = _lp_direction(h, scores, G, radius, active_frac=1.0)
        else:
            p = _lp_direction(h, scores, G, radius, active_frac=active_frac)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        iter_count += 1
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.45)
            if iter_count % 50 == 0:
                radius = max(radius, 0.08)
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5

def _softmax_direction(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _nesterov_subgrad(h, deadline, target_c5=0.375):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    y = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t_k = 1.0
    mom = np.zeros(n)
    lr0 = 0.08
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        t_k += 1.0
        beta = (1.0 - t_k) / (1.0 + t_k + 1.0)
        lr = lr0 / np.sqrt(t_k)
        y = h + beta * (y - h)
        grad = _softmax_direction(y, dx, 200.0)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            stall += 1
            continue
        grad = grad / gn
        step = lr * grad
        h_new = y + step
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            cur_c5 = c5_new
            h = h_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stal
ip = 0
        else:
            stall += 1
            if stall % 30 == 0:
                lr0 *= 0.9
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _simulated_anneal(h, deadline, initial_temp=0.003, final_temp=0.0001):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    total_iters = 800
    for it in range(total_iters):
        if time.time() > deadline:
            break
        progress = it / max(1, total_iters - 1)
        temp = initial_temp * (final_temp / initial_temp) ** progress
        scale = 0.02 + 0.04 * (1.0 - progress)
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        c5_pert = _exact_c5(h_pert)
        if c5_pert < cur_c5 - 1e-12:
            h = h_pert
            cur_c5 = c5_pert
            sub_dl = min(deadline, time.time() + 0.8)
            h, cur_c5 = _lp_loop(h, sub_dl, radius_init=0.06, active_frac=0.5)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        elif rng.random() < np.exp(-(c5_pert - cur_c5) / temp):
            h = h_pert
            cur_c5 = c5_pert
    return best_h, best_c5

def _best_coord_descent(h, deadline, max_sweeps=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            best_d = None
            best_c5_try = best_c5
            for j in range(n):
                if j == i:
                    continue
                if h[j] <= 1e-10 and h[i] < 0.5:
                    continue
                if h[j] >= 1.0 - 1e-10 and h[i] > 0.5:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                _, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5_try:
                    best_c5_try = c5_try
                    best_d = d
            if best_d is not None and best_c5_try < best_c5 - 1e-13:
                alpha, _ = _line_search(h, best_d, dx, 0.5)
                h = h + alpha * best_d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LP warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Nesterov subgradient exploration (10%)
    d2 = t0 + budget_s * 0.18
    h2, c5_2 = _nesterov_subgrad(best_h, d2, target_c5=0.375)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP with active set focusing (22%)
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.12, active_frac=0.4)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution search (28%)
    d4 = t0 + budget_s * 0.68
    best_n = len(best_h)
    n_candidates = [best_n + dn for dn in range(-24, 25, 2)]
    n_candidates = [x for x in n_candidates if 24 <= x <= 200 and x != best_n]
    rng.shuffle(n_candidates)
    tried = set()
    for n_try in n_candidates:
        if time.time() > d4:
            break
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r > best_c5 + 0.005:
            continue
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1, active_frac=0.5)
        rem = d4 - time.time()
        if rem > 4 and c5_r <= best_c5 + 0.001:
            sub_dl2 = min(d4, time.time() + max(2, budget_s * 0.006))
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.06, active_frac=1.0)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Simulated annealing with LP refinement (14%)
    d5 = t0 + budget_s * 0.82
    h5, c5_5 = _simulated_anneal(best_h, d5, initial_temp=0.004, final_temp=0.0001)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Coordinate descent (8%)
    d6 = t0 + budget_s * 0.90
    h6, c5_6 = _best_coord_descent(best_h, d6, max_sweeps=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final LP polish (8%)
    d7 = t0 + budget_s * 0.98
    h7, c5_7 = _lp_loop(best_h, d7, radius_init=0.05, active_frac=1.0)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
