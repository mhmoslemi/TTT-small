    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Gradient polish (10% budget)
    d2 = t0 + budget_s * 0.50
    if time.time() < d2:
        hg, c5g = grad_descent_polish(best_h, d2, lr=0.004, n_lags=4, patience=50)
        if c5g < best_c5:
            best_c5 = c5g
            best_h = hg

    # Phase 3: Pair sweep (15% budget)
    d3 = t0 + budget_s * 0.65
    if time.time() < d3:
        hp, c5p = pair_sweep(best_h, d3, max_passes=300, n_line=17)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp

    # Phase 4: FW with different parameters (10% budget)
    d4 = t0 + budget_s * 0.75
    if time.time() < d4:
        h4, c5_4 = fw_optimize(best_h, d4, eta_frac=0.015, n_active=4,
                               max_stall=25, perturb_scale=0.008, rng=rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: Resolution search (15% budget)
    d5 = t0 + budget_s * 0.90
    tried_n = set([len(best_h)])
    for dn in [0, -8, 8, -16, 16, -4, 4, -12, 12, 24, -24, 32]:
        if time.time() > d5:
            break
        n_try = n + dn
        if n_try < 16 or n_try > 128 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        c5_r = exact_c5(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.025)
        h_r2, c5_r2 = fw_optimize(h_r, sub_dl, eta_frac=0.03, n_active=5,
                                  max_stall=20, perturb_scale=0.01, rng=rng)
        if c5_r2 < c5_r:
            h_r = h_r2
            c5_r = c5_r2
        c5_r = exact_c5(h_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Final pair sweep + gradient (10% budget)
    d6 = deadline
    if time.time() < d6:
        hp2, c5p2 = pair_sweep(best_h, min(d6, time.time() + budget_s * 0.04),
                               max_passes=200, n_line=21)
        if c5p2 < best_c5:
            best_c5 = c5p2
            best_h = hp2
    if time.time() < d6:
        hg2, c5g2 = grad_descent_polish(best_h, d6, lr=0.002, n_lags=3, patience=30)
        if c5g2 < best_c5:
            best_c5 = c5g2
            best_h = hg2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
