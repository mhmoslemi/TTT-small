        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-13:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += -h[d:]
            g[d:] += (1.0 - h[:m])
    else:
        q = -d
        m = n - q
        if m > 0:
            g[:m] += (1.0 - h[q:])
            g[q:] += -h[:m]
    return g

def build_all_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        G[r] = build_grad_lag(h, d)
    return G

def line_search_exact(h, p, cur_c5, max_alpha):
    n = len(h)
    if max_alpha < 1e-12:
        return 0.0, cur_c5
    best_a = 0.0
    best_f = cur_c5
    n_grid = 14
    for i in range(1, n_grid + 1):
        a = max_alpha * i / (n_grid + 1)
        f = exact_c5(h + a * p)
        if f < best_f:
            best_f = f
            best_a = a
    if best_a == 0.0:
        return 0.0, cur_c5
    lo_a = max(0.0, best_a * 0.6)
    hi_a = min(max_alpha, best_a * 1.4)
    if hi_a - lo_a < 1e-13:
        return best_a, best_f
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(22):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_f = exact_c5(h + opt_a * p)
    if opt_f <= best_f:
        return opt_a, opt_f
    return best_a, best_f

def compute_alpha_max(h, p):
    amax = 1.0
    for j in range(len(h)):
        if p[j] > 1e-14:
            amax = min(amax, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            amax = min(amax, h[j] / (-p[j]))
    return max(amax, 1e-10)

def lp_direction(h, G, scores, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def topk_lp_direction(h, G, scores, topk, radius):
    n = len(h)
    sorted_idx = np.argsort(scores)[::-1]
    top_idx = sorted_idx[:topk]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((topk, n + 1))
    for i, idx in enumerate(top_idx):
        A_ub[i, :n] = G[idx]
        A_ub[i, n] = -1.0
    b_ub = -scores[top_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_loop(h, deadline, radius_init=0.12, topk_mode='all'):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 100:
            break
        scores = scores_array(h)
        G = build_all_grads(h)
        
        if topk_mode == 'all':
            p = lp_direction(h, G, scores, radius)
        else:
            p = topk_lp_direction(h, G, scores, topk_mode, radius)
        
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        
        amax = compute_alpha_max(h, p)
        alpha, new_c5 = line_search_exact(h, p, cur_c5, amax)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.08
    
    return best_h, best_c5

def gradient_descent_exact(h, deadline, topk=4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline:
        if stall > 100:
            break
        scores = scores_array(h)
        max_s = float(np.max(scores))
        sorted_idx = np.argsort(scores)[::-1]
        top_idx = sorted_idx[:topk]
        top_scores = scores[top_idx]
        gaps = top_scores - max_s
        tau = max(0.02 * max_s, 1e-4)
        weights = np.exp(gaps / tau)
        weights /= np.sum(weights)
        
        G = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            d = idx - (n - 1)
            G += w * build_grad_lag(h, d)
        
        mean_G = np.mean(G)
        p = mean_G - G
        p_norm = np.max(np.abs(p))
        if p_norm < 1e-14:
            stall += 1
            if stall > 50:
                break
            i, j = stall % n, (stall + 7) % n
            if i == j:
                j = (j + 1) % n
            p = np.zeros(n)
            p[i] = 1.0
            p[j] = -1.0
        else:
            p = p / p_norm
            stall = 0
        
        amax = compute_alpha_max(h, p)
        alpha, new_c5 = line_search_exact(h, p, cur_c5, amax)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
    
    return best_h, best_c5

def pair_polish(h, deadline, max_passes=80, sample_size=10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(12345)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        candidates = list(zip(*np.unique(rng.integers(0, n, size=(200, 2)), return_index=True)))
        # Filter i != j
        pairs = [(i, j) for i, j in rng.integers(0, n, size=(300, 2)) if i != j]
        seen = set()
        for i, j in pairs:
            key = (min(i,j), max(i,j))
            if key in seen:
                continue
            seen.add(key)
            if time.time() > deadline:
                break
            for sign in [1.0, -1.0]:
                if sign > 0:
                    amax = min(h[i], 1.0 - h[j])
                else:
                    amax = min(1.0 - h[i], h[j])
                if amax < 1e-12:
                    continue
                for frac in [1.0, 0.5, 0.25]:
                    a = amax * frac
                    h_t = h.copy()
                    h_t[i] += sign * a
                    h_t[j] -= sign * a
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_c5

def anneal_search(h, deadline, seed_anneal=99):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed_anneal)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = max(2e-4, 5e-4 * cur_c5)
    min_T = max(5e-7, 1e-7 * cur_c5)
    
    while time.time() < deadline and T > min_T:
        move_type = rng.random()
        if move_type < 0.7:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            if rng.random() < 0.5:
                amax = min(h[i], 1.0 - h[j])
                if amax < 1e-12:
                    T *= 0.995
                    continue
                a = amax * rng.uniform(0.05, 1.0)
                h_t = h.copy()
                h_t[i] += a
                h_t[j] -= a
            else:
                amax = min(1.0 - h[i], h[j])
                if amax < 1e-12:
                    T *= 0.995
                    continue
                a = amax * rng.uniform(0.05, 1.0)
                h_t = h.copy()
                h_t[i] -= a
                h_t[j] += a
        elif move_type < 0.85:
            blen = max(2, min(8, n // 8))
            a = rng.integers(0, n - blen + 1)
            h_t = h.copy()
            h_t[a:a+blen] = h_t[a:a+blen][::-1]
        else:
            blen = max(2, min(6, n // 10))
            a = rng.integers(0, n - blen + 1)
            h_t = h.copy()
            shift = rng.integers(1, blen)
            h_t[a:a+blen] = np.roll(h_t[a:a+blen], shift)
        
        c5_t = exact_c5(h_t)
        delta = c5_t - cur_c5
        
        if delta < -1e-12:
            h = h_t
            cur_c5 = c5_t
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            if delta < 0.01 * cur_c5:
                if rng.random() < np.exp(-delta / T):
                    h = h_t
                    cur_c5 = c5_t
        
        T *= 0.996
    
    return best_h, best_c5

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0
    
    # Phase 1: Full-lag LP (proven effective from parent)
    d1 = t0 + budget_s * 0.12
    h1, c1 = lp_loop(best_h, d1, radius_init=0.15, topk_mode='all')
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
    
    # Phase 2: Top-4 LP (focused on most critical lags)
    d2 = t0 + budget_s * 0.20
    h2, c2 = lp_loop(best_h, d2, radius_init=0.10, topk_mode=4)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()
    
    # Phase 3: Gradient descent with exact line search (top-k weighted)
    d3 = t0 + budget_s * 0.28
    h3, c3 = gradient_descent_exact(best_h, d3, topk=4)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3.copy()
    
    # Phase 4: Top-6 LP
    d4 = t0 + budget_s * 0.36
    h4, c4 = lp_loop(best_h, d4, radius_init=0.08, topk_mode=6)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4.copy()
    
    # Phase 5: Annealing for diversification
    d5 = t0 + budget_s * 0.52
    h5, c5 = anneal_search(best_h, d5, seed_anneal=seed + 1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()
    
    # Phase 6: Refine after annealing
    d6 = t0 + budget_s * 0.62
    h6, c6 = lp_loop(best_h, d6, radius_init=0.08, topk_mode='all')
    if c6 < best_c5:
        best_c5 = c6
        best_h = h6.copy()
    
    # Phase 7: More annealing from refined point
    d7 = t0 + budget_s * 0.70
    h7, c7 = anneal_search(best_h, d7, seed_anneal=seed + 2)
    if c7 < best_c5:
        best_c5 = c7
        best_h = h7.copy()
    
    # Phase 8: Multi-resolution search
    d8 = t0 + budget_s * 0.82
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -20, 20]
    tried_n = set([best_n])
    for dn in offsets:
        if time.time() > d8:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 160:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample(best_h, n_try)
        sub_dl = min(d8, time.time() + budget_s * 0.012)
        h_r, cr = lp_loop(h_r, sub_dl, radius_init=0.08, topk_mode='all')
        if cr < best_c5 - 1e-12:
            best_c5 = cr
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 9: Pair polish (targeted coordinate moves)
    d9 = t0 + budget_s * 0.90
    h9, c9 = pair_polish(best_h, d9, max_passes=150, sample_size=12)
    if c9 < best_c5:
        best_c5 = c9
        best_h = h9.copy()
    
    # Phase 10: Final gradient descent micro-refinement
    if time.time() < hard_deadline:
        h10, c10 = gradient_descent_exact(best_h, hard_deadline, topk=3)
        if c10 < best_c5:
            best_c5 = c10
            best_h = h10.copy()
    
    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    
    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0
    
    return best_h, final_c5, best_n
