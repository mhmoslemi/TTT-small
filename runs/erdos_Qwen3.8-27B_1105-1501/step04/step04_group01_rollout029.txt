def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _correlate_scores(h):
    """Return all correlation scores (not yet multiplied by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_obj(h, d, dx):
    """Compute the quadratic model of f_k along direction d.
    Returns arrays S0, T, C such that f_k(h+t*d) ≈ S0[k] + t*T[k] - t²*C[k]
    """
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    # T[k] = d^T grad_k where grad_k is gradient of f_k
    # f_k = sum_i h[i]*(1-h[i+k])
    # df_k/dh[j] = (1-h[j+k]) - h[j-k] (with boundary handling)
    # T = sum over k of (d^T grad_k) = ... 
    # Actually T[k] for each lag k:
    # f_k(h+td) = S0[k] + t*T[k] - t²*C[k]
    # where T[k] = sum_i (d[i]*(1-h[i+k]) - h[i]*d[i+k])
    #       C[k] = sum_i d[i]*d[i+k]
    # We can compute T and C using correlations:
    # T[k] = correlate(d, 1-h)[k] - correlate(h, d)[k]
    # C[k] = correlate(d, d)[k]
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    return S0, T, C

def _line_search(h, d, dx, trust_radius):
    """Find best step along direction d."""
    n = len(h)
    S0, T, C = _line_search_obj(h, d, dx)
    
    # Find valid range for t
    t_lo, t_hi = 0.0, trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _softmax_direction(h, scores, tau):
    """Compute gradient direction weighted by softmax of scores."""
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(tau * (scores - max_s))
    w /= float(np.sum(w))
    
    grad = np.zeros(n)
    # For each lag r (0 to 2n-2), corresponding to shift d = r - (n-1)
    # grad contribution: w[r] * [(1-h shifted) - h shifted]
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g+m])
                grad[g:] += w[r] * (1.0 - h[:m])
    
    # Center the gradient (subtract mean to maintain sum constraint approximately)
    grad -= np.mean(grad)
    return grad

def _projected_grad_step(h, grad, dx, trust_radius):
    """Take a projected gradient step with line search."""
    # Project gradient to feasible directions (sum=0, within box)
    # First, center: grad already centered
    # Scale to trust region
    max_g = float(np.max(np.abs(grad)))
    if max_g < 1e-15:
        return 0.0, _exact_c5(h)
    d = grad / max_g * trust_radius
    # Project to box: clip to feasible range
    pos_mask = d > 0
    neg_mask = d < 0
    d[pos_mask] = np.minimum(d[pos_mask], (1.0 - h[pos_mask]) / trust_radius * trust_radius)  # This is just d clipped
    # Actually the line search handles the box constraint, so just use d directly
    return _line_search(h, d, dx, trust_radius)

def _lp_topk(h, scores, topk, radius, deadline):
    """LP to find direction that reduces top-k scores."""
    n = len(h)
    num_lags = 2 * n - 1
    topk = min(topk, num_lags)
    
    # Get indices of top-k lags
    top_idx = np.argsort(scores)[-topk:]
    
    # Build gradient matrix rows for top-k lags only
    # G[r, j] = df_{lag_r}/dh[j]
    G = np.zeros((topk, n))
    for ii, r in enumerate(top_idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[ii, :m] += (1.0 - h[d:d+m])
                G[ii, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[ii, :m] += (-h[g:g+m])
                G[ii, g:] += (1.0 - h[:m])
    
    # LP: min t
    # s.t. G * d <= t * ones (scores decrease)
    #      sum(d) = 0
    #      -radius*ones <= d <= radius*ones
    # Variables: d[0..n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((topk, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(topk)  # G*d - t <= 0, i.e., G*d <= t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _coord_descent_1d(a_k, c_k):
    """Find x in [0,1] minimizing max_k(a_k + c_k * x).
    a_k: constant terms, c_k: slopes for each lag k.
    """
    # The function f(x) = max_k(a_k + c_k*x) is convex piecewise linear.
    # Minimum is at x=0, x=1, or at intersection of two lines that form the upper envelope.
    # Simple approach: check x=0, x=1, and candidate breakpoints.
    
    # More efficient: the subgradient at x is c_k* where k* = argmax_k(a_k + c_k*x)
    # At the minimum, 0 should be in the subgradient, meaning there exist two active
    # lines with c_k1 < 0 < c_k2 (or one with c=0).
    
    # Practical approach: ternary search (since f is convex)
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a_k + m1 * c_k))
        f2 = float(np.max(a_k + m2 * c_k))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _coord_descent_pass(h, dx):

    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_full(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (-h[g:g + m])
                G[r, g:] += (1.0 - h[:m])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 8})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _cd_1d_solve(a_k, c_k):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a_k + mid * c_k
        mx = float(np.max(vals))
        active = np.where(vals >= mx - 1e-14)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        min_slope = float(np.min(c_k[active]))
        max_slope = float(np.max(c_k[active]))
        if min_slope > 1e-10:
            hi = mid
        elif max_slope < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def _coord_descent_pass(h):
    n = len(h)
    target = n / 2.0
    for j in range(n):
        scores = np.correlate(h, 1.0 - h, mode='full')
        c_k = np.zeros(2 * n - 1)
        for r in range(2 * n - 1):
            d = r - (n - 1)
            coeff = 0.0
            jd = j + d
            if 0 <= jd < n:
                coeff += (1.0 - h[jd])
            jd2 = j - d
            if 0 <= jd2 < n:
                coeff += (-h[jd2])
            c_k[r] = coeff
        a_k = scores - c_k * h[j]
        x_new = _cd_1d_solve(a_k, c_k)
        h[j] = x_new
    h = _project(h, target)
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_phase(h, deadline):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    tau = 60.0
    lr = 0.06
    stall = 0
    mom = np.zeros(n)
    while time.time() < deadline:
        if stall > 50:
            break
        sc = _scores(cur)
        g = _softmax_grad(cur, sc, tau)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            tau *= 1.5
            continue
        g = g / gn
        mom = 0.85 * mom + 0.15 * g
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            tau *= 1.3
            continue
        mom = mom / mn
        d = lr * mom
        alpha, new_c5 = _line_search(cur, d, lr * 3.0)
        if new_c5 < cur_c5 - 1e-14:
            cur = cur + alpha * d
            cur = _project(cur, n / 2.0)
            cur_c5 = _exact_c5(cur)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
            stall = 0
            lr = min(lr * 1.06, 0.35)
        else:
            stall += 1
            lr *= 0.83
            tau *= 1.2
            if stall % 20 == 0:
                mom = np.zeros(n)
    return best, best_c5

def _lp_phase(h, deadline, use_topk=True):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    radius = 0.15
    topk = 12
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > 50:
            break
        sc = _scores(cur)
        if use_topk:
            p = _lp_topk(cur, sc, topk, radius)
        else:
            p = _lp_full(cur, sc, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.65
            if use_topk:
                topk = min(topk + 4, 50)
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search(cur, p, max(radius * 2, 0.1))
        if new_c5 < cur_c5 - 1e-13:
            cur = cur + alpha * p
            cur = _project(cur, n / 2.0)
            cur_c5 = _exact_c5(cur)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
            stall = 0
            radius = min(radius * 1.08, 0.35)
            if use_topk:
                topk = max(8, topk - 1)
        else:
            stall += 1
            radius *= 0.72
            if use_topk:
                topk = min(topk + 5, 60)
        it += 1
    return best, best_c5

def _cd_phase(h, deadline, max_passes=500):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        cur = _coord_descent_pass(cur)
        c5 = _exact_c5(cur)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = cur.copy()
        elif c5 >= best_c5 + 1e-14:
            break
        else:
            if _ >= 10:
                break
    return best, best_c5

def _perturb_phase(h, deadline, seed_val):
    n = len(h)
    rng = np.random.default_rng(seed_val)
    best = h.copy()
    best_c5 = _exact_c5(best)
    scale = 0.04
    attempts = 0
    while time.time() < deadline and attempts < 80:
        attempts += 1
        h_p = best + rng.normal(0, scale, n)
        h_p = _project(h_p, n / 2.0)
        sub_dl = min(deadline, time.time() + budget_s_global * 0.012)
        h_t, c5_t = _lp_phase(h_p, sub_dl, use_topk=True)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best = h_t.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.15)
    return best, best_c5

budget_s_global = 1100.0

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s_global
    budget_s_global = budget_s
    t0 = time.time()
    deadline = t0 + budget_s

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Gradient warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _grad_phase(h, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP top-k focused (22%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _lp_phase(best_h, d2, use_topk=True)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Coordinate descent (12%)
    d3 = t0 + budget_s * 0.42
    h3, c5_3 = _cd_phase(best_h, d3, max_passes=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP full (all lags) (10%)
    d4 = t0 + budget_s * 0.52
    h4, c5_4 = _lp_phase(best_h, d4, use_topk=False)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-n exploration (18%)
    d5 = t0 + budget_s * 0.70
    best_n = len(best_h)
    rng = np.random.default_rng(seed)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -3, 3, -5, 5, -7, 7, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 250:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r > best_c5 + 0.005:
            continue
        sub_dl = min(d5, time.time() + max(2, budget_s * 0.010))
        h_r, c5_r = _lp_phase(h_r, sub_dl, use_topk=True)
        sub_dl2 = min(d5, time.time() + max(1, budget_s * 0.006))
        h_r, c5_r = _cd_phase(h_r, sub_dl2, max_passes=80)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Perturbation + re-optimize (12%)
    d6 = t0 + budget_s * 0.82
    h6, c5_6 = _perturb_phase(best_h, d6, seed + 1)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final gradient + LP alternation (6%)
    d7 = t0 + budget_s * 0.88
    n_cur = len(best_h)
    cur = best_h.copy()
    cur_c5 = best_c5
    radius = 0.10
    stall = 0
    mode = 0
    while time.time() < d7:
        if stall > 35:
            break
        sc = _scores(cur)
        if mode == 0:
            g = _softmax_grad(cur, sc, 100.0)
            gn = np.linalg.norm(g)
            if gn < 1e-13:
                mode = 1
                continue
            d = g / gn * radius
        else:
            p = _lp_topk(cur, sc, 10, radius)
            if p is None:
                mode = 0
                continue
            d = p
        alpha, new_c5 = _line_search(cur, d, radius * 2.5)
        if new_c5 < cur_c5 - 1e-14:
            cur = cur + alpha * d
            cur = _project(cur, n_cur / 2.0)
            cur_c5 = _exact_c5(cur)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
            mode = 1 - mode
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.8
            mode = 1 - mode

    # Phase 8: Final coordinate descent (5%)
    d8 = t0 + budget_s * 0.96
    h8, c5_8 = _cd_phase(best_h, d8, max_passes=200)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
