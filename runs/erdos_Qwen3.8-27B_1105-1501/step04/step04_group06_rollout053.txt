import numpy as np

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _exact_c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _deriv_lag(h, n, r):
    s = r - (n - 1)
    j = np.arange(n)
    pd = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    pd[m1] = 1.0 - h[gs[m1]]
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    pd[m2] -= h[ps[m2]]
    return pd


def _epi_obj(z, n):
    return z[n]


def _epi_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _ineq_f(z, args):
    n, active = args
    h = z[:n]
    corr = _corr(h)
    return z[n] - corr[active]


def _ineq_g(z, args):
    n, active = args
    h = z[:n]
    k = len(active)
    J = np.zeros((k, n + 1))
    for idx in range(k):
        J[idx, :n] = -_deriv_lag(h, n, active[idx])
        J[idx, n] = 1.0
    return J


def _slsqp_active(h_init, n, deadline, max_rounds=12, K_init=8, eps_act=1e-8):
    h = _project(np.asarray(h_init, dtype=float), n * 0.5)
    corr = _corr(h)
    t_val = float(np.max(corr)) + 1e-4
    active = np.argsort(-corr)[:K_init]
    best_h, best_c5 = h.copy(), _exact_c5(h)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        z0 = np.concatenate([h, [t_val]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
        cons = [
            {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
            {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g, 'args': (n, active)}
        ]
        try:
            res = minimize(_epi_obj, z0, jac=_epi_grad, args=(n,),
                           bounds=bounds, constraints=cons, method='SLSQP',
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = _project(res.x[:n], n * 0.5)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:

            corr_new = _corr(h_new)

def _lp_direction(h, scores, n, radius):
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        G[r, :] = _deriv_lag(h, n, r)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for jj in range(n):
        lo_b = max(-radius, -h[jj])
        hi_b = min(radius, 1.0 - h[jj])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgrad_step(h, n, step_size=0.05, deadline=None):
    corr = _corr(h)
    r_max = int(np.argmax(corr))
    pd = _deriv_lag(h, n, r_max)
    pd -= np.mean(pd)
    norm = np.linalg.norm(pd)
    if norm < 1e-15:
        return h.copy(), _exact_c5(h)
    pd = pd / norm * step_size
    c5_cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), c5_cur
    for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
        h_try = _project(h - frac * pd, n * 0.5)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        if deadline is not None and time.time() > deadline:
            break
    return best_h, best_c5


def _lse_step(h, n, alpha, maxiter=80):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    def _lse_obj(x, n, alpha):
        h_x = np.asarray(x, dtype=float)
        c = _corr(h_x)
        ms = float(np.max(c))
        vals = np.clip(alpha * (c - ms), -400.0, 400.0)
        ev = np.exp(vals)
        return ms + (1.0 / alpha) * np.log(float(np.sum(ev)))
    res = minimize(_lse_obj, h, args=(n, alpha), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
    x = _project(res.x, target)
    return x, _exact_c5(x)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _gen_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)

def _gen_structured(n, mode, rng):
    x = np.arange(n)
    if mode == 0:
        f = rng.uniform(1.0, 4.0)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f * x / n + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        p = rng.permutation(n)
        h = np.zeros(n)
        h[p[:n // 2]] = 1.0
    elif mode == 2:
        f1, f2 = rng.uniform(1, 3), rng.uniform(3, 6)
        h = 0.5 + 0.2 * np.cos(2 * np.pi * f1 * x / n) * np.sin(2 * np.pi * f2 * x / n)
    else:
        h = rng.uniform(0, 1, n)
    return _project(h, n * 0.5)

    h_lp, c5_lp = _lp_loop(best_h, best_n, d8, radius_init=0.03,
                           max_iters=3000, stall_limit=80)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
