def _corr_vec(h):

    n = len(h)
    dx = 2.0 / n

def _exact_c5(h):

def _project(v, target):
    """Project v to {x: 0<=x_i<=1, sum(x)=target} via bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(v - mu, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n

def _corr_vec(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _exact_c5(h):
    cv = _corr_vec(h)
    return float(np.max(cv))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_lag(h, d):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    ad = abs(d)
    if ad >= n:

    if d > 0:
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
    else:
        k = -d
        g[:n-k] += dx * (1.0 - h[k:])
        g[k:] -= dx * h[:n-k]
    return g

def _fw_step(h, corr, K, eta):
    """One Frank-Wolfe step. Returns (h_new, improved)."""
    n = len(h)
    target = n / 2.0
    # Find top K lags
    top_idx = np.argsort(corr)[::-1][:K]
    # Softmax weights
    top_corr = corr[top_idx]
    weights = np.exp((top_corr - top_corr[0]) / eta)
    weights /= weights.sum()
    # Weighted gradient (subgradient)
    grad = np.zeros(n)
    for i in range(K):
        d = int(top_idx[i]) - (n - 1)
        grad += weights[i] * _grad_lag(h, d)
    # LP vertex: minimize grad · x s.t. 0≤x≤1, sum(x)=n/2
    order = np.argsort(grad)
    s = np.zeros(n)
    s[order[:int(target)]] = 1.0
    # Line search
    best_t = 0.0
    best_val = _exact_c5(h)
    alphas = np.linspace(0.0, 1.0, 21)
    vals = np.zeros(len(alphas))
    for i, t in enumerate(alphas):
        h_t = (1.0 - t) * h + t * s
        vals[i] = _exact_c5(h_t)
    best_i = int(np.argmin(vals))
    if vals[best_i] < best_val - 1e-12:
        # Refine around best
        if best_i > 0:
            t_lo = alphas[best_i - 1]
        else:
            t_lo = 0.0
        if best_i < len(alphas) - 1:
            t_hi = alphas[best_i + 1]
        else:
            t_hi = 1.0
        fine = np.linspace(t_lo, t_hi, 11)
        for t in fine:
            h_t = (1.0 - t) * h + t * s
            v = _exact_c5(h_t)
            if v < best_val:
                best_val = v
                best_t = float(t)
        # Project to ensure feasibility
        h_new = (1.0 - best_t) * h + best_t * s
        h_new = _project(h_new, target)
        final_val = _exact_c5(h_new)
        if final_val < _exact_c5(h) - 1e-12:
            return h_new, True
    return h, False

def _pair_optimize(h, n_pairs, deadline):
    """Systematic pair optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    # Compute gradient magnitudes to rank coordinates
    corr = _corr_vec(h)
    top_idx = np.argsort(corr)[::-1][:5]
    # Sum absolute gradient contributions per coordinate
    coord_weight = np.zeros(n)
    for qi in top_idx:
        d = int(qi) - (n - 1)
        g = _grad_lag(h, d)
        coord_weight += np.abs(g)
    # Rank coordinates by weight
    coord_rank = np.argsort(coord_weight)[::-1]
    # Try pairs from top coordinates
    tried = set()
    improved_total = False
    attempt = 0
    while attempt < n_pairs and time.time() < deadline:
        # Pick a pair
        i = coord_rank[attempt % n]
        j = coord_rank[(attempt + n // 3) % n]
        if i == j:
            j = (j + 1) % n
        if (min(i,j), max(i,j)) in tried:
            attempt += 1
            continue
        tried.add((min(i,j), max(i,j)))
        # Optimize pair (i, j)
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-9:
            attempt += 1
            continue
        # Grid search over t (new value of h[i])
        grid = np.linspace(lo, hi, 17)
        best_t = h[i]
        best_val = _exact_c5(h)
        for t in grid:
            h_t = h.copy()
            h_t[i] = t
            h_t[j] = s - t
            val = _exact_c5(h_t)
            if val < best_val:
                best_val = val
                best_t = t
        if best_val < best_c5 - 1e-12:
            h[i] = best_t
            h[j] = s - best_t
            best_c5 = best_val
            improved_total = True
            # Update coord weights
            corrupted = False
        attempt += 1
    # Refine: second pass with finer grid on improved pairs
    if improved_total:
        for (i, j) in list(tried):
            if time.time() >= deadline:
                break

            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-9:
                continue
            grid = np.linspace(lo, hi, 33)
            best_t = h[i]
            best_val = _exact_c5(h)
            for t in grid:
                h_t = h.copy()
                h_t[i] = t
