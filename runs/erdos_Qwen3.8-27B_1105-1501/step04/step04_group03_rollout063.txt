def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def true_corr_grad(h, d):
    """Compute true gradient of s(d) = sum_i h[i]*(1-h[i+d]) wrt h."""
    n = len(h)
    g = np.zeros(n)
    jp = np.arange(n) + d
    jm = np.arange(n) - d
    mask_p = (jp >= 0) & (jp < n)
    mask_m = (jm >= 0) & (jm < n)
    g[mask_p] += (1.0 - h[jp[mask_p]])
    g[mask_m] -= h[jm[mask_m]]
    return g

    A_eq = np.zeros((1, n+1))

def subgradient_loop(h, deadline):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    y = h.copy()  # Nesterov momentum
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t = 1.0
    alpha = 0.1
    beta = 0.9  # decay
    stagnant = 0
    
    while time.time() < deadline:
        # Nesterov step
        t_next = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        theta = (t - 1.0) / t_next
        
        # Compute scores and find active shifts
        scores = np.correlate(y, 1.0-y, mode='full')
        max_s = float(np.max(scores))

        t = t_next

def periodic_solve(n, p, deadline):
    """Solve for p-periodic h with total length n."""
    if n % p != 0 or p < 2 or p > n//2:
        return None, None
    nrep = n // p
    # Variables: a[0..p-1], constraint: sum(a) = p/2
    # Objective: max over distinct shifts
    
    from scipy.optimize import minimize as sp_min
    
    def obj(params):
        a = project(params, p/2.0)
        h = np.tile(a, nrep)
        return exact_c5(h)
    
    # Initialize
    a0 = np.ones(p) / 2.0
    # Perturb
    rng = np.random.default_rng(42)
    a0 = project(a0 + rng.normal(0, 0.1, p), p/2.0)
    
    bounds = [(0.0, 1.0)]*p
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - p/2.0,
             'jac': lambda x: np.ones(p)}]
    
    try:
        res = sp_min(obj, a0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': 200, 'ftol': 1e-12})
        if res.success or (hasattr(res, 'fun') and np.isfinite(res.fun)):
            a = project(res.x, p/2.0)
            h = np.tile(a, nrep)
            c5 = exact_c5(h)
            return h, c5
    except Exception:
        pass
    return None, None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old)+0.5)/n_old
    x_new = (np.arange(n_new)+0.5)/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new/2.0)

def coord_descent(h, deadline, max_iter=50000):

    n = len(h)

        order = np.random.default_rng(it).permutation(n)
        for j in order:
            if time.time() > deadline:
                break

            # Simple approach: try h[j] + delta, then project

                best_h = h.copy()

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

    else:

    t = 1.0
    alpha = 0.05
    stagnant = 0
    top_m = 8
    while time.time() < deadline:
        scores = np.correlate(y, 1.0 - y, mode='full')
        k_top = np.argsort(scores)[-top_m:]
        g = np.zeros(n)
        for k in k_top:
            d = k - (n - 1)
            g += subgrad_row(y, d)
        g /= top_m
        y_new = y - alpha * g
        y_new = project(y_new, target)
        c5_new = exact_c5(y_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = y_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                alpha *= 0.9
            if stagnant > 300:
                break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_next
        h = theta * y + (1.0 - theta) * h
        h = project(h, target)
        y = y_new
        t = t_next
        alpha = min(alpha * 1.005, 0.3)
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def periodic_trial(n_total, p, rng):
    if n_total % p != 0 or p < 2 or p > n_total // 2:
        return None, None
    nrep = n_total // p
    def obj(params):
        a = project(params, p / 2.0)
        h = np.tile(a, nrep)
        return exact_c5(h)
    a0 = project(np.ones(p) / 2.0 + rng.normal(0, 0.15, p), p / 2.0)
    bounds = [(0.0, 1.0)] * p
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - p / 2.0,
             'jac': lambda x: np.ones(p)}]
    try:
        res = minimize(obj, a0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-13})
        if hasattr(res, 'x') and np.isfinite(res.fun):
            a = project(res.x, p / 2.0)
            h = np.tile(a, nrep)
            c5 = exact_c5(h)
            return h, c5
    except Exception:
        pass
    return None, None

def pair_polish_fast(h, deadline, max_passes=20):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(123)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(200):
            if time.time() > deadline:
                break
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                continue
            for sign in [1.0, -1.0]:
                if sign > 0:
                    amax = min(best_h[j], 1.0 - best_h[k])
                else:
                    amax = min(1.0 - best_h[j], best_h[k])
                if amax < 1e-10:
                    continue
                for frac in [0.5, 1.0, 0.25]:
                    a = amax * frac
                    h_t = best_h.copy()
                    h_t[j] += sign * a
                    h_t[k] -= sign * a
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_t
                        improved = True
                        break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.955

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: LP with corrected gradient
    d1 = t0 + budget_s * 0.18
    h_lp, c5_lp = lp_loop(h, d1, radius_init=0.15)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: Nesterov subgradient for diversification
    d2 = t0 + budget_s * 0.30
    h_sg, c5_sg = nesterov_subgrad(best_h, d2)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()

    # Phase 3: LP refinement
    d3 = t0 + budget_s * 0.45
    h_lp2, c5_lp2 = lp_loop(best_h, d3, radius_init=0.08)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 4: Multi-resolution LP
    d4 = t0 + budget_s * 0.68
    for dn in [-10, -5, 5, 10, -15, 15, -20, 20, -30, 30, -40, 40]:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 180:
            continue
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.07)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Periodic solutions
    d5 = t0 + budget_s * 0.78
    for p in [3, 4, 5, 6, 7, 8, 10, 12, 16]:
        if time.time() > d5:
            break
        for n_mult in [10, 12, 15, 20, 25, 30]:
            n_p = p * n_mult
            if n_p < 30 or n_p > 200:
                continue
            if time.time() > d5:
                break
            h_per, c5_per = periodic_trial(n_p, p, rng)
            if h_per is not None and c5_per < best_c5 - 1e-14:
                best_c5 = c5_per
                best_h = h_per.copy()
                best_n = n_p

    # Phase 6: Perturb and re-optimize
    d6 = t0 + budget_s * 0.88
    for scale in [0.015, 0.04, 0.08, 0.12]:
        if time.time() > d6:
            break
        for trial in range(4):
            if time.time() > d6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = project(h_pert, best_n / 2.0)
            sub_dl = min(d6, time.time() + budget_s * 0.008)
            h_try, c5_try = lp_loop(h_pert, sub_dl, radius_init=0.05)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 7: Pair polish
    if time.time() < hard_deadline:
        d7 = min(hard_deadline, time.time() + budget_s * 0.03)
        h_pol, c5_pol = pair_polish_fast(best_h, d7, max_passes=15)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 8: Final LP micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.02)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-15:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
