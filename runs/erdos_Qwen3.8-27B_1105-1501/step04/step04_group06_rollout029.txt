    # Also check endpoints more carefully
    for x in [0.0, 1.0, 0.5]:
        v = _eval_M(x, corr, B, C0, old_val, n)
        if v < best_val:
            best_val = v
            best_x = x
    # Golden section refinement around best_x
    lo = max(0.0, best_x - 0.0625)
    hi = min(1.0, best_x + 0.0625)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _eval_M(c1, corr, B, C0, old_val, n)
    f2 = _eval_M(c2, corr, B, C0, old_val, n)
    for _ in range(50):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _eval_M(c1, corr, B, C0, old_val, n)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _eval_M(c2, corr, B, C0, old_val, n)
    x_opt = (lo + hi) / 2.0
    f_opt = _eval_M(x_opt, corr, B, C0, old_val, n)
    if f_opt > best_val:
        x_opt = best_x
        f_opt = best_val
    return x_opt, f_opt


def _coord_descent_step(h, corr, n, order):
    """One pass of exact coordinate descent."""
    h = h.copy()
    corr = corr.copy()
    j_arr = np.arange(n)
    
    for i in order:
        old_val = h[i]
        # Compute B_s for all lags
        s_arr = np.arange(2 * n - 1) - (n - 1)
        B = np.zeros(2 * n - 1)
        idx1 = i - s_arr
        mask1 = (idx1 >= 0) & (idx1 < n)
        B[mask1] += (1.0 - h[idx1[mask1]])
        idx2 = i + s_arr
        mask2 = (idx2 >= 0) & (idx2 < n)
        B[mask2] -= h[idx2[mask2]]
        
        C0 = corr[n - 1] - old_val + old_val * old_val
        
        x_opt, f_opt = _solve_1d(corr, B, C0, old_val, n)
        
        if abs(x_opt - old_val) > 1e-15:
            corr = corr + (x_opt - old_val) * B
            corr[n - 1] = C0 + x_opt - x_opt * x_opt
            h[i] = x_opt
    
    return h, corr


def _coord_descent_loop(h, n, max_sweeps=2000, seed=42, deadline=None):
    """Run coordinate descent with random coordinate ordering."""
    rng = np.random.default_rng(seed)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        h, corr = _coord_descent_step(h, corr, n, order)
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
    return best_h, best_c5


def _slsqp_solve(h_init, n, maxiter=300, ftol=1e-13):
    """SLSQP epigraph formulation."""
    h = np.asarray(h_init, dtype=float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    
    def _epi_obj(z):
        return z[n]
    
    def _epi_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def _eq_fun(z):
        return np.sum(z[:n]) - n * 0.5
    
    def _eq_jac(z):
        g = np.ones(n + 1)
        g[n] = 0.0
        return g
    
    def _ineq_fun(z):
        hz = z[:n]
        return z[n] - np.correlate(hz, 1.0 - hz, mode='full')
    
    def _ineq_jac(z):
        hz = z[:n]
        nc = 2 * n - 1
        J = np.zeros((nc, n + 1))
        j = np.arange(n)
        for r in range(nc):
            s = r - (n - 1)
            pd = np.zeros(n)
            gs = j - s
            m1 = (gs >= 0) & (gs < n)
            pd[m1] = 1.0 - hz[gs[m1]]
            ps = j + s
            m2 = (ps >= 0) & (ps < n)
            pd[m2] -= hz[ps[m2]]
            J[r, :n] = -pd
            J[r, n] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _gen_binary_start(n, rng, method=0):
    """Generate a binary-like starting vector."""
    if method == 0:
        # Random binary
        mask = rng.random(n) < 0.5
        # Adjust to get exactly n/2 ones
        n_ones = mask.sum()
        if n_ones > n // 2:
            one_idx = np.where(mask)[0]
            to_flip = rng.choice(one_idx, n_ones - n // 2, replace=False)
            mask[to_flip] = False
        elif n_ones < n // 2:
            zero_idx = np.where(~mask)[0]
            to_flip = rng.choice(zero_idx, n // 2 - n_ones, replace=False)
        for i in n_ones + (n // 2 - n_ones):
            mask[i] = True
        h = mask.astype(float)
    elif method == 1:
        # Quadratic residue based
        h = np.zeros(n)
        for i in range(n):
            val = (i * i) % n
            if val < n // 2:
                h[i] = 1.0
        # Adjust to get exactly n/2
        if h.sum() != n // 2:
            h = _project(h, n * 0.5)
    elif method == 2:
        # Alternating with perturbation
        h = np.zeros(n)
        for i in range(0, n, 2):
            h[i] = 1.0
        h = _project(h + rng.normal(0, 0.1, n), n * 0.5)
    else:
        # Smoothed version of random binary
        h = np.zeros(n)
        for i in range(0, n, 2):
            h[i] = 1.0
        h = _project(h + rng.normal(0, 0.3, n), n * 0.5)
    return _project(h, n * 0.5)


def _lp_polish(h, n, deadline, max_iters=2000):
    """LP-based local polishing."""
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    j_arr = np.arange(n)
    stall = 0
    radius = 0.1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        # Find descent direction via subgradient of max
        r_max = int(np.argmax(scores))
        s = r_max - (n - 1)
        pd = np.zeros(n)
        gs = j_arr - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j_arr + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        pd -= np.mean(pd)
        norm = np.linalg.norm(pd)
        if norm < 1e-14:
            break
        pd = pd / norm
        
        # Line search
        alpha_max = 0.3
        for jj in range(n):
            if pd[jj] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[jj]) / pd[jj])
            elif pd[jj] < -1e-14:
                alpha_max = min(alpha_max, h[jj] / (-pd[jj]))
        
        best_a, best_c5_try = 0.0, cur_c5
        for t in np.linspace(0.0, 1.0, 15):
            a = t * min(alpha_max, radius)
            if a < 1e-12:
                continue
            h_t = _project(h - a * pd, n * 0.5)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5_try:
                best_c5_try, best_a = c5_t, a
        
        if best_a > 0 and best_c5_try < cur_c5 - 1e-13:
            h = _project(h - best_a * pd, n * 0.5)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
        else:
            stall += 1
            radius *= 0.85
            if radius < 1e-5:
                break
    
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Initialize from parent
    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    def _elapsed():
        return time.time() - t0
    
    # Phase 1: Coordinate descent from parent (35% budget)
    d1 = t0 + budget_s * 0.35
    h_cd, c5_cd = _coord_descent_loop(h, n0, max_sweeps=5000, seed=seed, deadline=d1)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()
    
    # Phase 2: SLSQP polish (15% budget)
    d2 = t0 + budget_s * 0.50
    h_s, c5_s = _slsqp_solve(best_h, best_n, maxiter=400, ftol=1e-14)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    # More coord descent after SLSQP
    if time.time() < d2:
        h_cd2, c5_cd2 = _coord_descent_loop(best_h, best_n, max_sweeps=2000,
                                             seed=seed + 1, deadline=d2)
        if c5_cd2 < best_c5:
            best_c5, best_h = c5_cd2, h_cd2.copy()
    
    # Phase 3: LP polish (10% budget)
    d3 = t0 + budget_s * 0.60
    h_lp, c5_lp = _lp_polish(best_h, best_n, d3, max_iters=3000)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()
    
    # Phase 4: Multiple restarts with different initializations (25% budget)
    d4 = t0 + budget_s * 0.85
    methods = [0, 1, 2, 3]
    n_restarts = 40
    attempts = 0
    for m in range(4):
        if time.time() > d4:
            break
        for rep in range(n_restarts // 4):
            if time.time() > d4:
                break
            if best_c5 <= 0.3807:
                break
            h_start = _gen_binary_start(n0, rng, method=methods[m])
            # Quick coord descent
            sub_deadline = min(d4, time.time() + budget_s * 0.015)
            h_r, c5_r = _coord_descent_loop(h_start, n0, max_sweeps=500,
                                             seed=seed + 100 * m + rep,
                                             deadline=sub_deadline)
            # SLSQP if promising
            if c5_r < best_c5 + 0.002 and time.time() < d4 - budget_s * 0.01:
                sub_d2 = min(d4, time.time() + budget_s * 0.015)
                h_s2, c5_s2 = _slsqp_solve(h_r, n0, maxiter=200, ftol=1e-13)
                if c5_s2 < c5_r:
                    h_r, c5_r = h_s2, c5_s2
                # More coord descent
                sub_d3 = min(d4, time.time() + budget_s * 0.012)
                h_r2, c5_r2 = _coord_descent_loop(h_r, n0, max_sweeps=500,
                                                   seed=seed + 200 * m + rep,
                                                   deadline=sub_d3)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
            if c5_r < best_c5 - 1e-12:
                best_c5, best_h = c5_r, h_r.copy()
            attempts += 1
    
    # Phase 5: Resolution search (8% budget)
    d5 = t0 + budget_s * 0.93
    for dn in [2, -2, 4, -4, 6, -6, 8, 12, -8]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub1 = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _coord_descent_loop(h_r, n_try, max_sweeps=500,
                                         seed=seed + 500, deadline=sub1)
        sub2 = min(d5, time.time() + budget_s * 0.01)
        h_s, c5_s = _slsqp_solve(h_r, n_try, maxiter=150, ftol=1e-13)
        if c5_s < c5_r:
            h_r, c5_r = h_s, c5_s
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try
    
    # Phase 6: Final polish (7% budget)
    d6 = t0 + budget_s * 0.99
    h_f, c5_f = _coord_descent_loop(best_h, best_n, max_sweeps=3000,
                                     seed=seed + 999, deadline=d6)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    h_lp, c5_lp = _lp_polish(best_h, best_n, d6, max_iters=1000)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()
    h_s, c5_s = _slsqp_solve(best_h, best_n, maxiter=200, ftol=1e-15)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    h_final_cd, c5_final_cd = _coord_descent_loop(best_h, best_n, max_sweeps=1000,
                                                   seed=seed + 777,
                                                   deadline=min(d6, time.time() + 3))
    if c5_final_cd < best_c5:
        best_c5, best_h = c5_final_cd, h_final_cd.copy()
    
    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
