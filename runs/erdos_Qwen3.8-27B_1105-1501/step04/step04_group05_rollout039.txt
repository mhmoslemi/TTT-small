import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _g_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _g_k_grad(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        idx = np.arange(n - d)
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] -= h[idx]
    else:
        dp = -d
        idx = np.arange(dp, n)
        grad[idx] += (1.0 - h[idx - dp])
        grad[idx - dp] -= h[idx]
    return grad


def _slsqp_active(h, active, maxiter=600, ftol=1e-16):
    n = len(h)
    target = n / 2.0
    g_full = _g_vec(h)
    max_g = float(np.max(g_full))

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def ineq_fun(x):
        h = x[1:]
        g = _g_vec(h)
        return x[0] - g[active]

    def ineq_jac(x):
        h = x[1:]
        m = len(active)
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        for i, k in enumerate(active):
            J[i, 1:] = -_g_k_grad(h, k)
        return J

    def eq_fun(x):
        return x[1:].sum() - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    x0 = np.zeros(n + 1)
    x0[0] = max_g * 1.001 + 0.02
    x0[1:] = h

    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h.copy()


def _subgradient(h, top_k=5):
    g = _g_vec(h)
    top_idx = np.argsort(g)[-top_k:]
    grad = np.zeros(len(h))
    for k in top_idx:
        grad += _g_k_grad(h, int(k))
    grad /= len(top_idx)
    return grad


def _nesterov_loop(h, deadline, start_lr=0.005, min_lr=1e-6, top_k=5, patience=60):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    velocity = np.zeros(n)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = start_lr
    stall = 0
    mu = 0.85

    for _ in range(10000):
        if time.time() > deadline:
            break
        if stall > patience:
            break
        grad = _subgradient(h_cur, top_k)
        grad -= np.mean(grad)
        velocity = mu * velocity + lr * grad
        h_new = _project(h_cur - velocity, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            stall = 0
            lr = min(lr * 1.05, 0.03)
        elif c5_new < _exact_c5(h_cur) - 1e-14:
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr = max(lr * 0.85, min_lr)
            if lr <= min_lr:
                break
    return best_h, best_c5


def _pair_descent(h, deadline, max_passes=150, n_samp=16, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if rng is None:
        rng = np.random.default_rng(123)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 1e-7) & (h < 1.0 - 1e-7))[0]
        if len(interior) < 2:
            break

        order = np.arange(len(interior))
        rng.shuffle(order)
        idx_list = interior[order]

        for ai in range(len(idx_list) - 1):
            i = int(idx_list[ai])
            if time.time() > deadline:
                break
            h_i = h[i]
            if h_i <= 1e-7 or h_i >= 1.0 - 1e-7:
                continue
            for ak in range(ai + 1, len(idx_list)):
                j = int(idx_list[ak])
                h_j = h[j]
                if h_j <= 1e-7 or h_j >= 1.0 - 1e-7:
                    continue
                p = h_i + h_j
                lo = max(0.0, p - 1.0)
                hi = min(1.0, p)
                if hi - lo < 1e-10:
                    continue

                h[i] = lo
                h[j] = p - lo
                c5_lo = _exact_c5(h)
                h[i] = hi
                h[j] = p - hi
                c5_hi = _exact_c5(h)
                h[i] = h_i
                h[j] = h_j

                if c5_lo >= best_c5 - 1e-15 and c5_hi >= best_c5 - 1e-15:
                    t_mid = p * 0.5
                    h[i] = t_mid
                    h[j] = p - t_mid
                    c5_mid = _exact_c5(h)
                    h[i] = h_i
                    h[j] = h_j
                    if c5_mid >= best_c5 - 1e-15:
                        continue

                ts = np.linspace(lo, hi, n_samp)
                best_t = h_i
                best_local = best_c5
                for t in ts:
                    h[i] = t
                    h[j] = p - t
                    c5_t = _exact_c5(h)
                    if c5_t < best_local:
                        best_local = c5_t
                        best_t = t

                if best_local < best_c5 - 1e-14:
                    span = (hi - lo) / n_samp
                    for _r in range(4):
                        lo2 = max(lo, best_t - span)
                        hi2 = min(hi, best_t + span)
                        ts2 = np.linspace(lo2, hi2, 12)
                        for t in ts2:
                            h[i] = t
                            h[j] = p - t
                            c5_t = _exact_c5(h)
                            if c5_t < best_local:
                                best_local = c5_t
                                best_t = t
                    h[i] = best_t
                    h[j] = p - best_t
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, target)
                    best_c5 = best_local
                    best_h = h.copy()
                    improved = True

        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _get_active(g_full, frac=0.4, min_count=8):
    max_g = float(np.max(g_full))
    if max_g <= 0:
        return np.arange(len(g_full))
    threshold = max_g * frac
    active = np.where(g_full >= threshold)[0]
    if len(active) < min_count:
        k = min(min_count * 2, len(g_full))
        active = np.argsort(g_full)[-k:]
    return active
