            lo_b = max(-radius, -h[j])
            hi_b = min(radius, 1.0 - h[j])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        p = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'maxiter': 5000})
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def _weighted_grad(h, deadline, n_steps=200, gamma_init=20.0, gamma_end=300.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        gamma = gamma_init + (gamma_end - gamma_init) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(gamma * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.1
        improved = False
        for _ in range(35):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-11:
                break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(n, deadline, rng, n_iters=50000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    n_safe = max(n_iters, 1)
    cool = (T_min / T) ** (1.0 / n_safe)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _block_sa(n, deadline, rng, n_iters=30000, block_sizes=None):
    if block_sizes is None:
        block_sizes = [1, 2, 3, 4, 5, 8, 12]
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.04
    T_min = 0.0002
    n_safe = max(n_iters, 1)
    cool = (T_min / T) ** (1.0 / n_safe)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        bs = block_sizes[rng.integers(len(block_sizes))]
        bs = min(bs, n // 4)
        i = rng.integers(0, n - bs + 1)
        j = rng.integers(0, n - bs + 1)
        slice_i = slice(i, i + bs)
        slice_j = slice(j, j + bs)
        if np.any(np.arange(i, i+bs) >= np.arange(j, j+bs)) and (i + bs > j and j + bs > i):
            continue
        old_i = h[slice_i].copy()
        old_j = h[slice_j].copy()
        h[slice_i] = old_j
        h[slice_j] = old_i
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[slice_i] = old_i
            h[slice_j] = old_j
        T *= cool
    return best_h, best_c5

def _fourier_evolve(h, deadline, n_pop=20, K=10, n_gen=200, mutation_scale=0.1, rng=None):
    n = len(h)
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(2 * np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    target = n / 2.0
    a_init = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    pop_a = a_init + mutation_scale * rng.standard_normal((n_pop, 2*K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
        h_try = _project(h_try, target)
        pop_c5[i] = _exact_c5(h_try)
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = np.clip(0.5 + B @ pop_a[best_i], 0, 1)
    best_h = _project(best_h, target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        n_keep = max(2, n_pop // 2)
        pop_a[:n_keep] = pop_a[order[:n_keep]]
        pop_c5[:n_keep] = pop_c5[order[:n_keep]]
        for i in range(n_keep, n_pop):
            p1 = order[rng.integers(0, n_keep)]
            p2 = order[rng.integers(0, n_keep)]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1.0 - mix) * pop_a[order[p2]]
            pop_a[i] += mutation_scale * rng.standard_normal(2*K)
            h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
            h_try = _project(h_try, target)
            pop_c5[i] = _exact_c5(h_try)
        mutation_scale *= 0.995
        cur_best = int(np.argmin(pop_c5))
        if pop_c5[cur_best] < best_c5:
            best_c5 = pop_c5[cur_best]
            best_h = np.clip(0.5 + B @ pop_a[cur_best], 0, 1)
            best_h = _project(best_h, target)
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _coord_descent(h, deadline, max_rounds=10):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            a, c5_a = _line_search_1d(h, j, target)
            if c5_a < best_c5 - 1e-12:
                h[j] += a
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _line_search_1d(h, j, target):
    h2 = h.copy()
    v0 = h2[j]
    best_a = 0.0
    best_c5 = _exact_c5(h)
    for a in [0.02, 0.05, 0.1, 0.2, 0.3]:
        for sign in [1, -1]:
            h2[j] = v0 + sign * a
            if h2[j] < 0 or h2[j] > 1:
                continue
            h2 = _project(h2, target)
            c5_try = _exact_c5(h2)
            if c5_try < best_c5:
                best_a = sign * a
                best_c5 = c5_try
    return best_a, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP from initial with correct gradients
    dl1 = t0 + budget_s * 0.15
    h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.25, max_iters=3000, top_k=None)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Try with top-k as well
    if time.time() < dl1:
        h1b, c5_1b = _lp_optimize(best_h, min(dl1, time.time() + 60), radius_init=0.15,
                                   max_iters=2000, top_k=40)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Weighted gradient
    if time.time() < t0 + budget_s * 0.28:
        dl2 = t0 + budget_s * 0.32
        h2, c5_2 = _weighted_grad(best_h, dl2, n_steps=200, gamma_init=15.0, gamma_end=400.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        # LP polish after gradient
        if time.time() < dl2:
            h2b, c5_2b = _lp_optimize(best_h, min(dl2, time.time() + 40),
                                       radius_init=0.1, max_iters=1000, top_k=30)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Multi-resolution search
    dl3 = t0 + budget_s * 0.55
    resolutions = [96, 128, 160, 192, 256, 144, 176]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.75, 0.25]
    
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if best_c5 < 0.3806:
            break
        n_starts = 2
        for s in range(n_starts):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_periodic_init(n_try, alpha)
                h_init += rng.normal(0, 0.1, n_try)
                h_init = _project(h_init, n_try / 2.0)
            
            sub_dl = min(dl3, time.time() + budget_s * 0.04)
            h_try, c5_try = _lp_optimize(h_init, sub_dl, radius_init=0.2,
                                          max_iters=2000, top_k=None)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                if time.time() < dl3:
                    sub_dl2 = min(dl3, time.time() + budget_s * 0.015)
                    h_wg, c5_wg = _weighted_grad(h_try, sub_dl2, n_steps=100,
                                                   gamma_init=20, gamma_end=200)
                    if c5_wg < best_c5:
                        best_c5 = c5_wg
                        best_h = h_wg.copy()
                        best_n = n_try

    # Phase 4: Fourier evolution
    if time.time() < t0 + budget_s * 0.68:
        dl4 = t0 + budget_s * 0.74
        h_f, c5_f = _fourier_evolve(best_h, dl4, n_pop=25, K=12,
                                      n_gen=300, mutation_scale=0.08, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if time.time() < dl4:
            h_f2, c5_f2 = _fourier_evolve(h_start, dl4, n_pop=20, K=8,
                                            n_gen=200, mutation_scale=0.15, rng=rng)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()
        if time.time() < dl4 and best_c5 < 0.3810:
            sub_dl = min(dl4, time.time() + budget_s * 0.03)
            h_fp, c5_fp = _lp_optimize(best_h, sub_dl, radius_init=0.08,
                                         max_iters=1000, top_k=25)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 5: Binary SA with block moves
    if time.time() < t0 + budget_s * 0.80:
        dl5 = t0 + budget_s * 0.86
        for n_bin in [128, 160, 192]:
            if time.time() > dl5:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h_b, c5_b = _block_sa(n_bin, sub_dl, rng, n_iters=30000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
                h_br, c5_br = _lp_optimize(h_b, sub_dl2, radius_init=0.1,
                                             max_iters=800, top_k=20)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()
            if time.time() < dl5:
                sub_dl2 = min(dl5, time.time() + budget_s * 0.025)
                h_b2, c5_b2 = _binary_sa(n_bin, sub_dl2, rng, n_iters=30000)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_b2.copy()
                    best_n = n_bin

    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.93
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.08,
                                       max_iters=400, top_k=30)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 7: Final polishing
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.985
        hp, c5p = _lp_optimize(best_h, dl7, radius_init=0.02, max_iters=2000, top_k=None)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()
        if time.time() < dl7:
            hg, c5g = _weighted_grad(best_h, min(dl7, time.time() + 30),
                                       n_steps=100, gamma_init=100, gamma_end=500)
            if c5g < best_c5:
                best_c5 = c5g
                best_h = hg.copy()

    # Final: tiny LP refinement
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _lp_optimize(best_h, t0 + budget_s * 0.998,
                                       radius_init=0.005, max_iters=500, top_k=15)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
