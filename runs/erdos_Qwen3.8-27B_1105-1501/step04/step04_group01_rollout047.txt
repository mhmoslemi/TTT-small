Actually, I realize I should be more careful about the line search. In the pair case (d has only two nonzero entries), the correlation change is:
S0 + t*T - t²*C

where T and C can be computed efficiently for a pair direction.

For a pair direction d with d[i] = -1, d[j] = 1:
- The change in correlation at shift r is: -h[i]*(1-h[i+r-(n-1)]) ... no wait, it's more complex.

Let me just use the general formula and not optimize the pair case specially. The cost is O(n) per correlation evaluation (since we only need to update the correlations that involve indices i and j).

Actually, for the general line search, we need:
- S0[k] = corr_k (O(n) to compute each, or O(n²) total)
- T[k] = derivative of corr_k w.r.t. t (which is the directional derivative)
- C[k] = second derivative term

For a general direction d:
T[k] = ∑_i d_i (1 - h_{i+k}) + h_i (-d_{i+k}) ... this is the correlation of d with (1-h) plus correlation of (1-d) with h... 

Actually, from the previous code:
T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
C = np.correlate(d, d, mode='full')

Each of these is O(n²) using np.correlate. For n=112, that's fast.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _grad_row(h, r):
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d + m])
            g[d:] += -h[:m]
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += -h[gg:gg + m]
            g[gg:] += (1.0 - h[:m])
    return g

def _lp_dir(h, scores, radius, K=15):
    n = len(h)
    nsc = len(scores)
    if K >= nsc:
        idx = np.arange(nsc)
    else:
        idx = np.argsort(scores)[-K:]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    b_ub = np.zeros(len(idx))
    for ii, r in enumerate(idx):
        g = _grad_row(h, r)
        A_ub[ii, :n] = g
        A_ub[ii, n] = -1.0
        b_ub[ii] = -scores[r]
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b > hi_b:
            lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    K = 12
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_dir(h, scores, radius, K)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            K = min(K + 3, 2 * n - 1)
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.35)
            K = max(8, K - 1)
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.08
                K = min(K + 5, 2 * n - 1)
    return best_h, best_c5

def _coord_descent(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        free_idx = np.where(free)[0]
        if len(free_idx) < 2:
            break
        for ii in range(len(free_idx)):
            if time.time() > deadline:
                break
            i = free_idx[ii]
            for jj in range(ii + 1, len(free_idx)):
                j = free_idx[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _subgrad_descent(h, deadline, lr_init=0.015):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_s = float(np.max(scores))
        active = scores > max_s - 0.05 * max_s
        cnt = int(np.sum(active))
        if cnt == 0:
            break
        grad = np.zeros(n)
        for r in range(len(scores)):
            if active[r]:
                grad += _grad_row(h, r)
        grad /= cnt
        grad -= np.mean(grad)
        mom = 0.8 * mom + 0.2 * grad
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            stall += 1
            continue
        d = -mom / mn
        alpha, new_c5 = _line_search(h, d, dx, lr)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.3)
        else:
            stall += 1
            lr *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_init(n, seed):
    rng = np.random.default_rng(seed)
    h = rng.uniform(0, 1, n)
    h = _project(h, n / 2.0)
    return h
