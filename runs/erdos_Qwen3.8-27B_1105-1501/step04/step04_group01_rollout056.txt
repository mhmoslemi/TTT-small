import time

def _exact_c5(h):

def _all_c5_values(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_gradient(h, lag_idx):
    """Compute gradient of correlation at lag_idx with respect to h."""
    n = len(h)
    d = lag_idx - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:d+m])
            grad[d:] += -h[:m]
    else:
        g = -d
        m = n - g
        if m > 0:
            grad[:m] += -h[g:g+m]
            grad[g:] += (1.0 - h[:m])
    return grad

def _lp_minimax(h, scores, radius, max_active=None):
    """Solve LP: minimize t s.t. score_l(h+p) <= t, ||p||_inf <= radius, sum(p)=0, bounds on h+p."""
    n = len(h)
    n_lags = 2 * n - 1
    
    # Select active lags
    if max_active is not None and max_active < n_lags:
        sorted_idx = np.argsort(scores)[::-1]
        active = sorted_idx[:max_active]
    else:
        active = np.arange(n_lags)
    
    n_active = len(active)
    
    # c: minimize t (variable index n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: grad_l · p - t <= -score_l for each active lag
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for k, lag_idx in enumerate(active):
        g = _corr_gradient(h, lag_idx)
        A_ub[k, :n] = g
        A_ub[k, n] = -1.0
        b_ub[k] = -scores[lag_idx]
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on p
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def _quadratic_line_search(h, p, dx, radius):
    """Line search using quadratic model of the max correlation."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    # Linear term: d/dt of corr(h + t*p, 1 - h - t*p) at t=0
    T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
    # Quadratic term
    C = np.correlate(p, p, mode='full')
    
    # Bounds on t
    t_lo, t_hi = 0.0, radius
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_lo = max(0.0, float(np.max(h[neg] / (-p[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Golden section search on the quadratic model
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _logsumexp_gradient(h, beta, dx):
    """Compute gradient of smoothed max using log-sum-exp."""
    n = len(h)
    scores = _all_c5_values(h)
    max_s = float(np.max(scores))
    exp_s = np.exp(beta * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    grad = np.zeros(n)
    active_lags = np.where(w > 1e-10)[0]
    for r in active_lags:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g+m])
                grad[g:] += w[r] * (1.0 - h[:m])

def _nesterov_ascent(h, deadline, beta_list=(20, 50, 100, 200, 400), lr_init=0.02):

def _block_flip_perturb(h, rng, n_blocks=3, block_size=4):
    """Perturb by flipping contiguous blocks."""
    n = len(h)
    h_new = h.copy()
    for _ in range(n_blocks):
        start = rng.integers(0, max(1, n - block_size))
        end = min(n, start + block_size)
        # Flip: h -> 1-h in the block
        h_new[start:end] = 1.0 - h_new[start:end]
    return _project(h_new, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_snap(h, threshold=0.5):

    h_new = h.copy()
    h_new[h_new > threshold] = 1.0

    val = float(np.max(S0 + t_opt * T - t_opt*t_opt * C)) * dx
    return t_opt, val

def _lp_loop(h, deadline, radius_init=0.1, k_active=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _all_scores(h)
        p = _lp_active(h, scores, radius, k_active)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.65
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search_quad(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.5)
        else:
            stall += 1
            radius *= 0.68
            if stall % 25 == 0 and radius < 0.02:
                radius = 0.1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _block_perturb(h, rng, n_blocks=2, block_len=3):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_len)))
        end = start + block_len
        h_new[start:end] = 1.0 - h_new[start:end]
    return _project(h_new, n / 2.0)

def _sigmoid_perturb(h, rng, strength=0.3):
    n = len(h)
    shift = rng.normal(0, strength, n)
    h_new = 1.0 / (1.0 + np.exp(-8.0 * (h + shift - 0.5)))
    return _project(h_new, n / 2.0)

def _smoothed_grad(h, beta):
    n = len(h)
    scores = _all_scores(h)
    max_s = float(np.max(scores))
    exp_s = np.exp(beta * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g+m])
                grad[g:] += w[r] * (1.0 - h[:m])
    return grad

def _nesterov_phase(h, deadline, beta_start=30, beta_end=500):
    n = len(h)
    target = n / 2.0
    x = h.copy()
    y = h.copy()
    cur_c5 = _exact_c5(x)
    best_h = x.copy()
    best_c5 = cur_c5
    k = 0
    total_steps = 800
    while time.time() < deadline:
        k += 1
        frac = min(1.0, k / total_steps)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = 0.02 / (1.0 + 0.01 * k)
        grad = _smoothed_grad(y, beta)
        grad -= grad.mean()
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * float(k) * float(k))) / 2.0
        momentum = (t_k - 1.0) / (t_k + 1.0)
        y_new = y + momentum * (y - x) - lr * grad
        x = _project(y_new, target)
        y = x.copy()
        cur_c5 = _exact_c5(x)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = x.copy()
    return best_h, best_c5

def _pair_refine(h, deadline, rng, max_passes=20):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) < 2:
            break
        m = len(interior)
        n_try = min(m * (m - 1) // 2, 3000)
        for _ in range(n_try):
            if time.time() > deadline:
                break
            ii = int(rng.integers(0, m))
            jj = int(rng.integers(0, m))
            if ii == jj:
                continue
            i, j = interior[ii], interior[jj]
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search_quad(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _systematic_pairs(h, deadline, max_passes=15):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        m = len(interior)
        for ii in range(m):
            if time.time() > deadline:
                break
            i = interior[ii]
            for jj in range(ii + 1, m):
                j = interior[jj]
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search_quad(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Nesterov smoothed (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _nesterov_phase(h, d1, beta_start=20, beta_end=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Active-set LP (30%)
    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.12, k_active=35)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution (20%)
    d3 = t0 + budget_s * 0.58
    best_n = len(best_h)
    offsets = [-4, 4, -6, 6, -8, 8, -3, 3, -10, 10, -2, 2, -12, 12, -16, 16, -24, 24, -14, 14, -20, 20, -28, 28, -18, 18]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 36 or n_try > 220:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.014)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.1, k_active=30)
        rem = d3 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d3, time.time() + max(3, budget_s * 0.008))
            h_r, c5_r2 = _nesterov_phase(h_r, sub_dl2, beta_start=50, beta_end=200)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r  # keep
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: LP again with smaller radius (12%)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.06, k_active=50)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Perturbation + re-optimize (17%)
    d5 = t0 + budget_s * 0.87
    scale = 0.04
    for attempt in range(300):
        if time.time() > d5:
            break
        if best_c5 < 0.3805:
            break
        r = rng.random()
        if r < 0.35:
            h_pert = best_h + rng.normal(0, scale, len(best_h))
            h_pert = _project(h_pert, len(best_h) / 2.0)
        elif r < 0.65:
            nb = int(rng.integers(1, 5))
            bl = int(rng.integers(2, 8))
            h_pert = _block_perturb(best_h, rng, n_blocks=nb, block_len=bl)
        else:
            h_pert = _sigmoid_perturb(best_h, rng, strength=scale * 1.5)
        sub_dl = min(d5, time.time() + budget_s * 0.006)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08, k_active=30)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.008, scale * 0.87)
        else:
            scale = min(0.18, scale * 1.08)

    # Phase 6: Systematic pair refinement (7%)
    d6 = t0 + budget_s * 0.94
    h_p, c5_p = _systematic_pairs(best_h, d6, max_passes=12)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 7: Final LP polish
    d7 = t0 + budget_s * 0.98
    h_f, c5_f = _lp_loop(best_h, d7, radius_init=0.03, k_active=60)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
