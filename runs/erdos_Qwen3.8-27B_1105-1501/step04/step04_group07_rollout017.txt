    #   Gk @ delta - tau <= -sk  =>  Gk @ delta - tau + sk <= 0
    #   sum(delta) = 0

    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:g + m]
    return G * dx

def _lp_full(h, rho, top_k=None):
    n = len(h)
    L = 2 * n - 1
    scores = _scores(h)
    G = _grad_matrix(h)
    if top_k is not None and top_k < L:
        idx = np.argsort(scores)[-top_k:]
        sk = scores[idx]
        Gk = G[idx, :]
    else:
        sk = scores
        Gk = G
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0
    A_ub = np.zeros((len(sk), n_vars))
    A_ub[:, :n] = Gk
    A_ub[:, n] = -1.0
    b_ub = -sk
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_b = max(-rho, -h[i])
        hi_b = min(rho, 1.0 - h[i])
        if lo_b > hi_b:
            lo_b, hi_b = hi_b, lo_b
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if not res.success:
        return None
    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-14:
        return None
    return delta

def _line_search(h, delta):
    n = len(h)
    cur_c5 = _exact_c5(h)
    alpha_max = 1.0
    for i in range(n):
        if delta[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / delta[i])
        elif delta[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-delta[i]))
    alpha_max = min(alpha_max, 1.0)
    if alpha_max < 1e-10:
        return h, cur_c5, False
    n_a = 16
    alphas = np.linspace(0.05, alpha_max, n_a)
    best_a = 0.0
    best_c5 = cur_c5
    for a in alphas:
        h_try = h + a * delta
        if h_try.min() < -1e-9 or h_try.max() > 1.0 + 1e-9:
            continue
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_a = a
    if best_a == 0.0:
        return h, cur_c5, False
    idx_best = int(np.argmin(np.abs(alphas - best_a)))
    if 0 < idx_best < n_a - 1:
        lo = alphas[idx_best - 1]
        hi = alphas[idx_best + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(30):
            if hi - lo < 1e-10:
                break
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _exact_c5(h + c1 * delta)
            f2 = _exact_c5(h + c2 * delta)
            if f1 < f2:
                hi, f2 = c2, f1
            else:
                lo, f1 = c1, f2
        a_mid = (lo + hi) / 2.0
        c5_mid = _exact_c5(h + a_mid * delta)
        if c5_mid < best_c5:
            best_c5 = c5_mid
            best_a = a_mid
    new_h = h + best_a * delta
    new_h = _project(new_h, n / 2.0)
    final_c5 = _exact_c5(new_h)
    if final_c5 < cur_c5 - 1e-13:
        return new_h, final_c5, True
    return h, cur_c5, False

def _pnorm_direction(h, p):
    n = len(h)
    scores = _scores(h)
    s_max = float(scores.max())
    if s_max < 1e-15:
        return None
    ratio = scores / s_max
    w = ratio ** p
    sw = float(w.sum())
    if sw < 1e-300:
        w = np.zeros_like(scores)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return -grad / gn

def _main_loop(h, deadline, rho_init=0.12, use_pnorm=False, p_start=8.0, p_end=4096.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rho = rho_init
    stall = 0
    step_count = 0
    while time.time() < deadline:
        step_count += 1
        if use_pnorm and step_count % 3 == 0:
            frac = min(1.0, step_count / 300.0)
            p = p_start * (p_end / p_start) ** frac
            d = _pnorm_direction(h, p)
            if d is not None:
                new_h, new_c5, improved = _line_search(h, d * rho)
                if improved:
                    h = new_h
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    rho = min(rho * 1.01, 0.3)
                    continue
        delta = _lp_full(h, rho)
        if delta is None:
            stall += 1
            rho *= 0.75
            if stall > 60:
                if rho > 0.01:
                    rho = 0.08
                    stall = 0
                else:
                    break
            continue
        new_h, new_c5, improved = _line_search(h, delta)
        if improved:
            h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            rho = min(rho * 1.015, 0.3)
        else:
            stall += 1
            rho *= 0.7
            if stall > 80:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _symmetrize(h):
    n = len(h)
    h_sym = (h + h[::-1]) / 2.0
    return _project(h_sym, n / 2.0)

def _binary_init(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _binary_hill_climb(h_init, deadline, max_iters=50000):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng2 = np.random.default_rng(1)
        rng2.shuffle(idx)
        h[idx[ones_count - target:]] = 0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng2 = np.random.default_rng(1)
        rng2.shuffle(idx)
        h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 8000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline_final = t0 + budget_s * 0.99

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    rng = np.random.default_rng(seed)

    # Phase 1: Quick LP from parent (8%)
    dl1 = t0 + budget_s * 0.08
    h_try, c5_try = _main_loop(h0, dl1, rho_init=0.15, use_pnorm=False)
    if c5_try < best_c5:
        best_c5 = c5_try
        best_h = h_try.copy()

    # Phase 2: Symmetric attempt (7%)
    dl2 = t0 + budget_s * 0.15
    h_sym = _symmetrize(best_h)
    h_try, c5_try = _main_loop(h_sym, dl2, rho_init=0.12, use_pnorm=False)
    if c5_try < best_c5:
        best_c5 = c5_try
        best_h = h_try.copy()
        best_n = len(h_try)

    # Phase 3: P-norm + LP hybrid main engine (30%)
    dl3 = t0 + budget_s * 0.45
    h_try, c5_try = _main_loop(best_h, dl3, rho_init=0.10, use_pnorm=True, p_start=6.0, p_end=2048.0)
    if c5_try < best_c5:
        best_c5 = c5_try
        best_h = h_try.copy()
        best_n = len(h_try)

    # Phase 4: Pure full-lag LP refinement (15%)
    dl4 = t0 + budget_s * 0.60
    h_try, c5_try = _main_loop(best_h, dl4, rho_init=0.06, use_pnorm=False)
    if c5_try < best_c5:
        best_c5 = c5_try
        best_h = h_try.copy()
        best_n = len(h_try)

    # Phase 5: Resolution sweep (15%)
    dl5 = t0 + budget_s * 0.75
    for dn in [-2, -1, 1, 2, -4, 4, -6, 6, -8, 8, -12, 12]:
        if time.time() > dl5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 160:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + budget_s * 0.018)
        h_r, c5_r = _main_loop(h_r, sub_dl, rho_init=0.08, use_pnorm=False)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Binary exploration + continuous polish (10%)
    dl6 = t0 + budget_s * 0.85
    for nb in [best_n, best_n - 2, best_n + 2, best_n - 4, best_n + 4]:
        if time.time() > dl6:
            break
        if nb < 32 or nb > 160 or nb % 2 != 0:
            continue
        h_bin = _binary_init(nb, seed + nb)
        sub_dl = min(dl6, time.time() + budget_s * 0.015)
        h_bin, c5_bin = _binary_hill_climb(h_bin, sub_dl, max_iters=5000)
        if c5_bin < best_c5:
            sub_dl2 = min(dl6, time.time() + budget_s * 0.015)
            h_cont, c5_cont = _main_loop(h_bin, sub_dl2, rho_init=0.05, use_pnorm=False)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
                best_n = len(h_cont)

    # Phase 7: Random perturbation restarts (5%)
    dl7 = t0 + budget_s * 0.90
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl7:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.01)
        h_try, c5_try = _main_loop(h_pert, sub_dl, rho_init=0.04, use_pnorm=False)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.06)

    # Phase 8: Ultra-fine LP polish (5%)
    dl8 = deadline_final
    h_try, c5_try = _main_loop(best_h, dl8, rho_init=0.03, use_pnorm=False)
    if c5_try < best_c5:
        best_c5 = c5_try
        best_h = h_try.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    # Ensure exact sum
    s = float(np.sum(best_h))
    target = best_n / 2.0
    if abs(s - target) > 1e-10:
        diff = target - s
        interior = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            best_h[interior] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = _project(best_h, target)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
