    Find t in [t_lo, t_hi] minimizing max_i (a[i]*t + b[i]).
    a: array of slopes
    b: array of intercepts
    Returns (t_opt, f_opt)
    """
    n = len(a)
    # Sort by slope
    order = np.argsort(a)
    a_sorted = a[order]
    b_sorted = b[order]
    
    # Build upper envelope
    # The upper envelope consists of segments, each from one line.
    # For lines sorted by slope, the envelope is built using a stack.
    
    # Intersection of line i and line j (with a[i] < a[j]):
    # a[i]*t + b[i] = a[j]*t + b[j]
    # t = (b[i] - b[j]) / (a[j] - a[i])
    
    # Stack of (slope, intercept, start_t)
    env = []  # list of [slope, intercept, t_start]
    
    for idx in range(n):
        s = a_sorted[idx]
        c = b_sorted[idx]
        if not np.isfinite(s) or not np.isfinite(c):

            c_prev = env[-1][1]

                    continue  # reprocess this line
                else:
                    continue  # this line is never on top

            if t_int >= env[-1][2]:

                env.append([s, c, t_int])

                env.pop()

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 5})
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search_c5(h, p):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    alphas = np.linspace(0, alpha_max, 20)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if 0 < best_i < len(alphas) - 1:
        lo = alphas[best_i - 1]
        hi = alphas[best_i + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(20):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        oa = (lo + hi) / 2.0
        oc5 = _exact_c5(h + oa * p)
        if oc5 < best_c5:
            best_a, best_c5 = oa, oc5
    return best_a, best_c5

def _lp_local(h, deadline, top_k_list=None, radius_init=0.15, max_iters=3000):
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores_arr(h)
        p, t_val = _lp_direction(h, scores, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search_c5(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 10 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _weighted_grad(h, deadline, n_steps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 5.0 + 200.0 * frac
        scores = _scores_arr(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        dx = 2.0 / n
        grad = np.zeros(n)
        d_full = np.arange(-(n-1), n)
        for r in range(len(scores)):
            d = d_full[r]
            if d >= 0:
                m2 = n - d
                if m2 > 0:
                    grad[:m2] += w[r] * (1.0 - h[d:]) * dx
                    grad[d:] += w[r] * (-h[:m2]) * dx
            else:
                g = -d
                m2 = n - g
                if m2 > 0:
                    grad[:m2] += w[r] * (1.0 - h[g:]) * dx
                    grad[g:] += w[r] * (-h[:m2]) * dx
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.1
        for _ in range(30):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h = _project(h, n / 2.0)
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Exact pair CD on initial (most promising)
    dl1 = t0 + budget_s * 0.20
    h1, c5_1 = _pair_cd_h(h_start, dl1, max_sweeps=3000, n_pairs_per_sweep=600, rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP polish
    if time.time() < t0 + budget_s * 0.25:
        dl2 = t0 + budget_s * 0.30
        h2, c5_2 = _lp_local(best_h, dl2, top_k_list=[None, 15, 8, 5], radius_init=0.12, max_iters=2000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: More pair CD after LP
    if time.time() < t0 + budget_s * 0.32:
        dl3 = t0 + budget_s * 0.42
        h3, c5_3 = _pair_cd_h(best_h, dl3, max_sweeps=4000, n_pairs_per_sweep=800, rng=rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-resolution search
    dl4 = t0 + budget_s * 0.62
    resolutions = [96, 64, 80, 112, 160, 128, 192, 80, 64]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl4:
            break
        if best_c5 < 0.3805:
            break
        n_starts = 2
        for s in range(n_starts):
            if time.time() > dl4:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_periodic_init(n_try, alpha)
                h_init = h_init + rng.normal(0, 0.06, n_try)
                h_init = _project(h_init, n_try / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            ht, ct = _pair_cd_h(h_init, sub_dl, max_sweeps=2000, n_pairs_per_sweep=400, rng=rng)
            if time.time() < dl4:
                sub_dl2 = min(dl4, time.time() + budget_s * 0.012)
                ht2, ct2 = _lp_local(ht, sub_dl2, top_k_list=[10, 5, None], radius_init=0.10, max_iters=800)
                if ct2 < ct:
                    ht, ct = ht2, ct2
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = n_try

    # Phase 5: Weighted gradient + pair CD on best
    if time.time() < t0 + budget_s * 0.68:
        dl5 = t0 + budget_s * 0.72
        h5, c5_5 = _weighted_grad(best_h, dl5, n_steps=100)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < dl5:
            sub_dl5 = min(dl5, time.time() + budget_s * 0.03)
            h5b, c5_5b = _pair_cd_h(best_h, sub_dl5, max_sweeps=1500, n_pairs_per_sweep=500, rng=rng)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: Perturbation restarts with pair CD
    dl6 = t0 + budget_s * 0.88
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        ht, ct = _pair_cd_h(h_pert, sub_dl, max_sweeps=800, n_pairs_per_sweep=300, rng=rng)
        if ct < best_c5 - 1e-12:
            best_c5 = ct
            best_h = ht.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 7: Final LP + pair CD polish
    if time.time() < t0 + budget_s * 0.92:
        dl7 = t0 + budget_s * 0.95
        h7, c5_7 = _lp_local(best_h, dl7, top_k_list=[5, 3, 4, None], radius_init=0.05, max_iters=500)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    if time.time() < t0 + budget_s * 0.97:
        sub_dl = t0 + budget_s * 0.99
        h7b, c5_7b = _pair_cd_h(best_h, sub_dl, max_sweeps=2000, n_pairs_per_sweep=1000, rng=rng)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
