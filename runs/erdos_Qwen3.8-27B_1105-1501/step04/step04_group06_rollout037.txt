**Idea: Use a trust-region method with the exact nonsmooth objective.**

The idea is:
1. At current point h, compute the subdifferential of the max function
2. Find the steepest descent direction in the subdifferential
3. Take a step in that direction
4. Use a trust-region to limit the step size

The steepest descent direction for a max of functions is: find the vector in the subdifferential (convex hull of gradients of active functions) that has the smallest norm, then move in the negative of that direction.

For the max of corr_r functions:
Subdifferential at h = conv({∇corr_r(h) : r ∈ A}) where A = {r : corr_r(h) = max}

The steepest descent direction is -d where d = argmin ||v||² subject to v ∈ conv({∇corr_r : r ∈ A}).

This can be solved as a QP:
minimize ||v||²
subject to: v = Σ α_r ∇corr_r, Σ α_r = 1, α_r ≥ 0

This is a small QP (only |A| variables). The solution gives the steepest descent direction.

Then we take a step: h_new = h - t * d, projected to feasible set.

Use a backtracking line search to find a good t.

This is essentially the "steepest descent method for minimax problems" (Mitsos et al., or Hiriart-Urruty et al.).

Let me implement this. It's actually quite elegant and might be more effective than SLSQP for this particular problem structure.

```python
def _steepest_descent_dir(h, n, n_active=10):
    """Find steepest descent direction for min max(corr_r)."""
    c = _corr(h)
    mx = c.max()
    # Active set: top n_active + margin
    idx = np.where(c > mx - 0.01)[0]
    if len(idx) > n_active:
        idx = np.argsort(-c)[:n_active]
    
    j = np.arange(n)
    # Build gradient matrix for active constraints
    G = np.zeros((len(idx), n))
    for row, r in enumerate(idx):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        G[row, :] = pd
    
    # Solve QP: min ||G^T alpha||^2 s.t. sum(alpha)=1, alpha >= 0
    # = min alpha^T G G^T alpha s.t. 1^T alpha = 1, alpha >= 0
    M = G @ G.T
    m = len(idx)
    
    # Use scipy to solve this small QP
    # Or use a simple projection: the optimal alpha is the dual of projecting 0 onto the convex hull
    # Actually, min ||G^T alpha||^2 = min ||Σ alpha_i g_i||^2
    # This is equivalent to: find the point in conv({g_i}) closest to 0
    # Which can be solved by: alpha_i = max(0, ... ) or using a QP solver
    
    # Simple approach: use scipy.optimize for the small QP
    from scipy.optimize import minimize as _min
    alpha0 = np.ones(m) / m
    bounds = [(0, 1)] * m
    
    def _qp_obj(alpha):
        v = G.T @ alpha
        return np.dot(v, v)
    
    def _qp_grad(alpha):
        v = G.T @ alpha
        return 2.0 * G @ v
    
    try:
        res = _min(_qp_obj, alpha0, jac=_qp_grad, bounds=bounds,
                  constraints=[{'type': 'eq', 'fun': lambda a: a.sum() - 1}],
                  method='SLSQP', options={'maxiter': 100, 'ftol': 1e-14})
        alpha = res.x
        d = G.T @ alpha
    except Exception:
        d = G.mean(axis=0)
    
    # d is the steepest descent direction (we want to move in -d direction)
    # But we need to handle the sum constraint: sum(h) = n/2
    # Project d to be orthogonal to 1 (so that moving in -d preserves the sum)
    d = d - np.mean(d) * np.ones(n)
    
    return -d  # descent direction

def _steepest_descent_step(h, n, max_iters=50):
```

def _project(v, target):

        return np.zeros(n)

    diff = target - x.sum()
    if abs(diff) > 1e-13:

                if x[i] >= 1.0 - 1e-15:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    nidx = len(idx)
    j = np.arange(n)
    J = np.zeros((nidx, n + 1))
    for row, r in enumerate(idx):

        ps = j + s
        m2 = (ps >= 0) & (ps < n)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _fourier_perturb(h, n, rng, k_max=6, amp=0.05):
    x = (np.arange(n) + 0.5) / n
    p = np.zeros(n)
    for k in range(1, k_max + 1):
        a = rng.uniform(-amp, amp)
        b = rng.uniform(-amp, amp)
        p += a * np.sin(2.0 * np.pi * k * x) + b * np.cos(2.0 * np.pi * k * x)
    p -= p.mean()
    return _project(h + p, n * 0.5)

def _swap_perturb(h, n, rng, n_swaps=4, amp=0.05):
    h = h.copy()
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        amt = rng.uniform(-amp, amp)
        h[i] += amt
        h[j] -= amt
    return _project(h, n * 0.5)

def _steepest_dir(h, n, K=10):
    c = _corr(h)
    mx = c.max()
    idx = np.where(c > mx - 0.01)[0]
    if len(idx) > K:
        idx = np.argsort(-c)[:K]
    j = np.arange(n)
    grads = np.zeros((len(idx), n))
    for row, r in enumerate(idx):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        grads[row, :] = pd
    G = grads
    M = G @ G.T
    m = len(idx)
    ones = np.ones(m)
    try:
        M_inv_ones = np.linalg.solve(M + 1e-10 * np.eye(m), ones)
        alpha = M_inv_ones / ones @ M_inv_ones
        alpha = np.maximum(alpha, 0.0)
        alpha = alpha / alpha.sum()
    except Exception:
        alpha = ones / m
    d = G.T @ alpha
    d = d - np.mean(d)
    return d
