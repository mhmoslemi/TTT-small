def _project(v, target):
    """Project v onto simplex-like set: sum(v)=target, 0<=v[i]<=1"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:

        else:

        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(h, r):
    """Gradient of score r w.r.t. h (raw, not multiplied by dx)"""
    n = len(h)
    g = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        for i in range(m):
            g[i] += (1.0 - h[i+d])
            g[i+d] -= h[i]

        s = -d
        m = n - s
        for i in range(m, n):
            g[i] += (1.0 - h[i-s])
            g[i-s] -= h[i]
    return g

def _lp_direction_top(h, scores_raw, n_active, radius):
    """LP: find direction p minimizing predicted max over top n_active scores"""
    n = len(h)
    m = 2 * n - 1

    top_idx = np.argsort(scores_raw)[::-1][:n_active]

    A_rows = []
    b_rows = []
    cur_max = float(np.max(scores_raw))
    delta = cur_max * 1e-6 + 1e-10
    
    for r in top_idx:
        g = _row_grad(h, r)
        A_rows.append(g)
        b_rows.append(-(scores_raw[r] - delta))
    
    if not A_rows:
        return None
    
    A = np.array(A_rows)  # shape (n_active, n)
    b = np.array(b_rows)
    
    # LP: minimize t s.t. A @ p <= b, sum(p) = 0, |p_j| <= radius
    A_ub = np.column_stack([A, -np.ones(len(A_rows))])
    b_ub = b.copy()
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

        return float(np.sum(excess * excess))

        g = g * mask
        return g

    best_c5 = cur_c5

            break

def _resample_area(h, n_new):
    """Area-preserving resampling"""
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    
    h_new = np.zeros(n_new)
    for j in range(n_new):
        # New cell [j*dx_new, (j+1)*dx_new]
        lo = j * dx_new
        hi = (j + 1) * dx_new
        total = 0.0
        for i in range(n):
            olo = i * dx_old
            ohi = (i + 1) * dx_old
            overlap = min(hi, ohi) - max(lo, olo)
            if overlap > 0:
                total += h[i] * overlap
        h_new[j] = total / dx_new
    return _project(h_new, n_new / 2.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(h, r):
    n = len(h)
    g = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        g[:m] += (1.0 - h[d:])
        g[d:] -= h[:m]
    else:
        s = -d
        m = n - s
        g[s:] += (1.0 - h[:m])
        g[:m] -= h[s:]
    return g

def _line_search(h, p, c5_cur, alpha_max):
    n = len(h)
    if alpha_max <= 1e-12:
        return 0.0, c5_cur
    best_t = 0.0
    best_val = c5_cur
    n_grid = 35
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 10.0)
    for i in range(n_grid):
        h_t = h + ts[i] * p
        if h_t.min() >= -1e-10 and h_t.max() <= 1.0 + 1e-10:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 2)]
    hi_t = ts[min(n_grid - 1, idx + 2)]
    if hi_t <= lo_t:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)

    def feval(t):
        h_t = h + t * p
        if h_t.min() < -1e-10 or h_t.max() > 1.0 + 1e-10:
            return c5_cur + 10.0
        return _exact_c5(h_t)

    f1, f2 = feval(c1), feval(c2)
    for _ in range(28):
        if hi_t - lo_t < 1e-13:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = feval(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = feval(c2)
    f_opt = feval((lo_t + hi_t) / 2.0)
    if f_opt < best_val:

    return best_t, best_val

def _lp_active(h, scores_raw, n_active, radius):
    n = len(h)
    m = 2 * n - 1
    if n_active >= m:
        n_active = m - 1
    top_idx = np.argsort(scores_raw)[::-1][:n_active]
    A_rows = np.zeros((len(top_idx), n))
    for k, r in enumerate(top_idx):
        A_rows[k] = _row_grad(h, r)
    cur_max = float(np.max(scores_raw))
    delta = max(cur_max * 1e-5, 1e-8)
    b = cur_max - delta - scores_raw[top_idx]
    A_ub = np.column_stack([A_rows, -np.ones(len(top_idx))])
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            p = res.x[:n]
            if np.isfinite(p).all() and np.max(np.abs(p)) > 1e-14:
                return p
    except Exception:
        pass
    return None

def _td_obj(x):

def _td_grad(x):

def _target_drive(h, theta_raw, deadline, max_solves=10):

    target = n / 2.0

    _g_state['target'] = target
    constraints = [{'type': 'eq', 'fun': _td_con_fun, 'jac': _td_con_jac}]

                best_h = h_new.copy()

def _grad_desc(h, deadline, tau_init=12.0, lr_init=0.012):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_c5 = _exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    beta = 0.88
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        scores_raw = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        shifted = tau * (scores_raw - np.max(scores_raw))
        w = np.exp(shifted)
        w /= np.sum(w)
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] > 1e-11:
                grad += w[r] * _row_grad(h_cur, r)
        mom = beta * mom + (1.0 - beta) * grad
        h_new = _project(h_cur - lr * mom, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h_cur = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
            lr = min(lr * 1.02, 0.08)
            tau = min(tau * 1.015, 80.0)
        else:
            stall += 1
            lr *= 0.94
            if stall > 50:
                break
            if stall % 15 == 0:

def _pair_descent(h, deadline, max_passes=200):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = _exact_c5(h_cur)
    rng = np.random.default_rng(12345)

        w = np.exp(shifted)
        w /= np.sum(w)
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] > 1e-10:
                grad += w[r] * _row_grad(h_cur, r)
        order = np.argsort(grad)
        n_low = max(5, n // 4)
        n_high = max(5, n // 4)
        improved = False
        n_try = min(50, n)
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = int(order[rng.integers(0, n_low)])
            j = int(order[rng.integers(n - n_high, n)])
            if i == j:
                continue
            if h_cur[i] < 0.02 or h_cur[i] > 0.98:
                continue
            if h_cur[j] < 0.02 or h_cur[j] > 0.98:
                continue
            for sign in [1.0, -1.0]:
                if sign > 0:
                    max_d = min(1.0 - h_cur[i], h_cur[j])

                    max_d = min(h_cur[i], 1.0 - h_cur[j])
                if max_d < 0.01:

                best_delta = 0.0
                best_val = best_c5
                for frac in [0.15, 0.3, 0.5, 0.7, 1.0]:
                    d = max_d * frac
                    h_t = h_cur.copy()
                    h_t[i] += sign * d
                    h_t[j] -= sign * d
                    if h_t[i] < -1e-10 or h_t[i] > 1 + 1e-10:

                    if h_t[j] < -1e-10 or h_t[j] > 1 + 1e-10:

                        best_delta = d
                if best_delta > 0 and best_val < best_c5 - 1e-11:

                    h_cur[j] -= sign * best_delta
                    best_c5 = best_val
                    best_h = h_cur.copy()
                    improved = True
                    break

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        lo = j * dx_new
        hi = (j + 1) * dx_new
        total = 0.0
        i_start = max(0, int(lo / dx_old) - 1)
        i_end = min(n, int(hi / dx_old) + 2)
        for i in range(i_start, i_end):
            olo = i * dx_old
            ohi = (i + 1) * dx_old
            overlap = min(hi, ohi) - max(lo, olo)
            if overlap > 0:
                total += h[i] * overlap
        h_new[j] = total / dx_new
    return _project(h_new, n_new / 2.0)

    try_ns = sorted(set(n_cur + d for d in [-4, -2, -1, 1, 2, 3, 5, 6, 8, 10, -5, -6, 12]
                         if 30 <= n_cur + d <= 180))
    time_per = max(1.0, (d6 - time.time()) / max(len(try_ns), 1))
    for n_try in try_ns:
        if time.time() > d6:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + time_per * 0.5)
        h_r, c5_r = _active_descent(h_r, sub_dl, max_iter=30)
        if time.time() < d6 and (d6 - time.time()) > 2.0:
            sub_dl2 = min(d6, time.time() + time_per * 0.3)
            h_r, c5_r = _grad_desc(h_r, sub_dl2, tau_init=8.0, lr_init=0.008)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Perturbation (10%)
    d7 = t0 + budget_s * 0.93
    best_h, best_c5 = _perturb_loop(best_h, d7, budget_s, seed + 200)

    # Phase 8: Final polish (7%)
    d8 = t0 + budget_s * 0.98
    h, c5 = _active_descent(best_h, d8, max_iter=80)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < t0 + budget_s * 0.995:
        h, c5 = _grad_desc(best_h, t0 + budget_s * 0.995, tau_init=40.0, lr_init=0.004)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
