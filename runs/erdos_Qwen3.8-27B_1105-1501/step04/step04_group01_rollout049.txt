            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_for_r(h, r):
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d+m])
            g[d:] += -h[:m]
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:gg+m])
            g[gg:] += -h[:m]
    return g

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 60
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _active_lp_direction(h, scores, top_k, radius):
    n = len(h)
    top_idx = np.argsort(scores)[-top_k:]
    top_scores = scores[top_idx]
    G_sub = np.zeros((top_k, n))
    for i, r in enumerate(top_idx):
        G_sub[i, :] = _grad_for_r(h, r)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -top_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _active_lp_loop(h, deadline, radius_init=0.15, top_k=8):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = _corr_scores(h)
        p = _active_lp_direction(h, scores, top_k, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5

def _smooth_surr_gradient(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = scores[r]
        if abs(w) < 1e-15:
            continue
        grad += w * _grad_for_r(h, r)
    return grad * 2.0

def _smooth_gradient_descent(h, deadline, lr_init=0.015, mom_decay=0.9):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        if stall > 100:
            break
        g = _smooth_surr_gradient(h)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        mom = mom_decay * mom + (1.0 - mom_decay) * g
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.0)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.8
            if stall % 30 == 0:
                mom = np.zeros(n)
    return best_h, best_c5

def _fw_step(h, g, target):
    idx = np.argsort(g)
    d = np.zeros(len(h))
    d[idx[:int(target)]] = 1.0
    return d

def _frank_wolfe(h, deadline, top_k=5):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = 2.0 / 3.0
    stall = 0
    for it in range(5000):
        if time.time() > deadline or stall > 200:
            break
        scores = _corr_scores(h)
        top_idx = np.argsort(scores)[-top_k:]
        g = np.zeros(n)
        for r in top_idx:
            g += _grad_for_r(h, r)
        g -= np.mean(g)
        d = _fw_step(h, g, target)
        step = gamma * (d - h)
        alpha, new_c5 = _line_search(h, step, dx, 1.0)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * step
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = max(gamma * 0.995, 0.1)
        else:
            stall += 1
            gamma = min(gamma * 1.05, 0.9)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _step_param_optimize(h, deadline, max_iters=2000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    stall = 0
    lr = 2.0 / n
    while time.time() < deadline and stall < 150:
        scores = _corr_scores(h)
        top_idx = np.argsort(scores)[-10:]
        g = np.zeros(n)
        for r in top_idx:
            g += _grad_for_r(h, r)
        g -= np.mean(g)
        h_new = h - lr * g
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.1)
            stall = 0
        else:
            stall += 1
            lr *= 0.75
            if stall % 40 == 0:
                h = best_h + rng.normal(0, 0.01, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                lr = 2.0 / n
    return best_h, best_c5

def _pair_refine_fast(h, deadline, max_passes=60):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i_idx in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[i_idx]
            if h[i] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            for j_idx in range(i_idx + 1, len(interior)):
                j = interior[j_idx]
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _semidefinite_polish(h, deadline, top_k=6):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.06
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = _corr_scores(h)
        p = _active_lp_direction(h, scores, top_k, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.65
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.2)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Frank-Wolfe warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _frank_wolfe(h, d1, top_k=6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Active LP main (30%)
    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _active_lp_loop(best_h, d2, radius_init=0.18, top_k=8)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smooth surrogate gradient (12%)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _smooth_gradient_descent(best_h, d3, lr_init=0.02, mom_decay=0.88)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Re-apply active LP after smooth phase (12%)
    d4 = t0 + budget_s * 0.62
    h4, c5_4 = _active_lp_loop(best_h, d4, radius_init=0.12, top_k=10)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (18%)
    d5 = t0 + budget_s * 0.80
    best_n = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, -3, 3, -8, 8, -5, 5, -10, 10, -7, 7, -12, 12, -15, 15, -20, 20, -25, 25, -30, 30, 35, -35]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 250:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _active_lp_loop(h_r, sub_dl, radius_init=0.12, top_k=8)
        rem = d5 - time.time()
        if rem > 3 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d5, time.time() + max(2, budget_s * 0.006))
            h_r, c5_r = _semidefinite_polish(h_r, sub_dl2, top_k=6)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Perturbation restarts (10%)
    d6 = t0 + budget_s * 0.90
    scale = 0.04
    for attempt in range(200):
        if time.time() > d6:
            break
        if best_c5 < 0.3803:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.006)
        h_t, c5_t = _active_lp_loop(h_pert, sub_dl, radius_init=0.1, top_k=8)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.15, scale * 1.15)

    # Phase 7: Pair refinement (8%)
    d7 = t0 + budget_s * 0.98
    h_p, c5_p = _pair_refine_fast(best_h, d7, max_passes=50)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Final: tiny LP polish
    budget_left = budget_s - (time.time() - t0) - 0.5
    if budget_left > 1:
        h_f, c5_f = _semidefinite_polish(best_h, time.time() + budget_left, top_k=10)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
