    t = params[n]
    corr = np.correlate(h, 1.0 - h, mode='full')

Actually, let me try yet another angle. What if I formulate this as a **second-order cone program (SOCP)** relaxation?

The constraint t ≥ corr_s(h) can be written as:

The bilinear term h[i]*h[i+s] can be relaxed using the McCormick envelope if we know bounds on h[i] and h[i+s] (which we do: [0,1]). The McCormick relaxation gives:

h[i]*h[i+s] ≤ h[i+s]
h[i]*h[i+s] ≤ h[i]

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)

                diff += a
