import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def two_level_h(mask, d):
    return np.where(mask.astype(float), 1.0 - d, d)

def find_optimal_d(mask, n_grid=100):
    n = len(mask)
    d_vals = np.linspace(0.0, 0.5, n_grid)
    c5_vals = np.array([exact_c5(two_level_h(mask, d)) for d in d_vals])
    idx = np.argmin(c5_vals)
    if idx == 0:
        return 0.0
    if idx == n_grid - 1:
        return 0.5
    d_lo = d_vals[idx-1]
    d_hi = d_vals[idx+1]
    # Golden section
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    a, b = d_lo, d_hi
    c = b - (b-a)/phi
    g = a + (b-a)/phi
    fc = exact_c5(two_level_h(mask, c))
    fg = exact_c5(two_level_h(mask, g))
    for _ in range(50):
        if fc < fg:
            b = g
            g = c
            fg = fc
            c = b - (b-a)/phi
            fc = exact_c5(two_level_h(mask, c))
        else:
            a = c
            c = g
            fc = fg
            g = a + (b-a)/phi
            fg = exact_c5(two_level_h(mask, g))
    return (a+b)/2.0

def make_mask(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(-h)
    mask = np.zeros(n, dtype=int)
    mask[idx[:m]] = 1
    return mask

def pair_line_search(h, deadline, max_sweeps=50):
    """Pair line search for continuous h."""
    n = len(h)
    target = n / 2.0
    best_c5 = exact_c5(h)
    best_h = h.copy()
    
    for _sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Shuffle pair order for diversity
        pairs = []
        indices = np.arange(n)

        for i in range(n):
            for j in range(i+1, n):
                pairs.append((i, j))
        rng_local.shuffle(pairs)
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            p = best_h[i] + best_h[j]
            lo = max(0.0, p - 1.0)
            hi = min(1.0, p)
            if hi - lo < 1e-10:
                continue
            
            # Golden section search for optimal t
            phi = (1.0 + np.sqrt(5.0)) / 2.0
            a, b = lo, hi
            c = b - (b-a)/phi
            g = a + (b-a)/phi
            
            h_c = best_h.copy()
            h_c[i] = c
            h_c[j] = p - c
            fc = exact_c5(h_c)
            
            h_g = best_h.copy()
            h_g[i] = g
            h_g[j] = p - g
            fg = exact_c5(h_g)
            
            for _ in range(40):

                    b = g
                    g = c
                    fg = fc
                    c = b - (b-a)/phi
                    h_c = best_h.copy()
                    h_c[i] = c
                    h_c[j] = p - c
                    fc = exact_c5(h_c)
                else:
                    a = c
                    c = g
                    fc = fg
                    g = a + (b-a)/phi
                    h_g = best_h.copy()
                    h_g[i] = g
                    h_g[j] = p - g
                    fg = exact_c5(h_g)
            
            t_opt = (a+b)/2
            h_new = best_h.copy()
            h_new[i] = t_opt
            h_new[j] = p - t_opt
            h_new = project_to_sum(h_new, target)
            c5_new = exact_c5(h_new)
            
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def mask_descent(mask, deadline, refine_d=True):
    """Exhaustive swap descent on binary mask with d optimization."""
    n = len(mask)
    mask = mask.copy()
    
    d_opt = find_optimal_d(mask) if refine_d else 0.0
    h = two_level_h(mask, d_opt)
    best_c5 = exact_c5(h)
    
    ones = np.where(mask == 1)[0]
    zeros = np.where(mask == 0)[0]

            break
        best_swap_c5 = best_c5
        best_swap = None
        best_swap_d = d_opt

                break

                mask[j] = 1
                d_try = find_optimal_d(mask, n_grid=50)
                h_try = two_level_h(mask, d_try)
                c5_try = exact_c5(h_try)

                    best_swap_c5 = c5_try
                    best_swap = (i, j)
                    best_swap_d = d_try
                mask[i] = 1
                mask[j] = 0

            mask[i] = 0
            mask[j] = 1
            ones = np.where(mask == 1)[0]
            zeros = np.where(mask == 0)[0]
            d_opt = best_swap_d
            h = two_level_h(mask, d_opt)
            best_c5 = best_swap_c5
        else:
            break
    
    h_final = two_level_h(mask, d_opt)
    c5_final = exact_c5(h_final)
    return h_final, c5_final, mask, d_opt
