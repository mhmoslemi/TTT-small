    h[k] = old_k - t
    val = _exact_c5(h)
    h[j] = old_j
    h[k] = old_k
    return val

def _pair_golden(h, j, k, t_lo, t_hi, c_cur):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    if b - a < 1e-13:
        return 0.0, c_cur
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _c5_pair(h, j, k, c)
    fd = _c5_pair(h, j, k, d)
    for _ in range(45):
        if b - a < 1e-13:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = _c5_pair(h, j, k, c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = _c5_pair(h, j, k, d)
    t_opt = (a + b) / 2.0
    f_opt = _c5_pair(h, j, k, t_opt)
    best_t, best_f = t_opt, f_opt
    if fc < best_f:
        best_t, best_f = c, fc
    if fd < best_f:
        best_t, best_f = d, fd
    if best_f < c_cur:
        return best_t, best_f
    return 0.0, c_cur

def _pair_descent(h, deadline, max_sweeps=500):
    n = len(h)
    h = h.copy()
    c_cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c_cur
    rng = np.random.default_rng(12345)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 3:
            break
        # Random permutation of pairs
        idx = np.arange(n)
        rng.shuffle(idx)
        pair_list = []
        for ii in range(n):
            for jj in range(ii + 1, n):
                pair_list.append((idx[ii], idx[jj]))
        rng.shuffle(pair_list)
        improved_this_sweep = False
        for (j, k) in pair_list:
            if time.time() > deadline:
                break
            if h[j] < 1e-12 or h[j] > 1.0 - 1e-12 or h[k] < 1e-12 or h[k] > 1.0 - 1e-12:
                continue
            t_lo = max(-h[j] + 1e-14, h[k] - 1.0 + 1e-14)
            t_hi = min(1.0 - h[j] - 1e-14, h[k] - 1e-14)
            if t_hi - t_lo < 1e-11:
                continue
            t_star, c_new = _pair_golden(h, j, k, t_lo, t_hi, c_cur)
            if t_star != 0.0 and c_new < c_cur - 1e-13:
                h[j] += t_star
                h[k] -= t_star
                c_cur = c_new
                improved_this_sweep = True
                if c_cur < best_c5:
                    best_c5 = c_cur
                    best_h = h.copy()
        if improved_this_sweep:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _bw_descent(h, deadline, window=2, max_sweeps=300):
    """Block coordinate descent on adjacent pairs (banded)."""
    n = len(h)
    h = h.copy()
    c_cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c_cur
    rng = np.random.default_rng(999)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 5:
            break
        offset = sweep % window
        improved = False
        for start in range(offset, n - 1, window):
            if time.time() > deadline:
                break
            j, k = start, start + 1
            if h[j] < 1e-12 or h[j] > 1.0 - 1e-12 or h[k] < 1e-12 or h[k] > 1.0 - 1e-12:
                continue
            t_lo = max(-h[j] + 1e-14, h[k] - 1.0 + 1e-14)
            t_hi = min(1.0 - h[j] - 1e-14, h[k] - 1e-14)
            if t_hi - t_lo < 1e-11:
                continue
            t_star, c_new = _pair_golden(h, j, k, t_lo, t_hi, c_cur)
            if t_star != 0.0 and c_new < c_cur - 1e-13:
                h[j] += t_star
                h[k] -= t_star
                c_cur = c_new
                improved = True
                if c_cur < best_c5:
                    best_c5 = c_cur
                    best_h = h.copy()
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _grad_optimize(h, deadline, tau_init=15.0, lr_init=0.02, mom_beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            wr = w[r]
            if wr < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                grad[d:] += wr
                grad[d:] -= wr * h[:m]
                grad[:m] -= wr * h[d:]
            else:
                g = -d
                m = n - g
                grad[:m] += wr
                grad[:m] -= wr * h[g:]
                grad[g:] -= wr * h[:m]
        grad *= dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.12)
            tau = min(tau * 1.01, 150.0)
        else:
            stall += 1
            lr *= 0.93
            if stall > 60:
                break
            if stall % 25 == 0:
                tau = max(tau_init, tau * 0.85)
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, d:] += 1.0
            G[r, d:] -= h[:m]
            G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += 1.0
            G[r, :m] -= h[g:]
            G[r, g:] -= h[:m]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_local(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        # Line search
        best_t, best_val = 0.0, cur_c5
        ts = np.linspace(0.0, alpha_max, 20)
        for t in ts:
            h_t = h + t * p
            if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
                v = _exact_c5(h_t)
                if v < best_val:
                    best_val, best_t = v, t
        if best_val < cur_c5 - 1e-13:
            h = h + best_t * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.65
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_refine(h_best, deadline, budget_s, seed=42):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = 0.03
    time_per = max(1.0, budget_s * 0.015)
    attempt = 0
    while time.time() < deadline and attempt < 150:
        attempt += 1
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + time_per)
        h_try, c5_try = _pair_descent(h_pert, sub_dl, max_sweeps=30)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.2, scale * 1.15)
    return best_h, best_c5

def _weighted_avg_pool(candidates, n_target, seed=0):
    """Blend top candidates pairwise."""
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    for i in range(len(candidates)):
        for j in range(i + 1, len(candidates)):
            for alpha in [0.3, 0.5, 0.7]:
                h_blend = alpha * candidates[i][0] + (1 - alpha) * candidates[j][0]
                h_blend = _project(h_blend, n_target / 2.0)
                c = _exact_c5(h_blend)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_blend.copy()
    if best_h is not None:
        return best_h, best_c5
    return candidates[0]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_times if 'initial_h_values' not in dir() else initial_h_values, dtype=float)
    if 'initial_h_values' not in dir():
        n0 = 90
        h0 = np.full(n0, 0.5)
        h0[:n0//2] = 1.0
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick gradient warmup (4% budget)
    d1 = t0 + budget_s * 0.04
    h, c5 = _grad_optimize(h, d1, tau_init=12.0, lr_init=0.015, mom_beta=0.88)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Banded pair descent (quick structural fix, 6% budget)
    d2 = t0 + budget_s * 0.10
    h, c5 = _bw_descent(best_h, d2, window=1, max_sweeps=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Full pair descent (35% budget) - main workhorse
    d3 = t0 + budget_s * 0.45
    h, c5 = _pair_descent(best_h, d3, max_sweeps=500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: LP local search (12% budget)
    d4 = t0 + budget_s * 0.57
    h, c5 = _lp_local(best_h, d4, radius_init=0.12)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Another round of pair descent after LP (12% budget)
    d5 = t0 + budget_s * 0.69
    h, c5 = _pair_descent(best_h, d5, max_sweeps=300)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Multi-resolution (12% budget)
    d6 = t0 + budget_s * 0.81
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10]
    time_per = max(3.0, budget_s * 0.012)
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + time_per)
        h_r, c5_r = _pair_descent(h_r, sub_dl, max_sweeps=100)
        if time.time() < d6 and (d6 - time.time()) > 2.0:
            sub_dl2 = min(d6, time.time() + time_per * 0.5)
            h_r, c5_r = _bw_descent(h_r, sub_dl2, window=2, max_sweeps=50)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Perturbation + re-optimize (12% budget)
    d7 = t0 + budget_s * 0.93
    best_h, best_c5 = _perturb_refine(best_h, d7, budget_s, seed=seed + 200)

    # Phase 8: Final polish (7% budget)
    d8 = t0 + budget_s * 0.97
    h, c5 = _pair_descent(best_h, d8, max_sweeps=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Final LP micro-polish
    if time.time() < t0 + budget_s * 0.995:
        h, c5 = _lp_local(best_h, t0 + budget_s * 0.995, radius_init=0.03)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
