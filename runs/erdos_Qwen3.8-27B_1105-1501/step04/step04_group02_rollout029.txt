            g = -d
            m = n - g
            G[r, :m] += 1.0
            G[r, :m] -= h[g:]
            G[r, g:] -= h[:m]
    try:
        from scipy.optimize import linprog
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
        bounds.append((None, None))
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max):
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, 24)
    vals = np.full(24, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 2)]
    hi_t = ts[min(23, idx + 2)]
    if hi_t <= lo_t:
        lo_t, hi_t = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    f1, f2 = _f(c1), _f(c2)
    for _ in range(30):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_local(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline - 1.0 and stall < 60:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-8:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.6
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12.0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Alternative starting points
    starts = [h0.copy()]
    starts.append(h0[::-1].copy())
    for q in [0.3, 0.6, 1.5]:
        starts.append(_project(h0 - q * (h0 - 0.5), n0 / 2.0))
    # Binarized
    idx = np.argsort(h0)[::-1]
    h_bin = np.zeros(n0)
    h_bin[idx[:n0 // 2]] = 1.0
    starts.append(h_bin)
    # Smoothed
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    starts.append(_project(np.convolve(h0, np.flip(kernel), mode='same'), n0 / 2.0))

    # Phase 1: SLSQP at parent resolution (20% budget)
    d1 = t0 + budget_s * 0.20
    for s_i, h_s in enumerate(starts[:4]):
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.045)
        h_try, c5_try = _slsqp_iterate(h_s, sub_dl, max_rounds=3)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)

    # Phase 2: LP local search at best (10% budget)
    d2 = t0 + budget_s * 0.30
    h_try, c5_try = _lp_local(best_h, d2, radius_init=0.12)
    if c5_try < best_c5 - 1e-12:
        best_c5 = c5_try
        best_h = h_try.copy()

    # Phase 3: Multi-resolution SLSQP (25% budget)
    d3 = t0 + budget_s * 0.55
    offsets = [-4, 4, -6, 6, -8, 8, -2, 2, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 44 or n_try > 180:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.04)
        h_r2, c5_r2 = _slsqp_iterate(h_r, sub_dl, max_rounds=4)
        if c5_r2 < best_c5 - 1e-12:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = n_try

    # Phase 4: Subgradient + SLSQP at best (10% budget)
    d4 = t0 + budget_s * 0.65
    h_try, c5_try = _subgrad_descent(best_h, d4, lr_init=0.008, decay=0.985, momentum=0.88)
    if c5_try < best_c5 - 1e-12:
        best_c5 = c5_try
        best_h = h_try.copy()
    if time.time() < d4:
        h_try, c5_try = _slsqp_iterate(best_h, d4, max_rounds=3)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: LP + SLSQP polish (10% budget)
    d5 = t0 + budget_s * 0.75
    h_try, c5_try = _lp_local(best_h, d5, radius_init=0.08)
    if c5_try < best_c5 - 1e-12:
        best_c5 = c5_try
        best_h = h_try.copy()
    if time.time() < d5:
        h_try, c5_try = _slsqp_iterate(best_h, d5, max_rounds=3)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 6: Pair refinement (15% budget)
    d6 = t0 + budget_s * 0.90
    best_h, best_c5 = _pair_refine(best_h, d6, max_passes=300)

    # Phase 7: Final SLSQP + LP + pair (10% budget)
    d7 = deadline
    if time.time() < d7 - 10:
        h_try, c5_try = _slsqp_iterate(best_h, min(d7, time.time() + budget_s * 0.04), max_rounds=3)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    if time.time() < d7 - 5:
        h_try, c5_try = _lp_local(best_h, min(d7, time.time() + budget_s * 0.02), radius_init=0.05)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    best_h, best_c5 = _pair_refine(best_h, d7, max_passes=200)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
