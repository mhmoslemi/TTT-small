import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    v = np.clip(v.copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target - float(np.sum(v))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        v[mask] = np.clip(v[mask] + diff / cnt, 0.0, 1.0)
    return v

def _grad_smooth(h, beta, top_k=48):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = float(np.max(corr))
    scaled = np.clip(beta * (corr - max_c), -50, 50)
    exp_v = np.exp(scaled)
    weights = exp_v / (float(np.sum(exp_v)) + 1e-300)
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(weights)[-k:]
    grad = np.zeros(n)
    for r in top_idx:
        s = int(r) - (n - 1)
        d = abs(s)
        w = weights[r] * dx
        j = np.arange(0, n - d)
        grad[j + d] += w * (1.0 - h[j])
        grad[j] -= w * h[j + d]
    grad -= np.mean(grad)
    return grad, max_c
