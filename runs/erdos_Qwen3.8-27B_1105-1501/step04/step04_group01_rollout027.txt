def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):

    t_hi = min(t_hi, trust_radius)

    c1 = b - gr * (b - a)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction_k(h, scores, G, radius, k):
    n = len(h)
    num_rows = 2 * n - 1
    if k >= num_rows:
        k = num_rows
    top_idx = np.argsort(scores)[::-1][:k]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G[top_idx, :]
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_refine_adaptive(h, deadline, radius_init=0.15, k_start=6, k_max=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    k = k_start
    stall = 0
    total_stall = 0
    while time.time() < deadline:
        if total_stall > 120:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction_k(h, scores, G, radius, k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            total_stall += 1
            radius *= 0.7
            k = min(k + 4, k_max)
            if radius < 1e-5:
                if total_stall % 20 == 0:
                    radius = 0.12
                    k = k_start
                else:
                    break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            total_stall = 0
            radius = min(radius * 1.06, 0.40)
            k = max(k_start, k - 1)
        else:
            stall += 1
            total_stall += 1
            radius *= 0.72
            k = min(k + 5, k_max)
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.08
                k = k_start
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _combined_refine(h, deadline):
    """Alternate between LSA and LP methods."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    use_ls = True
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        if use_ls:
            d = _ls_exp_direction(h, 200, dx)
        else:
            scores = np.correlate(h, 1.0 - h, mode='full')
            G = _grad_matrix(h)
            p = _lp_direction_k(h, scores, G, 0.1, 12)
            d = p if p is not None else _ls_exp_direction(h, 200, dx)
        if d is None:
            use_ls = not use_ls
            continue
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            use_ls = not use_ls
            continue
        d = d / dn
        lr = 0.1
        alpha, c5_try = _line_search(h, lr * d, dx, lr)
        if c5_try < best_c5 - 1e-13:
            h = h + alpha * lr * d
            h = _project(h, target)
            c5_new = _exact_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
        use_ls = not use_ls
    return best_h, best_c5

def _shake_optimize(h, deadline, rng, n_trials=30):
    """Try random perturbations and optimize each briefly."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.04
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        c5_pert = _exact_c5(h_pert)
        sub_dl = min(deadline, time.time() + max(2, (deadline - time.time()) / max(1, n_trials - _)))
        if c5_pert < best_c5 + 0.01:
            h_t, c5_t = _lp_refine_adaptive(h_pert, sub_dl, radius_init=0.08, k_start=5, k_max=40)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.01, scale * 0.8)
            else:
                scale = min(0.15, scale * 1.1)
        else:
            scale = min(0.15, scale * 1.15)
    return best_h, best_c5

def _spectral_init(n, rng):
    """Generate a spectral-based initial candidate."""
    h = np.zeros(n)
    n_freqs = rng.integers(2, 6)
    for _ in range(n_freqs):
        freq = rng.uniform(1.0, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        h += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    h = _project(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Log-sum-exp refinement (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _ls_exp_refine(best_h, d1, gamma_start=20, gamma_end=6000, n_gammas=8, lr_init=0.10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Adaptive LP refinement (20%)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _lp_refine_adaptive(best_h, d2, radius_init=0.15, k_start=5, k_max=100)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution search (25%)
    d3 = t0 + budget_s * 0.65
    tried_ns = set()
    best_n = len(best_h)
    offsets = list(range(-8, 9)) + list(range(-16, 17, 2)) + [-20, 20, -24, 24, -12, 12]
    rng.shuffle(offsets)
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _ls_exp_refine(h_r, sub_dl, gamma_start=30, gamma_end=3000, n_gammas=5, lr_init=0.08)
        rem = d3 - time.time()
        if rem > 3 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d3, time.time() + max(2, budget_s * 0.008))
            h_r, c5_r = _lp_refine_adaptive(h_r, sub_dl2, radius_init=0.10, k_start=5, k_max=60)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            if best_c5 < 0.3806:
                break

    # Phase 4: Shake and optimize (15%)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = _shake_optimize(best_h, d4, rng, n_trials=40)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Combined refinement (10%)
    d5 = t0 + budget_s * 0.92
    h5, c5_5 = _combined_refine(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair refinement (5%)
    d6 = t0 + budget_s * 0.97
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=80)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.all(np.isfinite(best_h)) or np.min(best_h) < -1e-9 or np.max(best_h) > 1 + 1e-9:
        best_h = np.array(initial_h_values, dtype=float)
        best_h = _project(best_h, len(best_h) / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
