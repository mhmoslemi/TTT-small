def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(100):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:

        v[mask] += diff / cnt

def _compute_M(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _shift_direction(h, d):
    """Compute ∂M_d/∂h_k / dx for all k."""
    n = len(h)
    g = np.zeros(n)
    # For shift d: M_d = sum_{valid m} h[m] * (1-h[m+d])
    # ∂M_d/∂h[k] = (1-h[k+d]) if k+d valid - h[k-d] if k-d valid
    # k+d valid: 0 <= k < n and 0 <= k+d < n → -d <= k < n-d
    # k-d valid: 0 <= k-d < n and 0 <= k < n → d <= k < n+d
    lo1 = max(0, -d)
    hi1 = min(n, n - d)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
    lo2 = max(0, d)
    hi2 = min(n, n + d)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 - d:hi2 - d]
    return g
