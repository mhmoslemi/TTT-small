    # Actually: we want scores[r] + G[r] @ delta <= 0 for all selected r (to reduce max)
    # But we want to minimize the max, so: G[r] @ delta <= -scores[idx[r]] + eps for some small eps
    # Simpler: minimize t s.t. G[r] @ delta - t <= -scores[idx[r]], sum(delta)=0, bounds
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_round(h, n):
    """Round to binary with exactly n/2 ones."""
    target = n / 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:int(target)]] = 1.0
    return b

def _binary_local_search(b, max_iter=200, deadline=None):
    """Swap-based local search for binary h."""
    n = len(b)
    dx = 2.0 / n
    b = b.copy()
    cur = _exact_c5(b)
    best = cur
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Find improving swap: move a 1 to a 0 position
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        best_delta = 0
        best_swap = None
        S = _overlap(b)
        max_S = S.max()
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Try: b[i] = 0, b[j] = 1
                b[i] = 0.0
                b[j] = 1.0
                new_S = _overlap(b)
                new_max = float(new_S.max())
                b[i] = 1.0
                b[j] = 0.0
                if new_max < max_S - 1e-15:
                    delta = max_S - new_max
                    if delta > best_delta:
                        best_delta = delta
                        best_swap = (i, j)
        if best_swap is not None:
            b[best_swap[0]] = 0.0
            b[best_swap[1]] = 1.0
            cur = float(_overlap(b).max())
            if cur < best:
                best = cur
        else:
            break
    return b, best * dx

def _fract_from_binary(b, h_continuous, n):

    # Use the binary as a guide: positions that are 1 in binary get boosted
    alpha = 0.5
    h = alpha * b + (1 - alpha) * h_continuous
    return _project(h, n / 2.0)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

            lo = mid
        else:
            hi = mid

        if gn < 1e-14:
            break
        g = g / gn
        v = momentum * v - lr * g
        h_new = h + v
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, 0.15)
        else:
            stall += 1
            lr *= 0.85
            v = momentum * v
            if stall % 30 == 0:
                v = np.zeros(n)
    return best_h, best_c5

def _structured_direction(h, dx, k_modes=5):
    n = len(h)
    scores = _overlap(h) * dx
    max_s = float(np.max(scores))
    active_mask = scores > max_s - 0.05
    g = np.zeros(n)
    n_act = max(1, int(np.sum(active_mask)))
    for r in range(len(scores)):
        if not active_mask[r]:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d + m]) / n_act
                g[d:] += (-h[:m]) / n_act
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += (-h[gg:gg + m]) / n_act
                g[gg:] += (1.0 - h[:m]) / n_act
    g -= np.mean(g)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return np.zeros(n)
    return g / gn

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick LP polish (80s)
    d1 = t0 + 80
    h1, c5_1 = _lp_loop(h, d1, radius_init=0.1, top_k=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Nesterov subgradient (40s)
    d2 = t0 + 120
    h2, c5_2 = _nesterov_subgrad(best_h, d2, lr_init=0.008, momentum=0.75)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution search (500s)
    d3 = t0 + 620
    resolutions = []
    base_n = len(best_h)
    for dn in range(-32, 33, 4):
        n_try = base_n + dn
        if n_try >= 28 and n_try <= 240 and n_try % 2 == 0:
            resolutions.append(n_try)
    resolutions = sorted(set(resolutions), key=lambda x: abs(x - base_n))

    remaining = d3 - time.time()
    per_res = max(4.0, remaining / max(1, len(resolutions)))

    for n_try in resolutions:
        if time.time() > d3:
            break
        per_res_now = min(per_res, d3 - time.time())
        if per_res_now < 3:
            break
        h_r = _resample(best_h, n_try)
        sub_dl1 = time.time() + per_res_now * 0.35
        h_r, c5_r = _lp_loop(h_r, sub_dl1, radius_init=0.08, top_k=30)

        # Binary round + local search
        b = _binary_round(h_r, n_try)
        sub_dl2 = time.time() + per_res_now * 0.35
        b, c5_b = _binary_local_search(b, max_iter=200, deadline=sub_dl2, seed=n_try)
        if c5_b < c5_r:
            h_blend = 0.4 * b + 0.6 * h_r
            h_blend = _project(h_blend, n_try / 2.0)
            sub_dl3 = time.time() + per_res_now * 0.25
            h_r, c5_r = _lp_loop(h_blend, sub_dl3, radius_init=0.06, top_k=30)

        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Structured perturbation + re-optimize (200s)
    d4 = t0 + 820
    scales = [0.02, 0.04, 0.06, 0.1, 0.15]
    si = 0
    streak = 0
    while time.time() < d4:
        if best_c5 < 0.3806:
            break
        scale = scales[si % len(scales)]
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        # Add a structured component
        if streak % 3 == 1:
            mode = rng.integers(1, 6)
            h_pert += 0.02 * np.sin(np.pi * mode * np.arange(len(best_h)) / len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, top_k=30)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            streak = 0
            si = max(0, si - 1)
        else:
            streak += 1
            si += 1

    # Phase 5: Targeted pair moves (100s)
    d5 = t0 + 920
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = best_h.copy()
    cur_c5 = _exact_c5(h_cur)
    for _pass in range(100):
        if time.time() > d5:
            break
        improved = False
        ones_idx = np.where(h_cur > 0.6)[0]
        zeros_idx = np.where(h_cur < 0.4)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        best_move = None
        best_move_val = cur_c5
        n_try = min(500, len(ones_idx) * len(zeros_idx))
        for _ in range(n_try):
            i = ones_idx[rng.integers(len(ones_idx))]
            j = zeros_idx[rng.integers(len(zeros_idx))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h_cur, d, dx, 0.3)
            if c5_try < best_move_val - 1e-14:
                best_move_val = c5_try
                best_move = (i, j, d, alpha)
        if best_move is not None:
            h_cur = h_cur + best_move[2] * best_move[3]
            h_cur = _project(h_cur, target)
            cur_c5 = _exact_c5(h_cur)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
            improved = True
        if not improved:
            break

    # Phase 6: Final LP polish (rest of budget)
    d6 = t0 + budget_s - 3
    h_f, c5_f = _lp_loop(best_h, d6, radius_init=0.03, top_k=35)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
