    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10) * 0.999
    best_a = 0.0
    best_c5 = c5_cur
    # Coarse scan
    n_scan = 15
    for i in range(1, n_scan + 1):
        a = alpha_max * i / n_scan
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(25):
        if hi_a - lo_a < 1e-15:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lse_obj_and_grad(x, tau, target):
    """Log-sum-exp objective and gradient."""
    n = len(x)
    scores = corr_scores(x)
    max_s = np.max(scores)
    exp_s = np.exp((scores - max_s) / tau)
    Z = np.sum(exp_s)
    val = max_s + tau * np.log(Z)
    w = exp_s / Z
    # Gradient
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-15:
            g += w[k] * corr_grad(x, k)
    return val, g

def lbfgs_smooth(h0, tau, maxiter, target):

        v, _ = lse_obj_and_grad(x, tau, target)
        return v
    def grad(x):
        _, g = lse_obj_and_grad(x, tau, target)
        return g

                            maxiter=maxiter, maxfun=50000,
                            iprint=0, bounds=bounds)
        x_opt, f_opt, i = res
        # Handle the equality constraint by projecting
        x_opt = project(x_opt, target)
        return x_opt
    except Exception:
        return h0

                        for frac in [0.1, 0.15, 0.5, 0.8]:
                            t = amax * frac
                            h_t = best_h.copy()
                            h_t[i] += sign * t
                            h_t[j] -= sign * t
                            c5_t = exact_c5(h_t)
                            if c5_t < best_c5 - 1e-14:
                                found_better = True
                                break
                    if found_better:
                        # Golden section to find optimal t
                        lo_t, hi_t = 0.0, amax
                        gr = (np.sqrt(5.0) - 1.0) / 2.0
                        t1 = hi_t - gr * (hi_t - lo_t)
                        t2 = lo_t + gr * (hi_t - lo_t)
                        h1 = best_h.copy(); h1[i] += sign * t1; h1[j] -= sign * t1
                        f1 = exact_c5(h1)
                        h2 = best_h.copy(); h2[i] += sign * t2; h2[j] -= sign * t2
                        f2 = exact_c5(h2)
                        for _ in range(35):
                            if hi_t - lo_t < 1e-15:
                                break
                            if f1 < f2:
                                hi_t, t2, f2 = t2, t1, f1
                                t1 = hi_t - gr * (hi_t - lo_t)
                                h1 = best_h.copy(); h1[i] += sign * t1; h1[j] -= sign * t1
                                f1 = exact_c5(h1)
                            else:
                                lo_t, t1, f1 = t1, t2, f2
                                t2 = lo_t + gr * (hi_t - lo_t)
                                h2 = best_h.copy(); h2[i] += sign * t2; h2[j] -= sign * t2
                                f2 = exact_c5(h2)
                        opt_t = (lo_t + hi_t) / 2.0
                        h_opt = best_h.copy()
                        h_opt[i] += sign * opt_t
                        h_opt[j] -= sign * opt_t
                        f_opt = exact_c5(h_opt)
                        if f_opt < best_c5 - 1e-14:
                            best_c5 = f_opt
                            best_h = h_opt
                            improved = True
        if not improved:
            break
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.955

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: LP local search (conservative radius)
    d1 = t0 + budget_s * 0.10
    h_lp, c5_lp = lp_loop(h, d1, radius_init=0.08)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: L-BFGS-B with log-sum-exp (multiple tau)
    d2 = t0 + budget_s * 0.30
    tau_list = [0.6, 0.35, 0.2, 0.1, 0.05, 0.025, 0.012]
    h_lbg = best_h.copy()
    for tau in tau_list:
        if time.time() > d2:
            break
        h_lbg = lbfgs_smooth(h_lbg, tau, 300, best_n / 2.0)
        h_lbg = project(h_lbg, best_n / 2.0)
        c5_s = exact_c5(h_lbg)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_lbg.copy()

    # Phase 3: LP refinement after L-BFGS-B
    d3 = t0 + budget_s * 0.40
    h_lp2, c5_lp2 = lp_loop(best_h, d3, radius_init=0.04)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 4: Multi-n search with LP
    d4 = t0 + budget_s * 0.65
    n_offsets = [2, -2, 4, -4, 1, -1, 6, -6, 8, -8, 10, -10, 12, -12]
    tried_n = set([best_n])
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 160:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.05)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Pair polish (thorough)
    d5 = t0 + budget_s * 0.88
    h_pol, c5_pol = pair_polish(best_h, d5)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 6: Final LP micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.008)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
