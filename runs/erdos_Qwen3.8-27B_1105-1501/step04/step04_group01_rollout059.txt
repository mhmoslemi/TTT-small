        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:

    return None

def _subgrad_step(h, lr, n_active=5):
    """One subgradient step focusing on active shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    
    # Top n_active shifts
    top_idx = np.argsort(scores)[-n_active:]
    
    # Softmax weights over active set
    diff = scores[top_idx] - max_s
    w = np.exp(20 * diff)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i, r in enumerate(top_idx):
        if w[i] < 1e-10:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[i] * (1.0 - h[d:d+m])
                grad[d:] += w[i] * (-h[:m])
        else:
            g2 = -d
            m = n - g2
            if m > 0:
                grad[:m] += w[i] * (-h[g2:g2+m])
                grad[g2:] += w[i] * (1.0 - h[:m])
    
    grad -= np.mean(grad)
    return h - lr * grad, max_s

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n):

    # Add a sinusoidal perturbation
    h += 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    h = np.clip(h, 0, 1)
    return _project(h, n / 2.0)

def _pair_refine(h, deadline, max_passes=30):

    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0

                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = len(active_indices)
    A = np.zeros((k, n))
    for i, r in enumerate(active_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                A[i, :m] += (1.0 - h[d:d + m])
                A[i, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                A[i, :m] += (-h[g:g + m])
                A[i, g:] += (1.0 - h[:m])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -corr[active_indices]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _solve_full_lp(h, radius):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = 2 * n - 1
    A = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            g2 = n - d
            if g2 > 0:
                A[r, :g2] += (1.0 - h[d:d + g2])
                A[r, d:] += (-h[:g2])
        else:
            g2 = -d
            g3 = n - g2
            if g3 > 0:
                A[r, :g3] += (-h[g2:g2 + g3])
                A[r, g2:] += (1.0 - h[:g3])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -corr
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _cutting_plane_loop(h, deadline, ws_size=12, radius_init=0.12, use_full=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    working_set = list(np.argsort(corr)[-ws_size:])
    while time.time() < deadline:
        if stall > 35:
            break
        if use_full:
            p = _solve_full_lp(h, radius)
        else:
            p = _solve_active_lp(h, working_set, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.35)
            if not use_full:
                corr = np.correlate(h, 1.0 - h, mode='full')
                top = set(np.argsort(corr)[-ws_size:])
                new_set = set(working_set) | top
                if len(new_set) > ws_size:
                    arr = list(new_set)
                    sc = np.array([corr[x] for x in arr])
                    order = np.argsort(sc)[-ws_size:]
                    working_set = [arr[i] for i in order]
                else:
                    working_set = list(new_set)
        else:
            stall += 1
            radius *= 0.75
            if stall % 8 == 0 and radius < 0.015:
                radius = 0.05
            if not use_full:
                corr = np.correlate(h, 1.0 - h, mode='full')
                working_set = list(np.argsort(corr)[-ws_size:])
    return best_h, best_c5

def _subgrad_loop(h, deadline, lr_init=0.012, n_active=6):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        if stall > 40:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * (2.0 / n)
        max_s = float(np.max(scores))
        top_idx = np.argsort(scores)[-n_active:]
        diff = scores[top_idx] - max_s
        w = np.exp(15.0 * diff)
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(n_active):
            if w[i] < 1e-10:
                continue
            r = top_idx[i]
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[i] * (1.0 - h[d:d + m])
                    grad[d:] += w[i] * (-h[:m])
            else:
                g2 = -d
                m = n - g2
                if m > 0:
                    grad[:m] += w[i] * (-h[g2:g2 + m])
                    grad[g2:] += w[i] * (1.0 - h[:m])
        grad -= np.mean(grad)
        h_new = h - lr * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, 0.25)
        else:
            stall += 1
            lr *= 0.88
            if lr < 5e-4:
                lr = 0.004
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, seed_val=0):
    h = np.ones(n) * 0.5
    h += 0.25 * np.sin(2.0 * np.pi * 2.0 * np.arange(n) / n + seed_val)
    h += 0.15 * np.sin(2.0 * np.pi * 5.0 * np.arange(n) / n + seed_val * 1.3)
    h = np.clip(h, 0, 1)
    return _project(h, n / 2.0)

def _pair_refine(h, deadline, max_passes=25):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _multi_n_search(best_h, best_c5, deadline, rng):
    n0 = len(best_h)
    candidates = []
    for dn in range(-20, 21):
        if dn == 0:
            continue
        n_try = n0 + dn
        if 24 <= n_try <= 250:
            candidates.append(n_try)
    for small_n in [24, 28, 32, 36, 40, 44, 48, 52, 56, 64, 72, 80, 96, 128, 144, 160, 176, 192]:
        if small_n not in candidates:
            candidates.append(small_n)
    rng.shuffle(candidates)
    tried = set()
    for n_try in candidates:
        if time.time() > deadline:
            break
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        if n_try <= 48:
            h_s = _structured_init(n_try, seed_val=float(n_try))
            if _exact_c5(h_s) < _exact_c5(h_r):
                h_r = h_s
        per_time = min(8.0, (deadline - time.time()) / max(1, len(candidates) - len(tried)))
        sub_dl = min(deadline, time.time() + per_time)
        h_r, c5_r = _cutting_plane_loop(h_r, sub_dl, ws_size=8, radius_init=0.12)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Cutting plane on initial (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _cutting_plane_loop(h, d1, ws_size=12, radius_init=0.12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient polish (4%)
    d2 = t0 + budget_s * 0.22
    h2, c5_2 = _subgrad_loop(best_h, d2, lr_init=0.012, n_active=6)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.47
    best_h, best_c5 = _multi_n_search(best_h, best_c5, d3, rng)

    # Phase 4: Coarse-to-fine (8%)
    d4 = t0 + budget_s * 0.55
    if time.time() < d4:
        best_n = len(best_h)
        for frac in [4, 3, 2]:
            cn = max(12, best_n // frac)
            hc = _resample(best_h, cn)
            sub_dl = min(d4, time.time() + budget_s * 0.01)
            hc, c5c = _cutting_plane_loop(hc, sub_dl, ws_size=6, radius_init=0.15)
            hu = _resample(hc, best_n)
            c5u = _exact_c5(hu)
            if c5u < best_c5:
                best_c5 = c5u
                best_h = hu.copy()

    # Phase 5: Perturbation restarts (15%)
    d5 = t0 + budget_s * 0.70
    scale = 0.025
    n_best = len(best_h)
    for attempt in range(100):
        if time.time() > d5:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, n_best)
        h_pert = _project(h_pert, n_best / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.007)
        h_t, c5_t = _cutting_plane_loop(h_pert, sub_dl, ws_size=8, radius_init=0.08)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.09, scale * 1.18)

    # Phase 6: Full LP deep optimization (15%)
    d6 = t0 + budget_s * 0.85
    h6, c5_6 = _cutting_plane_loop(best_h, d6, ws_size=14, radius_init=0.06, use_full=True)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Pair refinement (8%)
    d7 = t0 + budget_s * 0.93
    h_p, c5_p = _pair_refine(best_h, d7, max_passes=20)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 8: Final full LP polish (5%)
    d8 = t0 + budget_s * 0.98
    h_f, c5_f = _cutting_plane_loop(best_h, d8, ws_size=16, radius_init=0.03, use_full=True)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
