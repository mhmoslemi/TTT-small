from scipy.optimize import linprog, minimize
import time

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0: continue
                a = min(1.0 - x[i], diff)
                x[i] += a; diff -= a
                if diff <= 1e-12: break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0: continue
                a = min(x[i], -diff)
                x[i] -= a; diff += a
                if diff >= -1e-12: break
    return x

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    return float(np.max(_corr(h))) * 2.0 / len(h)

def _grad_rows(h, idx, n):
    j = np.arange(n)
    rows = np.zeros((len(idx), n))
    for k, r in enumerate(idx):
        s = r - (n - 1)
        g = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        g[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        g[m2] -= h[ps[m2]]
        rows[k] = g
    return rows

def _lp_dir(h, corr, aidx, n, radius):
    J = _grad_rows(h, aidx, n)
    K = len(aidx)
    c = np.zeros(n + 1); c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = J; A_ub[:, n] = -1.0
    b_ub = -corr[aidx]
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[i]), min(radius, 1.0 - h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success: return res.x[:n]
    except: pass
    return None

def _ls(h, p, n, c5c):
    tgt = n * 0.5
    amax = 1.0
    for i in range(n):
        if p[i] > 1e-14: amax = min(amax, (1-h[i])/p[i])
        elif p[i] < -1e-14: amax = min(amax, h[i]/(-p[i]))
    amax = max(amax, 1e-10)
    ba, bc = 0.0, c5c
    for a in np.linspace(0, amax, 10):
        if a < 1e-10: continue
        c5n = _c5(_project(h + a*p, tgt))
        if c5n < bc: bc, ba = c5n, a
    if ba <= 1e-10: return ba, bc
    lo, hi = max(0, ba-amax*0.3), min(amax, ba+amax*0.3)
    gr = (np.sqrt(5)-1)/2
