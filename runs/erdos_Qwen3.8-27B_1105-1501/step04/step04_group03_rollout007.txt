def binary_local_search(h, deadline, max_rounds=100):
    n = len(h)
    # Create binary version
    threshold = np.sort(h)[::-1][n//2 - 1]
    binary = (h >= threshold).astype(float)
    # Ensure exactly n/2 ones
    num_ones = int(np.sum(binary))
    if num_ones != n // 2:
        idx = np.argsort(h)[::-1]
        binary = np.zeros(n)
        binary[idx[:n//2]] = 1.0
    
    best_c5 = exact_c5(binary)
    best_binary = binary.copy()
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try all single-bit flips (change 1 to 0 and 0 to 1 simultaneously - swap)
        ones = np.where(binary == 1)[0]
        zeros = np.where(binary == 0)[0]
        for i_idx in range(len(ones)):
            if time.time() > deadline:
                break
            for j_idx in range(len(zeros)):
                i, j = ones[i_idx], zeros[j_idx]
                binary[i] = 0
                binary[j] = 1
                c5 = exact_c5(binary)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_binary = binary.copy()
                    improved = True
                    # Restart search from new state
                    ones = np.where(binary == 1)[0]
                    zeros = np.where(binary == 0)[0]
                    break
                binary[i] = 1
                binary[j] = 0
            if improved:
                break
        if not improved:
            break
    return best_binary, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_corr_grad_subset(h, active_indices):
    """Build gradient matrix only for active correlation indices."""
    n = len(h)
    G = np.zeros((len(active_indices), n))
    for idx, r in enumerate(active_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def lp_direction_fast(h, scores, active_frac=0.5, radius=0.1):
    """LP direction using only top active constraints."""
    n = len(h)
    n_corr = 2 * n - 1
    # Select active constraints (top fraction)
    n_active = max(10, int(n_corr * active_frac))
    active_idx = np.argsort(scores)[::-1][:n_active]
    active_idx = np.sort(active_idx)
    G = build_corr_grad_subset(h, active_idx)
    active_scores = scores[active_idx]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -active_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search_fast(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(12):
        a = alpha_max * (i + 1) / 12.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(12):
        if hi_a - lo_a < 1e-12:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:

    return best_a, best_c5

def lp_loop(h, deadline, radius_init=0.1, active_frac=0.5, kick_prob=0.05):

    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline:
        if stall > 100:
            # Try a random kick
            if rng.random() < kick_prob:
                h = h + rng.normal(0, 0.05, n)
                h = project(h, target)

        p = lp_direction_fast(h, scores, active_frac, radius)

            stall += 1
            radius *= 0.8
            if radius < 1e-8:

            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.85

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_corr_grad_all(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def build_corr_grad_subset(h, active_r):
    n = len(h)
    G = np.zeros((len(active_r), n))
    for idx, r in enumerate(active_r):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def lp_direction(h, scores, radius, active_frac=0.5):
    n = len(h)
    n_corr = 2 * n - 1
    n_active = max(8, int(n_corr * active_frac))
    top_idx = np.argsort(scores)[::-1][:n_active]
    top_idx = np.sort(top_idx)
    G = build_corr_grad_subset(h, top_idx)
    a_scores = scores[top_idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(top_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -a_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(10):
        a = alpha_max * (i + 1) / 10.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(12):
        if hi_a - lo_a < 1e-13:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lp_loop(h, deadline, radius_init=0.1, active_frac=0.5, seed_offset=0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(42 + seed_offset)
    frac_cycle = [0.5, 0.3, 0.7, 1.0, 0.4]
    frac_i = 0
    while time.time() < deadline:
        if stall > 120:
            if rng.random() < 0.1:
                kick = 0.03 + 0.04 * rng.random()
                h = h + rng.normal(0, kick, n)
                h = project(h, target)
                cur_c5 = exact_c5(h)
                stall = 0
                radius = radius_init * 0.6
                frac_i = (frac_i + 1) % len(frac_cycle)
                continue
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        cur_frac = frac_cycle[frac_i]
        p = lp_direction(h, scores, radius, cur_frac)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.82
            if radius < 1e-8:
                radius = radius_init * 0.03
            if stall % 20 == 0:
                frac_i = (frac_i + 1) % len(frac_cycle)
            continue
        alpha, new_c5 = line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.4)
        else:
            stall += 1
            radius *= 0.88
    return best_h, best_c5

def binary_local_search(h, deadline, max_rounds=200):
    n = len(h)
    target = n // 2
    threshold = np.sort(h)[::-1][target - 1]
    binary = (h >= threshold).astype(float)
    num_ones = int(np.sum(binary))
    if num_ones != target:
        idx = np.argsort(h)[::-1]
        binary = np.zeros(n)
        binary[idx[:target]] = 1.0
    best_c5 = exact_c5(binary)
    best_binary = binary.copy()
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(best_binary == 1)[0]
        zeros = np.where(best_binary == 0)[0]
        for ii in range(len(ones)):
            if time.time() > deadline:
                break
            for jj in range(len(zeros)):
                i, j = ones[ii], zeros[jj]

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        omh = 1.0 - x
        scores = np.correlate(x, omh, mode='full') * (2.0 / n)
        max_s = float(np.max(scores))
        exp_s = np.exp((scores - max_s) / tau)
        Z = float(np.sum(exp_s))
        return max_s + tau * np.log(Z)

    def grad(x):
        omh = 1.0 - x
        scores_raw = np.correlate(x, omh, mode='full')
        scores = scores_raw * (2.0 / n)
        max_s = float(np.max(scores))
        exp_s = np.exp((scores - max_s) / tau)
        Z = float(np.sum(exp_s))
        w = exp_s / Z
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            d = k - (n - 1)
            js1 = max(0, -d)
            je1 = min(n - 1, n - 1 - d)
            if js1 <= je1:
                g[js1:je1 + 1] += wk * (1.0 - x[js1 + d:je1 + 1 + d])
            js2 = max(0, d)
            je2 = min(n - 1, n + d - 1)
            if js2 <= je2:
                g[js2:je2 + 1] -= wk * x[js2 - d:je2 + 1 - d]
        return (2.0 / n) * g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        a = amax * frac
                        h_t = best_h.copy()
                        h_t[j] += sign * a
                        h_t[k] -= sign * a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def ga_search(h_init, deadline, pop_size=30, seed=99):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        pert = h_init + rng.normal(0, 0.1, n)
        population.append(project(pert, target))
    # Also add some structured perturbations
    for i in range(min(5, pop_size - 1)):
        mask = rng.random(n) < 0.1
        pert = h_init.copy()
        pert[mask] = 1.0 - pert[mask]
        population[i + 1] = project(pert, target)
    population = population[:pop_size]
    scores = [exact_c5(p) for p in population]
    best_idx = int(np.argmin(scores))
    best_c5 = scores[best_idx]
    best_h = population[best_idx].copy()
    
    gen = 0
    while time.time() < deadline and gen < 500:
        gen += 1
        # Sort by fitness
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
        if population[0] is not best_h:
            if scores[0] < best_c5:
                best_c5 = scores[0]
                best_h = population[0].copy()
        
        new_pop = [population[0].copy()]
        for _ in range(pop_size - 1):
            if time.time() > deadline:
                break
            # Crossover
            p1_idx = int(rng.integers(0, max(1, pop_size // 3)))
            p2_idx = int(rng.integers(0, max(1, pop_size // 2)))
            alpha = rng.random()
            child = alpha * population[p1_idx] + (1 - alpha) * population[p2_idx]
            # Mutation
            if rng.random() < 0.3:
                child = child + rng.normal(0, 0.05, n)
            child = project(child, target)
            new_pop.append(child)
        
        # Evaluate new population
        new_scores = [exact_c5(p) for p in new_pop]
        # Selection
        combined = new_pop + population
        combined_scores = new_scores + scores
        combined = combined[:pop_size]
        combined_scores = combined_scores[:pop_size]
        # Keep top pop_size
        order2 = np.argsort(combined_scores)
        population = [combined[i] for i in order2[:pop_size]]
        scores = [combined_scores[i] for i in order2[:pop_size]]
    
    if scores[0] < best_c5:
        best_c5 = scores[0]
        best_h = population[0].copy()
    return best_h, best_c5
