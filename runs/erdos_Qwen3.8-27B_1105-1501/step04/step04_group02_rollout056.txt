import numpy as np
from scipy.optimize import minimize
import time

_N = 0

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return x[_N]

    g[_N] = 1.0
    return g

    T = x[_N]
    return T - _raw_scores(h)

    h = x[:n]
    m = 2 * n - 1
    jac = np.zeros((m, n + 1))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            jac[r, d:d+cnt] += 1.0 - h[:cnt]  # contribution from h[i+d] term
            jac[r, :cnt] += -h[d:d+cnt]  # contribution from h[i] term  
            # Wait, let me re-derive
            # S_r = sum_{i=0}^{n-d-1} h[i+d] * (1 - h[i])
            # dS_r/dh[j] for j: 
            #   If j < n-d (i.e., j is in the (1-h[i]) group): dS_r/dh[j] = -h[j+d]
            #   If j >= d (i.e., j is in the h[i+d] group): dS_r/dh[j] = 1 - h[j-d]
            #   If both (j < n-d AND j >= d): dS_r/dh[j] = 1 - h[j-d] - h[j+d]
            # In slice form:
            #   jac[r, :cnt] += -h[d:d+cnt]  (from the -h[i] group, positions 0..cnt-1)
            #   jac[r, d:d+cnt] += 1 - h[:cnt]  (from the h[i+d]*(1-h[i]) group, positions d..d+cnt-1)

            cnt = n - s

            # dS_r/dh[j]:
            #   If j < n-s: dS_r/dh[j] = 1 - h[j+s]
            #   If j >= s: dS_r/dh[j] = -h[j-s]
            #   If both: 1 - h[j+s] - h[j-s]

            jac[r, s:s+cnt] += -h[:cnt]
        jac[r, n] = 1.0

    dx_old = 2.0 / n

    # Old cell i spans [i*dx_old, (i+1)*dx_old]

    new_h = np.zeros(n_new)

            i_end = (i + 1) * dx_old
            overlap = min(j_end, i_end) - max(j_start, i_start)
            if overlap > 0:
                new_h[j] += h[i] * overlap / dx_new
    return _project(new_h, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

_N = 0

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _slsqp_obj(x):
    return x[_N]

def _slsqp_obj_grad(x):
    g = np.zeros(_N + 1)
    g[_N] = 1.0
    return g

def _slsqp_ineq(x):
    h = x[:_N]
    T = x[_N]
    return T - _raw_scores(h)

def _slsqp_ineq_jac(x):
    n = _N
    h = x[:n]
    m = 2 * n - 1
    jac = np.zeros((m, n + 1))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            if cnt > 0:
                jac[r, :cnt] += -h[d:d + cnt]
                jac[r, d:d + cnt] += 1.0 - h[:cnt]
        else:
            s = -d
            cnt = n - s
            if cnt > 0:
                jac[r, :cnt] += 1.0 - h[s:s + cnt]
                jac[r, s:s + cnt] += -h[:cnt]
        jac[r, n] = 1.0
    return jac

def _slsqp_eq(x):
    return x[:_N].sum() - _N / 2.0

def _slsqp_eq_jac(x):
    j = np.zeros(_N + 1)
    j[:_N] = 1.0
    return j

def _epigraph_solve(h0, deadline, ftol=1e-12, maxiter=250):
    global _N
    n = len(h0)
    _N = n
    h = _project(h0, n / 2.0)
    cur_c5 = _exact_c5(h)
    scores = _raw_scores(h)
    T0 = float(scores.max()) + 1e-5

    x0 = np.concatenate([h, [T0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    constraints = [
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac},
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}
    ]
    maxfev = maxiter * 15
    try:
        res = minimize(_slsqp_obj, x0, jac=_slsqp_obj_grad, bounds=bounds,
                       constraints=constraints, method='SLSQP',
                       options={'ftol': ftol, 'maxiter': maxiter,
                                'maxfev': maxfev, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_sol = _project(np.clip(res.x[:n], 0.0, 1.0), n / 2.0)
            c5_sol = _exact_c5(h_sol)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x, h, left=0.0, right=0.0)
    return _project(h_new, n_new / 2.0)

def _grad_step(h, w, lr, n):
    """One gradient step with weighted gradient."""
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += wr * (-h[d:d + m])
                grad[d:d + m] += wr * (1.0 - h[:m])
        else:
            s = -d
            m = n - s
            if m > 0:
                grad[:m] += wr * (1.0 - h[s:s + m])
                grad[s:s + m] += wr * (-h[:m])
    return h - lr * grad

    d3 = min(deadline, t0 + budget_s * 0.65)
    h_try, c5_try = _epigraph_solve_multi(best_h, d3, ftol=1e-13)
    if c5_try < best_c5:
        best_c5 = c5_try
        best_h = h_try.copy()

    # Phase 4: Gradient + SLSQP refinement (15% budget)
    d4 = min(deadline, t0 + budget_s * 0.80)
    h_g, c5_g = _softmax_grad_opt(best_h, min(d4, time.time() + budget_s * 0.05),
                                   tau=20.0, lr=0.008, mom=0.88)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
    if time.time() < d4:
        h_try, c5_try = _epigraph_solve(best_h, d4, ftol=1e-13, maxiter=200)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Perturbation restarts (10% budget)
    d5 = min(deadline, t0 + budget_s * 0.90)
    rng = np.random.default_rng(seed)
    n_best = len(best_h)
    scale = 0.03
    time_per_pert = budget_s * 0.015
    for attempt in range(50):
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, scale, n_best)
        h_pert = _project(h_pert, n_best / 2.0)
        dl_p = min(d5, time.time() + time_per_pert)
        h_try, c5_try = _epigraph_solve(h_pert, dl_p, ftol=1e-12, maxiter=100)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.7)
        else:
            scale = min(0.2, scale * 1.3)

    # Phase 6: Pair refinement (8% budget)
    d6 = min(deadline, t0 + budget_s * 0.96)
    h_p, c5_p = _pair_refine(best_h, d6, max_passes=200, seed=seed)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 7: Final SLSQP polish (4% budget)
    d7 = deadline
    if time.time() < d7 and (d7 - time.time()) > 2.0:
        h_try, c5_try = _epigraph_solve(best_h, d7, ftol=1e-14, maxiter=100)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return (best_h, final_c5, len(best_h))
