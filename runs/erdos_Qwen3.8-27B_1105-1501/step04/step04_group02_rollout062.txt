def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    """Compute all raw correlation scores matching np.correlate(h, 1-h, 'full')."""
    n = len(h)
    scores = np.zeros(2 * n - 1)
    one_minus_h = 1.0 - h
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            scores[r] = float(np.dot(h[d:], one_minus_h[:m]))
        else:
            s = -d
            m = n - s
            scores[r] = float(np.dot(h[:m], one_minus_h[s:]))

def _pair_refine(h, deadline, max_passes=50):
    """Local pair refinement preserving sum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(hash(_pass) & 0xFFFFFFFF)
        
        # Random pairs
        if n > 50:
            n_pairs = min(200, n * 4)
        else:
            n_pairs = n * (n - 1) // 2
        
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                continue
            # Try both directions
            for sign in [1, -1]:
                if sign > 0:
                    am = min(1.0 - h[j], h[k])
                else:
                    am = min(h[j], 1.0 - h[k])
                if am < 1e-10:
                    continue
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    h_t = h.copy()
                    h_t[j] += sign * am * frac
                    h_t[k] -= sign * am * frac
                    if h_t[j] < -1e-12 or h_t[j] > 1.0 + 1e-12:
                        continue
                    if h_t[k] < -1e-12 or h_t[k] > 1.0 + 1e-12:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

def _grad_smooth(h, deadline, tau=30.0, lr=0.01, max_iter=2000):
    """Gradient descent on smooth max approximation."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    mom = np.zeros(n)
    count = 0
    while time.time() < deadline and count < max_iter:
        count += 1
        scores = _raw_scores(h)
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        
        # Gradient of weighted sum of scores
        grad = np.zeros(n)
        for r in range(len(scores)):
            wr = w[r]
            if wr < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] += wr * (-h[d:])
            else:
                s = -d
                m = n - s
                grad[:m] += wr * (1.0 - h[s:])
                grad[s:] += wr * (-h[:m])
        
        mom = 0.9 * mom + 0.1 * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
        elif c5_new < _exact_c5(h) - 1e-13:
            h = h_new
        else:
            lr *= 0.9
            if lr < 1e-5:
                break
    return best_h, best_c5
