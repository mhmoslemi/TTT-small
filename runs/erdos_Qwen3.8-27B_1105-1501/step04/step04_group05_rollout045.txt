        return v

        return g

import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(v - mu, 0.0, 1.0)

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(_corr_vec(h)) * dx)

def _smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = _corr_vec(h) * dx
    m = float(np.max(c))
    shifted = (c - m) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    e = np.exp(shifted)
    se = float(np.sum(e))
    p = e / se
    val = m + tau * np.log(se)
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = p[q] * dx
        if w < 1e-10:
            continue
        s = q - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            d = -s
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n - d]
    return val, grad

def _smooth_slsqp(h0, tau, maxiter=300, ftol=1e-13):
    n = len(h0)
    target = n / 2.0

    def obj(h):
        v, _ = _smooth_obj_and_grad(h, tau)
        return v

    def jac(h):
        _, g = _smooth_obj_and_grad(h, tau)
        return g

    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h0.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swaps(h, deadline, max_rounds=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _c5(h)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        c = _corr_vec(h)
        top_lags = np.argsort(c)[-3:]
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-4 or h[j] > 1.0 - 1e-4:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-4 or h[k] > 1.0 - 1e-4:
                    continue
                for sgn in [1.0, -1.0]:
                    a_max = min(1.0 - h[j] if sgn > 0 else h[j],
                                (h[k] if sgn > 0 else 1.0 - h[k]))
                    if a_max < 1e-6:
                        continue
                    for frac in [0.5, 1.0, 0.25, 0.75, 0.1]:
                        a = frac * a_max
                        h_t = h.copy()

                        h_t[k] -= sgn * a
                        if h_t[j] < 0 or h_t[j] > 1 or h_t[k] < 0 or h_t[k] > 1:
                            continue

                            improved = True
                            break
                    if improved:
                        break
                if improved:

        if not improved:

    return h, best_c5

def _grad_descent_smooth(h, tau, deadline, lr_init=0.01, windowns=50):

    target = n / 2.0

    best_h = h.copy()

def _active_set_slsqp(h, active_frac=0.4, min_count=10, maxiter=200):
    n = len(h)
    target = n / 2.0
    g = _corr_vec(h)
    max_g = float(np.max(g))
    if max_g <= 0:
        return h.copy()
    threshold = max_g * active_frac
    active = np.where(g >= threshold)[0]
    if len(active) < min_count:
        k = min(min_count * 2, len(g))
        active = np.argsort(g)[-k:]
    max_c = max_g * (2.0 / n)

    x0 = np.zeros(n + 1)
    x0[0] = max_c * 1.01 + 0.02
    x0[1:] = h

    def slsqp_ineq(x, active):
        h_x = x[1:]
        g_x = _corr_vec(h_x) * (2.0 / len(h_x))
        return x[0] - g_x[active]

    def slsqp_ineq_jac(x, active):
        h_x = x[1:]
        n_x = len(h_x)
        dx = 2.0 / n_x
        J = np.zeros((len(active), n_x + 1))
        J[:, 0] = 1.0
        for idx, q in enumerate(active):
            s = q - (n_x - 1)
            if s >= 0:
                m1 = np.arange(s, n_x)
                J[idx, m1] += -dx * (1.0 - h_x[:n_x - s])
                m2 = np.arange(0, n_x - s)
                J[idx, m2] -= -dx * h_x[s:]
            else:
                d = -s
                m1 = np.arange(0, n_x - d)
                J[idx, m1] += -dx * (1.0 - h_x[d:])
                m2 = np.arange(d, n_x)
                J[idx, m2] -= -dx * h_x[:n_x - d]
        return J

    def slsqp_eq(x):
        return x[1:].sum() - target

    def slsqp_eq_jac(x):
        g_out = np.zeros(n + 1)
        g_out[1:] = 1.0
        return g_out

    def obj(x):
        return x[0]

    def obj_grad(x):
        g_out = np.zeros(n + 1)
        g_out[0] = 1.0
        return g_out

    constraints = [
        {'type': 'ineq', 'fun': lambda x: slsqp_ineq(x, active),
         'jac': lambda x: slsqp_ineq_jac(x, active)},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h
