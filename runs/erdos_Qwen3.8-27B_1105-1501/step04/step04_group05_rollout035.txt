import numpy as np
from scipy.optimize import minimize
import time

def _correlate_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _scaled_corr(h):
    n = len(h)
    return (2.0 / n) * _correlate_vec(h)

def _exact_c5(h):
    return float(np.max(_correlate_vec(h)) * (2.0 / len(h)))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _lse_value_grad(h, tau):
    """Compute value and gradient of tau * logsumexp(c/tau)."""
    n = len(h)
    c = _scaled_corr(h)
    m = c.max()
    log_p = (c - m) / tau
    p = np.exp(log_p)
    p /= p.sum()
    val = m + tau * np.log(p.sum())
    
    dx = 2.0 / n
    grad = np.zeros(n)
    # s = 0
    w = dx * p[n-1]
    grad += w * (1.0 - 2.0 * h)
    # s > 0
    for s in range(1, n):
        w = dx * p[n-1+s]
        grad[:n-s] += w * (1.0 - h[s:])
        grad[s:] -= w * h[:n-s]
    # s < 0
    for d in range(1, n):
        w = dx * p[n-1-d]
        grad[d:] += w * (1.0 - h[:-d])
        grad[:n-d] -= w * h[d:]
    return val, grad

def _lse_slsqp(h0, tau, maxiter=200, ftol=1e-12):
    """Solve min LSE with SLSQP."""
    n = len(h0)
    target = n / 2.0
    
    def obj(h):
        v, _ = _lse_value_grad(h, tau)
        return v
    
    def grad(h):
        _, g = _lse_value_grad(h, tau)
        return g
    
    def eq_fun(h):
        return h.sum() - target
    
    def eq_jac(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.success or res.x is not None:
            h_new = _project(res.x, target)
            return h_new
    except Exception:
        pass
    return h0

def _target_penalty_value_grad(h, target_c, rho, tau_lse):
    """
    Objective: sum of (max(0, c_k - target_c)/rho)^2 + tau_lse * LSE
    This penalizes constraints exceeding target_c.
    """
    n = len(h)
    c = _scaled_corr(h)
    excess = np.maximum(c - target_c, 0.0)
    penalty = np.sum((excess / rho) ** 2)
    
    # LSE part for smoothness
    m = c.max()
    log_p = (c - m) / tau_lse
    p_lse = np.exp(log_p)
    p_lse /= p_lse.sum()
    lse_val = m + tau_lse * np.log(p_lse.sum())
    
    val = penalty + 0.01 * lse_val
    
    # Gradient
    dx = 2.0 / n
    w = (2.0 * excess / rho**2)  # gradient of penalty w.r.t. c_k
    w += 0.01 * p_lse  # gradient of LSE w.r.t. c_k
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        k = n - 1 + s
        wk = dx * w[k]
        if wk == 0:
            continue
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n-s]
        else:
            d = -s
            grad[d:] += wk * (1.0 - h[:-d])
            grad[:n-d] -= wk * h[d:]
    return val, grad

def _target_slsqp(h0, target_c, rho, tau_lse, maxiter=200, ftol=1e-12):
    """Solve target penalty with SLSQP."""
    n = len(h0)
    target = n / 2.0
    
    def obj(h):
        v, _ = _target_penalty_value_grad(h, target_c, rho, tau_lse)
        return v
    
    def grad(h):
        _, g = _target_penalty_value_grad(h, target_c, rho, tau_lse)
        return g
    
    def eq_fun(h):
        return h.sum() - target
    
    def eq_jac(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None:
            h_new = _project(res.x, target)
            return h_new
    except Exception:
        pass
    return h0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_descent_lse(h, tau, lr, n_steps=100, deadline=None):
    """Simple projected gradient descent on LSE."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        _, grad = _lse_value_grad(h_cur, tau)
        # Remove mean to preserve sum (approximately)
        grad -= grad.mean()
        h_new = _project(h_cur - lr * grad, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def _correlate_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _scaled_corr(h):
    n = len(h)
    return (2.0 / n) * _correlate_vec(h)


def _exact_c5(h):
    return float(np.max(_correlate_vec(h)) * (2.0 / len(h)))


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _lse_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = dx * _correlate_vec(h)
    m = c.max()
    shifted = (c - m) / tau
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    val = m + tau * np.log(exp_shifted.sum())
    
    grad = np.zeros(n)
    # s = 0
    w = dx * p[n - 1]
    grad += w * (1.0 - 2.0 * h)
    # s > 0
    for s in range(1, n):
        w = dx * p[n - 1 + s]
        if w == 0.0:
            continue
        grad[:n - s] += w * (1.0 - h[s:])
        grad[s:] -= w * h[:n - s]
    # s < 0
    for d in range(1, n):
        w = dx * p[n - 1 - d]
        if w == 0.0:
            continue
        grad[d:] += w * (1.0 - h[:-d])
        grad[:n - d] -= w * h[d:]
    return val, grad


def _lse_obj(h, tau):
    v, _ = _lse_value_grad(h, tau)
    return v


def _lse_grad(h, tau):
    _, g = _lse_value_grad(h, tau)
    return g


def _sum_eq(h, target):
    return h.sum() - target


def _sum_jac(h, target):
    return np.ones(len(h))


def _solve_lse(h0, tau, maxiter=300, ftol=1e-13):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_lse_obj, h0.copy(), jac=_lse_grad, args=(tau,),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project(res.x, target)
    except Exception:
        pass
    return h0


def _target_penalty_vg(h, target_c, rho, tau_lse):
    n = len(h)
    dx = 2.0 / n
    c = dx * _correlate_vec(h)
    excess = np.maximum(c - target_c, 0.0)
    penalty = np.sum(excess * excess) / (rho * rho)
    
    m = c.max()
    shifted = (c - m) / tau_lse
    exp_s = np.exp(shifted)
    p_lse = exp_s / exp_s.sum()
    lse_val = m + tau_lse * np.log(exp_s.sum())
    
    val = penalty + 0.01 * lse_val
    
    # Gradient w.r.t. c_k
    w = (2.0 * excess / (rho * rho)) + 0.01 * p_lse
    
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        k = n - 1 + s
        wk = dx * w[k]
        if wk == 0.0:
            continue
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
        else:
            d = -s
            grad[d:] += wk * (1.0 - h[:-d])
            grad[:n - d] -= wk * h[d:]
    return val, grad


def _tp_obj(h, target_c, rho, tau_lse):
    v, _ = _target_penalty_vg(h, target_c, rho, tau_lse)
    return v


def _tp_grad(h, target_c, rho, tau_lse):
    _, g = _target_penalty_vg(h, target_c, rho, tau_lse)
    return g


def _solve_target(h0, target_c, rho, tau_lse, maxiter=200, ftol=1e-13):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_tp_obj, h0.copy(), jac=_tp_grad,
                      args=(target_c, rho, tau_lse),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project(res.x, target)
    except Exception:
        pass
    return h0


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _grad_step_lse(h, tau, lr, n_lags=0):
    """One projected gradient step on LSE objective."""
    n = len(h)
    target = n / 2.0
    _, grad = _lse_value_grad(h, tau)
    grad -= grad.mean()  # Project onto tangent space of sum constraint
    h_new = _project(h - lr * grad, target)
    return h_new


def _make_binary_h(n, seed):
    """Create a random binary pattern with exactly n/2 ones."""
    rng = np.random.default_rng(seed)
    pos = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0
    return h


def _make_block_h(n):
    """Create a block pattern: first n/2 are 1, rest are 0."""
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h


def _make_sin_h(n, freq=1.0):
    """Create a sinusoidal pattern."""
    x = np.arange(n) / n * 2.0  # [0, 2)
    h = 0.5 + 0.4 * np.sin(np.pi * freq * x)
    return _project(h, n / 2.0)
