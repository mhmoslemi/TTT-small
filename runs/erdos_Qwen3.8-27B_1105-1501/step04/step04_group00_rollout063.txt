import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] += dx * (1.0 - h[d:d+m])
            G[r, d:n] += -dx * h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += -dx * h[g:g+m]
            G[r, g:n] += dx * (1.0 - h[:m])
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    """Multi-scale log-sum-exp smoothed gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        gamma = gamma_start * (gamma_end / gamma_start) ** (frac ** 0.7)
        lr = lr_base / (1.0 + 0.3 * frac)
        G = _grad_matrix(h)
        stall = 0
        for step in range(steps_per_phase):
            if time.time() > deadline:
                break
            if stall > 15:
                break
            scores = _scores(h)
            shifted = scores - scores.max()
            w = np.exp(gamma * shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(len(scores))
                w[int(np.argmax(scores))] = 1.0
            else:
                w /= w_sum
            grad = G.T @ w
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                stall += 1
                continue
            grad /= gn
            alpha = lr
            found = False

def _fourier_opt(h, deadline, K=12, method='Nelder-Mead', n_restarts=3, rng=None):
    """Optimize in Fourier coefficient space."""
    n = len(h)
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(2 * np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        # Initial coefficients
        if restart == 0:
            a_init = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
        else:
            a_init = np.linalg.lstsq(B, best_h - 0.5, rcond=None)[0] + \
                     0.1 * rng.standard_normal(2*K)
        
        def objective(a):
            h_try = 0.5 + B @ a
            h_try = _project(h_try, target)
            return _exact_c5(h_try)
        
        try:
            res = minimize(objective, a_init, method=method,
                          options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.success:
                h_try = 0.5 + B @ res.x
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        except Exception:
            pass

def _block_sa(n, deadline, rng, n_iters=50000):

        bs = max(1, int(6 * (1 - frac) + 1))

        i = int(rng.integers(0, n - bs + 1))
        tries = 0
        while tries < 10:

        else:

        block_j = h[j:j+bs]
        if np.array_equal(block_i, block_j):
            continue
        
        # Swap
        h[i:i+bs] = block_j
        h[j:j+bs] = block_i
        
        new_c5 = _exact_c5(h)

        if delta < 0 or rng.random() < np.exp(-max(delta, -10) / T):

        else:
            # Undo
            h[i:i+bs] = block_i
            h[j:j+bs] = block_j

def _quasi_periodic_init(n, alpha):

    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    # Adjust to get exactly n//2 ones
    n_ones = int(h.sum())

        deficit = np.where(h == 0.0)[0]
        to_add = rng_global.choice(deficit, target - n_ones, replace=False)
        h[to_add] = 1.0
    return h

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):

        if d >= 0:
            m = n - d
            G[r, :m] += dx * (1.0 - h[d:d+m])
            G[r, d:n] += -dx * h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += -dx * h[g:g+m]
            G[r, g:n] += dx * (1.0 - h[:m])
    return G

    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, n_iters)
        bs = max(1, int(5 * (1 - frac) ** 1.5 + 1))
        i = int(rng.integers(0, n - bs + 1))
        j = int(rng.integers(0, n - bs + 1))
        if abs(i - j) < bs:
            continue
        bi = h[i:i+bs].copy()
        bj = h[j:j+bs].copy()
        if np.array_equal(bi, bj):
            continue
        h[i:i+bs] = bj
        h[j:j+bs] = bi
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-max(delta, -8.0) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i:i+bs] = bi
            h[j:j+bs] = bj
        T *= cool
    return best_h, best_c5

def _nm_fractional(h, deadline, max_iter=400):
    n = len(h)
    target = n / 2.0
    cur_c5 = _exact_c5(h)
    frac_mask = (h > 0.02) & (h < 0.98)
    frac_idx = np.where(frac_mask)[0]
    if len(frac_idx) == 0:
        return h, cur_c5
    if len(frac_idx) > 30:
        fracs = np.abs(h[frac_idx] - 0.5)
        order = np.argsort(-fracs)
        frac_idx = frac_idx[order[:30]]
    x0 = h[frac_idx].copy()
    def obj华侨(x):
        h_try = h.copy()
        h_try[frac_idx] = np.clip(x, 0, 1)
        h_try = _project(h_try, target)
        return _exact_c5(h_try)
    try:
        res = minimize(obj华侨, x0, method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-9, 'fatol': 1e-11})
        h_new = h.copy()
        h_new[frac_idx] = np.clip(res.x, 0, 1)
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5:
            return h_new, c5_new
    except Exception:
        pass
    return h, cur_c5

def _qp_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        excess = np.where(h == 1.0)[0]
        to_remove = rng.choice(excess, n_ones - target, replace=False)
        h[to_remove] = 0.0
    elif n_ones < target:
        deficit = np.where(h == 0.0)[0]
        to_add = rng.choice(deficit, target - n_ones, replace=False)
        h[to_add] = 1.0
    return h

    best_n = n0

    dl1 = t0 + budget_s * 0.12
