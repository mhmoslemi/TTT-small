    a_opt = (lo + hi) / 2.0
    return a_opt, _c5(h + a_opt * d)

def _subgrad_active(h, deadline, max_iters=5000, tol_frac=0.005):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    G = _grad_matrix(h)
    stall = 0
    tf = tol_frac
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        thresh = s_max * (1.0 - tf)
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        grad = G[active].mean(axis=0)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            stall += 1
            if stall > 80:
                break
            continue
        d = -grad / gn
        a_opt, new_c5 = _line_search(h, d)
        if new_c5 < cur_c5 - 1e-14:
            h = h + a_opt * d
            h = _project(h, target)
            cur_c5 = _c5(h)
            G = _grad_matrix(h)
            stall = 0
            tf = max(0.001, tf * 0.97)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            improved = False
            for scale in [0.5, 0.25, 0.1]:
                a2, nc2 = _line_search(h, d * scale)
                if nc2 < cur_c5 - 1e-14:
                    h = h + a2 * d * scale
                    h = _project(h, target)
                    cur_c5 = _c5(h)
                    G = _grad_matrix(h)
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                stall += 1
                tf = min(0.08, tf * 1.06)
    return best_h, best_c5

def _lp_refine(h, deadline, top_k=8, max_iters=4000):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        n_lags = 2 * n - 1
        tk = min(top_k, n_lags)
        idx = np.argsort(scores)[-tk:]
        sk = scores[idx]
        Gk = G[idx]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((tk + 2 * n, n + 1))
        b_ub = np.zeros(tk + 2 * n)
        row = 0
        for i in range(tk):
            A_ub[row, :n] = Gk[i]
            A_ub[row, n] = -1.0
            b_ub[row] = -sk[i]
            row += 1
        for j in range(n):
            A_ub[row, j] = 1.0
            b_ub[row] = 1.0
            row += 1
            A_ub[row, j] = -1.0
            b_ub[row] = 1.0
            row += 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                         bounds=[(None, None)] * (n + 1), method='highs')
        except Exception:
            stall += 1
            continue
        if not res.success:
            stall += 1
            continue
        p = res.x[:n]
        if np.max(np.abs(p)) < 1e-12:
            stall += 1
            continue
        a_opt, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = h + a_opt * p
            h = _project(h, target)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            improved = False
            for scale in [0.5, 0.25, 0.1, 0.05]:
                a2, nc2 = _line_search(h, p * scale)
                if nc2 < cur_c5 - 1e-13:
                    h = h + a2 * p * scale
                    h = _project(h, target)
                    cur_c5 = _c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
    return best_h, best_c5

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1500):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        improved = False
        for scale in [0.15, 0.1, 0.05, 0.025, 0.01]:
            a_opt, new_c5 = _line_search(h, d * scale)
            if new_c5 < cur_c5 - 1e-14:
                h = h + a_opt * d * scale
                h = _project(h, target)
                cur_c5 = _c5(h)
                G = _grad_matrix(h)
                stall = 0
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not improved:
            stall += 1
            if stall > 120:
                break
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = _c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-12:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 8:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_init(n, coeffs, bias=0.5):
    u = (np.arange(n) + 0.5) * 2.0 / n
    v = np.full(n, bias)
    for freq, (a, b) in enumerate(coeffs):
        v += a * np.cos(2.0 * np.pi * (freq + 1) * u)
        v += b * np.sin(2.0 * np.pi * (freq + 1) * u)
    return _project(v, n / 2.0)

def _binary_topn(v, n_target=None):
    n = len(v)
    if n_target is None:
        n_target = n // 2
    h = np.zeros(n)
    idx = np.zeros(n, dtype=bool)
    if n_target > 0:
        idx[np.argsort(v)[-n_target:]] = True
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    # Get initial h
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None
        n0 = 64

    if h0 is not None and len(h0) > 8:
        n0 = len(h0)
        h0 = _project(h0, n0 / 2.0)
    else:
        n0 = 64
        h0 = _project(np.full(n0, 0.5), n0 / 2.0)

    best_h = h0.copy()
    best_c5 = _c5(h0)
    best_n = n0

    # === Phase 1: Global initialization search (15%) ===
    dl1 = t0 + budget_s * 0.15
    # Try various Fourier initializations
    for K in range(2, 9):
        if time.time() > dl1:
            break
        coeffs = [(float(rng.uniform(-0.4, 0.4)), float(rng.uniform(-0.4, 0.4))) for _ in range(K)]
        for trial in range(3):
            if time.time() > dl1:
                break
            coeffs_t = [(float(rng.uniform(-0.5, 0.5)), float(rng.uniform(-0.5, 0.5))) for _ in range(K)]
            h_try = _fourier_init(n0, coeffs_t)
            c5_try = _c5(h_try)
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Try binary periodic patterns
    for period in [4, 5, 6, 7, 8, 10, 12, 16, 20, 24, 32]:
        if time.time() > dl1:
            break
        for phase in np.linspace(0, 1.0, 8, endpoint=False):
            if time.time() > dl1:
                break
            u = (np.arange(n0) + 0.5) / n0
            v = np.cos(2.0 * np.pi * period * (u + phase))
            h_bin = _binary_topn(v)
            c5_try = _c5(h_bin)
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_bin.copy()

    # === Phase 2: Intensive local search (40%) ===
    dl2 = t0 + budget_s * 0.55
    # Subgradient with active set
    hdr, c5r = _subgrad_active(best_h, min(dl2, t0 + budget_s * 0.20), max_iters=4000, tol_frac=0.005)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # LP refinement
    hdr, c5r = _lp_refine(best_h, min(dl2, t0 + budget_s * 0.35), top_k=8, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # P-norm descent
    hdr, c5r = _pnorm_descent(best_h, min(dl2, t0 + budget_s * 0.45), p_start=8.0, p_end=4096.0, n_steps=1200)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Another LP pass with fewer active lags
    hdr, c5r = _lp_refine(best_h, min(dl2, t0 + budget_s * 0.52), top_k=4, max_iters=2500)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # === Phase 3: Multi-start perturbation (15%) ===
    dl3 = t0 + budget_s * 0.70
    scale = 0.03
    for attempt in range(500):
        if time.time() > dl3:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl3, time.time() + budget_s * 0.012)
        h_try, c5_try = _lp_refine(h_pert, sub_dl, top_k=6, max_iters=400)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.12, scale * 1.08)

    # === Phase 4: Resolution sweep (10%) ===
    dl4 = t0 + budget_s * 0.80
    for dn in [-8, -6, -4, -3, -2, 2, 3, 4, 6, 8, 12, 16]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.010)
        h_r, c5_r = _lp_refine(h_r, sub_dl, top_k=6, max_iters=800)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # === Phase 5: Final intensive polish (15%) ===
    dl5 = deadline
    # Subgradient fine
    hdr, c5r = _subgrad_active(best_h, min(dl5, t0 + budget_s * 0.87), max_iters=3000, tol_frac=0.003)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # LP ultra-fine
    hdr, c5r = _lp_refine(best_h, min(dl5, t0 + budget_s * 0.93), top_k=3, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Pair polish
    sub_dl = min(dl5, t0 + budget_s * 0.96)
    hp, c5p = _pair_polish(best_h, sub_dl, max_passes=3)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, final_c5, int(best_n)
