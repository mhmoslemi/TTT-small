import numpy as np
import time

def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _all_M(h):
    """Compute all M_d values (the raw correlations times dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _active_set(M, tol=0.01):
    """Return indices of active lags (within tol of max)."""
    max_M = np.max(M)
    return np.where(M >= max_M - tol)[0]

def _gradient_max(M, h, active_idx):
    """Compute gradient of max M_d w.r.t. h at the active set."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in active_idx:
        d = r - (n - 1)
        low = max(0, -d)
        high = min(n, n - d)
        i = np.arange(low, high)
        j = i + d
        # d/dd_h_k [h_i(1-h_{i+d})] for k in the overlap:
        # if k = i: (1 - h_{i+d})
        # if k = i+d: -h_i
        grad[i] += dx * (1.0 - h[j])
        grad[j] += -dx * h[i]
    return grad

def _pair_move_exact(h, i, j, n_grid=50):
    """Exact 1D optimization for pair move h_i += tau, h_j -= tau."""
    n = len(h)
    target = n / 2.0
    tau_min = max(-h[i], h[j] - 1.0)
    tau_max = min(1.0 - h[i], h[j])
    if tau_max - tau_min < 1e-12:
        return h, _c5_value(h)
    
    best_h = h.copy()
    best_c5 = _c5_value(h)
    
    taus = np.linspace(tau_min, tau_max, n_grid)
    for t in taus:
        h[i] = h[i] + t
        h[j] = h[j] - t
        c5 = _c5_value(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h[i] = best_h[i]
        h[j] = best_h[j]
    
    return best_h, best_c5

        c = _c5(h)
        if c < best_c - 1e-15:
            best_c = c
            best_t = t
        h[i] = best_t and (hi0 + best_t) or hi0
        h[j] = best_t and (hj0 - best_t) or hj0
    h[i] = hi0
    h[j] = hj0
    return best_t

def _grad_step(h, n, ngrid=15):
    M = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    mx = M.max()
    active = np.where(M >= mx - 0.03 * mx)[0]
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        idx = np.arange(lo, hi)
        jdx = idx + d
        grad[idx] += (1.0 - h[jdx])
        grad[jdx] += -h[idx]
    grad *= dx / max(1, len(active))
    grad -= grad.mean()
    gn = np.max(np.abs(grad))
    if gn < 1e-15:
        return h, _c5(h)
    direction = -grad / gn
    best_h = h.copy()
    best_c = _c5(h)
    for a in np.geomspace(0.001, 0.3, ngrid):
        trial = _Proj(h + a * direction, n / 2.0)
        c = _c5(trial)
        if c < best_c - 1e-15:
            best_c = c
            best_h = trial.copy()
    return best_h, best_c

def _block_move(h, n, rng, blen=4):
    if blen >= n:
        return h, _c5(h)
    i = rng.integers(n - blen + 1)
    j = rng.integers(n - blen + 1)
    if abs(i - j) < blen:
        return h, _c5(h)
    block_i = slice(i, i + blen)
    block_j = slice(j, j + blen)
    tau_max = min(float(np.mean(h[block_j])) - float(np.min(h[block_j])),
                  1.0 - float(np.max(h[block_i])),
                  float(np.min(h[block_j])))
    tau_min = max(float(np.mean(h[block_i])) - float(np.max(h[block_i])),
                  -float(np.min(h[block_i])),
                  float(np.max(h[block_j])) - 1.0)
    if tau_max - tau_min < 1e-8:
        return h, _c5(h)
    best_h = h.copy()
    best_c = _c5(h)
    for t in np.linspace(tau_min, tau_max, 10):
        trial = h.copy()
        trial[block_i] = np.clip(trial[block_i] + t, 0, 1)
        trial[block_j] = np.clip(trial[block_j] - t, 0, 1)
        trial = _Proj(trial, n / 2.0)
        c = _c5(trial)
        if c < best_c - 1e-15:
            best_c = c
            best_h = trial
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _Proj(h_new, n_new / 2.0)

def _optimize_from(h, deadline, rng, label=""):
    n = len(h)
    best_h = h.copy()
    best_c5 = _c5(h)
    cur_c5 = best_c5
    temp = 0.005
    no_imp_streak = 0
    phase = 0

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 2:
            break

        # Determine which phase to use
        if no_imp_streak < 3:
            # Aggressive pair moves
            n_pairs = min(400, n * 3)
            for _ in range(n_pairs):
                if time.time() > deadline:
                    break
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                h2, c52 = _pair_move(h, i, j, ngrid=18)
                if c52 < cur_c5 - 1e-14:
                    h = h2
                    cur_c5 = c52
                    no_imp_streak = 0
                    if cur_c5 < best_c5:
                        best_c5 = c52
                        best_h = h.copy()
                elif temp > 1e-6 and c52 < cur_c5 + temp:
                    if rng.random() < np.exp(-(c52 - cur_c5) / max(temp, 1e-10)):
                        h = h2
                        cur_c5 = c52
            temp *= 0.97
        elif no_imp_streak < 8:
            # Gradient + pair mix
            h2, c52 = _grad_step(h, n)
            if c52 < cur_c5 - 1e-14:
                h = h2
                cur_c5 = c52
                no_imp_streak = 0
                if cur_c5 < best_c5:
                    best_c5 = c52
                    best_h = h.copy()
            for _ in range(100):
                if time.time() > deadline:
                    break
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                h2, c52 = _pair_move(h, i, j, ngrid=15)
                if c52 < cur_c5 - 1e-14:
                    h = h2
                    cur_c5 = c52
                    no_imp_streak = 0
                    if cur_c5 < best_c5:
                        best_c5 = c52
                        best_h = h.copy()
        else:
            # Perturb and restart
            n_pert = max(3, n // 16)
            idx = rng.integers(n, size=n_pert)
            for k in idx:
                h[k] = np.clip(h[k] + rng.uniform(-0.15, 0.15), 0.0, 1.0)
            h = _Proj(h, n / 2.0)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_imp_streak = 0
            temp = 0.01

        # Occasionally do block moves
        if rng.random() < 0.1:
            h2, c52 = _block_move(h, n, rng)
            if c52 < cur_c5 - 1e-14:
                h = h2
                cur_c5 = c52
                if cur_c5 < best_c5:
                    best_c5 = c52
                    best_h = h.copy()
                no_imp_streak = 0

        # Track no-improvement
        if best_c5 > cur_c5 + 1e-12 or no_imp_streak > 0:
            pass
        # Increment streak if no best improvement this round
        # (simplified: just always increment at end of loop)
        no_imp_streak += 0.5  # fractional to allow gradual escalation

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError):
        h0 = np.full(128, 0.5)

    n0 = len(h0)
    h0 = _Proj(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _c5(h0)
    best_n = n0

    # Phase 1: Main optimization from initial point (60% of budget)
    deadline1 = t0 + budget_s * 0.60
    h1, c5_1 = _optimize_from(h0, deadline1, rng, "main")
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Multi-resolution (25% of budget)
    deadline2 = t0 + budget_s * 0.85
    n_candidates = sorted(set([
        best_n - 16, best_n - 8, best_n - 4,
        best_n + 4, best_n + 8, best_n + 16,
        72, 80, 96, 112, 128, 144, 160, 192
    ]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 200 and x != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    per_n_budget = (deadline2 - time.time()) / max(1, len(n_candidates))
    for n_try in n_candidates:
        if time.time() > deadline2:
            break
        h_r = _resample(best_h, n_try)
        dl_r = min(deadline2, time.time() + max(per_n_budget, 5.0))
        h_r, c5_r = _optimize_from(h_r, dl_r, rng, f"resample_{n_try}")
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 3: Final intensive pair-move polish (10% of budget)
    deadline3 = t0 + budget_s * 0.97
    cur_h = best_h.copy()
    cur_c = best_c5
    n_cur = len(cur_h)
    while time.time() < deadline3:
        i = rng.integers(n_cur)
        j = rng.integers(n_cur)
        if i == j:
            continue
        h2, c52 = _pair_move(cur_h, i, j, ngrid=25)
        if c52 < cur_c - 1e-15:
            cur_h = h2
            cur_c = c52
            if cur_c < best_c5:
                best_c5 = cur_c
                best_h = cur_h.copy()

    # Final: try a few more different n near the best
    if time.time() < t0 + budget_s * 0.99:
        for dn in [-4, 4, -8, 8]:
            n_try = best_n + dn
            if n_try < 32 or n_try > 200:
                continue
            h_r = _resample(best_h, n_try)
            # Quick pair optimization
            n_quick = 500
            c5_r = _c5(h_r)
            for _ in range(n_quick):
                if time.time() > t0 + budget_s * 0.99:
                    break
                i = rng.integers(n_try)
                j = rng.integers(n_try)
                if i == j:
                    continue
                h2, c52 = _pair_move(h_r, i, j, ngrid=15)
                if c52 < c5_r - 1e-15:
                    h_r = h2
                    c5_r = c52
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Final safety
    best_h = _Proj(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = _c5(best_h)

    return best_h, best_c5, best_n
