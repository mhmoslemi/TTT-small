def _pair_golden(h, i, j, deadline):
    """Golden section search for optimal transfer from j to i."""
    n = len(h)
    lo = max(-h[j], 0 - 1.0 + h[i])  # h[j]-t >= 0 and h[i]+t <= 1
    hi = min(1.0 - h[i], h[j])

        return 0, _exact_c5(h)

    a = lo
    b = hi

        h_new = h.copy()
        h_new[i] += t
        h_new[j] -= t
        return _exact_c5(h_new)

            d = c

            fc = fd

def _ls_grad(h, beta, max_active=None):

    dx = 2.0 / n
    sc = _scores(h)
    L = len(sc)
    mx = np.max(sc)
    sh = np.clip(beta * (sc - mx), -700, 700)

    w = ex / (np.sum(ex) + 1e-300)
    if max_active is not None and max_active < L:

        w = w2 / (np.sum(w2) + 1e-300)

                break

            grad -= np.mean(grad)
            gn = float(np.max(np.abs(grad)))
            if gn < 1e-14:
                break
            d = -grad / gn
            best_a = 0.0
            best_tc = cur_c5
            for a in [lr, lr*0.6, lr*0.35, lr*0.15, lr*0.05]:
                trial = _project_box_sum(h + a * d, n / 2.0)

            if best_a > 0:

def _pair_eval(h, i, j, t):
    n = len(h)
    hi = h[i] + t
    hj = h[j] - t
    if hi < -1e-12 or hi > 1.0 + 1e-12 or hj < -1e-12 or hj > 1.0 + 1e-12:
        return 1e10
    h2 = h.copy()
    h2[i] = hi
    h2[j] = hj
    return _exact_c5(h2)

def _pair_line_search(h, i, j, deadline):
    lo = -h[j]
    hi = 1.0 - h[i]
    if lo >= hi:
        return 0.0
    n_grid = 13
    grid = np.linspace(lo, hi, n_grid)
    best_val = 1e10
    best_t = 0.0
    for t in grid:
        v = _pair_eval(h, i, j, t)
        if v < best_val:
            best_val = v
            best_t = t
    # Bisection refine around best
    idx_best = np.argmin([_pair_eval(h, i, j, t) for t in grid])
    if idx_best == 0:
        a, b = lo, grid[1]
    elif idx_best == n_grid - 1:
        a, b = grid[-2], hi
    else:
        a, b = grid[idx_best-1], grid[idx_best+1]
    for _ in range(18):
        if time.time() > deadline:
            break
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = _pair_eval(h, i, j, m1)
        f2 = _pair_eval(h, i, j, m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    best_t = (a + b) / 2.0
    bv = _pair_eval(h, i, j, best_t)
    if bv < best_val:
        best_val = bv
    return best_t

    n = len(h)
    best_h = h.copy()

    cur_c5 = best_c5

        if time.time() > deadline:
            break
        n_improve = 0
        grad = _ls_grad(h, 300.0)
        grad -= np.mean(grad)
        pos = np.where(grad > 0.01)[0]
        neg = np.where(grad < -0.01)[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        n_pairs = min(300, len(pos) * len(neg))
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i = pos[rng.integers(len(pos))]
            j = neg[rng.integers(len(neg))]
            if i == j:
                continue
            if grad[i] * grad[j] >= 0:
                continue
            t_best = _pair_line_search(h, i, j, deadline)
            if abs(t_best) < 1e-12:

            c5_try = _pair_eval(h, i, j, t_best)

                h[j] -= t_best

                n_improve += 1

                    best_h = h.copy()
        if n_improve == 0:
