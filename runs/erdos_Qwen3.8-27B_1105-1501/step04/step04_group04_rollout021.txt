def _rank_pairs(h, F, top_k):
    """Rank pairs by potential improvement using sensitivity analysis."""
    n = len(h)
    dx = 2.0 / n
    Fmax = F.max()
    active_thresh = Fmax * 0.98
    active = np.where(F >= active_thresh)[0]
    if len(active) == 0:
        active = np.where(F >= Fmax - 1e-6)[0]
    # Compute sensitivity: for each coord k, how much does F increase if h[k] increases?
    # coeff of h[k] in F_r: from "h" position: dx*(1-h[k-s]), from "1-h" position: -dx*h[k+s]
    # Net effect of increasing h[k] by 1: sum over active r of (contribution)
    sens = np.zeros(n)
    for r in active:
        s = r - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        for k in range(lo, hi):
            # h[k+s] appears as the "h" part: increasing h[k+s] increases F by dx*(1-h[k])
            # h[k] appears in (1-h[k]): increasing h[k] decreases F by dx*h[k+s]
            idx_s = k + s
            if 0 <= idx_s < n:
                sens[k] += dx * (1.0 - h[k])  # from h[k+s] position (k+s = idx)
            sens[k] -= dx * h[idx_s]  # from (1-h[k]) position
    # Pair (i,j): moving mass from i to j (h[i]-=d, h[j]+=d) changes C5 by ~ (sens[j]-sens[i])*d
    # We want sens[i] > sens[j] (decreasing high-sensitivity, increasing low-sensitivity)
    # Score for pair: sens[i] - sens[j] (positive means promising)
    scores = np.zeros((n, n))
    S = sens[:, None]
    scores = S - S.T  # scores[i,j] = sens[i] - sens[j]
    # Only consider interior points
    mask = (h > 0.01) & (h < 0.99)
    scores[~mask, :] = -np.inf
    scores[:, ~mask] = -np.inf
    np.fill_diagonal(scores, -np.inf)
    # Get top pairs
    flat_idx = np.argsort(scores.flatten())[::-1]
    pairs = []
    seen = set()
    for idx in flat_idx[:top_k * 3]:
        i, j = divmod(idx, n)
        key = (min(i,j), max(i,j))
        if key in seen:
            continue
        seen.add(key)
        pairs.append((i, j))
        if len(pairs) >= top_k:

    return pairs

def _active_gradient(h, F, beta=20.0):
    """Compute weighted gradient of active constraints."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    Fmax = F.max()
    # Soft weights
    shifted = F - Fmax
    weights = np.exp(beta * shifted)
    weights /= weights.sum()
    # Gradient of F_r w.r.t. h
    grad = np.zeros(n)
    for r in range(L):
        if weights[r] < 1e-10:
            continue
        s = r - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        w = weights[r] * dx
        for k in range(lo, hi):
            idx_s = k + s
            if 0 <= idx_s < n:
                # dF_r/dh[k+s] = dx*(1-h[k])
                grad[k + s] += w * (1.0 - h[k])
                # dF_r/dh[k] = -dx*h[k+s]
                grad[k] -= w * h[idx_s]
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

            if time.time() > dl2:
                break
            if h[i] <= 1e-8 or h[j] <= 1e-8 or h[i] >= 1-1e-8 or h[j] >= 1-1e-8:
                continue
            bt, bc5 = _ternary_pair(h, i, j, n_iter=40)
            if bt != 0.0 and bc5 < cur_c5 - 1e-13:
                h[i] += bt
                h[j] -= bt
                cur_c5 = bc5
                improved = True
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not improved:
            # Try direction step
            F = _compute_F(h)
            g = _active_gradient(h, F, beta=25.0)
            g = g - g.mean()
            if np.max(np.abs(g)) > 1e-10:
                _, dc5, dh = _direction_line_min(h, g, n_grid=40)
                if dc5 < cur_c5 - 1e-13:
                    h = dh.copy()
                    cur_c5 = dc5
                    improved = True
                    if cur_c5 < best_c5 - 1e-14:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            stall += 1
            # Add random perturbation to escape
            if stall % 10 == 0:
                idx = rng.integers(n)
                if 0.01 < h[idx] < 0.99:
                    h[idx] += rng.uniform(-0.05, 0.05)
                    h = _project_box_sum(h, n/2.0)
                    cur_c5 = _exact_c5(h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    
    # Phase 3: Multi-resolution exploration
    dl3 = t0 + budget_s * 0.70
    n_candidates = [48, 56, 64, 72, 80, 96, 112, 128, 144, 160, 192]
    n_candidates = [x for x in n_candidates if 32 <= x <= 200]
    # Sort by distance from current best_n
    n_candidates.sort(key=lambda x: abs(x - best_n))
    # Always include current best
    if best_n not in n_candidates:
        n_candidates.insert(0, best_n)
    
    time_per_n = budget_s * 0.02
    for n_try in n_candidates:
        if time.time() > dl3:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        # Quick pair descent on this resolution
        dl_r = min(dl3, time.time() + time_per_n)
        h_r, c5_r = _try_pairs_descent(h_r, dl_r, rng, max_rounds=200, pairs_per_round=30)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 4: Direction-based escape attempts
    dl4 = t0 + budget_s * 0.82
    h = best_h.copy()
    n = len(h)
    cur_c5 = _exact_c5(h)
    for attempt in range(100):
        if time.time() > dl4:
            break
        F = _compute_F(h)
        # Try different betas
        beta = 8.0 + 8.0 * attempt
        g = _active_gradient(h, F, beta=beta)
        g = g - g.mean()
        gn = np.max(np.abs(g))
        if gn < 1e-12:
            continue
        # Try multiple step sizes
        for scale in [1.0, 0.5, 0.25, 0.1]:
            d = g / gn * scale
            _, dc5, dh = _direction_line_min(h, d, n_grid=32)
            if dc5 < cur_c5 - 1e-13:
                h = dh.copy()
                cur_c5 = dc5
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
    
    # Phase 5: Final intensive pair descent
    dl5 = t0 + budget_s * 0.94
    h = best_h.copy()
    cur_c5 = _exact_c5(h)
    h, c5 = _try_pairs_descent(h, dl5, rng, max_rounds=2000, pairs_per_round=60)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 6: Ternary polish
    dl6 = t0 + budget_s * 0.97
    h = best_h.copy()
    n = len(h)
    cur_c5 = _exact_c5(h)
    stall = 0
    for rnd in range(500):
        if time.time() > dl6:
            break
        if stall > 20:
            break
        F = _compute_F(h)
        pairs = _rank_pairs(h, F, 25)
        improved = False
        for (i, j) in pairs:
            if h[i] <= 1e-8 or h[j] <= 1e-8 or h[i] >= 1-1e-8 or h[j] >= 1-1e-8:
                continue
            bt, bc5 = _ternary_pair(h, i, j, n_iter=45)
            if bt != 0.0 and bc5 < cur_c5 - 1e-13:
                h[i] += bt
                h[j] -= bt
                cur_c5 = bc5
                improved = True
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not improved:
            stall += 1
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    
    # Final validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        # Use the stored best
        best_c5 = _exact_c5(best_h)
    
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _exact_c5(best_h)
    
    return best_h, best_c5, best_n
