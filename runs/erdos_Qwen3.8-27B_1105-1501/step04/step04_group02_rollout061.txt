Let me now think about the `_resample` function more carefully. The mass-preserving approach:

Given h_old of length n_old on [0, 2], create h_new of length n_new on [0, 2].

Each cell i in old has width dx_old = 2/n_old and height h_old[i].
Each cell j in new has width dx_new = 2/n_new.

The mass in cell i is h_old[i] * dx_old.

    n = len(h)
    scores = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            scores[r] = np.dot(h[d:], 1.0 - h[:m])
        else:
            s = -d
            m = n - s
            scores[r] = np.dot(h[:m], 1.0 - h[s:])
    return scores

def _manual_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, :m] -= h[d:d + m]
            G[r, d:] += 1.0 - h[:m]
        else:
            s = -d
            m = n - s
            G[r, :m] += 1.0 - h[s:s + m]
            G[r, s:] -= h[:m]
    return G

class _Epi:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0

    def fun(self, x):
        return x[self.n]

    def funjac(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def cfun(self, x):
        h = np.clip(x[:self.n], 0.0, 1.0)
        T = x[self.n]
        return T - _manual_scores(h)

    def cjac(self, x):
        n = self.n
        h = np.clip(x[:n], 0.0, 1.0)
        m = 2 * n - 1
        G = _manual_grad(h)
        J = np.empty((m, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    def eqfun(self, x):
        return np.sum(x[:self.n]) - self.target

    def eqjac(self, x):
        J = np.zeros(self.n + 1)
        J[:self.n] = 1.0
        return J

def _slsqp_solve(h0, deadline, maxiter=80, ftol=1e-10):
    n = len(h0)
    if time.time() > deadline or n > 220:
        return None
    target = n / 2.0
    raw = _manual_scores(h0)
    T0 = float(np.max(raw)) + 0.1
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = T0
    epi = _Epi(n)
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'ineq', 'fun': epi.cfun, 'jac': epi.cjac},
        {'type': 'eq', 'fun': epi.eqfun, 'jac': epi.eqjac},
    ]
    try:
        res = minimize(epi.fun, x0, method='SLSQP', jac=epi.funjac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], target)
            return h_new
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    dts = [0.01, 0.02, 0.05, 0.08, 0.12, 0.2, 0.3, 0.5, 0.7, 1.0]
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(7919 * (p + 1) + 13)
        if n <= 70:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
        else:
            seen = set()
            pairs = []
            for _ in range(min(n * 4, 600)):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    key = (min(i, j), max(i, j))
                    if key not in seen:
                        seen.add(key)
                        pairs.append(key)
            rng.shuffle(pairs)
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            hi, hj = h[i], h[j]
            if hi < 1e-6 or hi > 1 - 1e-6 or hj < 1e-6 or hj > 1 - 1e-6:
                continue
            amax1 = min(1.0 - hi, hj)
            if amax1 < 1e-6:
                continue
            for dt in dts:
                a = dt * amax1
                if a < 1e-10:
                    continue
                h_t = h.copy()
                h_t[i] = hi + a
                h_t[j] = hj - a
                c5_t = _exact_c5(h_t)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    h = h_t.copy()
                    improved = True
                    break
            if not improved:
                amax2 = min(hi, 1.0 - hj)
                if amax2 < 1e-6:
                    continue
                for dt in dts:
                    a = dt * amax2
                    if a < 1e-10:
                        continue
                    h_t = h.copy()
                    h_t[i] = hi - a
                    h_t[j] = hj + a
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _gradient_step(h, deadline, lr=0.05, n_steps=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        tau = 20.0
        shifted = tau * (scores - max_s)
        w = np.exp(shifted)
        w /= float(np.sum(w))
        G = _manual_grad(h) * dx
        grad = G.T @ w
        h_new = h - lr * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
    return h, cur_c5

def _generate_transforms(h, n, seed=42):
    rng = np.random.default_rng(seed)
    candidates = [h.copy()]
    # Reversed
    candidates.append(h[::-1].copy())
    # Relaxed
    for q in [0.3, 0.5, 0.7]:
        h_r = h - q * (h - 0.5)
        candidates.append(_project(h_r, n / 2.0))
    # Sharpened
    for q in [0.3, 0.5, 0.8]:
        h_s = h.copy()
        mask_hi = h >= 0.5
        h_s[mask_hi] = h[mask_hi] + q * (1.0 - h[mask_hi]) * 0.4
        h_s[~mask_hi] = h[~mask_hi] - q * h[~mask_hi] * 0.4
        candidates.append(_project(h_s, n / 2.0))
    # Binarized
    sorted_idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    k = int(n // 2)
    h_b[sorted_idx[:k]] = 1.0
    candidates.append(_project(h_b, n / 2.0))
    # Small rolls
    for roll in [1, 2, 3, n // 4]:
        candidates.append(np.roll(h, roll).copy())
    # Random perturbations
    for scale in [0.02, 0.05, 0.1]:
        h_p = h + rng.normal(0, scale, n)
        candidates.append(_project(h_p, n / 2.0))
    # Smoothed versions
    from scipy.ndimage import gaussian_filter1d
    for sigma in [1.0, 2.0, 3.0]:
        h_smooth = gaussian_filter1d(h, sigma, mode='wrap')
        candidates.append(_project(h_smooth, n / 2.0))
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10.0

    # Initialize
    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float).copy()
        except NameError:
            n0 = 90
            x = (np.arange(n0) + 0.5) / n0 * 2.0
            h0 = np.clip(0.5 + 0.4 * np.sin(np.pi * x), 0, 1)
            h0 = _project(h0, n0 / 2.0)
    else:
        n0 = 90
        x = (np.arange(n0) + 0.5) / n0 * 2.0
        h0 = np.clip(0.5 + 0.4 * np.sin(np.pi * x), 0, 1)
        h0 = _project(h0, n0 / 2.0)

    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Phase 1: SLSQP on parent (30% budget)
    d1 = t0 + budget_s * 0.30
    h_slsqp = _slsqp_solve(best_h, d1, maxiter=100, ftol=1e-10)
    if h_slsqp is not None:
        c5 = _exact_c5(h_slsqp)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_slsqp.copy()
    # Second SLSQP from the result
    if time.time() < d1:
        h_slsqp2 = _slsqp_solve(best_h, d1, maxiter=80, ftol=1e-11)
        if h_slsqp2 is not None:
            c5 = _exact_c5(h_slsqp2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_slsqp2.copy()

    # Phase 2: Diverse transforms + SLSQP (25% budget)
    d2 = t0 + budget_s * 0.55
    transforms = _generate_transforms(best_h, best_n, seed=seed)
    # Sort by initial C5
    scored = []
    for t in transforms:
        c = _exact_c5(t)
        scored.append((c, t))
    scored.sort(key=lambda x: x[0])
    for c_init, h_t in scored:
        if time.time() > d2:
            break
        if c_init >= best_c5 + 0.01:
            continue
        cands = [h_t]
        if c_init < best_c5:
            best_c5 = c_init
            best_h = h_t.copy()
        for hc in cands:
            if time.time() > d2:
                break
            h_s = _slsqp_solve(hc, d2, maxiter=60, ftol=1e-10)
            if h_s is not None:
                c5 = _exact_c5(h_s)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_s.copy()

    # Phase 3: Multi-resolution (25% budget)
    d3 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    res_list = sorted(set([n_curr - 6, n_curr - 4, n_curr - 2, n_curr,
                            n_curr + 2, n_curr + 4, n_curr + 6,
                            n_curr - 10, n_curr + 10,
                            60, 72, 80, 84, 96, 108, 120]))
    res_list = [nn for nn in res_list if 40 <= nn <= 180]
    time_per_res = (d3 - t0) / max(1, len(res_list))
    for n_try in res_list:
        if time.time() > d3:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        c5_r_init = _exact_c5(h_r)
        if c5_r_init < best_c5:
            best_c5 = c5_r_init
            best_h = h_r.copy()
            best_n = n_try
        sub_dl = min(d3, time.time() + time_per_res)
        h_s = _slsqp_solve(h_r, sub_dl, maxiter=60, ftol=1e-10)
        if h_s is not None:
            c5_s = _exact_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n_try
        # Second pass at best resolution
        if time.time() < d3 and (d3 - time.time()) > 5:
            h_s2 = _slsqp_solve(best_h, d3, maxiter=50, ftol=1e-11)
            if h_s2 is not None and len(h_s2) == best_n:
                c5_s2 = _exact_c5(h_s2)
                if c5_s2 < best_c5:
                    best_c5 = c5_s2
                    best_h = h_s2.copy()

    # Phase 4: Intensive SLSQP at best resolution (15% budget)
    d4 = t0 + budget_s * 0.95
    n_best = len(best_h)
    for iter_i in range(5):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + 20)
        h_s = _slsqp_solve(best_h, sub_dl, maxiter=100, ftol=1e-12)
        if h_s is not None:
            c5 = _exact_c5(h_s)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_s.copy()
            else:
                break

    # Phase 5: Pair refinement (last 3%)
    d5 = t0 + budget_s * 0.98
    best_h, best_c5 = _pair_refine(best_h, d5, max_passes=50)

    # Final SLSQP polish
    if time.time() < t0 + budget_s * 0.99:
        h_s = _slsqp_solve(best_h, t0 + budget_s * 0.99, maxiter=40, ftol=1e-12)
        if h_s is not None:
            c5 = _exact_c5(h_s)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_s.copy()

    # Final projection and validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    # Safety: ensure we don't return worse than initial
    if final_c5 > _exact_c5(h0):
        best_h = h0.copy()
        final_c5 = _exact_c5(h0)
    return best_h, final_c5, len(best_h)
