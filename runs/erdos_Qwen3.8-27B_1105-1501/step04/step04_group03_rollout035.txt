    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_gradient(h, k_idx):
    """Build gradient of correlation at lag k_idx with respect to h."""
    n = len(h)
    d = k_idx - (n - 1)  # actual shift
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d+m])  # d(h[i](1-h[i+d])/dh[i]) = 1-h[i+d]
            g[d:d+m] += -h[:m]  # d(h[i](1-h[i+d])/dh[i+d]) = -h[i]
    else:
        g_idx = -d
        m = n - g_idx
        if m > 0:
            g[g_idx:] += (1.0 - h[:m])  # for shift -g_idx: h[i](1-h[i-g_idx])
            g[:m] += -h[g_idx:]  # dh[i-g_idx] term
    return g

def build_gradient_matrix(h):
    """Build full gradient matrix G where G[k, j] = d corr[k] / d h[j]."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        G[k] = build_gradient(h, k)
    return G

def active_set_lp(h, scores, G, radius, n_active_frac=0.15, target_reduction=0.0):
    """LP that focuses on active set of correlations."""
    n = len(h)
    max_score = np.max(scores)
    
    # Select active set: correlations within threshold of max
    threshold = max_score - (max_score * n_active_frac)
    active = scores > threshold
    n_active = np.sum(active)
    
    if n_active < 3:
        active = scores > max_score - max_score * 0.5
    
    # Target: reduce the max by target_reduction
    target = max_score - target_reduction
    
    # LP: minimize t
    # s.t. G_active @ p ≤ t - scores_active  (active correlations should be ≤ target)
    #      G_inactive @ p ≤ slack  (inactive ones just shouldn't blow up)
    #      sum(p) = 0
    #      |p| ≤ radius
    #      t ≥ 0 (or t can be negative, meaning we're reducing the max)
    
    # Actually, let's use: minimize sum(p) ... no.
    # Let's use: minimize t where:
    #   G_active[k] @ p + scores_active[k] ≤ target + t_for_k
    #   Simplification: just require G_active[k] @ p ≤ target - scores_active[k]
    #   And for inactive: G_inactive[k] @ p ≤ max_score * 0.3 - scores_inactive[k] (generous)
    
    n_constraints = 2*n - 1
    # Variables: p[0..n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((n_constraints, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    
    # For active: scores + G.p ≤ target + t → G.p - t ≤ target - scores
    # For inactive: scores + G.p ≤ (max_score + 0.1) + t → G.p - t ≤ (max_score + 0.1) - scores
    b_ub = np.zeros(n_constraints)
    b_ub[active] = target - scores[active]
    b_ub[~active] = (max_score + 0.05) - scores[~active]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def greedy_direction(h, scores, G, radius, top_k=5):
    """Find a descent direction using top-k active correlations."""
    n = len(h)
    max_score = np.max(scores)
    
    # Get top k active correlations
    top_indices = np.argsort(scores)[-top_k:]
    
    # Weighted combination of their gradients
    weights = scores[top_indices] - max_score + 1  # positive weights, larger for more active
    weights = weights / np.sum(weights)
    
    # Direction: negative weighted sum of gradients
    p = -np.sum([weights[i] * G[top_indices[i]] for i in range(top_k)], axis=0)
    
    # Center to satisfy sum(p) = 0
    p -= np.mean(p)
    
    # Clip to box
    for j in range(n):
        max_up = min(radius, 1.0 - h[j])
        max_down = max(-radius, -h[j])
        p[j] = np.clip(p[j], max_down, max_up)
    
    # Re-center (clipping might have broken sum=0)
    p -= np.sum(p) / n
    
    return p

def line_search_c5(h, p, c5_cur):
    """Exact line search for C₅ along direction p."""
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    best_a = 0.0
    best_c5 = c5_cur
    
    # Binary search / golden section
    lo_a, hi_a = 0.0, alpha_max
    c1 = lo_a + 0.382 * (hi_a - lo_a)
    c2 = lo_a + 0.618 * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    
    for _ in range(25):
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = lo_a + 0.382 * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + 0.618 * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
        if hi_a - lo_a < 1e-12:
            break
    
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return 0.0, c5_cur

def binary_round(h):
    """Round h to binary while preserving sum."""
    n = len(h)
    target = int(n / 2)
    # Sort by value, keep top 'target' as 1
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_swap_search(h_bin, max_iters=5000, deadline=None):
    """Local search on binary solution using swaps."""
    n = len(h_bin)
    target = n // 2
    h_cur = h_bin.copy()
    cur_c5 = exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    
    ones = np.where(h_cur == 1.0)[0]
    zeros = np.where(h_cur == 0.0)[0]
    
    iter_count = 0
    while iter_count < max_iters:
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Try random swaps
        for _ in range(50):
            if deadline is not None and time.time() > deadline:
                break
            i = np.random.randint(len(ones))
            j = np.random.randint(len(zeros))
            a, b = ones[i], zeros[j]
            if a == b:
                continue
            h_t = h_cur.copy()
            h_t[a] = 0.0
            h_t[b] = 1.0
            c5_t = exact_c5(h_t)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h_t.copy()
                h_cur = h_t.copy()
                cur_c5 = c5_t
                improved = True
                ones = np.where(h_cur == 1.0)[0]
                zeros = np.where(h_cur == 0.0)[0]
        iter_count += 1
        if not improved and iter_count > 100:
            break
    
    return best_h, best_c5

def continuous_refine_from_binary(h_bin, max_iters=200):
    """Refine a binary solution in continuous space."""
    n = len(h_bin)
    h = h_bin + 0.01 * (np.random.rand(n) - 0.5)
    h = project(h, n / 2.0)
    cur_c5 = exact_c5(h)
    
    scores = all_corrs(h)
    G = build_gradient_matrix(h)
    
    for it in range(max_iters):
        p, _ = active_set_lp(h, scores, G, radius=0.05, n_active_frac=0.2)
        if p is None or np.max(np.abs(p)) < 1e-13:
            # Try greedy
            p = greedy_direction(h, scores, G, radius=0.05, top_k=5)
            if np.max(np.abs(p)) < 1e-13:
                break
        a, new_c5 = line_search_c5(h, p, cur_c5)

            h = project(h, n / 2.0)
            cur_c5 = exact_c5(h)
            scores = all_corrs(h)
            G = build_gradient_matrix(h)
        else:
            break
    
    return h, cur_c5

def multi_start_lp(h, deadline, n_restarts=5, radius_init=0.1):
    """LP-based optimization with random restarts."""
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)

        h_cur = project(h_cur, n / 2.0)
        h_cur, c5_cur = lp_optimize_single(h_cur, sub_deadline, radius_init)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    
    return best_h, best_c5

def lp_optimize_single(h, deadline, radius_init=0.1):
    """Single LP optimization loop."""
    n = len(h)

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        j_range = np.arange(i_min, i_max + 1)
        G[k, j_range] = 1.0 - 2.0 * h[k - j_range]
    return G

def active_lp(h, scores, G, radius, frac=0.12):
    n = len(h)
    max_s = float(np.max(scores))
    thresh = max_s - max_s * frac
    active = scores > thresh
    n_c = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_c, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.empty(n_c)
    b_ub[active] = -scores[active]
    b_ub[~active] = -(scores[~active] + max_s * 0.08)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def greedy_dir(h, scores, G, radius, topk=5):
    n = len(h)
    max_s = np.max(scores)
    idx = np.argsort(scores)[-topk:]
    w = scores[idx] - max_s + 0.5
    w = np.maximum(w, 0.01)
    w /= np.sum(w)
    p = -np.zeros(n)
    for i in range(topk):
        p -= w[i] * G[idx[i]]
    p -= np.mean(p)
    for j in range(n):
        p[j] = np.clip(p[j], max(-radius, -h[j]), min(radius, 1.0 - h[j]))
    p -= np.sum(p) / n
    return p

def line_search(h, p, c5_cur):
    n = len(h)
    amax = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            amax = min(amax, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            amax = min(amax, h[j] / (-p[j]))
    amax = max(amax, 1e-10)
    if amax < 1e-12:
        return 0.0, c5_cur
    lo, hi = 0.0, amax
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * p)
    opt = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt * p)
    if opt_c5 < c5_cur:
        return opt, opt_c5
    return 0.0, c5_cur

def lp_loop(h, deadline, radius_init=0.1, frac=0.12):
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        scores = all_corrs(h)
        G = build_grad_matrix(h)
        p, t_val = active_lp(h, scores, G, radius, frac)
        if p is None or np.max(np.abs(p)) < 1e-13:
            p = greedy_dir(h, scores, G, radius, topk=5)
            if np.max(np.abs(p)) < 1e-13:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
        a, new_c5 = line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + a * p
            h = project(h, n / 2.0)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall > 60:
                break
            if stall % 25 == 0:
                radius = max(radius, 0.04)
    return best_h, best_c5

def binary_round(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def anneal_binary(h_bin, deadline, rng):
    n = len(h_bin)
    h_cur = h_bin.copy()
    cur_c5 = exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    ones = np.where(h_cur == 1.0)[0]
    zeros = np.where(h_cur == 0.0)[0]
    temp = 0.02
    cool = 0.9995
    while time.time() < deadline and temp > 1e-5:
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        a, b = ones[i], zeros[j]
        if a == b:
            continue
        h_t = h_cur.copy()
        h_t[a] = 0.0
        h_t[b] = 1.0
        c5_t = exact_c5(h_t)
        delta = c5_t - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            h_cur = h_t
            cur_c5 = c5_t
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
        temp *= cool
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)
