def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def get_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx, corr  # scaled, raw

def correlation_gradient(h):
    """Compute gradient of each correlation value w.r.t. h."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def focal_lp(h, scores, top_k, radius):
    """LP that only considers the top_k correlations."""
    n = len(h)
    top_idx = np.argsort(scores)[-top_k:]
    G = correlation_gradient(h)
    G_active = G[top_idx]
    s_active = scores[top_idx]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = -s_active
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(20):
        a = alpha_max * (i + 1) / 20.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2

            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    scores_raw = np.correlate(h, omh, mode='full')
    scores = scores_raw * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(np.clip((scores - max_s) / tau, -50, 50))
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wk * (1.0 - h[g:])
                grad[g:] -= wk * h[:m]
    return val, dx * grad

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, _ = soft_value_grad(x, tau)
        return v

    def grad(x):
        _, g = soft_value_grad(x, tau)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=20):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:

                        if c5_t < best_c5 - 1e-12:

                            best_h = h_t.copy()

            if improved:

        if not improved:

import numpy as np

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def get_scores(h):

def correlation_gradient(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def line_search(h, p, c5_cur):

            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = get_scores(h)
    max_s = float(np.max(scores))
    exp_s = np.exp(np.clip((scores - max_s) / tau, -50, 50))
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wk * (1.0 - h[g:])
                grad[g:] -= wk * h[:m]
    return val, dx * grad

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, _ = soft_value_grad(x, tau)
        return v

    def grad(x):
        _, g = soft_value_grad(x, tau)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=25):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        a = amax * frac
                        h_t = best_h.copy()
                        h_t[j] += sign * a
                        h_t[k] -= sign * a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

                    sol = res.x
            except Exception:
                pass
            if feasible:
                hi_t = mid_t
            else:
                lo_t = mid_t
        # If we found a feasible direction at hi_t, use it
        top_idx = np.argsort(scores)[-top_k:]
        G = correlation_gradient(h)
        G_active = G[top_idx]
        s_active = scores[top_idx]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub_full = np.zeros((top_k, n + 1))
        A_ub_full[:, :n] = G_active
        A_ub_full[:, n] = -1.0
        b_ub_full = -s_active
        A_eq_f = np.zeros((1, n + 1))
        A_eq_f[0, :n] = 1.0
        bounds2 = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds2.append((None, None))
        p = None
        try:
            res = linprog(c_vec, A_ub=A_ub_full, b_ub=b_ub_full, A_eq=A_eq_f, b_eq=np.zeros(1),
                          bounds=bounds2, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = line_search(h, p, exact_c5(h))
        if new_c5 < exact_c5(h) - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Focal LP - growing top-k (fast iterations, many of them)
    d1 = t0 + budget_s * 0.22
    for k in [3, 5, 8, 12]:
        sub_dl = min(d1, time.time() + budget_s * 0.05)
        h1, c5_1 = focal_lp_loop(h, sub_dl, top_k=k, radius_init=0.15)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Full LP for global direction (escape focal-LP traps)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = full_lp_loop(best_h, d2, radius_init=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Subgradient refinement
    d3 = t0 + budget_s * 0.40
    for k in [3, 5, 7]:
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h3, c5_3 = subgradient_method(best_h, sub_dl, top_k=k, lr_init=0.025)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: SLSQP smooth optimization
    d4 = t0 + budget_s * 0.52
    for tau in [0.3, 0.12, 0.05, 0.02]:
        if time.time() > d4:
            break
        h4 = slsqp_smooth(best_h, tau, 70)
        c5_4 = exact_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Focal LP again with tighter radius
    d5 = t0 + budget_s * 0.62
    for k in [4, 6, 10]:
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h5, c5_5 = focal_lp_loop(best_h, sub_dl, top_k=k, radius_init=0.08)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Multi-resolution search
    d6 = t0 + budget_s * 0.78
    for dn in [-10, -5, 5, 10, 15, -15, 20, -20, 8, -8]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 160:
            continue
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = focal_lp_loop(h_r, sub_dl, top_k=5, radius_init=0.12)
        if time.time() < d6 and c5_r < best_c5 - 1e-13:
            h_r = slsqp_smooth(h_r, 0.04, 40)
            c5_r = exact_c5(h_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Diversified restarts
    d7 = t0 + budget_s * 0.87
    for scale in [0.02, 0.05, 0.08, 0.12, 0.18]:
        if time.time() > d7:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.008)
        h_p, c5_p = focal_lp_loop(h_pert, sub_dl, top_k=5, radius_init=0.1)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 8: Pair polish
    d8 = t0 + budget_s * 0.93
    h_pol, c5_pol = pair_polish(best_h, d8, max_passes=30)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 9: Final micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = focal_lp_loop(best_h, hard_deadline, top_k=6, radius_init=0.03)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
