    h_new = np.interp(x_new, x_old, h)

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 32)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k_offset in range(step, n, max(1, n // 16)):
                k = (j + k_offset) % n
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    # Golden section search for best transfer amount
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(10):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time


def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _exact_c5(h):
    return float(np.max(_scores(h)))


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    P = 2 * n - 1
    M = np.max(scores)
    u = np.exp((scores - M) / tau)
    Z = np.sum(u)
    if Z < 1e-300:
        w = np.zeros(P)
        w[int(np.argmax(scores))] = 1.0
    else:
        w = u / Z
    grad = np.zeros(n)
    for r in range(P):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        wr_dx = w[r] * dx
        if d == 0:
            grad += wr_dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            if m > 0:
                grad[:m] -= wr_dx * h[d:]
                grad[d:] += wr_dx * (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wr_dx * (1.0 - h[g:])
                grad[g:] -= wr_dx * h[:m]
    return grad, scores


def _lp_direction(h, scores, radius, top_k=None):
    n = len(h)
    dx = 2.0 / n
    P = 2 * n - 1
    if top_k is not None and top_k < P:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(P)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((len(idx), n + 1))
    for j, r in enumerate(idx):
        d = r - (n - 1)
        if d == 0:
            A_ub[j, :n] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            if m > 0:
                A_ub[j, :m] -= dx * h[d:]
                A_ub[j, d:] += dx * (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                A_ub[j, :m] += dx * (1.0 - h[g:])
                A_ub[j, g:] -= dx * h[:m]
        A_ub[j, n] = 1.0
    b_ub = np.zeros(len(idx))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_b = max(-radius, -h[i] - 1e-12)
        hi_b = min(radius, 1.0 - h[i] + 1e-12)
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-12:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, 0.0


def _line_search_c5(h, p, alpha_max=None):

    n_samp = 13
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    if best_i == 0:
        return 0.0, vals[0]
    if best_i == n_samp - 1:
        return alphas[-1], vals[-1]
    lo = alphas[max(0, best_i - 1)]

    f1 = _exact_c5(h + c1 * p)

    for _ in range(20):

            c2 = c1

            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)

        tau_list = [0.05, 0.025, 0.0125, 0.006, 0.003, 0.0015, 0.0008]
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    trust = 0.12
    momentum = np.zeros(n)
    for tau in tau_list:

            break
        n_iter = 0
        stall = 0

            if mn < 1e-14:
                break
            momentum = momentum / mn
            h_try = _project(h + trust * momentum, target)
            c5_try = _exact_c5(h_try)
            accepted = False
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                accepted = True
            else:
                for frac in [0.6, 0.4, 0.25]:
                    h_try2 = _project(h + trust * frac * momentum, target)
                    c5_try2 = _exact_c5(h_try2)
                    if c5_try2 < cur_c5 - 1e-13:
                        h = h_try2
                        cur_c5 = c5_try2
                        accepted = True
                        break
            if accepted:
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                trust = min(trust * 1.04, 0.35)
            else:
                stall += 1
                trust *= 0.75
                momentum *= 0.5
                if trust < 1e-5:
                    break
            n_iter += 1
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 24)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k_off in range(step, n, max(1, n // 14)):
                k = (j + k_off) % n
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(9):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def _alt_optimize(h, deadline, target, n_rounds=50, tau_start=0.02, tau_end=0.001,
                  lp_top_k_list=None, radius_init=0.1):
    """Alternating smooth max and LP optimization."""
    if lp_top_k_list is None:
        lp_top_k_list = [15, 8, None]
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    momentum = np.zeros(n)
    n_tl = len(lp_top_k_list)
    tk_idx = 0

    for r in range(n_rounds):
        if time.time() > deadline:
            break
        frac = r / max(1, n_rounds - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        # Smooth max gradient step
        grad, _ = _smooth_grad(h, tau)
        grad_proj = grad - np.mean(grad)
        gn = np.linalg.norm(grad_proj)
        if gn > 1e-14:
            direction = -grad_proj / gn
            momentum = 0.6 * momentum + 0.4 * direction
            momentum -= np.mean(momentum)
            mn = np.max(np.abs(momentum))
            if mn > 1e-14:
                momentum /= mn
                h_try = _project(h + 0.1 * momentum, target)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-13:
                    h = h_try
                    cur_c5 = c5_try
                elif c5_try < cur_c5:
                    h = h_try
                    cur_c5 = c5_try

        # LP step
        scores = _scores(h)
        p, t_val = _lp_direction(h, scores, radius, top_k=lp_top_k_list[tk_idx % n_tl])
        if p is not None and float(np.max(np.abs(p))) > 1e-13:
            p -= p.mean()
            alpha, new_c5 = _line_search_c5(h, p)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = min(radius * 1.05, 0.4)
            else:
                radius *= 0.8
                tk_idx += 1
        else:
            radius *= 0.8
            tk_idx += 1

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    return best_h, best_c5


def _quasi_periodic_init(n, alpha, noise=0.05, rng=None):
    x = np.arange(n) / n
    h = (((x * alpha) % 1.0) < 0.5).astype(float)
    if rng is not None and noise > 0:
        h += rng.normal(0, noise, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Smooth max continuation on parent (0-20%)
    dl1 = t0 + budget_s * 0.20
    tau_list = [0.06, 0.035, 0.02, 0.012, 0.007, 0.004, 0.002, 0.001, 0.0006]
    h1, c5_1 = _smooth_anneal(h_start, dl1, n0 / 2.0, tau_list, rng, max_iters_per_tau=400)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refinement (20-35%)
    if time.time() < t0 + budget_s * 0.35:
        dl2 = t0 + budget_s * 0.35
        h2, c5_2 = _lp_optimize(best_h, dl2, top_k_list=[None, 18, 10, 5, 3],
                                radius_init=0.15, max_iters=4000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Multi-resolution alternating optimization (35-60%)
    dl3 = t0 + budget_s * 0.60
    resolutions = [128, 96, 112, 144, 96, 128, 112, 80]
    alphas_qp = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if n_try <= 0:
            continue
        if ri % 3 == 0:
            h_init = _resample(best_h, n_try)
        elif ri % 3 == 1:
            h_init = _quasi_periodic_init(n_try, alphas_qp[ri % len(alphas_qp)],
                                          noise=0.06, rng=rng)
        else:
            h_init = _resample(best_h, n_try)
            h_init += rng.normal(0, 0.03, n_try)
            h_init = _project(h_init, n_try / 2.0)

        sub_dl = min(dl3, time.time() + budget_s * 0.028)
        h_try, c5_try = _alt_optimize(h_init, sub_dl, n_try / 2.0,
                                      n_rounds=60, tau_start=0.03, tau_end=0.0008,
                                      radius_init=0.12)
        # LP polish
        if time.time() < dl3:
            sub_dl2 = min(dl3, time.time() + budget_s * 0.012)
            h_try, c5_try = _lp_optimize(h_try, sub_dl2,
                                         top_k_list=[10, 5, None, 3],
                                         radius_init=0.06, max_iters=800)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 4: Fine alternating on best + LP (60-78%)
    if time.time() < t0 + budget_s * 0.78:
        dl4 = t0 + budget_s * 0.78
        h4, c5_4 = _alt_optimize(best_h, dl4, best_n / 2.0,
                                 n_rounds=80, tau_start=0.015, tau_end=0.0004,
                                 lp_top_k_list=[10, 6, None, 4, 3],
                                 radius_init=0.06)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.02)
            h_lp, c5_lp = _lp_optimize(best_h, sub_dl,
                                       top_k_list=[8, 5, 3, None],
                                       radius_init=0.04, max_iters=1500)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()

    # Phase 5: Perturbation restarts (78-90%)
    dl5 = t0 + budget_s * 0.90
    scale = 0.025
    for attempt in range(80):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.01)
        h_try, c5_try = _alt_optimize(h_pert, sub_dl, best_n / 2.0,
                                      n_rounds=25, tau_start=0.01, tau_end=0.0005,
                                      radius_init=0.05)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.88)
        else:
            scale = min(0.08, scale * 1.15)

    # Phase 6: Pair polish (90-96%)
    if time.time() < t0 + budget_s * 0.96:
        dl6 = t0 + budget_s * 0.96
        hp, c5p = _pair_polish(best_h, dl6, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 7: Final LP refinement (96-99.5%)
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _lp_optimize(best_h, t0 + budget_s * 0.998,
                                     top_k_list=[4, 3, 5, None],
                                     radius_init=0.015, max_iters=600)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation and return
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
