    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _build_grad_block(h, r, n):
    """Build gradient row for shift index r."""
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d + m])
            g[d:] += -h[:m]
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:gg + m])
            g[gg:] += -h[:m]
    return g

def _reduced_lp(h, scores, m, radius):
    """Find LP direction using only top m active shifts."""
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:m]
    A = np.zeros((m, n))
    for ii, r in enumerate(top_idx):
        A[ii] = _build_grad_block(h, r, n)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx]
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b >= hi_b:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_fast(h, deadline, radius_init=0.15, m_init=12):
    """Fast LP loop with reduced active set."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    m = m_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = _scores(h)
        p = _reduced_lp(h, scores, m, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.65
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.12, 0.45)
            m = min(m + 1, 2 * n - 1)
        else:
            stall += 1
            radius *= 0.7
            m = max(m_init, m - 1)
            if stall % 25 == 0 and radius < 0.03:
                radius = 0.08
    return best_h, best_c5

def _softmax_dir(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = corr * dx
    mx = float(np.max(sc))
    ew = np.exp(tau * (sc - mx))
    w = ew / float(np.sum(ew))
    grad = np.zeros(n)
    thresh = 1e-12
    for r in range(2 * n - 1):
        if w[r] < thresh:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= grad.mean()
    return grad

def _grad_polish(h, deadline, tau_seq=(30, 80, 200, 500), lr0=0.03):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr0
    mom = np.zeros(n)
    stall = 0
    ti = 0
    while time.time() < deadline:
        if stall > 50:
            break
        tau = tau_seq[ti % len(tau_seq)]
        d = _softmax_dir(h, dx, tau)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            ti += 1
            continue
        d = d / dn
        mom = 0.82 * mom + 0.18 * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            ti += 1
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.8)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.8
            if stall % 20 == 0:
                ti += 1
                mom = np.zeros(n)
    return best_h, best_c5

def _diversify(h, deadline, rng, m_lp=10):
    """Try random perturbations and optimize each."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.05
    for _ in range(200):
        if time.time() > deadline:
            break
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, target)
        sub_dl = min(deadline, time.time() + max(2.0, (deadline - time.time()) * 0.15))
        h_t, c5_t = _lp_fast(h_p, sub_dl, radius_init=0.1, m_init=m_lp)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.01, scale * 0.8)
        else:
            scale = min(0.18, scale * 1.15)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine_fast(h, deadline, max_passes=20):
    """Only refine pairs where both values are strictly interior."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) < 2:
            break
        improved = False
        for ii in range(len(interior)):
            for jj in range(ii + 1, len(interior)):
                if time.time() > deadline:
                    break
                i, j = interior[ii], interior[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
                        improved = True
                        break
            if time.time() > deadline:
                break
        if not improved:
            break
    return best_h, best_c5

def _nelder_mead_1d(h, deadline, rng):
    """Optimize a single parameter: shift mass from random block A to block B."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(3000):
        if time.time() > deadline:
            break
        # Pick two random contiguous blocks
        size = rng.integers(2, max(3, n // 8))
        i1 = rng.integers(0, n - size)
        i2 = rng.integers(0, n - size)
        if abs(i1 - i2) < size:
            continue
        d = np.zeros(n)
        d[i1:i1 + size] = -1.0
        d[i2:i2 + size] = 1.0
        alpha, c5_try = _line_search(h, d, dx, 0.4)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick gradient polish (4% = ~44s)
    d1 = t0 + budget_s * 0.04
    h1, c5_1 = _grad_polish(h, d1, tau_seq=(40, 100, 300), lr0=0.025)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Fast LP with reduced active set (30% = ~330s)
    d2 = t0 + budget_s * 0.34
    h2, c5_2 = _lp_fast(best_h, d2, radius_init=0.18, m_init=10)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Diversification (14% = ~154s)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _diversify(best_h, d3, rng, m_lp=10)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution exploration (20% = ~220s)
    d4 = t0 + budget_s * 0.68
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16,
               -20, 20, -24, 24, -30, 30, -36, 36, -48, 48]
    tried_n = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_h.shape[0] + dn
        if n_try < 20 or n_try > 250:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        rem_total = d4 - time.time()
        if rem_total < 5:
            break
        sub_dl = min(d4, time.time() + max(3.0, rem_total * 0.18))
        h_r, c5_r = _lp_fast(h_r, sub_dl, radius_init=0.12, m_init=10)
        rem = d4 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d4, time.time() + max(2.0, rem * 0.25))
            h_r, c5_r = _grad_polish(h_r, sub_dl2, tau_seq=(50, 150, 400), lr0=0.02)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Block Moves (10% = ~110s)
    d5 = t0 + budget_s * 0.78
    h5, c5_5 = _nelder_mead_1d(best_h, d5, rng)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Second diversification round (8% = ~88s)
    d6 = t0 + budget_s * 0.86
    h6, c5_6 = _diversify(best_h, d6, rng, m_lp=8)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Pair refinement (5% = ~55s)
    d7 = t0 + budget_s * 0.91
    h7, c5_7 = _pair_refine_fast(best_h, d7, max_passes=15)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final LP polish (6% = ~66s)
    d8 = t0 + budget_s * 0.97
    h8, c5_8 = _lp_fast(best_h, d8, radius_init=0.06, m_init=15)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Projecting slightly changed values, re-check
        best_c5 = final_c5
    return best_h, best_c5, n_final
