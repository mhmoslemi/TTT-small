```python
def pair_slope(h, i, j):
    """Compute the slope a_r for each lag r when transferring mass from j to i."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    a = np.zeros(L)
    for r in range(L):
        d = r - (n - 1)
        # Contribution from h_i changing by +1
        if 0 <= i + d < n:
            a[r] += (1.0 - h[i + d])
        if 0 <= i - d < n:
            a[r] += -h[i - d]
        # Contribution from h_j changing by -1
        if 0 <= j + d < n:
            a[r] += -(1.0 - h[j + d])
        if 0 <= j - d < n:
            a[r] += h[j - d]
    a *= dx
    return a

def pair_optimal_delta(h, i, j, F):
    """Find optimal delta for transferring mass from j to i."""
    n = len(h)
    # Feasible range
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    if delta_max - delta_min < 1e-12:
        return 0.0
    
    a = pair_slope(h, i, j)
    
    # C(delta) = max_r (F[r] + a[r] * delta)
    # This is convex in delta. Find minimum.
    # Evaluate on a grid
    n_grid = 32
    deltas = np.linspace(delta_min, delta_max, n_grid)
    vals = np.max(F[None, :] + a[None, :] * deltas[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Refine with golden section
    if best_idx == 0:
        return delta_min
    if best_idx == n_grid - 1:
        return delta_max
    
    lo = deltas[max(0, best_idx - 1)]
    hi = deltas[min(n_grid - 1, best_idx + 1)]
    
    def c5_of_delta(d):
        return float(np.max(F + a * d))
    
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d_ = lo + gr * (hi - lo)
    fc = c5_of_delta(c)
    fd = c5_of_delta(d_)
    for _ in range(40):
        if fc < fd:
            hi = d_
            d_ = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = c5_of_delta(c)
        else:
            lo = c
            c = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = c5_of_delta(d_)
    
    return 0.5 * (lo + hi)
```

```python
def subgradient_fast(h, w):

    # If k-d_r is in [0,n): -w[r]*dx*h[k-d_r]

    # More precisely:

    #       - dx * sum_r w[r] * h[k-shifts[r]] for valid k-shifts[r]

        low = max(0, -d)
        high = min(n, n - d)
```

```python
def pair_slope_fast(h, i, j):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    shifts = np.arange(L) - (n - 1)
    
    # For index i
    ip = i + shifts
    im = i - shifts
    first_i = (np.maximum(0, -shifts) <= i) & (i < np.minimum(n, n - shifts))
    second_i = (np.maximum(0, shifts) <= i) & (i < np.minimum(n, n + shifts))
    
    grad_i = np.zeros(L)
    grad_i[first_i] = dx * (1.0 - h[np.clip(ip[first_i], 0, n-1)])
    grad_i[second_i] -= dx * h[np.clip(im[second_i], 0, n-1)]
    
    # For index j
    jp = j + shifts
    jm = j - shifts
    first_j = (np.maximum(0, -shifts) <= j) & (j < np.minimum(n, n - shifts))
    second_j = (np.maximum(0, shifts) <= j) & (j < np.minimum(n, n + shifts))
    
    grad_j = np.zeros(L)
    grad_j[first_j] = dx * (1.0 - h[np.clip(jp[first_j], 0, n-1)])
    grad_j[second_j] -= dx * h[np.clip(jm[second_j], 0, n-1)]
    
    return grad_i - grad_j
```

This is O(L) with vectorized operations. For n=128, L=255, this is very fast.

```python
    # term2[k] = sum_r w[r] * h[k-shifts[r]] * second_valid[k, r]

    # Wait, I'm confusing myself. Let me think again.

    # dF_r/dh[k] = dx*(1-h[k+d_r]) * [k is first-valid for r] - dx*h[k-d_r] * [k is second-valid for r]

    # First-valid for r (shift d): max(0,-d) <= k < min(n, n-d)
    # Equivalently: k >= max(0,-d) and k+d < n
    # So k+d is the index into h.
    
    # Let me use a different approach: iterate over r (only active ones)
    active = np.where(w > 1e-12)[0]

        if abs(wr) < 1e-15:
            continue
        d = shifts[r]
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        g[j] += wr * (1.0 - h[i])
        g[i] += -wr * h[j]
```

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    x = np.clip(v.copy(), 0.0, 1.0)
    for _ in range(200):
        s = float(np.sum(x))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _shifts_vec(n):
    L = 2 * n - 1
    return np.arange(L) - (n - 1)

def _subgrad(h, w):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    shifts = _shifts_vec(n)
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for r in active:
        wr = w[r] * dx
        if abs(wr) < 1e-15:
            continue
        d = shifts[r]
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        g[j] += wr * (1.0 - h[i])
        g[i] += -wr * h[j]
    return g

def _pair_slope(h, i, j):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    shifts = _shifts_vec(n)
    
    ip = i + shifts
    im = i - shifts
    first_i = (np.maximum(0, -shifts) <= i) & (i < np.minimum(n, n - shifts))
    second_i = (np.maximum(0, shifts) <= i) & (i < np.minimum(n, n + shifts))
    
    gi = np.zeros(L)
    gi[first_i] = dx * (1.0 - h[np.clip(ip[first_i], 0, n - 1)])
    gi[second_i] -= dx * h[np.clip(im[second_i], 0, n - 1)]
    
    jp = j + shifts
    jm = j - shifts
    first_j = (np.maximum(0, -shifts) <= j) & (j < np.minimum(n, n - shifts))
    second_j = (np.maximum(0, shifts) <= j) & (j < np.minimum(n, n + shifts))
    
    gj = np.zeros(L)
    gj[first_j] = dx * (1.0 - h[np.clip(jp[first_j], 0, n - 1)])
    gj[second_j] -= dx * h[np.clip(jm[second_j], 0, n - 1)]
    
    return gi - gj

def _pair_optimal(h, i, j, F):
    n = len(h)
    d_min = max(-h[i], h[j] - 1.0)
    d_max = min(1.0 - h[i], h[j])
    if d_max - d_min < 1e-12:
        return 0.0
    a = _pair_slope(h, i, j)
    n_pts = 25
    deltas = np.linspace(d_min, d_max, n_pts)
    vals = np.max(F[None, :] + a[None, :] * deltas[:, None], axis=1)
    bi = int(np.argmin(vals))
    if bi == 0:
        return d_min
    if bi == n_pts - 1:
        return d_max
    lo = deltas[max(0, bi - 1)]
    hi = deltas[min(n_pts - 1, bi + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    dd = lo + gr * (hi - lo)
    fc = float(np.max(F + a * c))
    fd = float(np.max(F + a * dd))
    for _ in range(30):
        if fc < fd:
            hi = dd
            dd = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = float(np.max(F + a * c))
        else:
            lo = c
            c = dd
            fc = fd
            dd = lo + gr * (hi - lo)
            fd = float(np.max(F + a * dd))
    return 0.5 * (lo + hi)

def _subgrad_descent(h, deadline, beta_start=20.0, beta_end=5000.0, n_beta=12):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    momentum = np.zeros(n)
    beta_list = np.geomspace(beta_start, beta_end, n_beta)
    step_size = 0.04
    
    for beta in beta_list:
        if time.time() > deadline:
            break
        stag = 0
        for it in range(10000):
            if time.time() > deadline:
                break
            if stag > 150:
                break
            scorer = _scores(h)
            mx = float(np.max(scorer))
            scaled = beta * (scorer - mx)
            scaled = np.clip(scaled, -500.0, 0.0)
            w = np.exp(scaled)
            ws = float(np.sum(w))
            if ws < 1e-300:
                w = np.zeros_like(scorer)
                w[np.argmax(scorer)] = 1.0
            else:
                w /= ws
            g = _subgrad(h, w)
            g -= np.mean(g)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
            direction = -g / gn
            momentum = 0.8 * momentum + 0.2 * direction
            mn = float(np.linalg.norm(momentum))
            if mn < 1e-15:
                momentum = direction
            else:
                momentum /= mn
            best_a = 0.0
            best_tc5 = cur_c5
            cands = [step_size, step_size * 0.7, step_size * 0.5, step_size * 0.3, step_size * 0.15]
            for a in cands:
                trial = _project(h + a * momentum, n / 2.0)
                tc5 = _exact_c5(trial)
                if tc5 < best_tc5 - 1e-14:
                    best_tc5 = tc5
                    best_a = a
            if best_a > 0:
                h = _project(h + best_a * momentum, n / 2.0)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stag = 0
                step_size = min(step_size * 1.03, 0.25)
            else:
                stag += 1
                step_size *= 0.88
                if step_size < 0.003:
                    break
    return best_h, best_c5

def _pair_descent(h, deadline, rng, n_pairs_per_round=500):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stag = 0
    
    while time.time() < deadline and stag < 2000:
        F = _scores(h)
        improved = False
        for _ in range(n_pairs_per_round):
            if time.time() > deadline:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            delta = _pair_optimal(h, i, j, F)
            if abs(delta) < 1e-12:
                continue
            h[i] += delta
            h[j] -= delta
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i] -= delta
                h[j] += delta
                continue
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                F = _scores(h)
            else:
                h[i] -= delta
                h[j] += delta
        if improved:
            stag = 0
        else:
            stag += 1
    return best_h, best_c5

def _local_coords(h, deadline, rng, max_iters=500000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    no_imp = 0
    idx = np.arange(n)
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_imp > 8000:
            break
        i = int(idx[rng.integers(n)])
        if h[i] <= 1e-9 or h[i] >= 1.0 - 1e-9:
            no_imp += 1
            continue
        delta = float(rng.uniform(-0.08, 0.08))
        new_val = max(0.0, min(1.0, h[i] + delta))
        if abs(new_val - h[i]) < 1e-12:
            continue
        d = new_val - h[i]
        h[i] = new_val
        c5_new = _exact_c5(h)
        if c5_new < cur_c5 - 1e-14:
            h = _project(h, n / 2.0)
            c5_new = _exact_c5(h)
            if c5_new < cur_c5 - 1e-14:
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_imp = 0
            else:
                h[i] = new_val - d
                no_imp += 1
        else:
            h[i] = new_val - d
            no_imp += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
```
