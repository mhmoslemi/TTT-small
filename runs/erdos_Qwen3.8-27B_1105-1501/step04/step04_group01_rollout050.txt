def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : sum(x) = target, 0 <= x <= 1}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

Now let me think about the **binary local search**. For a near-binary h (most values close to 0 or 1), we can:
1. For each pair (i, j) where h[i] is close to 0 and h[j] is close to 1 (or vice versa), try swapping
2. For each pair (i, j) where both are "interior" (0 < h < 1), try various moves

Actually, a simpler approach: for each pair (i, j), try the direction d with d[i] = 1, d[j] = -1 (move mass from j to i). Use line search to find the best step.

This is O(n²) directions per iteration, which for n=112 is about 6000 directions. Each line search takes about 40 correlation evaluations, so about 240,000 correlations per full pass. Each correlation is O(n²) ≈ 12,500 flops. So one full pass is about 3 billion flops. That might take about 30 seconds. Too slow.

Instead, I can do a **randomized** version: at each step, pick a random pair and try the move. This is O(n) per step (just the line search), and we can do many steps.

Let me also think about a **fast C5 evaluation** that avoids the full correlation:

Actually, for the line search, we already use the formula S0 + t*T - t²*C, which is O(n) after precomputing S0, T, C (each O(n²)). So the line search is fast once we have S0, T, C.

For the pair direction d with d[i]=1, d[j]=-1:
- T = correlate(d, 1-h) - correlate(h, d)
  = (1-h shifted appropriately) - (h shifted appropriately)
  
Actually, for a sparse direction (only 2 non-zero elements), T and C are very simple:
- C = d·d at lag 0 = d[i]² + d[j]² = 2 (at lag 0), and d[i]*d[j] at lag |i-j| and -(d[i]*d[j]) at lag -(i-j)
- T is similarly sparse

So the line search for a pair direction is very fast! We can compute the objective at any t in O(1) (just look at a few entries of S0, T, C).

Wait, that's not quite right. The objective is max_k (S0[k] + t*T[k] - t²*C[k]). Even if T and C are sparse (only a few non-zero entries), the max is over all k, and S0 is dense. So the objective at step t is max(S0 + t*T - t²*C), which is dense because of S0.

But we can vectorize: S0 + t*T - t²*C is a vector of length 2n-1, and we just take the max. This is O(n) per evaluation. The line search does ~40 evaluations, so O(40n) per line search. For n=112, that's about 4500 operations per line search. Very fast!

So the pair direction approach is actually fast. Let me implement it:

But wait, with n=112, there are only 112*111/2 ≈ 6200 unique pairs (up to sign). So after 6200 trials, we've tried all pairs at least once. After 100000 trials, we've tried each pair about 8 times. That should be sufficient for convergence.

But the issue is that _line_search with a pair direction might not find a good step if the trust region is too small. Let me use a larger trust region for pair moves (like 0.5 or even 1.0).

Actually, for a pair direction d[i]=1, d[j]=-1, the constraint h[i]+alpha ≤ 1 and h[j]-alpha ≥ 0 gives alpha ≤ min(1-h[i], h[j]). So the natural trust region is min(1-h[i], h[j]), which could be up to 1.0.

Let me use trust_radius = min(1.0 - h[i], h[j]) for the line search.

Actually, the _line_search function already handles this: it computes t_lo and t_hi based on the direction and current h values. So if d[i]=1, d[j]=-1, then:
- pos = {i} (since d[i] > 0)
- neg = {j} (since d[j] < 0)
- t_hi = min(trust_radius, (1-h[i])/1) = min(trust_radius, 1-h[i])
- t_lo = max(-trust_radius, h[j]/(-1)) = max(-trust_radius, -h[j])

So the line search will automatically respect the bounds. Good.

- Consecutive pairs (i, i+1) for all i
- Pairs at specific distances (i, i+k) for various k

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _softmax_dir(h, dx, tau):
