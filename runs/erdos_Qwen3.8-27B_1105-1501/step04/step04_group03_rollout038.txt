import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def to_binary(h):
    """Convert to binary (exactly n/2 ones)"""
    n = len(h)
    target = n // 2
    # Take top n/2 values
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_hill_climb(h, deadline, max_passes=5000):
    """Local search on binary vectors with exactly n/2 ones."""
    n = len(h)
    target = n // 2
    h = h.copy()
    # Ensure binary
    h = to_binary(h)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        # Try all pairs
        for i_idx in range(n_ones):
            for j_idx in range(n_zeros):
                i = ones[i_idx]
                j = zeros[j_idx]
                h[i] = 0
                h[j] = 1
                c5_t = exact_c5(h)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = h.copy()
                    improved = True
                    # Update tracking
                    ones[i_idx] = j  # i becomes zero, j becomes one
                    zeros[j_idx] = i
                    # Re-sort for efficiency (or just continue)
                    break
                else:
                    h[i] = 1
                    h[j] = 0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

**Quadratic model approach:**

At point h, for the active shift k*:
g_{k*}(h + Δh) ≈ g_{k*}(h) + ∇g_{k*}·Δh + ½ Δh^T H_{k*} Δh

where H_{k*} is the Hessian. For a single shift d:
H[j,j'] = -(δ_{j, j'+d} + δ_{j+d, j'}) (for valid indices)

This is a sparse matrix. We could use it for a Newton step.

But the complication is the MAX: we need all g_k to decrease, not just the active one.

**Alternative: Use the soft-max Hessian**

The soft-max F_τ(h) = τ log Σ exp(g_k/τ) has a well-defined Hessian:
∇²F_τ = (1/τ) Σ_k w_k ∇²g_k + (1/τ) Σ_k w_k (∇g_k)(∇g_k)^T - (1/τ) (Σ_k w_k ∇g_k)(Σ_k w_k ∇g_k)^T

where w_k = exp(g_k/τ) / Σ exp(g_j/τ).

The first term is a weighted average of Hessians (sparse), and the latter two terms are rank-1. So the Hessian is sparse + rank-1, which we can handle efficiently with the conjugate gradient method.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_hill_climb(h_bin, deadline):
    """Best-improvement hill climbing on binary vectors."""
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = np.max(corr)
        best_delta = -1e-14
        best_pair = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nc = np.correlate(h, 1.0 - h, mode='full')
                nm = np.max(nc)
                h[i] = 1.0
                h[j] = 0.0
                delta = nm - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = exact_c5(h)
        best_h = h.copy()
    
    return best_h, best_c5

def binary_anneal(h_bin, deadline, seed=99):
    """Simulated annealing on binary vectors."""
    n = len(h_bin)
    rng = np.random.default_rng(seed)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    
    # Adaptive temperature
    T = 0.005
    t_start = time.time()
    
    while time.time() < deadline:
        elapsed = time.time() - t_start
        # Cool down
        frac = elapsed / max(deadline - t_start, 1)
        T = 0.005 * (1e-4) ** frac  # geometric cooling from 0.005 to 5e-7
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = exact_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def lp_direction(h, scores, G, radius):
    """Find LP direction to decrease all correlation values."""
    from scipy.optimize import linprog
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def build_corr_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(20):
        a = alpha_max * (i + 1) / 20.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section
    lo_a = max(0.0, best_a * 0.7)
    hi_a = min(alpha_max, best_a * 1.3)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = build_corr_grad(h)
        p = lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.65
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.5)
        else:
            stall += 1
            radius *= 0.65
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    scores_raw = np.correlate(h, omh, mode='full')
    scores = scores_raw * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            grad[js1:je1 + 1] += wk * (1.0 - h[js1 + d:je1 + 1 + d])
        js2 = max(0, d)
        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wk * h[js2 - d:je2 + 1 - d]
    return val, dx * grad

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        v, _ = soft_value_grad(x, tau)
        return v
    
    def grad(x):
        _, g = soft_value_grad(x, tau)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=100):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue

import numpy as np

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_corr_grad(h):

    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:

                G[r, :m] += -h[d:]

            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    scores_raw = np.correlate(h, omh, mode='full')
    scores = scores_raw * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            grad[js1:je1 + 1] += wk * (1.0 - h[js1 + d:je1 + 1 + d])
        js2 = max(0, d)
        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wk * h[js2 - d:je2 + 1 - d]
    return val, dx * grad

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, _ = soft_value_grad(x, tau)
        return v

    def grad(x):
        _, g = soft_value_grad(x, tau)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):

    except Exception:
        pass
    return h0

    scales = [0.02, 0.04, 0.06, 0.1, 0.14, 0.2]
    for ns in scales:
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_try, c5_try = lp_loop(h_pert, sub_dl, radius_init=0.05)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Also try SLSQP on perturbation
        if time.time() < d6:
            h_try2 = slsqp_smooth(h_pert, 0.1, 50)
            h_try2 = project(h_try2, best_n / 2.0)
            c5_try2 = exact_c5(h_try2)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase 7: Multi-resolution
    d7 = t0 + budget_s * 0.88
    tried_n = set([best_n])
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -12, 12, -16, 16, -20, 20]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.04)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Pair polish
    d8 = t0 + budget_s * 0.92
    h_pol, c5_pol = pair_polish(best_h, d8, max_passes=60)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 9: Final LP micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.015)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
