def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-n + 1, n):
        r = n - 1 + d
        if d >= 0:
            m1_end = n - d
            if m1_end > 0:
                G[r, :m1_end] += -dx * h[d:d + m1_end]
            if d < n:
                G[r, d:] += dx * (1.0 - h[:n - d])
        else:
            g = -d
            m2_end = n - g
            if m2_end > 0:
                G[r, :m2_end] += dx * (1.0 - h[g:g + m2_end])
            if g < n:
                G[r, g:] += -dx * h[:n - g]
    return G

def find_optimal_t(scores, delta, t_max):
    if t_max < 1e-13:
        return 0.0
    lo = 0.0
    hi = t_max
    for _ in range(22):
        m1 = lo + (hi - lo) * (1.0 / 3.0)
        m2 = hi - (hi - lo) * (1.0 / 3.0)
        f1 = np.max(scores + m1 * delta)
        f2 = np.max(scores + m2 * delta)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def resample_h(h_old, n_new):

    if n_new == n_old:

    x_old = (np.arange(n_old) + 0.5) / n_old

    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)
