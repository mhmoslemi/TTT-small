def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:g + m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, max_alpha=0.5):

    if best_i in (0, n_s - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]

            f2 = _exact_c5(h + c2 * p)

                    h2[i] += sign * m2
                    h2[j] -= sign * m2
                    h2 = np.clip(h2, 0, 1)
                    f2 = _exact_c5(h2)
                    if f1 < f2:
                        hi_d = m2
                    else:
                        lo_d = m1
                d_opt = (lo_d + hi_d) / 2.0
                if d_opt > 1e-10:
                    h_try = h.copy()
                    h_try[i] += sign * d_opt
                    h_try[j] -= sign * d_opt
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: No-regret at original resolution
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _no_regret_opt(h_start, dl1, eta_init=0.5, alpha_init=0.04,
                               max_iters=30000, eta_growth=1.003, alpha_decay=0.9998)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refinement
    if time.time() < t0 + budget_s * 0.20:
        dl2 = t0 + budget_s * 0.22
        h2, c5_2 = _local_lp_opt(best_h, dl2, top_k_list=[None, 15, 8, 5],
                                   radius_init=0.12, max_iters=2000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Multi-resolution exploration
    dl3 = t0 + budget_s * 0.55
    resolutions = [160, 192, 224, 256, 144, 128]
    alphas_qp = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.56903, 0.44721]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if best_c5 < 0.3805:
            break
        n_starts = 2
        for s in range(n_starts):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas_qp[ri % len(alphas_qp)]
                h_init = _quasi_periodic_init(n_try, alpha)
                h_init += rng.normal(0, 0.05, n_try)
                h_init = _project(h_init, n_try / 2.0)

            sub_dl = min(dl3, time.time() + budget_s * 0.04)
            # No-regret first (fast, many iterations)
            h_nr, c5_nr = _no_regret_opt(h_init, sub_dl, eta_init=0.8, alpha_init=0.03,
                                          max_iters=20000, eta_growth=1.004, alpha_decay=0.9997)
            if c5_nr < best_c5:
                best_c5 = c5_nr
                best_h = h_nr.copy()
                best_n = n_try
            # Then stochastic subgrad
            if time.time() < dl3:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.025)
                h_sg, c5_sg = _stochastic_subgrad(h_nr, sub_dl2, max_iters=30000, top_m=5, seed=seed + ri * 10 + s)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()
                    best_n = n_try
            # Then LP polish
            if time.time() < dl3:
                sub_dl3 = min(dl3, time.time() + budget_s * 0.02)
                h_lp, c5_lp = _local_lp_opt(h_sg, sub_dl3, top_k_list=[10, 5, 8],
                                              radius_init=0.08, max_iters=800)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = n_try

    # Phase 4: Binary SA at various sizes
    if time.time() < t0 + budget_s * 0.72:
        dl4 = t0 + budget_s * 0.75
        for n_bin in [96, 128, 160, 192]:
            if time.time() > dl4:
                break
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=20000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                # LP polish the binary result
                if time.time() < dl4:
                    sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                    h_bp, c5_bp = _local_lp_opt(h_b, sub_dl2, top_k_list=[10, 5, None],
                                                  radius_init=0.1, max_iters=600)
                    if c5_bp < best_c5:
                        best_c5 = c5_bp
                        best_h = h_bp.copy()

    # Phase 5: Perturbation restarts from best
    dl5 = t0 + budget_s * 0.88
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        h_try, c5_try = _no_regret_opt(h_pert, sub_dl, eta_init=1.0, alpha_init=0.03,
                                        max_iters=5000, eta_growth=1.01, alpha_decay=0.998)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 6: LP + pair polish on best
    if time.time() < t0 + budget_s * 0.95:
        dl6 = t0 + budget_s * 0.96
        h_lp2, c5_lp2 = _local_lp_opt(best_h, dl6, top_k_list=[5, 8, 3, None],
                                        radius_init=0.05, max_iters=1000)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.99:
        dl7 = t0 + budget_s * 0.995
        hp, c5p = _pair_polish(best_h, dl7, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: tiny LP refinement
    if time.time() < t0 + budget_s * 0.998:
        h_fin, c5_fin = _local_lp_opt(best_h, t0 + budget_s * 0.999,
                                        top_k_list=[3, 4, 5],
                                        radius_init=0.01, max_iters=300)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
