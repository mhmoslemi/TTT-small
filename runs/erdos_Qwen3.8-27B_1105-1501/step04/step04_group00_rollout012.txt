        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

def grad_rows(h, active_idx):
    """Build gradient rows for given active indices."""
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    G = np.zeros((m, n))

            G[k, :length] -= dx * h[d:d+length]

            g = -d
            length = n - g
            G[k, :length] += dx * (1.0 - h[g:])
            G[k, g:g+length] -= dx * h[:length]
    return G

def qp_direction(h, scores_arr, top_k=None, trust=0.2, ridge=1e-3, rough=1e-3, use_both_sides=True):
    """Find QP direction to minimize active scores simultaneously."""
    n = len(h)
    s_all = scores_arr
    C = np.max(s_all)
    
    # Select active set
    if use_both_sides:
        # Include top scores on both positive and negative lags
        center = n - 1
        pos = np.argsort(s_all[center:]) + center
        neg = np.argsort(s_all[:center])
        if top_k is None:
            top_k = min(20, n)
        half = top_k // 2
        idx_pos = pos[:half]
        idx_neg = neg[:top_k - half]
        active = np.concatenate([idx_neg, idx_pos])
        active = np.unique(active)
    else:

        active = np.argsort(s_all)[-top_k:]

    s_active = s_all[active]

    #      1^T delta = 0

            A[i+1, i+1] += 2 * rough
            A[i, i+1] -= rough
            A[i+1, i] -= rough

    rhs = np.zeros(n+1)

    try:

    if max_abs < 1e-14:

    if max_abs > trust:
        delta *= trust / max_abs
    
    return delta

import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_rows(h, active_idx):
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    G = np.zeros((m, n))
    for k, r in enumerate(active_idx):
        d = int(r) - (n - 1)
        if d >= 0:
            length = n - d
            G[k, d:d+length] += dx * (1.0 - h[:length])
            G[k, :length] -= dx * h[d:d+length]
        else:
            g = -d
            length = n - g
            G[k, :length] += dx * (1.0 - h[g:])
            G[k, g:g+length] -= dx * h[:length]
    return G

def qp_direction(h, s, top_k, ridge, rough, target_alpha):
    n = len(h)
    C = float(np.max(s))
    if C < 1e-15:
        return None
    center = n - 1
    pos_rank = np.argsort(s[center:]) + center
    neg_rank = np.argsort(s[:center])
    half = max(2, top_k // 2)
    idx_pos = pos_rank[:half]
    idx_neg = neg_rank[:top_k - half]
    active = np.unique(np.concatenate([idx_neg, idx_pos]))
    G = grad_rows(h, active)
    s_act = s[active]
    target = target_alpha * C
    rhs_vec = s_act - target
    A = G.T @ G
    A += ridge * np.eye(n)
    if rough > 0:
        for i in range(n - 1):
            A[i, i] += 2.0 * rough
            A[i+1, i+1] += 2.0 * rough
            A[i, i+1] -= rough
            A[i+1, i] -= rough
    b = -G.T @ rhs_vec
    K = np.zeros((n+1, n+1))
    K[:n, :n] = A
    K[:n, n] = 1.0
    K[n, :n] = 1.0
    rhs = np.zeros(n+1)
    rhs[:n] = b
    try:
        sol = np.linalg.solve(K, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(K, rhs, rcond=1e-12)[0]
        except Exception:
            return None
    delta = sol[:n].copy()
    delta -= delta.mean()
    max_abs = float(np.max(np.abs(delta)))
    if max_abs < 1e-14:
        return None
    return delta / max_abs

def weighted_grad_dir(h, s, top_k):
    n = len(h)
    C = float(np.max(s))
    if C < 1e-15:
        return None
    center = n - 1
    pos_rank = np.argsort(s[center:]) + center
    neg_rank = np.argsort(s[:center])
    half = max(2, top_k // 2)
    active = np.unique(np.concatenate([neg_rank[:top_k-half], pos_rank[:half]]))
    G = grad_rows(h, active)
    beta = 5.0
    w = np.exp(beta * (s[active] - s[active].max()))
    w /= w.sum()
    p = -(G.T @ w)
    p -= p.mean()
    mx = float(np.max(np.abs(p)))
    if mx < 1e-14:
        return None
    return p / mx

def line_search(h, p, cur_c5):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-14)
    nsamp = 12
    alphas = np.linspace(0.0, alpha_max, nsamp)
    vals = np.empty(nsamp)
    for i in range(nsamp):
        vals[i] = exact_c5(np.clip(h + alphas[i] * p, 0.0, 1.0))
    best_i = int(np.argmin(vals))
    best_a = alphas[best_i]
    best_val = vals[best_i]
    if best_i in (0, nsamp - 1):
        return best_a, best_val
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(nsamp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
    f2 = exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
    for _ in range(25):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(np.clip(h + c1 * p, 0.0, 1.0))
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(np.clip(h + c2 * p, 0.0, 1.0))
    opt_a = (lo + hi) / 2.0
    opt_c = exact_c5(np.clip(h + opt_a * p, 0.0, 1.0))
    if opt_c < best_val:
        return opt_a, opt_c
    return best_a, best_val

def local_qp_opt(h, deadline, top_k_list, ridge_list, rough_list, target_alphas, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    momentum = np.zeros(n)
    mom_rate = 0.3
    it = 0
    ncfg = len(top_k_list)
    while it < max_iters:
        if time.time() > deadline:
            break
        if stall > 50:
            break
        cfg_i = it % ncfg
        tk = top_k_list[cfg_i]
        ridge = ridge_list[cfg_i % len(ridge_list)]
        rough = rough_list[cfg_i % len(rough_list)]
        ta = target_alphas[cfg_i % len(target_alphas)]
        s = scores_arr(h)
        p = qp_direction(h, s, tk, ridge, rough, ta)
        if p is None:
            p = weighted_grad_dir(h, s, tk)
        if p is None:
            p = weighted_grad_dir(h, s, max(5, tk // 2))
        if p is None:
            stall += 2
            it += 1
            continue
        momentum = mom_rate * momentum + (1.0 - mom_rate) * p
        momentum -= momentum.mean()
        norm_m = float(np.max(np.abs(momentum)))
        if norm_m < 1e-14:
            stall += 1
            it += 1
            continue
        p_use = momentum / norm_m
        alpha, new_c5 = line_search(h, p_use, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = np.clip(h + alpha * p_use, 0.0, 1.0)
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            h_try = np.clip(h + 0.3 * alpha * p, 0.0, 1.0)
            h_try = project(h_try, target)
            c5_try = exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                momentum *= 0.5
        it += 1
    return best_h, best_c5

def lp_direction(h, s, top_k, radius):
    n = len(h)
    center = n - 1
    pos_rank = np.argsort(s[center:]) + center
    neg_rank = np.argsort(s[:center])
    half = max(2, top_k // 2)
    active = np.unique(np.concatenate([neg_rank[:top_k-half], pos_rank[:half]]))
    G = grad_rows(h, active)
    s_act = s[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -s_act
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and float(np.max(np.abs(res.x[:n]))) > 1e-13:
            return res.x[:n]
    except Exception:
        pass
    return None

def local_lp_opt(h, deadline, top_k, radius_init, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        s = scores_arr(h)
        p = lp_direction(h, s, top_k, radius)
        if p is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = np.clip(h + alpha * p, 0.0, 1.0)
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.45)
        else:
            stall += 1
            radius *= 0.72
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def make_smooth_perturb(n, rng, scale, n_modes=5):
    x = np.arange(n) / n
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        amp = scale / k * rng.standard_normal()
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(2 * np.pi * k * x + phase)
    pert -= pert.mean()
    return pert

def pair_polish(h, deadline, max_passes=4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    step = max(1, n // 40)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(0, n, step):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(10):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0, 1)
                        f1 = exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0, 1)
                        f2 = exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    h0 = np.array(initial_h_values, dtype=float)

        h_pert = project(best_h + pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        h_try, c5_try = local_qp_opt(h_pert, sub_dl,
                                      top_k_list=[10, 14],
                                      ridge_list=[2e-3, 5e-3],
                                      rough_list=[1e-3, 0.0],
                                      target_alphas=[0.0, 0.5],

        if time.time() > dl6:

        h_bin = resample(best_h, n_bin)
        h_bin = (h_bin > 0.5).astype(float)
        h_bin = project(h_bin, n_bin / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.025)
        h_b, c5_b = local_qp_opt(h_bin, sub_dl,
                                  top_k_list=[12, 8],
                                  ridge_list=[5e-3],
                                  rough_list=[3e-3],
                                  target_alphas=[0.0],
                                  max_iters=1000)
        try_update(h_b, c5_b, n_bin)

    # Phase 7: Final QP polish
    dl7 = t0 + budget_s * 0.97
    h7, c5_7 = local_qp_opt(best_h, dl7,

    # Phase 8: LP polish

    # Phase 9: Pair polish

        try_update(hp, c5p, best_n)

    # Final projection and evaluation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
