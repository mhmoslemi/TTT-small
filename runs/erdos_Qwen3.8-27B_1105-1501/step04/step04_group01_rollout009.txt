def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_vals(h):

def _binary_sa(h_start, n, budget_s, rng, T_init=0.06, T_min=0.0005, cool=0.9993):
    """Simulated annealing on binary sequences."""
    target = n // 2
    h = np.zeros(n)
    # Use h_start to guide initial placement
    if len(h_start) >= n:
        h[:n] = (h_start[:n] >= 0.5).astype(float)
    else:
        h = (rng.random(n) > 0.5).astype(float)
    # Fix count
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    while len(ones) > target:
        idx = ones[rng.integers(0, len(ones))]
        h[idx] = 0.0
        ones = np.setdiff1d(ones, [idx])
    while len(ones) < target:
        idx = zeros[rng.integers(0, len(zeros))]
        h[idx] = 1.0
        zeros = np.setdiff1d(zeros, [idx])
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    t_start = time.time()

    while time.time() - t_start < budget_s and T > T_min:
        i = ones[rng.integers(0, len(ones))]
        j = zeros[rng.integers(0, len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones = np.setdiff1d(ones, [i])
            ones = np.union1d(ones, [j])
            zeros = np.setdiff1d(zeros, [j])
            zeros = np.union1d(zeros, [i])
        else:
            h[i], h[j] = h[j], h[i]
        T *= cool
    return best_h, best_c5

    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    t_start = time.time()

    # Try adjusting pairs of interior values
    while time.time() - t_start < budget_s:

        interior = np.where((h > 0.01) & (h < 0.99))[0]

            else:
                break

            if time.time() - t_start > budget_s:

            if i == j:
                continue
            # Try moving mass from i to j or j to i
            for sign in [1, -1]:
                d = np.zeros(n)
                d[i] = -sign * 0.05
                d[j] = sign * 0.05
                # Line search
                best_t = 0.0
                best_v = cur_c5

def _lp_local(h, budget_s, radius_init=0.1):
    """LP-based local search focused on worst shifts."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    t_start = time.time()

    while time.time() - t_start < budget_s and stall < 40:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = float(np.max(corr))
        # Find active shifts (within small epsilon of max)
        active = np.where(corr > max_val - 0.005)[0]
        # Compute gradient of max over active set (uniform weights)
        grad = np.zeros(n)
        for r in active:
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:d+m])
                    grad[d:] += -h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += (1.0 - h[g:g+m])
                    grad[g:] += -h[:m]
        grad /= len(active)
        # Project gradient to zero-sum
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-5:
                break
            continue
        grad = grad / gn * radius
        # Line search
        best_t = 0.0
        best_v = cur_c5
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T_arr = np.correlate(grad, 1.0 - h, mode='full') - np.correlate(h, grad, mode='full')
        C_arr = np.correlate(grad, grad, mode='full')
        # Find bounds
        pos = grad > 1e-14
        neg = grad < -1e-14
        t_max = radius
        if np.any(pos):
            t_max = min(t_max, float(np.min((1.0 - h[pos]) / grad[pos])))
        if np.any(neg):
            t_max = min(t_max, float(np.max(h[neg] / (-grad[neg]))))
        grid = np.linspace(0, t_max, 40)
        for t in grid:
            vals = S0 + t * T_arr - t * t * C_arr
            v = float(np.max(vals)) * dx
            if v < best_v:
                best_v = v
                best_t = t
        if best_t > 0 and best_v < cur_c5 - 1e-14:
            h = h + best_t * grad
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        zeros = np.where(h == 0)[0]
        while len(ones) > target:

            ones = np.setdiff1d(ones, [idx])
        while len(ones) < target:

    elif mode == 1:

        for i in range(n):

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
