def _project(v, target):

    if target <= 0:

    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _line_search(h, p, max_alpha=1.0):

    alpha_max = max_alpha
    for j in range(n):

    f1 = _exact_c5(h + c1 * p)

            f2 = f1
            c2 = c1

            f1 = f2
            c1 = c2

            best_a = c1
        if f2 < best_f:

            best_a = c2

def _full_lag_lp(h, rho, deadline):
    """Full-lag trust region LP. Returns (delta, model_tau) or (None, None)."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    scores = _scores(h)
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    
    # Variables: delta[0..n-1], tau (index n)
    # Minimize tau
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: G[r,:]·delta - tau <= -scores[r] for all r
    # => G[r,:]·delta + [-1]*tau <= -scores[r]
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on delta
    bounds = []
    for j in range(n):
        lo_b = max(-rho, -h[j])
        hi_b = min(rho, 1.0 - h[j])
        if lo_b > hi_b:
            lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))  # tau unbounded
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            return None, None
        delta = res.x[:n]
        tau = res.x[n]
        return delta, tau
    except Exception:
        return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 80:
                break
            continue
        d = -grad / gn
        alpha, new_c5 = _line_search(h, d, max_alpha=0.5)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            G = _grad_matrix(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            if stall > 100:
                break
    return best_h, best_c5

def _binary_init(n, seed):

def _binary_hill_climb(h_init, deadline, max_iters=20000):

def _pair_polish(h, deadline, max_passes=2):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(8):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5
