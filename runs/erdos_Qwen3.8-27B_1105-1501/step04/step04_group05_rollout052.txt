import numpy as np

import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _soft_value(h, tau):
    c = _corr_scaled(h)
    m = np.max(c)
    return float(m + tau * np.log(np.sum(np.exp((c - m) / tau))))


def _soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = _corr_scaled(h)
    m = np.max(c)
    exp_c = np.exp((c - m) / tau)
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        s = q - (n - 1)
        if s >= 0:
            L = n - s
            if L > 0:
                grad[s:] += w[q] * (1.0 - h[:L])
                grad[:L] += w[q] * (-h[s:])
        else:
            d = -s
            L = n - d
            if L > 0:
                grad[:L] += w[q] * (1.0 - h[d:])
                grad[d:] += w[q] * (-h[:L])
    return grad * dx

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            stall = 0
            lr = min(lr * 1.08, 0.08)
        elif c5_new < _exact_c5(h_cur) - 1e-13:
            h_cur = h_new
            stall = 0
            lr = min(lr * 1.03, 0.08)
        else:
            stall += 1
            lr = max(lr * 0.85, min_lr)
            if lr < min_lr:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 64
        h0 = np.zeros(n)
        h0[::2] = 1.0
        h0 = _project(h0, n / 2.0)

    n = len(h0)
    target = n / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Soft-max projected gradient (25% budget)
    d1 = t0 + budget_s * 0.25
    taus_pg = [0.06, 0.04, 0.025, 0.015, 0.01, 0.007, 0.005, 0.004]
    hp, c5p = _pg_loop_soft(best_h, d1, taus_pg, max_iters=3000, start_lr=0.06, min_lr=1e-5)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Phase 2: Soft-max SLSQP (20% budget)
    d2 = t0 + budget_s * 0.45
    taus_slsqp = [0.04, 0.025, 0.015, 0.01, 0.007, 0.005, 0.004, 0.003]
    for i, tau in enumerate(taus_slsqp):
        if time.time() > d2:
            break
        h_try = _slsqp_softmax(best_h, tau, maxiter=250, ftol=1e-13)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Also try from a perturbed version
        if time.time() < d2:
            h_pert = best_h + rng.normal(0, 0.02, len(best_h))
            h_pert = _project(h_pert, target)
            h_try2 = _slsqp_softmax(h_pert, tau, maxiter=150, ftol=1e-12)
            c5_try2 = _exact_c5(h_try2)
            if c5_try2 < best_c5 - 1e-13:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase 3: Threshold penalty (20% budget)
    d3 = t0 + budget_s * 0.65
    current_C = best_c5
    T = current_C * 0.999
    rho = 0.01
    for _iter in range(200):
        if time.time() > d3:
            break
        if T < 0.379:
            break
        h_try = _slsqp_thresh(best_h, T, rho, maxiter=200, ftol=1e-12)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            T = max(T * 0.998, best_c5 - 0.002)
            rho = max(rho * 0.9, 0.001)
        else:
            T = min(T + 0.0005, current_C)
            rho = min(rho * 1.3, 0.05)
            if T >= current_C:
                T = current_C * 0.999

    # Phase 4: Resolution search (15% budget)
    d4 = t0 + budget_s * 0.80
    n_try_candidates = sorted(
        [x for x in [n - 8, n - 4, n - 2, n + 2, n + 4, n + 8, n + 16, n + 32,
                     int(n * 1.5), int(n * 2), int(n * 0.75)]
         if 24 <= x <= 256 and x != n],
        key=lambda x: abs(x - n)
    )
    for n_try in n_try_candidates:
        if time.time() > d4:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.025)
        for tau_r in [0.02, 0.01, 0.006]:
            if time.time() > sub_dl:
                break
            h_r = _slsqp_softmax(h_r, tau_r, maxiter=150, ftol=1e-12)
        hg_r, c5g_r = _grad_loop_active(h_r, sub_dl, start_lr=0.005, min_lr=1e-6, n_lags=3, patience=30)
        if c5g_r < _exact_c5(h_r):
            h_r = hg_r
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            n = n_try
            target = n / 2.0

    # Phase 5: Active-set SLSQP + gradient refinement (15% budget)
    d5 = t0 + budget_s * 0.93
    c_full = _corr_scaled(best_h)
    active = _get_active(c_full, frac=0.4, min_count=12)
    for _round in range(10):
        if time.time() > d5:
            break
        h_act = _active_slsqp(best_h, active, maxiter=200, ftol=1e-13)
        c5_act = _exact_c5(h_act)
        if c5_act < best_c5 - 1e-13:
            best_c5 = c5_act
            best_h = h_act.copy()
            c_full = _corr_scaled(best_h)
            active = _get_active(c_full, frac=0.3, min_count=10)
        else:
            break
    hg5, c5g5 = _grad_loop_active(best_h, d5, start_lr=0.003, min_lr=1e-6, n_lags=3, patience=40)
    if c5g5 < best_c5:
        best_c5 = c5g5
        best_h = hg5.copy()

    # Phase 6: Pair refine (7% budget)
    d6 = t0 + budget_s * 0.97
    hp6, c5p6 = _pair_refine(best_h, d6, max_passes=300)
    if c5p6 < best_c5:
        best_c5 = c5p6
        best_h = hp6.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
