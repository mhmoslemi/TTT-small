corr(b,a)[n-1+d] = sum b[i]*a[i-d] = sum a[i]*b[i+d] (substituting j=i-d, i=j+d)
= corr(a,b)[n-1-d]

So corr(1-h, h)[n-1+d] = corr(h, 1-h)[n-1-d]

**Idea: Use the Hessian to get a better step length**

The quadratic model along direction d is:
f(t) = S0 + t*T - t²*C (elementwise, take max)
where S0 = corr(h, 1-h), T = gradient-direction, C = dᵀ Hessian d

The parent does golden section search on this. But we can do better:
- If we know which lag r* will be the maximizer at the optimum, we can solve the 1D quadratic exactly:

  Optimal t = T[r*] / (2*C[r*]) (if C[r*] > 0)

- We can iterate: start with r* = argmax S0, compute optimal t, check if r* is still the argmax at t, repeat.

```python
def _active_line_search(h, d, dx, trust_radius, max_iter=20):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find bounds
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = max(-trust_radius, float(np.max(h[neg] / d[neg]))) if np.any(neg) else -trust_radius
    t_hi = min(trust_radius, float(np.min((1.0 - h[pos]) / d[pos]))) if np.any(pos) else trust_radius
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Active set line search
    r_star = int(np.argmax(S0))
    t_best = 0.0
    f_best = float(np.max(S0)) * dx
    
    for _ in range(max_iter):
        Cs = C[r_star]
        if Cs > 1e-12:
            t_opt = T[r_star] / (2.0 * Cs)
        else:
            t_opt = t_hi if T[r_star] < 0 else 0.0
        t_opt = max(t_lo, min(t_hi, t_opt))
        
        # Evaluate
        f_val = float(np.max(S0 + t_opt * T - t_opt**2 * C)) * dx
        if f_val < f_best - 1e-14:
            f_best = f_val
            t_best = t_opt
            # Update active set
            new_r = int(np.argmax(S0 + t_opt * T - t_opt**2 * C))
            if new_r == r_star:
                break
            r_star = new_r
        else:
            break
    
    # Also check endpoints
    for t_check in [t_lo, t_hi, 0.0]:
        if t_lo <= t_check <= t_hi:
            fc = float(np.max(S0 + t_check * T - t_check**2 * C)) * dx
            if fc < f_best:
                f_best = fc
                t_best = t_check
    
    return t_best, f_best
```

Let me be extra careful about the active line search. The key insight is:

The quadratic model for the max objective along direction d with step t is:
F(t) = max_r [S0[r] + t*T[r] - t²*C[r]]

Each term is a (potentially) concave quadratic in t (if C[r] >= 0). The maximum of concave functions is generally not concave, but it is continuous and piecewise quadratic.

The active set method:
1. Start with r* = argmax S0
2. Solve the 1D quadratic for r*: t_opt = T[r*] / (2*C[r*]) (clipped to [t_lo, t_hi])
3. Evaluate F(t_opt). If r* = argmax at t_opt, we're at a local min of F.
4. Otherwise, update r* and repeat.

This converges in at most (number of distinct active sets) iterations, which is typically small.

But there's a subtlety: the "optimal" t for the active r* might not be the global min of F. We should also check a few other candidate t values (from the golden section or a coarse grid).

I'll combine both: do the active set search AND a coarse grid, then take the best.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _active_line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = max(-trust_radius, float(np.max(h[neg] / d[neg]))) if np.any(neg) else -trust_radius
    t_hi = min(trust_radius, float(np.min((1.0 - h[pos]) / d[pos]))) if np.any(pos) else trust_radius
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    r_star = int(np.argmax(S0))
    t_best = 0.0
    f_best = float(np.max(S0)) * dx
    for _ in range(30):
        Cs = C[r_star]
        if Cs > 1e-12:
            t_opt = T[r_star] / (2.0 * Cs)
        else:
            t_opt = t_hi if T[r_star] < 0 else 0.0
        t_opt = max(t_lo, min(t_hi, t_opt))
        f_val = float(np.max(S0 + t_opt * T - t_opt * t_opt * C)) * dx
        if f_val < f_best - 1e-15:
            f_best = f_val
            t_best = t_opt
            new_r = int(np.argmax(S0 + t_opt * T - t_opt * t_opt * C))
            if new_r == r_star:
                break
            r_star = new_r
        else:
            break
    n_grid = 30
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    gi = int(np.argmin(obj_grid))
    if obj_grid[gi] < f_best:
        f_best = obj_grid[gi]
        t_best = grid[gi]
    return t_best, f_best

def _softmax_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    grad = w @ G
    grad -= np.mean(grad)
    return grad

def _nesterov_phase(h, deadline, tau_start=30.0, tau_end=500.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    x_prev = h.copy()
    momentum = np.zeros(n)
    lr = 0.03
    stall = 0
    total_iters = 0
    t_start = time.time()
    while time.time() < deadline:
        elapsed_frac = min(1.0, (time.time() - t_start) / max(1.0, deadline - t_start))
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        g = _softmax_grad(h, dx, tau)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        momentum = 0.85 * momentum + 0.15 * g
        mn = np.linalg.norm(momentum)
        if mn < 1e-14:
            continue
        d = lr * momentum / mn
        alpha, new_c5 = _active_line_search(h, d, dx, lr * 2.0)
        if new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            cur_c5 = _exact_c5(h_new)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.8
            if lr < 0.002:
                break
        h = _project(h + alpha * d, target) if new_c5 < cur_c5 - 1e-14 else h
        cur_c5 = _exact_c5(h)
        total_iters += 1
        if stall > 50:
            break
    return best_h, best_c5

def _active_lp_loop(h, deadline, radius_init=0.15, top_k=10):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    k = top_k
    while time.time() < deadline:
        if stall > 40:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(scores)[-k:]
        G_full = _grad_matrix(h)
        G_active = G_full[top_idx, :]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((k, n + 1))
        A_ub[:, :n] = G_active
        A_ub[:, n] = -1.0
        b_ub = -scores[top_idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.success:
                p = res.x[:n]
                if float(np.max(np.abs(p))) > 1e-14:
                    alpha, new_c5 = _active_line_search(h, p, dx, radius)
                    if new_c5 < cur_c5 - 1e-14:
                        h = h + alpha * p
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.06, 0.30)
                        continue
            else:
                p = None
        except Exception:
            p = None
        stall += 1
        radius *= 0.75
        k = max(5, k - 1)
        if stall % 20 == 0 and radius < 0.015:
            radius = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _freq_perturb(h, rng, n_harmonics=5):
    n = len(h)
    x = np.arange(n)
    d = np.zeros(n)
    for k in range(1, n_harmonics + 1):
        freq = rng.uniform(0.5 * k, 1.5 * k)
        amp = rng.normal(0, 0.03 / k)
        phase = rng.uniform(0, 2 * np.pi)
        d += amp * np.sin(2 * np.pi * freq * x / n + phase)
    d -= np.mean(d)
    return d

def _perturb_escape(best_h, deadline, rng):
    n = len(best_h)
    best_h = best_h.copy()
    best_c5 = _exact_c5(best_h)
    scale = 0.03
    while time.time() < deadline:
        d = _freq_perturb(best_h, rng, n_harmonics=rng.integers(2, 8))
        h_pert = best_h + scale * d
        h_pert = _project(h_pert, n / 2.0)
        sub_dl = min(deadline, time.time() + 8.0)
        h_t, c5_t = _active_lp_loop(h_pert, sub_dl, radius_init=0.08, top_k=8)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.9)
        else:
            scale = min(0.15, scale * 1.15)
    return best_h, best_c5
