def _pair_optimize(h, i, j):
    """Exact golden section search on pair (i,j). Returns improved h and c5."""
    n = len(h)
    # Valid range for t: h[i]+t in [0,1], h[j]-t in [0,1]
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return h, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    h_save = h.copy()
    
    h[i] += c; h[j] -= c
    fc = _exact_c5(h)
    h[i] += (d - c); h[j] -= (d - c)
    fd = _exact_c5(h)
    
    best_t = 0.0
    best_f = _exact_c5(h_save)
    
    for _ in range(40):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - (b - a) / gr
            h[i] = h_save[i] + c; h[j] = h_save[j] - c
            fc = _exact_c5(h)
        else:
            a = c; c = d; fc = fd
            d = a + (b - a) / gr
            h[i] = h_save[i] + d; h[j] = h_save[j] - d
            fd = _exact_c5(h)
        if abs(b - a) < 1e-10:
            break
    
    # Evaluate at the best point
    t_opt = (a + b) / 2.0
    h[i] = h_save[i] + t_opt; h[j] = h_save[j] - t_opt
    f_opt = _exact_c5(h)
    
    if f_opt < best_f:
        return h, f_opt
    else:
        h[:] = h_save
        return h, best_f

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)

def _project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_smoothed(h, beta):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    Fs = F - np.max(F)
    expF = np.exp(np.clip(beta * Fs, -700.0, 0.0))
    w = expF / np.sum(expF)
    grad = np.zeros(n)
    for k in range(L):
        if w[k] < 1e-13:
            continue
        d = k - (n - 1)
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        ia = np.arange(low, high)
        ja = ia + d
        grad[ia] += w[k] * dx * (1.0 - h[ja])
        grad[ja] += w[k] * dx * (-h[ia])
    return grad

def _grad_active(h, top_k=8):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    Fmax = np.max(F)
    # Select top lags within a small fraction of max
    thresh = Fmax - 0.02
    active = np.where(F >= thresh)[0]
    if len(active) > top_k:
        active = active[np.argsort(-F[active])][:top_k]
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        ia = np.arange(low, high)

def _lp_trust_step(h, radius, top_k=10):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    # Select top lags
    order = np.argsort(-F)
    active = order[:top_k]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    
    for ai, k in enumerate(active):
        d = k - (n - 1)
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        ia = np.arange(low, high)
        ja = ia + d
        A_ub[ai, ia] += dx * (1.0 - h[ja])
        A_ub[ai, ja] += dx * (-h[ia])
        A_ub[ai, n] = -1.0
        b_ub[ai] = F[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_start=0.10, top_k=10, max_iters=10000):

    radius = radius_start
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        x_new = _lp_trust_step(h, radius, top_k)
        if x_new is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        x_new = _project_box_sum(x_new, target)
        best_step = 0.0
        best_tc5 = cur_c5
        for factor in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            trial = _project_box_sum(h + factor * (x_new - h), target)
            tc5 = _exact_c5(trial)

                best_tc5 = tc5

        if best_step > 0:

                best_h = h.copy()
            stall = 0

            stall += 1

    return best_h, best_c5

def _smoothed_descent(h, deadline, beta_start=15.0, beta_end=8000.0, n_beta=25,

        if time.time() > deadline:
            break

def _pair_golden(h, i, j, n_iter=35):
    """Exact golden section search on pair transfer. Returns (h, c5)."""
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return h, _exact_c5(h)
    
    h_save = h[i], h[j]
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    a, b = t_lo, t_hi
    c_pt = b - (b - a) / gr
    d_pt = a + (b - a) / gr
    
    h[i] = h_save[0] + c_pt
    h[j] = h_save[1] - c_pt
    fc = _exact_c5(h)
    h[i] = h_save[0] + d_pt
    h[j] = h_save[1] - d_pt
    fd = _exact_c5(h)
    
    for _ in range(n_iter):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - (b - a) / gr
            h[i] = h_save[0] + c_pt
            h[j] = h_save[1] - c_pt
            fc = _exact_c5(h)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + (b - a) / gr
            h[i] = h_save[0] + d_pt
            h[j] = h_save[1] - d_pt
            fd = _exact_c5(h)
    
    t_opt = (a + b) / 2.0
    h[i] = h_save[0] + t_opt
    h[j] = h_save[1] - t_opt
    f_opt = _exact_c5(h)
    if f_opt < _exact_c5(_project_box_sum(np.array(h), n/2.0)) + 1e-15:
        return h, f_opt
    else:
        h[i] = h_save[0]
        h[j] = h_save[1]
        return h, _exact_c5(h)

def _pair_search(h, deadline, rng, max_iters=50000, temp_start=0.0, temp_end=0.0):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _binary_threshold(h, frac=0.5):
    """Create a binary-like approximation of h."""
    n = len(h)
    sorted_h = np.sort(h)[::-1]
    k = int(n * frac)
    thresh = sorted_h[k - 1] if k > 0 else 0.5
    h_bin = (h >= thresh).astype(float)
    # Adjust to get exact sum
    deficit = n / 2.0 - np.sum(h_bin)
    if deficit > 0:
        # Need to add deficit: flip some zeros to ones or set partial
        zeros = np.where(h_bin < 0.5)[0]
        for idx in zeros[:int(deficit)]:
            h_bin[idx] = 1.0
        rem = deficit - int(deficit)
        if rem > 0 and len(zeros) > int(deficit):
            h_bin[zeros[int(deficit)]] = rem
    elif deficit < 0:
        excess = -deficit
        ones = np.where(h_bin > 0.5)[0]
        for idx in ones[:int(excess)]:
            h_bin[idx] = 0.0
        rem = excess - int(excess)
        if rem > 0 and len(ones) > int(excess):
            h_bin[ones[int(excess)]] = 1.0 - rem
    return _project_box_sum(h_bin, n / 2.0)

def _make_initial(n):
    """Create a deterministic initial feasible vector."""
    h = np.zeros(n)
    target = n / 2.0
    k = int(target)
    h[:k] = 1.0

    return h
