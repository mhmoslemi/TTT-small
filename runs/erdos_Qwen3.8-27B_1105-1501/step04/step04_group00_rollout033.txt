    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, lags):
    """Compute gradient rows for specific lags."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for idx, d in enumerate(lags):
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G * dx

def _lp_direction(h, scores, radius, top_k=None):
    """Find LP direction to reduce top-k scores."""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(n_lags)
    
    scores_k = scores[idx]
    G_k = _grad_rows(h, idx - (n-1))  # convert from full-correlate index to lag
    
    # LP: minimize t subject to G_k @ p - t <= -scores_k, sum(p)=0, |p|<=radius*(bound)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, n_eval=20):
    """Exact line search along direction p."""
    n = len(h)
    target = n / 2.0
    # Find max alpha
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    alpha_max = min(alpha_max, 2.0)
    
    # Coarse grid

    best_c5 = _exact_c5(h)
    best_a = 0.0

    for i in range(1, n_eval):

        c5_try = _exact_c5(h_try)
        vals.append(c5_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_a = alphas[i]
    
    # Golden section refinement
    if best_a > 0 and best_a < alpha_max:
        # Find bracket
        best_i = int(np.argmin(vals))
        lo = alphas[max(0, best_i - 1)]
        hi = alphas[min(n_eval - 1, best_i + 1)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(25):
            if hi - lo < 1e-12:

            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)

        opt_c5 = _exact_c5(_project(h + opt_a * p, target))
        if opt_c5 < best_c5:
            best_c5 = opt_c5
            best_a = opt_a

def _thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        # popcount(i) mod 2
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        h[i] = c % 2
    return _project(h, n / 2.0)

def _quasi_periodic(n, alpha):
    """Generate quasi-periodic binary sequence."""
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def _random_binary(n, rng):
    """Generate random binary sequence with exactly n/2 ones."""
    positions = rng.permutation(n)
    h = np.zeros(n)
    h[positions[:n//2]] = 1.0
    return h

def _resample(h, n_new):
    """Resample h to new resolution."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _local_optimize(h, deadline, top_k_list=None, radius_init=0.15, max_iters=5000):
    """LP-based local optimization."""
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5, 8, 15]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        p = _lp_direction(h, scores, radius, top_k=tk)
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, n_eval=20)
        
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.5)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 10 and radius < 0.02:
                radius = 0.05
    
    return best_h, best_c5

def _grad_descent_minimax(h, deadline, top_k=15, beta=30.0, max_iters=500):
    """Gradient-based minimax optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        w /= np.sum(w)
        
        # Get gradient of weighted scores
        lags = np.arange(-(n-1), n)
        G = _grad_rows(h, lags)
        grad = G.T @ w
        grad -= np.mean(grad)  # preserve sum
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        # Line search
        alpha = 0.1
        improved = False
        for _ in range(40):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-10:
                break
        
        if not improved:
            beta *= 1.3
            if beta > 10000:
                break
    
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    """Polish by trying pair exchanges."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    step = max(1, n // 24)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                    continue
                # Try moving mass from j to k and vice versa
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    
                    # Golden section for best a
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(10):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng, n_iters=30000):
    """Simulated annealing on binary sequences."""
    n = len(h_init)
    target = n // 2
    h = np.zeros(n)
    # Initialize from h_init by thresholding
    threshold = np.sort(h_init)[target]
    h = (h_init >= threshold).astype(float)
    if np.sum(h) != target:
        # Fix
        diffs = np.abs(h_init - threshold)
        order = np.argsort(diffs)
        # Set the closest to threshold
        h = np.zeros(n)
        h[order[target//2:]] = 1.0
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cool
    
    return best_h, best_c5

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

def _grad_rows(h, lags):
    n = len(h)

        x = i
