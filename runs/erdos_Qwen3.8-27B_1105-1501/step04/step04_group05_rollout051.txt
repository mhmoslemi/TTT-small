    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def grad_of_max(h, top_k=5):
    """Subgradient of max_q C_q(h), averaged over top_k lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k lags

        else:

            A = np.arange(0, n - k)
            B = np.arange(k, n)
            grad[A] += dx * (1.0 - h[B])
            grad[B] -= dx * h[A]
    return grad / len(top_idx)

    n = len(h)
    dx = 2.0 / n

    # Shift for numerical stability
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m)) - 1.0) / tau

def smoothed_grad(h, tau):

    n = len(h)
    dx = 2.0 / n

    m = np.max(corr)

    #             = sum(exp(tau*c_q) * grad c_q) / sum(exp(tau*c_q))
    tau_vals = tau * (corr - m)
    weights = np.exp(tau_vals)
    weights /= np.sum(weights)

            B = np.arange(k, n)

    g = grad_of_max(h, top_k)

                    best[i] = best_local
                    best[j] = p - best_local
                    best_val = best_local_val

def grad_top(h, k=5):
