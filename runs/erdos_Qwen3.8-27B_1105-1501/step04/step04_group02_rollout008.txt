    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max):
    n = len(h)
    best_t, best_val = 0.0, c5_cur
    n_grid = 24
    ts = np.linspace(0.0, alpha_max, n_grid)
    for t in ts:
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            v = _c5(h_t)
            if v < best_val:
                best_val, best_t = v, t
    lo = max(0.0, best_t - alpha_max / n_grid)
    hi = min(alpha_max, best_t + alpha_max / n_grid)
    if hi - lo < 1e-14:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = lo + (1 - gr) * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(h + c1 * p)
    f2 = _c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = lo + (1 - gr) * (hi - lo)
            f1 = _c5(h + c1 * p)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(h + c2 * p)
    opt = (lo + hi) / 2.0
    f_opt = _c5(h + opt * p)
    if f_opt < best_val:
        best_t, best_val = opt, f_opt
    return best_t, best_val

def _lp_local(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 70:
            break
        sc = _scores(h)
        p = _lp_direction(h, sc, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-8:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _proj(h, target)
            cur_c5 = _c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.12, 0.5)
        else:
            stall += 1
            radius *= 0.65
            if stall % 12 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _lattice_anneal(h0, deadline, m=8, tau0=0.003, tau_min=1e-6, decay=0.997, seed=42):
    """Stochastic lattice annealing that preserves sum exactly."""
    n = len(h0)
    target_sum = int(round(n * m / 2.0))
    if (n * m) % 2 != 0:
        m -= 1
        target_sum = int(round(n * m / 2.0))
        if m < 2:
            m = 2
            target_sum = int(round(n * m / 2.0))
    # Quantize
    q = np.round(h0 * m).astype(int)
    q = np.clip(q, 0, m).astype(int)
    # Adjust sum
    cur_sum = int(np.sum(q))
    diff = cur_sum - target_sum
    if diff > 0:
        idx = np.where(q > 0)[0]
        rng = np.random.default_rng(seed)
        rng.shuffle(idx)
        for i in idx[:diff]:
            q[i] -= 1
    elif diff < 0:
        idx = np.where(q < m)[0]
        rng = np.random.default_rng(seed)
        rng.shuffle(idx)
        for i in idx[:(-diff)]:
            q[i] += 1
    dx = 2.0 / n
    h_cur = q.astype(float) / m
    cur_c5 = _c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    T = tau0
    rng = np.random.default_rng(seed + 1)
    # Precompute which lags involve which positions (cache for speed)
    lag_pos = {}
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m_lag = n - d
            lag_pos[r] = (list(range(d, d + m_lag)), list(range(m_lag)))
        else:
            s = -d
            m_lag = n - s
            lag_pos[r] = (list(range(m_lag)), list(range(s, s + m_lag)))
    # Top-K tracking
    K_active = 6
    moves_done = 0
    stall = 0
    last_best_move = 0
    max_stall = 2000
    while time.time() < deadline:
        # Recompute active lags every 500 moves
        if moves_done % 500 == 0 or moves_done == 0:
            sc = _scores(h_cur)
            top_idx = np.argsort(sc)[::-1][:K_active]
            # Build pressure vector: how much each coordinate contributes to top lags
            pressure = np.zeros(n)
            for r in top_idx:
                sc_r = sc[r]
                w_r = np.exp((sc_r - sc[top_idx[0]]) * 50.0)
                inc_pos, dec_pos = lag_pos[r]
                for ip in inc_pos:
                    pressure[ip] += w_r * (1.0 - h_cur[dec_pos[ip]])
                for dp in dec_pos:
                    pressure[dp] -= w_r * h_cur[inc_pos[dp]]
            # Normalize pressure for sampling
            p_max = float(np.max(np.abs(pressure)))
            if p_max > 1e-10:
                pressure /= p_max
        # Propose move
        moves_done += 1
        if rng.random() < 0.3:
            # Uniform random pair
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
        else:
            # Biased: increase low-pressure, decrease high-pressure
            inc_probs = np.exp(-np.clip(pressure, -1, 1) * 2.0)
            inc_probs[q >= m] = 0
            s_ip = float(np.sum(inc_probs))
            if s_ip < 1e-10:
                i = int(rng.integers(0, n))
            else:
                inc_probs /= s_ip
                cum = np.cumsum(inc_probs)
                i = int(np.searchsorted(cum, rng.random()))
                if i >= n:
                    i = n - 1
            dec_probs = np.exp(np.clip(pressure, -1, 1) * 2.0)
            dec_probs[q <= 0] = 0
            dec_probs[i] = 0
            s_dp = float(np.sum(dec_probs))
            if s_dp < 1e-10:
                j = int(rng.integers(0, n))
                if j == i:
                    j = (j + 1) % n
            else:
                dec_probs /= s_dp
                cum = np.cumsum(dec_probs)
                j = int(np.searchsorted(cum, rng.random()))
                if j >= n:
                    j = n - 1
                if j == i:
                    j = (j + 1) % n
        if i >= n or j >= n or i == j:
            continue
        if q[i] >= m or q[j] <= 0:
            continue
        # Make move
        q[i] += 1
        q[j] -= 1
        h_new = q.astype(float) / m
        c5_new = _c5(h_new)
        delta = c5_new - cur_c5
        # Metropolis accept
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            h_cur = h_new
            cur_c5 = c5_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = q.astype(float) / m
                stall = 0
                last_best_move = moves_done
            else:
                stall += 1
        else:
            # Reject, undo
            q[i] -= 1
            q[j] += 1
            stall += 1
        # Temperature schedule
        T = max(tau_min, T * decay)
        if stall > max_stall:
            # Reheat
            T = min(tau0 * 0.5, T * 3.0)
            stall = 0
    return best_h, best_c5

def _pair_refine(h, deadline, K_focus=8):
    """Focused pair line search based on active lags."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    pass_count = 0
    while time.time() < deadline:
        pass_count += 1
        sc = _scores(h)
        top_idx = np.argsort(sc)[::-1][:K_focus]
        # For each active lag, get the positions
        candidate_pairs = []
        for r in top_idx:
            d = r - (n - 1)
            if d >= 0:
                m_l = n - d
                inc = list(range(d, d + m_l))
                dec = list(range(m_l))
            else:
                s = -d
                m_l = n - s
                inc = list(range(m_l))
                dec = list(range(s, s + m_l))
            # Suggest pairs: (inc_pos, dec_pos) to reduce this lag
            # Sample a few
            n_samp = min(4, len(inc))
            for _ in range(n_samp):
                ii = int(rng.integers(0, len(inc)))
                jj = int(rng.integers(0, len(dec)))
                candidate_pairs.append((inc[ii], dec[jj]))
        # Add some random pairs
        for _ in range(6):
            ii = int(rng.integers(0, n))
            jj = int(rng.integers(0, n))
            if ii != jj:
                candidate_pairs.append((ii, jj))
        improved = False
        for (i, j) in candidate_pairs:
            if time.time() > deadline:
                break
            if h[i] >= 1.0 - 1e-9 or h[j] <= 1e-9:
                continue
            t_max = min(1.0 - h[i], h[j])
            if t_max < 1e-10:
                continue
            # Line search
            best_t, best_val = 0.0, cur_c5
            n_g = 16
            ts = np.linspace(0.0, t_max, n_g)
            for t in ts:
                if t < 1e-12:
                    continue
                h_t = h.copy()
                h_t[i] += t
                h_t[j] -= t
                v = _c5(h_t)
                if v < best_val:
                    best_val, best_t = v, t
            if best_t > 1e-12 and best_val < cur_c5 - 1e-14:
                # Golden section refine
                lo = max(0.0, best_t - t_max / n_g)
                hi = min(t_max, best_t + t_max / n_g)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = lo + (1 - gr) * (hi - lo)
                c2 = lo + gr * (hi - lo)
                h1 = h.copy(); h1[i] += c1; h1[j] -= c1
                h2 = h.copy(); h2[i] += c2; h2[j] -= c2
                f1, f2 = _c5(h1), _c5(h2)
                for _ in range(25):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi, c2, f2 = c2, c1, f1
                        c1 = lo + (1 - gr) * (hi - lo)
                        h1 = h.copy(); h1[i] += c1; h1[j] -= c1
                        f1 = _c5(h1)
                    else:
                        lo, c1, f1 = c1, c2, f2
                        c2 = lo + gr * (hi - lo)
                        h2 = h.copy(); h2[i] += c2; h2[j] -= c2
                        f2 = _c5(h2)
                opt = (lo + hi) / 2.0
                h_opt = h.copy(); h_opt[i] += opt; h_opt[j] -= opt
                f_opt = _c5(h_opt)
                if f_opt < best_val:
                    best_val, best_t = f_opt, opt
                h[i] += best_t
                h[j] -= best_t
                h = _proj(h, target)
                cur_c5 = _c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not improved and pass_count > 80:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _multi_res(h_best, n_best, deadline, budget_s):
    best_h = h_best.copy()
    best_c5 = _c5(h_best)
    best_n = n_best
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, 12, -12, 16, -16, 20, -20]
    tried = set()
    time_per = max(budget_s * 0.012, 2.0)
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _c5(h_r)
        sub_dl = min(deadline, time.time() + time_per * 0.5)
        h_r, c5_r = _grad_opt(h_r, sub_dl, tau=0.03, lr=0.012, beta=0.88)
        if time.time() < deadline:
            sub_dl2 = min(deadline, time.time() + time_per * 0.3)
            h_r, c5_r = _lp_local(h_r, sub_dl2, radius_init=0.1)
        if time.time() < deadline:
            sub_dl3 = min(deadline, time.time() + time_per * 0.2)
            h_r, c5_r = _pair_refine(h_r, sub_dl3, K_focus=6)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    return best_h, best_c5, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    remaining = lambda: max(0, t0 + budget_s - time.time())

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _proj(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: Quick gradient (8% budget)
    d1 = t0 + budget_s * 0.08
    h, c5 = _grad_opt(h, d1, tau=0.02, lr=0.02, beta=0.88)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Lattice annealing (25% budget) - NEW
    d2 = t0 + budget_s * 0.33
    time_left = d2 - time.time()
    if time_left > 5:
        # Try multiple m values
        m_values = [4, 6, 8, 12]
        time_per_m = time_left / len(m_values)
        for mi, m in enumerate(m_values):
            if time.time() > d2 - 2:
                break
            sub_dl = min(d2 - 1, time.time() + time_per_m * 0.8)
            h_lat, c5_lat = _lattice_anneal(best_h, sub_dl, m=m,
                                             tau0=0.002, tau_min=5e-7,
                                             decay=0.9965, seed=seed + mi * 1000)
            if c5_lat < best_c5 - 1e-12:
                best_c5 = c5_lat
                best_h = h_lat.copy()
            elif time.time() < d2 - 1:
                # Try refining the lattice result in continuous space
                sub_dl2 = min(d2 - 0.5, time.time() + time_per_m * 0.15)
                h_try, c5_try = _grad_opt(h_lat, sub_dl2, tau=0.05, lr=0.008, beta=0.9)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_h = h_try.copy()

    # Phase 3: LP local search (17% budget)
    d3 = t0 + budget_s * 0.50
    h, c5 = _lp_local(best_h, d3, radius_init=0.15)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Pair refine (13% budget)
    d4 = t0 + budget_s * 0.63
    h, c5 = _pair_refine(best_h, d4, K_focus=8)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Multi-resolution (15% budget)
    d5 = t0 + budget_s * 0.78
    best_h, best_c5, best_n = _multi_res(best_h, best_n, d5, budget_s)

    # Phase 6: Second round of lattice at best resolution (10%)
    d6 = t0 + budget_s * 0.88
    if time.time() < d6 and remaining() > 10:
        n_cur = len(best_h)
        for m in [6, 10, 16]:
            if time.time() > d6 - 2:
                break
            sub_dl = min(d6 - 1, time.time() + budget_s * 0.03)
            h_lat, c5_lat = _lattice_anneal(best_h, sub_dl, m=m,
                                             tau0=0.001, tau_min=2e-7,
                                             decay=0.996, seed=seed + 5000 + m)
            if c5_lat < best_c5 - 1e-12:
                best_c5 = c5_lat
                best_h = h_lat.copy()

    # Phase 7: Final polish (10%)
    d7 = t0 + budget_s * 0.98
    h, c5 = _lp_local(best_h, min(d7, time.time() + budget_s * 0.03), radius_init=0.05)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d7:
        h, c5 = _pair_refine(best_h, min(d7, time.time() + budget_s * 0.03), K_focus=10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < d7:
        h, c5 = _grad_opt(best_h, min(d7, time.time() + budget_s * 0.015),
                          tau=0.5, lr=0.003, beta=0.92)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final feasibility check
    best_h = _proj(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, len(best_h)
