def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n/2.0
    
    def soft_obj(x):
        omh = 1.0 - x
        sc = np.correlate(x, omh, mode='full') * (2.0/n)
        mx = float(np.max(sc))
        es = np.exp((sc - mx) / tau)
        return mx + tau * np.log(float(np.sum(es)))
    
    def soft_grad(x):
        omh = 1.0 - x
        sc = np.correlate(x, omh, mode='full') * (2.0/n)
        mx = float(np.max(sc))
        es = np.exp((sc - mx) / tau)
        Z = float(np.sum(es))
        w = es / Z
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-15: continue
            d = k - (n-1)
            j1 = max(0, -d); j1e = min(n-1, n-1-d)
            if j1 <= j1e:
                g[j1:j1e+1] += wk * (1.0 - x[j1+d:j1e+1+d])
            j2 = max(0, d); j2e = min(n-1, n+d-1)
            if j2 <= j2e:
                g[j2:j2e+1] -= wk * x[j2-d:j2e+1-d]
        return (2.0/n) * g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(soft_obj, h0, jac=soft_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        pass
