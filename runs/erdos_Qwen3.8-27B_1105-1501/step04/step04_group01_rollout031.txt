```python
import numpy as np
from scipy.optimize import linprog
import time

def _waterfill(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target} using bisection."""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    # Find theta such that sum(clip(v-theta, 0, 1)) = target
    lo, hi = float(v.min()) - 1, float(v.max()) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo+hi)/2
    x = np.clip(v - theta, 0, 1)
    # Fix residual
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0: mask = x < 1
        else: mask = x > 0
        if mask.any(): x[mask] += res / mask.sum()
        x = np.clip(x, 0, 1)
    return x

def _c5(h):
    n = len(h)
    dx = 2.0/n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def _line_search(h, d, dx, t_max):
    """Find best step along direction d. Returns (best_t, best_c5)."""
    n = len(h)
    S0 = np.correlate(h, 1-h, mode='full')
    T = np.correlate(d, 1-h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Feasibility bounds
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = t_max
    t_lo = -t_max
    if pos.any(): t_hi = min(t_hi, float(np.min((1-h[pos])/d[pos])))
    if neg.any(): t_lo = max(t_lo, float(np.max(h[neg]/d[neg])))
    
    if t_hi <= t_lo: return 0.0, float(np.max(S0)*dx)
    
    def obj(t):
        return float(np.max(S0 + t*T - t*t*Q)) * dx
    
    # Golden section search
    gr = (np.sqrt(5)-1)/2
    a, b = t_lo, t_hi
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1, f2 = obj(c1), obj(c2)
    for _ in range(60):
        if b - a < 1e-12: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = obj(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = obj(c2)
    t_opt = (a+b)/2
    f_opt = obj(t_opt)
    # Also check endpoints
    for t_chk in [t_lo, t_hi, 0.0]:
        if t_lo <= t_chk <= t_hi and obj(t_chk) < f_opt:
            t_opt, f_opt = t_chk, obj(t_chk)
    return t_opt, f_opt
```

def _waterfill(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(x.sum())
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = float(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
