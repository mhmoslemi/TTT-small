def _c5(h):

def _corr_grad(h):

    where corr[r] = sum_k h[k] * (1-h[k+s]), s = r-(n-1)

def _lse_step(h, n, alpha, maxiter=200):
    """LSE smoothing step."""
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    
    def _obj_and_jac(x):

        vals = np.clip(vals, -500, 500)

            m1 = (gs >= 0) & (gs < n)
            grad[m1] -= w[r] * h[gs[m1]]

        grad -= np.mean(grad)

def _pair_move(h, n, rng):

def _lp_direction(h, n, radius):
    """Find LP descent direction to reduce max correlation."""
    from scipy.optimize import linprog
    
    corr = _corr_vec(h)
    G = _corr_grad(h)  # G[r, j] = d(corr[r])/dh[j]
    
    # Minimize t subject to t >= corr[r] + G[r,:] @ p for all r
    # i.e., G[r,:] @ p - t <= -corr[r]
    # sum(p) = 0, |p[j]| <= radius
    
    L = 2 * n - 1
    # Variables: p (n) + t (1) = n+1
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corr
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-10:
                return p
    except Exception:
        pass
    return None

def _resample(h, n_new):
    """Resample h to new resolution."""
    n = len(h)
    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

    """Generate a block pattern with given number of blocks."""

        val = rng.random()
        if val < 0.5:
            h[j*block_size:(j+1)*block_size] = 1.0
    return _project(h, n * 0.5)

                    break
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-14:
                    break
    return x


def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _corr_grad(h):
    """G[r,j] = d(corr[r])/dh[j] = (1-h[j-s])*[0<=j-s<n] - h[j+s]*[0<=j+s<n]"""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    j = np.arange(n)
    for r in range(L):
        s = r - (n - 1)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        G[r, m1] += (1.0 - h[gs[m1]])
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        G[r, m2] -= h[ps[m2]]
    return G


def _epi_obj(z, n):
    return z[n]


def _epi_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _eq_jac(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _ineq_f(z, n):
    h, t = z[:n], z[n]
    return t - np.correlate(h, 1.0 - h, mode='full')


def _ineq_jac(z, n):
    """J[r,j] = df[r]/dh[j] = h[j+s]*[0<=j+s<n] - (1-h[j-s])*[0<=j-s<n]"""
    h = z[:n]
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    j = np.arange(n)
    for r in range(L):
        s = r - (n - 1)
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        J[r, m2] += h[ps[m2]]
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        J[r, m1] -= (1.0 - h[gs[m1]])
        J[r, n] = 1.0
    return J


def _slsqp_solve(h_init, n, maxiter=300, ftol=1e-15):
    h = _project(np.asarray(h_init, dtype=float), n * 0.5)
    corr = _corr_vec(h)
    t0 = float(np.max(corr)) + 1e-3
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_jac, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _c5(x)
    except Exception:
        return h, _c5(h)


def _lse_obj_jac(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = _corr_vec(h)
    ms = float(np.max(corr))
    vals = alpha * (corr - ms)
    vals = np.clip(vals, -500, 500)
    exp_v = np.exp(vals)
    s = float(np.sum(exp_v))
    f = ms + (1.0 / alpha) * np.log(s)
    w = exp_v / s
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        s_shift = r - (n - 1)
        gs = j - s_shift
        m1 = (gs >= 0) & (gs < n)
        grad[m1] += w[r] * (1.0 - h[gs[m1]])
        ps = j + s_shift
        m2 = (ps >= 0) & (ps < n)
        grad[m2] -= w[r] * h[ps[m2]]
    grad -= np.mean(grad)
    return f, grad


def _lse_step(h, n, alpha, maxiter=200):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj_jac, h.copy(), args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'maxls': 30, 'iprint': 0})
        x = _project(res.x, target)
        return x, _c5(x)
    except Exception:
        return h.copy(), _c5(h)


def _lp_direction(h, n, radius):
    corr = _corr_vec(h)
    G = _corr_grad(h)
    L = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corr
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-12:
                return p
    except Exception:
        pass
    return None


def _lp_line_search(h, p, n):
    target = n * 0.5
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-8)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(_project(h + c1 * p, target))
    f2 = _c5(_project(h + c2 * p, target))
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(_project(h + c1 * p, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(_project(h + c2 * p, target))
    a = (lo + hi) / 2.0
    h_new = _project(h + a * p, target)
    c_new = _c5(h_new)
    if c_new < _c5(h):
        return h_new, c_new
    return h.copy(), _c5


def _lp_loop(h, n, deadline, max_iters=2000, stall_limit=30):
    h = h.copy()
    cur_c = _c5(h)
    best_h, best_c = h.copy(), cur_c
    radius = 0.1
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        p = _lp_direction(h, n, radius)
        if p is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        h_new, c_new = _lp_line_search(h, p, n)
        if c_new < cur_c - 1e-13:
            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.4)
        else:
            stall += 1
            radius *= 0.8
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c


def _subgrad_step(h, n, step_size):
    corr = _corr_vec(h)
    r_max = int(np.argmax(corr))
    s = r_max - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    g[m1] += (1.0 - h[gs[m1]])
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    g[m2] -= h[ps[m2]]
    g -= np.mean(g)
    norm = np.linalg.norm(g)
    if norm < 1e-15:
        return h.copy(), _c5(h)
    g = g / norm
    target = n * 0.5
    for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
        h_new = _project(h - frac * step_size * g, target)
        c_new = _c5(h_new)
        if c_new < _c5(h):
            return h_new, c_new
    return h.copy(), _c5(h)


def _subgrad_loop(h, n, deadline, initial_step=0.1, decay=0.97):
    h = h.copy()
    cur_c = _c5(h)
    best_h, best_c = h.copy(), cur_c
    step = initial_step
    stall = 0
    while time.time() < deadline and stall < 100:
        h_new, c_new = _subgrad_step(h, n, step)
        if c_new < cur_c - 1e-14:
            h = h_new
            cur_c = c_new
            stall = 0
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        else:
            stall += 1
            step *= decay
    return best_h, best_c


def _coord_descent(h, n, deadline):
    """Try setting each coordinate to 0 or 1 (re-projecting)."""
    h = h.copy()
    cur_c = _c5(h)
    best_h, best_c = h.copy(), cur_c
    target = n * 0.5
    order = list(range(n))
    total_iters = 0
    while time.time() < deadline:
        improved = False
        for j in order:
            total_iters += 1
            if time.time() > deadline:
                break
            for val in [0.0, 1.0]:
                if abs(h[j] - val) < 0.01:
                    continue
                h_try = h.copy()
                h_try[j] = val
                h_try = _project(h_try, target)
                c_try = _c5(h_try)
                if c_try < cur_c - 1e-13:
                    h = h_try
                    cur_c = c_try
                    improved = True
                    if cur_c < best_c:
                        best_c, best_h = cur_c, h.copy()
        if not improved:
            break
    return best_h, best_c


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _gen_sturmian(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i + 1) * alpha) % 1.0 < alpha else 0.0
    return _project(h, n * 0.5)


def _gen_sinusoidal(n, freq, phase):
    x = np.arange(n) / n
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x + phase)
    return _project(h, n * 0.5)


def _gen_blocks(n, n_blocks, rng):
    h = np.zeros(n)
    bs = n // n_blocks
    for j in range(n_blocks):
        if rng.random() < 0.5:
            h[j * bs:(j + 1) * bs] = 1.0
    return _project(h, n * 0.5)


def _refine_point(h, n, deadline):
    """Multi-method refinement of a single point."""
    target = n * 0.5
    best_h = h.copy()
    best_c = _c5(h)

    # LSE sweep
    for alpha in [10.0, 30.0, 100.0]:
        if time.time() > deadline:
            break
        h_l, c_l = _lse_step(best_h, n, alpha, maxiter=100)
        if c_l < best_c:
            best_c, best_h = c_l, h_l.copy()

    # SLSQP
    if time.time() < deadline - 3:
        h_s, c_s = _slsqp_solve(best_h, n, maxiter=300, ftol=1e-15)
        if c_s < best_c:
            best_c, best_h = c_s, h_s.copy()

    # LP loop
    if time.time() < deadline - 5:
        h_lp, c_lp = _lp_loop(best_h, n, min(deadline, time.time() + 30),
                              max_iters=1500, stall_limit=30)
        if c_lp < best_c:
            best_c, best_h = c_lp, h_lp.copy()

    # Subgradient
    if time.time() < deadline - 3:
        h_sg, c_sg = _subgrad_loop(best_h, n, min(deadline, time.time() + 10),
                                   initial_step=0.05)
        if c_sg < best_c:
            best_c, best_h = c_sg, h_sg.copy()

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: LSE warmup on parent (8% budget)
    d1 = t0 + budget_s * 0.08
    for alpha in [5.0, 15.0, 40.0, 100.0, 300.0]:
        if time.time() > d1:
            break
        h_l, c_l = _lse_step(h, n0, alpha, maxiter=100)
        if c_l < best_c5:
            best_c5, best_h = c_l, h_l.copy()

    # Phase 2: SLSQP from parent (12% budget)
    d2 = t0 + budget_s * 0.20
    for att in range(3):
        if time.time() > d2:
            break
        start = best_h.copy()
        if att > 0:
            start = _project(start + rng.normal(0, 0.015, n0), target)
        h_s, c_s = _slsqp_solve(start, n0, maxiter=500, ftol=1e-16)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s.copy()

    # Phase 3: LP polishing (12% budget)
    d3 = t0 + budget_s * 0.32
    h_lp, c_lp = _lp_loop(best_h, n0, d3, max_iters=3000, stall_limit=40)
    if c_lp < best_c5:
        best_c5, best_h = c_lp, h_lp.copy()

    # Phase 4: Multi-start with diverse patterns (20% budget)
    d4 = t0 + budget_s * 0.52
    alpha_golden = (np.sqrt(5.0) - 1.0) / 2.0
    candidates = []
    candidates.append(h0.copy())
    candidates.append(h0[::-1].copy())
    candidates.append(_gen_sturmian(n0, alpha_golden))
    candidates.append(_gen_sturmian(n0, 0.382))
    candidates.append(_gen_sturmian(n0, 0.618))
    for freq in [1.5, 2.0, 2.5, 3.0]:
        candidates.append(_gen_sinusoidal(n0, freq, rng.random() * 2.0 * np.pi))
    for nb in [4, 6, 8, 12]:
        candidates.append(_gen_blocks(n0, nb, rng))

    per_candidate_time = max(3, (d4 - time.time()) / max(1, len(candidates)))
    for cand in candidates:
        if time.time() > d4:
            break
        h_t = _project(cand, target)
        c_t = _c5(h_t)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t.copy()
        sub_deadline = min(d4, time.time() + per_candidate_time)
        h_r, c_r = _refine_point(h_t, n0, sub_deadline)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()

    # Phase 5: LP + subgradient on best (10% budget)
    d5 = t0 + budget_s * 0.62
    h_lp, c_lp = _lp_loop(best_h, n0, min(d5, time.time() + 20),
                          max_iters=2000, stall_limit=35)
    if c_lp < best_c5:
        best_c5, best_h = c_lp, h_lp.copy()
    h_sg, c_sg = _subgrad_loop(best_h, n0, min(d5, time.time() + 15),
                               initial_step=0.04, decay=0.95)
    if c_sg < best_c5:
        best_c5, best_h = c_sg, h_sg.copy()
    h_l, c_l = _lse_step(best_h, n0, 200.0, maxiter=100)
    if c_l < best_c5:
        best_c5, best_h = c_l, h_l.copy()

    # Phase 6: Resolution search (20% budget)
    d6 = t0 + budget_s * 0.82
    for dn in [16, 32, -16, 64, -32, 48]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 256 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        c_r = _c5(h_r)
        if c_r < best_c5:
            best_c5, best_h, best_n = c_r, h_r.copy(), n_try
        sub_d = min(d6, time.time() + max(5, budget_s * 0.03))
        h_rf, c_rf = _refine_point(h_r, n_try, sub_d)
        if c_rf < best_c5:
            best_c5, best_h, best_n = c_rf, h_rf.copy(), n_try

    # Phase 7: Coordinate descent (8% budget)
    d7 = t0 + budget_s * 0.90
    h_cd, c_cd = _coord_descent(best_h, best_n, d7)
    if c_cd < best_c5:
        best_c5, best_h = c_cd, h_cd.copy()

    # Phase 8: Final polish (10% budget)
    d8 = t0 + budget_s * 0.99
    h_f, c_f = _slsqp_solve(best_h, best_n, maxiter=400, ftol=1e-16)
    if c_f < best_c5:
        best_c5, best_h = c_f, h_f.copy()
    h_lp, c_lp = _lp_loop(best_h, best_n, min(d8, time.time() + 15),
                          max_iters=2000, stall_limit=50)
    if c_lp < best_c5:
        best_c5, best_h = c_lp, h_lp.copy()
    h_l, c_l = _lse_step(best_h, best_n, 500.0, maxiter=200)
    if c_l < best_c5:
        best_c5, best_h = c_l, h_l.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
