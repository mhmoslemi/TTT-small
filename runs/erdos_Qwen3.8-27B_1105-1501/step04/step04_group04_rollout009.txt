def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(1000):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _grad_focused(h, beta, top_k):
    """Gradient of smoothed max focusing on top_k lags."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    
    # Get top-k indices
    if top_k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(F, -top_k)[-top_k:]
    
    F_top = F[idx]
    F_shift = F_top - F_top.max()
    w_top = np.exp(beta * F_shift)
    w_top = w_top / w_top.sum()
    
    grad = np.zeros(n)
    for j, r_idx in enumerate(idx):
        wr = w_top[j]
        if wr < 1e-15:
            continue
        r = r_idx - (n - 1)
        if r >= 0:
            # dF_r/dh[j] for j in [0, n-r): contribution (1-h[j+r])*dx
            if n > r:
                grad[:n-r] += wr * dx * (1.0 - h[r:])

                grad[r:] -= wr * dx * h[:n-r]
        else:
            s = -r
            if n > s:
                grad[s:] += wr * dx * (1.0 - h[:n-s])
                grad[:n-s] -= wr * dx * h[s:]
    
    return grad

def _sim_anneal(h, deadline, rng, n_iters=50000, T_start=0.05, T_end=0.001):
    """Simulated annealing with pair moves."""
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    T = T_start
    cooling = (T_end / T_start) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T *= cooling
        
        # Pick two positions to swap mass
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(-0.15, 0.15)
        ni = cur[i] + d
        nj = cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        
        cur[i] = ni
        cur[j] = nj
        new_c5 = _exact_c5(cur)
        
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()

            # Metropolis acceptance
            delta = new_c5 - cur_c5
            if delta < T and rng.random() < math.exp(-delta / max(T, 1e-15)):
                cur_c5 = new_c5
            else:
                cur[i] = ni - d
                cur[j] = nj + d
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)
