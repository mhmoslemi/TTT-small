import numpy as np

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(x) if diff > 0 else np.argsort(-x)
        for i in idx:
            if diff > 0 and x[i] >= 1.0 - 1e-15:
                continue
            if diff < 0 and x[i] <= 1e-15:
                continue
            room = (1.0 - x[i]) if diff > 0 else x[i]
            step = min(room, abs(diff))
            x[i] += step if diff > 0 else -step
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
    return x

    return np.sum(z[:n]) - n * 0.5


def _eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def _ineq_fun(z, k, n):
    h = z[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return z[n] - corr[k]


def _ineq_jac(z, k, n):
    h = z[:n]
    s = k - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    g[m1] = (1.0 - h[gs[m1]])
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    g[m2] -= h[ps[m2]]
    out = np.zeros(n + 1)
    out[:n] = -g
    out[n] = 1.0
    return out


def _active_set_optimize(h_init, n, deadline, K_init=8, max_rounds=25, maxiter=200):
    h = _project(np.asarray(h_init, dtype=float), n * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = 2 * n - 1
    active = sorted(np.argsort(-corr)[:K_init].tolist())

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
        for k in active:
            cons.append({'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac, 'args': (k, n)})
        cur_corr = np.correlate(h, 1.0 - h, mode='full')
        t0 = float(np.max(cur_corr[active])) + 1e-5
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
        try:
            res = minimize(_obj_fn, z0, jac=_obj_grad, args=(n,),
                          bounds=bounds, constraints=cons, method='SLSQP',
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            h_new = _project(res.x[:n], n * 0.5)
        except Exception:
            break
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        t_val = float(np.max(corr_new))
        violated = [k for k in range(nc) if k not in active and corr_new[k] > t_val - 1e-6]
        if not violated:
            break
        violated.sort(key=lambda k: -corr_new[k])
        for k in violated[:3]:
            if k not in active:
                active.append(k)
        if len(active) > 30:
            active = sorted(np.argsort(-corr_new)[:30].tolist())

    return best_h, best_c5


def _subgradient_descent(h_init, n, deadline, max_iter=5000):
    h = _project(np.asarray(h_init, dtype=float), n * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_sizes = np.array([0.5, 0.3, 0.2, 0.1, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001])
    ss_idx = 2
    j = np.arange(n)
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(-corr)[:3]
        g = np.zeros(n)
        for kk in top_k:
            s = kk - (n - 1)
            gs = j - s
            m1 = (gs >= 0) & (gs < n)
            g[m1] += (1.0 - h[gs[m1]])
            ps = j + s
            m2 = (ps >= 0) & (ps < n)
            g[m2] -= h[ps[m2]]
        g /= len(top_k)
        g -= np.mean(g)
        norm = np.linalg.norm(g)
        if norm < 1e-15:
            break
        g = g / norm
        improved = False
        for ii in range(ss_idx, len(step_sizes)):
            ss = step_sizes[ii]
            h_try = _project(h - ss * g, n * 0.5)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                h = h_try
                best_h = h_try.copy()
                best_c5 = c5_try
                improved = True
                ss_idx = max(0, ii - 1)
                break
        if improved:
            stall = 0
        else:
            stall += 1
            ss_idx = min(len(step_sizes) - 1, ss_idx + 1)
            if stall > 30:
                h = _project(h + 0.02 * np.random.randn(n), n * 0.5)
                stall = 0
                ss_idx = 3

    return best_h, best_c5


def _coord_descent(h_init, n, deadline, max_rounds=50):
    h = _project(np.asarray(h_init, dtype=float), n * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    j = np.arange(n)

    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        nc = 2 * n - 1
        for i in range(n):
            if time.time() > deadline:
                break
            c = np.zeros(nc)
            for r in range(nc):
                s = r - (n - 1)
                val = 0.0
                if 0 <= i - s < n:
                    val += (1.0 - h[i - s])
                if 0 <= i + s < n:
                    val -= h[i + s]
                c[r] = val
            base = corr - h[i] * c
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = float(np.max(base + m1 * c))
                f2 = float(np.max(base + m2 * c))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            v_new = (lo + hi) / 2.0
            if abs(v_new - h[i]) < 1e-9:
                continue
            v_old = h[i]
            h[i] = v_new
            h = _project(h, n * 0.5)
            c5_new = _exact_c5(h)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                corr = np.correlate(h, 1.0 - h, mode='full')
            else:
                h = _project(np.concatenate([h[:i], [v_old], h[i+1:]]), n * 0.5)
        if not improved:
            break

    return best_h, best_c5


def _gen_binary(n, seed):
    rng = np.random.default_rng(seed)
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def _gen_sinusoidal(n, freq, phase):
    x = np.arange(n) / n
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x + phase)
    return _project(h, n * 0.5)


def _gen_structured(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    block = n // 4
    for b in range(4):
        val = 0.5 + 0.3 * rng.standard_normal()
        h[b * block:(b + 1) * block] = val
    return _project(h, n * 0.5)


def _refine_multi(h, n, deadline):
    """Multi-method refinement: active-set -> subgrad -> coord -> active-set."""
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    d1 = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    h1, c5_1 = _active_set_optimize(best_h, n, d1, K_init=8, max_rounds=20, maxiter=150)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    d2 = min(deadline, time.time() + (deadline - time.time()) * 0.3)
    h2, c5_2 = _subgradient_descent(best_h, n, d2, max_iter=3000)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    d3 = min(deadline, time.time() + (deadline - time.time()) * 0.2)
    h3, c5_3 = _coord_descent(best_h, n, d3, max_rounds=30)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    if time.time() < deadline:
        h4, c5_4 = _active_set_optimize(best_h, n, deadline, K_init=5, max_rounds=15, maxiter=200)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    h_parent = np.asarray(initial_h_values, dtype=float)
    n_parent = len(h_parent)

    best_h = h_parent.copy()
    best_c5 = _exact_c5(h_parent)
    best_n = n_parent

    # Phase 1: Refine parent with active-set (20% budget)
    d1 = t_start + budget_s * 0.20
    h1, c5_1 = _refine_multi(best_h, best_n, d1)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Diverse starts with active-set (35% budget)
    d2 = t_start + budget_s * 0.55
    starts = []
    for s in range(3):
        starts.append(_gen_binary(n_parent, seed + 200 + s))
    starts.append(_gen_sinusoidal(n_parent, 2, 0.0))
    starts.append(_gen_sinusoidal(n_parent, 3, np.pi / 3))
    starts.append(_gen_sinusoidal(n_parent, 5, 0.0))
    starts.append(_gen_sinusoidal(n_parent, 7, np.pi / 7))
    starts.append(_gen_structured(n_parent, seed + 300))
    starts.append(_gen_structured(n_parent, seed + 301))
    for s in range(2):
        hr = rng.uniform(0.2, 0.8, n_parent)
        starts.append(_project(hr, n_parent * 0.5))
    for s in range(2):
        hp = _project(h_parent + 0.08 * rng.standard_normal(n_parent), n_parent * 0.5)
        starts.append(hp)

    n_starts = len(starts)
    for i in range(n_starts):
        if time.time() > d2:
            break
        dl = min(d2, t_start + (i + 1.5) * (d2 - t_start) / n_starts)
        h_try, c5_try = _active_set_optimize(starts[i], n_parent, dl, K_init=6, max_rounds=12, maxiter=120)
        if c5_try < best_c5 - 1e-10:
            best_h, best_c5 = h_try, c5_try
            # Follow up with subgradient
            h_f, c5_f = _subgradient_descent(best_h, best_n, min(d2, time.time() + 5), max_iter=500)
            if c5_f < best_c5:
                best_h, best_c5 = h_f, c5_f

    # Phase 3: Refine best with different K values (10% budget)
    d3 = t_start + budget_s * 0.65
    for K in [4, 6, 12, 16, 20]:
        if time.time() > d3:
            break
        h_try, c5_try = _active_set_optimize(best_h, best_n, min(d3, time.time() + 20), K_init=K, max_rounds=15, maxiter=150)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 4: Resolution search (15% budget)
    d4 = t_start + budget_s * 0.80
    res_list = [best_n - 8, best_n - 4, best_n + 4, best_n + 8, best_n - 16, best_n + 16]
    for n_try in res_list:
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        if time.time() > d4:
            break
        x_old = (np.arange(best_n) + 0.5) / best_n
        x_new = (np.arange(n_try) + 0.5) / n_try
        h_res = np.interp(x_new, x_old, best_h)
        h_res = _project(h_res, n_try * 0.5)
        dl = min(d4, time.time() + budget_s * 0.015)
        h_try, c5_try = _refine_multi(h_res, n_try, dl)
        if c5_try < best_c5 - 1e-10:
            best_h, best_c5, best_n = h_try, c5_try, n_try

    # Phase 5: Coordinate descent polish (10% budget)
    d5 = t_start + budget_s * 0.90
    h_cd, c5_cd = _coord_descent(best_h, best_n, d5, max_rounds=80)
    if c5_cd < best_c5:
        best_h, best_c5 = h_cd, c5_cd

    # Phase 6: Final active-set + subgradient (10% budget)
    d6 = t_start + budget_s * 0.99
    h_f, c5_f = _active_set_optimize(best_h, best_n, d6, K_init=5, max_rounds=15, maxiter=250)
    if c5_f < best_c5:
        best_h, best_c5 = h_f, c5_f
    if time.time() < d6:
        h_sg, c5_sg = _subgradient_descent(best_h, best_n, d6, max_iter=2000)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
