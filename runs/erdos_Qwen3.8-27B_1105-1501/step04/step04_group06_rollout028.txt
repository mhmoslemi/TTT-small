import numpy as np

def _project(v, target):
    # Project v to [0,1]^n with sum = target
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(x)
            for i in idx:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            idx = np.argsort(-x)
            for i in idx:
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_deriv(h, s):
    # Derivative of correlation at shift s w.r.t. h
    n = len(h)
    j = np.arange(n)
    pd = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    pd[m1] = 1.0 - h[gs[m1]]
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    pd[m2] -= h[ps[m2]]
    return pd

def _slsqp_obj(z, n, active):

def _lp_direction(h, scores, n, radius):
    m = 2 * n - 1
    j = np.arange(n)
    G = np.zeros((m, n))
    for r in range(m):
        s = r - (n - 1)
        G[r, :] = _corr_deriv(h, s)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for jj in range(n):
        lo_b = max(-radius, -h[jj])
        hi_b = min(radius, 1.0 - h[jj])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _shuffle_binary(h, n, n_swaps, seed):
    # Swap some 1s and 0s in a near-binary vector
    rng = np.random.default_rng(seed)
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(n_swaps):
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        h[ones[i]], h[zeros[j]] = h[zeros[j]], h[ones[i]]
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    return h

        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(x)
            for i in idx:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-13:
                    break
        else:
            idx = np.argsort(-x)
            for i in idx:
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-13:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_deriv(h, s):
    n = len(h)
    j = np.arange(n)
    pd = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    pd[m1] = 1.0 - h[gs[m1]]
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    pd[m2] -= h[ps[m2]]
    return pd


def _build_G(h, n):
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        s = r - (n - 1)
        G[r, :] = _corr_deriv(h, s)
    return G


def _slsqp_obj(z, n, active):
    return z[n]


def _slsqp_jac(z, n, active):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _slsqp_eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _slsqp_eq_jac(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _slsqp_ineq_f(z, n, active):
    h = z[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return z[n] - corr[active]


def _slsqp_ineq_jac(z, n, active):
    h = z[:n]
    J = np.zeros((len(active), n + 1))
    for k, r in enumerate(active):
        s = r - (n - 1)
        J[k, :n] = -_corr_deriv(h, s)
        J[k, n] = 1.0
    return J


def _active_slsqp(h_init, n, K=14, maxiter=200, max_rounds=8):
    h = np.asarray(h_init, dtype=float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 1e-4
    active = list(np.argsort(-corr)[:K])
    best_h, best_c5 = h.copy(), _exact_c5(h)

    for rnd in range(max_rounds):
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
        cons = [
            {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': _slsqp_ineq_f, 'jac': _slsqp_ineq_jac, 'args': (n, active)}
        ]
        try:
            res = minimize(_slsqp_obj, z0, jac=_slsqp_jac, args=(n, active),
                           bounds=bounds, constraints=cons, method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            h = _project(res.x[:n], n * 0.5)
        except Exception:
            break

        c5_now = _exact_c5(h)
        if c5_now < best_c5:
            best_c5, best_h = c5_now, h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        t_val = float(np.max(corr))
        t0 = t_val + 1e-4
        violated = np.where(corr > t_val - 1e-12)[0]
        new_indices = [i for i in violated if i not in active]
        if not new_indices:
            break
        new_indices.sort(key=lambda i: -corr[i])
        add_k = max(3, len(active) // 3)
        active = active + new_indices[:add_k]
        active = list(set(active))[:min(len(active), 35)]

    return best_h, best_c5


def _lp_direction(h, scores, n, radius):
    G = _build_G(h, n)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for jj in range(n):
        lo_b = max(-radius, -h[jj])
        hi_b = min(radius, 1.0 - h[jj])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_line_search(h, p, n):
    cur_c5 = _exact_c5(h)
    alpha_max = 1.0
    for jj in range(n):
        if p[jj] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[jj]) / p[jj])
        elif p[jj] < -1e-14:
            alpha_max = min(alpha_max, h[jj] / (-p[jj]))
    alpha_max = max(alpha_max, 1e-10)
    best_a, best_c5 = 0.0, cur_c5
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-12:
            continue
        h_t = _project(h + a * p, n * 0.5)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a <= 0:
        return 0.0, cur_c5
    lo = max(0.0, best_a - alpha_max * 0.12)
    hi = min(alpha_max, best_a + alpha_max * 0.12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, n * 0.5))
    f2 = _exact_c5(_project(h + c2 * p, n * 0.5))
    for _ in range(14):
        if hi - lo < 1e-10 * max(alpha_max, 1.0):
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, n * 0.5))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, n * 0.5))
    a = (lo + hi) / 2.0
    c5_t = _exact_c5(_project(h + a * p, n * 0.5))
    if c5_t < best_c5:
        best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, n, deadline, radius_init=0.2, max_iters=5000, stall_limit=80):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, n, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.85
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _lp_line_search(h, p, n)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, n * 0.5)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.5)
        else:
            stall += 1
            radius *= 0.82
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.1
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _gen_binary_balanced(n, seed):
    rng = np.random.default_rng(seed)
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def _gen_wave(n, freq, phase):
    x = np.linspace(0.0, 2.0 * np.pi * freq, n)
    v = 0.5 + 0.35 * np.sin(x + phase) + 0.1 * np.sin(2.0 * x + phase * 0.7)
    return _project(v, n * 0.5)


def _gen_blocks(n, seed, nblocks):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    block_size = n // nblocks
    for b in range(nblocks):
        if rng.random() < 0.5:
            h[b * block_size:(b + 1) * block_size] = 1.0
    return _project(h, n * 0.5)


def _coarse_solve(n_coarse, seed, deadline):
    rng = np.random.default_rng(seed)
    best_h, best_c5 = None, 1.0
    for attempt in range(20):
        if time.time() > deadline:
            break
        h_try = _gen_binary_balanced(n_coarse, seed + attempt * 7)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
    h_lp, c5_lp = _lp_loop(best_h, n_coarse, deadline, radius_init=0.3,
                           max_iters=2000, stall_limit=40)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp
    return best_h, best_c5


def _subgrad_step(h, n, step_size=0.05):
    corr = np.correlate(h, 1.0 - h, mode='full')
    r_max = int(np.argmax(corr))
    s = r_max - (n - 1)
    pd = _corr_deriv(h, s)
    pd -= np.mean(pd)
    norm = np.linalg.norm(pd)
    if norm < 1e-15:
        return h.copy(), _exact_c5(h)
    direction = pd / norm
    c5_cur = _exact_c5(h)
    for mult in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
        h_new = _project(h - mult * direction, n * 0.5)
        c5_new = _exact_c5(h_new)
        if c5_new < c5_cur:
            return h_new, c5_new
    return h.copy(), c5_cur


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick LP on parent (8% budget)
    d1 = t0 + budget_s * 0.08
    h_lp, c5_lp = _lp_loop(h, n0, d1, radius_init=0.25, max_iters=800, stall_limit=40)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 2: Coarse-to-fine (10% budget)
    d2 = t0 + budget_s * 0.18
    n_coarse = max(16, n0 // 4)
    h_c, c5_c = _coarse_solve(n_coarse, seed + 500, d2)
    h_up = _resample(h_c, n0)
    h_up_s, c5_up_s = _active_slsqp(h_up, n0, K=12, maxiter=100, max_rounds=4)
    if c5_up_s < best_c5:
        best_c5, best_h = c5_up_s, h_up_s.copy()
    h_up_lp, c5_up_lp = _lp_loop(h_up_s, n0, d2, radius_init=0.15, max_iters=500, stall_limit=30)
    if c5_up_lp < best_c5:
        best_c5, best_h = c5_up_lp, h_up_lp.copy()

    # Phase 3: Diverse initialization (18% budget)
    d3 = t0 + budget_s * 0.36
    candidates = []
    candidates.append(best_h.copy())
    for s in range(8):
        candidates.append(_gen_binary_balanced(n0, seed + 1000 + s))
    for freq in [1, 2, 3, 4, 5, 7]:
        for phase in [0.0, 1.0, 2.0, 3.0, 4.0]:
            candidates.append(_gen_wave(n0, freq, phase))
    for nb in [2, 3, 4, 5, 6, 8]:
        for s in range(3):
            candidates.append(_gen_blocks(n0, seed + 2000 + nb * 10 + s, nb))
    # Symmetrized and antisymmetric variants of best
    h_sym = _project(0.5 * (best_h + best_h[::-1]), target)
    candidates.append(h_sym)
    h_asym = best_h.copy()
    h_asym[:n0 // 2] = 1.0 - best_h[n0 // 2 - 1::-1]
    h_asym[n0 // 2:] = best_h[:n0 // 2][::-1]
    candidates.append(_project(h_asym, target))
    # Reversed
    candidates.append(_project(best_h[::-1], target))

    # Evaluate and quick-optimize promising candidates
    cands_sorted = sorted(candidates, key=lambda c: _exact_c5(c))
    for cand in cands_sorted[:40]:
        if time.time() > d3:
            break
        c5_cand = _exact_c5(cand)
        if c5_cand > best_c5 + 0.04:
            break
        sub_d = min(d3, time.time() + 8.0)
        h_pol, c5_pol = _lp_loop(cand, n0, sub_d, radius_init=0.12,
                                  max_iters=300, stall_limit=25)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol.copy()

    # Phase 4: Active-set SLSQP refinement (8% budget)
    d4 = t0 + budget_s * 0.44
    for attempt in range(4):
        if time.time() > d4:
            break
        start = best_h.copy()
        if attempt > 0:
            start = _project(start + rng.normal(0, 0.015 + 0.005 * attempt, n0), target)
        h_s, c5_s = _active_slsqp(start, n0, K=14, maxiter=180, max_rounds=6)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    # Phase 5: Main aggressive LP (30% budget)
    d5 = t0 + budget_s * 0.74
    h_lp, c5_lp = _lp_loop(best_h, n0, d5, radius_init=0.35, max_iters=10000, stall_limit=120)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 6: Multi-resolution (12% budget)
    d6 = t0 + budget_s * 0.86
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, 12]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = _active_slsqp(h_r, n_try, K=10, maxiter=80, max_rounds=4)
        sub_d2 = min(d6, time.time() + budget_s * 0.015)
        h_r2, c5_r2 = _lp_loop(h_r, n_try, sub_d2, radius_init=0.1,
                                max_iters=400, stall_limit=25)
        if c5_r2 < c5_r:
            h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try
            # Refine at new resolution
            if time.time() < d6:
                sub_d3 = min(d6, time.time() + budget_s * 0.02)
                h_r3, c5_r3 = _lp_loop(best_h, best_n, sub_d3, radius_init=0.15,
                                        max_iters=800, stall_limit=40)
                if c5_r3 < best_c5:
                    best_c5, best_h = c5_r3, h_r3.copy()

    # Phase 7: Perturbation + subgradient escaping (6% budget)
    d7 = t0 + budget_s * 0.92
    for attempt in range(80):
        if time.time() > d7:
            break
        scale = 0.008 + 0.003 * (attempt % 7)
        h_try = _project(best_h + rng.normal(0, scale, best_n), best_n * 0.5)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 + 0.008:
            # Subgradient nudge
            h_sub, c5_sub = _subgrad_step(h_try, best_n, step_size=0.04)
            if c5_sub < c5_try:
                h_try, c5_try = h_sub, c5_sub
            sub_d = min(d7, time.time() + 5.0)
            h_pol, c5_pol = _lp_loop(h_try, best_n, sub_d, radius_init=0.08,
                                      max_iters=200, stall_limit=20)
            if c5_pol < best_c5:
                best_c5, best_h = c5_pol, h_pol.copy()

    # Phase 8: Final polish (8% budget)
    d8 = t0 + budget_s * 0.995
    h_s, c5_s = _active_slsqp(best_h, best_n, K=20, maxiter=250, max_rounds=8)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    h_lp, c5_lp = _lp_loop(best_h, best_n, d8, radius_init=0.06,
                            max_iters=4000, stall_limit=100)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
