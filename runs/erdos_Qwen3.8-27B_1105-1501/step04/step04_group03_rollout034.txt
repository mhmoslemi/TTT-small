def project(v, target):

        else:

import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += (-h[g:])
    return G

def lp_step(h, radius):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = build_G(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

                    cur_c5 = new_c5

        # Full step didn't work, try line search

                alpha_max = min(alpha_max, h[j] / (-d[j]))
        alpha_max = max(alpha_max, 1e-10)

        opt_a = (lo_a + hi_a) / 2.0

            radius *= 0.75

    return best_h, best_c5

def coord_descent(h, n_cycles=3, grid_pts=25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    for _ in range(n_cycles):
        for j in range(n):
            old_v = h[j]
            best_v = old_v
            best_c5 = cur_c5
            for frac in np.linspace(0.0, 1.0, grid_pts):
                if abs(frac - old_v) < 1e-6:
                    continue
                h[j] = frac
                c5_t = exact_c5(h)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_v = frac
            h[j] = best_v
        h = project(h, target)
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
    return h, cur_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):

            break
        improved = False

                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        omh = 1.0 - x
        scores = np.correlate(x, omh, mode='full') * dx
        max_s = float(np.max(scores))
        exp_s = np.exp((scores - max_s) / tau)
        Z = float(np.sum(exp_s))
        return max_s + tau * np.log(Z)

    def grad(x):
        omh = 1.0 - x
        scores_raw = np.correlate(x, omh, mode='full')
        scores = scores_raw * dx
        max_s = float(np.max(scores))
        exp_s = np.exp((scores - max_s) / tau)
        Z = float(np.sum(exp_s))
        w = exp_s / Z
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            d = k - (n - 1)
            js1 = max(0, -d)
            je1 = min(n - 1, n - 1 - d)
            if js1 <= je1:
                g[js1:je1 + 1] += wk * (1.0 - x[js1 + d:je1 + 1 + d])
            js2 = max(0, d)
            je2 = min(n - 1, n + d - 1)
            if js2 <= je2:
                g[js2:je2 + 1] -= wk * x[js2 - d:je2 + 1 - d]
        return dx * g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0
