def exact_1d_pair_step(h, i, j, scores):
    """Find optimal step a for pair (i,j)."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute b_k for all shifts k
    # b_k = ((1-h[i+k]) - h[i-k]) - ((1-h[j+k]) - h[j-k])
    # where terms are 0 if index is out of range
    
    # For shift d (where d ranges from -(n-1) to (n-1)):
    # The score at shift d is affected by h[i] if i is in [0, n-1] and i+d is in [0, n-1]
    # Change from h[i]: a * (1-h[i+d]) if i+d in [0,n-1] else 0, minus a*h[i-d] if i-d in [0,n-1] else 0
    
    # Let me compute the vector of linear coefficients
    # For each shift d, the coefficient of a in score_d is:
    # coeff_d = (1-h[i+d]) * [0 <= i+d < n] - h[i-d] * [0 <= i-d < n] - (1-h[j+d]) * [0 <= j+d < n] + h[j-d] * [0 <= j-d < n]
    
    # scores array is indexed 0 to 2n-2, corresponding to shifts d = -(n-1) to (n-1)
    # So score index = d + (n-1)
    
    b = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        coeff = 0.0
        if 0 <= i+d < n:
            coeff += (1.0 - h[i+d])
        if 0 <= i-d < n:
            coeff -= h[i-d]
        if 0 <= j+d < n:
            coeff -= (1.0 - h[j+d])
        if 0 <= j-d < n:
            coeff += h[j-d]
        b[d + n - 1] = coeff
    
    # Now find a that minimizes max_k (scores[k] + b[k] * a)
    # subject to: a_min <= a <= a_max
    a_min = max(-h[i], h[j] - 1.0)  # from 0 <= h[i]+a <= 1 and 0 <= h[j]-a <= 1
    a_max = min(1.0 - h[i], h[j])

    n = len(h)
    dx = 2.0 / n
    
    # Find active set
    sorted_scores = np.sort(scores)[::-1]
    threshold = sorted_scores[min(n_active, len(scores)) - 1]
    active = np.where(scores >= threshold - 1e-10)[0]

    grad = np.zeros(n)

def project(v, target):

        return np.zeros(n)

    n = len(h)
    # contrib[i][k] = (1-h[i+d]) * [0<=i+d<n] - h[i-d] * [0<=i-d<n]
    # where d = k - (n-1), i.e., k ranges from 0 to 2n-2
    contribs = np.zeros((n, 2*n-1))
    for i in range(n):
        for k in range(2*n-1):
            d = k - (n-1)
            val = 0.0
            if 0 <= i + d < n:
                val += (1.0 - h[i + d])
            if 0 <= i - d < n:
                val -= h[i - d]
            contribs[i, k] = val
    return contribs

def exact_1d_step(scores, b, a_min, a_max):
    """Find a in [a_min, a_max] that minimizes max_k (scores[k] + b[k]*a)."""
    if a_min >= a_max:
        return 0.0
    
    # Coarse grid
    n_grid = 64
    a_vals = np.linspace(a_min, a_max, n_grid)
    obj = np.max(scores[None, :] + b[None, :] * a_vals[:, None], axis=1)
    best_idx = int(np.argmin(obj))
    
    # Golden section refinement
    lo = a_vals[max(0, best_idx - 1)]
    hi = a_vals[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + b * c1))
    f2 = float(np.max(scores + b * c2))
    for _ in range(40):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + b * c1))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + b * c2))
    return float((lo + hi) / 2.0)

def cd_pass(h, scores, contribs, n_max_pairs=200, seed=42):
    """One pass of gradient-informed coordinate descent."""
    n = len(h)
    rng = np.random.default_rng(seed)
    cur_c5 = float(np.max(scores)) * (2.0 / n)
    best_h = h.copy()
    
    # Compute gradient (uniform subgradient over top active set)
    sorted_idx = np.argsort(scores)[::-1]
    n_active = max(3, min(8, len(scores)))
    active = sorted_idx[:n_active]
    
    grad = np.zeros(n)
    for k_idx in active:
        d = k_idx - (n - 1)
        for i in range(n):
            if 0 <= i + d < n:
                grad[i] += (1.0 - h[i + d])
            if 0 <= i - d < n:
                grad[i] -= h[i - d]
    grad /= n_active
    
    neg_idx = np.where((grad < -1e-8) & (h < 1.0 - 1e-6))[0]
    pos_idx = np.where((grad > 1e-8) & (h > 1e-6))[0]
    
    if len(neg_idx) == 0 or len(pos_idx) == 0:
        return best_h, cur_c5, False
    
    improved = False
    try_count = 0
    max_tries = n_max_pairs * 3
    
    while try_count < max_tries and time.time() < float('inf'):
        try_count += 1
        i = int(neg_idx[rng.integers(len(neg_idx))])
        j = int(pos_idx[rng.integers(len(pos_idx))])
        
        b = contribs[i] - contribs[j]
        a_min = max(-h[i], h[j] - 1.0)
        a_max = min(1.0 - h[i], h[j])
        
        if a_max - a_min < 1e-10:
            continue
        
        a = exact_1d_step(scores, b, a_min, a_max)
        if abs(a) < 1e-12:
            continue
        
        h_try = h.copy()
        h_try[i] += a
        h_try[j] -= a
        h_try = np.clip(h_try, 0, 1)
        
        scores_try = np.correlate(h_try, 1.0 - h_try, mode='full')
        c5_try = float(np.max(scores_try)) * (2.0 / n)
        
        if c5_try < cur_c5 - 1e-12:
            h = h_try
            scores = scores_try
            cur_c5 = c5_try
            best_h = h.copy()
            improved = True
            # Recompute contribs (approximate: only changed for i and j)
            for idx in [i, j]:
                for k_idx in range(2*n-1):
                    d = k_idx - (n-1)
                    val = 0.0
                    if 0 <= idx + d < n:
                        val += (1.0 - h[idx + d])
                    if 0 <= idx - d < n:
                        val -= h[idx - d]
                    contribs[idx, k_idx] = val
            # Update gradient
            sorted_idx = np.argsort(scores)[::-1]
            active = sorted_idx[:n_active]
            grad = np.zeros(n)
            for k_idx in active:
                d = k_idx - (n - 1)
                for ii in range(n):
                    if 0 <= ii + d < n:
                        grad[ii] += (1.0 - h[ii + d])
                    if 0 <= ii - d < n:
                        grad[ii] -= h[ii - d]
            grad /= n_active
            neg_idx = np.where((grad < -1e-8) & (h < 1.0 - 1e-6))[0]
            pos_idx = np.where((grad > 1e-8) & (h > 1e-6))[0]
            if len(neg_idx) == 0 or len(pos_idx) == 0:
                break
    
    return best_h, cur_c5, improved

def lp_loop(h, deadline, radius_init=0.1):
    from scipy.optimize import linprog
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    dx = 2.0 / n
    
    while time.time() < deadline:
        if stall > 100:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Build gradient matrix
        G = np.zeros((2*n-1, n))
        for d in range(-(n-1), n):
            r = n - 1 + d
            if d >= 0:
                m = n - d
                if m > 0:
                    G[r, d:] += (1.0 - h[:m])
                    G[r, :m] += -h[d:]
            else:
                g = -d
                m = n - g
                if m > 0:
                    G[r, :m] += (1.0 - h[g:])
                    G[r, g:] += -h[:m]
        
        # LP: find p that minimizes t subject to G@p - t <= -scores, sum(p)=0, |p| <= radius
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
            else:
                p = None
        except:
            p = None
        
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        best_a = 0.0
        best_c5_new = cur_c5
        for i in range(20):
            a = alpha_max * (i + 1) / 20.0
            c5_t = exact_c5(h + a * p)
            if c5_t < best_c5_new - 1e-13:
                best_c5_new = c5_t
                best_a = a
        
        if best_a > 0:
            h_new = h + best_a * p
            h_new = project(h_new, target)
            c5_new = exact_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.1, 0.5)
            else:
                stall += 1
                radius *= 0.7
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.06
    
    return best_h, best_c5

def full_pair_sweep(h, scores, contribs, deadline):
    """Systematic pair sweep with exact 1D steps."""
    n = len(h)
    dx = 2.0 / n
    cur_c5 = float(np.max(scores)) * dx
    best_h = h.copy()
    
    # Process pairs in order of "promise" (based on gradient)
    sorted_idx = np.argsort(scores)[::-1]
    n_active = max(3, min(8, len(scores)))
    active = sorted_idx[:n_active]
    
    grad = np.zeros(n)
    for k_idx in active:
        d = k_idx - (n-1)
        for i in range(n):
            if 0 <= i+d < n:
                grad[i] += (1.0 - h[i+d])
            if 0 <= i-d < n:
                grad[i] -= h[i-d]
    grad /= n_active
    
    # Build list of promising pairs
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            if (grad[i] < 0 and grad[j] > 0) or (grad[i] > 0 and grad[j] < 0):
                score = abs(grad[i]) + abs(grad[j])
                pairs.append((score, i, j))
    pairs.sort(reverse=True)
    
    for _, i, j in pairs:
        if time.time() > deadline:
            break
        
        b = contribs[i] - contribs[j]
        a_min = max(-h[i], h[j] - 1.0)
        a_max = min(1.0 - h[i], h[j])
        if a_max - a_min < 1e-10:
            continue
        
        a = exact_1d_step(scores, b, a_min, a_max)
        if abs(a) < 1e-12:
            continue
        
        h_try = h.copy()
        h_try[i] += a
        h_try[j] -= a
        h_try = np.clip(h_try, 0, 1)
        
        scores_try = np.correlate(h_try, 1.0-h_try, mode='full')
        c5_try = float(np.max(scores_try)) * dx
        
        if c5_try < cur_c5 - 1e-12:
            h = h_try
            scores = scores_try
            cur_c5 = c5_try
            best_h = h.copy()
            # Update contribs for i and j
            for idx in [i, j]:
                for k_idx2 in range(2*n-1):
                    d = k_idx2 - (n-1)
                    val = 0.0
                    if 0 <= idx+d < n:
                        val += (1.0 - h[idx+d])
                    if 0 <= idx-d < n:
                        val -= h[idx-d]
                    contribs[idx, k_idx2] = val
    
    return best_h, cur_c5

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_b(h, i, j):

        if 0 <= j - d < n:

    obj = np.max(scores[None, :] + b[None, :] * a_vals[:, None], axis=1)
    best_idx = int(np.argmin(obj))
    lo = a_vals[max(0, best_idx - 1)]
    hi = a_vals[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + b * c1))
    f2 = float(np.max(scores + b * c2))
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + b * c1))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + b * c2))
    return float((lo + hi) / 2.0)

def compute_subgrad(h, scores, n_active=5):
    n = len(h)
    sorted_idx = np.argsort(scores)[::-1]
    n_act = min(n_active, len(scores))
    active = sorted_idx[:n_act]
    grad = np.zeros(n)
    for k_idx in active:
        d = k_idx - (n - 1)
        for i in range(n):
            if 0 <= i + d < n:
                grad[i] += (1.0 - h[i + d])
            if 0 <= i - d < n:
                grad[i] -= h[i - d]
    grad /= float(n_act)
    return grad, active

def cd_round(h, scores, max_pairs=100, seed=42, deadline=None, n_active=5):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    cur_c5 = float(np.max(scores)) * dx
    best_h = h.copy()
    
    grad, active = compute_subgrad(h, scores, n_active)
    neg_idx = np.where((grad < -1e-8) & (h < 1.0 - 1e-6))[0]
    pos_idx = np.where((grad > 1e-8) & (h > 1e-6))[0]
    
    if len(neg_idx) == 0 or len(pos_idx) == 0:
        return best_h, cur_c5, False
    
    improved = False
    tries = 0
    max_tries = max_pairs * 4
    
    while tries < max_tries:
        if deadline is not None and time.time() > deadline:
            break
        tries += 1
        i = int(neg_idx[rng.integers(len(neg_idx))])
        j = int(pos_idx[rng.integers(len(pos_idx))])
        
        if abs(i - j) == 0:
            continue
        
        b = compute_b(h, i, j)
        a_min = max(-h[i], h[j] - 1.0)
        a_max = min(1.0 - h[i], h[j])
        if a_max - a_min < 1e-10:
            continue
        
        a = exact_1d_step(scores, b, a_min, a_max)
        if abs(a) < 1e-12:
            continue
        
        h_try = h.copy()
        h_try[i] += a
        h_try[j] -= a
        h_try = np.clip(h_try, 0.0, 1.0)
        
        scores_try = np.correlate(h_try, 1.0 - h_try, mode='full')
        c5_try = float(np.max(scores_try)) * dx
        
        if c5_try < cur_c5 - 1e-12:
            h = h_try
            scores = scores_try
            cur_c5 = c5_try
            best_h = h.copy()
            improved = True
            grad, active = compute_subgrad(h, scores, n_active)
            neg_idx = np.where((grad < -1e-8) & (h < 1.0 - 1e-6))[0]
            pos_idx = np.where((grad > 1e-8) & (h > 1e-6))[0]
            if len(neg_idx) == 0 or len(pos_idx) == 0:
                break
    
    return best_h, cur_c5, improved

def cd_round_multi_active(h, scores, max_pairs=80, seed=42, deadline=None):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    cur_c5 = float(np.max(scores)) * dx
    best_h = h.copy()
    
    for n_act in [3, 5, 8, 12]:
        if deadline is not None and time.time() > deadline:
            break

        pos_idx = np.where((grad > 1e-8) & (h > 1e-6))[0]
        if len(neg_idx) == 0 or len(pos_idx) == 0:
            continue
        
        tries = 0
        while tries < max_pairs:
            if deadline is not None and time.time() > deadline:
                break
            tries += 1
            i = int(neg_idx[rng.integers(len(neg_idx))])
            j = int(pos_idx[rng.integers(len(pos_idx))])
            if i == j:
                continue
            
            b = compute_b(h, i, j)
            a_min = max(-h[i], h[j] - 1.0)
            a_max = min(1.0 - h[i], h[j])
            if a_max - a_min < 1e-10:
                continue
            
            a = exact_1d_step(scores, b, a_min, a_max)
            if abs(a) < 1e-12:

            h_try = h.copy()

            h_try[j] -= a
            h_try = np.clip(h_try, 0.0, 1.0)
            scores_try = np.correlate(h_try, 1.0 - h_try, mode='full')

                scores = scores_try
                cur_c5 = c5_try

                neg_idx = np.where((grad < -1e-8) & (h < 1.0 - 1e-6))[0]
                pos_idx = np.where((grad > 1e-8) & (h > 1e-6))[0]
                if len(neg_idx) == 0 or len(pos_idx) == 0:
                    break
    
    return best_h, cur_c5

def lp_direction(h, scores, radius):
    n = len(h)
    dx = 2.0 / n
    scores_dx = scores * dx
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores_dx
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j] * 0.999), min(radius, (1.0 - h[j]) * 0.999)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = lp_direction(h, scores, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        best_a = 0.0
        best_c5_new = cur_c5
        for i in range(20):
            a = alpha_max * (i + 1) / 20.0
            c5_t = exact_c5(h + a * p)
            if c5_t < best_c5_new - 1e-13:
                best_c5_new = c5_t
                best_a = a
        
        if best_a > 0:
            h_new = project(h + best_a * p, target)
            c5_new = exact_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.1, 0.5)
            else:
                stall += 1
                radius *= 0.7
        else:
            stall += 1
            radius *= 0.7
    
    return best_h, best_c5

def full_sweep(h, deadline, max_passes=5):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = exact_c5(best_h)
    
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        scores = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        
        grad, _ = compute_subgrad(best_h, scores, 6)
        
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                if grad[i] * grad[j] > -1e-10:
                    continue
                if best_h[i] >= 1.0 - 1e-8 or best_h[i] <= 1e-8:
                    continue
                if best_h[j] >= 1.0 - 1e-8 or best_h[j] <= 1e-8:
                    continue
                
                b = compute_b(best_h, i, j)
                a_min = max(-best_h[i], best_h[j] - 1.0)
                a_max = min(1.0 - best_h[i], best_h[j])
                if a_max - a_min < 1e-10:
                    continue
                
                a = exact_1d_step(scores, b, a_min, a_max)
                if abs(a) < 1e-12:
                    continue
                
                h_try = best_h.copy()
                h_try[i] += a
                h_try[j] -= a
                h_try = np.clip(h_try, 0.0, 1.0)
                scores_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                c5_try = float(np.max(scores_try)) * dx
                
                if c5_try < best_c5 - 1e-12:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
                    grad, _ = compute_subgrad(best_h, scores_try, 6)
    
    return best_h, best_c5

def block_cd(h, scores, deadline, block_sizes=(2, 3, 4), seed=42):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = float(np.max(scores)) * dx
    
    for bs in block_sizes:
        n_blocks = n - bs + 1
        n_tries = min(200, max(50, n_blocks * 2))
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n_blocks))
            j = int(rng.integers(0, n_blocks))
            if j >= i and j < i + bs:
                j = (j + bs) % n_blocks
                if j >= i and j < i + bs:
                    continue
            
            a_min = min(best_h[i:i+bs])
            a_min2 = min(best_h[j:j+bs]) - 1.0
            a_max = 1.0 - max(best_h[i:i+bs])
            a_max2 = max(best_h[j:j+bs])
            a_lo = max(-a_min, a_min2)
            a_hi = min(a_max, a_max2)
            if a_hi - a_lo < 1e-10:
                continue
            
            b = np.zeros(2 * n - 1)
            for di in range(bs):
                b += compute_b(best_h, i + di, j, )
            # Fix: compute_b gives diff for (i,j). For block we need sum of individual diffs
            b = np.zeros(2 * n - 1)
            for di in range(bs):
                for dj in range(bs):
                    pass
            # Actually for a block move, we shift all h[i+di] by +a and all h[j+dj] by -a
            # The net effect on scores is sum of individual effects
            b = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                d = k - (n - 1)
                val = 0.0
                for di in range(bs):
                    ii = i + di
                    if 0 <= ii + d < n:
                        val += (1.0 - best_h[ii + d])
                    if 0 <= ii - d < n:
                        val -= best_h[ii - d]
                for dj in range(bs):
                    jj = j + dj
                    if 0 <= jj + d < n:
                        val -= (1.0 - best_h[jj + d])
                    if 0 <= jj - d < n:
                        val += best_h[jj - d]
                b[k] = val
            
            a = exact_1d_step(scores, b, a_lo, a_hi)
            if abs(a) < 1e-12:
                continue
            
            h_try = best_h.copy()
            h_try[i:i+bs] += a
            h_try[j:j+bs] -= a
            h_try = np.clip(h_try, 0.0, 1.0)
            # Check constraint
            if abs(np.sum(h_try) - n / 2.0) > 1e-8:
                continue
            scores_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            c5_try = float(np.max(scores_try)) * dx
            if c5_try < best_c5 - 1e-12:
                best_h = h_try
                best_c5 = c5_try
                scores = scores_try
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0
    
    # Phase 1: LP warm-up
    d1 = t0 + budget_s * 0.08
    h_lp, c5_lp = lp_loop(best_h, d1, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()
    
    # Phase 2: IntensCD with multiple active set sizes
    d2 = t0 + budget_s * 0.35
    scores = np.correlate(best_h, 1.0 - best_h, mode='full')
    for rnd in range(40):
        if time.time() > d2:
            break
        h_cd, c5_cd = cd_round_multi_active(best_h, scores, max_pairs=80, seed=seed + rnd, deadline=d2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            scores = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    # Phase 3: Full pair sweep
    d3 = t0 + budget_s * 0.52
    best_h, c5_sweep = full_sweep(best_h, d3, max_passes=5)
    if c5_sweep < best_c5:
        best_c5 = c5_sweep
        best_h = best_h.copy()
    
    # Phase 4: Block coordinate descent
    d4 = t0 + budget_s * 0.62
    scores = np.correlate(best_h, 1.0 - best_h, mode='full')
    h_blk, c5_blk = block_cd(best_h, scores, d4, block_sizes=(2, 3), seed=seed + 500)
    if c5_blk < best_c5:
        best_c5 = c5_blk
        best_h = h_blk.copy()
    
    # Phase 5: More CD rounds
    d5 = t0 + budget_s * 0.75
    scores = np.correlate(best_h, 1.0 - best_h, mode='full')
    for rnd in range(50):
        if time.time() > d5:
            break
        h_cd, c5_cd = cd_round(best_h, scores, max_pairs=60, seed=seed + 1000 + rnd, deadline=d5, n_active=4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            scores = np.correlate(best_h, 1.0 - best_h, mode='full')
        h_cd2, c5_cd2 = cd_round(best_h, scores, max_pairs=60, seed=seed + 2000 + rnd, deadline=d5, n_active=10)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            scores = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    # Phase 6: Diversified restarts
    d6 = t0 + budget_s * 0.87
    scales = [0.015, 0.03, 0.06, 0.1, 0.15, 0.25, 0.4]
    for s in scales:
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, s, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_try, c5_try = lp_loop(h_pert, sub_dl, radius_init=0.1)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 7: Multi-resolution
    d7 = t0 + budget_s * 0.91
    for dn in [-4, 4, -8, 8, -16, 16, -24, 24]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 140:
            continue
        x_old = (np.arange(best_n) + 0.5) / best_n
        x_new = (np.arange(n_try) + 0.5) / n_try
        h_r = np.interp(x_new, x_old, best_h)
        h_r = project(h_r, n_try / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.01)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.08)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 8: Final CD polish
    if time.time() < hard_deadline:
        scores = np.correlate(best_h, 1.0 - best_h, mode='full')
        for rnd in range(30):
            if time.time() > hard_deadline:
                break
            h_cd, c5_cd = cd_round(best_h, scores, max_pairs=40, seed=seed + 5000 + rnd, deadline=hard_deadline, n_active=3)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                scores = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    
    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0
    
    return best_h, final_c5, best_n
