                            best_val = c5_t
                            best_t = t
                if best_t != h[i]:
                    h[i] = best_t
                    h[j] = s_ij - best_t
                    if best_val < best_c5:
                        best_c5 = best_val
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5
