is a convex problem (minimize a variable subject to linear constraints in the variables, since corr[k] is linear in h). Wait, is corr[k] linear in h?

corr[k] = ∑_i h[i]*(1-h[i+k])

This is NOT linear in h! It's quadratic (the product h[i]*h[i+k] appears with a negative sign). So the problem is NOT convex.

Wait: corr[k] = ∑_i h[i] - ∑_i h[i]*h[i+k]

The first term is linear, the second is quadratic (bilinear). So the constraints corr[k] ≤ t are NOT linear. The problem is non-convex.

This is why SLSQP (a general NLP solver) is used rather than a convex solver.

Now, one thing I want to be careful about: the pair line search evaluates C₅ along a 1D line. The function C₅(t) = max_k corr_k(t) where each corr_k is quadratic in t (since we're only changing two coordinates). So C₅(t) is the max of quadratics, which is a convex function (max of convex functions is convex, and each quadratic here might be convex or concave).

Wait, is corr_k(t) convex or concave along the pair line? Let's see:
corr_k(t) = ∑_i h_i(t) * (1 - h_{i+k}(t))

Along the pair line, h_i(t) = t and h_j(t) = p - t for the pair (i,j), and all other h are fixed.

corr_k(t) changes only for the terms involving i or j (or i+k, j+k, i-k, j-k). It's a product of at most two linear functions of t, so it's a quadratic in t. The second derivative determines convexity.

For the specific term h_i * (1-h_{i+k}):
If k ≠ 0 and k ≠ j-i: h_i = t, h_{i+k} is fixed. So the term is t * (1 - h_{i+k}), which is linear in t.
If k = j-i: h_i = t, h_{i+k} = h_j = p-t. Term = t(1-(p-t)) = t(1-p+t) = t - pt + t². Second derivative = 2 > 0. Convex.
If k = -(j-i): h_i = t, h_{i+k} = h_{i-(j-i)} = h_{2i-j}. If 2i-j = j, i.e., i=j, impossible. Otherwise h_{2i-j} is fixed (unless 2i-j is i or j, which would mean i=j, impossible). So linear.
If k = 0: h_i(1-h_i) = t(1-t) = t - t². Second derivative = -2 < 0. Concave!

So the max corr_k(t) is the max of a mix of linear, convex quadratic, and concave quadratic functions. The overall max is NOT necessarily convex.

However, in practice, for small perturbations (the line search range is small), the function should be well-behaved. I'll use a dense grid + refinement approach.

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def two_level_h(mask, d):
    return np.where(mask, 1.0 - d, d)

def opt_d_for_mask(mask, n_grid=150, n_refine=50):
    n = len(mask)
    ds = np.linspace(0.0, 0.5, n_grid)
    best_obj = np.inf
    best_d = 0.25
    for d in ds:
        obj = exact_c5(two_level_h(mask, d))
        if obj < best_obj:
            best_obj = obj
            best_d = d
    step = 0.5 / (n_grid - 1)
    lo = max(0.0, best_d - step)
    hi = min(0.5, best_d + step)
    for _ in range(n_refine):
        if hi - lo < 1e-10:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if exact_c5(two_level_h(mask, m1)) < exact_c5(two_level_h(mask, m2)):
            hi = m2
        else:
            lo = m1
    best_d = (lo + hi) / 2.0
    return best_d, exact_c5(two_level_h(mask, best_d))

def get_active(g_full, frac=0.5, min_count=10):
    max_g = float(np.max(g_full))
    if max_g <= 0:
        return np.arange(len(g_full))
    threshold = max_g * frac
    active = np.where(g_full >= threshold)[0]
    if len(active) < min_count:
        k = min(min_count * 2, len(g_full))
        active = np.argsort(g_full)[-k:]
    return active

def _slsqp_obj(x, n):
    return x[0]

def _slsqp_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[0] = 1.0
    return g

def _g_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _g_jac(h):
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        d = j - (n - 1)
        if d >= 0:
            m1 = np.arange(d, n)
            J[j, m1] += (1.0 - h[m1 - d])
            if d < n:
                m2 = np.arange(0, n - d)
                J[j, m2] -= h[m2 + d]
        else:
            dp = -d
            m1 = np.arange(0, n - dp)
            J[j, m1] += (1.0 - h[m1 + dp])
            m2 = np.arange(dp, n)
            J[j, m2] -= h[m2 - dp]
    return J

def _slsqp_ineq(x, n, active):
    h = x[1:]
    g = _g_vec(h)
    return x[0] - g[active]

def _slsqp_ineq_jac(x, n, active):
    h = x[1:]
    Jg = _g_jac(h)
    J = np.zeros((len(active), n + 1))
    J[:, 0] = 1.0
    J[:, 1:] = -Jg[active, :]
    return J

def _slsqp_eq(x, n):
    return x[1:].sum() - n / 2.0

def _slsqp_eq_jac(x, n):
    g = np.zeros(n + 1)
    g[1:] = 1.0
    return g

def slsqp_solve(h, active_indices, maxiter=300, ftol=1e-12):
    n = len(h)
    target = n / 2.0
    g_full = _g_vec(h)
    max_g = float(np.max(g_full))
    x0 = np.zeros(n + 1)
    x0[0] = max_g * 1.005 + 0.05
    x0[1:] = h
    constraints = [
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac, 'args': (n, active_indices)},
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}
    ]
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, x0, jac=_slsqp_obj_grad, args=(n,),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = project(h_new, target)
            return h_new
    except Exception:
        pass
    return h

def pair_line_refine(h, i, j, n_refine=40, n_grid=15):
    """Refine a specific pair line search with golden section."""
    n = len(h)
    p = h[i] + h[j]
    lo = max(0.0, p - 1.0)

    best_t = ts[0]
    best_obj = np.inf
    for t in ts:
        h_try = h.copy()
        h_try[i] = t
        h_try[j] = p - t
        obj = exact_c5(h_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    # Golden section around best
    idx = np.argmin([exact_c5(h) for _ in [0]])  # placeholder
    step = (hi - lo) / n_grid
    glo = max(lo, best_t - step)
    ghi = min(hi, best_t + step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_refine):
        m1 = glo + (1 - gr) * (ghi - glo)

        h1 = h.copy()
        h1[i] = m1

        h2 = h.copy()
        h2[i] = m2

    h_new = h.copy()
    h_new[i] = best_t
    h_new[j] = p - best_t

def binary_descent_mask(mask, deadline, max_passes=2000):

    m = n // 2

    for _ in range(max_passes):

            break
        ones = np.where(best_mask)[0]
        zeros = np.where(~best_mask)[0]
        best_swap_obj = best_obj
        best_swap = None
        # Sample a subset of swaps for speed
        n_try = min(len(ones) * len(zeros), 200)
        idx = np.random.choice(len(ones) * len(zeros), min(n_try, len(ones)*len(zeros)), replace=False)
        for ii in idx:
            a = ones[ii // len(zeros)]

            trial = best_mask.copy()

            trial[b] = 1

            best_d = d_new

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
        # Try a few random mask descents
        for _ in range(5):
            if time.time() > d4:
                break
            mask_r = np.zeros(n)
            mask_r[rng.choice(n, n//2, replace=False)] = 1.0
            m_r, d_r, obj_r = binary_descent_mask(mask_r, d4, max_passes=1000)
            h_r = two_level_h(m_r, d_r)
            h_r = project(h_r, target)
            c5_r = exact_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 5: Multi-start SLSQP (10% budget)
    d5 = t0 + budget_s * 0.75
    if time.time() < d5:
        hm, c5m = multi_start_slqp(best_h, rng, d5, n_starts=80)
        if c5m < best_c5:
            best_c5 = c5m
            best_h = hm.copy()

    # Phase 6: Resolution search (15% budget)
    d6 = t0 + budget_s * 0.90
    for dn in [8, -8, 4, -4, 12, -12, 16, -16, 24]:
        if time.time() > d6:
            break
        n_try = n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.025)
        g_full = _g_vec(h_r)
        active = get_active(g_full, frac=0.4, min_count=10)
        for _ in range(8):
            if time.time() > sub_d:
                break
            h_r2 = slsqp_solve(h_r, active, maxiter=200, ftol=1e-12)
            c5_r2 = exact_c5(h_r2)
            if c5_r2 < exact_c5(h_r) - 1e-9:
                h_r = h_r2
                g_full = _g_vec(h_r)
                active = get_active(g_full, frac=0.3, min_count=8)
            else:
                break
        # Quick pair line search on new resolution
        sub_d2 = min(d6, time.time() + budget_s * 0.01)
        if time.time() < sub_d2:
            h_pr, c5_pr = pair_line_search(h_r, sub_d2, n_t=15, max_passes=50, n_lags=3)
            if c5_pr < exact_c5(h_r):
                h_r = h_pr
        c5_r = exact_c5(h_r)
        if c5_r < best_c5 - 1e-9:
            best_c5 = c5_r
            best_h = h_r.copy()
            n = n_try
            target = n / 2.0

    # Phase 7: Final polish (10% budget)
    d7 = deadline - 1.0
    if time.time() < d7:
        # SLSQP
        g_full = _g_vec(best_h)
        active = get_active(g_full, frac=0.5, min_count=15)
        h_f = slsqp_solve(best_h, active, maxiter=400, ftol=1e-14)
        c5_f = exact_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # Pair line search
        if time.time() < d7:
            sub_d = min(d7, time.time() + budget_s * 0.05)
            hp2, c5p2 = pair_line_search(best_h, sub_d, n_t=20, max_passes=300, n_lags=3)
            if c5p2 < best_c5:
                best_c5 = c5p2
                best_h = hp2.copy()
        # One more SLSQP
        if time.time() < d7:
            g_full = _g_vec(best_h)
            active = get_active(g_full, frac=0.6, min_count=20)
            h_f2 = slsqp_solve(best_h, active, maxiter=300, ftol=1e-14)
            c5_f2 = exact_c5(h_f2)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    # Ensure bounds
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, len(best_h)
