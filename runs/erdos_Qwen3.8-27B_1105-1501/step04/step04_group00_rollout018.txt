def project(v, target=None):

        else:
            hi = mid

def build_grad_rows(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    m = len(active_indices)
    G = np.zeros((m, n))
    for k in range(m):
        outidx = int(active_indices[k])
        d = outidx - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[k, d:d+length] += dx * (1.0 - h[:length])
                G[k, :length] -= dx * h[d:d+length]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[k, :length] += dx * (1.0 - h[g:])
                G[k, g:g+length] -= dx * h[:length]
    return G

def solve_kkt_direction(G, q, residual, ridge, rough, n):
    # G is m x n, q is m, residual is m
    # Solve: min sum q_i (G_i . p - residual_i)^2 + ridge*||p||^2 + rough*||Dp||^2
    # s.t. sum(p) = 0
    
    # H = G^T diag(q) G + ridge*I + rough*D^T D
    H = (G * q[:, None]).T @ G
    H += ridge * np.eye(n)
    
    if rough > 0:
        # D is (n-1) x n difference matrix
        idx = np.arange(n-1)
        H[idx, idx] += rough
        H[idx+1, idx+1] += rough
        H[idx, idx+1] -= rough
        H[idx+1, idx] -= rough
    
    # RHS: G^T diag(q) residual
    rhs_vec = G.T @ (q * residual)
    
    # KKT system:
    # [H  1] [p]   [rhs_vec]
    # [1^T 0] [mu] = [0]
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n+1)
    rhs[:n] = rhs_vec
    
    try:
        sol = np.linalg.solve(KKT, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(KKT, rhs, rcond=1e-14)[0]
        except Exception:
            return None
    
    p = sol[:n]
    p -= p.mean()  # ensure sum zero
    if not np.all(np.isfinite(p)):
        return None
    if np.max(np.abs(p)) < 1e-14:
        return None
    return p

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad_rows(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    m = len(active_indices)
    G = np.zeros((m, n))
    for k in range(m):
        outidx = int(active_indices[k])
        d = outidx - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[k, d:d+length] += dx * (1.0 - h[:length])
                G[k, :length] -= dx * h[d:d+length]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[k, :length] += dx * (1.0 - h[g:])
                G[k, g:g+length] -= dx * h[:length]
    return G

def _solve_kkt(G, q, residual, ridge, rough, n):
    H = (G * q[:, None]).T @ G
    H += ridge * np.eye(n)
    if rough > 0:
        idx = np.arange(n - 1)
        H[idx, idx] += rough
        H[idx+1, idx+1] += rough
        H[idx, idx+1] -= rough
        H[idx+1, idx] -= rough
    rhs_vec = G.T @ (q * residual)
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n+1)
    rhs[:n] = rhs_vec
    try:
        sol = np.linalg.solve(KKT, rhs)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(KKT, rhs, rcond=1e-14)[0]
        except Exception:
            return None
    p = sol[:n]
    p -= p.mean()
    if not np.all(np.isfinite(p)):
        return None
    if np.max(np.abs(p)) < 1e-14:
        return None
    return p

def _lp_direction(h, scores, active_idx, radius):
    n = len(h)
    G = _build_grad_rows(h, active_idx)
    s_act = scores[active_idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -s_act
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, trust=0.35):
    n = len(h)
    pm = np.max(np.abs(p))
    if pm < 1e-14:
        return 0.0, _exact_c5(h)
    p_unit = p / pm
    alpha_max = trust
    for j in range(n):
        if p_unit[j] > 1e-14:
            cap = (1.0 - h[j]) / (pm * p_unit[j])
            if cap < alpha_max:
                alpha_max = cap
        elif p_unit[j] < -1e-14:
            cap = h[j] / (-pm * p_unit[j])
            if cap < alpha_max:
                alpha_max = cap
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 12
    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i not in (0, n_samp - 1):
        lo = alphas[max(0, best_i - 1)]
        hi = alphas[min(n_samp - 1, best_i + 1)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(20):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        opt_a = (lo + hi) / 2.0
        opt_c5 = _exact_c5(h + opt_a * p)
        if opt_c5 < best_c5:
            best_a = opt_a
            best_c5 = opt_c5
    return best_a, best_c5

def _kkt_direction(h, scores, C, cfg):
    n = len(h)
    active_tol = cfg['active_tol']
    min_act = cfg['min_active']
    max_act = cfg['max_active']
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[scores[sorted_idx] >= C - active_tol]
    if len(active) < min_act:
        active = sorted_idx[:min_act]
    if len(active) > max_act:
        active = sorted_idx[:max_act]
    if len(active) < 2:
        return None
    s_act = scores[active]
    cmin_a = float(np.min(s_act))
    spread = C - cmin_a
    margin = max(cfg['margin_floor'], cfg['margin_factor'] * spread)
    target = cmin_a - margin
    gaps = C - s_act
    max_gap = float(np.max(gaps)) if np.max(gaps) > 1e-15 else 1.0
    q = 1.0 + cfg['priority_power'] * (gaps / max_gap) ** 2
    G = _build_grad_rows(h, active)
    if G is None or not np.all(np.isfinite(G)):
        return None
    residual = target - s_act
    p = _solve_kkt(G, q, residual, cfg['ridge'], cfg['rough'], n)
    if p is not None:
        return p
    p_fb = -(G.T @ q)
    p_fb -= p_fb.mean()
    if np.max(np.abs(p_fb)) > 1e-14:
        return p_fb
    return None

def _local_opt(h, deadline, variants, trust_init=0.30, use_lp=True, max_iter=5000):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    trust = trust_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        cfg = variants[it % len(variants)]
        scores = _scores_exact(h)
        C = float(np.max(scores))
        p = _kkt_direction(h, scores, C, cfg)
        method = 0
        if p is None and use_lp:
            active_tol = cfg['active_tol']
            sorted_idx = np.argsort(scores)[::-1]
            n_act = min(cfg['max_active'], max(8, int(0.15 * len(scores))))
            active = sorted_idx[:n_act]
            p = _lp_direction(h, scores, active, trust)
            method = 1
        if p is not None:
            alpha, new_c5 = _line_search(h, p, trust)
            if new_c5 < cur_c5 - 1e-13:
                h_new = h + alpha * p
                h_new = _project(h_new, n / 2.0)

                    h = h_new
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:

                    stall = 0
                    trust = min(trust * 1.04, 0.45)
                    continue

        trust *= 0.88
        if trust < 0.015:
            trust = 0.12
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, scale, rng):
    n = len(h)
    x = np.arange(n) / n
    noise = np.zeros(n)
    for k in range(1, 9):
        amp = scale / (k + 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        noise += amp * np.sin(2 * np.pi * k * x + phase)
    return _project(h + noise, n / 2.0)

def _binary_sa_init(n, alpha, rng):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

    # Phase 6: Ultra-tight polish (6%)
    dl6 = hard_dl
    for pass_i in range(4):
        if time.time() > dl6 - 0.3:
            break
        sub_dl = min(dl6 - 0.3, time.time() + budget_s * 0.015)
        h_try, c5_try = _local_opt(best_h, sub_dl, var_tight, trust_init=0.04 * (0.75 ** pass_i))
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Final
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
