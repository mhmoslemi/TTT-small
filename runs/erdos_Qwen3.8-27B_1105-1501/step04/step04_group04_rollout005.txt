def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(300):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v


def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -700, 700)
    exp_w = np.exp(shifted)
    w_sum = np.sum(exp_w)
    w = exp_w / w_sum
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n-1+k]
        if wk < 1e-14:
            continue
        lo1 = max(0, -k)
        hi1 = min(n-1, n-1-k)
        lo2 = max(0, k)
        hi2 = min(n-1, n-1+k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * dx * (1.0 - h[lo1+k:hi1+k+1])
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wk * dx * h[lo2-k:hi2-k+1]
    return grad, cmax

def _lp_step(h, radius, deadline=None):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i_arr = np.arange(low, high)
        j_arr = i_arr + d
        G[r, j_arr] += dx * (1.0 - h[i_arr])
        G[r, i_arr] += -dx * h[j_arr]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)


def _local_coordinate(h, deadline, rng, max_iters=500000):

        hi_val = h[i]

            no_imp += 1
            continue
        delta = float(rng.uniform(-0.05, 0.05))
        new_val = max(0.0, min(1.0, hi_val + delta))

            cur_c5 = tc

            no_imp = 0
        else:
