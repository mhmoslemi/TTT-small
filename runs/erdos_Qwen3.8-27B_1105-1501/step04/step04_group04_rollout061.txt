def _project_box_sum(v, target):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}."""
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(100):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-15
        else:
            mask = v > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _lp_step_active(h, radius, k_active=20):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    
    # Compute current scores
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Find top-k active indices
    k = min(k_active, L)
    active_idx = np.argsort(F)[-k:]
    
    # Build G for active rows only
    # For each active lag d, the gradient of F_d with respect to h
    G_active = np.zeros((k, n))
    for idx in range(k):
        r = active_idx[idx]
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            j_from = i
            j_to = i + d
            # ∂F_d/∂h[i] includes:
            # (1-h[i+d])*dx if i+d < n (first factor)
            # -h[i-d]*dx if i-d >= 0 (second factor, from term h[i-d]*(1-h[i]))
            G_active[idx, i] += dx * (1.0 - h[i + d])
            if i - d >= 0:
                G_active[idx, i] -= dx * h[i - d]
    
    # LP: minimize t
    # s.t. F[r] + G_active[idx] @ (x - h) <= t  for each active idx
    #      0 <= x_i <= 1
    #      sum(x) = n/2
    #      |x_i - h_i| <= radius (box constraint)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = G_active @ h - F[active_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _build_G_row(h, d, n):
    """Build gradient row for correlation at lag d."""
    dx = 2.0 / n
    row = np.zeros(n)
    lo = max(0, -d)
    hi = min(n, n - d)
    for i in range(lo, hi):
        row[i] += dx * (1.0 - h[i + d])
        if i - d >= 0:
            row[i] -= dx * h[i - d]
    return row

def _lp_active_step(h, radius, k_active):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    k = min(k_active, L)
    active_idx = np.argsort(F)[-k:]
    G = np.zeros((k, n))
    for idx in range(k):
        r = active_idx[idx]
        d = r - (n - 1)
        G[idx] = _build_G_row(h, d, n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 8000})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_start=0.12, k_active=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    radius = radius_start
    stall = 0
    factors = [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]
    max_iters = 100000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        x_new = _lp_active_step(h, radius, k_active)
        if x_new is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        x_new = _project_box_sum(x_new, n / 2.0)
        best_step = 0.0
        best_tc5 = cur_c5
        for factor in factors:
            trial = _project_box_sum(h + factor * (x_new - h), n / 2.0)
            tc5 = _exact_c5(trial)
            if tc5 < best_tc5 - 1e-13:
                best_tc5 = tc5
                best_step = factor
        if best_step > 0:
            h = _project_box_sum(h + best_step * (x_new - h), n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.30)
        else:
            stall += 1
            radius *= 0.70
    return best_h, best_c5

def _grad_descent_loop(h, deadline, beta_start=10.0, beta_end=2000.0, n_beta=10):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    beta_list = np.geomspace(beta_start, beta_end, n_beta)
    for beta in beta_list:
        if time.time() > deadline:
            break
        stall = 0
        step_size = 0.06
        max_inner = 3000
        for _ in range(max_inner):
            if time.time() > deadline:
                break
            if stall > 50:
                break
            scores = np.correlate(h, 1.0 - h, mode='full') * dx
            max_s = float(np.max(scores))
            shifted = scores - max_s
            exp_v = np.exp(beta * shifted)
            w = exp_v / (np.sum(exp_v) + 1e-300)
            g = np.zeros(n)
            L = 2 * n - 1
            for r in range(L):
                if w[r] < 1e-15:
                    continue
                d = r - (n - 1)
                row = _build_G_row(h, d, n)
                g += w[r] * row
            g -= np.mean(g)
            gn = float(np.max(np.abs(g)))
            if gn < 1e-15:
                break
            d = -g / gn * step_size
            best_a = 0.0
            best_tc5 = cur_c5
            for a in [1.0, 0.6, 0.3, 0.1]:
                trial = _project_box_sum(h + a * d, n / 2.0)
                tc5 = _exact_c5(trial)
                if tc5 < best_tc5 - 1e-13:
                    best_tc5 = tc5
                    best_a = a
            if best_a > 0:
                h = _project_box_sum(h + best_a * d, n / 2.0)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step_size = min(step_size * 1.06, 0.35)
            else:
                stall += 1
                step_size *= 0.75
    return best_h, best_c5

def _fast_local_search(h, deadline, rng, max_iters=500000):
    n = len(h)
    dx = 2.0 / n
    cur_scores = _scores_raw(h).copy()
    cur_c5 = float(np.max(cur_scores) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    step_size = 0.04
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 5000:
            break
        i = int(rng.integers(n))
        old = h[i]
        if old < 1e-9 or old > 1.0 - 1e-9:
            no_improve += 1
            continue
        delta = step_size * (2.0 * float(rng.random()) - 1.0)
        new_val = old + delta
        if new_val < 0.0 or new_val > 1.0:
            continue
        # Compute affected score indices
        k_lo_first = n - 1 - i
        k_hi_first = 2 * n - 2 - i
        k_lo_second = i
        k_hi_second = n - 1 + i
        lo = max(0, min(k_lo_first, k_lo_second))
        hi = min(2 * n - 2, max(k_hi_first, k_hi_second))
        if hi < lo:
            no_improve += 1
            continue
        kvals = np.arange(lo, hi + 1)
        # Compute deltas
        deltas = np.zeros(len(kvals))
        for jj in range(len(kvals)):
            kk = kvals[jj]
            d_lag = kk - (n - 1)
            contrib = 0.0
            if 0 <= i + d_lag < n:
                contrib += delta * (1.0 - h[i + d_lag])
            if 0 <= i - d_lag < n:
                contrib -= delta * h[i - d_lag]
            deltas[jj] = contrib
        cur_scores[kvals] += deltas
        h[i] = new_val
        new_c5 = float(np.max(cur_scores) * dx)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
            step_size = min(step_size * 1.04, 0.30)
        else:
            cur_scores[kvals] -= deltas
            h[i] = old
            no_improve += 1
            step_size *= 0.92
    # Final projection
    best_h = _project_box_sum(best_h, n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5 + 1e-12:
        best_c5 = final_c5
        best_h = best_h.copy()
    return best_h, best_c5

def _pair_search(h, deadline, rng, max_iters=200000):
    n = len(h)
    dx = 2.0 / n
    cur_scores = _scores_raw(h).copy()
    cur_c5 = float(np.max(cur_scores) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    step_size = 0.03
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 4000:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if h[i] < 1e-9 or h[i] > 1.0 - 1e-9 or h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
            no_improve += 1
            continue
        d = step_size * (2.0 * float(rng.random()) - 1.0)
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        # Update scores for both i and j
        k_lo_i = max(0, min(n - 1 - i, i))
        k_hi_i = min(2 * n - 2, max(2 * n - 2 - i, n - 1 + i))
        k_lo_j = max(0, min(n - 1 - j, j))
        k_hi_j = min(2 * n - 2, max(2 * n - 2 - j, n - 1 + j))
        lo = min(k_lo_i, k_lo_j)
        hi = max(k_hi_i, k_hi_j)
        kvals = np.arange(lo, hi + 1)
        deltas = np.zeros(len(kvals))
        for jj in range(len(kvals)):
            kk = kvals[jj]
            d_lag = kk - (n - 1)
            contrib = 0.0
            # i contribution
            if 0 <= i + d_lag < n:
                contrib += d * (1.0 - h[i + d_lag])
            if 0 <= i - d_lag < n:
                contrib -= d * h[i - d_lag]
            # j contribution (with -d)
            if 0 <= j + d_lag < n:
                contrib += (-d) * (1.0 - h[j + d_lag])
            if 0 <= j - d_lag < n:
                contrib -= (-d) * h[j - d_lag]
            deltas[jj] = contrib
        cur_scores[kvals] += deltas
        h[i] = ni
        h[j] = nj
        new_c5 = float(np.max(cur_scores) * dx)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
            step_size = min(step_size * 1.04, 0.25)
        else:
            cur_scores[kvals] -= deltas
            h[i] = ni - d
            h[j] = nj + d
            no_improve += 1
            step_size *= 0.92
    best_h = _project_box_sum(best_h, n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5 + 1e-12:
        best_c5 = final_c5
    return best_h, best_c5

def _optimize_at_resolution(h, deadline, rng):
    """Run a full optimization pipeline at a given resolution."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    # LP phase
    if time.time() < deadline - 30:
        dl = min(deadline - 30, time.time() + (deadline - time.time()) * 0.4)
        h1, c5_1 = _lp_loop(best_h, dl, radius_start=0.10, k_active=25)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    # Gradient descent
    if time.time() < deadline - 20:
        dl = min(deadline - 20, time.time() + (deadline - time.time()) * 0.3)
        h2, c5_2 = _grad_descent_loop(best_h, dl, beta_start=10.0, beta_end=1000.0, n_beta=8)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    # Local search
    if time.time() < deadline - 10:
        dl = min(deadline - 10, time.time() + (deadline - time.time()) * 0.5)
        h3, c5_3 = _fast_local_search(best_h, dl, rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    # Pair search
    if time.time() < deadline - 5:
        dl = min(deadline - 5, time.time() + (deadline - time.time()) * 0.5)
        h4, c5_4 = _pair_search(best_h, dl, rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    best_h = _project_box_sum(best_h, n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    return best_h, best_c5

def _make_diverse_starts(h, rng):
    """Create diverse starting points from h."""
    n = len(h)
    starts = [h.copy()]
    # Smoothed version
    kernel = np.array([1.0, 2.0, 3.0, 2.0, 1.0]) / 9.0
    smoothed = np.convolve(h, kernel, mode='same')
    starts.append(_project_box_sum(smoothed, n / 2.0))
    # Perturbed version
    perturbed = h + 0.05 * (rng.random(n) - 0.5)
    starts.append(_project_box_sum(perturbed, n / 2.0))
    # Reflected version
    starts.append(_project_box_sum(h[::-1].copy(), n / 2.0))
    # Complemented version
    starts.append(_project_box_sum(1.0 - h, n / 2.0))
    # Another perturbation
    perturbed2 = h + 0.1 * (rng.random(n) - 0.5)
    starts.append(_project_box_sum(perturbed2, n / 2.0))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        n0 = 64
        h0 = np.zeros(n0)
        h0[np.arange(0, n0, 2)] = 1.0
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP trust-region at parent resolution (25% of budget)
    deadline1 = t0 + budget_s * 0.22
    h1, c5_1 = _lp_loop(h, deadline1, radius_start=0.10, k_active=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient descent at parent resolution (12%)
    deadline2 = t0 + budget_s * 0.34
    h2, c5_2 = _grad_descent_loop(best_h, deadline2, beta_start=8.0, beta_end=2000.0, n_beta=10)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Fast local search at parent resolution (12%)
    deadline3 = t0 + budget_s * 0.46
    h3, c5_3 = _fast_local_search(best_h, deadline3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Resolution exploration (35%)
    deadline4 = t0 + budget_s * 0.81
    n_candidates = sorted(set([
        best_n, best_n - 16, best_n + 16,
        96, 112, 128, 144,
        64, 80, 48, 32, 24, 16
    ]))
    n_candidates = [x for x in n_candidates if 12 <= x <= 192]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    # Allocate time proportionally to closeness to current best_n
    remaining = deadline4 - time.time()
    if remaining > 30:
        # Give more time to smaller n (preferred) and n close to current best
        weights = {}
        for nc in n_candidates:
            if nc == best_n:
                weights[nc] = 0.15
            elif nc < best_n:
                weights[nc] = 0.20 / max(1, (best_n - nc) // 16 + 1)
            else:
                weights[nc] = 0.10 / max(1, (nc - best_n) // 16 + 1)
        total_w = sum(weights.values())
        for nc in n_candidates:
            if time.time() > deadline4 - 10:
                break
            if nc == best_n:
                continue
            t_alloc = remaining * weights[nc] / total_w
            t_alloc = max(15.0, min(t_alloc, remaining * 0.15))
            dl_r = min(deadline4, time.time() + t_alloc)
            h_r = _resample(best_h, nc)
            h_r, c5_r = _optimize_at_resolution(h_r, dl_r, rng)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = nc
            elif nc < best_n and c5_r < best_c5 + 1e-10:
                # Prefer smaller n if C5 is essentially tied
                best_h = h_r.copy()
                best_n = nc

    # Phase 5: Final LP polish at best resolution (10%)
    if time.time() < t0 + budget_s * 0.92:
        deadline5 = t0 + budget_s * 0.92
        h5, c5_5 = _lp_loop(best_h, deadline5, radius_start=0.05, k_active=25)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Final local search (5%)
    if time.time() < t0 + budget_s * 0.97:
        deadline6 = t0 + budget_s * 0.97
        h6, c5_6 = _fast_local_search(best_h, deadline6, rng)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final projection and validation
    bn = len(best_h)
    best_h = _project_box_sum(best_h, bn / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, bn / 2.0)
        best_c5 = _exact_c5(best_h)

    return best_h, best_c5, bn
