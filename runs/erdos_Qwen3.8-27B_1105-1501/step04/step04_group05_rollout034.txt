import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_correlation(h, m):
    """Gradient of C_m (m-th correlation value, scaled by dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    d = m - (n - 1)  # the lag
    g = np.zeros(n)
    if d >= 0:
        # C_m/dx = sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
        # d/dh[j] = dx * [(1-h[j-d]) if d<=j<=n-1] + dx * [(-h[j+d]) if 0<=j<=n-1-d]
        if d < n:
            # First term: j from d to n-1
            g[d:] += dx * (1.0 - h[:n-d])
            # Second term: j from 0 to n-1-d
            g[:n-d] -= dx * h[d:]
    else:
        p = -d
        # C_m/dx = sum_{i=p}^{n-1} h[i-p] * (1-h[i])
        # d/dh[j] = dx * [(1-h[j+p]) if 0<=j<=n-1-p] + dx * [(-h[j-p]) if p<=j<=n-1]
        if p < n:
            # First term: j from 0 to n-1-p
            g[:n-p] += dx * (1.0 - h[p:])
            # Second term: j from p to n-1
            g[p:] -= dx * h[:n-p]
    return g

def _lp_vertex(w, n):
    """Find vertex of {0<=x<=1, sum(x)=n/2} minimizing w^T x."""
    target = n / 2.0
    order = np.argsort(w, kind='stable')
    s = np.zeros(n)
    s[order[:int(target)]] = 1.0
    return s

def _fw_step(h, h_score, eta, top_k, deadline):
    """One FW iteration. Returns (new_h, new_score) or (h, h_score) if no improvement."""
    n = len(h)
    C = _corr_scaled(h)
    lam = float(np.max(C))
    
    # Softmax weights over top k correlations
    sort_idx = np.argsort(C)[::-1]
    top_idx = sort_idx[:top_k]
    top_vals = C[top_idx]
    shifted = (top_vals - lam) / eta
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Compute weighted subgradient
    g = np.zeros(n)
    for i, idx in enumerate(top_idx):
        g += weights[i] * _grad_correlation(h, idx)
    
    # LP vertex
    s = _lp_vertex(g, n)
    
    # Line search
    best_alpha = 0.0
    best_val = h_score
    alphas = np.linspace(0.0, 1.0, 33)
    for a in alphas:
        h_t = (1.0 - a) * h + a * s
        val = _exact_c5(h_t)
        if val < best_val - 1e-14:
            best_val = val
            best_alpha = a
    
    if best_alpha == 0.0:
        return h, h_score
    
    # Refine around best_alpha
    if best_alpha > 0 and best_alpha < 1:
        span = 1.0 / 32.0
        ref_alphas = np.linspace(max(0, best_alpha - span), min(1, best_alpha + span), 13)
        for a in ref_alphas:
            if time.time() > deadline:
                break
            h_t = (1.0 - a) * h + a * s
            val = _exact_c5(h_t)
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = a
    
    h_new = (1.0 - best_alpha) * h + best_alpha * s
    # Ensure exact feasibility
    h_new = _project(h_new, n / 2.0)
    final_val = _exact_c5(h_new)
    if final_val < h_score - 1e-14:
        return h_new, final_val
    return h, h_score

import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _grad_single(h, m):
    """Gradient of the m-th scaled correlation w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += dx * (1.0 - h[:n - d])
            g[:n - d] -= dx * h[d:]
    else:
        p = -d
        if p < n:
            g[:n - p] += dx * (1.0 - h[p:])
            g[p:] -= dx * h[:n - p]
    return g


def _lp_vertex(w, n):
    """Vertex of {0<=x<=1, sum(x)=n/2} minimizing w^T x."""
    target = int(n // 2)
    order = np.argsort(w, kind='stable')
    s = np.zeros(n)
    s[order[:target]] = 1.0
    return s


def _fw_single_step(h, h_score, eta, top_k, deadline):
    """One FW iteration."""
    n = len(h)
    C = _corr_scaled(h)
    lam = float(np.max(C))
    
    # Select top-k correlations
    k = min(top_k, 2 * n - 1)
    sort_idx = np.argsort(C)[::-1][:k]
    top_vals = C[sort_idx]
    
    # Softmax weights
    shifted = (top_vals - lam) / max(eta, 1e-10)
    w_exp = np.exp(shifted)
    w_sum = np.sum(w_exp)
    if w_sum < 1e-30:
        w_exp = np.ones(k) / k
    else:
        w_exp /= w_sum
    
    # Weighted subgradient
    g = np.zeros(n)
    for i in range(k):
        g += w_exp[i] * _grad_single(h, sort_idx[i])
    
    # LP vertex
    s = _lp_vertex(g, n)
    
    # Check if s == h (no progress possible)
    if np.max(np.abs(s - h)) < 1e-10:
        return h, h_score
    
    # Line search
    best_alpha = 0.0
    best_val = h_score
    n_grid = 25
    for a in np.linspace(0.0, 1.0, n_grid):
        if time.time() > deadline:
            break
        h_t = (1.0 - a) * h + a * s
        val = _exact_c5(h_t)
        if val < best_val - 1e-14:
            best_val = val
            best_alpha = a
    
    if best_alpha == 0.0:
        return h, h_score
    
    # Refine
    if 0 < best_alpha < 1.0:
        span = 1.0 / n_grid
        lo_r = max(0.0, best_alpha - span)
        hi_r = min(1.0, best_alpha + span)
        for a in np.linspace(lo_r, hi_r, 11):
            if time.time() > deadline:
                break
            h_t = (1.0 - a) * h + a * s
            val = _exact_c5(h_t)
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = a
    
    h_new = (1.0 - best_alpha) * h + best_alpha * s
    h_new = _project(h_new, n / 2.0)
    final_val = _exact_c5(h_new)
    
    if final_val < h_score - 1e-13:
        return h_new, final_val
    return h, h_score

Actually, I just realized something important. Let me re-examine the objective.

C5 = max_k integral h(x)(1-h(x+k)) dx

With the discretization, this is:
C5 = max_k sum_i h[i] * (1-h[(i+k) mod n]) * dx  ... if it's circular

But the problem uses linear correlation, not circular. So:
C5 = max_k sum_i h[i] * (1-h[i+k]) * dx

where the sum is over valid i (where i+k is in range).

This is the linear (non-periodic) version. For large lags |k|, the overlap is with a shorter window, so the contribution is smaller.

The interesting lags are the small ones (|k| << n), where the overlap window is large.

For the FW approach, the subgradient at a given point will mainly be influenced by the lags that are near the maximum. These are typically the small lags.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target: lo = mid
        else: hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def _corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_single(h, m):
    n = len(h)
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if 0 <= d < n:
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
    elif -n < d < 0:
        p = -d
        g[:n-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n-p]
    return g

def _lp_vertex(w, n):
    s = np.zeros(n)
    s[np.argsort(w, kind='stable')[:n//2]] = 1.0
    return s

def _fw_step(h, score, eta, k, deadline):
    n = len(h)
    C = _corr_scaled(h)
    lam = float(np.max(C))
    k = min(k, 2*n-1)
    idx = np.argsort(C)[::-1][:k]
    tv = C[idx]
    w = np.exp((tv - lam) / max(eta, 1e-10))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(k):
        g += w[i] * _grad_single(h, idx[i])
    s = _lp_vertex(g, n)
    if np.max(np.abs(s - h)) < 1e-10:
        return h, score
    best_a, best_v = 0.0, score
    for a in np.linspace(0, 1, 25):
        if time.time() > deadline: break
        v = _exact_c5((1-a)*h + a*s)
        if v < best_v - 1e-14: best_v, best_a = v, a
    if best_a == 0: return h, score
    # refine
    sp = 1.0/24
    for a in np.linspace(max(0,best_a-sp), min(1,best_a+sp), 11):
        if time.time() > deadline: break
        v = _exact_c5((1-a)*h + a*s)
        if v < best_v - 1e-14: best_v, best_a = v, a
    hn = (1-best_a)*h + best_a*s
    hn = _project(hn, n/2.0)
    fv = _exact_c5(hn)
    return (hn, fv) if fv < score - 1e-13 else (h, score)

def _fw_loop(h, deadline, max_iter=3000):
    n = len(h)
    best_h, best_c5 = h.copy(), _exact_c5(h)
    hc, cc = h.copy(), best_c5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline: break
        C = _corr_scaled(hc)
        lam = float(np.max(C))
        frac = min(it / 200.0, 1.0)
        eta = max(0.03 * lam * (1.0 - 0.7*frac), 1e-6)
        k = max(3, int(8 - 4*frac))
        hn, cn = _fw_step(hc, cc, eta, k, deadline)
        if cn < cc - 1e-13:
            hc, cc = hn, cn
            stall = 0
            if cn < best_c5: best_c5, best_h = cn, hc.copy()

            if stall > 80:

                    stall = 0
                    if cn2 < best_c5: best_c5, best_h = cn2, hc.copy()
                else:

    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    best_h, best_c5 = h.copy(), _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline: break
        improved = False
        C = _corr_scaled(best_h)
        # Compute sensitivity
        lam = float(np.max(C))
        active = np.where(C >= lam * 0.95)[0]
        sens = np.zeros(n)
        for m in active:
            g = np.abs(_grad_single(best_h, int(m)))
            sens = np.maximum(sens, g)
        # Pick top pairs by sensitivity product
        order = np.argsort(sens)[::-1]
        candidates = order[:min(24, n)]
        for ii in range(len(candidates)):
            if time.time() > deadline: break
            i = candidates[ii]
            for jj in range(ii+1, len(candidates)):
                j = candidates[jj]
                p = best_h[i] + best_h[j]
                lo = max(0.0, p - 1.0)
                hi = min(1.0, p)
                if hi - lo < 1e-9: continue
                best_z = (lo + hi) / 2
                best_v = _exact_c5(best_h)
                for z in np.linspace(lo, hi, 15):
                    ht = best_h.copy()
                    ht[i], ht[j] = z, p - z
                    v = _exact_c5(ht)
                    if v < best_v - 1e-13:
                        best_v, best_z = v, z
                if best_v < best_c5 - 1e-13:
                    best_h[i], best_h[j] = best_z, p - best_z
                    best_c5 = _exact_c5(best_h)
                    improved = True
                    break
            if improved: break
        if not improved: break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    xo = (np.arange(n) + 0.5) * 2.0 / n
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)
