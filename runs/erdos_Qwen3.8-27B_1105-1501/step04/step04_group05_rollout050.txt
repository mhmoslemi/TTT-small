                if hi - lo < 1e-12:
                    continue
                a, b = lo, hi
                c = b - (1.0 / inv_gr) * (b - a)
                d2 = a + (1.0 / inv_gr) * (b - a)
                h[i], h[j] = c, s - c
                fc = _score(h)
                h[i], h[j] = d2, s - d2
                fd = _score(h)
                for _ in range(golden_iters):
                    if fc < fd:
                        b = d2
                        d2 = c
                        fd = fc
                        c = b - (1.0 / inv_gr) * (b - a)
                        h[i], h[j] = c, s - c
                        fc = _score(h)
                    else:
                        a = c
                        c = d2
                        fc = fd
                        d2 = a + (1.0 / inv_gr) * (b - a)
                        h[i], h[j] = d2, s - d2
                        fd = _score(h)
                if fc <= fd:
                    h[i], h[j] = c, s - c
                    sc = fc
                else:
                    h[i], h[j] = d2, s - d2
                    sc = fd
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    best_h = _project(best_h, target)
    return best_h, _score(best_h)


def _greedy_mask(mask, n, m, max_rounds=200, deadline=None):
    """Iterated improvement on binary mask via best single swap."""
    h = mask.astype(float)
    cur = _score(h)
    best = cur
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        best_swap = None
        best_swap_score = cur
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = _score(h)
                h[oi] = 1.0
                h[zi] = 0.0
                if sc < best_swap_score - 1e-15:
                    best_swap_score = sc
                    best_swap = (oi, zi)
        if best_swap is None:
            break
        oi, zi = best_swap
        h[oi] = 0.0
        h[zi] = 1.0
        cur = best_swap_score
        if cur < best:
            best = cur
    return h, best
