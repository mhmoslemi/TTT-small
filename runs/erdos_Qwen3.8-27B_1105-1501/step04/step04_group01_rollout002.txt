```python
A_ub[:, n] = -1.0
b_ub = -scores
```

So the LP is:
minimize t (the last variable)
subject to: G @ p - t <= -scores  (i.e., G @ p + scores <= t)
            sum(p) = 0
            bounds on p

This is: find the minimum t such that scores + G@p <= t for all lags, with p being a perturbation. So t is the new maximum correlation value (linearized). The solution gives both the direction p and the target t.

Then a line search follows to actually move in direction p.

This is reasonable but it's a linearization. The key issue is that after the move, the linearization becomes stale.

**My improvements**:
1. **Iterative relinearization**: After each LP + line search, immediately recompute the gradient and try again. (The previous code does this in the loop, but I can make it more efficient.)

2. **Top-K active set**: Only use the top K lags in the LP, not all 2n-1. This makes the LP faster and more focused.

3. **Adaptive K**: Start with small K (e.g., 5) and increase when no progress is made.

4. **Block moves**: In addition to the LP direction, try moving contiguous blocks of h values.

5. **Grid size exploration**: Try a few different grid sizes.

6. **Better initial perturbation**: Instead of random Gaussian, use structured perturbations (e.g., sinusoidal).

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(40):
        if b - a < 1e-13: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    return opt_t, opt_val

def _build_grad_matrix(h, lags):
    """Build gradient matrix for specified lags only."""
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, lag in enumerate(lags):
        k = lag - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[idx, :m] += (1.0 - h[k:])
                G[idx, k:] += (-h[:m])
        else:
            g = -k
            m = n - g
            if m > 0:
                G[idx, :m] += (-h[g:])
                G[idx, g:] += (1.0 - h[:m])
    return G

def _lp_direction(h, scores, top_k, radius):
    """Find descent direction using top-k active lags."""
    n = len(h)
    # Get the top-k lags
    order = np.argsort(scores)[::-1]
    active = order[:top_k]
    active_scores = scores[active]
    
    G = _build_grad_matrix(h, active)
    
    # LP: min t s.t. G @ p <= active_scores - t, sum(p)=0, bounds
    # Wait, I want to minimize the max of the top-k correlations.
    # min t s.t. active_scores + G @ p <= t
    #          sum(p) = 0
    #          bounds on p
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -active_scores
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo_j = max(-radius, -h[j])
        hi_j = min(radius, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _optimize_active(h, deadline, top_k_init=8, radius_init=0.15):
    """Main optimization loop with active set management."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    
    while time.time() < deadline:
        if stall > 80:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, top_k, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            top_k = min(top_k + 2, 2*n - 1)
            if radius < 1e-6 and top_k >= 2*n - 1:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.30)
            top_k = max(4, top_k - 1)
        else:
            stall += 1
            radius *= 0.80
            top_k = min(top_k + 1, 2*n - 1)
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.08
                top_k = max(4, top_k - 3)
    
    return best_h, best_c5

def _block_move(h, dx, deadline):
    """Try contiguous block moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    
    while time.time() < deadline and stall < 100:
        # Try a random block move: increase a block, decrease another
        block_len = int(rng.integers(2, max(3, n // 8) + 1))
        i = int(rng.integers(0, n - 2 * block_len))
        j = int(rng.integers(0, n - 2 * block_len))
        if j < i + block_len:
            j = int(rng.integers(0, n - 2 * block_len))
            if j < i + block_len:
                continue
        
        d = np.zeros(n)
        d[i:i+block_len] = -1.0
        d[j:j+block_len] = 1.0
        
        alpha, new_c5 = _line_search(h, d, dx, 0.3)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _sinusoidal_perturb(h, n_harmonics=5, amp=0.05):
    """Generate a sinusoidal perturbation direction."""
    n = len(h)
    d = np.zeros(n)
    for k in range(1, n_harmonics + 1):
        d += np.sin(2 * np.pi * k * np.arange(n) / n + np.pi / 4)
    d -= d.mean()
    return d / (np.linalg.norm(d) + 1e-15)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
