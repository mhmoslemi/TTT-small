def _lp_direction(h, active_indices, grad_matrix, radius):
    """Find LP direction that minimizes max linearized active score increase.
    
    grad_matrix: shape (n_active, n), row r is the gradient of active score r.
    """
    n = len(h)
    n_act = len(active_indices)
    
    # Variables: d_0, ..., d_{n-1}, s
    # Minimize s
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: grad_r · d <= s for each active r
    # => grad_r · d - s <= 0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = grad_matrix
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_act)
    
    # Mean constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Box constraints on d
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))  # s is free
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            if float(np.max(np.abs(d))) > 1e-12:
                return d
    except Exception:
        pass
    return None

        if d == 0:
            G[idx] = 1.0 - 2.0 * h
        elif d > 0:
            j1 = np.arange(d, n)
            G[idx, j1] += (1.0 - h[j1 - d])
            j2 = np.arange(0, n - d)
            G[idx, j2] += (-h[j2 + d])
        else:
            g = -d
            j1 = np.arange(0, n - g)
            G[idx, j1] += (1.0 - h[j1 + g])
            j2 = np.arange(g, n)
            G[idx, j2] += (-h[j2 - g])
    return G

def _lp_dir(h, G, radius):
    n = len(h)
    n_act = G.shape[0]
    if n_act == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_act)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            if float(np.max(np.abs(d))) > 1e-12:
                return d
    except Exception:
        pass
    return None

def _subgrad_dir(h, tau, gap_tol=0.08):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    active = np.where(scores >= max_s - gap_tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    a_s = scores[active]
    w = np.exp(tau * (a_s - max_s))
    w = w / float(np.sum(w))
    G = _build_grad_mat(h, active)
    sg = G.T @ w
    sg -= np.mean(sg)
    return sg

def _optimize(h, deadline, radius_init=0.18, gap_init=0.08):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    gap = gap_init
    stall = 0
    tau = 30.0
    use_lp = True
    while time.time() < deadline:
        if stall > 50:
            break
        scores_norm = np.correlate(h, 1.0 - h, mode='full') * dx
        max_sn = float(np.max(scores_norm))
        active = np.where(scores_norm >= max_sn - gap)[0]
        top_k = min(len(active), 30)
        if top_k > 30:
            idx = np.argsort(scores_norm[active])[-30:]
            active = active[idx]
        G = _build_grad_mat(h, active)
        d = None
        if use_lp:
            d = _lp_dir(h, G, radius)
        if d is not None and float(np.max(np.abs(d))) > 1e-12:
            alpha, new_c5 = _line_search(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.40)
                gap = max(gap * 0.92, 0.01)
                tau = min(tau * 1.15, 200.0)
                continue
        # Fallback: subgradient
        sg = _subgrad_dir(h, tau, gap)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm > 1e-13:
            sg = sg / sg_norm
            pos = sg > 1e-10
            neg = sg < -1e-10
            max_pos = float(np.min((1.0 - h[pos]) / sg[pos])) if np.any(pos) else 1.0
            max_neg = float(np.max(h[neg] / (-sg[neg]))) if np.any(neg) else 1.0
            sc = min(max_pos, max_neg, radius)
            if sc > 1e-10:
                d = sg * sc
                alpha, new_c5 = _line_search(h, d, dx, radius)
                if new_c5 < cur_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.04, 0.35)
                    continue
        stall += 1
        radius *= 0.82
        gap = max(gap * 0.7, 0.005)
        tau = max(tau * 0.8, 10.0)
        if stall % 12 == 0 and radius < 0.02:
            radius = 0.08
            gap = 0.04
        if stall % 20 == 0:
            use_lp = not use_lp
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    improved_any = True
    passes = 0
    while improved_any and passes < max_passes and time.time() < deadline:
        improved_any = False
        passes += 1
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                hi_vals = (h[i] - h[j])
                if hi_vals < 0:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved_any = True
    return best_h, best_c5

def _triple_exchange(h, deadline, n_trials=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i, j, k = rng.choice(n, size=3, replace=False)
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -0.5
        d[k] = -0.5
        alpha, c5_try = _line_search(h, d, dx, 0.4)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gen_profile(n, method, seed):
    rng = np.random.default_rng(seed)
    target = n / 2.0
    if method == 0:
        x = np.linspace(0, 1, n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 4) * x + rng.uniform(0, 6.28))
    elif method == 1:
        h = 0.5 + 0.15 * np.sin(2 * np.pi * 2.0 * np.linspace(0, 1, n)) + \
            0.1 * np.sin(2 * np.pi * 3.5 * np.linspace(0, 1, n) + rng.uniform(0, 6.28))
    elif method == 2:
        h = np.clip(rng.normal(0.5, 0.25, n), 0, 1)
        h = _project(h, target)
        kernel = np.ones(3) / 3.0
        for _ in range(3):
            h = np.convolve(h, kernel, mode='same')
        h = _project(h, target)
    elif method == 3:
        h = 0.5
        h = np.full(n, 0.5)
        for _ in range(n // 4):
            c = rng.integers(0, n)
            w = rng.integers(1, n // 4)
            idx = np.arange(max(0, c - w), min(n, c + w + 1))
            h[idx] += rng.choice([-1, 1]) * rng.uniform(0.1, 0.3)
        h = _project(h, target)
    else:
        h = np.clip(0.5 + rng.normal(0, 0.3, n), 0, 1)
        h = _project(h, target)
    return _project(h, target)

def _smooth_perturb(h, sigma, seed):
    rng = np.random.default_rng(seed)
    n = len(h)
    noise = np.zeros(n)
    for f in range(1, 5):
        noise += rng.uniform(-1, 1) * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
    noise /= float(np.abs(noise).max() + 1e-15)
    h_new = h + sigma * noise
    return _project(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Aggressive optimization from parent (45%)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _optimize(h, d1, radius_init=0.20, gap_init=0.10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair/triple refinement (10%)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _pair_refine(best_h, min(d2, time.time() + budget_s * 0.06), max_passes=40)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    d2b = min(d2, time.time() + budget_s * 0.04)
    h2b, c5_2b = _triple_exchange(best_h, d2b, n_trials=300)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()

    # Phase 3: Re-optimize after pair refinement (10%)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _optimize(best_h, d3, radius_init=0.10, gap_init=0.04)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution search (20%)
    d4 = t0 + budget_s * 0.85
    best_n = len(best_h)
    n_tries = []
    for dn in [-8, 8, -6, 6, -10, 10, -4, 4, -12, 12, -16, 16, -20, 20, -24, 24]:
        nt = best_n + dn
        if 40 <= nt <= 200:
            n_tries.append(nt)
    n_tries = list(dict.fromkeys(n_tries))
    for nt in n_tries:
        if time.time() > d4:
            break
        h_r = _resample(best_h, nt)
        remaining = d4 - time.time()
        if remaining < 5:
            break
        sub_dl = min(d4, time.time() + min(remaining * 0.35, budget_s * 0.025))
        h_r, c5_r = _optimize(h_r, sub_dl, radius_init=0.15, gap_init=0.08)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt
            # Quick pair refine on the new best
            rem = d4 - time.time()
            if rem > 3:
                h_tp, c5_tp = _pair_refine(best_h, min(d4, time.time() + rem * 0.4), max_passes=15)
                if c5_tp < best_c5:
                    best_c5 = c5_tp
                    best_h = h_tp.copy()

    # Phase 5: Multi-start exploration (8%)
    d5 = t0 + budget_s * 0.93
    n_starts = 8
    for si in range(n_starts):
        if time.time() > d5:
            break
        method = si % 5
        h_s = _smooth_perturb(best_h, 0.06 + 0.02 * si, seed + 100 + si)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_s, c5_s = _optimize(h_s, sub_dl, radius_init=0.12, gap_init=0.06)
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Final polish (7%)
    d6 = t0 + budget_s * 1.0
    rem = max(0, d6 - time.time() - 2)
    if rem > 2:
        h_f, c5_f = _optimize(best_h, min(d6, time.time() + rem * 0.5), radius_init=0.06, gap_init=0.02)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    rem2 = max(0, d6 - time.time() - 2)
    if rem2 > 2:
        h_p, c5_p = _pair_refine(best_h, min(d6, time.time() + rem2), max_passes=30)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final: ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    sanity_check = evaluate_erdos_solution(best_h, len(best_h))
    if sanity_check is not None and sanity_check < best_c5:
        best_c5 = sanity_check

    return best_h, best_c5, len(best_h)
