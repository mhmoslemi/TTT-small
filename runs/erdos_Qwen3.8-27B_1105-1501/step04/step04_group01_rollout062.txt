import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    # Same as previous - golden section on quadratic approx
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(50):
        if t_hi - t_lo < 1e-12:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0

        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-13:

            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:

                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.02:

    return best_h, best_c5

def _swap_loop(h, deadline, max_passes=30):

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = t_hi - gr * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(45):
        if t_hi - t_lo < 5e-13:
            break
        if f1 < f2:
            t_hi = c2; c2 = c1; f2 = f1
            c1 = t_hi - gr * (t_hi - t_lo)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            t_lo = c1; c1 = c2; f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (t_lo + t_hi) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 70:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.68
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.28)
        else:
            stall += 1
            radius *= 0.70
            if stall % 18 == 0 and radius < 0.015:
                radius = 0.05
    return best_h, best_c5

def _all_pair_swap(h, deadline, max_passes=8):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)

                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):
    idx = rng.permutation(n)[:n//2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h
