            pairs_to_try = [pairs[k] for k in subset]
        else:
            pairs_to_try = list(pairs)
            rng.shuffle(pairs_to_try)
        improved = False
        for (i, j) in pairs_to_try:
            if time.time() > deadline:
                break
            delta = _pair_delta(h, i, j, G, c)
            if abs(delta) > 1e-11:
                h[i] += delta
                h[j] -= delta
                h[i] = max(0.0, min(1.0, h[i]))
                h[j] = max(0.0, min(1.0, h[j]))
                improved = True
                changes_since_recompute += 1
                if changes_since_recompute >= recompute_every:
                    c = _compute_c(h)
                    G = _compute_G(h)
                    changes_since_recompute = 0
                    new_c5 = float(np.max(c))
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        stagnation = 0
                    else:
                        stagnation += 1
                    cur_c5 = new_c5
        if not improved:
            break
        if stagnation > 50:
            break
    c = _compute_c(h)
    new_c5 = float(np.max(c))
    if new_c5 < best_c5:
        best_c5 = new_c5
        best_h = h.copy()
    return best_h, best_c5

def _lattice_anneal(h, deadline, m=8, max_moves=5000000, T_init=3e-4, T_min=1e-7, T_decay=0.9995, seed=42):
    n = len(h)
    target_sum = n * m // 2
    rng = np.random.default_rng(seed)
    q = np.round(h * m).astype(int)
    q = np.clip(q, 0, m)
    while q.sum() > target_sum:
        cands = np.where(q > 0)[0]
        if len(cands) == 0:
            break
        idx = rng.integers(0, len(cands))
        q[cands[idx]] -= 1
    while q.sum() < target_sum:
        cands = np.where(q < m)[0]
        if len(cands) == 0:
            break
        idx = rng.integers(0, len(cands))
        q[cands[idx]] += 1
    h_curr = q.astype(float) / m
    c_curr = _exact_c5(h_curr)
    best_h = h_curr.copy()
    best_c5 = c_curr
    h_start = h.copy()
    c_start = _exact_c5(h_start)
    if c_start < best_c5:
        best_c5 = c_start
        best_h = h_start.copy()
    T = T_init
    inc_mask = np.where(q < m)[0]
    dec_mask = np.where(q > 0)[0]
    no_improve = 0
    for move in range(max_moves):
        if time.time() > deadline:
            break
        if len(inc_mask) == 0 or len(dec_mask) == 0:
            break
        i = int(inc_mask[rng.integers(0, len(inc_mask))])
        j = int(dec_mask[rng.integers(0, len(dec_mask))])
        if i == j:
            continue
        q[i] += 1
        q[j] -= 1
        h_new = q.astype(float) / m
        c_new = _exact_c5(h_new)
        if c_new <= c_curr + 1e-15:
            c_curr = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta_c = c_new - c_curr
            if T > T_min and delta_c / T < 700.0:
                if np.log(rng.random() + 1e-300) < -delta_c / T:
                    c_curr = c_new
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h_new.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    q[i] -= 1
                    q[j] += 1
            else:
                q[i] -= 1
                q[j] += 1
                no_improve += 1
        T = max(T_min, T * T_decay)
        if no_improve > 200000 and T > T_min * 10:
            T = min(T_init, T * 3.0)
            no_improve = 0
        inc_mask = np.where(q < m)[0]
        dec_mask = np.where(q > 0)[0]
    if c_start < best_c5:
        return h_start, c_start
    return best_h, best_c5

def _smooth_anneal(h, deadline, max_iters=300000, T_init=0.003, T_min=1e-6, seed=42):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    h_curr = h.copy()
    c_curr = _exact_c5(h_curr)
    best_h = h_curr.copy()
    best_c5 = c_curr
    T = T_init
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            i = (i + 1) % n
        delta = rng.uniform(-0.05, 0.05)
        h_new = h_curr.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        c_new = _exact_c5(h_new)
        if c_new <= c_curr + 1e-15:
            h_curr = h_new
            c_curr = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_curr.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            dc = c_new - c_curr
            if T > T_min and dc / T < 700.0:
                if np.log(rng.random() + 1e-300) < -dc / T:
                    h_curr = h_new
                    c_curr = c_new
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h_curr.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
            else:
                no_improve += 1
        T = max(T_min, T * 0.9997)
        if no_improve > 50000 and T > T_min * 10:
            T = min(T_init, T * 2.0)
            no_improve = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gradient_descent(h, deadline, tau_init=20.0, lr_init=0.015, mom_beta=0.9, max_iters=200000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c_curr = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c_curr
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        G = _compute_G(h)
        grad = (G.T @ w) * dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c_new = _exact_c5(h_new)
        if c_new < c_curr - 1e-14:
            h = h_new
            c_curr = c_new
            stall = 0
            if c_curr < best_c5:
                best_c5 = c_curr
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
            tau = min(tau * 1.01, 150.0)
        else:
            stall += 1
            lr *= 0.93
            if stall > 60:
                break
            if stall % 25 == 0:
                tau = max(tau_init, tau * 0.8)
    return best_h, best_c5

def _diverse_start(n, rng, seed):
    """Generate diverse starting points."""
    starts = []
    # Symmetric-ish pattern
    h1 = np.zeros(n)
    half = n // 2
    for i in range(n):
        h1[i] = 0.7 if (i % 4 < 2) else 0.3
    h1 = _project(h1, n / 2.0)
    starts.append(h1)
    # Random
    h2 = rng.uniform(0.2, 0.8, n)
    h2 = _project(h2, n / 2.0)
    starts.append(h2)
    # Oscillating
    h3 = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    h3 = _project(h3, n / 2.0)
    starts.append(h3)
    # Binary-ish
    h4 = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h4[ones] = 1.0
    starts.append(h4.copy())
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize
    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
        except NameError:
            h0 = np.full(90, 0.5)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Pair descent at original resolution (35% budget)
    d1 = t0 + budget_s * 0.35
    h1, c1 = _pair_descent(h, d1, max_rounds=30000, n_top=500, seed=seed)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 1b: Gradient descent from pair descent result
    if time.time() < d1:
        h1b, c1b = _gradient_descent(h1, d1, tau_init=15.0, lr_init=0.012, mom_beta=0.9)
        if c1b < best_c5:
            best_c5 = c1b
            best_h = h1b.copy()

    # Phase 2: Smooth annealing (20% budget)
    d2 = t0 + budget_s * 0.55
    h2, c2 = _smooth_anneal(best_h, d2, max_iters=400000, T_init=0.004, T_min=2e-6, seed=seed + 1)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()

    # Phase 2b: Pair descent after annealing
    if time.time() < d2:
        h2b, c2b = _pair_descent(h2, d2, max_rounds=20000, n_top=400, seed=seed + 2)
        if c2b < best_c5:
            best_c5 = c2b
            best_h = h2b.copy()

    # Phase 3: Lattice annealing (20% budget)
    d3 = t0 + budget_s * 0.75
    h3, c3 = _lattice_anneal(best_h, d3, m=8, max_moves=3000000, T_init=5e-4, T_decay=0.9997, seed=seed + 3)
    if c3 < best_c5:
        best_c5 = c3
        best_h = c3 if isinstance(c3, np.ndarray) else h3.copy()
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3.copy()

    # Phase 3b: Pair descent from lattice result
    if time.time() < d3:
        h3b, c3b = _pair_descent(h3, d3, max_rounds=20000, n_top=400, seed=seed + 4)
        if c3b < best_c5:
            best_c5 = c3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution (15% budget)
    d4 = t0 + budget_s * 0.90
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12]
    time_per_res = budget_s * 0.012
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + time_per_res)
        h_r, c_r = _pair_descent(h_r, sub_dl, max_rounds=8000, n_top=min(300, n_try * n_try // 4), seed=seed + 10 + dn)
        if time.time() < d4 and (d4 - time.time()) > 2.0:
            sub_dl2 = min(d4, time.time() + time_per_res * 0.4)
            h_r, c_r = _gradient_descent(h_r, sub_dl2, tau_init=10.0, lr_init=0.01, mom_beta=0.88)
        if c_r < best_c5 - 1e-12:
            best_c5 = c_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Final intensive polishing (10% budget)
    d5 = t0 + budget_s * 0.97
    h5, c5 = _pair_descent(best_h, d5, max_rounds=50000, n_top=200, seed=seed + 100)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()

    # Final micro-polish
    if time.time() < t0 + budget_s * 0.99:
        h5b, c5b = _pair_descent(best_h, t0 + budget_s * 0.995, max_rounds=20000, n_top=100, seed=seed + 200)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h5b.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
