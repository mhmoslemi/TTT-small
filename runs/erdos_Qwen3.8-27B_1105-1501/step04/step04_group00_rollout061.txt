```python
def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] += (1.0 - h[d:]) * dx
            G[r, d:] += -h[:m] * dx
        else:
            g = -d
            m = n - g
            G[r, :m] += (1.0 - h[g:]) * dx
            G[r, g:] += -h[:m] * dx
    return G
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = (x > 0.0) & (x < 1.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        if not mask.any():
            mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:]) * dx
                G[r, d:] += -h[:m] * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:]) * dx
                G[r, g:] += -h[:m] * dx
    return G

def _line_search(h, p):

        return best_a, best_c5

    c1 = hi - gr * (hi - lo)

def _pair_polish(h, deadline, max_passes=6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()

    step = max(1, n // 20)

        for j in range(0, n, step):

                if time.time() > deadline:
                    break

                    continue
                for sign in [1, -1]:
                    if sign > 0:

                    else:
                        a_max = min(h[j], 1.0 - h[k])

                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2

                        f2 = _exact_c5(h2)
                        if f1 < f2:

                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0

                        best_h = h_try.copy()
                        h = h_try.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return h.astype(float)

def _structured_init(n, period, duty):
    h = np.zeros(n)
    base = np.zeros(period)
    k = max(1, int(round(period * duty)))
    base[:k] = 1.0
    n_tiles = n // period
    h[:n_tiles * period] = np.tile(base, n_tiles)
    remainder = n - n_tiles * period
    if remainder > 0:
        h[n_tiles * period:] = base[:remainder]
    return _project(h, n / 2.0)

def _fourier_evolve(h, deadline, n_pop=25, K=10, n_gen=200, mutation_scale=0.1, rng=None):
    n = len(h)
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = 2.0 * np.cos(2.0 * np.pi * k * x)
        B[:, 2*(k-1)+1] = 2.0 * np.sin(2.0 * np.pi * k * x)
    B_centered = B - B.mean(axis=0)
    target = n / 2.0
    a_init = np.linalg.lstsq(B_centered, h - 0.5, rcond=None)[0]
    pop_a = a_init + mutation_scale * rng.standard_normal((n_pop, 2*K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        h_try = np.clip(0.5 + B_centered @ pop_a[i], 0, 1)
        h_try = _project(h_try, target)
        pop_c5[i] = _exact_c5(h_try)
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = _project(np.clip(0.5 + B_centered @ pop_a[best_i], 0, 1), target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        n_keep = max(3, n_pop // 3)
        pop_a[:n_keep] = pop_a[order[:n_keep]]
        pop_c5[:n_keep] = pop_c5[order[:n_keep]]
        for i in range(n_keep, n_pop):
            p1 = order[int(rng.integers(0, n_keep))]
            p2 = order[int(rng.integers(0, n_keep))]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1.0 - mix) * pop_a[order[p2]]
            pop_a[i] += mutation_scale * rng.standard_normal(2*K)
            h_try = np.clip(0.5 + B_centered @ pop_a[i], 0, 1)
            h_try = _project(h_try, target)
            pop_c5[i] = _exact_c5(h_try)
        mutation_scale *= 0.996
        cur_best = int(np.argmin(pop_c5))
        if pop_c5[cur_best] < best_c5:
            best_c5 = pop_c5[cur_best]
            best_h = _project(np.clip(0.5 + B_centered @ pop_a[cur_best], 0, 1), target)
    return best_h, best_c5

def _weighted_grad(h, deadline, n_steps=200, gamma_init=15.0, gamma_max=400.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        gamma = gamma_init + (gamma_max - gamma_init) * frac ** 0.6
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(gamma * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.08
        found = False
        for _ in range(35):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                found = True
                break
            alpha *= 0.6
            if alpha < 1e-13:
                break
        if not found:
            gamma *= 0.6
            if gamma < 3:
                break
    return best_h, best_c5
```
