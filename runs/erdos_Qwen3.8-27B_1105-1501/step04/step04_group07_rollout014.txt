def _project(v, target):

        return np.zeros(n)

        return np.ones(n)

    hi = float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))

        mask = (x < 1.0) if res > 0 else (x > 0.0)

    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius

            break
        if stall > 80 and r < 0.01:
            break

        G = _grad_matrix(h)
        n_lags = len(scores)
        if top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk + 1e-12
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []
        for j in range(n):
            lo_b = max(-r, -h[j] + 1e-10)
            hi_b = min(r, 1.0 - h[j] - 1e-10)
            if lo_b >= hi_b:
                bounds.append((0.0, 0.0))
            else:
                bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                if r < 1e-6:
                    break
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            if r < 1e-6:
                break
            continue
        
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-6:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, max_alpha=1.0, n_golden=20)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.3)
        else:
            stall += 1
            r *= 0.85
            if stall > 40:
                r = max(r, 0.05)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_hill_climb(h_init, deadline, max_iters=20000):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    rng = np.random.default_rng(1)
    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[ones_count - target:]] = 0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng2 = np.random.default_rng(99)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 4000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng2.integers(len(ones_idx))])
        j = int(zeros_idx[rng2.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def _annealing_pairs(h, deadline, T_start=0.02, T_end=0.001, max_iters=5000):
    """Simulated annealing on pair transfers."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(55)
    total_iters = max_iters
    for it in range(total_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(total_iters, 1))
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        if direction > 0:
            a_max = min(h[j], 1.0 - h[i])
        else:
            a_max = min(h[i], 1.0 - h[j])
        if a_max < 1e-10:
            continue
        a = rng.random() * a_max
        h[i] += direction * a
        h[j] -= direction * a
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] -= direction * a
            h[j] += direction * a
    return best_h, best_c5

def _multi_direction_descent(h, deadline, max_iters=1000):
    """Try multiple direction types in rotation."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(777)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        scores = _scores(h)
        s_max = float(scores.max())
        n_lags = len(scores)
        
        # Choose direction type based on iteration
        method = it % 3
        
        if method == 0:
            # Subgradient
            thresh = s_max * 0.95
            active = np.where(scores >= thresh)[0]
            G = _grad_matrix(h)
            g = G[active].mean(axis=0)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                stall += 10
                continue
            d = -g / gn
        elif method == 1:
            # Single worst lag gradient
            worst = int(np.argmax(scores))
            G = _grad_matrix(h)
            g = G[worst].copy()
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                stall += 10
                continue
            d = -g / gn
        else:
            # Random feasible direction with bias toward reducing active lags
            d = rng.normal(0, 1, n)
            d -= d.mean()
            # Project toward negative gradient of worst lags
            thresh = s_max * 0.98
            active = np.where(scores >= thresh)[0]
            G = _grad_matrix(h)
            g = G[active].mean(axis=0)
            g -= g.mean()
            d = d - 2.0 * g
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                stall += 10
                continue
            d = d / dn
        
        found = False
        for scale in [0.1, 0.05, 0.02, 0.01, 0.005]:
            alpha, new_c5 = _line_search(h, d * scale, max_alpha=1.0, n_golden=20)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Subgradient descent (25%)
    dl1 = t0 + budget_s * 0.25
    hdr, c5r = _subgrad_descent(h, dl1, max_iters=3000, active_frac=0.05)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 2: Pair descent with gradient-guided selection (20%)
    dl2 = t0 + budget_s * 0.45
    hdr, c5r = _pair_descent(best_h, dl2, max_passes=20, top_pairs=40)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 3: Active LP (15%)
    dl3 = t0 + budget_s * 0.60
    hdr, c5r = _active_lp(best_h, dl3, top_k=12, radius=0.06, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 4: Multi-direction descent (10%)
    dl4 = t0 + budget_s * 0.70
    hdr, c5r = _multi_direction_descent(best_h, dl4, max_iters=2000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 5: Full pair sweep (10%)
    dl5 = t0 + budget_s * 0.80
    hdr, c5r = _all_pair_sweep(best_h, dl5, max_passes=2)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 6: Multi-resolution exploration (10%)
    dl6 = t0 + budget_s * 0.90
    for dn in [-4, -2, -1, 1, 2, 4, 8, 12, 16, -8]:
        if time.time() > dl6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 160:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl6, time.time() + budget_s * 0.015)
        h_r, c5_r = _subgrad_descent(h_r, sub_dl, max_iters=500, active_frac=0.04)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 7: Annealing polish (5%)
    dl7 = t0 + budget_s * 0.95
    if best_c5 > 0.3807:
        hdr, c5r = _annealing_pairs(best_h, dl7, T_start=0.01, T_end=0.0005, max_iters=3000)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hdr.copy()
    
    # Phase 8: Final ultra-fine LP (5%)
    dl8 = t0 + budget_s * 0.995
    hdr, c5r = _active_lp(best_h, dl8, top_k=6, radius=0.015, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
