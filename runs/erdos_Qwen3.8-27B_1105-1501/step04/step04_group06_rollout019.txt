    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _smooth_step(h, n, w):
    kernel = np.ones(w) / w
    h_p = np.concatenate([h[-w:], h, h[:w]])
    h_s = np.convolve(h_p, kernel, mode='valid')[:n]
    return _project(h_s, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 * 0.5
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def _elapsed():
        return time.time() - t0

    # Phase 1: Initial LP polish (12% budget)
    d1 = t0 + budget_s * 0.12
    h, c5 = _lp_loop(h, n0, d1, K=24, margin=0.015, radius_init=0.12,
                     max_iters=3000, stall_limit=70)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Resolution search (20% budget)
    d2 = t0 + budget_s * 0.32
    cands = []
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, 10, -10, -14, 14, 16, -16, 20, -20]:
        nt = n0 + dn
        if nt >= 24 and nt <= 128 and nt % 2 == 0:
            cands.append(nt)
    cands = sorted(set(cands), key=lambda x: (abs(x - n0), x))

    for n_try in cands:
        if time.time() > d2:
            break
        if n_try == best_n and _exact_c5(best_h) == best_c5:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d2, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, n_try, sub_d, K=18, margin=0.02,
                             radius_init=0.1, max_iters=1500, stall_limit=50)
        if c5_r < best_c5 - 5e-5:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 3: Aggressive multi-K LP (25% budget)
    d3 = t0 + budget_s * 0.57
    param_set = [
        (20, 0.02, 0.10), (28, 0.012, 0.08), (32, 0.01, 0.06),
        (16, 0.03, 0.15), (24, 0.015, 0.10), (40, 0.008, 0.05)
    ]
    for K_p, m_p, r_p in param_set:
        if time.time() > d3:
            break
        h_, c5_ = _lp_loop(best_h, best_n, min(d3, time.time() + budget_s * 0.035),
                           K=K_p, margin=m_p, radius_init=r_p,
                           max_iters=4000, stall_limit=80)
        if c5_ < best_c5:
            best_c5, best_h = c5_, h_.copy()

    # Phase 4: Smoothed restart + LP (15% budget)
    d4 = t0 + budget_s * 0.72
    for w in [3, 5, 7]:
        if time.time() > d4:
            break
        h_s = _smooth_step(best_h, best_n, w)
        h_s, c5_s = _lp_loop(h_s, best_n, min(d4, time.time() + budget_s * 0.02),
                             K=24, margin=0.015, radius_init=0.06,
                             max_iters=2000, stall_limit=50)
        if c5_s < best_c5 - 1e-10:
            best_c5, best_h = c5_s, h_s.copy()

    # Phase 5: Random perturbation restarts (20% budget)
    d5 = t0 + budget_s * 0.92
    scale = 0.04
    no_improve = 0
    for attempt in range(300):
        if time.time() > d5:
            break
        h_p = _project(best_h + rng.normal(0, scale, best_n), best_n * 0.5)
        sub_d = min(d5, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_loop(h_p, best_n, sub_d, K=20, margin=0.02,
                             radius_init=0.06, max_iters=600, stall_limit=35)
        if c5_s < best_c5 - 1e-10:
            best_c5, best_h = c5_s, h_s.copy()
            scale = max(0.003, scale * 0.75)
            no_improve = 0
        else:
            no_improve += 1
            scale = min(0.1, scale * 1.15)
            if no_improve > 40:
                scale = 0.005
                no_improve = 0

    # Phase 6: Final deep polish (8% budget)
    d6 = t0 + budget_s * 0.99
    h_f, c5_f = _lp_loop(best_h, best_n, d6, K=32, margin=0.008,
                         radius_init=0.02, max_iters=8000, stall_limit=150)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    h_f2, c5_f2 = _lp_loop(best_h, best_n, min(d6, time.time() + budget_s * 0.02),
                           K=40, margin=0.005, radius_init=0.01,
                           max_iters=5000, stall_limit=200)
    if c5_f2 < best_c5:
        best_c5, best_h = c5_f2, h_f2.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
