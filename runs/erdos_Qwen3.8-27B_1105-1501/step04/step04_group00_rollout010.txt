from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _subgradient_momentum(h, deadline, max_iters=5000, gamma_init=5.0, gamma_max=200.0,
                           momentum=0.7, alpha_init=0.05):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5

    alpha = alpha_init
    stall = 0

        grad -= np.mean(grad)  # maintain sum constraint

            stall = 0

def _lp_cuts(h, deadline, max_iters=2000, radius_init=0.2):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

            break

            break

        G = _grad_matrix(h)

        top_k = min(n_lags, max(10, n_lags // 4))
        idx = np.argsort(scores)[-top_k:]
        idx = idx[np.argsort(scores[idx][::-1])]  # sort descending
        scores_k = scores[idx]
        G_k = G[idx]
        
        # Solve LP
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((top_k, n + 1))
        A_ub[:, :n] = G_k
        A_ub[:, n] = -1.0
        b_ub = -scores_k
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

def _exact_c5(h):

def _scores(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return _project(h_new, n_new / 2.0)

def _binary_pattern(n, pattern_type, param):
    h = np.zeros(n)
    if pattern_type == 0:
        # Quasi-periodic: floor(i*alpha) mod 1 < 0.5
        x = np.arange(n)
        h = ((x * param) % 1.0) < 0.5
    elif pattern_type == 1:
        # Modular: i mod q < q/2
        q = int(round(param))
        if q < 2:
            q = 2
        x = np.arange(n)
        h = (x % q) < (q // 2)
    elif pattern_type == 2:
        # Goldboden-like
        x = np.arange(n)
        h = (np.sin(2 * np.pi * param * x) > 0)
    elif pattern_type == 3:
        # Product of two quasi-periodic
        x = np.arange(n)
        h1 = ((x * param) % 1.0) < 0.5
        h2 = ((x * param * 2.13) % 1.0) < 0.5
        h = (h1 & h2) | (~h1 & ~h2)
    h = h.astype(float)
    # Adjust to have exactly n/2 ones
    target = n // 2
    cnt = int(np.sum(h))
    if cnt > target:
        order = np.argsort(h)
        for idx in order:
            if cnt <= target:
                break
            h[idx] = 0.0
            cnt -= 1
    elif cnt < target:
        order = np.argsort(-h)
        for idx in order:
            if cnt >= target:
                break
            h[idx] = 1.0
            cnt += 1
    return h

def _relaxed_binary(h, deadline, eps_list=None):
