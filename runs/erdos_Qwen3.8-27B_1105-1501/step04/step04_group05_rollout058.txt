def _compute_D(h, i, j):
    """Compute D(m) = A(m,i) - A(m,j) for all lags m.
    
    A(m, idx) represents the coefficient such that changing h[idx] by delta
    changes corr(m) by delta * A(m, idx).
    
    For lag m (where corr index = n-1+m):
    - Forward: h[idx] appears as h[idx]*(1-h[idx+m]) if idx+m is valid
      → coefficient of h[idx] is (1-h[idx+m])
    - Backward: h[idx] appears as h[idx-m]*(1-h[idx]) if idx-m is valid
      → coefficient of (1-h[idx]) is h[idx-m], so coefficient of h[idx] is -h[idx-m]
    
    So A(m, idx) = (1-h[idx+m])_valid - h[idx-m]_valid
    where (1-h[idx+m])_valid = (1-h[idx+m]) if 0 ≤ idx+m < n, else 0
    and h[idx-m]_valid = h[idx-m] if 0 ≤ idx-m < n, else 0
    """
    n = len(h)

    for m in range(-(n-1), n):

        a_mi = 0.0
        if 0 <= i + m < n:
            a_mi += 1.0 - h[i + m]
        if 0 <= i - m < n:
            a_mi -= h[i - m]
        # A(m, j)
        a_mj = 0.0
        if 0 <= j + m < n:
            a_mj += 1.0 - h[j + m]
        if 0 <= j - m < n:
            a_mj -= h[j - m]
        D[m + n - 1] = a_mi - a_mj
    return D

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score(h):
    n = len(h)
    return float(np.max(_corr_vec(h)) * 2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _compute_D(h, i, j):
    n = len(h)
    D = np.zeros(2 * n - 1)
    omh = 1.0 - h
    for p, sign in ((i, 1.0), (j, -1.0)):
        m_lo = max(-(n-1), -p)
        m_hi = min(n-1, n-1-p)
        if m_lo <= m_hi:
            D[m_lo+n-1:m_hi+n] += sign * omh[p+m_lo:p+m_hi+1]
        m_lo = max(-(n-1), p-(n-1))
        m_hi = min(n-1, p)
        if m_lo <= m_hi:
            mv = np.arange(m_lo, m_hi+1)
            D[mv+n-1] -= sign * h[p-mv]
    return D

def _optimal_t(corr, D, t0, t_lo, t_hi, n_iter=50):
    """Find t that minimizes max(corr + (t-t0)*D) subject to t in [t_lo, t_hi]."""
    delta_lo = t_lo - t0
    delta_hi = t_hi - t0
    
    # Lower bound on optimal max
    indiv = np.where(D > 0, corr + delta_lo * D,
             np.where(D < 0, corr + delta_hi * D, corr))
    C_lo = float(np.max(indiv))
    C_hi = float(np.max(corr))
    
    if C_lo >= C_hi - 1e-14:
        return t0, C_hi
    
    t_opt = t0
    pos = D > 1e-15
    neg = D < -1e-15
    any_pos = np.any(pos)
    any_neg = np.any(neg)
    
    for _ in range(n_iter):
        C_mid = (C_lo + C_hi) / 2.0
        tu = t_hi
        tl = t_lo
        if any_pos:
            cand = t0 + (C_mid - corr[pos]) / D[pos]
            m = float(np.min(cand))
            if m < tu:
                tu = m
        if any_neg:
            cand = t0 + (C_mid - corr[neg]) / D[neg]
            m = float(np.max(cand))
            if m > tl:
                tl = m
        if tl <= tu:
            C_hi = C_mid
            t_opt = (tl + tu) / 2.0
        else:
            C_lo = C_mid
    return t_opt, C_hi

def _pair_descent(h, deadline, max_passes=200, tol=1e-12):
    """Optimal pair coordinate descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    corr = _corr_vec(h)
    best_h = h.copy()
    best_c5 = float(np.max(corr) * 2.0 / n)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                s = h[i] + h[j]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-14:

                D = _compute_D(h, i, j)

    if final_c5 < best_c5:

        best_c5 = final_c5
    return best_h, best_c5

def _resample(h, n_new):
