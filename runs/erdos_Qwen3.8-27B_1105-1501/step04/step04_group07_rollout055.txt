Precomputation cost: O(n²) = O(4096). Trivial.

With this, the pair line search becomes:
1. a_arr = S[:, q] - S[:, p]  (O(n) vector operation)
2. b_arr is known (mostly zeros, with specific entries)

```python
def _precompute_S(h, n):
    d = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # i ranges 0..n-1, d ranges -(n-1)..n-1
    # i+d ranges from -(n-1) to 2n-2
    i = np.arange(n)  # shape (n,)
    idx = i[:, None] + d[None, :]  # shape (n, 2n-1)
    valid = (idx >= 0) & (idx < n)  # shape (n, 2n-1)
    h_ext = np.zeros(2*n-1+2*(n-1))  # padded
    h_ext[n-1:n-1+n] = h  # center the h
    # Actually, let me just use np.where
    h_lookup = np.where(valid, h[np.clip(idx, 0, n-1)], 0.0)
    S_fwd = h_lookup
    # Similarly for i-d
    idx2 = i[:, None] - d[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    h_lookup2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    S = S_fwd + h_lookup2
    return S
```

This is O(n²) but fully vectorized with numpy. For n=64, the array is 64×127 = 8128 elements, so it's instant.

Now, for the pair line search:
```python
def _pair_step(h, p, q, S, b_template, s_raw, dx, n):
    d0 = q - p
    a_arr = S[q] - S[p]  # shape (2n-1,)
    b_arr = b_template.copy()  # precomputed for this d0
    # s_raw is the raw correlation (not multiplied by dx)
    
    alpha_lo = max(-h[p], h[q] - 1.0)
    alpha_hi = min(1.0 - h[p], h[q])
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, float(s_raw.max())
    
    # g(alpha) = max(s_raw + alpha*a_arr + alpha^2*b_arr)
    # Use grid + golden section
    
    n_grid = 20
    grid = np.linspace(alpha_lo, alpha_hi, n_grid)
    gs = np.array([float(np.max(s_raw + a*a_arr + a*a*b_arr)) for a in grid])
    best_i = int(np.argmin(gs))
    best_val = float(gs[best_i])
    best_alpha = float(grid[best_i])
    
    # Refine
    if 0 < best_i < n_grid - 1:
        lo = float(grid[best_i-1])
        hi = float(grid[best_i+1])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr*(hi-lo)
        c2 = lo + gr*(hi-lo)
        f1 = float(np.max(s_raw + c1*a_arr + c1*c1*b_arr))
        f2 = float(np.max(s_raw + c2*a_arr + c2*c2*b_arr))
        for _ in range(40):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi, f2, c2 = c2, f1, c1
                c1 = hi - gr*(hi-lo)
                f1 = float(np.max(s_raw + c1*a_arr + c1*c1*b_arr))
            else:
                lo, f1, c1 = c1, f2, c2
                c2 = lo + gr*(hi-lo)
                f2 = float(np.max(s_raw + c2*a_arr + c2*c2*b_arr))
        opt = (lo+hi)/2.0
        fopt = float(np.max(s_raw + opt*a_arr + opt*opt*b_arr))
        if fopt < best_val:
            best_val = fopt
            best_alpha = opt
    
    return best_alpha, best_val
```

This should be fast. The bottleneck is the 20 grid evaluations + ~40 golden section evaluations, each O(n) = O(64-127) with numpy. Total: ~60 * 127 ≈ 7620 floating point ops per pair. With numpy overhead, maybe 5-10 µs per pair.

For 2016 pairs per sweep: 2016 * 10µs ≈ 20ms. Very fast!

But wait, after each step, I need to:
1. Update h (O(1))
2. Update S (O(n) to recompute the two rows S[p] and S[q])
3. Update s_raw (O(n²) for full correlation, or O(n) for incremental update)

Actually, updating s_raw incrementally: when h[p] changes by δ and h[q] changes by -δ, how does the correlation change?

c(d) = Σ_{valid i} h[i](1-h[i+d])

The change in c(d) when h[p] += δ:
- Terms where i=p: δ*(1-h[p+d]) (if p+d valid)
- Terms where i+d=p, i.e., i=p-d: h[p-d]*(-δ) = -h[p-d]*δ (if p-d valid)

Similarly for h[q] -= δ:
- Terms where i=q: -δ*(1-h[q+d]) (if q+d valid)
- Terms where i+d=q: -h[q-d]*δ (if q-d valid)

So Δc(d) = δ[(1-h[p+d]) - h[p-d] - (1-h[q+d]) + h[q-d]]
          = δ * [h[q+d] - h[p+d] - h[p-d] + h[q-d]]
          = δ * a(d; p, q)

Which is exactly the linear coefficient! So the update to s_raw is:
s_raw_new = s_raw + δ * a_arr

This is O(n) per step. 

And the update to S:
S[i, :] changes only for i = p and i = q (since only h[p] and h[q] changed).
S[p, :] new: h[p+d] changes by δ for each d, so S[p, d] += δ (for the forward part)
Actually S[i, d] = h[i+d] + h[i-d].
S[p, d] = h[p+d] + h[p-d]. After h[p] += δ:
- h[p+d] changes only when d=0: h[p+0] = h[p] += δ
- h[p-d] changes only when d=0: h[p-0] = h[p] += δ
So S[p, 0] += 2δ, and all other S[p, :] unchanged? Wait no:
S[p, d] = h[p+d] + h[p-d]. The value h[p+d] doesn't change when h[p] changes (unless d=0). Similarly h[p-d] doesn't change (unless d=0). So only S[p, n-1] (which is d=0) changes by 2δ.

Similarly, S[q, n-1] -= 2δ.

And for other rows i ≠ p, q:
S[i, d] = h[i+d] + h[i-d]. This changes if i+d = p (i.e., d = p-i) or i+d = q (i.e., d = q-i) or i-d = p (i.e., d = i-p) or i-d = q (i.e., d = i-q).

So for each row i, at most 4 entries change. Total changes: O(n) per step.

OK this is getting complex. For simplicity, let me just recompute S from scratch after each step. It's O(n²) = O(4096) which is fast enough.

Actually, let me precompute S once and only update the affected rows. For n=64, recomputing S takes 64*127 = 8128 operations, which is trivial.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return _raw_corr(h) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _make_S(h):
    n = len(h)
    d = np.arange(2*n-1) - (n-1)
    i = np.arange(n)
    idx_f = i[:, None] + d[None, :]
    idx_b = i[:, None] - d[None, :]
    vf = (idx_f >= 0) & (idx_f < n)
    vb = (idx_b >= 0) & (idx_b < n)
    hf = np.zeros_like(idx_f, dtype=float)
    hf[vf] = h[idx_f[vf]]
    hb = np.zeros_like(idx_b, dtype=float)
    hb[vb] = h[idx_b[vb]]
    return hf + hb

def _b_vec(n, d0):
    """Quadratic coefficient vector for pair with separation d0."""
    nv = 2*n - 1
    b = np.zeros(nv)
    b[n-1] = -2.0  # d=0
    r_pos = n - 1 + d0
    if 0 <= r_pos < nv:
        b[r_pos] = 1.0
    r_neg = n - 1 - d0
    if 0 <= r_neg < nv:
        b[r_neg] = 1.0
    return b

def _pair_best_alpha(s_raw, a_arr, b_arr, alpha_lo, alpha_hi):
    """Find alpha in [alpha_lo, alpha_hi] minimizing max(s_raw + alpha*a + alpha^2*b)."""
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, float(s_raw.max())
    
    n_grid = 25
    grid = np.linspace(alpha_lo, alpha_hi, n_grid)
    best_val = float('inf')
    best_a = 0.0
    for a in grid:
        val = float(np.max(s_raw + a * a_arr + a*a * b_arr))
        if val < best_val:
            best_val = val
            best_a = a
    
    # Refine around best_a
    lo = float(np.clip(best_a - (alpha_hi-alpha_lo)/n_grid, alpha_lo, alpha_hi))
    hi = float(np.clip(best_a + (alpha_hi-alpha_lo)/n_grid, alpha_lo, alpha_hi))
    if hi - lo < 1e-14:
        return best_a, best_val
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(s_raw + c1*a_arr + c1*c1*b_arr))
    f2 = float(np.max(s_raw + c2*a_arr + c2*c2*b_arr))
    for _ in range(50):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi, f2 = c2, f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(s_raw + c1*a_arr + c1*c1*b_arr))
        else:
            lo, f1 = c1, f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(s_raw + c2*a_arr + c2*c2*b_arr))
    
    opt = (lo + hi) / 2.0
    fopt = float(np.max(s_raw + opt*a_arr + opt*opt*b_arr))
    if fopt < best_val:
        return opt, fopt
    return best_a, best_val

def _pair_sweep(h, deadline, n_rounds=50, seed=42):
    """Cyclic pair-swap optimization."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    s_raw = _raw_corr(h)
    S = _make_S(h)
    cur_c5 = float(s_raw.max()) * dx
    
    # Precompute b vectors for all possible d0
    b_cache = {d: _b_vec(n, d) for d in range(1, n)}
    
    # Interior pairs
    def get_interior_pairs():
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        pairs = []
        for ii in range(len(interior)):
            for jj in range(ii+1, len(interior)):
                pairs.append((interior[ii], interior[jj]))
        return pairs
    
    rng = np.random.default_rng(seed)
    global_best_c5 = cur_c5
    global_best_h = h.copy()
    
    for round_num in range(n_rounds):
        if time.time() > deadline:
            break
        pairs = get_interior_pairs()
        if len(pairs) == 0:
            break
        rng.shuffle(pairs)
        
        improved = 0
        for (p, q) in pairs:
            if time.time() > deadline:
                break
            if h[p] <= 1e-8 or h[p] >= 1.0 - 1e-8:
                continue
            if h[q] <= 1e-8 or h[q] >= 1.0 - 1e-8:
                continue
            
            d0 = abs(q - p)
            a_arr = S[q] - S[p] if q > p else S[p] - S[q]
            # Wait, need to be careful with sign
            # If direction is h[p]+=alpha, h[q]-=alpha:
            # a_arr = S[q] - S[p] (since a(d;p,q) = S(q,d) - S(p,d))
            # But if p > q, the direction is h[p]+=alpha, h[q]-=alpha
            # and d0 = p - q, a(d;p,q) = S(q,d) - S(p,d) still
            # Let me just always use p < q ordering
            if p > q:
                p, q = q, p
                # Now direction is h[p]+=alpha, h[q]-=alpha
                a_arr = S[q] - S[p]
            else:
                a_arr = S[q] - S[p]
            
            b_arr = b_cache[d0]
            alpha_lo = max(-h[p], h[q] - 1.0)
            alpha_hi = min(1.0 - h[p], h[q])
            
            alpha, best_val = _pair_best_alpha(s_raw, a_arr, b_arr, alpha_lo, alpha_hi)
            
            if abs(alpha) > 1e-14 and best_val < s_raw.max() - 1e-12:
                h[p] += alpha
                h[q] -= alpha
                h = np.clip(h, 0.0, 1.0)
                s_raw = _raw_corr(h)
                S = _make_S(h)
                cur_c5_raw = float(s_raw.max())
                improved += 1
                if cur_c5_raw * dx < global_best_c5:
                    global_best_c5 = cur_c5_raw * dx
                    global_best_h = h.copy()
        
        if improved == 0:
            break
    
    return global_best_h, global_best_c5

Hmm wait, I'm recomputing S (O(n²)) after every single pair update. That might be too slow if many pairs improve in one sweep. Let me estimate: if 50 pairs improve in a sweep, that's 50 * O(n²) = 50 * 4096 = 200K ops. Still fast.

Actually, recomputing S is 64*127 multiplications + additions ≈ 16K flops. In numpy, this is essentially instant (sub-microsecond). So it's fine.

But recomputing s_raw = _raw_corr(h) is O(n²) as well (correlation). For n=64, that's also instant.

So the bottleneck is really the line search: 25 + 50 = 75 evaluations of max(s_raw + a*a_arr + a²*b_arr), each O(n) = O(127). Total: 75 * 127 ≈ 9500 flops per pair. With numpy overhead, maybe 10-20 µs per pair.

For 2016 pairs: 2016 * 15µs ≈ 30ms per sweep. Very fast! I can do hundreds of sweeps in the budget.

But wait, I'm only using interior pairs. If the solution has many values at 0 or 1, the number of interior pairs could be much smaller. Let me say ~20 interior values → 190 pairs per sweep → 3ms per sweep. Even faster.

This means the pair sweep will converge very quickly (maybe 10-20 sweeps). After that, I need the basin hopping and other techniques to make further progress.

Let me think about what happens after the pair sweep converges. At the local optimum, for every interior pair (p,q), the best alpha is 0 (or at the boundary). This means the solution is a local optimum of the pair-swap landscape.

To escape, I need:
1. Larger perturbations (basin hopping)
2. Higher-order moves (triple or quad swaps)
3. Gradient-based methods that can move in directions not achievable by single pair swaps

Actually, any zero-sum direction can be decomposed into pair swaps (since the space of zero-sum vectors has dimension n-1, and pair swaps span this space). So in principle, the pair sweep should be able to reach any point in the feasible set, given enough iterations. The issue is that it might converge to a local optimum faster than it can explore the full space.

The key insight is that at a local optimum of the pair-swap landscape, the solution satisfies a KKT-like condition: for each pair (p,q), the directional derivative of C5 in the (p,q) direction is non-positive. This is a necessary but not sufficient condition for a local minimum of the full problem.

To make progress beyond this, I need to either:
1. Move in a direction that's a combination of multiple pair directions (higher-order)
2. Accept temporary worsening (simulated annealing)
3. Jump to a distant point (basin hopping)

Let me implement all three.

For (1), I can try "block moves": select a subset of k pairs and find the best combined step. This is a k-dimensional optimization, which is harder but doable for k=2 or k=3.

For (2), I'll use a simulated annealing schedule: accept steps that worsen C5 by a small amount, with probability decaying over time.

For (3), I'll add random noise and re-optimize.

Let me now write the complete code, incorporating all these ideas:

                    improved += 1
                    if cur_max < global_best_max:
                        global_best_max = cur_max
                        global_best_h = h.copy()
                    if accept_worsening:
                        temp *= 0.995
        if improved == 0:
            break
    
    return global_best_h, global_best_max * dx

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _raw_corr(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 60:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.1, 0.05, 0.02, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = _project(h + alpha * d * scale, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 100:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)
