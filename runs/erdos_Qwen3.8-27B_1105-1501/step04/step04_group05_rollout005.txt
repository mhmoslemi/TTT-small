import numpy as np

def _c5_eval(h):
    """Compute C5 = max(corr) * dx for a feasible vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} using bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    p = np.clip(v - mu, 0.0, 1.0)
    # Fix residual
    residual = target - float(np.sum(p))
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            slack = 1.0 - p
            idx = np.argsort(-slack)
        else:
            slack = -p
            idx = np.argsort(-slack)
        for i in idx:
            if abs(residual) < 1e-14:
                break
            if residual > 0:
                adj = min(residual, 1.0 - p[i])
            else:
                adj = max(residual, -p[i])
            p[i] += adj
            residual -= adj
    return p

def _mask_to_blur(mask, d):
    """Convert binary mask with value d blur to vector."""
    return np.where(mask == 1.0, 1.0 - d, d)

def _blur_c5(mask, d):
    """C5 for the d-blur of a binary mask."""
    h = _mask_to_blur(mask, d)
    return _c5_eval(h)

def _opt_d_for_mask(mask, deadline=None):

    n = len(mask)

    best_c = _blur_c5(mask, 0.5)
    # Grid search
    ds = np.linspace(0.0, 0.5, 200)
    for d in ds:
        if deadline and time.time() > deadline:
            break
        c = _blur_c5(mask, d)

            best_c = c
            best_d = d
    # Refine with golden section
    if not (deadline and time.time() > deadline):
        a, b = max(0, best_d - 0.5/200), min(0.5, best_d + 0.5/200)
        gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = _blur_c5(mask, c1)
        f2 = _blur_c5(mask, c2)
        for _ in range(60):
            if deadline and time.time() > deadline:

            if f1 < f2:
                b = c2
                c2 = c1
                f2 = f1
                c1 = b - gr * (b - a)
            else:
                a = c1
                c1 = c2
                f1 = f2
                c2 = a + gr * (b - a)
                f2 = _blur_c5(mask, c2)
            best_d = (a + b) / 2
            best_c = min(f1, f2)
    return best_d, _mask_to_blur(mask, best_d), best_c

def _pair_line_search(h, i, j, n_iters=40):
    """
    For pair (i,j), find t that minimizes C(t) where:
    h_new[i] = t, h_new[j] = h[i] + h[j] - t
    Uses ternary search on the convex function C(t).
    Returns (h_new, c5_new) if improved, else (h, c5_old).
    """
    p = h[i] + h[j]
    t_lo = max(0.0, p - 1.0)
    t_hi = min(1.0, p)
    if t_hi - t_lo < 1e-12:
        return h, _c5_eval(h)
    
    c5_old = _c5_eval(h)
    best_t = (t_lo + t_hi) / 2.0
    best_c = c5_old  # don't accept unless strictly better
    
    # Ternary search
    a, b = t_lo, t_hi
    for _ in range(n_iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h1 = h.copy()
        h1[i] = m1
        h1[j] = p - m1
        c1 = _c5_eval(h1)
        
        h2 = h.copy()
        h2[i] = m2
        h2[j] = p - m2
        c2 = _c5_eval(h2)
        
        if c1 < c2:
            b = m2
            if c1 < best_c:
                best_c = c1
                best_t = m1
        else:
            a = m1
            if c2 < best_c:
                best_c = c2
                best_t = m2
    
    if best_c < c5_old - 1e-14:
        h_new = h.copy()
        h_new[i] = best_t
        h_new[j] = p - best_t
        return h_new, best_c
    return h, c5_old

def _full_pair_pass(h, deadline, max_passes=20):
    """Multiple passes of pair line search."""
    n = len(h)
    best_h = h.copy()
    best_c = _c5_eval(h)
    
    for pnum in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Randomize pair order
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
                improved = True
        if not improved:
            break
    return best_h, best_c

def _balanced_mask_random(n, rng):

    m = n // 2
    idx = rng.choice(n, size=m, replace=False)
    mask = np.zeros(n)
    mask[idx] = 1.0
    return mask

def _parent_to_mask(h, n):
    """Convert parent to threshold mask."""
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    mask = np.zeros(n)
    mask[idx] = 1.0
    return mask

def _resample(h, n_new):

    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
import time


def _c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    p = np.clip(v - mu, 0.0, 1.0)
    residual = target - float(np.sum(p))
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(-(1.0 - p))
        else:
            idx = np.argsort(p)
        for i in idx:
            if abs(residual) < 1e-13:
                break
            if residual > 0:
                adj = min(residual, 1.0 - p[i])
            else:
                adj = max(residual, -p[i])
            p[i] += adj
            residual -= adj
    return p


def _mask_blur_c5(mask, d):
    h = np.where(mask == 1.0, 1.0 - d, d)
    return _c5_eval(h)


def _opt_d_for_mask(mask, deadline=None):
    n = len(mask)
    best_d = 0.5
    best_c = _mask_blur_c5(mask, 0.5)
    ds = np.linspace(0.0, 0.5, 200)
    for d in ds:
        if deadline and time.time() > deadline:
            break
        c = _mask_blur_c5(mask, d)
        if c < best_c:
            best_c = c
            best_d = float(d)
    if not (deadline and time.time() > deadline):
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = max(0.0, best_d - 0.5 / 200.0), min(0.5, best_d + 0.5 / 200.0)
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = _mask_blur_c5(mask, c1)
        f2 = _mask_blur_c5(mask, c2)
        for _ in range(50):
            if deadline and time.time() > deadline:
                break
            if f1 < f2:
                b = c2
                c2 = c1
                f2 = f1
                c1 = b - gr * (b - a)
                f1 = _mask_blur_c5(mask, c1)
            else:
                a = c1
                c1 = c2
                f1 = f2
                c2 = a + gr * (b - a)
                f2 = _mask_blur_c5(mask, c2)
        best_d = (a + b) / 2.0
        best_c = min(f1, f2)
    h = np.where(mask == 1.0, 1.0 - best_d, best_d)
    return best_d, h, _c5_eval(h)


def _pair_line_search(h, i, j, n_iters=35):
    p = h[i] + h[j]
    t_lo = max(0.0, p - 1.0)
    t_hi = min(1.0, p)
    if t_hi - t_lo < 1e-12:
        return h, _c5_eval(h)
    c5_old = _c5_eval(h)
    best_t = (t_lo + t_hi) / 2.0
    best_c = c5_old
    a, b = t_lo, t_hi
    for _ in range(n_iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h1 = h.copy()
        h1[i] = m1
        h1[j] = p - m1
        c1 = _c5_eval(h1)
        h2 = h.copy()
        h2[i] = m2
        h2[j] = p - m2
        c2 = _c5_eval(h2)
        if c1 < c2:
            b = m2
            if c1 < best_c:
                best_c = c1
                best_t = m1
        else:
            a = m1
            if c2 < best_c:
                best_c = c2
                best_t = m2
    if best_c < c5_old - 1e-14:
        h_new = h.copy()
        h_new[i] = best_t
        h_new[j] = p - best_t
        return h_new, best_c
    return h, c5_old


def _full_pair_pass(h, deadline, max_passes=50, rng=None):
    n = len(h)
    best_h = h.copy()
    best_c = _c5_eval(h)
    for pnum in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        if rng is not None:
            for k in range(len(pairs) - 1, 0, -1):
                j2 = rng.integers(k + 1)
                pairs[k], pairs[j2] = pairs[j2], pairs[k]
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            h_new, c_new = _pair_line_search(best_h, i, j, n_iters=30)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
                improved = True
        if not improved:
            break
    return best_h, best_c


def _balanced_mask_random(n, rng):
    m = n // 2
    idx = rng.choice(n, size=m, replace=False)
    mask = np.zeros(n)
    mask[idx] = 1.0
    return mask


def _parent_to_mask(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    mask = np.zeros(n)
    mask[idx] = 1.0
    return mask


def _structured_mask(n, step, offset):
    m = n // 2
    mask = np.zeros(n)
    pos = set()
    for k in range(m):
        p = (step * k + offset) % n
        if p not in pos:
            pos.add(p)
            mask[p] = 1.0
    if mask.sum() < m:
        remaining = [i for i in range(n) if mask[i] == 0]
        for i in remaining[:m - int(mask.sum())]:
            mask[i] = 1.0
    if mask.sum() == m:
        return mask
    return None


def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _mask_metropolis(mask, n, n_steps, rng, deadline):
    m = n // 2
    h = np.where(mask == 1.0, 1.0, 0.0)
    c_cur = _c5_eval(h)
    best_h = h.copy()
    best_c = c_cur
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        k = rng.integers(1, 5)
        k = min(k, len(ones), len(zeros))
        i_ones = rng.choice(len(ones), size=k, replace=False)
        i_zeros = rng.choice(len(zeros), size=k, replace=False)
        flip_ones = ones[i_ones]
        flip_zeros = zeros[i_zeros]
        h_new = h.copy()
        h_new[flip_ones] = 0.0
        h_new[flip_zeros] = 1.0
        c_new = _c5_eval(h_new)
        if c_new < c_cur:
            h = h_new
            c_cur = c_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c


def _basin_hopping(mask, n, rng, deadline, start_temp=0.03):
    h = np.where(mask == 1.0, 1.0, 0.0)
    c_cur = _c5_eval(h)
    best_h = h.copy()
    best_c = c_cur
    temp = start_temp
    for _ in range(200):
        if time.time() > deadline:
            break
        k = rng.integers(1, 6)
        k = min(k, n // 2)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i_ones = rng.choice(len(ones), size=k, replace=False)
        i_zeros = rng.choice(len(zeros), size=k, replace=False)
        h_new = h.copy()
        h_new[ones[i_ones]] = 0.0
        h_new[zeros[i_zeros]] = 1.0
        c_new = _c5_eval(h_new)
        delta = c_new - c_cur
        if delta < 0 or (temp > 0.001 and rng.random() < np.exp(-delta / temp)):
            h = h_new
            c_cur = c_new
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()
        temp *= 0.92
    return best_h, best_c
