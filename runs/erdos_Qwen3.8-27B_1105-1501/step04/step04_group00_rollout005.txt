def _exact_c5(h):

def _scores(h):

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        if d == 0:
            G[r, :] = -h
        else:
            m = n - d
            G[r, d:] += (1.0 - h[:m])
            G[r, :m] += -h[d:]
        r2 = n - 1 - d
        if d > 0:
            m = n - d
            G[r2, :m] += (1.0 - h[d:])
            G[r2, d:] += -h[:m]
    return G * (2.0 / n)

def _project(v, target):

            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _lp_direction_full(h, scores, G, radius, target_t=None):

        target_t = float(np.max(scores))

    c[-1] = 1.0

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        cur_c5 = c5_try
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _weighted_subgrad(h, deadline, n_steps=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 5.0 + 200.0 * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.05
        for _ in range(25):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            alpha *= 0.65
            if alpha < 1e-11:
                break
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0)
    h = np.where(h < 0.5, 1.0, 0.0)
    return _project(h, n / 2.0)

def _binary_swap_sa(n, deadline, rng, n_iters=30000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _direct_minimax_lp(h, deadline, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    t_target = cur_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if t_target < best_c5 - 0.001:
            t_target = best_c5 - 0.0005
        scores = _scores(h)
        G = _grad_matrix(h)
        c = G.T @ np.ones(2 * n - 1) / (2 * n - 1)
        c = c - c.mean()
        cn = np.linalg.norm(c)
        if cn < 1e-15:
            break
        c /= cn
        c = -c
        A_ub = np.zeros((2 * n - 1, n))
        A_ub[:] = G
        b_ub = np.full(2 * n - 1, t_target)
        A_eq = np.zeros((1, n))
        A_eq[0, :] = 1.0
        b_eq = np.array([target])
        bounds = [(0.0, 1.0)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.success:
                h_new = res.x
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-13:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    t_target = max(t_target * 0.97, cur_c5 - 0.002)
                else:
                    t_target = (t_target + cur_c5) / 2.0
            else:
                t_target = (t_target + cur_c5) / 2.0
        except Exception:
            t_target = (t_target + cur_c5) / 2.0
        if t_target > cur_c5 - 1e-6:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)
    hard_deadline = t0 + budget_s * 0.985

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Aggressive LP local optimization from start
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _local_opt(h_start, dl1, max_iters=2000, radius_init=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Direct minimax LP
    if time.time() < t0 + budget_s * 0.20:
        dl2 = t0 + budget_s * 0.22
        h2, c5_2 = _direct_minimax_lp(best_h, dl2, max_iters=300)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Weighted subgradient
    if time.time() < t0 + budget_s * 0.28:
        dl3 = t0 + budget_s * 0.30
        h3, c5_3 = _weighted_subgrad(best_h, dl3, n_steps=80)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: LP polish after subgradient
    if time.time() < t0 + budget_s * 0.35:
        dl4 = t0 + budget_s * 0.38
        h4, c5_4 = _local_opt(best_h, dl4, max_iters=1500, radius_init=0.08)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Multi-resolution search
    dl5 = t0 + budget_s * 0.60
    resolutions = [96, 80, 112, 128, 64, 144]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl5:
            break
        if best_c5 < 0.3805:
            break
        # Start 1: resample current best
        h_init1 = _resample(best_h, n_try)
        sub_dl = min(dl5, time.time() + budget_s * 0.03)
        h_try, c5_try = _local_opt(h_init1, sub_dl, max_iters=1200, radius_init=0.12)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
        # Start 2: quasi-periodic
        if time.time() < dl5:
            h_init2 = _quasi_periodic_init(n_try, alphas[ri % len(alphas)])
            h_init2 += rng.normal(0, 0.1, n_try)
            h_init2 = _project(h_init2, n_try / 2.0)
            sub_dl2 = min(dl5, time.time() + budget_s * 0.025)
            h_try2, c5_try2 = _local_opt(h_init2, sub_dl2, max_iters=1000, radius_init=0.15)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()
                best_n = n_try

    # Phase 6: Binary SA at various sizes
    if time.time() < t0 + budget_s * 0.72:
        dl6 = t0 + budget_s * 0.75
        for n_bin in [96, 128, 80]:
            if time.time() > dl6:
                break
            sub_dl = min(dl6, time.time() + budget_s * 0.025)
            h_b, c5_b = _binary_swap_sa(n_bin, sub_dl, rng, n_iters=20000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(dl6, time.time() + budget_s * 0.02)
                h_br, c5_br = _local_opt(h_b, sub_dl2, max_iters=600, radius_init=0.1)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()

    # Phase 7: Perturbation restarts
    dl7 = t0 + budget_s * 0.85
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl7:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.008)
        h_try, c5_try = _local_opt(h_pert, sub_dl, max_iters=300, radius_init=0.05)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.15)

    # Phase 8: Direct minimax LP again
    if time.time() < t0 + budget_s * 0.92:
        dl8 = t0 + budget_s * 0.93
        h8, c5_8 = _direct_minimax_lp(best_h, dl8, max_iters=400)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: Pair polish
    if time.time() < t0 + budget_s * 0.96:
        dl9 = t0 + budget_s * 0.965
        hp, c5p = _pair_polish_smart(best_h, dl9, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 10: Final LP fine-tuning
    if time.time() < t0 + budget_s * 0.98:
        h_fin, c5_fin = _local_opt(best_h, hard_deadline, max_iters=500, radius_init=0.01)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
