from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(38):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _grad_rows(h, row_indices):
    n = len(h)
    G = np.zeros((len(row_indices), n))
    for idx, r in enumerate(row_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, :m] += (1.0 - h[d:d + m])
                G[idx, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:g + m])
                G[idx, g:] += -h[:m]
    return G

def _lp_direction_topk(h, scores, k, radius):
    n = len(h)
    total_rows = 2 * n - 1
    top_k = min(k, total_rows)
    sort_idx = np.argsort(scores)[::-1]
    top_rows = sort_idx[:top_k]
    G = _grad_rows(h, top_rows)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(top_k)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b > hi_b:
            lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((-10.0, 10.0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-7})
        if res.success:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-12:
                return d
    except Exception:
        pass
    return None

def _lp_loop_topk(h, deadline, k_list=(5, 10, 20, 35), radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    ki = 0
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _exact_scores(h)
        k = k_list[ki % len(k_list)]
        p = _lp_direction_topk(h, scores, k, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1

                ki += 1
            radius *= 0.82

                break
            continue

        if new_c5 < cur_c5 - 1e-13:

            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:

            stall = 0

def _grad_step_softmax(h, dx, tau):
    n = len(h)
    scores = _exact_scores(h)
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    all_rows = np.arange(2 * n - 1)
    G = _grad_rows(h, all_rows)
    grad = w @ G
    grad -= np.mean(grad)
    return grad

def _grad_descent_nested(h, deadline, tau_list=(40, 100, 250, 600)):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):

    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()

        if len(interior) < 2:

            i = interior[ii]
            for jj in range(ii + 1, len(interior)):

                d = np.zeros(n)

                    if cur < best_c5:

def _multi_point_refine(h, deadline, n_trials=500, rng_seed=42):

def _flatten_direction(h, dx):
    """Direction that tries to flatten the score vector."""
    n = len(h)
    scores = _exact_scores(h)
    mean_s = float(np.mean(scores))
    # Weight: proportional to how much each row is above mean
    w = np.maximum(scores - mean_s, 0)
    if w.sum() < 1e-14:
        return None
    w = w / w.sum()
    all_rows = np.arange(2 * n - 1)
    G = _grad_rows(h, all_rows)
    d = w @ G
    d -= np.mean(d)
    dn = np.linalg.norm(d)
    if dn < 1e-14:
        return None
    return d / dn
