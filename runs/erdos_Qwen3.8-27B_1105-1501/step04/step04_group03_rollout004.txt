        h_t = h + a * p
        c5_t = exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
        if c5_t >= prev_c5:
            # Parabolic interpolation
            if a > prev_a > 0:
                lam = (prev_c5 - c5_t) / (prev_c5 - 2*best_c5 + c5_t + 1e-30)
                if 0 < lam < 1:
                    a = prev_a + lam * (a - prev_a)
                else:
                    a = a * 0.5
            else:
                a *= 0.5
        else:
            a = min(a * 1.3, alpha_max)
        prev_c5 = c5_t
        prev_a = a
    # Refine with golden section around best
    if best_a > 0:
        lo_a = max(0.0, best_a * 0.7)
        hi_a = min(alpha_max, best_a * 1.3)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_a - gr * (hi_a - lo_a)
        c2 = lo_a + gr * (hi_a - lo_a)
        f1 = exact_c5(h + c1 * p)
        f2 = exact_c5(h + c2 * p)
        for _ in range(20):
            if hi_a - lo_a < 1e-12:
                break
            if f1 < f2:
                hi_a, c2, f2 = c2, c1, f1
                c1 = hi_a - gr * (hi_a - lo_a)
                f1 = exact_c5(h + c1 * p)
            else:
                lo_a, c1, f1 = c1, c2, f2
                c2 = lo_a + gr * (hi_a - lo_a)
                f2 = exact_c5(h + c2 * p)
        opt_a = (lo_a + hi_a) / 2.0
        opt_c5 = exact_c5(h + opt_a * p)
        if opt_c5 < best_c5:
            best_a, best_c5 = opt_a, opt_c5
    return best_a, best_c5

def sca_loop(h, deadline, radius_init=0.1):
    """Successive Convex Approximation loop."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 100:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = build_corr_grad(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = line_search_c5(h, p, cur_c5, max_alpha=radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.8)
        else:
            stall += 1
            radius *= 0.65
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        scores = np.correlate(x, 1.0 - x, mode='full') * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        Z = np.sum(exp_s)
        return max_s + tau * np.log(Z)

    def grad(x):
        scores = np.correlate(x, 1.0 - x, mode='full') * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        Z = np.sum(exp_s)
        w = exp_s / Z
        # Gradient via correlation gradient
        G = build_corr_grad(x)
        g = (w @ G) * dx
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    # Pass target via closure-free approach: use a module-level variable is not ideal
    # We'll use a different approach: fix the constraint inline
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq',
                       'fun': lambda x: np.sum(x) - target,
                       'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def subgrad_momentum_loop(h0, deadline, beta=0.6, lr_init=0.01):
    """Subgradient method with momentum on active lags."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    vel = np.zeros(n)
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        if stall > 150:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_s = np.max(scores)
        active_thresh = max_s * 0.88
        active = np.where(scores >= active_thresh)[0]
        G = build_corr_grad(h)
        grad = G[active].mean(axis=0)
        # Normalize gradient
        gnrm = np.linalg.norm(grad)
        if gnrm < 1e-15:
            break
        grad = grad / gnrm
        # Momentum update
        vel = beta * vel - lr * grad
        h_new = h + vel
        h_new = project(h_new, target)
        c5_new = exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.1)
        else:
            h = h_new  # still move (projected)
            stall += 1
            lr *= 0.7
            if lr < 1e-6:
                break
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    stretched = False
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    # Binary search for optimal fraction
                    lo_f, hi_f = 0.0, 1.0
                    best_f = 0.0
                    best_c5_try = best_c5
                    for _bs in range(8):
                        mid_f = (lo_f + hi_f) / 2.0
                        a = amax * mid_f
                        h_t = best_h.copy()
                        h_t[j] += sign * a
                        h_t[k] -= sign * a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5_try - 1e-13:
                            best_c5_try = c5_t
                            best_f = mid_f
                            lo_f = mid_f
                        else:
                            hi_f = mid_f
                    if best_f > 0 and best_c5_try < best_c5 - 1e-13:
                        a = amax * best_f
                        best_h[j] += sign * a
                        best_h[k] -= sign * a
                        best_c5 = best_c5_try
                        improved = True
                        break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def structured_perturb(h, rng, mode):
    """Generate structured perturbations."""
    n = len(h)
    x = np.arange(n) / n
    if mode == 0:  # Sine wave
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        return h + amp * np.sin(2 * np.pi * freq * x + phase)
    elif mode == 1:  # Gaussian bump
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.2)
        amp = rng.uniform(0.1, 0.3) * rng.choice([1, -1])
        return h + amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif mode == 2:  # Triangle wave
        freq = rng.integers(1, 4)
        phase = rng.uniform(0, 1)
        amp = rng.uniform(0.05, 0.15)
        t = (x * freq + phase) % 1.0
        tri = 1.0 - 2.0 * np.abs(t - 0.5)
        return h + amp * tri
    elif mode == 3:  # 1-h symmetry
        return 1.0 - h
    elif mode == 4:  # Mirror
        return h[::-1].copy()
    else:  # Gaussian noise
        scale = rng.uniform(0.02, 0.15)
        return h + rng.normal(0, scale, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    # Initialize
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: SCA loop (main engine) - 25%
    d1 = t0 + budget_s * 0.25
    h_sca, c5_sca = sca_loop(h, d1, radius_init=0.15)
    if c5_sca < best_c5:
        best_c5 = c5_sca
        best_h = h_sca.copy()

    # Phase 2: SLSQP smoothing - 15%
    d2 = t0 + budget_s * 0.40
    tau_list = [0.3, 0.15, 0.08, 0.04, 0.02]
    h_sl = best_h.copy()
    for tau in tau_list:
        if time.time() > d2:
            break
        maxit = 80 if tau > 0.1 else 60
        h_sl = slsqp_smooth(h_sl, tau, maxit)
        h_sl = project(h_sl, best_n / 2.0)
        c5_sl = exact_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 3: SCA refinement - 10%
    d3 = t0 + budget_s * 0.50
    h_sca2, c5_sca2 = sca_loop(best_h, d3, radius_init=0.08)
    if c5_sca2 < best_c5:
        best_c5 = c5_sca2
        best_h = h_sca2.copy()

    # Phase 4: Subgradient momentum - 10%
    d4 = t0 + budget_s * 0.60
    h_sg, c5_sg = subgrad_momentum_loop(best_h, d4, beta=0.5, lr_init=0.008)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()
    # SCA after subgrad
    if time.time() < d4 + budget_s * 0.05:
        h_sca3, c5_sca3 = sca_loop(best_h, min(d4 + budget_s * 0.05, hard_deadline), radius_init=0.06)
        if c5_sca3 < best_c5:
            best_c5 = c5_sca3
            best_h = h_sca3.copy()

    # Phase 5: Structured global exploration - 20%
    d5 = t0 + budget_s * 0.80
    n_pert = 0
    max_pert = 20
    while time.time() < d5 and n_pert < max_pert:
        mode = n_pert % 6
        h_pert = structured_perturb(best_h, rng, mode)
        h_pert = project(h_pert, best_n / 2.0)
        # Quick SCA
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_try, c5_try = sca_loop(h_pert, sub_dl, radius_init=0.08)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
        n_pert += 1

    # Phase 6: Multi-resolution - 10%
    d6 = t0 + budget_s * 0.90
    offsets = [-4, 4, -8, 8, -12, 12, -16, 16, -24, 24, -32, 32, -40, 48]
    tried_n = set([best_n])
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.010)
        h_r, c5_r = sca_loop(h_r, sub_dl, radius_init=0.07)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final pair polish - 7%
    d7 = t0 + budget_s * 0.97
    h_pol, c5_pol = pair_polish(best_h, d7, max_passes=40)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 8: Final micro-SCA
    if time.time() < hard_deadline:
        h_f, c5_f = sca_loop(best_h, hard_deadline, radius_init=0.03)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
