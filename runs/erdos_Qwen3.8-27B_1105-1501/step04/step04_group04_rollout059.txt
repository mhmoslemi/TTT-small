import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _make_binary(h, n):
    """Convert h to binary with exactly n//2 ones."""
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    return b

def _binary_counts(h):
    """Compute the per-lag 1→0 counts for binary h."""
    # h is binary, returns array of counts for each lag
    La = 1.0 - h
    corr = np.correlate(h, La, mode='full')
    return corr  # For binary h, these are integer counts

def _binary_local_search(h, deadline, rng, max_iters=2000000):

        if no_improve > 10000:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

    ones = list(np.where(h >= 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])

        temp *= cool
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        h[j] = 1.0
        c5_new = _exact_c5(h)

            zeros[j_idx] = i

                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

def _double_swap_search(h, deadline, rng, max_iters=500000):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

def _make_pattern(n, pattern_type='golden', seed=0):
    """Create initial binary patterns."""
    m = n // 2
    b = np.zeros(n)
    if pattern_type == 'golden':
        phi = (1 + 5**0.5) / 2
        positions = [int((k * phi) % 1.0 * n) for k in range(m)]
        # Ensure uniqueness
        positions = []
        for k in range(m):
            p = int((k * phi) % 1.0 * n)
            while p in positions:
                p = (p + 1) % n
            positions.append(p)
        b[positions] = 1.0
    elif pattern_type == 'quadratic':

            while p in positions:
                p = (p + 1) % n
            positions.append(p)
        b[positions] = 1.0
    elif pattern_type == 'random':

        positions = rng.choice(n, size=m, replace=False)
        b[positions] = 1.0
    elif pattern_type == 'alternating':
        b[::2] = 1.0
    return b

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(300):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v


def _make_binary(h, n):
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    return b


def _make_pattern(n, ptype, seed=0):
    m = n // 2
    b = np.zeros(n)
    if ptype == 'golden':
        phi = (1.0 + 5.0**0.5) / 2.0
        used = [False] * n
        cnt = 0
        for k in range(n):
            if cnt >= m:
                break
            p = int((k * phi) % 1.0 * n)
            if not used[p]:
                b[p] = 1.0
                used[p] = True
                cnt += 1
        if cnt < m:
            for k in range(n):
                if cnt >= m:
                    break
                if not used[k]:
                    b[k] = 1.0
                    cnt += 1
    elif ptype == 'quad':
        used = [False] * n
        cnt = 0
        for k in range(n * 2):
            if cnt >= m:
                break
            p = int((k * k + k * m) % n)
            if not used[p]:
                b[p] = 1.0
                used[p] = True
                cnt += 1
        if cnt < m:
            for k in range(n):
                if cnt >= m:
                    break
                if not used[k]:
                    b[k] = 1.0
                    cnt += 1
    elif ptype == 'random':
        rng = np.random.default_rng(seed)
        pos = rng.choice(n, size=m, replace=False)
        b[pos] = 1.0
    elif ptype == 'anti_alt':
        b[1::3] = 1.0
        if int(np.sum(b)) < m:
            b[2::4] = 1.0
        b = _make_binary(b, n)
    return b

def _pair_transfer(h, deadline, rng, max_iters=300000):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)
