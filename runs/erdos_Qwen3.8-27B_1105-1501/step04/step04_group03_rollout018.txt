            best_c5 = c_ann
            best_h = h_ann.copy()
        # Quick local polish after anneal
        if time.time() < d2:
            sub_dl2 = min(d2, time.time() + budget_s * 0.008)
            h_pol, c_pol = local_optimize(h_ann, sub_dl2, n_passes=2)
            if c_pol < best_c5 - 1e-14:
                best_c5 = c_pol
                best_h = h_pol.copy()

    # Phase 3: Multi-resolution search (25% budget)
    d3 = t0 + budget_s * 0.85
    n_candidates = [56, 60, 64, 68, 72, 76, 80, 84, 88, 92, 96, 100, 104, 108, 112]
    # Prioritize smaller n and near-parent n
    n_candidates.sort(key=lambda x: (abs(x - best_n), x))
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try < 40 or n_try > 140:
            continue
        if n_try == best_n:
            continue
        h_r = resample(best_h, n_try)
        # Quick local optimization
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_opt, c_opt = local_optimize(h_r, sub_dl, n_passes=3)
        if c_opt < best_c5 - 1e-12:
            best_c5 = c_opt
            best_h = h_opt.copy()
            best_n = n_try
        # Try annealing at this resolution
        if time.time() < d3:
            sub_dl2 = min(d3, time.time() + budget_s * 0.01)
            h_ann2, c_ann2 = anneal(h_opt, sub_dl2, rng)
            if c_ann2 < best_c5 - 1e-12:
                best_c5 = c_ann2
                best_h = h_ann2.copy()
                best_n = n_try
                h_opt, c_opt = local_optimize(h_ann2, sub_dl2, n_passes=2)
                if c_opt < best_c5:
                    best_c5 = c_opt
                    best_h = h_opt.copy()

    # Phase 4: Final intensive polish (10% budget)
    d4 = t0 + budget_s * 0.95
    sub_dl = min(d4, time.time() + budget_s * 0.03)
    h_final, c_final = guided_pair_sweep(best_h, sub_dl, max_passes=15)
    if c_final < best_c5:
        best_c5 = c_final
        best_h = h_final.copy()

    # More pair sweeps
    if time.time() < hard_deadline:
        sub_dl2 = min(hard_deadline, time.time() + budget_s * 0.02)
        h_f2, c_f2 = local_optimize(best_h, sub_dl2, n_passes=4)
        if c_f2 < best_c5:
            best_c5 = c_f2
            best_h = h_f2.copy()

    # Final short anneal
    if time.time() < hard_deadline:
        sub_dl3 = min(hard_deadline, time.time() + budget_s * 0.015)
        h_f3, c_f3 = anneal(best_h, sub_dl3, rng)
        if c_f3 < best_c5:
            best_c5 = c_f3
            best_h = h_f3.copy()

    # Safety: ensure we don't return worse than parent
    best_h = project_box(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0
    else:
        final_c5 = exact_c5(best_h)

    return best_h, final_c5, best_n
