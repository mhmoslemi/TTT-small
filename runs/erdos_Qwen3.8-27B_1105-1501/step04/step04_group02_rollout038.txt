import numpy as np
from scipy.optimize import linprog, minimize
import time

_G = {}

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _scores_phys(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        mask = (x < 1.0 - 1e-14) if res > 0 else (x > 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(h, r):
    n = len(h)
    g = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        g[:m] += 1.0 - h[d:]
        g[d:] -= h[:m]
    else:
        s = -d
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

def _pnorm_calc(h, p, target, pen):
    n = len(h)
    dx = 2.0 / n
    s = _scores_phys(h)
    s_pos = np.maximum(s, 1e-15)
    s_p = s_pos ** p
    sp = float(np.sum(s_p))
    val = sp ** (1.0 / p) if sp > 1e-30 else 0.0
    ssum = float(np.sum(h))
    val += pen * (ssum - target) ** 2
    if sp < 1e-30:
        return val, 2.0 * pen * (ssum - target) * np.ones(n)
    w = (s_pos ** (p - 1)) * dx / (sp ** (1.0 - 1.0 / p))
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] > 1e-14:
            g += w[r] * _row_grad(h, r)
    g += 2.0 * pen * (ssum - target)
    return val, g

def _obj_fn(x):
    return _pnorm_calc(x, _G['p'], _G['target'], _G['pen'])[0]

def _grad_fn(x):
    return _pnorm_calc(x, _G['p'], _G['target'], _G['pen'])[1]

def _lbfgs_opt(h, deadline, p_list, pen=100.0):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    bounds = [(0.0, 1.0)] * n
    for p in p_list:
        if time.time() > deadline:
            break
        _G['p'] = p
        _G['target'] = target
        _G['pen'] = pen
        try:
            res = minimize(_obj_fn, best_h, jac=_grad_fn, method='L-BFGS-B',
                          bounds=bounds, maxiter=500,
                          options={'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
            h_t = _project(np.clip(res.x, 0.0, 1.0), target)
            c5_t = _exact_c5(h_t)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_t.copy()
        except Exception:
            pass
    return best_h, best_c5

def _line_search(h, p, c5_cur, alpha_max):
    n = len(h)
    best_t, best_val = 0.0, c5_cur
    n_grid = 40
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i in range(n_grid):
        ht = h + ts[i] * p
        if ht.min() >= -1e-12 and ht.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(ht)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 3)]
    hi_t = ts[min(n_grid - 1, idx + 3)]
    if hi_t <= lo_t:
        lo_t, hi_t = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)

    def _feval(t):
        ht = h + t * p
        if ht.min() < -1e-12 or ht.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(ht)

    f1, f2 = _feval(c1), _feval(c2)
    for _ in range(40):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _feval(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _feval(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _feval(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_direction_active(h, scores, active_idx, radius):
    n = len(h)
    A = np.zeros((len(active_idx), n))
    for ii, r in enumerate(active_idx):
        A[ii] = _row_grad(h, r)
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -scores[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_local(h, deadline, radii, active_sizes):
    n = len(h)

def _resample_area(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x0_new = j * dx_new
        x1_new = (j + 1) * dx_new
        total = 0.0
        for i in range(n_old):
            x0_old = i * dx_old
            x1_old = (i + 1) * dx_old
            overlap = max(0.0, min(x1_new, x1_old) - max(x0_new, x0_old))
            total += h_old[i] * overlap
        h_new[j] = total / dx_new
    return _project(h_new, n_new / 2.0)

def _anneal(h, deadline, n_iter=5000, T_start=0.002, T_end=0.0001):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(12345)
    h_cur = h.copy()
    c5_cur = _exact_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    for i in range(n_iter):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (i / max(n_iter - 1, 1))
        scale = 0.03 + 0.02 * T / T_start
        h_new = h_cur + rng.normal(0, scale, n)
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h_cur = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        idx_j = np.random.default_rng(_pass * 7 + 1).integers(0, n, size=min(n, 50))
        idx_k = np.random.default_rng(_pass * 13 + 2).integers(0, n, size=min(n, 50))
        pairs = set()
        for jj in idx_j:
            for kk in idx_k:
                if jj != kk:
                    pairs.add((min(int(jj), int(kk)), max(int(jj), int(kk))))
        pair_list = list(pairs)
        np.random.default_rng(_pass).shuffle(pair_list)
        for (j, k) in pair_list:
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                am = (1.0 - h[j]) / dj if dj > 0 else h[j] / (-dj)
                am = min(am, (1.0 - h[k]) / dk if dk > 0 else h[k] / (-dk))
                if am <= 1e-12:
                    continue
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    ht = h.copy()
                    ht[j] += dj * am * frac
                    ht[k] += dk * am * frac
                    if ht[j] < 0 or ht[j] > 1 or ht[k] < 0 or ht[k] > 1:
                        continue
                    c5_t = _exact_c5(ht)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = ht.copy()
                        h = ht.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5
