import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0 - 1e-9
        else:
            mask = x > 1e-9
        cnt = float(np.sum(mask))
        if cnt > 0:
            x[mask] += residual / cnt
    return np.clip(x, 0.0, 1.0)

def soft_max_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = exp_s[k] / Z
        if pk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:d+m] += pk * (1.0 - h[:m])
                grad[:m] -= pk * h[d:d+m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += pk * (1.0 - h[g:g+m])
                grad[g:g+m] -= pk * h[:m]
    grad *= dx / tau
    return val, grad

def build_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d+m] += (1.0 - h[:m])
                G[r, :m] -= h[d:d+m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:g+m])
                G[r, g:g+m] -= h[:m]
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    max_sc = float(np.max(scores))
    tol = 1e-5 * max_sc + 1e-10
    active = np.where(scores >= max_sc - tol)[0]
    if len(active) == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((-np.inf, np.inf))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.fun < -1e-12:
        return res.x[:n]
    return None

def line_search(h, direction, c5_cur, n_golden=20):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(alpha_max, 0.0)
    if alpha_max < 1e-14:
        return 0.0, c5_cur
    lo, hi = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * direction)
    f2 = exact_c5(h + c2 * direction)
    for _ in range(n_golden):
        if hi - lo < 1e-12 * alpha_max:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * direction)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * direction)
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * direction)
    return opt_a, opt_c5

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        G = build_grad_matrix(h)
        p = lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue

            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def slsqp_smooth(h0, tau, maxiter, deadline=None):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        if deadline is not None and time.time() > deadline:
            return np.inf
        v, _ = soft_max_obj_grad(x, tau)
        return v

    def grad(x):
        if deadline is not None and time.time() > deadline:
            return np.zeros(n)
        _, g = soft_max_obj_grad(x, tau)
        return g

    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def ils_search(h0, deadline, rng, n_passes=50):
    """Iterated Local Search."""
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    cur_h = h0.copy()
    cur_c5 = best_c5
    scales = [0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25]
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        # Perturb
        scale = scales[rng.integers(len(scales))]
        h_pert = cur_h + rng.normal(0, scale, n)
        h_pert = project(h_pert, target)
        # Quick local search: a few LP iterations + pair CD
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.02)
        h_try, c5_try = lp_loop(h_pert, sub_dl, radius_init=0.05)
        if time.time() < deadline:
            h_try, c5_try = pair_cd(h_try, min(deadline, time.time() + 2.0),
                                     max_pairs_per_pass=40, max_passes=20)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            cur_h = h_try.copy()
            cur_c5 = c5_try
        else:
            # With small probability, accept worse (annealing)
            delta = c5_try - cur_c5
            if delta < 0.005 and rng.random() < 0.1:
                cur_h = h_try.copy()
                cur_c5 = c5_try
    return best_h, best_c5
