    """Solve the global LP for the discretized Erdős problem with n points."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Build G matrix: G[k, i] such that sum_i G[k,i] * h[i] = correlation at lag k
    # from the code: G is built in _grad_matrix but scaled by dx already
    G = np.zeros((n_lags, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += 1.0  # (1-h) part contributes 1 to these
                G[r, :m] += -1.0  # -h part contributes -1
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += 1.0
                G[r, g:] += -1.0
    G *= dx
    
    # LP: min t
    # s.t. G @ h <= t for all lags (i.e., G @ h - t <= 0)
    #      sum(h) = n/2
    #      0 <= h[i] <= 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: G @ h - t <= 0 for each lag
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_lags)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    return res

def _fast_coord_descent(h, deadline):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Precompute correlation at current h
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    for pass in range(100):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            # Compute coefficient of h[j] in each s(d)
            # For lag r (0-indexed, 0 to 2n-2), corresponding to d = r - (n-1)
            coeff = np.zeros(2*n-1)
            for r in range(2*n-1):
                d = r - (n-1)
                c = 0.0
                if d >= 0:
                    if j >= d and j + d <= n-1:  # h[j] in h[i+d] slot: need i=j-d >= 0 and i <= n-1-d
                        # wait, i = j-d, need 0 <= j-d <= n-1-d, i.e., d <= j <= n-1
                        c += (1.0 - h[j-d])
                    if j <= n-1-d:  # h[j] in (1-h[i]) slot: need j+d <= n-1
                        c += -h[j+d]
                else:
                    g = -d
                    if j <= n-1-g:  # h[j] in h[i] slot: need j+g <= n-1
                        c += (1.0 - h[j+g])
                    if j >= g:  # h[j] in (1-h[i+g]) slot: need j-g >= 0
                        c += -h[j-g]
                coeff[r] = c
            # Now s(d; x) = corr[r] + dx * (x - old) * coeff[r]
            # Find x in [0,1] minimizing max_r (corr[r] + dx*(x-old)*coeff[r])
            # This is min_x max_r (a_r + b_r * x) where a_r = corr[r] - dx*old*coeff[r], b_r = dx*coeff[r]
            # Convex piecewise linear in x
            # Minimum is where the "upper envelope" is lowest
            # Use golden section since it's convex
            a_arr = corr - dx * old * coeff
            b_arr = dx * coeff
            
            def f(x):
                return np.max(a_arr + b_arr * x)
            
            # Golden section

            gr = (5.0**0.5 - 1.0) / 2.0

            c2 = lo + gr*(hi-lo)

                    f1 = f(c1)
                else:
                    lo = c1; c1 = c2; f1 = f2
                    c2 = lo + gr*(hi-lo)

            best_n = rn
    
    # Phase 4: LP local refinement (10%)
    dl4 = t0 + budget_s * 0.82
    hdr, c5r = _lp_local(best_h, dl4, top_k=6, radius=0.04, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    # Coordinate descent after LP
    dl4b = t0 + budget_s * 0.87
    hdr, c5r = _fast_coord_descent(best_h, dl4b, max_passes=3000, seed=seed + 2)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 5: Multi-start with coordinate descent (8%)
    dl5 = t0 + budget_s * 0.95
    hdr, c5r = _multi_start_coord_descent(best_h, best_n, dl5, n_starts=80, seed=seed + 10)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 6: Final coordinate descent polish (5%)
    if time.time() < t0 + budget_s * 0.99:
        hdr, c5r = _fast_coord_descent(best_h, t0 + budget_s * 0.995, max_passes=3000, seed=seed + 99)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hdr.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > 0.381:
        # Fallback: return the initial if something went wrong
        final_c5 = _exact_c5(h0)
        best_h = h0.copy()
    return best_h, final_c5, best_n
