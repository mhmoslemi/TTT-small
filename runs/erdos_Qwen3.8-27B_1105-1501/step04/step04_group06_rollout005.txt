import numpy as np
from scipy.optimize import minimize, linprog
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _grad_matrix(h, n):
    j = np.arange(n)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        G[r, :] = pd
    return G


def _lp_minimax(h, n, radius, top_k=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(-corr)[:top_k]
    else:
        idx = np.arange(2 * n - 1)
    G_full = _grad_matrix(h, n)
    G = G_full[idx, :]
    c_sub = corr[idx]
    m = len(idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -c_sub
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n], float(res.x[-1])
    return None, None


def _line_search(h, p, n, c5_cur, max_alpha=3.0):
    alpha_max = max_alpha
    for jj in range(n):
        if p[jj] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[jj]) / p[jj])
        elif p[jj] < -1e-14:
            alpha_max = min(alpha_max, h[jj] / (-p[jj]))
    alpha_max = max(alpha_max, 1e-10)
    best_a, best_c5 = 0.0, c5_cur
    grid = np.linspace(0, alpha_max, 15)
    for a in grid:
        if a < 1e-10:
            continue
        h_t = _project(h + a * p, n * 0.5)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, n * 0.5))
        f2 = _exact_c5(_project(h + c2 * p, n * 0.5))
        for _ in range(15):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, n * 0.5))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, n * 0.5))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(_project(h + a * p, n * 0.5))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_optimize(h, n, deadline, radius_init=0.2, top_k=None):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        if stall > 150:
            break
        p, pred_t = _lp_minimax(h, n, radius, top_k=top_k)
        if p is None or np.max(np.abs(p)) < 1e-14:
            radius *= 0.6
            if radius < 1e-6:
                break
            stall += 3
            continue
        alpha, new_c5 = _line_search(h, p, n, cur_c5, max_alpha=4.0)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, n * 0.5)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.2, 0.5)
        else:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            if stall % 30 == 0:
                radius = max(radius, 0.1)
    return best_h, best_c5


def _slsqp_epi(z, nn):
    return z[nn]


def _slsqp_epi_grad(z, nn):
    g = np.zeros(nn + 1)
    g[nn] = 1.0
    return g


def _slsqp_eq_f(z, nn):
    return np.sum(z[:nn]) - nn * 0.5


def _slsqp_eq_g(z, nn):
    g = np.ones(nn + 1)
    g[nn] = 0.0
    return g


def _slsqp_ineq_f(z, nn):
    h, t = z[:nn], z[nn]
    return t - np.correlate(h, 1.0 - h, mode='full')


def _slsqp_ineq_g(z, nn):
    h = z[:nn]
    G = _grad_matrix(h, nn)
    J = np.zeros((2 * nn - 1, nn + 1))
    J[:, :nn] = G
    J[:, nn] = 1.0
    return J


def _slsqp_solve(h_init, n, maxiter=400, ftol=1e-14):
    corr = np.correlate(h_init, 1.0 - h_init, mode='full')
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h_init, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _slsqp_ineq_f, 'jac': _slsqp_ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_slsqp_epi, z0, jac=_slsqp_epi_grad, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h_init.copy(), _exact_c5(h_init)


def _lse_obj(z, n, alpha):
    h = z[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = alpha * (corr - ms)
    vals = np.clip(vals, -500.0, 0.0)
    f = ms + (1.0 / alpha) * np.log(np.sum(np.exp(vals)))
    j = np.arange(n)
    w = np.exp(vals)
    wsum = np.sum(w)
    w = w / wsum
    grad = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        grad[r] = w[r] * np.dot(pd, h)
    # Proper gradient
    grad_h = np.zeros(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        grad_h += w[r] * pd
    grad_h -= np.mean(grad_h)
    return f, grad_h


def _lse_step(h, n, alpha, maxiter=80):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 50, 'iprint': 0})
        x = _project(res.x, target)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _subgradient_step(h, n, eta=0.05):
    corr = np.correlate(h, 1.0 - h, mode='full')
    r_max = int(np.argmax(corr))
    s = r_max - (n - 1)
    j = np.arange(n)
    pd = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    pd[m1] = 1.0 - h[gs[m1]]
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    pd[m2] -= h[ps[m2]]
    pd -= np.mean(pd)
    norm = np.linalg.norm(pd)
    if norm < 1e-15:
        return h.copy(), _exact_c5(h)
    h_new = _project(h - eta * pd / norm, n * 0.5)
    return h_new, _exact_c5(h_new)
