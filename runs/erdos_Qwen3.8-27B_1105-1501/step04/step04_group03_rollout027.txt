def exact_c5(h):

def compute_scores(h):

def project(v, target):

def active_gradient(h, scores, top_k=5, tau=0.05):
    """Compute gradient weighted by active lags only."""
    n = len(h)
    dx = 2.0 / n
    max_s = float(np.max(scores))
    # Softmax weights over all lags
    exp_s = np.exp((scores - max_s) / max(tau, 1e-8))
    Z = np.sum(exp_s)
    w = exp_s / Z
    
    # Zero out low-weight lags for sparsity
    w[w < 0.01] = 0.0
    if np.sum(w) > 0:
        w = w / np.sum(w)
    else:
        # Fallback: uniform over top K
        w = np.zeros(len(scores))
        idx = np.argsort(scores)[-top_k:]
        w[idx] = 1.0 / top_k
    
    G = np.zeros(n)
    center = n - 1
    for r in range(len(scores)):
        if w[r] < 1e-12:
            continue
        d = r - center
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            # score = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
            # d/dh[j] for j in 0..m-1: contribution from h[j] in (1-h[j]) term -> -h[j+d]
            # d/dh[j] for j in d..n-1: contribution from h[j] in h[j] term -> (1-h[j-d])
            G[0:m] += w[r] * (-h[d:d+m])
            G[d:n] += w[r] * (1.0 - h[0:m])

            G[0:m] += w[r] * (1.0 - h[q:q+m])
            G[q:n] += w[r] * (-h[0:m])

    A_ub = np.zeros((K, m + 1))
    A_ub[:, :m] = G_a * dx
    A_ub[:, m] = -1.0
    b_ub = -scores[top_idx]
    bounds = []
    for j in range(m):
        bounds.append((max(-radius, -a[j]), min(radius, 1.0 - a[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.status == 0:
            return res.x[:m]
    except Exception:
        pass
    return None

def sym_local_search(a, deadline, mode='lp', K=6, radius=0.12, tau=0.03):
    m = len(a)
    a = a.copy()
    cur_c5 = sym_c5(a)
    best_a = a.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    while time.time() < deadline:
        if stall > 80:
            break
        if mode == 'lp':
            p = sym_reduce_lp(a, K=K, radius=r)
            if p is None or np.max(np.abs(p)) < 1e-14:
                stall += 1
                r *= 0.7
                if r < 1e-7:
                    break
                continue
        else:
            G = sym_gradient(a, tau=tau, K=8)
            p = -G
            pn = np.max(np.abs(p))
            if pn < 1e-14:
                stall += 1
                tau *= 0.8
                if stall > 50:
                    break
                continue
            p /= pn
        alpha, new_c5 = line_search_sym(a, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            a = np.clip(a + alpha * p, 0.0, 1.0)
            cur_c5 = sym_c5(a)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_a = a.copy()
            stall = 0
            if mode == 'lp':
                r = min(r * 1.1, 0.5)
            else:
                tau *= 0.97
        else:
            stall += 1
            if mode == 'lp':
                r *= 0.7
                if r < 1e-7:
                    break
            else:
                tau *= 0.85
    return best_a, best_c5

def sym_pair_polish(a, deadline, max_passes=30):
    m = len(a)
    a = a.copy()
    best_a = a.copy()
    best_c5 = sym_c5(a)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(m):
            if time.time() > deadline:
                break
            for k in range(j + 1, m):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_a[j], 1.0 - best_a[k])
                    else:
                        amax = min(1.0 - best_a[j], best_a[k])
                    if amax < 1e-10:
                        continue
                    for frac in [0.5, 1.0]:
                        al = amax * frac
                        at = best_a.copy()
                        at[j] += sign * al
                        at[k] -= sign * al
                        c5_t = sym_c5(at)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_a = at.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_a, best_c5

def sym_anneal(a, deadline, n_walks=6):
    m = len(a)
    rng = np.random.default_rng(123)
    a_cur = a.copy()
    c5_cur = sym_c5(a_cur)
    best_a = a.copy()
    best_c5 = c5_cur
    T = 0.004
    T_min = 0.0001
    for _walk in range(n_walks):
        if time.time() > deadline:
            break
        a_cur = best_a.copy()
        c5_cur = best_c5
        for _step in range(300):
            if time.time() > deadline:
                break
            i1 = rng.integers(m)
            i2 = rng.integers(m)
            if i1 == i2:
                continue
            if a_cur[i1] < a_cur[i2]:
                amax = min(1.0 - a_cur[i1], a_cur[i2])
                sign = 1.0
            else:
                amax = min(1.0 - a_cur[i2], a_cur[i1])
                sign = -1.0
            if amax < 1e-8:
                continue
            al = amax * rng.uniform(0.3, 1.0)
            at = a_cur.copy()
            at[i1] += sign * al
            at[i2] -= sign * al
            c5_t = sym_c5(at)
            delta = c5_t - c5_cur
            if delta < -1e-12:
                a_cur = at
                c5_cur = c5_t
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_a = a_cur.copy()
            elif delta < 0.005 and rng.random() < math.exp(-max(delta, 0) / T):
                a_cur = at
                c5_cur = c5_t
        T = max(T * 0.6, T_min)
    return best_a, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.95

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(h0)
    best_h = h0.copy()
    best_c5 = parent_c5
    best_n = n0

    # Extract symmetric component (for even n)
    n_sym = n0 if n0 % 2 == 0 else n0 - 1
    m = n_sym // 2
    if n0 % 2 != 0:
        h_sym_init = resample_h(h0, n_sym)
    else:
        h_sym_init = h0.copy()
    a_init = np.zeros(m)
    a_init[:] = (h_sym_init[:m] + (1.0 - h_sym_init[m:][::-1])) / 2.0
    a_init = np.clip(a_init, 0.0, 1.0)

    # Phase 1: LP local search on symmetric subspace
    d1 = t0 + budget_s * 0.22
    a_lp, c5_lp = sym_local_search(a_init, d1, mode='lp', K=6, radius=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = sym_encode(a_lp)
        best_n = 2 * m

    # Phase 2: Gradient local search (refinement)
    d2 = t0 + budget_s * 0.38
    a_g, c5_g = sym_local_search(a_lp, d2, mode='grad', K=8, tau=0.03)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = sym_encode(a_g)
        best_n = 2 * m

    # Phase 3: Annealing
    d3 = t0 + budget_s * 0.55
    a_ann, c5_ann = sym_anneal(a_g, d3, n_walks=8)
    if c5_ann < best_c5:
        best_c5 = c5_ann
        best_h = sym_encode(a_ann)
        best_n = 2 * m

    # Phase 4: LP refinement after anneal
    d4 = t0 + budget_s * 0.68
    a_lp2, c5_lp2 = sym_local_search(a_ann, d4, mode='lp', K=5, radius=0.08)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = sym_encode(a_lp2)
        best_n = 2 * m

    # Phase 5: Multi-resolution search
    d5 = t0 + budget_s * 0.82
    current_a = a_lp2
    for dn in [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 160 or n_try % 2 != 0:
            continue
        m_try = n_try // 2
        h_cur = sym_encode(current_a)
        h_r = resample_h(h_cur, n_try)
        a_r = np.zeros(m_try)
        a_r[:] = (h_r[:m_try] + (1.0 - h_r[m_try:][::-1])) / 2.0
        a_r = np.clip(a_r, 0.0, 1.0)
        sub_dl = min(d5, time.time() + budget_s * 0.018)
        a_r, c5_r = sym_local_search(a_r, sub_dl, mode='lp', K=5, radius=0.08)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = sym_encode(a_r)
            best_n = n_try
            current_a = a_r

    # Phase 6: Pair polish on best
    d6 = t0 + budget_s * 0.90
    m_final = best_n // 2
    a_final = np.zeros(m_final)
    a_final[:] = (best_h[:m_final] + (1.0 - best_h[m_final:][::-1])) / 2.0
    a_final = np.clip(a_final, 0.0, 1.0)
    a_pol, c5_pol = sym_pair_polish(a_final, d6, max_passes=25)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = sym_encode(a_pol)
        best_n = 2 * m_final

    # Phase 7: Final micro LP polish
    if time.time() < hard_deadline:
        m_fin = best_n // 2
        a_fin = np.zeros(m_fin)
        a_fin[:] = (best_h[:m_fin] + (1.0 - best_h[m_fin:][::-1])) / 2.0
        a_fin = np.clip(a_fin, 0.0, 1.0)
        a_f, c5_f = sym_local_search(a_fin, hard_deadline, mode='lp', K=4, radius=0.03)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = sym_encode(a_f)
            best_n = 2 * m_fin

    # Final validation
    final_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(final_h)

    # Safety: never worse than parent
    if final_c5 > parent_c5 + 1e-13:
        final_h = h0.copy()
        final_c5 = parent_c5
        best_n = n0

    return final_h, final_c5, best_n
