import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_mean(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_ABCD(mask, n):
    """Precompute A_m, B_m, C_m, D_m for all lags m=1..n-1."""
    # For lag m, over i in [0, n-1-m]:
    # A_m = #{i: mask[i]=1, mask[i+m]=1}
    # B_m = #{i: mask[i]=1, mask[i+m]=0}
    # C_m = #{i: mask[i]=0, mask[i+m]=1}
    # D_m = #{i: mask[i]=0, mask[i+m]=0}
    A = np.zeros(n)  # A[0] unused
    B = np.zeros(n)
    C = np.zeros(n)
    D = np.zeros(n)
    for m in range(1, n):
        valid = n - m
        a = mask[:valid]
        b = mask[m:m+valid]
        A[m] = np.sum((a == 1) & (b == 1))
        B[m] = np.sum((a == 1) & (b == 0))
        C[m] = np.sum((a == 0) & (b == 1))
        D[m] = np.sum((a == 0) & (b == 0))
    return A, B, C, D

def two_level_score(ABCD, d, n):
    """Compute C5 for two-level construction with given d."""
    A, B, C, D = ABCD
    one_minus_d = 1.0 - d
    # c[n-1-m] = (1-d)*d*A + (1-d)²*B + d²*C + d*(1-d)*D
    c_fwd = one_minus_d * d * A + one_minus_d**2 * B + d**2 * C + d * one_minus_d * D
    # c[n-1+m] = (1-d)*d*A + (1-d)²*C + d²*B + d*(1-d)*D
    c_bwd = one_minus_d * d * A + one_minus_d**2 * C + d**2 * B + d * one_minus_d * D
    # Note: c[0] (lag 0) = sum h[i]*(1-h[i]) which for two-level is:
    # (1-d)*d * (n/2) + d*(1-d) * (n/2) = d*(1-d)*n
    # Wait, for lag 0: h[i]*(1-h[i]) = (1-d)*d for mask=1, d*(1-d) for mask=0
    # So c[0] = (1-d)*d*(n/2) + d*(1-d)*(n/2) = d*(1-d)*n
    # But this is already included in A[0]... wait no, I only computed m=1..n-1.
    # For m=0: A[0]=n/2, B[0]=0, C[0]=0, D[0]=n/2
    # c_fwd[0] = (1-d)*d*(n/2) + 0 + 0 + d*(1-d)*(n/2) = d*(1-d)*n
    # This is the self-overlap, which is always the same for all two-level constructions.
    # Actually wait - the problem has the max over ALL k including k=0.
    # For the original problem, k=0 gives integral h(x)*(1-h(x)) dx
    # For two-level: = (1-d)*d*(n/2)*dx + d*(1-d)*(n/2)*dx = d*(1-d)*n*dx = d*(1-d)*2
    # So for d=0.5, this is 0.5. For d=0, this is 0.
    # But we need to include this in the max!
    c0 = d * (1.0 - d) * n  # This is the lag-0 correlation value
    max_corr = max(np.max(c_fwd[1:]), np.max(c_bwd[1:]), c0)
    return float(max_corr * 2.0 / n)

def optimize_d(ABCD, n, d_min=0.0, d_max=0.5):
    """Find optimal d for two-level construction."""
    # Coarse grid
    d_grid = np.linspace(d_min, d_max, 200)
    scores = np.array([two_level_score(ABCD, d, n) for d in d_grid])
    best_idx = np.argmin(scores)
    best_d = d_grid[best_idx]
    best_score = scores[best_idx]
    
    # Fine grid around best
    if best_idx > 0 and best_idx < len(d_grid) - 1:
        d_fine = np.linspace(d_grid[best_idx-1], d_grid[best_idx+1], 500)
        scores_fine = np.array([two_level_score(ABCD, d, n) for d in d_fine])
        fine_idx = np.argmin(scores_fine)
        if scores_fine[fine_idx] < best_score:
            best_d = d_fine[fine_idx]
            best_score = scores_fine[fine_idx]
    
    return best_d, best_score

def two_level_h(mask, d, n):
    """Construct the two-level vector."""
    h = np.where(mask == 1, 1.0 - d, d).astype(float)
    return h

def binary_local_search(mask, deadline, max_rounds=500):
    """Best-improvement single-swap local search on binary mask."""
    n = len(mask)
    mask = mask.astype(int).copy()
    ones = list(np.where(mask == 1)[0])
    zeros = list(np.where(mask == 0)[0])

            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            best_score = best_s
            best_mask = mask.copy()
        else:
            break
    return best_mask, best_score
