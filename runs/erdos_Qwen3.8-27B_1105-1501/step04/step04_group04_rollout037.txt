def exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def raw_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(300):
        s = np.sum(v)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def feasible_interval(h, v):
    """Compute the feasible interval [t_min, t_max] for h + t*v."""
    n = len(h)
    t_min = 0.0
    t_max = 0.0
    pos = v > 1e-15
    neg = v < -1e-15

        t_max = min(t_max, np.max(-h[neg] / v[neg]))

def active_gradient(h, beta=20.0, top_k=None):
    """Compute the soft-min weight gradient of the active lags."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    scores = raw_scores(h)
    max_s = np.max(scores)
    
    # Soft weights
    exp_v = np.exp(beta * (scores - max_s))
    w = exp_v / np.sum(exp_v)
    
    # Build gradient
    g = np.zeros(n)
    for r in range(L):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        g[j] += w[r] * dx * (1.0 - h[i])
        g[i] += -w[r] * dx * h[j]
    
    return g

def _quadratic_coeffs(h, v, active_indices):

    # And sum_i h_i * v_{i+k} = ... this is related to correlate(h, v) at lag -k

    b = np.zeros(len(active_indices))
    c = np.zeros(len(active_indices))
    for idx, r in enumerate(active_indices):
        d = r - (n-1)
        lo_i = max(0, -d)
        hi_i = min(n, n-d)
        if lo_i >= hi_i:
            continue
        i = np.arange(lo_i, hi_i)
        j = i + d
        a[idx] = np.sum(h[i] * (1.0 - h[j])) * dx
        b[idx] = np.sum(v[i] * (1.0 - h[j]) - h[i] * v[j]) * dx
        c[idx] = -np.sum(v[i] * v[j]) * dx
    return a, b, c

My final approach:
1. **Core loop**: Generate directions, do exact 1D line search with golden section
2. **Direction types**: 
   - Active gradient (LSE weighted)
   - Individual lag gradients (top 5)
   - LP direction (every 5th iteration)
   - Random zero-sum
   - Negated previous direction (to explore both sides)
3. **Line search**: 
   - Compute feasible interval
   - Evaluate at t=0, t=mid, t=lo, t=hi, and a few intermediate points
   - Golden section refinement around best point
   - Total ~20-30 evaluations per direction
4. **ILS**: Every 200 iterations without improvement, apply a random perturbation and restart
5. **Multi-resolution**: At the end, try a few nearby resolutions
6. **Time management**: Track elapsed time throughout, return best found before deadline

import numpy as np

def _c5(h):

def _scores_raw(h):

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(800):
        diff = target - np.sum(v)
        if abs(diff) < 1e-15:

        m = v < 1.0 if diff > 0 else v > 0.0
        c = int(np.sum(m))
        if c == 0:
            break
        v[m] += diff / c
        v = np.clip(v, 0.0, 1.0)
    return v

def _t_range(h, v):
    p = v > 1e-14
    q = v < -1e-14
    lo = 0.0
    hi = 0.0
    if np.any(p):

def _golden_section(h, v, a, b, deadline, n_max=20):

    best_t = 0.0
    best_c = _c5(h)
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = _c5(h + c1 * v)
    f2 = _c5(h + c2 * v)

        if deadline is not None and time.time() > deadline:
            break
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = _c5(h + c1 * v)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = _c5(h + c2 * v)
        if f1 < best_c:
            best_c, best_t = f1, c1
        if f2 < best_c:
            best_c, best_t = f2, c2
    return best_t, best_c

def _line_search(h, v, deadline, n_evals=35):
    n = len(h)
    t_lo, t_hi = _t_range(h, v)
    if t_hi - t_lo < 1e-13:
        return 0.0, _c5(h)
    best_t = 0.0
    best_c = _c5(h)
    n_grid = min(n_evals - 10, 16)
    ts = np.linspace(t_lo, t_hi, n_grid)
    for t in ts:
        if deadline is not None and time.time() > deadline:
            break
        c = _c5(h + t * v)
        if c < best_c:
            best_c, best_t = c, t
    if best_c < _c5(h) - 1e-15:
        best_idx = int(np.argmin([_c5(h + t * v) for t in ts])) if len(ts) > 2 else 0
    if np.any(ts) and best_t != 0.0:
        seg = max(abs(t_hi - t_lo) / n_grid, 1e-12)
        a = max(t_lo, best_t - seg)

            gt, gc = _golden_section(h, v, a, b, deadline, n_max=18)
            if gc < best_c:
                best_t, best_c = gt, gc
    return best_t, best_c

def _active_grad(h, beta=30.0):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    scores = _scores_raw(h)
    max_s = np.max(scores)
    exp_v = np.exp(beta * (scores - max_s))
    w = exp_v / (np.sum(exp_v) + 1e-300)
    g = np.zeros(n)
    for r in range(L):
        if w[r] < 1e-18:
            continue
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        i = np.arange(lo, hi)
        j = i + d
        g[j] += w[r] * dx * (1.0 - h[i])
        g[i] += -w[r] * dx * h[j]
    return g, scores

def _lag_grad(h, r):
    n = len(h)
    dx = 2.0 / n
    d = r - (n - 1)
    lo = max(0, -d)
    hi = min(n, n - d)
    if lo >= hi:
        return np.zeros(n)
    i = np.arange(lo, hi)
    j = i + d
    g = np.zeros(n)
    g[j] += dx * (1.0 - h[i])
    g[i] += -dx * h[j]
    return g

def _normalize_zero_sum(v):
    v = v - np.mean(v)
    m = np.max(np.abs(v))
    if m < 1e-14:
        return None
    return v / m

def _lp_dir(h, radius=0.08):
    from scipy.optimize import linprog
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    scores = _scores_raw(h)
    max_s = np.max(scores)
    active_mask = scores >= max_s * (1.0 - 0.005)
    rows = np.where(active_mask)[0]
    G = np.zeros((len(rows), n))
    F_active = scores[active_mask].copy()
    for idx, r in enumerate(rows):
        d = r - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        i = np.arange(lo, hi)
        j = i + d
        G[idx, j] += dx * (1.0 - h[i])
        G[idx, i] += -dx * h[j]
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((len(rows), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F_active
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([n / 2.0])
    bounds = [(max(0.0, h[i] - radius), min(1.0, h[i] + radius)) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None:
            return res.x[:n] - h
    except Exception:
        pass
    return None

def _main_loop(h, deadline, rng, use_lp=True, lp_every=4):

            best_c5 = ct

            cur_h = ht.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _try_resolution(h_best, c_best, n_best, deadline, rng):

    candidates = sorted(set(candidates), key=lambda x: abs(x - n_best))
    per_time = max(1.0, (deadline - time.time()) / max(1, len(candidates)))
    for nn in candidates:
        if time.time() > deadline:

        if nn == n_best:

        h_r = _resample(h_best, nn)
        c_r = _c5(h_r)
        sub_dl = min(deadline, time.time() + per_time * 0.7)
        if sub_dl - time.time() < 2.0:

        h_opt, c_opt = _main_loop(h_r, sub_dl, rng, use_lp=True, lp_every=3)
        if c_opt < c_best - 1e-12:
            c_best = c_opt
            h_best = h_opt
            n_best = nn
    return h_best, c_best, n_best
