def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def random_seed(n, rng):
    """Generate a random feasible h with structure."""
    # Mix of binary and smooth
    if rng.random() < 0.5:
        # Binary with structure
        h = np.zeros(n)
        n_ones = n // 2
        # Try to make it "interleaved" or "clustered"
        if rng.random() < 0.3:
            # Clustered
            start = rng.integers(0, n)
            for i in range(n_ones):
                h[(start + i) % n] = 1.0
        elif rng.random() < 0.5:
            # Interleaved
            for i in range(0, n, 2):
                h[i] = 1.0
            # Fix count
            n_set = np.count_nonzero(h)
            if n_set > n_ones:
                idx = np.where(h == 1)[0]
                to_remove = rng.choice(idx, n_set - n_ones, replace=False)
                h[to_remove] = 0
            elif n_set < n_ones:
                idx = np.where(h == 0)[0]
                to_add = rng.choice(idx, n_ones - n_set, replace=False)
                h[to_add] = 1
        else:
            # Random binary
            h[rng.choice(n, n_ones, replace=False)] = 1.0
    else:
        # Smooth: cosine-like or random smooth
        phase = rng.random() * 2 * np.pi
        h = 0.5 + 0.5 * np.cos(2 * np.pi * rng.integers(1, 5) * np.arange(n) / n + phase)
        h = project(h, n / 2.0)
    return project(h, n / 2.0)

    # Phase 3: Multi-resolution (12% of budget)
    d3 = t0 + budget_s * 0.82
    offsets = [-4, 4, -8, 8, -12, 12, -20, 20, -16, 16, -6, 6]
    tried_n = set([best_n])
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        c5_r = exact_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_n = n_try
        sub_dl = min(d3, time.time() + budget_s * 0.010)
        h_r2, c5_r2 = pair_descent(h_r, sub_dl, max_iters=3000, n_active=6, top_pairs=20)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = n_try
    
    # Phase 4: Fine pair polish (10% of budget)
    d4 = t0 + budget_s * 0.92
    h_pol, c5_pol = fine_pair_polish(best_h, d4, max_passes=60)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    
    # Phase 5: Final micro-descent
    if time.time() < hard_deadline:
        h_f, c5_f = pair_descent(best_h, hard_deadline, max_iters=30000, n_active=12, top_pairs=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    
    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-12:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0
    
    return best_h, final_c5, best_n
