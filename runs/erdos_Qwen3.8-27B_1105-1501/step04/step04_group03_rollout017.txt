def project(v, target):

def build_corr_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g_ = -d
            m = n - g_
            if m > 0:
                G[r, :m] += (1.0 - h[g_:])
                G[r, g_:] += -h[:m]
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        h = h + best_a * p
        h = project(h, target)
        cur_c5 = exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        stall = 0
        radius = min(radius * 1.1, 0.5)
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / (n_old * 2.0)
    x_new = (np.arange(n_new) + 0.5) / (n_new * 2.0)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def simulated_annealing(h, deadline, T0=0.01, T_mult=0.9995, min_T=1e-6):
    """SA with pair moves and exact evaluation."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = T0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline and T > min_T:
        T *= T_mult
        # Propose a random pair move
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        sign = 1.0 if rng.random() < 0.5 else -1.0
        if sign > 0:
            t_max = min(h[i], 1.0 - h[j])
        else:
            t_max = min(h[j], 1.0 - h[i])
        if t_max < 1e-10:
            continue
        t = t_max * rng.random() * 0.5
        
        h_new = h.copy()
        h_new[i] += sign * t
        h_new[j] -= sign * t
        c5_new = exact_c5(h_new)
        
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5
