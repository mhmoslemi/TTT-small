def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-15
        else:
            mask = v > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        # Avoid overshoot
        if diff > 0:
            # Can only add up to (1-v) at each position
            max_add = 1.0 - v[mask]
            actual_step = min(step, max_add.mean())
        else:
            max_sub = v[mask]
            actual_step = max(step, -max_sub.mean())
        v[mask] += actual_step
        v = np.clip(v, 0.0, 1.0)
    return v

def _try_pair(h, i, j, scores, n):
    """Try two-point exchange. Returns (tau, new_c5) or (None, None)."""
    hi_val = h[i]
    hj_val = h[j]
    tau_lo = max(-hi_val, hj_val - 1.0)
    tau_hi = min(1.0 - hi_val, hj_val)
    if tau_hi <= tau_lo + 1e-12:
        return None, None
    
    # Compute slope vector a_r
    # a_r (for lag d = r - (n-1)) = [h[j-d] - h[i-d]] + [h[j+d] - h[i+d]]
    # with out-of-range = 0
    d = np.arange(-(n-1), n)
    # h[j-d]: index j-d, valid if 0 <= j-d < n
    jd = j - d
    id_ = i - d
    jd2 = j + d
    id2 = i + d
    
    a = np.zeros(2*n - 1)
    # h[j-d]
    m1 = (jd >= 0) & (jd < n)
    a[m1] += h[jd[m1]]
    # -h[i-d]
    m2 = (id_ >= 0) & (id_ < n)
    a[m2] -= h[id_[m2]]
    # h[j+d]
    m3 = (jd2 >= 0) & (jd2 < n)
    a[m3] += h[jd2[m3]]
    # -h[i+d]
    m4 = (id2 >= 0) & (id2 < n)
    a[m4] -= h[id2[m4]]
    
    # f(tau) = max(scores + a*tau) * (2/n)
    # Convex PL. Minimize over [tau_lo, tau_hi]
    # Evaluate at endpoints and at crossing points

    dx = 2.0 / n

    cands = [tau_lo, tau_hi]
    
    # Find top-2 at tau_lo and tau_hi
    val_lo = s_scaled + a * tau_lo
    val_hi = s_scaled + a * tau_hi
    
    r1_lo = int(np.argmax(val_lo))
    r1_hi = int(np.argmax(val_hi))
    
    # Crossing of top terms
    if r1_lo != r1_hi and a[r1_lo] != a[r1_hi]:
        tc = (s_scaled[r1_hi] - s_scaled[r1_lo]) / (a[r1_lo] - a[r1_hi])
        if tau_lo < tc < tau_hi:
            cands.append(tc)
    
    # Also try: crossing of each top-3 term at lo with top-3 at hi
    top_lo = np.argsort(val_lo)[-3:][::-1]
    top_hi = np.argsort(val_hi)[-3:][::-1]
    for rl in top_lo:
        for rh in top_hi:
            if rl == rh:
                continue
            if a[rl] != a[rh]:
                tc = (s_scaled[rh] - s_scaled[rl]) / (a[rl] - a[rh])
                if tau_lo < tc < tau_hi:
                    cands.append(tc)
    
    # Evaluate all candidates
    best_tau = None
    best_val = cur_max
    for tc in cands:
        tc = max(tau_lo, min(tau_hi, tc))
        v = float(np.max(s_scaled + a * tc)) * dx
        if v < best_val - 1e-14:
            best_val = v
            best_tau = tc
    
    if best_tau is None:
        return None, None
    return best_tau, best_val

def _try_pair_fast(h, i, j, scores, n):
    """Try two-point exchange using linear approximation + exact evaluation."""
    hi_val = h[i]
    hj_val = h[j]
    tau_lo = max(-hi_val, hj_val - 1.0)
    tau_hi = min(1.0 - hi_val, hj_val)
    if tau_hi <= tau_lo + 1e-12:
        return None, None
    
    # Compute linear slope a_r
    d = np.arange(-(n-1), n)
    jd = j - d; id_ = i - d; jd2 = j + d; id2 = i + d
    a = np.zeros(2*n - 1)
    m1 = (jd >= 0) & (jd < n); a[m1] += h[jd[m1]]
    m2 = (id_ >= 0) & (id_ < n); a[m2] -= h[id_[m2]]
    m3 = (jd2 >= 0) & (jd2 < n); a[m3] += h[jd2[m3]]
    m4 = (id2 >= 0) & (id2 < n); a[m4] -= h[id2[m4]]
    
    dx = 2.0 / n
    cur_max = float(np.max(scores)) * dx
    
    # Find candidate taus
    # The linear prediction says f_lin(tau) = max(scores + a*tau) * dx
    # Find where the max changes
    cands = np.array([tau_lo, tau_hi])
    
    # Top terms at endpoints
    val_lo = scores + a * tau_lo
    val_hi = scores + a * tau_hi
    r_lo = int(np.argmax(val_lo))
    r_hi = int(np.argmax(val_hi))
    
    if r_lo != r_hi and abs(a[r_lo] - a[r_hi]) > 1e-15:
        tc = (scores[r_hi] - scores[r_lo]) / (a[r_lo] - a[r_hi])
        if tau_lo < tc < tau_hi:
            cands = np.append(cands, tc)
    
    # Add a few quantiles
    cands = np.append(cands, tau_lo + np.array([0.25, 0.5, 0.75]) * (tau_hi - tau_lo))
    
    # Exact evaluation at candidates
    best_tau = None
    best_val = cur_max
    
    for tc in cands:
        tc = max(tau_lo, min(tau_hi, tc))
        # Update two entries
        h_old_i = h[i]
        h_old_j = h[j]
        h[i] = h_old_i + tc
        h[j] = h_old_j - tc
        # Recompute scores (O(n^2) but n=128 so ~16K ops)
        # Actually, let's use the linear update + correction
        new_scores = scores + a * tc
        # Correct the overlapping lags
        # d=0: index n-1
        new_scores[n-1] = 0  # placeholder, will fix
        # Actually, let me just recompute fully for correctness
        new_scores = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(new_scores)) * dx
        # Restore
        h[i] = h_old_i
        h[j] = h_old_j
        
        if c5_val < best_val - 1e-14:
            best_val = c5_val
            best_tau = tc
    
    if best_tau is None:
        return None, None
    return best_tau, best_val
