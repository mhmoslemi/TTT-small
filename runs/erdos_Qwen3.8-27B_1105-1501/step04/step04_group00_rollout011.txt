Let me think about what makes this different from the previous program:

The previous program used LP direction finding (find a direction that decreases all top-k constraints simultaneously). The new approach should use an **active equality** method: find a direction that makes the top active constraints equal to a common lower value. This is a quadratic subproblem (weighted least squares on the active rows) rather than a linear program.

Key differences:
- The KKT system: min ||G_active^T δ - target||^2 + ridge||δ||² + roughness, subject to 1^T δ = 0
- This produces directions that specifically target equalization of the active constraints
- Multiple variant configurations with different tolerance, margin, ridge parameters

Let me think about the implementation details:

For the KKT system:
- G is (m × n) where m = number of active constraints
- H = G^T diag(q) G + ridge * I + roughness_matrix
- Solve [H, 1; 1^T, 0] [δ; μ] = [G^T q * (target_a - s_a); 0]

The target for each active constraint is `c_min - sigma` where sigma is a small margin.

For the roughness penalty, I'll use a simple discrete Laplacian:
- H[i,i] += 2*rho
- H[i,i+1] += -rho (if i+1 < n)
- H[i+1,i] += -rho (if i+1 < n)

def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def scores_exact(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_h(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Handle non-finite
    v = np.where(np.isfinite(v), v, 0.5)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    # Clean up near-boundary values
    x[np.abs(x) < 1e-15] = 0.0
    x[np.abs(1.0 - x) < 1e-15] = 1.0
    return x

def build_grad_rows(h, active_indices):
    """Build gradient rows for selected correlation output indices."""
    n = len(h)
    dx = 2.0 / n
    m = len(active_indices)
    G = np.zeros((m, n))
    for k in range(m):
        r = active_indices[k]
        d = r - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[k, d:d+length] += dx * (1.0 - h[:length])
                G[k, :length] -= dx * h[d:d+length]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[k, :length] += dx * (1.0 - h[g:g+length])
                G[k, g:g+length] -= dx * h[:length]
    return G

def roughness_matrix(n, rho):
    """Build discrete Laplacian roughness penalty matrix."""
    if rho == 0.0:
        return np.zeros((n, n))
    H = np.zeros((n, n))
    H[np.arange(n), np.arange(n)] = 2.0 * rho
    if n > 1:
        idx = np.arange(n - 1)
        H[idx, idx + 1] = -rho
        H[idx + 1, idx] = -rho
    return H

def active_direction(h, scores, cfg):
    """Solve active-constraint equality KKT system for direction."""
    n = len(h)
    c_max = float(np.max(scores))
    
    # Select active set
    active_tol = cfg['active_tol']
    min_active = cfg['min_active']
    max_active = cfg['max_active']
    
    # Sort by score descending
    sorted_idx = np.argsort(scores)[::-1]
    active_mask = scores >= (c_max - active_tol)
    active_indices = sorted_idx[active_mask]
    
    # Ensure minimum active count
    if len(active_indices) < min_active:
        active_indices = sorted_idx[:min_active]
    # Cap at max
    if len(active_indices) > max_active:
        active_indices = sorted_idx[:max_active]
    
    m = len(active_indices)
    if m < 2:
        return None
    
    s_active = scores[active_indices]
    c_min_a = float(np.min(s_active))
    
    # Target: push all active constraints to c_min_a - margin
    spread = c_max - c_min_a
    margin = max(cfg['margin_floor'], cfg['margin_factor'] * (spread + cfg['margin_floor']))
    target_a = c_min_a - margin
    
    # Weights: prioritize higher-scoring constraints
    gaps = c_max - s_active
    max_gap = max(float(np.max(gaps)), 1e-15)
    q = 1.0 + cfg['priority'] * (gaps / max_gap) ** 2
    
    # Build gradient rows
    G = build_grad_rows(h, active_indices)
    
    # Build H = G^T diag(q) G + ridge*I + roughness
    Gq = G * q[:, None]
    H = Gq.T @ G
    H += cfg['ridge'] * np.eye(n)
    H += roughness_matrix(n, cfg['rough'])
    
    # RHS: G^T q * (target_a - s_active)
    rhs_vec = Gq.T @ (target_a - s_active)
    
    # KKT system: [H, 1; 1^T, 0] [δ; μ] = [rhs_vec; 0]
    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n + 1)
    rhs[:n] = rhs_vec
    
    try:
        sol = np.linalg.solve(KKT, rhs)
        p = sol[:n]
    except np.linalg.LinAlgError:
        # Fallback: increase ridge and retry
        H2 = H + 0.1 * np.eye(n)
        KKT2 = np.zeros((n + 1, n + 1))
        KKT2[:n, :n] = H2
        KKT2[:n, n] = 1.0
        KKT2[n, :n] = 1.0
        try:
            sol = np.linalg.solve(KKT2, rhs)
            p = sol[:n]
        except np.linalg.LinAlgError:
            # Fallback: weighted gradient direction
            p = -Gq.T @ np.ones(m)
    
    # Remove mean
    p = p - np.mean(p)
    
    if float(np.max(np.abs(p))) < 1e-12:
        return None
    
    # Normalize
    p = p / float(np.max(np.abs(p)))
    return p

def line_search(h, p, c_current, trust):

    # Bound: ensure h + alpha*p stays in [0,1]

    alpha_max = max(alpha_max, 1e-12)
    
    # Grid scan

    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.array([exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c = vals[best_i]
    best_a = alphas[best_i]
    
    if best_i in (0, n_samp - 1):
        return best_a, best_c

    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    f1 = exact_c5(h + c1 * p)

            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new / 2.0)

def weighted_grad_dir(h, scores, top_k=None):

    idx = np.argsort(scores)[-top_k:]

    w = np.exp(50.0 * (s_top - np.max(s_top)))
    w /= np.sum(w)

import numpy as np
import time

def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_h(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.where(np.isfinite(v), v, 0.5)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    x[np.abs(x) < 1e-15] = 0.0
    x[np.abs(1.0 - x) < 1e-15] = 1.0
    return x

def build_grad_rows(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    m = len(active_indices)
    G = np.zeros((m, n))
    for k in range(m):
        r = active_indices[k]
        d = r - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[k, d:d + length] += dx * (1.0 - h[:length])
                G[k, :length] -= dx * h[d:d + length]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[k, :length] += dx * (1.0 - h[g:g + length])
                G[k, g:g + length] -= dx * h[:length]
    return G

def roughness_matrix(n, rho):
    if rho == 0.0:
        return np.zeros((n, n))
    H = np.zeros((n, n))
    idx = np.arange(n)
    H[idx, idx] = 2.0 * rho
    if n > 1:
        j = np.arange(n - 1)
        H[j, j + 1] = -rho
        H[j + 1, j] = -rho
    return H

def make_cfg(tol, min_a, max_a, pri, mfac, mflr, ridge, rough):
    return {'active_tol': tol, 'min_active': min_a, 'max_active': max_a,
            'priority': pri, 'margin_factor': mfac, 'margin_floor': mflr,
            'ridge': ridge, 'rough': rough}

def active_direction(h, scores, cfg):
    n = len(h)
    c_max = float(np.max(scores))
    sorted_idx = np.argsort(scores)[::-1]
    active_mask = scores >= (c_max - cfg['active_tol'])
    active_indices = sorted_idx[active_mask]
    if len(active_indices) < cfg['min_active']:
        active_indices = sorted_idx[:cfg['min_active']]
    if len(active_indices) > cfg['max_active']:
        active_indices = sorted_idx[:cfg['max_active']]
    m = len(active_indices)
    if m < 2:
        return None
    s_active = scores[active_indices]
    c_min_a = float(np.min(s_active))
    spread = c_max - c_min_a
    margin = max(cfg['margin_floor'], cfg['margin_factor'] * (spread + cfg['margin_floor']))
    target_a = c_min_a - margin
    gaps = c_max - s_active
    max_gap = max(float(np.max(gaps)), 1e-15)
    q = 1.0 + cfg['priority'] * (gaps / max_gap) ** 2
    G = build_grad_rows(h, active_indices)
    Gq = G * q[:, None]
    H = Gq.T @ G
    H += cfg['ridge'] * np.eye(n)
    if cfg['rough'] > 0:
        H += roughness_matrix(n, cfg['rough'])
    rhs_vec = Gq.T @ (target_a - s_active)
    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n + 1)
    rhs[:n] = rhs_vec
    try:
        sol = np.linalg.solve(KKT, rhs)
        p = sol[:n]
    except np.linalg.LinAlgError:
        H2 = H + 0.1 * np.eye(n)
        KKT2 = np.zeros((n + 1, n + 1))
        KKT2[:n, :n] = H2
        KKT2[:n, n] = 1.0
        KKT2[n, :n] = 1.0
        try:
            sol = np.linalg.solve(KKT2, rhs)
            p = sol[:n]
        except np.linalg.LinAlgError:
            p = -Gq.T @ np.ones(m)
    p = p - np.mean(p)
    pmax = float(np.max(np.abs(p)))
    if pmax < 1e-12:
        return None
    return p / pmax

def weighted_grad_dir(h, scores, top_k=10):
    n = len(h)
    idx = np.argsort(scores)[::-1][:top_k]
    s_top = scores[idx]
    w = np.exp(50.0 * (s_top - np.max(s_top)))
    w /= np.sum(w)
    G = build_grad_rows(h, idx)
    grad = G.T @ w
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return grad / gn

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new / 2.0)

    x = np.arange(n) / n

def fourier_seed(n, k_list, phases, rng):
    x = np.arange(n) / n
    h = np.zeros(n)
    for k, phi in zip(k_list, phases):
        h += np.sin(2.0 * np.pi * k * x + phi)
    h = 0.5 + 0.25 * h / max(1.0, float(np.max(np.abs(h))))
    return project_h(h, n / 2.0)

                best_h = h_wg.copy()

    best_h = project_h(best_h, best_n / 2.0)
    final_c = exact_c5(best_h)
    return best_h, final_c, best_n
