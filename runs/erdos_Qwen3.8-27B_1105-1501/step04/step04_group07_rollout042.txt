```python
import numpy as np
import time

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _exact_c5(h):
    return float(_scores(h).max())

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
def _find_best_alpha(s0, L, Q_pos, Q_neg, alpha_lo, alpha_hi):
    # s0: scores at alpha=0 (length n_lags)
    # L: linear coefficients (length n_lags)
    # Q_pos, Q_neg: the k indices with Q=+1 and Q=-2
    # alpha_lo, alpha_hi: bounds on alpha
    # Returns: (alpha, f_alpha)
    
    def f(a):
        s = s0 + a * L
        # Apply quadratic corrections
        # Q=+1 at two positions, Q=-2 at one position
        s[Q_neg] += a * a * (-2.0)
        for qp in Q_pos:
            s[qp] += a * a * 1.0
        return float(np.max(s))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = alpha_lo
    b = alpha_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(40):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    mid = (a + b) / 2.0
    return mid, f(mid)
```

This is fast because f(a) only requires O(n) operations (add two quadratic corrections to a linear combination).

```python
    lo_a = a_lo
    hi_a = a_hi
    c_a = hi_a - gr * (hi_a - lo_a)
    d_a = lo_a + gr * (hi_a - lo_a)
    fc = _eval_f(raw_s0, L, c_a, Qneg, Qpos1, Qpos2)
    fd = _eval_f(raw_s0, L, d_a, Qneg, Qpos1, Qpos2)
    for _ in range(n_iter):
        if hi_a - lo_a < 1e-11:
            break
        if fc < fd:
            hi_a = d_a
            d_a = c_a
            fd = fc
            c_a = hi_a - gr * (hi_a - lo_a)
            fc = _eval_f(raw_s0, L, c_a, Qneg, Qpos1, Qpos2)
        else:
            lo_a = c_a
            c_a = d_a
            fc = fd
            d_a = lo_a + gr * (hi_a - lo_a)
            fd = _eval_f(raw_s0, L, d_a, Qneg, Qpos1, Qpos2)
    mid = (lo_a + hi_a) / 2.0
    fm = _eval_f(raw_s0, L, mid, Qneg, Qpos1, Qpos2)
    # Also check a few extra points near the minimum
    for test_a in [lo_a, hi_a, lo_a + 0.1*(hi_a-lo_a), hi_a - 0.1*(hi_a-lo_a)]:
        ft = _eval_f(raw_s0, L, test_a, Qneg, Qpos1, Qpos2)
        if ft < fm:
            fm = ft
            mid = test_a
    return mid, fm

def _pair_search(h, deadline, max_sweeps=20000, verbose=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    dx = 2.0 / n
    raw_s0 = _scores_raw(h)
    V = _precompute_V(n, h)
    
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        
        best_imp = 0.0
        best_p = -1
        best_q = -1
        best_a = 0.0
        
        max_k = int(np.argmax(raw_s0))
        
        for p in range(n):
            hp = h[p]
            if hp < 1e-15:
                a_lo_p = 0.0
            else:
                a_lo_p = -hp
            for q in range(p + 1, n):
                a_hi = 1.0 - h[q]
                if a_hi < 1e-15:
                    a_lo = a_lo_p
                    a_hi_t = a_hi
                    if a_lo >= a_hi_t - 1e-12:
                        continue
                else:
                    a_lo = a_lo_p
                    a_hi_t = a_hi
                
                L = V[p] - V[q]
                Lmax = L[max_k]
                
                # Screening: if the dominant lag won't decrease, skip
                if Lmax > 1e-12 and a_lo >= -1e-14:
                    continue
                if Lmax < -1e-12 and a_hi_t <= 1e-14:
                    continue
                if abs(Lmax) < 1e-14 and max_k == n - 1:
                    pass  # quadratic only at max lag, might help
                elif abs(Lmax) < 1e-14 and max_k != n - 1:
                    # Q=0 at max lag, so no change there; check if it helps other lags
                    # Less likely to be the best, but don't skip
                    pass
                
                Qneg = n - 1
                Qpos1 = n - 1 + p - q
                Qpos2 = n - 1 + q - p
                
                alpha, f_val = _gold_min_f(raw_s0, L, a_lo, a_hi_t, Qneg, Qpos1, Qpos2, n_iter=30)
                
                if f_val * dx < cur_c5 - 1e-13:
                    imp = cur_c5 - f_val * dx
                    if imp > best_imp:
                        best_imp = imp
                        best_a = alpha
                        best_p = p
                        best_q = q
        
        if best_p >= 0:
            h[best_p] += best_a
            h[best_q] -= best_a
            h = np.clip(h, 0.0, 1.0)
            # Light projection to fix floating point drift
            s = float(h.sum())
            diff = target - s
            if abs(diff) > 1e-10:
                mask = (h < 1.0) if diff > 0 else (h > 0.0)
                cnt = int(np.sum(mask))
                if cnt > 0:
                    h[mask] += diff / cnt
                    h = np.clip(h, 0.0, 1.0)
            raw_s0 = _scores_raw(h)
            cur_c5 = float(np.max(raw_s0)) * dx
            V = _precompute_V(n, h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _binary_local_search(h_init, deadline, max_iters=200000):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count != target:
        rng = np.random.default_rng(7)
        if ones_count > target:
            idx = np.where(h == 1)[0]
            rng.shuffle(idx)
            h[idx[target:]] = 0
        else:
            idx = np.where(h == 0)[0]
            rng.shuffle(idx)
            h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 8000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_refine(h, deadline, max_iters=3000):
    from scipy.optimize import linprog
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    radius = 0.1
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores_raw(h) * dx
        s_max = float(scores.max())
        # Top active lags
        k_active = max(3, int(n_lags * 0.02))
        top_idx = np.argsort(scores)[-k_active:]
        # Build gradient matrix for active lags
        # Each active lag r has gradient: d(score_r)/d(h_j)
        # score_r = sum_i h[i]*(1-h)[i-d_r] * dx where d_r = r-(n-1)
        # d(score_r)/d(h_j) = dx * [(1-h)[j-d_r] - h[j+d_r] - (delta terms)]
        # Let's just compute numerically for the active set
        G = np.zeros((len(top_idx), n))
        for ii, r in enumerate(top_idx):
            d = r - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j - d < n:
                    g += (1.0 - h[j - d])
                if 0 <= j + d < n and d != 0:
                    g -= h[j + d]
                elif d == 0:
                    g -= h[j]
                G[ii, j] = g
        G *= dx
        # LP: minimize t s.t. G@p <= -t + scores[top], -radius <= p_j <= radius, sum(p)=0
        # Actually: minimize max_r (score_r + G_r @ p)
        # = minimize t s.t. G_r @ p - t <= -score_r for all r
        # With t >= 0, sum(p) = 0, |p_j| <= radius
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(top_idx), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[top_idx]
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((0, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        # Line search along p
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
        lo_a, hi_a = 0.0, alpha_max
        c_a = hi_a - gr_phi * (hi_a - lo_a)
        d_a = lo_a + gr_phi * (hi_a - lo_a)
        fc = _exact_c5(h + c_a * p)
        fd = _exact_c5(h + d_a * p)
        for _ in range(25):
            if hi_a - lo_a < 1e-12:
                break
            if fc < fd:
                hi_a = d_a; d_a = c_a; fd = fc
                c_a = hi_a - gr_phi * (hi_a - lo_a)
                fc = _exact_c5(h + c_a * p)
            else:
                lo_a = c_a; c_a = d_a; fc = fd
                d_a = lo_a + gr_phi * (hi_a - lo_a)
                fd = _exact_c5(h + d_a * p)
        best_a = (lo_a + hi_a) / 2.0
        best_f = _exact_c5(h + best_a * p)
        if best_f < cur_c5 - 1e-12:
            h = h + best_a * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.8
            if stall > 20 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Pair local search (main engine, 40%)
    dl1 = t0 + budget_s * 0.40
    hdr, c5r = _pair_search(h, dl1, max_sweeps=30000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 2: LP refinement (15%)
    dl2 = t0 + budget_s * 0.55
    hdr, c5r = _lp_refine(best_h, dl2, max_iters=2000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 3: More pair search after LP (20%)
    dl3 = t0 + budget_s * 0.75
    hdr, c5r = _pair_search(best_h, dl3, max_sweeps=30000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 4: Resolution sweep (12%)
    dl4 = t0 + budget_s * 0.87
    for dn in [-4, -2, -1, 1, 2, 4, 6, 8, -6, -8]:
        if time.time() > dl4:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.01)
        h_r, c5_r = _pair_search(h_r, sub_dl, max_sweeps=2000)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 5: Random restarts (8%)
    dl5 = t0 + budget_s * 0.95
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        h_try, c5_try = _pair_search(h_pert, sub_dl, max_sweeps=1500)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.1)
    
    # Phase 6: Final pair polish (5%)
    if time.time() < t0 + budget_s * 0.98:
        hdr, c5r = _pair_search(best_h, t0 + budget_s * 0.995, max_sweeps=10000)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hdr.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
