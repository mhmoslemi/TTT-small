    return corr * dx

def _project(v, target):
    # Project onto {v: 0<=v<=1, sum(v)=target}
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_b_coeffs(h, p, q):
    """Compute b_k and d_k for pair direction V[p]=1, V[q]=-1."""
    n = len(h)
    n_lags = 2 * n - 1
    b = np.zeros(n_lags)
    d = np.zeros(n_lags)
    
    # b_k = (1-h[p+d]) - h[p-d] - (1-h[q+d]) + h[q-d]
    # where d = k - (n-1), so k = n-1+d
    # d ranges from -(n-1) to n-1
    
    for d in range(-(n-1), n):
        k = n - 1 + d
        bk = 0.0
        if 0 <= p + d < n:
            bk += (1.0 - h[p + d])
        if 0 <= p - d < n:
            bk -= h[p - d]
        if 0 <= q + d < n:
            bk -= (1.0 - h[q + d])
        if 0 <= q - d < n:
            bk += h[q - d]
        b[k] = bk
        d_k = 0.0
        if d == 0:
            d_k = -2.0
        elif d == q - p:
            d_k = 1.0
        elif d == p - q:
            d_k = 1.0
        d[k] = d_k
    
    return b, d

def _pair_optimal_alpha(h, s, p, q, b, d, n_samples=200):
    """Find alpha that minimizes max_k (s_k + alpha*b_k + alpha^2*d_k) 
    subject to 0 <= h[p]+alpha <= 1 and 0 <= h[q]-alpha <= 1."""
    a_min = max(0.0, -h[q])  # h[q]-alpha >= 0 → alpha <= h[q]
    a_max = min(1.0 - h[p], h[q])  # h[p]+alpha <= 1, h[q]-alpha >= 0
    
    if a_max <= a_min:
        return 0.0, float(s.max())
    
    # Sample
    alphas = np.linspace(a_min, a_max, n_samples)
    best_alpha = alphas[0]
    best_val = float('inf')
    
    for a in alphas:
        new_s = s + a * b + a * a * d
        v = float(new_s.max())
        if v < best_val:
            best_val = v
            best_alpha = a
    
    # Refine with golden section
    lo = 0.0
    hi = 1.0
    if best_alpha == alphas[0]:
        lo, hi = a_min, alphas[1]
    elif best_alpha == alphas[-1]:
        lo, hi = alphas[-2], a_max
    else:
        idx = int(np.argmax(alphas == best_alpha))
        lo = alphas[max(0, idx-1)]
        hi = alphas[min(n_samples-1, idx+1)]
    
    g = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - g * (hi - lo)
    c2 = lo + g * (hi - lo)
    f1 = float((s + c1 * b + c1*c1 * d).max())
    f2 = float((s + c2 * b + c2*c2 * d).max())
    
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - g * (hi - lo)
            f1 = float((s + c1 * b + c1*c1 * d).max())
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + g * (hi - lo)
            f2 = float((s + c2 * b + c2*c2 * d).max())
    
    opt_a = (lo + hi) / 2.0
    opt_v = float((s + opt_a * b + opt_a*opt_a * d).max())
    if opt_v < best_val:
        best_alpha = opt_a
        best_val = opt_v
    
    return best_alpha, best_val

def _pair_descent(h, deadline, max_iters=5000, top_k_pairs=50):
    """Main optimization loop using exact pair line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    s = _scores(h)
    cur_c5 = float(s.max())
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        # Find critical lags
        s_max = float(s.max())
        crit_thresh = s_max * 0.995
        crit_lags = np.where(s >= crit_thresh)[0]
        
        # Generate candidate pairs
        # Strategy: pairs where one has high h and other has low h
        # Focus on pairs that affect critical lags
        rng = np.random.default_rng(it + 42)
        
        # Build a set of candidate pairs
        pairs = []
        
        # Pair with critical lag influence
        # For a critical lag k (shift d=k-(n-1)), pairs (p,q) where
        # p+d, p-d, q+d, q-d are relevant
        # Simple heuristic: try random pairs biased toward high/low h
        order = np.argsort(h)
        low_half = order[:n//2]
        high_half = order[n//2:]
        
        n_candidates = max(top_k_pairs, 20)
        for _ in range(n_candidates):
            p = int(high_half[rng.integers(len(high_half))])
            q = int(low_half[rng.integers(len(low_half))])
            if p != q:
                pairs.append((p, q))
        
        # Also add some structured pairs
        for p in range(n):
            for q in range(p+1, min(p+5, n)):
                if h[p] > 0.5 and h[q] < 0.5:
                    pairs.append((p, q))
        
        # Try each candidate pair
        best_alpha = 0.0
        best_new_s = s.max()
        best_pair = None
        
        for (p, q) in pairs:
            if h[p] <= h[q]:
                continue  # We want to move from high to low
            b, d = _pair_b_coeffs(h, p, q)
            alpha, new_max = _pair_optimal_alpha(h, s, p, q, b, d)
            if alpha > 1e-10 and new_max < best_new_s:
                best_new_s = new_max
                best_alpha = alpha
                best_pair = (p, q)
        
        if best_pair is None:
            # Try all pairs with alpha < 0 (reverse direction)
            for (p, q) in pairs:
                if h[p] >= h[q]:
                    continue
                # Swap: V[p]=-1, V[q]=1
                b, d = _pair_b_coeffs(h, q, p)  # This gives V[q]=1, V[p]=-1
                # Actually we need V[p]=-1, V[q]=1, so swap p and q in the call
                b, d = _pair_b_coeffs(h, p, q)  # V[p]=1, V[q]=-1
                # For the reverse, we negate:
                b = -b
                d = d  # d doesn't change sign (it's -VV which is symmetric)
                # Wait, if V → -V, then b → -b and d (-VV) stays the same
                alpha, new_max = _pair_optimal_alpha(h, s, q, p, -b, d)
                # Actually let me just call with swapped p,q
                if alpha > 1e-10 and new_max < best_new_s:
                    best_new_s = new_max
                    best_alpha = alpha
                    best_pair = (q, p)  # means V[q]=1, V[p]=-1
            
        if best_pair is None:
            stall += 1
            continue
        
        # Apply the step
        p, q = best_pair
        h[p] += best_alpha
        h[q] -= best_alpha
        h = np.clip(h, 0.0, 1.0)
        h = _project(h, target)
        
        s = _scores(h)
        cur_c5 = float(s.max())
        
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

Wait, I realize there's an issue with my approach. When I do the pair transfer and then project, the projection might undo some of the improvement. Let me instead do the exact computation: after the step h[p] += α, h[q] -= α, verify that the constraint sum(h) = n/2 is still satisfied (it should be, since we're just transferring mass). The only constraint we need to check is 0 ≤ h ≤ 1, which is handled by the α bounds.

So actually, we don't need to re-project! The pair transfer automatically preserves the sum. We just need to make sure 0 ≤ h[p]+α ≤ 1 and 0 ≤ h[q]-α ≤ 1, which is what the α bounds ensure.

Let me also think about efficiency. For n=64, we have 127 lags. Computing b and d for a pair requires O(n) work (iterating over all d values). Then finding the optimal α requires O(n * n_samples) work for the initial sampling, plus O(n * 30) for the golden section refinement. So per pair: O(n * n_samples + n * 30) ≈ O(n * 230). For 50 pairs per iteration: O(n * 50 * 230) = O(64 * 11500) ≈ 736,000 operations. That's very fast.

We could afford to try more pairs per iteration, say 200-500.

Let me also think about a "multi-pair" direction. Instead of just one pair at a time, we could try a direction that's a combination of multiple pairs. But that makes the 1D optimization more complex (the quadratic term becomes denser). Let me stick with single pairs for now.

Another idea: **momentum**. Keep track of the recent direction and add it to the new direction.

Let me also think about what happens at convergence. The algorithm stops when no single pair transfer can improve the objective. This is a "pairwise" local optimum. It's stronger than a coordinate-wise local optimum but weaker than a global optimum.

To escape local optima, I can:
1. Add random perturbations
2. Try different resolutions
3. Use a different initialization

The current solution uses n=64. Can we do better with a smaller n? Let's think about what's achievable.
