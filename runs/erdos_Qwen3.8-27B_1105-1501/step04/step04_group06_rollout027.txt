    return float(np.max(corr) * 2.0 / n)

def _full_grad(h, n):
    nc = 2 * n - 1
    G = np.zeros((nc, n))
    j = np.arange(n)
    for r in range(nc):
        s = r - (n - 1)
        ps = j + s
        m1 = (ps >= 0) & (ps < n)
        G[r, m1] += (1.0 - h[ps[m1]])
        gs = j - s
        m2 = (gs >= 0) & (gs < n)
        G[r, m2] -= h[gs[m2]]
    return G

def _lp_step(h, n, radius=0.1):

    M = float(corr.max())
    G = _full_grad(h, n)

    c = np.zeros(n + 1)

    A_ub[:, :n] = G
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return h.copy(), M
    p = res.x[:n]
    if np.max(np.abs(p)) < 1e-14:
        return h.copy(), M

    alpha_max = radius  # since p is already bounded by radius
    # Actually p is the direction, we need to think about this differently
    # The LP already incorporates the trust region, so p itself is a step
    # But we should verify that h+p is feasible
    
    # Check feasibility of h + p

    if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:

        for i in range(n):
            if p[i] > 0 and h[i] + p[i] > 1:
                scale = min(scale, (1.0 - h[i]) / p[i])

        if c5_t < best_c5:

def _lp_loop(h, n, deadline, radius=0.1, max_iters=5000, stall_limit=100):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius

        else:
            stall += 1
            r *= 0.8

def _slsqp_solve(h, n, maxiter=300, ftol=1e-13):
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(corr.max()) + 1e-4
    
    def obj(z):
        return z[n]
    def obj_g(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def ineq_f(z):
        return z[n] - np.correlate(z[:n], 1.0 - z[:n], mode='full')
    def ineq_g(z):
        G = _full_grad(z[:n], n)
        J = np.zeros((2*n-1, n+1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J
    def eq_f(z):
        return np.sum(z[:n]) - n * 0.5
    def eq_g(z):
        g = np.ones(n + 1)
        g[n] = 0.0
        return g
    
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, n)]
    cons = [{'type': 'eq', 'fun': eq_f, 'jac': eq_g},
            {'type': 'ineq', 'fun': ineq_f, 'jac': ineq_g}]
    
    try:
        res = minimize(obj, z0, jac=obj_g, bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except:
        return h.copy(), _exact_c5(h)

def _lse_step(h, n, alpha, maxiter=100):
    def lse_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        ms = corr.max()
        vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
        exp_v = np.exp(vals)
        s = exp_v.sum()
        f = ms + (1.0/alpha) * np.log(s)
        w = exp_v / s
        grad = np.zeros(n)
        j = np.arange(n)
        for r in range(2*n-1):
            sc = r - (n-1)
            ps = j + sc
            m1 = (ps >= 0) & (ps < n)
            grad[m1] += w[r] * (1.0 - x[ps[m1]])
            gs = j - sc
            m2 = (gs >= 0) & (gs < n)
            grad[m2] -= w[r] * x[gs[m2]]
        grad -= grad.mean()
        return f, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except:
        return h.copy(), _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _binarize(h, n):
    """Try to push h towards binary values."""
    target = n * 0.5
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Adjust to meet constraint
    s = h_bin.sum()
    if s > target:
        # Set some 1s to 0s (those closest to 0.5 first)
        ones = np.where(h_bin == 1.0)[0]
        ones = ones[np.argsort(h[ones])]  # ones with smallest h values
        k = int(s - target)
        for i in range(min(k, len(ones))):
            h_bin[ones[i]] = 0.0
    elif s < target:
        zeros = np.where(h_bin == 0.0)[0]
        zeros = zeros[np.argsort(-h[zeros])]  # zeros with largest h values
        k = int(target - s)
        for i in range(min(k, len(zeros))):
            h_bin[zeros[i]] = 1.0
    return h_bin, _exact_c5(h_bin)

def _bit_flip_search(h_bin, n, max_iters=5000, deadline=None):
    """Try single bit flips (pair swaps) to improve."""
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Try all (one, zero) pairs - swap them
        one_list = list(ones)
        zero_list = list(zeros)
        for oi in one_list:
            for zi in zero_list:
                h[oi] = 0.0
                h[zi] = 1.0
                c5_try = _exact_c5(h)
                h[oi] = 1.0
                h[zi] = 0.0
                if c5_try < best_c5 - 1e-14:
                    h[oi] = 0.0
                    h[zi] = 1.0
                    ones.discard(oi)
                    ones.add(zi)
                    zeros.discard(zi)
                    zeros.add(oi)
                    best_h = h.copy()
                    best_c5 = c5_try
                    cur_c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    x = np.clip(v - mid, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)


def _full_grad(h, n):
    nc = 2 * n - 1
    G = np.zeros((nc, n))
    j = np.arange(n)
    for r in range(nc):
        s = r - (n - 1)
        ps = j + s
        m1 = (ps >= 0) & (ps < n)
        G[r, m1] += (1.0 - h[ps[m1]])
        gs = j - s
        m2 = (gs >= 0) & (gs < n)
        G[r, m2] -= h[gs[m2]]
    return G


def _epi_obj(z, n):
    return z[n]


def _epi_obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _epi_ineq_f(z, n):
    return z[n] - np.correlate(z[:n], 1.0 - z[:n], mode='full')


def _epi_ineq_g(z, n):
    G = _full_grad(z[:n], n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def _epi_eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _epi_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _slsqp_solve(h, n, maxiter=300, ftol=1e-13):
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(corr.max()) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'eq', 'fun': _epi_eq_f, 'jac': _epi_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _epi_ineq_f, 'jac': _epi_ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_obj_grad, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _lse_obj(x, n, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    ms = float(corr.max())
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    exp_v = np.exp(vals)
    s = float(exp_v.sum())
    f = ms + (1.0 / alpha) * np.log(s)
    w = exp_v / s
    grad = np.zeros(n)
    j = np.arange(n)
    for r in range(2 * n - 1):
        sc = r - (n - 1)
        ps = j + sc
        m1 = (ps >= 0) & (ps < n)
        grad[m1] += w[r] * (1.0 - x[ps[m1]])
        gs = j - sc
        m2 = (gs >= 0) & (gs < n)
        grad[m2] -= w[r] * x[gs[m2]]
    grad -= grad.mean()
    return f, grad


def _lse_step(h, n, alpha, maxiter=80):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 50, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _lp_step(h, n, radius):
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = float(corr.max())
    G = _full_grad(h, n)
    nc = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((nc, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - corr
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h.copy(), M
    if not res.success or res.x is None:
        return h.copy(), M
    p = res.x[:n]
    if np.max(np.abs(p)) < 1e-14:
        return h.copy(), M
    h_try = h + p
    scale = 1.0
    for i in range(n):
        if p[i] > 1e-14 and h_try[i] > 1.0:
            scale = min(scale, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-14 and h_try[i] < 0.0:
            scale = min(scale, -h[i] / p[i])
    p = p * min(scale, 1.0) * 0.99
    best_h = h.copy()
    best_c5 = M
    for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
        h_t = _project(h + t * p, n * 0.5)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5


def _lp_loop(h, n, deadline, radius=0.1, max_iters=5000, stall_limit=80):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        h, new_c5 = _lp_step(h, n, r)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.03, 0.4)
        else:
            stall += 1
            r *= 0.85
            if r < 0.003:
                break
            if stall % 15 == 0:
                r = 0.06
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _binarize(h, n):
    target = n * 0.5
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    s = h_bin.sum()
    if s > target:
        ones = np.where(h_bin == 1.0)[0]
        ones = ones[np.argsort(h[ones])]
        k = int(s - target)
        for i in range(min(k, len(ones))):
            h_bin[ones[i]] = 0.0
    elif s < target:
        zeros = np.where(h_bin == 0.0)[0]
        zeros = zeros[np.argsort(-h[zeros])]
        k = int(target - s)
        for i in range(min(k, len(zeros))):
            h_bin[zeros[i]] = 1.0
    return h_bin, _exact_c5(h_bin)


def _bit_flip_search(h_bin, n, max_iters=3000, deadline=None):
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for _ in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                c5_try = _exact_c5(h)
                if c5_try < best_c5 - 1e-14:
                    ones.remove(oi)
                    ones.append(zi)
                    zeros.remove(zi)
                    zeros.append(oi)
                    best_h = h.copy()
                    best_c5 = c5_try
                    improved = True
                    break
                h[oi] = 1.0
                h[zi] = 0.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def _optimize_n(h_start, n, deadline):
    target = n * 0.5
    if len(h_start) == n:
        h = _project(h_start, target)
    else:
        h = _resample(h_start, n)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    d1 = min(deadline, time.time() + 20)
    for alpha in [30.0, 80.0, 150.0]:
        if time.time() > d1:
            break
        h, c5 = _lse_step(h, n, alpha, maxiter=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    d2 = min(deadline, time.time() + 50)
    h_s, c5_s = _slsqp_solve(best_h, n, maxiter=300, ftol=1e-14)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()
    if time.time() < d2:
        h_s2, c5_s2 = _slsqp_solve(best_h, n, maxiter=200, ftol=1e-15)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()

    d3 = min(deadline, time.time() + 100)
    h_lp, c5_lp = _lp_loop(best_h, n, d3, radius=0.1, max_iters=5000, stall_limit=80)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    if time.time() < deadline:
        h_bin, c5_bin = _binarize(best_h, n)
        if c5_bin < best_c5 + 0.02:
            d_bin = min(deadline, time.time() + 40)
            h_b2, c5_b2 = _bit_flip_search(h_bin, n, max_iters=2000, deadline=d_bin)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
            h_u, c5_u = _slsqp_solve(h_b2, n, maxiter=200, ftol=1e-13)
            if c5_u < best_c5:
                best_c5 = c5_u
                best_h = h_u.copy()
            if time.time() < deadline:
                h_u2, c5_u2 = _lp_loop(best_h, n, min(deadline, time.time() + 30), radius=0.05)
                if c5_u2 < best_c5:
                    best_c5 = c5_u2
                    best_h = h_u2.copy()

    if time.time() < deadline:
        h_f, c5_f = _slsqp_solve(best_h, n, maxiter=200, ftol=1e-15)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        h_f2, c5_f2 = _lp_loop(best_h, n, deadline, radius=0.03, max_iters=3000, stall_limit=60)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    return best_h, best_c5


def _multi_slsqp(h, n, rng, deadline, max_rounds=5):
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.03
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, n * 0.5)
        h_s, c5_s = _slsqp_solve(h_p, n, maxiter=200, ftol=1e-14)
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_s.copy()
            scale = max(0.005, scale * 0.7)
        else:
            scale = min(0.2, scale * 1.3)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)

    # Phase 1: Optimize the parent construction (12% budget)
    d1 = t0 + budget_s * 0.12
    best_h, best_c5, best_n = _optimize_n(h0, n0, d1)

    # Phase 2: Explore nearby n values (45% budget)
    d2 = t0 + budget_s * 0.57
    n_candidates = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 16, -10, -12, 20]:
        nn = n0 + dn
        if nn % 2 == 0 and 32 <= nn <= 160:
            n_candidates.append(nn)
    # Remove duplicates, sort by distance
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n0))
    remaining = max(d2 - time.time(), 60)
    per_n = remaining / max(len(n_candidates), 1)
    for nn in n_candidates:
        if time.time() > d2:
            break
        sub_dead = min(d2, time.time() + per_n)
        h_try, c5_try = _optimize_n(best_h, nn, sub_dead)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = nn

    # Phase 3: Multi-start SLSQP + LP from perturbed (20% budget)
    d3 = t0 + budget_s * 0.77
    best_h, best_c5 = _multi_slsqp(best_h, best_n, rng, d3, max_rounds=30)
    h_lp, c5_lp = _lp_loop(best_h, best_n, min(d3, time.time() + 60),
                           radius=0.08, max_iters=3000, stall_limit=60)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 4: Try structural variations (15% budget)
    d4 = t0 + budget_s * 0.92
    # Reversed
    h_rev = best_h[::-1].copy()
    h_rev = _project(h_rev, best_n * 0.5)
    h_q, c5_q = _slsqp_solve(h_rev, best_n, maxiter=200, ftol=1e-13)
    if c5_q < best_c5:
        best_c5 = c5_q
        best_h = h_q.copy()
    # Random block swap
    for _ in range(5):
        if time.time() > d4:
            break
        if best_n < 16:
            break
        bl = best_n // 4
        seg = np.random.default_rng(int(rng.integers(0, 2**31))).permutation(best_n)
        h_swap = best_h[seg].copy()
        h_swap = _project(h_swap, best_n * 0.5)
        h_q2, c5_q2 = _slsqp_solve(h_swap, best_n, maxiter=150, ftol=1e-13)
        if c5_q2 < best_c5:
            best_c5 = c5_q2
            best_h = h_q2.copy()
    # Binary search variant
    if time.time() < d4:
        h_bin, c5_bin = _binarize(best_h, best_n)
        h_b2, c5_b2 = _bit_flip_search(h_bin, best_n, max_iters=3000,
                                        deadline=min(d4, time.time() + 60))
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
        h_u, c5_u = _slsqp_solve(best_h, best_n, maxiter=200, ftol=1e-14)
        if c5_u < best_c5:
            best_c5 = c5_u
            best_h = h_u.copy()

    # Phase 5: Final polish (8% budget)
    d5 = t0 + budget_s * 0.99
    h_f, c5_f = _slsqp_solve(best_h, best_n, maxiter=400, ftol=1e-15)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    h_f2, c5_f2 = _lp_loop(best_h, best_n, d5, radius=0.04,
                           max_iters=5000, stall_limit=100)
    if c5_f2 < best_c5:
        best_c5 = c5_f2
        best_h = h_f2.copy()

    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
