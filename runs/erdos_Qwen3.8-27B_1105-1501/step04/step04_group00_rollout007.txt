        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def build_active_gradients(h, active_idx):
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    G = np.zeros((m, n))
    for k in range(m):
        r = active_idx[k]
        d = r - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                G[k, d:d+m2] += (1.0 - h[:m2])
                G[k, :m2] -= h[d:d+m2]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                G[k, :m2] += (1.0 - h[g:g+m2])
                G[k, g:g+m2] -= h[:m2]
    G *= dx
    return G

def roughness_mat(n, rho):
    if rho == 0:
        return np.zeros((n, n))
    H = np.zeros((n, n))
    for i in range(n - 1):
        H[i, i] += rho
        H[i+1, i+1] += rho
        H[i, i+1] -= rho
        H[i+1, i] -= rho
    return H

def logsumexp_grad(h, beta):
    scores = scores_exact(h)
    n = len(h)
    dx = 2.0 / n
    shifted = scores - np.max(scores)
    w = np.exp(beta * shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                grad[d:d+m2] += w[r] * dx * (1.0 - h[:m2])
                grad[:m2] += w[r] * (-dx) * h[d:d+m2]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                grad[:m2] += w[r] * dx * (1.0 - h[g:g+m2])
                grad[g:g+m2] += w[r] * (-dx) * h[:m2]
    return grad

def qp_direction(h, scores, active_idx, priority, margin_f, margin_floor, ridge, rough):
    n = len(h)
    s_act = scores[active_idx]
    c_max = float(np.max(s_act))
    c_min = float(np.min(s_act))
    spread = max(c_max - c_min, 1e-15)
    sigma = max(margin_floor, min(3e-4, margin_f * spread))
    target_level = c_min - sigma
    gaps = c_max - s_act
    max_gap = max(float(np.max(gaps)), 1e-15)
    q = 1.0 + priority * (gaps / max_gap) ** 2
    q = np.maximum(q, 1e-6)
    r_vec = target_level - s_act
    G = build_active_gradients(h, active_idx)
    if not np.all(np.isfinite(G)):
        return None
    Gq = G * q[:, None]
    H = Gq.T @ G
    H += ridge * np.eye(n)
    if rough > 0:
        H += roughness_mat(n, rough)
    rhs = G.T @ (q * r_vec)
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs_full = np.zeros(n+1)
    rhs_full[:n] = rhs
    try:
        sol = np.linalg.solve(KKT, rhs_full)
        p = sol[:n]
        if not np.all(np.isfinite(p)):
            raise ValueError
    except Exception:
        p = -(G.T @ q)
        p -= p.mean()
    p -= p.mean()
    if float(np.max(np.abs(p))) < 1e-14:
        return None
    return p

def line_search(h, p, cur_c5, trust=0.3):
    n = len(h)
    p = p - p.mean()
    pmax = float(np.max(np.abs(p)))
    if pmax < 1e-14:
        return 0.0, cur_c5
    p_unit = p / pmax
    alpha_max = min(trust, 0.5)
    for j in range(n):
        if p_unit[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p_unit[j])
        elif p_unit[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p_unit[j]))
    alpha_max = max(alpha_max, 1e-10)
    alpha_max = min(alpha_max, 1.0)
    if alpha_max < 1e-12:
        return 0.0, cur_c5
    n_grid = 14
    alphas = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, np.inf)
    for i in range(n_grid):
        h_try = h + alphas[i] * p_unit
        if h_try.min() >= -1e-10 and h_try.max() <= 1.0 + 1e-10:
            vals[i] = exact_c5(h_try)
    best_idx = int(np.argmin(vals))
    best_a = alphas[best_idx]
    best_val = vals[best_idx]
    if best_val >= cur_c5:
        return 0.0, cur_c5
    if 0 < best_idx < n_grid - 1:
        lo = alphas[best_idx - 1]
        hi = alphas[best_idx + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = exact_c5(h + c1 * p_unit)
        f2 = exact_c5(h + c2 * p_unit)
        for _ in range(22):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = exact_c5(h + c1 * p_unit)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = exact_c5(h + c2 * p_unit)
        opt_a = (lo + hi) / 2.0
        opt_val = exact_c5(h + opt_a * p_unit)
        if opt_val < best_val:
            best_a = opt_a
            best_val = opt_val
    return best_a, best_val

def local_opt(h, deadline, use_lse=False, beta_lse=40.0):
    n = len(h)
    target = n / 2.0
    h = project(h.copy(), target)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    trust_scale = 1.0
    variants = [
        (3e-4, 12, 48, 100.0, 0.25, 2e-5, 5e-3, 3e-3, 0.25),
        (1e-3, 16, 64, 40.0, 0.35, 3e-5, 8e-3, 5e-3, 0.22),
        (1e-4, 8, 28, 500.0, 0.15, 1e-5, 2e-3, 1e-3, 0.15),
        (5e-5, 6, 20, 2000.0, 0.08, 8e-6, 8e-4, 0.0, 0.10),
        (7e-5, 10, 36, 250.0, 0.12, 1.5e-5, 3e-3, 2e-3, 0.18),
    ]
    n_var = len(variants)
    max_iters = 3000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        if use_lse and it % 3 == 2:
            grad = logsumexp_grad(h, beta_lse)
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                stall += 1
                continue
            p = -grad / gn
            a, new_c5 = line_search(h, p, cur_c5, trust=0.12 * trust_scale)
            if new_c5 < cur_c5 - 1e-12:
                h = project(h + a * p, target)
                cur_c5 = exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
            continue
        v = variants[it % n_var]
        atol, min_a, max_a, pri, mf, mfl, ridge, rough, trust = v
        scores = scores_exact(h)
        c_max = float(np.max(scores))
        sorted_idx = np.argsort(scores)[::-1]
        active_mask = scores >= (c_max - atol)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < min_a:
            active_idx = sorted_idx[:min(min_a, 2*n-1)]
        if len(active_idx) > max_a:
            active_idx = active_idx[:max_a]
        p = qp_direction(h, scores, active_idx, pri, mf, mfl, ridge, rough)
        if p is None:
            stall += 1
            trust_scale *= 0.8
            continue
        a, new_c5 = line_search(h, p, cur_c5, trust=trust * trust_scale)
        if new_c5 < cur_c5 - 1e-12:
            h = project(h + a * p, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            trust_scale = min(trust_scale * 1.04, 1.2)
        else:
            stall += 1
            trust_scale *= 0.88
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def smooth_perturb(h, scale, rng, n_modes=4):
    n = len(h)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        amp = scale / k
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(2 * np.pi * k * x + phase)
        phase2 = rng.uniform(0, 2 * np.pi)
        pert += amp * np.sin(2 * np.pi * k * x + phase2)
    return project(h + pert, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.98

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = project(h0.copy(), n0 / 2.0)
    c5_start = exact_c5(h_start)

    best_h = h_start.copy()
    best_c5 = c5_start
    best_n = n0

    if best_c5 > 0.381:
        pass

    # Phase 1: Intensive local optimization on parent
    dl1 = t0 + budget_s * 0.15
    h1, c5_1 = local_opt(h_start, dl1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Second pass with LSE
    if time.time() < t0 + budget_s * 0.20:
        dl1b = min(t0 + budget_s * 0.22, hard_deadline)
        h1b, c5_1b = local_opt(best_h, dl1b, use_lse=True, beta_lse=50.0)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Multi-resolution
    dl2 = t0 + budget_s * 0.55
    resolutions = [96, 112, 128, 144, 160, 104, 120, 136]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl2:
            break
        if n_try > 256:
            continue
        sub_dl = min(dl2, time.time() + budget_s * 0.04)
        if ri == 0:
            h_init = resample(best_h, n_try)
        else:
            h_init = resample(best_h, n_try)
            h_init = smooth_perturb(h_init, 0.03, rng, n_modes=3)
        h_r, c5_r = local_opt(h_init, sub_dl)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
        # LSE variant on same
        if time.time() < dl2:
            sub_dl2 = min(dl2, time.time() + budget_s * 0.025)
            h_r2, c5_r2 = local_opt(h_r, sub_dl2, use_lse=True, beta_lse=35.0)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = n_try

    # Phase 3: Diverse starts at best_n
    dl3 = t0 + budget_s * 0.75
    for attempt in range(30):
        if time.time() > dl3:
            break
        scale = 0.02 + 0.03 * (attempt % 5)
        h_pert = smooth_perturb(best_h, scale, rng, n_modes=3 + attempt % 3)
        sub_dl = min(dl3, time.time() + budget_s * 0.025)
        h_try, c5_try = local_opt(h_pert, sub_dl)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: LP-based direction (different from QP) for diversity
    dl4 = t0 + budget_s * 0.85
    if time.time() < dl4:
        h_lp_start = best_h.copy()
        sub_dl = min(dl4, time.time() + budget_s * 0.05)
        n = len(h_lp_start)
        target = n / 2.0
        for r_lag in [None, 30, 15, 8]:
            if time.time() > sub_dl:
                break
            scores = scores_exact(h_lp_start)
            c_max = float(np.max(scores))
            if r_lag is None:
                top_idx = np.argsort(scores)[::-1][:50]
            else:
                top_idx = np.argsort(scores)[::-1][:r_lag]
            G = build_active_gradients(h_lp_start, top_idx)
            scores_k = scores[top_idx]
            # LP: minimize t s.t. G p - t <= 0 (decrease all), ||p||_something
            # Simpler: minimize -p[0] (arbitrary) s.t. G p <= 0, sum p = 0
            # Actually let's just use: min sum(p^2) s.t. G p <= -0.005*scores_k, sum p = 0
            # Too complex. Use a simple approach:
            # Find direction that decreases all top constraints:
            # p = -G^T G p => solve (G^T G + I) p = -G^T G * ones
            # No, just use weighted gradient:
            w = np.exp(30.0 * (scores_k - c_max))
            w /= w.sum()
            grad = G.T @ w
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn > 1e-14:
                p = -grad / gn
                a, new_c5 = line_search(h_lp_start, p, exact_c5(h_lp_start), trust=0.15)
                if new_c5 < best_c5:
                    h_new = project(h_lp_start + a * p, target)
                    c5_new = exact_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    h_lp_start = h_new

    # Phase 5: Final polish
    dl5 = t0 + budget_s * 0.975
    h_fin, c5_fin = local_opt(best_h, dl5)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()

    # Final project and evaluate
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
