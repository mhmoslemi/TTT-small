import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {sum=target, 0<=x<=1} via bisection on shift."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    return np.clip(v - mu, 0.0, 1.0)

def _pair_line_optimize(h, i, j, n_samples=12):
    """
    Given feasible h, find best t in feasible range for pair (i,j).
    h[i]=t, h[j]=s-t where s=h[i]+h[j].
    Returns (best_h, best_c5) or (h, c5) if no improvement.
    """
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-12:
        return h, _exact_c5(h)
    
    o_i, o_j = h[i], h[j]
    best_t = t_lo + (t_hi - t_lo) * 0.5  # initial: keep as is (t = o_i)
    best_t = o_i  # actually start at current
    best_c5 = _exact_c5(h)
    
    # Sample t values
    ts = np.linspace(t_lo, t_hi, n_samples)
    for t in ts:
        h[i], h[j] = t, s - t
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_t = t
    h[i], h[j] = best_t, s - best_t
    return h, best_c5

def _pair_sweep(h, max_passes=50, deadline=None, time_per_pass=None):
    """Repeated pair sweep for local optimization."""
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    rng = np.random.default_rng(12345)
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        # Shuffle pairs for diversity
        if p > 0:
            rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            if deadline is not None and time.time() > deadline:
                break
            h, c5 = _pair_line_optimize(h, i, j)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                improved = True
        if not improved:
            break
    return h, best_c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum
    residual = target - p.sum()
    if abs(residual) > 1e-10:
        # Distribute residual
        slack = 1.0 - p if residual > 0 else p
        order = np.argsort(slack)[::-1] if residual > 0 else np.argsort(slack)
        per = residual / len(order)
        for idx in order:
            adj = min(abs(residual), abs(per))
            if residual > 0:
                p[idx] = min(1.0, p[idx] + residual)
            else:
                p[idx] = max(0.0, p[idx] + residual)
            residual = target - p.sum()
            if abs(residual) < 1e-12:
                break
    return p

def _mask_to_twolevel(mask, d):
    return np.where(mask == 1, 1.0 - d, d)

def _optimize_d(mask, n_grid=50):
    n = len(mask)
    ds = np.linspace(0.0, 0.5, n_grid)
    best_d = 0.0
    best_c5 = float('inf')
    for d in ds:
        h = _mask_to_twolevel(mask, d)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_d = d
    # Refine
    if 0 < best_d < 0.5:
        span = 0.5 / n_grid
        ds2 = np.linspace(max(0, best_d - span), min(0.5, best_d + span), 30)
        for d in ds2:
            h = _mask_to_twolevel(mask, d)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_d = d
    h = _mask_to_twolevel(mask, best_d)
    return h, best_c5

def _pair_sweep(h, max_passes=50, deadline=None, seed=42, n_t=14):
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Generate pair order
        if p == 0:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
        
        for (i, j) in pairs:
            if deadline is not None and time.time() > deadline:
                break
            s = h[i] + h[j]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-10:
                continue
            o_i = h[i]
            ts = np.linspace(t_lo, t_hi, n_t)
            best_t = o_i
            local_best = best_c5
            for t in ts:
                h[i] = t
                h[j] = s - t
                c5 = _exact_c5(h)
                if c5 < local_best:
                    local_best = c5
                    best_t = t
            h[i] = best_t
            h[j] = s - best_t
            if local_best < best_c5 - 1e-13:
                best_c5 = local_best
                improved = True
        if not improved:
            break
    return h.copy(), best_c5

def _metropolis_mask(mask, n_swaps, T_start, T_end, seed=42):
    """
    Metropolis search on balanced binary masks.
    Returns best mask found.
    """
    n = len(mask)
    m = int(mask.sum())
    rng = np.random.default_rng(seed)
    
    ones = np.where(mask == 1)[0]
    zeros = np.where(mask == 0)[0]
    
    best_mask = mask.copy()
    best_c5 = _exact_c5(mask.astype(float))
    cur_c5 = best_c5
    
    # Temperature schedule
    T = T_start
    
    for step in range(n_swaps):
        T = T_start * (T_end / T_start) ** (step / max(n_swaps, 1))
        
        # Pick random 1 and random 0 to swap
        oi = rng.integers(len(ones))
        zi = rng.integers(len(zeros))
        i = ones[oi]
        j = zeros[zi]
        
        # Perform swap
        mask[i] = 0
        mask[j] = 1
        ones[oi] = j
        zeros[zi] = i
        
        new_c5 = _exact_c5(mask.astype(float))
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_mask = mask.copy()
        else:
            # Revert
            mask[j] = 0
            mask[i] = 1
            ones[oi] = i
            zeros[zi] = j
    
    return best_mask, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
import time
from scipy.optimize import minimize


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - p.sum()
    if abs(residual) > 1e-12:
        slack = (1.0 - p) if residual > 0 else p.copy()
        order = np.argsort(slack)[::-1] if residual > 0 else np.argsort(slack)
        for idx in order:
            if residual > 1e-14:
                adj = min(residual, 1.0 - p[idx])
                p[idx] += adj
                residual -= adj
            elif residual < -1e-14:
                adj = max(residual, -p[idx])
                p[idx] += adj
                residual -= adj
            else:
                break
    return p


def _two_level_vec(mask, d):
    return np.where(mask == 1, 1.0 - d, d)


def _optimize_d(mask, n_grid=50):
    n = len(mask)
    best_d = 0.0
    best_c5 = float('inf')
    ds = np.linspace(0.0, 0.5, n_grid)
    for d in ds:
        h = _two_level_vec(mask, d)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_d = d
    if 0 < best_d < 0.5:
        span = 0.5 / n_grid
        ds2 = np.linspace(max(0.0, best_d - span), min(0.5, best_d + span), 30)
        for d in ds2:
            h = _two_level_vec(mask, d)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_d = d
    h = _two_level_vec(mask, best_d)
    return h, best_c5


def _g_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _g_jac(h):
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        d = j - (n - 1)
        if d >= 0:
            if d < n:
                m1 = np.arange(d, n)
                J[j, m1] += (1.0 - h[m1 - d])
        else:
            dp = -d
            m2 = np.arange(0, n - dp)
            J[j, m2] += (1.0 - h[m2 + dp])
            if dp < n:
                m1 = np.arange(dp, n)
                J[j, m1] -= h[m1 - dp]
    return J


def _get_active(g_full, frac=0.4, min_count=8):
    max_g = float(np.max(g_full))
    if max_g <= 0:
        return np.arange(len(g_full))
    threshold = max_g * frac
    active = np.where(g_full >= threshold)[0]
    if len(active) < min_count:
        k = min(min_count * 2, len(g_full))
        active = np.argsort(g_full)[-k:]
    return active


def _slsqp_obj(x, n):
    return x[0]


def _slsqp_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[0] = 1.0
    return g


def _slsqp_ineq(x, n, active):
    h = x[1:]
    g = _g_vec(h)
    return x[0] - g[active]


def _slsqp_ineq_jac(x, n, active):
    h = x[1:]
    Jg = _g_jac(h)
    J = np.zeros((len(active), n + 1))
    J[:, 0] = 1.0
    J[:, 1:] = -Jg[active, :]
    return J


def _slsqp_eq(x, n):
    return x[1:].sum() - n / 2.0


def _slsqp_eq_jac(x, n):
    g = np.zeros(n + 1)
    g[1:] = 1.0
    return g


def _slsqp_solve(h, active_indices, maxiter=300, ftol=1e-13):
    n = len(h)
    target = n / 2.0
    g_full = _g_vec(h)
    max_g = float(np.max(g_full))
    x0 = np.zeros(n + 1)
    x0[0] = max_g * 1.005 + 0.05
    x0[1:] = h
    constraints = [
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac, 'args': (n, active_indices)},
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}
    ]
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, x0, jac=_slsqp_obj_grad, args=(n,),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = np.clip(res.x[1:], 0.0, 1.0)
            h_new = _project(h_new, target)
            return h_new
    except Exception:
        pass
    return h


def _pair_sweep(h, max_passes=50, deadline=None, seed=42, n_t=14):
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        if p > 0:
            rng.shuffle(pairs)
        for (i, j) in pairs:
            if deadline is not None and time.time() > deadline:
                break
            s = h[i] + h[j]
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-10:
                continue
            o_i = h[i]
            ts = np.linspace(t_lo, t_hi, n_t)
            best_t = o_i
            local_best = best_c5
            for t in ts:
                h[i] = t
                h[j] = s - t
                c5 = _exact_c5(h)
                if c5 < local_best:
                    local_best = c5
                    best_t = t
            h[i] = best_t
            h[j] = s - best_t
            if local_best < best_c5 - 1e-13:
                best_c5 = local_best
                improved = True
        if not improved:
            break
    return h.copy(), best_c5


def _metropolis_binary(n, m, n_steps, T_start, T_end, seed, top_k=50):
    """Metropolis on balanced binary masks. Tracks top-k by binary C5."""
    rng = np.random.default_rng(seed)
    mask = np.zeros(n)
    mask[:m] = 1.0
    rng.shuffle(mask)

    ones = np.where(mask == 1)[0].astype(int)
    zeros = np.where(mask == 0)[0].astype(int)

    cur_c5 = _exact_c5(mask)
    best_c5 = cur_c5
    best_mask = mask.copy()

    top_masks = {}
    key0 = tuple(mask.astype(int))
    top_masks[key0] = cur_c5

    for step in range(n_steps):
        if step % 50000 == 0 and step > 0:
            frac = step / n_steps
            T = T_start * (T_end / T_start) ** frac
        else:
            frac = step / n_steps
            T = T_start * (T_end / T_start) ** frac

        oi = rng.integers(len(ones))
        zi = rng.integers(len(zeros))
        i = ones[oi]
        j = zeros[zi]

        mask[i] = 0.0
        mask[j] = 1.0
        new_c5 = _exact_c5(mask)
        delta = new_c5 - cur_c5

        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones[oi] = j
            zeros[zi] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_mask = mask.copy()
            key = tuple(mask.astype(int))
            if key not in top_masks or new_c5 < top_masks[key]:
                top_masks[key] = new_c5
                if len(top_masks) > top_k:
                    worst_key = max(top_masks, key=top_masks.get)
                    del top_masks[worst_key]
        else:
            mask[j] = 0.0
            mask[i] = 1.0

    return best_mask, best_c5, top_masks


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _structured_masks(n, m):
    """Generate structured binary masks."""
    masks = []
    # Arithmetic progression masks
    for step in [1, 3, 5, 7, 11, 13, 17, 19]:
        for offset in range(0, min(step, 4)):
            mask = np.zeros(n)
            positions = set()
            for k in range(m):
                pos = (step * k + offset) % n
                if pos not in positions:
                    positions.add(pos)
                    mask[pos] = 1.0
            if mask.sum() == m:
                masks.append(mask)
    # Periodic masks
    for period in [4, 6, 8, 10, 12, 16]:
        if n % period == 0:
            ones_per_period = round(m * period / n)
            if ones_per_period * (n // period) == m and 0 < ones_per_period < period:
                for shift in range(min(period, 3)):
                    mask = np.zeros(n)
                    for block in range(n // period):
                        for k in range(ones_per_period):
                            mask[block * period + ((k + shift) % period)] = 1.0
                    if mask.sum() == m:
                        masks.append(mask)
    return masks

    if c5_opt < best_c5:
        best_c5 = c5_opt

    # Structured masks
    struct_masks = _structured_masks(n, int(target))
    for sm in struct_masks[:30]:
        if time.time() > d1:
            break
        sh, sc = _optimize_d(sm)
        if sc < best_c5:
            best_c5 = sc
            best_h = sh.copy()

    # Random masks (a few)
    for _ in range(10):
        if time.time() > d1:
            break
        rm = np.zeros(n)
        idx = rng_master.choice(n, size=int(target), replace=False)
        rm[idx] = 1.0
        rh, rc = _optimize_d(rm)
        if rc < best_c5:
            best_c5 = rc
            best_h = rh.copy()

    # Phase 2: Metropolis on binary masks (40% budget)
    d2 = t0 + budget_s * 0.45
    top_mask_pool = {}
    n_chains = 4
    steps_per_chain = 80000
    for ci in range(n_chains):
        if time.time() > d2:
            break
        T_start = 0.02 - 0.005 * ci
        T_end = 0.001
        start_mask = mask0.copy() if ci == 0 else None
        if ci > 0:
            start_mask = np.zeros(n)
            idx = rng_master.choice(n, size=int(target), replace=False)
            start_mask[idx] = 1.0
            # Optionally do a quick SLSQP to get a better starting mask
        bm, bc, top_m = _metropolis_binary(n, int(target), steps_per_chain,
                                           T_start, T_end, seed=seed + 100 * ci,
                                           top_k=30)
        for k, v in top_m.items():
            top_mask_pool[k] = v
        # Optimize d for best mask from this chain
        bh, bc2 = _optimize_d(bm)
        if bc2 < best_c5:
            best_c5 = bc2
            best_h = bh.copy()

    # Phase 3: Optimize d for top masks from pool (10% budget)
    d3 = t0 + budget_s * 0.55
    if len(top_mask_pool) > 0:
        sorted_masks = sorted(top_mask_pool.items(), key=lambda x: x[1])[:20]
        for (key, val) in sorted_masks:
            if time.time() > d3:
                break
            m = np.array(key, dtype=float).reshape(n)
            mh, mc = _optimize_d(m)
            if mc < best_c5:
                best_c5 = mc
                best_h = mh.copy()

    # Phase 4: Local optimization - SLSQP active-set loop (25% budget)
    d4 = t0 + budget_s * 0.80
    for iteration in range(200):
        if time.time() > d4:
            break
        g_full = _g_vec(best_h)
        max_g = float(np.max(g_full))
        if max_g < 1e-10:

            c5_new2 = _exact_c5(h_new2)

                best_c5 = c5_new2
                best_h = h_new2.copy()

                if c5_pw < best_c5 - 1e-14:
                    best_c5 = c5_pw
                    best_h = h_pw.copy()
                else:

    # Phase 5: Pair sweep polish (10% budget)
    d5 = t0 + budget_s * 0.90
    if time.time() < d5:

                                seed=seed + 9000, n_t=16)
        if c5_p < best_c5:

            best_h = h_p.copy()

        for dn in [4, -4, 8, -8, 12, -12]:
            if time.time() > d6:

            n_try = n + dn
            if n_try < 16 or n_try > 128 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            sub_dead = min(d6, time.time() + budget_s * 0.02)
            h_r2, c5_r2 = _pair_sweep(h_r, max_passes=15, deadline=sub_dead,
                                      seed=seed + 10000 + n_try, n_t=12)
            g_r = _g_vec(h_r2)
            act_r = _get_active(g_r, frac=0.3, min_count=8)

        act_f = _get_active(g_f, frac=0.5, min_count=12)
        h_f = _slsqp_solve(best_h, act_f, maxiter=300, ftol=1e-14)
        c5_f = _exact_c5(h_f)

                                      seed=seed + 20000, n_t=20)
            if c5_pf < best_c5:

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
