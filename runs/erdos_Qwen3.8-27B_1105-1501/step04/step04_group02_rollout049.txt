def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    n = len(h)
    m = 2 * n - 1

    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            scores[r] = np.dot(h[d:], 1.0 - h[:cnt])
        else:
            g = -d
            cnt = n - g
            scores[r] = np.dot(h[:cnt], 1.0 - h[g:])

    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            G[r, :cnt] -= h[d:d+cnt]
            G[r, d:d+cnt] += 1.0 - h[:cnt]
        else:
            g = -d
            cnt = n - g
            G[r, :cnt] += 1.0 - h[g:g+cnt]
            G[r, g:g+cnt] -= h[:cnt]
    return G

def _slsqp_solve(h, deadline, max_iter=80):
    n = len(h)
    target = n / 2.0
    h_init = _project(h.copy(), target)
    scores_init = _manual_scores(h_init)
    T_init = float(np.max(scores_init)) + 0.005
    x0 = np.concatenate([h_init, [T_init]])
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
    bounds.append((0.0, float(n)))

    def obj_fun(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def ineq_fun(x):
        h_vec = x[:-1]
        T_val = x[-1]
        sc = _manual_scores(h_vec)
        return T_val - sc

    def ineq_jac(x):
        h_vec = x[:-1]
        G = _manual_grad_matrix(h_vec)
        nv = len(h_vec)
        jac = np.zeros((2 * nv - 1, nv + 1))
        jac[:, :nv] = -G
        jac[:, nv] = 1.0
        return jac

    def eq_fun(x):
        return np.array([np.sum(x[:-1]) - target])

    def eq_jac(x):
        nv = len(x) - 1
        jac = np.zeros((1, nv + 1))
        jac[0, :nv] = 1.0
        return jac

    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]

    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    n_iter = max(10, max_iter)
    try:
        res = minimize(obj_fun, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': n_iter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass
    return best_h, best_c5

def _slsqp_topk(h, deadline, k_init=15, k_max=40):
    n = len(h)
    target = n / 2.0
    h_cur = _project(h.copy(), target)
    best_h = h_cur.copy()
    best_c5 = _exact_c5(h_cur)
    scores = _manual_scores(h_cur)
    top_idx = np.argsort(scores)[::-1][:k_init]

    for outer in range(30):
        if time.time() > deadline:
            break
        active = top_idx.copy()
        scores_cur = _manual_scores(h_cur)
        T0 = float(np.max(scores_cur)) + 0.005
        x0 = np.concatenate([h_cur, [T0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]

        def obj_fun2(x):
            return x[-1]

        def obj_grad2(x):
            g = np.zeros(len(x))
            g[-1] = 1.0
            return g

        def ineq_fun2(x):
            hv = x[:-1]
            sc = _manual_scores(hv)
            return x[-1] - sc[active]

        def ineq_jac2(x):
            hv = x[:-1]
            G = _manual_grad_matrix(hv)
            nv = len(hv)
            jac = np.zeros((len(active), nv + 1))
            jac[:, :nv] = -G[active]
            jac[:, nv] = 1.0
            return jac

        def eq_fun2(x):
            return np.array([np.sum(x[:-1]) - target])

        def eq_jac2(x):
            nv = len(x) - 1
            jac = np.zeros((1, nv + 1))
            jac[0, :nv] = 1.0
            return jac

        constrs = [
            {'type': 'ineq', 'fun': ineq_fun2, 'jac': ineq_jac2},
            {'type': 'eq', 'fun': eq_fun2, 'jac': eq_jac2}
        ]
        try:
            res = minimize(obj_fun2, x0, jac=obj_grad2, method='SLSQP',
                           bounds=bounds, constraints=constrs,
                           options={'maxiter': 50, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _project(res.x[:n], target)
                sc_all = _manual_scores(h_new)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                # Check violations among all lags
                viol = np.where(sc_all > res.x[-1] + 1e-5)[0]
                if len(viol) == 0:
                    # All constraints satisfied, check if solution is good
                    h_cur = h_new
                    break
                # Update active set: add violated, keep top k_max
                all_idx = np.unique(np.concatenate([active, viol]))
                sc_all_idx = sc_all[all_idx]
                order = np.argsort(sc_all_idx)[::-1]
                top_idx = all_idx[order[:k_max]]
                h_cur = h_new
            else:
                break
        except Exception:
            break
        if time.time() > deadline:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    # Area-preserving: mass in each new cell
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_lo = j * dx_new
        x_hi = (j + 1) * dx_new
        mass = 0.0
        for i in range(n):
            i_lo = i * dx_old
            i_hi = (i + 1) * dx_old
            overlap = min(x_hi, i_hi) - max(x_lo, i_lo)
            if overlap > 0:
                mass += h[i] * overlap
        h_new[j] = mass / dx_new
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = _exact_c5(h_cur)
    improved_any = True
    for _pass in range(max_passes):
        if time.time() > deadline or not improved_any:
            break
        improved_any = False
        scores = _manual_scores(h_cur)
        top_k = min(15, 2 * n - 1)
        active = np.argsort(scores)[::-1][:top_k]
        # Build gradient for active set
        G = _manual_grad_matrix(h_cur)
        grad = np.zeros(n)
        for r in active:
            grad += G[r]
        grad /= len(active)
        # Sort indices by gradient magnitude
        idx_order = np.argsort(-np.abs(grad))
        n_try = min(n, 60)
        trial_idx = idx_order[:n_try]
        for j in trial_idx:
            if time.time() > deadline:
                break
            if h_cur[j] <= 1e-8 or h_cur[j] >= 1.0 - 1e-8:
                continue
            # Try moving h[j] up or down, compensating at another index
            for direction in ([1.0, -1.0] if grad[j] > 0 else [-1.0, 1.0]):
                # Find best partner
                for k in trial_idx:
                    if k == j:
                        continue
                    if h_cur[k] <= 1e-8 or h_cur[k] >= 1.0 - 1e-8:
                        continue
                    am = 0.05
                    if direction > 0:
                        am = min(am, (1.0 - h_cur[j]) / 1.0)
                    else:
                        am = min(am, h_cur[j] / 1.0)
                    if direction > 0:
                        am = min(am, h_cur[k] / 1.0)
                    else:
                        am = min(am, (1.0 - h_cur[k]) / 1.0)
                    if am < 1e-10:
                        continue
                    h_t = h_cur.copy()
                    h_t[j] += direction * am
                    h_t[k] -= direction * am
                    h_t = np.clip(h_t, 0.0, 1.0)
                    h_t = _project(h_t, target)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h_cur = h_t.copy()
                        improved_any = True
                        break
                if improved_any:
                    break
            if improved_any:
                break
    return best_h, best_c5

def _subgradient_optimize(h, deadline, n_active=8, lr_init=0.02):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = _exact_c5(h_cur)
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        scores = _manual_scores(h_cur)
        max_s = float(np.max(scores))
        # Active set: lags within 5% of max
        thresh = max_s - 0.05 * max_s
        active = np.where(scores >= thresh)[0]
        if len(active) > n_active:
            order = np.argsort(scores[active])[::-1]
            active = active[order[:n_active]]
        G = _manual_grad_matrix(h_cur)
        grad = G[active].mean(axis=0)
        h_new = h_cur - lr * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            stale = 0
            lr = min(lr * 1.02, 0.1)
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if stall > 60:
                break
    return best_h, best_c5

def _line_search_pair(h, j, k, sign, deadline):
    """Line search along direction: h[j] += t*sign, h[k] -= t*sign."""
    n = len(h)
    target = n / 2.0
    # Find max feasible t
    amax = 0.5
    if sign > 0:
        amax = min(amax, (1.0 - h[j]), h[k])
    else:
        amax = min(amax, h[j], (1.0 - h[k]))
    if amax < 1e-10:
        return None, None
    # Golden section on [0, amax]
    c5_cur = _exact_c5(h)
    best_t = 0.0
    best_val = c5_cur
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = amax
    # Quick grid
    n_grid = 20
    ts = np.linspace(0.0, amax, n_grid)
    for t in ts:
        h_t = h.copy()
        h_t[j] += sign * t
        h_t[k] -= sign * t
        if h_t.min() < -1e-12 or h_t.max() > 1 + 1e-12:
            continue
        c5_t = _exact_c5(h_t)
        if c5_t < best_val:
            best_val = c5_t
            best_t = t
    # Refine with golden section
    t1 = hi - gr * (hi - lo)
    t2 = lo + gr * (hi - lo)

    def _f(t):
        h_t = h.copy()
        h_t[j] += sign * t
        h_t[k] -= sign * t
        if h_t.min() < -1e-12 or h_t.max() > 1 + 1e-12:
            return best_val + 1.0
        return _exact_c5(h_t)

    if best_t > 1e-15:
        lo = max(0.0, best_t - amax / n_grid)
        hi = min(amax, best_t + amax / n_grid)
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        f1 = _f(t1)
        f2 = _f(t2)
        for _ in range(25):
            if hi - lo < 1e-8:
                break
            if f1 < f2:
                hi = t2
                t2 = t1
                f2 = f1
                t1 = hi - gr * (hi - lo)
                f1 = _f(t1)
            else:
                lo = t1
                t1 = t2
                f1 = f2
                t2 = lo + gr * (hi - lo)
                f2 = _f(t2)
        opt_t = (lo + hi) / 2.0
        f_opt = _f(opt_t)
        if f_opt < best_val:
            best_val = f_opt
            best_t = opt_t
    return best_t, best_val

def _grad_lp_hybrid(h, deadline):
    """Alternating between gradient and LP-like step."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = _exact_c5(h_cur)
    lr = 0.015
    stall = 0
    while time.time() < deadline:
        scores = _manual_scores(h_cur)
        max_s = float(np.max(scores))
        thresh = max_s - 0.03 * max_s
        active = np.where(scores >= thresh)[0]
        if len(active) > 12:
            order = np.argsort(scores[active])[::-1]
            active = active[order[:12]]
        G = _manual_grad_matrix(h_cur)
        grad = G[active].mean(axis=0)
        # Try both gradient step and anti-gradient step for top lag
        top_r = int(np.argmax(scores))
        dir1 = -grad
        dir2 = -G[top_r]
        for d in [dir1, dir2]:
            # Normalize
            d_max = float(np.max(np.abs(d)))
            if d_max < 1e-12:
                continue
            d = d / d_max
            # Find feasible alpha
            amax = lr * 2
            for j in range(n):
                if d[j] > 1e-14:
                    amax = min(amax, (1.0 - h_cur[j]) / d[j])
                elif d[j] < -1e-14:
                    amax = min(amax, h_cur[j] / (-d[j]))
            amax = max(amax, 1e-10)
            # Quick line search
            ts = np.linspace(0, amax, 12)
            best_t = 0.0
            best_v = best_c5
            for t in ts:
                h_t = h_cur + t * d
                if h_t.min() < -1e-12 or h_t.max() > 1 + 1e-12:
                    continue
                h_t = _project(h_t, target)
                c5_t = _exact_c5(h_t)
                if c5_t < best_v:
                    best_v = c5_t
                    best_t = t
            if best_t > 1e-10:
                h_new = _project(h_cur + best_t * d, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cur = h_new
                    lr = min(lr * 1.03, 0.08)
                    stall = 0
                    break
        else:
            stall += 1
            lr *= 0.93
            if stall > 80:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97

    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float, copy=True)
    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = _project(h0, target0)

    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    def _try_update(cand_h):
        nonlocal best_h, best_c5, best_n
        ch = _project(cand_h, len(cand_h) / 2.0)
        c5_cand = _exact_c5(ch)
        if c5_cand < best_c5 - 1e-12:
            best_c5 = c5_cand
            best_h = ch.copy()
            best_n = len(ch)

    # Generate diverse initial candidates
    candidates = []
    candidates.append(h0.copy())
    candidates.append(h0[::-1].copy())
    # Relaxed
    for q in [0.3, 0.5, 0.7]:
        hr = h0 - q * (h0 - 0.5)
        candidates.append(_project(hr, target0))
    # Sharpened
    for q in [0.3, 0.5]:
        hs = np.where(h0 >= 0.5, h0 + q * (1.0 - h0) * 0.4, h0 - q * h0 * 0.4)
        candidates.append(_project(hs, target0))
    # Binarized
    sorted_idx = np.argsort(-h0)
    n_on = int(target0 // 1)
    h_bin = np.zeros(n0)
    h_bin[sorted_idx[:n_on]] = 1.0
    rem = target0 - n_on
    if rem > 1e-10:
        h_bin[sorted_idx[n_on]] = rem
    candidates.append(h_bin.copy())
    # Shifted
    for shift in [1, 2, 3, n0 // 4]:
        candidates.append(np.roll(h0, shift).copy())
    # Random perturbations
    for s in [0.03, 0.06, 0.1]:
        hp = h0 + rng.normal(0, s, n0)
        candidates.append(_project(hp, target0))

    # Evaluate and sort candidates
    cand_scores = []
    for c in candidates:
        c5_c = _exact_c5(c)
        cand_scores.append(c5_c)
    order = np.argsort(cand_scores)
    candidates = [candidates[i] for i in order]

    # Phase 1: SLSQP top-k at parent resolution (30% budget)
    d1 = t0 + budget_s * 0.30
    for ci in range(min(5, len(candidates))):
        if time.time() > d1:
            break
        hc = candidates[ci]
        sub_dl = min(d1, time.time() + (d1 - t0) * 0.2)
        h_new, c5_new = _slsqp_topk(hc, sub_dl, k_init=15, k_max=35)
        _try_update(h_new)
        # Also try full SLSQP if n is small enough
        if best_n <= 100 and time.time() < d1:
            sub_dl2 = min(d1, time.time() + (d1 - t0) * 0.15)
            h_new2, c5_new2 = _slsqp_solve(h_new, sub_dl2, max_iter=60)
            _try_update(h_new2)

    # Phase 2: Multi-resolution SLSQP (35% budget)
    d2 = t0 + budget_s * 0.65
    res_list = []
    base = best_n
    for dn in [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, 32, -32, 40]:
        nt = base + dn
        if 40 <= nt <= 180 and nt % 2 == 0:
            if nt not in res_list:
                res_list.append(nt)
    # Also add some fixed sizes
    for sz in [56, 64, 72, 80, 88, 96, 104, 112, 120, 128, 136]:
        if sz not in res_list and 40 <= sz <= 180:
            res_list.append(sz)
    res_list.sort(key=lambda x: abs(x - base))
    # Remove duplicates
    seen = set()
    res_list_f = []
    for r in res_list:
        if r not in seen:
            seen.add(r)
            res_list_f.append(r)
    res_list = res_list_f

    time_per_res = budget_s * 0.03
    for rn in res_list:
        if time.time() > d2:
            break
        if rn == best_n:
            continue
        h_res = _resample(best_h, rn)
        sub_dl = min(d2, time.time() + time_per_res)
        h_r, c5_r = _slsqp_topk(h_res, sub_dl, k_init=12, k_max=30)
        _try_update(h_r)
        if c5_r < best_c5 + 0.001 and time.time() < d2:
            # Try gradient hybrid on this
            sub_dl2 = min(d2, time.time() + time_per_res * 0.4)
            h_r2, c5_r2 = _grad_lp_hybrid(h_r, sub_dl2)
            _try_update(h_r2)

    # Phase 3: Pair refinement at best resolution (15% budget)
    d3 = t0 + budget_s * 0.80
    h_pr, c5_pr = _pair_refine(best_h, d3, max_passes=150)
    _try_update(h_pr)

    # Phase 4: Further SLSQP + subgradient at best resolution (10% budget)
    d4 = t0 + budget_s * 0.90
    if time.time() < d4:
        h_sg, c5_sg = _subgradient_optimize(best_h, min(d4, time.time() + budget_s * 0.04), n_active=6, lr_init=0.01)
        _try_update(h_sg)
    if time.time() < d4:
        h_sl, c5_sl = _slsqp_topk(best_h, min(d4, time.time() + budget_s * 0.04), k_init=20, k_max=40)
        _try_update(h_sl)

    # Phase 5: Perturbation restarts (5% budget)
    d5 = t0 + budget_s * 0.95
    scale = 0.03
    for attempt in range(30):
        if time.time() > d5:
            break
        hp = best_h + rng.normal(0, scale, best_n)
        hp = _project(hp, best_n / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_p, c5_p = _slsqp_topk(hp, sub_dl, k_init=12, k_max=25)
        if c5_p < best_c5 - 1e-12:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = len(h_p)
            scale = max(0.01, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.3)

    # Final polishing
    if time.time() < hard_deadline:
        h_fin, c5_fin = _pair_refine(best_h, hard_deadline, max_passes=50)
        _try_update(h_fin)

    # Ensure feasibility of final answer
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
