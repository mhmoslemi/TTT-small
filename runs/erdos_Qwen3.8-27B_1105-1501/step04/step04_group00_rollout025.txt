def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    S = _scores(h)  # shape (2n-1,)
    M = np.max(S)
    u = np.exp((S - M) / tau)
    Z = np.sum(u)
    w = u / Z  # shape (2n-1,)
    
    # Filter to only compute for significant weights
    threshold = 1e-10
    active = w > threshold
    if not np.any(active):
        return M + tau * np.log(Z), np.zeros(n)
    
    F = M + tau * np.log(Z)
    grad = np.zeros(n)
    
    for r in range(2*n-1):
        if not active[r]:
            continue
        d = r - (n-1)
        wr = w[r]
        if d == 0:
            grad += wr * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += wr * dx * (1.0 - h[d:])
            grad[d:] -= wr * dx * h[:m]
        else:
            g = -d
            m = n - g
            grad[g:] += wr * dx * (1.0 - h[:m])
            grad[:m] -= wr * dx * h[g:]
    
    return F, grad

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj(h, tau):
    S = _scores(h)
    M = np.max(S)
    u = np.exp((S - M) / tau)
    Z = np.sum(u)
    return M + tau * np.log(Z)

def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    S = _scores(h)
    M = np.max(S)
    u = np.exp((S - M) / tau)
    Z = np.sum(u)
    F = M + tau * np.log(Z)
    w = u / Z
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        wr = w[r]
        if d == 0:
            grad += wr * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            if m > 0:
                grad[:m] += wr * dx * (1.0 - h[d:])
                grad[d:] -= wr * dx * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[g:] += wr * dx * (1.0 - h[:m])
                grad[:m] -= wr * dx * h[g:]
    return F, grad

def _safe_line(h, p, tau, alpha_max, n_grid=15, n_golden=12):
    best_alpha = 0.0
    best_val = _smooth_obj(h, tau)
    if alpha_max < 1e-14:
        return best_alpha, best_val
    alphas = np.linspace(0.0, alpha_max, n_grid)
    vals = np.array([_smooth_obj(h + a * p, tau) for a in alphas])
    best_i = int(np.argmin(vals))
    if 0 < best_i < n_grid - 1:
        lo = alphas[best_i - 1]
        hi = alphas[best_i + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _smooth_obj(h + c1 * p, tau)
        f2 = _smooth_obj(h + c2 * p, tau)
        for _ in range(n_golden):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _smooth_obj(h + c1 * p, tau)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _smooth_obj(h + c2 * p, tau)
        opt_a = (lo + hi) / 2.0
        opt_v = _smooth_obj(h + opt_a * p, tau)
        if opt_v < best_val:
            best_alpha, best_val = opt_a, opt_v
    elif best_i == n_grid - 1:
        best_alpha, best_val = alpha_max, vals[-1]
    else:
        best_alpha, best_val = 0.0, vals[0]

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

def _lp_direction(h, scores, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        sc = scores[idx]
    else:
        idx = np.arange(n_lags)
        sc = scores
    G = np.zeros((len(idx), n + 1))
    dx = 2.0 / n
    for ii, r in enumerate(idx):
        d = r - (n - 1)
        if d == 0:
            G[ii, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            G[ii, :m] += dx * (1.0 - h[d:])
            G[ii, d:] -= dx * h[:m]
        else:
            g = -d
            m = n - g
            G[ii, g:] += dx * (1.0 - h[:m])
            G[ii, :m] -= dx * h[g:]
    G[:, -1] = -1.0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = G
    b_ub = -sc
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _binary_sa(n, deadline, rng, n_iters=30000):

    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0004
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                else:
                    sub_dl2 = min(dl4, time.time() + budget_s * 0.012)
                    h_lp, c5_lp = _lp_optimize(h_try, sub_dl2, radius=0.1,
                                               max_iters=800, top_k_list=[10, 5, None])
                    if c5_lp < best_c5:
                        best_c5 = c5_lp
                        best_h = h_lp.copy()
                        best_n = n_try

    # Phase 5: Fourier optimization
    if time.time() < deadline - budget_s * 0.05:
        dl5 = min(deadline, time.time() + budget_s * 0.10)
        h_f, c5_f = _fourier_optimize(best_h, dl5, K=14, n_steps=2000,
                                       tau_schedule=[0.03, 0.015, 0.008, 0.004, 0.002])
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if time.time() < dl5:
            h_f2, c5_f2 = _fourier_optimize(h_start, dl5, K=10, n_steps=1500,
                                             tau_schedule=[0.05, 0.025, 0.01, 0.005])
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    # Phase 6: Binary SA + quasi-periodic
    if time.time() < deadline - budget_s * 0.03:
        dl6 = min(deadline, time.time() + budget_s * 0.08)
        for n_bin in [96, 128]:
            if time.time() > dl6:
                break
            sub_dl = min(dl6, time.time() + budget_s * 0.025)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=20000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
            if time.time() < dl6:
                alpha = 0.6180339887
                h_qp = _quasi_init(n_bin, alpha)
                sub_dl2 = min(dl6, time.time() + budget_s * 0.02)
                tau_q = [0.04, 0.02, 0.01, 0.005]
                h_qp2, c5_qp = _soft_anneal(h_qp, sub_dl2, tau_q, max_iters_per_tau=100)
                if c5_qp < best_c5:
                    best_c5 = c5_qp
                    best_h = h_qp2.copy()
                    best_n = n_bin

    # Phase 7: Perturbation restarts
    if time.time() < deadline - budget_s * 0.02:
        dl7 = min(deadline, time.time() + budget_s * 0.06)
        scale = 0.02
        for attempt in range(200):
            if time.time() > dl7:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl7, time.time() + budget_s * 0.012)
            tau_p = [0.008, 0.004, 0.002]
            h_try, c5_try = _soft_anneal(h_pert, sub_dl, tau_p, max_iters_per_tau=80)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.005, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.1)

    # Phase 8: Final LP + smooth polish
    if time.time() < deadline:
        dl8 = min(deadline, time.time() + budget_s * 0.03)
        h_lp, c5_lp = _lp_optimize(best_h, dl8, radius=0.05, max_iters=1000,
                                   top_k_list=[5, 8, None])
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 9: Final smooth annealing (very tight tau)
    if time.time() < deadline:
        dl9 = min(deadline, time.time() + budget_s * 0.02)
        tau_final = [0.003, 0.0015, 0.0008, 0.0004]
        h_fin, c5_fin = _soft_anneal(best_h, dl9, tau_final, max_iters_per_tau=200)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 10: Pair polish
    if time.time() < deadline:
        dl10 = min(deadline, time.time() + budget_s * 0.02)
        hp, c5p = _pair_polish(best_h, dl10, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
