def _exact_c5(h):

    dx = 2.0 / n

def _scores(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))

        else:

            x = np.clip(x, 0.0, 1.0)

def _grad_matrix(h):

    n = len(h)
    target = n / 2.0
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    alpha = alpha_init
    stall = 0

        if time.time() > deadline:
            break
        if stall > 100:
            break
        frac = min(1.0, it / max(1, n_iters - 1))
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)

        # Softmax weights

        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        # Backtracking line search
        h_new = None
        c5_new = cur_c5
        for _bt in range(20):

            c5_try = _exact_c5(h_try)

                c5_new = c5_try
                break

            alpha = min(alpha * 1.05, 0.5)

    return best_h, best_c5

def _differential_evolution(h_best, deadline, n_pop=30, K=10, n_gen=200,
                             F=0.8, CR=0.5, rng=None):

    n = len(h_best)
    # Build Fourier basis
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):

        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    
    target = n / 2.0
    a_init = np.linalg.lstsq(B, h_best - 0.5, rcond=None)[0]
    
    # Initialize population
    pop_a = np.zeros((n_pop, 2*K))
    pop_a[0] = a_init
    for i in range(1, n_pop):
        pop_a[i] = a_init + 0.15 * rng.standard_normal(2*K)
    
    pop_h = np.zeros((n_pop, n))

        pop_c5[i] = _exact_c5(h_try)

            # Crossover
            cross_mask = rng.random(2*K) < CR

                pop_h[i] = h_try
                pop_c5[i] = c5_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        # Adapt F and CR
        F = max(0.3, F * 0.995)
        CR = 0.3 + 0.4 * np.random.random()  # Actually use rng

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_samp = 12
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(16):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _local_optimize(h, deadline, top_k_list=None, radius_init=0.15, max_iters=3000, rng=None):
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)

            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break

        alpha, new_c5 = _line_search(h, p)

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0

        else:
            stall += 1

            ti += 1

def _smooth_grad_opt(h, deadline, beta_start=8.0, beta_end=500.0,
                     max_iters=2500, alpha_init=0.04, rng=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = alpha_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        frac = min(1.0, it / max(1, max_iters - 1))
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        exp_s = np.exp(beta * (scores - max_s))
        w = exp_s / np.sum(exp_s)
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        h_new = None
        alpha_try = alpha
        for _bt in range(25):
            h_try = _project(h - alpha_try * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h_new = h_try
                break
            alpha_try *= 0.7
            if alpha_try < 1e-13:
                break
        if h_new is not None:
            h = h_new
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            alpha = min(alpha / 0.7 * 1.03, 0.3)
        else:
            stall += 1
            alpha *= 0.85
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_basis(n, K):
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(2 * np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    return B

def _de_optimize(h_best, deadline, n_pop=25, K=8, n_gen=300, rng=None):
    n = len(h_best)
    B = _fourier_basis(n, K)
    target = n / 2.0
    a_init = np.linalg.lstsq(B, h_best - 0.5, rcond=None)[0]
    pop_a = np.zeros((n_pop, 2*K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        if i == 0:
            pop_a[i] = a_init
        else:
            pop_a[i] = a_init + 0.12 * rng.standard_normal(2*K)
        h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
        h_try = _project(h_try, target)
        pop_c5[i] = _exact_c5(h_try)
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = np.clip(0.5 + B @ pop_a[best_i], 0, 1)
    best_h = _project(best_h, target)
    F = 0.7
    CR = 0.5
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(n_pop):
            a, b, c_idx = list(rng.choice(n_pop, 3, replace=False))
            mutant = pop_a[a] + F * (pop_a[b] - pop_a[c_idx])
            cross_mask = rng.random(2*K) < CR
            if not cross_mask.any():
                cross_mask[rng.integers(0, 2*K)] = True
            trial_a = np.where(cross_mask, mutant, pop_a[i])
            h_try = np.clip(0.5 + B @ trial_a, 0, 1)
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try <= pop_c5[i]:
                pop_a[i] = trial_a
                pop_c5[i] = c5_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        F = max(0.4, F * 0.998)
        CR = 0.3 + 0.4 * rng.random()
    return best_h, best_c5

def _binary_sa(n, deadline, rng, n_iters=30000, T_init=0.004):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    T_min = 0.00002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones_list = list(positions[:target])
    zeros_list = list(positions[target:])
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 4 == 0:
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
                cur_c5 = new_c5
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            if len(ones_list) < 2 or len(zeros_list) < 2:
                continue
            i1 = ones_list[rng.integers(len(ones_list))]
            i2 = ones_list[rng.integers(len(ones_list))]
            j1 = zeros_list[rng.integers(len(zeros_list))]
            j2 = zeros_list[rng.integers(len(zeros_list))]
            if i1 == i2 or j1 == j2:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
                cur_c5 = new_c5
                ones_list.remove(i1)
                ones_list.remove(i2)
                ones_list.extend([j1, j2])
                zeros_list.remove(j1)
                zeros_list.remove(j2)
                zeros_list.extend([i1, i2])
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        T *= cool
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 24)
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + step, n, step):
                if time.time() > deadline:
                    break
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _structured_binary_init(n, rng):
    h = np.zeros(n)
    target = n // 2
    # Use a structured pattern based on modular arithmetic
    step = int(np.ceil(np.sqrt(n)))
    ones = set()
    for i in range(n):
        if (i * step + i * i) % n < target:
            ones.add(i)
    # Fill or remove to get exact count
    remaining = list(set(range(n)) - ones)
    if len(ones) < target:
        for i in remaining[:target - len(ones)]:
            ones.add(i)
    elif len(ones) > target:
        for i in list(ones)[:len(ones) - target]:
            ones.remove(i)
    for i in ones:
        h[i] = 1.0
    return h

                    best_h = h_bp.copy()
                    best_n = n_bin
            # Smooth grad on binary
            if time.time() < dl4:
                sub_dl3 = min(dl4, time.time() + budget_s * 0.015)
                h_bs, c5_bs = _smooth_grad_opt(h_b, sub_dl3,
                                                beta_start=20.0, beta_end=400.0,
                                                max_iters=600, alpha_init=0.02, rng=rng)
                if c5_bs < best_c5:
                    best_c5 = c5_bs
                    best_h = h_bs.copy()
                    best_n = n_bin

    # Phase 5: DE on best at current resolution + resample up/down
    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.88
        # DE at best_n
        sub_dl_de = min(dl5, time.time() + budget_s * 0.03)
        h_de2, c5_de2 = _de_optimize(best_h, sub_dl_de, n_pop=30, K=10,
                                      n_gen=300, rng=rng)
        if c5_de2 < best_c5:
            best_c5 = c5_de2
            best_h = h_de2.copy()
        # Resample up then down (resolution boosting)
        if time.time() < dl5:
            n_up = min(best_n + 48, 256)
            h_up = _resample(best_h, n_up)
            sub_dl_up = min(dl5, time.time() + budget_s * 0.02)
            h_up_opt, c5_up = _local_optimize(h_up, sub_dl_up,
                                               top_k_list=[None, 15, 8],
                                               radius_init=0.1, max_iters=800, rng=rng)
            h_dn = _resample(h_up_opt, best_n)
            c5_dn = _exact_c5(h_dn)
            if c5_dn < best_c5:
                best_c5 = c5_dn
                best_h = h_dn.copy()
            # LP on resampled-down
            if time.time() < dl5:
                sub_dl_dn = min(dl5, time.time() + budget_s * 0.015)
                h_dn2, c5_dn2 = _local_optimize(h_dn, sub_dl_dn,
                                                 top_k_list=[8, 4, None],
                                                 radius_init=0.05, max_iters=600, rng=rng)
                if c5_dn2 < best_c5:
                    best_c5 = c5_dn2
                    best_h = h_dn2.copy()

    # Phase 6: Perturbation restarts with LP + smooth grad
    dl6 = t0 + budget_s * 0.94
    scale = 0.025
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.standard_normal(best_n) * scale
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        if attempt % 3 == 0:
            h_try, c5_try = _local_optimize(h_pert, sub_dl,
                                             top_k_list=[5, 8, None],
                                             radius_init=0.06, max_iters=300, rng=rng)
        else:
            h_try, c5_try = _smooth_grad_opt(h_pert, sub_dl,
                                              beta_start=30.0, beta_end=300.0,
                                              max_iters=400, alpha_init=0.02, rng=rng)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.9)
        else:
            scale = min(0.08, scale * 1.1)

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.985
        hp, c5p = _pair_polish(best_h, dl7, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: tiny LP refinement
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _local_optimize(best_h, t0 + budget_s * 0.998,
                                         top_k_list=[3, 5, 4],
                                         radius_init=0.01, max_iters=400, rng=rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
