import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)
            idx = 0
            while diff > 1e-12 and idx < n:
                i = order[idx]
                if x[i] >= 1.0:
                    idx += 1
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                idx += 1
        else:
            order = np.argsort(-x)
            idx = 0
            while diff < -1e-12 and idx < n:
                i = order[idx]
                if x[i] <= 0.0:
                    idx += 1
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                idx += 1
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _coord_desc(h, n, max_sweeps=10000, tol=1e-11, deadline=None):
    """Coordinate descent with exact 1D minimization per coordinate."""
    h = np.asarray(h, dtype=float).copy()
    target = n * 0.5
    best_c5 = _exact_c5(h)
    s_arr = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        c5_before = _exact_c5(h)
        for i in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            # Compute D[r] for each lag r
            D = np.zeros(2 * n - 1)
            idx1 = np.where((i - s_arr >= 0) & (i - s_arr < n))[0]
            if len(idx1) > 0:
                D[idx1] += 1.0 - h[i - s_arr[idx1]]
            idx2 = np.where((i + s_arr >= 0) & (i + s_arr < n))[0]
            if len(idx2) > 0:
                D[idx2] -= h[i + s_arr[idx2]]
            # corr_r(x) = corr_r + D_r * (x - h[i]) for x in [0,1]
            a = corr - D * h[i]
            # Minimize max(a + D*x) for x in [0,1]
            v0 = float(np.max(a))
            v1 = float(np.max(a + D))
            best_v = min(v0, v1)
            best_x = 0.0 if v0 <= v1 else 1.0
            pos = D > 1e-14
            neg = D < -1e-14
            if np.any(pos) and np.any(neg):
                ap = a[pos]
                an = a[neg]
                Dp = D[pos]
                Dn = D[neg]
                xp = (an[None, :] - ap[:, None]) / (Dp[:, None] - Dn[None, :])
                vp = ap[:, None] + Dp[:, None] * xp
                valid = (xp >= 0.0) & (xp <= 1.0)
                if np.any(valid):
                    vv = np.where(valid, vp, np.inf)
                    mi = np.unravel_index(np.argmin(vv), vv.shape)
                    if vv[mi] < best_v:
                        best_v = float(vv[mi])
                        best_x = float(xp[mi])
            if abs(best_x - h[i]) > 1e-15:
                h[i] = best_x
                h = _project(h, target)
        c5_after = _exact_c5(h)
        if c5_after < best_c5:
            best_c5 = c5_after
        if c5_before - c5_after < tol:
            break
    return h, _exact_c5(h)


def _pair_coord_desc(h, n, max_sweeps=5000, deadline=None):
    """Pair coordinate descent: update two coords maintaining mass exactly."""
    h = np.asarray(h, dtype=float).copy()
    target = n * 0.5
    s_arr = np.arange(-(n - 1), n)
    best_c5 = _exact_c5(h)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            j = (i + 1) % n
            c = h[i] + h[j]
            lo_x = max(0.0, c - 1.0)
            hi_x = min(1.0, c)
            if hi_x - lo_x < 1e-12:
                continue
            corr = np.correlate(h, 1.0 - h, mode='full')
            # D for coord i, E for coord j
            Di = np.zeros(2 * n - 1)
            E = np.zeros(2 * n - 1)
            idx1i = np.where((i - s_arr >= 0) & (i - s_arr < n))[0]
            idx2i = np.where((i + s_arr >= 0) & (i + s_arr < n))[0]
            if len(idx1i) > 0:
                Di[idx1i] += 1.0 - h[i - s_arr[idx1i]]
            if len(idx2i) > 0:
                Di[idx2i] -= h[i + s_arr[idx2i]]
            idx1j = np.where((j - s_arr >= 0) & (j - s_arr < n))[0]
            idx2j = np.where((j + s_arr >= 0) & (j + s_arr < n))[0]
            if len(idx1j) > 0:
                E[idx1j] += 1.0 - h[j - s_arr[idx1j]]
            if len(idx2j) > 0:
                E[idx2j] -= h[j + s_arr[idx2j]]
            # corr_r(x) where x=h[i], h[j]=c-x:
            # = base_r + Di_r * x + E_r * (c - x)
            # = (base_r + E_r * c) + (Di_r - E_r) * x
            a = corr - Di * h[i] - E * h[j] + E * c
            d = Di - E
            # Minimize max(a + d*x) for x in [lo_x, hi_x]
            v_lo = float(np.max(a + d * lo_x))
            v_hi = float(np.max(a + d * hi_x))
            best_v = min(v_lo, v_hi)
            best_x = lo_x if v_lo <= v_hi else hi_x
            pos = d > 1e-14
            neg = d < -1e-14
            if np.any(pos) and np.any(neg):
                ap = a[pos]
                an = a[neg]
                dp = d[pos]
                dn = d[neg]
                xp = (an[None, :] - ap[:, None]) / (dp[:, None] - dn[None, :])
                vp = ap[:, None] + dp[:, None] * xp
                valid = (xp >= lo_x - 1e-14) & (xp <= hi_x + 1e-14)
                if np.any(valid):
                    vv = np.where(valid, vp, np.inf)
                    mi = np.unravel_index(np.min(vv), vv.shape)
                    if vv[mi] < best_v:
                        best_v = float(vv[mi])
                        best_x = float(np.clip(xp[mi], lo_x, hi_x))
            if abs(best_x - h[i]) > 1e-14:
                h[i] = best_x
                h[j] = c - best_x
                h = _project(h, target)
        c5_cur = _exact_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
    return h, _exact_c5(h)


def _epi_f(z, n):
    return z[n]


def _epi_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _ineq_f(z, n):
    h, t = z[:n], z[n]
    return t - np.correlate(h, 1.0 - h, mode='full')


def _ineq_g(z, n):
    h = z[:n]
    nc = 2 * n - 1
    J = np.zeros((nc, n + 1))
    j = np.arange(n)
    for r in range(nc):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        J[r, :n] = -pd
        J[r, n] = 1.0
    return J


def _slsqp_solve(h_init, n, maxiter=400, ftol=1e-14):
    h = np.asarray(h_init, dtype=float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_f, z0, jac=_epi_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return _project(h, n * 0.5), _exact_c5(h)


def _lse_objective(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    exp_v = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_v)))
    w = exp_v / float(np.sum(exp_v))
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        grad += w[r] * pd
    grad -= np.mean(grad)
    return f, grad


def _lse_step(h, n, alpha, maxiter=100):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_objective, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 50, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return _project(h, n * 0.5), _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _ftl_refine(h, n, steps=2000, deadline=None):
    """Follow-the-leader with decreasing step size."""
    h = np.asarray(h, dtype=float).copy()
    target = n * 0.5
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h_avg = h.copy()
    for t in range(1, steps + 1):
        if deadline is not None and time.time() > deadline:
            break
        eta = 0.5 / (1.0 + 0.01 * t)
        corr = np.correlate(h, 1.0 - h, mode='full')
        r_max = int(np.argmax(corr))
        s = r_max - (n - 1)
        j = np.arange(n)
        g = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        g[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        g[m2] -= h[ps[m2]]
        g -= np.mean(g)
        h_new = _project(h - eta * g, target)
        h_avg = (h_avg * t + h_new) / (t + 1)
        c5_cur = _exact_c5(h_new)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_new.copy()
        if t % 100 == 0:
            c5_avg = _exact_c5(h_avg)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg.copy()
        h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def _elapsed():
        return time.time() - t0

    # Phase 1: Coordinate descent on initial state (20% budget)
    d1 = t0 + budget_s * 0.20
    h_cd, c5_cd = _coord_desc(h, n0, max_sweeps=8000, deadline=d1)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 2: Pair coordinate descent (10% budget)
    d2 = t0 + budget_s * 0.30
    h_pc, c5_pc = _pair_coord_desc(best_h, n0, max_sweeps=4000, deadline=d2)
    if c5_pc < best_c5:
        best_c5, best_h = c5_pc, h_pc.copy()

    # Phase 3: SLSQP polish (15% budget)
    d3 = t0 + budget_s * 0.45
    for attempt in range(3):
        if time.time() > d3:
            break
        start = best_h.copy()
        if attempt > 0:
            start = _project(start + rng.normal(0, 0.01, n0), target)
        h_s, c5_s = _slsqp_solve(start, n0, maxiter=500, ftol=1e-15)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    # Phase 4: More coordinate descent after SLSQP (10% budget)
    d4 = t0 + budget_s * 0.55
    h_cd2, c5_cd2 = _coord_desc(best_h, n0, max_sweeps=10000, deadline=d4)
    if c5_cd2 < best_c5:
        best_c5, best_h = c5_cd2, h_cd2.copy()

    # Phase 5: LSE smoothing + SLSQP from perturbed starts (15% budget)
    d5 = t0 + budget_s * 0.70
    scale = 0.04
    for attempt in range(30):
        if time.time() > d5:
            break
        if best_c5 <= 0.3806:
            break
        h_p = best_h + rng.normal(0, scale, n0)
        h_p = _project(h_p, target)
        h_lse, c5_lse = _lse_step(h_p, n0, 30.0, maxiter=60)
        h_s, c5_s = _slsqp_solve(h_lse, n0, maxiter=200, ftol=1e-14)
        h_cd3, c5_cd3 = _coord_desc(h_s, n0, max_sweeps=3000, deadline=d5)
        if c5_cd3 < best_c5 - 1e-10:
            best_c5, best_h = c5_cd3, h_cd3.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.15)

    # Phase 6: Resolution search (15% budget)
    d6 = t0 + budget_s * 0.85
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, -10]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 160:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.018)
        h_r, c5_r = _coord_desc(h_r, n_try, max_sweeps=5000, deadline=sub_d)
        sub_d2 = min(d6, time.time() + budget_s * 0.015)
        h_s, c5_s = _slsqp_solve(h_r, n_try, maxiter=300, ftol=1e-14)
        if c5_s < c5_r:
            h_r, c5_r = h_s.copy(), c5_s
        sub_d3 = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _coord_desc(h_r, n_try, max_sweeps=3000, deadline=sub_d3)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: FTL + final coordinate descent (10% budget)
    d7 = t0 + budget_s * 0.95
    h_ft, c5_ft = _ftl_refine(best_h, best_n, steps=3000, deadline=min(d7, time.time() + budget_s * 0.04))
    if c5_ft < best_c5:
        best_c5, best_h = c5_ft, h_ft.copy()
    h_final, c5_final = _coord_desc(best_h, best_n, max_sweeps=10000, deadline=d7)
    if c5_final < best_c5:
        best_c5, best_h = c5_final, h_final.copy()

    # Final SLSQP polish
    if time.time() < t0 + budget_s * 0.99:
        h_sf, c5_sf = _slsqp_solve(best_h, best_n, maxiter=200, ftol=1e-15)
        if c5_sf < best_c5:
            best_c5, best_h = c5_sf, h_sf.copy()
            h_final2, c5_final2 = _coord_desc(best_h, best_n, max_sweeps=3000,
                                               deadline=t0 + budget_s * 0.995)
            if c5_final2 < best_c5:
                best_c5, best_h = c5_final2, h_final2.copy()

    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
