    best_n = n0

    # Phase 1: Pair optimization (55% budget)
    dl1 = t0 + budget_s * 0.55
    h1, c5_1 = _pair_optimize(h, dl1, max_passes=50000, tol=1e-13, n_grid=128)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-direction refinement (10% budget)
    dl2 = t0 + budget_s * 0.65
    h2, c5_2 = _multi_dir_optimize(best_h, dl2, max_iters=2000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Another pair pass on refined solution (10% budget)
    dl3 = t0 + budget_s * 0.75
    h3, c5_3 = _pair_optimize(best_h, dl3, max_passes=30000, tol=1e-14, n_grid=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Resolution sweep (15% budget)
    dl4 = t0 + budget_s * 0.90
    resolutions = [48, 80, 96, 112, 128, 32, 40, 56, 72, 88, 104, 120, 144]
    for nr in resolutions:
        if time.time() > dl4:
            break
        if nr == best_n:
            continue
        h_r = _resample(best_h, nr)
        sub_dl = min(dl4, time.time() + budget_s * 0.018)
        h_r, c5_r = _pair_optimize(h_r, sub_dl, max_passes=20000, tol=1e-13, n_grid=128)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nr

    # Phase 5: Perturbation restarts (8% budget)
    dl5 = t0 + budget_s * 0.98
    scale = 0.02
    for _att in range(100):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        h_t, c5_t = _pair_optimize(h_pert, sub_dl, max_passes=5000, tol=1e-13, n_grid=128)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.1)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
