def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _grad_matrix(h):

    return best_h, best_c5

def _lp_optimize(h, deadline, top_k=None, radius=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-7:

            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.5)
        else:
            stall += 1
            r *= 0.8
            if stall > 20 and r < 0.02:
                r = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, max_iters=1000000):
    """Simulated annealing on binary version."""
    n = len(h_init)
    if n % 2 != 0:
        # Make it even by adding or removing one element
        if n % 2 == 1:
            h_init = h_init[:-1]
            n = len(h_init)

    h = (h_init >= 0.5).astype(float)

    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng2 = np.random.default_rng(42)
        rng2.shuffle(idx)
        h[idx[ones_count - target:]] = 0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng2 = np.random.default_rng(42)
        rng2.shuffle(idx)
        h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T_start = 0.01
    T_end = 1e-6

        if time.time() > deadline:
            break
        frac = it / max(1, max_iters - 1)
        T = T_start * (T_end / T_start) ** frac
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])

        h[j] = 1.0

                best_h = h.copy()

            h[j] = 0.0

def _pair_polish(h, deadline, max_passes=6):

                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)

                        best_h = h_try.copy()
                        h = h_try.copy()

    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, c5_cur, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            a_j = (1.0 - h[j]) / p[j]
            if a_j < alpha_max:
                alpha_max = a_j
        elif p[j] < -1e-14:
            a_j = h[j] / (-p[j])
            if a_j < alpha_max:
                alpha_max = a_j
    alpha_max = max(alpha_max, 1e-12)
    n_s = 25
    alphas = np.linspace(0, alpha_max, n_s)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_s - 1):
        return best_a, best_c5
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    oa = (lo + hi) / 2.0
    oc = _exact_c5(h + oa * p)
    if oc < best_c5:
        return oa, oc
    return best_a, best_c5

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 80:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.15, 0.10, 0.06, 0.03, 0.015, 0.007]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-15:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 150:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _softmax_descent(h, deadline, n_steps=2000, beta_start=50.0, beta_end=3000.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        shifted = (scores - s_max) * beta / max(s_max, 1e-15)
        w = np.exp(shifted)
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.12, 0.06, 0.03, 0.015, 0.007]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-15:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 120:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_optimize(h, deadline, top_k=None, radius=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.06, 0.5)
        else:
            stall += 1
            r *= 0.75
            if stall > 15 and r < 0.015:
                r = 0.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, max_iters=800000):
    n = len(h_init)
    if n % 2 != 0:
        h_init = h_init[:n-1]
        n = len(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng2 = np.random.default_rng(42)
        rng2.shuffle(idx)
        h[idx[ones_count - target:]] = 0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng2 = np.random.default_rng(42)
        rng2.shuffle(idx)
        h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T_start = 0.008
    T_end = 1e-7
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(1, max_iters - 1)
        T = T_start * (T_end / T_start) ** frac
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-11:
                        continue
                    lo_b, hi_b = 0.0, a_max
                    for _ in range(14):
                        m1 = lo_b + (hi_b - lo_b) / 3.0
                        m2 = hi_b - (hi_b - lo_b) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_b = m2
                        else:
                            lo_b = m1
                    a_opt = (lo_b + hi_b) / 2.0
                    if a_opt < 1e-13:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _topk_subgrad(h, deadline, n_steps=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        lr = 0.08 * (0.002 / 0.08) ** frac
        scores = _scores(h)
        k_top = max(2, int(3 + 5 * (1.0 - frac)))
        idx = np.argsort(scores)[-k_top:]
        w = np.zeros(n_lags)
        w[idx] = 1.0 / k_top
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [lr, lr * 0.5, lr * 0.25]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-15:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 120:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Initial LP (5%)
    dl1 = t0 + budget_s * 0.05
    hdr, c5r = _lp_optimize(h, dl1, top_k=None, radius=0.2, max_iters=800)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 2: P-norm medium (15%)
    dl2 = t0 + budget_s * 0.20
    hdr, c5r = _pnorm_descent(best_h, dl2, p_start=6.0, p_end=512.0, n_steps=4000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 3: Softmax descent (10%)
    dl3 = t0 + budget_s * 0.30
    hdr, c5r = _softmax_descent(best_h, dl3, n_steps=2500, beta_start=80.0, beta_end=3000.0)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 4: Top-k subgradient (10%)
    dl4 = t0 + budget_s * 0.40
    hdr, c5r = _topk_subgrad(best_h, dl4, n_steps=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 5: LP refinement (10%)
    dl5 = t0 + budget_s * 0.50
    hdr, c5r = _lp_optimize(best_h, dl5, top_k=6, radius=0.06, max_iters=2500)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 6: Binary SA (10%)
    dl6 = t0 + budget_s * 0.60
    h_b, c5_b = _binary_sa(best_h, dl6, max_iters=600000)
    if c5_b < best_c5 - 1e-11:
        best_h = h_b.copy()
        best_c5 = c5_b
        # Continuous refinement after binary improvement
        dl6b = min(t0 + budget_s * 0.66, time.time() + budget_s * 0.06)
        hdr, c5r = _lp_optimize(best_h, dl6b, top_k=6, radius=0.1, max_iters=1500)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hdr.copy()

    # Phase 7: P-norm high (8%)
    dl7 = t0 + budget_s * 0.74
    hdr, c5r = _pnorm_descent(best_h, dl7, p_start=128.0, p_end=16384.0, n_steps=1500)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 8: Multi-start (12%)
    dl8 = t0 + budget_s * 0.86
    scale = 0.025
    for attempt in range(600):
        if time.time() > dl8:
            break
        if best_c5 < 0.3805:
            break
        if attempt % 4 == 0:
            h_pert = best_h + rng.normal(0, scale, best_n)
        elif attempt % 4 == 1:
            h_pert = best_h.copy()
            n_flips = max(1, int(best_n * scale * 2))
            flip_idx = rng.choice(best_n, size=n_flips, replace=False)
            for fi in flip_idx:
                h_pert[fi] = 1.0 - h_pert[fi]
        elif attempt % 4 == 2:
            h_pert = best_h + rng.normal(0, scale * 2.0, best_n)
        else:
            # Unround: snap to {0, 0.5, 1} then add noise
            h_pert = np.round(best_h * 2.0) / 2.0
            h_pert += rng.normal(0, scale * 0.5, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl8, time.time() + budget_s * 0.014)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k=5, radius=0.09, max_iters=350)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.87)
        else:
            scale = min(0.12, scale * 1.05)

    # Phase 9: Resolution sweep (6%)
    dl9 = t0 + budget_s * 0.92
    for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12]:
        if time.time() > dl9:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl9, time.time() + budget_s * 0.008)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k=6, radius=0.1, max_iters=200)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 10: Final ultra-fine LP (3%)
    dl10 = t0 + budget_s * 0.95
    hdr, c5r = _lp_optimize(best_h, dl10, top_k=3, radius=0.006, max_iters=4000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 11: Pair polish (4%)
    if time.time() < t0 + budget_s * 0.96:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.998, max_passes=5)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
