    # For m=7: x^7 + x^3 + 1 (one of the primitive polynomials)

    L = (1 << m) - 1

        state = (state >> 1) | (feedback << (m-1))

    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)

    w = np.exp(beta * (scores - max_s))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros_like(scores)
        w[np.argmax(scores)] = 1.0
    else:
        w /= w_sum
    d_range = np.arange(-(n - 1), n, dtype=np.int64)
    grad = np.zeros(n)
    h_pad = np.concatenate([np.zeros(n), h, np.zeros(n)])
    for r in range(len(d_range)):
        d = d_range[r]
        w_r = w[r]
        if w_r < 1e-15:
            continue
        # dF_d/dh_k = dx * [(1-h_{k+d})*1_{valid(k+d)} - h_{k-d}*1_{valid(k-d)}]
        k_plus = np.arange(n) + d
        valid_plus = (k_plus >= 0) & (k_plus < n)
        k_minus = np.arange(n) - d
        valid_minus = (k_minus >= 0) & (k_minus < n)
        term_plus = np.zeros(n)
        term_plus[valid_plus] = 1.0 - h[k_plus[valid_plus]]
        term_minus = np.zeros(n)
        term_minus[valid_minus] = h[k_minus[valid_minus]]
        grad += w_r * dx * (term_plus - term_minus)
    grad -= np.mean(grad)
    return grad
