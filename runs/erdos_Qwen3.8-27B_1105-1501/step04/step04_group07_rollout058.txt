        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _f(c2)

    opt_c5 = _f(opt_a)
    if opt_c5 < best_c5:

    else:

        for a_chk in [a_min, a_max, 0.0]:
            c5_chk = _f(a_chk)
            if c5_chk < best_c5:
                best_a, best_c5 = a_chk, c5_chk

def _compute_B(h, i, j):
    n = len(h)
    n_lags = 2 * n - 1
    d = np.arange(-(n-1), n)  # all shifts
    B = np.zeros(n_lags)
    # h[j-d]: valid when 0 <= j-d < n, i.e., j-n+1 <= d <= j
    mask1 = (j - d >= 0) & (j - d < n)
    B[mask1] += h[j - d[mask1]]
    # -h[i-d]: valid when 0 <= i-d < n
    mask2 = (i - d >= 0) & (i - d < n)
    B[mask2] -= h[i - d[mask2]]
    # -h[i+d]: valid when 0 <= i+d < n
    mask3 = (i + d >= 0) & (i + d < n)
    B[mask3] -= h[i + d[mask3]]
    # h[j+d]: valid when 0 <= j+d < n
    mask4 = (j + d >= 0) & (j + d < n)
    B[mask4] += h[j + d[mask4]]
    return B

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid

    res = target - float(x.sum())
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt

def _compute_B(h, i, j):
    n = len(h)
    d = np.arange(-(n - 1), n, dtype=np.int64)

    m1 = (j - d >= 0) & (j - d < n)
    if m1.any():
        B[m1] += h[j - d[m1]]
    m2 = (i - d >= 0) & (i - d < n)
    if m2.any():
        B[m2] -= h[i - d[m2]]
    m3 = (i + d >= 0) & (i + d < n)
    if m3.any():

    m4 = (j + d >= 0) & (j + d < n)

    C = np.zeros(nl)
    C[n - 1] = -2.0
    c1 = n - 1 + (i - j)
    if 0 <= c1 < nl:
        C[c1] = 1.0
    c2 = n - 1 + (j - i)
    if 0 <= c2 < nl and c2 != c1:

def _pair_best(h, A, B, C, i, j):

    dx = 2.0 / n
    a_lo = max(-h[i], h[j] - 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        s_max = float(scores.max())

        if top_k < nl:
            idx = np.argsort(scores)[-top_k:]
        else:
            idx = np.arange(nl)

        # G[idx, j] = d(score_idx)/d(h_j)
        # score_d = sum h[i+d](1-h[i-d])... this is complex
        # Simpler: use finite differences or the analytical formula
        # For small perturbations dh:
        # score_d(h+dh) ≈ score_d(h) + sum_j G[d,j]*dh[j]
        # G[d,j] = (1-h[j-d])*delta(j+d valid) - h[j+d]*delta(j-d valid) ... 
        # Actually, let's just use the formula:
        # score_d = sum_n h[n+d](1-h[n])
