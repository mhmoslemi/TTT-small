import numpy as np
from scipy.optimize import linprog
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_and_G(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = 2 * n - 1
    G = np.zeros((m, n))
    j = np.arange(n)
    for r in range(m):
        s = r - (n - 1)
        js = j - s
        mask1 = (js >= 0) & (js < n)
        if np.any(mask1):
            G[r, mask1] += (1.0 - h[js[mask1]])
        jp = j + s
        mask2 = (jp >= 0) & (jp < n)
        if np.any(mask2):
            G[r, mask2] -= h[jp[mask2]]
    return corr, G


def _solve_lp_minimax(corr, G, n, radius):
    m = 2 * n - 1
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corr.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-radius, radius)] * n + [(None, None)]
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], float(res.fun)
    return None, float('inf')


def _line_search(h, p, n, cur_c5):
    alpha_max = 1e10
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = min(alpha_max, 1.0)
    if alpha_max < 1e-12:
        return 0.0, cur_c5
    best_a, best_c5 = 0.0, cur_c5
    for t in np.linspace(0.0, 1.0, 10):
        a = t * alpha_max
        if a < 1e-12:
            continue
        h_t = _project(h + a * p, n * 0.5)
        c5_t = _exact_c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.2)
        hi = min(alpha_max, best_a + alpha_max * 0.2)
        if hi - lo < 1e-12:
            return best_a, best_c5
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, n * 0.5))
        f2 = _exact_c5(_project(h + c2 * p, n * 0.5))
        for _ in range(14):
            if hi - lo < 1e-11:

            if f1 < f2:

                c1 = hi - gr * (hi - lo)

                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, n * 0.5))
        a = (lo + hi) / 2.0
        c5_t = _exact_c5(_project(h + a * p, n * 0.5))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_minimax_loop(h, n, deadline, radius_init=0.08, max_iters=50000, stall_limit=100):

    best_h, best_c5 = h.copy(), cur_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _top_k_lp(h, n, K, radius):
    """LP using only top-K tightest constraints for speed at large n."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:K]
    corr_k = corr[top_idx]
    m = len(top_idx)
    G_k = np.zeros((m, n))
    j = np.arange(n)
    for ii, r in enumerate(top_idx):
        s = r - (n - 1)
        js = j - s
        mask1 = (js >= 0) & (js < n)
        if np.any(mask1):
            G_k[ii, mask1] += (1.0 - h[js[mask1]])
        jp = j + s
        mask2 = (jp >= 0) & (jp < n)
        if np.any(mask2):
            G_k[ii, mask2] -= h[jp[mask2]]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -corr_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-radius, radius)] * n + [(None, None)]
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None
