def project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - np.sum(v)
        if abs(diff) < 1e-15:

            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    maxF = F.max()
    w = np.exp(np.clip(beta * (F - maxF), -50, 50))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(F)
        w[F.argmax()] = 1.0
    else:
        w /= s
    g = np.zeros(n)
    for ri in range(2 * n - 1):
        wv = w[ri] * dx
        if wv < 1e-15:
            continue
        r = ri - (n - 1)
        lo = max(0, -r)
        hi = min(n, n - r)
        idx = np.arange(lo, hi)
        g[idx] += wv * (1.0 - h[idx + r])
        idx0 = idx - r
        m0 = (idx0 >= 0) & (idx0 < n)
        g[idx[m0]] -= wv * h[idx0[m0]]
    return g

def lse_descent(h, deadline, beta_start=5, beta_end=400, n_beta=10):

    target = n / 2.0
    best_h = h.copy()

    cur_c5 = best_c5
    for beta in np.geomspace(beta_start, beta_end, n_beta):
        if time.time() > deadline:
            break
        step = 0.05
        stall = 0
        for _ in range(500):
            if time.time() > deadline or stall > 100:
                break

            g -= g.mean()
            gn = np.max(np.abs(g))
            if gn < 1e-15:
                break
            d = -g / gn
            best_a, best_tc = 0.0, cur_c5
            for a in [step, step*0.6, step*0.3, step*0.1]:
                t = project_box_sum(h + a * d, target)
                c = exact_c5(t)
                if c < best_tc - 1e-14:

            if best_a > 0:

                cur_c5 = best_tc
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()
                stall = 0
                step = min(step * 1.08, 0.3)

def pair_descent(h, deadline, rng, max_no_improve=3000):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    no_imp = 0
    while time.time() < deadline and no_imp < max_no_improve:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        if hi < 1e-9 or hi > 1-1e-9 or hj < 1e-9 or hj > 1-1e-9:
            no_imp += 1
            continue
        # Try several transfers
        candidates = [0.03, -0.03, 0.06, -0.06, 0.1, -0.1]
        best_t, best_tc = 0.0, cur_c5
        for t in candidates:
            ni, nj = hi + t, hj - t
            if ni < -1e-10 or ni > 1+1e-10 or nj < -1e-10 or nj > 1+1e-10:
                continue
            h[i], h[j] = ni, nj
            c = exact_c5(h)
            if c < best_tc - 1e-14:

        if best_t != 0:

            h[j] -= best_t
            cur_c5 = best_tc
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            no_imp = 0
        else:
            no_imp += 1
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)
