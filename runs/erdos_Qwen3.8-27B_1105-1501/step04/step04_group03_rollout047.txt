from scipy.optimize import minimize, linprog

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def get_scores(h):
    n = len(h)
    dx = 2.0 / n

def lp_solve(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b > hi_b:
            lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def soft_max_obj(h, tau):
    scores = get_scores(h)
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    return max_s + tau * np.log(Z)

def soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = get_scores(h)
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    G = build_G(h)
    grad = (G.T @ w) * dx
    return grad

def slsqp_smooth(h0, tau, maxiter, deadline):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return soft_max_obj(x, tau)
    def grad(x):
        return soft_max_grad(x, tau)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def pair_move_opt(h, i, j, n_evals=12):
    n = len(h)
    dx = 2.0 / n
    a_min = -h[i] + 1e-15
    a_max = min(h[j], 1.0 - h[i]) - 1e-15
    if a_max - a_min < 1e-13:
        return 0.0, exact_c5(h)
    best_a = 0.0
    best_c5 = exact_c5(h)
    grid = np.linspace(a_min, a_max, n_evals)
    for a in grid:
        if a <= a_min + 1e-14 or a >= a_max - 1e-14:
            continue
        h_t = h.copy()
        h_t[i] += a
        h_t[j] -= a
        c5_t = exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, best_c5
    idx = int(round((best_a - a_min) / (a_max - a_min) * (n_evals - 1)))
    lo = grid[max(0, idx - 1)]
    hi = grid[min(n_evals - 1, idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    def ev(a):
        h_t = h.copy()
        h_t[i] += a
        h_t[j] -= a
        return exact_c5(h_t)
    f1 = ev(c1)
    f2 = ev(c2)
    for _ in range(15):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = ev(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = ev(c2)
    best_a = (lo + hi) / 2.0
    best_c5 = ev(best_a)
    if best_c5 >= exact_c5(h):

    return best_a, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:

    x_old = (np.arange(n_old) + 0.5) / n_old

    h_new = np.interp(x_new, x_old, h_old)

    hard_deadline = t0 + budget_s * 0.96

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: LP optimization (35%)
    d1 = t0 + budget_s * 0.30
    h_lp, c5_lp = lp_loop(h, d1, radius_init=0.15)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: SLSQP with soft-max (20%)
    d2 = t0 + budget_s * 0.45
    tau_list = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
    iter_list = [120, 100, 90, 80, 70, 60, 50, 40]
    h_s = best_h.copy()
    for i in range(len(tau_list)):
        if time.time() > d2:
            break
        h_s = slsqp_smooth(h_s, tau_list[i], iter_list[i], d2)
        h_s = project(h_s, best_n / 2.0)
        c5_s = exact_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: LP again (10%)
    d3 = t0 + budget_s * 0.52
    h_lp2, c5_lp2 = lp_loop(best_h, d3, radius_init=0.1)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 4: Subgradient + momentum (8%)
    d4 = t0 + budget_s * 0.60
    h_sg, c5_sg = subgrad_momentum(best_h, d4, tau=0.05, lr_init=0.001, momentum=0.9, max_iter=3000)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()

    # Phase 5: Greedy pair moves (15%)
    d5 = t0 + budget_s * 0.75
    h_pr, c5_pr = greedy_pairs(best_h, d5, max_passes=500, sample_size=250)
    if c5_pr < best_c5:
        best_c5 = c5_pr
        best_h = h_pr.copy()

    # Phase 6: LP refinement (7%)
    d6 = t0 + budget_s * 0.82
    h_lp3, c5_lp3 = lp_loop(best_h, d6, radius_init=0.06)
    if c5_lp3 < best_c5:
        best_c5 = c5_lp3
        best_h = h_lp3.copy()

    # Phase 7: Pair moves again (7%)
    d7 = t0 + budget_s * 0.89
    h_pr2, c5_pr2 = greedy_pairs(best_h, d7, max_passes=400, sample_size=200)
    if c5_pr2 < best_c5:
        best_c5 = c5_pr2
        best_h = h_pr2.copy()

    # Phase 8: Multi-resolution (3%)
    d8 = t0 + budget_s * 0.93
    offsets = [-8, 8, -12, 12, -16, 16, -24, 24, -36, 36]
    for dn in offsets:
        if time.time() > d8:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d8, time.time() + budget_s * 0.008)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.08)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 9: Final LP micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.02)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final pair polish
    if time.time() < hard_deadline - 5:
        sub_dl = min(hard_deadline, time.time() + 10)
        h_fp, c5_fp = greedy_pairs(best_h, sub_dl, max_passes=100, sample_size=150)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
