def _pair_optimize(h, p, q):
    """Find optimal alpha for pair transfer h[p]+=alpha, h[q]-=alpha."""
    n = len(h)
    V = np.zeros(n)
    V[p] = 1.0
    V[q] = -1.0
    
    # Compute quadratic coefficients using evaluations at alpha=0,1,2
    raw0 = np.correlate(h, 1.0 - h, mode='full')
    h1 = h + V  # alpha=1 (may go out of [0,1] but that's fine for algebra)
    raw1 = np.correlate(h1, 1.0 - h1, mode='full')
    h2 = h + 2*V  # alpha=2
    raw2 = np.correlate(h2, 1.0 - h2, mode='full')
    
    # C = (raw2 - 2*raw1 + raw0) / 2
    # B = raw1 - raw0 - C  (actually: raw1 = A + B + C, so B = raw1 - A - C = raw1 - raw0 - C)
    # A = raw0
    C = (raw2 - 2*raw1 + raw0) / 2.0
    B = raw1 - raw0 - C
    A = raw0
    
    # Feasibility bounds on alpha
    a_min = max(-h[p], h[q] - 1.0)
    a_max = min(1.0 - h[p], h[q])
    
    if a_max <= a_min:
        return 0.0, _exact_c5(h)
    
    # Minimize max_r (A[r] + B[r]*alpha + C[r]*alpha^2) over alpha in [a_min, a_max]
    # Use grid + golden section
    # First, coarse grid
    n_grid = 50
    alphas = np.linspace(a_min, a_max, n_grid)
    best_alpha = a_min
    best_val = np.inf
    for a in alphas:
        vals = A + B*a + C*a*a
        mv = float(np.max(vals))
        if mv < best_val:
            best_val = mv
            best_alpha = a
    
    # Refine with golden section around best
    idx = int(np.argmin([float(np.max(A + B*a + C*a*a)) for a in alphas]))
    if idx == 0:
        lo_g, hi_g = a_min, alphas[1]
    elif idx == n_grid - 1:
        lo_g, hi_g = alphas[-2], a_max
    else:
        lo_g, hi_g = alphas[idx-1], alphas[idx+1]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_g - gr*(hi_g - lo_g)
    c2 = lo_g + gr*(hi_g - lo_g)
    f1 = float(np.max(A + B*c1 + C*c1*c1))
    f2 = float(np.max(A + B*c2 + C*c2*c2))
    for _ in range(40):
        if hi_g - lo_g < 1e-12: break
        if f1 < f2:
            hi_g = c2; f2 = f1; c2 = c1
            c1 = hi_g - gr*(hi_g - lo_g)
            f1 = float(np.max(A + B*c1 + C*c1*c1))
        else:
            lo_g = c1; f1 = f2; c1 = c2
            c2 = lo_g + gr*(hi_g - lo_g)
            f2 = float(np.max(A + B*c2 + C*c2*c2))
    
    best_alpha = (lo_g + hi_g) / 2.0
    best_val = float(np.max(A + B*best_alpha + C*best_alpha*best_alpha))
    
    # Check if this is better than current
    cur_val = float(np.max(raw0))
    if best_val < cur_val - 1e-14:
        h_new = h.copy()
        h_new[p] += best_alpha
        h_new[q] -= best_alpha
        return best_alpha, _exact_c5(h_new)
    return 0.0, _exact_c5(h)

                vq = (q + shifts >= 0) & (q + shifts < n)
                B[vp] += (1.0 - h[p + shifts[vp]])
                B[vq] -= (1.0 - h[q + shifts[vq]])
                vp2 = (p - shifts >= 0) & (p - shifts < n)
                vq2 = (q - shifts >= 0) & (q - shifts < n)
                B[vp2] -= h[p - shifts[vp2]]
                B[vq2] += h[q - shifts[vq2]]
                
                # Optimize alpha
                n_grid = 25
                alphas = np.linspace(a_min, a_max, n_grid)
                best_a = 0.0
                best_val = cur_max_raw
                
                for a in alphas:
                    if a == 0.0:
                        continue
                    vals = raw0 + B * a
                    vals[n-1] += -2.0 * a * a
                    if 0 <= n-1+d < n_lags:
                        vals[n-1+d] += a * a
                    if 0 <= n-1-d < n_lags:
                        vals[n-1-d] += a * a
                    mv = float(np.max(vals))
                    if mv < best_val:
                        best_val = mv
                        best_a = a
                
                if best_val < cur_max_raw - 1e-14 and best_a != 0.0:
                    h[p] += best_a
                    h[q] -= best_a
                    h[p] = np.clip(h[p], 0.0, 1.0)
                    h[q] = np.clip(h[q], 0.0, 1.0)
                    improved = True
                    cur_max_raw = best_val
        
        new_c5 = _exact_c5(h)
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

Let me also think about the binary local search more carefully. For a binary h with n/2 ones:
- A "move" is to swap a 1 at position p with a 0 at position q
- This changes exactly the lags that involve p or q
- The change in score_r is: new_score_r - old_score_r

For binary h, the score at lag r (shift d) is:
score_r = sum_i h[i]*(1-h[i+d])

If we swap h[p]: 1→0 and h[q]: 0→1:
- For lag with shift d:
  - If h[p]=1 and h[p+d]=0: this contributed 1, now h[p]=0 so it contributes 0 → score decreases by 1
  - If h[p]=1 and h[p-d]=0: (the "other direction") this contributed 1, now h[p]=0 → score decreases by 1
  - If h[q]=0 and h[q+d]=1: this contributed 0, now h[q]=1 so it contributes 1 → score increases by 1
  - etc.

This is O(1) per lag per swap, so O(n) per swap to evaluate the full score change. With n/2 * (n/2) possible swaps, one full pass is O(n^3). For n=64: 262K. Very fast.
