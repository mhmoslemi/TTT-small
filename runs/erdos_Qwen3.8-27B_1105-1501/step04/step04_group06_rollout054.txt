import numpy as np
from scipy.optimize import minimize, linprog
import time

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)
            for i in order:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(-x)
            for i in order:
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_and_grad(h, r):
    """Compute corr[r] and its gradient w.r.t. h."""
    n = len(h)
    s = r - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    grad[m1] += 1.0 - h[gs[m1]]
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    grad[m2] -= h[ps[m2]]
    # Compute corr[r]
    valid = (j >= 0) & (j < n) & (j - s >= 0) & (j - s < n)
    corr_val = np.sum(h[j[valid]] * (1 - h[j[valid] - s]))
    return corr_val, grad

def _all_grads(h):
    """Compute gradients of all correlations w.r.t. h."""
    n = len(h)
    j = np.arange(n)
    grads = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        s = r - (n-1)
        g = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        g[m1] += 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        g[m2] -= h[ps[m2]]
        grads[r] = g
    return grads

def _binary_local_search(h_bin, max_iter=5000, deadline=None):
    """Local search on binary sequences: try all pair swaps."""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        if stall > 100:
            break
        improved = False
        # Try all swaps (or a random subset for large n)
        if n <= 64:
            swaps = [(i, j) for i in ones for j in zeros]
            np.random.shuffle(swaps)  # Actually use rng
        else:
            swaps = [(np.random.choice(ones), np.random.choice(zeros)) for _ in range(200)]
        
        for i, j in swaps:
            if h[i] < 0.5 or h[j] > 0.5:
                continue  # Already changed
            h[i], h[j] = h[j], h[i]
            c5_new = _exact_c5(h)
            if c5_new < cur_c5 - 1e-12:
                cur_c5 = c5_new
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break  # Restart scan after improvement
            else:
                h[i], h[j] = h[j], h[i]  # Revert
        if improved:
            stall = 0
        else:
            stall += 1

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}."""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
