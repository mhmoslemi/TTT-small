```python
import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(100):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        adj = diff / cnt
        v[mask] += adj
        v = np.clip(v, 0.0, 1.0)
    return v

def compute_scores(h):
    """Compute all scaled correlation values F_r = dx * corr[r]"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def gradient_F_r(h, d):
    """Compute gradient of F_r where r = n-1+d (d is the shift)"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    ad = abs(d)
    if ad == 0:
        g = dx * (1.0 - 2.0 * h)
        return g
    if d > 0:
        # F = dx * sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
        # dF/dh[j] = dx * [(1-h[j-d]) if d<=j<=n-1] - dx * [h[j+d] if 0<=j<=n-1-d]
        if n - 1 > 0:
            # First term: j from d to n-1
            j1 = np.arange(d, n)
            g[j1] += dx * (1.0 - h[j1 - d])
            # Second term: j from 0 to n-1-d
            j2 = np.arange(0, n - d)
            g[j2] -= dx * h[j2 + d]
    else:
        # d < 0, d' = -d
        d2 = -d
        # F = dx * sum_{i=0}^{n-1-d2} h[i] * (1-h[i+d2])
        # dF/dh[j] = dx * [(1-h[j+d2]) if 0<=j<=n-1-d2] - dx * [h[j-d2] if d2<=j<=n-1]
        j1 = np.arange(0, n - d2)
        g[j1] += dx * (1.0 - h[j1 + d2])
        j2 = np.arange(d2, n)
        g[j2] -= dx * h[j2 - d2]
    return g

def center_normalize(v):
    """Center v (make zero sum) and normalize (max abs = 1)"""
    v = v - np.mean(v)
    m = np.max(np.abs(v))
    if m < 1e-14:
        return None
    return v / m

def feasible_interval(h, v):
    """Find the feasible interval [t_min, t_max] for h + t*v to stay in [0,1]^n"""
    n = len(h)
    t_min = 0.0
    t_max = 0.0
    # For v_i > 0: h_i + t*v_i <= 1 => t <= (1-h_i)/v_i
    # For v_i < 0: h_i + t*v_i >= 0 => t >= -h_i/v_i = h_i/(-v_i)
    pos = v > 1e-14
    neg = v < -1e-14
    if np.any(pos):
        t_max = np.min((1.0 - h[pos]) / v[pos])
    if np.any(neg):
        t_min = np.max(-h[neg] / v[neg])
    # Ensure t_min < t_max
    if t_min >= t_max - 1e-12:
        return None
    return t_min, t_max

def line_search(h, v, n_samples=25, refine_rounds=3, refine_points=9):
    """Exact 1D line search for C5(h + t*v) over feasible interval"""
    n = len(h)
    interval = feasible_interval(h, v)
    if interval is None:
        return 0.0, exact_c5(h), h.copy()
    t_lo, t_hi = interval
    # Add small margin
    margin = 1e-10 * (1.0 + abs(t_hi - t_lo))
    t_lo += margin
    t_hi -= margin
    if t_lo >= t_hi:
        return 0.0, exact_c5(h), h.copy()
    
    # Coarse grid
    ts = np.linspace(t_lo, t_hi, n_samples)
    # Ensure 0 is included
    if t_lo < 0 < t_hi:
        ts = np.unique(np.concatenate([ts, [0.0]]))
    else:
        ts = np.concatenate([ts, [0.0]]) if (abs(0 - t_lo) > abs(0 - t_hi)) else ts
    
    best_t = 0.0
    best_c = exact_c5(h)
    for t in ts:
        h_trial = h + t * v
        # Clip for safety (shouldn't be needed with feasible interval)
        c = exact_c5(h_trial)
        if c < best_c - 1e-14:
            best_c = c
            best_t = t
    
    # Refinement
    if best_t == 0.0:
        return 0.0, best_c, h.copy()
    
    for _ in range(refine_rounds):
        span = t_hi - t_lo
        if span < 1e-10:
            break
        # New interval around best_t
        new_lo = max(t_lo, best_t - span * 0.3)
        new_hi = min(t_hi, best_t + span * 0.3)
        if new_lo >= new_hi:
            break
        ts_ref = np.linspace(new_lo, new_hi, refine_points)
        for t in ts_ref:
            h_trial = h + t * v
            c = exact_c5(h_trial)
            if c < best_c - 1e-14:
                best_c = c
                best_t = t
        t_lo, t_hi = new_lo, new_hi
    
    h_new = h + best_t * v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_box_sum(h_new, n / 2.0)
    final_c = exact_c5(h_new)
    if final_c < best_c:
        return best_t, final_c, h_new
```

```python
        j2 = np.arange(d, n)
```

```python
        j2 = np.arange(0, n - d)
        grad[rp, j2] -= dx * h[j2 + d]
        rn = n - 1 - d
        j1 = np.arange(0, n - d)
        grad[rn, j1] = dx * (1.0 - h[j1 + d])
        j2 = np.arange(d, n)
        grad[rn, j2] -= dx * h[j2 - d]
    return grad


def center_normalize(v):
    v = v - np.mean(v)
    m = np.max(np.abs(v))
    if m < 1e-14:
        return None
    return v / m


def feasible_interval(h, v):
    pos = v > 1e-14
    neg = v < -1e-14
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / v[pos]))
    if np.any(neg):
        t_lo = float(np.max(-h[neg] / v[neg]))
    if t_lo >= t_hi - 1e-12:
        return None
    return t_lo, t_hi


def line_search(h, v, n_coarse=25, refine_rounds=3, refine_pts=11):
    n = len(h)
    interval = feasible_interval(h, v)
    if interval is None:
        return 0.0, h.copy()
    t_lo, t_hi = interval
    t_lo = max(t_lo, -5.0)
    t_hi = min(t_hi, 5.0)
    if t_lo >= t_hi - 1e-10:
        return 0.0, h.copy()
    cur_c = exact_c5(h)
    best_t = 0.0
    best_c = cur_c
    ts = np.linspace(t_lo, t_hi, n_coarse)
    if t_lo < -1e-10 or t_hi < 1e-10:
        ts = np.append(ts, 0.0)
        ts = np.unique(ts)
    for t in ts:
        if abs(t) < 1e-14:
            continue
        ht = h + t * v
        if np.any(ht < -1e-12) or np.any(ht > 1.0 + 1e-12):
            continue
        c = exact_c5(ht)
        if c < best_c - 1e-14:
            best_c = c
            best_t = t
    if best_t == 0.0:
        return 0.0, h.copy()
    for _ in range(refine_rounds):
        span = t_hi - t_lo
        if span < 1e-13:
            break
        frac = 0.3
        new_lo = max(t_lo, best_t - span * frac)
        new_hi = min(t_hi, best_t + span * frac)
        if new_lo >= new_hi - 1e-14:
            break
        ts_r = np.linspace(new_lo, new_hi, refine_pts)
        for t in ts_r:
            if abs(t) < 1e-14:
                continue
            ht = h + t * v
            if np.any(ht < -1e-12) or np.any(ht > 1.0 + 1e-12):
                continue
            c = exact_c5(ht)
            if c < best_c - 1e-14:
                best_c = c
                best_t = t
        t_lo, t_hi = new_lo, new_hi
    if abs(best_t) < 1e-12:
        return 0.0, h.copy()
    h_new = h + best_t * v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_box_sum(h_new, n / 2.0)
    final_c = exact_c5(h_new)
    if final_c < cur_c - 1e-12:
        return best_t, h_new
    return 0.0, h.copy()


def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)


def gen_directions(h, G, scores, rng, n):
    directions = []
    L = 2 * n - 1
    max_s = float(np.max(scores))
    tol = max(5e-4, 0.02 * max_s)
    active = np.where(scores >= max_s - tol)[0]
    active_sorted = active[np.argsort(-scores[active])]
    top_k = min(10, len(active_sorted))
    for beta in [3, 8, 20, 50, 100]:
        w = np.exp(beta * (scores - max_s))
        sw = float(np.sum(w))
        if sw < 1e-30:
            continue
        w = w / sw
        g = G.T @ w
        v = center_normalize(-g)
        if v is not None:
            directions.append(v)
    for r in active_sorted[:top_k]:
        g = G[r]
        v = center_normalize(-g)
        if v is not None:
            directions.append(v)
    w = np.exp(8.0 * (scores - max_s))
    sw = float(np.sum(w))
    if sw > 1e-30:
        w = w / sw
        g_soft = G.T @ w
    else:
        g_soft = np.zeros(n)
    gidx = np.argsort(-np.abs(g_soft))
    for _ in range(16):
        i = gidx[rng.integers(min(20, n))]
        j = gidx[rng.integers(min(20, n))]
        if i == j:
            continue
        if g_soft[i] >= g_soft[j]:
            v = np.zeros(n)
            v[i] = -1.0
            v[j] = 1.0
        else:
            v = np.zeros(n)
            v[i] = 1.0
            v[j] = -1.0
        directions.append(v)
    for _ in range(10):
        i = rng.integers(n)
        j = (i + 1 + rng.integers(0, max(2, n // 8))) % n
        if i == j:
            continue
        d = g_soft[i] - g_soft[j]
        v = np.zeros(n)
        if d >= 0:
            v[i] = -1.0
            v[j] = 1.0
        else:
            v[i] = 1.0
            v[j] = -1.0
        directions.append(v)
    x = (np.arange(n) + 0.5) / n
    for freq in [1.0, 2.0, 3.0]:
        for ph in [0.0, 0.7, 1.4]:
            v = center_normalize(np.sin(2.0 * np.pi * freq * x + ph))
            if v is not None:
                directions.append(v)
    for _ in range(4):
        v = center_normalize(rng.standard_normal(n))
        if v is not None:
            directions.append(v)
    if not directions:
        return directions
    unique = [directions[0]]
    for v in directions[1:]:
        dup = False
        for u in unique:
            nv = float(np.linalg.norm(v))
            nu = float(np.linalg.norm(u))
            if nv < 1e-14 or nu < 1e-14:
                continue
            cos = np.dot(v, u) / (nv * nu)
            if abs(cos) > 0.99:
                dup = True
                break
        if not dup:
            unique.append(v)
    return unique


def polish(h, deadline, rng, max_cycles=500):
    n = len(h)
    cur_c = exact_c5(h)
    best_c = cur_c
    best_h = h.copy()
    stall = 0
    for _ in range(max_cycles):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = compute_scores(h)
        G = compute_all_gradients(h)
        dirs = gen_directions(h, G, scores, rng, n)
        improved = False
        best_dir_c = cur_c
        best_dir_h = h.copy()
        for v in dirs:
            if time.time() > deadline:
                break
            _, h_new = line_search(h, v, n_coarse=21, refine_rounds=3, refine_pts=9)
            if h_new is not h:
                c_new = exact_c5(h_new)
                if c_new < best_dir_c - 1e-12:
                    best_dir_c = c_new
                    best_dir_h = h_new
        if best_dir_c < cur_c - 1e-12:
            h = best_dir_h
            cur_c = best_dir_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            stall = 0
            improved = True
        else:
            stall += 1
            if stall % 3 == 0:
                v = center_normalize(rng.standard_normal(n))
                if v is not None:
                    _, h_p = line_search(h, v, n_coarse=15, refine_rounds=2, refine_pts=7)
                    c_p = exact_c5(h_p)
                    if c_p < cur_c - 1e-12:
                        h = h_p
                        cur_c = c_p
                        if cur_c < best_c:
                            best_c = cur_c
                            best_h = h.copy()
                        stall = 0
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    reserve = max(15.0, 0.05 * budget_s)
    hard_deadline = t0 + budget_s - reserve

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError, Exception):
        h0 = None

    if h0 is None or not np.all(np.isfinite(h0)) or len(h0) < 16:
        n0 = 128
        h0 = np.zeros(n0)
        for i in range(0, n0, 2):
            h0[i] = 1.0

    n0 = len(h0)
    h0 = project_box_sum(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c = exact_c5(h0)
    best_n = n0

    # Phase 1: Main polish on n0
    deadline1 = t0 + budget_s * 0.55
    h1, c1 = polish(best_h, deadline1, rng, max_cycles=500)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Try rolled/reversed variants
    deadline2 = t0 + budget_s * 0.70
    variants = []
    for shift in [n0 // 4, n0 // 3, n0 // 2]:
        v = np.roll(best_h, shift)
        c = exact_c5(v)
        if c < best_c + 0.01:
            variants.append(v.copy())
    v = best_h[::-1].copy()
    cv = exact_c5(v)
    if cv < best_c + 0.01:
        variants.append(v)
    per_var = max(5.0, (deadline2 - time.time()) / max(1, len(variants)))
    for var in variants:
        if time.time() > deadline2:
            break
        dl = min(deadline2, time.time() + per_var)
        hv, cv = polish(var, dl, rng, max_cycles=100)
        if cv < best_c:
            best_c = cv
            best_h = hv.copy()

    # Phase 3: Resolution exploration
    deadline3 = t0 + budget_s * 0.88
    n_candidates = sorted(set([
        best_n - 16, best_n - 8, best_n + 8, best_n + 16,
        best_n + 32, best_n - 32,
        64, 80, 96, 112, 128, 144, 160
    ]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 200]
    n_candidates.sort(key=lambda x: abs(x - best_n))
    remaining = max(0.0, deadline3 - time.time())
    per_n = remaining / max(1, len(n_candidates))
    for n_try in n_candidates:
        if time.time() > deadline3:
            break
        if n_try == best_n:
            continue
        if per_n < 3.0:
            per_n = 3.0
        h_r = resample(best_h, n_try)
        c_r0 = exact_c5(h_r)
        if c_r0 > best_c + 0.005:
            continue
        dl_r = min(deadline3, time.time() + per_n)
        h_r, c_r = polish(h_r, dl_r, rng, max_cycles=80)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Final polish on best
    if time.time() < hard_deadline:
        h_f, c_f = polish(best_h, hard_deadline, rng, max_cycles=300)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c = exact_c5(best_h)
    if final_c <= best_c + 1e-10:
        best_c = final_c
    else:
        # Something went wrong, use the pre-projection value
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c = final_c

    return best_h, best_c, best_n
```
