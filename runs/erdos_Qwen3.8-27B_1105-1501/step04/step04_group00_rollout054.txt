    dx = 2.0 / n

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:n] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:n] += (1.0 - h[:m])
                G[r, :m] += -h[g:g+m]
    return G * dx

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j] + 1e-12), min(radius, 1.0 - h[j] - 1e-12)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, max_alpha=0.5):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] * 0.99)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.99)
    alpha_max = max(alpha_max, 1e-12)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max

    c2 = lo + gr * (hi - lo)

    f2 = _exact_c5(h + c2 * p)

            hi = c2
            f2 = f1

            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _thue_morse_init(n):
    h = np.zeros(n)

        v = i
        c = 0
        while v:
            c ^= 1
            v >>= 1
        h[i] = 1.0 - c

                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _weighted_grad(h, deadline, n_steps=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 5.0 + 200.0 * frac
        scores = _scores(h)
        max_s = np.max(scores)
        scaled = beta * (scores - max_s)
        scaled = np.clip(scaled, -500, 500)
        w = np.exp(scaled)
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.1
        found = False
        for _ in range(35):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                found = True
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP optimization from parent (0-10%)
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _local_lp(h_start, dl1, top_k_list=[None, 20, 12, 6, 4, 3],
                         radius_init=0.25, max_iters=8000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution diverse search (10-50%)
    dl2 = t0 + budget_s * 0.50
    resolutions = [64, 80, 96, 112, 128, 96, 80, 112, 128, 144]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.2928932188]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl2:
            break
        if best_c5 < 0.3804:
            break
        n_starts = 3
        for s in range(n_starts):
            if time.time() > dl2:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_periodic_init(n_try, alpha)
                h_init = h_init + rng.normal(0, 0.05, n_try)
                h_init = _project(h_init, n_try / 2.0)
            else:
                h_init = _thue_morse_init(n_try)
                h_init = h_init * 0.3 + 0.7 * rng.uniform(0, 1, n_try)
                h_init = _project(h_init, n_try / 2.0)

            sub_dl = min(dl2, time.time() + budget_s * 0.03)
            h_try, c5_try = _local_lp(h_init, sub_dl,
                                      top_k_list=[None, 20, 10, 5, 3],
                                      radius_init=0.2, max_iters=6000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                # Follow up with subgradient
                sub_dl2 = min(dl2, time.time() + budget_s * 0.015)
                h_sg, c5_sg = _subgrad_nesterov(h_try, sub_dl2, n_steps=500)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()
                    best_n = n_try
            else:
                sub_dl2 = min(dl2, time.time() + budget_s * 0.01)
                h_wg, c5_wg = _weighted_grad(h_try, sub_dl2, n_steps=100)
                if c5_wg < best_c5:
                    best_c5 = c5_wg
                    best_h = h_wg.copy()
                    best_n = n_try

    # Phase 3: Subgradient + weighted grad on best (50-65%)
    if time.time() < t0 + budget_s * 0.65:
        dl3 = t0 + budget_s * 0.65
        sub_dl = min(dl3, time.time() + budget_s * 0.05)
        h_sg, c5_sg = _subgrad_nesterov(best_h, sub_dl, n_steps=2000,
                                        beta_init=5.0, beta_end=400.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        # LP after subgradient
        if time.time() < dl3:
            sub_dl2 = min(dl3, time.time() + budget_s * 0.04)
            h_lp2, c5_lp2 = _local_lp(best_h, sub_dl2,
                                      top_k_list=[15, 8, 5, 3],
                                      radius_init=0.15, max_iters=5000)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h_lp2.copy()

    # Phase 4: Binary SA + quasi-periodic at specific resolutions (65-80%)
    if time.time() < t0 + budget_s * 0.80:
        dl4 = t0 + budget_s * 0.80
        for n_bin in [80, 96, 112]:
            if time.time() > dl4:
                break
            sub_dl = min(dl4, time.time() + budget_s * 0.03)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=30000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                h_br, c5_br = _local_lp(h_b, sub_dl2,
                                        top_k_list=[15, 8, 5, None],
                                        radius_init=0.12, max_iters=3000)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()
            # Quasi-periodic + LP
            if time.time() < dl4:
                h_qp = _quasi_periodic_init(n_bin, 0.6180339887)
                h_qp += rng.normal(0, 0.03, n_bin)
                h_qp = _project(h_qp, n_bin / 2.0)
                sub_dl3 = min(dl4, time.time() + budget_s * 0.02)
                h_qp2, c5_qp = _local_lp(h_qp, sub_dl3,
                                         top_k_list=[None, 12, 6, 3],
                                         radius_init=0.2, max_iters=4000)
                if c5_qp < best_c5:
                    best_c5 = c5_qp
                    best_h = h_qp2.copy()
                    best_n = n_bin

    # Phase 5: Perturbation restarts (80-93%)
    dl5 = t0 + budget_s * 0.93
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        h_try, c5_try = _local_lp(h_pert, sub_dl,
                                  top_k_list=[10, 5, 3, None],
                                  radius_init=0.08, max_iters=2000)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.88)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 6: Final polish (93-98%)
    if time.time() < t0 + budget_s * 0.98:
        dl6 = t0 + budget_s * 0.98
        hp, c5p = _pair_polish(best_h, dl6, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 7: Tiny LP refinement (98-99.7%)
    if time.time() < t0 + budget_s * 0.997:
        h_fin, c5_fin = _local_lp(best_h, t0 + budget_s * 0.997,
                                  top_k_list=[5, 3, 4],
                                  radius_init=0.01, max_iters=2000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
