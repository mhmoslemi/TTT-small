def _fw_step(h, dx, beta=50):
    n = len(h)
    R = np.correlate(h, 1.0-h, mode='full')
    scores = R * dx
    max_s = np.max(scores)
    # Softmax weights for near-active lags
    w = np.exp(beta * (scores - max_s))
    w /= np.sum(w)
    # Weighted subgradient
    g = np.zeros(n)
    for r in range(2*n-1):
        if w[r] < 1e-14:
            continue
        d = r - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += w[r] * (1.0 - h[d:d+m])
                g[d:] += w[r] * (-h[:m])
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += w[r] * (-h[gg:gg+m])
                g[gg:] += w[r] * (1.0 - h[:m])
    # Oracle: argmin g·x subject to 0≤x≤1, sum(x)=n/2
    # Sort g ascending
    order = np.argsort(g)
    x_star = np.zeros(n)
    x_star[order[:n//2]] = 1.0
    # Frank-Wolfe gap
    fw_gap = np.dot(g, h - x_star)
    return x_star, fw_gap, g

def _exact_c5(h):

def _waterfill(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

def _line_search(h, d, dx, trust):

    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

    if np.any(neg):

    t_lo = max(t_lo, -trust)
    t_hi = min(t_hi, trust)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Coarse grid

    grid = np.linspace(t_lo, t_hi, n_grid)

    best_val = obj_grid[best_i]

            c2 = a + gr * (b - a)

def _grad_matrix(h):
    """Compute gradient of each R_lag with respect to h."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):

        else:

            m = n - g
            if m > 0:
                G[r, :m] += -h[g:g+m]
                G[r, g:] += (1.0 - h[:m])
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _active_lp_direction(h, scores, G, radius, top_k=None):

    # Select top-k lags
    thresh = np.sort(scores)[-top_k][0]

    if not np.any(active):
        active[np.argmax(scores)] = True
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = G[active]
    A_ub_full = np.zeros((int(np.sum(active)), n+1))
    A_ub_full[:, :n] = A_ub
    A_ub_full[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
    bounds.append((None, None))
    try:

                      bounds=bounds, method='highs')
        if res.success:

    except Exception:

    return _lp_direction(h, scores, G, radius)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _waterfill(h_new, n_new / 2.0)

def _svd_directions(h, n_dirs=8):
    """Generate directions from SVD of the gradient matrix."""
    n = len(h)
    G = _grad_matrix(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    # Weight rows by score
    max_s = np.max(scores)
    w = np.exp(20.0 * (scores - max_s))
    w /= np.sum(w)
    Gw = w[:, None] * G
    # SVD
    try:
        U, S, Vt = np.linalg.svd(Gw, full_matrices=False)
        dirs = []
        for i in range(min(n_dirs, Vt.shape[0])):
            d = Vt[i].copy()
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                dirs.append(d / dn)
    except Exception:
        pass
    return dirs

            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += -h[g:g+m]
                G[r, g:] += (1.0 - h[:m])
    return G

def _fw_oracle(g, n):
    order = np.argsort(g)
    x_star = np.zeros(n)
    half = n // 2
    x_star[order[:half]] = 1.0
    return x_star

def _compute_weighted_grad(h, scores, beta):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(scores)] = 1.0
    else:
        w /= wsum
    g = np.zeros(n)
    for r in range(2*n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += w[r] * (1.0 - h[d:d+m])
                g[d:] += w[r] * (-h[:m])
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += w[r] * (-h[gg:gg+m])
                g[gg:] += w[r] * (1.0 - h[:m])
    return g

def _lp_solve(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _active_lp_solve(h, scores, G, radius, top_k=15):
    n = len(h)
    total_lags = 2*n - 1
    if top_k >= total_lags:
        return _lp_solve(h, scores, G, radius)
    sorted_scores = np.sort(scores)
    thresh = sorted_scores[-top_k]
    active = scores >= thresh
    if not np.any(active):
        active[np.argmax(scores)] = True
    n_act = int(np.sum(active))
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_act, n+1))
    A_ub[:, :n] = G[active]
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return _lp_solve(h, scores, G, radius)

def _svd_directions(h, n_dirs=6):
    n = len(h)
    G = _grad_matrix(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_s = np.max(scores)
    w = np.exp(15.0 * (scores - max_s))
    w /= np.sum(w)
    Gw = w[:, None] * G
    try:
        U, S, Vt = np.linalg.svd(Gw, full_matrices=False)
        dirs = []
        for i in range(min(n_dirs, Vt.shape[0])):
            d = Vt[i].copy()
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn > 1e-12:
                dirs.append(d / dn)
    except Exception:
        pass
    return dirs

def _harmon directions(h, n_dirs=4):
    n = len(h)
    dirs = []
    x = np.arange(n)
    for k in range(1, n_dirs+1):
        d = np.sin(2.0 * np.pi * k * x / n)
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for k in range(1, n_dirs+1):
        d = np.cos(2.0 * np.pi * k * x / n)
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    return dirs

def _grad_descent_step(h, scores, beta, trust):
    n = len(h)
    dx = 2.0 / n
    g = _compute_weighted_grad(h, scores, beta)
    g -= g.mean()
    g += 1e-4 * np.random.randn(n) * 0  # no noise, deterministic
    dn = np.linalg.norm(g)
    if dn < 1e-13:
        return None, 0.0, float('inf')
    d = -g / dn * trust
    d -= d.mean()
    alpha, c5_try = _line_search(h, d, dx, trust)
    return d, alpha, c5_try

def _fw_step(h, scores, beta, trust_frac=1.0):
    n = len(h)
    dx = 2.0 / n
    g = _compute_weighted_grad(h, scores, beta)
    x_star = _fw_oracle(g, n)
    d = x_star - h
    dm = float(np.max(np.abs(d)))
    if dm < 1e-10:
        return None, 0.0, float('inf')
    d *= trust_frac
    alpha, c5_try = _line_search(h, d, dx, trust_frac)
    return d, alpha, c5_try

def _pair_step_gradient(h, scores, G, max_tries=20):
    n = len(h)
    dx = 2.0 / n
    max_s = np.max(scores)
    beta = 80.0
    w = np.exp(beta * (scores - max_s))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(scores)] = 1.0
    else:
        w /= wsum
    g = (w[:, None] * G).sum(axis=0)
    order_desc = np.argsort(-g)
    order_asc = np.argsort(g)
    best_alpha = 0.0
    best_c5 = float('inf')
    best_d = None
    for ii in range(min(12, n)):
        for jj in range(min(12, n)):
            i = order_desc[ii]
            j = order_asc[jj]
            if i == j:
                continue
            if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_alpha = alpha
                best_d = d
    return best_d, best_alpha, best_c5

def _combined_loop(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.12
    stall = 0
    stall_lp = 0
    t_start = time.time()
    dur = deadline - t_start
    it = 0
    
    while time.time() < deadline:
        it += 1
        progress = min(1.0, (time.time() - t_start) / max(dur, 1e-8))
        beta = 25.0 + 275.0 * progress
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        scores_dx = scores * (2.0/n)
        
        # Choose method based on stall and iteration
        stall_total = stall
        if stall_total > 30:
            method = 'explore'
        elif stall_total > 10:
            method = 'fw'
        elif it % 10 < 3:
            method = 'lp'
        elif it % 10 < 5:
            method = 'fw'
        elif it % 10 < 7:
            method = 'svd'
        elif it % 10 < 9:
            method = 'grad'
        else:
            method = 'pair'
        
        p = None
        alpha = 0.0
        new_c5 = float('inf')
        
        if method == 'explore':
            # Random perturbation + many directions
            rng_local = np.random.default_rng(it * 7919 + 13)
            for _ in range(8):
                d = rng_local.standard_normal(n)
                d -= d.mean()
                dn = np.linalg.norm(d)
                if dn < 1e-12:
                    continue
                d = d / dn * 0.15
                a2, c5_2 = _line_search(h, d, 2.0/n, 0.15)
                if c5_2 < new_c5:
                    new_c5 = c5_2
                    alpha = a2
                    p = d
            if p is None:
                # Try harmonic
                for hd in _harmon_directions(h, 4):
                    a2, c5_2 = _line_search(h, hd * 0.2, 2.0/n, 0.2)
                    if c5_2 < new_c5:
                        new_c5 = c5_2
                        alpha = a2
                        p = hd * 0.2
                    a2, c5_2 = _line_search(h, -hd * 0.2, 2.0/n, 0.2)
                    if c5_2 < new_c5:
                        new_c5 = c5_2
                        alpha = a2
                        p = -hd * 0.2
            if p is None:
                stall += 2
                radius *= 0.8
                continue
        elif method == 'lp':
            G = _grad_matrix(h)
            p = _active_lp_solve(h, scores, G, radius, top_k=12)
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall_lp += 1
                if stall_lp > 5:
                    radius *= 0.75
                    if radius < 1e-6:
                        break
                continue
            stall_lp = 0
            alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        elif method == 'fw':
            result = _fw_step(h, scores_dx, beta, trust_frac=0.8)
            if result[0] is None:
                stall += 1
                continue
            p, alpha, new_c5 = result
        elif method == 'svd':
            dirs = _svd_directions(h, 5)
            if not dirs:
                stall += 1
                continue
            best_d = None
            best_a = 0.0
            best_v = float('inf')
            for d in dirs:
                a2, c5_2 = _line_search(h, d * 0.25, 2.0/n, 0.25)
                if c5_2 < best_v:
                    best_v = c5_2
                    best_a = a2
                    best_d = d * 0.25
                a2, c5_2 = _line_search(h, -d * 0.25, 2.0/n, 0.25)
                if c5_2 < best_v:
                    best_v = c5_2
                    best_a = a2
                    best_d = -d * 0.25
            if best_d is not None:
                p = best_d
                alpha = best_a
                new_c5 = best_v
            else:
                stall += 1
                continue
        elif method == 'grad':
            result = _grad_descent_step(h, scores_dx, beta, 0.2)
            if result[0] is None:
                stall += 1
                continue
            p, alpha, new_c5 = result
        else:  # pair
            G = _grad_matrix(h)
            result = _pair_step_gradient(h, scores, G)
            if result[0] is None:
                stall += 1
                continue
            p, alpha, new_c5 = result
        
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _waterfill(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.35)
        else:
            stall += 1
            if method == 'lp':
                radius *= 0.88
            if stall > 80:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _waterfill(h_new, n_new / 2.0)

def _exhaustive_pairs(h, deadline, max_passes=25):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i+1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _waterfill(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _fw_loop(h, deadline, beta_start=40, beta_end=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t_start = time.time()
    dur = max(deadline - t_start, 1e-8)
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        progress = min(1.0, (time.time() - t_start) / dur)
        beta = beta_start + (beta_end - beta_start) * progress
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        g = _compute_weighted_grad(h, scores, beta)
        x_star = _fw_oracle(g, n)
        d = x_star - h
        if float(np.max(np.abs(d))) < 1e-10:
            stall += 1
            continue
        alpha, new_c5 = _line_search(h, d, 2.0/n, 1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _waterfill(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _symmetrize(h):
    n = len(h)
    h_sym = (h + h[::-1]) / 2.0
    return _waterfill(h_sym, n/2.0)

def _construct_alt_init(n, seed):
    """Construct alternative initial profiles."""
    rng = np.random.default_rng(seed)
    # Type 1: random smooth
    f = np.random.randn(n, 3)
    x = np.arange(n, dtype=float)
    for k in range(3):
        f[:, k] *= np.sin(2.0*np.pi*(k+1)*x/n + rng.uniform(0, 6.28))
    h = 0.5 + 0.4 * np.tanh(f.sum(axis=1) / (3.0 * np.sqrt(3.0)))
    return _waterfill(h, n/2.0)

def _construct_block_init(n, n_blocks, seed):
    """Construct block-structured initial profile."""
    rng = np.random.default_rng(seed)
    block_size = n // n_blocks
    h = np.zeros(n)
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        val = 1.0 if rng.random() > 0.5 else 0.0
        h[start:end] = val
        # Add some randomness
        h[start:end] += rng.uniform(-0.15, 0.15, end - start)
    h = np.clip(h, 0, 1)
    return _waterfill(h, n/2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _waterfill(h0, n0/2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Try symmetrized
    h_sym = _symmetrize(h0)
    c5_sym = _exact_c5(h_sym)
    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_sym.copy()
    
    # Phase 1: Main combined optimization (45%)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _combined_loop(best_h, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Multi-resolution (20%)
    d2 = t0 + budget_s * 0.65
    n_best = len(best_h)
    n_list = []
    for dn in range(-16, 17, 1):
        if dn == 0:
            continue
        nt = n_best + dn
        if 72 <= nt <= 160:
            n_list.append(nt)
    # Sort by priority: prefer smaller resolutions first (faster)
    n_list.sort(key=lambda x: abs(x - n_best))
    rng.shuffle(n_list)
    
    for nt in n_list:
        if time.time() > d2:
            break
        h_r = _resample(best_h, nt)
        sub_dl = min(d2, time.time() + max(4, budget_s * 0.02))
        h_r, c5_r = _combined_loop(h_r, sub_dl)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 3: Alternative initializations (10%)
    d3 = t0 + budget_s * 0.75
    n_curr = len(best_h)
    alt_seeds = [100, 200, 300, 400, 500, 600, 700, 800]
    for s in alt_seeds:
        if time.time() > d3:
            break
        for constr_type in range(3):
            if time.time() > d3:
                break
            if constr_type == 0:
                h_alt = _construct_alt_init(n_curr, s * 10)
            elif constr_type == 1:
                h_alt = _construct_block_init(n_curr, 4, s * 20)
            else:
                h_alt = _construct_block_init(n_curr, 6, s * 30)
            sub_dl = min(d3, time.time() + max(3, budget_s * 0.008))
            h_t, c5_t = _combined_loop(h_alt, sub_dl)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_t.copy()
    
    # Phase 4: Perturbation restarts (10%)
    d4 = t0 + budget_s * 0.85
    scale = 0.025
    for attempt in range(300):
        if time.time() > d4:
            break
        if best_c5 < 0.3802:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _waterfill(h_pert, len(best_h)/2.0)
        sub_dl = min(d4, time.time() + max(2, budget_s * 0.005))
        h_t, c5_t = _combined_loop(h_pert, sub_dl)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.8)
        else:
            scale = min(0.1, scale * 1.08)
    
    # Phase 5: FW polish (5%)
    d5 = t0 + budget_s * 0.90
    h_f, c5_f = _fw_loop(best_h, d5, beta_start=60, beta_end=500)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    # Phase 6: Exhaustive pair refinement (5%)
    d6 = t0 + budget_s * 0.97
    h_p, c5_p = _exhaustive_pairs(best_h, d6, max_passes=20)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()
    
    # Final: combined polish on remaining time
    d7 = t0 + budget_s * 0.995
    h_f2, c5_f2 = _combined_loop(best_h, d7)
    if c5_f2 < best_c5:
        best_c5 = c5_f2
        best_h = h_f2.copy()
    
    # Final validation
    best_h = _waterfill(best_h, len(best_h)/2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
