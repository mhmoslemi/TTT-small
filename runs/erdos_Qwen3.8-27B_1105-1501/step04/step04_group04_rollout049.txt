O(d) = Σ h[i]*(1-h[i+d]) * dx

This measures "how much of h's mass is at positions where h is 0 at the shifted position." To make this small, we want h[i] and h[i+d] to be both 1 (or both 0) as much as possible.

The optimal trade-off is to have h be a "smooth" function (not 0/1 valued) that varies slowly, so that h[i] ≈ h[i+d] for small d.

If h is very smooth (like a bump function), then h[i] ≈ h[i+d] for small d, so h[i]*(1-h[i+d]) ≈ h[i]*(1-h[i]) which is maximized when h[i] = 1/2. But the integral of h*(1-h) is maximized when h = 1/2 everywhere, giving O(0) = ∫(1/2)(1/2)dx = 1/2.

            m = n - shift

            b = h[shift:shift + m]
            scores[idx] = dx * np.dot(a, 1.0 - b)
            grads[idx, :m] += dx * (1.0 - b)

            scores[idx] = dx * np.dot(a, 1.0 - b)

            grads[idx, :m] += -dx * a
    return scores, grads

def _lp_step(h, K, radius):
    n = len(h)
    target = n / 2.0
    S = _scores_arr(h)
    max_s = float(np.max(S))
    active = np.where(S >= max_s - 0.05 * max_s - 1e-10)[0]
    if len(active) > K:
        top = np.argsort(S[active])[::-1][:K]
        active = active[top]
    F, G = _active_grads_vec(h, active)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi)) + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            h[j] = h[j] + d

        if cn < cur_c5 - 1e-14:
            cur_c5 = cn
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_imp = 0
        else:
            h[i] = old
            h = _project(h, target)
            no_imp += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_escape(h, rng, scale=0.05):
    n = len(h)
    target = n / 2.0
    p = np.zeros(n)
    idx = rng.integers(0, n, size=n // 4 + 1)
    p[idx] = rng.uniform(-scale, scale, size=len(idx))
    p -= np.mean(p)
    return _project(h + p, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except (NameError, TypeError):
        n0 = 64
        h = np.zeros(n0)
        h[0::2] = 1.0

    n0 = len(h)
    h = _project(h, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: LP trust region
    dl1 = t0 + budget_s * 0.28
    h1, c1 = _lp_loop(h, dl1, K=20, radius_start=0.12)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Smoothed subgradient
    dl2 = t0 + budget_s * 0.42
    betas = np.geomspace(5, 500, 10)
    h2, c2 = _smoothed_subgrad(best_h, dl2, betas)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()

    # Phase 3: LP again
    dl3 = t0 + budget_s * 0.56
    h3, c3 = _lp_loop(best_h, dl3, K=25, radius_start=0.08)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3.copy()

    # Phase 4: Subgradient continuation (higher beta)
    dl4 = t0 + budget_s * 0.66
    betas2 = np.geomspace(100, 5000, 8)
    h4, c4 = _smoothed_subgrad(best_h, dl4, betas2)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4.copy()

    # Phase 5: Pair descent
    dl5 = t0 + budget_s * 0.76
    h5, c5 = _pair_descent(best_h, dl5, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()

    # Phase 6: Single coordinate descent
    dl6 = t0 + budget_s * 0.82
    h6, c6 = _single_descent(best_h, dl6, rng)
    if c6 < best_c5:
        best_c5 = c6
        best_h = h6.copy()

    # Phase 7: Multi-resolution
    dl7 = t0 + budget_s * 0.93
    n_tries = set()
    for dn in [-16, -8, -4, 4, 8, 16, 32, -32]:
        nt = best_n + dn
        if 32 <= nt <= 200:
            n_tries.add(nt)
    for nt in [48, 56, 64, 80, 96, 112, 128, 144, 160, 192]:
        if 32 <= nt <= 200:
            n_tries.add(nt)
    n_tries = sorted(n_tries, key=lambda x: abs(x - best_n))
    per_time = max(3.0, (dl7 - time.time()) / max(len(n_tries), 1) * 0.8)
    for nt in n_tries:
        if time.time() > dl7:
            break
        if nt == best_n:
            continue
        hr = _resample(best_h, nt)
        cr = _c5(hr)
        dl_r = min(dl7, time.time() + per_time)
        hr2, cr2 = _lp_loop(hr, dl_r, K=15, radius_start=0.10)
        if cr2 < best_c5 - 1e-10:
            best_c5 = cr2
            best_h = hr2.copy()
            best_n = nt
            # Quick secondary polish at this resolution
            if time.time() < dl7:
                hr3, cr3 = _lp_loop(best_h, min(dl7, time.time() + per_time * 0.4),
                                    K=20, radius_start=0.05)
                if cr3 < best_c5:
                    best_c5 = cr3
                    best_h = hr3.copy()

    # Phase 8: Final LP polish
    dl8 = t0 + budget_s * 0.97
    if time.time() < dl8:
        hf, cf = _lp_loop(best_h, dl8, K=30, radius_start=0.04)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Phase 9: Final pair descent
    dl9 = t0 + budget_s * 0.99
    if time.time() < dl9:
        hl, cl = _pair_descent(best_h, dl9, rng, max_iters=50000)
        if cl < best_c5:
            best_c5 = cl
            best_h = hl.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    if not np.all(best_h >= -1e-12) or not np.all(best_h <= 1.0 + 1e-12):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project(best_h, best_n / 2.0)
        best_c5 = _c5(best_h)

    return best_h, best_c5, best_n
