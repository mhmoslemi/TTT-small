- h[i] += d, h[j] -= d
- Constraint: -h[j] <= d <= 1-h[i] (and 0 <= h[j]-d, 0 <= h[i]+d)
- New corr[r] = corr[r] + (G[r,i] - G[r,j]) * d

I want to minimize max_r (corr[r] + (G[r,i] - G[r,j]) * d) over the feasible range of d.

This is a 1D convex piecewise-linear minimization. The minimum is at a point where the slope changes sign, or at a boundary.

The slope at any point is determined by which constraint is active. The function is:
f(d) = max_r (corr[r] + a_r * d) where a_r = G[r,i] - G[r,j]

The subgradient of f at d is {sum_r w_r * a_r : w in simplex, w_r > 0 only if corr[r] + a_r * d = f(d)}

At the optimum, 0 is in the subgradient, meaning there exist active constraints with both positive and negative slopes (or one with zero slope).

A simple approach: evaluate f at d=0, and at the boundaries, and at the "breakpoints" (where two constraints become simultaneously active). The breakpoints are at d = (corr[r1] - corr[r2]) / (a_r2 - a_r1) for pairs of constraints.

For n=64, there are 127 correlations, so up to 127*126/2 = ~8000 breakpoints. That's manageable.

import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 1e-15:
        return np.zeros(n)
    if target >= n - 1e-15:
        return np.ones(n)
    v = np.asarray(v, dtype=float).copy()
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(x)
        else:
            order = np.argsort(-x)
        for i in order:
            if diff > 0:
                a = min(1.0 - x[i], diff)
                if a <= 1e-15:
                    continue
                x[i] += a
                diff -= a
            else:
                a = min(x[i], -diff)
                if a <= 1e-15:
                    continue
                x[i] -= a
                diff += a
            if abs(diff) <= 1e-12:
                break
    return x


def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    k = np.arange(n)
    for r in range(L):
        s = r - (n - 1)
        ks = k - s
        m1 = (ks >= 0) & (ks < n)
        G[r, m1] += 1.0 - h[ks[m1]]
        kp = k + s
        m2 = (kp >= 0) & (kp < n)
        G[r, m2] -= h[kp[m2]]
    return G


def _slsqp_f(z, n):
    return z[n]


def _slsqp_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _slsqp_ineq_f(z, n):
    h = z[:n]
    t = z[n]
    return t - _corr(h)


def _slsqp_ineq_g(z, n):
    h = z[:n]
    G = _grad_matrix(h)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def _slsqp_eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _slsqp_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _solve_slqp(h_init, n, deadline, maxiter=300):
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _slsqp_ineq_f, 'jac': _slsqp_ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_slsqp_f, z0, jac=_slsqp_jac, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        c = _c5(x)
        return x, c
    except Exception:
        return h, _c5(h)


def _lse_f(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = _corr(h)
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    ev = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(ev)))
    return f


def _lse_jac(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = _corr(h)
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    ev = np.exp(vals)
    w = ev / float(np.sum(ev))
    G = _grad_matrix(h)
    grad = G.T @ w
    return grad


def _solve_lse(h_init, n, alpha, maxiter=100):
    h = np.asarray(h_init, dtype=float).copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_f, h, jac=_lse_jac, args=(n, alpha),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _c5(x)
    except Exception:
        return h, _c5(h)
