        else:

        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

    """Find the interval of t such that 0 <= h + t*v <= 1."""
    pos = v > 1e-15
    neg = v < -1e-15
    t_min = 0.0
    t_max = 0.0
    if np.any(pos):
        t_min = max(t_min, -np.min(h[pos] / v[pos]))
    if np.any(neg):
        t_max = min(t_max, np.min((1.0 - h[neg]) / (-v[neg])))
    return t_min, t_max

def line_minimize_1d(h, v, n_samples=33, n_refine=3):
    """Minimize C5(h + t*v) over feasible t. Returns (t_best, c_best)."""
    t_min, t_max = feasible_interval(h, v)
    if t_max - t_min < 1e-12:
        return 0.0, exact_c5(h)
    
    # Coarse grid
    ts = np.linspace(t_min, t_max, n_samples)
    cvals = np.array([exact_c5(h + t * v) for t in ts])
    
    # Find best region
    idx = np.argmin(cvals)
    
    # Refine around best
    for _ in range(n_refine):
        lo = max(t_min, ts[max(0, idx-2)])
        hi = min(t_max, ts[min(len(ts)-1, idx+2)])
        if hi - lo < 1e-10:
            break
        ts = np.linspace(lo, hi, n_samples)
        cvals = np.array([exact_c5(h + t * v) for t in ts])

    """Compute weighted gradient of active correlation terms."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1

        w_masked[top_idx] = weights[top_idx]

def _grad_weighted(h, sc, beta):

def _grad_lag(h, k):

    v[j] -= h[i]
    v = -v

    m = np.abs(v).max()

    return v / m
