def _project(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target} via bisection on threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    for dn in [1, -1, 2, -2, 3, -3, 4, -4, 6, -6, 8, -8]:
        if time_left() < 5:
            break
        n_try = n + dn
        if n_try < 16 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_deadline = min(d5, time.time() + budget_s * 0.015)
        # Optimize at new resolution
        corr_r = _corr_scaled(h_r)
        active_r = _get_active(corr_r, frac=0.3, min_count=8)
        h_r2 = _slsqp_with_active(h_r, active_r, n_try, n_try / 2.0, sub_deadline, maxiter=200, ftol=1e-12)
        h_r3 = _slsqp_with_lse(h_r2, 0.008, n_try, n_try / 2.0, sub_deadline, maxiter=100, ftol=1e-12)
        c5_r = min(_exact_c5(h_r2), _exact_c5(h_r3))
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r2 if _exact_c5(h_r2) <= _exact_c5(h_r3) else h_r3
            n = n_try

    # === Phase 6: Final polish (remaining time) ===
    if time_left() > 5:
        corr = _corr_scaled(best_h)
        active = _get_active(corr, frac=0.5, min_count=10)
        h_f = _slsqp_with_active(best_h, active, n, n / 2.0, deadline, maxiter=400, ftol=1e-14)
        c5_f = _exact_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # Final gradient descent
        if time_left() > 5:
            hg, c5g = _grad_desc_lse(best_h, 0.003, n, n / 2.0, 0.003, deadline, max_steps=50, patience=15)
            if c5g < best_c5:
                best_c5 = c5g
                best_h = hg.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
