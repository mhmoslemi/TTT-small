def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_active(h, lags):
    n = len(h)
    m = len(lags)
    G = np.zeros((m, n))
    for idx in range(m):
        d = int(lags[idx]) - (n - 1)
        if d >= 0:
            lo1, hi1 = 0, n - 1 - d
            if lo1 <= hi1:

            if lo1 <= hi1:

            lo2, hi2 = 0, n - 1 - g

def _line_search(h, d, dx, t_max):
    n = len(h)
    S0 = _corr(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15

    t_hi = t_max
    if np.any(pos):

    if t_hi <= t_lo + 1e-15:

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)

    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx

def _lp_step(h, radius, top_k=25):
    n = len(h)
    dx = 2.0 / n
    scores = _corr(h)
    max_s = float(np.max(scores))
    if top_k < len(scores):
        active = np.argsort(scores)[-top_k:]
    else:
        active = np.arange(len(scores))
    G = _grad_active(h, active)
    m = len(active)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = max_s - scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, 0.0

def _softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = _corr(h) * dx
    max_c = float(np.max(corr))
    shifted = tau * (corr - max_c)
    m = float(np.max(shifted))
    exp_s = np.exp(shifted - m)
    w = exp_s / float(np.sum(exp_s))
    top_idx = np.argsort(w)[-15:]
    G = _grad_active(h, top_idx)
    grad = (w[top_idx] * dx) @ G
    grad -= np.mean(grad)
    return grad, max_c
