import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _score_vec_fast(h, V):
    """Compute score_d for h + alpha*V as quadratic coefficients (A_d, B_d, C_d)
    where score_d(alpha) = A_d + alpha*B_d - alpha^2*C_d"""
    n = len(h)
    n_lags = 2 * n - 1
    # A_d = sum_i h[i](1-h[i+d])
    A = np.correlate(h, 1.0 - h, mode='full')
    # B_d = sum_i [V[i](1-h[i+d]) - h[i]V[i+d]]
    B1 = np.correlate(V, 1.0 - h, mode='full')
    B2 = np.correlate(h, V, mode='full')
    B = B1 - B2
    # C_d = sum_i V[i]V[i+d]
    C = np.correlate(V, V, mode='full')
    dx = 2.0 / n
    return A * dx, B * dx, C * dx

def _max_quadratic(A, B, C, alpha):
    """Compute max_d (A_d + alpha*B_d - alpha^2*C_d)"""
    return float(np.max(A + alpha * B - alpha * alpha * C))

def _minimize_max_quad(A, B, C, a_lo, a_hi, n_grid=40):
    """Minimize max of quadratics over [a_lo, a_hi] using grid + golden section."""
    alphas = np.linspace(a_lo, a_hi, n_grid)
    vals = np.array([_max_quadratic(A, B, C, a) for a in alphas])
    bi = int(np.argmin(vals))
    if bi == 0:
        return a_lo, float(vals[0])
    if bi == n_grid - 1:
        return a_hi, float(vals[-1])
    lo = float(alphas[bi - 1])
    hi = float(alphas[bi + 1])
    best_c = float(vals[bi])
    best_a = float(alphas[bi])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _max_quadratic(A, B, C, c1)
    f2 = _max_quadratic(A, B, C, c2)
    for _ in range(35):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _max_quadratic(A, B, C, c1)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _max_quadratic(A, B, C, c2)
    mid = (lo + hi) / 2.0
    fm = _max_quadratic(A, B, C, mid)
    if fm < best_c:
        return mid, fm
    return best_a, best_c

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

            break
        improved = False

                    pairs.append((i, j))
        rng.shuffle(pairs)
        # Limit pairs per round for speed
        max_pairs = min(len(pairs), 500)
        for (p, q) in pairs[:max_pairs]:

            if f_opt < cur_c5 - 1e-14:

                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
            # Reverse: h[q] += a, h[p] -= a
            a_max_r = min(1.0 - h[q], h[p])
            if a_max_r < 1e-12:
                continue
            V[p] = -1.0
            V[q] = 1.0

def _multi_coordinate_descent(h, deadline, max_rounds=100, k_coords=3):

    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n
    
    # Phase 1: LP refinement (8%)
    dl1 = t0 + budget_s * 0.08
    hdr, c5r = _lp_optimize(h, dl1, top_k=None, radius=0.15, max_iters=800)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 2: P-norm descent main (25%)
    dl2 = t0 + budget_s * 0.33
    hdr, c5r = _pnorm_descent(best_h, dl2, p_start=4.0, p_end=4096.0, n_steps=2500)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 3: Pair sweep (12%)
    dl3 = t0 + budget_s * 0.45
    hdr, c5r = _pair_sweep(best_h, dl3, max_rounds=30)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 4: Active set polish (10%)
    dl4 = t0 + budget_s * 0.55
    hdr, c5r = _active_set_polish(best_h, dl4, max_rounds=300)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 5: Multi-coordinate descent (8%)
    dl5 = t0 + budget_s * 0.63
    hdr, c5r = _multi_coordinate_descent(best_h, dl5, max_rounds=150, k_coords=3)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 6: LP fine-tune (8%)
    dl6 = t0 + budget_s * 0.71
    hdr, c5r = _lp_optimize(best_h, dl6, top_k=8, radius=0.04, max_iters=2000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 7: Multi-start perturbation (15%)
    dl7 = t0 + budget_s * 0.86
    scale = 0.02
    n_pert = len(best_h)
    for attempt in range(500):
        if time.time() > dl7:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, n_pert)
        h_pert = _project(h_pert, n_pert / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.015)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k=8, radius=0.05, max_iters=200)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.06)
    
    # Phase 8: Resolution sweep (6%)
    dl8 = t0 + budget_s * 0.92
    best_n_trial = best_n
    for dn in [-6, -4, -2, 2, 4, 6, 8, 12, -8, 16]:
        if time.time() > dl8:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl8, time.time() + budget_s * 0.01)
        # Quick optimization at new resolution
        h_r2, c5_r2 = _lp_optimize(h_r, sub_dl, top_k=8, radius=0.08, max_iters=150)
        if c5_r2 < best_c5 - 1e-12:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = n_try
    
    # Phase 9: Final pair sweep + active set (6%)
    dl9 = t0 + budget_s * 0.96
    hdr, c5r = _pair_sweep(best_h, dl9, max_rounds=20)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 10: Ultra-fine LP (4%)
    dl10 = t0 + budget_s * 0.995
    hdr, c5r = _lp_optimize(best_h, dl10, top_k=4, radius=0.01, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
