def _quasi_periodic(n, alpha):
    """Create a quasi-periodic binary pattern."""
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return h.astype(float)

def _block_pattern(n, block_sizes):
    """Create a periodic block pattern."""
    total = sum(block_sizes)
    h = np.zeros(n)
    pos = 0
    for i, bs in enumerate(block_sizes):
        x_start = int(round(pos / total * n))
        x_end = int(round((pos + bs) / total * n))
        if i % 2 == 0:
            h[x_start:x_end] = 1.0
        pos += bs
    return _project(h, n / 2.0)

One more idea: the **dual variable** approach. In the minimax problem, at the optimum, there exist non-negative weights λ_r (summing to 1) on the active constraints such that:
