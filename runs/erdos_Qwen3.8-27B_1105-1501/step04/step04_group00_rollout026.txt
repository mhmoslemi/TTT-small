                G[r, d:] += (1.0 - h[:m])  # first factor: j from d to n-1, h[j-d] from h[0] to h[m-1]

        h2b, c5_2b = _local_lp(best_h, dl2b, max_iters=1000, radius_init=0.1)
        if c5_2b < best_c5:
            best_c5, best_h = c5_2b, h2b.copy()

    # Phase 3: Multi-resolution search
    dl3 = deadline_at(0.55)
    for n_try in [96, 128, 80, 112, 64, 144, 128, 160, 96, 128]:
        if time.time() > dl3:
            break
        if best_c5 <= 0.3805:
            break
        h_init = _resample(best_h, n_try)
        sub_dl = min(dl3, time.time() + remaining() * 0.12)
        h_try, c5_try = _local_lp(h_init, sub_dl, max_iters=2500, radius_init=0.18)
        if c5_try < best_c5:
            best_c5, best_h, best_n = c5_try, h_try.copy(), n_try
        else:
            sub_dl2 = min(dl3, time.time() + remaining() * 0.06)
            h_sm, c5_sm = _smooth_grad_step(h_try, sub_dl2, beta_start=60, beta_end=500, n_steps=100)
            if c5_sm < best_c5:
                best_c5, best_h, best_n = c5_sm, h_sm.copy(), n_try
            if time.time() < dl3:
                sub_dl3 = min(dl3, time.time() + remaining() * 0.04)
                h_try2, c5_try2 = _local_lp(h_sm, sub_dl3, max_iters=1000, radius_init=0.08)
                if c5_try2 < best_c5:
                    best_c5, best_h, best_n = c5_try2, h_try2.copy(), n_try

    # Phase 4: Diverse restarts (quasi-periodic, binary SA, random)
    if time.time() < deadline_at(0.72):
        dl4 = deadline_at(0.75)
        alphas = [0.6180339887, 0.4142135624, 0.5, 0.3819660113, 0.7071067812]
        for n_try in [best_n, 96, 64]:
            if time.time() > dl4:
                break
            for ai in range(2):
                if time.time() > dl4:
                    break
                h_q = _quasi_init(n_try, alphas[(n_try + ai) % len(alphas)])
                h_q = _project(h_q + rng.normal(0, 0.05, n_try), n_try / 2.0)
                sub_dl = min(dl4, time.time() + remaining() * 0.05)
                h_try, c5_try = _local_lp(h_q, sub_dl, max_iters=1500, radius_init=0.2)
                if c5_try < best_c5:
                    best_c5, best_h, best_n = c5_try, h_try.copy(), n_try
        for n_bin in [best_n, 96]:
            if time.time() > dl4:
                break
            sub_dl = min(dl4, time.time() + remaining() * 0.04)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=20000)
            if c5_b < best_c5 + 0.01:
                h_b_cont = (1 - 0.04) * h_b + 0.02
                h_b_cont = _project(h_b_cont, n_bin / 2.0)
                sub_dl2 = min(dl4, time.time() + remaining() * 0.03)
                h_b2, c5_b2 = _local_lp(h_b_cont, sub_dl2, max_iters=800, radius_init=0.12)
                if c5_b2 < best_c5:
                    best_c5, best_h, best_n = c5_b2, h_b2.copy(), n_bin

    # Phase 5: Basin hopping on best solution
    if time.time() < deadline_at(0.88):
        dl5 = deadline_at(0.90)
        scale = 0.025
        for attempt in range(200):
            if time.time() > dl5:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl5, time.time() + remaining() * 0.015)
            h_try, c5_try = _local_lp(h_pert, sub_dl, max_iters=400, radius_init=0.06)
            if c5_try < best_c5 - 1e-13:
                best_c5, best_h = c5_try, h_try.copy()
                scale = max(0.005, scale * 0.82)
            else:
                scale = min(0.08, scale * 1.08)

    # Phase 6: Final intensive LP polish
    if time.time() < deadline_at(0.96):
        dl6 = deadline_at(0.97)
        h_fin, c5_fin = _local_lp(best_h, dl6, max_iters=5000, radius_init=0.03)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin.copy()
        # Smooth gradient fine-tune
        if time.time() < deadline_at(0.985):
            h_sg, c5_sg = _smooth_grad_step(best_h, deadline_at(0.985),
                                            beta_start=200, beta_end=1000, n_steps=100)
            if c5_sg < best_c5:
                best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 7: Pair polish
    if time.time() < deadline_at(0.995):
        hp, c5p = _pair_polish(best_h, deadline_at(0.995), max_passes=4)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
