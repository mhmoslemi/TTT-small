from scipy.optimize import minimize


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_scaled(h):
    """Compute the scaled correlation vector (len 2n-1)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(v, target):
    """Project v onto {x : 0<=x_i<=1, sum(x)=target} via bisection on shift."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _lp_vertex(w, target):
    """Solve min w·s s.t. 0<=s<=1, sum(s)=target. Returns a vertex."""
    n = len(w)
    idx = np.argsort(w, kind='stable')
    s = np.zeros(n)
    if isinstance(target, float) and abs(target - int(target)) < 1e-12:
        target = int(target)
    s[idx[:target]] = 1.0
    return s


def _grad_corr_q(h, q):
    """Gradient of the q-th scaled correlation w.r.t. h. q is index in [0, 2n-2]."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    d = q - (n - 1)
    if d >= 0:
        # C_q = dx * sum_{i=0}^{n-d-1} h_{i+d} * (1 - h_i)
        if d < n:
            A = np.arange(d, n)
            B = np.arange(0, n - d)
            g[A] += dx * (1.0 - h[B])
            g[B] -= dx * h[A]
    else:
        k = -d
        if k < n:
            A = np.arange(0, n - k)
            B = np.arange(k, n)
            g[A] += dx * (1.0 - h[B])
            g[B] -= dx * h[A]
    return g


def _fw_line_search(h, s, target):
    """Exact line search along h -> s. Returns best h_alpha and its score."""
    n = len(h)
    # Coarse grid
    alphas = np.linspace(0.0, 1.0, 51)
    scores = np.zeros(len(alphas))
    for i, a in enumerate(alphas):
        h_a = (1.0 - a) * h + a * s
        scores[i] = _exact_c5(h_a)
    
    best_i = int(np.argmin(scores))
    best_alpha = alphas[best_i]
    best_score = scores[best_i]
    best_h = (1.0 - best_alpha) * h + best_alpha * s
    
    # Refine around best
    if 0 < best_i < len(alphas) - 1:
        lo_a = alphas[max(0, best_i - 1)]
        hi_a = alphas[min(len(alphas) - 1, best_i + 1)]
        fine_alphas = np.linspace(lo_a, hi_a, 21)
        for a in fine_alphas:
            h_a = (1.0 - a) * h + a * s
            sc = _exact_c5(h_a)
            if sc < best_score:
                best_score = sc
                best_alpha = a
                best_h = (1.0 - a) * h + a * s
    
    # Ensure feasibility
    best_h = _project(best_h, target)
    return best_h, best_score


def _fw_step(h, target, eta, max_active, deadline=None):
    """One Frank-Wolfe step with softmax-weighted subgradient."""
    n = len(h)
    corr = _corr_scaled(h)
    lam = float(np.max(corr))
    
    # Softmax weights over all constraints (or top ones)
    # Use temperature to focus on active set
    diff = corr - lam
    weights = np.exp(diff / eta)
    total_w = float(np.sum(weights))
    if total_w < 1e-30:
        weights = np.zeros_like(corr)
        top_idx = np.argsort(corr)[-max_active:]
        weights[top_idx] = 1.0 / max_active
    else:
        weights /= total_w
    
    # Compute subgradient
    w_sub = np.zeros(n)
    # Only compute gradients for constraints with significant weight
    active = np.where(weights > 1e-6)[0]
    for q in active:
        w_sub += weights[q] * _grad_corr_q(h, q)
    
    # LP oracle: find vertex s minimizing w_sub · s
    s = _lp_vertex(w_sub, target)
    
    # Line search
    h_new, score_new = _fw_line_search(h, s, target)
    return h_new, score_new, lam


def _excess_gradient_step(h, target, T, top_k, deadline=None):
    """Excess penalty step: reduce correlations that exceed target T."""
    n = len(h)
    corr = _corr_scaled(h)
    excess = np.maximum(corr - T, 0.0)
    
    if float(np.sum(excess)) < 1e-14:
        return h, _exact_c5(h)
    
    # Softmax weights over excess
    # Focus on top excess constraints
    sorted_idx = np.argsort(corr)[-top_k:]
    ex = excess[sorted_idx]
    if float(np.sum(ex)) < 1e-14:
        return h, _exact_c5(h)
    
    w = ex / float(np.sum(ex))
    
    # Gradient
    g = np.zeros(n)
    for q in sorted_idx:
        g += w[list(sorted_idx).index(q)] * _grad_corr_q(h, q)
    
    # Projected gradient step with multiple step sizes
    dx = 2.0 / n
    norm_g = float(np.linalg.norm(g))
    if norm_g < 1e-14:
        return h, _exact_c5(h)
    
    g = g / norm_g  # normalize
    cur_score = _exact_c5(h)
    best_h = h
    best_score = cur_score
    
    for scale_factor in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
        step = scale_factor * dx
        h_try = _project(h - step * g, target)
        sc = _exact_c5(h_try)
        if sc < best_score - 1e-13:
            best_score = sc
            best_h = h_try
    
    return best_h, best_score


def _pair_line_search(h, i, j, target, deadline=None):
    """Exact line search over pair (i,j) keeping h_i + h_j constant."""
    p = h[i] + h[j]
    lo = max(0.0, p - 1.0)
    hi = min(1.0, p)
    if hi - lo < 1e-10:
        return h, _exact_c5(h)
    
    cur_score = _exact_c5(h)
    best_t = h[i]
    best_score = cur_score
    
    # Grid search
    grid = np.linspace(lo, hi, 25)
    for t in grid:
        h[i] = t
        h[j] = p - t
        sc = _exact_c5(h)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    # Refine
    if best_t != h[i]:
        # Find neighborhood
        idx = int(np.argmin(np.abs(grid - best_t)))
        lo_r = grid[max(0, idx - 1)]
        hi_r = grid[min(len(grid) - 1, idx + 1)]
        fine = np.linspace(lo_r, hi_r, 15)
        for t in fine:
            h[i] = t
            h[j] = p - t
            sc = _exact_c5(h)
            if sc < best_score:
                best_score = sc
                best_t = t
    
    h[i] = best_t
    h[j] = p - best_t
    return h, best_score


def _pair_refine(h, target, deadline, max_passes=50):

    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Precompute influence: which pairs have large gradient contribution
    # Use a simple heuristic: pairs with both values in (0,1)

    for _pass in range(max_passes):

        improved = False
        # Shuffle for diversity
        import random
        random.shuffle(interior)

            if time.time() > deadline:
                break

            for j in interior:

                    improved = True
                    # Recompute interior
                    interior = [x for x in range(n) if 1e-8 < h[x] < 1.0 - 1e-8]
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5

            g[B] -= dx * h[A]
    return g


def _fw_line_search(h, s, target, n_coarse=31):
    n = len(h)
    alphas = np.linspace(0.0, 1.0, n_coarse)
    best_score = _exact_c5(h)
    best_a = 0.0
    for a in alphas[1:]:
        h_a = (1.0 - a) * h + a * s
        sc = _exact_c5(h_a)
        if sc < best_score:
            best_score = sc
            best_a = a
    if best_a > 0.0:
        idx = int(round(best_a * (n_coarse - 1)))
        lo_a = alphas[max(0, idx - 1)]
        hi_a = alphas[min(n_coarse - 1, idx + 1)]
        for a in np.linspace(lo_a, hi_a, 17):
            if a < 1e-15:
                continue
            h_a = (1.0 - a) * h + a * s
            sc = _exact_c5(h_a)
            if sc < best_score:
                best_score = sc
                best_a = a
    h_new = (1.0 - best_a) * h + best_a * s
    h_new = _project(h_new, target)
    return h_new, best_score


def _fw_step(h, target, eta_rel, max_active, deadline):
    n = len(h)
    corr = _corr_scaled(h)
    lam = float(np.max(corr))
    eta = max(eta_rel * max(lam, 1e-6), 1e-8)
    diff = (corr - lam) / eta
    w_softmax = np.exp(np.minimum(diff, 0.0))
    tw = float(np.sum(w_softmax))
    if tw > 1e-30:
        w_softmax /= tw
    else:
        top = np.argsort(corr)[-max_active:]
        w_softmax = np.zeros_like(corr)
        w_softmax[top] = 1.0 / max_active
    g = np.zeros(n)
    active = np.where(w_softmax > 1e-7)[0]
    for q in active:
        g += w_softmax[q] * _grad_q(h, q)
    s = _lp_vertex(g, target)
    h_new, sc_new = _fw_line_search(h, s, target)
    return h_new, sc_new


def _fw_multi_temp(h, target, base_eta, deadline):
    n = len(h)
    best_h = h.copy()
    best_sc = _exact_c5(h)
    multipliers = [0.4, 0.6, 1.0, 1.5, 2.5, 4.0]
    for m in multipliers:
        if time.time() > deadline:
            break
        h_try, sc_try = _fw_step(h, target, base_eta * m, 8, deadline)
        if sc_try < best_sc - 1e-13:
            best_sc = sc_try
            best_h = h_try
    return best_h, best_sc


def _pair_line(h, i, j, target):
    p = h[i] + h[j]
    lo = max(0.0, p - 1.0)
    hi = min(1.0, p)
    if hi - lo < 1e-9:
        return _exact_c5(h)
    orig_i, orig_j = h[i], h[j]
    best_score = _exact_c5(h)
    best_t = orig_i
    grid = np.linspace(lo, hi, 21)
    for t in grid:
        h[i], h[j] = t, p - t
        sc = _exact_c5(h)
        if sc < best_score:
            best_score = sc
            best_t = t
    if best_t != orig_i:
        idx = int(np.argmin(np.abs(grid - best_t)))
        lo_r = grid[max(0, idx - 1)]
        hi_r = grid[min(len(grid) - 1, idx + 1)]
        for t in np.linspace(lo_r, hi_r, 13):
            h[i], h[j] = t, p - t
            sc = _exact_c5(h)
            if sc < best_score:
                best_score = sc
                best_t = t
    h[i], h[j] = best_t, p - best_t
    return best_score


def _pair_refine(h, target, deadline, n_passes=60):
    n = len(h)
    h = h.copy()
    best = _exact_c5(h)
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = [i for i in range(n) if 1e-6 < h[i] < 1.0 - 1e-6]
        if len(interior) < 2:
            break
        np.random.shuffle(interior)
        for i in interior:
            if time.time() > deadline:
                break
            if not (1e-6 < h[i] < 1.0 - 1e-6):
                continue
            for j in interior:
                if j == i or not (1e-6 < h[j] < 1.0 - 1e-6):
                    continue
                sc = _pair_line(h, i, j, target)
                if sc < best - 1e-12:
                    best = sc
                    improved = True
                    interior = [x for x in range(n) if 1e-6 < h[x] < 1.0 - 1e-6]
                    break
            if improved:
                break
        if not improved:
            break
    return h, best


def _excess_step(h, target, T, top_k, deadline):
    n = len(h)
    corr = _corr_scaled(h)
    excess = np.maximum(corr - T, 0.0)
    if float(np.max(excess)) < 1e-14:
        return h, _exact_c5(h)
    idx = np.argsort(corr)[-top_k:]
    ex = excess[idx]
    if float(np.sum(ex)) < 1e-14:
        return h, _exact_c5(h)
    w = ex / float(np.sum(ex))
    g = np.zeros(n)
    for ii, q in enumerate(idx):
        g += w[ii] * _grad_q(h, q)
    cur = _exact_c5(h)
    best_h, best_sc = h, cur
    norm = float(np.linalg.norm(g))
    if norm < 1e-14:
        return h, cur
    g = g / norm
    dx = 2.0 / n
    for sf in [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.5]:
        h_try = _project(h - sf * dx * g, target)
        sc = _exact_c5(h_try)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_h = h_try
    return best_h, best_sc


def _slsqp_polish(h, target, maxiter=200, ftol=1e-13, deadline=None):
    from scipy.optimize import minimize
    n = len(h)
    corr_full = _corr_scaled(h)
    max_corr = float(np.max(corr_full))
    threshold = max_corr * 0.7
    active = list(np.where(corr_full >= threshold)[0])
    if len(active) < 5:
        active = list(np.argsort(corr_full)[-15:])

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def ineq_fun(x):
        hh = x[1:]
        c = _corr_scaled(hh)
        return x[0] - c[active]

    def ineq_jac(x):
        hh = x[1:]
        J = np.zeros((len(active), n + 1))
        J[:, 0] = 1.0
        for ai, q in enumerate(active):
            J[ai, 1:] = -_grad_q(hh, q)
        return J

    def eq_fun(x):
        return x[1:].sum() - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    x0 = np.zeros(n + 1)
    x0[0] = max_corr * 1.002 + 0.01
    x0[1:] = h
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[1:], target)
            sc = _exact_c5(h_new)
            if sc < _exact_c5(h) - 1e-14:
                return h_new, sc
    except Exception:
        pass
    return h, _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _perturb_reoptimize(h, target, sigma, deadline, n_fw=10):
    n = len(h)
    rng = np.random.default_rng()
    h_p = _project(h + rng.normal(0, sigma, n), target)
    tr = target
    for _ in range(n_fw):
        if time.time() > deadline:
            break
        h_p, sc = _fw_step(h_p, tr, 0.02, 6, deadline)
    sc = _exact_c5(h_p)
    if sc < _exact_c5(h) - 1e-12:
        return h_p, sc
    return h, _exact_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    h_best = _project(h0, target)
    best_c5 = _exact_c5(h_best)

    # Phase 1: FW multi-temperature (25% budget)
    d1 = t0 + budget_s * 0.25
    eta_base = 0.035
    stall = 0
    for it in range(300):
        if time.time() > d1:
            break
        if stall > 40:
            break
        h_new, sc_new = _fw_multi_temp(h_best, target, eta_base, d1)
        if sc_new < best_c5 - 1e-13:
            best_c5 = sc_new
            h_best = h_new
            stall = 0
            eta_base = max(0.006, eta_base * 0.97)
        else:
            stall += 1
            if stall % 10 == 0:
                eta_base = min(0.06, eta_base * 1.1)

    # Phase 2: Pair refinement (15% budget)
    d2 = t0 + budget_s * 0.40
    h_p, c5_p = _pair_refine(h_best, target, d2, n_passes=80)
    if c5_p < best_c5:
        best_c5 = c5_p
        h_best = h_p

    # Phase 3: Excess penalty (15% budget)
    d3 = t0 + budget_s * 0.55
    T = best_c5 * 1.0005
    gamma = 0.0003
    for it in range(800):
        if time.time() > d3:
            break
        T = max(best_c5 * 0.985, T - gamma)
        h_new, sc_new = _excess_step(h_best, target, T, 12, d3)
        if sc_new < best_c5 - 1e-13:
            best_c5 = sc_new
            h_best = h_new
        else:
            gamma = max(5e-6, gamma * 0.85)

    # Phase 4: SLSQP polish (10% budget)
    d4 = t0 + budget_s * 0.65
    if time.time() < d4:
        h_s, sc_s = _slsqp_polish(h_best, target, maxiter=300, ftol=1e-14, deadline=d4)
        if sc_s < best_c5:
            best_c5 = sc_s
            h_best = h_s

    # Phase 5: FW + pair alternating (15% budget)
    d5 = t0 + budget_s * 0.80
    for it in range(80):
        if time.time() > d5:
            break
        h_new, sc_new = _fw_step(h_best, target, 0.012, 6, d5)
        if sc_new < best_c5 - 1e-13:
            best_c5 = sc_new
            h_best = h_new
        else:
            sub_d = min(d5, time.time() + budget_s * 0.01)
            h_p2, c5_p2 = _pair_refine(h_best, target, sub_d, n_passes=20)
            if c5_p2 < best_c5 - 1e-13:
                best_c5 = c5_p2
                h_best = h_p2
            else:
                # Try perturbation
                if time.time() < d5:
                    h_pt, sc_pt = _perturb_reoptimize(h_best, target, 0.02, d5, n_fw=8)
                    if sc_pt < best_c5 - 1e-12:
                        best_c5 = sc_pt
                        h_best = h_pt
                    else:
                        break

    # Phase 6: Resolution search (12% budget)
    d6 = deadline - 15
    n_cur = len(h_best)
    resolutions = [-4, 4, -2, 2, -6, 6, -8, 8, -12, 12, 16, -16, 20]
    for dn in resolutions:
        if time.time() > d6:
            break
        n_try = n_cur + dn
        if n_try < 20 or n_try > 128:
            continue
        h_r = _resample(h_best, n_try)
        tr = n_try / 2.0
        sub_d = min(d6, time.time() + budget_s * 0.012)
        for _ in range(30):
            if time.time() > sub_d:
                break
            h_r2, sc_r2 = _fw_step(h_r, tr, 0.02, 6, sub_d)
            if sc_r2 < _exact_c5(h_r) - 1e-10:
                h_r = h_r2
            else:
                break
        sub_d2 = min(d6, time.time() + budget_s * 0.005)
        h_r, sc_r = _pair_refine(h_r, tr, sub_d2, n_passes=15)
        if sc_r < best_c5 - 1e-8:
            best_c5 = sc_r
            h_best = h_r
            n_cur = n_try

    # Phase 7: Final polish (8% budget)
    d7 = deadline - 2
    if time.time() < d7:
        h_fin, sc_fin = _pair_refine(h_best, target, d7, n_passes=50)
        if sc_fin < best_c5:
            best_c5 = sc_fin
            h_best = h_fin
    if time.time() < deadline - 1:
        h_sl, sc_sl = _slsqp_polish(h_best, target, maxiter=200, ftol=1e-14, deadline=deadline - 1)
        if sc_sl < best_c5:
            best_c5 = sc_sl
            h_best = h_sl

    h_best = _project(h_best, len(h_best) / 2.0)
    final_c5 = _exact_c5(h_best)
    return h_best, final_c5, len(h_best)
