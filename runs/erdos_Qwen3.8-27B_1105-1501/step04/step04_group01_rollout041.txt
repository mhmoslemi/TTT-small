```python
def _evaluate_flip(h, i, corr):
    """Evaluate the effect of flipping h[i] (0->1 or 1->0) on the correlation."""
    n = len(h)
    old = h[i]
    new = 1.0 - old
    delta = new - old  # +1 or -1
    
    # corr[r] = sum_x h[x]*(1-h[x+d]) where d = r-(n-1)
    # The change: for each valid (x, x+d) pair involving position i:
    # - If x = i: term was old*(1-h[i+d]), becomes new*(1-h[i+d]), change = delta*(1-h[i+d])
    # - If x+d = i: term was h[x]*(1-old), becomes h[x]*(1-new), change = -delta*h[x]
    
    # For shift d (r = n-1+d):
    # valid pairs: max(0,-d) <= x < min(n, n-d)
    # position i is involved if:
    #   - i >= max(0,-d) and i < min(n, n-d) [as x]
    #   - i-d >= max(0,-d) and i-d < min(n, n-d) [as x+d, i.e., x = i-d]
    
    # Let me compute the delta for each shift:
    delta_corr = np.zeros(2*n-1)
    for r in range(2*n-1):
        d = r - (n-1)
        # As x = i: valid if 0 <= i < n and 0 <= i+d < n
        if 0 <= i+d < n:
            # old contribution: h[i]*(1-h[i+d]), new: (1-h[i])*(1-h[i+d])
            # delta = (1-h[i])*(1-h[i+d]) - h[i]*(1-h[i+d]) = (1-2*h[i])*(1-h[i+d])
            delta_corr[r] += (1.0 - 2.0*old) * (1.0 - h[i+d])
        # As x+d = i, so x = i-d: valid if 0 <= i-d < n
        if 0 <= i-d < n:
            # old contribution: h[i-d]*(1-h[i]), new: h[i-d]*(1-(1-h[i])) = h[i-d]*h[i]
            # delta = h[i-d]*h[i] - h[i-d]*(1-h[i]) = h[i-d]*(2*h[i]-1)
            delta_corr[r] += h[i-d] * (2.0*old - 1.0)
    
    new_corr = corr + delta_corr
    new_c5 = float(np.max(new_corr)) * (2.0/n)
    return new_c5, new_corr, delta_corr
```

```python
def _sa_optimize(h, deadline, n_target=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    best_h = h.copy()
    
    # Ensure h is 0-1
    h = (h > 0.5).astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    best_h = h.copy()
    
    # Precompute for each position i, the ranges of shifts it affects
    # _single_delta can be computed quickly
    
    rng = np.random.default_rng(42)
    T = 0.005
    T_min = 1e-6
    cool_rate = 0.99995
    
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        di = 1.0 - 2.0 * h[i]
        dj = 1.0 - 2.0 * h[j]
        
        # Compute delta
        delta = np.zeros(2*n-1)
        # _single_delta for i
        d1 = np.arange(-i, n-i)
        r1 = n-1 + d1
        delta[r1] += di * (1.0 - h[i + d1])
        d2 = np.arange(i-n+1, i+1)
        r2 = n-1 + d2
        delta[r2] += -di * h[i - d2]
        # _single_delta for j
        d1j = np.arange(-j, n-j)
        r1j = n-1 + d1j
        delta[r1j] += dj * (1.0 - h[j + d1j])
        d2j = np.arange(j-n+1, j+1)
        r2j = n-1 + d2j
        delta[r2j] += -dj * h[j - d2j]
        # Cross term
        delta[n-1+(j-i)] += -di*dj
        delta[n-1+(i-j)] += -di*dj
        
        new_c5 = float(np.max(corr + delta)) * dx
        dc5 = new_c5 - cur_c5
        
        if dc5 < 0 or (T > T_min and rng.random() < np.exp(-dc5 / T)):
            h[i] += di  # h[i] becomes 1-h[i]
            h[j] += dj
            corr = corr + delta
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        
        T *= cool_rate
    
    return best_h, best_c5
```

    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

        else:
