import cvxpy as cp

def qp_direction(h, scores, G, radius, active_idx=None):
    n = len(h)
    if active_idx is not None:
        G_sub = G[active_idx, :]
        b_sub = -scores[active_idx]
    else:
        G_sub = G
        b_sub = -scores
    p = cp.Variable(n)
    t = cp.Variable()
    obj = cp.Minimize(cp.sum_squares(p) + 100*t)
    cons = [G_sub @ p <= b_sub + t,
            cp.sum(p) == 0]
    for j in range(n):
        cons.append(p[j] >= max(-radius, -h[j]) - 1e-8)
        cons.append(p[j] <= min(radius, 1-h[j]) + 1e-8)
    prob = cp.Problem(obj, cons)
    prob.solve(solver=cp.ECOS, max_iter=500)
    if p.value is not None:
        return np.asarray(p.value).flatten()
    return None

Let me also try: **using the exact subgradient of the max function.**

The subgradient of f(h) = max_k f_k(h) at h is:
∂f(h) = convex hull of {∇f_k(h) : f_k(h) = f(h)}

If multiple k achieve the max, we can use their average gradient, which is in the subdifferential. This gives a potentially better direction than just the LP.

2. Identify "active" lags (within some tolerance of the max)
3. Compute the average gradient over active lags
4. Take a step in the negative gradient direction
5. Project
6. Line search

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    v = v.copy()
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(x)
    if abs(r) > 1e-12:
        if r > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0: x[mask] += r / cnt
    x = np.clip(x, 0.0, 1.0)
    r2 = target - np.sum(x)
    if abs(r2) > 1e-14:
        idx = int(np.argmax(1.0 - x)) if r2 > 0 else int(np.argmin(x))
        x[idx] = np.clip(x[idx] + r2, 0, 1)
    return x

def active_subgrad(h, tol=0.001):
    """Compute subgradient of max_k f_k(h) at h, using active lags within tol of max."""
    n = len(h)
    dx = 2.0 / n
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = scores_raw * dx
    max_s = np.max(scores)
    active = np.where(scores >= max_s - tol)[0]
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        ad = abs(d)
        m = n - ad
        if ad == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:m] += dx * (1.0 - h[d:d+m])
            grad[d:] -= dx * h[:m]
        else:
            g = -d
            grad[g:] += dx * (1.0 - h[:m])
            grad[:m] -= dx * h[g:]
    return grad / len(active), scores, active

def lse_val_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = scores_raw * dx
    max_s = np.max(scores)
    exp_s = np.exp((scores - max_s) / tau)
    Z = np.sum(exp_s)
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12: continue
        d = k - (n - 1)
        ad = abs(d)
        m = n - ad
        if ad == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:m] += wk * dx * (1.0 - h[d:d+m])
            grad[d:] -= wk * dx * h[:m]
        else:
            g = -d
            grad[g:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[g:]
    return val, grad

def projected_grad_step(h, alpha, grad, target):
    """Take a gradient step and project."""
    h_new = h - alpha * grad
    return project(h_new, target)

def backtracking_line_search(h, direction, c5_cur, dx_step, target, max_iter=30):
    """Line search along direction from h, with projection."""
    n = len(h)
    # Find max step size to stay in [0,1]
    alpha_max = np.inf
    for j in range(n):
        if direction[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / direction[j])
        elif direction[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-direction[j]))
    alpha_max = max(min(alpha_max, dx_step), 1e-10)
    
    best_a = 0.0

    alpha = dx_step

            best_a = alpha
        alpha *= 0.7
        if alpha < 1e-10:
            break
    return best_a, best_c5, project(h + best_a * direction, target)

def lse_optimize(h, target, deadline, tau_init=0.3, tau_final=0.01, max_iters=2000):

def build_corr_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        ad = abs(d)
        m = n - ad
        if ad == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[r, :m] += dx * (1.0 - h[d:d+m])
            G[r, d:] += -dx * h[:m]
        else:
            g = -d
            G[r, g:] += dx * (1.0 - h[:m])
            G[r, :m] += -dx * h[g:]
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lb = max(-radius, -h[j])
        ub = min(radius, 1.0 - h[j])
        bounds.append((lb, ub))
    bounds.append((0, None))  # t >= 0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old: return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = v.copy()
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(x)
    if abs(r) > 1e-12:
        if r > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += r / cnt
    x = np.clip(x, 0.0, 1.0)
    r2 = target - np.sum(x)
    if abs(r2) > 1e-14:
        idx = int(np.argmax(1.0 - x)) if r2 > 0 else int(np.argmin(x))
        x[idx] = np.clip(x[idx] + r2, 0, 1)
    return x

def lse_val_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = scores_raw * dx
    max_s = np.max(scores)
    exp_s = np.exp((scores - max_s) / tau)
    Z = np.sum(exp_s)
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        if d == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += wk * dx * (1.0 - h[d:d + m])
            grad[d:] -= wk * dx * h[:m]
        else:
            g = -d
            m = n - g
            grad[g:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[g:]
    return val, grad

def build_corr_grad(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            G[r, :m] += dx * (1.0 - h[d:d + m])
            G[r, d:] += -dx * h[:m]
        else:
            g = -d
            m = n - g
            G[r, g:] += dx * (1.0 - h[:m])
            G[r, :m] += -dx * h[g:]
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lb = max(-radius, -h[j] - 1e-10)
        ub = min(radius, 1.0 - h[j] + 1e-10)
        bounds.append((lb, ub))
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search_lp(h, p, c5_cur, target):
    n = len(h)
    alpha_max = np.inf
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(min(alpha_max, 2.0), 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(20):
        a = alpha_max * (i + 1) / 20.0
        h_t = project(h + a * p, target)
        c5_t = exact_c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = 0.0
    hi_a = alpha_max
    best_mid = best_a
    for _ in range(25):
        mid1 = lo_a + (hi_a - lo_a) * 0.382
        mid2 = lo_a + (hi_a - lo_a) * 0.618
        f1 = exact_c5(project(h + mid1 * p, target))
        f2 = exact_c5(project(h + mid2 * p, target))
        if f1 < f2:
            hi_a = mid2
            if f1 < best_c5 - 1e-15:
                best_c5 = f1
                best_mid = mid1
        else:
            lo_a = mid1
            if f2 < best_c5 - 1e-15:
                best_c5 = f2
                best_mid = mid2
        if hi_a - lo_a < 1e-14:
            break
    final_a = (lo_a + hi_a) / 2.0
    final_c5 = exact_c5(project(h + final_a * p, target))
    if final_c5 < best_c5 - 1e-15:
        best_c5 = final_c5
        best_mid = final_a
    return best_mid, best_c5

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 120:
            break
        scores_raw = np.correlate(h, 1.0 - h, mode='full')
        scores = scores_raw * (2.0 / n)
        G = build_corr_grad(h)
        p = lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.65
            if radius < 1e-9:
                break
            continue
        alpha, new_c5 = line_search_lp(h, p, cur_c5, target)
        if alpha > 0 and new_c5 < cur_c5 - 1e-13:
            h = project(h + alpha * p, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 25 == 0 and radius < 0.015:
                radius = 0.06
    return best_h, best_c5

def lse_projected_grad(h, target, deadline, tauSchedule, max_iters):
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha_bb = 0.1
    prev_h = h.copy()
    prev_grad = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(it / max(max_iters - 1, 1), 1.0)
        idx = int(frac * (len(tauSchedule) - 1))
        if idx >= len(tauSchedule) - 1:
            tau = tauSchedule[-1]
        else:
            t_frac = frac * (len(tauSchedule) - 1) - idx
            tau = tauSchedule[idx] * (1 - t_frac) + tauSchedule[idx + 1] * t_frac
        val, grad = lse_val_grad(h, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        if prev_grad is not None:
            s = h - prev_h
            y = grad - prev_grad
            ys = np.dot(s, y)
            yy = np.dot(y, y)
            ss = np.dot(s, s)
            if yy > 1e-16 and ys > 1e-16:
                alpha_bb = max(min(ss / ys, 2.0 / (yy + 1e-8)), 0.001)
            elif ss > 1e-16:
                alpha_bb = max(ss / max(ys, 1e-6), 0.01)
        alpha = min(alpha_bb, 0.5)
        h_new = project(h - alpha * grad, target)
        c5_new = exact_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            prev_h = h.copy()
            prev_grad = grad.copy()
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            alpha_bb *= 0.6
            prev_h = h.copy()
            prev_grad = grad.copy()
            h_t = project(h - alpha_bb * grad, target)
            c5_t = exact_c5(h_t)
            if c5_t < cur_c5 - 1e-15:
                h = h_t
                cur_c5 = c5_t

def pair_polish(h, deadline, max_passes=80, rng=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    if rng is None:
        rng = np.random.default_rng(0)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        rng.shuffle(indices)
        for ia in range(n):
            if time.time() > deadline:
                break
            ii = indices[ia]
            for jb in range(n):
                if ia == jb:
                    continue
                jj = indices[jb]
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[jj], 1.0 - best_h[ii])
                    else:
                        amax = min(1.0 - best_h[jj], best_h[ii])
                    if amax < 1e-12:
                        continue
                    lo_a = 0.0
                    hi_a = amax
                    best_local = best_c5
                    best_alpha = 0.0
                    for _bs in range(18):
                        mid_a = (lo_a + hi_a) * 0.5
                        if mid_a < 1e-14:
                            hi_a = mid_a
                            continue
                        h_t = best_h.copy()
                        h_t[ii] += sign * mid_a
                        h_t[jj] -= sign * mid_a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_local - 1e-14:
                            best_local = c5_t
                            best_alpha = mid_a
                            lo_a = mid_a
                        else:
                            hi_a = mid_a
                    if best_alpha > 0 and best_local < best_c5 - 1e-14:
                        best_h[ii] += sign * best_alpha
                        best_h[jj] -= sign * best_alpha
                        best_c5 = best_local
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:

    x_old = (np.arange(n_old) + 0.5) / n_old

    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)
