def _binary_local_search(h, deadline, max_iters=1000000):
    n = len(h)
    h = (h >= 0.5).astype(float)
    cur_scores = _scores_vec(h)
    cur_max = cur_scores.max()
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Find active shifts (top few)
        top_k = np.argsort(cur_scores)[-5:]
        
        # Generate candidate swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try swaps from active shifts
        candidates = []
        for k_abs in top_k:
            d = k_abs - (n - 1)  # actual shift
            if d > 0:
                # Problem pairs: i where h[i]=0, h[i+d]=1
                for i in range(n - d):
                    if h[i] == 0 and h[i+d] == 1:
                        # Direct swap: h[i]=1, h[i+d]=0
                        candidates.append((i, i+d))
            elif d < 0:
                dd = -d
                for i in range(n - dd):
                    if h[i] == 1 and h[i+dd] == 0:
                        candidates.append((i+dd, i))
        
        # Also add some random swaps
        for _ in range(20):
            i = zeros[rng.integers(len(zeros))]
            j = ones[rng.integers(len(ones))]
            candidates.append((i, j))
        
        # Try each candidate
        best_swap = None
        best_new_max = cur_max
        for (i, j) in candidates[:50]:  # Limit to 50 per iteration
            if i == j:
                continue
            if h[i] != 0 or h[j] != 1:
                continue
            h[i] = 1
            h[j] = 0
            new_scores = _scores_vec(h)
            new_max = new_scores.max()
            h[i] = 0
            h[j] = 1
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 1
            h[j] = 0
            cur_scores = _scores_vec(h)
            cur_max = cur_scores.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            # No improvement found, try a random perturbation
            if rng.random() < 0.1:
                i = zeros[rng.integers(len(zeros))]
                j = ones[rng.integers(len(ones))]
                h[i] = 1
                h[j] = 0
                cur_scores = _scores_vec(h)
                cur_max = cur_scores.max()
    
    return best_h, best_max * (2.0/n)
