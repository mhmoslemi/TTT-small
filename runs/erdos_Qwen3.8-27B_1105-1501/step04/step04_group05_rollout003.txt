    target = n // 2
    order = np.argsort(w, kind='stable')
    s = np.zeros(n)

    """Line search along h -> s. Returns best alpha and point."""
    alphas = np.linspace(0.0, 1.0, n_alphas)
    best_alpha = 0.0
    best_val = float('inf')
    for a in alphas:
        h_t = (1.0 - a) * h + a * s

    max_c = corr.max()

    weights = np.exp(shifted)
    total_w = weights.sum()
    if total_w < 1e-30:
        return h, max_c, False
    weights /= total_w

def pair_refine(h, target, deadline, max_passes=50):
    """Pairwise coordinate exchange refinement"""
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try all pairs (or a random subset for large n)
        if n > 50:
            rng_p = np.random.default_rng(_pass)
            pairs = [(int(a), int(b)) for a, b in 
                     zip(*rng_p.choice(n, size=(2, 400), replace=False).T) if a < b]
        else:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        
        for i, j in pairs:
            if time.time() > deadline:
                break
            p = best_h[i] + best_h[j]
            lo = max(0.0, p - 1.0)
            hi = min(1.0, p)
            if hi - lo < 1e-10:
                continue
            # Binary search for best split
            # C5 as function of z = h[i] (h[j] = p - z) is piecewise quadratic
            # Just do a fine grid
            grid = np.linspace(lo, hi, 25)
            best_z = best_h[i]
            best_val = best_c5
            for z in grid:
                if z < lo - 1e-10 or z > hi + 1e-10:
                    continue
                h_t = best_h.copy()
                h_t[i] = z
                h_t[j] = p - z
                v = exact_c5(h_t)
                if v < best_val:
                    best_val = v
                    best_z = z
            if best_val < best_c5 - 1e-12:
                best_h[i] = best_z
                best_h[j] = p - best_z
                best_c5 = best_val
                improved = True
    
        if not improved:
            break
    
    return best_h, best_c5
