import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_score(h, lag):
    """Gradient of score at given lag with respect to h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += -dx * h[lag:lag+m]
            g[lag:] += dx * (1.0 - h[:m])
    else:
        d = -lag
        m = n - d
        if m > 0:
            g[:m] += dx * (1.0 - h[d:d+m])
            g[d:] += -dx * h[:m]
    return g

def lp_direction(h, active_scores, active_grads, radius):
    """Find LP direction to reduce active scores."""
    n = len(h)
    na = len(active_scores)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((na, n + 1))
    A_ub[:, :n] = active_grads
    A_ub[:, n] = -1.0
    b_ub = -active_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, p, n_eval=30):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j] + 1e-10) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, (h[j] - 1e-10) / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    alphas = np.linspace(0, alpha_max, n_eval)
    vals = np.array([exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_eval - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_eval - 1, best_i + 1)]
    gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr_phi * (hi - lo)
    c2 = lo + gr_phi * (hi - lo)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr_phi * (hi - lo)
            f1 = exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr_phi * (hi - lo)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def local_optimize(h, deadline, top_k=4, radius_init=0.05, max_iters=5000):
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = compute_scores(h)
        order = np.argsort(scores)[::-1]
        ak = order[:top_k]
        active_scores = scores[ak]
        active_grads = np.array([grad_score(h, int(r) - (n-1)) for r in ak])
        p = lp_direction(h, active_scores, active_grads, radius)
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * p
            h_new = project(h_new, n / 2.0)
            h = h_new
            cur_c5 = exact_c5(h)
            stall = 0
            radius = min(radius * 1.03, 0.3)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            radius *= 0.7
            if stall > 10 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = float(c & 1)
    return t

def golden_ratio_pattern(n, alpha):
    x = np.arange(n) / n
    return ((x * alpha) % 1.0)

def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def active_set_optimize(h, deadline, top_k_seq=None, radius_init=0.03):
    """Iteratively optimize with decreasing active set sizes."""
    if top_k_seq is None:
        top_k_seq = [5, 4, 3, 3, 2, 3, 4, 3]
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    for idx, tk in enumerate(top_k_seq * 20):
        if time.time() > deadline:
            break
        h_res, c5_res = local_optimize(h, deadline, top_k=tk,
                                        radius_init=radius_init, max_iters=300)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            cur_c5 = c5_res
        h = h_res
        if c5_res >= cur_c5 - 1e-13 and idx > 5:
            if time.time() > deadline - 5:
                break
    return best_h, best_c5

def basin_jump(h, deadline, rng, n_jumps=50):
    """Try large perturbations to escape local minima."""
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    scale = 0.15
    jump_count = 0
    for _ in range(n_jumps):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = project(h_pert, n / 2.0)
        sub_dl = min(deadline, time.time() + budget_remaining(deadline, 0.02))
        h_try, c5_try = local_optimize(h_pert, sub_dl, top_k=3,
                                        radius_init=0.04, max_iters=500)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            jump_count += 1
            scale = max(0.02, scale * 0.8)
        else:
            scale = min(0.25, scale * 1.15)
        if jump_count > 10 and scale < 0.03:
            break
    return best_h, best_c5

def budget_remaining(deadline, frac):
    return max(deadline - time.time(), 0.0) * frac

def score_targeting_step(h, deadline, max_steps=200):
    """Target the single worst score with greedy coordinate moves."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    for step in range(max_steps):
        if time.time() > deadline:
            break
        scores = compute_scores(h)
        worst_idx = int(np.argmax(scores))
        worst_lag = worst_idx - (n - 1)
        worst_val = scores[worst_idx]
        g = grad_score(h, worst_lag)
        # Find direction that reduces worst score most
        p = -g
        p -= p.mean()
        p_norm = np.linalg.norm(p)
        if p_norm < 1e-15:
            break
        p /= p_norm
        alpha, new_c5 = line_search(h, p, n_eval=20)
        if new_c5 < cur_c5 - 1e-13:
            h_new = project(h + alpha * p, n / 2.0)
            h = h_new
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5
