    idx = d_range + (n-1)  # correlation indices

    beta = np.zeros(L)

        d_a = np.arange(d_lo_a, d_hi_a+1)
        idx_a = d_a + (n-1)
        alpha[idx_a] = 1.0 - h[j + d_a]

    d_lo_b = max(-(n-1), j-n+1)
    d_hi_b = min(n-1, j)
    if d_lo_b <= d_hi_b:
        d_b = np.arange(d_lo_b, d_hi_b+1)
        idx_b = d_b + (n-1)
        beta[idx_b] = h[j - d_b]

    # F_k(1) = corr[k]*dx + (1-h_j)*slopes[k]

    #     slope_k = slopes[k]

            hi = mid

            lo = mid

    # Clamp
    t_opt = max(0.0, min(1.0, t_opt))
    
    return t_opt

    """Find exact optimal h[j] given all other coordinates fixed. Box constraint only."""
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    h_j = h[j]

    # alpha_d = (1-h[j+d]) if 0 <= j+d < n, else 0
    # beta_d = h[j-d] if 0 <= j-d < n, else 0

def _pair_search(h, rng, deadline, max_iters=50000):
    """Random pair moves with exact line search maintaining sum."""
    n = len(h)
    target = n / 2.0
    best_c5 = _eval_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 2000:
            break
        
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        s = h[i] + h[j]
        # h[i] in [max(0, s-1), min(1, s)]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-12:
            no_improve += 1
            continue
        
        # Grid search: 12 points
        grid = np.linspace(lo, hi, 12)
        best_t = h[i]
        best_c5_local = best_c5
        
        for t in grid:
            h[i] = t
            h[j] = s - t
            c = _eval_c5(h)
            if c < best_c5_local:
                best_c5_local = c
                best_t = t
        
        h[i] = best_t
        h[j] = s - best_t
        c5_new = _eval_c5(h)
        
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        elif c5_new > best_c5 + 1e-10:
            # Revert
            h[i] = h[i] + (h[j] - h[j])  # undo
            h[i], h[j] = best_h[i], best_h[j]
            no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

    """Simulated annealing with coordinate moves."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = _eval_c5(h)
    best_h = h.copy()
    temp = 0.01
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        
        j = rng.integers(n)
        h_old = h[j]
        h_new = h_old + rng.uniform(-0.15, 0.15)

        h_proj = _project_box_sum(h, target)

        h = np.tile(0.5 + 0.3 * np.sin(np.linspace(0, 2*np.pi, block)), n // block + 1)[:n]
        h = _project_box_sum(h, target)
        starts.append(h)
    
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h0 = None
    
    best_h = None
    best_c5 = np.inf
    best_n = 0
    
    if h0 is not None and len(h0) > 0:
        n0 = len(h0)
        h_init = _project_box_sum(h0, n0 / 2.0)
        c5_init = _eval_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
        best_n = n0
    
    # Phase 1: Coordinate descent on parent (main driver)
    deadline1 = t0 + budget_s * 0.25
    h1, c5_1 = _coord_sweep(best_h, rng, deadline1, max_sweeps=3000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Pair search
    deadline2 = t0 + budget_s * 0.38
    h2, c5_2 = _pair_search(best_h, rng, deadline2, max_iters=80000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: More coordinate descent from improved point
    deadline3 = t0 + budget_s * 0.50
    h3, c5_3 = _coord_sweep(best_h, rng, deadline3, max_sweeps=3000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Try multiple alternative starts at parent resolution
    deadline4 = t0 + budget_s * 0.65
    if best_n >= 4:
        per_start = max(1.0, (deadline4 - time.time()) / 8.0)
        for start_h in _generate_starts(best_n, rng)[:6]:
            if time.time() > deadline4 - per_start:
                break
            dl = min(deadline4, time.time() + per_start)
            hs, cs = _coord_sweep(start_h, rng, dl, max_sweeps=500)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()
    
    # Phase 5: Multi-resolution exploration
    deadline5 = t0 + budget_s * 0.82
    n_candidates = [40, 48, 64, 80, 96, 112, 128, 144, 160, 192, 256]
    n_candidates = [x for x in n_candidates if x >= 32 and x <= 256]
    # Sort by distance to current best_n
    n_candidates.sort(key=lambda x: abs(x - best_n))
    
    per_res = max(2.0, (deadline5 - time.time()) / max(1, len(n_candidates)))
    for n_try in n_candidates:
        if n_try == best_n:
            continue
        if time.time() > deadline5 - per_res:
            break
        
        h_r = _resample(best_h, n_try)
        dl_r = min(deadline5, time.time() + per_res)
        hr, cr = _coord_sweep(h_r, rng, dl_r, max_sweeps=1000)
        
        # Also try pair search briefly
        if cr < best_c5 and time.time() < deadline5 - 2:
            dl_r2 = min(deadline5, time.time() + per_res * 0.4)
            hr2, cr2 = _pair_search(hr, rng, dl_r2, max_iters=20000)
            if cr2 < cr:
                hr, cr = hr2, cr2
        
        # Prefer smaller n on ties
        if cr < best_c5 - 1e-10 or (abs(cr - best_c5) <= 1e-10 and n_try < best_n):
            best_c5 = cr
            best_h = hr.copy()
            best_n = n_try
    
    # Phase 6: Final coordinate descent polish
    deadline6 = t0 + budget_s * 0.93
    h6, c5_6 = _coord_sweep(best_h, rng, deadline6, max_sweeps=5000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    # Phase 7: Final pair search
    deadline7 = t0 + budget_s * 0.97
    h7, c5_7 = _pair_search(best_h, rng, deadline7, max_iters=30000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    
    # Final projection and validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _eval_c5(best_h)
    
    # Safety: use the better of computed vs stored
    if final_c5 > best_c5 + 1e-12:
        # Projection may have slightly increased; use the pre-projection value if valid
        # but we need a valid point, so use projected
        best_c5 = final_c5
    
    # Ensure validity
    if not np.all(best_h >= -1e-12) or not np.all(best_h <= 1.0 + 1e-12):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _eval_c5(best_h)
    
    # Last check: ensure sum constraint
    s = float(np.sum(best_h))
    if abs(s - best_n / 2.0) > 1e-8:
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _eval_c5(best_h)
    
    return best_h, best_c5, best_n
