def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    return float(np.max(_corr(h)) * (2.0 / len(h)))


def _proj(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0, 1)


def _softmax_grad(h, beta):
    """Gradient of (1/beta)*log(sum exp(beta*corr_j)) scaled by dx."""
    n = len(h)
    corr = _corr(h)
    dx = 2.0 / n
    bcorr = beta * corr
    bcorr -= bcorr.max()
    w = np.exp(bcorr)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-16:
            continue
        d = j - (n - 1)
        if d >= 0:
            if d == 0:
                grad += w[j] * (1.0 - 2.0 * h)
            else:
                grad[:n - d] += w[j] * (1.0 - h[d:])
                grad[d:] -= w[j] * h[:n - d]
        else:
            m = -d
            grad[m:] += w[j] * (1.0 - h[:n - m])
            grad[:n - m] -= w[j] * h[m:]
    return grad * dx


def _active_grad(h, n_top=5):
    """Subgradient: average gradient of top n_active constraints."""
    n = len(h)
    corr = _corr(h)
    dx = 2.0 / n
    top = np.argsort(corr)[-n_top:]
    grad = np.zeros(n)
    for j in top:
        d = j - (n - 1)
        if d >= 0:
            if d == 0:
                grad += (1.0 - 2.0 * h)
            else:
                grad[:n - d] += (1.0 - h[d:])
                grad[d:] -= h[:n - d]
        else:
            m = -d
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    grad /= len(top)
    return grad * dx


def _phase_softmax(h, deadline):
    """Optimize with increasing beta using projected gradient."""
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c = _c5(h)
    x = h.copy()
    v = np.zeros(n)
    momentum = 0.7
    
    betas = [10, 30, 80, 200, 500, 1000]
    for beta in betas:
        t_limit = min(deadline, time.time() + 30.0)  # ~30s per beta
        lr = 0.02
        it = 0
        while time.time() < t_limit:
            it += 1
            g = _softmax_grad(x, beta)
            g -= g.mean()  # preserve sum
            v = momentum * v - lr * g
            x_new = _proj(x + v, target)
            c = _c5(x_new)
            if c < best_c:
                best_c = c
                best = x_new.copy()
            x = x_new
            # Adaptive lr
            if it % 50 == 0:
                lr *= 0.7
    return best, best_c


def _phase_nesterov(h, deadline, n_top=5):
    """Nesterov-style accelerated subgradient method."""
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c = _c5(h)
    x = h.copy()
    x_prev = h.copy()
    t = 1.0
    lr = 0.03
    it = 0
    while time.time() < deadline:
        it += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        # Extrapolate
        y = _proj(x + (t - 1.0) / t_new * (x - x_prev), target)
        # Subgradient step
        g = _active_grad(y, n_top)
        g -= g.mean()
        x_new = _proj(y - lr * t_new * g, target)
        c = _c5(x_new)
        if c < best_c:
            best_c = c
            best = x_new.copy()
        x_prev = x
        x = x_new
        t = t_new
        # Slow down lr
        lr = 0.03 / (1.0 + 0.002 * it)
        if lr < 1e-4:
            break
    return best, best_c
