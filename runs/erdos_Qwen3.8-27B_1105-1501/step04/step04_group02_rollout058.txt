import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')  # raw scores, length 2n-1

def _score_grad_single(h, d):
    """Gradient of S_r w.r.t. h for lag d = r - (n-1)."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        g[d:] += 1.0 - h[:m]
        g[:m] -= h[d:]
    else:
        s = -d
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    return g

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _score_compute(h, d):
    n = len(h)
    if d >= 0:
        m = n - d
        if m <= 0:
            return 0.0
        return float(np.dot(h[d:], 1.0 - h[:m]))
    else:
        s = -d
        m = n - s
        if m <= 0:
            return 0.0
        return float(np.dot(h[:m], 1.0 - h[s:]))

def _score_grad_single(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += 1.0 - h[:m]
            g[:m] -= h[d:]
    else:
        s = -d
        m = n - s
        if m > 0:
            g[:m] += 1.0 - h[s:]
            g[s:] -= h[:m]
    return g

def _weighted_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:

        else:
            s = -d
            m = n - s
            if m <= 0:
                continue
            grad[:m] += wr * (1.0 - h[s:])
            grad[s:] -= wr * h[:m]
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

class _NAS:
    def __init__(self, n, lags, target):
        self.n = n
        self.lags = lags
        self.target = target

    def obj(self, x):
        return x[self.n]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def ineq(self, x):
        h = x[:self.n]
        T = x[self.n]
        out = np.empty(len(self.lags))
        for i, d in enumerate(self.lags):
            out[i] = T - _score_compute(h, d)
        return out

    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        nlags = len(self.lags)
        J = np.empty((nlags, n + 1))
        for i, d in enumerate(self.lags):
            J[i, :n] = -_score_grad_single(h, d)
            J[i, n] = 1.0
        return J

    def eq(self, x):
        return np.sum(x[:self.n]) - self.target

    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def _slsqp_solve(h, active_lags, n, target, maxiter=35):
    prog = _NAS(n, active_lags, target)
    raw = _all_scores(h)
    T0 = float(np.max(raw)) + 1e-5
    x0 = np.concatenate([h, [T0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [{'type': 'ineq', 'fun': prog.ineq, 'jac': prog.ineq_jac},
            {'type': 'eq', 'fun': prog.eq, 'jac': prog.eq_jac}]
    try:
        res = minimize(prog.obj, x0, jac=prog.obj_grad, constraints=cons,
                       bounds=bounds, method='SLSQP', maxiter=maxiter, ftol=1e-14)
        x_out = res.x
        if x_out is not None and np.all(np.isfinite(x_out)):
            h_new = _project(x_out[:n], target)
            return h_new
    except Exception:
        pass
    return _project(h, target)

    best_c5 = _exact_c5(h_best)
    best_n = n_best
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, 12, -12, 16, -16]
    tried = set()
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n_best + dn
        if n_try < 36 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
        sub_dl = min(deadline, time.time() + per_time)
        h_r, c5_r = _active_set_loop(h_r, sub_dl, k_init=14, max_outer=20)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
        if time.time() < deadline:
            sub_dl2 = min(deadline, time.time() + per_time * 0.5)
            h_r, c5_r = _grad_optimize(h_r, sub_dl2, tau_init=15.0, lr_init=0.01, max_iter=300)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    return best_h, best_c5, best_n

def _perturb_refine(h_best, deadline, per_time=2.0, seed=100):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = 0.03
    attempt = 0
    while time.time() < deadline and attempt < 100:
        attempt += 1
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + per_time)
        h_try, c5_try = _active_set_loop(h_pert, sub_dl, k_init=12, max_outer=15)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.3)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safety = max(5, int(0.03 * budget_s))
    hard_deadline = t0 + budget_s - safety

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Gradient warmup
    d1 = t0 + budget_s * 0.06
    h, c5 = _grad_optimize(h, d1, tau_init=15.0, lr_init=0.02, mom_beta=0.88, max_iter=500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Active set SLSQP
    d2 = t0 + budget_s * 0.22
    h, c5 = _active_set_loop(best_h, d2, k_init=16, max_outer=35)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Gradient polish after SLSQP
    d3 = t0 + budget_s * 0.30
    h, c5 = _grad_optimize(best_h, d3, tau_init=40.0, lr_init=0.008, mom_beta=0.9, max_iter=800)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: More SLSQP with different k
    d4 = t0 + budget_s * 0.40
    h, c5 = _active_set_loop(best_h, d4, k_init=24, max_outer=25)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Multi-resolution
    d5 = t0 + budget_s * 0.65
    best_h, best_c5, best_n = _multi_resolution(best_h, best_n, d5, per_time=2.5)

    # Phase 6: Active set at best resolution
    d6 = t0 + budget_s * 0.74
    h, c5 = _active_set_loop(best_h, d6, k_init=20, max_outer=30)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 7: Direction search
    d7 = t0 + budget_s * 0.80
    h, c5 = _direction_search(best_h, d7, n_dirs=10, radius=0.25, seed=seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 8: Perturbation
    d8 = t0 + budget_s * 0.88
    best_h, best_c5 = _perturb_refine(best_h, d8, per_time=2.0, seed=seed + 50)

    # Phase 9: Final SLSQP polish
    d9 = t0 + budget_s * 0.94
    h, c5 = _active_set_loop(best_h, d9, k_init=30, max_outer=20)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 10: Pair refinement
    d10 = hard_deadline
    h, c5 = _pair_refine(best_h, d10, max_passes=300, n_samples=80, seed=999)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Final SLSQP
    if time.time() < hard_deadline - 2:
        h, c5 = _active_set_loop(best_h, hard_deadline, k_init=20, max_outer=10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
