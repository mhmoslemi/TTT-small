import numpy as np

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _jac_corr(h):
    """Jacobian of correlation vector w.r.t. h. Shape (2n-1, n)."""
    n = len(h)
    nc = 2 * n - 1
    J = np.zeros((nc, n))
    j = np.arange(n)
    for r in range(nc):
        s = r - (n - 1)
        # d(c_s)/d(h_j) = (1 - h_{j+s}) * 1[0<=j+s<n] - h_{j-s} * 1[0<=j-s<n]
        idx1 = j + s
        m1 = (idx1 >= 0) & (idx1 < n)
        J[r, m1] += (1.0 - h[idx1[m1]])
        idx2 = j - s
        m2 = (idx2 >= 0) & (idx2 < n)
        J[r, m2] -= h[idx2[m2]]
    return J

def _lse_objective(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = _corr(h)
    ms = float(np.max(corr))
    shifted = alpha * (corr - ms)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_v = np.exp(shifted)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_v)))
    w = exp_v / float(np.sum(exp_v))
    # Gradient
    n = len(h)
    nc = 2 * n - 1
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(nc):
        s = r - (n - 1)
        idx1 = j + s
        m1 = (idx1 >= 0) & (idx1 < n)
        d = np.zeros(n)
        d[m1] = (1.0 - h[idx1[m1]])
        idx2 = j - s
        m2 = (idx2 >= 0) & (idx2 < n)
        d[m2] -= h[idx2[m2]]
        grad += w[r] * d
    # Remove mean to handle the constraint
    grad -= np.mean(grad)
    return f, grad

def _newton_active(h, active_lags, max_iter=50, tol=1e-12):
    """
    Newton's method on the active set.
    Given active lags, solve for h and t such that c_k = t for k in active_lags, sum(h) = n/2.
    """
    n = len(h)
    h = h.copy().astype(float)
    target = n * 0.5
    t = float(np.max(_corr(h)))
    
    for it in range(max_iter):
        corr = _corr(h)
        # Update t to be the max of active lags
        t = float(np.max(corr[active_lags]))
        
        # Residuals: for each active lag, c_k - t. Plus sum(h) - target.
        m = len(active_lags)
        F = np.zeros(m + 1)
        F[:m] = corr[active_lags] - t
        F[m] = np.sum(h) - target
        
        if np.max(np.abs(F)) < tol:
            break
        
        # Jacobian of F w.r.t. (h, t)
        Jc = _jac_corr(h)  # (2n-1, n)
        J_active = Jc[active_lags, :]  # (m, n)
        
        # Build the full Jacobian: (m+1) x (n+1)
        J_full = np.zeros((m + 1, n + 1))
        J_full[:m, :n] = J_active
        J_full[:m, n] = -1.0  # d(c_k - t)/dt = -1
        J_full[m, :n] = 1.0   # d(sum(h))/dh = 1
        
        # Newton step
        try:
            step = np.linalg.solve(J_full, -F)
        except np.linalg.LinAlgError:
            break
        
        # Damped step
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.01]:
            h_new = h + alpha * step[:n]
            t_new = t + alpha * step[n]
            h_new = _project(h_new, target)
            c_new = _corr(h_new)
            if np.max(c_new) < np.max(corr) + 1e-10:
                h, t = h_new, float(np.max(c_new))
                break
        else:
            break
    
    return h, _exact_c5(h)

def _slsqp_solve(h_init, n, maxiter=300, ftol=1e-13):
    """SLSQP epigraph formulation."""
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    
    def _epi_f(z, nn):
        return z[nn]
    
    def _epi_g(z, nn):
        g = np.zeros(nn + 1)
        g[nn] = 1.0
        return g
    
    def _eq_f(z, nn):
        return np.sum(z[:nn]) - nn * 0.5
    
    def _eq_g(z, nn):
        g = np.ones(nn + 1)
        g[nn] = 0.0
        return g
    
    def _ineq_f(z, nn):
        h, t = z[:nn], z[nn]
        return t - _corr(h)
    
    def _ineq_g(z, nn):
        h = z[:nn]
        return np.concatenate([-_jac_corr(h), np.ones(2*nn-1)[:, None]], axis=1)
    
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_f, z0, jac=_epi_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _binary_seed(n, rng):
    """Generate a balanced binary seed."""
    positions = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h

def _smooth_seed(n, rng, kind='gaussian'):
    """Generate a smooth seed with correct mass."""
    x = np.linspace(0, 1, n)
    if kind == 'gaussian':
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.1, 0.3)
        h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif kind == 'cosine':
        h = 0.5 + 0.5 * np.cos(2 * np.pi * x * rng.integers(1, 4))
    else:
        h = rng.uniform(0, 1, n)
    return _project(h / h.sum() * (n / 2), n / 2)

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _jac_corr(h):
    n = len(h)
    nc = 2 * n - 1
    J = np.zeros((nc, n))
    j = np.arange(n)
    for r in range(nc):
        s = r - (n - 1)
        idx1 = j + s
        m1 = (idx1 >= 0) & (idx1 < n)
        J[r, m1] += (1.0 - h[idx1[m1]])
        idx2 = j - s
        m2 = (idx2 >= 0) & (idx2 < n)
        J[r, m2] -= h[idx2[m2]]
    return J

    c2 = a + gr * (b - a)
    f1 = float(np.max(c + c1 * d))
    f2 = float(np.max(c + c2 * d))
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2
            f2 = f1
            c2 = c1
            f1 = float(np.max(c + c1 * d))
            c1 = b - gr * (b - a)
        else:
            a = c1
            f1 = f2
            c1 = c2
            f2 = float(np.max(c + c2 * d))
            c2 = a + gr * (b - a)
    return (a + b) / 2.0


def _pairwise_cd(h, n, max_rounds=200, n_pairs=800, rng=None, sys_sweep=False):
    """Pairwise coordinate descent preserving mass exactly."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0

    for rnd in range(max_rounds):
        c = _corr(h)
        cur_max = float(np.max(c))
        improved = False

        if sys_sweep:
            pairs = []
            for i in range(n):
                for j in range(i + 1, n):
                    pairs.append((i, j))
            rng.shuffle(pairs) if rng else None
            n_pairs = len(pairs)
        else:
            pairs = []
            for _ in range(n_pairs):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    pairs.append((i, j))

        for (i, j) in pairs:
            nc = 2 * n - 1
            d = np.zeros(nc)
            for r in range(nc):
                s = r - (n - 1)
                di = 0.0
                if 0 <= i + s < n:
                    di += 1.0 - h[i + s]
                if 0 <= i - s < n:
                    di -= h[i - s]
                dj = 0.0
                if 0 <= j + s < n:
                    dj += 1.0 - h[j + s]
                if 0 <= j - s < n:
                    dj -= h[j - s]
                d[r] = di - dj

            delta = _pairwise_delta(c, d, h[i], h[j])
            if abs(delta) > 1e-10:
                h[i] += delta
                h[j] -= delta
                c = _corr(h)
                new_max = float(np.max(c))
                if new_max < cur_max - 1e-12:
                    improved = True
                    cur_max = new_max
                    if new_max * 2.0 / n < best_c5:
                        best_c5 = new_max * 2.0 / n
                        best_h = h.copy()

        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0

    return best_h, best_c5


def _slsqp_solve(h_init, n, maxiter=300, ftol=1e-14):
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]

    def _f(z, nn):
        return z[nn]

    def _g(z, nn):
        g = np.zeros(nn + 1)
        g[nn] = 1.0
        return g

    def _eq_f(z, nn):
        return np.sum(z[:nn]) - nn * 0.5

    def _eq_g(z, nn):
        g = np.ones(nn + 1)
        g[nn] = 0.0
        return g

    def _ineq_f(z, nn):
        hh = z[:nn]
        t = z[nn]
        return t - _corr(hh)

    def _ineq_g(z, nn):
        hh = z[:nn]
        J = _jac_corr(hh)
        return np.concatenate([np.column_stack([-J, np.ones(2 * nn - 1)])], axis=1)

    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_f, z0, jac=_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)


def _lse_step(h, n, alpha, maxiter=60):
    bounds = [(0.0, 1.0)] * n

    def _obj(x, nn, al):
        hh = np.asarray(x, dtype=float)
        cc = _corr(hh)
        ms = float(np.max(cc))
        vals = np.clip(al * (cc - ms), -500.0, 500.0)
        ev = np.exp(vals)
        f = ms + (1.0 / al) * np.log(float(np.sum(ev)))
        w = ev / float(np.sum(ev))
        nn2 = len(hh)
        j = np.arange(nn2)
        grad = np.zeros(nn2)
        nc2 = 2 * nn2 - 1
        for r in range(nc2):
            s = r - (nn2 - 1)
            d = np.zeros(nn2)
            idx1 = j + s
            m1 = (idx1 >= 0) & (idx1 < nn2)
            d[m1] = 1.0 - hh[idx1[m1]]
            idx2 = j - s
            m2 = (idx2 >= 0) & (idx2 < nn2)
            d[m2] -= hh[idx2[m2]]
            grad += w[r] * d
        grad -= np.mean(grad)
        return f, grad

    try:
        res = minimize(_obj, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 40, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _binary_seed(n, rng):
    positions = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h


def _smooth_seed(n, rng, kind=0):
    x = np.linspace(0, 1, n)
    if kind == 0:
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.08, 0.25)
        h = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif kind == 1:
        f = int(rng.integers(1, 4))
        h = 0.5 + 0.5 * np.cos(2 * np.pi * x * f)
    else:
        h = 0.3 + 0.7 * np.abs(np.sin(np.pi * x * rng.integers(1, 5)))
    h = _project(h / np.sum(h) * (n / 2), n / 2)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 * 0.5
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def _elapsed():
        return time.time() - t0

    # Phase 1: LSE smoothing (8% budget)
    d1 = t0 + budget_s * 0.08
    for alpha in [20.0, 50.0, 100.0, 200.0]:
        if time.time() > d1:
            break
        h, c5 = _lse_step(h, n0, alpha, maxiter=50)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Pairwise CD from parent (12% budget)
    d2 = t0 + budget_s * 0.20
    h_cd, c5_cd = _pairwise_cd(best_h, n0, max_rounds=100, n_pairs=600, rng=rng)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 3: SLSQP (15% budget)
    d3 = t0 + budget_s * 0.35
    for attempt in range(4):
        if time.time() > d3:
            break
        if attempt == 0:
            start = best_h.copy()
        else:
            sc = 0.008 * (1.0 + 0.5 * attempt)
            start = _project(best_h + rng.normal(0, sc, n0), target0)
        h_s, c5_s = _slsqp_solve(start, n0, maxiter=400, ftol=1e-14)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
            # Follow with pairwise CD
            if time.time() < d3:
                h_cd, c5_cd = _pairwise_cd(h_s, n0, max_rounds=50, n_pairs=400, rng=rng)
                if c5_cd < best_c5:
                    best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 4: Diverse seeds (20% budget)
    d4 = t0 + budget_s * 0.55
    seeds = []
    for _ in range(4):
        seeds.append(_binary_seed(n0, rng))
    seeds.append(_smooth_seed(n0, rng, 0))
    seeds.append(_smooth_seed(n0, rng, 1))
    seeds.append(_smooth_seed(n0, rng, 2))
    # Symmetrized
    sym = (best_h + best_h[::-1]) / 2.0
    seeds.append(_project(sym, target0))
    # Parent with structured perturbation
    wt = np.sin(np.pi * np.arange(n0) / n0)
    seeds.append(_project(best_h + 0.05 * wt, target0))
    seeds.append(_project(best_h + 0.1 * wt, target0))

    for sh in seeds:
        if time.time() > d4:
            break
        sh, c5_sh = _lse_step(sh, n0, 50.0, maxiter=30)
        sh, c5_sh = _pairwise_cd(sh, n0, max_rounds=30, n_pairs=300, rng=rng)
        if c5_sh < best_c5:
            best_c5, best_h = c5_sh, sh.copy()
        if time.time() < d4:
            h_s, c5_s = _slsqp_solve(sh, n0, maxiter=200, ftol=1e-13)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()
                h_cd, c5_cd = _pairwise_cd(h_s, n0, max_rounds=40, n_pairs=400, rng=rng)
                if c5_cd < best_c5:
                    best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 5: Resolution search (15% budget)
    d5 = t0 + budget_s * 0.70
    for dn in [-16, -8, -12, 8, -4, 4, -20, 12, -32]:
        if time.time() > d5:
            break
        n_try = n0 + dn
        if n_try < 16 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        tr = n_try * 0.5
        h_r, c5_r = _lse_step(h_r, n_try, 40.0, maxiter=30)
        h_r, c5_r = _pairwise_cd(h_r, n_try, max_rounds=50, n_pairs=400, rng=rng)
        h_s, c5_s = _slsqp_solve(h_r, n_try, maxiter=200, ftol=1e-13)
        if c5_s < c5_r:
            h_r, c5_r = h_s, c5_s
            h_r, c5_r = _pairwise_cd(h_r, n_try, max_rounds=40, n_pairs=300, rng=rng)
        if c5_r < best_c5 - 1e-12:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Intensive final polish (25% budget)
    d6 = t0 + budget_s * 0.97
    target_best = best_n * 0.5

    # SLSQP with tight tolerance
    h_f, c5_f = _slsqp_solve(best_h, best_n, maxiter=500, ftol=1e-15)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()

    # Systematic pairwise sweep
    if time.time() < d6:
        h_cd, c5_cd = _pairwise_cd(best_h, best_n, max_rounds=50,
                                    n_pairs=best_n * (best_n - 1) // 2,
                                    rng=rng, sys_sweep=True)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # More SLSQP + CD iterations
    for attempt in range(30):
        if time.time() > d6:
            break
        sc = 0.003 + 0.004 * rng.random()
        h_p = _project(best_h + rng.normal(0, sc, best_n), target_best)
        h_s, c5_s = _slsqp_solve(h_p, best_n, maxiter=150, ftol=1e-13)
        if c5_s < best_c5 - 1e-11:
            best_c5, best_h = c5_s, h_s.copy()
            if time.time() < d6:
                h_cd, c5_cd = _pairwise_cd(h_s, best_n, max_rounds=30,
                                            n_pairs=300, rng=rng)
                if c5_cd < best_c5:
                    best_c5, best_h = c5_cd, h_cd.copy()

    # Final pairwise CD sweep
    if time.time() < d6:
        h_cd, c5_cd = _pairwise_cd(best_h, best_n, max_rounds=40,
                                    n_pairs=500, rng=rng)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Ultimate polish: alternating SLSQP and CD
    for _ in range(3):
        if time.time() > d6:
            break
        h_s, c5_s = _slsqp_solve(best_h, best_n, maxiter=200, ftol=1e-15)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        h_cd, c5_cd = _pairwise_cd(best_h, best_n, max_rounds=30,
                                    n_pairs=400, rng=rng)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
