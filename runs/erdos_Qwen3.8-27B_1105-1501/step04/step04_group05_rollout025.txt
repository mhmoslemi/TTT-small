def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto [0,1]^n with sum = target using bisection."""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _binary_score(mask):
    """Score of a binary mask as a step function."""
    n = len(mask)
    h = mask.astype(float)
    return _score(h)

def _two_level_score(mask, d):
    """Score of two-level mask with values d (zeros) and 1-d (ones)."""
    n = len(mask)
    h = np.where(mask > 0.5, 1.0 - d, d)
    return _score(h)

def _optimal_d(mask, n_samples=200):
    """Find optimal d for a two-level mask via golden section search."""
    n = len(mask)
    # d in [0, 0.5], d=0 is binary, d=0.5 is uniform (worst)
    # We expect optimal d to be small, near 0
    a, b = 0.0, 0.5
    invphi = (np.sqrt(5) - 1) / 2
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = _two_level_score(mask, c)
    fd = _two_level_score(mask, d)
    for _ in range(n_samples):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - invphi * (b - a)
            fc = _two_level_score(mask, c)
        else:
            a = c; c = d; fc = fd
            d = b - invphi * (b - a)
            fd = _two_level_score(mask, d)

def _binary_local_search(mask, deadline, max_rounds=5000):
    """Local search on binary mask: try all single swaps, keep best."""
    n = len(mask)
    mask = mask.copy()
    ones = set(np.where(mask)[0].tolist())
    zeros = set(np.where(~mask)[0].tolist())
    curr_score = _binary_score(mask)
    best_mask = mask.copy()
    best_score = curr_score

        best_swap = None
        best_swap_score = curr_score
        # Try all single swaps

- Total per round: 1024 * ~256 ops ≈ 262K ops
- With numpy overhead, maybe 1-5ms per round
- 10000 rounds = 10-50 seconds

That should be fine for 30% of 1100s = 330s.

Actually, I can vectorize over all swaps at once! For a fixed o, I can compute dc for all z simultaneously. Let me think...

For a fixed o, the terms involving o are:
- v_ok (h[o+k]): fixed, doesn't depend on z
- v_omk (h[o-k]): fixed

The terms involving z are:
- v_zk (h[z+k]): depends on z
- v_zmk (h[z-k]): depends on z

So dc = [fixed_part] + [1 - v_zmk] - v_zk

where fixed_part = v_ok - (1 - v_omk) = v_ok - 1 + v_omk

And for each z, I need v_zk and v_zmk.

I can precompute v_zk for all z at once! For each z, v_zk is h[z+k] for k in [-(n-1), n-1], which is a shifted version of h.

Let me create a 2D array where row z is v_zk for that z:
```python
# For all z, compute h[z+k] for k in [-(n-1), n-1]
# This is like a "correlation" of h with itself, but with zero-padding
# Actually it's just: result[z, i] = h[z + (-(n-1) + i)] if valid, else 0

# Can use scipy.signal or just numpy indexing
all_zk = np.zeros((n, 2*n-1))
for z in range(n):
    id_z = z + ks
    valid = (id_z >= 0) & (id_z < n)
    all_zk[z, valid] = h[id_z[valid]]
all_zmk = all_zk[:, ::-1]  # reverse each row to get h[z-k]
```

This is O(n²) = O(4096) for n=64, which is trivial.
