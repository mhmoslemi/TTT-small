import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(carr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _sqp_step(h, rho, G=None, scores=None):
    """Full-lag SQP step. Returns (delta, tau_model) or (None, None) on failure."""
    n = len(h)
    L = 2 * n - 1
    if G is None:
        G = _grad_matrix(h)
    if scores is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * (2.0 / n)
    
    # Variables: delta (n) + tau (1) = n+1 total
    # Objective: minimize tau
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize tau
    
    # Inequality constraints: scores[r] + G[r] @ delta <= tau
    # => scores[r] + G[r] @ delta - tau <= 0
    # => G[r] @ delta - tau <= -scores[r]
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds: 0 <= h_i + delta_i <= 1 => -h_i <= delta_i <= 1-h_i
    # Plus trust region: |delta_i| <= rho
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        if lo > hi:
            lo = hi = 0.0  # degenerate, shouldn't happen if rho is reasonable
        bounds.append((lo, hi))
    bounds.append((None, None))  # tau is free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([0.0]),
                  bounds=bounds, method='highs')
    if not res.success:
        return None, None, res
    delta = res.x[:n]
    tau_model = res.x[n]
    return delta, tau_model, res
