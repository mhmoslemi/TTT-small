For a specific index j:
- j appears as h_{i+d} when i = j-d, which is valid if 0 ≤ j-d ≤ n-d-1, i.e., d ≤ j ≤ n-1

So:
∂C_q/∂h_j = dx * [1{j in [d, n-1]} * (1 - h_{j-d}) - 1{j in [0, n-d-1]} * h_{j+d}]

∂C_q/∂h_j = dx * [1{j in [0, n-k-1]} * (1 - h_{j+k}) - 1{j in [k, n-1]} * h_{j-k}]

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_gradient(h, q):
    """Gradient of C_q(h) = dx * corr[h, 1-h, full][q]"""
    n = len(h)
    dx = 2.0 / n
    d = q - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        # C_q = dx * sum_{i=0}^{n-d-1} h[i+d] * (1-h[i])
        if d < n:
            # j in [d, n-1]: contributes dx*(1-h[j-d])
            g[d:] += dx * (1.0 - h[:n-d])
            # j in [0, n-d-1]: contributes -dx*h[j+d]
            g[:n-d] -= dx * h[d:]
    else:
        k = -d
        # C_q = dx * sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
        if k < n:
            # j in [0, n-k-1]: contributes dx*(1-h[j+k])
            g[:n-k] += dx * (1.0 - h[k:])
            # j in [k, n-1]: contributes -dx*h[j-k]
            g[k:] -= dx * h[:n-k]
    return g

def _lp_vertex(w, target):
    """Solve min w·x s.t. 0<=x_i<=1, sum x_i = target. Target is integer n/2."""
    order = np.argsort(w)
    x = np.zeros(len(w))
    x[order[:int(target)]] = 1.0
    return x

def _fw_step(h, deadline, rng, n_active=6, eta_init=0.04, max_iter=100):
    """Frank-Wolfe step with LP vertex oracle."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_c5 = _exact_c5(h_cur)
    c_all = _corr_scaled(h_cur)
    lam = float(np.max(c_all))
    eta = eta_init * max(lam, 0.01)
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 25:
            break
            
        # Softmax weights over active constraints
        # Focus on top constraints
        idx = np.argsort(c_all)[::-1]
        top = idx[:n_active]
        weights = np.exp((c_all[top] - lam) / eta)
        weights /= weights.sum()
        
        # Build direction vector (negative gradient direction)
        w = np.zeros(n)
        for i, q in enumerate(top):
            w += weights[i] * _corr_gradient(h_cur, q)
        
        # LP vertex: minimizes w·x over constraint set
        s = _lp_vertex(w, target)
        
        # Line search: h_alpha = (1-alpha)*h_cur + alpha*s
        best_alpha = 0.0
        best_val = best_c5
        alphas = np.linspace(0, 1, 33)
        for a in alphas:
            if a == 0:
                continue
            h_t = (1 - a) * h_cur + a * s
            if abs(h_t.sum() - target) > 1e-9:
                h_t = _project(h_t, target)
            val = _exact_c5(h_t)
            if val < best_val - 1e-13:
                best_val = val
                best_alpha = a
        
        if best_alpha > 0 and best_val < best_c5 - 1e-13:
            # Refine around best_alpha
            lo_a = max(0, best_alpha - 1/33.0)
            hi_a = min(1.0, best_alpha + 1/33.0)
            for a in np.linspace(lo_a, hi_a, 17):
                h_t = (1 - a) * h_cur + a * s
                if abs(h_t.sum() - target) > 1e-9:
                    h_t = _project(h_t, target)
                val = _exact_c5(h_t)
                if val < best_val - 1e-13:
                    best_val = val
                    best_alpha = a
            
            h_cur = (1 - best_alpha) * h_cur + best_alpha * s
            if abs(h_cur.sum() - target) > 1e-9:
                h_cur = _project(h_cur, target)
            best_c5 = _exact_c5(h_cur)
            c_all = _corr_scaled(h_cur)
            lam = float(np.max(c_all))
            stall = 0
            eta = max(eta * 0.85, 0.002 * max(lam, 0.01))
        else:
            stall += 1
            eta = eta * 0.9
            # Try increasing n_active
            n_active = min(n_active + 1, 2 * n - 1)
    
    return h_cur, best_c5

def _grad_descent(h, deadline, lr=0.003, n_lags=5, patience=30):
    """Protected projected gradient descent."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_c5 = _exact_c5(h_cur)
    stall = 0
    
    for _ in range(5000):
        if time.time() > deadline:
            break
        if stall > patience:
            break
        
        c_all = _corr_scaled(h_cur)
        top = np.argsort(c_all)[::-1][:n_lags]
        
        # Average gradient of top constraints
        g = np.zeros(n)
        for q in top:
            g += _corr_gradient(h_cur, q)
        g /= len(top)
        
        # Remove mean to preserve sum
        g -= np.mean(g)
        
        h_new = _project(h_cur - lr * g, target)
        c_new = _exact_c5(h_new)
        
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            h_cur = h_new
            stall = 0
            lr = min(lr * 1.05, 0.03)
        elif c_new < _exact_c5(h_cur) - 1e-13:
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr = max(lr * 0.85, 1e-6)
            if lr < 1e-6:
                break
    
    return h_cur, best_c5

def _targeted_leveling(h, deadline, gamma_init=0.003, max_iter=200):
    """Level out the top correlation peaks by targeting a reduced maximum."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_c5 = _exact_c5(h_cur)
    c_all = _corr_scaled(h_cur)
    lam = float(np.max(c_all))
    gamma = gamma_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        
        # Target: reduce max by gamma
        T = lam * (1 - gamma)
        
        # Residual: how much each constraint exceeds target
        r = np.maximum(c_all - T, 0.0)
        active = np.where(r > 1e-12)[0]
        
        if len(active) == 0:
            gamma *= 0.7
            continue
        
        # Weighted gradient of excess
        w = np.zeros(n)
        total_r = r[active].sum()
        if total_r < 1e-15:
            # All at target, use uniform weights
            for q in active:
                w += _corr_gradient(h_cur, q)
            w /= len(active)
        else:
            for q in active:
                w += (r[q] / total_r) * _corr_gradient(h_cur, q)
        
        # Remove mean
        w -= np.mean(w)
        
        # Try multiple step sizes
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.125, 2.0]:
            h_try = _project(h_cur - 0.01 * scale * w, target)
            c_try = _exact_c5(h_try)
            if c_try < best_c5 - 1e-13:
                best_c5 = c_try
                h_cur = h_try
                c_all = _corr_scaled(h_cur)
                lam = float(np.max(c_all))
                stall = 0
                improved = True
                break
        
        if improved:
            gamma = min(gamma * 1.1, 0.02)
        else:
            stall += 1
            gamma = max(gamma * 0.7, 0.0005)
            if gamma < 0.0005:
                break
    
    return h_cur, best_c5

def _pair_refine(h, deadline, max_passes=50):
    """Compensated pair coordinate refinement."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Focus on interior points
        interior = [j for j in range(n) if 1e-6 < h[j] < 1.0 - 1e-6]
        
        for j in interior:
            if time.time() > deadline:
                break
            if not (1e-6 <= h[j] <= 1.0 - 1e-6):
                continue
            for k in range(j + 1, n):
                if not (1e-6 <= h[k] <= 1.0 - 1e-6):
                    continue
                # Keep sum constant: h_j + h_k = p
                p = h[j] + h[k]
                lo = max(0.0, p - 1.0)
                hi = min(1.0, p)
                if hi - lo < 1e-9:
                    continue
                
                # Scan for best split
                best_z = h[j]
                best_val = best_c5
                for z in np.linspace(lo, hi, 21):
                    h_t = h.copy()
                    h_t[j] = z
                    h_t[k] = p - z
                    val = _exact_c5(h_t)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_z = z
                
                if best_val < best_c5 - 1e-12 and abs(best_z - h[j]) > 1e-8:
                    h[j] = best_z
                    h[k] = p - best_z
                    best_c5 = best_val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _corr_grad(h, q):
    n = len(h)
    dx = 2.0 / n
    d = q - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += dx * (1.0 - h[:n - d])
            g[:n - d] -= dx * h[d:]
    else:
        k = -d
        if k < n:
            g[:n - k] += dx * (1.0 - h[k:])
            g[k:] -= dx * h[:n - k]
    return g


def _lp_vertex(w, target):
    order = np.argsort(w)
    x = np.zeros(len(w))
    x[order[:int(round(target))]] = 1.0
    return x

def _pair_refine(h, deadline, max_passes=80):

    h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):

        improved = False
        interior = [j for j in range(n) if 1e-6 < h[j] < 1.0 - 1e-6]

                break
            if not (1e-6 <= h[j] <= 1.0 - 1e-6):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _shift_eval(h):

    best = (h, _exact_c5(h))

        c = _exact_c5(h_s)

        c5_r = min(c5_r, c5_r2)
        if c5_r < best_c5 - 1e-8:
            best_c5 = c5_r
            best_h = h_r.copy()
            n = n_try

    # Final polish
    if time.time() < deadline - 1:
        h_f, c5_f = _fw_optimize(best_h, deadline - 1, rng,
                                 n_active=10, eta_frac=0.01, max_iter=60, line_pts=41)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_f2, c5_f2 = _pair_refine(best_h, deadline - 1, max_passes=30)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
