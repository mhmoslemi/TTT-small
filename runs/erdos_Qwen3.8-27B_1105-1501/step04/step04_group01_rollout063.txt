- h[i] appears in (1-h[n]) when n = i (if valid)

After changing h[i] to x:

     = h[i-r]*(h[i]-x) + (x-h[i])*(1-h[i+r])
     = (x-h[i]) * (1-h[i+r] - h[i-r])

(Where we take h[j] = 0 for j < 0 or j >= n, so terms with invalid indices are 0.)

So for coordinate i, changing h[i] to x changes f_r by (x-h[i]) * A_r where A_r = 1-h[i+r]-h[i-r] (with appropriate boundary handling).

The new C5 = max_r (f_r + (x-h[i])*A_r) * dx

To minimize this, we need to find x that minimizes max_r (f_r + (x-h[i])*A_r).

Let δ = x - h[i]. Then we want to minimize max_r (f_r + δ*A_r) over δ ∈ [max(-h[i], -h[i]... well, δ ∈ [-h[i], 1-h[i]].

This is a 1D minimax of lines! The optimal δ is where the "envelope" of lines f_r + δ*A_r is minimized at its peak.

This can be solved efficiently:
- The max is a convex piecewise-linear function of δ
- Its minimum is where the slope changes from negative to positive
- Equivalently, where the line with the largest A (among those achieving the max) crosses the line with the smallest A

More precisely: at the optimum, the two lines that achieve the maximum should be equal (if the minimum is interior):
f_r1 + δ*A_r1 = f_r2 + δ*A_r2 = max
→ δ = (f_r2 - f_r1) / (A_r1 - A_r2)

If the minimum is at a boundary (δ = -h[i] or δ = 1-h[i]), then set x to that boundary.

This is a very efficient coordinate optimization! Much faster than the LP approach.

```python
def _coord_optimize(h, i, corr):
    """Find optimal value for h[i] given all others fixed."""
    n = len(h)
    dx = 2.0 / n
    # Compute A_r for all r
    # A_r = 1 - h[i+r] - h[i-r] (with h[j]=0 for j<0 or j>=n)
    A = np.ones(2*n-1)
    r_vals = np.arange(-(n-1), n)
    # i+r valid means 0 <= i+r < n
    idx1 = r_vals + i  # positions of h[i+r]
    valid1 = (idx1 >= 0) & (idx1 < n)
    A[valid1] -= h[idx1[valid1]]
    # i-r valid means 0 <= i-r < n
    idx2 = i - r_vals  # positions of h[i-r]
    valid2 = (idx2 >= 0) & (idx2 < n)
    A[valid2] -= h[idx2[valid2]]
    
    # minimize max_r (corr[r] + delta * A[r]) for delta in [-h[i], 1-h[i]]
    lo_d = -h[i]
    hi_d = 1.0 - h[i]
    
    # The function g(delta) = max_r (corr[r] + delta * A[r]) is convex piecewise linear
    # Find minimum by bisection on the subgradient
    # Subgradient at delta contains {A[r] : r active at delta}
    # Minimum is where 0 is in the subgradient, i.e., min active A <= 0 <= max active A
    
    best_delta = 0.0
    best_val = float(np.max(corr))
    
    # Bisection
    for _ in range(50):
        mid = (lo_d + hi_d) / 2.0
        vals = corr + mid * A
        max_v = float(np.max(vals))
        # Subgradient: A[r] for r where vals[r] is close to max
        active = vals >= max_v - 1e-10
        min_A = float(np.min(A[active]))
        max_A = float(np.max(A[active]))
        if min_A <= 0 <= max_A:
            best_delta = mid
            break
        elif min_A > 0:
            # All active slopes positive, move left
            hi_d = mid
        else:
            # All active slopes negative, move right
            lo_d = mid
    
    # Verify by trying the exact candidates
    # The breakpoints are where two lines cross
    # For efficiency, just evaluate at a few candidates
    candidates = [lo_d, hi_d, best_delta]
    # Add intersections of top lines
    top_r = np.argsort(corr)[-10:]
    for a in range(len(top_r)):
        for b in range(a+1, len(top_r)):
            ra, rb = top_r[a], top_r[b]
            if abs(A[ra] - A[rb]) > 1e-15:
                delta_cross = (corr[rb] - corr[ra]) / (A[ra] - A[rb])
                if lo_d <= delta_cross <= hi_d:
                    candidates.append(delta_cross)
    
    best_val = float('inf')
    for dc in candidates:
        dc = max(lo_d, min(hi_d, dc))
        val = float(np.max(corr + dc * A)) * dx
        if val < best_val:
            best_val = val
            best_delta = dc
    
    return best_delta, best_val
```

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_full(h):

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    r_vals = np.arange(-(n - 1), n)
    for j in range(n):
        idx_h1 = j - r_vals
        valid1 = (idx_h1 >= 0) & (idx_h1 < n)
        G[:, j] = (1.0 - h[idx_h1[np.arange(2*n-1)]]) * 0.0
        for idx in range(2 * n - 1):
            r = r_vals[idx]
            val = 0.0
            jmr = j - r
            if 0 <= jmr < n:
                val += (1.0 - h[jmr])
            jpr = j + r
            if 0 <= jpr < n:
                val -= h[jpr]
            G[idx, j] = val
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, max_stall=40):

            break

def _fw_direction(h, scores, top_k=5):
    n = len(h)
    max_s = float(np.max(scores))
    sorted_idx = np.argsort(scores)[::-1][:top_k]
    w = np.zeros(2 * n - 1)
    for idx in sorted_idx:
        w[idx] = max(0.1, scores[idx] - max_s + 1.0)
    G = _grad_matrix(h)
    c_vec = w @ G
    k = n // 2
    order = np.argsort(c_vec)
    v = np.zeros(n)
    v[order[:k]] = 1.0
    d = v - h
    dn = np.linalg.norm(d)
    if dn < 1e-12:
        return None
    return d / dn

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)

        h_r, c5_r = _hybrid_loop(h_r, sub_dl, radius_init=0.08)
        rem = d4 - time.time()
        if rem > 4 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d4, time.time() + max(2, budget_s * 0.006))
            h_r, c5_r = _coord_pass(h_r, deadline=sub_dl2, max_rounds=50)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Perturbation + re-optimization (15%)
    d5 = t0 + budget_s * 0.90
    scale = 0.03
    for attempt in range(100):
        if time.time() > d5:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d5, time.time() + max(2, budget_s * 0.008))
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.06, max_stall=20)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.004, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.15)

    # Phase 6: Final coordinate + LP polish (7%)
    d6 = t0 + budget_s * 0.94
    h_c, c5_c = _coord_pass(best_h, deadline=d6, max_rounds=100)
    if c5_c < best_c5:
        best_c5 = c5_c
        best_h = h_c.copy()

    d7 = t0 + budget_s * 0.98
    h_f, c5_f = _lp_loop(best_h, d7, radius_init=0.03, max_stall=25)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
