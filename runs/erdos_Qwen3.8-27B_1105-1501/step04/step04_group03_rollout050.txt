def subgrad_descent(h0, deadline, tau=0.02, lr_init=0.5, momentum=0.7):

def build_G(h):
    """Build the gradient matrix G where G[k, j] = d(corr_raw[k])/dh[j]"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            # j+d valid: j in [0, n-1-d]
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d+m])  # (1-h[j+d]) for j=0..m-1
                # j-d valid: j in [d, n-1]
                G[r, d:] += -h[:m]  # -h[j-d] for j=d..n-1
        else:
            g = -d
            m = n - g
            if m > 0:
                # j+d valid: j+d = j-g, valid when j in [g, n-1]
                G[r, g:] += (1.0 - h[:m])  # (1-h[j-g]) for j=g..n-1
                # j-d valid: j-d = j+g, valid when j in [0, n-1-g]
                G[r, :m] += -h[g:g+m]  # -h[j+g] for j=0..m-1
    return G

def lp_direction(h, scores, G, radius):
    """Find direction p minimizing max_k (scores[k] + G[k]@p) subject to constraints."""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def line_search_c5(h, p, c5_cur, max_iter=25):

        elif p[j] < -1e-14:

    alpha_max = max(alpha_max, 1e-10)

    best_c5 = c5_cur
    # Golden section
    lo_a = 0.0
    hi_a = alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(max_iter):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    # Also check endpoints
    f0 = exact_c5(h)
    if f0 < best_c5:
        return 0.0, f0
    fal = exact_c5(h + alpha_max * p)
    if fal < best_c5:
        return alpha_max, fal
    return best_a, best_c5

    A_ub[:, n] = -1.0
    b_ub = -b_sel
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, p, c5_cur):
    n = len(h)
    amax = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            amax = min(amax, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            amax = min(amax, h[j] / (-p[j]))
    amax = max(amax, 1e-10)
    lo_a, hi_a = 0.0, amax
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < c5_cur:
        return opt_a, opt_c5
    return 0.0, c5_cur

def lp_loop(h, deadline, radius_init=0.1, max_stall=50, top_k=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > max_stall:
            break
        scores = compute_scores(h)
        G = build_G(h)
        p = lp_dir_active(h, scores, G, radius, top_k=top_k)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            radius *= 0.72
            if stall % 25 == 0 and radius < 0.005:
                radius = 0.04
    return best_h, best_c5

def subgrad_loop(h0, deadline, init_tau=0.15, final_tau=0.003):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    vel = np.zeros(n)
    momentum = 0.75
    step = 0
    while time.time() < deadline:
        step += 1
        frac = min(1.0, step / 4000.0)
        tau = init_tau * (final_tau / init_tau) ** frac
        max_s, grad = subgradient_of_max(h, tau)
        gnorm_sq = float(np.dot(grad, grad))
        if gnorm_sq < 1e-22:
            break
        gap = max_s - best_c5
        if gap < 1e-8:
            gap = 1e-8
        step_size = min(2.0 * gap / gnorm_sq, 0.3)
        vel = momentum * vel - step_size * grad
        h = project(h + vel, target)
        c5_cur = exact_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def pair_polish(h, deadline, max_passes=60):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline:
                break
            for k in range(j + 1, n, 2):
                if time.time() > deadline:
                    break
                amax = min(best_h[j], 1.0 - best_h[k])
                if amax > 1e-10:
                    for a in [amax * 0.5, amax]:
                        h_t = best_h.copy()
                        h_t[j] += a
                        h_t[k] -= a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            improved = True
                            break
                if improved:
                    break
                amax2 = min(1.0 - best_h[j], best_h[k])
                if amax2 > 1e-10:
                    for a in [amax2 * 0.5, amax2]:
                        h_t = best_h.copy()
                        h_t[k] += a
                        h_t[j] -= a
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-12:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            improved = True
                            break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: LP with active set
    d1 = t0 + budget_s * 0.22
    h1, c5_1 = lp_loop(h, d1, radius_init=0.12, max_stall=45, top_k=40)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Subgradient exploration
    d2 = t0 + budget_s * 0.32
    h2, c5_2 = subgrad_loop(best_h, d2, init_tau=0.12, final_tau=0.003)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: LP refinement (full constraints for precision)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = lp_loop(best_h, d3, radius_init=0.06, max_stall=60, top_k=None)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Multi-n search
    d4 = t0 + budget_s * 0.72
    n_offsets = sorted(set([d for d in range(-24, 25, 2) if 48 <= best_n + d <= 140]))
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.013)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.05, max_stall=25, top_k=30)
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, h_r, n_try

    # Phase 5: Random restarts
    d5 = t0 + budget_s * 0.82
    for _ in range(15):
        if time.time() > d5:
            break
        scale = rng.uniform(0.015, 0.12)
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.006)
        h_p, c5_p = lp_loop(h_pert, sub_dl, radius_init=0.04, max_stall=15, top_k=30)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p

    # Phase 6: Pair polish
    d6 = t0 + budget_s * 0.92
    h_pol, c5_pol = pair_polish(best_h, d6, max_passes=80)
    if c5_pol < best_c5:
        best_c5, best_h = c5_pol, h_pol

    # Phase 7: Final LP micro-polish
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.015, max_stall=25, top_k=None)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0.copy(), n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-15:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
