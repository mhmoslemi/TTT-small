from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, direction, max_alpha=1.0):
    """Line search along direction from h. Returns (best_alpha, best_c5)."""
    n = len(h)
    # Clamp max_alpha for feasibility
    if max_alpha > 0:
        for j in range(n):
            if direction[j] > 1e-14:
                max_alpha = min(max_alpha, (1.0 - h[j]) / direction[j])
            elif direction[j] < -1e-14:
                max_alpha = min(max_alpha, h[j] / (-direction[j]))
    max_alpha = max(max_alpha, 0.0)
    
    # Grid search
    n_grid = 16
    alphas = np.linspace(0, max_alpha, n_grid)
    vals = np.array([_exact_c5(h + a * direction) for a in alphas])
    best_i = int(np.argmin(vals))
    best_a = float(alphas[best_i])
    best_c5 = float(vals[best_i])
    
    # Golden section refinement
    if 0 < best_i < n_grid - 1:
        lo = float(alphas[best_i - 1])
        hi = float(alphas[best_i + 1])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * direction)
        f2 = _exact_c5(h + c2 * direction)
        for _ in range(20):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi, f2, c2 = c1, f1, c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * direction)
            else:
                lo, f1, c1 = c1, f2, c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * direction)
        a_opt = (lo + hi) / 2.0
        c5_opt = _exact_c5(h + a_opt * direction)
        if c5_opt < best_c5:
            best_a, best_c5 = a_opt, c5_opt
    
    return best_a, best_c5

def _full_lp_step(h, G, scores, rho):
    """Solve full-lag LP trust region. Returns direction delta or None."""
    n = len(h)
    n_lags = 2 * n - 1
    
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: G[r]·delta - t <= -scores[r] for all r
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -scores.copy()
    
    # Box constraints on delta
    lo_bounds = np.maximum(h, -rho)  # delta >= -min(h, rho)
    # Wait, delta >= -h and delta >= -rho, so delta >= -min(h, rho)
    # delta <= 1-h and delta <= rho, so delta <= min(1-h, rho)
    
    # I'll use the 'bounds' parameter for the box constraints
    bounds = []
    for j in range(n):
        lo = -min(h[j], rho)
        hi = min(1.0 - h[j], rho)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize_full(h, deadline, rho_start=0.3, max_iters=5000):
    """Main optimization: full-lag LP trust region."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rho = rho_start
    stall = 0
    
    G = _grad_matrix(h)
    scores = _scores(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Solve LP
        delta = _full_lp_step(h, G, scores, rho)
        if delta is None:
            rho *= 0.7
            stall += 1
            if stall > 50:
                break
            continue
        
        # Check if delta is meaningful
        if np.max(np.abs(delta)) < 1e-14:
            rho *= 0.7
            stall += 1
            if stall > 50:
                break
            continue
        
        # Line search
        alpha, new_c5 = _line_search(h, delta, max_alpha=1.0)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * delta
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            G = _grad_matrix(h)
            scores = _scores(h)
            stall = 0
            # Increase rho on success
            rho = min(rho * 1.15, 1.0)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            rho *= 0.75
            if stall > 80 and rho < 0.01:
                break
    
    return best_h, best_c5

    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())

            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.1, 0.05, 0.02, 0.01]:

def _binary_exchange(h_init, deadline, max_iters=50000):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng2 = np.random.default_rng(1)
        rng2.shuffle(idx)
        h[idx[ones_count - target:]] = 0.0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng2 = np.random.default_rng(1)
        rng2.shuffle(idx)
        h[idx[:target - ones_count]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    stall = 0
    T = 0.02 * cur_c5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 6000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta_c5 = new_c5 - cur_c5
        T *= 0.9998
        if delta_c5 < 0 or (rng.random() < np.exp(-delta_c5 / max(T, 1e-10))):
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _symmetrize(h):
    n = len(h)
    h_sym = (h + h[::-1]) / 2.0
    return _project(h_sym, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Symmetrize + initial LP (7%)
    dl1 = t0 + budget_s * 0.07
    h_sym = _symmetrize(h)
    c5_sym = _exact_c5(h_sym)
    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_sym.copy()
    hdr, c5r = _lp_optimize_full(best_h, dl1, rho_start=0.4, max_iters=5000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 2: Main full-LP optimization (35%)
    dl2 = t0 + budget_s * 0.42
    hdr, c5r = _lp_optimize_full(best_h, dl2, rho_start=0.25, max_iters=20000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 3: P-norm smoothing (12%)
    dl3 = t0 + budget_s * 0.54
    hdr, c5r = _pnorm_descent(best_h, dl3, p_start=6.0, p_end=2048.0, n_steps=2000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 4: LP again (10%)
    dl4 = t0 + budget_s * 0.64
    hdr, c5r = _lp_optimize_full(best_h, dl4, rho_start=0.1, max_iters=10000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 5: Noise restarts with LP (20%)
    dl5 = t0 + budget_s * 0.84
    scale = 0.03
    n_try = best_n
    for attempt in range(500):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, n_try)
        h_pert = _project(h_pert, n_try / 2.0)
        # Quick LP polish
        sub_dl = min(dl5, time.time() + budget_s * 0.015)
        h_try, c5_try = _lp_optimize_full(h_pert, sub_dl, rho_start=0.15, max_iters=2000)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 6: Multi-resolution exploration (8%)
    dl6 = t0 + budget_s * 0.92
    for dn in [2, 3, 4, -2, -3, -4, 6, -6, 8]:
        if time.time() > dl6:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl6, time.time() + budget_s * 0.01)
        h_r, c5_r = _lp_optimize_full(h_r, sub_dl, rho_start=0.15, max_iters=3000)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Binary exchange (5%)
    if time.time() < t0 + budget_s * 0.97:
        h_bin, c5_bin = _binary_exchange(best_h, min(t0 + budget_s * 0.97, time.time() + budget_s * 0.04))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Polish binary result with continuous LP
        sub_dl = min(t0 + budget_s * 0.985, time.time() + budget_s * 0.015)
        h_pol, c5_pol = _lp_optimize_full(best_h, sub_dl, rho_start=0.2, max_iters=5000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 8: Final ultra-fine LP (2%)
    if time.time() < t0 + budget_s * 0.995:
        hdr, c5r = _lp_optimize_full(best_h, t0 + budget_s * 0.995, rho_start=0.03, max_iters=5000)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hdr.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
