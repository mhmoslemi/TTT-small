    j = np.arange(n)

For a given j, the value of h[j] that minimizes max_s c[s] can be found by:
- Computing c[s] as a function of h[j] for each s

This is a "Chebyshev approximation" in 1D, solvable by checking all pairwise intersections. If K is the number of relevant lags, this takes O(K²) per coordinate, or O(K) with a sweep.

This is a 1D minimax problem with linear functions. The solution is at a point where the maximum is minimized, which occurs at:
- The left or right boundary of the feasible interval for h[j], or
- A point where two lines cross and this crossing value is the maximum

Algorithm for 1D minimax of linear functions:
1. For each s, we have line L_s(x) = A_s' + D_s * x for x ∈ [x_min, x_max]
2. Find x that minimizes max_s L_s(x)
3. This is equivalent to: find the smallest t such that L_s(x) ≤ t for all s, and x ∈ [x_min, x_max]
4. The optimal t is the minimum over x of max_s L_s(x)

A simple approach: 
- Evaluate max_s L_s(x) at x_min, x_max, and at all pairwise intersections of the lines within [x_min, x_max]
- Take the minimum

For K lines, there are O(K²) intersections. If K ~ 64 (all lags), that's ~2000 evaluations, each O(K) = 64, so ~130K operations per coordinate. For n=64 coordinates, total ~8M operations per sweep. That's fast!

But we don't need all lags, just the ones that could be near the maximum. So K = top-K (say 16) should suffice.

    mid = (lo + hi) / 2
    vals = A + D * mid

- If sum(h) < n/2: decrease ν (make it more negative)
- If sum(h) > n/2: increase ν

            old_val = h[j]
            D = np.zeros(2*n-1)

        elif current_sum < target:

            best_c5 = c5
            # decay nu_step
            nu_step *= 0.95

def project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(x)
        else:
            order = np.argsort(-x)
        for i in order:
            if diff > 0:
                a = min(1.0 - x[i], diff)
            else:
                a = max(x[i] + diff, 0.0) - x[i]
            x[i] += a
            diff -= a
            if abs(diff) < 1e-12:
                break
    return x


def eval_c5(h):

def _minimax_1d(corr, D, y_lo, y_hi, K):
    mid = (y_lo + y_hi) / 2.0
    vals = corr + D * mid
    top = np.argsort(-vals)[:K]
    A_k = corr[top]
    D_k = D[top]
    nk = len(A_k)
    candidates = [y_lo, y_hi]
    for i in range(nk):
        for j in range(i + 1, nk):
            dd = D_k[i] - D_k[j]
            if abs(dd) > 1e-14:
                xc = (A_k[j] - A_k[i]) / dd
                if y_lo - 1e-12 <= xc <= y_hi + 1e-12:
                    xc = np.clip(xc, y_lo, y_hi)
                    candidates.append(xc)
    best_x = mid
    best_v = np.inf
    for x in candidates:
        v = float(np.max(A_k + D_k * x))
        if v < best_v:
            best_v = v
            best_x = x
    return best_x

    pd = pd / norm * step_size
    cur_c5 = eval_c5(h)
    h_new = project(h - pd, n * 0.5)
    c5_new = eval_c5(h_new)
    if c5_new < cur_c5:
        return h_new, c5_new
    for frac in [0.5, 0.25, 0.1, 0.05]:
        h_try = project(h - frac * pd, n * 0.5)
        c5_try = eval_c5(h_try)
        if c5_try < cur_c5:
            return h_try, c5_try
    return h.copy(), cur_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = project(h0, target)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    best_n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: LSE warm-up (10% budget)
    d1 = t0 + budget_s * 0.10
    for alpha in [15.0, 30.0, 60.0, 120.0]:
        if time.time() > d1:
            break
        h, c5 = _lse_step(h, n0, alpha, maxiter=60)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Active-set SLSQP (15% budget)
    d2 = t0 + budget_s * 0.25
    for attempt in range(4):
        if time.time() > d2:
            break
        start = best_h.copy()
        if attempt > 0:
            start = project(start + rng.normal(0, 0.015, n0), target)
        h_s, c5_s = _aset_slsqp(start, n0, K=12, maxiter=300, ftol=1e-14)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    # Phase 3: Coordinate descent (20% budget) - NOVEL
    d3 = t0 + budget_s * 0.45
    h_cd, c5_cd = coord_descent(best_h, n0, K=20, max_sweeps=500, deadline=d3)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()
    # Second pass with smaller K for speed
    if time.time() < d3:
        h_cd2, c5_cd2 = coord_descent(best_h, n0, K=12, max_sweeps=300, deadline=d3)
        if c5_cd2 < best_c5:
            best_c5, best_h = c5_cd2, h_cd2.copy()

    # Phase 4: LP local search (15% budget)
    d4 = t0 + budget_s * 0.60
    h_lp, c5_lp = _lp_loop(best_h, n0, d4, radius_init=0.08,
                           max_iters=5000, stall_limit=50)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 5: Subgradient + SLSQP iterations (10% budget)
    d5 = t0 + budget_s * 0.70
    for it in range(50):
        if time.time() > d5:
            break
        if best_c5 <= 0.3806:
            break
        h_sg, c5_sg = _subgrad_step(best_h, n0, step_size=0.03)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()
        h_s, c5_s = _aset_slsqp(best_h, n0, K=10, maxiter=150, ftol=1e-14)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        h_cd3, c5_cd3 = coord_descent(best_h, n0, K=16, max_sweeps=50,
                                      deadline=min(d5, time.time() + 5))
        if c5_cd3 < best_c5:
            best_c5, best_h = c5_cd3, h_cd3.copy()

    # Phase 6: Ternary rounding + local search (10% budget)
    d6 = t0 + budget_s * 0.80
    h_tern, c5_tern = _ternary_round(best_h, n0)
    if c5_tern < best_c5:
        best_c5, best_h = c5_tern, h_tern.copy()
    h_tl, c5_tl = _ternary_local_search(best_h, n0, max_iters=30000, deadline=d6)
    if c5_tl < best_c5:
        best_c5, best_h = c5_tl, h_tl.copy()

    # Phase 7: Multi-resolution search (10% budget)
    d7 = t0 + budget_s * 0.90
    for dn in [0, 2, -2, 4, -4, 8, -8, 6, -6]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d7, time.time() + budget_s * 0.02)
        h_r, c5_r = _lse_step(h_r, n_try, 40.0, maxiter=40)
        h_cd_r, c5_cd_r = coord_descent(h_r, n_try, K=16, max_sweeps=100, deadline=sub_d)
        if c5_cd_r < c5_r:
            h_r, c5_r = h_cd_r, c5_cd_r
        sub_d2 = min(d7, time.time() + budget_s * 0.015)
        h_s, c5_s = _aset_slsqp(h_r, n_try, K=10, maxiter=150)
        if c5_s < c5_r:
            h_r, c5_r = h_s, c5_s
        sub_d3 = min(d7, time.time() + budget_s * 0.012)
        h_lp_r, c5_lp_r = _lp_loop(h_r, n_try, sub_d3, radius_init=0.05,
                                    max_iters=1000, stall_limit=40)
        if c5_lp_r < c5_r:
            h_r, c5_r = h_lp_r, c5_lp_r
        if c5_r < best_c5 - 1e-11:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 8: Final polish (10% budget)
    d8 = t0 + budget_s * 0.99
    for _ in range(5):
        if time.time() > d8:
            break
        h_s, c5_s = _aset_slsqp(best_h, best_n, K=10, maxiter=200, ftol=1e-15)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        h_cd_f, c5_cd_f = coord_descent(best_h, best_n, K=16, max_sweeps=100,
                                         deadline=min(d8, time.time() + 3))
        if c5_cd_f < best_c5:
            best_c5, best_h = c5_cd_f, h_cd_f.copy()
        sub_d = min(d8, time.time() + budget_s * 0.03)
        h_lp_f, c5_lp_f = _lp_loop(best_h, best_n, sub_d, radius_init=0.02,
                                    max_iters=2000, stall_limit=80)
        if c5_lp_f < best_c5:
            best_c5, best_h = c5_lp_f, h_lp_f.copy()
        if best_c5 <= 0.3806:
            break

    best_h = project(best_h, best_n * 0.5)
    final_c5 = eval_c5(best_h)
    return best_h, final_c5, best_n
