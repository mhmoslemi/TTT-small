    n = len(h)
    n_lags = len(scores)
    max_s = np.max(scores)
    # Active set: lags within some tolerance of max
    tol = 1e-6
    active = np.where(scores >= max_s - tol)[0]
    if len(active) == 0:
        return np.argmax(scores), np.ones(1)
    if len(active) == 1:
        return active[0], np.ones(1)

    # This is a QP. Use simplex or quadratic programming.
    G_a = grad_matrix[active].T  # shape (n, len(active))
    # Minimize λ^T (G_a G_a^T) λ s.t. sum(λ)=1, λ>=0
    Q = G_a @ G_a.T  # shape (len(active), len(active))
    
    # Use scipy minimize with SLSQP or just project
    from scipy.optimize import minimize
    m = len(active)
    x0 = np.ones(m) / m
    result = minimize(

        x0,

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

# KKT-based direction
def _kkt_direction(h, scores, G, tolerance=1e-4):
    """Find direction using KKT conditions on active set."""
    n = len(h)
    max_s = np.max(scores)
    active = np.where(scores >= max_s - tolerance)[0]
    
    if len(active) == 0:
        return None
    if len(active) == 1:
        # Just gradient of the single active constraint
        grad = G[active[0]]
    else:
        # Find weights that minimize ||sum lam_k * g_k||^2
        # subject to sum(lam) = 1, lam >= 0
        G_a = G[active].T  # (n, len(active))
        m = len(active)
        Q = G_a @ G_a.T  # (m, m)
        
        # Use simple projected gradient for the QP
        lam = np.ones(m) / m
        for _ in range(100):
            grad_lam = 2 * Q @ lam
            lam = lam - 0.01 * grad_lam
            # Project onto simplex
            lam = np.maximum(lam, 0)
            lam /= lam.sum()
        
        grad = G_a @ lam  # (n,)
    
    # Remove mean to satisfy sum constraint
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return grad / gn

The new score is s_r + χ_r * a. The new C5 is max_r (s_r + χ_r * a).

This is the max of linear functions of a! So it's convex and piecewise linear in a. To minimize it over [0, a_max], I need to find the point where the "upper envelope" is minimized.

The breakpoints occur where s_{r1} + χ_{r1}*a = s_{r2} + χ_{r2}*a, i.e., a = (s_{r2} - s_{r1}) / (χ_{r1} - χ_{r2}).

Algorithm for exact pair optimization:
1. Compute χ_r for all r (2n-1 values)
2. Compute all breakpoints a_r = (s_r - s_{r'}) / (χ_{r'} - χ_r) for pairs where χ_r ≠ χ_{r'}
3. Filter breakpoints to [0, a_max]
4. Evaluate max(s + χ*a) at each breakpoint and at 0 and a_max
5. Return the minimum

But there are O((2n)²) breakpoints which is ~65000 for n=128. That's too many to evaluate all of them for each pair.

Better approach: since we only care about the TOP scores (which determine the max), we can:
1. Sort lags by current score (descending)
2. The max at a=0 is determined by the top lag
3. As a increases, the top lag might change at specific breakpoints
4. Use a "megiddo" style parametric search

Actually, for practical purposes, the ternary search with 10-12 iterations is fine because:
- The function is convex piecewise linear
- 12 iterations of ternary search gives interval of length a_max / 3^12 ≈ a_max / 531441
- This is more than enough precision

So the ternary search is already near-optimal. The issue isn't the search precision but whether we're checking the right pairs.

Let me reconsider. The pair polish checks ALL pairs (j,k) with 0<h<1. For n=128, that's C(128,2) ≈ 8000 pairs, each with 2 directions. So ~16000 ternary searches per pass, each with ~10 evaluations. That's 160000 evaluations of _exact_c5 per pass. Each evaluation is O(n) = O(128) for the correlate. Total: ~20M operations per pass. That's very fast (maybe 0.1s per pass).

So the pair polish should be able to find any 2-coordinate improvement. The question is whether the optimum requires a 3+ coordinate move.

I think the strategy should be:
1. Get close with LP (within ~0.001)
2. Use pair polish to get within ~1e-5
3. If still not at target, try triple moves or a small LP

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p):
    n = len(h)
    alpha_max = 0.8
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(28):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 > f1:
        opt_a, opt_c5 = c1, f1
    if opt_c5 > f2:
        opt_a, opt_c5 = c2, f2
    return opt_a, opt_c5

def _local_opt(h, deadline, top_k_list=None, radius_init=0.2, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 20, 10, 5, 3]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    dir_mom = np.zeros(n)
    n_mom = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.55
            ti += 1
            if radius < 1e-7:
                break
            continue
        
        # Add momentum
        if n_mom > 0:
            p = 0.7 * p + 0.3 * dir_mom
            p = p - p.mean()
        
        alpha, new_c5 = _line_search(h, p)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            dir_mom = 0.7 * dir_mom + 0.3 * p
            dir_mom = dir_mom - dir_mom.mean()
            n_mom = min(n_mom + 1, 5)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.35)
        else:
            stall += 1
            radius *= 0.6
            ti += 1
            n_mom = max(0, n_mom - 1)
            if stall > 12 and radius < 0.008:
                radius = 0.08
    
    return best_h, best_c5

def _weighted_grad(h, deadline, n_steps=300, beta_start=5, beta_end=250):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    vel = np.zeros(n)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        vel = 0.9 * vel + 0.1 * grad
        vel = vel - vel.mean()
        vn = np.linalg.norm(vel)
        if vn < 1e-15:
            break
        vel /= vn
        
        alpha = 0.12
        improved = False
        for _ in range(45):
            h_new = _project(h - alpha * vel, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-16:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-13:
                break
        if not improved:
            for _ in range(15):
                h_new = _project(h - alpha * vel, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-16:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    break
                alpha *= 0.6
                if alpha < 1e-14:
                    break
    
    return best_h, best_c5

def _kkt_dir(h, scores, G, tol=1e-4):
    n = len(h)
    max_s = np.max(scores)
    active = np.where(scores >= max_s - tol)[0]
    if len(active) == 0:
        return None
    if len(active) == 1:
        grad = G[active[0]]
    else:
        G_a = G[active].T
        m = len(active)
        Q = G_a @ G_a.T
        lam = np.ones(m) / m
        for _ in range(80):
            gl = 2.0 * Q @ lam
            lam = lam - 0.02 * gl
            lam = np.maximum(lam, 0.0)
            s = lam.sum()
            if s < 1e-15:
                lam = np.ones(m) / m
            else:
                lam /= s
        grad = G_a @ lam
    grad = grad - grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return grad / gn

def _pair_polish(h, deadline, max_passes=12):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-10 or h[j] > 1.0 - 1e-10:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-10 or h[k] > 1.0 - 1e-10:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-12:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(12):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _qp_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    return _project(h.astype(float), n / 2.0)

def _fourier_init(n, coeffs, phases):
    x = np.arange(n) / n
    h = 0.5
    for c, p in zip(coeffs, phases):
        h += c * np.cos(2 * np.pi * x + p)
    return _project(h, n / 2.0)
