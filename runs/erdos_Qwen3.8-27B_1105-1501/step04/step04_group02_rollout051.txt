def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    n = len(h)
    scores = np.zeros(2 * n - 1)
    for d in range(n):  # d = 0, 1, ..., n-1
        m = n - d
        if d == 0:
            scores[n - 1] = float(np.sum(h[:m] * (1.0 - h[:m])))
        else:
            scores[n - 1 + d] = float(np.sum(h[d:] * (1.0 - h[:m])))
            scores[n - 1 - d] = float(np.sum(h[:m] * (1.0 - h[d:])))
    return scores

def _compute_grad_row(h, d):
    """Gradient of score at lag d w.r.t. h. Returns array of length n."""
    n = len(h)
    grad = np.zeros(n)
    m = n - d
    if d >= 0:
        # Score = sum_{i=0}^{m-1} h[i+d] * (1-h[i])
        # d/dh[i] = -h[i+d] for i < m
        # d/dh[i+d] = 1 - h[i] for i < m
        if m > 0:
            if d > 0:
                grad[:m] += -h[d:d+m]
                grad[d:d+m] += 1.0 - h[:m]
            else:
                # d == 0: grad[i] = -h[i] + (1-h[i]) = 1 - 2*h[i]
                grad[:] = 1.0 - 2.0 * h
    else:
        g = -d
        # Score = sum_{i=0}^{m-1} h[i] * (1-h[i+g])
        # d/dh[i] = 1 - h[i+g]
        # d/dh[i+g] = -h[i]
        if g > 0 and m > 0:
            grad[:m] += 1.0 - h[g:g+m]
            grad[g:g+m] += -h[:m]
    return grad

    """SLSQP solve using top-K active constraints."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    def objective(x):
        return x[n]  # T
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def constraints_func(x):
        h_vec = x[:n]
        T = x[n]
        scores = _raw_scores(h_vec)
        return scores - T
    
    def constraints_jac(x):
        h_vec = x[:n]
        G = np.zeros((2 * n - 1, n))
        for d in range(-n + 1, n):
            r = n - 1 + d
            G[r, :] = _compute_grad_row(h_vec, d)
        jac = np.zeros((2 * n - 1, n + 1))
        jac[:, :n] = G
        jac[:, n] = -1.0
        return jac

        row[:n] = 1.0

Given h_old of length n_old on [0,2] (dx_old = 2/n_old), and we want h_new of length n_new:
- Each old cell i covers [i*dx_old, (i+1)*dx_old]
- Each new cell j covers [j*dx_new, (j+1)*dx_new]
- The mass in new cell j = sum of (fraction of old cell i overlapping new cell j) * h_old[i] * dx_old / dx_new

For simplicity, I'll use the standard approach:
- Old centers: x_i = (i + 0.5) * dx_old
- New centers: y_j = (j + 0.5) * dx_new
- Use linear interpolation: h_new[j] = interp(y_j, old_centers, h_old)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def _lp_direction(h, scores, radius):

    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-n + 1, n):
        r = n - 1 + d
        if d >= 0:
            mm = n - d
            if mm > 0:
                G[r, d:d+mm] += 1.0
                G[r, d:d+mm] -= h[:mm]
                G[r, :mm] -= h[d:d+mm]
        else:
            g = -d
            mm = n - g
            if mm > 0:
                G[r, :mm] += 1.0
                G[r, :mm] -= h[g:g+mm]
                G[r, g:g+mm] -= h[:mm]
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    """Line search: find best alpha in [0, alpha_max]."""
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_eval)
    best_i = 0
    best_v = c5_cur
    for i, t in enumerate(ts):

        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            v = _exact_c5(h_t)
            if v < best_v:

                best_i = i

    if 0 < best_i < n_eval - 1:

        hi = ts[best_i + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)

            h_t = h + t * p
            if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1.0e-12:
                return c5_cur + 1.0
            return _exact_c5(h_t)

                c1 = hi - gr * (hi - lo)

                c2 = lo + gr * (hi - lo)

            best_t = opt_t
        else:

def _slsqp_topk(h, deadline, K_init=8, K_max=30, maxiter=None):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

        maxiter = max(20, min(60, 2000 // n))

    stall = 0

        target_T = n / 2.0  # just for bounds
        scores = _raw_scores(h)
        T_init = float(np.max(scores)) + 0.2
        top_k = np.argsort(scores)[::-1][:K]
        
        x0 = np.concatenate([h, [T_init]])
        
        # We'll use a subset of constraints
        # For SLSQP, we need to provide all constraints
        # Use only top-K
        def obj(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        # Constraints: S_r - T <= 0 for top-K r's
        cons = []
        for idx in top_k:
            d = idx - (n - 1)
            # Create constraint function for this specific lag
            def cfun(x, d=d):
                hv = x[:n]
                T = x[n]
                m = n - abs(d)
                if d >= 0:
                    s = float(np.sum(hv[d:d+m] * (1.0 - hv[:m])))
                else:
                    g = -d
                    s = float(np.sum(hv[:m] * (1.0 - hv[g:g+m])))
                return s - T
            
            def cjac(x, d=d):
                hv = x[:n]
                m = n - abs(d)
                row = np.zeros(n + 1)
                if d >= 0:
                    if d > 0:
                        row[d:d+m] += 1.0
                        row[d:d+m] -= hv[:m]
                        row[:m] -= hv[d:d+m]
                    else:
                        row[:] = 1.0 - 2.0 * hv
                else:
                    g = -d
                    row[:m] += 1.0
                    row[:m] -= hv[g:g+m]
                    row[g:g+m] -= hv[:m]
                row[n] = -1.0
                return row
            
            cons.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})
        
        # Equality: sum(h) = target
        cons.append({'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target,
                     'jac': lambda x: np.concatenate([np.ones(n), [0.0]]).reshape(1, -1)})
        
        bounds = [(0.0, 1.0)] * n + [(max(0, float(np.max(scores)) - 1.0), float(n))]
        
        try:
            with warnings.catch_warnings():
                warnings.simplefilter("ignore")
                result = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': maxiter, 'ftol': 1e-9})
            if result.x is not None and np.all(np.isfinite(result.x)):

                    best_h = h_new.copy()
                    best_c5 = c5_new

                        K = min(K * 2, K_max)
                    else:
                        K = max(K // 2, K_init)
                else:
                    stall += 1
                    K = min(K * 2, K_max)
        except Exception:
            stall += 1
            K = min(K * 2, K_max)
    
    return best_h, best_c5
