import numpy as np

_DE_STATE = {}

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0).astype(float).copy()
    for _ in range(500):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        G[r, j] += dx * (1.0 - h[i])
        G[r, i] += -dx * h[j]
    return G

def _lp_step(h, radius, k_active):
    n = len(h)
    L = 2 * n - 1
    F = _c5_vec(h)
    G = _build_G(h)
    idx = np.argsort(F)[::-1][:k_active]
    n_var = n + 1
    c = np.zeros(n_var)
    c[n] = 1.0
    A_ub = np.zeros((len(idx), n_var))
    b_ub = np.zeros(len(idx))
    for ii, s in enumerate(idx):
        A_ub[ii, :n] = G[s]
        A_ub[ii, n] = -1.0
        b_ub[ii] = F[s]
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-radius, radius)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

    for it in range(max_iters):

            break
        if stall > 80:
            break
        d = _lp_step(h, radius, min(64, 2*n-1))
        if d is None:

            radius *= 0.7

            continue

def _harmonic_h(params, n, freqs):
    M = len(freqs)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    z = np.full(n, params[0])
    for j in range(M):
        fj = freqs[j]
        z += params[1 + 2*j] * np.cos(2.0 * math.pi * fj * x)
        z += params[2 + 2*j] * np.sin(2.0 * math.pi * fj * x)
    z = np.clip(z, -18.0, 18.0)
    h_temp = 1.0 / (1.0 + np.exp(-z))
    return _project(h_temp, n / 2.0)

def _harmonic_c5(params, n, freqs):
    h = _harmonic_h(params, n, freqs)
    return _c5(h)

    h_fit = np.interp(x, x_p, h_parent)
    h_clip = np.clip(h_fit, 0.02, 0.98)
    z_target = np.log(h_clip / (1.0 - h_clip))
    M = len(freqs)
    cols = [np.ones(n)]
    for j in range(M):
        cols.append(np.cos(2.0 * math.pi * freqs[j] * x))
        cols.append(np.sin(2.0 * math.pi * freqs[j] * x))
    A = np.column_stack(cols)
    try:
        sol, _, _, _ = np.linalg.lstsq(A, z_target, rcond=None)
        sol = np.clip(sol, -3.0, 3.0)
        return sol
    except Exception:
        return np.zeros(2 * M + 1)

def _active_pair_search(h, deadline, rng, max_iters=200000):

            best_h = h_r.copy()
            best_n = n_try
        elif c5_r <= best_c5 + 1e-9 and n_try < best_n:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final LP polish at best resolution
    if time.time() < deadline_total - 30:
        dl7 = min(deadline_total - 15, t0 + budget_s * 0.95)
        h_f, c5_f = _lp_loop(best_h, dl7, radius_start=0.03)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Final active pair polish
    if time.time() < deadline_total - 10:
        h_fp, c5_fp = _active_pair_search(best_h, deadline_total - 5, rng, max_iters=50000)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Final validation and projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10) or not np.isfinite(final_c5):
        best_h = _project(np.clip(h0, 0, 1), n0 / 2.0)
        best_n = n0
        final_c5 = _c5(best_h)

    return best_h, final_c5, int(len(best_h))
