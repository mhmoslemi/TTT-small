    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad_rows(h, indices):
    n = len(h)
    K = len(indices)
    G = np.zeros((K, n))
    for r in range(K):
        d = indices[r] - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] = (1.0 - h[:m])
                G[r, :m] = -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] = (1.0 - h[g:])
                G[r, g:] = -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 20
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = float(obj_grid[best_i])
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(18):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t = opt_t
        best_val = opt_val
    return best_t, best_val

def _fast_lp(h, scores, k_active, radius):
    n = len(h)
    top_k = np.argsort(scores)[-k_active:]
    G = _compute_grad_rows(h, top_k)
    s_k = scores[top_k]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k_active, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.10, k_active=8):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    k = k_active
    while time.time() < deadline:
        if stall > 35:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _fast_lp(h, scores, k, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            if stall % 10 == 0:
                k = min(k + 2, 20)
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.04, 0.28)
        else:
            stall += 1
            radius *= 0.70
            if stall % 10 == 0:
                k = min(k + 3, 22)
            if stall > 15 and radius < 0.015:
                radius = 0.04
                k = k_active
    return best_h, best_c5

def _subgradient_step(h, dx, lr):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    active = scores >= max_s - dx * 0.15
    if not np.any(active):
        active = scores == max_s
    idx = np.where(active)[0]
    n_act = len(idx)
    G = _compute_grad_rows(h, idx)
    grad = G.T.shape[1] and (G.sum(axis=0) / n_act)
    grad -= np.mean(grad)
    return grad

def _grad_descent(h, deadline, lr_init=0.015, mom=0.85):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    m_vec = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        g = _subgradient_step(h, dx, lr)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        m_vec = mom * m_vec + (1.0 - mom) * g
        mn = np.linalg.norm(m_vec)
        if mn < 1e-14:
            continue
        m_vec = m_vec / mn
        alpha, new_c5 = _line_search(h, lr * m_vec, dx, lr * 1.3)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * m_vec
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.04, 0.30)
        else:
            stall += 1
            lr *= 0.85
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=25):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _multi_resolution(h_best, c5_best, deadline, budget_s):
    best_h = h_best.copy()
    best_c5 = c5_best
    best_n = len(best_h)
    n_base = best_n
    offsets = []
    for d in range(0, 45):
        offsets.append(n_base + d)
        if d > 0:
            offsets.append(n_base - d)
    offsets = [x for x in offsets if 20 <= x <= 220]
    rng = np.random.default_rng(123)
    rng.shuffle(offsets)
    for n_try in offsets:
        if time.time() > deadline:
            break
        h_r = _resample(best_h, n_try)
        c5_cur = _exact_c5(h_r)
        if c5_cur > best_c5 + 0.008:
            continue
        sub_dl = min(deadline, time.time() + max(2.0, budget_s * 0.015))
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, k_active=8)
        rem = deadline - time.time()
        if rem > 3 and c5_r < best_c5 + 0.001:
            h_r, c5_r = _grad_descent(h_r, min(deadline, time.time() + 2.0))
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Gradient descent warmup (4%)
    d1 = t0 + budget_s * 0.04
    h1, c5_1 = _grad_descent(h, d1, lr_init=0.012)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP local search with small K (18%)
    d2 = t0 + budget_s * 0.22
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.10, k_active=6)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP with larger K (16%)
    d3 = t0 + budget_s * 0.38
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.10, k_active=12)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Gradient refinement (6%)
    d4 = t0 + budget_s * 0.44
    h4, c5_4 = _grad_descent(best_h, d4, lr_init=0.008)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (26%)
    d5 = t0 + budget_s * 0.70
    best_h, best_c5 = _multi_resolution(best_h, best_c5, d5, budget_s)

    # Phase 6: Perturbation + re-optimize (14%)
    d6 = t0 + budget_s * 0.84
    scale = 0.035
    for attempt in range(300):
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.006)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.08, k_active=8)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.87)
        else:
            scale = min(0.16, scale * 1.09)

    # Phase 7: Pair refinement (8%)
    d7 = t0 + budget_s * 0.92
    h_p, c5_p = _pair_refine(best_h, d7, max_passes=20)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 8: Final LP polish (6%)
    d8 = t0 + budget_s * 0.98
    h_f, c5_f = _lp_loop(best_h, d8, radius_init=0.03, k_active=10)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
