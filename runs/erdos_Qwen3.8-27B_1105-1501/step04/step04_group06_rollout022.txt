import numpy as np

import time


def _project(v, target):
    """Project v onto {0<=x_i<=1, sum(x)=target} using bisection on lambda."""
    v = np.asarray(v, dtype=float).copy()
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Find lambda s.t. sum(clip(v + lambda, 0, 1)) = target
    lo = float(v.min()) - 1.0 - 1.0
    hi = float(v.max()) + 1.0 + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(v + lam, 0.0, 1.0)
    # Fix residual
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(x)  # most room first
        else:
            order = np.argsort(-x)  # least first
        for i in order:
            if abs(diff) <= 1e-12:
                break
            if diff > 0:
                room = 1.0 - x[i]
                step = min(room, diff)
                if step > 1e-15:
                    x[i] += step
                    diff -= step
            else:
                room = x[i]
                step = min(room, -diff)
                if step > 1e-15:
                    x[i] -= step
                    diff += step
    return x


def _corr(h):
    """Compute correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    """Compute C5 value."""
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _build_grad_matrix(h, active_r):
    """Build gradient matrix for given active correlation indices."""
    n = len(h)
    j = np.arange(n)
    J = np.zeros((len(active_r), n))
    for k, r in enumerate(active_r):
        s = r - (n - 1)  # offset
        pd = np.zeros(n)
        # d(corr_r)/d(h_m) = [valid m+s](1-h[m+s]) - [valid m-s]h[m-s]
        gs = j + s  # m + s must be valid
        m1 = (gs >= 0) & (gs < n)
        pd[m1] += (1.0 - h[gs[m1]])
        ps = j - s  # m - s must be valid
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        J[k, :] = pd
    return J


def _lp_direction(h, corr_vals, n, radius, active_r, reg=0.01):
    """Solve LP to find descent direction."""
    n_act = len(active_r)
    J = _build_grad_matrix(h, active_r)
    # Variables: p (n dims) + t (1 dim)
    # Minimize: t + reg * ||p||_inf (approximated by t + reg*t_extra)
    # Actually: minimize t + reg * sum(|p_i|) -- but that's non-smooth
    # Better: minimize t + reg * p_pos + reg * p_neg where p = p_pos - p_neg
    # But that doubles variables. Let's just minimize t with box constraints on p.
    
    # Constraints: J @ p - t <= -corr_vals[active_r]  (linearized: we want to reduce corr)
    # Actually: at h+alpha*p, corr_r ≈ corr_r(h) + alpha * J[r] @ p
    # We want: corr_r(h) + alpha * J[r] @ p <= t for all active r
    # => J[r] @ p - t <= -corr_r(h)  ... wait that's not right either
    # 
    # The linearized constraint is: corr_r + p @ grad_r <= t
    # => grad_r @ p - t <= -corr_r
    # We want to minimize t subject to these, with bound on p.
    
    m = n_act
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = J
    A_ub[:, n] = -1.0
    b_ub = -np.array(corr_vals[active_r])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(p) = 0 to preserve feasibility
    
    bounds = []
    for i in range(n):
        lo_b = max(-radius, -h[i])
        hi_b = min(radius, 1.0 - h[i])
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, 10.0 * n))  # t is free-ish
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n]
    return None


def _line_search(h, p, n, max_alpha=None):
    """Line search along direction p from h."""
    target = n * 0.5
    if max_alpha is None:
        max_alpha = 0.5
        for i in range(n):
            if p[i] > 1e-14:
                max_alpha = min(max_alpha, (1.0 - h[i]) / p[i])
            elif p[i] < -1e-14:
                max_alpha = min(max_alpha, h[i] / (-p[i]))
        max_alpha = max(max_alpha, 1e-10)
    
    best_a = 0.0
    best_c5 = _c5(h)
    # Grid search
    alphas = np.linspace(0.0, max_alpha, 20)
    for a in alphas:
        if a < 1e-12:
            continue
        h_t = _project(h + a * p, target)
        c5_t = _c5(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    
    if best_a <= 0:
        return 0.0, best_c5
    
    # Golden section refinement
    lo = max(0.0, best_a - max_alpha * 0.15)
    hi = best_a + max_alpha * 0.15
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(_project(h + c1 * p, target))
    f2 = _c5(_project(h + c2 * p, target))
    for _ in range(15):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(_project(h + c1 * p, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(_project(h + c2 * p, target))
    a = 0.5 * (lo + hi)
    c5_t = _c5(_project(h + a * p, target))
    if c5_t < best_c5:
        best_a, best_c5 = a, c5_t
    
    return best_a, best_c5


def _tp_loop(h, n, deadline, radius=0.1, max_iters=5000, stall_limit=80):
    """Trust-region LP loop."""
    h = h.copy()
    target = n * 0.5
    cur_c5 = _c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    stall = 0

            break

            radius *= 0.7

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _slsqp_solve(h_init, n, maxiter=200, ftol=1e-14):
    """SLSQP epigraph solve."""
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-3
    
    def _obj(z):
        return z[n]
    
    def _obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def _eq_f(z):
        return z[:n].sum() - n * 0.5
    
    def _eq_g(z):
        g = np.ones(n + 1)
        g[n] = 0.0
        return g
    
    def _ineq_f(z):
        return z[n] - np.max(_corr(z[:n]))
    
    # Can't easily get gradient of max, so use top-K
    def _ineq_g(z):
        h_z = z[:n]
        corr_z = _corr(h_z)
        top = int(np.argmax(corr_z))
        J = _build_grad_matrix(h_z, [top])
        g = np.zeros(n + 1)
        g[:n] = -J[0]
        g[n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    z0 = np.concatenate([h, [t0]])
    
    try:
        res = minimize(_obj, z0, jac=_obj_grad,
                       bounds=bounds,
                       constraints=[
                           {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g},
                           {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g}
                       ],
                       method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _c5(x)
    except Exception:
        return h, _c5(h)

import numpy as np
from scipy.optimize import linprog, minimize
import time


def _project(v, target):
    v = np.asarray(v, dtype=float).copy()
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.5
    hi = float(v.max()) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(v + lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-11:
        if diff > 0:
            order = np.argsort(x)
        else:
            order = np.argsort(-x)
        for i in order:
            if abs(diff) <= 1e-12:
                break
            if diff > 0:
                step = min(1.0 - x[i], diff)
                if step > 1e-15:
                    x[i] += step
                    diff -= step
            else:
                step = min(x[i], -diff)
                if step > 1e-15:
                    x[i] -= step
                    diff += step
    return x


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _grad_matrix(h, r_list):
    n = len(h)
    j = np.arange(n)
    K = len(r_list)
    J = np.zeros((K, n))
    for k, r in enumerate(r_list):
        s = int(r) - (n - 1)
        pd = np.zeros(n)
        gs = j + s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] += (1.0 - h[gs[m1]])
        ps = j - s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        J[k, :] = pd
    return J


def _lp_dir(h, corr, n, radius, active):

    m = len(active)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = J
    A_ub[:, n] = -1.0
    b_ub = -np.array([corr[r] for r in active])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for i in range(n):
        lo_b = max(-radius, -h[i])
        hi_b = min(radius, 1.0 - h[i])
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, 50.0 * n))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

            max_a = min(max_a, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-14:
            max_a = min(max_a, h[i] / (-p[i]))
    max_a = max(max_a, 1e-10)
    best_a, best_v = 0.0, _c5(h)
    for t in np.linspace(0.0, 1.0, 25):
        a = t * max_a
        if a < 1e-12:
            continue
        h_t = _project(h + a * p, target)
        v = _c5(h_t)
        if v < best_v:
            best_v, best_a = v, a
    if best_a <= 0:
        return 0.0, best_v
    lo = max(0.0, best_a - 0.15 * max_a)
    hi = best_a + 0.15 * max_a
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(_project(h + c1 * p, target))
    f2 = _c5(_project(h + c2 * p, target))
    for _ in range(18):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(_project(h + c1 * p, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(_project(h + c2 * p, target))
    a = 0.5 * (lo + hi)
    v = _c5(_project(h + a * p, target))
    if v < best_v:
        best_a, best_v = a, v
    return best_a, best_v


def _tp(h, n, t_end, r0=0.1, maxit=5000, stall_lim=100):
    h = h.copy()
    target = n * 0.5
    cur = _c5(h)
    best_h, best_c5 = h.copy(), cur
    radius = r0
    stall = 0
    n_corr = 2 * n - 1
    for _ in range(maxit):
        if time.time() > t_end:
            break
        if stall > stall_lim:
            break
        corr = _corr(h)
        raw_max = float(np.max(corr))
        sorted_idx = np.argsort(-corr)
        K = min(30, n_corr)
        active = sorted_idx[:K].astype(int)
        margin = max(1e-7, 0.005 * raw_max)
        near = np.where(corr >= raw_max - margin)[0]
        for ri in near:
            if ri not in active:
                active = np.append(active, ri)
        active = np.unique(active.astype(int))
        if len(active) > 40:
            best_40 = np.argsort(-corr[active])[:40]
            active = np.sort(active[best_40])
        p = _lp_dir(h, corr, n, radius, active)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.65
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _ls(h, p, n)
        if new_c5 < cur - 1e-14:
            h = _project(h + alpha * p, target)
            cur = _c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.4)
        else:
            stall += 1
            radius *= 0.6
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5


def _rsmp(h, n2):
    n = len(h)
    if n2 == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n2) + 0.5) / n2
    h2 = np.interp(xn, xo, h)
    return _project(h2, n2 * 0.5)


def _slsq_f(z, n):
    return z[n]


def _slsq_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _slsq_eq_f(z, n):
    return z[:n].sum() - n * 0.5


def _slsq_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _slsq_ineq_f(z, n):
    cc = _corr(z[:n])
    return z[n] - cc[int(np.argmax(cc))]


def _slsq_ineq_g(z, n):
    hz = z[:n]
    cc = _corr(hz)
    top = int(np.argmax(cc))
    J = _grad_matrix(hz, [top])
    g = np.zeros(n + 1)
    g[:n] = -J[0]
    g[n] = 1.0
    return g


def _slsq(h_init, n, maxiter=200):
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-3
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 20.0)]
    try:
        res = minimize(_slsq_f, z0, jac=_slsq_g, args=(n,),
                       bounds=bounds,
                       constraints=[
                           {'type': 'eq', 'fun': _slsq_eq_f, 'jac': _slsq_eq_g, 'args': (n,)},
                           {'type': 'ineq', 'fun': _slsq_ineq_f, 'jac': _slsq_ineq_g, 'args': (n,)}
                       ],
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _c5(x)
    except Exception:
        return h, _c5(h)


def _subgrad(h, n, step=0.05):
    corr = _corr(h)
    r_max = int(np.argmax(corr))
    J = _grad_matrix(h, [r_max])
    pd = J[0] - np.mean(J[0])
    nm = np.linalg.norm(pd)
    if nm < 1e-14:
        return h.copy(), _c5(h)
    pd = pd / nm * step
    best_h, best_c5 = h.copy(), _c5(h)
    for frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
        h_t = _project(h - frac * pd, n * 0.5)
        c5_t = _c5(h_t)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 * 0.5)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: Initial LP polish (8%)
    d1 = t0 + budget_s * 0.08
    h_p, c5_p = _tp(h, n0, d1, r0=0.1, maxit=3000, stall_lim=60)
    if c5_p < best_c5:
        best_h, best_c5 = h_p, c5_p

    # Phase 2: SLSQP + LP (12%)
    d2 = t0 + budget_s * 0.20
    h_s, c5_s = _slsq(best_h, best_n, maxiter=300)
    if c5_s < best_c5:
        best_h, best_c5 = h_s, c5_s
    h_p, c5_p = _tp(best_h, best_n, d2, r0=0.08, maxit=3000, stall_lim=50)
    if c5_p < best_c5:
        best_h, best_c5 = h_p, c5_p

    # Phase 3: Multi-start exploration (45%)
    d3 = t0 + budget_s * 0.65
    scale = 0.015
    it_count = 0
    while _tl() > budget_s * 0.32:
        it_count += 1
        # Diverse perturbations
        if it_count % 7 == 0:
            # Binary snap
            h_try = (best_h > 0.5).astype(float)
            h_try = _project(h_try, best_n * 0.5)
        elif it_count % 11 == 0:
            # Reverse
            h_try = _project(best_h[::-1], best_n * 0.5)
        else:
            h_try = _project(best_h + rng.normal(0, scale, best_n), best_n * 0.5)

        # Run LP
        sub_d = min(d3, time.time() + budget_s * 0.025)
        h_lp, c5_lp = _tp(h_try, best_n, sub_d, r0=0.08, maxit=800, stall_lim=35)
        if c5_lp < best_c5 - 1e-11:
            best_h, best_c5 = h_lp, c5_lp
            scale = max(0.003, scale * 0.7)
        else:
            scale = min(0.12, scale * 1.12)

        # Occasional SLSQP
        if it_count % 6 == 0 and _tl() > budget_s * 0.28:
            sub_d2 = min(d3, time.time() + budget_s * 0.015)
            h_s2, c5_s2 = _slsq(h_lp, best_n, maxiter=150)
            if c5_s2 < best_c5 - 1e-11:
                best_h, best_c5 = h_s2, c5_s2

        # Occasional subgradient escape
        if it_count % 13 == 0 and _tl() > budget_s * 0.30:
            h_sg, c5_sg = _subgrad(best_h, best_n, step=0.03)
            if c5_sg < best_c5 - 1e-11:
                best_h, best_c5 = h_sg, c5_s

    # Phase 4: Resolution search (15%)
    d4 = t0 + budget_s * 0.82
    for dn in [-2, 2, -4, 4, -6, 6, 1, -1, -8, 8]:
        if _tl() < budget_s * 0.18:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 160:
            continue
        h_r = _rsmp(best_h, n_try)
        sub_d = min(d4, time.time() + budget_s * 0.02)
        h_lp, c5_lp = _tp(h_r, n_try, sub_d, r0=0.08, maxit=600, stall_lim=30)
        if c5_lp < best_c5 - 1e-10:
            best_h, best_c5, best_n = h_lp, c5_lp, n_try
            sub_d2 = min(d4, time.time() + budget_s * 0.015)
            h_lp2, c5_lp2 = _tp(h_lp, best_n, sub_d2, r0=0.05, maxit=800, stall_lim=40)
            if c5_lp2 < best_c5:
                best_h, best_c5 = h_lp2, c5_lp2

    # Phase 5: Intensive final polish (15%)
    d5 = t0 + budget_s * 0.98
    # Tight LP
    h_p, c5_p = _tp(best_h, best_n, d5, r0=0.05, maxit=10000, stall_lim=80)
    if c5_p < best_c5:
        best_h, best_c5 = h_p, c5_p
    # SLSQP
    if _tl() > 10:
        h_s, c5_s = _slsq(best_h, best_n, maxiter=300)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
    # Final tight LP
    d5b = t0 + budget_s * 0.99
    if _tl() > 5:
        h_p2, c5_p2 = _tp(best_h, best_n, d5b, r0=0.02, maxit=5000, stall_lim=60)
        if c5_p2 < best_c5:
            best_h, best_c5 = h_p2, c5_p2

    # Validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
