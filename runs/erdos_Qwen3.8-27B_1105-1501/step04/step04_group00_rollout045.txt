    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search(h, p, max_iter=40):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j] * 0.99)
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]) * 0.99)
    alpha_max = max(alpha_max, 1e-10)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    h_c1 = h + c1 * p
    h_c2 = h + c2 * p
    h_c1 = _project(h_c1, n / 2.0)
    h_c2 = _project(h_c2, n / 2.0)
    f1 = _exact_c5(h_c1)
    f2 = _exact_c5(h_c2)
    best_a = c1 if f1 < f2 else c2
    best_c5 = min(f1, f2)
    for _ in range(max_iter):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            h_c1 = _project(h + c1 * p, n / 2.0)
            f1 = _exact_c5(h_c1)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            h_c2 = _project(h + c2 * p, n / 2.0)
            f2 = _exact_c5(h_c2)
        if f1 < best_c5:
            best_c5 = f1
            best_a = c1
        if f2 < best_c5:
            best_c5 = f2
            best_a = c2
    opt_a = (lo + hi) / 2.0
    h_opt = _project(h + opt_a * p, n / 2.0)
    c5_opt = _exact_c5(h_opt)
    if c5_opt < best_c5:
        return opt_a, c5_opt
    return best_a, best_c5

def _local_optimize(h, deadline, top_k_schedule=None, radius_init=0.15, max_iters=5000):
    if top_k_schedule is None:
        top_k_schedule = [None, 30, 15, 8, 5, 12]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    ti = 0
    n_sched = len(top_k_schedule)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        if cur_c5 < best_c5 - 1e-10:
            best_h = h.copy()
            best_c5 = cur_c5
        
        tk = top_k_schedule[ti % n_sched]
        scores = _scores(h)
        
        if tk is None:
            active_idx = np.arange(len(scores))
        else:
            active_idx = np.argsort(scores)[-tk:]
        
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, active_idx)
        
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-7:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            # Adaptive radius: increase if good progress
            if alpha > 0.5 * radius:
                radius = min(radius * 1.15, 0.5)
            else:
                radius = min(radius * 1.02, 0.5)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 10 and radius < 0.01:
                radius = 0.08
                ti = 0
    
    if _exact_c5(h) < best_c5:
        best_h = h.copy()
        best_c5 = cur_c5
    return best_h, best_c5

def _softmax_grad_step(h, deadline, gamma_start=10.0, gamma_end=500.0, n_steps=300, step_size_start=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        gamma = gamma_start * (gamma_end / gamma_start) ** frac
        step_size = step_size_start * (1.0 - 0.8 * frac)
        
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp((scores - max_s) / (gamma * max_s + 1e-10))
        sw = np.sum(w)
        w /= sw
        
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        alpha = step_size
        improved = False
        for _ in range(25):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.65
            if alpha < 1e-10:
                break
        if not improved:
            # Try smaller step or skip
            continue
    
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    h = np.zeros(n)
    x = (np.arange(n) + 0.5) / n
    h = ((x * 2 * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _staircase_init(n, n_steps=8):
    # Create a staircase pattern
    h = np.zeros(n)
    per_step = n // n_steps
    for i in range(n_steps):
        start = i * per_step
        end = min(start + per_step, n)
        frac = (i + 1) / n_steps
        h[start:end] = frac
    return _project(h, n / 2.0)

def _random_binary_init(n, rng):
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:n//2]] = 1.0
    return h
