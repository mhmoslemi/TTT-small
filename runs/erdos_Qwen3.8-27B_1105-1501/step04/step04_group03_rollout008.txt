def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx, dx

def exact_c5(h):

    return float(np.max(scores))

def project_equal(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Binary search for threshold
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:

    x = np.clip(x, 0.0, 1.0)
    # Final correction
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-14:
        idx = int(np.argmax(1.0 - x)) if residual2 > 0 else int(np.argmin(x))

def pair_line_search(h, i, j, sign, t_max, n_eval=20):
    """Find best t in [0, t_max] for direction sign*(e_i - e_j)."""
    best_t = 0.0
    best_c5 = exact_c5(h)
    
    # Coarse grid
    ts = np.linspace(0, t_max, n_eval)
    c5s = np.zeros(n_eval)
    h_trial = h.copy()
    for idx, t in enumerate(ts):
        h_trial[i] = h[i] + sign * t
        h_trial[j] = h[j] - sign * t
        c5s[idx] = exact_c5(h_trial)
    h_trial[i] = h[i]
    h_trial[j] = h[j]
    
    best_idx = int(np.argmin(c5s))
    if best_idx == 0:
        return 0.0, best_c5
    
    # Refine around best
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(n_eval - 1, best_idx + 1)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = lo
    b = hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = _eval_pair(h, i, j, sign, c1)
    f2 = _eval_pair(h, i, j, sign, c2)
    
    for _ in range(30):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = _eval_pair(h, i, j, sign, c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = _eval_pair(h, i, j, sign, c2)
    
    opt_t = (a + b) / 2.0
    opt_c5 = _eval_pair(h, i, j, sign, opt_t)
    if opt_c5 < c5s[best_idx]:
        return opt_t, opt_c5
    return ts[best_idx], c5s[best_idx]

def _eval_pair(h, i, j, sign, t):
    h[i] = h[i] + sign * t
    h[j] = h[j] - sign * t
    c5 = exact_c5(h)
    h[i] -= sign * t
    h[j] += sign * t
    return c5

def metropolis_phase(h, deadline, start_temp=0.003, end_temp=0.0001, 
                     decay=0.999, max_iter=20000):
    """Metropolis-style exploration with exact objective."""
    n = len(h)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    temp = start_temp
    accepted = 0
    evaluated = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if temp < end_temp:
            break
        
        # Choose random pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        sign = 1.0 if rng.random() < 0.5 else -1.0
        
        # Choose step size
        if sign > 0:
            t_max = min(1.0 - h[i], h[j])
        else:
            t_max = min(h[i], 1.0 - h[j])
        
        if t_max < 1e-6:
            temp *= decay
            continue
        
        # Adaptive step: use sqrt of current C5 as scale
        t = t_max * rng.random() * 0.1  # Small steps
        
        if t < 1e-8:
            temp *= decay
            continue
        
        # Evaluate
        h[i] += sign * t
        h[j] -= sign * t
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
        
        new_c5 = exact_c5(h)
        evaluated += 1
        
        if new_c5 <= cur_c5 + temp * rng.random():
            # Accept
            cur_c5 = new_c5
            accepted += 1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Reject - revert
            h[i] -= sign * t
            h[j] += sign * t
            h[i] = np.clip(h[i], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)
        
        temp *= decay
    
    return best_h, best_c5

def multi_start_descent(h, deadline, n_starts=5):
    """Try multiple random perturbations and descend from each."""
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(777)
    
    for s in range(n_starts):
        if time.time() > deadline:
            break
        # Perturb
        scale = 0.05 * (s + 1) / n_starts
        h_pert = h + rng.normal(0, scale, n)
        h_pert = project_equal(h_pert, n / 2.0)
        
        # Short descent
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) / (n_starts - s))
        h_try, c5_try = descent_phase(h_pert, sub_deadline, max_iter=500)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return best_h, best_c5

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_equal(h_new, n_new / 2.0)

def metropolis_search(h, deadline, start_temp=0.003, end_temp=5e-5,
                      decay=0.9995, max_iter=50000, step_scale=0.05):
    """Simulated annealing with paired moves and exact objective."""
    n = len(h)
    rng = np.random.default_rng(999)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    temp = start_temp
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if temp < end_temp:
            break
        
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        sign = 1.0 if rng.random() < 0.5 else -1.0
        
        if sign > 0:
            t_max = min(1.0 - h[i], h[j])
        else:
            t_max = min(h[i], 1.0 - h[j])
        
        if t_max < 1e-8:
            temp *= decay
            continue
        
        t = t_max * step_scale * rng.random()
        if t < 1e-9:
            temp *= decay
            continue
        
        # Apply move
        oi = h[i]
        oj = h[j]
        h[i] = oi + sign * t
        h[j] = oj - sign * t
        
        new_c5 = exact_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = oi
            h[j] = oj
        
        temp *= decay
    
    return best_h, best_c5
