import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search on the quadratic-in-t objective
    # f(t) = max_k (S0[k] + t*T[k] - t^2*C[k])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgrad_h(h, scores, dx, tau):
    """Compute smoothed subgradient using softmax over shifts."""
    n = len(h)
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    # Subgradient: sum_k w_k * grad f_k
    # grad f_k = G[k] (from grad matrix)
    # But we can compute more efficiently:
    # For shift d (where d = r - (n-1)):
    #   d >= 0: grad = (1-h[d:]) in first (n-d) positions, -h[:n-d] in last d positions... 
    # Actually let me use the grad matrix approach but exploit sparsity
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g+m])
                grad[g:] += w[r] * (1.0 - h[:m])
    return grad

def _policy_gradient(h, deadline, tau_init=30.0, tau_max=500.0, tau_growth=1.03):
    """Subgradient method with adaptive softmax temperature."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    lr = 0.15
    mom = np.zeros(n)
    mom_decay = 0.8
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            # Reset with perturbation
            rng = np.random.default_rng(hash(tuple(h[:10])) % 2**32)
            h = h + rng.normal(0, 0.05, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            mom = np.zeros(n)
            stall = 0
            lr = 0.1
            tau = tau_init * 0.5
            continue
        scores = _all_c5(h)
        d = _subgrad_h(h, scores, dx, tau)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            tau *= tau_growth
            continue
        d = d / dn
        mom = mom_decay * mom + (1.0 - mom_decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.2)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            tau = min(tau * tau_growth, tau_max)
            lr = min(lr * 1.03, 0.5)
        else:
            stall += 1
            lr *= 0.9
            if stall > 20:
                tau = max(tau_init, tau * 0.8)
                mom = 0.5 * mom
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()

                best_c5 = cur_c5
                best_h = h.copy()

        else:

    return best_h, best_c5

def _interpolate(h1, h2, n_target):
    """Interpolate between two solutions and find best point on the segment."""
    if len(h1) != len(h2):
        # Resample to common size
        if len(h1) < len(h2):
            h1 = _resample(h1, len(h2))
        else:
            h2 = _resample(h2, len(h1))
    n = len(h1)
    if n_target is not None and n_target != n:
        n = n_target
        x = (np.arange(n) + 0.5) / n
        h1_r = np.interp(x, (np.arange(len(h1)) + 0.5) / len(h1), h1)
        h2_r = np.interp(x, (np.arange(len(h2)) + 0.5) / len(h2), h2)
        h1, h2 = _project(h1_r, n/2), _project(h2_r, n/2)
    
    # Additive workflow:
    # Try lambda from 0 to 1
    best_c5 = min(_exact_c5(h1), _exact_c5(h2))
    best_h = h1.copy() if _exact_c5(h1) < _exact_c5(h2) else h2.copy()
    
    for lam in np.linspace(0, 1, 21):
        h_mid = lam * h1 + (1 - lam) * h2
        h_mid = _project(h_mid, n / 2.0)
        c5_mid = _exact_c5(h_mid)
        if c5_mid < best_c5:
            best_c5 = c5_mid
            best_h = h_mid.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine_fast(h, deadline, max_passes=30):

def _drift_minimization(h, deadline):
    """Try to equalize the active shifts to reduce the max."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    while time.time() < deadline:
        scores = _all_c5(h)
        max_s = float(np.max(scores))
        # Find active shifts (within 10% of max)
        active = scores > 0.9 * max_s
        # Compute gradient of only the active shifts (uniform weight)
        grad = np.zeros(n)
        n_active = int(np.sum(active))
        for r in range(2 * n - 1):
            if not active[r]:
                continue
            w = 1.0 / n_active
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w * (1.0 - h[d:d+m])
                    grad[d:] += w * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w * (-h[g:g+m])
                    grad[g:] += w * (1.0 - h[:m])
        
        # Also add a small regularization to reduce variance of scores
        var_grad = np.zeros(n)
        for r in range(2 * n - 1):
            diff = scores[r] - max_s
            if abs(diff) < 1e-10:
                continue
            w = diff / (n_active * max_s) if n_active > 0 else 0
            if abs(w) < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    var_grad[:m] += w * (1.0 - h[d:d+m])
                    var_grad[d:] += w * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    var_grad[:m] += w * (-h[g:g+m])
                    var_grad[g:] += w * (1.0 - h[:m])
        
        total_grad = grad + 0.3 * var_grad
        gn = np.linalg.norm(total_grad)
        if gn < 1e-13:
            break
        total_grad /= gn
        alpha, new_c5 = _line_search(h, 0.1 * total_grad, dx, 0.12)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * 0.1 * total_grad
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

from scipy.optimize import linprog
import cvxpy as cp
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(55):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, max_stall=60):

    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    stall = 0

def _cvx_start(n, timeout=60):
    """Solve the McCormick relaxation to get a good starting point."""
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    for d in range(1, n):
        m = n - d
        constraints.append(cp.sum(cp.pos(h[:m] - h[d:])) <= t / dx)
        constraints.append(cp.sum(cp.pos(h[d:] - h[:m])) <= t / dx)
    
    prob = cp.Problem(cp.Minimize(t + 0.01 * cp.norm(h, 2)), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, float('inf')
    
    if h.value is not None:
        h_val = np.clip(h.value, 0, 1)
        h_val = _project(h_val, n / 2.0)
        return h_val, _exact_c5(h_val)
    return None, float('inf')

def _subgrad_loop(h, deadline, tau_init=30.0, tau_max=400.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    lr = 0.12
    mom = np.zeros(n)
    mom_decay = 0.82
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        scores = _all_scores(h)
        max_s = float(np.max(scores))
        exp_s = np.exp(tau * (scores - max_s))
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:d+m])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (-h[g:g+m])
                    grad[g:] += w[r] * (1.0 - h[:m])
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            tau = min(tau * 1.1, tau_max)
            continue
        d = grad / gn
        mom = mom_decay * mom + (1.0 - mom_decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.3)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            tau = min(tau * 1.04, tau_max)
            lr = min(lr * 1.02, 0.35)
        else:
            stall += 1
            lr *= 0.88
            if stall % 20 == 0:
                tau = max(tau_init, tau * 0.7)
                mom *= 0.5
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_round(h):
    """Round h to a binary solution with sum = n/2."""
    n = len(h)
    target = int(n / 2)
    # Sort by value, take top target as 1
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def _pair_refine(h, deadline, max_passes=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        free_idx = np.where(free)[0]
        if len(free_idx) < 2:
            break
        n_tries = min(300, len(free_idx) * 3)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i, j = rng.choice(free_idx, size=2, replace=False)
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.4)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import cvxpy as cp
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(55):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(scores)[::-1][:top_k]
        G_use = G[idx]
        b_use = -scores[idx]
    else:
        G_use = G
        b_use = -scores
    n_rows = G_use.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_rows, n + 1))
    A_ub[:, :n] = G_use
    A_ub[:, n] = -1.0
    b_ub = b_use
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, active_frac=0.4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    k_ratio = active_frac
    while time.time() < deadline:
        if stall > 50:
            if k_ratio < 1.0:
                k_ratio = min(1.0, k_ratio + 0.2)
                stall = 0
                continue
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        top_k = max(int(k_ratio * (2 * n - 1)), 10)
        p = _lp_direction(h, scores, G, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            k_ratio = max(0.2, k_ratio * 0.9)
            radius = min(radius * 1.05, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall > 15:
                k_ratio = min(1.0, k_ratio + 0.15)
    return best_h, best_c5

def _subgrad_loop(h, deadline, tau_init=30.0, tau_max=500.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    lr = 0.12
    mom = np.zeros(n)
    mom_decay = 0.82
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        scores = _all_scores(h)
        max_s = float(np.max(scores))
        exp_s = np.exp(tau * (scores - max_s))
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-12:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:d + m])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (-h[g:g + m])
                    grad[g:] += w[r] * (1.0 - h[:m])
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            tau = min(tau * 1.1, tau_max)
            continue
        d = grad / gn
        mom = mom_decay * mom + (1.0 - mom_decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.3)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            tau = min(tau * 1.04, tau_max)
            lr = min(lr * 1.02, 0.35)
        else:
            stall += 1
            lr *= 0.88
            if stall % 15 == 0:
                tau = max(tau_init, tau * 0.7)
                mom *= 0.5
    return best_h, best_c5

def _cvx_start(n, deadline):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    max_shift = min(n - 1, 60)
    for d in range(1, max_shift + 1):
        m = n - d
        constraints.append(cp.sum(cp.pos(h_var[:m] - h_var[d:])) <= t_var / dx)
        constraints.append(cp.sum(cp.pos(h_var[d:] - h_var[:m])) <= t_var / dx)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    if time.time() > deadline:
        return None
    if h_var.value is not None:
        h_val = np.clip(h_var.value, 0, 1)
        h_val = _project(h_val, n / 2.0)
        return h_val
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(2024)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        free_idx = np.where(free)[0]
        if len(free_idx) < 2:
            break
        n_tries = min(200, len(free_idx) * 2)
        for _ in range(n_tries):
            if time.time() > deadline:
                break
            i, j = rng.choice(free_idx, size=2, replace=False)
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.4)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _focus_refine(h, deadline):
    """Iteratively focus on the worst shift and reduce it."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    while time.time() < deadline and stall < 40:
        scores = _all_scores(h)
        max_s = float(np.max(scores))
        active = np.where(scores >= 0.95 * max_s)[0]
        if len(active) == 0:
            break
        grad = np.zeros(n)
        for r in active:
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:d + m])
                    grad[d:] += (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += (-h[g:g + m])
                    grad[g:] += (1.0 - h[:m])
        grad /= len(active)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        d = grad / gn
        alpha, new_c5 = _line_search(h, 0.08 * d, dx, 0.1)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * 0.08 * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5
