**Approach: Directly optimize using the quadratic structure**

Since the objective along any zero-sum direction is a max of quadratics, I can formulate the problem as:
- Minimize t
- Subject to: for all lags d, y_d(h + V) ≤ t
- For all i: 0 ≤ (h + V)_i ≤ 1
- sum(h + V) = n/2

This is a QP (or even an SOCP if we use SOS1 constraints). But the number of constraints is O(n) for the lags plus O(n) for the box, which is manageable.

Actually, this is non-convex because y_d(h) is a non-convex quadratic (it's -h^T B_d h + 1^T h, and B_d is not PSD in general).

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_grid = 16
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_grid - 1):
        return best_a, best_c5
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    oa = (lo + hi) / 2.0
    oc = _exact_c5(h + oa * p)
    if oc < best_c5:
        return oa, oc
    return best_a, best_c5

def _pnorm_descent(h, deadline, p_start=6.0, p_end=4096.0, n_steps=1500, momentum=0.3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    v = np.zeros(n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        v = momentum * v + (1 - momentum) * d
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            v = d
            vn = 1.0
        else:
            v /= vn
        found = False
        for scale in [0.15, 0.10, 0.06, 0.03, 0.015]:
            alpha, new_c5 = _line_search(h, v * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * v * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            v = 0.7 * v
            stall += 1
            if stall > 150:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_subgrad(h, deadline, top_k=12, radius=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.75
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.75
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.75
            if r < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.03, 0.5)
        else:
            stall += 1
            r *= 0.82
            if stall > 40 and r < 0.015:
                r = 0.06
    return best_h, best_c5

def _pair_descent_fast(h, deadline, max_rounds=20):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    scores = _scores(h)
    n_lags = 2 * n - 1
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for p_idx in range(n):
            if time.time() > deadline:
                break
            if h[p_idx] < 0.01 or h[p_idx] > 0.99:
                continue
            for q_idx in range(n):
                if q_idx == p_idx:
                    continue
                if h[q_idx] < 0.01 or h[q_idx] > 0.99:
                    continue
                alpha_max = min(1.0 - h[p_idx], h[q_idx])
                if alpha_max < 0.005:
                    continue
                p_dir = np.zeros(n)
                p_dir[p_idx] = 1.0
                p_dir[q_idx] = -1.0
                alpha_opt, c5_opt = _line_search(h, p_dir, alpha_max, max_alpha=alpha_max)
                if c5_opt < cur_c5 - 1e-12:
                    h = h + alpha_opt * p_dir
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, n / 2.0)
                    cur_c5 = _exact_c5(h)
                    scores = _scores(h)
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_init(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h

def _periodic_init(n, period, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    block = np.zeros(period)
    ones_per_block = max(1, period // 2)
    idx = rng.choice(period, size=ones_per_block, replace=False)
    block[idx] = 1.0
    for i in range(n):
        h[i] = block[i % period]
    return _project(h, n / 2.0)
```
