def resample_h(h, n_new):

    if n_new == n:

    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return project_to_box_sum(h_new, n_new / 2.0)

def quasi_periodic_init(n, alpha, rng):

            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            k = (j + step_size) % n
            if k == j:
                continue
            if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                continue
            for sign in [1, -1]:
                if sign > 0:
                    a_max = min(1.0 - h[j], h[k])
                else:
                    a_max = min(h[j], 1.0 - h[k])
                if a_max < 1e-10:
                    continue
                lo_a, hi_a = 0.0, a_max
                for _ in range(10):
                    m1 = lo_a + (hi_a - lo_a) / 3.0
                    m2 = hi_a - (hi_a - lo_a) / 3.0
                    h1 = h.copy()
                    h1[j] += sign * m1
                    h1[k] -= sign * m1
                    h1 = np.clip(h1, 0, 1)
                    f1 = exact_c5(h1)
                    h2 = h.copy()
                    h2[j] += sign * m2
                    h2[k] -= sign * m2
                    h2 = np.clip(h2, 0, 1)
                    f2 = exact_c5(h2)
                    if f1 < f2:
                        hi_a = m2
                    else:
                        lo_a = m1
                a_opt = (lo_a + hi_a) / 2.0
                h_try = h.copy()
                h_try[j] += sign * a_opt
                h_try[k] -= sign * a_opt
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_to_box_sum(h_try, n / 2.0)
                c5_try = exact_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_sa(n, deadline, rng, n_iters=30000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.04
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 500 == 0:
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cool, T_min)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = project_to_box_sum(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = exact_c5(h_start)
    best_n = n0

    # Phase 1: Optimize parent with active LS
    if time.time() < hard_deadline:
        dl1 = t0 + budget_s * 0.10
        h1, c5_1 = optimize_active(h_start, dl1, max_iters=3000)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Multi-resolution diverse search
    dl2 = t0 + budget_s * 0.48
    resolutions = [64, 80, 96, 112, 128, 144, 160]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.7236]
    res_idx = 0
    for n_try in resolutions:
        if time.time() > dl2:
            break
        if best_c5 < 0.3795:
            break
        n_starts = 3
        for s in range(n_starts):
            if time.time() > dl2:
                break
            sub_dl = min(dl2, time.time() + budget_s * 0.04)
            if s == 0:
                h_init = resample_h(best_h, n_try)
            elif s == 1:
                alpha = alphas[res_idx % len(alphas)]
                h_init = quasi_periodic_init(n_try, alpha, rng)
            else:
                h_init = smooth_random_init(n_try, rng, k=3)
            res_idx += 1
            h_try, c5_try = optimize_active(h_init, sub_dl, max_iters=2500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 3: Binary SA at several resolutions
    if time.time() < t0 + budget_s * 0.62:
        dl3 = t0 + budget_s * 0.65
        for n_bin in [64, 96, 128]:
            if time.time() > dl3:
                break
            sub_dl = min(dl3, time.time() + budget_s * 0.03)
            h_b, c5_b = binary_sa(n_bin, sub_dl, rng, n_iters=30000)
            if time.time() < dl3 and c5_b < best_c5 + 0.005:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.02)
                h_bp, c5_bp = optimize_active(h_b, sub_dl2, max_iters=1500)
                if c5_bp < best_c5:
                    best_c5 = c5_bp
                    best_h = h_bp.copy()
                    best_n = n_bin

    # Phase 4: Perturbation restarts around best
    dl4 = t0 + budget_s * 0.82
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl4:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = project_to_box_sum(h_pert, best_n / 2.0)
        sub_dl = min(dl4, time.time() + budget_s * 0.012)
        h_try, c5_try = optimize_active(h_pert, sub_dl, max_iters=800)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.15)

    # Phase 5: Fine polish with tighter parameters
    if time.time() < t0 + budget_s * 0.93:
        dl5 = t0 + budget_s * 0.94
        fine_variants = [
            dict(tol=5e-5, min_a=5, max_a=12, pp=5000, mf=0.05, mfl=3e-6, ridge=5e-4, rough=0.0, trust=0.05),
            dict(tol=1e-4, min_a=8, max_a=20, pp=2000, mf=0.08, mfl=5e-6, ridge=1e-3, rough=1e-4, trust=0.08),
            dict(tol=2e-4, min_a=10, max_a=25, pp=500, mf=0.12, mfl=1e-5, ridge=2e-3, rough=2e-3, trust=0.12),
        ]
        h_f, c5_f = optimize_active(best_h, dl5, variants=fine_variants, max_iters=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: Couple polish
    if time.time() < t0 + budget_s * 0.96:
        dl6 = t0 + budget_s * 0.97
        hp, c5p = couple_polish(best_h, dl6, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: tiny refinement
    if time.time() < hard_deadline:
        final_variants = [
            dict(tol=2e-5, min_a=4, max_a=10, pp=8000, mf=0.03, mfl=2e-6, ridge=3e-4, rough=0.0, trust=0.03),
        ]
        h_fin, c5_fin = optimize_active(best_h, hard_deadline, variants=final_variants, max_iters=1000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_box_sum(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
