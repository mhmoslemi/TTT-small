def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    # Project v onto {x: 0<=x_i<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def fw_step(h, g, target, gamma):
    """Frank-Wolfe step: find v in feasible set minimizing <g,v>, then interpolate."""
    # The optimal v for min <g,v> s.t. 0<=v<=1, sum=v=target:
    # Sort g, set v[i]=1 for smallest g, v[i]=0 for largest g
    idx = np.argsort(g)
    v = np.zeros(len(h))
    v[idx[:int(target)]] = 1.0
    # Handle fractional part
    frac = target - int(target)
    if frac > 1e-10:
        v[idx[int(target)]] = frac
    return (1-gamma)*h + gamma*v

def frank_wolfe(h, deadline, gamma_init=1.0, max_iter=5000):

def binary_round(h):
    """Round h to binary (exactly n/2 ones)."""
    n = len(h)
    target = int(n / 2.0)
    h_sorted = np.sort(h)[::-1]
    b = np.zeros(n)
    # Take the top n/2
    thresh = h_sorted[target - 1]
    b[h > thresh] = 1.0
    # Handle ties
    n_ones = int(np.sum(b))
    if n_ones < target:
        ties = np.where((h == thresh) & (b == 0))[0]
        for i in range(target - n_ones):
            if i < len(ties):
                b[ties[i]] = 1.0
    elif n_ones > target:
        ties = np.where((h == thresh) & (b == 1))[0]
        for i in range(n_ones - target):
            if i < len(ties):
                b[ties[i]] = 0.0
    return b

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def shift_grad(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            g[0:m] += (1.0 - h[d:n])
            g[d:n] -= h[0:m]
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[gg:n] += (1.0 - h[0:n-gg])
            g[0:n-gg] -= h[gg:n]
    return g

def topk_subgrad(h, k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:k]
    g = np.zeros(n)
    for idx in top_idx:
        d = idx - (n - 1)
        g += shift_grad(h, d)
    return dx * g / k

def fw_vertex(h, g):
    n = len(h)
    target = n / 2.0
    idx = np.argsort(g)
    v = np.zeros(n)
    k = int(target)
    v[idx[:k]] = 1.0
    frac = target - k
    if frac > 1e-12:
        v[idx[k]] = frac
    return v

def frank_wolfe(h, deadline, gamma_init=1.0, max_iter=20000, topk=3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = topk_subgrad(h, topk)
        v = fw_vertex(h, g)
        best_g = 0.0
        best_c5_s = cur_c5
        for frac in [0.3, 0.5, 0.7, 1.0]:
            a = gamma * frac
            h_t = project((1 - a) * h + a * v, target)
            c5_t = exact_c5(h_t)
            if c5_t < best_c5_s:
                best_c5_s = c5_t
                best_g = a
        if best_g > 0 and best_c5_s < cur_c5 - 1e-14:
            h = project((1 - best_g) * h + best_g * v, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.02, 1.0)
        else:
            stall += 1
            gamma *= 0.9
            if stall > 100:
                break
    return best_h, best_c5
