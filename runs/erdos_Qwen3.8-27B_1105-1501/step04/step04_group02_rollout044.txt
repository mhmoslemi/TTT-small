def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        if s > target:
            lo = mid
        else:

        if res > 0:

        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def _score_grad_matrix(h):
    """Compute (2n-1, n) matrix of gradients of raw scores."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        if d >= 0:
            m = n - d
            G[r, :m] -= h[d:d+m]
            G[r, d:d+m] += 1.0 - h[:m]
        else:
            s = -d
            m = n - s
            G[r, :m] += 1.0 - h[s:s+m]
            G[r, s:s+m] -= h[:m]
    return G
