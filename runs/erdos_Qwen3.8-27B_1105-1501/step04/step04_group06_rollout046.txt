Note that corr[s + n-1] + corr[-s + n-1] = Σ_j h[j] + Σ_j h[j] - n = ... hmm, not quite.

corr[-s+(n-1)] = Σ_j h[j](1-h[j+s]) where j ranges over valid indices.

import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_grad(h, s, n):
    """Gradient of corr at shift s wrt h. corr_s = sum_j h[j](1-h[j-s])."""
    j = np.arange(n)
    g = np.zeros(n)
    # Term 1: d/dh[j] of h[j]*(1-h[j-s]) at j=i gives (1-h[i-s]) if i-s valid
    idx1 = (j - s >= 0) & (j - s < n)
    g[idx1] += (1.0 - h[j[idx1] - s])
    # Term 2: d/dh[i] of h[j]*(1-h[i]) where j = i+s, gives -h[i+s] if i+s valid
    idx2 = (j + s >= 0) & (j + s < n)
    g[idx2] -= h[j[idx2] + s]
    return g


def _obj(z, n):
    return z[n]


def _obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq_fun(z, n):
    return np.sum(z[:n]) - n * 0.5


def _eq_grad(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _ineq_fun(z, active, n):
    h, t = z[:n], z[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr[active]


def _ineq_grad(z, active, n):
    h = z[:n]
    n_act = len(active)
    J = np.zeros((n_act, n + 1))
    for k, s_idx in enumerate(active):
        s = s_idx - (n - 1)
        g = _corr_grad(h, s, n)
        J[k, :n] = -g
        J[k, n] = 1.0
    return J


def _active_slsqp(h, n, deadline, K_start=4, K_max=12, max_outer=50):
    """Active-set SLSQP: start with few constraints, add violated ones."""
    h = h.copy()
    target = n * 0.5
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        t_val = float(np.max(corr))
        # Select top-K active constraints
        K = min(K_start + outer * 2, K_max, 2 * n - 1)
        top_idx = np.argsort(-corr)[:K]
        t0 = t_val + 1e-6
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
        cons = [
            {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_grad, 'args': (n,)},
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_grad, 'args': (top_idx, n,)}
        ]
        try:
            res = minimize(_obj, z0, jac=_obj_grad, args=(n,),
                           bounds=bounds, constraints=cons, method='SLSQP',
                           options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_new = _project(res.x[:n], target)
        except Exception:
            h_new = h.copy()
        # Check for violated constraints
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        t_new = float(np.max(corr_new))
        # Are all non-active constraints satisfied?
        active_mask = np.zeros(2 * n - 1, dtype=bool)
        active_mask[top_idx] = True
        violated = ~active_mask & (corr_new > t_new - 1e-10)
        if not np.any(violated):
            h = h_new
            if _exact_c5(h) < t_new * 2.0 / n + 1e-12:
                break
            # Check KKT-like condition: all corr <= t_new
            if np.max(corr_new) <= t_new + 1e-9:
                break
        else:
            # Add violated constraints for next iteration
            K_start = min(K + 2, K_max)
            h = h_new
        h = h_new
    return h, _exact_c5(h)


def _balance_step(h, n, n_active=5, max_iters=50):
    """Level active correlations by solving a linear system."""
    h = h.copy()
    target = n * 0.5
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        t_val = float(np.max(corr))
        # Get top active
        idx = np.argsort(-corr)[:n_active]
        active_corrs = corr[idx]
        # Target: level them to their mean
        t_target = float(np.mean(active_corrs))
        # Build system: G d = b
        # For each active s: sum_i grad_corr_s[i] * d[i] = active_corr[s] - t_target
        # Plus: sum_i d[i] = 0
        m = len(idx) + 1
        G = np.zeros((m, n))
        b = np.zeros(m)
        for k, s_idx in enumerate(idx):
            s = s_idx - (n - 1)
            g = _corr_grad(h, s, n)
            G[k, :] = g
            b[k] = active_corrs[k] - t_target
        G[m - 1, :] = 1.0
        b[m - 1] = 0.0
        # Solve least squares
        try:
            d, _, _, _ = np.linalg.lstsq(G, b, rcond=None)
        except Exception:
            break
        # Line search
        best_t = t_val
        best_h = h.copy()
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            h_try = _project(h + alpha * d, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_t * 2.0 / n:
                best_t = c5_try * n / 2.0
                best_h = h_try
        if np.allclose(best_h, h, atol=1e-14):
            break
        h = best_h
    return h, _exact_c5(h)


def _coord_descent(h, n, deadline, n_sweeps=200):
    """Coordinate-wise line search: for each coordinate, try to reduce max corr."""
    h = h.copy()
    target = n * 0.5

                    h[i] = new_val
                    # Adjust another coordinate to maintain sum
                    # Find coordinate with most room
                    diff = target - np.sum(h)

                        pass
                    elif diff > 0:
                        # Need to increase somewhere

                        if j == i:
                            h[i] = old_val
                            break
                        adj = min(1.0 - h[j], diff)
                        h[j] += adj
                        diff -= adj
                        if abs(diff) > 1e-10:
                            # Can't fully fix, skip
                            h[i] = old_val
                            h[j] -= adj
                            break
                    else:
                        j = np.argmin(h)
                        if j == i:
                            h[i] = old_val
                            break
                        adj = min(h[j], -diff)
                        h[j] -= adj
                        diff += adj
                        if abs(diff) > 1e-10:

                        best_hi = new_val

def _perturb_refine(h, n, deadline, rng, n_trials=10000):

    best_c5 = cur_c5
    best_h = h.copy()
    scale = 0.05

        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try
            h = h_try
            scale = max(0.005, scale * 0.95)
        else:
            scale = min(0.15, scale * 1.02)
    return best_h, best_c5

def _gen_binary_init(n, pattern, rng):
    """Generate binary initialization with exactly n/2 ones."""
    n_half = n // 2
    if pattern == 'random':
        h = np.zeros(n)
        idx = rng.permutation(n)[:n_half]
        h[idx] = 1.0
    elif pattern == 'blocks':
        h = np.zeros(n)
        h[:n_half] = 1.0
        # Randomly permute
        h = h[rng.permutation(n)]
    else:  # alternating
        h = np.zeros(n)
        h[::2] = 1.0
        if n_half != n // 2:
            h[1] = 1.0 - h[1]
    return h
