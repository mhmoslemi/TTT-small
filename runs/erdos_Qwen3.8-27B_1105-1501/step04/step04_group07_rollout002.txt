def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

So minimizing max_k ∑_j h[j]·(1-h[j+k])·dx is equivalent to maximizing min_k ∑_j h[j]·h[j+k]·dx!

This is a much nicer formulation! We want to find h (with ∫h=1, 0≤h≤1) that maximizes the minimum autocorrelation.

Let me denote A_k(h) = ∑_j h[j]·h[j+k]·dx. We want to maximize min_k A_k(h).

This is a "max-min" problem, which is dual to the "min-max" problem we had before. The max-min formulation might be easier to work with.

For the coordinate descent: changing h[i] affects A_k for various k. The coefficient of h[i] in A_k is:
- From j=i: h[i+k] (if i+k < n)
- From j+k=i: h[i-k] (if i-k ≥ 0)

So A_k = base_k + h[i]·(h[i+k]·[i+k<n] + h[i-k]·[i-k≥0])

And we want to max_v min_k (base_k + c_k·v) over v∈[0,1].

This is a concave function of v (min of affines is concave), and we're maximizing it. The maximum is at a point where the subdifferential contains 0, or at a boundary.

This is analogous to before but with max→min and min→max. The algorithm is the same: find all breakpoints of the lower envelope, evaluate at each, take the best.

This formulation might be slightly better numerically because we're maximizing (finding the "top" of the lower envelope) rather than minimizing (finding the "bottom" of the upper envelope).

def _autocorr_scores(h):
    """Compute A_k(h) = sum_j h[j]*h[j+k]*dx for all k."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    return corr * dx

def _coord_optimal_maxmin(h, i):
    """Find optimal h[i] to maximize min_k A_k, given others fixed."""
    n = len(h)
    m = 2 * n - 1
    scores = _autocorr_scores(h)
    
    c = np.zeros(m)
    for d in range(n - i):
        c[n-1+d] += h[i+d]
    for d in range(-i, 0):
        c[n-1+d] += h[i+d]
    
    base = scores - c * h[i]
    
    # Maximize min_k (base[k] + c[k]*v) over v in [0,1]
    # This is a concave piecewise linear function of v
    # Maximum is at a vertex of the lower envelope or at a boundary
    
    candidates = [0.0, 1.0]
    c_diff = c[:, None] - c[None, :]
    b_diff = base[None, :] - base[:, None]
    mask = np.abs(c_diff) > 1e-15
    tri = np.triu(np.ones((m, m), dtype=bool), k=1)
    mask &= tri
    if np.any(mask):
        bpts = b_diff[mask] / c_diff[mask]
        bpts = bpts[(bpts >= -1e-10) & (bpts <= 1.0 + 1e-10)]
        if len(bpts) > 0:
            candidates.extend(bpts.tolist())
    
    best_v = 0.0
    best_val = -float('inf')
    for v in candidates:
        v = max(0.0, min(1.0, v))
        val = float(np.min(base + c * v))
        if val > best_val:
            best_val = val
            best_v = v
    
    return best_v

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _autocorr_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coord_optimal_maxmin(h, i):
    n = len(h)
    m = 2 * n - 1
    scores = _autocorr_scores(h)
    c = np.zeros(m)
    for d in range(n - i):
        c[n - 1 + d] += h[i + d]
    for d in range(-i, 0):
        c[n - 1 + d] += h[i + d]
    base = scores - c * h[i]
    candidates = [0.0, 1.0]
    c_diff = c[:, None] - c[None, :]
    b_diff = base[None, :] - base[:, None]
    mask = np.abs(c_diff) > 1e-14
    tri = np.triu(np.ones((m, m), dtype=bool), k=1)
    mask &= tri
    if np.any(mask):
        bpts = b_diff[mask] / c_diff[mask]
        bpts = bpts[(bpts >= -1e-10) & (bpts <= 1.0 + 1e-10)]
        if len(bpts) > 0:

    for v in candidates:
        v = max(0.0, min(1.0, v))
        val = float(np.min(base + c * v))
        if val > best_val:
            best_val = val
            best_v = v
    return best_v

def _coord_descent(h, deadline, max_sweeps=500, order='sequential'):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stalls = 0
    rng = np.random.default_rng(7)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stalls > 15:
            break
        if order == 'random' or (sweep > 10 and sweep % 3 == 0):
            idx = np.arange(n)
            rng.shuffle(idx)
        else:
            idx = np.arange(n)
        sweep_improved = False
        for i in idx:
            if time.time() > deadline:
                break
            v_opt = _coord_optimal_maxmin(h, i)
            if abs(v_opt - h[i]) > 1e-11:
                h[i] = v_opt
                sweep_improved = True
        if sweep_improved:

            new_c5 = _exact_c5(h)
            if new_c5 < best_c5 - 1e-14:

                best_h = h.copy()
            stalls = 0
        else:

    return best_h, best_c5

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, max_alpha=0.5):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_s = 30
    alphas = np.linspace(0, alpha_max, n_s)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_s - 1):
        return best_a, best_c5
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    oa = (lo + hi) / 2.0
    oc = _exact_c5(h + oa * p)
    if oc < best_c5:
        return oa, oc
    return best_a, best_c5

def _lp_optimize(h, deadline, top_k=None, radius=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.03, 0.5)
        else:
            stall += 1
            r *= 0.8
            if stall > 40 and r < 0.02:
                r = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pnorm_gradient_step(h, scores, G, p_power, alpha_scale):
    n = len(h)
    n_lags = len(scores)
    s_max = float(scores.max())
    if s_max < 1e-15:
        return None
    ratio = scores / s_max
    w = ratio ** p_power
    sw = float(w.sum())
    if sw < 1e-300:
        w = np.zeros(n_lags)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    grad = G.T @ w
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return -grad / gn * alpha_scale

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1000):

    target = n / 2.0

    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        d = _pnorm_gradient_step(h, scores, G, p, 0.1)
        if d is None:

            if stall > 60:
                break
            continue
        found = False
        for scale in [1.0, 0.5, 0.25, 0.1]:

            if new_c5 < cur_c5 - 1e-14:

                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 80:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5
