def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _grad_rows(h, rows, n):
    j = np.arange(n)
    G = np.zeros((len(rows), n))
    for idx, r in enumerate(rows):
        s = r - (n - 1)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pm = gs[m1]
        G[idx, j[m1]] += (1.0 - h[pm])
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        G[idx, j[m2]] -= h[ps[m2]]
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _pair_search(h, n, deadline, max_rounds=200):

def _lse_smooth(h, n, alpha, maxiter=60):
    """Log-sum-exp smoothing step using L-BFGS-B."""
    def lse_obj(x):
        hx = _project(x, n * 0.5)
        corr = _corr(hx)
        ms = float(np.max(corr))
        vals = np.clip(alpha * (corr - ms), -400.0, 400.0)
        exp_v = np.exp(vals)
        s_exp = float(np.sum(exp_v))
        f = ms + (1.0 / alpha) * np.log(s_exp)
        w = exp_v / s_exp
        j = np.arange(n)
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            sv = w[r]
            if sv < 1e-15:
                continue
            st = r - (n - 1)
            gs = j - st
            m1 = (gs >= 0) & (gs < n)
            grad[j[m1]] += sv * (1.0 - hx[gs[m1]])
            ps = j + st
            m2 = (ps >= 0) & (ps < n)
            grad[j[m2]] -= sv * hx[ps[m2]]
        grad -= np.mean(grad)
        return f, grad

    bounds = [(0.0, 1.0)] * n
    try:
        from scipy.optimize import minimize as _min
        res = _min(lse_obj, h, jac=True, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 40, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _c5(x)
    except Exception:
        return h.copy(), _c5(h)
