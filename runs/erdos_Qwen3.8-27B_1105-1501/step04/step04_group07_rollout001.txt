    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, cur_c5, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    if alpha_max < 1e-15:
        return 0.0, cur_c5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
    f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    for _ in range(32):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    best_a = (lo + hi) / 2.0
    best_c5 = _exact_c5(np.clip(h + best_a * p, 0, 1))
    if f1 < best_c5:
        best_c5, best_a = f1, c1
    if f2 < best_c5:
        best_c5, best_a = f2, c2
    return best_a, best_c5

def _full_lag_step(h, rho):
    n = len(h)
    scores = _scores(h)
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lo_j = max(-rho, -h[j])
        hi_j = min(rho, 1.0 - h[j])
        bounds.append((lo_j, hi_j))
    bounds.append((-1.0, 3.0))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                      b_eq=np.zeros(1), bounds=bounds, method='highs')
        if not res.success:
            return None
        return res.x[:n]
    except Exception:
        return None

def _full_lag_optimize(h, deadline, rho_init=0.4, max_iters=8000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rho = rho_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        p = _full_lag_step(h, rho)
        if p is None:
            stall += 5
            rho *= 0.5
            if rho < 1e-9:
                break
            continue
        if np.max(np.abs(p)) < 1e-14:
            stall += 10
            rho *= 0.7
            if rho < 1e-9:
                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            new_h = _project(np.clip(h + alpha * p, 0, 1), target)
            actual_c5 = _exact_c5(new_h)
            if actual_c5 < cur_c5 - 1e-15:
                h = new_h
                cur_c5 = actual_c5
                stall = 0
                if alpha > 0.7:
                    rho = min(rho * 1.25, 1.5)
                elif alpha > 0.2:
                    rho = rho * 1.05
                else:
                    rho = rho * 0.85
            else:
                h = np.clip(h + alpha * p, 0, 1)
                cur_c5 = _exact_c5(h)
                stall = 0
                rho *= 0.75
        else:
            stall += 1
            rho *= 0.55
            if rho < 1e-9:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _lse_descent(h, deadline, beta_init=30.0, beta_max=800.0, max_iters=4000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        beta = min(beta_init * (1.0 + it / 80.0), beta_max)
        scores = _scores(h)
        max_s = float(scores.max())
        shifted = scores - max_s
        exp_s = np.exp(beta * shifted)
        w = exp_s / float(exp_s.sum())
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            stall += 5
            continue
        d = -grad / gn
        found = False
        for scale in [0.06, 0.03, 0.015, 0.007, 0.003]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-13:
                h = _project(np.clip(h + alpha * d * scale, 0, 1), target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _active_subgrad(h, deadline, max_iters=4000, thresh_frac=0.995):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        scores = _scores(h)
        s_max = float(scores.max())
        thresh = s_max * thresh_frac
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        w = np.zeros(n_lags)
        w[active] = 1.0 / len(active)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            stall += 5
            continue
        d = -grad / gn
        found = False
        for scale in [0.04, 0.02, 0.01, 0.005]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = _project(np.clip(h + alpha * d * scale, 0, 1), target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_hill_climb(h_init, deadline, max_iters=50000):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    h = (h_init >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count > target:
        idx = np.where(h == 1)[0]
        rng2 = np.random.default_rng(7)
        rng2.shuffle(idx)
        h[idx[ones_count - target:]] = 0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        rng2 = np.random.default_rng(7)
        rng2.shuffle(idx)
        h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(99)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 8000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=3):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0.copy(), n0 / 2.0)
    
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    # Phase 1: Full-lag LP from start (most important)
    dl1 = t0 + budget_s * 0.15
    hdr, c5r = _full_lag_optimize(h_start, dl1, rho_init=0.5, max_iters=6000)
    if c5r < best_c5:
        best_c5, best_h, best_n = c5r, hdr, len(hdr)
    
    # Phase 2: LSE descent
    dl2 = t0 + budget_s * 0.27
    hdr, c5r = _lse_descent(best_h, dl2, beta_init=20.0, beta_max=600.0, max_iters=3000)
    if c5r < best_c5:
        best_c5, best_h = c5r, hdr
    
    # Phase 3: More full-lag LP (tighter radius)
    dl3 = t0 + budget_s * 0.40
    hdr, c5r = _full_lag_optimize(best_h, dl3, rho_init=0.2, max_iters=8000)
    if c5r < best_c5:
        best_c5, best_h = c5r, hdr
    
    # Phase 4: Active subgradient
    dl4 = t0 + budget_s * 0.48
    hdr, c5r = _active_subgrad(best_h, dl4, max_iters=3000, thresh_frac=0.99)
    if c5r < best_c5:
        best_c5, best_h = c5r, hdr
    
    # Phase 5: Multi-start with diverse perturbations
    dl5 = t0 + budget_s * 0.65
    scale = 0.03
    for attempt in range(500):
        if time.time() > dl5:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.01)
        h_try, c5_try = _full_lag_optimize(h_pert, sub_dl, rho_init=0.15, max_iters=400)
        if c5_try < best_c5 - 1e-12:
            best_c5, best_h = c5_try, h_try
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)
    
    # Phase 6: Symmetrization
    dl6 = t0 + budget_s * 0.70
    h_sym = (best_h + best_h[::-1]) / 2.0
    h_sym = _project(h_sym, best_n / 2.0)
    h_sym, c5_sym = _full_lag_optimize(h_sym, dl6, rho_init=0.2, max_iters=2000)
    if c5_sym < best_c5:
        best_c5, best_h = c5_sym, h_sym
    
    # Phase 7: Binary exploration
    dl7 = t0 + budget_s * 0.78
    h_bin, c5_bin = _binary_hill_climb(best_h, dl7, max_iters=20000)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin
    # Polish binary result continuously
    if c5_bin < best_c5 + 0.005:
        h_pol, c5_pol = _full_lag_optimize(h_bin, dl7, rho_init=0.3, max_iters=1000)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol
    
    # Phase 8: Resolution sweep
    dl8 = t0 + budget_s * 0.86
    for dn in [-3, -2, -1, 1, 2, 3, 4, -4, 6, 8, -6]:
        if time.time() > dl8:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl8, time.time() + budget_s * 0.012)
        h_r, c5_r = _full_lag_optimize(h_r, sub_dl, rho_init=0.25, max_iters=800)
        if c5_r < best_c5 - 1e-12:
            best_c5, best_h, best_n = c5_r, h_r, n_try
    
    # Phase 9: Final full-lag LP polish
    dl9 = t0 + budget_s * 0.93
    hdr, c5r = _full_lag_optimize(best_h, dl9, rho_init=0.08, max_iters=6000)
    if c5r < best_c5:
        best_c5, best_h = c5r, hdr
    
    # Phase 10: LSE refinement
    dl10 = t0 + budget_s * 0.96
    hdr, c5r = _lse_descent(best_h, dl10, beta_init=200.0, beta_max=2000.0, max_iters=2000)
    if c5r < best_c5:
        best_c5, best_h = c5r, hdr
    
    # Phase 11: Pair polish
    if time.time() < t0 + budget_s * 0.985:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.99, max_passes=2)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp
    
    # Final projection and validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
