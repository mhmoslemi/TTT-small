```python
def _grad_matrix(h, n):
    """G[r, j] = d(corr_r)/d(h[j]) for all r, j."""
    G = np.zeros((2*n-1, n))
    j = np.arange(n)
    for r in range(2*n-1):
        s = r - (n-1)
        # (1-h[j+s]) for valid j+s
        js = j + s
        m1 = (js >= 0) & (js < n)
        tmp = np.zeros(n)
        tmp[m1] = 1.0 - h[js[m1]]
        # -h[j-s] for valid j-s
        js2 = j - s
        m2 = (js2 >= 0) & (js2 < n)
        tmp[m2] -= h[js2[m2]]
        G[r] = tmp
    return G
```

```python
            idx = np.argsort(-x)
            for i in idx:
                if x[i] <= 0.0: continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12: break
    return x

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_matrix(h, n):
    G = np.zeros((2*n-1, n))
    j = np.arange(n)
    for r in range(2*n-1):
        s = r - (n-1)
        js = j + s
        m1 = (js >= 0) & (js < n)
        tmp = np.zeros(n)
        tmp[m1] = 1.0 - h[js[m1]]
        js2 = j - s
        m2 = (js2 >= 0) & (js2 < n)
        tmp[m2] -= h[js2[m2]]
        G[r] = tmp
    return G

def _lp_descent(h, n, deadline, radius=0.15, max_iters=5000, stall_limit=80):
    """LP-based descent with adaptive radius and multi-constraint objective."""
    h = h.copy()
    target = n * 0.5
    cur_c5_raw = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    best_h, best_raw = h.copy(), cur_c5_raw
    stall = 0
    r = radius
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
            
        corr = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h, n)
        max_val = float(np.max(corr))
        
        # Weighted objective: weights based on proximity to max
        weights = np.exp(-5.0 * (max_val - corr))
        weights /= np.sum(weights)
        
        # Objective: minimize sum_r weights[r] * G[r] · p
        c = -G.T @ weights  # minimize c · p = -weights^T G p
        # Wait: we want to minimize sum w_r * (corr_r + G_r · p)
        # = sum w_r * corr_r + (G^T w) · p
        # So objective coefficient is (G^T w), and we want to minimize it
        c = G.T @ weights
        
        # Constraints: G[r] · p ≤ max_val - corr[r] + slack for all r
        # (to not violate any constraint too much)
        slack = 0.5 * max_val  # generous slack
        b_ub = max_val - corr + slack
        
        # Actually, let's be more careful. We want:
        # After step: corr_r + G_r · p ≤ t for all r
        # We want t to be as small as possible
        # Linearized: minimize t
        # subject to G_r · p - t ≤ -corr_r for all r
        #           sum(p) = 0
        #           |p|_j ≤ r for all j
        
        m = 2 * n - 1
        c_lp = np.zeros(n + 1)
        c_lp[n] = 1.0  # minimize t
        
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            stall += 1
            continue
        
        if not res.success or res.x is None:
            stall += 1
            continue
        
        p = res.x[:n]
        if np.max(np.abs(p)) < 1e-14:
            stall += 1
            r *= 0.9
            if r < 1e-6:
                break
            continue
        
        # Line search
        h_new = _project(h + p, target)
        new_raw = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full')))
        
        if new_raw < cur_c5_raw - 1e-13:
            h = h_new
            cur_c5_raw = new_raw
            if cur_c5_raw < best_raw:
                best_raw = cur_c5_raw
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.4)
        else:
            # Try smaller step
            improved = False
            for frac in [0.5, 0.25, 0.125]:
                h_try = _project(h + frac * p, target)
                raw_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))
                if raw_try < cur_c5_raw - 1e-13:
                    h = h_try
                    cur_c5_raw = raw_try
                    if cur_c5_raw < best_raw:
                        best_raw = cur_c5_raw
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                r = min(r * 1.02, 0.3)
            else:
                stall += 1
                r *= 0.85
                if r < 1e-5:
                    break
    
    return best_h, best_raw * 2.0 / n
```

```python
def _swap_search(h_bin, n, max_iters=200, deadline=None):

    target = n * 0.5

    best = cur

            break
        if no_improve > 50:

        zeros = np.where(h == 0.0)[0]

        best_val = cur

            for j in zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                val = float(np.max(np.correlate(h, 1.0 - h, mode='full')))

                h[j] = 0.0

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            cur = best_val
            no_improve = 0
        else:
            no_improve += 1
    
    return h, cur * 2.0 / n
```

Let me be more careful about the LP formulation. The LP is:
minimize t
subject to: G @ p - t <= -corr  (where p = h_new - h, so G@p ≈ corr_new - corr)

Wait, I need to be more careful. If p = h_new - h, then:
- h_new = h + p
- We need 0 <= h + p <= 1, so -h[j] <= p[j] <= 1 - h[j]
- But we also want |p[j]| <= r for the trust region
- So: max(-r, -h[j]) <= p[j] <= min(r, 1-h[j])

And the correlation constraint: at the new point, corr_new[r] <= t
Linearized: corr[r] + G[r] @ p <= t
i.e., G[r] @ p - t <= -corr[r]

This is correct. The LP finds the best linearized improvement within the trust region.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(x)
            for i in idx:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            idx = np.argsort(-x)
            for i in idx:
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x

def _raw_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _c5(h):
    n = len(h)
    return _raw_max(h) * 2.0 / n

def _grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        js = j + s
        m1 = (js >= 0) & (js < n)
        tmp = np.zeros(n)
        tmp[m1] = 1.0 - h[js[m1]]
        js2 = j - s
        m2 = (js2 >= 0) & (js2 < n)
        tmp[m2] -= h[js2[m2]]
        G[r] = tmp
    return G

def _lse_obj(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 0.0)
    ev = np.exp(vals)
    s = float(np.sum(ev))
    f = ms + (1.0 / alpha) * np.log(s)
    w = ev / s
    G = _grad_matrix(h, n)
    g = G.T @ w
    g -= np.mean(g)
    return f, g

def _lse_opt(h, n, alpha, maxiter=300):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h.copy(), args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 100, 'iprint': 0})
        x = _project(res.x, target)
        return x, _c5(x)
    except Exception:
        return h.copy(), _c5(h)

def _lp_descent(h, n, deadline, r0=0.25, max_iters=20000, stall_limit=300):
    h = h.copy()
    target = n * 0.5
    cur_raw = _raw_max(h)
    best_h, best_raw = h.copy(), cur_raw
    stall = 0
    r = r0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h, n)
        m = 2 * n - 1
        c_lp = np.zeros(n + 1)
        c_lp[n] = 1.0
        A_ub = np.empty((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j_ in range(n):
            lo_b = max(-r, -h[j_])
            hi_b = min(r, 1.0 - h[j_])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        try:
            res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            stall += 1
            continue
        if not res.success or res.x is None:
            stall += 1
            continue
        p = res.x[:n]
        if np.max(np.abs(p)) < 1e-14:
            stall += 1
            r *= 0.9
            if r < 1e-6:
                break
            continue
        h_new = _project(h + p, target)
        new_raw = _raw_max(h_new)
        if new_raw < cur_raw - 1e-14:
            h = h_new
            cur_raw = new_raw
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_h = h.copy()
            stall = 0
            r = min(r * 1.04, 0.6)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.125]:
                h_try = _project(h + frac * p, target)
                raw_try = _raw_max(h_try)
                if raw_try < cur_raw - 1e-14:
                    h = h_try
                    cur_raw = raw_try
                    if cur_raw < best_raw:
                        best_raw = cur_raw
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                r = min(r * 1.02, 0.4)
            else:
                stall += 1
                r *= 0.88
                if r < 1e-5:
                    break
    return best_h, best_raw * 2.0 / n

def _binary_round(h, n):
    h_bin = np.zeros(n)
    idx = np.argsort(-h)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def _push_binary(h, n, iters=8):
    target = n * 0.5
    for _ in range(iters):
        h = _project(4.0 * (h - 0.5) + 0.5, target)
    return h

def _swap_search(h, n, deadline, max_rounds=2000):
    h_b = _binary_round(h, n)
    cur_raw = _raw_max(h_b)
    no_imp = 0
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        if no_imp > 40:
            break
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        best_sv = cur_raw
        best_sw = None
        for i in ones:
            for j in zeros:
                h_b[i] = 0.0
                h_b[j] = 1.0
                val = _raw_max(h_b)
                h_b[i] = 1.0
                h_b[j] = 0.0
                if val < best_sv - 1e-15:
                    best_sv = val
                    best_sw = (i, j)
        if best_sw is not None:
            h_b[best_sw[0]] = 0.0
            h_b[best_sw[1]] = 1.0
            cur_raw = best_sv
            no_imp = 0
        else:
            no_imp += 1
    return h_b, cur_raw * 2.0 / n

def _mixed_swap_search(h, n, deadline, max_rounds=200, n_restarts=5):
    """Swap search with random restarts from perturbed binary solutions."""
    target = n * 0.5
    best_h, best_raw = h.copy(), _c5(h)
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart == 0:
            h_b = _binary_round(h, n)
        else:
            h_tmp = _push_binary(h, n, iters=6)
            h_b = _binary_round(h_tmp, n)
            # Random perturbation: flip a few bits
            n_flip = 2 + restart * 2
            ones = np.where(h_b == 1.0)[0]
            zeros = np.where(h_b == 0.0)[0]
            for _ in range(n_flip):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = np.random.randint(len(ones))
                j = np.random.randint(len(zeros))
                h_b[ones[i]] = 0.0
                h_b[zeros[j]] = 1.0
                ones = np.where(h_b == 1.0)[0]
                zeros = np.where(h_b == 0.0)[0]
        h_b, cw = _swap_search(h_b, n, deadline, max_rounds=max_rounds)
        if cw < best_raw:
            best_raw = cw
            best_h = h_b.copy()
    return best_h, best_raw

def _slsqp_multi(h, n, maxiter=400):
    """SLSQP with per-constraint formulation (top active set)."""
    def _obj_f(z, nn):
        return z[nn]
    def _obj_g(z, nn):
        g = np.zeros(nn + 1)
        g[nn] = 1.0
        return g
    def _eq_f(z, nn):
        return np.sum(z[:nn]) - nn * 0.5
    def _eq_g(z, nn):
        g = np.ones(nn + 1)
        g[nn] = 0.0
        return g
    K = 10
    def _active_idx(hh, nn):
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        return np.argsort(-corr)[:K]
    def _ineq_f(z, nn):
        hh, tt = z[:nn], z[nn]
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        idx = np.argsort(-corr)[:K]
        return tt - np.max(corr[idx])
    def _ineq_g(z, nn):
        hh = z[:nn]
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        r_max = int(np.argmax(corr))
        s = r_max - (nn - 1)
        j = np.arange(nn)
        pd = np.zeros(nn)
        js = j + s
        m1 = (js >= 0) & (js < nn)
        pd[m1] = 1.0 - hh[js[m1]]
        js2 = j - s
        m2 = (js2 >= 0) & (js2 < nn)
        pd[m2] -= hh[js2[m2]]
        g = np.zeros(nn + 1)
        g[:nn] = pd
        g[nn] = 1.0
        return g
    h = np.asarray(h, dtype=float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    t0 = float(np.max(corr)) + 0.005
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_obj_f, z0, jac=_obj_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _c5(x)
    except Exception:
        return h, _c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _subgrad_step(h, n, step=0.05):
    """One subgradient step on the exact max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    r_max = int(np.argmax(corr))
    s = r_max - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    js = j + s
    m1 = (js >= 0) & (js < n)
    g[m1] = 1.0 - h[js[m1]]
    js2 = j - s
    m2 = (js2 >= 0) & (js2 < n)
    g[m2] -= h[js2[m2]]
    g -= np.mean(g)
    norm = np.linalg.norm(g)
    if norm < 1e-15:
        return h.copy()
    h_new = _project(h - step * g / norm, n * 0.5)
    return h_new
```
