def _line_search(h, d, dx, t_max):
    """Exact line search along direction d from h, with bound h+td in [0,1]"""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find feasible range for t
    t_lo = 0.0
    t_hi = t_max
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx

    return opt_t, opt_val

def _active_set_lp(h, top_k, radius):
    """LP to find a direction that reduces the top-k active correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    # Get top-k indices
    top_idx = np.argsort(scores)[-top_k:][::-1]
    max_score = float(scores[top_idx[0]])
    
    # Build gradient matrix for active constraints only
    G = np.zeros((top_k, n))
    for i, r in enumerate(top_idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[i, :m] += (1.0 - h[d:d+m])
                G[i, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[i, :m] += -h[g:g+m]
                G[i, g:] += (1.0 - h[:m])
    
    # LP: minimize t s.t. G*(h+p) <= max_score - t, |p| <= radius, sum(p) = 0
    # Rearranged: G*p + t <= -epsilon (we want to reduce by at least a bit)
    # Actually: minimize t s.t. G*p <= -t, sum(p)=0, -radius <= p <= radius
    # This means: find p such that all active scores decrease by at least t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0  # G*p + t <= 0 means p reduces the score
    
    b_ub = np.zeros(top_k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0, None))  # t >= 0
    
    try:

        if res.success and res.x[-1] > 1e-12:

def _smooth_descent(h, deadline, tau=100):
    """Projected gradient descent on smooth max approximation"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_obj
    lr = 0.01
    mom = np.zeros(n)
    beta = 0.9
    stall = 0
    
    while time.time() < deadline:
        if stall > 50:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * dx
        max_s = float(np.max(scores))
        exp_s = np.exp(tau * (scores - max_s))
        w = exp_s / float(np.sum(exp_s))

        grad = np.zeros(n)
        for r in np.where(w > 1e-12)[0]:

                m = n - g

                    grad[g:] += w[r] * (1.0 - h[:m])

        # We want to minimize it, so descend

def _grad_rows(h, indices):
    """Compute gradient rows of corr_r for given correlation indices."""
    n = len(h)

def _smooth_grad_dir(h, tau):

    best_c5 = _exact_c5(h)

    # Phase 1: Active-set LP with varying k (35%)
    d1 = t0 + budget_s * 0.35
    k_schedule = [5, 3, 7, 4, 6, 8, 3, 5]
    for ki in k_schedule:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.04)
        h1, c5_1 = _lp_loop_active(h, sub_dl, top_k=ki, radius_init=0.15)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            h = h1.copy()

    # Phase 2: Smooth gradient (12%)
    d2 = t0 + budget_s * 0.47
    for tau in [100, 250, 500]:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h2, c5_2 = _smooth_descent(h, sub_dl, tau=tau)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            h = h2.copy()

    # Phase 3: More LP (18%)
    d3 = t0 + budget_s * 0.65
    for ki in [3, 5, 4, 6, 3]:
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + budget_s * 0.035)
        h3, c5_3 = _lp_loop_active(h, sub_dl, top_k=ki, radius_init=0.12)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            h = h3.copy()

    # Phase 4: Multi-resolution (18%)
    d4 = t0 + budget_s * 0.83
    best_n = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, -3, 3, -8, 8, -5, 5, -10, 10, -12, 12, -7, 7, -14, 14]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        nt = best_n + dn
        if nt < 24 or nt > 180 or nt in tried:
            continue
        tried.add(nt)
        h_r = _resample(best_h, nt)
        sub_dl = min(d4, time.time() + budget_s * 0.011)
        h_r, c5_r = _lp_loop_active(h_r, sub_dl, top_k=5, radius_init=0.1)
        rem = d4 - time.time()
        if rem > 4 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d4, time.time() + max(2, budget_s * 0.006))
            h_r, c5_r = _lp_loop_active(h_r, sub_dl2, top_k=3, radius_init=0.06)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt

    # Phase 5: Structured perturbation (9%)
    d5 = t0 + budget_s * 0.92
    h5, c5_5 = _structured_perturb(best_h, 80, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair refinement (6%)
    d6 = t0 + budget_s * 0.98
    h_p, c5_p = _pair_refine(best_h, d6)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Final: tiny LP polish with remaining time
    if time.time() < t0 + budget_s * 0.995:
        h_f, c5_f = _lp_loop_active(best_h, t0 + budget_s * 0.995, top_k=3, radius_init=0.03)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
