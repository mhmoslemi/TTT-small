```python
        i_start = max(0, -d)
        i_end = min(n - 1, n - 1 - d)

            g[i + d] -= dx * h[i]
```

```python
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    d = k_idx - (n - 1)

    for i in range(i_start, i_end + 1):
        g[i] += dx * (1.0 - h[i + d])
        g[i + d] -= dx * h[i]
    return g

    """Average gradient over active lags (those within epsilon of max)."""
    corr = corr_all(h)
    max_c = float(np.max(corr))
    active = np.where(corr >= max_c - epsilon)[0]
    g = np.zeros(len(h))
    for k in active:
        g += gradient_at_lag(h, k)
    g /= len(active)

    best_h = h.copy()
    best_c5 = cur_c5

    epsilon = epsilon_init

    while time.time() < deadline and stall < 200:
        t += 1
        g, max_c, n_active = active_gradient(h, epsilon)
        
        # Backtracking line search
        step = alpha / np.sqrt(t + 1)
        improved = False
        for trial in range(20):

            c5_new = exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:

                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break
            step *= 0.7
        
        if improved:
            stall = 0
            # Decrease epsilon to focus more
            epsilon = max(0.005, epsilon * 0.97)
        else:
            stall += 1
            alpha *= 0.8
            epsilon = min(0.1, epsilon * 1.1)
            if alpha < 1e-8:
                break
    
    return best_h, best_c5
```

```python
def pair_optimize_fast(h, j, k):
    n = len(h)
    dx = 2.0 / n
    corr = corr_all(h)  # shape (2n-1,)
    
    # Compute d_l for all lags
    d_upgrade_j = np.zeros(2*n-1)  # effect of increasing h[j] by 1
    d_downgrade_k = np.zeros(2*n-1)  # effect of increasing h[k] by 1
    
    for l in range(2*n-1):
        d = l - (n-1)
        # h[j] contribution to c_l:
        # if j+d in [0,n-1]: +(1-h[j+d])*dx (from h[j] term)
        # if j-d in [0,n-1]: -h[j-d]*dx (from (1-h[j]) term)
        coeff = 0.0
        if 0 <= j+d < n:
            coeff += (1.0 - h[j+d]) * dx
        if 0 <= j-d < n:
            coeff -= h[j-d] * dx
        d_upgrade_j[l] = coeff
        
        coeff_k = 0.0
        if 0 <= k+d < n:
            coeff_k += (1.0 - h[k+d]) * dx
        if 0 <= k-d < n:
            coeff_k -= h[k-d] * dx
        d_downgrade_k[l] = coeff_k
    
    # Net effect of transferring 1 unit from k to j:
    d_net = d_upgrade_j - d_downgrade_k  # c_l(a) = corr[l] + d_net[l] * a
    
    a_lo = max(-h[j], h[k] - 1.0)
    a_hi = min(1.0 - h[j], h[k])
    
    if a_lo >= a_hi:
        return 0.0, float(np.max(corr))
    
    # Find min of max_l (corr[l] + d_net[l] * a) for a in [a_lo, a_hi]
    # This is a convex piecewise-linear function (max of linear functions)
    # The minimum is at a breakpoint or boundary
    
    # Collect all breakpoints
    breakpoints = set()
    breakpoints.add(a_lo)
    breakpoints.add(a_hi)
    
    # For efficiency, only consider top 20 lags for breakpoints
    top_idx = np.argsort(corr)[-20:]
    for i in range(len(top_idx)):
        for j2 in range(i+1, len(top_idx)):
            li, lj = top_idx[i], top_idx[j2]
            denom = d_net[li] - d_net[lj]
            if abs(denom) > 1e-15:
                bp = (corr[lj] - corr[li]) / denom
                if a_lo <= bp <= a_hi:
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        mask = (x < 1.0 - 1e-10) if residual > 0 else (x > 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        idx = int(np.argmax(1.0 - x)) if residual2 > 0 else int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_dnet(h, j, k):
    """Compute d_net vector: effect of transferring 1 unit from k to j on all correlations."""
    n = len(h)
    dx = 2.0 / n
    d_j = np.zeros(2 * n - 1)
    d_k = np.zeros(2 * n - 1)
    for l in range(2 * n - 1):
        d = l - (n - 1)
        cj = 0.0
        if 0 <= j + d < n:
            cj += (1.0 - h[j + d]) * dx
        if 0 <= j - d < n:
            cj -= h[j - d] * dx
        d_j[l] = cj
        ck = 0.0
        if 0 <= k + d < n:
            ck += (1.0 - h[k + d]) * dx
        if 0 <= k - d < n:
            ck -= h[k - d] * dx
        d_k[l] = ck
    return d_j - d_k

def pair_exact_solve(h, j, k, corr):
    """Find optimal transfer from k to j using exact 1D piecewise-linear minimization."""
    n = len(h)
    d_net = compute_dnet(h, j, k)
    a_lo = max(-h[j], h[k] - 1.0)
    a_hi = min(1.0 - h[j], h[k])
    if a_lo >= a_hi - 1e-14:
        return 0.0, float(np.max(corr))
    top_m = min(15, 2 * n - 1)
    top_idx = np.argsort(corr)[-top_m:]
    cands = [a_lo, a_hi]
    for i in range(len(top_idx)):
        for jj in range(i + 1, len(top_idx)):
            li, lj = top_idx[i], top_idx[jj]
            denom = d_net[li] - d_net[lj]
            if abs(denom) > 1e-14:
                bp = (corr[lj] - corr[li]) / denom
                if a_lo - 1e-12 <= bp <= a_hi + 1e-12:
                    cands.append(max(a_lo, min(a_hi, bp)))
    cands = [max(a_lo, min(a_hi, c)) for c in cands]
    best_a = 0.0
    best_val = float(np.max(corr))
    for a in cands:
        val = float(np.max(corr + d_net * a))
        if val < best_val:
            best_val = val
            best_a = a
    return best_a, best_val

def pair_sweep(h, deadline, max_rounds=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = exact_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = corr_all(h)
        improved = False
        rng_local = np.random.default_rng(rnd + 1000)
        num_pairs = min(200, n * (n - 1) // 2)
        for _ in range(num_pairs):
            if time.time() > deadline:
                break
            j = int(rng_local.integers(0, n))
            k = int(rng_local.integers(0, n))
            if j == k:
                continue
            a, val = pair_exact_solve(h, j, k, corr)
            if abs(a) > 1e-12 and val < best_c5 - 1e-13:
                h[j] += a
                h[k] -= a
                h = np.clip(h, 0.0, 1.0)
                h = project(h, target)
                best_c5 = exact_c5(h)
                corr = corr_all(h)
                improved = True
        if not improved:
            break
    return h, best_c5

def lp_refine(h, deadline, radius_init=0.03, top_k=8):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline and stall < 60:
        corr = corr_all(h)
        max_c = float(np.max(corr))
        threshold = max_c - radius * 0.5
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            top_idx = np.argsort(corr)[-top_k:]
            active = top_idx
        elif len(active) > top_k * 3:
            active = np.argsort(corr)[-top_k * 3:]
        m = len(active)
        A = np.zeros((m, n))
        c_vals = np.zeros(m)
        for idx, kk in enumerate(active):
            d = kk - (n - 1)
            i_s = max(0, -d)
            i_e = min(n - 1, n - 1 - d)
            for i in range(i_s, i_e + 1):
                A[idx, i] += (1.0 - h[i + d])
                A[idx, i + d] -= h[i]
            c_vals[idx] = corr[kk]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = c_vals - A @ h
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(max(0.0, h[j] - radius), min(1.0, h[j] + radius)) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                h_new = project(h_new, target)
                c5_new = exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    radius = min(radius * 1.05, 0.3)

                    stall += 1
                    radius *= 0.75

                radius *= 0.75

                    break
        except Exception:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
    return best_h, best_c5

    if n_new < 10:
```
