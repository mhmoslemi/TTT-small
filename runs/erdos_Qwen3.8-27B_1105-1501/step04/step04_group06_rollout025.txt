def _project(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(-x if diff > 0 else x)
        for i in idx:
            if diff > 0:
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
            else:

                x[i] -= a
                diff += a
            if abs(diff) < 1e-10:

    return x

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _lse_loss(h, alpha):
    """Log-sum-exp approximation to max(corr(h)), with gradient."""
    corr = _corr(h)
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    loss = max_c + (1.0/alpha) * np.log(sum_exp)
    weights = exp_vals / sum_exp
    # Gradient
    n = len(h)
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2*n - 1):
        s = r - (n - 1)
        # d(corr[r])/d(h[j]) = (1-h[j-s]) * I(j-s valid) - h[j+s] * I(j+s valid)
        if weights[r] < 1e-12:

        grad[m2] -= w * h[ps[m2]]
    return loss, grad

    h = _project(h_init, n * 0.5)

We want to minimize max_r corr_new[r] over δ (subject to box constraints).

This is a 1D minimax problem, which can be solved by binary search on t:

import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in np.argsort(x):
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-14:
                    break
        else:
            for i in np.argsort(-x):
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-14:
                    break
    return x


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    return float(np.max(_corr(h)) * 2.0 / len(h))


def _lse_fg(x, n, alpha, mp):
    corr = _corr(x)
    mc = float(np.max(corr))
    e = np.exp(alpha * (corr - mc))
    se = float(np.sum(e))
    w = e / se
    loss = mc + np.log(se) / alpha
    n_loc = len(x)
    j = np.arange(n_loc)
    grad = np.zeros(n_loc)
    for r in range(2 * n_loc - 1):
        if w[r] < 1e-14:
            continue
        s = r - (n_loc - 1)
        gs = j - s
        m1 = (gs >= 0) & (gs < n_loc)
        grad[m1] += w[r] * (1.0 - x[gs[m1]])
        ps = j + s
        m2 = (ps >= 0) & (ps < n_loc)
        grad[m2] -= w[r] * x[ps[m2]]
    md = np.sum(x) - n_loc * 0.5
    loss += mp * md * md
    grad = grad + 2.0 * mp * md
    return loss, grad


def _lse_optimize(h, n, alphas, maxiter=200, mp=50.0):
    h_cur = _project(h, n * 0.5)
    best_h = h_cur.copy()
    best_c5 = _c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def fg(x):
            return _lse_fg(x, n, alpha, mp)
        try:
            res = minimize(fg, h_cur, jac=True, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-16,
                                    'gtol': 1e-10, 'maxls': 50, 'iprint': 0})
            h_cur = _project(res.x, n * 0.5)
        except Exception:
            h_cur = _project(h_cur, n * 0.5)
        c5 = _c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    return best_h, best_c5


def _s_obj(z, n):
    return z[n]


def _s_obj_j(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _s_eq(z, n):
    return np.sum(z[:n]) - n * 0.5


def _s_eq_j(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _s_ineq(z, n, active):

    return z[n] - cc[active]


def _s_ineq_j(z, n, active):
    hh = z[:n]
    n_loc = len(hh)
    j = np.arange(n_loc)
    m = len(active)
    J = np.zeros((m, n_loc + 1))
    for ii in range(m):
        r = active[ii]
        s = r - (n_loc - 1)
        d = np.zeros(n_loc)
        gs = j - s
        m1 = (gs >= 0) & (gs < n_loc)
        d[m1] = 1.0 - hh[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n_loc)
        d[m2] -= hh[ps[m2]]
        J[ii, :n_loc] = -d
        J[ii, n_loc] = 1.0
    return J


def _slsqp_active(h, n, K=6, maxiter=400, max_rounds=4):
    h = _project(h, n * 0.5)
    best_h = h.copy()
    best_c5 = _c5(h)
    corr = _corr(h)
    active = np.argsort(-corr)[:K]

    for rnd in range(max_rounds):
        act = active.copy()
        t0 = float(np.max(_corr(h))) + 1e-3
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
        cons = [
            {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_j, 'args': (n,)},
            {'type': 'ineq', 'fun': _s_ineq, 'jac': _s_ineq_j, 'args': (n, act)}
        ]
        try:
            res = minimize(_s_obj, z0, jac=_s_obj_j, args=(n,),
                           bounds=bounds, constraints=cons, method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            h = _project(res.x[:n], n * 0.5)
        except Exception:
            pass
        c5 = _c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        corr = _corr(h)
        t_max = float(np.max(corr))
        viol = np.where(corr > t_max - 1e-4)[0]
        new_act = np.sort(np.unique(np.concatenate([active, viol])))
        if len(new_act) <= len(active) or len(new_act) > 2 * K or len(new_act) > 20:
            break
        active = new_act
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new * 0.5)


def _binary_start(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def _periodic_start(n, period, rng):
    h = np.zeros(n)
    pattern_len = period
    n_ones = max(1, period // 2)
    ones = rng.choice(pattern_len, size=n_ones, replace=False)
    pat = np.zeros(pattern_len)
    pat[ones] = 1.0
    for i in range(n):
        h[i] = pat[i % period]
    return _project(h, n * 0.5)


def _cosine_start(n, freq, amp, rng):
    x = np.linspace(0, 1, n, endpoint=False)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + amp * np.cos(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return _project(h, n * 0.5)


def _nudge(h, n, max_sweeps=15):
    """Coordinate nudging: adjust one element at a time with mass correction."""
    h = _project(h, n * 0.5)
    best_c5 = _c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Try nudging h[i] up or down, with uniform compensation
            for direction in [1.0, -1.0]:
                # Max step size
                if direction > 0:
                    dmax = (1.0 - h[i]) / 1.0
                else:
                    dmax = h[i] / 1.0
                dmax = min(dmax, 0.1)
                if dmax < 1e-8:
                    continue
                # Golden section on delta in [0, dmax]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                lo = 0.0
                hi = dmax
                d1 = hi - gr * (hi - lo)
                d2 = lo + gr * (hi - lo)

                def _eval_d(d):
                    h_t = h.copy()
                    h_t[i] += direction * d
                    rem = n - 1
                    if rem <= 0:
                        return 1.0
                    adj = -direction * d / rem
                    h_t[0:i] += adj
                    h_t[i+1:] += adj
                    h_t = _project(h_t, n * 0.5)
                    return _c5(h_t)

                f1 = _eval_d(d1)
                f2 = _eval_d(d2)
                for _ in range(20):
                    if hi - lo < 1e-10:
                        break
                    if f1 < f2:
                        hi, d2, f2 = d2, d1, f1
                        d1 = hi - gr * (hi - lo)
                        f1 = _eval_d(d1)
                    else:
                        lo, d1, f1 = d1, d2, f2
                        d2 = lo + gr * (hi - lo)
                        f2 = _eval_d(d2)
                d_best = (lo + hi) / 2.0
                h_t = h.copy()
                h_t[i] += direction * d_best
                rem = n - 1
                adj = -direction * d_best / rem
                h_t[0:i] += adj
                h_t[i+1:] += adj
                h_t = _project(h_t, n * 0.5)
                c5_t = _c5(h_t)
                if c5_t < best_c5 - 1e-10:
                    h = h_t
                    best_c5 = c5_t
                    improved = True
        if not improved:
            break
    return h, best_c5
