    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = np.clip(0.5 + B @ pop_a[best_i], 0, 1)
    best_h = _project(best_h, target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        n_keep = max(2, n_pop // 2)
        pop_a[:n_keep] = pop_a[order[:n_keep]]
        pop_c5[:n_keep] = pop_c5[order[:n_keep]]
        for i in range(n_keep, n_pop):
            p1 = order[rng.integers(0, n_keep)]
            p2 = order[rng.integers(0, n_keep)]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1.0 - mix) * pop_a[order[p2]]
            pop_a[i] += mutation_scale * rng.standard_normal(2*K)
            h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
            h_try = _project(h_try, target)
            pop_c5[i] = _exact_c5(h_try)
        mutation_scale *= 0.996
        cur_best = int(np.argmin(pop_c5))
        if pop_c5[cur_best] < best_c5:
            best_c5 = pop_c5[cur_best]
            best_h = np.clip(0.5 + B @ pop_a[cur_best], 0, 1)
            best_h = _project(best_h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Strong LP optimization from initial (all constraints)
    if time.time() < t0 + budget_s * 0.15:
        dl1 = t0 + budget_s * 0.15
        h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.18, max_iters=5000, radius_min=0.001)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Smoothed gradient with momentum
    if time.time() < t0 + budget_s * 0.25:
        dl2 = t0 + budget_s * 0.25
        h2, c5_2 = _smoothed_grad_momentum(best_h, dl2, beta_start=15, beta_end=400, n_steps=400, step_init=0.04)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        # Follow up with LP
        if time.time() < t0 + budget_s * 0.30:
            dl2b = t0 + budget_s * 0.30
            h2b, c5_2b = _lp_optimize(best_h, dl2b, radius_init=0.10, max_iters=3000, radius_min=0.001)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b.copy()

    # Phase 3: Multi-resolution exploration
    if time.time() < t0 + budget_s * 0.45:
        dl3 = t0 + budget_s * 0.45
        resolutions = [64, 80, 96, 112, 128, 96, 80, 64]
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.6180339887, 0.3819660113, 0.7071067812]
        for ri, n_try in enumerate(resolutions):
            if time.time() > dl3:
                break
            if best_c5 < 0.3806:
                break
            for s in range(2):
                if time.time() > dl3:
                    break
                if s == 0:
                    h_init = _resample(best_h, n_try)
                else:
                    alpha = alphas[ri % len(alphas)]
                    h_init = _quasi_periodic_init(n_try, alpha, rng)
                sub_dl = min(dl3, time.time() + budget_s * 0.03)
                h_try, c5_try = _lp_optimize(h_init, sub_dl, radius_init=0.15, max_iters=3000, radius_min=0.003)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try

    # Phase 4: Another round of LP + momentum from best
    if time.time() < t0 + budget_s * 0.55:
        dl4 = t0 + budget_s * 0.55
        h4, c5_4 = _smoothed_grad_momentum(best_h, dl4, beta_start=30, beta_end=500, n_steps=300, step_init=0.03)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        if time.time() < t0 + budget_s * 0.62:
            dl4b = t0 + budget_s * 0.62
            h4b, c5_4b = _lp_optimize(best_h, dl4b, radius_init=0.08, max_iters=4000, radius_min=0.001)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: Fourier evolution + binary SA for diversity
    if time.time() < t0 + budget_s * 0.70:
        dl5 = t0 + budget_s * 0.70
        h_f, c5_f = _fourier_evolve(best_h, dl5, n_pop=20, K=10, n_gen=200, mutation_scale=0.08, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if time.time() < t0 + budget_s * 0.75:
            sub_dl = min(t0 + budget_s * 0.75, time.time() + budget_s * 0.02)
            h_fp, c5_fp = _lp_optimize(best_h, sub_dl, radius_init=0.06, max_iters=1500, radius_min=0.002)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()
        # Binary SA
        if time.time() < t0 + budget_s * 0.80:
            for n_bin in [64, 96]:
                if time.time() > t0 + budget_s * 0.80:
                    break
                sub_dl = min(t0 + budget_s * 0.80, time.time() + budget_s * 0.025)
                h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=20000)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    best_n = n_bin
                    sub_dl2 = min(t0 + budget_s * 0.80, time.time() + budget_s * 0.015)
                    h_br, c5_br = _lp_optimize(h_b, sub_dl2, radius_init=0.10, max_iters=1000, radius_min=0.005)
                    if c5_br < best_c5:
                        best_c5 = c5_br
                        best_h = h_br.copy()

    # Phase 6: Perturbation restarts with LP
    dl6 = t0 + budget_s * 0.90
    scale = 0.025
    for attempt in range(300):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, radius_init=0.06, max_iters=500, radius_min=0.005)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.08, scale * 1.10)

    # Phase 7: Coordinate polish
    if time.time() < t0 + budget_s * 0.95:
        dl7 = t0 + budget_s * 0.95
        hc, c5c = _coordinate_polish(best_h, dl7, n_cycles=2)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.97:
        dl8 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, dl8, n_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 9: Final fine LP
    if time.time() < t0 + budget_s * 0.995:
        dl9 = t0 + budget_s * 0.998
        h_fin, c5_fin = _lp_optimize(best_h, dl9, radius_init=0.01, max_iters=2000, radius_min=0.0005)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
