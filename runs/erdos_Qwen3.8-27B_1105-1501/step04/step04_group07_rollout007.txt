        return np.zeros(n)
    if target >= n:

    lo = float(v.min()) - 2.0

The trust-region LP at point h with current scores s:
- Variables: δ[0..n-1], τ
- Minimize: τ
- Subject to:
  - For each lag r: s[r] + G[r]·δ ≤ τ
  - sum(δ) = 0
  - -ρ ≤ δ[i] ≤ ρ for all i
  - -(h[i]) ≤ δ[i] ≤ (1-h[i]) for all i (to keep h+δ in [0,1])

Combined: max(-ρ, -h[i]) ≤ δ[i] ≤ min(ρ, 1-h[i])

After solving, we get a direction δ. Then we do a line search:
g(α) = C₅(h + α*δ)
Find α* that minimizes g(α) for α ∈ [0, α_max] where α_max ensures feasibility.

For the correlation c_d = sum h[n+d](1-h[n]), the second-order part is:

c_d(h+δ) = c_d(h) + G[d]·δ - (autocorr of δ at lag d)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        # Part 1: derivative from h[n+d] term
        j1 = max(0, d)
        j1e = min(n-1, n-1+d) + 1
        if j1e > j1:
            G[r, j1:j1e] += (1.0 - h[j1-d:j1e-d])
        # Part 2: derivative from (1-h[n]) term  
        j2 = max(0, -d)
        j2e = min(n-1, n-1-d) + 1
        if j2e > j2:
            G[r, j2:j2e] += -h[j2+d:j2e+d]
    return G * dx

def _line_search(h, p, c5_cur, max_alpha=None):
    n = len(h)
    if max_alpha is None:
        max_alpha = 1.0
    # Find feasible max alpha
    for j in range(n):
        if p[j] > 1e-14:
            max_alpha = min(max_alpha, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            max_alpha = min(max_alpha, h[j] / (-p[j]))
    max_alpha = max(max_alpha, 1e-12)
    # Coarse search
    n_s = 16
    alphas = np.linspace(0, max_alpha, n_s)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    # Refine with golden section if not at boundary
    if best_i > 0 and best_i < n_s - 1:
        lo = float(alphas[max(0, best_i - 1)])
        hi = float(alphas[min(n_s - 1, best_i + 1)])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(20):
            if hi - lo < 1e-10:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        oa = (lo + hi) / 2.0
        oc = _exact_c5(h + oa * p)
        if oc < best_c5:
            best_a = oa
            best_c5 = oc
    return best_a, best_c5

def _lp_step(h, radius, max_iters=2000, deadline=None):
    """Full-lag trust-region LP optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    r = radius
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        L = 2 * n - 1
        # Build LP
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize tau
        A_ub = np.zeros((L, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-r, -h[j])
            hi = min(r, 1.0 - h[j])
            if lo <= hi:
                bounds.append((lo, hi))
            else:
                bounds.append((0.0, 0.0))
        bounds.append((-2.0, None))  # tau lower bound
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-6:
                break
            continue
        # Line search
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            r = min(r * 1.05, 0.5)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            r *= 0.75
            if stall > 20 and r < 0.01:
                r = 0.05
    return best_h, best_c5

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 60:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.1, 0.06, 0.03, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 100:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _chirp_init(n, freq=1.0, seed=0):
    """Create a chirp-like initial condition."""
    i = np.arange(n)
    v = np.sin(2 * np.pi * freq * i * i / (2 * n))
    h = 0.5 + 0.4 * v
    return _project(h, n / 2.0)

def _random_init(n, seed=42):
    rng = np.random.default_rng(seed)
    h = rng.random(n)
    return _project(h, n / 2.0)
