    v = np.asarray(v, dtype=float)
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0 - 1e-14:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 1e-14:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _coord_optimize(h, j, n):
    old_val = h[j]
    r0 = n - 1
    h[j] = 0.0
    c0_arr = _corr_vals(h)
    h[j] = 1.0
    c1_arr = _corr_vals(h)
    h[j] = old_val
    lin_coeff = c1_arr - c0_arr
    quad_coeff = -1.0
    # corr_r(x) = c0_arr[r] + lin_coeff[r]*x + quad_coeff*x^2 (quad only for r=r0)
    # Verify: at r0, corr = c0_arr[r0] + x - x^2
    # Check: c1_arr[r0] = c0_arr[r0] + 1 - 1 = c0_arr[r0]. So lin_coeff[r0] should be 1.
    # But actually corr_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
    # With h[j]=x: (n/2 - old + x) - (S + x^2) where S = sum_{i!=j} h[i]^2
    # = const + x - x^2. At x=0: const. At x=1: const + 1 - 1 = const. So c1_arr[r0]=c0_arr[r0].
    # lin_coeff[r0] = 0, but actual linear term is 1. So we need to handle r0 specially.
    # For r0: corr(x) = c0_arr[r0] + x - x^2
    # For r != r0: corr(x) = c0_arr[r] + lin_coeff[r] * x

    def eval_F(x):
        c = c0_arr + lin_coeff * x
        c[r0] = c0_arr[r0] + x - x * x
        return float(np.max(c))

    # Find critical points
    crit = [0.0, 1.0, 0.5]
    # Intersections of quad with each linear
    for r in range(2 * n - 1):
        if r == r0:
            continue
        a = 1.0
        b = lin_coeff[r] - 1.0
        c = c0_arr[r] - c0_arr[r0]
        disc = b * b - 4.0 * a * c
        if disc >= 0:
            sq = np.sqrt(disc)
            x1 = (-b + sq) / (2.0 * a)
            x2 = (-b - sq) / (2.0 * a)
            if 0.0 <= x1 <= 1.0:
                crit.append(x1)
            if 0.0 <= x2 <= 1.0:
                crit.append(x2)

    # Intersections of pairs of linears (only nearby pairs to save time)
    # Actually for n=64, 126 linears, pairs = ~8000. Each eval is O(1). Fine.
    # But we only need to check if the intersection is in [0,1] and is a local min candidate
    # So let's just add all line-line intersections in [0,1]
    n_lins = 2 * n - 1
    lins = [r for r in range(n_lins) if r != r0]
    for ii in range(len(lins)):
        r1 = lins[ii]
        for ii2 in range(ii + 1, len(lins)):
            r2 = lins[ii2]
            dc = lin_coeff[r1] - lin_coeff[r2]
            if abs(dc) < 1e-14:
                continue
            x_int = (c0_arr[r2] - c0_arr[r1]) / dc
            if 0.0 <= x_int <= 1.0:
                crit.append(x_int)

    # Also check the midpoint between adjacent grid points for safety
    grid = np.linspace(0.0, 1.0, 11)
    crit.extend(grid.tolist())

    best_x = old_val
    best_f = eval_F(old_val)
    for xc in crit:
        xc = max(0.0, min(1.0, xc))
        f = eval_F(xc)
        if f < best_f:
            best_f = f
            best_x = xc

    # Golden section refinement around best
    if 0.0 < best_x < 1.0:
        lo = max(0.0, best_x - 0.05)
        hi = min(1.0, best_x + 0.05)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1, f2 = eval_F(c1), eval_F(c2)
        for _ in range(20):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_F(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_F(c2)
        x_new = 0.5 * (lo + hi)
        f_new = eval_F(x_new)
        if f_new < best_f:
            best_f = f_new
            best_x = x_new

    h[j] = best_x
    return best_f


def _coord_descent(h, n, max_sweeps=200, tol=1e-11, deadline=None):
    h = h.copy()
    target = n * 0.5
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    rng = np.random.default_rng(12345)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improvement = 0.0
        order = np.arange(n)
        for j in order:
            if deadline is not None and time.time() > deadline:
                return best_h, best_f
            old_f = _exact_c5(h)
            _coord_optimize(h, j, n)
            h = _project(h, target)
            new_f = _exact_c5(h)
            imp = old_f - new_f
            if imp > 0:
                improvement += imp
        if improvement < tol:
            break
        if best_f - cur_f > 10 * tol:
            pass
    h = _project(h, target)
    final_f = _exact_c5(h)
    if final_f < best_f:
        best_h, best_f = h.copy(), final_f
    return best_h, best_f


def _paired_coord(h, n, max_sweeps=50, deadline=None):
    """Pair coordinate descent: optimize pairs (j,k) together."""
    h = h.copy()
    target = n * 0.5
    cur_f = _exact_c5(h)
    best_h, best_f = h.copy(), cur_f
    rng = np.random.default_rng(999)
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        imp_total = 0.0
        for _ in range(n // 2):
            if deadline is not None and time.time() > deadline:
                return best_h, best_f
            j, k = rng.integers(0, n, 2)
            if j == k:
                k = (j + 1) % n
            old_f = _exact_c5(h)
            # Optimize (h[j], h[k]) via small grid
            vj, vk = h[j], h[k]
            best_x, best_y, best_f2 = vj, vk, old_f
            for dj in [-0.1, -0.05, 0.0, 0.05, 0.1]:
                for dk in [-0.1, -0.05, 0.0, 0.05, 0.1]:
                    x_try = vj + dj
                    y_try = vk + dk
                    if x_try < 0 or x_try > 1 or y_try < 0 or y_try > 1:
                        continue
                    h[j] = x_try
                    h[k] = y_try
                    h = _project(h, target)
                    f_try = _exact_c5(h)
                    if f_try < best_f2:
                        best_f2 = f_try
                        best_x, best_y = h[j], h[k]
            imp_total += old_f - best_f2
        cur_f = _exact_c5(h)
        if cur_f < best_f:
            best_f, best_h = cur_f, h.copy()
        if imp_total < 1e-11:
            break
    return best_h, best_f


def _subgrad_momentum(h, n, deadline, max_iters=5000):
    """Subgradient method with momentum and adaptive step."""
    h = h.copy()
    target = n * 0.5
    cur_f = _exact_c5(h)
    best_h, best_f = h.copy(), cur_f
    v = np.zeros(n)
    step = 0.1
    momentum = 0.9
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = _corr_vals(h)
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-4)[0]
        grad = np.zeros(n)
        j = np.arange(n)
        for r in active:
            s = r - (n - 1)
            g = np.zeros(n)
            gi = j - s
            m1 = (gi >= 0) & (gi < n)
            g[m1] += (1.0 - h[gi[m1]])
            pi = j + s
            m2 = (pi >= 0) & (pi < n)
            g[m2] -= h[pi[m2]]
            grad += g
        grad /= len(active)
        # Center gradient (projection onto sum=const hyperplane)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        v = momentum * v - step * grad
        h_new = _project(h + v, target)
        f_new = _exact_c5(h_new)
        if f_new < cur_f - 1e-13:
            h = h_new
            cur_f = f_new
            if cur_f < best_f:
                best_f, best_h = cur_f, h.copy()
            stall = 0
            step = min(step * 1.01, 0.3)
        else:
            stall += 1
            step *= 0.85
            if stall > 100:
                step = 0.1
                v *= 0.5
                stall = 0
    return best_h, best_f


def _slsqp_solve(h_init, n, maxiter=300, ftol=1e-14):
    """SLSQP epigraph formulation."""
    n_pts = n
    r0 = n - 1
    n_corr = 2 * n - 1

    def _epi_obj(z):
        return z[n_pts]

    def _epi_jac(z):
        g = np.zeros(n_pts + 1)
        g[n_pts] = 1.0
        return g

    def _eq_fun(z):
        return np.sum(z[:n_pts]) - n_pts * 0.5

    def _eq_jac(z):
        g = np.ones(n_pts + 1)
        g[n_pts] = 0.0
        return g

    def _ineq_fun(z):
        hh = z[:n_pts]
        cc = np.correlate(hh, 1.0 - hh, mode='full')
        return z[n_pts] - cc

    def _ineq_jac(z):
        hh = z[:n_pts]
        J = np.zeros((n_corr, n_pts + 1))
        j = np.arange(n_pts)
        for r in range(n_corr):
            s = r - (n_pts - 1)
            pd = np.zeros(n_pts)
            gi = j - s
            m1 = (gi >= 0) & (gi < n_pts)
            pd[m1] = 1.0 - hh[gi[m1]]
            pi = j + s
            m2 = (pi >= 0) & (pi < n_pts)
            pd[m2] -= hh[pi[m2]]
            J[r, :n_pts] = -pd
            J[r, n_pts] = 1.0
        return J

    corr = np.correlate(h_init, 1.0 - h_init, mode='full')
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h_init, [t0]])
    bounds = [(0.0, 1.0)] * n_pts + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}
    ]
    try:
        res = minimize(_epi_obj, z0, jac=_epi_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n_pts], n_pts * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h_init.copy(), _exact_c5(h_init)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _lse_step(h, n, alpha, maxiter=100):
    def _lse_obj(x):
        hh = np.asarray(x, dtype=float)
        cc = np.correlate(hh, 1.0 - hh, mode='full')
        ms = float(np.max(cc))
        vals = np.clip(alpha * (cc - ms), -500.0, 500.0)
        ev = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(ev)))
        w = ev / float(np.sum(ev))
        j = np.arange(n)
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            s = r - (n - 1)
            pd = np.zeros(n)
            gi = j - s
            m1 = (gi >= 0) & (gi < n)
            pd[m1] = 1.0 - hh[gi[m1]]
            pi = j + s
            m2 = (pi >= 0) & (pi < n)
            pd[m2] -= hh[pi[m2]]
            grad += w[r] * pd
        grad -= np.mean(grad)
        return f, grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h, jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    def _elapsed():
        return time.time() - t_start

    # Phase 1: Subgradient with momentum (15% budget)
    d1 = t_start + budget_s * 0.15
    h, c5 = _subgrad_momentum(h, n0, d1, max_iters=3000)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Coordinate descent intensive (25% budget)
    d2 = t_start + budget_s * 0.40
    h_cd, c5_cd = _coord_descent(best_h, n0, max_sweeps=500, tol=1e-12, deadline=d2)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 3: Paired coordinate descent (10% budget)
    d3 = t_start + budget_s * 0.50
    h_pc, c5_pc = _paired_coord(best_h, n0, max_sweeps=100, deadline=d3)
    if c5_pc < best_c5:
        best_c5, best_h = c5_pc, h_pc.copy()
    # Coordinate descent again after paired
    d3b = t_start + budget_s * 0.55
    h_cd2, c5_cd2 = _coord_descent(best_h, n0, max_sweeps=200, tol=1e-12, deadline=d3b)
    if c5_cd2 < best_c5:
        best_c5, best_h = c5_cd2, h_cd2.copy()

    # Phase 4: SLSQP polish (15% budget)
    d4 = t_start + budget_s * 0.70
    for attempt in range(3):
        if time.time() > d4:
            break
        if attempt == 0:
            start = best_h.copy()
        else:
            start = _project(best_h + rng.normal(0, 0.01, n0), target)
        h_s, c5_s = _slsqp_solve(start, n0, maxiter=400, ftol=1e-15)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        # Coordinate descent after SLSQP
        if time.time() < d4:
            h_c, c5_c = _coord_descent(best_h, n0, max_sweeps=100, tol=1e-13,
                                       deadline=d4)
            if c5_c < best_c5:
                best_c5, best_h = c5_c, h_c.copy()

    # Phase 5: LSE + SLSQP from perturbed starts (10% budget)
    d5 = t_start + budget_s * 0.80
    for attempt in range(30):
        if time.time() > d5:
            break
        scale = 0.02 + 0.03 * np.exp(-attempt * 0.1)
        h_p = _project(best_h + rng.normal(0, scale, n0), target)
        h_l, c5_l = _lse_step(h_p, n0, alpha=50.0, maxiter=50)
        h_s, c5_s = _slsqp_solve(h_l, n0, maxiter=200, ftol=1e-14)
        h_c, c5_c = _coord_descent(h_s, n0, max_sweeps=50, tol=1e-12, deadline=d5)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()

    # Phase 6: Try higher resolution (12% budget)
    d6 = t_start + budget_s * 0.92
    for n_try in [80, 96, 128, 84, 100]:
        if time.time() > d6:
            break
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _subgrad_momentum(h_r, n_try, sub_d, max_iters=500)
        sub_d2 = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _coord_descent(h_r, n_try, max_sweeps=100, tol=1e-12, deadline=sub_d2)
        sub_d3 = min(d6, time.time() + budget_s * 0.015)
        h_s, c5_s = _slsqp_solve(h_r, n_try, maxiter=200, ftol=1e-14)
        if c5_s < c5_r:
            h_r, c5_r = h_s, c5_s
        sub_d4 = min(d6, time.time() + budget_s * 0.015)
        h_c, c5_c = _coord_descent(h_r, n_try, max_sweeps=100, tol=1e-12, deadline=sub_d4)
        if c5_c < c5_r:
            h_r, c5_r = h_c, c5_c
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: Final polish (8% budget)
    d7 = t_start + budget_s * 0.99
    h_fp, c5_fp = _coord_descent(best_h, best_n, max_sweeps=300, tol=1e-13, deadline=d7)
    if c5_fp < best_c5:
        best_c5, best_h = c5_fp, h_fp.copy()
    h_s, c5_s = _slsqp_solve(best_h, best_n, maxiter=300, ftol=1e-15)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    h_fp2, c5_fp2 = _coord_descent(best_h, best_n, max_sweeps=200, tol=1e-13, deadline=d7)
    if c5_fp2 < best_c5:
        best_c5, best_h = c5_fp2, h_fp2.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
