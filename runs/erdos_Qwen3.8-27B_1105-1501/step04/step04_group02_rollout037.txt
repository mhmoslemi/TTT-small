import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    x = np.clip(v - alpha, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    n = len(h)
    out = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            out[r] = float(np.dot(h[d:], 1.0 - h[:m]))
        else:
            g = -d
            m = n - g
            out[r] = float(np.dot(h[:m], 1.0 - h[g:]))
    return out

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, :m] += -h[d:d+m]
            G[r, d:d+m] += 1.0 - h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += 1.0 - h[g:g+m]
            G[r, g:g+m] += -h[:m]
    return G

# SLSQP constraint functions (top-level, no closures)
def _slsqp_ineq_fun(x, n, active):
    hv = np.clip(x[:n], 0.0, 1.0)
    T = x[n]
    out = np.zeros(len(active))
    for j, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            out[j] = T - float(np.dot(hv[d:], 1.0 - hv[:m]))
        else:
            g = -d
            m = n - g
            out[j] = T - float(np.dot(hv[:m], 1.0 - hv[g:]))
    return out

def _slsqp_ineq_jac(x, n, active):
    hv = np.clip(x[:n], 0.0, 1.0)
    J = np.zeros((len(active), n + 1))
    for j, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            J[j, :m] += hv[d:d+m]
            J[j, d:d+m] += -(1.0 - hv[:m])
        else:
            g = -d
            m = n - g
            J[j, :m] += -(1.0 - hv[g:g+m])
            J[j, g:g+m] += hv[:m]
        J[j, n] = 1.0
    return J

def _slsqp_eq_fun(x, n, target):
    return float(np.sum(x[:n]) - target)

def _slsqp_eq_jac(x, n, target):
    row = np.zeros(n + 1)
    row[:n] = 1.0
    return row

def _slsqp_obj(x, n):
    return x[n]

def _slsqp_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _slsqp_epigraph(h, deadline, top_k=None, max_iter=100, ftol=1e-12):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    m_total = 2 * n - 1
    if top_k is None or top_k >= m_total:
        active = list(range(m_total))
    else:
        sc = _scores(h)
        active = list(np.argsort(sc)[::-1][:top_k])
    
    T0 = float(np.max(_scores(h))) * 1.0001 + 1e-6
    
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = T0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    
    cons = [
        {'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac, 'args': (n, active)},
        {'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac, 'args': (n, target)},
    ]
    
    try:
        res = minimize(
            fun=_slsqp_obj, x0=x0, method='SLSQP',
            jac=lambda x: _slsqp_obj_grad(x, n),
            constraints=cons, bounds=bounds,
            options={'maxiter': max_iter, 'ftol': ftol, 'disp': False}
        )
    except Exception:
        pass
    else:
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    x = np.clip(v - alpha, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    n = len(h)
    out = np.zeros(2 * n - 1)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            out[r] = float(np.dot(h[d:], 1.0 - h[:m]))
        else:
            g = -d
            m = n - g
            out[r] = float(np.dot(h[:m], 1.0 - h[g:]))
    return out

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[r, :m] += -h[d:d+m]
            G[r, d:d+m] += 1.0 - h[:m]
        else:
            g = -d
            m = n - g
            G[r, :m] += 1.0 - h[g:g+m]
            G[r, g:g+m] += -h[:m]
    return G

# SLSQP helper functions (top-level, no closures)
def _slsqp_obj(x, n):
    return x[n]

def _slsqp_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _slsqp_ineq_fun(x, n, active):
    hv = np.clip(x[:n], 0.0, 1.0)
    T = x[n]
    out = np.zeros(len(active))
    for j, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            out[j] = T - float(np.dot(hv[d:], 1.0 - hv[:m]))
        else:
            g = -d
            m = n - g
            out[j] = T - float(np.dot(hv[:m], 1.0 - hv[g:]))
    return out

def _slsqp_ineq_jac(x, n, active):
    hv = np.clip(x[:n], 0.0, 1.0)
    J = np.zeros((len(active), n + 1))
    for j, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            J[j, :m] += hv[d:d+m]
            J[j, d:d+m] += -(1.0 - hv[:m])
        else:
            g = -d
            m = n - g
            J[j, :m] += -(1.0 - hv[g:g+m])
            J[j, g:g+m] += hv[:m]
        J[j, n] = 1.0
    return J

def _slsqp_eq_fun(x, n, target):
    return float(np.sum(x[:n]) - target)

def _slsqp_eq_jac(x, n, target):
    row = np.zeros(n + 1)
    row[:n] = 1.0
    return row

def _slsqp_epigraph(h, deadline, top_k=None, max_iter=100, ftol=1e-12):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    if time.time() > deadline:
        return best_h, best_c5
    m_total = 2 * n - 1
    if top_k is None or top_k >= m_total:
        active = tuple(range(m_total))
    else:
        sc = _scores(h)
        active = tuple(np.argsort(sc)[::-1][:top_k])
    max_s = float(np.max(_scores(h)))
    T0 = max_s * 1.0001 + 1e-6
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = T0
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac, 'args': (n, active)},
        {'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac, 'args': (n, target)},
    ]
    try:
        res = minimize(
            fun=_slsqp_obj, x0=x0, args=(n,), method='SLSQP',
            jac=_slsqp_obj_grad, constraints=cons, bounds=bounds,
            options={'maxiter': max_iter, 'ftol': ftol, 'disp': False}
        )
    except Exception:
        res = None
    if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
        h_new = _project(res.x[:n], target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    n = len(h)
    G = _grad_matrix(h)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max):
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    n_grid = 25
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 2)]
    hi_t = ts[min(n_grid - 1, idx + 2)]
    if hi_t <= lo_t:
        lo_t, hi_t = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi_t - lo_t < 1e-13:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _exact_c5(h + c2 * p)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _exact_c5(h + opt_t * p)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_local(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        sc = _scores(h)
        dx = 2.0 / n
        sc_phys = sc * dx
        p = _lp_direction(h, sc_phys, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.65
            if stall % 12 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(12345 + _pass * 31)
        if n > 50:
            idx_j = rng.integers(0, n, size=min(n, 35))
            idx_k = rng.integers(0, n, size=min(n, 35))
            pairs = set()
            for jj in idx_j:
                for kk in idx_k:
                    if jj != kk:
                        pairs.add((min(int(jj), int(kk)), max(int(jj), int(kk))))
            pair_list = list(pairs)
        else:
            pair_list = [(j, k) for j in range(n) for k in range(j + 1, n)]
        rng.shuffle(pair_list)
        for (j, k) in pair_list:
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                am = 1e10
                if dj > 0:
                    am = min(am, (1.0 - h[j]) / dj)
                else:
                    am = min(am, h[j] / (-dj))
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-12:
                    continue
                for frac in [0.3, 0.6, 1.0]:
                    h_t = h.copy()
                    h_t[j] += dj * am * frac
                    h_t[k] += dk * am * frac
                    if h_t[j] < -1e-12 or h_t[j] > 1 + 1e-12 or h_t[k] < -1e-12 or h_t[k] > 1 + 1e-12:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smoothed_grad(h, tau):
    n = len(h)
    sc = _scores(h)
    max_s = float(np.max(sc))
    shifted = tau * (sc - max_s)
    exp_s = np.exp(shifted)
    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    return w @ G

def _smoothed_descend(h, deadline, tau=50.0, lr=0.01, mom=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    velocity = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        if stall > 50:
            break
        grad = _smoothed_grad(h, tau) * (2.0 / n)
        velocity = mom * velocity + (1.0 - mom) * grad
        h_new = h - lr * velocity
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.12)
        else:
            stall += 1
            lr *= 0.9
    return best_h, best_c5
