    dx = 2.0 / n

    return float(np.max(exact_scores(h)))

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    x = np.clip(v - mid, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_value_and_weights(scores, tau):
    M = np.max(scores)
    u = np.exp((scores - M) / tau)
    Z = np.sum(u)
    val = M + tau * np.log(Z)
    w = u / Z
    return val, w

def smooth_grad(h, tau):

    dx = 2.0 / n

    grad = np.zeros(n)

        if wr < 1e-14:
            continue
        m = n - d
        grad[d:] += wr * (1.0 - h[:m])
        grad[:m] -= wr * h[d:]
    # d<0
    for d in range(1, n):

        if wr < 1e-14:

        m = n - d
