def _project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    return _project(h_new, n_new / 2.0)

def _fourier_optimize(h_init, deadline, K=12, n_restarts=5, rng=None):
    """Optimize in Fourier coefficient space using scipy minimize"""
    n = len(h_init)
    target = n / 2.0
    x = np.arange(n) / n
    
    # Build Fourier basis
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(2 * np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    B_mean = B.mean(axis=0)
    B_centered = B - B_mean
    
    # Initial coefficients
    h_centered = h_init - 0.5
    a_init = np.linalg.lstsq(B_centered, h_centered, rcond=None)[0]
    
    best_c5 = _exact_c5(h_init)
    best_h = h_init.copy()
    
    def objective(a):
        h = 0.5 + B_centered @ a
        h = _project(h, target)
        return _exact_c5(h)
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart == 0:
            a0 = a_init.copy()
        else:
            a0 = a_init + 0.15 * rng.standard_normal(2 * K)
        
        try:
            res = minimize(objective, a0, method='Nelder-Mead',
                          options={'maxiter': 500, 'fatol': 1e-8, 'xatol': 1e-8,
                                  'adaptive': True})
            if res.fun < best_c5:
                h_opt = 0.5 + B_centered @ res.x
                h_opt = _project(h_opt, target)
                c5_opt = _exact_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except Exception:
            pass
    
    return best_h, best_c5

    x = np.arange(n) / n
    h = ((x * alpha + offset) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _binary_sa(n, deadline, rng, n_iters=30000):
    target_count = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target_count]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones_set = set(positions[:target_count].tolist())
    zeros_set = set(positions[target_count:].tolist())
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if len(ones_set) == 0 or len(zeros_set) == 0:
            break
        i = list(ones_set)[rng.integers(len(ones_set))]
        j = list(zeros_set)[rng.integers(len(zeros_set))]
        h[i] = 0.0
        h[j] = 1.0
        ones_set.remove(i)
        ones_set.add(j)
        zeros_set.remove(j)
        zeros_set.add(i)
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_set.remove(j)
            ones_set.add(i)
            zeros_set.remove(i)
            zeros_set.add(j)
        T *= cool
    return best_h, best_c5

def _perturb_restart(h, scale, rng):
    """Structured perturbation: block flips + Gaussian noise"""
    n = len(h)
    h_p = h.copy()
    # Gaussian noise
    h_p += scale * rng.standard_normal(n)

    L = rng.integers(max(4, n // 16), max(5, n // 4))
    start = rng.integers(0, n - L)
    h_p[start:start+L] = 1.0 - h_p[start:start+L]
    return _project(h_p, n / 2.0)

        c5 = _exact_c5(h)
        if c5 < best_c5_val:
            best_c5_val = c5
            best_pattern = h.copy()
        if best_c5_val < 0.37:
            break
    return best_pattern, best_c5_val

def _binary_sa_fast(n, deadline, rng, n_iters=40000):
    target_count = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target_count]] = 1.0
    ones_idx = positions[:target_count].copy()
    zeros_idx = positions[target_count:].copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.035
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ii = rng.integers(len(ones_idx))
        jj = rng.integers(len(zeros_idx))
        i = ones_idx[ii]
        j = zeros_idx[jj]
        h[i] = 0.0
        h[j] = 1.0
        ones_idx[ii] = j
        zeros_idx[jj] = i
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_idx[ii] = i
            zeros_idx[jj] = j
        T *= cool
    return best_h, best_c5

def _structured_perturb(h, scale, rng):
    n = len(h)
    h_p = h.copy()
    h_p += scale * rng.standard_normal(n)
    n_flips = rng.integers(1, 4)
    for _ in range(n_flips):
        L = rng.integers(max(4, n // 20), max(5, n // 5))
        start = rng.integers(0, n - L)
        h_p[start:start+L] = 1.0 - h_p[start:start+L]
    return _project(h_p, n / 2.0)

def _subgradient_step(h, deadline, n_steps=100, beta_start=20.0, beta_end=200.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.1
        for _ in range(25):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            alpha *= 0.65
            if alpha < 1e-10:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0
    
    # Phase 1: LP polish from initial (7%)
    dl = t0 + budget_s * 0.07
    h1, c5_1 = _local_opt(h_start, dl, top_k_list=[None, 30, 15, 10, 6, 4, 3],
                           radius_init=0.25, max_iters=4000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Subgradient + Fourier (13%)
    if time.time() < t0 + budget_s * 0.20:
        dl = t0 + budget_s * 0.20
        h_sg, c5_sg = _subgradient_step(best_h, dl, n_steps=80, beta_start=15.0, beta_end=150.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    if time.time() < t0 + budget_s * 0.25:
        dl = t0 + budget_s * 0.25
        h_f, c5_f = _fourier_optimize(best_h, dl, K=9, n_restarts=4, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Phase 3: Multi-resolution diverse search (30%)
    dl = t0 + budget_s * 0.55
    resolutions = [96, 112, 128, 144, 160, 128, 144, 112]
    alphas = [0.6180339887, 0.4142135624, 0.3819660113, 0.7071067812,
              0.2928932188, 0.5, 0.4450418679, 0.3568220898]
    
    # Try periodic patterns first
    for L in [8, 12, 16]:
        if time.time() > dl:
            break
        for n_try in [L * 10, L * 12]:
            if n_try > 200:
                continue
            pat, pat_c5 = _periodic_patterns(n_try, L, rng)
            if pat is not None and pat_c5 < best_c5:
                best_c5 = pat_c5
                best_h = pat.copy()
                best_n = n_try
            elif pat is not None:
                sub_dl = time.time() + min(budget_s * 0.02, (dl - time.time()) * 0.1)
                h_try, c5_try = _local_opt(pat, sub_dl,
                                             top_k_list=[None, 15, 8, 5],
                                             radius_init=0.2, max_iters=1500)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
    
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl:
            break
        if best_c5 < 0.3806:
            break
        n_starts = 3
        for s in range(n_starts):
            if time.time() > dl:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                alpha = alphas[ri % len(alphas)]
                offset = rng.random()
                h_init = _quasi_init(n_try, alpha, offset)
                h_init += rng.normal(0, 0.1, n_try)
                h_init = _project(h_init, n_try / 2.0)
            else:
                h_init = _structured_perturb(_resample(best_h, n_try), 0.07, rng)
            
            remaining = max(0.1, dl - time.time())
            per_start = min(remaining / max(1, n_starts - s), budget_s * 0.038)
            sub_dl = time.time() + per_start
            
            h_try, c5_try = _local_opt(h_init, sub_dl,
                                         top_k_list=[None, 25, 12, 8, 5, 3],
                                         radius_init=0.2, max_iters=2500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
    
    # Phase 4: Binary SA (10%)
    dl = t0 + budget_s * 0.65
    for n_bin in [96, 128, 160]:
        if time.time() > dl:
            break
        if best_c5 < 0.3806:
            break
        sub_dl = time.time() + min(budget_s * 0.028, max(1.0, (dl - time.time()) / 3))
        h_b, c5_b = _binary_sa_fast(n_bin, sub_dl, rng, n_iters=35000)
        if c5_b < best_c5:
            sub_dl2 = time.time() + min(budget_s * 0.02, max(0.5, (dl - time.time()) / 4))
            h_bp, c5_bp = _local_opt(h_b, sub_dl2,
                                       top_k_list=[15, 8, 5, 3, None],
                                       radius_init=0.12, max_iters=1200)
            if c5_bp < best_c5:
                best_c5 = c5_bp
                best_h = h_bp.copy()
                best_n = n_bin
            elif c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
    
    # Phase 5: Aggressive restart loop (20%)
    dl = t0 + budget_s * 0.85
    scale = 0.035
    for attempt in range(400):
        if time.time() > dl:
            break
        h_pert = _structured_perturb(best_h, scale, rng)
        remaining = max(0.05, dl - time.time())
        per_attempt = min(remaining / max(1, 400 - attempt), budget_s * 0.01)
        sub_dl = time.time() + per_attempt
        h_try, c5_try = _local_opt(h_pert, sub_dl,
                                     top_k_list=[10, 6, 4, 3, 8],
                                     radius_init=0.08, max_iters=500)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(best_h)
            scale = max(0.004, scale * 0.88)
        else:
            scale = min(0.13, scale * 1.06)
    
    # Phase 6: Final intensive polish (13%)
    dl = t0 + budget_s * 0.98
    # LP with all constraints, small radius
    h_fin, c5_fin = _local_opt(best_h, dl,
                                 top_k_list=[None, 10, 6, 4, 3, 5],
                                 radius_init=0.02, max_iters=3000)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()
    
    # Fourier polish
    if time.time() < dl - 1.0:
        h_fp, c5_fp = _fourier_optimize(best_h, dl, K=8, n_restarts=2, rng=rng)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()
    
    # Final LP micro-polish
    if time.time() < t0 + budget_s * 0.995:
        h_ult, c5_ult = _local_opt(best_h, t0 + budget_s * 0.997,
                                     top_k_list=[3, 4, 5, 3, 4],
                                     radius_init=0.005, max_iters=500)
        if c5_ult < best_c5:
            best_c5 = c5_ult
            best_h = h_ult.copy()
    
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
