def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_slopes(h):
    """Compute the slope matrix: slopes[d, j] = dS_d/dh[j].
    d goes from 0 to 2n-2 (corresponding to lag d-(n-1)).
    Returns array of shape (2n-1, n).
    """
    n = len(h)
    K = 2 * n - 1
    slopes = np.zeros((K, n))
    
    for d in range(K):
        lag = d - (n - 1)
        g = abs(lag)
        if g == 0:
            # S_0 = sum h[i](1-h[i]), d/dh[j] = 1 - 2h[j]
            slopes[d, :] = 1.0 - 2.0 * h
        elif lag > 0:
            # S = sum_{m=0}^{n-1-g} h[m+g](1-h[m])
            # d/dh[j] = (1-h[j-g]) * I(g <= j <= n-1) - h[j+g] * I(0 <= j <= n-1-g)
            # Term 1: j from g to n-1, slope = 1 - h[j-g]
            for j in range(g, n):
                slopes[d, j] += (1.0 - h[j - g])
            # Term 2: j from 0 to n-1-g, slope = -h[j+g]
            for j in range(0, n - g):
                slopes[d, j] += -h[j + g]
        else:
            # lag = -g, S = sum_{m=0}^{n-1-g} h[m](1-h[m+g])
            # d/dh[j] = (1-h[j+g]) * I(0 <= j <= n-1-g) - h[j-g] * I(g <= j <= n-1)
            for j in range(0, n - g):
                slopes[d, j] += (1.0 - h[j + g])
            for j in range(g, n):
                slopes[d, j] += -h[j - g]
    
    return slopes

Now for the 1D minimax solver. Given:
- S: current values (length K)
- b: direction vector (length K), where b[d] = slopes[d,j] - slopes[d,l]
- t_lo, t_hi: bounds on t

We want to minimize f(t) = max_d(S[d] + t*b[d]) for t ∈ [t_lo, t_hi].

The function f is convex and piecewise linear. The minimum is at a breakpoint or at a bound.

Efficient algorithm:
1. Evaluate f at t=0 (or t_lo if 0 is out of range)
2. Find the active set A = {d: S[d] + t*b[d] = f(t)}
3. If min_{d∈A} b[d] ≤ 0 ≤ max_{d∈A} b[d], we're at the minimum
4. Otherwise:
   - If all b[d] > 0 for d∈A, we need to decrease t. Find the nearest breakpoint below t.
   - If all b[d] < 0 for d∈A, we need to increase t. Find the nearest breakpoint above t.
5. Jump to the breakpoint and repeat.

Finding the nearest breakpoint: for the current active set A, the breakpoints are where a line in A crosses a line not in A. The nearest one in the direction we want is:
t_bp = min/max over d∈A, e∉A of (S[e] - S[d]) / (b[d] - b[e]) (in the appropriate direction)

This is O(K) per step, and at most O(K) steps, so O(K²) total.

def _minimax_1d(S, b, t_lo, t_hi):

    f_lo = float(np.max(S + t_lo * b))
    f_hi = float(np.max(S + t_hi * b))

    f_0 = float(np.max(S + t_0 * b))

            c = d
            fc = fd

            best_f = f_check
            best_t = t_check

        vals = S + t * b
        max_val = float(np.max(vals))
        active = (vals >= max_val - 1e-12)

        min_active_b = float(np.min(b[active]))

            # We want the largest t_bp < t (nearest to the left)

                        continue

                        best_bp = bp
            t = best_bp

    f_lo = float(np.max(S + t_lo * b))

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_slopes(h):
    n = len(h)
    K = 2 * n - 1
    slopes = np.empty((K, n))
    for d in range(K):
        lag = d - (n - 1)
        if lag == 0:
            slopes[d, :] = 1.0 - 2.0 * h
        elif lag > 0:
            g = lag
            m = n - g
            slopes[d, :] = 0.0
            if m > 0:
                slopes[d, g:g+m] += (1.0 - h[:m])
                slopes[d, :m] += (-h[g:])
        else:
            g = -lag

            slopes[d, :] = 0.0

def _minimax_1d(S, b, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        t0 = max(t_lo, min(t_hi, 0.0))
        return t0, float(np.max(S + t0 * b))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, br = t_lo, t_hi
    c = br - gr * (br - a)
    d2 = a + gr * (br - a)
    fc = float(np.max(S + c * b))
    fd = float(np.max(S + d2 * b))
    for _ in range(55):
        if br - a < 1e-13:
            break
        if fc < fd:
            br = d2
            d2 = c
            fd = fc
            c = br - gr * (br - a)
            fc = float(np.max(S + c * b))
        else:
            a = c
            c = d2
            fc = fd
            d2 = a + gr * (br - a)
            fd = float(np.max(S + d2 * b))
    t_opt = 0.5 * (a + br)
    f_opt = float(np.max(S + t_opt * b))
    for t_chk in (t_lo, t_hi, 0.0):
        if t_lo <= t_chk <= t_hi:
            f_chk = float(np.max(S + t_chk * b))
            if f_chk < f_opt:
                t_opt, f_opt = t_chk, f_chk
    return t_opt, f_opt

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, br = t_lo, t_hi
    c = br - gr * (br - a)
    d2 = a + gr * (br - a)
    fc = float(np.max(S0 + c * T - c * c * C)) * dx
    fd = float(np.max(S0 + d2 * T - d2 * d2 * C)) * dx
    for _ in range(40):
        if br - a < 1e-12:
            break
        if fc < fd:
            br = d2
            d2 = c
            fd = fc

        else:
            a = c
            c = d2
            fc = fd

    t_opt = 0.5 * (a + br)
    f_opt = float(np.max(S0 + t_opt * T - t_opt * t_opt * C)) * dx
    if f_opt < fc:

    return t_opt, f_opt

def _lp_direction(h, scores, dx, radius, top_k=30):
    n = len(h)
    K = 2 * n - 1
    max_s = float(np.max(scores))
    tol = 0.005 * max_s + 1e-6
    active = scores >= max_s - tol
    n_active = int(np.sum(active))
    if n_active == 0:
        return None
    if n_active > top_k:
        idx = np.argsort(scores)[-top_k:]
        active = np.zeros(K, dtype=bool)
        active[idx] = True
    active_idx = np.where(active)[0]
    G = np.zeros((len(active_idx), n))
    for ii, r in enumerate(active_idx):
        lag = r - (n - 1)
        if lag == 0:
            G[ii, :] = 1.0 - 2.0 * h
        elif lag > 0:
            g = lag
            m = n - g
            if m > 0:
                G[ii, g:g+m] = (1.0 - h[:m])
                G[ii, :m] += (-h[g:])
        else:
            g = -lag
            m = n - g
            if m > 0:
                G[ii, :m] = (1.0 - h[g:])
                G[ii, g:g+m] += (-h[:m])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -(scores[active_idx] - max_s + 1e-6)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, max_stall=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gradient_step(h, deadline, tau_list=(50, 120, 300), lr_init=0.02, mom_decay=0.85, max_stall=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    ti = 0
    while time.time() < deadline - 2:
        if stall > max_stall:
            break
        tau = tau_list[ti % len(tau_list)]
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * dx
        max_s = float(np.max(scores))
        exp_s = np.exp(tau * (scores - max_s))
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-14:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:d+m])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (-h[g:g+m])
                    grad[g:] += w[r] * (1.0 - h[:m])
        grad -= np.mean(grad)
        dn = np.linalg.norm(grad)
        if dn < 1e-13:
            ti += 1
            continue
        grad = grad / dn
        mom = mom_decay * mom + (1.0 - mom_decay) * grad
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.5)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.4)
        else:
            stall += 1
            lr *= 0.8
            if stall % 20 == 0:
                ti += 1
                mom = np.zeros(n)
    return best_h, best_c5

def _validate(h):
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-9 or np.max(h) > 1.0 + 1e-9:
        return False
    target = len(h) / 2.0
    if abs(np.sum(h) - target) > 1e-6:
        return False
    return True
