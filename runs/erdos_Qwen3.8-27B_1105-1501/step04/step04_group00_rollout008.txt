import numpy as np
import time
from scipy.optimize import linprog

def exact_c5(h):
    n = len(h)
    if n <= 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project(v, target=None):
    v = np.asarray(v, dtype=float).copy()
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    x[np.abs(x) < 1e-16] = 0.0
    x[np.abs(1.0 - x) < 1e-16] = 1.0
    return x

def build_grad_rows(h, active_idx):
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    G = np.zeros((m, n))
    for k in range(m):
        r = int(active_idx[k])
        d = r - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                G[k, d:d+m2] += dx * (1.0 - h[:m2])
                G[k, :m2] -= dx * h[d:d+m2]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                G[k, :m2] += dx * (1.0 - h[g:g+m2])
                G[k, g:g+m2] -= dx * h[:m2]
    return G

def kkt_direction(h, scores, active_idx, margin_factor, margin_floor, ridge, rough, priority_power):
    n = len(h)
    s_active = scores[active_idx]
    c_max = float(np.max(s_active))
    c_min = float(np.min(s_active))
    spread = c_max - c_min
    sigma = max(margin_floor, min(3e-4, margin_factor * (spread + 2 * margin_floor)))
    target_level = c_min - sigma
    gaps = c_max - s_active
    max_gap = float(np.max(gaps)) if len(gaps) > 0 else 1.0
    if max_gap < 1e-15:
        q = np.ones(len(active_idx))
    else:
        q = 1.0 + priority_power * (gaps / max_gap) ** 2
        q = np.maximum(q, 1e-6)
    G = build_grad_rows(h, active_idx)
    if not np.all(np.isfinite(G)):
        return None
    residual = target_level - s_active
    H = G.T @ (q[:, None] * G) + ridge * np.eye(n)
    if rough > 0:
        for i in range(n):
            H[i, i] += 2.0 * rough
            if i > 0:
                H[i, i-1] -= rough
                H[i-1, i] -= rough
    rhs = G.T @ (q * residual)
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs_vec = np.zeros(n + 1)
    rhs_vec[:n] = rhs
    try:
        sol = np.linalg.solve(KKT, rhs_vec)
        delta = sol[:n]
    except np.linalg.LinAlgError:
        try:
            KKT2 = KKT.copy()
            KKT2[:n, :n] += 0.05 * np.eye(n)
            sol = np.linalg.solve(KKT2, rhs_vec)
            delta = sol[:n]
        except Exception:
            delta = -G.T @ q
            delta -= delta.mean()
    if not np.all(np.isfinite(delta)):
        return None
    delta -= delta.mean()
    if np.max(np.abs(delta)) < 1e-14:
        return None
    return delta

def line_search(h, direction, max_alpha=0.3):
    n = len(h)
    d = direction.copy()
    d_max = float(np.max(np.abs(d)))
    if d_max < 1e-15:
        return 0.0, exact_c5(h)
    d_unit = d / d_max
    alpha_max = max_alpha
    for j in range(n):
        if d_unit[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d_unit[j])
        elif d_unit[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d_unit[j]))
    alpha_max = max(alpha_max, 1e-14) * 0.99
    if alpha_max < 1e-14:
        return 0.0, exact_c5(h)
    n_samp = 10
    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.empty(n_samp)
    for i in range(n_samp):
        vals[i] = exact_c5(h + alphas[i] * d_unit)
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * d_unit)
    f2 = exact_c5(h + c2 * d_unit)
    for _ in range(18):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, f2, c2, c1 = c2, f1, c1, hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * d_unit)
        else:
            lo, f1, c1, c2 = c1, f2, c2, lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * d_unit)
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * d_unit)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def active_optimize(h, deadline, variants=None, max_iters=3000):
    if variants is None:
        variants = [
            (2e-3, 8, 40, 0.3, 5e-5, 1e-2, 1e-2, 20, 0.25),
            (5e-4, 6, 24, 0.15, 2e-5, 5e-3, 5e-3, 200, 0.15),
            (1e-5, 4, 14, 0.06, 8e-6, 2e-3, 0.0, 2000, 0.06),
            (8e-4, 5, 18, 0.1, 1e-5, 3e-3, 2e-3, 150, 0.1),
        ]
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_var = len(variants)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        v = variants[ti % n_var]
        atol, min_act, max_act, mf, mfl, ridge, rough, prio, trust = v
        scores = scores_exact(h)
        c_max = float(np.max(scores))
        sorted_idx = np.argsort(scores)[::-1]
        active_mask = scores >= c_max - atol
        active_idx = sorted_idx[active_mask]
        if len(active_idx) < min_act:
            active_idx = sorted_idx[:min_act]
        if len(active_idx) > max_act:
            active_idx = active_idx[:max_act]
        if len(active_idx) == 0:
            stall += 1
            continue
        direction = kkt_direction(h, scores, active_idx, mf, mfl, ridge, rough, prio)
        if direction is None or np.max(np.abs(direction)) < 1e-13:
            G0 = build_grad_rows(h, [sorted_idx[0]])
            direction = -G0[0]
            direction -= direction.mean()
        if direction is None or np.max(np.abs(direction)) < 1e-13:
            stall += 1
            ti += 1
            continue
        alpha, new_c5 = line_search(h, direction, max_alpha=trust)
        if new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha * (direction / np.max(np.abs(direction)))
            h_new = project(h_new, n / 2.0)
            cur_c5 = exact_c5(h_new)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_new.copy()
            h = h_new
            stall = 0
            trust = min(trust * 1.03, 0.35)
        else:
            stall += 1
            trust = max(0.03, trust * 0.7)
            ti += 1
    return best_h, best_c5

def lp_direction(h, scores, top_k, radius=0.1):
    n = len(h)
    n_lags = len(scores)
    idx = np.argsort(scores)[-top_k:]
    G = build_grad_rows(h, idx)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def pair_polish(h, deadline, max_passes=10):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 24)
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-10 or h[j] > 1.0 - 1e-10:
                continue
            for k in range(0, n, step):
                if k == j:
                    continue
                if h[k] < 1e-10 or h[k] > 1.0 - 1e-10:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(8):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = np.array([((x[j] * alpha) % 1.0) < 0.5 for j in range(n)], dtype=float)
    return project(h, n / 2.0)

def smooth_perturb(h, scale, rng):
    n = len(h)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for k in range(1, 7):
        phase = rng.uniform(0, 2 * np.pi)
        pert += (scale / k) * np.cos(2 * np.pi * k * x + phase)
    return project(h + pert, n / 2.0)

def weighted_grad_optimize(h, deadline, n_steps=200):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = 5.0 + 200.0 * frac
        scores = scores_exact(h)
        max_s = float(np.max(scores))
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = build_grad_rows(h, np.arange(len(scores)))
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        for alpha_try in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            h_new = project(h - alpha_try * grad, n / 2.0)
            c5_new = exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
    return best_h, best_c5

    return best_h, final_c5, best_n
