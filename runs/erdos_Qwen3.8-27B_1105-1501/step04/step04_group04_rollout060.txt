def _gradient_descent(h, deadline, beta_start=20, beta_end=500, n_beta=20):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    betas = np.geomspace(beta_start, beta_end, n_beta)
    
    for beta in betas:
        if time.time() > deadline:
            break
        cur_c5 = best_c5
        step = 0.05
        stall = 0
        for it in range(5000):
            if time.time() > deadline:
                break
            if stall > 100:
                break
            grad = _smoothed_max_gradient(h, beta)
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                stall += 1000
                break
            grad = grad / gnorm
            
            # Try different step sizes
            best_trial = None
            best_trial_c5 = cur_c5
            for f in [step, step*0.5, step*0.25, step*0.1]:
                trial = _project_box_sum(h - f * grad, target)
                tc5 = _exact_c5(trial)
                if tc5 < best_trial_c5 - 1e-14:
                    best_trial_c5 = tc5
                    best_trial = trial
            if best_trial is not None:
                h = best_trial
                cur_c5 = best_trial_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step = min(step * 1.02, 0.5)
            else:
                stall += 1
                step *= 0.8
    
    return best_h, best_c5
