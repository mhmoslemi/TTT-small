import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)
            for i in order:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(-x)
            for i in order:
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5_val(h, n):
    return float(np.max(_corrs(h)) * 2.0 / n)


def _grad_corr(h, n, s):
    """Gradient of correlation at shift s."""
    j = np.arange(n)
    g = np.zeros(n)
    gs = j + s
    m1 = (gs >= 0) & (gs < n)
    g[m1] += 1.0 - h[gs[m1]]
    gs2 = j - s
    m2 = (gs2 >= 0) & (gs2 < n)
    g[m2] -= h[gs2[m2]]
    return g


def _topk_grad(h, n, K):
    """Weighted average gradient of top-K correlations."""
    c = _corrs(h)
    n_corr = len(c)
    top_idx = np.argsort(-c)[:K]
    # Weight by exponential of (c_i - c_max) for smoothness
    c_top = c[top_idx]
    w = np.exp(10.0 * (c_top - c_top[0]))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(K):
        s = top_idx[k] - (n - 1)
        g += w[k] * _grad_corr(h, n, s)
    return g, c, top_idx


def _lse_fg(x, n, alpha):
    """Log-sum-exp smoothed objective and gradient."""
    h = np.asarray(x, dtype=float)
    c = _corrs(h)
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_v = np.exp(shifted)
    f = c_max + (1.0 / alpha) * np.log(exp_v.sum())
    w = exp_v / exp_v.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for r in range(len(c)):
        if w[r] < 1e-15:
            continue
        s = r - (n - 1)
        gs = j + s
        m1 = (gs >= 0) & (gs < n)
        g[m1] += w[r] * (1.0 - h[gs[m1]])
        gs2 = j - s
        m2 = (gs2 >= 0) & (gs2 < n)
        g[m2] -= w[r] * h[gs2[m2]]
    return f, g


def _lse_optimize(h, n, alpha, maxiter=200):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_fg, h.copy(), args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxls': 50, 'iprint': 0})
        x = _project(res.x, target)
        return x, _c5_val(x, n)
    except Exception:
        return h.copy(), _c5_val(h, n)


def _continuation_lse(h, n, deadline, alphas=None):
    """Continuation method: gradually increase alpha (sharpen the max)."""
    if alphas is None:
        alphas = np.linspace(5.0, 200.0, 30)
    best_h, best_c5 = h.copy(), _c5_val(h, n)
    cur_h = h.copy()
    for alpha in alphas:
        if time.time() > deadline:
            break
        # Use more iterations for larger alpha (sharper)
        iters = min(300, int(50 + 5.0 * alpha))
        cur_h, c5 = _lse_optimize(cur_h, n, alpha, maxiter=iters)
        if c5 < best_c5:
            best_c5, best_h = c5, cur_h.copy()
    return best_h, best_c5


def _bb_descent(h, n, deadline, max_iter=5000, K=8):
    """Barzilai-Borwein gradient descent on exact max with top-K averaging."""
    target = n * 0.5
    h = h.copy()
    cur_c5 = _c5_val(h, n)
    best_h, best_c5 = h.copy(), cur_c5
    prev_h = h.copy()
    prev_g = _topk_grad(h, n, K)[0]
    use_bb1 = True
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        g, c, _ = _topk_grad(h, n, K)
        # BB step size
        s = h - prev_h
        y = g - prev_g
        sy = s @ y
        if sy > 1e-14:
            if use_bb1:
                ss = s @ s
                alpha_bb = ss / sy if ss > 1e-14 else 0.01
            else:
                yy = y @ y
                alpha_bb = sy / yy if yy > 1e-14 else 0.01
            use_bb1 = not use_bb1
        else:
            alpha_bb = 0.01
        alpha_bb = np.clip(alpha_bb, 1e-6, 0.1)

        # Try the BB step with line search
        best_a, best_c5_try = 0.0, cur_c5
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a = alpha_bb * frac
            h_try = _project(h - a * g, target)
            c5_try = _c5_val(h_try, n)
            if c5_try < best_c5_try:
                best_c5_try, best_a = c5_try, a
        if best_a > 0 and best_c5_try < cur_c5 - 1e-14:
            h = _project(h - best_a * g, target)
            cur_c5 = _c5_val(h, n)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 200:
                # Perturb to escape
                h = _project(h + 0.01 * np.random.randn(n), target)
                cur_c5 = _c5_val(h, n)
                stall = 0
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()

        prev_h, prev_g = h.copy(), g

    return best_h, best_c5


def _slsqp_active(h, n, K=10, maxiter=300):
    """SLSQP with only top-K correlation constraints + t variable."""
    c = _corrs(h)
    top_idx = np.argsort(-c)[:K]
    t0 = float(c.max()) + 1e-4
    z0 = np.concatenate([h, [t0]])
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]

    def _obj(z, nn):
        return z[nn]

    def _obj_jac(z, nn):
        gz = np.zeros(nn + 1)
        gz[nn] = 1.0
        return gz

    def _eq_f(z, nn):
        return np.sum(z[:nn]) - nn * 0.5

    def _eq_jac(z, nn):
        gz = np.ones(nn + 1)
        gz[nn] = 0.0
        return gz

    def _ineq_f(z, idx, nn):
        hh, t = z[:nn], z[nn]
        cc = np.correlate(hh, 1.0 - hh, mode='full')
        return t - cc[idx]

    def _ineq_jac(z, idx, nn):
        hh = z[:nn]
        j = np.arange(nn)
        J = np.zeros((len(idx), nn + 1))
        for row, r in enumerate(idx):
            s = r - (nn - 1)
            gs = j + s
            m1 = (gs >= 0) & (gs < nn)
            J[row, m1] += 1.0 - hh[gs[m1]]
            gs2 = j - s
            m2 = (gs2 >= 0) & (gs2 < nn)
            J[row, m2] -= hh[gs2[m2]]
            J[row, nn] = 1.0
        return J

    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_jac, 'args': (top_idx, n)}
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        x = _project(res.x[:n], target)
        return x, _c5_val(x, n)
    except Exception:
        return h.copy(), _c5_val(h, n)


def _binary_round(h, n):
    """Round to nearest binary with exact mass n/2."""
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def _binary_local_search(h, n, deadline, max_passes=200):
    """Local search on binary configurations using swaps."""
    target = n // 2
    b = _binary_round(h, n)
    cur_c5 = _c5_val(b, n)
    best_b, best_c5 = b.copy(), cur_c5
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try random swaps
        n_swaps = min(200, len(ones) * len(zeros))
        for _ in range(n_swaps):
            if time.time() > deadline:
                break
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            if i == j:
                continue
            b[i], b[j] = 0.0, 1.0
            c5_try = _c5_val(b, n)
            if c5_try < cur_c5 - 1e-15:
                cur_c5 = c5_try
                improved = True
                if c5_try < best_c5:
                    best_c5, best_b = c5_try, b.copy()
                ones = np.where(b == 1.0)[0]
                zeros = np.where(b == 0.0)[0]
            else:
                b[i], b[j] = 1.0, 0.0
        if not improved:
            break

    return best_b, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _slsqp_active_g(z, idx, nn):
    h = z[:nn]
    j = np.arange(nn)
    J = np.zeros((len(idx), nn + 1))
    for row, r in enumerate(idx):
        s = r - (nn - 1)
        gs = j + s
        m1 = (gs >= 0) & (gs < nn)
        J[row, m1] += 1.0 - h[gs[m1]]
        gs2 = j - s
        m2 = (gs2 >= 0) & (gs2 < nn)
        J[row, m2] -= h[gs2[m2]]
        J[row, nn] = 1.0
    return J


def _slsqp_full(h, n, maxiter=200, ftol=1e-15):
    """SLSQP with all constraints (for final polish)."""
    c = _corrs(h)
    t0 = float(c.max()) + 1e-3
    z0 = np.concatenate([h, [t0]])
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]

    def _obj(z, nn):
        return z[nn]

    def _obj_jac(z, nn):
        gz = np.zeros(nn + 1)
        gz[nn] = 1.0
        return gz

    def _eq_f(z, nn):
        return np.sum(z[:nn]) - nn * 0.5

    def _eq_jac(z, nn):
        gz = np.ones(nn + 1)
        gz[nn] = 0.0
        return gz

    n_corr = 2 * n - 1
    all_idx = np.arange(n_corr)

    def _ineq_f(z, idx, nn):
        hh, t = z[:nn], z[nn]
        cc = np.correlate(hh, 1.0 - hh, mode='full')
        return t - cc[idx]

    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _slsqp_active_g, 'args': (all_idx, n)}
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], target)
        return x, _c5_val(x, n)
    except Exception:
        return h.copy(), _c5_val(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _c5_val(h, n0)
    best_n = n0

    def _elapsed():
        return time.time() - t_start

    # Phase 1: Continuation LSE (20% budget)
    d1 = t_start + budget_s * 0.20
    cur_h, cur_c5 = _continuation_lse(h, n0, d1)
    if cur_c5 < best_c5:
        best_c5, best_h = cur_c5, cur_h.copy()

    # Phase 2: BB descent (15% budget)
    d2 = t_start + budget_s * 0.35
    cur_h, cur_c5 = _bb_descent(best_h, n0, d2, max_iter=10000, K=8)
    if cur_c5 < best_c5:
        best_c5, best_h = cur_c5, cur_h.copy()

    # Phase 3: SLSQP with reduced active set (15% budget)
    d3 = t_start + budget_s * 0.50
    for attempt in range(4):
        if time.time() > d3:
            break
        start = best_h.copy()
        if attempt > 0:
            start = _project(start + rng.normal(0, 0.015, n0), target)
        K_try = 8 + 4 * attempt
        h_s, c5_s = _slsqp_active(start, n0, K=K_try, maxiter=400)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    # Phase 4: Binary local search + continuous refinement (15% budget)
    d4 = t_start + budget_s * 0.65
    b, c5_b = _binary_local_search(best_h, n0, d4, max_passes=100)
    if c5_b < best_c5:
        # Refine from binary
        h_rf, c5_rf = _lse_optimize(b, n0, alpha=50.0, maxiter=200)
        if c5_rf < best_c5:
            best_c5, best_h = c5_rf, h_rf.copy()
        if time.time() < d4:
            h_s, c5_s = _slsqp_active(h_rf, n0, K=10, maxiter=300)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()

    # Phase 5: More continuation from perturbed starts (15% budget)
    d5 = t_start + budget_s * 0.80
    for attempt in range(5):
        if time.time() > d5:
            break
        scale = 0.02 + 0.01 * attempt
        h_p = _project(best_h + rng.normal(0, scale, n0), target)
        sub_d = min(d5, time.time() + budget_s * 0.025)
        h_c, c5_c = _continuation_lse(h_p, n0, sub_d,
                                       alphas=np.linspace(20.0, 150.0, 15))
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()

    # Phase 6: Resolution search (15% budget)
    d6 = t_start + budget_s * 0.95
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, 12]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = _lse_optimize(h_r, n_try, alpha=40.0, maxiter=100)
        if time.time() < d6:
            c = _corrs(h_r)
            K_r = int(np.sum(c > c.max() - 0.05))
            K_r = max(6, min(16, K_r))
            h_r, c5_r = _slsqp_active(h_r, n_try, K=K_r, maxiter=200)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 7: Final polish (5% budget)
    d7 = t_start + budget_s * 0.99
    h_f, c5_f = _slsqp_active(best_h, best_n, K=10, maxiter=300, )
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    if time.time() < d7:
        h_bb, c5_bb = _bb_descent(best_h, best_n, d7, max_iter=5000, K=5)
        if c5_bb < best_c5:
            best_c5, best_h = c5_bb, h_bb.copy()
    if time.time() < d7:
        h_full, c5_full = _slsqp_full(best_h, best_n, maxiter=100, ftol=1e-15)
        if c5_full < best_c5:
            best_c5, best_h = c5_full, h_full.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _c5_val(best_h, best_n)
    return best_h, final_c5, best_n
