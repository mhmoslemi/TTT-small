def _pair_slopes(h, i, j):
    n = len(h)
    dx = 2.0 / n
    d = np.arange(-(n-1), n)  # all shifts
    # h_at(i+d)
    idx1 = i + d
    a = np.ones(len(d))  # 1 - h_out = 1 - 0 = 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    a[valid1] = 1.0 - h[idx1[valid1]]
    # h_at(i-d)
    idx2 = i - d
    b = np.zeros(len(d))
    valid2 = (idx2 >= 0) & (idx2 < n)
    b[valid2] = h[idx2[valid2]]
    # h_at(j+d)
    idx3 = j + d
    c = np.ones(len(d))
    valid3 = (idx3 >= 0) & (idx3 < n)
    c[valid3] = 1.0 - h[idx3[valid3]]
    # h_at(j-d)
    idx4 = j - d
    e = np.zeros(len(d))
    valid4 = (idx4 >= 0) & (idx4 < n)
    e[valid4] = h[idx4[valid4]]
    
    slopes = dx * (a - b - c + e)
    return slopes
