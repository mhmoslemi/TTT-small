def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _proj(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search_quad(h, d, dx, trust):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = max(0.0, -trust)
    t_hi = trust
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    grid = np.linspace(t_lo, t_hi, 40)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None]**2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if 0 < best_i < 39:
        a, b = grid[best_i - 1], grid[best_i + 1]

    gr = (np.sqrt(5.0) - 1.0) / 2.0

        if b - a < 1e-12:
            break

            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    if opt_val < best_val:

def _active_gradient(h, top_k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for r in idx:
        d_off = r - (n - 1)
        if d_off >= 0:
            m = n - d_off
            if m > 0:
                grad[:m] += (1.0 - h[d_off:d_off+m])
                grad[d_off:] += -h[:m]
        else:
            g = -d_off
            m = n - g
            if m > 0:
                grad[:m] += -h[g:g+m]
                grad[g:] += (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad, idx, corr

def _bb_descent(h, deadline, top_k=5, max_iter=2000):

        if stall > 40:
            break

                pair_scores[g:, :m] += b_val[:, None] * a_val[None, :]

def _anneal(h, deadline, rng):

    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    cool = 0.995
    steps = 0
    while time.time() < deadline and T > 0.0005:
        for _ in range(8):
            if time.time() > deadline:

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amount = rng.uniform(-0.15, 0.15)

            h_new[i] -= amount
            h_new[j] += amount

                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                steps += 1

def _lp_dir(h, scores, radius):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores.astype(float)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.1):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _combined_refine(h, deadline):

    best_c5 = _exact_c5(h)
    phases = [('bb', 0.35), ('vanilla', 0.2), ('pair', 0.2), ('lp', 0.15), ('anneal', 0.1)]

        if c5_r < best_c5 - 1e-14:
