def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix rounding
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = (x < 1.0 - 1e-12) & (x > 1e-12)
        else:
            mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_rows(h, lags):
    """Compute gradient rows for specified lag indices (0 to 2n-2)."""
    n = len(h)
    dx = 2.0 / n
    rows = []
    for r in lags:
        d = r - (n - 1)
        row = np.zeros(n)
        if d >= 0:
            m = n - d
            if m > 0:
                row[d:] += (1.0 - h[:m])
                row[:m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                row[:m] += (1.0 - h[g:])
                row[g:] += -h[:m]
        rows.append(row * dx)
    return np.array(rows)

def _lp_direction(h, scores, radius, top_k=None):
    """Find LP direction to reduce max score."""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(n_lags)
    
    G_active = _grad_rows(h, idx)
    s_active = scores[idx]
    
    # LP: min t s.t. G_active @ p - t <= -s_active, ||p||_inf <= radius, 1'p = 0
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(idx), n_vars))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = -s_active
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search_cubic(h, p):
    """Cubic line search for h + alpha * p."""
    n = len(h)
    # Find max alpha
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    
    # Sample at several points
    n_samp = 20
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(np.clip(h + a * p, 0, 1)) for a in alphas])
    best_i = int(np.argmin(vals))
    
    if best_i <= 0 or best_i >= n_samp - 1:
        best_a = alphas[best_i]
        return best_a, vals[best_i]
    
    # Golden section refinement
    lo = alphas[best_i - 1]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
    f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    
    for _ in range(25):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(np.clip(h + opt_a * p, 0, 1))
    best_a = alphas[best_i]
    if opt_c5 < vals[best_i]:
        return opt_a, opt_c5
    return best_a, vals[best_i]

        if time.time() > deadline:
            break
        improved = False
        step_size = max(1, n // 24)
        for j in range(0, n, step_size):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(0, n, step_size):
                if k == j:
                    continue
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-11:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(10):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0, 1)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0, 1)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _fourier_evolve(h, deadline, n_pop=20, K=8, n_gen=200, mutation_scale=0.10, rng=None):
    n = len(h)
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(2 * np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    target = n / 2.0
    a_init = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    pop_a = a_init[None, :] + mutation_scale * rng.standard_normal((n_pop, 2*K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
        h_try = _project(h_try, target)
        pop_c5[i] = _exact_c5(h_try)
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = np.clip(0.5 + B @ pop_a[best_i], 0, 1)
    best_h = _project(best_h, target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        n_keep = max(2, n_pop // 2)
        pop_a[:n_keep] = pop_a[order[:n_keep]]
        pop_c5[:n_keep] = pop_c5[order[:n_keep]]
        for i in range(n_keep, n_pop):
            p1 = order[int(rng.integers(n_keep))]
            p2 = order[int(rng.integers(n_keep))]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1.0 - mix) * pop_a[order[p2]]
            pop_a[i] += mutation_scale * rng.standard_normal(2*K)
            h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
            h_try = _project(h_try, target)
            pop_c5[i] = _exact_c5(h_try)
        mutation_scale *= 0.996
        cur_best = int(np.argmin(pop_c5))
        if pop_c5[cur_best] < best_c5:
            best_c5 = pop_c5[cur_best]
            best_h = np.clip(0.5 + B @ pop_a[cur_best], 0, 1)
            best_h = _project(best_h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Aggressive LP from initial (12% budget)
    dl1 = t0 + budget_s * 0.12
    h1, c5_1 = _local_lp(h_start, dl1,
                         top_k_list=[None, 30, 20, 15, 10, 8, 5, 5, 3],
                         radius_init=0.3, max_iters=4000, rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n0

    # Phase 2: Weighted subgradient refinement (8%)
    if time.time() < t0 + budget_s * 0.22:
        dl2 = t0 + budget_s * 0.22
        h2, c5_2 = _weighted_subgrad(best_h, dl2, n_steps=120, gamma_start=8.0, gamma_end=150.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        # Quick LP polish on weighted result
        if time.time() < dl2:
            h2p, c5_2p = _local_lp(best_h, dl2, top_k_list=[10, 8, 5],
                                    radius_init=0.1, max_iters=500, rng=rng)
            if c5_2p < best_c5:
                best_c5 = c5_2p
                best_h = h2p.copy()

    # Phase 3: Multi-resolution diverse search (38%)
    dl3 = t0 + budget_s * 0.62
    resolutions = [96, 128, 160, 128, 192, 128, 160, 160, 256, 128]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.5, 0.3819660113, 0.65]
    ri = 0
    for n_try in resolutions:
        if time.time() > dl3:
            break
        n_starts = 3
        for s in range(n_starts):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_init(n_try, alpha)
                h_init += 0.06 * rng.standard_normal(n_try)
                h_init = _project(h_init, n_try / 2.0)
            else:
                h_init = _resample(best_h, n_try)
                h_init += 0.04 * rng.standard_normal(n_try)
                h_init = _project(h_init, n_try / 2.0)

            sub_dl = min(dl3, time.time() + budget_s * 0.03)
            h_try, c5_try = _local_lp(h_init, sub_dl,
                                       top_k_list=[None, 20, 15, 10, 8, 5],
                                       radius_init=0.22, max_iters=2500, rng=rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            # Follow-up weighted subgrad
            if time.time() < dl3 and c5_try < best_c5 + 0.002:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.012)
                h_wg, c5_wg = _weighted_subgrad(h_try, sub_dl2, n_steps=60,
                                                 gamma_start=15.0, gamma_end=100.0)
                if c5_wg < best_c5:
                    best_c5 = c5_wg
                    best_h = h_wg.copy()
                    best_n = n_try
            ri += 1

    # Phase 4: Fourier evolution (12%)
    if time.time() < t0 + budget_s * 0.75:
        dl4 = t0 + budget_s * 0.75
        h_f, c5_f = _fourier_evolve(best_h, dl4, n_pop=22, K=10,
                                     n_gen=200, mutation_scale=0.08, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # LP polish on Fourier result
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.025)
            h_fp, c5_fp = _local_lp(best_h, sub_dl,
                                     top_k_list=[12, 8, 5, 3],
                                     radius_init=0.08, max_iters=800, rng=rng)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 5: Binary SA at various resolutions (10%)
    if time.time() < t0 + budget_s * 0.85:
        dl5 = t0 + budget_s * 0.85
        for n_bin in [128, 160, 96]:
            if time.time() > dl5:
                break
            sub_dl = min(dl5, time.time() + budget_s * 0.025)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=20000)
            if c5_b < best_c5 + 0.01:
                # LP refine the binary solution
                sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
                h_br, c5_br = _local_lp(h_b, sub_dl2,
                                         top_k_list=[15, 10, 8, 5],
                                         radius_init=0.15, max_iters=1000, rng=rng)
                if c5_br < best_c5:
                    best_c5 = c5_br
                    best_h = h_br.copy()
                    best_n = n_bin

    # Phase 6: Perturbation restarts (8%)
    dl6 = t0 + budget_s * 0.93
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + scale * rng.standard_normal(best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        h_try, c5_try = _local_lp(h_pert, sub_dl,
                                   top_k_list=[10, 8, 5, 3],
                                   radius_init=0.08, max_iters=300, rng=rng)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.008, scale * 0.82)
        else:
            scale = min(0.12, scale * 1.15)

    # Phase 7: Pair polish (5%)
    if time.time() < t0 + budget_s * 0.975:
        dl7 = t0 + budget_s * 0.975
        hp, c5p = _pair_polish(best_h, dl7, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: tiny LP refinement
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _local_lp(best_h, t0 + budget_s * 0.997,
                                   top_k_list=[5, 3, 4, 3],
                                   radius_init=0.015, max_iters=600, rng=rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
