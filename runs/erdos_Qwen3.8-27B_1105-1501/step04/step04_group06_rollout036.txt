import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _jacobian_corr(h):
    n = len(h)
    nc = 2 * n - 1
    J = np.zeros((nc, n))
    j = np.arange(n)
    for r in range(nc):
        s = r - (n - 1)
        row = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        row[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        row[m2] -= h[ps[m2]]
        J[r] = row
    return J


def _lse_obj_jac(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    exp_v = np.exp(vals)
    f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_v)))
    total = float(np.sum(exp_v))
    w = exp_v / total
    J = _jacobian_corr(h)
    grad = J.T @ w
    grad -= np.mean(grad)
    return f, grad

def _active_slsqp(h, n, K, maxiter, ftol):
    h = np.asarray(h, dtype=float).copy()
    corr = _corr_vals(h)
    top_idx = np.argsort(-corr)[:K]
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_j, 'args': (n,)},
        {'type': 'ineq', 'fun': _slsqp_ineq_f, 'jac': _slsqp_ineq_j,
         'args': (n, top_idx)}
    ]
    try:
        res = minimize(_slsqp_f, z0, jac=_slsqp_jac_f, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)


def _slsqp_loop(h, n, deadline, K_start, K_max):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)
