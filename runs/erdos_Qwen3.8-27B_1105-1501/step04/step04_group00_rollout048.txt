def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):

                G[r, :m] += (1.0 - h[g:])

            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    return opt_a, opt_c5

def _lp_optimize(h, deadline, radius_init=0.25, max_it=5000, use_all=True, topk=30):
    """Main LP-based local optimizer."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        if use_all or (it % 3 == 0):
            p, t_val = _lp_all(h, radius)
        else:
            p, t_val = _lp_topk(h, radius, topk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.75
            if stall > 10 and radius < 0.03:
                radius = 0.1
    return best_h, best_c5

def _subgrad_step(h, gamma, deadline=None):
    """One subgradient step with softmax weighting."""
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    grad = G.T @ w
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, _exact_c5(h)
    grad /= gn
    # Line search
    alpha = 0.1
    cur_c5 = _exact_c5(h)
    best_a = 0.0
    best_c5 = cur_c5
    for _ in range(25):
        h_new = _project(h - alpha * grad, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_a = alpha
        alpha *= 0.7
        if alpha < 1e-10:
            break
    if best_a > 0:
        h_new = _project(h - best_a * grad, target)
        return h_new, best_c5
    return h, cur_c5

def _subgrad_optimize(h, deadline, gamma_init=10, gamma_max=500, max_it=2000):
    """Subgradient optimization with increasing gamma."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        h_new, c5_new = _subgrad_step(h, gamma)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.02, gamma_max)
        else:
            stall += 1
            gamma = min(gamma * 1.05, gamma_max)
    return best_h, best_c5

def _fw_step(h, gamma, deadline=None):
    """Frank-Wolfe step: find v minimizing subgradient dot v over polytope."""
    n = len(h)
    target = n / 2.0
    scores = _scores(h)
    max_s = np.max(scores)
    w = np.exp(gamma * (scores - max_s))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(scores))
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    G = _grad_matrix(h)
    grad = G.T @ w  # subgradient of C5 at h
    # LMO: min grad . v subject to 0<=v<=1, sum(v)=n/2
    # Optimal: v[i]=1 for the (n/2) indices with smallest grad[i]
    order = np.argsort(grad)
    v = np.zeros(n)
    # We need exactly n/2 ones (assuming n is even)
    n_ones = int(target)
    v[order[:n_ones]] = 1.0
    # Line search from h to v
    d = v - h
    if np.max(np.abs(d)) < 1e-14:
        return h, _exact_c5(h)
    # The feasible line is h + alpha*d (stays in [0,1] and sum=n/2 for alpha in [0,1])
    cur_c5 = _exact_c5(h)
    best_a = 0.0
    best_c5 = cur_c5
    # Golden section on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * d)
    f2 = _exact_c5(h + c2 * d)
    for _ in range(25):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * d)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * d)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * d)
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_a = opt_a
    # Also try full step
    full_c5 = _exact_c5(v)
    if full_c5 < best_c5:
        best_c5 = full_c5
        best_a = 1.0
    if best_a > 0:
        h_new = h + best_a * d
        h_new = _project(h_new, target)
        return h_new, _exact_c5(h_new)
    return h, cur_c5

def _fw_optimize(h, deadline, gamma_init=20, gamma_max=300, max_it=1500):
    """Frank-Wolfe optimization."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = gamma_init
    stall = 0
    for it in range(max_it):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        h_new, c5_new = _fw_step(h, gamma)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = min(gamma * 1.03, gamma_max)
        else:
            stall += 1
            gamma = min(gamma * 1.08, gamma_max)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_periodic(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def _binary_sa(n, deadline, rng, n_iters=30000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.05
    T_min = 0.001
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _perturb_restart(h, scale, deadline, rng):
    """Perturb and re-optimize."""
    n = len(h)
    target = n / 2.0
    h_p = h + rng.normal(0, scale, n)
    h_p = _project(h_p, target)
    h_opt, c5_opt = _lp_optimize(h_p, deadline, radius_init=0.1, max_it=500, use_all=True, topk=20)
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Start from initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Aggressive LP from initial (all constraints)
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _lp_optimize(h_start, dl1, radius_init=0.30, max_it=5000, use_all=True, topk=40)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Subgradient refinement
    dl2 = t0 + budget_s * 0.20
    h2, c5_2 = _subgrad_optimize(best_h, dl2, gamma_init=5, gamma_max=300, max_it=3000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP again after subgradient
    dl3 = t0 + budget_s * 0.30
    h3, c5_3 = _lp_optimize(best_h, dl3, radius_init=0.15, max_it=4000, use_all=True, topk=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Try higher resolutions
    dl4 = t0 + budget_s * 0.55
    higher_ns = [160, 192, 224, 144, 176]
    for n_try in higher_ns:
        if time.time() > dl4:
            break
        if best_c5 < 0.3806:
            break
        h_res = _resample(best_h, n_try)
        sub_dl = min(dl4, time.time() + budget_s * 0.04)
        h_try, c5_try = _lp_optimize(h_res, sub_dl, radius_init=0.2, max_it=3000, use_all=True, topk=30)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            # Continue optimizing at this resolution
            sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
            h_try2, c5_try2 = _lp_optimize(h_try, sub_dl2, radius_init=0.1, max_it=2000, use_all=True, topk=20)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase 5: FW optimization
    if time.time() < t0 + budget_s * 0.65:
        dl5 = t0 + budget_s * 0.72
        h_fw, c5_fw = _fw_optimize(best_h, dl5, gamma_init=10, gamma_max=200, max_it=2000)
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw.copy()
        # LP polish after FW
        sub_dl = min(dl5, time.time() + budget_s * 0.02)
        h_fp, c5_fp = _lp_optimize(best_h, sub_dl, radius_init=0.08, max_it=1500, use_all=True, topk=20)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Phase 6: Quasi-periodic + binary SA at various resolutions
    if time.time() < t0 + budget_s * 0.78:
        dl6 = t0 + budget_s * 0.82
        alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.25]
        for n_qp in [128, 160, 192]:
            if time.time() > dl6:
                break
            for a in alphas[:3]:
                if time.time() > dl6:
                    break
                h_qp = _quasi_periodic(n_qp, a)
                sub_dl = min(dl6, time.time() + budget_s * 0.02)
                h_qo, c5_qo = _lp_optimize(h_qp, sub_dl, radius_init=0.2, max_it=2000, use_all=True, topk=25)
                if c5_qo < best_c5:
                    best_c5 = c5_qo
                    best_h = h_qo.copy()
                    best_n = n_qp
        # Binary SA
        if time.time() < dl6:
            for n_bin in [128, 160]:
                if time.time() > dl6:
                    break
                sub_dl = min(dl6, time.time() + budget_s * 0.02)
                h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=20000)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    best_n = n_bin
                    sub_dl2 = min(dl6, time.time() + budget_s * 0.015)
                    h_br, c5_br = _lp_optimize(h_b, sub_dl2, radius_init=0.1, max_it=1000, use_all=True, topk=15)
                    if c5_br < best_c5:
                        best_c5 = c5_br
                        best_h = h_br.copy()

    # Phase 7: Perturbation restarts
    dl7 = t0 + budget_s * 0.93
    scale = 0.04
    for attempt in range(200):
        if time.time() > dl7:
            break
        sub_dl = min(dl7, time.time() + budget_s * 0.01)
        h_try, c5_try = _perturb_restart(best_h, scale, sub_dl, rng)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 8: Final aggressive LP polish
    if time.time() < t0 + budget_s * 0.97:
        dl8 = t0 + budget_s * 0.985
        h_fin, c5_fin = _lp_optimize(best_h, dl8, radius_init=0.05, max_it=3000, use_all=True, topk=15)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and evaluation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
