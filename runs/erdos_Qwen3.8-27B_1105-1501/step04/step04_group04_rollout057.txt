import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(100):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _lse_grad(h, beta):
    """Compute LSE-smoothed gradient."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    Fm = F - np.max(F)
    Fm = np.clip(Fm, -700.0, 0.0)
    w = np.exp(beta * Fm)
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.zeros(L)
        w[np.argmax(F)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    omh = 1.0 - h
    for r in range(L):
        wr = w[r]
        if wr < 1e-14:
            continue
        k = r - (n - 1)
        low = max(0, -k)
        high = min(n, n - k)
        if low >= high:
            continue
        t1lo = max(0, low + k)
        t1hi = min(n, high + k)
        if t1lo < t1hi:
            grad[t1lo:t1hi] += wr * dx * omh[t1lo-k:t1hi-k]
        grad[low:high] -= wr * dx * h[low+k:high+k]
    return grad

def _grad_descent(h, deadline, beta_start=8.0, beta_end=5000.0, n_betas=25):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for beta in betas:
        if time.time() > deadline:
            break
        step = 0.03
        mom = np.zeros(n)
        stall = 0
        for it in range(8000):
            if time.time() > deadline:
                break
            if stall > 40:
                break
            g = _lse_grad(h, beta)
            g -= np.mean(g)
            mom = 0.85 * mom - step * g
            tri = _project_box_sum(h + mom, target)
            tc = _exact_c5(tri)
            if tc < cur_c5 - 1e-13:
                h = tri
                cur_c5 = tc
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step = min(step * 1.02, 0.15)
            else:
                ok = False
                for f in [0.5, 0.25, 0.1]:
                    tri = _project_box_sum(h + f * mom, target)
                    tc = _exact_c5(tri)
                    if tc < cur_c5 - 1e-13:
                        h = tri
                        cur_c5 = tc
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        step *= f
                        stall = 0
                        ok = True
                        break
                if not ok:
                    stall += 1
                    step *= 0.6
                    mom *= 0.5
    return best_h, best_c5

            break

        # dj = dx * [(1-h[j-k])*I(lo_r <= j-k < hi_r) - h[j+k]*I(lo_r <= j < hi_r)]

        # Need to safely index h

        jpk_clipped = np.clip(jpk, 0, n-1)
        ik_clipped = np.clip(ik, 0, n-1)
        ipk_clipped = np.clip(ipk, 0, n-1)
        dj = dx * ((1.0 - h[jk_clipped]) * jv1 - h[jpk_clipped] * jv2)
        di = dx * ((1.0 - h[ik_clipped]) * iv1 - h[ipk_clipped] * iv2)
        g = dj - di
        # Evaluate C at grid points
        n_grid = 15
        grid = np.linspace(lo, hi, n_grid)
        # C(delta) = max_r (F[r] + delta * g[r])
        # Shape: (n_grid, L)
        vals = F + np.outer(grid, g)
        c_vals = np.max(vals, axis=1)
        best_idx = np.argmin(c_vals)
        if c_vals[best_idx] < cur_c5 - 1e-11:
            # Refine
            if best_idx > 0:
                rlo = grid[max(0, best_idx-1)]
            else:
                rlo = lo
            if best_idx < n_grid - 1:
                rhi = grid[min(n_grid-1, best_idx+1)]
            else:
                rhi = hi
            grid2 = np.linspace(rlo, rhi, 15)
            vals2 = F + np.outer(grid2, g)
            c_vals2 = np.max(vals2, axis=1)
            b2 = np.argmin(c_vals2)
            bd = grid2[b2]
            if c_vals2[b2] < cur_c5 - 1e-11:
                h[i] -= bd
                h[j] += bd
                h = _project_box_sum(h, target)
                cur_c5 = _exact_c5(h)
                F = np.correlate(h, 1.0 - h, mode='full') * dx
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_imp = 0
                continue
        no_imp += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    hn = np.interp(xn, xo, h)
    return _project_box_sum(hn, n_new / 2.0)

def _update_F_after_transfer(F, h, i, j, delta, n):
    """Update correlation scores after transferring delta from h[i] to h[j]."""
    dx = 2.0 / n
    L = 2 * n - 1
    F_new = F.copy()
    for r in range(L):
        k = r - (n - 1)
        low = max(0, -k)
        high = min(n, n - k)
        if low >= high:
            continue
        delta_r = 0.0
        # h[i] decreased by delta:
        # i appears as h[n+k] when n = i-k, valid if low <= i-k < high
        if low <= i - k < high and 0 <= i - k < n:
            delta_r += -delta * (1.0 - h[i - k])  # before the change
        # i appears as h[n] when n = i, valid if low <= i < high
        if low <= i < high and 0 <= i + k < n:
            delta_r += -delta * h[i + k]  # before the change (but h[i+k] might be h[j] if i+k==j)
        # h[j] increased by delta:
        if low <= j - k < high and 0 <= j - k < n:
            delta_r += delta * (1.0 - h[j - k])
        if low <= j < high and 0 <= j + k < n:
            delta_r += -delta * h[j + k]
        F_new[r] += dx * delta_r
    return F_new

F_r = dx * sum_{n=low}^{high-1} h[n+k] * (1-h[n])

After transfer: h[i] -> h[i]-δ, h[j] -> h[j]+δ

The change in F_r:
ΔF_r = dx * [sum over n in [low,high) of (h[n+k]_new * (1-h[n]_new) - h[n+k] * (1-h[n]))]

The old term: h[n+k] * (1-h[n])
The new term: h[n+k]' * (1-h[n])'

where h[x]' = h[x] - δ if x==i, h[x] + δ if x==j, h[x] otherwise.

Let me define:
a_n = h[n+k], b_n = h[n]
a_n' = a_n - δ*I(n+k==i) + δ*I(n+k==j)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(100):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    Fm = F - np.max(F)
    Fm = np.clip(Fm, -700.0, 0.0)
    w = np.exp(beta * Fm)
    ws = float(np.sum(w))
    if ws < 1e-300:
        w = np.zeros(L)
        w[np.argmax(F)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    omh = 1.0 - h
    for r in range(L):
        wr = w[r]
        if wr < 1e-14:
            continue
        k = r - (n - 1)
        low = max(0, -k)
        high = min(n, n - k)
        if low >= high:
            continue
        t1lo = max(0, low + k)
        t1hi = min(n, high + k)
        if t1lo < t1hi:
            grad[t1lo:t1hi] += wr * dx * omh[t1lo - k:t1hi - k]
        grad[low:high] -= wr * dx * h[low + k:high + k]
    return grad

def _topk_grad(h, k=5):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(F)[-k:]
    grad = np.zeros(n)
    omh = 1.0 - h
    for r in top_idx:
        wr = 1.0 / k
        k_lag = int(r) - (n - 1)
        low = max(0, -k_lag)
        high = min(n, n - k_lag)
        if low >= high:
            continue
        t1lo = max(0, low + k_lag)
        t1hi = min(n, high + k_lag)
        if t1lo < t1hi:
            grad[t1lo:t1hi] += wr * dx * omh[t1lo - k_lag:t1hi - k_lag]
        grad[low:high] -= wr * dx * h[low + k_lag:high + k_lag]
    return grad

def _grad_descent(h, deadline, beta_start=8.0, beta_end=5000.0, n_betas=25, use_topk=False):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for beta in betas:
        if time.time() > deadline:
            break
        step = 0.03
        mom = np.zeros(n)
        stall = 0
        for it in range(8000):
            if time.time() > deadline:
                break
            if stall > 40:
                break
            if use_topk:
                g = _topk_grad(h, k=5)
            else:
                g = _lse_grad(h, beta)
            g -= np.mean(g)
            mom = 0.85 * mom - step * g
            tri = _project_box_sum(h + mom, target)
            tc = _exact_c5(tri)
            if tc < cur_c5 - 1e-13:
                h = tri
                cur_c5 = tc
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step = min(step * 1.02, 0.15)
            else:
                ok = False
                for f in [0.5, 0.25, 0.1]:
                    tri = _project_box_sum(h + f * mom, target)
                    tc = _exact_c5(tri)
                    if tc < cur_c5 - 1e-13:
                        h = tri
                        cur_c5 = tc
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        step *= f
                        stall = 0
                        ok = True
                        break
                if not ok:
                    stall += 1
                    step *= 0.6
                    mom *= 0.5
    return best_h, best_c5

def _coord_descent(h, deadline, max_iters=100000, rng=None):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    no_imp = 0
    k_vals = np.arange(-(n - 1), n, dtype=np.float64)
    lo_vals = np.maximum(0, -k_vals)
    hi_vals = np.minimum(n, n - k_vals)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_imp > 3000:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi - lo < 1e-9:
            no_imp += 1
            continue
        jk = j - k_vals
        jpk = j + k_vals
        ik = i - k_vals
        ipk = i + k_vals
        jv1 = (jk >= lo_vals) & (jk < hi_vals) & (jk >= 0) & (jk < n)
        jv2 = (j >= lo_vals) & (j < hi_vals)
        iv1 = (ik >= lo_vals) & (ik < hi_vals) & (ik >= 0) & (ik < n)
        iv2 = (i >= lo_vals) & (i < hi_vals)
        jk_c = np.clip(jk, 0, n - 1).astype(int)
        jpk_c = np.clip(jpk, 0, n - 1).astype(int)
        ik_c = np.clip(ik, 0, n - 1).astype(int)
        ipk_c = np.clip(ipk, 0, n - 1).astype(int)
        dj = dx * ((1.0 - h[jk_c]) * jv1 - h[jpk_c] * jv2)
        di = dx * ((1.0 - h[ik_c]) * iv1 - h[ipk_c] * iv2)
        g = dj - di
        nonz = np.abs(g) > 1e-18
        if not np.any(nonz):
            no_imp += 1
            continue
        ng = 12
        grid = np.linspace(lo, hi, ng)
        F_sub = F[nonz]
        g_sub = g[nonz]
        vals = np.max(F_sub + np.outer(grid, g_sub), axis=1)
        bidx = int(np.argmin(vals))
        if vals[bidx] < cur_c5 - 1e-11:
            if bidx > 0:
                rlo = grid[bidx - 1]
            else:
                rlo = lo
            if bidx < ng - 1:
                rhi = grid[bidx + 1]
            else:
                rhi = hi
            grid2 = np.linspace(rlo, rhi, 12)
            vals2 = np.max(F_sub + np.outer(grid2, g_sub), axis=1)
            b2 = int(np.argmin(vals2))
            bd = grid2[b2]
            if vals2[b2] < cur_c5 - 1e-11:
                h[i] -= bd
                h[j] += bd
                h = _project_box_sum(h, target)
                cur_c5 = _exact_c5(h)
                F = np.correlate(h, 1.0 - h, mode='full') * dx
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_imp = 0
                continue
        no_imp += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    hn = np.interp(xn, xo, h)
    return _project_box_sum(hn, n_new / 2.0)

def _anneal_refine(h, deadline, rng, n_iter=2000, noise_scale=0.02):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    temp = noise_scale
    for it in range(n_iter):
        if time.time() > deadline:
            break
        temp = max(temp * 0.999, 0.001)
        trial = h + rng.normal(0, temp, size=n)
        trial = _project_box_sum(trial, target)
        tc = _exact_c5(trial)
        if tc < cur_c5:
            h = trial
            cur_c5 = tc
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        elif rng.random() < np.exp(-(tc - cur_c5) / (0.005 * max(temp, 0.001))):
            h = trial
            cur_c5 = tc
    return best_h, best_c5
