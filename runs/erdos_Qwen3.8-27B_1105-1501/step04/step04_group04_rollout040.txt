    """
    Solve: minimize max_k (F_active[k] - dx * slopes[k] * delta)

    slopes: array of a_d values for active lags

def _minimax_1d(F, b, dx, delta_min, delta_max):
    """
    Find delta that minimizes max_k (F[k] + dx * b[k] * delta)
    over delta in [delta_min, delta_max].
    
    Returns (best_delta, best_value).
    """
    L = len(F)
    # Lines: y = m_k * delta + c_k where m_k = dx * b[k], c_k = F[k]
    m = dx * b
    c = F
    
    # Sort by slope
    order = np.argsort(m)
    m_sorted = m[order]
    c_sorted = c[order]
    
    # Build upper envelope using stack
    # Each entry in stack is (index, x_intersect_with_previous)
    # x_intersect of line i and line j (m_i < m_j) is: (c_i - c_j) / (m_j - m_i)
    
    stack = []  # list of (line_index, x_start) where x_start is where this line becomes the upper envelope
    
    for idx in range(L):
        mi, ci = m_sorted[idx], c_sorted[idx]
        if not stack:
            stack.append((idx, -np.inf))
            continue
        # Check if new line is below current top at x_start of top
        # If so, new line never appears on envelope
        # Actually, the standard algorithm:
        # While len(stack) >= 2 and intersection(new, stack[-1]) <= x_start of stack[-1]:
        #     pop stack[-1]
        
        while len(stack) >= 2:

The standard algorithm for the upper envelope of lines (also known as the "convex hull of lines" or "Li Chao tree" in 1D):

Given lines sorted by slope (increasing), we maintain a stack. For each new line:
- While the stack has at least 2 lines and the intersection of the new line with the top of the stack is at or before the intersection of the top two stack lines:

- Add the new line to the stack with its intersection x-coordinate

The x-coordinate of the intersection of lines (m1, c1) and (m2, c2) where m1 < m2 is:
x = (c1 - c2) / (m2 - m1)

This is where line 2 (steeper) catches up to line 1 (less steep).
