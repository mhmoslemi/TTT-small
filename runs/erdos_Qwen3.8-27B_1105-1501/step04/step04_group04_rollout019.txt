        diff = target - s

        else:

        if cnt == 0:

        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)

def feasible_interval(h, v):
    """Find [t_lo, t_hi] such that 0 <= h[t] + t*v[t] <= 1 for all t."""
    pos = v > 1e-15
    neg = v < -1e-15
    t_lo = -np.inf
    t_hi = np.inf
    if pos.any():
        t_lo = max(t_lo, float(np.max(-h[pos] / v[pos])))
    if neg.any():
        t_hi = min(t_hi, float(np.max(h[neg] / (-v[neg]))))
    # Clamp to reasonable range
    t_lo = max(t_lo, -10.0)
    t_hi = min(t_hi, 10.0)
    if t_lo >= t_hi:
        return None
    return t_lo, t_hi

def exact_1d_minimize(h, v, n_grid=25, n_refine=4):

    n = len(h)
    fi = feasible_interval(h, v)
    if fi is None:
        return 0.0, exact_c5(h), h
    t_lo, t_hi = fi
    if t_hi - t_lo < 1e-12:
        mid = 0.5 * (t_lo + t_hi)
        h_try = _project_box_sum(h + mid * v, n / 2.0)
        return mid, exact_c5(h_try), h_try

def weighted_active_gradient(h, scores, beta, k_top):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    idx = np.argsort(scores)[::-1][:k_top]
    active = scores[idx]
    shifted = active - active.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    
    g = np.zeros(n)
    for k in range(k_top):
        r = idx[k]
        s = r - (n - 1)
        # (1-h[a+s]) for valid a
        a_lo1 = max(0, -s)
        a_hi1 = min(n, n - s)
        if a_hi1 > a_lo1:
            g[a_lo1:a_hi1] += w[k] * (1.0 - h[a_lo1+s:a_hi1+s])
        # -h[a-s] for valid a
        a_lo2 = max(0, s)
        a_hi2 = min(n, n + s)
        if a_hi2 > a_lo2:
            g[a_lo2:a_hi2] -= w[k] * h[a_lo2-s:a_hi2-s]
    g *= dx
    g -= g.mean()
    max_abs = np.max(np.abs(g))
    if max_abs > 1e-15:
        g /= max_abs

    if fi is None:
        return 0.0, c0, h
    t_lo, t_hi = fi
    
    if t_hi - t_lo < 1e-10:
        t = 0.5 * (t_lo + t_hi)
        h_new = project_box_sum(h + t * v, n / 2.0)
        return t, exact_c5(h_new), h_new
    
    # Grid search
    n_grid = 21
    ts = np.linspace(t_lo, t_hi, n_grid)
    best_t = 0.0
    best_c = c0
    
    for t in ts:
        h_try = h + t * v
        c_try = exact_c5(h_try)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_t = t
    
    if best_t == 0.0:
        return 0.0, c0, h
    
    # Refine: ternary search in bracket
    step = (t_hi - t_lo) / (n_grid - 1)
    lo = max(t_lo, best_t - 1.5 * step)
    hi = min(t_hi, best_t + 1.5 * step)
    
    for _ in range(30):
        if hi - lo < 1e-10:
