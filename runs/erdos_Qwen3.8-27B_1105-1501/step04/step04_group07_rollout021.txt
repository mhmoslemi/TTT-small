                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _line_search(h, p, max_alpha=0.5):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_f = _exact_c5(h + opt_a * p)
    # Also check 0
    f0 = _exact_c5(h)
    if f0 < opt_f:
        return 0.0, f0
    return opt_a, opt_f

def _lp_refine(h, deadline, top_k=None, radius=0.05, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.5)
        else:
            stall += 1
            r *= 0.8
    return best_h, best_c5

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 60:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.15, 0.1, 0.05, 0.02, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 100:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-14:
            stall += 1
            r *= 0.7
            if r < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.04, 0.5)
        else:
            stall += 1
            r *= 0.82
            if stall > 40 and r < 0.015:
                r = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall_global = 0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                continue
            for k in range(j + 1, n):
                if time.time() > deadline:
                    break
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-11:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(14):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-13:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            stall_global += 1
            if stall_global > 2:
                break
        else:
            stall_global = 0
    return best_h, best_c5

def _softmax_descent(h, deadline, mu_start=0.05, mu_end=0.003, lr=0.03, n_steps=600):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        mu = mu_start * (mu_end / mu_start) ** frac
        scores = _scores(h)
        s_shifted = (scores - scores.max()) / mu
        w = np.exp(s_shifted)
        w /= w.sum()
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        t_adam += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        found = False
        for scale in [1.0, 0.5, 0.25, 0.1]:
            h_try = h - scale * update
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = c5_try
                stall = 0
                found = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not found:
            stall += 1
            if stall > 80:
                break
    return best_h, best_c5

def _active_set_method(h, deadline, thresh_frac=0.99, n_steps=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        s_max = float(scores.max())
        thresh = s_max * thresh_frac
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            break
        weights = np.zeros(n_lags)
        gaps = s_max - scores[active]
        w_sum = float(gaps.sum())
        if w_sum < 1e-15:
            weights[active] = 1.0 / len(active)
        else:
            weights[active] = gaps / w_sum
        G = _grad_matrix(h)
        grad = G.T @ weights
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 50:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.1, 0.05, 0.02, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-15:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
                found = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not found:
            stall += 1
            thresh_frac = min(thresh_frac + 0.005, 0.999)
            if stall > 80:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Coarse-to-fine (8%)
    dl1 = t0 + budget_s * 0.08
    for n_try in [16, 32]:
        if time.time() > dl1:
            break
        h_c = _resample(best_h, n_try)
        sub_dl = min(dl1, time.time() + budget_s * 0.02)
        h_c, c5_c = _pnorm_descent(h_c, sub_dl, p_start=4.0, p_end=1024.0, n_steps=300)
        if c5_c < best_c5 * (n_try / n0) * (n0 / n_try):
            pass
        h_up = _resample(h_c, n0)
        if _exact_c5(h_up) < best_c5:
            best_c5 = _exact_c5(h_up)
            best_h = h_up.copy()

    # Phase 2: P-norm descent main (22%)
    dl2 = t0 + budget_s * 0.30
    hdr, c5r = _pnorm_descent(best_h, dl2, p_start=4.0, p_end=8192.0, n_steps=2500)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 3: Softmax descent (12%)
    dl3 = t0 + budget_s * 0.42
    hdr, c5r = _softmax_descent(best_h, dl3, mu_start=0.04, mu_end=0.002, lr=0.025, n_steps=500)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 4: LP refinement (15%)
    dl4 = t0 + budget_s * 0.57
    hdr, c5r = _lp_refine(best_h, dl4, top_k=10, radius=0.06, max_iters=4000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 5: Active set method (10%)
    dl5 = t0 + budget_s * 0.67
    hdr, c5r = _active_set_method(best_h, dl5, thresh_frac=0.995, n_steps=300)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 6: High-p P-norm polish (10%)
    dl6 = t0 + budget_s * 0.77
    hdr, c5r = _pnorm_descent(best_h, dl6, p_start=512.0, p_end=262144.0, n_steps=600)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 7: Multi-start perturbation (8%)
    dl7 = t0 + budget_s * 0.85
    scale = 0.015
    for attempt in range(400):
        if time.time() > dl7:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.012)
        h_try, c5_try = _lp_refine(h_pert, sub_dl, top_k=8, radius=0.05, max_iters=150)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.04)

    # Phase 8: Resolution sweep (5%)
    dl8 = t0 + budget_s * 0.90
    for dn in [-4, -2, -1, 1, 2, 4, 6, 8, 12, 16]:
        if time.time() > dl8:
            break
        n_try = best_n + dn
        if n_try < 16 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl8, time.time() + budget_s * 0.008)
        h_r, c5_r = _lp_refine(h_r, sub_dl, top_k=8, radius=0.08, max_iters=100)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 9: Ultra-fine LP (5%)
    dl9 = t0 + budget_s * 0.95
    hdr, c5r = _lp_refine(best_h, dl9, top_k=4, radius=0.008, max_iters=6000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 10: Pair polish (5%)
    if time.time() < t0 + budget_s * 0.995:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.998, max_passes=5)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
