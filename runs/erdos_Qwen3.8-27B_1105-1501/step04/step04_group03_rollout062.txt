def compute_a_coeffs(h, i):
    """Compute the linear coefficients a_d for each shift d when varying h[i]."""
    n = len(h)
    a = np.zeros(2*n-1)
    for d in range(-(n-1), n):
        if d == 0:
            continue
        k = d + n - 1
        if d > 0:
            coef = 0.0
            if d <= i:  # i-d >= 0
                coef += (1.0 - h[i-d])
            if i + d < n:  # i+d valid
                coef -= h[i+d]
            a[k] = coef
        else:
            g = -d
            coef = 0.0
            if i + g < n:  # i = j-g, j = i+g valid
                coef += (1.0 - h[i+g])
            if i - g >= 0:  # i >= g, so i-g >= 0
                coef -= h[i-g]
            a[k] = coef
    return a

def min_1d_score(scores, a, h_i):
    """Find t in [0,1] minimizing max_d (score_d + (t-h_i)*a_d) with d=0 handled quadratically."""
    n = len(scores) // 2  # n-1... actually scores has 2n-1 elements
    # scores[k] for k=0..2n-2, d = k-(n-1)
    # a[k] is the coefficient for shift d=k-(n-1)
    # For d=0 (k=n-1): quadratic term
    
    def g(t):
        s = t - h_i
        # Build the array of scores at t
        vals = scores + s * a
        # Override d=0 with quadratic
        vals[n-1] = scores[n-1] + s*(1.0 - 2.0*h_i) - s*s
        return np.max(vals)
    
    # Golden section
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1, f2 = g(c1), g(c2)
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 <= f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr*(hi-lo)
            f1 = g(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr*(hi-lo)
            f2 = g(c2)
    result = (lo + hi) / 2.0
    # Check endpoints
    if g(0.0) < g(result): result = 0.0
    if g(1.0) < g(result): result = 1.0
    return result

def compute_a_coeffs(h, i):
