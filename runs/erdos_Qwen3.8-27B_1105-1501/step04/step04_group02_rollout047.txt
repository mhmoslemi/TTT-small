    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _active_set_loop(h0, deadline, max_rounds=10, top_k=15):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for round in range(max_rounds):

            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        T = float(np.max(scores))
        # Find top K active constraints
        idx = np.argsort(scores)[::-1][:top_k]
        active_idx = tuple(idx)
        
        h_new, c5_new = _slsqp_solve(h, n, active_idx, maxiter=50, ftol=1e-11)
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Check if all constraints are satisfied
        scores = np.correlate(h, 1.0 - h, mode='full')
        T_sol = float(np.max(scores))
        # If the solver reduced T below the current max, we made progress
        if T_sol < T - 1e-6:

        else:
            # Check convergence
            if T_sol < T - 1e-8:
                continue
            else:
                break
