def _solve_1d_convex(a, b, lo, hi):
    """Solve min_{x in [lo,hi]} max_i (a[i]*x + b[i])
    Returns (x_opt, f_opt)
    """
    # Binary search on t = optimal value
    # For given t, feasible iff:
    #   exists x in [lo,hi] such that a[i]*x + b[i] <= t for all i
    #   => for a[i] > 0: x <= (t - b[i]) / a[i]
    #   => for a[i] < 0: x >= (t - b[i]) / a[i]
    #   => for a[i] == 0: b[i] <= t
    
    # Lower bound on t: max_i b[i] (at x=0... wait no)
    # Actually, t >= max_i (a[i]*lo + b[i]) and t >= max_i (a[i]*hi + b[i])
    # Hmm, not quite. t >= max of the function at the boundaries AND at interior points.
    
    # Let's just binary search.
    # t_low = max of (a[i]*lo + b[i]) and (a[i]*hi + b[i]) ... no.
    # t_low = min possible value of max_i (a[i]*x + b[i])
    # A lower bound: for any x, max_i (a[i]*x + b[i]) >= max_i b[i] (not tight)
    # Better lower bound: the dual problem. But let's just use a reasonable range.
    
    f_lo = float(np.max(a * lo + b))
    f_hi = float(np.max(a * hi + b))
    
    # The optimal value is at most min(f_lo, f_hi)
    # and at least... let's think. 
    # The minimum of max_i (a_i x + b_i) over x is the value where the 
    # upper envelope has its minimum.
    
    # Lower bound: for any dual variables (weights) w_i >= 0, sum(w_i) = 1:
    #   max_i (a_i x + b_i) >= sum_i w_i (a_i x + b_i) = (sum w_i a_i) x + sum w_i b_i
    #   min_x of RHS = sum w_i b_i (if sum w_i a_i = 0) or at boundary
    # This is the LP dual. But computing it is expensive.
    
    # Simpler: just use f_lo and f_hi as upper bounds, and 0 as lower bound.
    t_hi = min(f_lo, f_hi)
    t_lo = float(np.min(b))  # rough lower bound (could be improved)
    
    # Actually, a better lower bound:
    # At any x, max_i (a_i x + b_i) >= (sum_i (a_i x + b_i)) / len(a) (by AM>=GM... no, that's not right)
    # Actually max >= mean, so:
    t_lo = float(np.mean(b) + 0)  # still rough
    
    # Let me just use a wider range and binary search
    t_lo = min(float(np.min(a * lo + b)), float(np.min(a * hi + b)))
    t_hi = min(f_lo, f_hi)
    
    if t_hi - t_lo < 1e-14:
        # Check which boundary is better
        if f_lo <= f_hi:
            return lo, f_lo
        else:
            return hi, f_hi
    
    # Binary search on t
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2.0
        # Check feasibility: exists x in [lo, hi] with a_i x + b_i <= t_mid for all i
        # For a_i > 0: x <= (t_mid - b_i) / a_i
        # For a_i < 0: x >= (t_mid - b_i) / a_i
        # For a_i == 0: need b_i <= t_mid
        
        pos_mask = a > 1e-15
        neg_mask = a < -1e-15
        zero_mask = ~pos_mask & ~neg_mask
        
        feasible = True
        if np.any(zero_mask) and float(np.max(b[zero_mask])) > t_mid:
            feasible = False
        else:
            x_upper = hi
            if np.any(pos_mask):
                x_upper = min(x_upper, float(np.min((t_mid - b[pos_mask]) / a[pos_mask])))
            x_lower = lo
            if np.any(neg_mask):
                x_lower = max(x_lower, float(np.max((t_mid - b[neg_mask]) / a[neg_mask])))
            if x_lower > x_upper + 1e-15:
                feasible = False
        
        if feasible:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    t_opt = (t_lo + t_hi) / 2.0
    
    # Now find x_opt: the x that achieves value t_opt
    # x must satisfy: a_i x + b_i <= t_opt for all i
    # x_upper = min(hi, min_{a_i>0} (t_opt - b_i) / a_i)
    # x_lower = max(lo, max_{a_i<0} (t_opt - b_i) / a_i)
    # Any x in [x_lower, x_upper] works. Pick the midpoint.
    
    pos_mask = a > 1e-15
    neg_mask = a < -1e-15
    
    x_upper = hi
    if np.any(pos_mask):
        x_upper = min(x_upper, float(np.min((t_opt - b[pos_mask]) / a[pos_mask])))
    x_lower = lo
    if np.any(neg_mask):
        x_lower = max(x_lower, float(np.max((t_opt - b[neg_mask]) / a[neg_mask])))
    
    x_opt = (x_lower + x_upper) / 2.0
    x_opt = max(lo, min(hi, x_opt))
    
    return x_opt, t_opt

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _solve_1d(a, b, lo, hi):
    """Solve min_{x in [lo,hi]} max_i (a[i]*x + b[i]). Returns x_opt."""
    if hi - lo < 1e-14:
        return (lo + hi) / 2.0
    f_lo = float(np.max(a * lo + b))
    f_hi = float(np.max(a * hi + b))
    t_upper = min(f_lo, f_hi)
    avg_a = float(np.mean(a))
    avg_b = float(np.mean(b))
    if avg_a > 0:
        t_lower = avg_a * lo + avg_b
    elif avg_a < 0:
        t_lower = avg_a * hi + avg_b
    else:
        t_lower = avg_b
    t_lower = min(t_lower, float(np.min(b)) - abs(avg_a) * max(abs(lo), abs(hi)))
    if t_upper - t_lower < 1e-15:
        return lo if f_lo <= f_hi else hi
    t_lo = max(t_lower, -1e6)
    t_hi = t_upper
    pos_mask = a > 1e-14
    neg_mask = a < -1e-14
    any_pos = np.any(pos_mask)
    any_neg = np.any(neg_mask)
    any_zero = np.any(~(pos_mask | neg_mask))
    for _ in range(55):
        t_mid = (t_lo + t_hi) / 2.0
        feasible = True
        x_ub = hi
        if any_pos:
            vals = (t_mid - b[pos_mask]) / a[pos_mask]
            x_ub = min(x_ub, float(np.min(vals)))
        x_lb = lo
        if any_neg:
            vals = (t_mid - b[neg_mask]) / a[neg_mask]
            x_lb = max(x_lb, float(np.max(vals)))
        if any_zero:
            zm = ~(pos_mask | neg_mask)
            if float(np.max(b[zm])) > t_mid:
                feasible = False
        if not feasible:
            t_lo = t_mid
            continue
        if x_lb > x_ub + 1e-14:
            t_lo = t_mid
        else:
            t_hi = t_mid
    t_opt = (t_lo + t_hi) / 2.0
    x_ub = hi
    if any_pos:
        vals = (t_opt - b[pos_mask]) / a[pos_mask]
        x_ub = min(x_ub, float(np.min(vals)))
    x_lb = lo
    if any_neg:
        vals = (t_opt - b[neg_mask]) / a[neg_mask]
        x_lb = max(x_lb, float(np.max(vals)))
    if x_lb > x_ub:
        return lo if f_lo <= f_hi else hi
    x_opt = (x_lb + x_ub) / 2.0
    return max(lo, min(hi, x_opt))

def _coord_descent(h, deadline, mode='nearest_half', max_passes=50000):
    """Pair coordinate descent with exact 1D subproblem solving."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    n_lags = 2 * n - 1
    dx = 2.0 / n
    d_arr = np.arange(n_lags) - (n - 1)
    stall = 0
    order = list(range(n))
    for p in range(max_passes):
        if time.time() > deadline:
            break
        if p % 50 == 0 and p > 0:
            # Shuffle order periodically
            np.random.shuffle(order)
        for j in order:
            if time.time() > deadline:
                break
            if mode == 'nearest_half':
                k = int(np.argmin(np.abs(h - 0.5)))
            elif mode == 'round_robin':
                k = (j + 1) % n
            else:
                k = (j * 7 + 3) % n
            if j == k:
                k = (j + 1) % n
            S = h[j] + h[k]
            x_lo = max(0.0, S - 1.0)
            x_hi = min(1.0, S)
            if x_hi - x_lo < 1e-13:
                continue
            jd = j - d_arr
            jp = j + d_arr
            kd = k - d_arr
            kp = k + d_arr
            vjd = (jd >= 0) & (jd < n)
            vjp = (jp >= 0) & (jp < n)
            vkd = (kd >= 0) & (kd < n)
            vkp = (kp >= 0) & (kp < n)
            a_j = np.zeros(n_lags)
            if np.any(vjd):
                a_j[vjd] = (1.0 - h[jd[vjd]]) * dx
            if np.any(vjp):
                a_j[vjp] += (-h[jp[vjp]]) * dx
            a_k = np.zeros(n_lags)
            if np.any(vkd):
                a_k[vkd] = (1.0 - h[kd[vkd]]) * dx
            if np.any(vkp):
                a_k[vkp] += (-h[kp[vkp]]) * dx
            A = a_j - a_k
            corr = np.correlate(h, 1.0 - h, mode='full')
            B = corr.astype(float) - A * h[j]
            x_opt = _solve_1d(A, B, x_lo, x_hi)
            if abs(x_opt - h[j]) > 1e-13:
                h[j] = x_opt
                h[k] = S - x_opt
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        if abs(s - target) > 1e-10:
            h = _project(h, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5 - 1e-15:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best_h, best_c5

def _lp_optimize(h, deadline, top_k=None, radius=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        d_arr = np.arange(n_lags) - (n - 1)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            d_sel = d_arr[idx]
        else:
            sk = scores
            d_sel = d_arr
        G = np.zeros((len(sk), n))
        for li, d in enumerate(d_sel):
            dd = int(d)
            jd = np.arange(n) - dd
            jp = np.arange(n) + dd
            vjd = (jd >= 0) & (jd < n)
            vjp = (jp >= 0) & (jp < n)
            col = np.zeros(n)
            if np.any(vjd):
                col[jd[vjd]] += (1.0 - h[np.arange(n)[vjd]])
            if np.any(vjp):
                col[np.arange(n)[vjp]] += (-h[jp[vjp]])
            G[li] = col * dx
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.8
            if r < 1e-7:
                break
            continue
        # Line search
        h_try = h + p
        h_try = _project(h_try, target)
        new_c5 = _exact_c5(h_try)
        if new_c5 < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.5)
        else:
            # Try half step
            h_try2 = h + 0.5 * p
            h_try2 = _project(h_try2, target)
            new_c5_2 = _exact_c5(h_try2)
            if new_c5_2 < cur_c5 - 1e-14:
                h = h_try2
                cur_c5 = new_c5_2
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                r *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=3):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.995, max_passes=2)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
