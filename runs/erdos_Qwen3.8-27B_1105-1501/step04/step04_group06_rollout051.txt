def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-10:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-10:
                    break
    return x

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_jac(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    J = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        row = np.zeros(n)
        row[m1] += (1.0 - h[gs[m1]])
        row[m2] -= h[ps[m2]]
        J[r, :] = row
    return corr, J

def _s_eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5

def _s_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g

def _s_obj_f(z, n):
    return z[n]

def _s_obj_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _s_ineq_f(z, n, idx):
    hh = z[:n]
    tt = z[n]
    cc = np.correlate(hh, 1.0 - hh, mode='full')
    return tt - cc[idx]

def _s_ineq_g(z, n, idx):
    hh = z[:n]
    _, J = _corr_jac(hh, n)
    Jf = np.zeros((len(idx), n + 1))
    Jf[:, :n] = J[idx, :]
    Jf[:, n] = 1.0
    return Jf

def _run_slsqp(h, n, K, maxiter=200, ftol=1e-14):
    h = np.asarray(h, dtype=float).copy()
    target = n * 0.5
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:K]
    t0 = float(np.max(corr)) + 1e-5
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _s_eq_f, 'jac': _s_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _s_ineq_f, 'jac': _s_ineq_g, 'args': (n, top_idx)}
    ]
    try:
        res = minimize(_s_obj_f, z0, jac=_s_obj_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], target)
        return x, _exact_c5(x), res
    except Exception:
        return h.copy(), _exact_c5(h), None

        cc = np.correlate(hh, 1.0 - hh, mode='full')
        ms = float(np.max(cc))
        vals = np.clip(a * (cc - ms), -500.0, 500.0)
        e = np.exp(vals)
        se = float(np.sum(e))
        f = ms + (1.0 / a) * np.log(se)
        w = e / se
        j = np.arange(nn)
        grad = np.zeros(nn)
        for r in range(2 * nn - 1):
            s = r - (nn - 1)
            gs = j - s
            m1 = (gs >= 0) & (gs < nn)
            ps = j + s
            m2 = (ps >= 0) & (ps < nn)
            pd = np.zeros(nn)
            pd[m1] = 1.0 - hh[gs[m1]]
            pd[m2] -= hh[ps[m2]]
            grad += w[r] * pd
        grad -= np.mean(grad)
        return f, grad

        row[m2] -= h[ps[m2]]

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

        h_r, c5_r = _active_set_slsqp(h_r, n_try, sub_deadline2,
                                       K_start=10, K_max=30, max_rounds=25)
        sub_deadline3 = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, n_try, sub_deadline3,
                             radius_init=0.08, max_iters=1000, stall_limit=25)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final polish (5% budget)
    d7 = t0 + budget_s * 0.99
    h_f, c5_f = _run_slsqp(best_h, best_n, K=25, maxiter=200, ftol=1e-15)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    h_lp2, c5_lp2 = _lp_loop(best_h, best_n, d7, radius_init=0.04,
                             max_iters=2000, stall_limit=50)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
