    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_t_optimal(h, i, j, scores, n_eval=25):
    """Find optimal t for pair (i,j) given current scores."""
    n = len(h)
    dx = 2.0 / n
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi - 1e-15:
        return 0.0
    
    # Compute linear and quad coefficients
    L = np.zeros(2*n-1)
    Q = np.zeros(2*n-1)
    for d in range(-(n-1), n):
        r = n - 1 + d
        # Find affected positions
        affected = []
        if 0 <= i < n and 0 <= i+d < n:
            affected.append(i)
        if 0 <= j < n and 0 <= j+d < n:
            if j not in affected:  # avoid duplicate
                affected.append(j)
        nd = i - d
        if 0 <= nd < n and nd not in affected:
            # Check: is this position valid? Need 0<=nd<n and 0<=nd+d<n = 0<=i<n
            if 0 <= i < n:
                affected.append(nd)
        nd = j - d
        if 0 <= nd < n and nd not in affected:
            if 0 <= j < n:
                affected.append(nd)
        
        lin = 0.0
        quad = 0.0
        for nn in affected:
            nn_d = nn + d
            # Verify nn and nn_d are both valid
            if not (0 <= nn < n and 0 <= nn_d < n):
                continue
            d1 = 1.0 if nn == i else (-1.0 if nn == j else 0.0)
            d2 = 1.0 if nn_d == i else (-1.0 if nn_d == j else 0.0)
            lin += d1 * (1.0 - h[nn_d]) - h[nn] * d2
            quad += -d1 * d2
        L[r] = lin * dx
        Q[r] = quad * dx
    
    # Now minimize max_d (scores[d] + t*L[d] + t²*Q[d]) over t in [t_lo, t_hi]
    # Use sampling + golden section
    n_s = n_eval
    ts = np.linspace(t_lo, t_hi, n_s)
    # Evaluate f(t) = max_d(scores + t*L + t²*Q)
    best_t = t_lo
    best_f = -np.inf
    for t in ts:
        vals = scores + t * L + t*t * Q
        f = float(np.max(vals))
        if f < best_f:
            best_f = f
            best_t = t
    
    # Golden section refinement around best_t
    half = (t_hi - t_lo) / n_s
    lo = max(t_lo, best_t - half)
    hi = min(t_hi, best_t + half)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(scores + c1 * L + c1*c1 * Q))
    f2 = float(np.max(scores + c2 * L + c2*c2 * Q))
    for _ in range(30):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(scores + c1 * L + c1*c1 * Q))
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(scores + c2 * L + c2*c2 * Q))
    
    t_opt = (lo + hi) / 2.0
    return t_opt

def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    if abs(res) > 1e-12:

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx
