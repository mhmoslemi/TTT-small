    """Minimize max_l (scores[l] + D[l]*t) for t in [t_min, t_max]."""
    for _ in range(100):
        t_mid = (t_min + t_max) / 2.0
        vals = scores + D * t_mid
        g_val = np.max(vals)
        # Find active set (within tolerance)
        active = np.abs(vals - g_val) < 1e-10 * max(1.0, abs(g_val))

        else:

from scipy.optimize import linprog
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_1d(v, low, high, target):
    """Project v onto {x: low <= x <= high, sum(x) = target}."""
    n = len(v)
    if target <= n * low:
        return np.full(n, low)
    if target >= n * high:
        return np.full(n, high)
    lo, hi = low, high
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, low, high))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, low, high)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < high - 1e-12
        else:
            mask = x > low + 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    return np.clip(x, low, high)

def project(v):
    n = len(v)
    return project_1d(v, 0.0, 1.0, n / 2.0)

def solve_pair(h, scores, j, k):
    """Solve 1D subproblem for pair (j, k). Returns (a_opt, b_opt, improved_c5)."""
    n = len(h)
    dx = 2.0 / n
    S = h[j] + h[k]
    a_min = max(0.0, S - 1.0)
    a_max = min(1.0, S)
    if a_max - a_min < 1e-14:
        return None
    
    # Compute D for all lags
    D = np.zeros(2 * n - 1)
    for l in range(2 * n - 1):
        d = l - (n - 1)
        val = 0.0
        jp = j + d
        jm = j - d
        kp = k + d
        km = k - d
        if 0 <= jp < n:
            val += (1.0 - h[jp])
        if 0 <= jm < n:
            val -= h[jm]
        if 0 <= kp < n:
            val -= (1.0 - h[kp])
        if 0 <= km < n:
            val += h[km]
        D[l] = dx * val
    
    # Minimize max_l (scores[l] + D[l] * t) for t in [t_min, t_max]
    t_min = a_min - h[j]
    t_max = a_max - h[j]
    
    # Binary search on subgradient
    for _ in range(80):
        t_mid = (t_min + t_max) / 2.0
        vals = scores + D * t_mid
        g_val = np.max(vals)
        tol = 1e-10 * max(1.0, abs(g_val))
        active = np.abs(vals - g_val) <= tol
        if not np.any(active):
            active = np.zeros(len(vals), dtype=bool)
            active[np.argmax(vals)] = True
        min_D = np.min(D[active])
        max_D = np.max(D[active])
        if min_D > 1e-12:
            t_max = t_mid
        elif max_D < -1e-12:
            t_min = t_mid
        else:
            break
        if t_max - t_min < 1e-15:
            break
    
    t_opt = (t_min + t_max) / 2.0
    a_opt = h[j] + t_opt
    b_opt = S - a_opt
    
    # Validate
    if a_opt < a_min - 1e-12 or a_opt > a_max + 1e-12:
        return None
    
    # Evaluate new C5
    h_new = h.copy()
    h_new[j] = a_opt
    h_new[k] = b_opt
    new_scores = compute_scores(h_new)
    new_c5 = float(np.max(new_scores))
    
    return a_opt, b_opt, new_c5

def small_lp(h, scores, idx_set, c5_current, max_time=0.5):
    """Solve small LP for variables at indices in idx_set."""
    n = len(h)
    m = len(idx_set)
    if m < 2 or m > 10:
        return None
    
    S = np.sum(h[idx_set])
    
    # For each lag l, c_l as a function of h[idx_set]:
    # c_l = base_l + sum_{j in idx_set} coeff[l, j] * h[j]
    # where coeff[l, j] = dx * [(1-h[j+d])_valid - h[j-d]_valid]
    # (same formula as D but for a single variable)
    
    # Build coefficient matrix
    coeffs = np.zeros((2 * n - 1, m))
    for j_local, j in enumerate(idx_set):
        for l in range(2 * n - 1):
            d = l - (n - 1)
            val = 0.0
            jp = j + d
            jm = j - d
            if 0 <= jp < n:
                val += (1.0 - h[jp])
            if 0 <= jm < n:
                val -= h[jm]
            coeffs[l, j_local] = (2.0 / n) * val
    
    # base_l = scores[l] - sum_{j in idx_set} coeffs[l, j] * h[j]
    base = scores - coeffs @ h[idx_set]
    
    # LP: minimize t
    # s.t. base_l + coeffs[l] @ x <= t  for all l
    #      sum(x) = S
    #      0 <= x_j <= 1
    
    # Variables: x (m vars) + t (1 var) = m+1 vars
    # Objective: minimize t → c = [0, ..., 0, 1]
    
    from scipy.optimize import linprog
    c = np.zeros(m + 1)
    c[-1] = 1.0
    
    # Inequality constraints: base_l + coeffs[l] @ x - t <= 0
    # A_ub @ [x; t] <= b_ub
    A_ub = np.zeros((2 * n - 1, m + 1))
    A_ub[:, :m] = coeffs
    A_ub[:, m] = -1.0
    b_ub = -base
    
    # Equality: sum(x) = S
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0.0, 1.0)] * m + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time})
        if res.success:
            x_opt = res.x[:m]
            # Reconstruct h
            h_new = h.copy()
            h_new[idx_set] = x_opt
            new_c5 = exact_c5(h_new)
            return h_new, new_c5
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project(v):
    n = len(v)
    target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    return np.clip(x, 0.0, 1.0)

def solve_pair_1d(h, scores, j, k):
    n = len(h)
    dx = 2.0 / n
    S = h[j] + h[k]
    a_min = max(0.0, S - 1.0)
    a_max = min(1.0, S)
    if a_max - a_min < 1e-14:
        return None
    D = np.zeros(2 * n - 1)
    for l in range(2 * n - 1):
        d = l - (n - 1)
        val = 0.0
        jp = j + d
        jm = j - d
        kp = k + d
        km = k - d
        if 0 <= jp < n:
            val += (1.0 - h[jp])
        if 0 <= jm < n:
            val -= h[jm]
        if 0 <= kp < n:
            val -= (1.0 - h[kp])
        if 0 <= km < n:
            val += h[km]
        D[l] = dx * val
    t_min = a_min - h[j]
    t_max = a_max - h[j]
    for _ in range(100):
        t_mid = (t_min + t_max) / 2.0
        vals = scores + D * t_mid
        g_val = np.max(vals)
        tol = 1e-9 * max(1.0, abs(g_val))
        active = np.abs(vals - g_val) <= tol
        if not np.any(active):
            active = np.zeros(len(vals), dtype=bool)
            active[np.argmax(vals)] = True
        min_D = np.min(D[active])
        max_D = np.max(D[active])
        if min_D > 1e-11:
            t_max = t_mid
        elif max_D < -1e-11:
            t_min = t_mid
        else:
            break
        if t_max - t_min < 1e-15:
            break
    t_opt = (t_min + t_max) / 2.0
    a_opt = h[j] + t_opt
    b_opt = S - a_opt
    a_opt = np.clip(a_opt, a_min, a_max)
    b_opt = S - a_opt
    h_new = h.copy()
    h_new[j] = a_opt
    h_new[k] = b_opt
    new_c5 = exact_c5(h_new)
    return a_opt, b_opt, new_c5

def solve_small_lp(h, scores, idx_arr, max_time=0.3):
    n = len(h)
    m = len(idx_arr)
    if m < 2 or m > 8:
        return None
    S = np.sum(h[idx_arr])
    coeffs = np.zeros((2 * n - 1, m))
    for jl, j in enumerate(idx_arr):
        for l in range(2 * n - 1):
            d = l - (n - 1)
            val = 0.0
            jp = j + d
            jm = j - d
            if 0 <= jp < n:
                val += (1.0 - h[jp])
            if 0 <= jm < n:
                val -= h[jm]
            coeffs[l, jl] = (2.0 / n) * val
    base = scores - coeffs @ h[idx_arr]
    c = np.zeros(m + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, m + 1))
    A_ub[:, :m] = coeffs
    A_ub[:, m] = -1.0
    b_ub = -base
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([S])
    bounds = [(0.0, 1.0)] * m + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': max_time})
        if res.success:
            x_opt = np.clip(res.x[:m], 0.0, 1.0)
            h_new = h.copy()
            h_new[idx_arr] = x_opt
            h_new = project(h_new)
            new_c5 = exact_c5(h_new)
            return h_new, new_c5
    except Exception:
        pass
    return None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new)

def phase_rapid_pairs(h, best_h, best_c5, rng, deadline):
    n = len(h)
    scores = compute_scores(h)
    cur_c5 = float(np.max(scores))
    stall = 0
    iters = 0
    while time.time() < deadline:
        iters += 1
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        result = solve_pair_1d(h, scores, j, k)
        if result is None:
            continue
        a_opt, b_opt, new_c5 = result
        if new_c5 < cur_c5 - 1e-13:
            h[j] = a_opt
            h[k] = b_opt
            h = project(h)
            scores = compute_scores(h)
            cur_c5 = float(np.max(scores))
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000 and iters % 10000 == 0:
            h = project(h + rng.normal(0, 0.01, n))
            scores = compute_scores(h)
            cur_c5 = float(np.max(scores))
            stall = 0
    return h, best_h, best_c5

def phase_directed_pairs(h, best_h, best_c5, rng, deadline):
    n = len(h)
    scores = compute_scores(h)
    cur_c5 = float(np.max(scores))
    batch_size = 30
    while time.time() < deadline:
        candidates = []
        for _ in range(batch_size):
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if j == k:
                continue
            result = solve_pair_1d(h, scores, j, k)
            if result is not None:
                candidates.append((result[2], result[0], result[1], j, k))
        if not candidates:
            break
        candidates.sort(key=lambda x: x[0])
        best_new_c5 = candidates[0][0]
        if best_new_c5 < cur_c5 - 1e-12:
            _, a_opt, b_opt, j, k = candidates[0]
            h[j] = a_opt
            h[k] = b_opt
            h = project(h)
            scores = compute_scores(h)
            cur_c5 = float(np.max(scores))
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return h, best_h, best_c5

def phase_small_lp(h, best_h, best_c5, rng, deadline):
    n = len(h)
    scores = compute_scores(h)
    cur_c5 = float(np.max(scores))
    sizes = [3, 4, 5, 6]
    while time.time() < deadline:
        m = int(rng.choice(sizes))
        subset = rng.choice(n, size=m, replace=False)
        subset = np.sort(subset)
        result = solve_small_lp(h, scores, subset, max_time=0.2)
        if result is not None:
            h_new, new_c5 = result
            if new_c5 < cur_c5 - 1e-12:
                h = h_new
                scores = compute_scores(h)
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if time.time() > deadline:
            break
    return h, best_h, best_c5

def phase_final(h, best_h, best_c5, rng, deadline):
    n = len(h)
    # More 1D pair optimization with smaller perturbations for diversification
    scores = compute_scores(h)
    cur_c5 = float(np.max(scores))
    # Try multi-resolution
    tried = {n}
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -30, 30, -40, 40]
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n + dn
        if n_try < 40 or n_try > 200 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample_h(h, n_try)
        # Quick optimization at new resolution
        r_scores = compute_scores(h_r)
        r_c5 = float(np.max(r_scores))
        # A few rounds of 1D pairs
        for _ in range(2000):
            if time.time() > deadline:
                break
            j = int(rng.integers(0, n_try))
            k = int(rng.integers(0, n_try))
            if j == k:
                continue
            result = solve_pair_1d(h_r, r_scores, j, k)
            if result is not None and result[2] < r_c5 - 1e-13:
                h_r[j] = result[0]
                h_r[k] = result[1]
                h_r = project(h_r)
                r_scores = compute_scores(h_r)
                r_c5 = float(np.max(r_scores))
        if r_c5 < best_c5 - 1e-12:
            best_c5 = r_c5
            best_h = h_r.copy()
            n = n_try
            h = h_r.copy()
    # Final rapid pairs on best
    h = project(best_h.copy())
    n = len(h)
    scores = compute_scores(h)
    cur_c5 = float(np.max(scores))
    while time.time() < deadline:
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        result = solve_pair_1d(h, scores, j, k)
        if result is not None and result[2] < cur_c5 - 1e-13:
            h[j] = result[0]
            h[k] = result[1]
            h = project(h)
            scores = compute_scores(h)
            cur_c5 = float(np.max(scores))
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return h, best_h, best_c5
