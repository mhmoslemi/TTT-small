import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_max_grad(h, tau):
    """Compute gradient of F_tau(h) = tau * log(sum exp(S_r/tau))"""
    n = len(h)
    dx = 2.0 / n
    S = _scores(h)
    P = len(S)  # 2n-1

    M = np.max(S)
    u = np.exp((S - M) / tau)
    Z = np.sum(u)
    w = u / Z  # soft weights, sum to 1
    
    # Aggregate weights by |lag|
    # Output index r corresponds to lag r - (n-1)
    nags = n  # lags from 0 to n-1
    W = np.zeros(nags)
    center = n - 1
    for a in range(nags):
        if a == 0:
            W[0] = w[center]
        else:
            W[a] = w[center + a] + w[center - a]
    
    # Compute gradient

    for a in range(nags):
        wa = W[a]
        if wa < 1e-14:
            continue
        m = n - a
        grad[a:] += wa * (1.0 - h[:m])
        grad[:m] += -wa * h[a:]
    grad *= dx
    return grad, S, w

def _line_search_exact(h, p, max_alpha):
    """Line search: find best alpha in [0, max_alpha] minimizing exact C5"""
    n = len(h)
    # Compute max feasible alpha
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12) * 0.9999
    
    # Coarse grid
    n_samp = 15
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    
    # Golden section refinement
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(15):

            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

    cur_c5 = _exact_c5(h)

    best_c5 = cur_c5

        stall = 0

                stall = 0

            else:
                stall += 1
                trust *= 0.75
                if trust < 1e-5:
                    break

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _top_k_lp_direction(h, scores, radius, top_k):

    n_lags = len(scores)
    G = np.zeros((n_lags, n))

        ad = abs(d)

        idx = np.argsort(scores)[-top_k:]

        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))

            sub_dl2 = min(dl4, time.time() + budget_s * 0.025)
            h_q, c5_q = _smooth_anneal(h_qp, taus_main[:3], sub_dl2, trust_init=0.2, max_iters_per_tau=250)
            if c5_q < best_c5:
                best_c5 = c5_q
                best_h = h_q.copy()

    # Phase 5: Fourier init + anneal
    if time.time() < t0 + budget_s * 0.72:
        dl5 = t0 + budget_s * 0.75
        for n_f in [128, 96, 112]:
            if time.time() > dl5:
                break
            h_f = _fourier_init(n_f, rng, K=10, scale=0.12)
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h_fr, c5_fr = _smooth_anneal(h_f, taus_main[:4], sub_dl, trust_init=0.2, max_iters_per_tau=250)
            if c5_fr < best_c5:
                best_c5 = c5_fr
                best_h = h_fr.copy()
                best_n = n_f
            else:
                best_n = n0 if len(best_h) == n0 else len(best_h)

    # Phase 6: Perturbation restarts
    dl6 = t0 + budget_s * 0.88
    best_n = len(best_h)
    scale = 0.04
    for attempt in range(100):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.018)
        h_try, c5_try = _smooth_anneal(h_pert, taus_fine, sub_dl, trust_init=0.1, max_iters_per_tau=200)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 7: LP + smooth combined fine-tuning
    if time.time() < t0 + budget_s * 0.94:
        dl7 = t0 + budget_s * 0.95
        h7, c5_7 = _lp_polish(best_h, dl7, radius_init=0.05, max_iters=200)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
        if time.time() < dl7:
            h7b, c5_7b = _smooth_anneal(best_h, [0.001, 0.0005, 0.0003], dl7, trust_init=0.05, max_iters_per_tau=150)
            if c5_7b < best_c5:
                best_c5 = c5_7b
                best_h = h7b.copy()

    # Phase 8: Pair polish
    if time.time() < t0 + budget_s * 0.975:
        dl8 = t0 + budget_s * 0.98
        hp, c5p = _pair_polish(best_h, dl8, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final projection and evaluation
    best_n_final = len(best_h)
    best_h = _project(best_h, best_n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = h_start.copy()
        final_c5 = _exact_c5(h_start)
    return best_h, final_c5, len(best_h)
