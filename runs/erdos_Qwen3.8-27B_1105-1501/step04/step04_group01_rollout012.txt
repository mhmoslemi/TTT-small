def _waterfill(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _line_search(h, d, dx, trust):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust)
    t_hi = min(t_hi, trust)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = float(obj_grid[best_i])
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx

        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)

        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx

        best_val = opt_val
    return best_t, best_val

def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b > hi_b:
            lo_b = 0.0
            hi_b = 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _subgrad_dir(h, dx, k_active, rng):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    threshold = max_s - 0.02
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    if len(active) > k_active:
        idx = rng.choice(len(active), size=k_active, replace=False)
        active = active[idx]
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += (1.0 - h[d:d + m])
                grad[d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += -h[g:g + m]
                grad[g:] += (1.0 - h[:m])
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return np.zeros(n)
    return -grad / norm

def _smoothed_grad_dir(h, dx, beta):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(scores))
    v = beta * (scores - max_s)
    v = np.clip(v, -50, 50)
    exp_v = np.exp(v)
    w = exp_v / float(np.sum(exp_v))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d + m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g + m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return np.zeros(n)
    return -grad / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _waterfill(h_new, n_new / 2.0)

def _crossover(h1, h2, rng):

    alpha = rng.uniform(0.25, 0.75)
    h_new = alpha * h1 + (1.0 - alpha) * h2
    return _waterfill(h_new, n / 2.0)

            if len(pop) > pop_size:
                worst = int(np.argmax(pop_c5))
                pop.pop(worst)
                pop_c5.pop(worst)

    # Phase 4: Annealing perturbation restarts (16%)
    d4 = t0 + budget_s * 0.82
    scale = 0.025
    best_scale = scale
    best_improve = 0.0
    for attempt in range(100):
        if time.time() > d4:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _waterfill(h_pert, len(best_h) / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.009)
        h_t, c5_t = _local_optimize(h_pert, sub_dl, rng, trust_init=0.05, use_lp=True)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_scale = scale
            best_improve = best_improve + 1
            scale = max(0.005, scale * 0.88)
        else:
            scale = min(0.12, scale * 1.18)
            if scale > best_scale * 3.0 and best_improve > 0:
                scale = best_scale

    # Phase 5: Try diverse structural transforms (8%)
    d5 = t0 + budget_s * 0.90
    transforms = []
    # Complement symmetry
    hc = 1.0 - best_h
    transforms.append(hc)
    # Time reversal
    hr = best_h[::-1].copy()
    transforms.append(hr)
    # Symmetrized
    hsym = (best_h + best_h[::-1]) / 2.0
    transforms.append(_waterfill(hsym, len(hsym) / 2.0))
    # Slight Gaussian smooth
    kernel_size = max(3, len(best_h) // 20)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4.0))
    kernel = kernel / kernel.sum()
    hsmooth = np.convolve(best_h, kernel, mode='same')
    transforms.append(_waterfill(hsmooth, len(hsmooth) / 2.0))

    for h_transform in transforms:
        if time.time() > d5:
            break
        h_transform = _waterfill(h_transform, len(h_transform) / 2.0)
        c5_transform = _exact_c5(h_transform)
        if c5_transform < best_c5 + 0.002:
            sub_dl = min(d5, time.time() + budget_s * 0.012)
            h_t, c5_t = _local_optimize(h_transform, sub_dl, rng, trust_init=0.05, use_lp=True)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 6: Pair refinement (7%)
    d6 = t0 + budget_s * 0.97
    h_p, c5_p = _pair_refine(best_h, d6, rng)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Final validation
    best_h = _waterfill(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
