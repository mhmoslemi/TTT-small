    dx = 2.0 / n

def _scores(h):

    return np.correlate(h, 1.0 - h, 'full') * (2.0 / n)

def _project(v, n):
    target = n / 2.0
    # Water-filling projection
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Exact correction
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _fourier_h(params, n, K):
    i = (np.arange(n) + 0.5) / n
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = params[2*(k-1)]
        b = params[2*(k-1)+1]
        h += a * np.cos(2.0 * np.pi * k * i)
        h += b * np.sin(2.0 * np.pi * k * i)
    return h
