        mid = (l + r) / 2.0
        f_mid = float(np.max(a + b * mid))
        # Subgradient: b values where a + b*mid = f_mid
        active = np.abs(a + b * mid - f_mid) < 1e-12
        if not np.any(active):
            active = (a + b * mid) == f_mid
        if np.any(active):
            min_b = float(np.min(b[active]))
            max_b = float(np.max(b[active]))
        else:
            # Fallback: use argmax
            idx = int(np.argmax(a + b * mid))
            min_b = max_b = b[idx]
        if min_b > 1e-12:
            # f increasing at mid, optimum is to the left
            r = mid
        elif max_b < -1e-12:
            # f decreasing at mid, optimum is to the right
            l = mid
        else:
            # mid is optimal
            best_d = mid
            best_f = f_mid
            break
        if f_mid < best_f:
            best_f = f_mid
            best_d = mid
    # Also check endpoints
    for cand in [lo, hi]:
        fc = float(np.max(a + b * cand))
        if fc < best_f:
            best_f = fc
            best_d = cand
    return best_d

def pair_optimize_pass(h, corr, G, deadline, top_k_pairs=200):
    """One pass of greedy pair optimization. Returns improved h or None."""
    n = len(h)
    dx = 2.0 / n
    cur_max = float(np.max(corr))
    improved = False
    stall = 0
    
    while time.time() < deadline:
        if stall > 500:
            break
        best_improve = 1e-15
        best_pair = None
        best_delta = 0.0
        best_new_h = None
        
        # Sample a subset of pairs for efficiency
        n_try = min(top_k_pairs, n * (n - 1) // 2)
        if n_try < n * (n - 1) // 2:
            pairs_idx = np.random.default_rng(int(time.time() * 1000) % 100000).choice(
                n * (n - 1) // 2, size=n_try, replace=False)
            pair_list = [(pairs_idx[i] // (n - 1 - pairs_idx[i] // (n - 1)), 
                          pairs_idx[i] % (n - 1 - pairs_idx[i] // (n - 1))) 
                         for i in range(n_try)]
            # Fix: generate pairs properly
            pair_list = []
            all_pairs = []
            for i in range(n):
                for j in range(i + 1, n):
                    all_pairs.append((i, j))
            if len(all_pairs) > n_try:
                idx_sel = np.random.default_rng(int(time.time() * 7919) % 999999).choice(
                    len(all_pairs), size=n_try, replace=False)
                pair_list = [all_pairs[i] for i in idx_sel]
            else:
                pair_list = all_pairs
        else:
            pair_list = [(i, j) for i in range(n) for j in range(i + 1, n)]
        
        for (i, j) in pair_list:
            lo_d = max(-h[i], h[j] - 1.0)
            hi_d = min(1.0 - h[i], h[j])
            if hi_d - lo_d < 1e-12:
                continue
            diff = G[:, i] - G[:, j]
            delta = find_optimal_delta(corr, diff, lo_d, hi_d)
            new_max = float(np.max(corr + delta * diff))
            improve = cur_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_pair = (i, j)
                best_delta = delta
                best_new_h = h.copy()
                best_new_h[i] += delta
                best_new_h[j] -= delta
        
        if best_pair is not None and best_new_h is not None:
            h = best_new_h
            h = project(h, n / 2.0)
            corr = compute_corrs(h)
            G = build_G(h)
            cur_max = float(np.max(corr))
            improved = True
            stall = 0
        else:
            stall += 1
    
    return h, improved

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    def obj_grad(x):
        omh = 1.0 - x
        scores_raw = np.correlate(x, omh, mode='full')
        scores = scores_raw * dx
        max_s = float(np.max(scores))
        exp_s = np.exp((scores - max_s) / tau)
        Z = float(np.sum(exp_s))
        w = exp_s / Z
        val = max_s + tau * np.log(Z)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            d = k - (n - 1)
            js1 = max(0, -d)
            je1 = min(n - 1, n - 1 - d)
            if js1 <= je1:
                grad[js1:je1 + 1] += wk * (1.0 - x[js1 + d:je1 + 1 + d])
            js2 = max(0, d)
            je2 = min(n - 1, n + d - 1)
            if js2 <= je2:
                grad[js2:je2 + 1] -= wk * x[js2 - d:je2 + 1 - d]
        return val, dx * grad

    def obj(x):
        return obj_grad(x)[0]

    def grad(x):
        return obj_grad(x)[1]

    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 100:
            break
        corr = compute_corrs(h)
        G = build_G(h)
        scores = corr.copy()
        
        # LP direction: find p minimizing max_r (scores[r] + p·G[r,:]) * dx
        # subject to sum(p) = 0, ||p||_inf <= radius
        m = 2 * n - 1
        # Variables: p[0..n-1], t
        # Minimize t
        # s.t. G[r,:]·p + scores[r] <= t for all r
        #      sum(p) = 0
        #      -radius <= p[i] <= radius
        #      Also maintain h+p in [0,1]: -h[i] <= p[i] <= 1-h[i]
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo_b = max(-radius, -h[j])
            hi_b = min(radius, 1.0 - h[j])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        
        p = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            continue
        
        # Line search
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        best_a = 0.0
        best_c5 = cur_c5
        for i in range(12):
            a = alpha_max * (i + 1) / 12.0
            c5_t = exact_c5(h + a * p)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_a = a
        
        if best_a > 0:
            lo_a = max(0.0, best_a * 0.5)
            hi_a = min(alpha_max, best_a * 1.5)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_a - gr * (hi_a - lo_a)
            c2 = lo_a + gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
            f2 = exact_c5(h + c2 * p)
            for _ in range(12):
                if hi_a - lo_a < 1e-13:
                    break
                if f1 < f2:
                    hi_a, c2, f2 = c2, c1, f1
                    c1 = hi_a - gr * (hi_a - lo_a)
                    f1 = exact_c5(h + c1 * p)
                else:
                    lo_a, c1, f1 = c1, c2, f2
                    c2 = lo_a + gr * (hi_a - lo_a)
                    f2 = exact_c5(h + c2 * p)
            opt_a = (lo_a + hi_a) / 2.0
            opt_c5 = exact_c5(h + opt_a * p)
            if opt_c5 < best_c5:
                best_a, best_c5 = opt_a, opt_c5
        
        if best_a > 0 and best_c5 < cur_c5 - 1e-13:
            h = h + best_a * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.5)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def intensive_pair_opt(h, deadline, n_passes_max=500):
    """Intensive pair optimization with periodic restarts."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(12345)
    
    for _pass in range(n_passes_max):
        if time.time() >= deadline:
            break
        h = best_h.copy()
        corr = compute_corrs(h)
        G = build_G(h)
        cur_max = float(np.max(corr))
        
        # Multiple rounds within one pass
        sub_deadline = min(deadline, time.time() + 3.0)
        improved_any = False
        for _round in range(200):
            if time.time() >= sub_deadline:
                break
            # Find best pair to swap
            n_try = min(150, n * (n - 1) // 2)
            all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            if len(all_pairs) > n_try:
                sel = rng.choice(len(all_pairs), size=n_try, replace=False)
                trial_pairs = [all_pairs[i] for i in sel]
            else:
                trial_pairs = all_pairs
            
            best_improve = 1e-14
            best_pi = best_pj = best_delta = 0
            for (i, j) in trial_pairs:
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d - lo_d < 1e-12:
                    continue
                diff = G[:, i] - G[:, j]
                delta = find_optimal_delta(corr, diff, lo_d, hi_d)
                new_max = float(np.max(corr + delta * diff))
                imp = cur_max - new_max
                if imp > best_improve:
                    best_improve = imp
                    best_pi, best_pj, best_delta = i, j, delta
            
            if best_improve > 1e-14:
                h[best_pi] += best_delta
                h[best_pj] -= best_delta
                h = project(h, target)
                corr = compute_corrs(h)
                G = build_G(h)
                cur_max = float(np.max(corr))
                improved_any = True
            else:
                break
        
        c5_cur = exact_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        if not improved_any:
            # Stuck - try perturbation
            if time.time() < deadline:
                scale = 0.03 + 0.04 * (1.0 - (deadline - time.time()) / max(deadline - t0_global, 1e-6))
                h_pert = best_h + rng.normal(0, scale, n)
                h_pert = project(h_pert, target)
                c5_pert = exact_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_pert
    
    return best_h, best_c5

t0_global = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0 = time.time()
    t0_global = t0
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    # Initialize
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: LP to establish good baseline (15%)
    d1 = t0 + budget_s * 0.15
    h_lp, c5_lp = lp_loop(h, d1, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: SLSQP smoothing (10%)
    d2 = t0 + budget_s * 0.25
    tau_list = [0.3, 0.15, 0.08, 0.04, 0.02]
    iter_list = [80, 80, 70, 60, 50]
    h_s = best_h.copy()
    for i in range(len(tau_list)):
        if time.time() > d2:
            break
        h_s = slsqp_smooth(h_s, tau_list[i], iter_list[i])
        h_s = project(h_s, best_n / 2.0)
        c5_s = exact_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Intensive pair optimization (45%)
    d3 = t0 + budget_s * 0.70
    h_pair, c5_pair = intensive_pair_opt(best_h, d3)
    if c5_pair < best_c5:
        best_c5 = c5_pair
        best_h = h_pair.copy()

    # Phase 4: Multi-resolution (10%)
    d4 = t0 + budget_s * 0.80
    offsets = [0, -6, 6, -10, 10, -15, 15, -20, 20, -25, 25, -30, 35, -40, 40]
    tried = set([best_n])
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 160:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample_h(best_h, n_try)
        # Quick LP
        sub_dl = min(d4, time.time() + budget_s * 0.01)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.06)
        # Quick pair
        sub_dl2 = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r2 = intensive_pair_opt(h_r, sub_dl2, n_passes_max=50)
        c5_r = min(c5_r, c5_r2)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Final polishing (15%)
    d5 = hard_deadline
    # LP micro
    sub_dl = min(d5, time.time() + budget_s * 0.04)
    h_f, c5_f = lp_loop(best_h, sub_dl, radius_init=0.03)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    # Pair polish
    sub_dl2 = min(d5, time.time() + budget_s * 0.04)
    h_p, c5_p = intensive_pair_opt(best_h, sub_dl2, n_passes_max=100)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
