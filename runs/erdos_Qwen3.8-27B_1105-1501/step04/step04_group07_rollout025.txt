```python
def _pair_optimal_delta(scores, G_col_i, G_col_j, delta_min, delta_max):
    # Find delta that minimizes max_k (scores_k + delta * (G_col_i[k] - G_col_j[k]))
    u = G_col_i - G_col_j
    # Objective: max_k (scores_k + delta * u_k)
    # This is convex in delta
    # Use ternary search
    a, b = delta_min, delta_max
    for _ in range(60):  # ~18 decimal digits of precision
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        f1 = np.max(scores + m1 * u)
        f2 = np.max(scores + m2 * u)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad(h):
    """Compute gradient matrix G where G[k, i] = df_k/dh_i"""
    n = len(h)
    dx = 2.0 / n
    # For f_k = sum_j h[j+k]*(1-h[j]) * dx
    # df_k/dh[i] involves terms where h[i] appears
    # h[i] appears in f_k as:
    #   - h[i]*(1-h[i-k]) when j = i-k (so j+k = i)
    #   - (1-h[i])*h[i+k] when j = i
    # So df_k/dh[i] = (1-h[i-k])*dx - h[i+k]*dx
    # (with appropriate bounds)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        d = k - (n - 1)  # actual lag
        for i in range(n):
            # Term 1: h[i] appears as h[j+k] where j = i-k, so j = i-d
            j1 = i - d
            if 0 <= j1 < n and j1 + d == i:  # j1+k_offset = i means h[i] is the "h[j+k]" term
                # f_k has term h[i]*(1-h[j1])
                # df_k/dh[i] += (1-h[j1])*dx
                G[k, i] += (1.0 - h[j1]) * dx
```

```python
def _grad_matrix_fast(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        d = k - (n - 1)
        # Term 1: (1-h[i-d]) for valid i-d
        i_vals = np.arange(n)
        i_minus_d = i_vals - d
        mask1 = (i_minus_d >= 0) & (i_minus_d < n)
        term1 = np.zeros(n)
        term1[mask1] = 1.0 - h[i_minus_d[mask1]]
        # Term 2: -h[i+d] for valid i+d
        i_plus_d = i_vals + d
        mask2 = (i_plus_d >= 0) & (i_plus_d < n)
        term2 = np.zeros(n)
        term2[mask2] = -h[i_plus_d[mask2]]
        G[k] = dx * (term1 + term2)
    return G
```

```python
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix_fast(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        d = k - (n - 1)
        i_vals = np.arange(n)
        imd = i_vals - d
        m1 = (imd >= 0) & (imd < n)
        t1 = np.zeros(n)
        t1[m1] = 1.0 - h[imd[m1]]
        ipd = i_vals + d
        m2 = (ipd >= 0) & (ipd < n)
        t2 = np.zeros(n)
        t2[m2] = -h[ipd[m2]]
        G[k] = dx * (t1 + t2)
    return G

def _pair_local_search(h, deadline, max_iters=500000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    it = 0
    stall = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        scores = _scores(h)
        G = _grad_matrix_fast(h)
        
        best_imp = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in range(n):
            hi_val = h[i]
            if hi_val < 1e-12 or hi_val > 1.0 - 1e-12:
                continue
            for j in range(i + 1, n):
                hj_val = h[j]
                if hj_val < 1e-12 or hj_val > 1.0 - 1e-12:
                    continue
                u = G[:, i] - G[:, j]
                d_lo = max(-hi_val, hj_val - 1.0)
                d_hi = min(1.0 - hi_val, hj_val)
                if d_hi - d_lo < 1e-12:
                    continue
                a, b = d_lo, d_hi
                for _ in range(20):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    f1 = float(np.max(scores + m1 * u))
                    f2 = float(np.max(scores + m2 * u))
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                delta_opt = (a + b) / 2.0
                pred = float(np.max(scores + delta_opt * u))
                imp = cur_c5 - pred
                if imp > best_imp:
                    best_imp = imp
                    best_i = i
                    best_j = j
                    best_delta = delta_opt
        
        if best_i < 0 or best_imp < 1e-14:
            stall += 1
            if stall > 30:
                break
            break
        
        h[best_i] += best_delta
        h[best_j] -= best_delta
        np.clip(h, 0.0, 1.0, out=h)
        cur_c5 = _exact_c5(h)
        stall = 0
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5

def _pnorm_descent(h, deadline, p_start=4.0, p_end=1024.0, n_steps=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix_fast(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.15, 0.1, 0.05, 0.025, 0.01]:
            h_try = h + d * scale
            h_try = _project(h_try, target)
            new_c5 = _exact_c5(h_try)
            if new_c5 < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = new_c5
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 150:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _subgrad_descent(h, deadline, taus=[0.005, 0.01, 0.02, 0.05, 0.1]):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    n_lags = 2 * n - 1
    stall = 0
    t_idx = 0
    it = 0
    while time.time() < deadline and it < 50000:
        it += 1
        tau = taus[t_idx % len(taus)]
        if it % 200 == 0:
            t_idx += 1
        scores = _scores(h)
        exp_s = np.exp(scores / tau - scores.max())
        ws = exp_s / exp_s.sum()
        G = _grad_matrix_fast(h)
        grad = G.T @ ws
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 200:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
            h_try = h + d * scale
            h_try = _project(h_try, target)
            new_c5 = _exact_c5(h_try)
            if new_c5 < cur_c5 - 1e-15:
                h = h_try
                cur_c5 = new_c5
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 200:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_refine(h, deadline, top_k=8, radius=0.08, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores(h)
        G = _grad_matrix_fast(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-7:
                break
            continue
        best_alpha = 0.0
        best_new_c5 = cur_c5
        for fr in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_try = h + p * fr
            h_try = _project(h_try, target)
            nc = _exact_c5(h_try)
            if nc < best_new_c5:
                best_new_c5 = nc
                best_alpha = fr
        if best_new_c5 < cur_c5 - 1e-13:
            h = h + p * best_alpha
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.03, 0.3)
        else:
            stall += 1
            r *= 0.8
            if stall > 20 and r < 0.02:
                r = 0.06
    return best_h, best_c5
```
