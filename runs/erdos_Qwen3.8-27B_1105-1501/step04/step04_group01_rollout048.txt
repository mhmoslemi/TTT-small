def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] = np.clip(x[mask] + res / cnt, 0.0, 1.0)
    return x

def _exact_c5(h):

def _shift_values(h):

def _grad_for_shifts(h, shift_indices):
    n = len(h)
    G = np.zeros((len(shift_indices), n))
    for idx, r in enumerate(shift_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[idx, :m] += (1.0 - h[d:d + m])
            G[idx, d:] += -h[:m]
        else:
            e = -d
            m = n - e
            G[idx, e:] += (1.0 - h[:m])
            G[idx, :m] += -h[e:e + m]
    return G

def _lp_direction(h, scores, top_k=20, radius=0.15):

    G = _grad_for_shifts(h, active)
    c = np.zeros(n + 1)
    A_ub = np.zeros((K + 1, n + 1))
    A_ub[:K, :n] = G
    A_ub[:K, n] = -1.0
    A_ub[K, :n] = 1.0
    b_ub = np.zeros(K + 1)
    b_ub[K] = 0.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    try:

        if res.success and res.x[n] > 1e-12:

    return None, 0.0

def _softmax_gradient(h, dx, tau):
    n = len(h)
    sv = np.correlate(h, 1.0 - h, mode='full') * dx
    sv_max = float(np.max(sv))
    exp_sv = np.exp(tau * (sv - sv_max))
    w = exp_sv / float(np.sum(exp_sv))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += w[r] * (1.0 - h[d:d + m])
            grad[d:] += w[r] * (-h[:m])
        else:
            e = -d
            m = n - e
            grad[e:] += w[r] * (1.0 - h[:m])
            grad[:m] += w[r] * (-h[e:e + m])
    grad -= float(np.sum(grad)) / n
    return grad
