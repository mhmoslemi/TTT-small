def exact_c5(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_rows(h, active_indices):
    """Build gradient rows for active correlation indices."""
    n = len(h)
    m = len(active_indices)
    dx = 2.0 / n
    G = np.zeros((m, n))
    for k in range(m):
        outidx = active_indices[k]
        d = outidx - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                G[k, d:] += dx * (1.0 - h[:m2])
                G[k, :m2] -= dx * h[d:]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                G[k, :m2] += dx * (1.0 - h[g:])
                G[k, g:] -= dx * h[:m2]
    return G

def line_search(h, p, cur_c5, max_alpha=0.4):
    """Exact line search along direction p from point h."""
    n = len(h)
    # Compute max feasible alpha
    alpha_max = max_alpha

        elif p[j] < -1e-14:

def active_equalize_direction(h, scores, k_active, margin_factor, ridg, rough_pen):
    """Solve the active constraint equalization QP."""
    n = len(h)
    # Select active indices
    sorted_idx = np.argsort(scores)[::-1]
    c_max = scores[sorted_idx[0]]
    c_min_active = scores[sorted_idx[min(k_active, len(scores))-1]]
    spread = c_max - c_min_active
    
    # Active set: top k or those within tolerance
    active = sorted_idx[:k_active]
    s_active = scores[active]
    
    # Target: drive all active toward min - margin
    margin = margin_factor * max(spread, 1e-4)
    target = s_active.min() - margin
    
    # Weights: prioritize constraints further above target
    gaps = target - s_active  # all negative (since target < min(s_active))
    # Actually: we want to decrease each s_i to target, so residual = target - s_i < 0
    # Weight more on those that are furthest above target
    max_gap = np.max(s_active) - target
    if max_gap < 1e-15:
        q = np.ones(len(active))
    else:
        q = 1.0 + 50.0 * ((s_active - target) / max_gap) ** 2
    
    # Build gradient matrix
    G = build_grad_rows(h, active)
    
    # Build QP: min (Gp - r)^T W (Gp - r) + ridge*||p||^2 + rough*||Dp||^2
    # where r = target - s_active (negative residuals)
    r = target - s_active  # this is negative for all active
    
    # H = G^T W G + ridge*I + rough*D^T D
    W = np.diag(q)
    H = G.T @ (W @ G)
    H += ridg * np.eye(n)
    
    # Roughness: simple second difference
    if rough_pen > 1e-15:
        for i in range(n):
            H[i, i] += 2.0 * rough_pen
            if i > 0:
                H[i, i-1] -= rough_pen
                H[i-1, i] -= rough_pen
            if i < n - 1:
                H[i, i+1] -= rough_pen
                H[i+1, i] -= rough_pen
    
    # RHS: G^T W r
    rhs = G.T @ (q * r)
    
    # KKT system: [H  1; 1^T 0] [p; mu] = [rhs; 0]
    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs_full = np.zeros(n + 1)
    rhs_full[:n] = rhs
    
    try:
        sol = np.linalg.solve(KKT, rhs_full)
        p = sol[:n]
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(KKT, rhs_full, rcond=None)[0]
            p = sol[:n]
        except Exception:
            return None
    
    p -= np.mean(p)  # Ensure zero mean (redundant but safe)
    
    if not np.all(np.isfinite(p)):
        return None
    if np.max(np.abs(p)) < 1e-12:
        return None
    
    return p

def weighted_grad_direction(h, scores, k_active):
    """Fallback: weighted negative gradient of top-k active constraints."""
    n = len(h)
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:k_active]
    s_active = scores[active]
    c_max = s_active[0]

    G = build_grad_rows(h, active)
    p = -(G.T @ q)
    p -= np.mean(p)
    
    if not np.all(np.isfinite(p)) or np.max(np.abs(p)) < 1e-12:
        return None
    return p

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def smooth_perturb(h, rng, scale):
    """Add smooth zero-mean perturbation."""
    n = len(h)
    # Low-frequency noise
    noise = np.zeros(n)
    for k in [1, 2, 3, 5, 8]:
        amp = scale / k
        phase = rng.random() * 2 * np.pi
        noise += amp * np.sin(2 * np.pi * k * np.arange(n) / n + phase)
    noise -= noise.mean()
    return project(h + noise, len(h) / 2.0)

def binary_sa(n, deadline, rng, n_iters=30000):
    """Simulated annealing on binary vectors."""
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5 or (T > T_min and rng.random() < np.exp((cur_c5 - new_c5) / T)):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    
    return best_h, best_c5

def exact_c5(h):

def project(v, target):

        return np.ones(n)

    hi = float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def build_grad_rows(h, active_indices):
    n = len(h)
    m = len(active_indices)
    if m == 0:
        return np.zeros((0, n))
    dx = 2.0 / n
    G = np.zeros((m, n))
    for k in range(m):
        outidx = active_indices[k]
        d = outidx - (n - 1)
        if d >= 0:
            m2 = n - d
            if m2 > 0:
                G[k, d:] += dx * (1.0 - h[:m2])
                G[k, :m2] -= dx * h[d:]
        else:
            g = -d
            m2 = n - g
            if m2 > 0:
                G[k, :m2] += dx * (1.0 - h[g:])
                G[k, g:] -= dx * h[:m2]
    return G

def qp_direction(h, scores, k_active, margin_factor, ridge_val, rough_val, weight_pow):
    n = len(h)
    sorted_idx = np.argsort(scores)[::-1]
    active = sorted_idx[:k_active]
    s_active = scores[active]
    c_max = s_active[0]
    target = s_active.min() - margin_factor * max(c_max - s_active.min(), 1e-4)
    gaps = c_max - s_active
    max_gap = max(gaps[0], 1e-15)
    q = 1.0 + weight_pow * (gaps / max_gap) ** 2
    G = build_grad_rows(h, active)
    r = target - s_active
    H = G.T @ (q[:, None] * G)
    H += ridge_val * np.eye(n)
    if rough_val > 1e-15:
        dv = np.zeros(n)
        dv[0] = -1.0; dv[1] = 1.0
        dv[-1] = 1.0; dv[-2] = -1.0
        for i in range(1, n - 1):

            dv[i-1] = 0.0; dv[i] = 0.0; dv[i+1] = 0.0

    KKT[:n, n] = 1.0

            return None

            [4, 0.03, 3e-4, 0.0, 300.0],
            [6, 0.02, 2e-4, 0.0, 500.0],
            [5, 0.04, 4e-4, 2e-4, 200.0],
            [8, 0.05, 8e-4, 5e-4, 80.0],
        ]
        h_fin, c5_fin = local_optimize(best_h, dl6, cfg6, trust_init=0.06, max_iters=3000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = len(h_fin)

        # One more pair polish pass
        if time.time() < hard_dl:
            hp2, c5p2 = pair_polish_light(best_h, hard_dl, n_passes=2)
            if c5p2 < best_c5:
                best_c5 = c5p2
                best_h = hp2.copy()
                best_n = len(hp2)

    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
