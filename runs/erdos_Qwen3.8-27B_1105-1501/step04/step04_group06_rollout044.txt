    rng = np.random.default_rng(seed)
    k = n // 2
    idx = rng.choice(n, size=k, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def _gen_perturbed(h_base, n, seed, scale):
    rng = np.random.default_rng(seed)
    v = h_base + rng.normal(0, scale, n)
    return _project(v, float(n) * 0.5)


def _coord_descent(h, n, max_sweeps=50, deadline=None):
    """Coordinate descent: adjust pairs to reduce max correlation."""
    h = np.asarray(h, dtype=float).copy()
    target = n * 0.5
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        r_max = int(np.argmax(corr))
        s = r_max - (n - 1)
        # Find indices that contribute most to the max correlation
        j = np.arange(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        contrib = np.zeros(n)
        contrib[m1] = h[gs[m1]] * (1.0 - h[j[m1]])
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        contrib2 = np.zeros(n)
        contrib2[m2] = (1.0 - h[j[m2]]) * h[ps[m2]]
        # Try to reduce by moving mass from high h at g positions to low h at p positions
        # Simple: try swapping values between a high-contrib index and a low one
        if s >= 0:
            hi_idx = np.argmax(contrib)
            lo_idx = np.argmin(contrib)
        else:
            hi_idx = np.argmax(contrib2)
            lo_idx = np.argmin(contrib2)
        if hi_idx == lo_idx:
            continue
        # Try transfer
        delta = (h[hi_idx] - h[lo_idx]) * 0.5
        if delta < 1e-10:
            continue
        h[hi_idx] -= delta
        h[lo_idx] += delta
        h = _project(h, target)
        c5_new = _exact_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h = best_h.copy()

    return best_h, best_c5


def _multi_slsqp(h_init, n, sub_deadline):
    """Run active SLSQP then subgrad then SLSQP again."""
    h1, c5_1 = _active_slsqp(h_init, n, K_start=6, K_max=30,
                             max_outer=20, maxiter=100,
                             deadline=sub_deadline)
    if time.time() > sub_deadline:
        return h1, c5_1
    h2, c5_2 = _subgrad_steps(h1, n, n_steps=100, deadline=sub_deadline)
    if c5_2 < c5_1:
        h1, c5_1 = h2, c5_2
    if time.time() > sub_deadline:
        return h1, c5_1
    h3, c5_3 = _active_slsqp(h1, n, K_start=10, K_max=35,
                             max_outer=15, maxiter=150,
                             deadline=sub_deadline)
    if c5_3 < c5_1:
        h1, c5_1 = h3, c5_3
    return h1, c5_1


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    main_deadline = t0 + budget_s * 0.96
    final_deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = float(n0) * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Active SLSQP from parent (most promising)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _multi_slsqp(best_h, n0, d1)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1.copy(), n0

    # Phase 2: More SLSQP with growing K from current best
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _active_slsqp(best_h, n0, K_start=12, K_max=50,
                             max_outer=20, maxiter=200, deadline=d2)
    if c5_2 < best_c5:
        best_c5, best_h, best_n = c5_2, h2.copy(), n0

    # Phase 3: Diverse starts - binary balanced sequences
    d3 = t0 + budget_s * 0.65
    for biter in range(12):
        if time.time() > d3:
            break
        h_b = _gen_binary_balanced(n0, seed + 100 + biter)
        # Smooth it a bit
        h_s = _project(h_b + rng.normal(0, 0.3, n0), target)
        sub_d = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _active_slsqp(h_s, n0, K_start=8, K_max=30,
                                  max_outer=12, maxiter=80,
                                  deadline=sub_d)
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n0

    # Phase 4: Perturbed starts from best
    d4 = t0 + budget_s * 0.80
    for piter in range(30):
        if time.time() > d4:
            break
        scale = 0.02 + 0.04 * (piter % 5)
        h_p = _gen_perturbed(best_h, n0, seed + 500 + piter, scale)
        sub_d = min(d4, time.time() + budget_s * 0.008)
        h_r, c5_r = _active_slsqp(h_p, n0, K_start=8, K_max=25,
                                  max_outer=10, maxiter=80,
                                  deadline=sub_d)
        if c5_r < best_c5 - 1e-12:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n0

    # Phase 5: Multi-resolution search
    d5 = t0 + budget_s * 0.92
    for dn in [2, -2, 4, -4, 6, -6, 8, 16, -8]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d5, time.time() + budget_s * 0.02)
        h_r, c5_r = _active_slsqp(h_r, n_try, K_start=8, K_max=30,
                                  max_outer=12, maxiter=100,
                                  deadline=sub_d)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Final intensive polish
    d6 = final_deadline
    h_f, c5_f = _multi_slsqp(best_h, best_n, d6)
    if c5_f < best_c5:
        best_c5, best_h, best_n = c5_f, h_f.copy(), best_n

    # Subgradient polish at the very end
    h_g, c5_g = _subgrad_steps(best_h, best_n, n_steps=300, deadline=final_deadline)
    if c5_g < best_c5:
        best_c5, best_h, best_n = c5_g, h_g.copy(), best_n

    # One more SLSQP if time remains
    if time.time() < t0 + budget_s * 0.99:
        h_m, c5_m = _active_slsqp(best_h, best_n, K_start=15, K_max=45,
                                  max_outer=10, maxiter=200,
                                  deadline=t0 + budget_s * 0.995)
        if c5_m < best_c5:
            best_c5, best_h, best_n = c5_m, h_m.copy(), best_n

    # Final validation
    best_h = _project(best_h, float(best_n) * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
