    target = n / 2.0
    h = h.copy()
    
    for it in range(max_iters):
        # Compute gradient: dx * (1 - 2 * T_lambda @ h)
        # (T_lambda @ h)[i] = sum_k lam[k] * h[i+k] (valid indices only)
        Th = np.zeros(n)
        n_lags = len(lam)

            k = idx - (n - 1)  # lag value
            w = lam[idx]

                    Th[:m] += w * h[k:k+m]  # h[i+k] for i=0..m-1

                m = n - g

                    Th[g:] += w * h[:m]  # h[i-k] for i=g..n-1, h[i-g] with i-g=0..m-1

        alpha = 0.5

        # Actually f_λ(h) = dx * sum(h) - dx * h @ (T_λ h)
