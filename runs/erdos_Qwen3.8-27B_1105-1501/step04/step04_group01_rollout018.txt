def _project(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

def _lp_direction(h, scores, radius):
    n = len(h)
    # Identify active constraints (top ones)
    sorted_idx = np.argsort(scores)[::-1]
    max_s = scores[sorted_idx[0]]
    active = [i for i in sorted_idx if scores[i] > max_s - 0.01]
    if not active:
        active = [sorted_idx[0]]
    active = active[:30]  # Cap
    
    # Build gradient matrix for active rows
    G = np.zeros((len(active), n))
    for k, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[k, :m] = (1.0 - h[d:d+m])
                G[k, d:] += (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[k, :m] = (-h[g:g+m])
                G[k, g:] += (1.0 - h[:m])
    
    # LP: minimize s s.t. G @ d <= s, |d| <= radius, sum(d) = 0
    # Variables: [d (n), s (1)]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:

            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick subgradient warmup (6%)
    d1 = t0 + budget_s * 0.06
    h1, c5_1 = _subgrad_loop(h, d1, tau_list=(50, 150, 500), lr_init=0.02, mom_decay=0.85)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP optimization from parent (14%)
    d2 = t0 + budget_s * 0.20
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (30%)
    d3 = t0 + budget_s * 0.50
    n_best = len(best_h)
    n_candidates = sorted(set([
        48, 56, 64, 72, 80, 88, 96,
        max(40, n_best - 8), max(40, n_best - 4),
        n_best,
        min(160, n_best + 4), min(160, n_best + 8), min(160, n_best + 12),
    ]))
    time_per_n = max(3, (d3 - time.time()) / max(1, len(n_candidates)))
    for n_try in n_candidates:
        rem = d3 - time.time()
        if rem < 3:
            break
        if n_try < 32 or n_try > 200:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl1 = min(d3, time.time() + time_per_n * 0.5)
        h_r, c5_r = _subgrad_loop(h_r, sub_dl1, tau_list=(40, 150, 500), lr_init=0.02)
        rem2 = d3 - time.time()
        if rem2 > 3 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d3, time.time() + rem2 * 0.5)
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.08)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 4: Diversification (18%)
    d4 = t0 + budget_s * 0.68
    n_div = len(best_h)
    max_attempts = int((d4 - time.time()) / 6)
    max_attempts = max(4, min(max_attempts, 60))
    for attempt in range(max_attempts):
        rem = d4 - time.time()
        if rem < 5:
            break
        gen_type = attempt % 5
        if gen_type == 0:
            h_new = _gen_random_binary(n_div, rng)
        elif gen_type == 1:
            h_new = _gen_sinusoidal(n_div, rng)
        elif gen_type == 2:
            h_new = _gen_block(n_div, rng)
        elif gen_type == 3:
            h_new = _gen_perturb(best_h, rng, 0.12 + 0.03 * (attempt % 4))
        else:
            # Try at a different resolution
            n_alt = n_div + (attempt % 3 - 1) * 8
            if n_alt < 32:
                n_alt = 48
            h_new = _resample(best_h, n_alt)
            h_new = _gen_perturb(h_new, rng, 0.1)
            n_use = len(h_new)
        sub_dl = min(d4, time.time() + max(2, rem * 0.35))
        h_t, c5_t = _subgrad_loop(h_new, sub_dl, tau_list=(30, 100, 300), lr_init=0.02)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 5: Intensive LP + subgradient on best (15%)
    d5 = t0 + budget_s * 0.83
    h5, c5_5 = _lp_loop(best_h, min(d5, time.time() + budget_s * 0.07), radius_init=0.1)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    sub_dl5 = min(d5, time.time() + budget_s * 0.07)
    h5b, c5_5b = _subgrad_loop(best_h, sub_dl5, tau_list=(80, 250, 700), lr_init=0.015)
    if c5_5b < best_c5:
        best_c5 = c5_5b
        best_h = h5b.copy()

    # Phase 6: Pair and triple refinement (8%)
    d6 = t0 + budget_s * 0.91
    h6, c5_6 = _pair_refine(best_h, min(d6, time.time() + budget_s * 0.04), max_passes=30)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    sub_dl6 = min(d6, time.time() + budget_s * 0.04)
    h6b, c5_6b = _triple_refine(best_h, sub_dl6, max_passes=15)
    if c5_6b < best_c5:
        best_c5 = c5_6b
        best_h = h6.copy()

    # Phase 7: Final LP polish + pair (5%)
    d7 = t0 + budget_s * 0.96
    h7, c5_7 = _lp_loop(best_h, min(d7, time.time() + budget_s * 0.025), radius_init=0.05)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    sub_dl7 = min(d7, time.time() + budget_s * 0.022)
    h7b, c5_7b = _pair_refine(best_h, sub_dl7, max_passes=20)
    if c5_7b < best_c5:
        best_c5 = c5_7b
        best_h = h7b.copy()

    # Final subgradient polish
    d8 = t0 + budget_s * 0.99
    h8, c5_8 = _subgrad_loop(best_h, d8, tau_list=(200, 600), lr_init=0.01)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 > 0.5:
        best_h = _project(h0, n0 / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
