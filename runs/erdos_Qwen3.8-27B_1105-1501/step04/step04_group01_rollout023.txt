def _lp_direction_topk(h, G, scores, k=20, radius=0.1):
    n = len(h)
    # Get top k rows
    top_k = np.argsort(scores)[-k:]
    # LP: minimize s subject to G[top_k] @ d - s <= -scores[top_k]
    #     sum(d) = 0
    #     radius bounds on d
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G[top_k]
    A_ub[:, n] = -1.0
    b_ub = -scores[top_k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n]
    return None

            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _compute_G(h)
        p = _lp_direction(h, scores, G, k, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
            if stall == 0 and int(np.sum((alpha * p) ** 2)) % 50 == 0:
                pass
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.03:
                radius = 0.08
    return best_h, best_c5

def _softmax_direction(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    G = _compute_G(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _grad_descent(h, deadline, tau_list=(50, 120, 300), lr_init=0.03, mom_decay=0.85):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    ti = 0
    while time.time() < deadline:
        if stall > 50:
            break
        tau = tau_list[ti % len(tau_list)]
        d = _softmax_direction(h, dx, tau)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            ti += 1
            continue
        d = d / dn
        mom = mom_decay * mom + (1.0 - mom_decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.5)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.8
            if stall % 20 == 0:
                ti += 1
                mom = np.zeros(n)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_round(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[target:]] = 1.0
    return h_bin

def _perturb_and_optimize(h, rng, scale, deadline, method='lp'):
    n = len(h)
    h_pert = h + rng.normal(0, scale, n)
    h_pert = _project(h_pert, n / 2.0)
    if method == 'lp':
        h_opt, c5_opt = _lp_local_search(h_pert, deadline, radius_init=0.1, k=20)
    else:
        h_opt, c5_opt = _grad_descent(h_pert, deadline, tau_list=(80, 200), lr_init=0.02)
    return h_opt, c5_opt

def _pair_refine(h, deadline, max_passes=20):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) < 2:
            break
        for pi in range(len(interior)):
            if time.time() > deadline:
                break
            i = interior[pi]
            for pj in range(pi + 1, len(interior)):
                j = interior[pj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _structured_directions(h, rng, dx):
    n = len(h)
    directions = []
    for tau in [30, 80, 300]:
        d = _softmax_direction(h, dx, tau)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            directions.append(d / dn)
    for _ in range(8):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j:
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            directions.append(d)
    for _ in range(4):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        d = np.sin(np.arange(n) * freq + phase)
        d -= d.mean()
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            directions.append(d / dn)
    return directions

def _structured_search(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    trust = 0.1
    stall = 0
    while time.time() < deadline:
        if stall > 35:
            break
        directions = _structured_directions(h, rng, dx)
        improved = False
        for d in directions:
            if time.time() > deadline:
                break
            alpha, new_c5 = _line_search(h, trust * d, dx, trust)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * trust * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
        if improved:
            trust = min(trust * 1.03, 0.3)
            stall = 0
        else:
            stall += 1
            trust *= 0.82
            if trust < 0.005:
                trust = 0.04
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick gradient warmup (3% of budget)
    d1 = t0 + budget_s * 0.03
    h1, c5_1 = _grad_descent(h, d1, tau_list=(60, 150), lr_init=0.02, mom_decay=0.85)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP local search with top-K (22%)
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _lp_local_search(best_h, d2, radius_init=0.15, k=30)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More LP with different K values (10%)
    d3 = t0 + budget_s * 0.35
    for k_try in [15, 10, 50, 80]:
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h3, c5_3 = _lp_local_search(best_h, sub_dl, radius_init=0.1, k=k_try)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Structured direction search (10%)
    d4 = t0 + budget_s * 0.45
    h4, c5_4 = _structured_search(best_h, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution exploration (25%)
    d5 = t0 + budget_s * 0.70
    best_n = len(best_h)
    n_candidates = sorted(set([best_n + dn for dn in
        range(-30, 31, 4) if 48 <= best_n + dn <= 180]))
    n_candidates.sort(key=lambda x: abs(x - best_n))
    for n_try in n_candidates:
        if time.time() > d5:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_local_search(h_r, sub_dl, radius_init=0.1, k=20)
        rem = d5 - time.time()
        if rem > 5 and c5_r < best_c5 + 0.003:
            sub_dl2 = min(d5, time.time() + max(2, budget_s * 0.006))
            h_r, c5_r = _structured_search(h_r, sub_dl2, rng)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Binary rounding exploration (8%)
    d6 = t0 + budget_s * 0.78
    h_bin = _binary_round(best_h)
    h_bin, c5_bin = _lp_local_search(h_bin, min(d6, time.time() + budget_s * 0.02), radius_init=0.3, k=20)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 7: Perturbation-based restarts (17%)
    d7 = t0 + budget_s * 0.95
    scale = 0.04
    for attempt in range(300):
        if time.time() > d7:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.006)
        h_t, c5_t = _lp_local_search(h_pert, sub_dl, radius_init=0.08, k=15)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.005, scale * 0.88)
        else:
            scale = min(0.15, scale * 1.1)

    # Phase 8: Pair refinement (3%)
    d8 = t0 + budget_s * 0.98
    h_p, c5_p = _pair_refine(best_h, d8, max_passes=15)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 > 1.0:
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
