            b = list(zeros)[np.random.randint(len(zeros))]

For a swap of a (1→0) and b (0→1):

For h[a]: 1→0
- As first index in c[s]: contributes (1-h[a+s]) to c[s] for valid a+s. After change: contributes 0. Delta: -(1-h[a+s]).
- As second index in c[s]: the term is h[a-s]*(1-h[a]) = h[a-s]*0 = 0. After change: h[a-s]*(1-0) = h[a-s]. Delta: +h[a-s].

- As second index in c[s]: the term is h[b-s]*(1-h[b]) = h[b-s]*1 = h[b-s]. After change: h[b-s]*0 = 0. Delta: -h[b-s].

def _lse_objective(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -300.0, 300.0)
    exp_v = np.exp(vals)
    s = float(np.sum(exp_v))
    f = ms + (1.0 / alpha) * np.log(s)
    w = exp_v / s
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        s_val = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s_val
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s_val
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        grad += w[r] * pd
    grad -= np.mean(grad)
    return f, grad

def _continuous_refine(h, n, deadline, maxiter_per_alpha=100):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for alpha in [15.0, 40.0, 80.0, 160.0]:
        if time.time() > deadline:
            break
        try:
            res = minimize(_lse_objective, h, args=(n, alpha), jac=True,
                         method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14, 'gtol': 1e-10})
            x = _project(res.x, n * 0.5)
            c5 = _exact_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
                h = x.copy()
        except Exception:
            pass
    
    return best_h, best_c5
