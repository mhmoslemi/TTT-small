import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_con = len(scores_k)
    A_ub = np.zeros((n_con, n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        lb = max(-radius, -h[j] + 1e-12)
        ub = min(radius, 1.0 - h[j] - 1e-12)
        bounds.append((lb, ub))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _line_search(h, p, target):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    c5_0 = _exact_c5(h)
    best_a = 0.0
    best_c5 = c5_0
    n_samp = 12
    alphas = np.linspace(0, alpha_max, n_samp)
    for a in alphas:
        h_try = _project(h + a * p, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_a = a
    if best_a in (0.0, alpha_max):
        return best_a, best_c5
    lo = alphas[max(0, alphas.tolist().index(best_a) - 1)]
    hi = alphas[min(n_samp - 1, alphas.tolist().index(best_a) + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(14):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(_project(h + opt_a * p, target))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            continue
        alpha, new_c5 = _line_search(h, p, target)
        if new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 10 and radius < 0.05:
                radius = 0.1
                ti = 0
    return best_h, best_c5

def _kkt_grad_step(h, deadline, max_iters=2000, gamma_start=10.0, gamma_end=200.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        frac = min(1.0, it / max(1, max_iters - 1))
        gamma = gamma_start * (gamma_end / gamma_start) ** frac
        scores = _scores(h)
        max_s = float(np.max(scores))
        w = np.exp(gamma * (scores - max_s))
        sw = float(np.sum(w))
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.05 / (1.0 + 0.5 * frac)
        improved = False
        for _ in range(25):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                stall = 0
                break
            alpha *= 0.7
            if alpha < 1e-12:
                break
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng, noise=0.0):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    if noise > 0:
        h += rng.normal(0, noise, n)
    return _project(h, n / 2.0)

def _binary_init(n, rng):
    h = np.zeros(n)
    pos = rng.permutation(n)
    h[pos[:n // 2]] = 1.0
    return h

def _block_init(n, rng, n_blocks=4):
    h = np.zeros(n)
    blocks = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            h[b * blocks:(b + 1) * blocks] = 1.0
    h += rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = max(1, n // 32)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, step):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(0, n, step):
                if k == j or (k - j) % step < 2:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in (1, -1):
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(6):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _fourier_evolve(h, deadline, n_pop=20, K=8, n_gen=200, mut_scale=0.12, rng=None):
    n = len(h)
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2 * (k - 1)] = np.cos(2 * np.pi * k * x)
        B[:, 2 * (k - 1) + 1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    target = n / 2.0
    a_init = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    pop_a = a_init + mut_scale * rng.standard_normal((n_pop, 2 * K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        h_try = _project(np.clip(0.5 + B @ pop_a[i], 0, 1), target)
        pop_c5[i] = _exact_c5(h_try)
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = _project(np.clip(0.5 + B @ pop_a[best_i], 0, 1), target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        n_keep = max(2, n_pop // 2)
        pop_a[:n_keep] = pop_a[order[:n_keep]]
        pop_c5[:n_keep] = pop_c5[order[:n_keep]]
        for i in range(n_keep, n_pop):
            p1 = order[rng.integers(0, n_keep)]
            p2 = order[rng.integers(0, n_keep)]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1.0 - mix) * pop_a[order[p2]]
            pop_a[i] += mut_scale * rng.standard_normal(2 * K)
            h_try = _project(np.clip(0.5 + B @ pop_a[i], 0, 1), target)
            pop_c5[i] = _exact_c5(h_try)
        mut_scale *= 0.996
        cur_best = int(np.argmin(pop_c5))
        if pop_c5[cur_best] < best_c5:
            best_c5 = pop_c5[cur_best]
            best_h = _project(np.clip(0.5 + B @ pop_a[cur_best], 0, 1), target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    if best_c5 < 0.3808:
        return best_h, best_c5, best_n

    # Phase 1: Quick LP from parent (8%)
    dl1 = t0 + budget_s * 0.08
    h1, c5_1 = _local_lp(h_start, dl1, top_k_list=[None, 16, 10],
                          radius_init=0.15, max_iters=500)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Aggressive multi-resolution LP search (35%)
    dl2 = t0 + budget_s * 0.43
    resolutions = [48, 56, 64, 72, 80]
    alphas_list = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.6614378278]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl2:
            break
        if best_c5 < 0.3806:
            break
        n_starts = 4
        for s in range(n_starts):
            if time.time() > dl2:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                alpha = alphas_list[(ri + s) % len(alphas_list)]
                h_init = _quasi_init(n_try, alpha, rng, noise=0.06)
            elif s == 2:
                h_init = _binary_init(n_try, rng)
            else:
                h_init = _block_init(n_try, rng, n_blocks=ri % 3 + 3)
            sub_dl = min(dl2, time.time() + budget_s * 0.04)
            h_try, c5_try = _local_lp(h_init, sub_dl,
                                       top_k_list=[None, 12, 8],
                                       radius_init=0.25, max_iters=8000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                sub_dl2 = min(dl2, time.time() + budget_s * 0.015)
                h_k, c5_k = _kkt_grad_step(h_try, sub_dl2, max_iters=500)
                if c5_k < best_c5:
                    best_c5 = c5_k
                    best_h = h_k.copy()
                    best_n = n_try

    # Phase 3: Upsample best and refine (12%)
    if time.time() < t0 + budget_s * 0.55:
        dl3 = t0 + budget_s * 0.55
        for n_up in [96, 112, 128]:
            if time.time() > dl3:
                break
            if n_up == best_n:
                continue
            h_up = _resample(best_h, n_up)
            sub_dl = min(dl3, time.time() + budget_s * 0.03)
            h_up2, c5_up = _local_lp(h_up, sub_dl,
                                      top_k_list=[None, 20, 12, 8],
                                      radius_init=0.12, max_iters=2000)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up2.copy()
                best_n = n_up

    # Phase 4: Fourier exploration (15%)
    if time.time() < t0 + budget_s * 0.70:
        dl4 = t0 + budget_s * 0.70
        n_f = best_n if best_n <= 96 else 80
        h_f = _resample(best_h, n_f) if len(best_h) != n_f else best_h
        h_f2, c5_f = _fourier_evolve(h_f, dl4, n_pop=24, K=10,
                                      n_gen=300, mut_scale=0.10, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f2.copy()
            best_n = n_f
        if time.time() < dl4:
            sub_dl = min(dl4, time.time() + budget_s * 0.02)
            h_fp, c5_fp = _local_lp(h_f2, sub_dl,
                                     top_k_list=[10, 6, None],
                                     radius_init=0.08, max_iters=500)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 5: KKT refinement at best resolution (12%)
    if time.time() < t0 + budget_s * 0.82:
        dl5 = t0 + budget_s * 0.82
        h_k, c5_k = _kkt_grad_step(best_h, dl5, max_iters=3000,
                                    gamma_start=8.0, gamma_end=300.0)
        if c5_k < best_c5:
            best_c5 = c5_k
            best_h = h_k.copy()
        if time.time() < dl5 and best_c5 < 0.3808:
            sub_dl = min(dl5, time.time() + budget_s * 0.03)
            h_lp2, c5_lp2 = _local_lp(best_h, sub_dl,
                                       top_k_list=[8, 6, None],
                                       radius_init=0.06, max_iters=3000)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h_lp2.copy()

    # Phase 6: Perturbation restarts (10%)
    if time.time() < t0 + budget_s * 0.92:
        dl6 = t0 + budget_s * 0.92
        scale = 0.02
        for _ in range(80):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.012)
            h_try, c5_try = _local_lp(h_pert, sub_dl,
                                       top_k_list=[6, 4, None],
                                       radius_init=0.05, max_iters=400)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.003, scale * 0.8)
            else:
                scale = min(0.08, scale * 1.15)

    # Phase 7: Pair polish (5%)
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, dl7, max_passes=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: tiny LP refinement
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _local_lp(best_h, t0 + budget_s * 0.998,
                                   top_k_list=[4, 6, 5],
                                   radius_init=0.01, max_iters=300)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
