One important detail: the `_gradient_lse` function needs to correctly compute the gradient. Let me verify the formula.

The objective is C(h) = max_d F_d(h) where F_d(h) = dx * sum_{valid i} h[i] * (1-h[i+d]).

The log-sum-exp approximation is: LSE(F, β) = (1/β) ln(sum_d exp(β F_d))

The gradient is: ∇LSE = sum_d w_d * ∇F_d where w_d = exp(βF_d) / sum_e exp(βF_e)

For F_d = dx * sum_{i in valid range for lag d} h[i] * (1-h[i+d]):

∂F_d/∂h[k] = dx * [(1-h[k+d]) * 1(k+d valid) + (-h[k-d]) * 1(k-d valid)]

Wait, I need to be careful. F_d involves pairs (i, i+d) where both i and i+d are in [0, n).

For a fixed k, k appears in F_d as:
- The "first" element: when i = k and i+d = k+d is valid. Contribution: h[k]*(1-h[k+d]), derivative w.r.t. h[k] is (1-h[k+d])*dx.
- The "second" element: when i+d = k, i.e., i = k-d, and i = k-d is valid. Contribution: h[k-d]*(1-h[k]), derivative w.r.t. h[k] is -h[k-d]*dx.

So: ∂F_d/∂h[k] = dx * [(1-h[k+d]) * 1(0≤k+d<n) - h[k-d] * 1(0≤k-d<n)]

And the full gradient: ∇LSE[k] = sum_d w_d * dx * [(1-h[k+d])*1(0≤k+d<n) - h[k-d]*1(0≤k-d<n)]

This can be computed as a loop over d, or more efficiently using correlation/convolution.

            best_h = x.copy()
        v = np.zeros(n)
        alpha = 0.04
        stall = 0
        for it in range(2500):
            if time.time() > deadline:
                break
            if stall > 70:
                break
            g, F, w = _grad_lse(x, beta)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
            direction = -g / gn
            best_trial = None
            best_tc5 = cur_c5
            for a in [alpha, alpha * 0.6, alpha * 0.3, alpha * 0.1, alpha * 1.5, alpha * 3.0]:
                trial = _project_box_sum(x + a * direction, n / 2.0)
                tc5 = _exact_c5(trial)
                if tc5 < best_tc5 - 1e-13:
                    best_tc5 = tc5
                    best_trial = trial
            if best_trial is not None:
                v = 0.65 * v + 0.35 * (best_trial - x)
                x = best_trial
                cur_c5 = best_tc5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = x.copy()
                stall = 0
                alpha = min(alpha * 1.04, 0.8)
            else:
                v *= 0.4
                stall += 1
                alpha *= 0.65
                if alpha < 1e-9:
                    break
        h = _project_box_sum(x, n / 2.0)
    return best_h, best_c5

def _pair_search(h, deadline, rng, max_iters=60000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    idx = np.arange(n)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 4000:
            break
        i = int(idx[rng.integers(n)])
        j = int(idx[rng.integers(n)])
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        if hi <= 1e-9 or hi >= 1.0 - 1e-9 or hj <= 1e-9 or hj >= 1.0 - 1e-9:
            no_improve += 1
            continue
        t_low = max(-hi, hj - 1.0)
        t_high = min(1.0 - hi, hj)
        if t_high - t_low < 1e-12:
            continue
        best_t = 0.0
        best_tc5 = cur_c5
        samples = np.linspace(t_low, t_high, 11)
        for t in samples:
            h[i] = hi + t
            h[j] = hj - t
            tc5 = _exact_c5(h)
            if tc5 < best_tc5 - 1e-14:
                best_tc5 = tc5
                best_t = t
        if abs(best_t) > 1e-12:
            lo2 = max(t_low, best_t - (t_high - t_low) * 0.1)
            hi2 = min(t_high, best_t + (t_high - t_low) * 0.1)
            for t in np.linspace(lo2, hi2, 15):
                h[i] = hi + t
                h[j] = hj - t
                tc5 = _exact_c5(h)
                if tc5 < best_tc5 - 1e-14:
                    best_tc5 = tc5
                    best_t = t
            h[i] = hi + best_t
            h[j] = hj - best_t
            cur_c5 = best_tc5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
        else:
            h[i] = hi
            h[j] = hj
            no_improve += 1
    return best_h, best_c5

def _single_coord_search(h, deadline, rng, max_iters=80000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, -0.02, -0.01, -0.005, -0.002, -0.001]
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 6000:
            break
        i = int(rng.integers(n))
        d = float(rng.choice(deltas))
        new_val = h[i] + d
        if new_val < 0.0 or new_val > 1.0:
            no_improve += 1
            continue
        h[i] = new_val
        c5_new = _exact_c5(h)
        if c5_new < cur_c5 - 1e-14:
            h = _project_box_sum(h, n / 2.0)
            c5_new = _exact_c5(h)
            if c5_new < cur_c5 - 1e-14:
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_improve = 0
            else:
                h[i] = new_val
        else:
            h[i] = h[i] - d
            no_improve += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _binary_start(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _smooth_start(n, rng):
    x = (np.arange(n) + 0.5) / n
    h = 0.5 + 0.4 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 2 * np.pi))
    h = np.clip(h, 0.0, 1.0)
    return _project_box_sum(h, n / 2.0)

def _alternating_structure(n):
    h = np.zeros(n)
    period = rng_period = min(n, 8)
    for i in range(n):
        phase = (i % period) / period
        h[i] = 0.5 + 0.45 * np.cos(2 * np.pi * phase)
    h = np.clip(h, 0.0, 1.0)
    return _project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline_final = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        n0 = 128
        h0 = _binary_start(n0, rng)
    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LP active-set (most effective for fine tuning)
    dl1 = t0 + budget_s * 0.28
    h1, c5_1 = _lp_loop(h, dl1, radius_start=0.06)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothed momentum descent
    dl2 = t0 + budget_s * 0.40
    h2, c5_2 = _smoothed_momentum(best_h, dl2, beta_start=15.0, beta_end=4000.0, n_beta=10)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP again from improved point
    dl3 = t0 + budget_s * 0.52
    h3, c5_3 = _lp_loop(best_h, dl3, radius_start=0.04)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair search
    dl4 = t0 + budget_s * 0.62
    h4, c5_4 = _pair_search(best_h, dl4, rng, max_iters=40000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: More LP
    dl5 = t0 + budget_s * 0.68
    h5, c5_5 = _lp_loop(best_h, dl5, radius_start=0.03)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Try alternative starts and resolutions
    dl6 = t0 + budget_s * 0.82
    n_candidates = sorted(set([max(48, best_n - 16), max(48, best_n - 8), best_n,
                                best_n + 8, best_n + 16, best_n + 32]))
    n_candidates = [x for x in n_candidates if 48 <= x <= 256]
    per_n = budget_s * 0.02
    # Also try a couple of novel starts
    starts = [best_h.copy()]
    if n0 >= 64:
        starts.append(_binary_start(n0, rng))
        starts.append(_smooth_start(n0, rng))

    for start_h in starts[:3]:
        if time.time() > dl6:
            break
        for n_try in n_candidates:
            if time.time() > dl6:
                break
            if n_try == best_n and start_h is best_h:
                continue
            h_r = _resample(start_h, n_try)
            dl_r = min(dl6, time.time() + per_n)
            h_r, c5_r = _lp_loop(h_r, dl_r, radius_start=0.05)
            h_r, c5_r = _pair_search(h_r, min(dl6, time.time() + per_n * 0.5), rng, max_iters=5000)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 7: Final LP polish
    if time.time() < dl_final - 30:
        h7, c5_7 = _lp_loop(best_h, min(dl_final - 15, time.time() + budget_s * 0.06), radius_start=0.02)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final pair search
    if time.time() < dl_final - 10:
        h8, c5_8 = _pair_search(best_h, min(dl_final - 5, time.time() + budget_s * 0.03), rng, max_iters=30000)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: Single coordinate final
    if time.time() < dl_final - 5:
        h9, c5_9 = _single_coord_search(best_h, min(dl_final - 2, time.time() + budget_s * 0.02), rng, max_iters=50000)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final validation and projection
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        # Revert to pre-projection if projection hurt
        best_c5 = min(best_c5, final_c5)

    # Safety: ensure feasibility
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _exact_c5(best_h)

    # Verify sum constraint
    s = float(np.sum(best_h))
    if abs(s - best_n / 2.0) > 1e-10:
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _exact_c5(best_h)

    return best_h.tolist(), best_c5, best_n
