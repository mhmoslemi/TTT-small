```python
import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _weighted_corr_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    grad += weights[n - 1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[n - 1 + d]
        wn = weights[n - 1 - d]
        grad[d:] += wp * (1.0 - h[:n - d])
        grad[:n - d] -= wp * h[d:]
        grad[:n - d] += wn * (1.0 - h[d:])
        grad[d:] -= wn * h[:n - d]
    return grad


def _lse_val_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    t = (corr - m) / tau
    exps = np.exp(t)
    s = float(np.sum(exps))
    val = (m + tau * np.log(s)) * dx
    w = exps / s
    grad = _weighted_corr_grad(h, w) * dx
    return val, grad


def _slsqp_lse_obj(x, tau):
    h = np.clip(x, 0.0, 1.0)
    val, _ = _lse_val_grad(h, tau)
    return val


def _slsqp_lse_grad(x, tau):
    h = np.clip(x, 0.0, 1.0)
    _, g = _lse_val_grad(h, tau)
    return g


def _eq_fun(x, n):
    return x.sum() - n / 2.0


def _eq_jac(x, n):
    return np.ones(n)


def _solve_lse_slsqp(h0, tau, maxiter=300, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_slsqp_lse_obj, h0.copy(), jac=_slsqp_lse_grad, args=(tau,),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project(np.clip(res.x, 0.0, 1.0), n / 2.0)
    except Exception:
        pass
    return h0.copy()


def _binary_project(h):
    n = len(h)
    k = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:k]] = 1.0
    return h_bin


def _binary_greedy(h_bin, max_passes=200, deadline=None):
    n = len(h_bin)
    h = h_bin.copy()
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0.0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5_t = _exact_c5(h)
                h[i] = 1
                h[j] = 0
                if c5_t < best_delta:
                    best_delta = c5_t
                    best_pair = (i, j)
        if best_pair is not None and best_delta < best_c5 - 1e-12:
            h[best_pair[0]] = 0
            h[best_pair[1]] = 1
            best_c5 = best_delta
            best_h = h.copy()
        else:
            break
    return best_h, best_c5
