    dx = 2.0 / n

    best_h = h.copy()

            break

        max_score = float(np.max(scores))
        # Find active rows (within tolerance)
        active_tol = 1e-6
        active_rows = np.where(scores > max_score - active_tol)[0]

        # For the max, the subgradient is a convex combination of gradients of active rows
        # Use uniform weighting over active rows
        G = np.zeros(n)
        for r in active_rows:
            G += _active_grad(h, int(r), n)
        G /= len(active_rows)
        
        # Find best pair: i with most positive G, j with most negative G
        # (move mass from high G to low G)
        pos_candidates = np.where((h > 0.01) & (G > 0))[0]
        neg_candidates = np.where((h < 0.99) & (G < 0))[0]
        
        if len(pos_candidates) == 0 or len(neg_candidates) == 0:
            break
        
        i = pos_candidates[np.argmax(G[pos_candidates])]
        j = neg_candidates[np.argmin(G[neg_candidates])]
        
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, c5_try = _line_search(h, d, dx, min(h[i], 1-h[j]) * 0.99)

            d[j] = -1.0

                # Remove i from consideration for next step (or try next best)
                pass
