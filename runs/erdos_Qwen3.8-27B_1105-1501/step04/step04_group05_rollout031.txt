    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = [j for j in range(n) if 1e-4 < h[j] < 1.0 - 1e-4]
        for j in interior:
            if time.time() > deadline:
                break
            if not (1e-4 <= h[j] <= 1.0 - 1e-4):
                continue
            for k in interior:
                if k == j:
                    continue
                if not (1e-4 <= h[k] <= 1.0 - 1e-4):
                    continue
                a_max = min(1.0 - h[j], h[k])
                if a_max < 1e-8:
                    continue
                for frac in [0.5, 1.0, 0.25]:
                    a = frac * a_max
                    ht = h.copy()
                    ht[j] += a
                    ht[k] -= a
                    ht = np.clip(ht, 0.0, 1.0)
                    ht = _project(ht, target)
                    c5_t = _exact_c5(ht)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = ht.copy()
                        h = ht
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def _accelerated_smooth_descent(h, tau_list, deadline, max_steps=300):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for tau in tau_list:
        if time.time() > deadline:
            break
        m = np.zeros(n)
        h_cur = h.copy()
        alpha = 0.01
        for step in range(max_steps):
            if time.time() > deadline:
                break
            y = h_cur + m
            grad = _smooth_grad(y, tau)
            h_new = _project(h_cur - alpha * grad, target)
            m = 0.9 * m - alpha * grad
            c5_new = _exact_c5(h_new)
            if c5_new < _exact_c5(h_cur) - 1e-13 or step < 5:
                h_cur = h_new
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
            else:
                h_cur = _project(h_cur + 0.5 * alpha * grad, target)
            if step % 25 == 24:
                alpha = min(alpha * 1.2, 0.03)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Smooth log-sum-exp descent (40% budget)
    d1 = t0 + budget_s * 0.40
    tau_list = [0.06, 0.04, 0.025, 0.015, 0.010, 0.007, 0.005, 0.003]
    h_s, c5_s = _smooth_descent(best_h, tau_list[:4], d1, max_steps=500, alpha_init=0.006)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()

    # Phase 1b: Continuation with smaller tau
    if time.time() < d1:
        h_s2, c5_s2 = _smooth_descent(best_h, tau_list[4:], d1, max_steps=400, alpha_init=0.004)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()

    # Phase 2: SLSQP active-set refinement (15% budget)
    d2 = t0 + budget_s * 0.55
    for iteration in range(50):
        if time.time() > d2:
            break
        c_full = _corr_vec(best_h)
        frac = max(0.25 - 0.05 * iteration / 30.0, 0.1)
        active = _get_active(best_h, frac=frac, min_count=6)
        h_new = _slsqp_minmax(best_h, active, np.max(c_full), maxiter=250)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            active2 = _get_active(best_h, frac=0.08, min_count=15)
            h_new2 = _slsqp_minmax(best_h, active2, np.max(c_full), maxiter=200)
            c5_new2 = _exact_c5(h_new2)
            if c5_new2 < best_c5 - 1e-12:
                best_c5 = c5_new2
                best_h = h_new2.copy()
            else:
                break

    # Phase 3: Gradient loop (10% budget)
    d3 = t0 + budget_s * 0.65
    if time.time() < d3:
        hg, c5g = _grad_loop(best_h, d3, start_lr=0.003, min_lr=1e-6, top_k=4, patience=40)
        if c5g < best_c5:
            best_c5 = c5g
            best_h = hg.copy()

    # Phase 4: Multi-start with smooth descent (15% budget)
    d4 = t0 + budget_s * 0.80
    scale = 0.03
    tau_small = [0.015, 0.008, 0.004]
    for attempt in range(200):
        if time.time() > d4:
            break
        if best_c5 < 0.3806:
            break
        # Various perturbation strategies
        if attempt % 3 == 0:
            h_p = best_h + rng.normal(0, scale, n)
        elif attempt % 3 == 1:
            # Complement symmetry perturbation
            h_p = 1.0 - best_h + rng.normal(0, scale * 0.5, n)
        else:
            # Directional perturbation along top gradient
            h_p = _gradient_step(best_h, scale * 10, top_k=5) + rng.normal(0, scale * 0.3, n)
        h_p = _project(h_p, target)
        c5_p = _exact_c5(h_p)
        if c5_p < best_c5 + 0.005:  # Only refine if close
            h_r, c5_r = _smooth_descent(h_p, tau_small, min(d4, time.time() + 15), max_steps=100, alpha_init=0.004)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                scale = max(0.008, scale * 0.7)
            else:
                scale = min(0.12, scale * 1.15)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 5: Resolution search (10% budget)
    d5 = t0 + budget_s * 0.90
    try_n_values = [n + 8, n - 8, n + 16, n - 16, n + 32, n + 64, 48, 80, 96, 128]
    seen_n = set()
    for n_try in try_n_values:
        if time.time() > d5:
            break
        if n_try < 20 or n_try > 200 or n_try in seen_n or n_try == n:
            continue
        seen_n.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r_init = _exact_c5(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        # Quick smooth descent at new resolution
        tau_quick = [0.03, 0.015, 0.008]
        h_r2, c5_r2 = _smooth_descent(h_r, tau_quick, sub_dl, max_steps=80, alpha_init=0.005)
        if c5_r2 < best_c5 - 1e-4:
            # Polish with SLSQP
            active = _get_active(h_r2, frac=0.3, min_count=8)
            h_r3 = _slsqp_minmax(h_r2, active, np.max(_corr_vec(h_r2)), maxiter=200)
            c5_r3 = _exact_c5(h_r3)
            if c5_r3 < best_c5 - 1e-4:
                best_c5 = c5_r3
                best_h = h_r3.copy()
                n = n_try
                target = n / 2.0

    # Phase 6: Pair refine (4% budget)
    d6 = t0 + budget_s * 0.94
    if time.time() < d6:
        hp, c5p = _pair_refine(best_h, d6, max_passes=100)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 7: Final polish (4% budget)
    d7 = deadline
    if time.time() < d7:
        active = _get_active(best_h, frac=0.5, min_count=12)
        h_f = _slsqp_minmax(best_h, active, np.max(_corr_vec(best_h)), maxiter=400)
        c5_f = _exact_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if time.time() < d7:
            hg2, c5g2 = _grad_loop(best_h, d7, start_lr=0.002, min_lr=1e-7, top_k=3, patience=25)
            if c5g2 < best_c5:
                best_c5 = c5g2
                best_h = hg2.copy()

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, n_final
