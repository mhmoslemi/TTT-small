```python
    n = len(v)

    n = n_total
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(v, target):
    n = len(v)
    t = min(max(target, 0.0), float(n))
    if t <= 0:
        return np.zeros(n)
    if t >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > t:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def weighted_grad(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = weights[q]
        if w < 1e-12:
            continue
        d = q - (n - 1)
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            if n - d > 0:
                grad[d:] += w * dx * (1.0 - h[:n-d])
                grad[:n-d] -= w * dx * h[d:]
        else:
            k = -d
            if n - k > 0:
                grad[:n-k] += w * dx * (1.0 - h[k:])
                grad[k:] -= w * dx * h[:n-k]
    return grad

def lp_vertex(w, target):
    n = len(w)
    order = np.argsort(w)
    s = np.zeros(n)
    nf = int(target)
    frac = target - nf
    s[order[:nf]] = 1.0
    if frac > 1e-12 and nf < n:
        s[order[nf]] = frac
    return s

def fw_step(h, corr_vals, n, top_k=6, eta_frac=0.05):
    target = n / 2.0
    lam = float(np.max(corr_vals))
    eta = max(eta_frac * lam, 1e-5)
    
    sorted_idx = np.argsort(corr_vals)[::-1][:top_k]
    top_vals = corr_vals[sorted_idx]
    w = np.exp((top_vals - top_vals[0]) / eta)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(sorted_idx)):
        grad += w[i] * single_grad(h, sorted_idx[i], n)
    
    s = lp_vertex(grad, target)
    best_alpha = 0.0
    best_val = compute_c5(h)
    for alpha in [1.0, 0.5, 0.25, 0.75, 0.125]:
        h_try = (1.0 - alpha) * h + alpha * s
        val = compute_c5(h_try)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    if best_alpha == 0.0:
        return h, best_val
    return project_feasible((1.0 - best_alpha) * h + best_alpha * s, target), best_val

def single_grad(h, q, n):
    dx = 2.0 / n
    g = np.zeros(n)
    d = q - (n - 1)
    if d == 0:
        g = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
    else:
        k = -d
        g[:n-k] += dx * (1.0 - h[k:])
        g[k:] -= dx * h[:n-k]
    return g
```

```python
def optimize(h, n, deadline):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    
    # Temperature schedule: start high, decay to low
    mu_max = 0.15
    mu_min = 0.003
    lr = 0.5
    momentum = np.zeros(n)
    beta = 0.9
    
    it = 0
    stall = 0
    
    while time.time() < deadline:
        it += 1
        if stall > 200:
            break
        
        corr = compute_corr(h_cur)
        c5_cur = float(np.max(corr))
        
        # Adaptive temperature
        progress = min(1.0, it / 5000.0)
        mu = mu_max * (mu_min / mu_max) ** progress
        # Or: mu = max(mu_min, mu_max * 0.98**it)
        
        # Softmax weights
        shifted = corr - c5_cur
        weights = np.exp(shifted / mu)
        weights /= weights.sum()
        
        # Weighted gradient
        grad = weighted_grad(h_cur, weights, n)
        grad -= np.mean(grad)  # preserve sum
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        direction = -momentum
        direction -= np.mean(direction)
        
        # Backtracking line search
        improved = False
        for _ in range(50):
            h_new = project_feasible(h_cur + lr * direction, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-13:
                h_cur = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                stall = 0
                lr = min(lr * 1.1, 2.0)
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            stall += 1
            lr = max(lr * 0.8, 1e-6)
            if lr < 1e-6:
                break
        
        # Periodic FW step
        if it % 100 == 0 and time.time() < deadline - 10:
            h_fw, c5_fw = fw_step(h_cur, corr, n, top_k=min(8, 2*n-1), eta_frac=mu / c5_cur)
            if c5_fw < best_c5:
                best_c5 = c5_fw
                best_h = h_fw.copy()
                h_cur = h_fw.copy()
                corr = compute_corr(h_cur)
    
    return best_h, best_c5
```

```python
def pair_optimal_split(h, i, j, n):
    """Find optimal h[i], h[j] given h[i]+h[j]=const, all else fixed."""
    c = h[i] + h[j]
    # t = h[i], so h[j] = c - t
    # t range: max(0, c-1) <= t <= min(1, c)
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-10:
        return h[i], h[j]
    
    dx = 2.0 / n
    # For each lag q, C_q is linear in t
    # C_q(t) = a_q * t + b_q (where a_q, b_q depend on other h values)
    # We need to find t that minimizes max_q (a_q * t + b_q)
    
    # Compute a_q and b_q for each q
    # C_q = dx * sum_{valid} h[p](1-h[r]) where p, r depend on q and the indices
    # When we change h[i] -> t and h[j] -> c-t:
    # The change in C_q is: dx * [coefficient_of_t_q * t - coefficient_of_c_minus_t_q * (c-t)]
    
    # Actually, let's just evaluate C_q at t=lo and t=hi, then the linear functions
    # are determined. The max of linear functions is minimized where the
    # "upper envelope" has a minimum.
    
    # More efficient: evaluate C_q at a few candidate t values and use the
    # geometry of the upper envelope.
    
    # Simple approach: ternary search on t (since max of affine is convex)
    best_t = (lo + hi) / 2
    best_val = float('inf')
    
    # Sample at multiple points
    for _ in range(40):
        mid = (lo + hi) / 2
        # Evaluate at lo, mid, hi
        h_lo = h.copy(); h_lo[i] = lo; h_lo[j] = c - lo
        h_mid = h.copy(); h_mid[i] = mid; h_mid[j] = c - mid
        h_hi = h.copy(); h_hi[i] = hi; h_hi[j] = c - hi
        
        v_lo = compute_c5(h_lo)
        v_mid = compute_c5(h_mid)
        v_hi = compute_c5(h_hi)
        
        if v_mid < best_val:
            best_val = v_mid
            best_t = mid
        
        # Since max of affine is convex, use ternary search
        t1 = lo + (hi - lo) / 3
        t2 = hi - (hi - lo) / 3
        h_t1 = h.copy(); h_t1[i] = t1; h_t1[j] = c - t1
        h_t2 = h.copy(); h_t2[i] = t2; h_t2[j] = c - t2
        v1 = compute_c5(h_t1)
        v2 = compute_c5(h_t2)
        
        if v1 < v2:
            hi = t2
        else:
            lo = t1
        if hi - lo < 1e-8:
            break
    
    return best_t, c - best_t
```
