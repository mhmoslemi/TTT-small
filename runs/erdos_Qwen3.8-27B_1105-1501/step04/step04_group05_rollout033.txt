def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    """Project v onto {0<=x<=1, sum=x=target} using bisection on clipping."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    p = np.clip(v - mu, 0.0, 1.0)
    # Fix rounding
    residual = target - float(np.sum(p))
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - p)
        else:
            order = np.argsort(p)
        remaining = abs(residual)
        for idx in order:
            if remaining <= 1e-15:
                break
            if residual > 0:

                p[idx] += delta
                remaining -= delta
            else:
                delta = min(-remaining, p[idx])
                p[idx] -= delta
                remaining += delta
    return p

def _project(v, target):

At ~10⁸ ops/sec in numpy, that's about 1.2 seconds per pass. With 40 passes, that's about 48 seconds. Should fit in the budget.

Actually, the correlation is O(n) for a single lag (it's a dot product of length n), and there are 2n-1 lags, so one evaluation is O(n²) = O(4096) for n=64. But numpy.correlate is implemented in C and is very fast. Let me not worry too much about the exact timing.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    p = np.clip(v - mu, 0.0, 1.0)
    residual = target - float(np.sum(p))
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - p)
        else:
            order = np.argsort(p)
        remaining = residual
        for idx in order:
            if abs(remaining) <= 1e-15:
                break
            if remaining > 0:
                delta = min(remaining, 1.0 - p[idx])
                p[idx] += delta
                remaining -= delta
            else:
                delta = min(-remaining, p[idx])
                p[idx] -= delta
                remaining += delta
    return p

def _two_level(mask, d):
    return np.where(mask == 1, 1.0 - d, d)

            h_minus = _project(h_minus, target)

            fm = float(np.max(np.correlate(h_minus, 1-h_minus, mode='full')))
            grad[i] = (fp - fm) / (2*eps)
        # Remove mean to preserve sum
        grad -= np.mean(grad)
        # Line search
        for trial in range(20):

            h_new = _project(h_cur - step * grad, target)

                h_cur = h_new
                break
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_s
