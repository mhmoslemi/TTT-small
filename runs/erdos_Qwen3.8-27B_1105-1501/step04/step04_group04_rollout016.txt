import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0).copy()
    for _ in range(300):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _smooth_max_gradient(h, beta, active_k=None):
    """Compute gradient of log-sum-exp(beta * F) with respect to h"""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    F_max = np.max(F)
    exp_F = np.exp(beta * (F - F_max))
    w = exp_F / (np.sum(exp_F) + 1e-300)
    if active_k is not None:
        w[active_k] = 0.0  # mask some out? No, let's not
    # Gradient of F_k with respect to h
    # F_k = dx * sum_{i} h_i * (1 - h_{i+k}) for appropriate i
    # dF_k/dh_i = dx * (1 - h_{i+k}) - dx * h_{i-k} (when indices are valid)
    # Build gradient
    g = np.zeros(n)
    L = 2 * n - 1
    for r in range(L):
        if w[r] < 1e-15:
            continue
        k = r - (n - 1)  # lag
        low = max(0, -k)
        high = min(n, n - k)
        for i in range(low, high):
            j = i + k
            # d/dh_i: dx * (1 - h[j])
            g[i] += w[r] * dx * (1.0 - h[j])
            # d/dh_j: -dx * h[i]
            g[j] += w[r] * (-dx) * h[i]
    return g, w, F

def _smooth_max_gradient_fast(h, beta):
    """Faster version using the matrix structure"""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    F_max = np.max(F)
    exp_F = np.exp(beta * (F - F_max))
    w = exp_F / (np.sum(exp_F) + 1e-300)
    
    # Build gradient using the structure
    # For each lag k = r - (n-1):
    #   g[i] += w[r] * dx * (1 - h[i+k]) for valid i
    #   g[i+k] += w[r] * (-dx) * h[i] for valid i
    g = np.zeros(n)
    # Use convolution-like approach
    # g[i] = sum_r w[r] * dx * (term from h[i] as first) + sum_r w[r] * dx * (term from h[i] as second)
    # As first term: h[i]*(1-h[i+k]) -> derivative wrt h[i] is (1-h[i+k])
    #   contribution to g[i] from lag k: w[n-1+k] * dx * (1-h[i+k])
    # As second term: h[i-k]*(1-h[i]) -> derivative wrt h[i] is -h[i-k]
    #   contribution to g[i] from lag k: w[n-1-k] * dx * (-h[i-k])
    # So g[i] = dx * sum_k w[n-1+k]*(1-h[i+k]) + dx * sum_k w[n-1+k]*(-h[i-k])
    #          = dx * [conv(w, (1-h))[i] - conv(w, h)[i shifted]]
    
    # More carefully:
    # g[i] = dx * sum_{k: valid} w[n-1+k] * (1 - h[i+k]) - dx * sum_{k: valid} w[n-1-k] * h[i-k]
    
    # Let's use np.convolve or manual
    # w has length 2n-1, indexed 0..2n-2
    # w[n-1+k] for k = -(n-1)..(n-1) gives w[0]..w[2n-2]
    
    # First sum: sum_k w[n-1+k] * (1-h[i+k]) where i+k in [0,n)
    # This is like a cross-correlation of w with (1-h)
    # Second sum: sum_k w[n-1-k] * h[i-k] where i-k in [0,n)
    # This is like a cross-correlation of w with h (reversed w)
    
    ones_minus_h = 1.0 - h
    # First term: for each i, sum over k where 0 <= i+k < n: w[n-1+k] * (1-h[i+k])
    # Let j = i+k, so k = j-i, and 0 <= j < n, -(n-1) <= j-i <= n-1
    # w[n-1+j-i] * (1-h[j])
    # This is a convolution-like operation
    
    # I'll just do it with a loop over lags for clarity (n is at most ~128)
    L = 2 * n - 1
    for r in range(L):
        wr = w[r]
        if wr < 1e-16:
            continue
        k = r - (n - 1)
        low = max(0, -k)
        high = min(n, n - k)
        i = np.arange(low, high)
        j = i + k
        g[i] += wr * dx * (1.0 - h[j])
        g[j] += wr * dx * (-1.0) * h[i]
    return g, w, F

def _adam_smooth_max(h, deadline, beta_start=10.0, beta_end=5000.0, 
                     n_beta=20, lr=0.02, beta1=0.9, beta2=0.999,
                     max_inner=500, improve_tol=1e-13):
    """Adam optimizer on smooth-max objective"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    betas = np.geomspace(beta_start, beta_end, n_beta)
    
    for bi, beta in enumerate(betas):
        if time.time() > deadline:
            break
        m = np.zeros(n)  # first moment
        v = np.zeros(n)  # second moment
        t = 0
        stall = 0
        
        for inner in range(max_inner):
            if time.time() > deadline:
                break
            if stall > 50:
                break
            
            g, w, F = _smooth_max_gradient_fast(h, beta)
            # Subtract mean to handle sum constraint (tangent space)
            g = g - np.mean(g)
            
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g * g
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Try different step sizes
            best_trial_c5 = cur_c5
            best_factor = 0.0
            for factor in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                trial = _project_box_sum(h - factor * step, n / 2.0)
                tc5 = _exact_c5(trial)
                if tc5 < best_trial_c5 - improve_tol:
                    best_trial_c5 = tc5
                    best_factor = factor
            
            if best_factor > 0:
                h = _project_box_sum(h - best_factor * step, n / 2.0)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5 - improve_tol:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                lr *= 0.8
                if lr < 1e-6:
                    break
    
    return best_h, best_c5

def _binary_projection_and_repair(h, deadline, rng, max_iters=50000):
    """Try to find a good binary pattern, then locally optimize fractional values"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    # Start from current h, threshold to binary
    for thresh in [0.5, 0.3, 0.7, 0.4, 0.6]:
        if time.time() > deadline:
            break
        h_bin = (h >= thresh).astype(float)
        # Adjust to have exactly n/2 ones
        n_ones = int(np.sum(h_bin))
        if n_ones != n // 2:
            # Adjust
            if n_ones < n // 2:
                # Add ones at positions where h is largest
                zero_mask = h_bin == 0
                zero_indices = np.where(zero_mask)[0]
                h_sorted = h[zero_indices]
                add_count = n // 2 - n_ones
                if len(zero_indices) >= add_count:
                    top = np.argsort(h_sorted)[-add_count:]
                    h_bin[zero_indices[top]] = 1.0
            elif n_ones > n // 2:
                one_mask = h_bin == 1
                one_indices = np.where(one_mask)[0]
                h_sorted = h[one_indices]
                remove_count = n_ones - n // 2
                if len(one_indices) >= remove_count:
                    bottom = np.argsort(h_sorted)[:remove_count]
                    h_bin[one_indices[bottom]] = 0.0
        cur_c5 = _exact_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()
    
    # Now do local coordinate descent on the best found
    h = best_h.copy()
    cur_c5 = best_c5
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        # Try changing h[i]
        for delta in [-0.1, -0.05, 0.05, 0.1, -0.2, 0.2]:
            new_val = h[i] + delta
            if new_val < 0 or new_val > 1:
                continue
            h[i] = new_val
            h = _project_box_sum(h, n / 2.0)
            tc5 = _exact_c5(h)
            if tc5 < cur_c5 - 1e-13:
                cur_c5 = tc5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
            else:
                h[i] -= delta
                h = _project_box_sum(h, n / 2.0)
    
    return best_h, best_c5

def _pair_descent(h, deadline, rng, max_iters=100000):
    """Pair coordinate descent: move mass between pairs (i,j)"""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 2000:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = rng.uniform(-0.08, 0.08)
        if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
            no_improve += 1
            continue
        h[i] += d
        h[j] -= d
        tc5 = _exact_c5(h)
        if tc5 < cur_c5 - 1e-14:
            cur_c5 = tc5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            no_improve = 0
        else:
            h[i] -= d
            h[j] += d
            no_improve += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _lp_trust_step(h, radius):
    """LP trust region step using scipy"""
    from scipy.optimize import linprog
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Build linearized objective
    # F_k(x) ≈ F_k(h) + G_k[(x-h)]
    # where G_k[i] = dx*(1-h[i+k]) - dx*h[i-k] (with bounds)
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        G[r, j] += dx * (1.0 - h[i])
        G[r, i] += -dx * h[j]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_start=0.10, max_iters=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    radius = radius_start
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        x_new = _lp_trust_step(h, radius)
        if x_new is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        x_new = _project_box_sum(x_new, n / 2.0)
        best_step = 0.0
        best_trial_c5 = cur_c5
        for factor in [1.0, 0.7, 0.5, 0.3, 0.15]:
            trial = _project_box_sum(h + factor * (x_new - h), n / 2.0)
            tc5 = _exact_c5(trial)
            if tc5 < best_trial_c5 - 1e-13:
                best_trial_c5 = tc5
                best_step = factor
        if best_step > 0:
            h = _project_box_sum(h + best_step * (x_new - h), n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.2)
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5

def _structured_initialization(n, seed):
    """Create a structured initial pattern"""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    # Use a quasi-random pattern
    golden = 0.6180339887498949
    positions = sorted(((i * golden) % 1.0 for i in range(n)))
    # Map to binary: first n/2 positions get 1
    for i in range(n // 2):
        h[int(positions[i] * n) % n] = 1.0
    # Add some noise

def _exact_c5(h):

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0).copy()
    for _ in range(300):

        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    F_max = np.max(F)

        i = np.arange(low, high)
        j = i + k
        g[i] += wr * dx * (1.0 - h[j])
        g[j] -= wr * dx * h[i]
    g -= np.mean(g)

def _lp_trust_step(h, radius):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        G[r, j] += dx * (1.0 - h[i])
        G[r, i] += -dx * h[j]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: L-BFGS multi-beta
    deadline3 = t0 + budget_s * 0.50
    h3, c5_3 = _multi_beta_lbfgs(best_h, deadline3, beta_list=[80, 200, 500, 1500])
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP refinement
    deadline4 = t0 + budget_s * 0.58
    h4, c5_4 = _lp_loop(best_h, deadline4, radius_start=0.05)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: High-beta Adam
    deadline5 = t0 + budget_s * 0.66
    h5, c5_5 = _adam_smooth_max(best_h, deadline5, beta_start=100.0,
                                 beta_end=15000.0, n_beta=12, lr=0.012,
                                 b1=0.9, b2=0.999, max_inner=200)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pair descent
    deadline6 = t0 + budget_s * 0.74
    h6, c5_6 = _pair_descent(best_h, deadline6, rng, max_iters=200000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Multi-resolution
    deadline7 = t0 + budget_s * 0.86
    n_tries = [48, 56, 64, 72, 80, 96, 112, 128, 144, 160]
    n_tries = [x for x in n_tries if 24 <= x <= 160 and x != best_n]
    n_tries.sort(key=lambda x: abs(x - best_n))
    per_n_time = budget_s * 0.018
    for n_try in n_tries:
        if time.time() > deadline7:
            break
        h_r = _resample(best_h, n_try)
        dl_r = min(deadline7, time.time() + per_n_time)
        hr, cr = _adam_smooth_max(h_r, dl_r, beta_start=20.0,
                                   beta_end=2000.0, n_beta=8, lr=0.02, max_inner=80)
        if cr < best_c5 - 1e-10:
            best_c5 = cr
            best_h = hr.copy()
            best_n = n_try
            # Quick LP polish
            if time.time() < deadline7:
                hp, cp = _lp_loop(best_h, min(deadline7, time.time() + per_n_time * 0.4),
                                   radius_start=0.04)
                if cp < best_c5:
                    best_c5 = cp
                    best_h = hp.copy()

    # Phase 8: Random restarts
    deadline8 = t0 + budget_s * 0.93
    h8, c5_8 = _random_restart_optimize(best_h, deadline8, rng, n_restarts=4)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Phase 9: Final LP
    if time.time() < t0 + budget_s * 0.96:
        h_f, c5_f = _lp_loop(best_h, t0 + budget_s * 0.975, radius_start=0.025)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 10: Final pair
    if time.time() < t0 + budget_s * 0.98:
        h_l, c5_l = _pair_descent(best_h, t0 + budget_s * 0.99, rng, max_iters=30000)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l.copy()

    # Final validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        pass
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _exact_c5(best_h)

    return best_h, best_c5, best_n
