    return float(np.max(corr) * dx)

def _project(v, target):

import numpy as np
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _two_level_vec(d, mask):
    n = len(mask)
    return np.where(mask == 1, 1.0 - d, d)


def _optimize_d(mask, n, deadline):
    """Find optimal d for two-level vector with given mask."""
    best_d = 0.5
    best_s = _score(_two_level_vec(0.5, mask))
    # Coarse
    for d in np.linspace(0.0, 0.5, 101):
        s = _score(_two_level_vec(d, mask))
        if s < best_s:
            best_s = s
            best_d = d
    # Fine
    lo = max(0.0, best_d - 0.005)
    hi = min(0.5, best_d + 0.005)
    for d in np.linspace(lo, hi, 201):
        s = _score(_two_level_vec(d, mask))
        if s < best_s:
            best_s = s
            best_d = d
    # Ultra-fine
    lo = max(0.0, best_d - 0.0005)
    hi = min(0.5, best_d + 0.0005)
    for d in np.linspace(lo, hi, 201):
        s = _score(_two_level_vec(d, mask))
        if s < best_s:
            best_s = s
            best_d = d
    h = _two_level_vec(best_d, mask)
    h = _project(h, n / 2.0)
    return h, _score(h)


def _mask_score(mask):
    h = np.where(mask == 1, 1.0, 0.0)
    return _score(h)


def _binary_local_search(mask, n, deadline, max_iters=10000):
    """Best-improvement local search on binary masks."""
    mask = mask.copy().astype(np.int8)
    ones = np.where(mask == 1)[0]
    zeros = np.where(mask == 0)[0]
    cur_s = _mask_score(mask)
    best_mask = mask.copy()
    best_s = cur_s
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 300:
            break
        improved = False
        best_gain = 0.0
        best_i = -1
        best_j = -1

        # Adaptive: try all pairs if fast enough, else sample
        if len(ones) * len(zeros) < 5000:
            i_list = ones
            j_list = zeros
        else:
            i_list = ones[:min(20, len(ones))]
            j_list = zeros[:min(20, len(zeros))]

        for i in i_list:
            if time.time() > deadline:
                break
            mask[i] = 0
            for j in j_list:
                if time.time() > deadline:
                    break
                if mask[j] == 1:
                    continue
                mask[j] = 1
                s = _mask_score(mask)
                gain = cur_s - s
                if gain > best_gain + 1e-15:
                    best_gain = gain
                    best_i = i
                    best_j = j
                mask[j] = 0
            mask[i] = 1

        if best_i >= 0:
            mask[best_i] = 0
            mask[best_j] = 1
            cur_s -= best_gain
            ones = np.where(mask == 1)[0]
            zeros = np.where(mask == 0)[0]
            no_improve = 0
            if cur_s < best_s:
                best_s = cur_s
                best_mask = mask.copy()
        else:
            no_improve += 1

    h = np.where(best_mask == 1, 1.0, 0.0)
    return h, _score(h)


def _binary_sa(mask, n, deadline, seed_sa=777):
    """Simulated annealing on binary masks."""
    rng = np.random.default_rng(seed_sa)
    mask = mask.copy().astype(np.int8)
    ones = np.where(mask == 1)[0]
    zeros = np.where(mask == 0)[0]
    cur_s = _mask_score(mask)
    best_mask = mask.copy()
    best_s = cur_s

    T = 0.3
    T_min = 0.005
    cooling = 0.9997

    while time.time() < deadline and T > T_min:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        mask[i] = 0
        mask[j] = 1
        s = _mask_score(mask)
        delta = s - cur_s
        if delta < 0 or rng.random() < np.exp(-max(delta, -2.0) / max(T, 1e-8)):
            cur_s = s
            ones = np.where(mask == 1)[0]
            zeros = np.where(mask == 0)[0]
            if s < best_s:
                best_s = s
                best_mask = mask.copy()
        else:
            mask[i] = 1
            mask[j] = 0
        T *= cooling

    h = np.where(best_mask == 1, 1.0, 0.0)
    return h, _score(h)


def _pair_line_search(h, deadline, max_sweeps=200, n_samples=13):
    """Pair line search: move along h[i]+h[j]=const."""
    n = len(h)
    h = h.copy()
    cur_s = _score(h)
    target = n / 2.0
    rng = np.random.default_rng(42)
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if no_improve_sweeps > 20:
            break
        improved = False
        order = list(range(n))
        rng.shuffle(order)

        for a in order:
            if time.time() > deadline:
                break
            for b in range(a + 1, n):
                if time.time() > deadline:
                    break
                p = h[a] + h[b]
                lo = max(0.0, p - 1.0)
                hi = min(1.0, p)
                if hi - lo < 1e-10:
                    continue
                best_t = h[a]
                best_s = cur_s
                ts = np.linspace(lo, hi, n_samples)
                for t in ts:
                    h[a] = t
                    h[b] = p - t
                    s = _score(h)
                    if s < best_s:
                        best_s = s
                        best_t = t
                # Refine around best
                if best_t > lo and best_t < hi:
                    lo2 = max(lo, best_t - (hi - lo) / n_samples)
                    hi2 = min(hi, best_t + (hi - lo) / n_samples)
                    for t in np.linspace(lo2, hi2, 9):
                        h[a] = t
                        h[b] = p - t
                        s = _score(h)
                        if s < best_s:
                            best_s = s
                            best_t = t

                h[a] = best_t
                h[b] = p - best_t
                if best_s < cur_s - 1e-12:
                    cur_s = best_s
                    improved = True

        if not improved:
            no_improve_sweeps += 1
        else:
            no_improve_sweeps = 0

    h = _project(h, target)
    return h, _score(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 16:
        n_new = 16
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _make_mask(h, n):
    half = n // 2
    idx = np.argsort(h)[::-1][:half]
    mask = np.zeros(n, dtype=np.int8)
    mask[idx] = 1
    return mask


def _grad_descend(h, deadline, start_lr=0.01, min_lr=1e-6, patience=50):
    """Projected subgradient descent on the max."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_s = _score(h)
    lr = start_lr
    stall = 0

    while time.time() < deadline:
        if stall > patience:
            break
        g = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        # Subgradient: average of gradients at top-3 active lags
        top3 = np.argsort(g)[-3:]
        grad = np.zeros(n)
        for lag_idx in top3:
            d = lag_idx - (n - 1)
            if d >= 0:
                for i in range(d, n):
                    grad[i] += (1.0 - h_cur[i - d])
                for i in range(0, n - d):
                    grad[i] -= h_cur[i + d]
            else:
                dp = -d
                for i in range(0, n - dp):
                    grad[i] += (1.0 - h_cur[i + dp])
                for i in range(dp, n):
                    grad[i] -= h_cur[i - dp]
        grad /= 3.0
        grad -= np.mean(grad)  # preserve sum
        h_new = _project(h_cur - lr * grad, target)
        s_new = _score(h_new)
        if s_new < best_s - 1e-14:
            best_s = s_new
            best_h = h_new.copy()
            h_cur = h_new
            stall = 0
            lr = min(lr * 1.05, 0.05)
        elif s_new < _score(h_cur) - 1e-14:
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            lr = max(lr * 0.85, min_lr)
            if lr < min_lr:
                break
    return best_h, best_s


def _random_kick(h, rng, scale=0.05):
    n = len(h)
    target = n / 2.0
    h_p = h + rng.normal(0, scale, n)
    return _project(h_p, target)
