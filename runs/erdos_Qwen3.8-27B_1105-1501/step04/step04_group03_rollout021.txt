def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

        # i-k+n-1 in [0, n-1] => k in [i-n+1, i]
        lo1 = max(0, i - n + 1)
        hi1 = min(2*n - 2, i)
        indices = i - np.arange(lo1, hi1+1) + n - 1
        M[lo1:hi1+1, i] += (1.0 - h[indices])
        
        # h[i] appears as the "(1-h)" term: contributes -h[m] where m = i+k-n+1
        # i+k-n+1 in [0, n-1] => k in [n-1-i, 2n-2-i]
        lo2 = max(0, n - 1 - i)
        hi2 = min(2*n - 2, 2*n - 2 - i)
        indices2 = i + np.arange(lo2, hi2+1) - n + 1
        M[lo2:hi2+1, i] -= h[indices2]
    return M

def pair_cd(h, deadline, max_passes=500):

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    M = compute_sensitivity(h)
    s_base = np.correlate(h, 1.0 - h, mode='full') * dx
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                d_min = -h[j]
                d_max = 1.0 - h[i]
                if d_max - d_min < 1e-12:
                    continue
                ds = M[:, i] - M[:, j]
                # f(δ) = max(s_base + δ * ds)
                # Quick check: is there improvement?
                # f(0) = max(s_base) = current max
                # f at midpoint
                mid = 0.5 * (d_min + d_max)
                f_mid = np.max(s_base + mid * ds)
                f_0 = np.max(s_base)
                if f_mid >= f_0 - 1e-14 and d_min > -1e-14 and d_max < 1 + 1e-14:
                    # Try small steps in both directions
                    small = 0.1 * (d_max - d_min)
                    f_s1 = np.max(s_base + small * ds)
                    f_s2 = np.max(s_base - small * ds)
                    if f_s1 >= f_0 - 1e-14 and f_s2 >= f_0 - 1e-14:
                        continue
                
                # Find optimum via grid + refine
                best_r = np.max(s_base)
                best_d = 0.0
                grid = np.linspace(d_min, d_max, 13)
                for d_try in grid:
                    f_val = np.max(s_base + d_try * ds)
                    if f_val < best_r:
                        best_r = f_val
                        best_d = d_try
                
                if best_r < best_c5 - 1e-13:
                    # Apply
                    h[i] += best_d
                    h[j] -= best_d
                    h = np.clip(h, 0.0, 1.0)
                    # Re-project to maintain sum constraint exactly
                    h = project(h, target)
                    s_base = np.correlate(h, 1.0 - h, mode='full') * dx
                    M = compute_sensitivity(h)
                    new_c5 = exact_c5(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def build_corr_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def resample_h(h_old, n_new):

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.96
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: LP from initial (12% budget)
    d1 = t0 + budget_s * 0.12
    h_lp, c5_lp = lp_loop(h, d1, radius_init=0.15)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: Pairwise coordinate descent (30% budget)
    d2 = t0 + budget_s * 0.42
    h_cd, c5_cd = pair_cd(best_h, d2, max_passes=500, grid_size=25)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 3: LP refinement after CD (10% budget)
    d3 = t0 + budget_s * 0.52
    h_lp2, c5_lp2 = lp_loop(best_h, d3, radius_init=0.1)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 4: Second CD pass (10% budget)
    d4 = t0 + budget_s * 0.62
    h_cd2, c5_cd2 = pair_cd(best_h, d4, max_passes=400, grid_size=21)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    # Phase 5: Diversified restarts (20% budget)
    d5 = t0 + budget_s * 0.82
    scales = [0.02, 0.04, 0.07, 0.1, 0.15, 0.2, 0.3]
    for ns in scales:
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        # Quick CD
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_try, c5_try = pair_cd(h_pert, sub_dl, max_passes=100, grid_size=17)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Quick LP
        if time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.008)
            h_try2, c5_try2 = lp_loop(h_pert, sub_dl2, radius_init=0.08)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase 6: Multi-resolution search (10% budget)
    d6 = t0 + budget_s * 0.92
    offsets = [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24]
    tried_n = set([best_n])
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 160:
            continue
        if n_try in tried_n:
            continue
        if n_try % 2 != 0:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.008)
        h_r, c5_r = pair_cd(h_r, sub_dl, max_passes=80, grid_size=17)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final polish with CD + LP (8% budget)
    if time.time() < hard_deadline:
        h_f, c5_f = pair_cd(best_h, hard_deadline, max_passes=200, grid_size=21)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
