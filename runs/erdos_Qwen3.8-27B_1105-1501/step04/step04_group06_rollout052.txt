def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    # Bisection to find threshold
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)[::-1]  # smallest first for adding
            order = np.argsort(x)
            for i in order:
                if x[i] >= 1.0: continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12: break
        else:
            order = np.argsort(-x)
            for i in order:
                if x[i] <= 0.0: continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12: break
