def _area_resample(h_old, n_new):
    """Area-preserving resampling: preserve the integral (area) of h."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    # For each new cell, compute the overlap with old cells
    # New cell j: [j*dx_new, (j+1)*dx_new]
    # Old cell i: [i*dx_old, (i+1)*dx_old]
    
    h_new = np.zeros(n_new)
    for j in range(n_new):
        # New cell j spans [j*dx_new, (j+1)*dx_new]
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        # Find overlapping old cells
        i_start = max(0, int(x_start // dx_old))
        i_end = min(n_old - 1, int((x_end - 1e-15) // dx_old))
        area = 0.0
        for i in range(i_start, i_end + 1):
            # Overlap of [i*dx_old, (i+1)*dx_old] and [x_start, x_end]
            ol = max(i * dx_old, x_start)
            oa = min((i + 1) * dx_old, x_end)
            if oa > ol:
                area += h_old[i] * (oa - ol)
        h_new[j] = area / dx_new
        # Clip to [0, 1] (should be automatically in range if h_old is)
        h_new[j] = min(max(h_new[j], 0.0), 1.0)
    
    return _project(h_new, n_new / 2.0)

        top_idx = np.argsort(scores)[::-1][:n_top]
        T0 = float(np.max(scores)) + 1e-5
        x0 = np.concatenate([h, [T0]])
        bounds_s = [(0.0, 1.0)] * n + [(0.0, float(n))]
        def _obj(x):
            return x[n]
        def _obj_g(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        def _ineq(x):
            hx = x[:n]
            sx = _raw_scores(hx)
            return x[n] - sx[top_idx]
        def _ineq_j(x):
            hx = x[:n]
            jac = np.zeros((n_top, n + 1))
            for i, r in enumerate(top_idx):
                G = _G_row(hx, r)
                jac[i, :n] = G
                jac[i, n] = -1.0
            return jac
        def _eq(x):
            return float(np.sum(x[:n]) - target)
        def _eq_j(x):
            v = np.zeros(n + 1)
            v[:n] = 1.0
            return v
        constr = [
            {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_j},
            {'type': 'eq', 'fun': _eq, 'jac': _eq_j}
        ]
        maxiter = max(20, min(70, int(4000 / max(n, 1))))
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            try:
                res = minimize(_obj, x0, method='SLSQP', jac=_obj_g,
                              bounds=bounds_s, constraints=constr,
                              options={'maxiter': maxiter, 'ftol': 1e-11, 'disp': False})
            except Exception:
                break
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            elif c5_new < _exact_c5(h) - 1e-12:
                h = h_new
        else:
            break
    return best_h, best_c5

def _area_resample(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        xs = j * dx_new
        xe = (j + 1) * dx_new
        i_start = max(0, int(xs // dx_old))
        i_end = min(n_old - 1, int((xe - 1e-15) // dx_old))
        area = 0.0
        for i in range(i_start, i_end + 1):
            ol = max(i * dx_old, xs)
            oa = min((i + 1) * dx_old, xe)
            if oa > ol:
                area += h_old[i] * (oa - ol)
        h_new[j] = area / dx_new
        h_new[j] = min(max(h_new[j], 0.0), 1.0)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(hash(_pass * 7919 + 13) & 0xFFFFFFFF)
        if n > 50:
            n_sample = min(n, 50)
            idx_j = rng.integers(0, n, size=n_sample)
            idx_k = rng.integers(0, n, size=n_sample)
            pairs = set()
            for jj in idx_j:
                for kk in idx_k:
                    if jj != kk:
                        pairs.add((min(int(jj), int(kk)), max(int(jj), int(kk))))
            pair_list = list(pairs)
        else:
            pair_list = [(j, k) for j in range(n) for k in range(j + 1, n)]
        rng.shuffle(pair_list)
        for (j, k) in pair_list:
            if time.time() > deadline:
                break
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6:
                continue
            if h[k] <= 1e-6 or h[k] >= 1.0 - 1e-6:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if dj > 0:
                    am = (1.0 - h[j]) / dj
                else:
                    am = h[j] / (-dj)
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-12:
                    continue
                for frac in [0.3, 0.6, 0.85, 1.0]:
                    h_t = h.copy()
                    h_t[j] += dj * am * frac
                    h_t[k] += dk * am * frac
                    if h_t[j] < -1e-12 or h_t[j] > 1.0 + 1e-12:
                        continue
                    if h_t[k] < -1e-12 or h_t[k] > 1.0 + 1e-12:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _grad_descend(h, deadline, tau_init=15.0, lr_init=0.015, mom_beta=0.9, max_iter=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    for _it in range(max_iter):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        scores = _raw_scores(h) * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            wr = w[r]
            if wr < 1e-15:
                continue
            G = _G_row(h, r)
            grad += wr * G
        grad *= dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
            tau = min(tau * 1.01, 150.0)
        else:
            stall += 1
            lr *= 0.93
            if stall > 100:
                break
            if stall % 40 == 0:
                tau = max(tau_init, tau * 0.8)
    return best_h, best_c5

def _diverse_starts(h_base, n, seed):
    """Generate diverse starting points from a base."""
    rng = np.random.default_rng(seed)
    candidates = []
    target = n / 2.0
    candidates.append(_project(h_base.copy(), target))
    # Reversed
    candidates.append(_project(h_base[::-1].copy(), target))
    # Binarized
    h_bin = (h_base > 0.5).astype(float)
    # Fix sum
    n_ones = int(np.sum(h_bin))
    if n_ones != int(target):
        order = np.argsort(-h_base)
        h_bin = np.zeros(n)
        h_bin[order[:int(target)]] = 1.0
    # Add fractional part
    rem = target - np.sum(h_bin)
    if abs(rem) > 1e-10:
        partial = order[int(target)] if int(target) < n else order[0]
        h_bin[partial] = rem if rem > 0 else 1.0 + rem
    h_bin = np.clip(h_bin, 0, 1)
    h_bin = _project(h_bin, target)
    candidates.append(h_bin.copy())
    # Smoothed
    kernel = np.array([0.25, 0.5, 0.25])
    h_smooth = np.convolve(h_base, kernel, mode='same')
    candidates.append(_project(h_smooth, target))
    # Sharpened
    h_sharp = np.where(h_base > 0.5, 0.5 + 0.6 * (h_base - 0.5), 0.5 - 0.6 * (0.5 - h_base))
    candidates.append(_project(h_sharp, target))
    # Small noise
    h_noisy = h_base + rng.normal(0, 0.03, n)
    candidates.append(_project(h_noisy, target))
    # Perturbed noise
    h_pert = h_base + rng.normal(0, 0.08, n)
    candidates.append(_project(h_pert, target))
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0, n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    best_n = n0

    # Phase 1: LP warmup at parent resolution (15%)
    d1 = t0 + budget_s * 0.15
    h, c5 = _lp_optimize(h_init, d1, n_active=50, radius_init=0.15, max_stall=80)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n0

    # Phase 2: SLSQP at parent resolution (15%)
    d2 = t0 + budget_s * 0.30
    h, c5 = _slsqp_optimize(best_h, d2, n_top=40, max_rounds=6)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n0

    # Phase 3: LP after SLSQP (10%)
    d3 = t0 + budget_s * 0.40
    h, c5 = _lp_optimize(best_h, d3, n_active=50, radius_init=0.1, max_stall=60)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n0

    # Phase 4: Multi-resolution search (30%)
    d4 = t0 + budget_s * 0.70
    try_res = [n0, n0-4, n0+4, n0-8, n0+8, n0-12, n0+12, n0-16, n0+16,
               n0-20, n0+20, 70, 72, 76, 80, 84, 96, 100, 108, 112, 120]
    tried = set()
    time_per_res = budget_s * 0.02
    for n_try in sorted(try_res):
        if time.time() > d4:
            break
        if n_try < 40 or n_try > 160 or n_try in tried or n_try == best_n:
            continue
        tried.add(n_try)
        h_res = _area_resample(best_h, n_try)
        # Quick check: does resampling alone already improve?
        c5_res = _exact_c5(h_res)
        sub_dl = min(d4, time.time() + time_per_res)
        h_try, c5_try = _slsqp_optimize(h_res, sub_dl, n_top=min(35, n_try // 2 // 2), max_rounds=4)
        if c5_try < c5_res:
            c5_res = c5_try
            h_res = h_try
        if time.time() < d4 and (d4 - time.time()) > time_per_res * 0.4:
            sub_dl2 = min(d4, time.time() + time_per_res * 0.5)
            h_try2, c5_try2 = _lp_optimize(h_res, sub_dl2, n_active=min(40, n_try), radius_init=0.12, max_stall=40)
            if c5_try2 < c5_res:
                c5_res = c5_try2
                h_res = h_try2
        if c5_res < best_c5 - 1e-11:
            best_c5 = c5_res
            best_h = h_res.copy()
            best_n = n_try

    # Phase 5: Intensive optimization at best resolution (20%)
    d5 = t0 + budget_s * 0.90
    h, c5 = _lp_optimize(best_h, min(d5, time.time() + budget_s * 0.06), n_active=60, radius_init=0.12, max_stall=80)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d5:
        h, c5 = _slsqp_optimize(best_h, min(d5, time.time() + budget_s * 0.05), n_top=45, max_rounds=6)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < d5:
        h, c5 = _lp_optimize(best_h, min(d5, time.time() + budget_s * 0.04), n_active=60, radius_init=0.06, max_stall=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Pair refinement + gradient (10%)
    d6 = t0 + budget_s * 0.96
    h, c5 = _pair_refine(best_h, min(d6, time.time() + budget_s * 0.04), max_passes=150)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d6:
        h, c5 = _grad_descend(best_h, min(d6, time.time() + budget_s * 0.02), tau_init=30.0, lr_init=0.005, mom_beta=0.92)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < d6:
        h, c5 = _pair_refine(best_h, d6, max_passes=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: Final LP polish
    if time.time() < hard_deadline:
        h, c5 = _lp_optimize(best_h, hard_deadline, n_active=50, radius_init=0.03, max_stall=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    
    # Compare with parent
    parent_c5 = _exact_c5(_project(h0, n0 / 2.0))
    if parent_c5 < final_c5:
        return _project(h0, n0 / 2.0), parent_c5, n0
    
    return best_h, final_c5, best_n
