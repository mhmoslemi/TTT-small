def _exact_c5(h):

def _scores(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_direction(h, scores, G, radius, top_k, l1_penalty=0.0):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
    else:
        idx = np.arange(n_lags)
    scores_k = scores[idx]
    G_k = G[idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    if l1_penalty > 0:
        c[:n] = l1_penalty
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgradient_opt(h_in, deadline, n_steps=500, gamma_start=20.0, gamma_end=200.0,
                     momentum=0.5, lr_init=0.05):
    n = len(h_in)
    target = n / 2.0
    h = h_in.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        gamma = gamma_start * (gamma_end / gamma_start) ** frac
        lr = lr_init * (1.0 - 0.5 * frac)
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(gamma * (scores - max_s))
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        g = G.T @ w
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        v = momentum * v - lr * g
        h_new = _project(h + v, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            v *= 0.5
    return best_h, best_c5

def _resample(h, n_new):

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _fourier_evolve(h, deadline, n_pop=20, K=8, n_gen=200, mutation_scale=0.10, rng=None):
    n = len(h)
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(2 * np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    a0 = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    pop_a = a0 + mutation_scale * rng.standard_normal((n_pop, 2*K))
    pop_c5 = np.array([_exact_c5(_project(np.clip(0.5 + B @ pop_a[i], 0, 1), n/2.0))
                       for i in range(n_pop)])
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]

def _adaptive_subgrad(h_in, deadline, n_cycles=10):
    """Subgradient with adaptive gamma and restarts."""
    n = len(h_in)
    target = n / 2.0
    h = h_in.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    gamma = 10.0
    lr = 0.06
    v = np.zeros(n)
    stall = 0
    for cycle in range(n_cycles):
        cur_c5 = _exact_c5(h)
        if time.time() > deadline:
            break
        n_steps = 50
        for step in range(n_steps):
            if time.time() > deadline:
                break
            scores = _scores(h)
            max_s = np.max(scores)
            w = np.exp(gamma * (scores - max_s))
            sw = w.sum()
            if sw < 1e-300:
                w = np.zeros(len(scores))
                w[int(np.argmax(scores))] = 1.0
            else:
                w /= sw
            G = _grad_matrix(h)
            g = G.T @ w
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g /= gn
            v = 0.5 * v - lr * g
            h_new = _project(h + v, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                v = 0.5 * v  # dampen on success
            else:
                v *= 0.3
        if _exact_c5(h) < best_c5:
            best_c5 = _exact_c5(h)
            best_h = h.copy()
            gamma *= 1.3
            lr *= 0.85
            stall = 0
        else:
            stall += 1
            gamma = max(5.0, gamma * 0.8)
            lr = min(0.08, lr * 1.2)
            h = best_h + 0.01 * (h - best_h)
            h = _project(h, target)
        if stall > 4:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h0 = _quasi_periodic_init(128, 0.6180339887, rng)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: Quick LP warmup from initial (5% time)
    dl1 = t0 + budget_s * 0.05
    h1, c5_1 = _multi_lp_optimize(h_start, dl1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution diverse search (25% time)
    dl2 = t0 + budget_s * 0.30
    resolutions = [64, 80, 96, 112, 128, 144]
    alphas_list = [0.6180339887, 0.4142135624, 0.3819660113, 0.7071067812, 0.5, 0.455]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl2:
            break
        if best_c5 < 0.3805:
            break
        alpha = alphas_list[ri % len(alphas_list)]

        # Start 1: resample best
        h_init = _resample(best_h, n_try)
        sub_dl = min(dl2, time.time() + budget_s * 0.025)
        h_try, c5_try = _multi_lp_optimize(h_init, sub_dl)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

        # Start 2: quasi-periodic
        if time.time() < dl2:
            h_qp = _quasi_periodic_init(n_try, alpha, rng)
            h_qp += rng.normal(0, 0.05, n_try)
            h_qp = _project(h_qp, n_try / 2.0)
            sub_dl2 = min(dl2, time.time() + budget_s * 0.025)
            h_qp2, c5_qp = _multi_lp_optimize(h_qp, sub_dl2)
            if c5_qp < best_c5:
                best_c5 = c5_qp
                best_h = h_qp2.copy()
                best_n = n_try

        # Start 3: smooth quasi-periodic
        if time.time() < dl2 and best_c5 > 0.3810:
            h_sq = _smooth_quasi_init(n_try, alpha, 0.1, rng)
            sub_dl3 = min(dl2, time.time() + budget_s * 0.02)
            h_sq2, c5_sq = _multi_lp_optimize(h_sq, sub_dl3)
            if c5_sq < best_c5:
                best_c5 = c5_sq
                best_h = h_sq2.copy()
                best_n = n_try

    # Phase 3: Deep LP optimization on best (25% time)
    dl3 = t0 + budget_s * 0.55
    if time.time() < dl3:
        h_deep, c5_deep = _multi_lp_optimize(best_h, dl3, l1_pen=0.0)
        if c5_deep < best_c5:
            best_c5 = c5_deep
            best_h = h_deep.copy()

    # Phase 4: Subgradient + LP interleaved (20% time)
    dl4 = t0 + budget_s * 0.72
    if time.time() < dl4:
        # Subgradient phase
        sub_dl4a = min(dl4, time.time() + budget_s * 0.08)
        h_sg, c5_sg = _adaptive_subgrad(best_h, sub_dl4a, n_cycles=8)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

        # LP refinement after subgradient
        if time.time() < dl4:
            sub_dl4b = min(dl4, time.time() + budget_s * 0.08)
            h_lp4, c5_lp4 = _multi_lp_optimize(best_h, sub_dl4b)
            if c5_lp4 < best_c5:
                best_c5 = c5_lp4
                best_h = h_lp4.copy()

    # Phase 5: Binary SA + structured inits (10% time)
    dl5 = t0 + budget_s * 0.82
    if time.time() < dl5 and best_c5 > 0.3808:
        for n_bin in [best_n if best_n in (64, 80, 96, 112, 128) else 128, 96, 64]:
            if time.time() > dl5:
                break
            sub_dl5 = min(dl5, time.time() + budget_s * 0.025)
            h_b, c5_b = _binary_sa(n_bin, sub_dl5, rng, n_iters=20000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                # Quick LP on binary result
                if time.time() < dl5:
                    sub_dl5b = min(dl5, time.time() + budget_s * 0.015)
                    h_bl, c5_bl = _multi_lp_optimize(h_b, sub_dl5b)
                    if c5_bl < best_c5:
                        best_c5 = c5_bl
                        best_h = h_bl.copy()

    # Phase 6: Fourier evolution (8% time)
    dl6 = t0 + budget_s * 0.90
    if time.time() < dl6 and best_c5 > 0.3806:
        h_f, c5_f = _fourier_evolve(best_h, dl6, n_pop=15, K=10,
                                     n_gen=150, mutation_scale=0.08, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        if time.time() < dl6 and best_c5 < 0.3810:
            sub_dl6 = min(dl6, time.time() + budget_s * 0.02)
            h_fp, c5_fp = _multi_lp_optimize(h_f, sub_dl6)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 7: Pair polish (5% time)
    dl7 = t0 + budget_s * 0.95
    if time.time() < dl7:
        hp, c5p = _pair_polish(best_h, dl7, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 8: Final tiny LP refinement (5% time)
    dl8 = t0 + budget_s * 0.995
    if time.time() < dl8:
        h_fin, c5_fin = _multi_lp_optimize(best_h, dl8)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and verification
    best_h = _project(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, best_n
