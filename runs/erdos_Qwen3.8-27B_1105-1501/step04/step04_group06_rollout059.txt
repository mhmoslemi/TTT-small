import numpy as np
from scipy.optimize import minimize, linprog
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _grad_corr(h, r):
    n = len(h)
    s = r - (n - 1)
    j = np.arange(n)
    pd = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    pd[m1] = 1.0 - h[gs[m1]]
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    pd[m2] -= h[ps[m2]]
    return pd


def _lse_fn(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(corr.max())
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    ev = np.exp(vals)
    tot = float(ev.sum())
    f = ms + (1.0 / alpha) * np.log(tot)
    w = ev / tot
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-13:
            continue
        s = r - (n - 1)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd = np.zeros(n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        grad += w[r] * pd
    grad -= np.mean(grad)
    return f, grad


def _lse_opt(h, n, alpha, maxiter=200):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_fn, h, args=(n, alpha), jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                               'maxls': 100, 'iprint': 0})
        x = _project(res.x, target)
        return x, _c5(x)
    except Exception:
        return h.copy(), _c5(h)


def _epi_f(z, n):
    return z[n]


def _epi_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _act_f(z, n, idx):
    h = z[:n]
    t = z[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - corr[idx]


def _act_g(z, n, idx):
    h = z[:n]
    k = len(idx)
    J = np.zeros((k, n + 1))
    j = np.arange(n)
    for ii in range(k):
        r = idx[ii]
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        J[ii, :n] = -pd
        J[ii, n] = 1.0
    return J


def _active_slsqp(h_init, n, maxiter=200, ftol=1e-13, active_tol=0.005, max_active=20):

            x_new = _project(res.x[:n], n * 0.5)

def _lp_dir(h, n, radius, scores=None):
    if scores is None:
        scores = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        G[r, :] = pd
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for jj in range(n):
        lo_b = max(-radius, -h[jj])
        hi_b = min(radius, 1.0 - h[jj])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _sturmian(n, alpha):
    return np.array([int(np.floor((i + 1) * alpha) - np.floor(i * alpha))
                     for i in range(n)], dtype=float)


def _thue_morse(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)


def _diverse_starts(n, rng, best_h):
    starts = [best_h.copy()]
    # Sturmian sequences
    for alpha in [0.51, 0.49, (1 + np.sqrt(5)) / 4 - 0.31, 0.4, 0.6]:
        s = _sturmian(n, alpha)
        s = _project(s, n * 0.5)
        starts.append(s)
    # Thue-Morse
    tm = _thue_morse(n)
    tm = _project(tm, n * 0.5)
    starts.append(tm)
    # Alternating
    alt = np.tile([1.0, 0.0], n // 2)
    starts.append(alt.copy())
    # Random balanced
    for _ in range(3):
        r = rng.random(n)
        r = _project(r, n * 0.5)
        starts.append(r)
    # Symmetrized best_h
    sym = 0.5 * (best_h + best_h[::-1])
    sym = _project(sym, n * 0.5)
    starts.append(sym)
    return starts


def _subgrad_step(h, n, step_size=0.05):
    corr = np.correlate(h, 1.0 - h, mode='full')
    r_max = int(np.argmax(corr))
    pd = _grad_corr(h, r_max)
    pd -= np.mean(pd)
    norm = np.linalg.norm(pd)
    if norm < 1e-15:
        return h.copy(), _c5(h)
    pd = pd / norm * step_size
    best_h, best_c5 = h.copy(), _c5(h)
    for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
        h_try = _project(h - frac * pd, n * 0.5)
        c5_try = _c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c5_try
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 * 0.5
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: LSE warmup with increasing sharpness (12% budget)
    d1 = t0 + budget_s * 0.12
    for alpha in [30.0, 80.0, 150.0, 300.0]:
        if time.time() > d1:
            break
        h, c5 = _lse_opt(h, n0, alpha, maxiter=400)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Active-set SLSQP from best (18% budget)
    d2 = t0 + budget_s * 0.30
    for attempt in range(8):
        if time.time() > d2:
            break
        h_a, c5_a = _active_slsqp(best_h, n0, maxiter=300, ftol=1e-14,
                                  active_tol=0.003, max_active=18)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a.copy()
        else:
            h_p = _project(best_h + rng.normal(0, 0.03, n0), target)
            h_a, c5_a = _active_slsqp(h_p, n0, maxiter=200, ftol=1e-13,
                                      active_tol=0.01, max_active=15)
            if c5_a < best_c5:
                best_c5, best_h = c5_a, h_a.copy()

    # Phase 3: LP polishing (18% budget)
    d3 = t0 + budget_s * 0.48
    h_lp, c5_lp = _lp_loop(best_h, n0, d3, radius_init=0.12,
                           max_iters=5000, stall_limit=60)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 4: Diverse starts and refinement (18% budget)
    d4 = t0 + budget_s * 0.66
    starts = _diverse_starts(n0, rng, best_h)
    for s in starts:
        if time.time() > d4:
            break
        s_c5 = _c5(s)
        if s_c5 >= best_c5 + 0.02:
            continue
        # Quick LSE
        s, sc = _lse_opt(s, n0, 100.0, maxiter=200)
        if sc < best_c5:
            best_c5, best_h = sc, s.copy()
        # SLSQP
        if time.time() < d4:
            h_a, c5_a = _active_slsqp(s, n0, maxiter=200, ftol=1e-13,
                                      active_tol=0.005, max_active=15)
            if c5_a < best_c5:
                best_c5, best_h = c5_a, h_a.copy()
        # LP
        if time.time() < d4:
            sub_d = min(d4, time.time() + budget_s * 0.01)
            h_lp2, c5_lp2 = _lp_loop(best_h, n0, sub_d, radius_init=0.08,
                                     max_iters=800, stall_limit=30)
            if c5_lp2 < best_c5:
                best_c5, best_h = c5_lp2, h_lp2.copy()

    # Phase 5: Resolution search (16% budget)
    d5 = t0 + budget_s * 0.82
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, 10]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _lse_opt(h_r, n_try, 100.0, maxiter=200)
        if time.time() < d5:
            h_a, c5_a = _active_slsqp(h_r, n_try, maxiter=200, ftol=1e-13,
                                      active_tol=0.005, max_active=15)
            if c5_a < c5_r:
                h_r, c5_r = h_a, c5_a
        if time.time() < d5:
            sub_d2 = min(d5, time.time() + budget_s * 0.008)
            h_lp3, c5_lp3 = _lp_loop(h_r, n_try, sub_d2, radius_init=0.06,
                                     max_iters=500, stall_limit=30)
            if c5_lp3 < c5_r:
                h_r, c5_r = h_lp3, c5_r
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Final sharp polish (14% budget)
    d6 = t0 + budget_s * 0.99
    h, c5 = _lse_opt(best_h, best_n, 400.0, maxiter=500)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    if time.time() < d6:
        h_a, c5_a = _active_slsqp(best_h, best_n, maxiter=300, ftol=1e-14,
                                  active_tol=0.001, max_active=25)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a.copy()
    if time.time() < d6:
        h_lp4, c5_lp4 = _lp_loop(best_h, best_n, min(d6, time.time() + budget_s * 0.03),
                                 radius_init=0.03, max_iters=3000, stall_limit=80)
        if c5_lp4 < best_c5:
            best_c5, best_h = c5_lp4, h_lp4.copy()
    # Final subgradient nudges
    if time.time() < d6:
        for _ in range(50):
            if time.time() > d6:
                break
            h_s, c5_s = _subgrad_step(best_h, best_n, step_size=0.02)
            if c5_s < best_c5 - 1e-14:
                best_c5, best_h = c5_s, h_s.copy()
            else:
                break
        # One more LSE at very high alpha
        h_f, c5_f = _lse_opt(best_h, best_n, 500.0, maxiter=300)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
