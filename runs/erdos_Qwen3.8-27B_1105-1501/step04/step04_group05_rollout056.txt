import numpy as np
import time
from scipy.optimize import minimize

def _c5_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_to_target(v, target):
    """Project v to [0,1]^n with sum = target using bisection."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0 + v.min()
    hi = 2.0 + v.max()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _two_level_from_mask(mask, a):
    """Create a two-level vector from a binary mask and parameter a."""
    n = len(mask)
    h = np.where(mask, 1.0 - a, a)
    return h

def _optimize_a_for_mask(mask, n_targets=200):
    """Find the a that minimizes C5 for a given mask."""
    n = len(mask)
    # a ranges from 0 to 0.5
    # Coarse grid
    a_vals = np.linspace(0.0, 0.5, n_targets)
    best_a = 0.0
    best_c5 = np.inf
    for a in a_vals:
        h = _two_level_from_mask(mask, a)
        c5 = _c5_score(h)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
    # Fine grid around best
    if 0 < best_a < 0.5:
        fine = np.linspace(max(0, best_a - 0.005), min(0.5, best_a + 0.005), 100)
    else:
        fine = np.linspace(0, 0.01, 50) if best_a == 0 else np.linspace(0.49, 0.5, 50)
    for a in fine:
        h = _two_level_from_mask(mask, a)
        c5 = _c5_score(h)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a
    return best_a, best_c5

    n = len(mask)
    mask = mask.copy()
    best_a, best_c5 = _optimize_a_for_mask(mask)
    
    # Get current evaluated vector
    h_best = _two_level_from_mask(mask, best_a)
    
    ones = np.where(mask == 1)[0]
    zeros = np.where(mask == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap = None
        best_score = best_c5
        
        # Try each (one, zero) swap
        for i in ones:
            for j in zeros:
                if time.time() > deadline:
                    break
                mask[i] = 0
                mask[j] = 1
                a_try, c5_try = _optimize_a_for_mask(mask)
                mask[i] = 1
                mask[j] = 0
                if c5_try < best_score - 1e-10:
                    best_score = c5_try
                    best_swap = (i, j, a_try)
        
        if best_swap is None:
            break
        
        i, j, new_a = best_swap
        mask[i] = 0
        mask[j] = 1
        best_a = new_a
        best_c5 = best_score
        ones = np.where(mask == 1)[0]
        zeros = np.where(mask == 0)[0]
    
    h_best = _two_level_from_mask(mask, best_a)
    # Verify
    if abs(np.sum(h_best) - n/2) > 1e-6:
        h_best = _project_to_target(h_best, n/2)
    return h_best, _c5_score(h_best)
