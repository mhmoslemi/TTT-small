def _project(v, target):

    scores = a[None, :] + b[None, :] * alphas[:, None] + c[None, :] * (alphas[:, None])**2

    best_val = max_scores[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_samples - 1:
        lo = alphas[best_i - 1]
        hi = alphas[best_i + 1]

        if fm < best_val:
            best_alpha = mid
            best_val = fm
    
    return best_alpha, best_val

So ∫_0^1 (1-0) dx = 1

The maximum is 1 (at k=1 or k=2). That's terrible.

What about h(x) = 1/2 for all x? Then ∫ h(x)(1-h(x+k)) dx = ∫_0^2 (1/2)(1/2) dx = 1/2. That's better but still 0.5.

The optimal solution is somewhere in between. The fact that the best known is ~0.381 suggests a fairly complex structure.

For n=64, a "every other" pattern gives 32 ones. The autocorrelation at lag 2 would be high, leading to low overlap at lag 2 but potentially high at other lags.

Actually, the best binary patterns for this type of problem are related to **Barker sequences** or **m-sequences** (maximal length linear feedback shift register sequences). These have the property that their autocorrelation is as flat as possible.

An m-sequence of length 2^m - 1 has the property that its periodic autocorrelation is -1 at all non-zero lags. If I can find a subset of such a sequence with exactly n/2 elements, it might give a good starting point.

For n=64 (which is 2^6), I can't use a perfect m-sequence (which would be length 63), but I can approximate.

Let me try a different approach: use a **golden ratio** or **irrational rotation** to place the "ones". For example, place ones at positions floor(i * n / φ) mod n for i = 0, ..., n/2-1 where φ is the golden ratio. This gives a low-discrepancy set.

Actually, for this problem, what we want is a set S of size n/2 such that for every shift d, the number of "conflicts" (pairs (x, x+d) where x ∈ S and x+d ∉ S, or x ∉ S and x+d ∈ S) is as small as possible.

This is related to the **discrepancy** of the set S. A set with low discrepancy in all shifts is what we want.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _eval_max_quadratic(a, b, c, alpha):
    """Evaluate max_r (a[r] + b[r]*alpha + c[r]*alpha^2) for a scalar alpha."""
    s = a + b * alpha + c * alpha * alpha
    return float(s.max())

def _solve_pair_step(h, i, j, dir_sign=1.0, n_samples=200):
    """
    For pair (i,j), find best alpha to move mass.
    dir_sign=1: move from i to j (h[i] decreases, h[j] increases)
    dir_sign=-1: move from j to i
    Returns (best_alpha, best_raw_score) where raw_score is max corr value (before *dx).
    """
    n = len(h)
    # Feasible range
    if dir_sign > 0:
        alpha_max = min(h[i], 1.0 - h[j])
    else:
        alpha_max = min(h[j], 1.0 - h[i])
    if alpha_max < 1e-10:
        return 0.0, _eval_max_quadratic(
            np.correlate(h, 1.0-h, mode='full'),
            np.zeros(2*n-1), np.zeros(2*n-1), 0.0)
    
    # Compute a, b, c
    a = np.correlate(h, 1.0 - h, mode='full')
    
    # For direction V: V[i] = -dir_sign, V[j] = dir_sign
    # b_r = corr(V, 1-h)[r] - corr(h, V)[r]
    # = -dir_sign * (1-h)[i+r-n+1] + dir_sign * (1-h)[j+r-n+1] 
    #   + dir_sign * h[i-r+n-1] - dir_sign * h[j-r+n-1]
    b = np.zeros(2*n-1)
    
    # Term: -dir_sign * (1-h)[i+r-n+1]
    # valid r: 0 <= i+r-n+1 < n => n-1-i <= r <= 2n-2-i
    r_lo = max(0, n - 1 - i)
    r_hi = min(2*n-2, 2*n-2-i)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        idx = i + rr - n + 1
        b[rr] += -dir_sign * (1.0 - h[idx])
    
    # Term: +dir_sign * (1-h)[j+r-n+1]
    r_lo = max(0, n - 1 - j)
    r_hi = min(2*n-2, 2*n-2-j)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        idx = j + rr - n + 1
        b[rr] += dir_sign * (1.0 - h[idx])
    
    # Term: +dir_sign * h[i-r+n-1]
    r_lo = max(0, i - n + 1)
    r_hi = min(2*n-2, i)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        idx = i - rr + n - 1
        b[rr] += dir_sign * h[idx]
    
    # Term: -dir_sign * h[j-r+n-1]
    r_lo = max(0, j - n + 1)
    r_hi = min(2*n-2, j)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        idx = j - rr + n - 1
        b[rr] -= dir_sign * h[idx]
    
    # c_r: -corr(V,V,full)[r]
    # V[i] = -dir_sign, V[j] = dir_sign
    # corr(V,V)[n-1] = V[i]^2 + V[j]^2 = 2
    # corr(V,V)[n-1+j-i] = V[i]*V[j] = -1
    # corr(V,V)[n-1+i-j] = V[j]*V[i] = -1
    c = np.zeros(2*n-1)
    c[n-1] = -2.0
    c[n-1+j-i] = 1.0
    c[n-1+i-j] = 1.0
    
    # Solve 1D: min_{alpha in [0, alpha_max]} max_r (a[r] + b[r]*alpha + c[r]*alpha^2)
    alphas = np.linspace(0.0, alpha_max, n_samples)
    # Vectorized: scores[r, s] = a[r] + b[r]*alphas[s] + c[r]*alphas[s]^2
    scores = a[:, None] + b[:, None] * alphas[None, :] + c[:, None] * (alphas[None, :])**2
    max_scores = scores.max(axis=0)
    best_i = int(np.argmin(max_scores))
    best_alpha = float(alphas[best_i])
    best_val = float(max_scores[best_i])
    
    # Refine with golden section
    if 0 < best_i < n_samples - 1:
        lo = float(alphas[best_i - 1])
        hi = float(alphas[best_i + 1])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _eval_max_quadratic(a, b, c, c1)
        f2 = _eval_max_quadratic(a, b, c, c2)
        for _ in range(30):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2; f2 = f1; c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _eval_max_quadratic(a, b, c, c1)
            else:
                lo = c1; f1 = f2; c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _eval_max_quadratic(a, b, c, c2)
        mid = (lo + hi) / 2.0
        fm = _eval_max_quadratic(a, b, c, mid)
        if fm < best_val:
            best_alpha = mid
            best_val = fm
    
    return best_alpha, best_val

def _fast_c5_raw(h):
    """Return max of raw correlation (without dx factor)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max())

def _pair_descent(h, deadline, max_rounds=50, n_samples=200):
    """
    Full pair descent: cycle through all pairs, find optimal alpha for each.
    """
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_raw = _fast_c5_raw(h)
    cur_raw = best_raw
    dx = 2.0 / n
    
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        round_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] < 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] > 1.0 - 1e-10:
                    continue
                # Try both directions
                for ds in [1.0, -1.0]:
                    if ds > 0:
                        a_max = min(h[i], 1.0 - h[j])
                    else:
                        a_max = min(h[j], 1.0 - h[i])
                    if a_max < 1e-10:
                        continue
                    best_alpha, best_val = _solve_pair_step(h, i, j, ds, n_samples)
                    if best_val < cur_raw - 1e-8:
                        if ds > 0:
                            h[i] -= best_alpha
                            h[j] += best_alpha
                        else:
                            h[j] -= best_alpha
                            h[i] += best_alpha
                        h = np.clip(h, 0.0, 1.0)
                        h = _project(h, target)
                        cur_raw = _fast_c5_raw(h)
                        if cur_raw < best_raw:
                            best_raw = cur_raw
                            best_h = h.copy()
                        round_improved = True
        
        if round_improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    
    return best_h, best_raw * dx

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_local(h, deadline, radius=0.05, top_k=8, max_iters=2000):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    n_lags = 2 * n - 1
    stall = 0
    r = radius
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        scores = _scores(h)
        # Gradient matrix (sparse-ish)
        G = np.zeros((n_lags, n))
        for d in range(-(n-1), n):
            r_idx = n - 1 + d
            if d >= 0:
                m = n - d
                G[r_idx, d:] += (1.0 - h[:m])
                G[r_idx, :m] += -h[d:]
            else:
                g = -d
                m = n - g
                G[r_idx, :m] += (1.0 - h[g:])
                G[r_idx, g:] += -h[:m]
        G *= (2.0 / n)
        
        if top_k and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
        else:
            idx = np.arange(n_lags)
        sk = scores[idx]
        Gk = G[idx]
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[jj]), min(r, 1.0 - h[jj])) for jj in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                          b_eq=np.zeros(1), bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        
        if np.max(np.abs(p)) < 1e-12:
            stall += 1
            r *= 0.7
            if r < 1e-8:
                break
            continue
        
        # Line search
        alpha_max = 1.0
        for jj in range(n):
            if p[jj] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[jj]) / p[jj])
            elif p[jj] < -1e-14:
                alpha_max = min(alpha_max, h[jj] / (-p[jj]))
        alpha_max = max(alpha_max, 1e-12)
        
        alphas = np.linspace(0, alpha_max, 30)
        best_a = 0.0
        best_v = cur_c5
        for a in alphas:
            h_try = h + a * p
            h_try = _project(h_try, target)
            v = _exact_c5(h_try)
            if v < best_v:
                best_v = v
                best_a = a
        
        if best_v < cur_c5 - 1e-12:
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.3)
        else:
            stall += 1
            r *= 0.8
            if stall > 20 and r < 0.01:
                r = 0.05
    
    return best_h, best_c5

def _pdescent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1500):
    """P-norm soft-max descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    n_lags = 2 * n - 1
    stall = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        # Gradient
        G = np.zeros((n_lags, n))
        for d in range(-(n-1), n):
            r_idx = n - 1 + d
            if d >= 0:
                m = n - d
                G[r_idx, d:] += (1.0 - h[:m])
                G[r_idx, :m] += -h[d:]
            else:
                g = -d
                m = n - g
                G[r_idx, :m] += (1.0 - h[g:])
                G[r_idx, g:] += -h[:m]
        G *= (2.0 / n)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 60:
                break
            continue
        d_vec = -grad / gn
        # Line search
        alpha_max = 0.5
        for jj in range(n):
            if d_vec[jj] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[jj]) / d_vec[jj])
            elif d_vec[jj] < -1e-14:
                alpha_max = min(alpha_max, h[jj] / (-d_vec[jj]))
        alpha_max = max(alpha_max, 1e-12)
        alphas = np.linspace(0, alpha_max, 20)
        best_a = 0.0
        best_v = cur_c5
        for a in alphas:
            h_try = _project(h + a * d_vec, target)
            v = _exact_c5(h_try)
            if v < best_v:
                best_v = v
                best_a = a
        if best_v < cur_c5 - 1e-13:
            h = _project(h + best_a * d_vec, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            if stall > 80:
                break
    return best_h, best_c5

def _make_binary_seed(n, seed):
    """Create a structured binary initial point."""
    rng = np.random.default_rng(seed)
    # Try golden ratio placement
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    ones = n // 2
    positions = set()
    for i in range(ones):
        pos = int((i * phi * n) % n)
        positions.add(pos)
    # If we don't have enough, add random
    while len(positions) < ones:
        pos = rng.integers(n)
        positions.add(pos)
    pos_list = list(positions)[:ones]
    h = np.zeros(n)
    h[pos_list] = 1.0
    return h

def _make_symmetric_init(n, seed):
    """Create a symmetric initial point: h[i] + h[n-1-i] = 1."""
    rng = np.random.default_rng(seed)
    if n % 2 != 0:
        return _make_binary_seed(n, seed)
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        val = rng.uniform(0, 1)
        h[i] = val
        h[n - 1 - i] = 1.0 - val
    return h

def _pair_descent_v2(h, deadline, max_rounds=100, n_samples=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_raw = float(np.correlate(h, 1.0-h, mode='full').max())
    cur_raw = best_raw
    dx = 2.0 / n
    
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        a = np.correlate(h, 1.0-h, mode='full')  # Precompute once per round
        cur_raw = float(a.max())
        round_improved = False
        
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] < 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] > 1.0 - 1e-10:
                    continue
                for ds in (1.0, -1.0):
                    if ds > 0:
                        a_max = h[i] if h[i] < 1.0 - h[j] else 1.0 - h[j]
                    else:
                        a_max = h[j] if h[j] < 1.0 - h[i] else 1.0 - h[i]
                    if a_max < 1e-10:
                        continue
                    # Compute b, c for this pair
                    b, c = _pair_bc(h, i, j, ds, n)
                    # 1D solve
                    best_alpha, best_val = _solve_1d(a, b, c, 0.0, a_max, n_samples)
                    if best_val < cur_raw - 1e-8:
                        if ds > 0:
                            h[i] -= best_alpha
                            h[j] += best_alpha
                        else:
                            h[j] -= best_alpha
                            h[i] += best_alpha
                        h = np.clip(h, 0.0, 1.0)
                        # Quick sum correction
                        diff = target - float(h.sum())
                        if abs(diff) > 1e-12:
                            if diff > 0:
                                mask = h < 1.0
                            else:
                                mask = h > 0.0
                            cnt = int(mask.sum())
                            if cnt > 0:
                                h[mask] += diff / cnt
                                h = np.clip(h, 0.0, 1.0)
                        # Recompute scores
                        a = np.correlate(h, 1.0-h, mode='full')
                        cur_raw = float(a.max())
                        if cur_raw < best_raw:
                            best_raw = cur_raw
                            best_h = h.copy()
                        round_improved = True
        
        if round_improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    
    return best_h, best_raw * dx

And `_pair_bc` computes just b and c (not a, since a is precomputed):

```python
def _pair_bc(h, i, j, ds, n):
    b = np.zeros(2*n - 1)
    r_lo = max(0, n - 1 - i)
    r_hi = min(2*n-2, 2*n-2-i)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        b[rr] += -ds * (1.0 - h[i + rr - n + 1])
    
    r_lo = max(0, n - 1 - j)
    r_hi = min(2*n-2, 2*n-2-j)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        b[rr] += ds * (1.0 - h[j + rr - n + 1])
    
    r_lo = max(0, i - n + 1)
    r_hi = min(2*n-2, i)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        b[rr] += ds * h[i - rr + n - 1]
    
    r_lo = max(0, j - n + 1)
    r_hi = min(2*n-2, j)
    if r_lo <= r_hi:
        rr = np.arange(r_lo, r_hi+1)
        b[rr] -= ds * h[j - rr + n - 1]
    
    c = np.zeros(2*n - 1)
    c[n-1] = -2.0
    c[n-1+j-i] = 1.0
    c[n-1+i-j] = 1.0
    return b, c
```
