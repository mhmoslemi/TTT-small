    # Using the correlation structure
    C = np.correlate(h, 1.0 - h, mode='full')  # C_k
    # B_k = corr(d, 1-h) - corr(h, d) ... let me verify
    # For shift k (in correlate 'full' indexing, position n-1+k corresponds to shift k):
    # C_{n-1+k} = sum_{i} h[i] * (1-h[i+k]) for valid i
    # B_{n-1+k} = sum_{i} d[i]*(1-h[i+k]) - h[i]*d[i+k]

def _grad_mat(h):
    """Compute G where G[r, j] = df_k/ dh_j for shift k = r - (n-1)"""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m_ = n - d
            G[r, d:] += (1.0 - h[:m_])
            G[r, :m_] += -h[d:]
        else:
            g = -d
            m_ = n - g
            G[r, :m_] += (1.0 - h[g:])
            G[r, g:] += -h[:m_]
    return G

def _lp_dir(h, scores, G, radius):
    """Find LP direction to reduce max overlap"""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_exact_loop(h, deadline, radius_init=0.15):

    n = len(h)

    target = n / 2.0

        p = _lp_dir(h, scores, G, radius)

            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue

        pos = p > 1e-14

            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        
        alpha, new_c5 = _exact_line_search(h, p, t_lo, t_hi)

            h = h + alpha * p
            h = _project(h, target)

                best_c5 = cur
                best_h = h.copy()

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

        if new_c5 < cur - 1e-14:

            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.5)
        else:
            stall += 1
            radius *= 0.7
            if stall % 25 == 0 and radius < 0.03:
                radius = 0.1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_refine(h, deadline, rng, n_pert=40):
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n = len(h)
    scale = 0.04
    for i in range(n_pert):
        if time.time() > deadline:
            break
        h_p = best_h + rng.normal(0.0, scale, n)
        h_p = _project(h_p, n / 2.0)
        sub_dl = min(deadline, time.time() + max(2.0, (deadline - time.time()) / max(1, n_pert - i)))
        h_r, c5_r = _lp_exact_loop(h_p, sub_dl, radius_init=0.08)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            scale = max(0.002, scale * 0.82)
        else:
            scale = min(0.18, scale * 1.18)
    return best_h, best_c5

def _pair_sweep(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    passes = 0
    while time.time() < deadline and passes < 60:
        passes += 1
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            if h[i] <= 1e-12 or h[i] >= 1.0 - 1e-12:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                t_lo = max(-h[i], -h[j])
                t_hi = min(1.0 - h[i], 1.0 - h[j])
                if t_hi - t_lo < 1e-14:
                    continue
                alpha, c5_try = _exact_line_search(h, d, t_lo, t_hi)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _soft_dir(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    G = _grad_mat(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    return grad

def _sd_loop(h, deadline, tau_list=(80, 200, 500), lr_init=0.03):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    lr = lr_init
    mom = np.zeros(n)
    decay = 0.85
    stall = 0
    ti = 0
    while time.time() < deadline:
        if stall > 80:
            break
        tau = tau_list[ti % len(tau_list)]
        d = _soft_dir(h, dx, tau)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            ti += 1
            continue
        d = d / dn
        mom = decay * mom + (1.0 - decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            continue
        mom = mom / mn
        step = lr * mom
        pos = step > 1e-14
        neg = step < -1e-14
        t_hi = lr
        t_lo = -lr
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / step[pos])))
        if np.any(neg):
            t_lo = max(t_lo, float(np.max(h[neg] / step[neg])))
        if t_hi - t_lo < 1e-14:
            stall += 1
            lr *= 0.8
            continue
        alpha, new_c5 = _exact_line_search(h, step, t_lo, t_hi)
        if new_c5 < cur - 1e-14:
            h = h + alpha * step
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.35)
        else:
            stall += 1
            lr *= 0.82
            if stall % 30 == 0:
                ti += 1
                mom = np.zeros(n)
    return best_h, best_c5

def _step_function(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: SD warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _sd_loop(h, d1, tau_list=(100, 250), lr_init=0.025)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP + exact line search (22%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _lp_exact_loop(best_h, d2, radius_init=0.18)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution (25%)
    d3 = t0 + budget_s * 0.55
    best_n = len(best_h)
    n_offsets = sorted(range(-30, 31), key=lambda x: abs(x))
    for dn in n_offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 250:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.011)
        h_r, c5_r = _lp_exact_loop(h_r, sub_dl, radius_init=0.12)
        rem = d3 - time.time()
        if rem > 2 and c5_r < best_c5 + 0.004:
            sub_dl2 = min(d3, time.time() + max(2.0, budget_s * 0.005))
            h_r, c5_r = _lp_exact_loop(h_r, sub_dl2, radius_init=0.06)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Perturbation restarts (17%)
    d4 = t0 + budget_s * 0.72
    best_h, best_c5 = _perturb_refine(best_h, d4, rng, n_pert=50)

    # Phase 5: SD on current best (10%)
    d5 = t0 + budget_s * 0.82
    h5, c5_5 = _sd_loop(best_h, d5, tau_list=(60, 150, 400), lr_init=0.02)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Alternative starts (10%)
    d6 = t0 + budget_s * 0.92
    best_n = len(best_h)
    alt_starts = [
        _step_function(best_n),
        _step_function(best_n + 4),
        _step_function(best_n - 4),
    ]
    for alt in alt_starts:
        if time.time() > d6:
            break
        na = len(alt)
        alt = _project(alt, na / 2.0)
        c5_alt = _exact_c5(alt)
        if c5_alt > best_c5 + 0.02:
            continue
        sub_dl = min(d6, time.time() + max(3.0, budget_s * 0.015))
        h_alt, c5_alt2 = _lp_exact_loop(alt, sub_dl, radius_init=0.2)
        if c5_alt2 < best_c5 - 1e-13:
            best_c5 = c5_alt2
            best_h = h_alt.copy()

    # Phase 7: Pair sweep (6%)
    d7 = t0 + budget_s * 0.98
    h_p, c5_p = _pair_sweep(best_h, d7)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 8: Final LP polish
    d8 = t0 + budget_s * 0.995
    h_f, c5_f = _lp_exact_loop(best_h, d8, radius_init=0.04)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
