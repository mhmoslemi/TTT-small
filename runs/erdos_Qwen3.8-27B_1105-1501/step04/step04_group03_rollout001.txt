    exp_arg = np.clip((sc - c_max) / temperature, -500.0, 0.0)
    w = np.exp(exp_arg)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        top_idx = np.argsort(sc)[-10:]
        w = np.zeros(len(sc))
        w[top_idx] = 1.0 / len(top_idx)
    else:
        w = w / w_sum
    center = n - 1
    g = np.zeros(n)
    for r in range(len(sc)):
        wk = w[r]
        if wk < 1e-15:
            continue
        d = r - center
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            g[:m] += wk * (-h[d:d + m])
            g[d:] += wk * (1.0 - h[:m])
        else:
            q = -d
            m = n - q
            if m <= 0:
                continue
            g[:m] += wk * (1.0 - h[q:q + m])
            g[q:] += wk * (-h[:m])
    dx = 2.0 / n
    return dx * g

def line_search_exact(h, p, target):
    n = len(h)
    best_c5 = exact_c5(h)
    best_alpha = 0.0
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-15:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_grid = 24
    besti = 0
    bestv = best_c5
    for i in range(1, n_grid + 1):
        a = alpha_max * i / n_grid
        ht = h + a * p
        if np.min(ht) < -1e-12 or np.max(ht) > 1.0 + 1e-12:
            continue
        c5t = exact_c5(ht)
        if c5t < bestv:
            bestv = c5t
            besti = i
    if besti == 0:
        return 0.0, best_c5, best_c5
    lo_a = alpha_max * max(0, besti - 1) / n_grid
    hi_a = alpha_max * min(n_grid, besti + 1) / n_grid
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(18):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < bestv:
        return opt_a, opt_c5, best_c5
    a_best = alpha_max * besti / n_grid
    return a_best, bestv, best_c5

def gradient_polish(h, target, deadline, tau_init=0.05, tau_min=0.001, max_iters=50000):
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = active_gradient(h, tau)
        p = g - np.mean(g)
        pnorm = np.max(np.abs(p))
        if pnorm < 1e-12:
            tau *= 0.7
            stall += 1
            if stall > 50:
                break
            continue
        p = p / pnorm
        a, new_c5, _ = line_search_exact(h, p, target)
        if a > 1e-12 and new_c5 < cur_c5 - 1e-14:
            h = h + a * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            tau *= 0.98
            tau = max(tau, tau_min)
        else:
            stall += 1
            tau *= 0.6
            tau = max(tau, tau_min)
            if stall > 100:
                break
    return best_h, best_c5

def pair_polish_exact(h, target, deadline, max_rounds=200):
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _trial in range(n * 2):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            amt = 0.05 + 0.45 * rng.random()
            if cur_c5 - exact_c5(h) > 0.01:
                break
            for sign in [1.0, -1.0]:
                if sign > 0:
                    amax = min(h[i], 1.0 - h[j]) * amt
                else:
                    amax = min(1.0 - h[i], h[j]) * amt
                if amax < 1e-10:
                    continue
                ht = h.copy()
                ht[i] += sign * amax
                ht[j] -= sign * amax
                ht = project(ht, target)
                c5t = exact_c5(ht)
                if c5t < cur_c5 - 1e-13:
                    h = ht
                    cur_c5 = c5t
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def segment_moves(h, target, deadline, rng, max_moves=500):
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(max_moves):
        if time.time() > deadline:
            break
            a = int(rng.integers(0, n - 2))
        length = int(rng.integers(2, min(15, n - a)))
        b = a + length
        ht = h.copy()
        ht[a:b] = h[a:b][::-1]
        ht = project(ht, target)
        c5t = exact_c5(ht)
        if c5t < cur_c5 - 1e-13:
            h = ht
            cur_c5 = c5t
            if cur_c5 < best_c5:
                best_c5 = c5t
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def basin_hopping(h_best, c5_best, target, deadline, rng, n_perturbs=20):
    h = h_best.copy()
    best_h = h.copy()
    best_c5 = c5_best
    for idx in range(n_perturbs):
        if time.time() > deadline:
            break
        scale = 0.03 + 0.1 * (1.0 - idx / max(1, n_perturbs))
        h_pert = h_best + rng.normal(0, scale, len(h_best))
        h_pert = project(h_pert, target)
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.35)
        h_try, c5_try = gradient_polish(h_pert, target, sub_dl, tau_init=0.03, tau_min=0.002, max_iters=3000)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.try.copy() if hasattr(h_try, 'copy') else h_try.copy()
            h_best = h_try.copy()
    return best_h, best_c5

def initial_construction(n, seed):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / n
    h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x + 0.5)
    return project(h, n / 2.0)

def symmetric_variants(h):
    n = len(h)
    variants = []
    variants.append(h.copy())
    variants.append(np.cos(h).copy())
    if n % 2 == 0:
        half = n // 2
        t = (h[:half] + h[half:]) / 2.0
        g = np.zeros(n)
        g[:half] = t
        g[half:] = 1.0 - t
        variants.append(project(g, n / 2.0))
    t2 = (h + h[::-1]) / 2.0
    variants.append(t2.copy())
    variants.append(project(1.0 - h, n / 2.0).copy())
    return variants

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    # Initialize from parent or fallback
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        n0 = 92
        h0 = initial_construction(n0, seed)
    h0 = project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    best_n = n0
    target = n0 / 2.0

    # Phase 1: Aggressive gradient polish from parent
    d1 = t0 + budget_s * 0.15
    h_gp, c5_gp = gradient_polish(best_h, target, d1, tau_init=0.08, tau_min=0.001, max_iters=200000)
    if c5_gp < best_c5:
        best_c5 = c5_gp
        best_h = h_gp.copy()

    # Phase 2: Second gradient polish with finer parameters
    d2 = t0 + budget_s * 0.28
    h_gp2, c5_gp2 = gradient_polish(best_h, target, d2, tau_init=0.02, tau_min=0.0005, max_iters=200000)
    if c5_gp2 < best_c5:
        best_c5 = c5_gp2
        best_h = h_gp2.copy()

    # Phase 3: Pair polish (stochastic local search)
    d3 = t0 + budget_s * 0.40
    h_pp, c5_pp = pair_polish_exact(best_h, target, d3, max_rounds=500)
    if c5_pp < best_c5:
        best_c5 = c5_pp
        best_h = h_pp.copy()

    # Phase 4: Basin hopping
    d4 = t0 + budget_s * 0.70
    best_h, best_c5 = basin_hopping(best_h, best_c5, target, d4, rng, n_perturbs=30)

    # Phase 5: Re-polish after basin hopping
    d5 = t0 + budget_s * 0.80
    h_gp3, c5_gp3 = gradient_polish(best_h, target, d5, tau_init=0.015, tau_min=0.0003, max_iters=200000)
    if c5_gp3 < best_c5:
        best_c5 = c5_gp3
        best_h = h_gp3.copy()

    # Phase 6: Multi-resolution search
    d6 = t0 + budget_s * 0.90
    offsets = [-10, -8, -6, -5, -4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 8, 10, 14, 20]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        h_r = resample_h(best_h, n_try)
        t_r = n_try / 2.0
        sub_dl = min(d6, time.time() + budget_s * 0.008)
        h_r2, c5_r = gradient_polish(h_r, t_r, sub_dl, tau_init=0.03, tau_min=0.001, max_iters=10000)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r2.copy()
            best_n = n_try
            target = t_r

    # Phase 7: Final micro-polish
    h_final, c5_final = gradient_polish(best_h, target, hard_deadline, tau_init=0.005, tau_min=0.0002, max_iters=100000)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_c5 = exact_c5(h0)
    if final_c5 > parent_c5 + 1e-14:
        best_h = h0.copy()
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
