The key challenge is that at the optimum, multiple lags are likely tied or nearly tied at the maximum. The smooth max approach handles this well because it naturally distributes gradient across all near-max constraints.

Let me also think about whether there's a structural insight. The problem is:
minimize max_d ∫ h(x)(1-h(x+d)) dx

For the continuous version, if h is a {0,1} function (indicator of a set A), then:
∫_A (1-h(x+d)) dx = measure(A \ (A-d))

So we want to minimize the maximum over all shifts d of the measure of A minus (A shifted by d). This is related to the "holes" in the set when shifted.

For a periodic set with period p, the overlap at shift d is related to the discrepancy between the set and its shift by d.

The golden ratio (1/phi ≈ 0.618) is often optimal for such problems because it maximizes the minimum distance between points in a cyclic group.

import numpy as np
from scipy.optimize import linprog
import time

def exact_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_max_grad(h, tau):
    """
    Compute F_tau(h) and its gradient.
    F_tau(h) = tau * log(sum_r exp(S_r / tau))
    where S_r = dx * corr_r
    """
    n = len(h)
    dx = 2.0 / n
    scores = exact_scores(h)
    P = len(scores)
    
    # Stable log-sum-exp
    M = np.max(scores)
    u = np.exp((scores - M) / tau)
    Z = np.sum(u)
    F = M + tau * np.log(Z)
    w = u / Z
    
    # Compute gradient
    # For lag d (where d = r - (n-1)):
    # If d == 0: dS/dh[j] = dx * (1 - 2*h[j]) for all j
    # If d > 0: dS_d/dh[j] = dx * [(1-h[j-d]) if j>=d, -h[j+d] if j<=n-1-d]
    # If d < 0: dS_{-d}/dh[j] = dx * [(1-h[j+|d|]) if j<=n-1-|d|, -h[j-|d|] if j>=|d|]
    
    grad = np.zeros(n)
    for r in range(P):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        wr = w[r] * dx
        if d == 0:
            grad += wr * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            # h[j+d] term: j in [0, m-1], contributes (1-h[j]) to h[j+d] position
            # So for position j+d (j in [0,m-1], i.e., position in [d, n-1]): += (1-h[position-d])  wait
            # Let me redo: S_d = sum_{j=0}^{m-1} h[j+d]*(1-h[j])
            # d/dh[p] S_d:
            #   if p in [0, m-1]: h[p] appears in (1-h[j]) with j=p, contributes -h[p+d]
            #   if p in [d, n-1]: h[p] appears in h[j+d] with j=p-d, contributes (1-h[p-d])
            #   overlap if p in [d, m-1]: both terms
            grad[d:] += wr * (1.0 - h[:m])  # positions d to n-1 get (1-h[0] to 1-h[m-1])
            grad[:m] += -wr * h[d:]  # positions 0 to m-1 get -h[d] to -h[n-1]
        else:
            g = -d
            m = n - g
            # S_{-g} = sum_{j=0}^{m-1} h[j]*(1-h[j+g])
            # d/dh[p] S_{-g}:
            #   if p in [0, m-1]: h[p] appears directly, contributes (1-h[p+g])
            #   if p in [g, n-1]: h[p] appears in (1-h[j+g]) with j+g=p, j=p-g, contributes -h[p-g]
            grad[:m] += wr * (1.0 - h[g:])  # positions 0 to m-1 get (1-h[g] to 1-h[n-1])
            grad[g:] += -wr * h[:m]  # positions g to n-1 get -h[0] to -h[m-1]
    
    return F, grad

def constrained_direction(grad, n):
    """Project gradient onto zero-sum subspace."""
    p = -grad.copy()
    p -= np.mean(p)
    # Normalize to infinity norm
    pmax = np.max(np.abs(p))
    if pmax < 1e-15:
        return None
    p /= pmax
    return p

def safe_alpha_max(h, p):
    """Find maximum step size that keeps h + alpha*p in [0,1]."""
    alpha_max = 1.0  # since p is inf-norm normalized
    for j in range(len(h)):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    return max(alpha_max, 1e-14) * 0.999

def line_search(h, p, tau, trust):
    """
    Line search along h + alpha*p for alpha in [0, min(trust, alpha_max)].
    Returns (best_alpha, best_c5, best_F).
    """
    amax = min(trust, safe_alpha_max(h, p))
    if amax < 1e-14:
        return 0.0, exact_c5(h), smooth_max_grad(h, tau)[0]
    
    # Grid search
    n_pts = 15
    alphas = np.linspace(0, amax, n_pts)
    c5s = np.array([exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(c5s))
    best_c5 = c5s[best_i]
    best_a = alphas[best_i]
    
    if best_i == 0 or best_i == n_pts - 1:
        return float(best_a), float(best_c5), 0
    
    # Golden section refinement
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_pts - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return float(opt_a), float(opt_c5), 0
    return float(best_a), float(best_c5), 0

def soft_anneal(h, deadline, tau_schedule=None, rng=None):

    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    stall_count = 0
    max_stall = 50

        max_iters = 500

            if stall_count > max_stall:

            else:

                trust *= 0.7

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def weighted_grad_step(h, deadline, n_steps=100):

        beta = 5.0 + 200.0 * frac
        scores = exact_scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        w /= sw
        # Compute weighted gradient (same as smooth max grad but with different weights)
        dx = 2.0 / n
        grad = np.zeros(n)
        P = len(scores)
        for r in range(P):
            if w[r] < 1e-15:

            elif d > 0:

                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] += -wr * h[d:]
            else:
                g = -d
                m = n - g
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] += -wr * h[:m]
        
        p = -grad
        p -= np.mean(p)

        if pmax < 1e-15:

        p /= pmax

def pair_polish(h, deadline, max_passes=4):

                stall += 1
                trust *= 0.7
                if trust < 0.004:
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = exact_c5(h_start)
    best_n = n0

    # Phase 1: Smooth max annealing (primary method)
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = soft_anneal(h_start, dl1,
                           tau_list=[0.10, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008],
                           rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP-based annealing
    if time.time() < t0 + budget_s * 0.42:
        dl2 = t0 + budget_s * 0.45
        h2, c5_2 = lp_anneal(best_h, dl2, radius_init=0.12, max_iters=1500)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Weighted (spiky) gradient
    if time.time() < t0 + budget_s * 0.52:
        dl3 = t0 + budget_s * 0.56
        h3, c5_3 = weighted_grad_opt(best_h, dl3, n_steps=120)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Mixed direction (smooth + LP alternating)
    if time.time() < t0 + budget_s * 0.68:
        dl4 = t0 + budget_s * 0.72
        h4, c5_4 = mixed_direction_anneal(best_h, dl4,
                                           tau_list=[0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005],
                                           rng=rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Multi-resolution search
    if time.time() < t0 + budget_s * 0.80:
        dl5 = t0 + budget_s * 0.82
        for n_try in [96, 112, 144]:
            if time.time() > dl5:
                break
            if n_try == n0:
                continue
            h_init = resample(best_h, n_try)
            sub_dl = min(dl5, time.time() + budget_s * 0.06)
            h_try, c5_try = soft_anneal(h_init, sub_dl,
                                         tau_list=[0.04, 0.02, 0.01, 0.005, 0.002, 0.001],
                                         rng=rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 6: Perturbation restarts
    if time.time() < t0 + budget_s * 0.90:
        dl6 = t0 + budget_s * 0.92
        scale = 0.025
        for attempt in range(80):
            if time.time() > dl6:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = project(h_pert, best_n / 2.0)
            sub_dl = min(dl6, time.time() + budget_s * 0.015)
            h_try, c5_try = soft_anneal(h_pert, sub_dl,
                                         tau_list=[0.015, 0.008, 0.004, 0.002, 0.001],
                                         rng=rng)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                scale = max(0.005, scale * 0.82)
            else:
                scale = min(0.06, scale * 1.1)

    # Phase 7: Pair polish
    if time.time() < t0 + budget_s * 0.96:
        dl7 = t0 + budget_s * 0.975
        hp, c5p = pair_polish(best_h, dl7, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final fine refinement
    if time.time() < t0 + budget_s * 0.99:
        h_fin, c5_fin = soft_anneal(best_h, t0 + budget_s * 0.995,
                                     tau_list=[0.002, 0.001, 0.0005, 0.0003],
                                     rng=rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
