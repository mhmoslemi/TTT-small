    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)

        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _coord_descent(h, n, n_sweeps=30, n_grid=40, deadline=None):
    h = h.copy()
    target = n * 0.5
    best_c5 = _exact_c5(h)
    grid = np.linspace(0.0, 1.0, n_grid)
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            best_val = old_val
            best_obj = best_c5
            for v in grid:
                if abs(v - old_val) < 1e-10:
                    continue
                h[i] = v
                h_p = _project(h, target)
                c = _exact_c5(h_p)
                if c < best_obj - 1e-14:
                    best_obj = c
                    best_val = v
            h[i] = best_val
            if best_val != old_val:
                h = _project(h, target)
                improved = True
                if best_obj < best_c5:
                    best_c5 = best_obj
        if not improved and sweep > 3:
            break
    return h, best_c5


def _pair_swap(h, n, n_rounds=10, deadline=None):
    h = h.copy()
    target = n * 0.5
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    for rnd in range(n_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for _ in range(50):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            base = best_c5
            best_delta = 0.0
            best_c = base
            for delta in np.linspace(-0.5, 0.5, 21):
                if delta == 0:
                    continue
                hi_new = h[i] + delta
                hj_new = h[j] - delta
                if hi_new < -1e-10 or hi_new > 1 + 1e-10:
                    continue
                if hj_new < -1e-10 or hj_new > 1 + 1e-10:
                    continue
                h[i], h[j] = hi_new, hj_new
                c = _exact_c5(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_delta = delta
            if best_delta != 0:
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                improved = True
            else:
                h[i], h[j] = h[i], h[j]  # no change
        if not improved:
            break
    return h, best_c5


def _build_grad_matrix(h, n, active_idx):
    """Build gradient rows for active correlation indices."""
    K = len(active_idx)
    J = np.zeros((K, n))
    j = np.arange(n)
    for ki, r in enumerate(active_idx):
        s = r - (n - 1)
        row = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        row[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        row[m2] -= h[ps[m2]]
        J[ki, :] = row
    return J


def _lp_step(h, n, radius, top_k=30):
    corr = _corr_vec(h)
    raw_max = float(np.max(corr))
    # Active set: top K plus near-max
    threshold = raw_max * 0.9
    active = np.where(corr >= threshold)[0]
    if len(active) > top_k:
        order = np.argsort(-corr[active])[:top_k]
        active = active[order]
    
    J = _build_grad_matrix(h, n, active)
    K = len(active)
    scores = corr[active]
    
    # LP: min T s.t. J@p - T <= -scores + raw_max (i.e., J@p <= scores - raw_max... 
    # Actually we want to find p such that for active r: corr_r + J_r @ p <= raw_max - eps
    # Linearized: scores_r + J_r @ p <= raw_max - 1e-6
    # => J_r @ p <= raw_max - 1e-6 - scores_r
    # Minimize ||p|| (use T as surrogate: min max_r (scores_r + J_r@p - raw_max))
    # Simpler: minimize sum_r (scores_r + J_r@p) = minimize J_sum @ p
    # Subject to: J_r @ p <= (raw_max - scores_r) - 1e-6 for all active r
    #             sum(p) = 0
    #             |p_i| <= radius, and h_i + p_i in [0,1]
    
    m = len(active)
    c_obj = np.zeros(n)
    A_ub = J.copy()
    b_ub = raw_max - scores - 1e-4
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_b = max(-radius, -h[i])
        hi_b = min(radius, 1.0 - h[i])
        bounds.append((lo_b, hi_b))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            p = res.x
            if np.max(np.abs(p)) > 1e-10:
                return p
    except Exception:
        pass
    return None


def _line_search(h, p, n, c5_cur):
    target = n * 0.5
    alpha_max = 1.0
    for i in range(n):
        if p[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    alpha_max = max(alpha_max, 1e-8) * 0.99
    
    best_a, best_c5 = 0.0, c5_cur
    # Coarse search
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-12:
            continue
        h_try = _project(h + a * p, target)
        c_try = _exact_c5(h_try)
        if c_try < best_c5:
            best_c5, best_a = c_try, a
    
    if best_a <= 0:
        return 0.0, c5_cur
    
    # Golden section refinement
    lo = max(0.0, best_a - alpha_max * 0.15)
    hi = min(alpha_max, best_a + alpha_max * 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(14):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    a = (lo + hi) / 2.0
    c5_final = _exact_c5(_project(h + a * p, target))
    if c5_final < best_c5:
        best_a, best_c5 = a, c5_final
    return best_a, best_c5


def _lp_loop(h, n, deadline, radius_init=0.08, top_k=30, stall_limit=60):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        p = _lp_step(h, n, radius, top_k)
        if p is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            if stall % 15 == 0:
                radius = max(radius, 0.03)
            continue
        alpha, new_c5 = _line_search(h, p, n, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, n * 0.5)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5


def _slsqp_solve(h_init, n, maxiter=300, ftol=1e-13):
    h = np.asarray(h_init, dtype=float).copy()
    corr = _corr_vec(h)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    
    def _epi_f(z, nn):
        return z[nn]
    
    def _epi_g(z, nn):
        g = np.zeros(nn + 1)
        g[nn] = 1.0
        return g
    
    def _eq_f(z, nn):
        return np.sum(z[:nn]) - nn * 0.5
    
    def _eq_g(z, nn):
        g = np.ones(nn + 1)
        g[nn] = 0.0
        return g
    
    def _ineq_f(z, nn):
        hh = z[:nn]
        return z[nn] - np.max(_corr_vec(hh))
    
    def _ineq_g(z, nn):
        hh = z[:nn]
        cc = _corr_vec(hh)
        r_max = int(np.argmax(cc))
        s = r_max - (nn - 1)
        j = np.arange(nn)
        pd = np.zeros(nn)
        gs = j - s
        m1 = (gs >= 0) & (gs < nn)
        pd[m1] = 1.0 - hh[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < nn)
        pd[m2] -= hh[ps[m2]]
        g = np.zeros(nn + 1)
        g[:nn] = -pd
        g[nn] = 1.0
        return g
    
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_f, z0, jac=_epi_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _generate_fourier_seed(n, seed):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / n
    h = np.zeros(n)
    n_harm = rng.integers(2, 6)
    for _ in range(n_harm):
        k = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        h += amp * np.sin(2 * np.pi * k * x + phase)
    h += 0.5
    return _project(h, n * 0.5)


def _generate_binary_seed(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def _generate_smooth_binary(n, seed):
    """Generate a smooth binary-like seed with block structure."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    # Random blocks of 1s and 0s with varying widths
    pos = 0
    while pos < n:
        width = rng.integers(2, max(3, n // 8))
        if rng.random() > 0.5:
            h[pos:pos + width] = 1.0
        pos += width
    # Smooth slightly
    kernel = np.array([0.2, 0.6, 0.2])
    h_smooth = np.convolve(h, kernel, mode='same')
    return _project(h_smooth, n * 0.5)

    for s in range(2):
        starts.append(_generate_smooth_binary(best_n, seed + 300 + s))
    
    for si, h_start in enumerate(starts):
        if time.time() > d3:
            break
        sub_budget = min(d3, time.time() + budget_s * 0.04)
        # Quick coord descent
        h_s, c5_s = _coord_descent(h_start, best_n, n_sweeps=8, n_grid=25, deadline=sub_budget)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s
            sub_budget = min(d3, time.time() + budget_s * 0.03)
        # LP step
        if time.time() < d3:
            h_s2, c5_s2 = _lp_loop(h_s, best_n, sub_budget, radius_init=0.08,
                                    top_k=25, stall_limit=30)
            if c5_s2 < best_c5:
                best_c5, best_h = c5_s2, h_s2
    
    # Phase 4: SLSQP refinement (10% budget)
    d4 = deadline(0.73)
    for attempt in range(3):
        if time.time() > d4:
            break
        h_s, c5_s = _slsqp_solve(best_h, best_n, maxiter=200, ftol=1e-14)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
    
    # Phase 5: Resolution search (15% budget)
    d5 = deadline(0.88)
    for dn in [0, -2, 2, -4, 4, -6, 6, 8, -8, -10, 10]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = _coord_descent(h_r, n_try, n_sweeps=8, n_grid=25, deadline=sub_d)
        sub_d2 = min(d5, time.time() + budget_s * 0.015)
        h_r2, c5_r2 = _lp_loop(h_r, n_try, sub_d2, radius_init=0.06, top_k=20, stall_limit=25)
        if c5_r2 < c5_r:
            h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try
    
    # Phase 6: Final intensive polish (12% budget)
    d6 = deadline(0.99)
    # LP with smaller radius
    h_f1, c5_f1 = _lp_loop(best_h, best_n, min(d6, time.time() + budget_s * 0.04),
                            radius_init=0.03, top_k=40, stall_limit=80)
    if c5_f1 < best_c5:
        best_c5, best_h = c5_f1, h_f1
    # Coord descent
    h_f2, c5_f2 = _coord_descent(best_h, best_n, n_sweeps=20, n_grid=50, deadline=d6)
    if c5_f2 < best_c5:
        best_c5, best_h = c5_f2, h_f2
    # Final LP
    if time.time() < d6:
        h_f3, c5_f3 = _lp_loop(best_h, best_n, d6, radius_init=0.02, top_k=50, stall_limit=100)
        if c5_f3 < best_c5:
            best_c5, best_h = c5_f3, h_f3
    
    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
