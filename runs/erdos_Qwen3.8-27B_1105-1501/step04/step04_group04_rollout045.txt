import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(50):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_gradient(h, beta):
    """Compute gradient of LSE(beta * F) with softmax weights."""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    # softmax weights
    F_shifted = F - F.max()
    w = np.exp(beta * F_shifted)
    w = w / w.sum()
    # Compute gradient
    g = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        wr = w[r] * dx
        if wr < 1e-18:
            continue
        low = max(0, -d)
        high = min(n, n - d)
        i = np.arange(low, high)
        g[i] += wr * (1.0 - h[i + d])
        g[i + d] -= wr * h[i]
    return g, F, w

def lse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    F_shifted = F - F.max()
    return (1.0/beta) * np.log(np.exp(beta * F_shifted).sum()) + F.max()
