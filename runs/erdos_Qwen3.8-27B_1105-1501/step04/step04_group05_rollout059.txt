```python
def _pair_line_search(h, i, j):
    """Exact 1D line search for pair (i,j) with h[i]+t, h[j]-t."""
    n = len(h)
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return h, 0.0
    
    best_t = 0.0
    best_val = _c5(h)
    
    # Dense grid
    ts = np.linspace(t_lo, t_hi, 33)
    vals = []
    for t in ts:
        h_t = h.copy()
        h_t[i] += t
        h_t[j] -= t
        vals.append(_c5(h_t))
    vals = np.array(vals)
    best_idx = np.argmin(vals)
    best_t = ts[best_idx]
    best_val = vals[best_idx]
    
    # Refine around best
    if best_idx > 0 and best_idx < len(ts) - 1:
        t_a = ts[best_idx - 1]
        t_b = ts[best_idx + 1]
        ts2 = np.linspace(t_a, t_b, 21)

            h_t[i] += t
```

```python
def _grad_cq(h, q, n):
    """Gradient of C_q with respect to h."""
    dx = 2.0 / n
    d = q - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        # C_q = dx * sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
        m = n - 1 - d
        for i in range(m + 1):
            g[i + d] += dx * (1.0 - h[i])
            g[i] -= dx * h[i + d]

        m = n - 1 - k
        for i in range(m + 1):
            g[i] += dx * (1.0 - h[i + k])
            g[i + k] -= dx * h[i]
    return g
```

```python
def _golden_pair(h, i, j, n, max_iters=40):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-14:
        return 0.0, None
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f_t(t):
        hi, hj = h[i], h[j]
        h[i] = hi + t
        h[j] = hj - t
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
        h[i] = hi
        h[j] = hj
        return val
    
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_t(c)
    fd = f_t(d)
    
    for _ in range(max_iters):
        if fc <= fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_t(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_t(d)
    
    t_best = (a + b) / 2.0
    f_best = f_t(t_best)
    return t_best, f_best
```

This is clean and safe. The function doesn't modify h (except temporarily during evaluation).

The correlation np.correlate(h, 1-h, mode='full') has length 2n-1. The entries correspond to lags from -(n-1) to +(n-1).

For the Erdős problem, we're looking at:
C₅ = max_k ∫₀² h(x)(1-h(x+k)) dx

where k ranges over all real values (or at least all values in [0,2] or [-2,2]). In the discretization, k takes values that are multiples of dx = 2/n.

The correlation gives us ∫ h(x)(1-h(x+k)) dx for discrete k values. The max over these discrete k values is our objective.

Now, here's a key observation: the integral ∫₀² h(x)(1-h(x+k)) dx for arbitrary k (not just discrete multiples) could be larger than the max over discrete k. But since we're working with the discrete approximation, we just maximize over the discrete lags.

import numpy as np

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def _project(v, n):
    target = n / 2.0
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _golden_pair(h, i, j, n, max_iters=40):
    hi_val = h[i]
    hj_val = h[j]
    t_lo = max(-hi_val, hj_val - 1.0)
    t_hi = min(1.0 - hi_val, hj_val)
    if t_hi - t_lo < 1e-14:
        return 0.0, None
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    dx = 2.0 / n
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] = hi_val + c; h[j] = hj_val - c
    fc = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    h[i] = hi_val; h[j] = hj_val
    h[i] = hi_val + d; h[j] = hj_val - d
    fd = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    h[i] = hi_val; h[j] = hj_val

        if fc <= fd:

            c = b - gr * (b - a)
