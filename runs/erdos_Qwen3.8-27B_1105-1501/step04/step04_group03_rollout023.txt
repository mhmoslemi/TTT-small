def project(v, target):

        return np.zeros(n)

        else:

    n = len(h)
    scores = corr_scores(h)
    p = active_lp(h, scores, top_k, radius)
    if p is None:
        return h, exact_c5(h)
    
    # Line search
    dx = 2.0 / n
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    c5_cur = exact_c5(h)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(20):
        a = alpha_max * (i + 1) / 20.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    
    if best_a > 0:
        h_new = project(h + best_a * p, n / 2.0)
        c5_new = exact_c5(h_new)
        if c5_new < c5_cur:
            return h_new, c5_new
    return h, c5_cur

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    dx = 2.0 / n
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                # Try moving mass from j to k and vice versa
                amax = min(best_h[j], 1.0 - best_h[k])
                if amax < 1e-10:
                    continue
                # Try a few step sizes
                for frac in [0.5, 1.0, 0.25]:
                    a = amax * frac
                    h_t = best_h.copy()
                    h_t[j] -= a
                    h_t[k] += a
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
                    # Try reverse
                    h_t = best_h.copy()
                    h_t[j] += a
                    h_t[k] -= a
                    # But h_t[k] might go negative
                    if h_t[k] >= -1e-10:
                        h_t[k] = max(0, h_t[k])
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def corr_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def build_G(h, idx_list):
    n = len(h)
    G = np.zeros((len(idx_list), n))
    for ii, k in enumerate(idx_list):
        d = int(k) - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            G[ii, js1:je1+1] += (1.0 - h[js1+d:je1+1+d])
        js2 = max(0, d)
        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            G[ii, js2:je2+1] += -h[js2-d:je2+1-d]
    return G

def smoothed_grad(h, tau, top_k=None):
    n = len(h)
    dx = 2.0 / n
    scores = corr_scores(h)
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(scores)[-top_k:]
        s_use = scores[idx]
    else:
        idx = np.arange(2 * n - 1)
        s_use = scores
    max_s = float(np.max(s_use))
    exp_s = np.exp((s_use - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    G = build_G(h, idx)
    grad = (G.T @ w) * dx
    return val, grad

def momentum_descent(h0, target, tau_start, tau_end, n_iter, deadline, top_k=None, momentum=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    lr = 0.008
    lr_decay = 0.999

    for i in range(n_iter):
        if time.time() > deadline:
            break
        frac = i / max(n_iter - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        val, grad = smoothed_grad(h, tau, top_k)
        v = momentum * v - lr * grad
        lr *= lr_decay

        # Backtracking line search
        alpha = 1.0
        h_trial = project(h + alpha * v, target)
        c5_trial = exact_c5(h_trial)
        for _ in range(8):
            if c5_trial < val + 0.5 * tau * np.log(2 * n - 1) or alpha < 1e-6:
                break
            alpha *= 0.5
            h_trial = project(h + alpha * v, target)
            c5_trial = exact_c5(h_trial)

        c5_trial = exact_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        h = h_trial

    return best_h, best_c5

def active_lp_direction(h, top_k=10, radius=0.1):
    n = len(h)
    dx = 2.0 / n
    scores = corr_scores(h)
    idx = np.argsort(scores)[-top_k:]
    G = build_G(h, idx)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = dx * G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(top_k)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-7:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_step(h, top_k=8, radius=0.1):
    n = len(h)
    p = active_lp_direction(h, top_k, radius)
    if p is None:
        return h, exact_c5(h)

    c5_cur = exact_c5(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)

    best_a = 0.0
    best_c5 = c5_cur
    for i in range(20):
        a = alpha_max * (i + 1) / 20.0
        h_t = h + a * p
        h_t = project(h_t, n / 2.0)
        c5_t = exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a

    if best_a > 0:
        h_new = project(h + best_a * p, n / 2.0)
        c5_new = exact_c5(h_new)
        if c5_new < c5_cur - 1e-14:
            return h_new, c5_new
    return h, c5_cur

def lp_loop(h, deadline, top_k=8, radius_init=0.12, max_stall=50):
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline and stall < max_stall:
        h_try, c5_try = lp_step(h, top_k, radius)
        if c5_try < cur_c5 - 1e-13:
            h = h_try
            cur_c5 = c5_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if radius < 0.015:
                radius = 0.06
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=80):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                amax = min(best_h[j], 1.0 - best_h[k])
                if amax < 1e-12:
                    continue
                for frac in [0.5, 1.0]:
                    a = amax * frac
                    h_t = best_h.copy()
                    h_t[j] -= a
                    h_t[k] += a
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def binary_snap(h, n_target):
    """Try to push values toward 0 or 1 while maintaining sum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    # Sort by distance to nearest endpoint
    dist = np.minimum(h, 1.0 - h)
    order = np.argsort(dist)
    for i in range(len(order)):
        idx = order[i]
        if h[idx] < 0.5:
            h[idx] = 0.0
        else:
            h[idx] = 1.0
    h = project(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Momentum gradient descent - global exploration
    d1 = t0 + budget_s * 0.22
    configs1 = [
        (0.5, 0.05, 250, 12, 0.85),
        (0.25, 0.02, 250, 10, 0.88),
        (0.12, 0.01, 200, 8, 0.9),
        (0.06, 0.005, 150, 8, 0.9),
        (0.03, 0.002, 100, 6, 0.92),
    ]
    for cfg in configs1:
        if time.time() > d1:
            break
        ts, te, ni, tk, mom = cfg
        h_try, c5_try = momentum_descent(best_h, best_n / 2.0, ts, te, ni, d1, top_k=tk, momentum=mom)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 2: Active-set LP
    d2 = t0 + budget_s * 0.40
    h_lp, c5_lp = lp_loop(best_h, d2, top_k=10, radius_init=0.15, max_stall=60)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: Fine momentum descent
    d3 = t0 + budget_s * 0.50
    configs3 = [
        (0.02, 0.002, 100, 6, 0.92),
        (0.01, 0.001, 80, 5, 0.93),
    ]
    for cfg in configs3:
        if time.time() > d3:
            break
        ts, te, ni, tk, mom = cfg
        h_try, c5_try = momentum_descent(best_h, best_n / 2.0, ts, te, ni, d3, top_k=tk, momentum=mom)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: LP micro-refinement
    d4 = t0 + budget_s * 0.60
    h_lp2, c5_lp2 = lp_loop(best_h, d4, top_k=6, radius_init=0.06, max_stall=40)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 5: Diversified restarts
    d5 = t0 + budget_s * 0.78
    scales = [0.015, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35]
    for ns in scales:
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        # Quick momentum
        sub_dl = min(d5, time.time() + budget_s * 0.025)
        h_try, c5_try = momentum_descent(h_pert, best_n / 2.0, 0.15, 0.01, 50, sub_dl, top_k=10)
        # Quick LP
        sub_dl2 = min(d5, time.time() + budget_s * 0.02)
        h_try2, c5_try2 = lp_loop(h_try, sub_dl2, top_k=8, radius_init=0.1, max_stall=20)
        if c5_try2 < best_c5:
            best_c5 = c5_try2
            best_h = h_try2.copy()

    # Phase 6: Multi-resolution
    d6 = t0 + budget_s * 0.87
    tried_n = {best_n}
    for dn in [0, -6, 6, -12, 12, -18, 18, -24, 24, -30, 30, -40, 40, -50, 50]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 180 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.02)
        h_r, c5_r = momentum_descent(h_r, n_try / 2.0, 0.1, 0.01, 40, sub_dl, top_k=10)
        sub_dl2 = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = lp_loop(h_r, sub_dl2, top_k=8, radius_init=0.08, max_stall=20)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Pair polish
    d7 = t0 + budget_s * 0.94
    h_pol, c5_pol = pair_polish(best_h, d7, max_passes=100)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Final LP micro-polish if time remains
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, top_k=5, radius_init=0.03, max_stall=20)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
