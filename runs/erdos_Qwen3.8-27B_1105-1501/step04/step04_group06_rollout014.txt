import numpy as np
from scipy.optimize import minimize, linprog
import time


def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}."""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(x)
            for i in idx:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            idx = np.argsort(-x)
            for i in idx:
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_and_grad_max(h, alpha):
    """Compute LSE-smoothed max of correlations and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    shifted = alpha * (corr - mc)
    exp_v = np.exp(np.clip(shifted, -300, 300))
    Z = float(np.sum(exp_v))
    w = exp_v / Z
    f = mc + (1.0 / alpha) * np.log(Z)
    
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-10:
            continue
        s = r - (n - 1)
        if s >= 0:
            # Term 1: (1-h[j-s]) for j in [s, n-1]
            # Term 2: -h[j+s] for j in [0, n-1-s]
            grad[s:] += w[r] * (1.0 - h[:n-s])
            grad[:n-s] -= w[r] * h[s:]
        else:

            ss = -s
            # Term 1: (1-h[j+s]) for j in [0, n-1-ss]  (j < n-ss)
            # Term 2: -h[j+s] for j in [ss, n-1]
            grad[:n-ss] += w[r] * (1.0 - h[ss:])
            grad[ss:] -= w[r] * h[:n-ss]
    
    return f, grad, corr


def _lse_optimize(h, n, alpha, maxiter=200, ftol=1e-14):
    """Optimize using LSE smoothing with L-BFGS-B + penalty for sum constraint."""
    target = n * 0.5
    
    def objective(x):
        f, g, _ = _corr_and_grad_max(x, alpha)
        # Add penalty for sum constraint
        pen = 1000.0 * (np.sum(x) - target) ** 2
        gpen = 2000.0 * (np.sum(x) - target)
        return f + pen, g + gpen * np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                               'maxls': 30, 'iprint': 0})
        x = _project(res.x, target)
        return x, _exact_c5(x)
    except Exception:

def _binary_from_h(h):
    """Convert h to binary by keeping top n/2 values."""
    n = len(h)
    k = n // 2
    idx = np.argsort(-h)[:k]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def _binary_exchange_search(h, n, max_rounds=200, deadline=None):
    """Local exchange search on binary solution."""
    b = _binary_from_h(h)
    cur_c5 = _exact_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    ones = np.where(b == 1)[0]
    
    for round_ in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Try all pairs (swap a 1 position with a 0 position)
        zeros = np.where(b == 0)[0]
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                b[i], b[j] = b[j], b[i]
                c5 = _exact_c5(b)
                if c5 < best_c5 - 1e-12:
                    best_c5, best_b = c5, b.copy()
                    improved = True
                else:
                    b[i], b[j] = b[j], b[i]
        if not improved:
            break
    
    return best_b, best_c5

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(x)
            for i in idx:
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:

    cur_c5 = _exact_c5(cur)
    best_b, best_c5 = cur.copy(), cur_c5

    h_new = np.interp(x_new, x_old, h)

                    break
                h_s, c5_s = _lse_optimize(h_p, n, alpha, 200, 1e-15, 2000.0)
                if c5_s < best_c5:
                    best_c5, best_h = c5_s, h_s.copy()

    # Phase E: Final SLSQP polish
    if time.time() < deadline:
        h_s, c5_s = _slsqp_active(best_h, n, 10, 300, 1e-15, 3)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()

    best_h = _project(best_h, n * 0.5)
    return best_h, _exact_c5(best_h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)

    best_h = _project(h0, n0 * 0.5)
    best_c5 = _exact_c5(best_h)
    best_n = n0

    # Main optimization at base resolution
    d1 = t0 + budget_s * 0.50
    h_r, c5_r = _full_pipeline(h0, n0, d1, rng)
    if c5_r < best_c5:
        best_c5, best_h, best_n = c5_r, h_r.copy(), n0

    # Try neighboring resolutions
    d2 = t0 + budget_s * 0.85
    resolutions = [n0 - 8, n0 + 8, n0 - 16, n0 + 16, n0 - 4, n0 + 4]
    for dn in resolutions:
        if time.time() > d2:
            break
        if dn < 16 or dn > 128 or dn % 2 != 0:
            continue
        h_res = _resample(best_h, dn)
        sub_dl = min(d2, time.time() + budget_s * 0.06)
        h_r, c5_r = _full_pipeline(h_res, dn, sub_dl, rng)
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, h_r.copy(), dn

    # Final polish at best resolution
    d3 = t0 + budget_s * 0.98
    h_p, c5_p = _slsqp_active(best_h, best_n, 12, 300, 1e-15, 3)
    if c5_p < best_c5:
        best_c5, best_h = c5_p, h_p.copy()

    # LSE final touch
    if time.time() < d3:
        h_l, c5_l = _lse_optimize(best_h, best_n, 200.0, 100, 1e-16, 1000.0)
        if c5_l < best_c5:
            best_c5, best_h = c5_l, h_l.copy()

    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
