def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def soft_max_optimize(h0, tau, maxiter):
    """SLSQP with LogSumExp smoothing."""
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        scores = corr * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        Z = np.sum(exp_s)
        return max_s + tau * np.log(Z)
    
    def grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        scores = corr * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        Z = np.sum(exp_s)
        w = exp_s / Z
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            d = k - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[d:] += wk * (1.0 - x[:m])
                    g[:m] += -wk * x[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    g[:m] += wk * (1.0 - x[gg:])
                    g[gg:] += -wk * x[:m]
        return dx * g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

But the O(n) cost of computing scores is O(n²) (from correlate), and trying pairs is O(k² * n²) where k is the number of pairs tried. def objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def build_active_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = scores_raw * dx
    top_k = np.argsort(scores)[::-1][:K]
    G = np.zeros((K, n))
    for idx, lag in enumerate(top_k):
        d = lag - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G, scores, top_k

def active_lp(h, K, radius):
    n = len(h)
    G, scores, top_k = build_active_grad(h, K)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[top_k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search_exact(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    num_coarse = 12
    for i in range(num_coarse):
        a = alpha_max * (i + 1) / num_coarse
        c5_t = objective(h + a * p)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.6)
    hi_a = min(alpha_max, best_a * 1.4)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = objective(h + c1 * p)
    f2 = objective(h + c2 * p)
    for _ in range(18):
        if hi_a - lo_a < 1e-13:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = objective(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = objective(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = objective(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def soft_max_optimize(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        scores = corr * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        Z = np.sum(exp_s)
        return max_s + tau * np.log(Z)
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        scores = corr * dx
        max_s = np.max(scores)
        exp_s = np.exp((scores - max_s) / tau)
        Z = np.sum(exp_s)
        w = exp_s / Z
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            d = k - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    g[d:] += wk * (1.0 - x[:m])
                    g[:m] += -wk * x[d:]
            else:
                gg = -d
                m = n - gg
                if m > 0:
                    g[:m] += wk * (1.0 - x[gg:])
                    g[gg:] += -wk * x[:m]
        return dx * g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def anneal_pairs(h, deadline, target, start_temp=0.004, end_temp=1e-6):
    n = len(h)
    h = h.copy()
    cur_c5 = objective(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    t_start = time.time()
    t_dur = max(deadline - t_start, 1.0)
    while time.time() < deadline:
        progress = min((time.time() - t_start) / t_dur, 1.0)
        temp = start_temp * (end_temp / start_temp) ** progress
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        sign = 1 if h[j] > h[i] else -1
        if sign > 0:
            t_max = h[j]
        else:
            t_max = h[i]
        if t_max < 1e-10:
            continue
        t = t_max * rng.random() * 0.7
        if t < 1e-12:
            continue
        h_t = h.copy()
        h_t[i] += sign * t
        h_t[j] -= sign * t
        c5_t = objective(h_t)
        delta = c5_t - cur_c5
        if delta < -1e-13 or (temp > 1e-9 and rng.random() < np.exp(-delta / temp)):
            h = h_t
            cur_c5 = c5_t
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def make_binary_seed(n, rng):
    h = np.zeros(n)
    positions = np.arange(n)
    rng.shuffle(positions)
    k = n // 2
    h[positions[:k]] = 1.0
    return project(h, n / 2.0)

def make_block_seed(n, rng):
    h = np.zeros(n)
    num_blocks = int(rng.integers(3, 8))
    block_sizes = np.zeros(num_blocks, dtype=int)
    remaining = n // 2
    for i in range(num_blocks - 1):
        s = int(rng.integers(1, max(2, remaining // (num_blocks - i))))
        block_sizes[i] = s
        remaining -= s
    block_sizes[-1] = remaining
    positions = np.zeros(n, dtype=int)
    rng.shuffle(positions)
    pos = np.arange(n)
    for i in range(num_blocks):
        start = int(positions[i * n // num_blocks])
        end = min(start + max(1, block_sizes[i]), n)
        h[start:end] = 1.0
    return project(h, n / 2.0)
