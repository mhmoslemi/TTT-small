def _binary_exchange(rng, n, n_iters=2000):
    """Generate a binary vector via random init + exchange moves."""
    target = n // 2
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:target]] = 1.0
    c5 = _exact_c5(h)
    for _ in range(n_iters):
        # Try random exchange: pick a 1 and a 0, swap them
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = h[j], h[i]
        c5_new = _exact_c5(h)
        if c5_new < c5:
            c5 = c5_new
        else:
            h[i], h[j] = h[j], h[i]  # undo
    return h, c5

    bounds = [(0.0, 1.0)] * n
    target = n * 0.5
    try:
        res = minimize(_lse_obj, h, jac=True, args=(n, alpha),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'maxls': 50, 'iprint': 0})
        x = _project(res.x, target)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _binary_exchange(rng, n, n_iters=5000):
    target = n // 2
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    c5 = _exact_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for _ in range(n_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        a, b = int(ones[i]), int(zeros[j])
        h[a], h[b] = h[b], h[a]
        c5_new = _exact_c5(h)
        if c5_new < c5:
            c5 = c5_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[a], h[b] = h[b], h[a]
    return h, c5


def _sinusoidal_init(n, m, phase, rng):
    x = (np.arange(n) + 0.5) / n * 2.0
    h = 0.5 + 0.4 * np.sin(np.pi * m * x + phase) + 0.1 * rng.normal(0, 1, n)
    return _project(h, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 * 0.5
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LSE warm-up (12% budget)
    d1 = t0 + budget_s * 0.12
    alphas = [30.0, 80.0, 150.0, 300.0]
    for alpha in alphas:
        if time.time() > d1:
            break
        h, c5 = _lse_step(h, n0, alpha, maxiter=200)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Active-set optimization (30% budget)
    d2 = t0 + budget_s * 0.42
    h_a, c5_a = _slsqp_active(best_h, n0, d2, K_start=4, K_step=3,
                               maxiter=200, ftol=1e-14, max_rounds=30)
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a.copy()

    # Phase 3: Full SLSQP polish (15% budget)
    d3 = t0 + budget_s * 0.57
    h_f, c5_f = _slsqp_full(best_h, n0, maxiter=500, ftol=1e-15)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()

    # Phase 4: Diversified restarts (15% budget)
    d4 = t0 + budget_s * 0.72
    attempts = []
    scales = [0.015, 0.03, 0.05, 0.02, 0.04]
    for sc in scales:
        h_p = _project(best_h + rng.normal(0, sc, n0), target0)
        attempts.append(h_p)
    # Sinusoidal starts
    for m in [3, 5, 7]:
        h_s = _sinusoidal_init(n0, m, float(rng.uniform(0, 2 * np.pi)), rng)
        attempts.append(h_s)
    # Antisymmetric transform
    h_anti = _project(1.0 - best_h[::-1].copy(), target0)
    attempts.append(h_anti)

    for h_try in attempts:
        if time.time() > d4:
            break
        c5_try = _exact_c5(h_try)
        if c5_try > best_c5 * 1.05:
            continue
        sub_d = min(d4, time.time() + budget_s * 0.03)
        h_opt, c5_opt = _slsqp_active(h_try, n0, sub_d, K_start=4,
                                       K_step=3, maxiter=150, ftol=1e-13,
                                       max_rounds=15)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 5: Resolution sweep (12% budget)
    d5 = t0 + budget_s * 0.84
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, 12, -12]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_d = min(d5, time.time() + budget_s * 0.02)
        h_r, c5_r = _lse_step(h_r, n_try, 100.0, maxiter=80)
        sub_d2 = min(d5, time.time() + budget_s * 0.025)
        h_r, c5_r = _slsqp_active(h_r, n_try, sub_d2, K_start=4,
                                   K_step=3, maxiter=120, ftol=1e-13,
                                   max_rounds=12)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Final intensive polish (12% budget)
    d6 = t0 + budget_s * 0.96
    # Multiple active-set rounds with tight tolerance
    h_f, c5_f = _slsqp_active(best_h, best_n, d6, K_start=3, K_step=2,
                               maxiter=300, ftol=1e-15, max_rounds=25)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()
    # Full SLSQP
    if time.time() < d6 - 10:
        h_f, c5_f = _slsqp_full(best_h, best_n, maxiter=400, ftol=1e-15)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, best_n
