def exact_c5(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_exact(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    v = np.asarray(v, dtype=float)
    v[~np.isfinite(v)] = 0.5
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))

        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    x[np.abs(x) < 1e-16] = 0.0
    x[np.abs(x - 1.0) < 1e-16] = 1.0
    return x

def build_active_grads(h, active_idx):
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    G = np.zeros((m, n))
    for k in range(m):
        r = active_idx[k]
        d = r - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[k, d:d+length] += dx * (1.0 - h[:length])
                G[k, :length] -= dx * h[d:d+length]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[k, :length] += dx * (1.0 - h[g:])
                G[k, g:g+length] -= dx * h[:length]
    return G

def kkt_direction(h, scores, active_idx, margin_factor, margin_floor, ridge, rough, priority_power):
    n = len(h)
    m = len(active_idx)
    if m == 0:
        return None
    s_active = scores[active_idx]
    c_max = float(np.max(s_active))
    c_min = float(np.min(s_active))
    spread = c_max - c_min
    margin = max(margin_floor, margin_factor * spread)
    target = c_min - margin
    r = target - s_active

    gaps = c_max - s_active
    max_gap = float(np.max(gaps))
    if max_gap < 1e-15:
        q = np.ones(m)
    else:
        q = 1.0 + priority_power * (gaps / max_gap) ** 2

    G = build_active_grads(h, active_idx)
    if not np.all(np.isfinite(G)):
        return None

    H = G.T @ (q[:, None] * G)
    H += ridge * np.eye(n)
    if rough > 1e-15:
        for i in range(n):
            H[i, i] += 2.0 * rough
            if i > 0:
                H[i, i-1] -= rough
                H[i-1, i] -= rough
            if i < n - 1:
                H[i, i+1] -= rough
                H[i+1, i] -= rough

    rhs = G.T @ (q * r)

    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs_full = np.zeros(n + 1)
    rhs_full[:n] = rhs

    try:
        sol = np.linalg.solve(KKT, rhs_full)
        delta = sol[:n]
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(KKT, rhs_full, rcond=1e-14)[0]
            delta = sol[:n]
        except Exception:
            return None

    if not np.all(np.isfinite(delta)):
        return None
    delta -= delta.mean()
    pm = float(np.max(np.abs(delta)))
    if pm < 1e-13:
        return None
    return delta / pm

def fallback_weighted_grad(h, scores, active_idx, priority_power):

def lp_direction(h, scores, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = build_active_grads(h, idx)
    else:
        scores_k = scores
        G_k = build_active_grads(h, np.arange(n_lags))

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def line_search(h, p, trust=0.5):
    n = len(h)
    p_mu = float(np.max(np.abs(p)))
    if p_mu < 1e-15:
        return 0.0, exact_c5(h)
    p_unit = p / p_mu
    alpha_max = trust
    for j in range(n):
        if p_unit[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p_unit[j])
        elif p_unit[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p_unit[j]))
    alpha_max = max(alpha_max, 1e-14)
    cur_c5 = exact_c5(h)
    n_samp = 12
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([exact_c5(h + a * p_unit) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_samp - 1):
        return best_a * p_mu, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * p_unit)
    f2 = exact_c5(h + c2 * p_unit)
    for _ in range(16):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * p_unit)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * p_unit)
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * p_unit)
    if opt_c5 < best_c5:
        best_c5 = opt_c5
        best_a = opt_a
    return best_a * p_mu, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)

def quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return project_feasible(h, n / 2.0)

def smooth_perturb(h, rng, scale, n_harmonics=5):
    n = len(h)
    x = np.arange(n) / n
    noise = np.zeros(n)
    for k in range(1, n_harmonics + 1):
        a = rng.normal(0, scale / np.sqrt(k))
        b = rng.normal(0, scale / np.sqrt(k))
        noise += a * np.cos(2 * np.pi * k * x) + b * np.sin(2 * np.pi * k * x)
    return project_feasible(h + noise, n / 2.0)

                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = n_try

    # Phase 5: Perturbation restarts around best
    dl5 = t0 + budget_s * 0.82
    scale = 0.025
    for attempt in range(200):
        if time.time() > dl5:
            break
        if attempt % 4 == 0:
            h_pert = smooth_perturb(best_h, rng, scale, n_harmonics=3)
        else:
            h_pert = project_feasible(best_h + rng.normal(0, scale * 0.5, best_n), best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        vi = attempt % 5
        h_try, c5_try = optimize_active(h_pert, sub_dl, variant_idx=vi, max_iters=400)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.1)

    # Phase 6: Interleaved LP + active on best with small trust
    dl6 = t0 + budget_s * 0.92
    for vi in range(5):
        if time.time() > dl6:
            break
        sub_dl = min(dl6, time.time() + budget_s * 0.015)
        h_try, c5_try = optimize_active(best_h, sub_dl, variant_idx=vi, max_iters=600)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        if time.time() > dl6:
            break
        sub_dl2 = min(dl6, time.time() + budget_s * 0.008)
        h_lp, c5_lp = optimize_lp(best_h, sub_dl2, max_iters=400,
                                   top_k_list=[5, 3, 4], radius_init=0.05)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 7: Final tight polish
    if time.time() < deadline:
        for vi in [2, 3, 0]:
            if time.time() > deadline:
                break
            sub_dl = min(deadline, time.time() + budget_s * 0.008)
            h_try, c5_try = optimize_active(best_h, sub_dl, variant_idx=vi, max_iters=300)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
