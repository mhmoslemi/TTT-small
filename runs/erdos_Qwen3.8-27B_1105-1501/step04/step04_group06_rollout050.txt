    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_at_lag(h, s):
    """Gradient of c_s at h, where s is the lag (can be negative)."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s  # number of valid pairs
        if m <= 0:
            return g
        g[:m] += (1.0 - h[s:])
        g[s:] -= h[:m]
    else:
        s2 = -s
        m = n - s2
        if m <= 0:
            return g
        g[s2:] += (1.0 - h[:m])
        g[:m] -= h[s2:]
    return g

def _lp_active_set(h, n, K, deadline, max_iters=5000):
    """LP-based active set method for minimizing max correlation."""
    h = h.copy()
    target = n * 0.5
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corrs = _corr_vec(h)
        M = float(np.max(corrs))
        
        # Active set: top K lags
        top_idx = np.argsort(-corrs)[:K]
        
        # Compute gradients for active lags
        grads = np.zeros((K, n))
        for i, r in enumerate(top_idx):
            s = r - (n - 1)
            grads[i] = _grad_at_lag(h, s)
        
        # LP: min t s.t. grads[i]·(x-h) + corrs[top_idx[i]] <= t, 0<=x<=1, sum(x)=n/2
        # Equivalent: min t s.t. grads[i]·x <= t + (corrs[top_idx[i]] - grads[i]·h)
        # Variables: [x (n), t (1)]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        for i in range(K):
            A_ub[i, :n] = grads[i]
            A_ub[i, n] = -1.0
            b_ub[i] = corrs[top_idx[i]] - grads[i] @ h
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success or res.x is None:
            K = max(3, K // 2)  # Reduce active set
            continue
        
        x_lp = res.x[:n]
        d = x_lp - h
        
        if np.max(np.abs(d)) < 1e-12:
            # Can't improve with this active set
            K = min(2*n-1, K + 2)
            if K > 2*n - 1:
                break
            continue
        
        # Line search
        best_a = 0.0
        best_c5_ls = cur_c5
        # Coarse
        for t in np.linspace(0.0, 1.0, 11):
            a = t
            h_try = _project(h + a * d, target)
            c5_try = _c5(h_try)
            if c5_try < best_c5_ls:
                best_c5_ls = c5_try
                best_a = a
        
        # Golden section refinement
        if best_a > 0:
            lo = max(0.0, best_a - 0.1)
            hi = min(1.0, best_a + 0.1)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = _c5(_project(h + c1 * d, target))
            f2 = _c5(_project(h + c2 * d, target))
            for _ in range(15):
                if hi - lo < 1e-8:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _c5(_project(h + c1 * d, target))
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _c5(_project(h + c2 * d, target))
            a = (lo + hi) / 2.0
            h_try = _project(h + a * d, target)
            c5_try = _c5(h_try)
            if c5_try < best_c5_ls:
                best_c5_ls = c5_try
                best_a = a

            cur_c5 = _c5(h)

                best_h = h.copy()
        else:
            # Stuck - try more active constraints or break
            K = min(2*n - 1, K + 3)
            if K >= 2*n - 1:
                break
    
    return best_h, best_c5
