import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_1d(h, V, active_threshold=0.02):
    """Exact 1D minimization of max score along direction V (must have sum≈0)."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Feasible range
    amin, amax = 0.0, 0.0
    for i in range(n):
        vi = V[i]
        if vi > 1e-14:
            amin = max(amin, -h[i] / vi)
            amax = min(amax, (1.0 - h[i]) / vi)
        elif vi < -1e-14:
            amax = min(amax, -h[i] / vi)
            amin = max(amin, (1.0 - h[i]) / vi)
    
    if amin > 0:
        amin = 0.0
    if amax < 0:
        amax = 0.0
    if amin >= amax:
        # Can't move at all
        # Shrink to stay feasible
        if amax < 0:
            amax = 0.0
        if amin > 0:
            amin = 0.0
        if amax <= amin:
            return 0.0, _exact_c5(h)
    
    # Quadratic coefficients
    c_vec = np.correlate(h, 1.0 - h, mode='full') * dx
    b_vec = (np.correlate(V, 1.0 - h, mode='full') - np.correlate(h, V, mode='full')) * dx
    a_vec = -np.correlate(V, V, mode='full') * dx
    
    # Current scores
    s_max = float(np.max(c_vec))
    
    # Identify active lags for intersection computation
    thresh = s_max * (1.0 - active_threshold)
    active_mask = c_vec >= thresh
    active_idx = np.where(active_mask)[0]
    
    # If too few active, add some more
    if len(active_idx) < 5:
        order = np.argsort(c_vec)[::-1]
        active_idx = order[:15]
    
    # Limit to top 25 for speed
    if len(active_idx) > 25:
        active_idx = active_idx[np.argsort(c_vec[active_idx])[::-1][:25]]
    
    # Candidates
    candidates = [amin, amax, 0.0]
    
    # Vertices (all lags - cheap since it's just O(n_lags))
    for r in range(n_lags):
        if abs(a_vec[r]) > 1e-14:
            v = -b_vec[r] / (2.0 * a_vec[r])
            if v >= amin - 1e-10 and v <= amax + 1e-10:
                v = max(amin, min(amax, v))
                candidates.append(v)
    
    # Intersections among active lags
    m = len(active_idx)
    for i in range(m):
        ri = active_idx[i]
        for j in range(i + 1, m):
            rj = active_idx[j]
            da = a_vec[ri] - a_vec[rj]
            db = b_vec[ri] - b_vec[rj]
            dc = c_vec[ri] - c_vec[rj]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    root = -dc / db
                    if root >= amin and root <= amax:
                        candidates.append(root)
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= 0:
                    sq = np.sqrt(disc)
                    r1 = (-db + sq) / (2.0 * da)
                    r2 = -r1 - db / da  # actually r2 = (-db - sq) / (2*da)
                    r2 = (-db - sq) / (2.0 * da)
                    if r1 >= amin and r1 <= amax:
                        candidates.append(r1)
                    if r2 >= amin and r2 <= amax:
                        candidates.append(r2)
    
    # Deduplicate and evaluate
    candidates = np.array(candidates)
    candidates = np.clip(candidates, amin, amax)
    # Round to avoid duplicates
    candidates = np.round(candidates, 12)
    unique_candidates = np.unique(candidates)
    
    best_c5 = float('inf')
    best_alpha = 0.0
    for alpha in unique_candidates:
        f = float(np.max(c_vec + b_vec * alpha + a_vec * alpha * alpha))
        if f < best_c5:
            best_c5 = f
            best_alpha = alpha
    
    return best_alpha, best_c5

    n = len(h)

    s_max = float(scores.max())

    sw = w.sum()
    if sw < 1e-300:

        w[int(np.argmax(scores))] = 1.0

    # = δ*correlate(e_i, 1-h) - δ*correlate(h, e_i)

    # correlate(e_i, 1-h)[r] = (1-h)[i - (r - (n-1))] if valid, else 0
    # correlate(h, e_i)[r] = h[i + (r - (n-1))] if valid, else 0

    # correlate(e_i, 1-h) has (1-h) shifted... 
    # Actually correlate(a, b)[k] = sum_n a[n+k] * b[n] (for valid n)
    # So correlate(e_i, 1-h)[k] = sum_n e_i[n+k] * (1-h)[n] = (1-h)[n] where n+k=i, so n=i-k
    # = (1-h)[i-k] if 0 <= i-k < n, else 0
    #
    # correlate(h, e_i)[k] = sum_n h[n+k] * e_i[n] = h[n+k] where n=i, so h[i+k]
    # = h[i+k] if 0 <= i+k < n, else 0

    for r in range(n_lags):
        if w[r] < 1e-15:
            continue

        # d(score_r)/d(h[i]) = dx * ((1-h)[i-d] if 0<=i-d<n else 0) - dx * (h[i+d] if 0<=i+d<n else 0)
        # For all i:
        # term1[i] = (1-h)[i-d] if 0 <= i-d < n else 0
        # term2[i] = h[i+d] if 0 <= i+d < n else 0

            idx1 = np.arange(d, n)
            term1 = np.zeros(n)
            term1[idx1] = 1.0 - h[:n-d]
            term2 = np.zeros(n)
            if n - d > 0:
                term2[:n-d] = h[d:]
        else:
            g = -d
            # term1[i] = (1-h)[i+g] for i <= n-1-g
            # term2[i] = h[i-g] for i >= g
            term1 = np.zeros(n)
            if n - g > 0:
                term1[:n-g] = 1.0 - h[g:]
            term2 = np.zeros(n)
            term2[g:] = h[:n-g]
        
        grad += w[r] * dx * (term1 - term2)
    
    grad -= np.mean(grad)  # ensure sum = 0
    return -grad  # descent direction

def _compute_grad(h, w):
    """Compute grad[i] = sum_k w[k] * d(score_k)/d(h[i]) * dx.
    w is the weight vector over lags (already normalized, sum=1)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        wv = w[k]
        if abs(wv) < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            # term1: (1-h[i-d]) for i in [d, n-1]
            grad[d:] += wv * dx * (1.0 - h[:n - d])
            # term2: -h[i+d] for i in [0, n-1-d]
            if n - d > 0:
                grad[:n - d] -= wv * dx * h[d:]
        else:
            g = -d  # g > 0
            # term1: (1-h[i+g]) for i in [0, n-1-g]
            if n - g > 0:
                grad[:n - g] += wv * dx * (1.0 - h[g:])
            # term2: -h[i-g] for i in [g, n-1]
            grad[g:] -= wv * dx * h[:n - g]
    
    return grad

def _optimize_direction(h, deadline, max_iters=5000):
    """Main optimization loop using p-norm direction + exact 1D search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    
    p = 10.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute direction
        grad = _pnorm_grad(h, p)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        V = -grad / gn
        
        # Exact 1D search
        alpha, new_c5 = _exact_1d(h, V)
        
        if new_c5 < cur_c5 - 1e-14:

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start_descent(h, deadline, seed, n_starts=100):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    scale = 0.05
    for trial in range(n_starts):
        if time.time() > deadline:
            break
        if best_c5 < 0.3806:
            break
        h_pert = h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + budget_s_global * 0.02)
        h_try, c5_try = _main_descent(h_pert, sub_dl, p_start=20, p_end=20000, max_iters=2000)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.1)
    return best_h, best_c5

budget_s_global = 1100.0

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s_global
    budget_s_global = budget_s
    t0 = time.time()

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: Main p-norm descent (35%)
    dl1 = t0 + budget_s * 0.35
    hdr, c5r = _main_descent(h, dl1, p_start=8.0, p_end=50000.0, max_iters=8000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 2: Pair descent (15%)
    dl2 = t0 + budget_s * 0.50
    hdr, c5r = _pair_descent(best_h, dl2, max_sweeps=100)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 3: Another p-norm pass with sharpened p (10%)
    dl3 = t0 + budget_s * 0.60
    hdr, c5r = _main_descent(best_h, dl3, p_start=100.0, p_end=100000.0, max_iters=5000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 4: Active set gradient descent (10%)
    dl4 = t0 + budget_s * 0.70
    hdr, c5r = _main_descent(best_h, dl4, p_start=5.0, p_end=500.0, max_iters=5000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 5: Random direction exploration (10%)
    dl5 = t0 + budget_s * 0.80
    hdr, c5r = _random_descent(best_h, dl5, seed=seed + 1, n_trials=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 6: Multi-start perturbation (7%)
    dl6 = t0 + budget_s * 0.87
    hdr, c5r = _multi_start_descent(best_h, dl6, seed=seed + 2, n_starts=80)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
        best_n = len(hdr)

    # Phase 7: Resolution sweep (5%)
    dl7 = t0 + budget_s * 0.92
    for dn in [-2, -1, 1, 2, 4, 6, 8, 12]:
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try % 2 != 0 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + budget_s * 0.025)
        h_r, c5_r = _main_descent(h_r, sub_dl, p_start=10, p_end=10000, max_iters=3000)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final pair polish (4%)
    dl8 = t0 + budget_s * 0.96
    hdr, c5r = _pair_descent(best_h, dl8, max_sweeps=50)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 9: Ultra-fine final descent (2%)
    dl9 = t0 + budget_s * 0.99
    hdr, c5r = _main_descent(best_h, dl9, p_start=500.0, p_end=200000.0, max_iters=3000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
