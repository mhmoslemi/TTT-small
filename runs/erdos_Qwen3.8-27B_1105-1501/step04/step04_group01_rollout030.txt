def _lp_dir_full(h, G, radius):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _ls(h, d, dx, tr):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -tr)
    t_hi = min(t_hi, tr)
    if t_hi - t_lo < 1e-14:
        return 0.0, _c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-11:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    t = (a + b) / 2.0
    return t, float(np.max(S0 + t * T - t * t * C)) * dx

def _lp_loop(h, deadline, r0=0.15, ntop=18, use_full=False, full_every=8):
    n = len(h)
    tgt = n / 2.0
    h = h.copy()
    best = h.copy()
    bc = _c5(h)
    r = r0
    stall = 0
    it = 0
    while time.time() < deadline:
        if stall > 100:
            break
        G = _grad_matrix(h)
        it += 1
        if use_full and it % full_every == 0:
            p = _lp_dir_full(h, G, r)
        else:
            p = _lp_dir(h, G, r, ntop)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.65
            if r < 1e-7:
                break
            continue
        alpha, nc = _ls(h, p, 2.0 / n, r)
        if nc < _c5(h) - 1e-14:
            h = _project(h + alpha * p, tgt)
            c = _c5(h)
            if c < bc:
                bc = c
                best = h.copy()
            stall = 0
            r = min(r * 1.04, 0.40)
        else:
            stall += 1
            r *= 0.72
            if stall % 20 == 0 and r < 0.03:
                r = 0.08
    return best, bc

def _interp_rs(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _blowup(h, factor=2):
    return _project(np.repeat(h, factor), len(h) * factor // 2)

def _staggered_blowup(h):
    """Interleaved blow-up for better structure at 2x resolution"""
    n = len(h)
    h2 = np.zeros(2 * n)
    h2[0::2] = h
    h2[1::2] = 0.5 * (h + np.roll(h, -1))
    return _project(h2, n)

def _pair_ref(h, deadline, max_p=60):
    n = len(h)
    dx = 2.0 / n
    tgt = n / 2.0
    h = h.copy()
    best = h.copy()
    bc = _c5(h)
    for _ in range(max_p):
        if time.time() > deadline:
            break
        imp = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                a, c = _ls(h, d, dx, 0.5)
                if c < bc - 1e-14:
                    h = _project(h + a * d, tgt)
                    nc = _c5(h)
                    if nc < bc:
                        bc = nc
                        best = h.copy()
                        imp = True
        if not imp:
            break
    return best, bc

def _multi_res(h, deadline):
    """Try multiple resolutions via blow-up and interpolation"""
    best_h = h.copy()
    best_c5 = _c5(h)
    n_cur = len(h)
    tried = set()

    # Try blow-up to 2x
    if time.time() < deadline:
        h2 = _blowup(h, 2)
        sub_dl = min(deadline, time.time() + max(5, (deadline - time.time()) * 0.45))
        h2o, c5_2o = _lp_loop(h2, sub_dl, r0=0.12, ntop=20)
        if c5_2o < best_c5 - 1e-12:
            best_h = h2o.copy()
            best_c5 = c5_2o
            n_cur = 2 * n_cur
            tried.add(2 * n_cur)

    # Try staggered blow-up
    if time.time() < deadline:
        h2s = _staggered_blowup(h)
        n2 = len(h2s)
        if n2 not in tried:
            tried.add(n2)
            sub_dl = min(deadline, time.time() + max(4, (deadline - time.time()) * 0.35))
            h2so, c5_2so = _lp_loop(h2s, sub_dl, r0=0.10, ntop=20)
            if c5_2so < best_c5 - 1e-12:
                best_h = h2so.copy()
                best_c5 = c5_2so

    # Try interpolation to various n
    nc = len(best_h)
    offsets = [-6, 6, -12, 12, -18, 18, -24, 24, -36, 36, -50, 50]
    for off in offsets:
        if time.time() > deadline:
            break
        nt = nc + off
        if nt < 30 or nt > 600 or nt in tried:
            continue
        tried.add(nt)
        hr = _interp_rs(best_h, nt)
        sub_dl = min(deadline, time.time() + max(3, budget_global * 0.013))
        hr, cr = _lp_loop(hr, sub_dl, r0=0.10, ntop=16)
        if cr < best_c5 - 1e-12:
            best_h = hr.copy()
            best_c5 = cr
            nc = nt
    return best_h, best_c5

def _perturb_round(h, deadline, rng, n_att=60):
    best_h = h.copy()
    best_c5 = _c5(h)
    sc = 0.05
    # Also try structured perturbations
    n = len(h)
    for att in range(n_att):
        if time.time() > deadline:
            break
        mode = att % 4
        if mode == 0:
            hp = best_h + rng.normal(0, sc, n)
        elif mode == 1:
            k = rng.integers(1, 5)
            hp = best_h + sc * np.sin(2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 6.28))
        elif mode == 2:
            # Block perturbation
            bl = rng.integers(4, max(5, n // 4))
            st = rng.integers(0, n - bl)
            hp = best_h.copy()
            hp[st:st+bl] += rng.normal(0, sc, bl)
        else:
            # Swap-based
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            hp = best_h.copy()
            hp[i], hp[j] = hp[j], hp[i]
        hp = _project(hp, n / 2.0)
        sub_dl = min(deadline, time.time() + max(2, budget_global * 0.011))
        ht, ct = _lp_loop(hp, sub_dl, r0=0.08, ntop=15)
        if ct < best_c5 - 1e-12:
            best_h = ht.copy()
            best_c5 = ct
            sc = max(0.004, sc * 0.8)
        else:
            sc = min(0.15, sc * 1.13)
    return best_h, best_c5

budget_global = 1100

def run(seed=42, budget_s=1100, **kwargs):
    global budget_global
    budget_global = budget_s
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h = np.array(initial_h_values, dtype=float)
    h = _project(h, len(h) / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)

    # Phase 1: Initial LP optimization at n=112 (10%)
    d1 = t0 + budget_s * 0.10
    h1, c1 = _lp_loop(h, d1, r0=0.15, ntop=18, use_full=True, full_every=6)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Multi-resolution exploration (25%)
    d2 = t0 + budget_s * 0.35
    h2, c2 = _multi_res(best_h, d2)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2

    # Phase 3: Deep LP at best resolution (15%)
    d3 = t0 + budget_s * 0.50
    h3, c3 = _lp_loop(best_h, d3, r0=0.12, ntop=20, use_full=True, full_every=5)
    if c3 < best_c5:
        best_h, best_c5 = h3, c3

    # Phase 4: Another multi-resolution pass (15%)
    d4 = t0 + budget_s * 0.65
    h4, c4 = _multi_res(best_h, d4)
    if c4 < best_c5:
        best_h, best_c5 = h4, c4

    # Phase 5: Perturbation diversification (18%)
    d5 = t0 + budget_s * 0.83
    h5, c5 = _perturb_round(best_h, d5, rng, n_att=80)
    if c5 < best_c5:
        best_h, best_c5 = h5, c5

    # Phase 6: Pair refinement (7%)
    d6 = t0 + budget_s * 0.90
    h6, c6 = _pair_ref(best_h, d6, max_p=50)
    if c6 < best_c5:
        best_h, best_c5 = h6, c6

    # Phase 7: Final LP polish (7%)
    d7 = t0 + budget_s * 0.97
    h7, c7 = _lp_loop(best_h, d7, r0=0.05, ntop=12, use_full=False)
    if c7 < best_c5:
        best_h, best_c5 = h7, c7

    # Phase 8: Very final pair micro-opt (3%)
    d8 = t0 + budget_s * 1.0
    h8, c8 = _pair_ref(best_h, d8, max_p=30)
    if c8 < best_c5:
        best_h, best_c5 = h8, c8

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, len(best_h)
