import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _active_grad(h, scores, eps=0.001):
    """Compute weighted gradient of active scores."""
    n = len(h)
    s_max = float(np.max(scores))
    active = np.where(scores >= s_max - eps)[0]
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d == 0:
            # sum h[j]*(1-h[j])
            grad -= 2.0 * h + 1.0
        elif d > 0:
            m = n - d
            grad[d:] += 1.0 - h[:m]
            grad[:m] -= h[d:]
        else:
            g = -d
            m = n - g
            grad[:m] += 1.0 - h[g:]
            grad[g:] -= h[:m]
    grad -= np.mean(grad)  # enforce sum=0
    return grad

def _line_search(h, p, c5_cur, alpha_max, n_grid=20):
    """Line search along direction p from h, max step alpha_max."""
    n = len(h)
    # Grid search
    ts = np.linspace(0.0, alpha_max, n_grid)
    best_t = 0.0
    best_val = c5_cur
    for t in ts:
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            v = _exact_c5(h_t)
            if v < best_val:
                best_val = v
                best_t = t
    # Refine with golden section
    if best_t > 0 and best_t < alpha_max:
        lo = ts[max(0, int(best_t / alpha_max * (n_grid-1)) - 2)]
        hi = ts[min(n_grid-1, int(best_t / alpha_max * (n_grid-1)) + 2)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p) if (h + c1*p).min() >= -1e-12 else c5_cur + 1
        f2 = _exact_c5(h + c2 * p) if (h + c2*p).min() >= -1e-12 else c5_cur + 1
        for _ in range(30):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, f2, c2 = c2, f1, c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo, f1, c1 = c1, f2, c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        t_opt = min(max((lo+hi)/2.0, 0.0), alpha_max)
        h_opt = h + t_opt * p
        if h_opt.min() >= -1e-12 and h_opt.max() <= 1.0 + 1e-12:
            f_opt = _exact_c5(h_opt)
            if f_opt < best_val:
                best_val = f_opt
                best_t = t_opt
    return best_t, best_val

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def _active_set_search(h, deadline, eps_init=0.001, max_iter=5000):
    """Active-set line search for min-max."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    eps = eps_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline or stall > 100:
            break
        scores = _scores(h)
        grad = _active_grad(h, scores, eps)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-10:
            eps *= 0.5
            if eps < 1e-8:
                break
            continue
        
        p = -grad / gnorm  # normalize
        # Compute alpha_max
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = min(alpha_max, 1.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _anneal_exchange(h, deadline, m=8, n_steps=5000, T_init=0.001):
    """Stochastic annealing with exact exchange moves."""
    n = len(h)
    target = n / 2.0
    # Quantize to m levels
    q = np.round(h * m).astype(int)
    # Fix sum constraint
    q_sum = int(np.sum(q))
    target_q = int(target * m)
    diff = target_q - q_sum
    # Adjust to fix sum
    if diff > 0:
        # Need to increase sum by diff
        order = np.argsort(-h * m + np.random.default_rng(0).uniform(0, 0.3, n))
        for i in range(min(diff, n)):
            if q[order[i]] < m:
                q[order[i]] += 1
    elif diff < 0:
        order = np.argsort(h * m + np.random.default_rng(0).uniform(0, 0.3, n))
        for i in range(min(-diff, n)):
            if q[order[i]] > 0:
                q[order[i]] -= 1
    
    h_cur = q.astype(float) / m
    c5_cur = _exact_c5(h_cur)
    best_q = q.copy()
    best_c5 = c5_cur
    
    rng = np.random.default_rng(42)
    T = T_init
    rng_state = 0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Propose exchange
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Transfer one unit from j to i (or vice versa)
        if q[i] >= m or q[j] <= 0:
            continue
        q[i] += 1
        q[j] -= 1
        h_new = q.astype(float) / m
        c5_new = _exact_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_q = q.copy()
        else:
            q[i] -= 1
            q[j] += 1
        T *= 0.9995
    
    return best_q.astype(float) / m, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _active_grad(h, scores, eps=0.001):
    n = len(h)
    s_max = float(np.max(scores))
    active = np.where(scores >= s_max - eps)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(scores))])
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d == 0:
            grad += 1.0 - 2.0 * h
        elif d > 0:
            m = n - d
            grad[d:] += 1.0 - h[:m]
            grad[:m] -= h[d:]
        else:
            g = -d
            m = n - g
            grad[:m] += 1.0 - h[g:]
            grad[g:] -= h[:m]
    grad -= np.mean(grad)
    return grad

def _line_search(h, p, c5_cur, alpha_max, n_grid=16):
    if alpha_max < 1e-14:
        return 0.0, c5_cur
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_grid)
    for t in ts:
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            v = _exact_c5(h_t)
            if v < best_val:
                best_val = v
                best_t = t
    if best_t > 0.0 and best_t < alpha_max:
        idx = int(best_t / alpha_max * (n_grid - 1))
        lo = ts[max(0, idx - 2)]
        hi = ts[min(n_grid - 1, idx + 2)]
        if hi <= lo:
            lo, hi = 0.0, alpha_max
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(25):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, f2, c2 = c2, f1, c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo, f1, c1 = c1, f2, c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        t_opt = min(max((lo + hi) / 2.0, 0.0), alpha_max)
        h_opt = h + t_opt * p
        if h_opt.min() >= -1e-12 and h_opt.max() <= 1.0 + 1e-12:
            f_opt = _exact_c5(h_opt)
            if f_opt < best_val:
                best_val = f_opt
                best_t = t_opt
    return best_t, best_val

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def _active_set_search(h, deadline, eps_init=0.002, max_iter=8000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    eps = eps_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        scores = _scores(h)
        grad = _active_grad(h, scores, eps)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-12:
            eps *= 0.5
            if eps < 1e-9:
                break
            continue
        p = -grad / gnorm
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                am = (1.0 - h[j]) / p[j]
                if am < alpha_max:
                    alpha_max = am
            elif p[j] < -1e-14:
                am = h[j] / (-p[j])
                if am < alpha_max:
                    alpha_max = am
        if alpha_max < 1e-12:
            stall += 1
            eps *= 0.8
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            eps = min(eps * 1.08, 0.015)
        else:
            stall += 1
            eps *= 0.65
            if eps < 1e-9:
                break
    return best_h, best_c5

def _resample(h, n_new, mode='mid'):
    n = len(h)
    if n_new == n:
        return h.copy()
    if mode == 'mid':
        x_old = (np.arange(n) + 0.5) / n
        x_new = (np.arange(n_new) + 0.5) / n_new
    else:
        x_old = np.arange(n) / n
        x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_redescent(h_best, c5_best, deadline, seed=42, max_attempts=30):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = c5_best
    scale = 0.06
    for attempt in range(max_attempts):
        if time.time() > deadline:
            break
        if best_c5 < 0.3806:
            break
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, target)
        sub_d = min(deadline, time.time() + (deadline - t0_global) * 0.012)
        h_r, c5_r = _active_set_search(h_p, sub_d, eps_init=0.001, max_iter=3000)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            scale = max(0.008, scale * 0.85)
        else:
            scale = min(0.3, scale * 1.3)
            if scale > 0.25:
                scale = 0.06
    return best_h, best_c5

def _block_anneal(h, deadline, m=32, n_block=6, T_init=0.0008):
    n = len(h)
    target = n / 2.0
    q = np.round(h * m).astype(int)
    q = np.clip(q, 0, m)
    q_sum = int(np.sum(q))
    target_q = int(round(target * m))
    diff = target_q - q_sum
    if diff != 0:
        idx = np.argsort(np.abs(q - m / 2.0))[::-1]
        if diff > 0:
            for i in idx:
                if q[i] < m and diff > 0:
                    q[i] += 1
                    diff -= 1
        else:
            for i in idx:
                if q[i] > 0 and diff < 0:
                    q[i] -= 1
                    diff += 1
    h_cur = q.astype(float) / m
    c5_cur = _exact_c5(h_cur)
    best_q = q.copy()
    best_c5 = c5_cur
    rng = np.random.default_rng(77)
    T = T_init
    for step in range(80000):
        if time.time() > deadline:
            break
        i0 = rng.integers(0, n - n_block + 1)
        block = slice(i0, i0 + n_block)
        block_sum = int(np.sum(q[block]))
        new_block = rng.integers(0, m + 1, n_block)
        new_sum = int(np.sum(new_block))
        if new_sum == block_sum:
            q_new = q.copy()
            q_new[block] = new_block
            h_new = q_new.astype(float) / m
            c5_new = _exact_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-12)):
                q = q_new
                h_cur = h_new
                c5_cur = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_q = q.copy()
        T *= 0.9997
        if T < 1e-10:
            T = 1e-10
    return best_q.astype(float) / m, best_c5

def _pair_refine(h, deadline, n_rounds=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(99)
    stall = 0
    for _round in range(n_rounds):
        if time.time() > deadline or stall > 40:
            break
        improved = False
        for _ in range(80):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            p = np.zeros(n)
            p[i] = 1.0
            p[j] = -1.0
            alpha_max = min(1.0 - h[i], h[j])
            if alpha_max < 1e-10:
                continue
            alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max, n_grid=12)
            if new_c5 < cur_c5 - 1e-12:
                h[i] += alpha
                h[j] -= alpha
                cur_c5 = _exact_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _topk_objective(h, k=5):
    n = len(h)
    s = _scores(h)
    return float(np.sum(np.sort(s)[-k:]))

def _topk_descent(h, deadline, k=5, max_iter=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        s = _scores(h)
        top_idx = np.argsort(s)[-k:]
        grad = np.zeros(n)
        for r in top_idx:
            d = r - (n - 1)
            if d == 0:
                grad += 1.0 - 2.0 * h
            elif d > 0:
                m = n - d
                grad[d:] += 1.0 - h[:m]
                grad[:m] -= h[d:]
            else:
                g = -d
                m = n - g
                grad[:m] += 1.0 - h[g:]
                grad[g:] -= h[:m]
        grad -= np.mean(grad)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-12:
            break
        p = -grad / gnorm
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                am = (1.0 - h[j]) / p[j]
                if am < alpha_max:
                    alpha_max = am
            elif p[j] < -1e-14:
                am = h[j] / (-p[j])
                if am < alpha_max:
                    alpha_max = am
        if alpha_max < 1e-12:
            break
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            if _topk_objective(h, k) > _topk_objective(best_h, k) + 1e-8:
                break
    return best_h, best_c5
