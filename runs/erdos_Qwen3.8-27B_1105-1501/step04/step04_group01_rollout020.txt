Now for the exact 1D minimization. Given h, direction d (with sum(d)=0), dx, and trust radius rho:

The objective along the line is:
g(t) = max_r (S0[r] + t*T[r] - t²*Q[r]) * dx

where:
- S0 = correlate(h, 1-h, 'full')
- T = correlate(d, 1-h, 'full') - correlate(h, d, 'full')
- Q = correlate(d, d, 'full')

Feasible interval: [t_lb, t_ub] where:
- t_ub = min((1-h[pos])/d[pos]) for d[pos] > 0
- t_lb = max(h[neg]/d[neg]) for d[neg] < 0

The exact minimum of g(t) on [t_lb, t_ub]:

Each component f_r(t) = S0[r] + t*T[r] - t²*Q[r] is a quadratic in t.
- If Q[r] > 0: f_r is concave, vertex at t_r = T[r]/(2*Q[r]) (maximum)
- If Q[r] < 0: f_r is convex, vertex at t_r = T[r]/(2*Q[r]) (minimum)
- If Q[r] = 0: f_r is linear

The upper envelope g(t) = max_r f_r(t). Its minimum on [t_lb, t_ub] occurs at:
- An endpoint, OR
- A point where two components intersect AND both are ≥ all others (on the envelope)

As I argued, checking all pairwise intersections + endpoints suffices.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_M(v, n):
    M = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = r - (n-1)
        if d >= 0:
            M[r, :n-d] = v[d:n]
        else:
            g = -d
            M[r, g:] = v[0:n-g]
    return M

def _alt_lp_step(h, n):
    """One step of alternating LP: fix v=1-h, optimize h."""
    v = 1.0 - h
    M = _build_M(v, n)
    target = n / 2.0
    
    # LP: min t s.t. M@h - t <= 0, sum(h) = target, 0 <= h <= 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def _exact_1d_line_search(h, d, dx, rho):
    """Exact 1D minimization using intersection points."""
    n = len(h)
    d = d - d.mean()
    if np.max(np.abs(d)) < 1e-15:
        return 0.0, _exact_c5(h)
    
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = float(np.min((1.0 - h[pos]) / d[pos])) if np.any(pos) else rho
    t_lo = float(np.max(h[neg] / d[neg])) if np.any(neg) else -rho
    t_lo = max(t_lo, -rho)
    t_hi = min(t_hi, rho)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    n_lags = len(S0)
    candidates = [t_lo, t_hi]
    
    i_idx, j_idx = np.triu_indices(n_lags, k=1)
    a = Q[j_idx] - Q[i_idx]
    b = T[i_idx] - T[j_idx]
    c_vec = S0[i_idx] - S0[j_idx]
    
    mask_q = np.abs(a) > 1e-15
    if np.any(mask_q):
        aq = a[mask_q]
        bq = b[mask_q]
        cq = c_vec[mask_q]
        disc = bq*bq - 4.0*aq*cq
        valid = disc >= 0
        if np.any(valid):
            sq = np.sqrt(disc[valid])
            t1 = (-bq[valid] + sq) / (2.0*aq[valid])
            t2 = (-bq[valid] - sq) / (2.0*aq[valid])
            candidates.extend(t1.tolist())
            candidates.extend(t2.tolist())
    mask_l = ~mask_q
    if np.any(mask_l):
        bl = b[mask_l]
        cl = c_vec[mask_l]
        valid = np.abs(bl) > 1e-15
        if np.any(valid):
            candidates.extend((-cl[valid] / bl[valid]).tolist())
    
    cand_arr = np.array(candidates)
    cand_arr = cand_arr[np.isfinite(cand_arr)]
    cand_arr = cand_arr[(cand_arr >= t_lo - 1e-10) & (cand_arr <= t_hi + 1e-10)]
    cand_arr = np.clip(cand_arr, t_lo, t_hi)
    
    # Also add a fine grid
    grid = np.linspace(t_lo, t_hi, 300)
    all_pts = np.concatenate([cand_arr, grid])
    
    best_val = np.inf
    best_t = t_lo
    chunk = 5000
    for st in range(0, len(all_pts), chunk):
        pts = all_pts[st:st+chunk]
        F = S0[None,:] + pts[:,None]*T[None,:] - (pts[:,None]**2)*Q[None,:]
        vals = np.max(F, axis=1) * dx
        bi = np.argmin(vals)
        if vals[bi] < best_val:
            best_val = vals[bi]
            best_t = float(pts[bi])
    
    return best_t, best_val

def _subgrad_direction(h, dx, beta=100):
    """Compute subgradient of C5 using softmax weighting."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(beta * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    
    grad = np.zeros(n)
    for r in range(2*n-1):
        if w[r] < 1e-14:
            continue
        d = r - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g+m])
                grad[g:] += w[r] * (1.0 - h[:m])
    grad -= np.mean(grad)
    return grad

def _resample(h, n_new):

    if n_new == n:
