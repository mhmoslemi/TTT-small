def _project(v, target):
    # Water-filling projection onto [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_count = int(np.max(corr))
    return max_count * dx, max_count, corr

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=800):

            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores_full(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = np.maximum(scores, 0) / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw

        # Compute gradient direction
        grad = np.zeros(n)
        for r in range(len(scores)):
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[d:])
                    grad[d:] += w[r] * (-h[:m])
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += w[r] * (1.0 - h[g:])
                    grad[g:] += w[r] * (-h[:m])
        grad = (2.0 / n) * grad
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue

        d = -grad / gn
        # Line search
        alpha_max = 0.5

        alpha_max = max(alpha_max, 1e-12)

        n_s = 15
        alphas = np.linspace(0, alpha_max, n_s)
        vals = np.array([_exact_c5(h + a * d) for a in alphas])
        best_i = int(np.argmin(vals))
        best_a = float(alphas[best_i])
        best_val = float(vals[best_i])

        # Golden section refine
        if 0 < best_i < n_s - 1:
            lo = float(alphas[best_i - 1])
            hi = float(alphas[best_i + 1])
            gr = (np.sqrt(5.0) - 1.0) / 2.0

                else:
                    lo = c1
                    f1 = f2
                    c1 = c2
                    c2 = lo + gr * (hi - lo)

def _lp_refine(h, deadline, top_k=8, radius=0.05, max_iters=2000):

        A_eq = np.ones((1, n + 1))
