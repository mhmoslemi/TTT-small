def _solve_qp(h0, active_indices, n):
    t = cp.Variable()
    d = cp.Variable(n)
    corr = _compute_correlations(h0)
    G = _compute_gradients(h0, active_indices)
    constraints = []
    for k_idx, r in enumerate(active_indices):
        # corr[r] + G[k_idx] @ d <= t
        constraints.append(corr[r] + G[k_idx] @ d <= t)
    constraints.append(cp.sum(d) == 0)
    constraints.append(d >= -h0)
    constraints.append(d <= 1 - h0)
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    if d.value is None:
        return None, None
    h_new = h0 + d.value
    return h_new, t.value

    c = np.zeros(n + 1)

    if not res.success or res.x is None:
        return None
    return res.x[:n], float(res.x[n])

def _sqp_optimize(h0, n, deadline, max_iter=2000, init_K=40):
    """Active-set SQP for the min-overlap problem."""
    h = h0.copy()
    c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    corr = _corr(h)
    K = min(init_K, 2 * n - 1)
    active = np.argsort(-corr)[:K]
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        sol = _solve_subproblem(h, active, n)
        if sol is None:
            stall += 5
            continue
        d, t_lp = sol
        
        # Check if d is essentially zero
        if np.max(np.abs(d)) < 1e-12:
            # At a KKT point of the linearized problem
            # Check if active set is complete
            corr = _corr(h)
            t_max = np.max(corr)
            missing = np.where(corr > t_max * (1 + 1e-10) - 1e-12)[0]
            # Actually, check if any constraint is violated
            violated = np.where(corr[active] > t_lp + 1e-10)[0]
            if len(violated) == 0 and not missing.size:
                break
            # Add violated constraints
            all_missing = np.where(corr > t_lp + 1e-8)[0]
            if len(all_missing) > len(active):
                active = np.union1d(active, all_missing)
                if len(active) > 80:
                    order = np.argsort(-corr[active])
                    active = active[order[:80]]
                stall = max(0, stall - 10)
            else:
                stall += 10
            continue
        
        # Line search
        best_alpha = 0.0
        best_c5_ls = c5
        best_h_ls = h.copy()
        alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.01]
        for alpha in alphas:
            h_try = _project(h + alpha * d, n * 0.5)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5_ls:
                best_c5_ls = c5_try
                best_alpha = alpha
                best_h_ls = h_try
        
        if best_alpha > 0 and best_c5_ls < c5 - 1e-14:
            h = best_h_ls
            c5 = best_c5_ls
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
            # Update active set
            corr = _corr(h)
            t_max = np.max(corr)
            new_active = np.where(corr >= t_max - 1e-6)[0]
            active = np.union1d(active, new_active)
            if len(active) > 80:
                order = np.argsort(-corr[active])
                active = active[order[:80]]
        else:
            stall += 1
            # Try adding all near-max constraints
            corr = _corr(h)
            t_max = np.max(corr)
            new_active = np.where(corr >= t_max - 5e-4)[0]
            active = np.unique(np.concatenate([active, new_active]))
            if len(active) > 100:
                order = np.argsort(-corr[active])
                active = active[order[:100]]
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _binary_search(n, deadline, n_starts=5):
    """Try to find good binary starting points."""
    best_h = None
    best_c5 = _exact_c5(h0) if 'h0' in globals() else 1.0
    for start in range(n_starts):
        if time.time() > deadline:
            break
        rng = np.random.default_rng(42 + start * 1000)
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
        c5 = _exact_c5(h)
        # Simple local search
        for _ in range(50000):
            if time.time() > deadline:
                break
            corr = _corr(h)
            max_k = np.argmax(corr)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            improved = False
            for _ in range(20):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i], h[j] = h[j], h[i]
                new_c5 = _exact_c5(h)
                if new_c5 < c5:
                    c5 = new_c5
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
            if not improved:
                break
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
