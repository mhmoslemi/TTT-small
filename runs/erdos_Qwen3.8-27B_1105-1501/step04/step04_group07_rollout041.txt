def _smooth_descent(h, deadline, T_start=0.05, T_end=0.005, n_steps=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    
    # Barzilai-Borwein parameters
    prev_grad = None
    prev_h = None
    step_size = 0.05
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        frac = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        scores = _scores(h)
        
        # Softmax weights (numerically stable)
        s_shifted = scores - scores.max()
        exp_s = np.exp(s_shifted / T)
        w = exp_s / exp_s.sum()
        
        # Gradient
        grad = G.T @ w
        grad -= np.mean(grad)  # Project onto sum=0 subspace
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # BB step size
        if prev_grad is not None:
            yh = h - prev_h
            ys = grad - prev_grad
            denom = np.dot(ys, ys)
            if denom > 1e-15:
                bb = np.dot(yh, ys) / denom
                step_size = max(0.01, min(0.2, bb))
        
        prev_grad = grad.copy()
        prev_h = h.copy()
        
        # Direction
        d = -grad / gn
        
        # Line search
        alpha, new_c5 = _line_search(h, d * step_size, cur_c5)
        
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * d * step_size
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            G = _grad_matrix(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

def _fourier_construction(n, coeffs, phases):
    x = (np.arange(n) + 0.5) / n
    v = np.zeros(n)
    for k, (a, phi) in enumerate(zip(coeffs, phases)):
        v += a * np.cos(2 * np.pi * (k + 1) * x + phi)
    return _project(v, n / 2.0)

def _block_construction(n, n_blocks, pattern):
    # pattern is a list of 0s and 1s of length n_blocks
    block_size = n // n_blocks
    h = np.zeros(n)
    for i, p in enumerate(pattern):
        if i < n_blocks - 1:
            h[i*block_size:(i+1)*block_size] = p
        else:
            h[i*block_size:] = p
    return _project(h, n / 2.0)

def _modular_construction(n, p, m, q):
    # h[i] = 1 if (i*m mod p) < q
    h = np.zeros(n)
    for i in range(n):
        if (i * m) % p < q:
            h[i] = 1.0
    return _project(h, n / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, d, cur_c5, max_evals=50):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_s = min(30, max_evals)
    alphas = np.linspace(0, alpha_max, n_s)
    vals = np.empty(n_s)
    for i, a in enumerate(alphas):
        vals[i] = _exact_c5(h + a * d)
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if 0 < best_i < n_s - 1:
        lo = float(alphas[best_i - 1])
        hi = float(alphas[best_i + 1])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * d)
        f2 = _exact_c5(h + c2 * d)
        for _ in range(15):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi = c2; f2 = f1; c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * d)
            else:
                lo = c1; f1 = f2; c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * d)
        oa = (lo + hi) / 2.0
        oc = _exact_c5(h + oa * d)
        if oc < best_c5:
            best_a, best_c5 = oa, oc
    return best_a, best_c5

def _smooth_descent(h, deadline, T_start=0.08, T_end=0.003, max_steps=10000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    prev_h = h.copy()
    prev_grad = None
    step_size = 0.05
    stall = 0
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        frac = step / max(1, max_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        
        # Softmax weights
        s_sh = scores - s_max
        exp_s = np.exp(s_sh / T)
        w = exp_s / float(exp_s.sum())
        
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        
        # BB step size
        if prev_grad is not None:
            yh = h - prev_h
            ys = grad - prev_grad
            denom = float(np.dot(ys, ys))
            if denom > 1e-15:
                bb = float(np.dot(yh, ys)) / denom
                if bb > 0:
                    step_size = max(0.01, min(0.15, bb))
        
        prev_h = h.copy()
        prev_grad = grad.copy()
        
        d = -grad / gn
        found = False
        for scale in [1.0, 0.5, 0.25, 0.12, 0.06]:
            alpha, new_c5 = _line_search(h, d * step_size * scale, cur_c5)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * step_size * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        
        if not found:
            stall += 1
            if stall > 200:
                break
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5

def _lp_step(h, deadline, radius=0.1, top_k=20, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        
        scores = _scores(h)
        G = _grad_matrix(h)
        
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-6:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.4)
        else:
            stall += 1
            r *= 0.85
            if stall > 40 and r < 0.03:
                r = 0.08
    
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=5):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n):
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi_ = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi_ - lo) / 3.0
                        m2 = hi_ - (hi_ - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_ = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi_) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _line_search(h, d, cur_c5, max_evals=50):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if d[j] > 1e-14:
            am = (1.0 - h[j]) / d[j]
            if am < alpha_max:
                alpha_max = am
        elif d[j] < -1e-14:
            am = h[j] / (-d[j])
            if am < alpha_max:
                alpha_max = am
    alpha_max = max(alpha_max, 1e-12)
    n_s = min(30, max_evals)
    alphas = np.linspace(0.0, alpha_max, n_s)
    vals = np.empty(n_s)
    for i in range(n_s):
        vals[i] = _exact_c5(h + alphas[i] * d)
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if 0 < best_i < n_s - 1:
        lo = float(alphas[best_i - 1])
        hi = float(alphas[best_i + 1])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * d)
        f2 = _exact_c5(h + c2 * d)
        for _ in range(18):
            if hi - lo < 1e-11:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * d)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * d)
        oa = (lo + hi) / 2.0
        oc = _exact_c5(h + oa * d)
        if oc < best_c5:
            best_a, best_c5 = oa, oc
    return best_a, best_c5

def _smooth_descent(h, deadline, T_start=0.06, T_end=0.002, max_steps=15000, use_momentum=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    prev_h = h.copy()
    prev_grad = None
    step_size = 0.04
    momentum_vec = np.zeros(n)
    beta = 0.9
    stall = 0
    n_stall_limit = 300

    for step in range(max_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, max_steps - 1)
        T = T_start * (T_end / T_start) ** frac

        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break

        s_sh = scores - s_max
        exp_s = np.exp(s_sh / T)
        w = exp_s / float(exp_s.sum())

        grad = G.T @ w
        grad -= np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            stall += 1
            if stall > n_stall_limit:
                break
            continue

        if prev_grad is not None:
            yh = h - prev_h
            ys = grad - prev_grad
            denom = float(np.dot(ys, ys))
            if denom > 1e-15:
                bb = float(np.dot(yh, ys)) / denom
                if 0.005 < bb < 0.2:
                    step_size = bb

        prev_h = h.copy()
        prev_grad = grad.copy()

        d = -grad / gn
        if use_momentum:
            d_new = beta * momentum_vec + d
            momentum_vec = d_new
            d = d_new / max(1.0, np.linalg.norm(d_new))

        found = False
        for scale in [1.0, 0.6, 0.35, 0.18, 0.08]:
            alpha, new_c5 = _line_search(h, d * step_size * scale, cur_c5)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * step_size * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break

        if not found:
            stall += 1
            beta = max(0.5, beta * 0.995)
            if stall > n_stall_limit:
                break

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    return best_h, best_c5

def _lp_optimize(h, deadline, radius=0.1, top_k=15, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    r = radius
    n_lags = 2 * n - 1

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 120:
            break

        scores = _scores(h)
        G = _grad_matrix(h)

        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G

        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo_b = max(-r, -h[j])
            hi_b = min(r, 1.0 - h[j])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue

        if float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-6:
                break
            continue

        alpha, new_c5 = _line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.04, 0.4)
        else:
            stall += 1
            r *= 0.82
            if stall > 35 and r < 0.025:
                r = 0.08

    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=6):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        frac_idx = [j for j in range(n) if 0.01 < h[j] < 0.99]
        for j in frac_idx:
            if time.time() > deadline:
                break
            for k in range(n):
                if k == j:
                    continue
                if not (0.01 < h[k] < 0.99) and not (h[j] > 0.01 and h[j] < 0.99):
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi_ = 0.0, a_max
                    for _ in range(12):
                        m1 = lo + (hi_ - lo) / 3.0
                        m2 = hi_ - (hi_ - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_ = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi_) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = _project(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(best_h)
    best_n = n0

    # Phase 1: Binary hill climb (8%)
    dl1 = t0 + budget_s * 0.08
    bh, bc5 = _binary_hill_climb(best_h, dl1, max_iters=20000)
    if bc5 < best_c5:
        best_c5 = bc5
        best_h = bh.copy()

    # Phase 2: Global search (7%)
    dl2 = t0 + budget_s * 0.15
    gh, gc5 = _global_search(n0, dl2, seed + 1)
    if gh is not None and gc5 < best_c5:
        best_c5 = gc5
        best_h = gh.copy()

    # Phase 3: LP coarse (12%)
    dl3 = t0 + budget_s * 0.27
    lh, lc5 = _lp_optimize(best_h, dl3, radius=0.12, top_k=12, max_iters=2000)
    if lc5 < best_c5:
        best_c5 = lc5
        best_h = lh.copy()

    # Phase 4: Smooth descent main (33%)
    dl4 = t0 + budget_s * 0.60
    sh, sc5 = _smooth_descent(best_h, dl4, T_start=0.05, T_end=0.002,
                              max_steps=20000, use_momentum=True)
    if sc5 < best_c5:
        best_c5 = sc5
        best_h = sh.copy()

    # Phase 5: Active set equalization (10%)
    dl5 = t0 + budget_s * 0.70
    ah, ac5 = _active_equalize(best_h, dl5, n_steps=4000)
    if ac5 < best_c5:
        best_c5 = ac5
        best_h = ah.copy()

    # Phase 6: LP fine (12%)
    dl6 = t0 + budget_s * 0.82
    lh, lc5 = _lp_optimize(best_h, dl6, radius=0.04, top_k=6, max_iters=3000)
    if lc5 < best_c5:
        best_c5 = lc5
        best_h = lh.copy()

    # Phase 7: Multi-start perturbation (8%)
    dl7 = t0 + budget_s * 0.90
    scale = 0.025
    for attempt in range(400):
        if time.time() > dl7:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, time.time() + budget_s * 0.015)
        h_try, c5_try = _smooth_descent(h_pert, sub_dl, T_start=0.02,
                                        T_end=0.003, max_steps=2000, use_momentum=False)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.82)
        else:
            scale = min(0.10, scale * 1.08)

    # Phase 8: Resolution sweep (5%)
    dl8 = t0 + budget_s * 0.95
    for dn in [-6, -4, -2, 2, 4, 6, 8, 12]:
        if time.time() > dl8:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 120:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl8, time.time() + budget_s * 0.015)
        h_r, c5_r = _smooth_descent(h_r, sub_dl, T_start=0.03,
                                    T_end=0.005, max_steps=3000, use_momentum=True)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 9: Final pair polish (5%)
    if time.time() < t0 + budget_s * 0.98:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.995, max_passes=4)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
