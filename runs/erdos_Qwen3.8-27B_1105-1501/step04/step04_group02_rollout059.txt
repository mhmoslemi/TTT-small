        h_new = _project(y_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
            if stall == 0 and cur_c5 < best_c5:
                mom_beta = min(mom_beta * 1.01, 0.92)
        else:
            stall += 1
            lr *= 0.9
            if stall > 100:
                break
            if stall % 30 == 0:
                mom_beta = max(mom_init, mom_beta * 0.95)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_resolution(h_best, c5_best, n_best, deadline, total_budget):
    best_h = h_best.copy()
    best_c5 = c5_best
    best_n = n_best
    candidates = set()
    for dn in [-8, -6, -4, -2, 0, 2, 4, 6, 8, 10, 12, 16]:
        nt = n_best + dn
        if 36 <= nt <= 200:
            candidates.add(nt)
    candidates = sorted(candidates)
    time_per = total_budget * 0.018
    for nt in candidates:
        if time.time() > deadline:
            break
        if nt == best_n and nt == n_best:
            continue
        h_r = _resample(best_h, nt)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt
        sub_dl = min(deadline, time.time() + time_per)
        h_r, c5_r = _active_lp_optimize(h_r, sub_dl, top_k_init=10, radius_init=0.12, max_stall=30)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt
    return best_h, best_c5, best_n

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if n <= 100:
            pairs = [(j, k) for j in range(n) for k in range(j + 1, n)]
            rng.shuffle(pairs)
            pairs = pairs[:200]
        else:
            pairs = []
            for _ in range(200):
                j = int(rng.integers(0, n))
                k = int(rng.integers(0, n))
                if j != k:
                    pairs.append((min(j,k), max(j,k)))
        for (j, k) in pairs:
            if time.time() > deadline:
                break
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8 or h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for (sj, sk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if sj > 0:
                    am = (1.0 - h[j]) / sj
                else:
                    am = h[j] / (-sj)
                if sk > 0:
                    am = min(am, (1.0 - h[k]) / sk)
                else:
                    am = min(am, h[k] / (-sk))
                if am < 1e-12:
                    continue
                for frac in [0.3, 0.6, 1.0]:
                    h_t = h.copy()
                    h_t[j] += sj * am * frac
                    h_t[k] += sk * am * frac
                    if h_t[j] < -1e-10 or h_t[j] > 1+1e-10 or h_t[k] < -1e-10 or h_t[k] > 1+1e-10:
                        continue
                    h_t = np.clip(h_t, 0.0, 1.0)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _perturb_optimize(h_best, deadline, total_budget, seed=42):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = 0.03
    time_per = total_budget * 0.015
    attempt = 0
    while time.time() < deadline and attempt < 300:
        attempt += 1
        if best_c5 < 0.3803:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + time_per)
        h_try, c5_try = _active_lp_optimize(h_pert, sub_dl, top_k_init=10, radius_init=0.1, max_stall=25)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.15)
    return best_h, best_c5

def _topk_subgrad_wrap(h, deadline, top_k=6, lr=0.02, mom=0.88):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    y = h.copy()
    vel = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        sp = _scores_phys(y)
        idx = np.argsort(sp)[::-1][:top_k]
        G = np.mean([_grad_r(y, int(r)) for r in idx], axis=0) * (2.0 / n)
        vel = mom * vel + (1.0 - mom) * G
        y_new = y - lr * vel
        h_new = _project(y_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            y = y_new
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.015, 0.08)
        else:
            stall += 1
            lr *= 0.92
            if stall > 80:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s - 5.0

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick subgradient warmup (5%)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _topk_subgrad_wrap(h, d1, top_k=8, lr=0.02, mom=0.85)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Active LP optimization (40%)
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _active_lp_optimize(best_h, d2, top_k_init=12, radius_init=0.2, max_stall=80)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Subgradient refinement (10%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _topk_subgrad_wrap(best_h, d3, top_k=5, lr=0.008, mom=0.9)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.75
    best_h, best_c5, best_n = _multi_resolution(best_h, best_c5, best_n, d4, budget_s)

    # Phase 5: Intensive LP at best resolution (12%)
    d5 = t0 + budget_s * 0.87
    h5, c5_5 = _active_lp_optimize(best_h, d5, top_k_init=10, radius_init=0.1, max_stall=60)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Perturbation search (7%)
    d6 = t0 + budget_s * 0.94
    h6, c5_6 = _perturb_optimize(best_h, d6, budget_s, seed=seed + 77)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final LP polish (3%)
    d7 = min(hard_deadline, t0 + budget_s * 0.97)
    h7, c5_7 = _active_lp_optimize(best_h, d7, top_k_init=20, radius_init=0.05, max_stall=30)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final pair polishing
    if time.time() < hard_deadline - 0.5:
        d8 = hard_deadline - 0.1
        h8, c5_8 = _pair_refine(best_h, d8, max_passes=60)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    return best_h, best_c5, best_n
