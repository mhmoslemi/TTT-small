```python
def _mask_correlation_coeffs(mask, n):
    """Precompute A, B, C coefficients for each lag of a two-level vector."""
    A = np.zeros(2 * n - 1)
    B = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    
    for s in range(-(n-1), n):
        # s is the lag (negative means we shift h to the left)
        if s >= 0:
            # corr = sum_{j=s}^{n-1} h[j]*(1-h[j-s])
            js = range(s, n)
        else:
            dp = -s
            # corr = sum_{j=0}^{n-1-dp} h[j]*(1-h[j+dp])
            js = range(0, n - dp)
            # In this case, "h[j] * (1-h[j-s])" = h[j] * (1-h[j+dp])
            # mask[j] and mask[j+dp]
        
        n11 = 0
        n10 = 0
        n01 = 0
        n00 = 0
        
        if s >= 0:
            for j in range(s, n):
                mj = mask[j]
                ms = mask[j - s]
                if mj and ms: n11 += 1
                elif mj and not ms: n10 += 1
                elif not mj and ms: n01 += 1
                else: n00 += 1
        else:
            dp = -s
            for j in range(0, n - dp):
                mj = mask[j]
                ms = mask[j + dp]
                if mj and ms: n11 += 1
                elif mj and not ms: n10 += 1
                elif not mj and ms: n01 += 1
                else: n00 += 1
        
        idx = s + (n - 1)  # index into the full correlation
        A[idx] = n10 + n01 - n11 - n00
        B[idx] = n11 + n00 - 2 * n10
        C[idx] = n10
    
    return A, B, C

def _c5_two_level_fast(A, B, C, d):
    """Compute C5 for two-level vector given precomputed coefficients."""
    n = (len(A) + 1) // 2
    vals = A * d * d + B * d + C
    return float(np.max(vals) * (2.0 / n))

def _opt_d_fast(coeffs, n, n_grid=500):
    """Find optimal p for two-level vector given precomputed coefficients."""
    A, B, C = coeffs
    ps = np.linspace(0, 0.5, n_grid)
    # Vectorized: compute all correlations for all p values
    # vals[p_idx, lag_idx] = A[lag_idx] * p[p_idx]^2 + B[lag_idx] * p[p_idx] + C[lag_idx]
    # Shape: (n_grid, 2n-1)
    vals = A[np.newaxis, :] * (ps[:, np.newaxis] ** 2) + B[np.newaxis, :] * ps[:, np.newaxis] + C[np.newaxis, :]
    max_vals = np.max(vals, axis=1) * (2.0 / n)
    best_idx = np.argmin(max_vals)
    best_p = ps[best_idx]
    best_val = max_vals[best_idx]
    
    # Refine
    lo = max(0, best_p - 0.005)
    hi = min(0.5, best_p + 0.005)
    ps_fine = np.linspace(lo, hi, 200)
    vals_fine = A[np.newaxis, :] * (ps_fine[:, np.newaxis] **

2) + B[np.newaxis, :] * ps_fine[:, np.newaxis] + C[np.newaxis, :]
    max_fine = np.max(vals_fine, axis=1) * (2.0 / n)
    best_idx_f = np.argmin(max_fine)
    if max_fine[best_idx_f] < best_val:
        best_p = ps_fine[best_idx_f]

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _tl(mask, p):
    return np.where(mask, 1.0 - p, p)

def _coeffs(mask):
    """Compute (A, B, C) coefficient arrays for two-level correlation."""
    n = len(mask)
    size = 2 * n - 1
    A = np.zeros(size)
    B = np.zeros(size)
    C = np.zeros(size)
    
    for s in range(-(n-1), n):
        idx = s + (n - 1)
        if s >= 0:
            n11 = n10 = n01 = n00 = 0
            for j in range(n - s):
                mj = mask[j]
                ms = mask[j + s]
                if mj:
                    if ms: n11 += 1
                    else: n10 += 1
                else:
                    if ms: n01 += 1
                    else: n00 += 1
        else:
            dp = -s
            n11 = n10 = n01 = n00 = 0
