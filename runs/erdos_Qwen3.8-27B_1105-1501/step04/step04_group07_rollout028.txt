def _thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    tm = np.zeros(n)
    for i in range(n):
        cnt = 0
        x = i
        while x > 0:
            cnt += x & 1
            x >>= 1
        tm[i] = 1.0 if cnt % 2 == 1 else 0.0
    # Adjust to have exactly n//2 ones
    ones = int(tm.sum())
    target = n // 2
    if ones > target:
        idx = np.where(tm == 1)[0]
        np.random.shuffle(idx)
        tm[idx[:ones - target]] = 0
    elif ones < target:
        idx = np.where(tm == 0)[0]
        np.random.shuffle(idx)
        tm[idx[:target - ones]] = 1
    return tm

def _golden_init(n, seed):
    """Initialize using golden ratio distribution."""
    rng = np.random.default_rng(seed)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.zeros(n)
    for i in range(n):
        positions[i] = (i / phi) % 1.0
    # Sort and assign: first n//2 get 1, rest get 0
    order = np.argsort(positions)
    h = np.zeros(n)
    h[order[:n//2]] = 1.0
    return h

    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):

        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = Gk
    A_ub[:, n] = -1.0
    b_ub = -sk
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                       bounds=bounds, method='highs')
        if not res.success:
            return None
        return res.x[:n]
    except Exception:
        return None

def _lp_optimize(h, deadline, top_k=None, radius=0.1, max_iters=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    r = radius
    n_lags = 2 * n - 1
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        if top_k is not None and top_k < n_lags:
            scores = _scores(h)
            idx = np.argsort(scores)[-top_k:]
        else:
            idx = np.arange(n_lags)
        d = _lp_solve(h, idx, r)
        if d is None:
            stall += 1
            r *= 0.8
            continue
        if np.max(np.abs(d)) < 1e-13:
            stall += 1
            r *= 0.8
            if r < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, d, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.02, 0.25)
        else:
            stall += 1
            r *= 0.85
            if stall > 15 and r < 0.02:
                r = 0.06
    return best_h, best_c5

def _block_descent(h, deadline, n_blocks=8):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    n_lags = 2 * n - 1
    stall = 0
    block_size = n // n_blocks
    blocks = [slice(i * block_size, min(n, (i + 1) * block_size)) for i in range(n_blocks)]
    if block_size * n_blocks < n:
        blocks[-1] = slice(blocks[-1].start, n)
    for it in range(50000):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        bi = it % n_blocks
        sl = blocks[bi]
        bs = sl.stop - sl.start
        if bs < 2:
            continue
        scores = _scores(h)
        idx = np.argsort(scores)[-min(20, n_lags):]
        G = _grad_matrix(h)
        Gk = G[idx, sl]
        sk = scores[idx]
        c = np.zeros(bs + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(idx), bs + 1))
        A_ub[:, :bs] = Gk
        A_ub[:, bs] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, bs + 1))
        A_eq[0, :bs] = 1.0
        bounds = [(max(-0.15, -h[sl][j]), min(0.15, 1.0 - h[sl][j])) for j in range(bs)]
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                           bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                continue
            d_block = res.x[:bs]
        except Exception:
            stall += 1
            continue
        if np.max(np.abs(d_block)) < 1e-13:
            stall += 1
            continue
        d_full = np.zeros(n)
        d_full[sl] = d_block
        alpha, new_c5 = _line_search(h, d_full, max_alpha=1.0)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d_full
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
    return best_h, best_c5

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=800):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 50:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.08, 0.05, 0.025, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 80:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _active_equalize(h, deadline, n_steps=400):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        s_max = float(scores.max())
        thresh = s_max * 0.995
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.where(scores >= s_max * 0.98)[0]
        if len(active) == 0:
            break
        weights = np.zeros(n_lags)
        for idx in active:
            weights[idx] = 1.0
        weights /= weights.sum()
        grad = G.T @ weights
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 40:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.06, 0.03, 0.015]:
            alpha, new_c5 = _line_search(h, d * scale, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 60:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=2):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline:
                break
            for k in range(j + 1, n, 3):
                if h[j] < 1e-9 or h[j] > 1.0 - 1e-9:
                    continue
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo, hi = 0.0, a_max
                    for _ in range(8):
                        m1 = lo + (hi - lo) / 3.0
                        m2 = hi - (hi - lo) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi = m2
                        else:
                            lo = m1
                    a_opt = (lo + hi) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = _project(h_try, n / 2.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target = n0 / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Full LP (all lags) - 12%
    dl1 = t0 + budget_s * 0.12
    hdr, c5r = _lp_optimize(h, dl1, top_k=None, radius=0.12, max_iters=5000)
    if c5r < best_c5 - 1e-12:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 2: Block coordinate descent - 18%
    dl2 = t0 + budget_s * 0.30
    hdr, c5r = _block_descent(best_h, dl2, n_blocks=8)
    if c5r < best_c5 - 1e-12:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 3: Adaptive top-k LP - 20%
    dl3 = t0 + budget_s * 0.50
    for k in [25, 15, 10, 7, 5]:
        if time.time() >= dl3:
            break
        sub_dl = min(dl3, time.time() + budget_s * 0.04)
        hdr, c5r = _lp_optimize(best_h, sub_dl, top_k=k, radius=0.06, max_iters=4000)
        if c5r < best_c5 - 1e-12:
            best_c5 = c5r
            best_h = hdr.copy()

    # Phase 4: Active set equalization - 10%
    dl4 = t0 + budget_s * 0.60
    hdr, c5r = _active_equalize(best_h, dl4, n_steps=500)
    if c5r < best_c5 - 1e-12:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 5: P-norm descent - 12%
    dl5 = t0 + budget_s * 0.72
    hdr, c5r = _pnorm_descent(best_h, dl5, p_start=16.0, p_end=8192.0, n_steps=600)
    if c5r < best_c5 - 1e-12:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 6: Multi-start perturbation - 13%
    dl6 = t0 + budget_s * 0.85
    scales = [0.015, 0.02, 0.03, 0.02, 0.01]
    scale_idx = 2
    for attempt in range(500):
        if time.time() >= dl6:
            break
        if best_c5 < 0.3800:
            break
        scale = scales[scale_idx % len(scales)]
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.015)
        h_try, c5_try = _lp_optimize(h_pert, sub_dl, top_k=8, radius=0.05, max_iters=400)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale_idx = max(0, scale_idx - 1)
        else:
            scale_idx = min(len(scales) - 1, scale_idx + 1)

    # Phase 7: Resolution sweep - 8%
    dl7 = t0 + budget_s * 0.93
    for dn in [-2, -1, 1, 2, 4, -4, 6]:
        if time.time() >= dl7:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + budget_s * 0.01)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, top_k=10, radius=0.08, max_iters=500)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final ultra-fine LP - 5%
    dl8 = t0 + budget_s * 0.98
    hdr, c5r = _lp_optimize(best_h, dl8, top_k=3, radius=0.008, max_iters=5000)
    if c5r < best_c5 - 1e-13:
        best_c5 = c5r
        best_h = hdr.copy()

    # Phase 9: Pair polish - 2%
    if time.time() < t0 + budget_s * 0.99:
        hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.999, max_passes=2)
        if c5p < best_c5 - 1e-13:
            best_c5 = c5p
            best_h = hp.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = _project(best_h, best_n / 2.0)
        final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
