    if target >= n:

    hi = float(v.max()) + 2.0

            lo = mid
        else:

    if abs(res) > 1e-12:

        else:
            mask = x > 0.0

            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute gradient matrix G where G[r, j] = dS_{lag}/dh[j]*dx for lag = r-(n-1)"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                # S_d = sum_{i=0}^{m-1} h[i+d]*(1-h[i]), dS_d/dh[j]:
                # dh[j] appears as h[i+d] when j = i+d, i.e., i = j-d (if j >= d)
                #   contributes (1-h[j-d])
                # dh[j] appears as h[i] when j = i (if j < m = n-d)
                #   contributes -h[j+d]
                G[r, d:] += (1.0 - h[:m])  # for j >= d: (1-h[j-d])
                G[r, :m] += -h[d:]          # for j < m: -h[j+d]

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_step(h, radius=0.15, top_k=None, beta=50.0):
    """LP-based step with soft active set."""
    n = len(h)
    scores = _scores(h)
    G = _grad_matrix(h)
    n_lags = len(scores)
    
    # Soft active set: use all lags but weight by proximity to max
    max_s = np.max(scores)
    weights = np.exp(beta * (scores - max_s))
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        w = np.zeros(n_lags)
        w[np.argmax(scores)] = 1.0
    else:
        w = weights / w_sum
    
    # Select top constraints by weight
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(w)[-top_k:]
    else:
        idx = np.arange(n_lags)
    
    G_k = G[idx] * w[idx, None]  # weighted rows
    scores_k = scores[idx]
    
    # LP: min t s.t. G_k @ p <= -scores_k - t, sum(p) = 0, box constraints
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-12)
        hi_b = min(radius, 1.0 - h[j] - 1e-12)
        if lo_b > hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, max_alpha=0.5):
    n = len(h)
    p_norm = np.linalg.norm(p)
    if p_norm < 1e-15:
        return 0.0, _exact_c5(h)
    p = p / p_norm
    
    for j in range(n):
        if p[j] > 1e-14:
            max_alpha = min(max_alpha, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            max_alpha = min(max_alpha, h[j] / (-p[j]))
    max_alpha = max(max_alpha, 1e-12) * 0.99
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, max_alpha
    f_lo = _exact_c5(h)
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    best_a = opt_a if opt_c5 < f_lo else 0.0
    return best_a, min(opt_c5, f_lo)

When we swap positions i (0→1) and j (1→0), the score at lag k changes by:
ΔS_k = effect of h[i]+1 and h[j]-1 on S_k

For S_k (k≥0): S_k = Σ_{t=0}^{n-k-1} h[t+k](1-h[t])
- h[i] appears as h[t] when t=i (if i < n-k): contributes -(1-h[i+k]) to S_k, now h[i]=1 so new contribution is -1... wait this is confusing.

Let me think differently. Let's denote the old values as h_old and new as h_new where h_new differs from h_old only at positions i and j.

ΔS_k = S_k(h_new) - S_k(h_old)

For k ≥ 0: S_k = Σ_{t=0}^{n-k-1} h[t+k](1-h[t])

Position i:
- As h[t+k] (when t+k=i, i.e., t=i-k, valid if i≥k and i-k < n-k, i.e., i < n): contributes (1-h[i-k]) old → now h_new[i]=1, so new contribution is (1-h[i-k]) still... wait no. h_new[i] = 1, so the term h_new[t+k](1-h_new[t]) when t+k=i becomes 1*(1-h_new[t]) = 1*(1-h[i-k]) (assuming i≠j so h[i-k] is unchanged). Old was h_old[i]*(1-h[i-k]) = 0*(1-h[i-k]) = 0. So change is +(1-h[i-k]) if i≥k.
- As h[t] (when t=i, valid if i < n-k, i.e., k < n-i): the term h[t+k](1-h[t]) = h[i+k](1-h[i]). Old: h[i+k]*(1-0) = h[i+k]. New: h_new[i+k]*(1-1) = 0. Change is -h[i+k] if k < n-i.

Position j (h[j]: 1→0):
- As h[t+k] (when t+k=j, i.e., t=j-k, valid if j≥k): old contribution h_old[j]*(1-h[j-k]) = 1*(1-h[j-k]). New: 0*(1-h[j-k]) = 0. Change is -(1-h[j-k]) if j≥k.
- As h[t] (when t=j, valid if j < n-k): old: h[j+k]*(1-h[j]) = h[j+k]*0 = 0. New: h_new[j+k]*(1-h_new[j]) = h[j+k]*1 = h[j+k]. Change is +h[j+k] if j < n-k.

So for k ≥ 0:
ΔS_k = [1_{i≥k}*(1-h[i-k]) - 1_{k<n-i}*h[i+k]] + [-1_{j≥k}*(1-h[j-k]) + 1_{k<n-j}*h[j+k]]

Similarly for k < 0 (let m = -k > 0): S_{-m} = Σ_{t=0}^{n-m-1} h[t](1-h[t+m])

Position i (0→1):
- As h[t] (when t=i, valid if i < n-m): old: 0*(1-h[i+m]) = 0. New: 1*(1-h[i+m]). Change: +(1-h[i+m]) if i < n-m.
- As h[t+m] (when t+m=i, i.e., t=i-m, valid if i≥m): old: h[i-m]*(1-0) = h[i-m]. New: h_new[i-m]*(1-1) = 0. Change: -h[i-m] if i≥m.

Position j (1→0):
- As h[t] (when t=j, valid if j < n-m): old: 1*(1-h[j+m]) = 1-h[j+m]. New: 0*(1-h[j+m]) = 0. Change: -(1-h[j+m]) if j < n-m.
- As h[t+m] (when t+m=j, i.e., t=j-m, valid if j≥m): old: h[j-m]*(1-1) = 0. New: h_new[j-m]*(1-0) = h[j-m]. Change: +h[j-m] if j≥m.

So for m > 0 (lag = -m):
ΔS_{-m} = [1_{i<n-m}*(1-h[i+m]) - 1_{i≥m}*h[i-m]] + [-1_{j<n-m}*(1-h[j+m]) + 1_{j≥m}*h[j-m]]

This allows O(1) computation of ΔS_k for each k, and O(n) to compute all ΔS_k. Then we can update the scores array in O(n) per swap trial.

This makes the best-improvement search O(n² * n) = O(n³) per iteration (n² swaps × n lags to check). For n=128, that's about 2M operations per iteration. With ~1μs per operation in numpy, that's ~2ms per iteration. We can do ~500K iterations in 1100s. More than enough.

But actually, we don't need to check ALL lags. We only need to check if max_k (S_k + ΔS_k) < current_max. We can early-exit if any lag already exceeds.

Even better: we can precompute which lags are affected by positions i and j. Position i affects lags k where |k| < n (all lags actually, since i can appear in the overlap window for any lag). So we do need to check all lags.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _swap_delta(h, i, j):
    """Compute change in all scores when h[i]:0->1 and h[j]:1->0. Returns array of deltas."""
    n = len(h)
    dx = 2.0 / n
    deltas = np.zeros(2 * n - 1)
    # For positive lags k>=0: S_k = sum_{t=0}^{n-k-1} h[t+k](1-h[t])
    # Position i (0->1): +1_{i>=k}*(1-h[i-k]) - 1_{k<n-i}*h[i+k]
    # Position j (1->0): -1_{j>=k}*(1-h[j-k]) + 1_{k<n-j}*h[j+k]
    for k in range(1, n):
        di = 0.0
        dj = 0.0
        if i >= k:
            di += (1.0 - h[i - k])
        if k < n - i:
            di -= h[i + k]
        if j >= k:
            dj -= (1.0 - h[j - k])
        if k < n - j:
            dj += h[j + k]
        deltas[n - 1 + k] = dx * (di + dj)
        # For negative lag -k: S_{-k} = sum_{t=0}^{n-k-1} h[t](1-h[t+k])
        di2 = 0.0
        dj2 = 0.0
        if i < n - k:
            di2 += (1.0 - h[i + k])
        if i >= k:
            di2 -= h[i - k]
        if j < n - k:
            dj2 -= (1.0 - h[j + k])
        if j >= k:
            dj2 += h[j - k]
        deltas[n - 1 - k] = dx * (di2 + dj2)
    return deltas

def _local_lp(h, deadline, radius_init=0.15, max_iters=2000, beta=30.0, top_k=None):
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        ws = np.sum(w)
        if ws < 1e-300:
            w = np.zeros(len(scores))
            w[np.argmax(scores)] = 1.0
        else:
            w /= ws
        if top_k is not None and top_k < len(scores):
            idx = np.argsort(w)[-top_k:]
        else:
            idx = np.arange(len(scores))
        Gk = G[idx] * w[idx, None]
        sk = scores[idx]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            lo_b = max(-radius, -h[jj] + 1e-10)
            hi_b = min(radius, 1.0 - h[jj] - 1e-10)
            if lo_b > hi_b:
                lo_b, hi_b = 0.0, 0.0
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        p = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        if p is None or np.max(np.abs(p)) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        # Line search
        pn = p / (np.linalg.norm(p) + 1e-30)
        alpha_max = 0.5
        for jj in range(n):
            if pn[jj] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[jj]) / pn[jj])
            elif pn[jj] < -1e-14:
                alpha_max = min(alpha_max, h[jj] / (-pn[jj]))
        alpha_max = max(alpha_max, 1e-12) * 0.99
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, alpha_max
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * pn)
        f2 = _exact_c5(h + c2 * pn)
        f0 = cur_c5
        for _ in range(30):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * pn)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * pn)
        opt_a = (lo + hi) / 2.0
        opt_c5 = _exact_c5(h + opt_a * pn)
        candidates = [(f0, 0.0), (f1, c1), (f2, c2), (opt_c5, opt_a)]
        best_f, best_a = min(candidates, key=lambda x: x[0])
        if best_f < cur_c5 - 1e-13:
            h = _project(h + best_a * pn, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.72
            if stall % 5 == 0:
                beta *= 1.5
                if beta > 200:
                    beta = 30.0
    return best_h, best_c5

def _combo_search(h, deadline, max_iters=200000, rng=None):
    """Best-improvement swap search on binary solution."""
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    s = np.sum(h_bin)
    if s != n / 2:
        if s > n / 2:
            idx = np.where(h_bin == 1.0)[0]
            rng.shuffle(idx)
            h_bin[idx[:int(s - n/2)]] = 0.0
        else:
            idx = np.where(h_bin == 0.0)[0]
            rng.shuffle(idx)
            h_bin[idx[:int(n/2 - s)]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    stall = 0
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        # Number of trials per iteration
        n_trials = min(200, len(ones) * len(zeros))
        best_delta = 1e-14
        best_i, best_j = -1, -1
        for _ in range(n_trials):
            i = zeros[rng.integers(len(zeros))]
            j = ones[rng.integers(len(ones))]
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            c5_try = _exact_c5(h_bin)
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            if c5_try < cur_c5 - best_delta:
                best_delta = cur_c5 - c5_try
                best_i, best_j = i, j
        if best_i >= 0:
            h_bin[best_i] = 1.0
            h_bin[best_j] = 0.0
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            cur_c5 -= best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lsexp_grad_step(h, beta, n_steps=50):
    """Log-sum-exp gradient descent with projection."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        ws = np.sum(w)
        if ws < 1e-300:
            break
        w /= ws
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = 0.1
        improved = False
        for _ in range(25):
            h_new = _project(h - alpha * grad, n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.6
            if alpha < 1e-10:
                break
        if not improved:
            break
    return best_h, best_c5
