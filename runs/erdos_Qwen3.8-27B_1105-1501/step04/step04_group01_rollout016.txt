def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else:
            hi = mid

def _line_search_quad(h, d, dx, trust_radius):

    n = len(h)

def _grad_matrix(h):
    """Build gradient matrix G where G[r, j] = ∂(corr_r)/∂h_j."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])

    return G

        scores_use = scores[top_idx]
        m_use = len(top_idx)
    else:
        G_use = G
        scores_use = scores
        m_use = m
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m_use, n + 1))
    A_ub[:, :n] = G_use
    A_ub[:, n] = -1.0
    b_ub = -scores_use
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
