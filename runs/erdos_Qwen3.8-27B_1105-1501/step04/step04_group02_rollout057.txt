def _manual_scores(h):
    n = len(h)
    m = 2 * n - 1
    scores = np.empty(m)
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            scores[r] = np.dot(h[d:], 1.0 - h[:cnt])
        else:
            g = -d
            cnt = n - g
            scores[r] = np.dot(h[:cnt], 1.0 - h[g:])
    return scores

def _manual_gradient(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            G[r, d:] += 1.0 - h[:cnt]
            G[r, :cnt] += -h[d:]
        else:
            g = -d
            cnt = n - g
            G[r, :cnt] += 1.0 - h[g:]

def _project(v, target):
    n = len(v)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _manual_scores(h):
    n = len(h)
    m = 2 * n - 1
    scores = np.empty(m)
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            scores[r] = np.dot(h[d:], 1.0 - h[:cnt])
        else:
            g = -d
            cnt = n - g
