import numpy as np

import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):

    dx = 2.0 / n

    return corr * dx, corr  # scaled, raw

def _proj(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0: mask = x < 1.0 - 1e-14
        else: mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    """Compute full gradient matrix G of shape (2n-1, n). G[r, j] = dc_r/dh_j."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            mm = n - d
            G[r, :mm] = 1.0 - h[d:d+mm]  # first term: (1-h[j+d]) for j in [0, mm-1]
            G[r, d:] -= h[:mm]  # second term: -h[j-d] for j in [d, n-1]
        else:
            g = -d
            mm = n - g
            G[r, g:] = 1.0 - h[:mm]  # first term: (1-h[j-g]) for j in [g, n-1]
            G[r, :mm] -= h[g:g+mm]  # second term: -h[j+g] for j in [0, mm-1]
    return G

def _line_search_1d(h, scores_raw, G, j, k, direction, n_eval=20):
    """
    1D line search along direction: h[j] += direction*delta, h[k] -= direction*delta.
    Returns best delta and best c5.
    """
    n = len(h)
    dx = 2.0 / n
    # Feasible range
    if direction > 0:
        delta_max = min(1.0 - h[j], h[k])
    else:
        delta_max = min(h[j], 1.0 - h[k])
    if delta_max < 1e-10:
        return 0.0, _c5(h)
    
    # Slopes for each lag
    slopes = direction * (G[:, j] - G[:, k])
    
    # Sample
    best_delta = 0.0
    best_c5 = _c5(h)
    alphas = np.linspace(0, delta_max, n_eval)
    for a in alphas[1:]:
        h_t = h.copy()
        h_t[j] += direction * a
        h_t[k] -= direction * a
        if h_t[j] < -1e-12 or h_t[j] > 1 + 1e-12 or h_t[k] < -1e-12 or h_t[k] > 1 + 1e-12:
            continue

def _coord_descent(h, deadline, n_passes=50, n_pairs_per_pass=200, seed=0):
    """Cyclic coordinate descent with pair moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    rng = np.random.default_rng(seed)
    
    for _pass in range(n_passes):
        if time.time() > deadline:
            break
        G = _grad_matrix(h)
        improved = False
        # Random pairs
        pairs = []
        for _ in range(n_pairs_per_pass):
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j != k and 0 < h[j] < 1 and 0 < h[k] < 1:
                pairs.append((j, k))
        
        for (j, k) in pairs:
            if time.time() > deadline:
                break
            for direction in [1, -1]:

                    h = _proj(h, target)
                    best_c5 = _c5(h)
                    best_h = h.copy()
                    improved = True
                    G = _grad_matrix(h)  # Rebuild

        if not improved:

    return best_h, best_c5

def _active_lp(h, top_k=10, radius=0.1):
    """LP direction on top-k active lags."""
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(scores_raw)[::-1][:top_k]
    G = _grad_matrix(h)
    G_active = G[idx, :]  # shape (top_k, n)
    c_active = scores_raw[idx]
    
    # Minimize t subject to: G_active @ p <= t - c_active, for each active lag
    # i.e., G_active[r] @ p + c_active[r] <= t
    # Variables: p (n), t (1)
    # min t
    # A_ub: G_active rows, with -1 in t column
    # b_ub: zeros (since G_active @ p + c_active <= t → G_active @ p - t <= -c_active)
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((top_k, n_vars))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = -c_active
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 10})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

                a, c1, f1 = c1, c2, f2

def _valid(h):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    # Area-preserving resampling
    x_old = np.arange(n) / n
    x_new = np.arange(n_new) / n_new
    # Each new sample is the average of old samples in its cell
    h_new = np.zeros(n_new)

def _proj(v, target):

def _grad_matrix(h):

    m = 2 * n - 1
    G = np.zeros((m, n))

                G[r, d:] -= h[:mm]
        else:
            g = -d
            mm = n - g
            if mm > 0:
                G[r, g:] = 1.0 - h[:mm]
                G[r, :mm] -= h[g:g + mm]

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    h_new = np.zeros(n_new)
    for j in range(n_new):
        lo_f = j / n_new
        hi_f = (j + 1) / n_new
        i_lo = int(lo_f * n)
        i_hi = int(np.ceil(hi_f * n))
        i_lo = max(0, min(i_lo, n - 1))
        i_hi = max(i_lo + 1, min(i_hi, n))
        if i_hi == i_lo:
            h_new[j] = h[i_lo]
        else:
            overlaps = np.zeros(i_hi - i_lo)
            for i in range(i_lo, i_hi):
                o_lo = i / n
                o_hi = (i + 1) / n
                ov = max(0.0, min(hi_f, o_hi) - max(lo_f, o_lo))
                overlaps[i - i_lo] = ov
            tot = overlaps.sum()
            if tot > 1e-12:
                h_new[j] = float(np.dot(h[i_lo:i_hi], overlaps) / tot)
            else:
                h_new[j] = h[i_lo]
    return _proj(h_new, n_new / 2.0)
