def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.array(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.5, float(v.max()) + 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        if mask.sum() > 0:
            x[mask] += res / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

    """Compute all row values (unnormalized)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _line_search_quad(h, d, dx, trust_radius):
    """Exact line search along direction d (mean-neutral).
    Returns (best_t, best_value)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible range for t
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # Golden section search on [t_lo, t_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(50):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _gradient_row(h, d, dx):
    """Compute gradient of row with lag d."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            # q_d = dx * sum_{i=0}^{m-1} h[i+d]*(1-h[i])
            # d/dh[j]: for j in [d, n-1]: dx*(1-h[j-d])
            #           for j in [0, m-1]: -dx*h[j+d]
            g[d:] += dx * (1.0 - h[:m])
            g[:m] += -dx * h[d:d+m]
    else:
        g_abs = -d
        m = n - g_abs
        if m > 0:
            # q_{-g} = dx * sum_{i=g}^{n-1} h[i-g]*(1-h[i])
            # Let j = i-g, so sum_{j=0}^{m-1} h[j]*(1-h[j+g])
            # d/dh[j]: for j in [0, m-1]: dx*(1-h[j+g])
            #           for j in [g, n-1]: -dx*h[j-g]
            g[:m] += dx * (1.0 - h[g_abs:g_abs+m])
            g[g_abs:] += -dx * h[:m]
    return g

def _subgradient(h, dx, k=5, tol=1e-6):
    """Compute subgradient: average of gradients of top-k rows."""
    n = len(h)
    rows = _row_values(h) * dx
    max_val = float(np.max(rows))
    active = np.where(rows >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(rows)])
    if len(active) > k:
        idx = np.argsort(rows[active])[::-1][:k]
        active = active[idx]
    g = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        g += _gradient_row(h, d, dx)
    g /= len(active)
    g -= np.mean(g)  # Make mean-neutral
    return g, len(active)

    res = target - x.sum()
    if abs(res) > 1e-12:
        mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        if mask.sum() > 0:
            x[mask] += res / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = -trust_radius, trust_radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_, b_ = t_lo, t_hi
    c1 = b_ - gr * (b_ - a_)
    c2 = a_ + gr * (b_ - a_)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(45):
        if b_ - a_ < 1e-12: break
        if f1 < f2:
            b_, c2, f2 = c2, c1, f1
            c1 = b_ - gr * (b_ - a_)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a_, c1, f1 = c1, c2, f2
            c2 = a_ + gr * (b_ - a_)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a_ + b_) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _grad_matrix(h):
    """Build the (2n-1) x n gradient matrix for all rows."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius):
    """Find LP direction to minimize max increase in active rows."""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[n] < 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, radius_init=0.12):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 40: break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6: break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
