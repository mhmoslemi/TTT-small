            break
        interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(interior) < 2:
            break
        best_d, best_v = 0.0, cur_c5
        best_pq = None
        npairs = min(len(interior)*(len(interior)-1)//2, 200)
        tried = set()
        for _ in range(npairs):
            p = int(interior[rng.integers(len(interior))])
            q = int(interior[rng.integers(len(interior))])
            if p == q:
                continue
            key = (min(p,q), max(p,q))
            if key in tried:
                continue
            tried.add(key)
            a0, bv, cv = _pair_effects(h, p, q)
            d, v = _pair_opt_delta(a0, bv, cv, p, q, h)
            if v < best_v - 1e-13:
                best_v, best_d, best_pq = v, d, key
        if best_pq is not None:
            p, q = best_pq
            h[p] -= best_d
            h[q] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _lp_local(h, deadline, top_k=12, radius=0.08, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    cur_c5 = best_c5
    r = radius
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        scores = _scores(h)
        n_lags = 2 * n - 1
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
        else:
            idx = np.arange(n_lags)
        G = np.zeros((len(idx), n))
        for ii, r in enumerate(idx):
            d = r - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            G[ii, i_lo:i_hi+1] += (1.0 - h[i_lo+d:i_hi+d+1])
            j_lo, j_hi = i_lo + d, i_hi + d
            G[ii, j_lo:j_hi+1] += -h[j_lo-d:j_hi-d+1]
        G *= (2.0 / n)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(idx), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores[idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-12:
            stall += 1
            r *= 0.7
            if r < 1e-6:
                break
            continue
        alpha = 1.0
        for _ in range(20):
            h_try = _project(h + alpha * p, target)
            c5_try = _c5(h_try)
            if c5_try < cur_c5 - 1e-12:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
                r = min(r * 1.05, 0.4)
                break
            alpha *= 0.6
        else:
            stall += 1
            r *= 0.8
    return best_h, best_c5

def _es_optimize(h_init, deadline, pop_size=160, sigma_init=0.06, sigma_min=0.002):
    n = len(h_init)
    target = n / 2.0
    rng = np.random.default_rng(200)
    pop = [h_init.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        pop[i] = _project(pop[i] + rng.normal(0, sigma_init, n), target)
    fitness = [_c5(p) for p in pop]
    best_i = int(np.argmin(fitness))
    best_h = pop[best_i].copy()
    best_c5 = fitness[best_i]
    cur_sigma = sigma_init
    stall = 0
    gen = 0
    while time.time() < deadline:
        gen += 1
        if stall > 80:
            break
        offspring = []
        for i in range(pop_size // 2):
            pi = pop[i]
            child = _project(pi + rng.normal(0, cur_sigma, n), target)
            offspring.append(child)
            child2 = _project(pi + rng.normal(0, cur_sigma, n), target)
            offspring.append(child2)
        off_fit = [_c5(o) for o in offspring]
        combined = pop[:] + offspring
        combined_fit = fitness[:] + off_fit
        order = np.argsort(combined_fit)
        pop = [combined[i].copy() for i in order[:pop_size]]
        fitness = [combined_fit[i] for i in order[:pop_size]]
        bi = int(np.argmin(fitness))
        if fitness[bi] < best_c5 - 1e-12:
            best_c5 = fitness[bi]
            best_h = pop[bi].copy()
            cur_sigma = max(cur_sigma * 0.97, sigma_min)
            stall = 0
        else:
            cur_sigma = max(cur_sigma * 0.95, sigma_min)
            stall += 1
        if cur_sigma < sigma_min * 1.1:
            if stall > 40:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_perturb(h, rng):
    n = len(h)
    target = n / 2.0
    h2 = h.copy()
    kind = rng.integers(4)
    if kind == 0:
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.02, 0.08)
        h2 += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
    elif kind == 1:
        k = rng.integers(4, 16)
        h2 += np.tile(np.random.default_rng(int(rng.integers(1e9))).uniform(-0.05, 0.05, k), n // k + 1)[:n]
    elif kind == 2:
        h2 += rng.normal(0, 0.04, n)
    else:
        h2 += np.linspace(-0.05, 0.05, n) * rng.uniform(0.5, 1.5)
    return _project(h2, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: Subgradient descent (8%)
    dl = t0 + budget_s * 0.08
    hs, cs = _subgrad_descent(h, dl, n_steps=30000, thresh_frac=0.99)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()

    # Phase 2: Pair descent (8%)
    dl = t0 + budget_s * 0.16
    hs, cs = _pair_descent(best_h, dl, max_rounds=2000)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()

    # Phase 3: LP local (12%)
    dl = t0 + budget_s * 0.28
    hs, cs = _lp_local(best_h, dl, top_k=16, radius=0.1, max_iters=2500)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()

    # Phase 4: ES optimization (25%)
    dl = t0 + budget_s * 0.53
    hs, cs = _es_optimize(best_h, dl, pop_size=120, sigma_init=0.05, sigma_min=0.003)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()

    # Phase 5: Subgradient refine (7%)
    dl = t0 + budget_s * 0.60
    hs, cs = _subgrad_descent(best_h, dl, n_steps=20000, thresh_frac=0.995)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()

    # Phase 6: LP + pair polish (10%)
    dl = t0 + budget_s * 0.70
    hs, cs = _lp_local(best_h, dl, top_k=8, radius=0.05, max_iters=2000)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()
    if time.time() < t0 + budget_s * 0.73:
        hs, cs = _pair_descent(best_h, t0 + budget_s * 0.73, max_rounds=1500)
        if cs < best_c5:
            best_c5, best_h = cs, hs.copy()

    # Phase 7: Multi-resolution ES (15%)
    dl = t0 + budget_s * 0.88
    for dn in [-16, -8, -4, -2, 2, 4, 8, 16]:
        if time.time() > dl:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl, time.time() + budget_s * 0.015)
        h_r, c5_r = _es_optimize(h_r, sub_dl, pop_size=60, sigma_init=0.04, sigma_min=0.005)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Perturbation + restart (7%)
    dl = t0 + budget_s * 0.95
    scale = 0.03
    for attempt in range(200):
        if time.time() > dl:
            break
        if best_c5 < 0.3805:
            break
        h_p = _structured_perturb(best_h, rng)
        sub_dl = min(dl, time.time() + budget_s * 0.012)
        h_p, c5_p = _lp_local(h_p, sub_dl, top_k=10, radius=0.08, max_iters=300)
        if c5_p < best_c5 - 1e-10:
            best_c5 = c5_p
            best_h = h_p.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 9: Final ultra-fine polish (5%)
    dl = t0 + budget_s * 0.99
    hs, cs = _lp_local(best_h, dl, top_k=4, radius=0.015, max_iters=3000)
    if cs < best_c5:
        best_c5, best_h = cs, hs.copy()
    if time.time() < t0 + budget_s * 0.995:
        hs, cs = _subgrad_descent(best_h, t0 + budget_s * 0.998, n_steps=10000, thresh_frac=0.999)
        if cs < best_c5:
            best_c5, best_h = cs, hs.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
