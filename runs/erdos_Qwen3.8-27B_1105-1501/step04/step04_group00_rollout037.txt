import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] = np.clip(x[mask] + res / cnt, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += -dx * h[d:]
                G[r, d:] += dx * (1.0 - h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += dx * (1.0 - h[g:])
                G[r, g:] += -dx * h[:m]
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diffset_seq(n, p):
    """Construct a binary sequence from quadratic residues mod p, padded to length n."""
    h = np.zeros(n)
    if p > 0 and p % 4 == 1:
        k = (p - 1) // 2
        residues = set()
        for x in range(1, p):
            residues.add((x * x) % p)
        for r in range(min(p, n)):
            if r in residues:
                h[r] = 1.0
        # Adjust to get exactly n/2 ones
        n_ones = int(np.sum(h))
        needed = n // 2 - n_ones
        if needed > 0:
            zero_positions = np.where(h == 0)[0]
            if len(zero_positions) >= needed:
                h[zero_positions[:needed]] = 1.0
        elif needed < 0:
            one_positions = np.where(h == 1)[0]
            if len(one_positions) >= -needed:
                h[one_positions[-needed:]] = 0.0
    return h

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
        G_k = G[idx]
    else:
        scores_k = scores
        G_k = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(scores_k), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

    lo = alphas[max(0, best_i - 1)]

    return best_a, best_c5

def _lp_local(h, deadline, top_k_list=None, radius_init=0.18, max_iters=2000):
    if top_k_list is None:
        top_k_list = [None, 16, 8, 4, 12, 6]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 50:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, t_val = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            ti += 1
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.75
            ti += 1
            if stall > 10 and radius < 0.03:
                radius = 0.06
    return best_h, best_c5

def _subgrad_optimize(h, deadline, top_k=8, init_alpha=0.3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = init_alpha
    stall = 0
    beta_sm = 15.0
    while time.time() < deadline:
        scores = _scores(h)
        max_s = np.max(scores)
        k_act = min(top_k, len(scores))
        idx = np.argsort(scores)[-k_act:]
        active_s = scores[idx]
        w = np.exp(beta_sm * (active_s - max_s))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(k_act)
            w[0] = 1.0
        else:
            w /= w_sum
        G = _grad_matrix(h)
        g = G[idx].T @ w
        g -= g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        found = False
        a = alpha
        for _ in range(25):
            h_try = _project(h - a * g, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-14:
                h = h_try
                cur_c5 = c5_try
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                alpha = min(a * 1.04, 2.0)
                stall = 0
                found = True
                break
            a *= 0.65
            if a < 1e-9:
                break
        if not found:
            stall += 1
            alpha *= 0.75
            if stall > 60:
                break
        beta_sm *= 1.01
    return best_h, best_c5

def _binary_sa_binary(h_init, deadline, rng, n_iters=30000):
    n = len(h_init)
    target = n // 2
    h = np.zeros(n)
    if np.sum(h_init) == target:
        h = h_init.astype(float).copy()
    else:
        positions = rng.permutation(n)
        h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        scores = _scores(h)
        worst_lag = int(np.argmax(scores))
        d = worst_lag - (n - 1)
        if abs(d) > 0 and abs(d) < n:
            # Targeted move: find a swap that affects the worst lag
            if d > 0:
                # N_d^+ = #{i: h[i]=1, h[i+d]=0}
                bad_ones = [i for i in ones if i + d < n and h[i + d] == 0]
                if len(bad_ones) > 0:
                    i = bad_ones[rng.integers(len(bad_ones))]
                    # Try setting h[i+d]=1 (if it's 0)
                    if h[i + d] == 0:
                        j = i + d
                        h[i] = 0.0
                        h[j] = 1.0
                        new_c5 = _exact_c5(h)
                        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
                            cur_c5 = new_c5
                            ones = np.where(h == 1.0)[0]
                            zeros = np.where(h == 0.0)[0]
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                        else:
                            h[i] = 1.0
                            h[j] = 0.0
                        T *= cool
                        continue
            # Fallback: random swap
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
                cur_c5 = new_c5
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
                cur_c5 = new_c5
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        T *= cool
    return best_h, best_c5

def _binary_to_continuous(h_bin, n_cont, rng):
    """Convert binary solution to continuous by adding small noise and projecting."""
    h_cont = _resample(h_bin, n_cont)
    h_cont += rng.normal(0, 0.05, n_cont)
    return _project(h_cont, n_cont / 2.0)

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = max(1, n // 24)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(j + 1, n, step):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(7):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        h1 = np.clip(h1, 0, 1)
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        h2 = np.clip(h2, 0, 1)
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _quasi_periodic_init(n, alpha):
    h = np.zeros(n)
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    # Phase 1: LP polish from initial (10%)
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _lp_local(h_start, dl1, top_k_list=[None, 16, 8, 4],
                         radius_init=0.18, max_iters=2000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient from initial and best (12%)
    if time.time() < t0 + budget_s * 0.22:
        dl2 = t0 + budget_s * 0.22
        h2, c5_2 = _subgrad_optimize(best_h, dl2, top_k=10, init_alpha=0.25)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Multi-resolution diverse search (30%)
    dl3 = t0 + budget_s * 0.52
    resolutions = [64, 80, 96, 112, 128, 144, 160, 96, 80, 112]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5, 0.555, 0.455]

    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3:
            break
        if best_c5 < 0.3804:
            break
        n_starts = 3
        for s in range(n_starts):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_periodic_init(n_try, alpha)
                h_init += rng.normal(0, 0.06, n_try)
                h_init = _project(h_init, n_try / 2.0)
            else:
                # Difference set based
                primes_near = [127, 113, 89, 79, 71, 67]
                p = primes_near[ri % len(primes_near)]
                h_init = _diffset_seq(n_try, p)
                h_init += rng.normal(0, 0.04, n_try)
                h_init = _project(h_init, n_try / 2.0)

            sub_dl = min(dl3, time.time() + budget_s * 0.03)
            h_try, c5_try = _lp_local(h_init, sub_dl,
                                       top_k_list=[None, 12, 6, 4],
                                       radius_init=0.15, max_iters=1500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            else:
                if time.time() < dl3:
                    sub_dl2 = min(dl3, time.time() + budget_s * 0.012)
                    h_sg, c5_sg = _subgrad_optimize(h_try, sub_dl2, top_k=8, init_alpha=0.2)
                    if c5_sg < best_c5:
                        best_c5 = c5_sg
                        best_h = h_sg.copy()
                        best_n = n_try

    # Phase 4: Binary SA at multiple resolutions (20%)
    if time.time() < t0 + budget_s * 0.72:
        dl4 = t0 + budget_s * 0.72
        for n_bin in [128, 96, 160, 80]:
            if time.time() > dl4:
                break
            sub_dl = min(dl4, time.time() + budget_s * 0.04)
            h_b, c5_b = _binary_sa_binary(best_h if len(best_h) == n_bin else _resample(best_h, n_bin),
                                           sub_dl, rng, n_iters=30000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_n = n_bin
                # Convert to continuous and refine
                h_cont = _binary_to_continuous(h_b, n_bin, rng)
                sub_dl2 = min(dl4, time.time() + budget_s * 0.02)
                h_rc, c5_rc = _lp_local(h_cont, sub_dl2,
                                         top_k_list=[8, 4, None],
                                         radius_init=0.1, max_iters=800)
                if c5_rc < best_c5:
                    best_c5 = c5_rc
                    best_h = h_rc.copy()
                    best_n = n_bin
            else:
                # Try from best_h resampled
                h_b2, c5_b2 = _binary_sa_binary(_resample(best_h, n_bin),
                                                  sub_dl, rng, n_iters=15000)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = _resample(h_b2, n_bin)
                    best_n = n_bin

    # Phase 5: LP + Subgradient on best (10%)
    if time.time() < t0 + budget_s * 0.82:
        dl5 = t0 + budget_s * 0.82
        h5, c5_5 = _lp_local(best_h, dl5,
                              top_k_list=[None, 10, 6, 4, 8],
                              radius_init=0.12, max_iters=1500)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if time.time() < dl5:
            h5b, c5_5b = _subgrad_optimize(best_h, dl5, top_k=6, init_alpha=0.15)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: Perturbation restarts (10%)
    dl6 = t0 + budget_s * 0.92
    scale = 0.025
    for attempt in range(200):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        h_try, c5_try = _lp_local(h_pert, sub_dl,
                                   top_k_list=[6, 4, 8, None],
                                   radius_init=0.06, max_iters=300)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.004, scale * 0.82)
        else:
            scale = min(0.08, scale * 1.10)

    # Phase 7: Pair polish (6%)
    if time.time() < t0 + budget_s * 0.97:
        dl7 = t0 + budget_s * 0.985
        hp, c5p = _pair_polish(best_h, dl7, max_passes=5)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: tiny LP refinement
    if time.time() < t0 + budget_s * 0.99:
        h_fin, c5_fin = _lp_local(best_h, t0 + budget_s * 0.998,
                                   top_k_list=[3, 4, 5],
                                   radius_init=0.01, max_iters=400)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, best_n
