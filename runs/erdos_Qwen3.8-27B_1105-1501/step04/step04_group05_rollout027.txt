def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_max(h):
    """Return max of raw correlation (without dx factor) and the argmax lag."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)), int(np.argmax(corr))

def _project(v, target):

    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)

    threshold = max_val - 1e-6
    active = np.where(corr >= threshold)[0]
