The least-squares solution minimizes ||G_A @ p + e||², which tries to reduce all active constraints simultaneously.

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(r, h):
    n = len(h)
    grad = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        grad[d:] += 1.0 - h[:m]
        grad[:m] -= h[d:]
    else:
        g = -d
        m = n - g
        grad[g:] += 1.0 - h[:m]
        grad[:m] -= h[g:]
    return grad

def _alpha_max_bounds(h, p):
    am = 10.0

            am = min(am, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            am = min(am, h[j] / (-p[j]))
    return max(am, 1e-14)

def _gn_solve(G, e, radius, lam):
    n = G.shape[1]
    GTG = G.T @ G + lam * np.eye(n)
    GTe = G.T @ e
    try:
        p = np.linalg.solve(GTG, -GTe)
    except np.linalg.LinAlgError:
        try:
            p = np.linalg.lstsq(GTG, -GTe, rcond=None)[0]
        except Exception:
            return None
    if not np.all(np.isfinite(p)):
        return None
    p = p - np.mean(p)
    m = np.max(np.abs(p))
    if m < 1e-15:
        return None
    return p / m * radius

def _svd_directions(G, n_dirs, radius):
    n = G.shape[1]
    try:
        U, s, Vt = np.linalg.svd(G, full_matrices=False)
    except Exception:
        return []
    dirs = []
    nd = min(n_dirs, Vt.shape[0])
    for i in range(nd):
        p = Vt[i].copy()
        p = p - np.mean(p)
        m = np.max(np.abs(p))
        if m < 1e-15:
            continue
        p = p / m * radius
        dirs.append(p)
        dirs.append(-p)
    return dirs

def _roll_dir(h, k, radius):
    p = np.roll(h, k) - h
    p = p - np.mean(p)
    m = np.max(np.abs(p))
    if m < 1e-15:

    return p / m * radius

def _weighted_grad_direction(h, scores, radius, tau):
    n = len(h)
    max_s = float(np.max(scores))
    w = np.exp((scores - max_s) / tau)
    sw = float(np.sum(w))
    if sw < 1e-30:
        return None
    w = w / sw
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        grad += w[r] * _row_grad(r, h)
    grad = grad - np.mean(grad)
    m = np.max(np.abs(grad))
    if m < 1e-14:
        return None
    return -grad / m * radius

def _resample(h, n_new):

def _alternating_transforms(h, n, deadline, seed=0):

    cands.append(c1)

    cands.append(hb)

    for shift in [1, -1, 2, -2]:
        cands.append(_roll_dir(h, shift, 0.01) + h)

def _pair_refine(h, deadline, seed=42, max_passes=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_pairs = min(n * 3, 80)
        idx_j = rng.integers(0, n, size=n_pairs)
        idx_k = rng.integers(0, n, size=n_pairs)
        seen = set()
        for pi in range(n_pairs):
            if time.time() > deadline:
                break
            j, k = int(idx_j[pi]), int(idx_k[pi])
            if j == k:
                continue
            key = (min(j, k), max(j, k))
            if key in seen:
                continue
            seen.add(key)
            if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                continue
            if h[k] <= 1e-8 or h[k] >= 1.0 - 1e-8:
                continue
            for sign in [1.0, -1.0]:
                p = np.zeros(n)
                p[j] = sign
                p[k] = -sign
                am = _alpha_max_bounds(h, p)
                if am < 1e-12:
                    continue
                am = min(am, 0.5)
                t, v = _line_search(h, p, best_c5, am)
                if t > 1e-12 and v < best_c5 - 1e-13:
                    h = h + t * p
                    h = _project(h, target)
                    best_c5 = _exact_c5(h)
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

    # Initialize
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick core optimization at initial resolution (8%)
    d1 = t0 + budget_s * 0.08
    h, c5 = _core_optimize(h, d1, seed=seed, active_sizes=(8, 16, 24))
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Alternating transforms (3%)
    d2 = t0 + budget_s * 0.11
    h, c5 = _alternating_transforms(best_h, best_n, d2, seed=seed + 10)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        # Quick optimize the improved transform
        if time.time() < d2:
            h2, c52 = _core_optimize(h, d2, seed=seed + 20, active_sizes=(16, 24))
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2.copy()

    # Phase 3: Multi-resolution screening (20%)
    d3 = t0 + budget_s * 0.31
    best_h, best_c5, best_n = _multi_resolution(best_h, best_n, d3, time_per_frac=0.015)

    # Phase 4: Intensive core at best resolution (30%)
    d4 = t0 + budget_s * 0.61
    h, c5 = _core_optimize(best_h, d4, seed=seed + 100, active_sizes=(8, 16, 24, 36, 48))
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Another multi-res pass from improved best (12%)
    d5 = t0 + budget_s * 0.73
    best_h, best_c5, best_n = _multi_resolution(best_h, best_n, d5, time_per_frac=0.012)

    # Phase 6: Perturbation and restart (12%)
    d6 = t0 + budget_s * 0.85
    best_h, best_c5 = _perturb_refine(best_h, best_n, d6, seed=seed + 200)

    # Phase 7: Final core + pair polish (10%)
    d7 = t0 + budget_s * 0.95
    h, c5 = _core_optimize(best_h, min(d7, time.time() + budget_s * 0.05), seed=seed + 300, active_sizes=(8, 16, 24, 36))
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    if time.time() < d7:
        h, c5 = _pair_refine(best_h, d7, seed=seed + 400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final polish with very small steps
    if time.time() < hard_deadline:
        h, c5 = _core_optimize(best_h, hard_deadline, seed=seed + 500, active_sizes=(8, 16))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
