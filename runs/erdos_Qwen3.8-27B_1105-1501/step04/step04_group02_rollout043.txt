def _anneal(h, best_c5, deadline, seed):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    
    # Temperature schedule
    T = 0.002  # initial temperature
    T_min = 1e-6
    cool_rate = 0.9999  # per iteration
    
    # Move sizes
    base_delta = 0.03
    
    it = 0
    while time.time() < deadline:
        it += 1
        # Choose move type
        if rng.random() < 0.7:
            # Pair exchange
            i = rng.integers(n)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            delta = rng.uniform(-1e-4, 1e-4)  # symmetric
            # Actually let's use random sign
            delta = rng.uniform(0.005, base_delta) * (1 if rng.random() < 0.5 else -1)
            if h[i] + delta < 0 or h[i] + delta > 1:
                continue
            if h[j] - delta < 0 or h[j] - delta > 1:
                continue
            h[i] += delta
            h[j] -= delta
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                # Accept
            elif new_c5 > best_c5 + 0.005:
                # Reject: too far from best
                h[i] -= delta
                h[j] += delta
            else:
                # Metropolis acceptance
                if rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                    cur_c5 = new_c5
                else:
                    h[i] -= delta
                    h[j] += delta
        else:
            # Single move (move mass from one position to another)
            # This is the same as pair exchange really
            pass
        
        T = max(T_min, T * cool_rate)
    
    return h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_optimize(h, deadline, tau_init=15.0, lr_init=0.02, mom_beta=0.9):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            wr = w[r]
            if wr < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] -= wr * h[d:]
            else:
                g = -d
                m = n - g
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] -= wr * h[:m]
        grad *= dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.15)
            tau = min(tau * 1.02, 200.0)
        else:
            stall += 1
            lr *= 0.92
            if stall > 80:
                break

                tau = max(tau_init, tau * 0.8)
    return best_h, best_c5

def _lp_local(h, deadline, radius_init=0.15):

    target = n / 2.0

        b_ub = -scores
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):

            hi_j = min(radius, 1.0 - h[j])

        # Choose move: biased toward active positions
        if n_active > 4 and rng.random() < 0.6:
            active_mask = (h > 0.01) & (h < 0.99)
            candidates = np.where(active_mask)[0]
            i = int(candidates[rng.integers(len(candidates))])
            all_idx = np.arange(n)
            j = int(all_idx[rng.integers(n)])
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
        if i == j:
            continue
        delta = delta_scale * rng.standard_normal()
        hi_new = h[i] + delta
        hj_new = h[j] - delta
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            # Try smaller
            delta *= 0.3
            hi_new = h[i] + delta
            hj_new = h[j] - delta
            if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
                no_improve += 1
                continue
        h[i] = hi_new
        h[j] = hj_new
        new_c5 = _exact_c5(h)
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            ratio = (new_c5 - cur_c5) / T
            if ratio < 30 and new_c5 < best_c5 + 0.003:
                if rng.random() < np.exp(-ratio):
                    cur_c5 = new_c5
                    no_improve += 1
                else:
                    h[i] = hi_new - delta
                    h[j] = hj_new + delta
                    no_improve += 1
            else:
                h[i] = hi_new - delta
                h[j] = hj_new + delta
                no_improve += 1
        T = max(T_min, T * cool)
        # Restart if stuck
        if no_improve > 400000:
            noise = 0.005 + 0.005 * min(no_improve // 1000000, 3)
            h = best_h + rng.normal(0, noise, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            T = max(T, 0.0008)

    return best_h, best_c5

def _three_way_sweep(h_init, c5_init, deadline):

    best_c5 = c5_init
    best_h = h_init.copy()
    tol = 1e-6
    active = np.where((h > tol) & (h < 1.0 - tol))[0]

        return h, best_c5
    # Try triples from active set
    n_act = min(len(active), 25)
    act = active[:n_act]
    for a in range(n_act):
        if time.time() > deadline:
            break
        for b in range(a + 1, n_act):
            if time.time() > deadline:
                break
            for c in range(b + 1, n_act):
                i, j, k = act[a], act[b], act[c]
                # Move small amount: h[i]-t, h[j]+t, normalize via h[k]
                # Keep sum: h[i]-t, h[j]+t, h[k] unchanged (already sum-preserving)
                t_max = min(h[i], 1.0 - h[j])
                if t_max < 1e-8:

                # Try a few values

                    h[i] -= t
                    h[j] += t
                    v = _exact_c5(h)
                    if v < best_c5:
                        best_c5 = v
                        best_h = h.copy()
                    else:
                        h[i] += t
                        h[j] -= t
    return best_h, best_c5
