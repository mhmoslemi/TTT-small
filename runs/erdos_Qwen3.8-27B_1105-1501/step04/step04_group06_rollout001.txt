def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(x)
            for i in idx:
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            idx = np.argsort(-x)
            for i in idx:
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x

    ps = j + s
    m2 = (ps >= 0) & (ps < n)

Let me also think about the combinatorial local search more carefully.

**Pair swap local search**:
- Pick two indices i, j
- Consider moving mass δ from i to j: h[i] -= δ, h[j] += δ
- Find the optimal δ by checking the objective at the endpoints of the feasible interval
- The feasible interval for δ is: max(0, h[j]-1) <= δ <= min(h[i], 1-h[j])
- Actually, we want to minimize max_k c_k, so we should do a line search over δ

For efficiency:
- Try random pairs (i, j)
- For each pair, do a small grid search over δ
- Accept if the exact objective improves

This is O(n^2) in the worst case but we can sample pairs.

Actually, a better approach: for each coordinate i, try moving mass to the coordinate j that gives the best improvement. This is O(n^2) evaluations per sweep, each evaluation being O(n) for the correlation. So O(n^3) per sweep which for n=64 is about 262K operations. That's feasible.

But actually, we can be smarter. The change in c_k when we move δ from i to j is:
c_k(new) = c_k(old) + δ * (grad_c_k[i] - grad_c_k[j])

where grad_c_k is the gradient of c_k with respect to h.

So the change is linear in δ! This means the optimum δ for a given pair (i,j) and a given active lag k is at one of the endpoints of the feasible interval.

But since we have multiple active lags, the effective change is:
max_k [c_k + δ * (grad_c_k[i] - grad_c_k[j])]

This is the max of linear functions in δ, which is convex in δ. We can find the minimum by ternary search or by checking the breakpoints.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, s):
    n = len(h)
    j = np.arange(n)
    g = np.zeros(n)
    gs = j - s
    m1 = (gs >= 0) & (gs < n)
    if np.any(m1):
        g[m1] = 1.0 - h[gs[m1]]
    ps = j + s
    m2 = (ps >= 0) & (ps < n)
    if np.any(m2):
        g[m2] -= h[ps[m2]]
    return g

def _grad_matrix(h):
    n = len(h)
    nc = 2 * n - 1
    G = np.zeros((nc, n))
    for r in range(nc):
        s = r - (n - 1)
        j = np.arange(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        if np.any(m1):
            G[r, m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        if np.any(m2):
            G[r, m2] -= h[ps[m2]]
    return G
```

For the SLSQP with active constraints, I need to define the objective, gradient, and constraint functions. The key is that these must be top-level functions (no closures). I can pass the active indices via the `args` parameter.

```python
def _slsqp_obj(z, n):
    return z[n]

def _slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _slsqp_eq_f(z, n):
    return z[:n].sum() - n * 0.5

def _slsqp_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g

def _slsqp_ineq_f(z, n, active_idx):
    t = z[n]
    hz = z[:n]
    cc = _corr(hz)
    return t - cc[active_idx]

def _slsqp_ineq_g(z, n, active_idx):
    hz = z[:n]
    K = len(active_idx)
    J = np.zeros((K, n + 1))
    for i, r in enumerate(active_idx):
        s = r - (n - 1)
        g = _grad_lag(hz, s)
        J[i, :n] = -g
        J[i, n] = 1.0
    return J
```

Then the solver:
```python
def _slsqp_solve_active(h_init, n, active_idx, maxiter=200, ftol=1e-14):
    target = n * 0.5
    h = h_init.copy()
    corr = _corr(h)
    t0 = float(np.max(corr)) + 1e-3
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _slsqp_ineq_f, 'jac': _slsqp_ineq_g, 'args': (n, tuple(active_idx))}
    ]
    try:
        res = minimize(_slsqp_obj, z0, jac=_slsqp_obj_jac, args=(n,),
                      bounds=bounds, constraints=cons, method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], target)
        return x, _c5(x)
    except Exception:
        return h, _c5(h)
```

```python
def _slsqp_loop(h_init, n, deadline, K=20, maxiter=150):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    stall = 0
    while time.time() < deadline and stall < 15:
        corr = _corr(h)
        active = np.argsort(-corr)[:K]
        h_new, c5_new = _slsqp_solve_active(h, n, active, maxiter=maxiter)
        h_new = _project(h_new, n * 0.5)
        c5_new = _c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
            if c5_new < _c5(h):
                h = h_new
    return best_h, best_c5
```

For the projected subgradient method:
```python
def _proj_subgrad(h_init, n, deadline, K=10, temp_init=1.0, temp_decay=0.95):
    h = h_init.copy()
    target = n * 0.5
    best_h = h.copy()
    best_c5 = _c5(h)
    temp = temp_init
    step = 0.1
    while time.time() < deadline:
        corr = _corr(h)
        cmax = corr.max()
        # Softmax weights over all lags, or top-K
        w = np.exp((corr - cmax) / temp)
        w /= w.sum()
        # Weighted gradient
        G = _grad_matrix(h)
        g = w @ G  # shape (n,)
        g -= g.mean()  # center to maintain sum constraint
        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(30):
            h_new = _project(h - alpha * g, target)
            c5_new = _c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_h = h.copy()
                best_c5 = c5_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            step = min(step * 1.1, 0.5)
        else:
            step = min(step * 1.05, 0.5)
        temp *= temp_decay
        if temp < 0.01:
            temp = 0.01
    return best_h, best_c5
```

Wait, this computes the full gradient matrix every iteration, which is O(n^2). For n=64, that's 4096 operations per iteration, which is fine.

But the line search also calls `_c5` which is O(n^2) (correlation of two length-n arrays). So each line search step is O(n^2), and we do up to 30 steps, so O(30*n^2) per outer iteration. For n=64, that's about 78K operations per outer iteration. At maybe 1000 outer iterations, that's 78M operations, which should take a few seconds. That's fine.

Actually, let me reconsider. The correlation `np.correlate(h, 1-h, mode='full')` is O(n^2) in the worst case, but numpy implements it efficiently (possibly using FFT for large n). For n=64, it's probably just a direct computation, O(n^2) = O(4096).

The gradient matrix computation is also O(n^2) with the loop.

So the projected subgradient should be fast enough.

def _pair_swap(h_init, n, deadline, rng, max_rounds=100):
    h = h_init.copy()
    target = n * 0.5
    best_h = h.copy()
    best_c5 = _c5(h)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try random pairs
        for trial in range(200):
            if time.time() > deadline:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # Move mass from i to j
            # h[i] -= d, h[j] += d
            # Feasible range: d in [max(0, h[j]-1), min(h[i], 1-h[j])]
            lo_d = max(0.0, h[j] - 1.0)
            hi_d = min(h[i], 1.0 - h[j])
            if hi_d - lo_d < 1e-10:
                continue
            # Also try the reverse: h[i] += d, h[j] -= d
            # d in [max(0, h[i]-1), min(h[j], 1-h[i])]
            # Try several values of d
            for d in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                if lo_d <= d <= hi_d:
                    h_try = h.copy()
                    h_try[i] -= d
                    h_try[j] += d
                    h_try = _project(h_try, target)
                    c5_try = _c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        h = h_try
                        best_h = h.copy()
                        best_c5 = c5_try
                        improved = True
                        break
                # Reverse direction
                lo_d2 = max(0.0, h[i] - 1.0)
                hi_d2 = min(h[j], 1.0 - h[i])
                if lo_d2 <= d <= hi_d2:
                    h_try = h.copy()
                    h_try[i] += d
                    h_try[j] -= d
                    h_try = _project(h_try, target)
                    c5_try = _c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        h = h_try
                        best_h = h.copy()
