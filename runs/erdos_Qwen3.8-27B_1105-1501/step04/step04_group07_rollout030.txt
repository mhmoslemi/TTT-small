def _c5_binary(h):
    """Compute C5 for a binary (or general) step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _scores_full(h):
    """Compute all lag scores (not yet multiplied by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _project_to_binary(h, seed=0):
    """Project h to nearest binary with exactly n/2 ones."""
    n = len(h)
    if n % 2 != 0:
        return h.copy()
    target = n // 2
    # Rank by value
    indices = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[indices[:target]] = 1.0
    return result

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    if n % 2 != 0:
        return h
    target = n // 2
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def _local_search_binary(h, deadline, max_sweeps=10000):
    """Best-improvement pair swap local search for binary h."""
    n = len(h)
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sweeps = 0
    while time.time() < deadline:
        sweeps += 1
        if sweeps > max_sweeps:
            break
        improved = False
        cur_scores = _scores(h)
        cur_max = cur_scores.max()
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_max = _scores(h).max()
                h[i] = 1.0
                h[j] = 0.0
                delta = cur_max - new_max
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is not None and best_delta > 0:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c5

def _simulated_annealing_binary(h, deadline, T0=1.0, T1=0.01, n_iter=100000):
    """Simulated annealing for binary h using pair swaps."""
    n = len(h)
    h = h.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    cur_scores = _scores(h)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iter))
        # Pick random swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            cur_scores = _scores(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5
