    """Find t in [t_lo, t_hi] minimizing max_d (A[d]*t + B[d])"""
    n = len(A)
    order = np.argsort(A)
    A_s = A[order]
    B_s = B[order]
    
    # Build upper envelope
    hull_A = []
    hull_B = []
    hull_x = []
    
    for i in range(n):
        a, b = A_s[i], B_s[i]
        
        if hull_A and a == hull_A[-1]:
            if b > hull_B[-1]:
                hull_B[-1] = b  # same slope, higher intercept wins
            continue
        
        if not hull_A:
            hull_A.append(a)
            hull_B.append(b)
            hull_x.append(-np.inf)
            continue
        
        # Compute intersection with top
        # a*t + b = hull_A[-1]*t + hull_B[-1]
        # t = (hull_B[-1] - b) / (a - hull_A[-1])
        while hull_A:
            x_int = (hull_B[-1] - b) / (a - hull_A[-1])
            if x_int > hull_x[-1]:
                break
            hull_A.pop()
            hull_B.pop()
            hull_x.pop()
        
        if not hull_A:
            hull_A.append(a)
            hull_B.append(b)
            hull_x.append(-np.inf)
        else:
            hull_A.append(a)
            hull_B.append(b)
            hull_x.append(x_int)

    # The upper envelope has increasing slopes
    # Find where slope transitions from ≤ 0 to ≥ 0
    
    m = len(hull_A)
    if m == 0:
        return t_lo
    
    # Find the transition index
    # hull_A[i] is the slope of the i-th segment
    # The segment i is active on [hull_x[i], hull_x[i+1]) (last segment: [hull_x[i], inf))
    
    # Find the first index where slope > 0
    # The minimum is at the start of that segment (or at the end of the previous, same thing)
    
    # If all slopes ≤ 0: min at t_hi
    if hull_A[-1] <= 0:
        return t_hi
    # If all slopes > 0: min at t_lo
    if hull_A[0] > 0:
        return t_lo
    
    # Find transition: last index with slope ≤ 0
    for i in range(m - 1):
        if hull_A[i] <= 0 and hull_A[i+1] > 0:
            x_min = hull_x[i+1]  # where the positive-slope segment starts

            # Return the start of this segment (or t_lo if the segment starts before t_lo)
            x_min = max(t_lo, hull_x[i])
            return x_min
    
    # Fallback (shouldn't reach here)
    return (t_lo + t_hi) / 2

2. **Coordinate descent phase**: 
   - For each sweep:

       - Update h[j]

3. **Pair swap phase**:

     - Find optimal t using min_max_1d
     - Update h[j] -= t, h[k] += t
