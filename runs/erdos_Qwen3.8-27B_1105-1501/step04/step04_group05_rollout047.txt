def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _max_corr(h):
    """Return max of raw correlation (without dx scaling)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm.astype(np.float64)

    positions = rng.permutation(n)
    mask[positions[:n//2]] = 1.0
    return mask

def _optimal_blur(mask):
    """Find best blur parameter d for a binary mask using golden section."""
    n = len(mask)
    def f(d):
        h = _blur(mask, d)
        return _max_corr(h)
    # Coarse grid
    ds = np.linspace(0.0, 0.5, 50)
    vals = [f(d) for d in ds]
    idx = int(np.argmin(vals))
    a = ds[max(0, idx-1)]
    b = ds[min(len(ds)-1, idx+1)]
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    dd = a + gr * (b - a)
    fc = f(c)
    fd = f(dd)
    for _ in range(60):
        if fc < fd:
            b = dd; dd = c; fd = fc; c = b - gr*(b-a); fc = f(c)
        else:
            a = c; c = dd; fc = fd; dd = a + gr*(b-a); fd = f(dd)
    best_d = (a + b) / 2.0
    return best_d, f(best_d)
