def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _raw_scores(h):
    """Compute raw correlation sums (without dx factor)."""
    return np.correlate(h, 1.0 - h, mode='full')

    A, B, C = _abc_matrices(h, V)
    n_lags = len(A)
    
    candidates = [alpha_lo, alpha_hi]
    
    # Stationary points of individual quadratics
    for r in range(n_lags):
        if C[r] < -1e-15:
            a_star = -B[r] / (2.0 * C[r])
            if alpha_lo <= a_star <= alpha_hi:
                candidates.append(a_star)
    
    # Pairwise crossings
    for i in range(n_lags):
        for j in range(i + 1, n_lags):
            a2 = C[i] - C[j]
            b1 = B[i] - B[j]
            c0 = A[i] - A[j]
            if abs(a2) < 1e-18:
                if abs(b1) > 1e-18:
                    a_cross = -c0 / b1
                    if alpha_lo <= a_cross <= alpha_hi:
                        candidates.append(a_cross)
            else:
                disc = b1 * b1 - 4.0 * a2 * c0
                if disc >= 0:
                    sq = np.sqrt(disc)
                    a1 = (-b1 + sq) / (2.0 * a2)
                    a2r = (-b1 - sq) / (2.0 * a2)
                    if alpha_lo <= a1 <= alpha_hi:
                        candidates.append(a1)
                    if alpha_lo <= a2r <= alpha_hi:
                        candidates.append(a2r)
    
    candidates = np.array(candidates)
    # Evaluate at all candidates
    vals = A[None, :] + B[None, :] * candidates[:, None] + C[None, :] * candidates[:, None]**2
    max_vals = vals.max(axis=1)
    best_idx = int(np.argmin(max_vals))
    best_alpha = float(candidates[best_idx])
    best_val = float(max_vals[best_idx])
    
    return best_alpha, best_val  # return raw value (without dx)

def _exact_1d_search_fast(h, V, alpha_lo, alpha_hi):
    A, B, C = _abc_matrices(h, V)
    n_lags = len(A)
    
    # All pairwise differences (upper triangle)
    # Use broadcasting
    C_diff = C[:, None] - C[None, :]  # n_lags x n_lags
    B_diff = B[:, None] - B[None, :]
    A_diff = A[:, None] - A[None, :]
    
    # disc = B_diff^2 - 4 * C_diff * A_diff
    disc = B_diff**2 - 4.0 * C_diff * A_diff
    
    # Only look at upper triangle where C_diff != 0
    mask = np.triu(np.ones((n_lags, n_lags), dtype=bool), k=1)
    mask &= (np.abs(C_diff) > 1e-18)
    disc[mask] = np.where(disc[mask] >= 0, disc[mask], -1)
    
    # For C_diff == 0 (linear case), handled separately
    linear_mask = np.triu(np.ones((n_lags, n_lags), dtype=bool), k=1)
    linear_mask &= (np.abs(C_diff) <= 1e-18)
    linear_mask &= (np.abs(B_diff) > 1e-18)
    
    candidates = [alpha_lo, alpha_hi]
    
    # Quadratic crossings
    valid = mask & (disc >= 0)
    if valid.any():
        sq = np.sqrt(disc[valid])
        c_diff_v = C_diff[valid]
        b_diff_v = B_diff[valid]
        a1 = (-b_diff_v + sq) / (2.0 * c_diff_v)
        a2 = (-b_diff_v - sq) / (2.0 * c_diff_v)
        in_range = (a1 >= alpha_lo) & (a1 <= alpha_hi)
        candidates.extend(a1[in_range].tolist())
        in_range2 = (a2 >= alpha_lo) & (a2 <= alpha_hi)
        candidates.extend(a2[in_range2].tolist())
    
    # Linear crossings
    if linear_mask.any():
        a_cross = -A_diff[linear_mask] / B_diff[linear_mask]
        in_range = (a_cross >= alpha_lo) & (a_cross <= alpha_hi)
        candidates.extend(a_cross[in_range].tolist())
    
    # Stationary points
    for r in range(n_lags):
        if C[r] < -1e-15:
            a_star = -B[r] / (2.0 * C[r])
            if alpha_lo <= a_star <= alpha_hi:
                candidates.append(a_star)
    
    if len(candidates) < 2:
        candidates = [alpha_lo, alpha_hi]
    
    candidates = np.array(candidates)
    # Vectorized evaluation
    vals = A[None, :] + B[None, :] * candidates[:, None] + C[None, :] * candidates[:, None]**2
    max_vals = vals.max(axis=1)
    best_idx = int(np.argmin(max_vals))
    
    return float(candidates[best_idx]), float(max_vals[best_idx])

import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_raw_scores(h)) * (2.0 / n))

def _grad_matrix_rows(h, lags):
    """Compute G[lags] where G[r, i] = d(score_r)/d(h_i)"""
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((len(lags), n))
    h_1m = 1.0 - h
    for ii, r in enumerate(lags):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[ii, d:] += h_1m[:m]
                G[ii, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[ii, :m] += h_1m[g:]
                G[ii, g:] += -h[:m]
    return G

def _exact_1d(h, V, alpha_lo, alpha_hi):
    """Exact 1D minimization of max_r (A_r + B_r*a + C_r*a^2) * dx"""
    n = len(h)
    dx = 2.0 / n
    A = _raw_scores(h)
    B = np.correlate(V, 1.0 - h, mode='full') - np.correlate(h, V, mode='full')
    C = -np.correlate(V, V, mode='full')
    n_lags = len(A)
    
    candidates = [alpha_lo, alpha_hi]
    
    # Stationary points
    for r in range(n_lags):
        if C[r] < -1e-15:
            a_star = -B[r] / (2.0 * C[r])
            if alpha_lo <= a_star <= alpha_hi:
                candidates.append(a_star)
    
    # Pairwise crossings (vectorized)
    C_diff = C[:, None] - C[None, :]
    B_diff = B[:, None] - B[None, :]
    A_diff = A[:, None] - A[None, :]
    triu_mask = np.triu(np.ones((n_lags, n_lags), dtype=bool), k=1)
    
    quad_mask = triu_mask & (np.abs(C_diff) > 1e-18)
    disc = B_diff[quad_mask]**2 - 4.0 * C_diff[quad_mask] * A_diff[quad_mask]
    valid = disc >= 0
    if valid.any():
        sq = np.sqrt(disc[valid])
        cd = C_diff[quad_mask][valid]
        bd = B_diff[quad_mask][valid]
        a1 = (-bd + sq) / (2.0 * cd)
        a2 = (-bd - sq) / (2.0 * cd)
        m1 = (a1 >= alpha_lo) & (a1 <= alpha_hi)
        m2 = (a2 >= alpha_lo) & (a2 <= alpha_hi)
        if m1.any():
            candidates.extend(a1[m1].tolist())
        if m2.any():
            candidates.extend(a2[m2].tolist())
    
    lin_mask = triu_mask & (np.abs(C_diff) <= 1e-18) & (np.abs(B_diff) > 1e-18)
    if lin_mask.any():
        ac = -A_diff[lin_mask] / B_diff[lin_mask]
        m = (ac >= alpha_lo) & (ac <= alpha_hi)
        if m.any():
            candidates.extend(ac[m].tolist())
    
    candidates = np.array(candidates)
    vals = A[None, :] + B[None, :] * candidates[:, None] + C[None, :] * candidates[:, None]**2
    max_vals = vals.max(axis=1)
    best_idx = int(np.argmin(max_vals))
    return float(candidates[best_idx]), float(max_vals[best_idx] * dx)

def _lp_direction(h, active_lags, bound=1.0):
    """Find direction V minimizing max rate of change on active lags."""
    n = len(h)
    G = _grad_matrix_rows(h, active_lags)
    n_active = len(active_lags)
    # LP: min t s.t. G@V <= t, sum(V)=0, -bound <= V <= bound
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_active)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-bound, bound)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, 0.0

def _subgrad_direction(h, active_lags):
    """Negative mean subgradient of active lags."""
    n = len(h)
    G = _grad_matrix_rows(h, active_lags)
    V = -G.mean(axis=0)
    V -= V.mean()  # make zero-sum
    norm = np.linalg.norm(V)
    if norm < 1e-15:
        return None
    return V / norm

def _optimize(h, deadline, n_max_iter=50000):

            stall += 1
            continue

        alpha_opt, new_c5 = _exact_1d(h, V, alpha_lo, alpha_hi)
        
        if new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha_opt * V
            h = _project(h_new, target)
            # Recompute after projection (projection might change the value slightly)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            # Try a random pair swap as escape
            if stall % 20 == 0 and stall < 200:
                ones = np.where(h > 0.01)[0]
                zeros = np.where(h < 0.99)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = int(np.random.default_rng(it).integers(len(ones)))
                    j = int(np.random.default_rng(it+1).integers(len(zeros)))
                    i, j = ones[i], zeros[j]
                    # Try small transfer
                    for amt in [0.01, 0.02, 0.05]:
                        h_try = h.copy()
                        h_try[i] = np.clip(h_try[i] - amt, 0, 1)
                        h_try[j] = np.clip(h_try[j] + amt, 0, 1)
                        h_try = _project(h_try, target)
                        c5_try = _exact_c5(h_try)
                        if c5_try < cur_c5 - 1e-12:
                            h = h_try
                            cur_c5 = c5_try
                            stall = 0
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _exact_c5(h):
    n = len(h)
    return float(np.max(_raw_scores(h)) * (2.0 / n))

def _grad_rows(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    h1 = 1.0 - h
    for ii, r in enumerate(lags):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[ii, d:] += h1[:m]
                G[ii, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[ii, :m] += h1[g:]
                G[ii, g:] += -h[:m]
    return G

def _exact_1d(h, V, alpha_lo, alpha_hi, A=None):
    n = len(h)
    dx = 2.0 / n
    if A is None:
        A = _raw_scores(h)
    h1 = 1.0 - h

    C = -np.correlate(V, V, mode='full')
    n_lags = len(A)
    candidates = [alpha_lo, alpha_hi]
    for r in range(n_lags):

            a_star = -B[r] / (2.0 * C[r])

                candidates.append(float(a_star))
    if n_lags <= 300:

        Bd = B[:, None] - B[None, :]
        Ad = A[:, None] - A[None, :]
        tri = np.triu(np.ones((n_lags, n_lags), dtype=bool), k=1)
        qm = tri & (np.abs(Cd) > 1e-18)

            disc = Bd[qm]**2 - 4.0 * Cd[qm] * Ad[qm]
            valid = disc >= 0

                cd = Cd[qm][valid]
                bd = Bd[qm][valid]
                a1 = (-bd + sq) / (2.0 * cd)
                a2 = (-bd - sq) / (2.0 * cd)
                m1 = (a1 >= alpha_lo) & (a1 <= alpha_hi)
                m2 = (a2 >= alpha_lo) & (a2 <= alpha_hi)
                if m1.any():

                    candidates.extend(a2[m2].tolist())
        lm = tri & (np.abs(Cd) <= 1e-18) & (np.abs(Bd) > 1e-18)

    candidates = np.array(candidates)

    max_vals = vals.max(axis=1)

def _lp_dir(h, lags, bound=1.0):
    n = len(h)
    G = _grad_rows(h, lags)
    na = len(lags)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((na, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(na)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bnds = [(-bound, bound)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bnds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, 0.0

def _subgrad_dir(h, lags):
    n = len(h)
    G = _grad_rows(h, lags)
    V = -G.mean(axis=0)
    V -= V.mean()
    norm = np.linalg.norm(V)
    if norm < 1e-15:
        return None
    return V / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 4:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

        h_try, c5_try = _optimize(h_pert, sub_dl, max_iter=5000, active_frac=0.99)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.04)

    # Phase 5: Final optimization + polish (10%)
    dl5 = t0 + budget_s * 0.94
    hh, cc = _optimize(best_h, dl5, max_iter=20000, active_frac=0.998)
    if cc < best_c5:
        best_c5 = cc
        best_h = hh.copy()

    if time.time() < t0 + budget_s * 0.99:
        hp, cp = _pair_polish(best_h, t0 + budget_s * 0.995, max_passes=2)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()

    # Try self-dual transform
    n = best_n
    h_sd = (best_h + 1.0 - best_h[::-1]) / 2.0
    h_sd = _project(h_sd, n / 2.0)
    c5_sd = _exact_c5(h_sd)
    if c5_sd < best_c5:
        best_c5 = c5_sd
        best_h = h_sd.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
