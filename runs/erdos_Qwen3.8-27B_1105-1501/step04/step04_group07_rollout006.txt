        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    center = n - 1
    for d in range(-(n - 1), n):
        r = center + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d+m] += (1.0 - h[:m]) * dx
                G[r, :m] += -h[d:d+m] * dx
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:g+m]) * dx
                G[r, g:g+m] += -h[:m] * dx
    return G

def _line_search(h, d, cur_c5, max_alpha=2.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if d[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / d[j])
        elif d[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-d[j]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Try a grid first
    n_pts = 16
    alphas = np.linspace(0, alpha_max, n_pts)
    vals = np.array([_c5(h + a * d) for a in alphas])
    best_i = int(np.argmin(vals))
    
    if best_i == 0:
        return 0.0, cur_c5
    if best_i == n_pts - 1:
        return alpha_max, float(vals[best_i])
    
    # Refine with golden section
    lo = float(alphas[best_i - 1])
    hi = float(alphas[best_i + 1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5(h + c1 * d)
    f2 = _c5(h + c2 * d)
    for _ in range(20):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5(h + c1 * d)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5(h + c2 * d)
    best_a = (lo + hi) / 2.0
    best_c5 = _c5(h + best_a * d)
    
    # Also check if a slightly larger step works (extrapolation)
    ext_a = best_a * 1.3
    if ext_a <= alpha_max:
        ext_c5 = _c5(h + ext_a * d)
        if ext_c5 < best_c5:
            best_a = ext_a
            best_c5 = ext_c5
    
    return best_a, best_c5

def _lp_minimax(h, deadline, max_iters=5000, radius=0.15):
    """LP-based minimax optimization using all constraints."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    cur_c5 = best_c5
    stall = 0
    r = radius
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        
        scores = _scores(h)
        G = _grad_matrix(h)
        n_lags = 2 * n - 1
        
        # Find active lags (within 5% of max)
        s_max = float(scores.max())
        thresh = s_max * 0.95
        active = np.where(scores >= thresh)[0]
        
        if len(active) == 0:
            break
        
        # LP: min t
        # s.t. G[active, :] @ d <= t - scores[active]  (i.e., G@*d + scores <= t)
        #      |d_j| <= r  (box constraint)
        #      sum(d) = 0
        #      t <= s_max (we want to reduce)
        #      t >= 0
        
        # Variables: [d_0, ..., d_{n-1}, t]
        # Minimize: t  (c = [0, 0, ..., 0, 1])
        # Constraints:
        #   G[active, :] @ d - t <= -scores[active]
        #   d_j <= r
        #   -d_j <= r
        #   sum(d) = 0
        #   t <= s_max (to ensure we're trying to reduce)
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # G[active] @ d - t <= -scores[active]
        A_ub = np.zeros((len(active) + 2*n, n + 1))
        b_ub = np.zeros(len(active) + 2*n)
        
        A_ub[:len(active), :n] = G[active]
        A_ub[:len(active), n] = -1.0
        b_ub[:len(active)] = -scores[active]
        
        # d_j <= r
        A_ub[len(active):len(active)+n, :n] = np.eye(n)
        b_ub[len(active):len(active)+n] = r
        
        # -d_j <= r
        A_ub[len(active)+n:len(active)+2*n, :n] = -np.eye(n)
        b_ub[len(active)+n:len(active)+2*n] = r
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        
        bounds = [(None, None)] * n + [(0, s_max)]
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 0.5})
            if not res.success:
                stall += 1
                r *= 0.8
                if r < 0.01:
                    break
                continue
            
            d = res.x[:n]
            t_opt = float(res.x[-1])
        except Exception:
            stall += 1
            r *= 0.8
            continue
        
        if t_opt >= s_max - 1e-10:
            # Can't reduce further with this radius
            stall += 1
            r *= 0.85
            if stall > 50 and r < 0.02:
                r = 0.1
            continue
        
        if np.max(np.abs(d)) < 1e-12:
            stall += 1
            continue
        
        # Take the step with line search
        alpha, new_c5 = _line_search(h, d, cur_c5, max_alpha=2.0)
        if new_c5 < cur_c5 - 1e-12:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            cur_c5 = _c5(h_new)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_new.copy()
            h = h_new
            stall = 0
            r = min(r * 1.05, 0.4)
        else:
            stall += 1
            r *= 0.9
            if stall > 80 and r < 0.03:
                r = 0.08
    
    return best_h, best_c5

The optimal t will be the minimum possible value of max_r (scores_r + G[r,:]*d) over d in the box. If the LP finds t_opt < s_max, that means there exists a direction d within the box that can reduce the max score. But the actual reduction depends on the step size α.

The issue is that the LP gives the OPTIMAL direction within the box, but we then do a line search in that direction. The line search might not achieve the full reduction predicted by the LP (because of the nonlinear terms).

A better approach: after the LP gives us direction d, we know that for small α:
scores(α) ≈ scores + α * G@*d
So the predicted max at step α is: s_max + α * (G@*d).max() (approximately)

If t_opt < s_max, then (G@*d).max() < 0, which means for small α, the max score decreases. The line search will find the actual optimal α.
