def _project(v, target):
    n = len(v)

    grad = np.zeros(n)

    A_ub[:, n] = -1.0
    b_ub = -np.full(len(top_idx), delta)

            stall = 0
            lr = min(lr * 1.05, 0.02)
        else:
            stall += 1
            lr *= 0.9
            if stall > 30:
                break
        # Coordinate descent after subgradient
        for _ in range(3):
            if time.time() > deadline2:
                break
            h, _ = _coord_descent_pass(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
    
    # Phase 3: LP-based local search (15%)
    deadline3 = t0 + budget_s * 0.70
    h = best_h.copy()
    radius = 0.1
    stall = 0
    while time.time() < deadline3:
        if _remaining() < 15:
            break
        if stall > 40:
            break
        scores = _get_raw_scores(h) * (2.0 / len(h))
        p = _lp_direction(h, scores, radius, top_k=6)
        if p is None:
            stall += 1
            radius *= 0.7
            if radius < 0.005:
                break
            continue
        alpha_max = radius
        for j in range(len(h)):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        alpha, new_c5 = _line_search_exact(h, p, alpha_max)
        if new_c5 < _exact_c5(h) - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            stall = 0
            radius = min(radius * 1.2, 0.3)
            c5 = _exact_c5(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
        else:
            stall += 1
            radius *= 0.65
        # Periodic coordinate descent
        if stall % 10 == 5:
            for _ in range(2):
                if time.time() > deadline3:
                    break
                h, _ = _coord_descent_pass(h, target)
    
    # Phase 4: Pair refinement (15%)
    deadline4 = t0 + budget_s * 0.85
    h = best_h.copy()
    h, c5 = _pair_refine(h, target, deadline4, max_pairs=300)
    if c5 < best_c5 - 1e-13:
        best_c5 = c5
        best_h = h.copy()
    # Follow with coordinate descent
    for _ in range(10):
        if time.time() > deadline4:
            break
        h, improved = _coord_descent_pass(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    
    # Phase 5: Multi-resolution (10%)
    deadline5 = t0 + budget_s * 0.95
    offsets = [-2, -1, 1, 2, -4, 4, -6, 6, -8, 8, 10, -10, 12, -12, 16, -16, 20, -20, 24, -24]
    time_per = budget_s * 0.008
    tried = set()
    for dn in offsets:
        if time.time() > deadline5 or _remaining() < 10:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample_area(best_h, n_try)
        # Quick coordinate descent at this resolution
        sub_dl = min(deadline5, time.time() + time_per * 2)
        for _ in range(30):
            if time.time() > sub_dl:
                break
            h_r, imp = _coord_descent_pass(h_r, n_try / 2.0)
            if not imp:
                break
        c5_r = _exact_c5(h_r)
        if time.time() < deadline5 and _remaining() > 10:
            # Pair refinement
            sub_dl2 = min(deadline5, time.time() + time_per)
            h_r, c5_r = _pair_refine(h_r, n_try / 2.0, sub_dl2, max_pairs=50)
        _try_update(h_r)
    
    # Phase 6: Final polishing (remaining)
    deadline6 = t0 + budget_s * 0.99
    h = best_h.copy()
    for _ in range(100):
        if time.time() > deadline6:
            break
        h, improved = _coord_descent_pass(h, best_n)
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        elif not improved:
            break
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Projection made it worse, try without
        h_alt = np.clip(best_h, 0.0, 1.0)
        h_alt = _project(h_alt, best_n / 2.0)
        final_c5 = _exact_c5(h_alt)
        best_h = h_alt
    
    return best_h, final_c5, best_n
