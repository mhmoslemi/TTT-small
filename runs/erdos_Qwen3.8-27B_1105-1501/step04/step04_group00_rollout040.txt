import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, tau, weight_thresh=1e-8):
    """Compute gradient of F_tau(h) = tau * log(sum_r exp(S_r/tau))"""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    P = len(scores)  # 2n-1
    
    # Stable log-sum-exp
    M = np.max(scores)
    u = np.exp((scores - M) / tau)
    Z = np.sum(u)
    w = u / Z  # soft weights
    
    # Skip rows with negligible weight
    active = w > weight_thresh
    n_active = int(np.sum(active))
    
    grad = np.zeros(n)
    
    for r in np.where(active)[0]:
        d = r - (n - 1)
        wr = w[r] * dx
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] += -wr * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] += -wr * h[:m]
    
    return grad

def _line_search(h, p, max_alpha, n_scan=15):
    """Find best alpha in [0, max_alpha] minimizing C5(h + alpha*p)"""
    alphas = np.linspace(0, max_alpha, n_scan)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    
    if best_i in (0, n_scan - 1):
        return best_a, best_c5
    
    lo = float(alphas[max(0, best_i - 1)])
    hi = float(alphas[min(n_scan - 1, best_i + 1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    
    for _ in range(15):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_grad(h, tau, thresh=1e-10):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    M = float(np.max(scores))
    u = np.exp((scores - M) / tau)
    Z = float(np.sum(u))
    w = u / Z
    
    grad = np.zeros(n)
    active_idx = np.where(w > thresh)[0]
    
    for r in active_idx:
        d = int(r) - (n - 1)
        wr = float(w[r]) * dx
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] += -wr * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] += -wr * h[:m]
    
    return grad

def _max_step(h, p):
    """Maximum alpha such that h + alpha*p stays in [0,1]"""
    alpha_max = 1.0  # default upper bound
    for j in range(len(h)):
        if p[j] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-15:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    return max(alpha_max, 1e-12)

def _line_search(h, p, max_alpha, n_scan=13):
    if max_alpha < 1e-14:
        return 0.0, _exact_c5(h)
    alphas = np.linspace(0, max_alpha, n_scan)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    
    if best_i in (0, n_scan - 1):
        return best_a, best_c5
    
    lo = float(alphas[best_i - 1])
    hi = float(alphas[min(n_scan - 1, best_i + 1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    
    for _ in range(12):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_grad(h, tau, thresh=1e-10):
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    M = float(np.max(scores))
    u = np.exp((scores - M) / tau)
    Z = float(np.sum(u))
    w = u / Z
    grad = np.zeros(n)
    active_idx = np.where(w > thresh)[0]
    for r in active_idx:
        d = int(r) - (n - 1)
        wr = float(w[r]) * dx
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] += -wr * h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] += -wr * h[:m]
    return grad

def _max_step(h, p):
    alpha_max = 1.0
    pos = p > 1e-15
    neg = p < -1e-15
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-p[neg]))))
    return max(alpha_max, 1e-12)

def _line_search(h, p, max_alpha, n_scan=13):
    if max_alpha < 1e-14:
        return 0.0, _exact_c5(h)
    alphas = np.linspace(0, max_alpha, n_scan)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i in (0, n_scan - 1):
        return best_a, best_c5
    lo = float(alphas[best_i - 1])
    hi = float(alphas[min(n_scan - 1, best_i + 1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(12):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _smooth_optimize(h, deadline, taus, mom=0.5, trust_init=0.3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    momentum = np.zeros(n)
    trust = trust_init

    for tau in taus:
        if time.time() > deadline:
            break
        tau_stall = 0
        max_it = max(40, int(6000 * tau))
        for it in range(max_it):
            if time.time() > deadline:
                break
            if tau_stall > 25:
                break
            grad = _smooth_grad(h, tau)
            grad -= grad.mean()
            gn = float(np.max(np.abs(grad)))
            if gn < 1e-15:
                tau_stall += 10
                continue
            p = -grad / gn
            p = 0.5 * p + mom * momentum
            pn = float(np.max(np.abs(p)))
            if pn < 1e-15:
                tau_stall += 5
                continue
            p /= pn
            p -= p.mean()
            max_a = _max_step(h, p)
            a_max = min(max_a * 0.99, trust)
            alpha, new_c5 = _line_search(h, p, a_max)
            if new_c5 < cur_c5 - 1e-13:
                h = _project(h + alpha * p, target)
                cur_c5 = _exact_c5(h)
                momentum = 0.7 * momentum + 0.3 * p
                tau_stall = 0
                trust = min(trust * 1.03, 0.8)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                tau_stall += 1
                trust *= 0.85
                if trust < 0.005:
                    break
    return best_h, best_c5

def _gp(newv, i, j, a, approx=True):
    N = len(newv)
    p = i - j
    if p == 0:
        return 0.0
    if p > 0:
        m = N - p
        if m <= 0:
            return 0.0
        dp = newv[:m] - newv[p:]
        ok = (dp > 1e-12) | (dp < -1e-12)
        if np.any(ok):
            v = dp[ok]
            c = (np.abs(v) - np.abs(v - a)) / v
            return c.sum()
        return 0.0
    else:
        p2 = -p
        m = N - p2
        if m <= 0:
            return 0.0
        dode = dp2 = newv[p2:] - newv[:m]
        ok = (dp2 > 1e-12) | (dp2 < -1e-12)
        if np.any(ok):
            v = dp2[ok]
            c = (np.abs(v) - np.abs(v + a)) / v
            return c.sum()
        dode2 = newv[:m] - newv[p2:]
        ok2 = (dode2 > 1e-12) | (dode2 < -1e-12)
        if np.any(ok2):
            v2 = dode2[ok2]
            c2 = (np.abs(v2) - np.abs(v2 + a)) / v2
            return c2.sum()
        return 0.0

def _gp_full(h_orig, i, j, a):
    newv = h_orig.copy()

def _pair_line(newv, i, j, a_max, iters=10):
    c5_0 = _exact_c5(newv)
    lo, hi = 0.0, a_max
    f_lo = c5_0
    for _ in range(iters):
        if hi - lo < 1e-15:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = newv.copy(); v1[i] += m1; v1[j] -= m1
        v2 = newv.copy(); v2[i] += m2; v2[j] -= m2
        f1 = _exact_c5(v1)
        f2 = _exact_c5(v2)
        if f1 < f2:
            hi, f_lo = m2, f1
        else:
            lo = m1
    a_opt = (lo + hi) / 2.0
    v_opt = newv.copy(); v_opt[i] += a_opt; v_opt[j] -= a_opt
    f_opt = _exact_c5(v_opt)
    if f_opt < f_lo:
        return a_opt, f_opt, v_opt
    best_a = 0.0 if f_lo <= _exact_c5(newv) else (lo + hi) / 2.0
    return best_a, f_min if (f_min := f_lo) < c5_0 else c5_0, v_opt if f_lo < c5_0 else newv

def _coord_polish(h, deadline, max_passes=15):

            gi = grad[i]

                sign = -1.0

                sign = 1.0

                newv = h.copy()

                c5_try = _exact_c5(v_try)
