def _scores_arr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_eval(h, V, a):
    """Evaluate C5 at h + a*V"""
    return _exact_c5(h + a * V)

def _golden_search(h, V, a_lo, a_hi, f_lo, f_hi, n_iter=20):
    """Golden section search on [a_lo, a_hi] for f(a) = C5(h + a*V)"""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_a = a_lo if f_lo < f_hi else a_hi
    best_f = min(f_lo, f_hi)
    for _ in range(n_iter):
        if a_hi - a_lo < 1e-10:
            break
        a1 = a_hi - gr * (a_hi - a_lo)
        a2 = a_lo + gr * (a_hi - a_lo)
        f1 = _c5_eval(h, V, a1)
        f2 = _c5_eval(h, V, a2)
        if f1 < f2:
            a_hi = a2
            f_hi = f1
            if f1 < best_f:
                best_f = f1
                best_a = a1
        else:
            a_lo = a1
            f_lo = f2
            if f2 < best_f:
                best_f = f2
                best_a = a2
    return best_a, best_f

def _pnorm_direction(h, scores, p):
    """Compute P-norm descent direction."""
    n = len(h)
    n_lags = 2 * n - 1
    s_max = float(scores.max())
    if s_max < 1e-15:
        return np.zeros(n)
    ratio = scores / s_max
    w = ratio ** p
    sw = float(w.sum())
    if sw < 1e-300:
        w = np.zeros(n_lags)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= sw
    # Gradient of score r w.r.t. h
    G = _build_grad(h)
    grad = G.T @ w
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return np.zeros(n)
    return -grad / gn

def _build_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores_arr(h)
        d = _pnorm_direction(h, scores, p)
        if np.linalg.norm(d) < 1e-14:
            stall += 1
            if stall > 50:
                break
            continue
        found = False
        for scale in [0.15, 0.08, 0.04, 0.02]:
            h_try = h + scale * d
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < cur_c5 - 1e-13:
                h = h_try
                cur_c5 = c5_try
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 80:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_refine(h, deadline, top_k=None, radius=0.1, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    n_lags = 2 * n - 1
    r = radius
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        scores = _scores_arr(h)
        G = _build_grad(h)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = scores[idx]
            Gk = G[idx]
        else:
            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(sk), n + 1))
        A_ub[:, :n] = Gk
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                          b_eq=np.zeros(1), bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-7:
                break
            continue
        # Line search along p
        best_a = 0.0
        best_f = cur_c5
        for a_s in np.linspace(0, 1.0, 12):
            h_try = _project(h + a_s * p, target)
            f_s = _exact_c5(h_try)
            if f_s < best_f:
                best_f = f_s
                best_a = a_s
        if 0 < best_a < 1.0:
            lo = max(0, best_a - 1.0 / 12.0)
            hi = min(1.0, best_a + 1.0 / 12.0)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(12):
                if hi - lo < 1e-9:
                    break
                a1 = hi - gr * (hi - lo)
                a2 = lo + gr * (hi - lo)
                h1 = _project(h + a1 * p, target)
                h2 = _project(h + a2 * p, target)
                f1 = _exact_c5(h1)
                f2 = _exact_c5(h2)
                if f1 < f2:
                    hi = a2
                else:
                    lo = a1
                mid = (lo + hi) / 2.0
                hm = _project(h + mid * p, target)
                fm = _exact_c5(hm)
                if fm < best_f:
                    best_f = fm
                    best_a = mid
        if best_f < cur_c5 - 1e-12:
            h = _project(h + best_a * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            r = min(r * 1.05, 0.5)
        else:
            stall += 1
            r *= 0.8
            if stall > 25 and r < 0.03:
                r = 0.08
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_rand(n, seed):
    rng = np.random.default_rng(seed)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _binary_local(h, deadline, max_iters=20000):
    n = len(h)
    if n % 2 != 0:
        return h, _exact_c5(h)
    target = n // 2
    h = (h >= 0.5).astype(float)
    ones_count = int(h.sum())
    if ones_count != target:
        if ones_count > target:
            idx = np.where(h == 1)[0]
            np.random.seed(0)
            np.random.shuffle(idx)
            h[idx[target:]] = 0
        else:
            idx = np.where(h == 0)[0]
            np.random.seed(0)
            np.random.shuffle(idx)
            h[idx[:target - ones_count]] = 1
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(7)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 4000:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    return best_h, best_c5

def _intensity_transform(h, t):
    """Apply intensity transform: h -> clip(t*(h - 0.5) + 0.5, 0, 1) then project"""
    n = len(h)
    h_new = t * (h - 0.5) + 0.5
    h_new = np.clip(h_new, 0.0, 1.0)
    return _project(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    
    # Phase 1: Quick LP + P-norm (12%)
    dl1 = t0 + budget_s * 0.06
    hdr, c5r = _lp_refine(h, dl1, top_k=15, radius=0.15, max_iters=1000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    dl1b = t0 + budget_s * 0.12
    hdr, c5r = _pnorm_descent(best_h, dl1b, p_start=6, p_end=512, n_steps=300)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 2: Intensive pair search (35%)
    dl2 = t0 + budget_s * 0.47
    hdr, c5r = _pair_search(best_h, dl2, max_rounds=60)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 3: Coordinate descent (10%)
    dl3 = t0 + budget_s * 0.57
    hdr, c5r = _coord_descent(best_h, dl3, max_passes=6)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 4: P-norm + LP refinement (13%)
    dl4 = t0 + budget_s * 0.64
    hdr, c5r = _pnorm_descent(best_h, dl4, p_start=32, p_end=8192, n_steps=400)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    dl4b = t0 + budget_s * 0.70
    hdr, c5r = _lp_refine(best_h, dl4b, top_k=8, radius=0.05, max_iters=2000)
    if c5r < best_c5:
        best_c5 = c5r
        best_h = hdr.copy()
    
    # Phase 5: Multi-start with diverse perturbations (18%)
    dl5 = t0 + budget_s * 0.88
    for attempt in range(200):
        if time.time() > dl5:
            break
        if best_c5 < 0.3805:
            break
        # Different perturbation strategies
        strategy = attempt % 4
        if strategy == 0:
            scale = 0.02 + 0.03 * (attempt % 5) / 5.0
            h_pert = best_h + rng.normal(0, scale, best_n)
        elif strategy == 1:
            t = 0.3 + 0.7 * rng.random()
            h_pert = _intensity_transform(best_h, t)
        elif strategy == 2:
            h_pert = (best_h + 1.0 / best_n) % 1.0
            h_pert = _project(h_pert, best_n / 2.0)
        else:
            h_bin = (best_h >= 0.5).astype(float)
            flip = best_n // 16
            idx = rng.choice(best_n, size=flip, replace=False)
            h_bin[idx] = 1.0 - h_bin[idx]
            ones = int(h_bin.sum())
            if ones > best_n // 2:
                zi = np.where(h_bin == 1)[0]
                rng.shuffle(zi)
                h_bin[zi[best_n // 2:]] = 0
            elif ones < best_n // 2:
                zi = np.where(h_bin == 0)[0]
                rng.shuffle(zi)
                h_bin[zi[:best_n // 2 - ones]] = 1
            h_pert = h_bin.astype(float)
        
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl5, time.time() + budget_s * 0.012)
        h_try, c5_try = _pair_search(h_pert, sub_dl, max_rounds=4)
        if c5_try > best_c5:
            # Try LP if pair search didn't help
            sub_dl2 = min(dl5, time.time() + budget_s * 0.008)
            h_try, c5_try = _lp_refine(h_try, sub_dl2, top_k=10, radius=0.06, max_iters=200)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 6: Resolution sweep (8%)
    dl6 = t0 + budget_s * 0.96
    for dn in [1, -1, 2, -2, 3, -3, 4, -4, 6, -6, 8, 12, 16]:
        if time.time() > dl6:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 160:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl6, time.time() + budget_s * 0.010)
        h_r, c5_r = _pair_search(h_r, sub_dl, max_rounds=3)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 7: Final polish (4%)
    if time.time() < t0 + budget_s * 0.98:
        dl7 = t0 + budget_s * 0.995
        hdr, c5r = _pair_search(best_h, dl7, max_rounds=5)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = hdr.copy()
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
