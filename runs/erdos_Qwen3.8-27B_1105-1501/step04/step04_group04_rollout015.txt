import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        G[r, j] += dx * (1.0 - h[i])
        G[r, i] += -dx * h[j]
    return G

def centered(v):
    """Make v zero-sum by subtracting mean, then normalize."""
    v = v - np.mean(v)
    norm = np.max(np.abs(v))
    if norm < 1e-14:
        return None
    return v / norm

    neg_mask = direction < -1e-14
    
    t_min = 0.0

            break

        c5_t = exact_c5(h_t)
        evals.append((t, c5_t))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_t = t
    
    if best_c5 >= cur_c5 - 1e-12:
        return 0.0, cur_c5, h.copy()
    
    # Refine around best

    sorted_evals = sorted(evals, key=lambda x: x[1])
    for t_c, c_c in sorted_evals[:3]:
        if time.monotonic() > deadline:
            break
        # Local refinement via golden section
        t_lo = max(t_min, t_c - (t_max - t_min) / n_grid * 1.5)

            c2 = exact_c5(h + m2 * direction)
            if c1 < c2:
                t_hi = m2
            else:
                t_lo = m1
            if min(c1, c2) < best_c5:
                best_c5 = min(c1, c2)
                best_t = (t_lo + t_hi) / 2

def lse_weights(scores, beta, top_k=None):
    """Compute LSE (log-sum-exp) weights over top-k active constraints."""
    if top_k is not None and len(scores) > top_k:
        idx = np.argsort(scores)[::-1][:top_k]
        scores = scores[idx]
    max_s = np.max(scores)
    exp_v = np.exp(beta * (scores - max_s))
    w = exp_v / np.sum(exp_v)
    return w

def generate_directions(h, G, scores, beta, rng, n_pairs=16):
    """Generate a pool of candidate directions."""
    n = len(h)
    L = 2 * n - 1
    directions = []
    
    # 1. Weighted gradient direction
    top_k = max(8, min(32, int(0.15 * n) + 4))
    w = lse_weights(scores, beta, top_k)
    v = -w @ G  # negative weighted gradient
    v_c = centered(v)
    if v_c is not None:
        directions.append(v_c)
    
    # 2. Individual top active constraint gradients
    idx_top = np.argsort(scores)[::-1][:min(8, L)]
    for idx in idx_top:
        v = -G[idx]
        v_c = centered(v)
        if v_c is not None:
            directions.append(v_c)
    
    # 3. Pair exchange directions
    # Find pairs where one has high h and other has low h, weighted by sensitivity
    # Use gradient magnitudes to find sensitive coordinates
    g_mag = np.sum(np.abs(G), axis=0)
    # Find high-sensitivity high-h and high-sensitivity low-h coordinates
    high_h_mask = h > 0.5
    low_h_mask = h < 0.5
    high_h_idx = np.where(high_h_mask & (g_mag > np.median(g_mag)))[0]
    low_h_idx = np.where(low_h_mask & (g_mag > np.median(g_mag)))[0]
    
    if len(high_h_idx) > 0 and len(low_h_idx) > 0:
        # Random pairs
        for _ in range(n_pairs):
            i = high_h_idx[rng.integers(len(high_h_idx))]
            j = low_h_idx[rng.integers(len(low_h_idx))]
            if i != j:
                v = np.zeros(n)
                v[i] = 1.0
                v[j] = -1.0
                directions.append(v)
    
    # 4. Random zero-sum directions
    for _ in range(4):
        v = rng.standard_normal(n)
        v_c = centered(v)
        if v_c is not None:
            directions.append(v_c)
    
    # 5. Sinusoidal patterns
    x = (np.arange(n) + 0.5) / n
    for freq in [1, 2, 3]:
        for phase in [0, np.pi/2]:
            v = np.sin(2 * np.pi * freq * x + phase)
            v_c = centered(v)
            if v_c is not None:
                directions.append(v_c)
    
    return directions

def resample(h, n_new):
    n = len(h)
    if n_new == n:

    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)

Let me also think about the structure of optimal solutions. The Erdős overlap problem has some known structure. For C₁ (one shift), the optimal is a 50/50 split. For C₅ (five shifts), the optimal should have some periodic structure.

Looking at the problem: we want to minimize max over all shifts k of ∫ h(x)(1-h(x+k)) dx. This means we want h to be "incompatible" with all its shifts simultaneously. A function that is 50% on average should achieve this optimally if it has the right frequency content.

def compute_quadratic_coefficients(h, v):
    """
    For h + t*v, compute coefficients a, b, c such that
    F_r(t) = a[r] + b[r]*t + c[r]*t^2 for each lag r.
    F_r = dx * sum_i h[i] * (1-h[i+d]) where d = r-(n-1)
    """
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    a = np.zeros(L)
    b = np.zeros(L)
    c = np.zeros(L)
    
    for d in range(-(n-1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        a[r] = dx * np.sum(h[i] * (1.0 - h[j]))
        b[r] = dx * np.sum(v[i] * (1.0 - h[j]) - h[i] * v[j])
        c[r] = dx * np.sum(-v[i] * v[j])
    
    return a, b, c

def c5_from_quadratic(a, b, c, t):
    vals = a + b * t + c * t * t
    return float(np.max(vals))

import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(300):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        G[r, j] += dx * (1.0 - h[i])
        G[r, i] += -dx * h[j]
    return G

def centered(v):
    v = v - np.mean(v)
    norm = np.max(np.abs(v))
    if norm < 1e-14:
        return None
    return v / norm

def compute_quad_coeffs(h, v):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    a = np.zeros(L)
    b = np.zeros(L)
    c = np.zeros(L)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        a[r] = dx * np.sum(h[i] * (1.0 - h[j]))
        b[r] = dx * np.sum(v[i] * (1.0 - h[j]) - h[i] * v[j])
        c[r] = dx * np.sum(-v[i] * v[j])
    return a, b, c

def c5_from_quad(a, b, c, t):
    return float(np.max(a + b * t + c * t * t))

def feasible_interval(h, v):
    pos = v > 1e-14
    neg = v < -1e-14
    t_max = 2.0
    t_min = -2.0
    if np.any(pos):
        t_max = min(t_max, np.min((1.0 - h[pos]) / v[pos]))
    if np.any(neg):
        t_min = max(t_min, np.max(-h[neg] / v[neg]))
    if t_max - t_min < 1e-12:
        return None
    return t_min, t_max

def line_minimize_fast(h, v, deadline, n_grid=30):
    fi = feasible_interval(h, v)
    if fi is None:
        return 0.0, exact_c5(h), h
    t_min, t_max = fi
    a, b, c = compute_quad_coeffs(h, v)
    
    best_c5 = c5_from_quad(a, b, c, 0.0)
    best_t = 0.0
    
    ts = np.linspace(t_min, t_max, n_grid)
    for t in ts:
        if time.monotonic() > deadline:
            break
        cv = c5_from_quad(a, b, c, t)
        if cv < best_c5:
            best_c5 = cv
            best_t = t
    
    if best_c5 >= c5_from_quad(a, b, c, 0.0) - 1e-13:
        return 0.0, c5_from_quad(a, b, c, 0.0), h
    
    # Refine: find best neighborhood from grid
    vals = a + b * ts + c * ts * ts
    maxes = np.max(vals, axis=0) if vals.ndim > 1 else np.zeros(n_grid)
    # Recompute properly
    maxes = np.array([c5_from_quad(a, b, c, t) for t in ts])
    sorted_idx = np.argsort(maxes)
    
    for si in sorted_idx[:4]:
        if time.monotonic() > deadline:
            break
        t_c = ts[si]
        step = (t_max - t_min) / n_grid
        t_lo = max(t_min, t_c - step * 2)
        t_hi = min(t_max, t_c + step * 2)
        
        for _ in range(30):
            if time.monotonic() > deadline:
                break
            if t_hi - t_lo < 1e-10:
                break
            m1 = t_lo + 0.381966 * (t_hi - t_lo)
            m2 = t_hi - 0.381966 * (t_hi - t_lo)
            c1 = c5_from_quad(a, b, c, m1)
            c2 = c5_from_quad(a, b, c, m2)
            if c1 < c2:
                t_hi = m2
            else:
                t_lo = m1
            cm = c5_from_quad(a, b, c, (t_lo + t_hi) / 2)
            if cm < best_c5:
                best_c5 = cm
                best_t = (t_lo + t_hi) / 2
    
    # Verify with exact computation
    h_best = h + best_t * v
    exact_val = exact_c5(h_best)
    if exact_val < exact_c5(h) - 1e-12:
        return best_t, exact_val, h_best
    return best_t, c5_from_quad(a, b, c, best_t), h + best_t * v

def lse_weights(scores, beta, top_k=None):
    if top_k is not None and len(scores) > top_k:
        idx = np.argsort(scores)[::-1][:top_k]
        scores = scores[idx]
    max_s = np.max(scores)
    exp_v = np.exp(np.clip(beta * (scores - max_s), -700, 700))
    w = exp_v / (np.sum(exp_v) + 1e-300)
    return w

def generate_directions(h, G, scores, beta, rng, n=0):
    if n == 0:
        n = len(h)
    L = 2 * n - 1
    directions = []
    
    top_k = max(8, min(40, int(0.2 * n) + 4))
    w = lse_weights(scores, beta, top_k)
    v = -(w @ G)
    vc = centered(v)
    if vc is not None:
        directions.append(vc)
    
    # Individual top active gradients
    idx_top = np.argsort(scores)[::-1]
    for idx in idx_top[:6]:
        v = -G[idx]
        vc = centered(v)
        if vc is not None:
            directions.append(vc)
    
    # Active set average
    active_thresh = scores[-(top_k // 2)] if top_k >= 4 else np.max(scores) - 0.01
    active_mask = scores >= active_thresh
    if np.sum(active_mask) > 0:
        v = -np.sum(G[active_mask], axis=0) / np.sum(active_mask)
        vc = centered(v)
        if vc is not None:
            directions.append(vc)
    
    # Pair directions: transfer from high-h to low-h among sensitive coordinates
    g_mag = np.sum(np.abs(G), axis=0)
    med_g = np.median(g_mag)
    hi_idx = np.where((h > 0.55) & (g_mag > med_g))[0]
    lo_idx = np.where((h < 0.45) & (g_mag > med_g))[0]
    n_pairs = min(20, len(hi_idx) * len(lo_idx))
    if n_pairs > 0:
        for _ in range(min(n_pairs, 20)):
            i = hi_idx[rng.integers(len(hi_idx))]
            j = lo_idx[rng.integers(len(lo_idx))]
            if i != j:
                v = np.zeros(n)
                v[i] = 1.0
                v[j] = -1.0
                directions.append(v)
    
    # Random zero-sum
    for _ in range(5):
        v = rng.standard_normal(n)
        vc = centered(v)
        if vc is not None:
            directions.append(vc)
    
    # Sinusoidal
    x = (np.arange(n) + 0.5) / n
    for freq in [1, 2, 3, 5]:
        ph = rng.uniform(0, 2 * np.pi)
        v = np.sin(2 * np.pi * freq * x + ph)
        vc = centered(v)
        if vc is not None:
            directions.append(vc)
    
    return directions

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)

def polish_lines(h, deadline, rng, beta_start=4, max_cycles=500):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    cur_c5 = best_c5
    beta = float(beta_start)
    stall = 0
    
    for cyc in range(max_cycles):
        if time.monotonic() > deadline:
            break
        if stall > 40:
            break
        
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        G = build_gradient_matrix(h)
        dirs = generate_directions(h, G, scores, beta, rng)
        
        reordered = rng.permutation(len(dirs))
        improved = False
        t_inner = time.monotonic()
        
        for di in reordered:
            if time.monotonic() > deadline:
                break
            v = dirs[di].copy()
            # Ensure exact zero-sum
            vs = np.sum(v)
            if abs(vs) > 1e-12 * np.sum(np.abs(v)) + 1e-15:
                v = v - vs / n
            
            tl, c5l, hl = line_minimize_fast(h, v, deadline)
            if c5l < cur_c5 - 1e-12:
                # Project and verify
                hl = project_box_sum(hl, n / 2.0)
                c5v = exact_c5(hl)
                if c5v < cur_c5 - 1e-12:
                    h = hl
                    cur_c5 = c5v
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Increase beta to sharpen focus
            beta = min(beta * 1.4, 128.0)
            if stall > 20:
                # Try lower beta to escape
                beta = max(4.0, beta * 0.3)
    
    return best_h, best_c5

        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h0 = np.zeros(128)
        h0[np.arange(0, 128, 2)] = 1.0
    
    n0 = len(h0)
    h = project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    # Phase 1: Main line-search polishing (largest budget)
    dl1 = t0 + budget_s * 0.45
    h1, c5_1 = polish_lines(h, dl1, rng, beta_start=4)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Continue with different beta focus
    if time.monotonic() < deadline:
        dl2 = min(deadline, time.monotonic() + budget_s * 0.12)
        h2, c5_2 = polish_lines(best_h, dl2, rng, beta_start=32)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Phase 3: Pair polish
    if time.monotonic() < deadline:
        dl3 = min(deadline, time.monotonic() + budget_s * 0.10)
        h3, c5_3 = pair_polish(best_h, dl3, rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    
    # Phase 4: Multi-resolution search
    if time.monotonic() < deadline:
        n_cur = len(best_h)
        n_tries = sorted(set([max(32, n_cur - 32), max(48, n_cur - 16),
                               n_cur + 16, min(192, n_cur + 32),
                               64, 80, 96, 112, 128, 144, 160]))
        n_tries = [x for x in n_tries if 32 <= x <= 192 and x != n_cur]
        n_tries.sort(key=lambda x: abs(x - n_cur))
        remaining = deadline - time.monotonic()
        per_n = max(10.0, remaining / max(1, len(n_tries)))
        
        for n_try in n_tries:
            if time.monotonic() > deadline:
                break
            if remaining < 8.0:
                break
            h_r = resample(best_h, n_try)
            dl_r = min(deadline, time.monotonic() + per_n * 0.7)
            h_r, c5_r = polish_lines(h_r, dl_r, rng, beta_start=8)
            
            # Quick coordinate polish
            if time.monotonic() < deadline and c5_r <= best_c5 + 0.001:
                dl_rp = min(deadline, time.monotonic() + per_n * 0.3)
                h_r, c5_r = coordinate_polish(h_r, dl_rp, rng)
            
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
            elif c5_r < best_c5 + 1e-9 and n_try < n_cur:
                # Prefer smaller n on tie
                best_c5 = c5_r
                best_h = h_r.copy()
    
    # Phase 5: Final coordinate polish on best
    if time.monotonic() < deadline:
        dl5 = min(deadline, time.monotonic() + budget_s * 0.08)
        h5, c5_5 = coordinate_polish(best_h, dl5, rng)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    
    # Phase 6: Final line-search polish
    if time.monotonic() < deadline:
        dl6 = min(deadline, time.monotonic() + budget_s * 0.05)
        h6, c5_6 = polish_lines(best_h, dl6, rng, beta_start=64)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 <= best_c5 + 1e-11:
        best_c5 = final_c5
    else:
        best_c5 = final_c5
    
    # Safety
    if not np.all(best_h >= -1e-9) or not np.all(best_h <= 1.0 + 1e-9):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_box_sum(best_h, len(best_h) / 2.0)
        best_c5 = exact_c5(best_h)
    
    return best_h, best_c5, len(best_h)
