import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    v = np.array(v, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)
            for i in order:
                if x[i] >= 1.0:
                    break
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff < 1e-12:
                    break
        else:
            order = np.argsort(-x)
            for i in order:
                if x[i] <= 0.0:
                    break
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff > -1e-12:
                    break
    return x


def _c5_val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _f_obj(z, n):
    return z[n]


def _g_obj(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _eq_f(z, n):
    return np.sum(z[:n]) - 0.5 * n

def _act_f(z, n, k):
    h = z[:n]

def _act_g(z, n, k):
    h = z[:n]
    s = k - (n - 1)

def _active_slsqp(h, n, K=12, maxiter=150, ftol=1e-14, deadline=None):
    h = np.array(h, dtype=float)
    corr = _corr(h)
    t0 = float(corr.max()) + 1e-3
    active_indices = sorted(np.argsort(-corr)[:K].tolist())

    for outer in range(25):
        if deadline is not None and time.time() > deadline:
            break
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
        cons = [
            {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        ]
        for k in active_indices:
            cons.append({'type': 'ineq', 'fun': _act_f, 'jac': _act_g, 'args': (n, k)})

        try:
            res = minimize(_f_obj, z0, jac=_g_obj, args=(n,),
                           bounds=bounds, constraints=cons, method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        except Exception:
            break

        if not res.success:
            break
        h = _project(res.x[:n], n * 0.5)
        corr = _corr(h)
        t_new = float(corr.max())

        # Update active set: keep K largest correlations
        new_top = np.argsort(-corr)[:K]
        active_indices = sorted(set(active_indices + new_top.tolist()) &
                                set(new_top.tolist()))
        if len(active_indices) < K:

                if idx not in active_indices:

def _coord_descent(h, n, steps=3, eps=0.05, deadline=None):
    h = np.array(h, dtype=float)
    cur = _c5_val(h)
    for _ in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                for delta in [eps, -eps]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                    h_try[j] = np.clip(h_try[j] - delta, 0.0, 1.0)
                    c = _c5_val(h_try)
                    if c < cur - 1e-10:
                        h = h_try
                        cur = c
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    h = _project(h, n * 0.5)
    return h, _c5_val(h)


def _lp_direction(h, scores, n, radius):
    m = 2 * n - 1
    G = np.zeros((m, n))
    j = np.arange(n)
    for r in range(m):
        s = r - (n - 1)
        pd = np.zeros(n)
        m1 = (j + s >= 0) & (j + s < n)
        pd[j[m1]] = 1.0 - h[j[m1] + s]
        m2 = (j - s >= 0) & (j - s < n)
        pd[j[m2]] -= h[j[m2] - s]
        G[r, :] = pd
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for jj in range(n):
        lo_b = max(-radius, -h[jj])
        hi_b = min(radius, 1.0 - h[jj])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = minimize(c_vec, np.zeros(n + 1), method='SLSQP',
                       bounds=bounds,
                       constraints=[
                           {'type': 'ineq', 'fun': lambda z: (A_ub @ z - b_ub).min(),
                            'jac': lambda z: A_ub.T},
                           {'type': 'eq', 'fun': lambda z: A_eq @ z - b_eq,
                            'jac': lambda z: A_eq.T}
                       ])
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_polish(h, n, deadline, radius_init=0.05, max_iters=1000):
    h = np.array(h, dtype=float)
    cur = _c5_val(h)
    best_h, best_c5 = h.copy(), cur
    radius = radius_init
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 40:
            break
        scores = _corr(h)
        # Use subgradient direction (simplest: gradient at argmax)
        r_max = int(np.argmax(scores))
        s = r_max - (n - 1)
        j = np.arange(n)
        pd = np.zeros(n)
        m1 = (j + s >= 0) & (j + s < n)
        pd[j[m1]] = 1.0 - h[j[m1] + s]
        m2 = (j - s >= 0) & (j - s < n)
        pd[j[m2]] -= h[j[m2] - s]
        pd -= pd.mean()
        norm = np.linalg.norm(pd)
        if norm < 1e-14:
            break
        p = -pd / norm * radius

        # Line search
        alpha_max = 1.0
        for jj in range(n):
            if p[jj] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[jj]) / p[jj])
            elif p[jj] < -1e-14:
                alpha_max = min(alpha_max, h[jj] / (-p[jj]))
        alpha_max = max(alpha_max, 1e-8)

        best_a, best_c5_try = 0.0, cur
        for t in np.linspace(0, 1, 8):
            a = t * alpha_max
            h_t = _project(h + a * p, n * 0.5)
            c5_t = _c5_val(h_t)
            if c5_t < best_c5_try:
                best_c5_try, best_a = c5_t, a
        if best_a > 1e-6:
            h = _project(h + best_a * p, n * 0.5)
            cur = _c5_val(h)
            if cur < best_c5:
                best_c5, best_h = cur, h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.15)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.005:
                radius = 0.03
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _cosine_init(n, m, a, phi, rng=None):
    i = np.arange(n)
    h = 0.5 + a * np.cos(2.0 * np.pi * m * i / n + phi)
    return _project(h, n * 0.5)


def _binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


def _smooth_binary(h_bin, sigma):
    n = len(h_bin)
    kernel = np.zeros(n)
    for j in range(n):
        kernel[j] = np.exp(-0.5 * ((j / sigma) ** 2))
    # Circular convolution
    h_sm = np.convolve(h_bin, kernel, mode='same')
    h_sm = h_sm / h_sm.sum() * n * 0.5
    return _project(h_sm, n * 0.5)
