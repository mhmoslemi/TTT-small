```python
    n = len(h)
    dx = 2.0 / n
    cor, G = compute_corr_and_grads(h)
    t_cur = np.max(cor)
    
    # Build LP: min t s.t. cor[h] + G@(x-h) <= t, sum(x) = n/2, 0<=x<=1
    # Variables: x_0, ..., x_{n-1}, t (total n+1 variables)
    # Objective: minimize t → c = [0,...,0,1]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: cor[h] + G@(x-h) - t <= 0
    # i.e., G@x - t <= -cor[h] + G@h
    # In standard form: A_ub @ vars <= b_ub
    # A_ub = [G, -1], b_ub = G@h - cor
    
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = G @ h - cor
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= x_i <= 1, t free (or t >= 0)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
```

Actually, here's another idea that might be more effective: **Frank-Wolfe (conditional gradient) method**.

The problem min max_k f_k(h) s.t. h in P (where P = {h : sum(h)=n/2, 0<=h<=1}) can be approached with Frank-Wolfe:
1. Compute subgradient g of max_k f_k at current h (pick the argmax k, g = grad f_k)
2. Solve linear oracle: min g @ x s.t. x in P
3. Step: h_new = (1-α)h + α * x_oracle
4. Update α (e.g., α = 2/(t+2) for accelerated FW)

The linear oracle for P is:
min g @ x s.t. sum(x) = n/2, 0 <= x <= 1

This is a simple LP (or even solvable by sorting):
- Sort g in ascending order
- Set x_i = 1 for the smallest g_i values
- Set x_i = 0 for the largest
- The "middle" value gets a fractional assignment to meet the sum constraint

This is O(n log n) per iteration! Much faster than solving a full LP.

The convergence of FW is O(1/t), but the accelerated version (Cutting-plane FW or SFW) can be faster in practice.

Let me implement this:

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_single(h, r):
    """Compute gradient of cor[r] w.r.t. h."""
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[0:m] += (1.0 - h[d:d+m])
            g[d:d+m] += (-h[0:m])
    else:
        gval = -d
        m = n - gval
        if m > 0:
            g[0:m] += (1.0 - h[gval:gval+m])
            g[gval:gval+m] += (-h[0:m])
    return g

def sca_solve(h, k_active=50):
    """Solve one SCA iteration: linearize and solve LP."""
    n = len(h)
    target = n / 2.0
    cor = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    
    # Select active lags (top k by correlation value)
    if k_active < len(cor):
        top_idx = np.argsort(-cor)[:k_active]
    else:
        top_idx = np.arange(len(cor))
    
    # Build gradient matrix for active lags
    n_act = len(top_idx)
    G = np.zeros((n_act, n))
    for i, r in enumerate(top_idx):
        G[i] = compute_grad_single(h, r)
    
    # LP: min t
    # s.t. G@x - t <= G@h - cor[top]
    #      sum(x) = target
    #      0 <= x <= 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - cor[top_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, None
    return res.x[:n], res.x[n]

def sca_loop(h0, deadline, k_active=50, damp=1.0):

    n = len(h0)

    h = h0.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

        if stall > 50:
            break
        
        x_lp, t_lp = sca_solve(h, k_active)
        if x_lp is None:
            break
        
        # Line search from h to x_lp
        direction = x_lp - h
        if np.max(np.abs(direction)) < 1e-12:
            stall += 10
            break
        
        best_alpha = 0.0
        best_c5_try = cur_c5
        for alpha in [damp, damp*0.7, damp*0.5, damp*0.3, damp*0.1]:
            h_try = h + alpha * direction
            h_try = project(h_try, target)
            c5_try = exact_c5(h_try)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_alpha = alpha
        
        if best_alpha > 0 and best_c5_try < cur_c5 - 1e-13:

            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:

def lp_direction(h, scores, radius):
    """Find LP direction to reduce max correlation."""
    n = len(h)
    # Build gradient matrix
    total_lags = 2 * n - 1
    G = np.zeros((total_lags, n))
    for r in range(total_lags):
        G[r] = compute_grad_single(h, r)
    
    # LP: max t s.t. scores + G@(x-h) <= t, sum(x)=n/2, |x-h|<=radius, 0<=x<=1
    # Actually: min -t s.t. G@(x-h) - t <= -scores, ...
    # Or: min t s.t. G@x - t <= G@h - scores (same thing)
    # But we want to REDUCE t, so: min t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((total_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, x_target, c5_cur):
    """Line search from h toward x_target."""
    n = len(h)
    target = n / 2.0
    direction = x_target - h
    best_a = 0.0
    best_c5 = c5_cur
    for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
        h_try = h + frac * direction
        h_try = project(h_try, target)
        c5_try = exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_a = frac
    # Golden section refinement
    if best_a > 0:
        lo_a = best_a * 0.5
        hi_a = min(1.0, best_a * 1.5)
        if hi_a - lo_a > 1e-10:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_a - gr * (hi_a - lo_a)
            c2 = lo_a + gr * (hi_a - lo_a)
            f1 = exact_c5(project(h + c1 * direction, target))
            f2 = exact_c5(project(h + c2 * direction, target))
            for _ in range(12):
                if hi_a - lo_a < 1e-10:
                    break
                if f1 < f2:
                    hi_a, c2, f2 = c2, c1, f1
                    c1 = hi_a - gr * (hi_a - lo_a)
                    f1 = exact_c5(project(h + c1 * direction, target))
                else:
                    lo_a, c1, f1 = c1, c2, f2
                    c2 = lo_a + gr * (hi_a - lo_a)
                    f2 = exact_c5(project(h + c2 * direction, target))
            opt_a = (lo_a + hi_a) / 2.0
            opt_c5 = exact_c5(project(h + opt_a * direction, target))
            if opt_c5 < best_c5:
                best_a, best_c5 = opt_a, opt_c5
    return best_a, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h0, deadline, max_passes=50):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    a = amax * 0.5
                    h_t = best_h.copy()
                    h_t[j] += sign * a
                    h_t[k] -= sign * a
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5
```

```python
def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_single(h, r):
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[0:m] += (1.0 - h[d:d+m])
            g[d:d+m] += (-h[0:m])
    else:
        gv = -d
        m = n - gv
        if m > 0:
            g[0:m] += (1.0 - h[gv:gv+m])
            g[gv:gv+m] += (-h[0:m])
    return g

def build_grad_matrix(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for i, r in enumerate(lags):
        G[i] = compute_grad_single(h, r)
    return G

def sca_solve(h, k_active=60):
    n = len(h)
    target = n / 2.0
    cor = np.correlate(h, 1.0 - h, mode='full')
    if k_active < len(cor):
        top_idx = np.argsort(-cor)[:k_active]
    else:
        top_idx = np.arange(len(cor))
    n_act = len(top_idx)
    G = build_grad_matrix(h, top_idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - cor[top_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:

        return res.x[:n]

    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h0, deadline, max_passes=80):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        a_max = min(1.0 - best_h[j], best_h[k])
                    else:
                        a_max = min(best_h[j], 1.0 - best_h[k])
                    if a_max < 1e-12:
                        continue
                    lo, hi = 0.0, a_max
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    ht1 = best_h.copy()
                    ht1[j] += sign * c1
                    ht1[k] -= sign * c1
                    f1 = exact_c5(ht1)
                    ht2 = best_h.copy()
                    ht2[j] += sign * c2
                    ht2[k] -= sign * c2
                    f2 = exact_c5(ht2)
                    for _ in range(18):
                        if hi - lo < 1e-11:
                            break
                        if f1 < f2:
                            hi, c2, f2 = c2, c1, f1
                            c1 = hi - gr * (hi - lo)
                            ht1 = best_h.copy()
                            ht1[j] += sign * c1
                            ht1[k] -= sign * c1
                            f1 = exact_c5(ht1)
                        else:
                            lo, c1, f1 = c1, c2, f2
                            c2 = lo + gr * (hi - lo)
                            ht2 = best_h.copy()
                            ht2[j] += sign * c2
                            ht2[k] -= sign * c2
                            f2 = exact_c5(ht2)
                    opt_a = (lo + hi) / 2.0
                    ht = best_h.copy()
                    ht[j] += sign * opt_a
                    ht[k] -= sign * opt_a
                    f_opt = exact_c5(ht)
                    if f_opt < best_c5 - 1e-13:
                        best_c5 = f_opt
                        best_h = ht
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: SCA iterations (primary engine)
    d1 = t0 + budget_s * 0.22
    h_sca, c5_sca = sca_loop(h, d1, k_active=60, damp=1.0)
    if c5_sca < best_c5:
        best_c5 = c5_sca
        best_h = h_sca.copy()

    # Phase 2: LP local search
    d2 = t0 + budget_s * 0.35
    h_lp, c5_lp = lp_loop(best_h, d2, radius_init=0.1)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: SCA refinement with more active constraints
    d3 = t0 + budget_s * 0.45
    h_sca2, c5_sca2 = sca_loop(best_h, d3, k_active=100, damp=0.6)
    if c5_sca2 < best_c5:
        best_c5 = c5_sca2
        best_h = h_sca2.copy()

    # Phase 4: Pair polish (high precision)
    d4 = t0 + budget_s * 0.58
    h_pol, c5_pol = pair_polish(best_h, d4, max_passes=60)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 5: Multi-resolution search with SCA
    d5 = t0 + budget_s * 0.75
    offsets = [4, -4, 8, -8, 12, -12, 16, -16, 24, -24, 32, -32, 48, -48]
    tried_n = set([best_n])
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.018)
        h_r, c5_r = sca_loop(h_r, sub_dl, k_active=50, damp=0.8)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Diversified restarts
    d6 = t0 + budget_s * 0.87
    scales = [0.02, 0.04, 0.07, 0.12, 0.18, 0.3]
    for ns in scales:
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_try, c5_try = sca_loop(h_pert, sub_dl, k_active=50, damp=0.9)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 7: Final SCA micro-polish
    if time.time() < hard_deadline:
        sub_dl = min(hard_deadline, time.time() + budget_s * 0.025)
        h_f, c5_f = sca_loop(best_h, sub_dl, k_active=150, damp=0.3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Final pair polish
    if time.time() < hard_deadline:
        h_pol2, c5_pol2 = pair_polish(best_h, hard_deadline, max_passes=40)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
```
