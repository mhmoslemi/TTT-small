This is a weighted average of the gradients of individual lags, with weights proportional to their "importance" (softmax of scores).

Δscore[k] = -δ * ∂score[k]/∂h[i] + δ * ∂score[k]/∂h[j]

So the change in the max score depends on which lags are active.

def _grad_matrix(h):
    """Compute the (2n-1) x n matrix G where G[k,i] = d(score_k)/d(h_i)."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d  # row index
        if d == 0:
            G[r, :] = dx * (1.0 - 2.0*h)
        elif d > 0:
            m = n - d
            G[r, d:] += dx * (1.0 - h[:m])
            G[r, :m] += -dx * h[d:]
        else:
            g = -d
            m = n - g
            G[r, :m] += dx * (1.0 - h[g:])
            G[r, g:] += -dx * h[:m]
    return G

import numpy as np
from scipy.optimize import linprog
import time
from itertools import combinations

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0-h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo+hi)*0.5
        if np.sum(np.clip(v-mid, 0, 1)) > target: lo = mid
        else: hi = mid
    x = np.clip(v-(lo+hi)*0.5, 0, 1)
    r = target - x.sum()
    if abs(r) > 1e-10:
        m = x < 0.999 if r > 0 else x > 0.001
        c = m.sum()
        if c > 0: x[m] += r/c
    return np.clip(x, 0, 1)

def _G(h):
    """Gradient matrix: G[k,i] = d(score_k)/d(h_i)"""
    n = len(h); dx = 2.0/n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n-1+d
        if d == 0:
            G[r] = dx*(1-2*h)
        elif d > 0:
            m = n-d
            G[r, d:] += dx*(1-h[:m])
            G[r, :m] += -dx*h[d:]
        else:
            g = -d; m = n-g
            G[r, :m] += dx*(1-h[g:])
            G[r, g:] += -dx*h[:m]
    return G

def _ls(h, d, c5_ref):
    """Line search along direction d from h."""
    n = len(h)
    amax = 1.0
    p = d > 1e-14
    ng = d < -1e-14
    if p.any(): amax = min(amax, float(((1-h[p])/d[p]).min()))
    if ng.any(): amax = min(amax, float((-h[ng]/(-d[ng])).min()))
    amax = max(amax, 1e-15)
    # Sample
    ns = 16
    al = np.linspace(0, amax, ns)
    vs = np.array([_c5(h+a*d) for a in al])
    bi = vs.argmin()
    if bi in (0, ns-1): return float(al[bi]), float(vs[bi])
    lo, hi = al[bi-1], al[bi+1]
    gr = (5.0**0.5 - 1)*0.5
    c1 = hi - gr*(hi-lo); c2 = lo + gr*(hi-lo)
    f1 = _c5(h+c1*d); f2 = _c5(h+c2*d)
    for _ in range(20):
        if hi-lo < 1e-13: break
        if f1 < f2: hi,f2,c2 = c2,f1,c1; c1 = hi-gr*(hi-lo); f1 = _c5(h+c1*d)
        else: lo,f1,c1 = c2,f2,c2; c2 = lo+gr*(hi-lo); f2 = _c5(h+c2*d)
    return (lo+hi)*0.5, _c5(h+(lo+hi)*0.5*d)

def _smooth_desc(h, dl, bs=15, be=800, ns=800):
    n = len(h); tgt = n*0.5
    h = h.copy(); bh = h.copy(); bc = _c5(h); G = _G(h); stall = 0
    for s in range(ns):
        if time.time() > dl: break
        fr = s/max(1,ns-1)
        beta = bs*(be/bs)**fr
        sc = _scores(h)
        w = np.exp(beta*(sc-sc.max())); w /= w.sum()
        g = G.T @ w; g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            stall += 1
            if stall > 100: break
            continue
        d = -g/gn; found = False
        for sc_ in [0.12, 0.08, 0.04, 0.02]:

            if nc < bc - 1e-14:

                nc = _c5(h); G = _G(h)
                if nc < bc: bc = nc; bh = h.copy()
                stall = 0; found = True; break
        if not found:

            if stall > 150: break
    return bh, bc

def _lp_refine(h, dl, topk=16, rad=0.12, maxit=3000):
    n = len(h); tgt = n*0.5
    h = h.copy(); bc = _c5(h); bh = h.copy(); stall = 0
    G = _G(h)
    for it in range(maxit):
        if time.time() > dl: break
        if stall > 120: break
        sc = _scores(h)
        if topk and topk < 2*n-1:
            idx = np.argsort(sc)[-topk:]
        else:
            idx = np.arange(2*n-1)
        Gk = G[idx]; sk = sc[idx]
        # LP: min t s.t. Gk @ dh + sk <= t, sum(dh)=0, |dh| <= rad*(1-|h|) bounds
        # var: [dh_0,...,dh_{n-1}, t], size n+1
        c = np.zeros(n+1); c[n] = 1.0
        Au = np.zeros((len(sk), n+1))
        Au[:,:n] = Gk; Au[:,n] = -1.0
        bu = -sk
        Ae = np.zeros((1,n+1)); Ae[0,:n] = 1.0
        bnds = []
        for j in range(n):
            lo_j = -min(rad, h[j])
            hi_j = min(rad, 1.0-h[j])
            bnds.append((lo_j, hi_j))
        bnds.append((None, None))
        try:
            res = linprog(c, A_ub=Au, b_ub=bu, A_eq=Ae, b_eq=[0], bounds=bnds, method='highs')
            if not res.success:

        except:
            stall += 1; rad *= 0.75; continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1; rad *= 0.8
            if rad < 1e-5: break
            continue

        if nc < bc - 1e-13:
            h = _project(h+a*p, tgt)
            nc = _c5(h); G = _G(h)
            if nc < bc: bc = nc; bh = h.copy()
            stall = 0; rad = min(rad*1.03, 0.4)
        else:
            stall += 1; rad *= 0.85
            if stall > 40 and rad < 0.02: rad = 0.06
    return bh, bc

def _binary_swap(h, dl, max_swaps=20000):
    """Binary swap search: threshold h, then do swaps."""
    n = len(h)
    if n % 2 != 0: return h, _c5(h)
    hb = (h >= 0.5).astype(float).copy()
    oc = int(hb.sum()); tgt = n//2
    if oc > tgt:
        idx = np.where(hb==1)[0]
        for i in idx[oc-tgt:]: hb[i] = 0
    elif oc < tgt:
        idx = np.where(hb==0)[0]
        for i in idx[:tgt-oc]: hb[i] = 1
    cc = _c5(hb); bc = cc; bh = hb.copy()
    rng = np.random.default_rng(7)
    stall = 0
    for it in range(max_swaps):
        if time.time() > dl: break
        if stall > 3000: break
        oi = np.where(hb==1)[0]; zi = np.where(hb==0)[0]
        i = oi[rng.integers(len(oi))]; j = zi[rng.integers(len(zi))]
        hb[i] = 0; hb[j] = 1
        nc = _c5(hb)
        if nc < cc - 1e-14:
            cc = nc; stall = 0
            if nc < bc: bc = nc; bh = hb.copy()
        else:
            hb[i] = 1; hb[j] = 0; stall += 1
    return bh, bc

def _upsample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    xo = (np.arange(n)+0.5)/n
    xn = (np.arange(n_new)+0.5)/n_new
    hn = np.interp(xn, xo, h)
    return _project(hn, n_new*0.5)

def _exhaustive_binary(n, deadline):
    """Exhaustive search over all binary sequences with n/2 ones."""
    best_c5 = float('inf')
    best_h = None
    tgt = n//2
    total = 1
    for i in range(n):
        total *= (n-i)
    for i in range(tgt):
        total //= (i+1)
    # If too many, do random sampling
    if total > 2000000:
        rng = np.random.default_rng(1)
        seen = set()
        for _ in range(min(int(total), 2000000)):
            if time.time() > deadline: break
            idx = tuple(sorted(rng.choice(n, size=tgt, replace=False).tolist()))
            if idx in seen: continue
            seen.add(idx)
            h = np.zeros(n); h[list(idx)] = 1.0
            c5 = _c5(h)
            if c5 < best_c5: best_c5 = c5; best_h = h.copy()
    else:
        for comb in combinations(range(n), tgt):
            if time.time() > deadline: break
            h = np.zeros(n); h[list(comb)] = 1.0
            c5 = _c5(h)
            if c5 < best_c5: best_c5 = c5; best_h = h.copy()
    return best_h, best_c5
