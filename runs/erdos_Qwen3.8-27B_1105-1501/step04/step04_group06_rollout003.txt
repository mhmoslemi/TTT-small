    bounds = [(max(-radius, -h[i]), min(radius, 1.0 - h[i])) for i in range(n)]
    bounds.append((0.0, raw_max + 1.0))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            p = res.x[:n]
            if np.max(np.abs(p)) > 1e-12:
                # Line search
                return _line_search(h, p, n)
    except Exception:
        pass
    return h.copy(), _exact_c5(h)

def _line_search(h, p, n, max_evals=30):
    """Backtracking + golden section line search."""
    target = n * 0.5
    c5_cur = _exact_c5(h)
    # Find max step
    alpha_max = 1.0
    for i in range(n):
        if p[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    alpha_max = max(alpha_max, 1e-10)
    
    # Coarse grid
    best_a, best_f = 0.0, c5_cur
    alphas = np.linspace(0.1, 1.0, 10) * alpha_max
    for a in alphas:
        h_t = _project(h + a * p, target)
        f_t = _exact_c5(h_t)
        if f_t < best_f:
            best_f, best_a = f_t, a
    
    # Golden section around best
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(_project(h + c1 * p, target))
        f2 = _exact_c5(_project(h + c2 * p, target))
        for _ in range(15):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(_project(h + c1 * p, target))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(_project(h + c2 * p, target))
        a = (lo + hi) / 2.0
        f = _exact_c5(_project(h + a * p, target))
        if f < best_f:
            best_f, best_a = f, a
    
    if best_a > 1e-10 and best_f < c5_cur:
        return _project(h + best_a * p, target), best_f
    return h.copy(), c5_cur

def _pair_swap_search(h, n, max_sweeps=10):
    """Local search using pair swaps."""
    h = h.copy()
    c5_cur = _exact_c5(h)
    corr = _corr(h)
    raw_max = float(np.max(corr))
    active = np.where(corr >= raw_max - 1e-6)[0]
    G = _grad_matrix(h)
    
    for sweep in range(max_sweeps):
        improved = False
        # For each active constraint, find the best pair swap
        for r in active:
            g = G[r, :]  # gradient of corr_r
            # We want to find (i,j) such that g[i] - g[j] < 0 (to decrease corr_r)
            # And also check other active constraints
            # Simple approach: for each i, find j that minimizes the max increase
            best_pair = None
            best_score = 0.0
            for i in range(n):
                if h[i] <= 0.01:
                    continue
                # direction: decrease h[i], increase h[j]
                # effect on corr_r: -g[i] + g[j] (for decreasing h[i] by 1, increasing h[j] by 1)
                # We want this < 0, i.e., g[j] < g[i]
                # Effect on other active r': -g_r'[i] + g_r'[j]
                # We want all to be ≤ 0 (or at least the max to decrease)
                # Score: max over active r' of (-g_r'[i] + g_r'[j])
                # We want this to be < 0
                for j in range(n):
                    if j == i or h[j] >= 0.99:
                        continue
                    # Check effect on all active constraints
                    effects = -G[active, i] + G[active, j]
                    if np.max(effects) < -1e-8:  # all constraints improve
                        score = -np.max(effects)
                        if score > best_score:
                            best_score = score
                            best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                # Line search for step size
                p = np.zeros(n)
                p[i] = -1.0
                p[j] = 1.0
                h_new, c5_new = _line_search(h, p, n)
                if c5_new < c5_cur - 1e-10:
                    h = h_new
                    c5_cur = c5_new
                    corr = _corr(h)
                    raw_max = float(np.max(corr))
                    active = np.where(corr >= raw_max - 1e-6)[0]
                    G = _grad_matrix(h)
                    improved = True
        if not improved:
            break
    return h, c5_cur
