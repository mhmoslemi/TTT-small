import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_obj_grad(h, tau):
    """Returns (F_tau, grad) where F_tau = tau * log(sum exp(S_r/tau))"""
    n = len(h)
    dx = 2.0 / n
    scores = _scores(h)
    P = len(scores)
    
    M = float(np.max(scores))
    exp_vals = np.exp((scores - M) / tau)
    Z = float(np.sum(exp_vals))
    F = M + tau * np.log(Z)
    w = exp_vals / Z
    
    grad = np.zeros(n)
    for r in range(P):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            grad[d:] += w[r] * dx * (1.0 - h[:m])
            grad[:m] += w[r] * dx * (-h[d:])
        else:
            g = -d
            grad[:m] += w[r] * dx * (1.0 - h[g:])
            grad[g:] += w[r] * dx * (-h[:m])
    
    return F, grad

def _max_safe_step(h, p):
    """Maximum alpha such that h + alpha*p stays in [0,1]"""
    alpha = 1e10
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        alpha = min(alpha, (1.0 - h[pos] - 1e-12) / p[pos])
    if np.any(neg):
        alpha = min(alpha, (h[neg] - 1e-12) / (-p[neg]))
    return max(alpha, 0.0)

def _line_search(h, p, tau, max_alpha, n_grid=15):
    """Line search: minimize exact C5 (or smooth F) along h + alpha*p"""
    # Grid scan
    alphas = np.linspace(0, max_alpha, n_grid)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    
    if best_i == 0:
        return best_a, best_c5
    if best_i == n_grid - 1:
        return best_a, best_c5
    
    # Golden section refinement
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    
    for _ in range(20):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for tau in tau_schedule:
        if time.time() > deadline:
            break
        iter_count = 0
        stall = 0
        momentum = np.zeros(n)
        
        for it in range(max_iters_per_tau):
            if time.time() > deadline:
                break
            
            F, grad = _smooth_obj_grad(h, tau)
            
            # Project to zero mean
            grad = grad - np.mean(grad)
            
            # Momentum
            momentum = 0.5 * momentum - grad
            p = momentum.copy()
            
            # Normalize
            pmax = np.max(np.abs(p))
            if pmax < 1e-14:
                stall += 1
                break
            p = p / pmax
            
            # Safe step
            max_a = _max_safe_step(h, p)
            # Also limit by trust region
            max_a = min(max_a, 0.5)
            
            if max_a < 1e-12:
                # Try opposite direction
                p = -p
                max_a = _max_safe_step(h, p)
                max_a = min(max_a, 0.5)
                if max_a < 1e-12:
                    stall += 1
                    continue
            
            alpha, new_c5 = _line_search(h, p, tau, max_a)
            
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                momentum *= 0.5  # reduce momentum after step
            else:
                stall += 1
                momentum *= 0.5
                if stall > 30:
                    break
        
        # After each tau, check if we should reset
        if _exact_c5(h) > best_c5 + 0.001:
            h = best_h.copy()
            cur_c5 = best_c5
    
    return best_h, best_c5

def _line_search(h, p, max_alpha, n_grid=15):
    if max_alpha < 1e-14:
        return 0.0, _exact_c5(h)
    alphas = np.linspace(0, max_alpha, n_grid)
    vals = np.empty(n_grid)
    for i in range(n_grid):
        vals[i] = _exact_c5(h + alphas[i] * p)
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    
    if best_i == 0 or best_i == n_grid - 1:
        return best_a, best_c5
    
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    
    for _ in range(20):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, f2, c2 = c2, f1, c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo, f1, c1 = c1, f2, c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _anneal(h, deadline, tau_schedule, max_iters=800, momentum_beta=0.3):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    for tau in tau_schedule:
        if time.time() > deadline:
            break
        momentum = np.zeros(n)
        stall = 0
        
        for it in range(max_iters):
            if time.time() > deadline:
                break
            if stall > 50:
                break
            
            grad = _smooth_grad(h, tau)
            grad -= np.mean(grad)
            
            momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
            p = -momentum
            p -= np.mean(p)
            
            pmax = np.max(np.abs(p))
            if pmax < 1e-14:
                stall += 1
                continue
            p = p / pmax
            
            max_a = min(_max_safe_step(h, p), 0.5)
            if max_a < 1e-12:
                p = -p
                max_a = min(_max_safe_step(h, p), 0.5)
                if max_a < 1e-12:
                    stall += 1
                    continue
            
            alpha, new_c5 = _line_search(h, p, max_a)
            
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
                momentum *= 0.5
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quasi_init(n, alpha, rng):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)
