def subgradient_step(h, c_target, n_active, epsilon):
    """One subgradient step with momentum"""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = np.max(scores)
    # Active set: coefficients within epsilon of max
    active = scores > max_s - epsilon
    n_act = np.sum(active)
    
    # Compute subgradient (average over active set)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if not active[k]:
            continue
        d = k - (n - 1)
        # d can range from -(n-1) to (n-1)
        if d >= 0:
            m = n - d
            for j in range(m):
                grad[j] += (1.0 - h[j+d])
                grad[j+d] += -h[j]

            m = n - g
            for j in range(m):

                grad[j+g] += -h[j]
    
    grad /= n_act
    return grad, max_s

def coord_descent_reservoir(h, deadline, max_sweeps=500):
    """Coordinate descent using a free (reservoir) coordinate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Cycle the free coordinate for better coverage
        free_idx = sweep % n
        for i in range(n):
            if i == free_idx:
                continue
            sum_rest = np.sum(h) - h[i] - h[free_idx]
            hi_max = min(1.0, target - sum_rest)
            hi_min = max(0.0, target - sum_rest - 1.0)
            if hi_max - hi_min < 1e-12:
                continue
            
            lo = hi_min
            hi = hi_max
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            h[i] = c1
            h[free_idx] = target - sum_rest - c1
            f1 = exact_c5(h)
            h[i] = c2
            h[free_idx] = target - sum_rest - c2
            f2 = exact_c5(h)
            
            for _ in range(35):
                if hi - lo < 1e-13:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    h[i] = c1
                    h[free_idx] = target - sum_rest - c1
                    f1 = exact_c5(h)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    h[i] = c2
                    h[free_idx] = target - sum_rest - c2
                    f2 = exact_c5(h)
            
            opt_val = (lo + hi) / 2.0
            h[i] = opt_val
            h[free_idx] = target - sum_rest - opt_val
            h[free_idx] = max(0.0, min(1.0, h[free_idx]))
            new_c5 = exact_c5(h)
            
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                improved = True
        
        if not improved:
            break
    
    h = project(h, target)
    cur_c5 = exact_c5(h)
    return h, cur_c5

def subgradient_momentum(h, deadline, f_lb=0.30, beta=0.9):
    """Subgradient method with momentum for max-of-linear objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    t = 0
    
    # Adaptive epsilon for active set
    eps_init = 0.01
    eps_decay = 0.999
    
    eps = eps_init
    base_step = 0.05
    
    while time.time() < deadline:
        t += 1
        scores = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        scores_scaled = scores * dx
        max_s = float(np.max(scores_scaled))
        
        # Active set
        active = scores_scaled > max_s - eps
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(2*n-1, dtype=bool)
            active[np.argmax(scores_scaled)] = True
            n_active = 1
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in range(2*n - 1):
            if not active[k]:
                continue
            d = k - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:])
                    grad[d:] += -h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += (1.0 - h[g:])
                    grad[g:] += -h[:m]
        grad /= n_active
        grad *= dx  # Scale by dx since objective includes dx
        
        # Momentum
        mom = beta * mom + (1.0 - beta) * grad
        
        # Step size: Polyak-like
        gap = max_s - f_lb
        if gap < 1e-10:
            break
        norm_sq = float(np.dot(mom, mom))
        if norm_sq < 1e-20:
            break
        step = gap / (norm_sq * 1.0)
        step = min(step, 0.1)  # Cap step size
        step = max(step, 1e-6)
        
        # Take step
        h_new = h - step * mom
        h_new = project(h_new, target)
        
        new_c5 = exact_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
        cur_c5 = new_c5
        
        # Decay epsilon
        eps *= eps_decay
        if eps < 0.001:
            eps = 0.001
    
    return best_h, best_c5

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    scores_raw = np.correlate(h, omh, mode='full')
    scores = scores_raw * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            grad[js1:je1 + 1] += wk * (1.0 - h[js1 + d:je1 + 1 + d])
        js2 = max(0, d)
        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wk * h[js2 - d:je2 + 1 - d]
    return val, dx * grad

        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lp_loop(h, deadline, radius_init=0.1):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = build_corr_grad(h)
        p = lp_direction(h, scores, G, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def coord_descent_reservoir(h, deadline, max_sweeps=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stalls = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        free_idx = sweep % n
        for i in range(n):
            if i == free_idx:
                continue
            sum_rest = np.sum(h) - h[i] - h[free_idx]
            hi_max = min(1.0, target - sum_rest)
            hi_min = max(0.0, target - sum_rest - 1.0)
            if hi_max - hi_min < 1e-12:
                continue
            lo = hi_min
            hi = hi_max
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            h[i] = c1
            h[free_idx] = target - sum_rest - c1
            f1 = exact_c5(h)
            h[i] = c2
            h[free_idx] = target - sum_rest - c2
            f2 = exact_c5(h)
            for _ in range(25):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    h[i] = c1
                    h[free_idx] = target - sum_rest - c1
                    f1 = exact_c5(h)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    h[i] = c2
                    h[free_idx] = target - sum_rest - c2
                    f2 = exact_c5(h)
            opt_val = (lo + hi) / 2.0
            h[i] = opt_val
            h[free_idx] = target - sum_rest - opt_val
            if h[free_idx] < 0.0:
                h[free_idx] = 0.0
            elif h[free_idx] > 1.0:
                h[free_idx] = 1.0
            new_c5 = exact_c5(h)
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                improved = True
        if not improved:
            stalls += 1
            if stalls > 5:
                break
        else:
            stalls = 0

    h = project(h, target)
    cur_c5 = exact_c5(h)
    return h, cur_c5

def subgradient_momentum(h, deadline, f_lb=0.30, beta=0.88):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    eps = 0.015
    t = 0
    stall_count = 0

    while time.time() < deadline:
        t += 1
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        active = scores > max_s - eps
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[np.argmax(scores)] = True
            n_active = 1
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if not active[k]:
                continue
            d = k - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[:m] += (1.0 - h[d:])
                    grad[d:] += -h[:m]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += (1.0 - h[g:])
                    grad[g:] += -h[:m]
        grad /= n_active
        grad *= dx
        mom = beta * mom + (1.0 - beta) * grad
        gap = max_s - f_lb
        if gap < 1e-10:
            break
        norm_sq = float(np.dot(mom, mom))
        if norm_sq < 1e-20:
            break
        step = gap / norm_sq
        step = min(step, 0.08)
        step = max(step, 1e-6)
        h_new = h - step * mom
        h_new = project(h_new, target)
        new_c5 = exact_c5(h_new)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        h = h_new
        cur_c5 = new_c5
        eps = max(0.003, eps * 0.998)
        if stall_count > 200:
            break
    return best_h, best_c5

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    scores_raw = np.correlate(h, omh, mode='full')
    scores = scores_raw * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            grad[js1:je1 + 1] += wk * (1.0 - h[js1 + d:je1 + 1 + d])
        js2 = max(0, d)
        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wk * h[js2 - d:je2 + 1 - d]
    return val, dx * grad

def lfgs_smooth(h0, tau, maxiter, deadline):
    n = len(h0)
    target = n / 2.0
    penalty_w = 500.0
    try:
        h_work = h0.copy()
        for it in range(maxiter):
            if time.time() > deadline:
                break
            v, g = soft_value_grad(h_work, tau)
            s = np.sum(h_work) - target
            g_pen = g + 2.0 * penalty_w * s * np.ones(n)
            trial = h_work - 0.01 * g_pen
            trial = project(trial, target)
            v2, _ = soft_value_grad(trial, tau)
            if v2 < v + penalty_w * (np.sum(trial) - target) ** 2:
                h_work = trial
            else:
                trial = h_work - 0.003 * g_pen
                trial = project(trial, target)
                v2, _ = soft_value_grad(trial, tau)
                if v2 < v + penalty_w * (np.sum(trial) - target) ** 2:
                    h_work = trial
        return project(h_work, target)
    except Exception:
        return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def random_pair_cd(h, deadline, n_pairs=5000):
    """Random pair coordinate descent with golden section."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    rng_local = np.random.default_rng(12345)
    stalls = 0

    for it in range(n_pairs):
        if time.time() > deadline:
            break
        i = int(rng_local.integers(0, n))
        j = int(rng_local.integers(0, n))
        if i == j:
            continue
        s_const = h[i] + h[j]
        lo = max(0.0, s_const - 1.0)
        hi = min(1.0, s_const)
        if hi - lo < 1e-12:
            continue
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        h[i] = c1
        h[j] = s_const - c1
        f1 = exact_c5(h)
        h[i] = c2
        h[j] = s_const - c2
        f2 = exact_c5(h)
        for _ in range(20):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                h[i] = c1
                h[j] = s_const - c1
                f1 = exact_c5(h)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                h[i] = c2
                h[j] = s_const - c2
                f2 = exact_c5(h)
        opt = (lo + hi) / 2.0
        h[i] = opt
        h[j] = s_const - opt
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            stalls = 0
        else:
            stalls += 1
            if stalls > 500:
                break
    h = project(h, target)
    cur_c5 = exact_c5(h)
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase A: LP polish (10% budget)
    dA = t0 + budget_s * 0.10
    h_lp, c5_lp = lp_loop(h, dA, radius_init=0.12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase B: Reservoir coordinate descent (20% budget)
    dB = t0 + budget_s * 0.30
    h_cd, c5_cd = coord_descent_reservoir(best_h, dB, max_sweeps=10000)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase C: Subgradient with momentum (18% budget)
    dC = t0 + budget_s * 0.48
    h_sg, c5_sg = subgradient_momentum(best_h, dC, f_lb=0.30, beta=0.88)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()

    # Phase D: Random pair CD (12% budget)
    dD = t0 + budget_s * 0.60
    h_rp, c5_rp = random_pair_cd(best_h, dD, n_pairs=20000)
    if c5_rp < best_c5:
        best_c5 = c5_rp
        best_h = h_rp.copy()

    # Phase E: LP polish (12% budget)
    dE = t0 + budget_s * 0.72
    h_lp2, c5_lp2 = lp_loop(best_h, dE, radius_init=0.08)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase F: Diversification - perturb + optimize (15% budget)
    dF = t0 + budget_s * 0.87
    scales = [0.02, 0.05, 0.10, 0.20]
    for ns in scales:
        if time.time() > dF:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        # Quick reservoir CD
        sub_dl = min(dF, time.time() + budget_s * 0.025)
        h_try, c5_try = coord_descent_reservoir(h_pert, sub_dl, max_sweeps=500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Quick LP
        if time.time() < dF:
            sub_dl2 = min(dF, time.time() + budget_s * 0.015)
            h_try2, c5_try2 = lp_loop(h_pert, sub_dl2, radius_init=0.05)
            if c5_try2 < best_c5:
                best_c5 = c5_try2
                best_h = h_try2.copy()

    # Phase G: Multi-resolution (8% budget)
    dG = t0 + budget_s * 0.95
    tried_n = set([best_n])
    for dn in [-4, 4, -8, 8, -12, 12, -16, 16, -24, 24]:
        if time.time() > dG:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 250:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(dG, time.time() + budget_s * 0.012)
        h_r, c5_r = coord_descent_reservoir(h_r, sub_dl, max_sweeps=300)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            if time.time() < dG:
                sub_dl2 = min(dG, time.time() + budget_s * 0.008)
                h_r2, c5_r2 = lp_loop(h_r, sub_dl2, radius_init=0.04)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase H: Final polish
    dH = t0 + budget_s * 0.99
    if time.time() < dH:
        h_f, c5_f = coord_descent_reservoir(best_h, dH, max_sweeps=2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
