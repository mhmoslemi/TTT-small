    weights /= np.sum(weights)
    J = _jac_corr(h)
    grad = J.T @ weights

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_jac(h):
    n = len(h)
    dx = 2.0 / n
    nc = 2 * n - 1
    J = np.zeros((nc, n))
    for s in range(0, n):

        q_neg = n - 1 - s

        else:

            r2 = np.arange(s, n)
            J[q_pos, r1] += (1.0 - h[r1 + s])
            J[q_pos, r2] -= h[r2 - s]
            J[q_neg, r2] += (1.0 - h[r2 - s])
            J[q_neg, r1] -= h[r1 + s]
    return J * dx

    z = _scaled_corr(h)
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    exp_shifted = np.exp(shifted)
    log_sum = z_max + tau * np.log(np.sum(exp_shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    J = _corr_jac(h)
    grad = J.T @ weights
    return log_sum, grad

def _lse_slsqp(h, tau, maxiter=200, ftol=1e-12):

    def f(x):
        return _lse_value_and_grad(x, tau)[0]
    
    def g(x):
        return _lse_value_and_grad(x, tau)[1]

        return x.sum() - target

        return np.ones(n)
