def _raw_scores(h):

def _raw_grad_matrix(h):

def _resample(h, n_new):
    """Area-preserving resampling."""
    n = len(h)
    if n_new == n:
        return h.copy()
    # Area-preserving: each source cell [i*2/n, (i+1)*2/n] with height h[i]
    # maps to target cells [j*2/n_new, (j+1)*2/n_new]
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    x_old = (np.arange(n) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h)
    # Area is approximately preserved; project to exact
    return _project(h_new, n_new / 2.0)

from scipy.optimize import minimize, linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _raw_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            G[r, :cnt] += -h[d:d+cnt]
            G[r, d:d+cnt] += 1.0 - h[:cnt]
        else:
            g = -d
            cnt = n - g
            G[r, :cnt] += 1.0 - h[g:g+cnt]
            G[r, g:g+cnt] += -h[:cnt]
    return G

class _Ctx:
    def __init__(self, n):
        self.n = n
    def obj(self, x):
        return x[self.n]
    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    def ineq(self, x):
        h = x[:self.n]
        T = x[self.n]
        return _raw_scores(h) - T
    def ineq_jac(self, x):
        n = self.n
        h = x[:n]
        G = _raw_grad_matrix(h)
        jac = np.empty((2*n-1, n+1))
        jac[:, :n] = G
        jac[:, n] = -1.0
        return jac
    def eq(self, x):
        return np.array([np.sum(x[:self.n]) - self.n / 2.0])
    def eq_jac(self, x):
        J = np.zeros((1, self.n + 1))
        J[0, :self.n] = 1.0
        return J
    def bds(self):
        return [(0.0, 1.0)] * self.n + [(0.0, float(self.n))]

        return h0, _exact_c5(h0)

    try:

            ctx.obj, x0, jac=ctx.obj_grad,
            constraints=[
                {'type': 'ineq', 'fun': ctx.ineq, 'jac': ctx.ineq_jac},
                {'type': 'eq', 'fun': ctx.eq, 'jac': ctx.eq_jac},
            ],
            bounds=ctx.bds(),
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )

    bounds.append((None, None))
    try:
        r = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs')
        if r.success:
            return r.x[:n]
    except Exception:
        pass
    return None

def _pair_line_search(h, j, k, sign_j, sign_k, c5_cur):
    """1D line search for pair move h[j]+=t, h[k]-=t (or similar)."""
    if sign_j > 0:
        t_max = min(1.0 - h[j], h[k])
    else:
        t_max = min(h[j], 1.0 - h[k])
    if t_max < 1e-12:
        return 0.0, c5_cur
    # Grid search
    n_grid = 30
    ts = np.linspace(0.0, t_max, n_grid)
    best_t = 0.0
    best_val = c5_cur
    best_idx = 0
    for idx, t in enumerate(ts):
        if t < 1e-14:
            continue
        h_t = h.copy()
        h_t[j] += sign_j * t
        h_t[k] += sign_k * t
        if h_t[j] < -1e-12 or h_t[j] > 1 + 1e-12 or h_t[k] < -1e-12 or h_t[k] > 1 + 1e-12:
            continue
        v = _exact_c5(h_t)
        if v < best_val:
            best_val = v
            best_t = t
            best_idx = idx
    # Refine with golden section
    lo_t = ts[max(0, best_idx - 3)]
    hi_t = ts[min(n_grid - 1, best_idx + 3)]
    if hi_t - lo_t < 1e-15:
        return best_t, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f1 = _exact_c5(np.concatenate([h[:j], [h[j]+sign_j*c1], h[j+1:k], [h[k]+sign_k*c1], h[k+1:]])) if k > j else _exact_c5(np.concatenate([h[:k], [h[k]+sign_k*c1], h[k+1:j], [h[j]+sign_j*c1], h[j+1:]]))
    f2 = _exact_c5(np.concatenate([h[:j], [h[j]+sign_j*c2], h[j+1:k], [h[k]+sign_k*c2], h[k+1:]])) if k > j else _exact_c5(np.concatenate([h[:k], [h[k]+sign_k*c2], h[k+1:j], [h[j]+sign_j*c2], h[j+1:]]))
    for _ in range(25):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            h_c = h.copy()
            h_c[j] += sign_j * c1
            h_c[k] += sign_k * c1
            f1 = _exact_c5(h_c)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            h_c = h.copy()
            h_c[j] += sign_j * c2
            h_c[k] += sign_k * c2
            f2 = _exact_c5(h_c)
    opt_t = (lo_t + hi_t) / 2.0
    h_opt = h.copy()
    h_opt[j] += sign_j * opt_t
    h_opt[k] += sign_k * opt_t
    f_opt = _exact_c5(h_opt)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _pair_refine(h, deadline, max_passes=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0

    for _pass in range(max_passes):
        if time.time() > deadline or stall > 50:
            break
        improved = False
        # Compute variable importance from active lags
        scores = _raw_scores(h)
        max_s = float(np.max(scores))
        active_thresh = max_s - 0.05 * max_s
        var_imp = np.zeros(n)
        for r in range(2 * n - 1):
            if scores[r] < active_thresh:
                continue
            d = r - (n - 1)
            if d >= 0:
                cnt = n - d
                var_imp[:cnt] += scores[r]
                var_imp[d:d+cnt] += scores[r]
            else:
                g = -d
                cnt = n - g
                var_imp[:cnt] += scores[r]
                var_imp[g:g+cnt] += scores[r]

        # Select top variables
        top_k = min(n, 28)
        top_vars = np.argsort(var_imp)[::-1][:top_k].astype(int)
        rng = np.random.default_rng(hash(_pass) & 0xFFFFFFFF)

        # Generate candidate pairs
        pairs = set()
        for a in range(len(top_vars)):
            for b in range(a + 1, len(top_vars)):
                pairs.add((int(top_vars[a]), int(top_vars[b])))
        # Add some random pairs
        for _ in range(min(15, n)):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        pair_list = list(pairs)
        rng.shuffle(pair_list)
        # Limit pairs per pass
        max_pairs_per_pass = min(len(pair_list), 40)
        pair_list = pair_list[:max_pairs_per_pass]

        for (j, k) in pair_list:
            if time.time() > deadline:
                break
            if h[j] < 1e-7 or h[j] > 1 - 1e-7 or h[k] < 1e-7 or h[k] > 1 - 1e-7:
                continue
            for (sj, sk) in [(1, -1), (-1, 1)]:
                t, new_c5 = _pair_line_search(h, j, k, sj, sk, best_c5)
                if new_c5 < best_c5 - 1e-14:
                    h = h.copy()
                    h[j] += sj * t
                    h[k] += sk * t
                    h = _project(h, target)
                    best_c5 = _exact_c5(h)
                    best_h = h.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
        else:
            stall = 0

    return best_h, best_c5

def _grad_descent(h, deadline, tau_init=20.0, lr_init=0.01):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    dx = 2.0 / n

    while time.time() < deadline and stall < 70:
        scores = _raw_scores(h) * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        shifted -= np.max(shifted)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        G = _raw_grad_matrix(h)
        grad = G.T @ w * dx
        mom = 0.9 * mom + 0.1 * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.04, 0.08)
            tau = min(tau * 1.03, 200.0)
        else:
            stall += 1
            lr *= 0.88
            if stall % 30 == 0:
                tau = max(tau_init * 0.5, tau * 0.7)
    return best_h, best_c5

def _lp_local(h, deadline, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline and stall < 50:
        scores = _raw_scores(h) * (2.0 / n)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.55
            if radius < 1e-8:
                break
            continue
        # Feasible alpha
        alpha_max = 5.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        # Line search
        n_grid = 25
        ts = np.linspace(0.0, alpha_max, n_grid)
        best_a = 0.0
        best_v = cur_c5
        best_i = 0
        for i, t in enumerate(ts):
            if t < 1e-14:
                continue
            h_t = h + t * p
            if h_t.min() < -1e-12 or h_t.max() > 1 + 1e-12:
                continue
            v = _exact_c5(h_t)
            if v < best_v:
                best_v = v
                best_a = t
                best_i = i
        # Golden refine
        lo_t = ts[max(0, best_i - 2)]
        hi_t = ts[min(n_grid - 1, best_i + 2)]
        if hi_t - lo_t > 1e-14:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_t - gr * (hi_t - lo_t)
            c2 = lo_t + gr * (hi_t - lo_t)
            h_c1 = h + c1 * p
            h_c2 = h + c2 * p
            f1 = _exact_c5(h_c1) if h_c1.min() >= -1e-12 and h_c1.max() <= 1+1e-12 else cur_c5 + 1
            f2 = _exact_c5(h_c2) if h_c2.min() >= -1e-12 and h_c2.max() <= 1+1e-12 else cur_c5 + 1
            for _ in range(20):
                if hi_t - lo_t < 1e-14:
                    break
                if f1 < f2:
                    hi_t, c2, f2 = c2, c1, f1
                    c1 = hi_t - gr * (hi_t - lo_t)
                    h_c1 = h + c1 * p
                    f1 = _exact_c5(h_c1) if h_c1.min() >= -1e-12 and h_c1.max() <= 1+1e-12 else cur_c5 + 1
                else:
                    lo_t, c1, f1 = c1, c2, f2
                    c2 = lo_t + gr * (hi_t - lo_t)
                    h_c2 = h + c2 * p
                    f2 = _exact_c5(h_c2) if h_c2.min() >= -1e-12 and h_c2.max() <= 1+1e-12 else cur_c5 + 1
            opt_t = (lo_t + hi_t) / 2.0
            h_opt = h + opt_t * p
            if h_opt.min() >= -1e-12 and h_opt.max() <= 1+1e-12:
                f_opt = _exact_c5(h_opt)
                if f_opt < best_v:
                    best_v = f_opt
                    best_a = opt_t

        if best_v < cur_c5 - 1e-14:
            h = h + best_a * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.35)
        else:
            stall += 1
            radius *= 0.6
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _equalize_scores(h, deadline, n_iters=30):
    """Try to equalize the top scores by solving a small LP."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    dx = 2.0 / n

    for _it in range(n_iters):
        if time.time() > deadline:
            break
        scores = _raw_scores(h) * dx
        max_s = float(np.max(scores))
        # Find lags within 2% of max
        active = np.where(scores > max_s * 0.98)[0]
        if len(active) < 2 or len(active) > 40:
            active = np.argsort(scores)[::-1][:15]

        # Try to reduce the max by moving mass from high-contribution positions
        # to low-contribution positions
        # Compute per-variable contribution to active scores
        G = _raw_grad_matrix(h)
        active_grads = G[active]  # shape (n_active, n)
        # Average gradient over active lags
        avg_grad = np.mean(active_grads, axis=0) * dx
        # We want to decrease the active scores: move h in direction -avg_grad
        # But constrained to sum(p)=0 and box
        p = -avg_grad
        p -= np.mean(p)
        # Clip to feasible direction
        radius = 0.1
        scale = np.max(np.abs(p))
        if scale > 1e-14:
            p = p / scale * radius
        # Check feasibility
        alpha_max = 5.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        if alpha_max < 1e-10:
            continue
        # Try a few step sizes
        best_t = 0.0
        best_v = best_c5
        for frac in [0.3, 0.5, 0.8, 1.0]:
            t = alpha_max * frac
            h_t = h + t * p
            h_t = _project(h_t, target)
            v = _exact_c5(h_t)
            if v < best_v:
                best_v = v
                best_t = t
        if best_t > 0:
            h = _project(h + best_t * p, target)
            if _exact_c5(h) < best_c5:
                best_c5 = _exact_c5(h)
                best_h = h.copy()
            else:
                h = best_h.copy()
        else:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s - 8.0

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float, copy=True)
    else:
        n0_fb = 80
        x = (np.arange(n0_fb) + 0.5) / n0_fb * 2.0
        h0 = 0.5 + 0.35 * np.sin(np.pi * x) + 0.08 * np.sin(2 * np.pi * x)
        h0 = _project(h0, n0_fb / 2.0)

    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Phase 1: SLSQP + LP at parent resolution
    d1 = min(hard_deadline, t0 + budget_s * 0.18)
    if n0 <= 140:
        h1, c5_1 = _slsqp_solve(best_h, d1, maxiter=60)
        if c5_1 < best_c5 - 1e-12:
            best_c5 = c5_1
            best_h = h1.copy()
    if time.time() < d1:
        h1b, c5_1b = _lp_local(best_h, d1, radius_init=0.15)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Gradient descent for broader search
    d2 = min(hard_deadline, t0 + budget_s * 0.28)
    if time.time() < d2:
        h2, c5_2 = _grad_descent(best_h, d2, tau_init=15.0, lr_init=0.012)
        if c5_2 < best_c5 - 1e-12:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Multi-resolution search
    d3 = min(hard_deadline, t0 + budget_s * 0.58)
    n_list = sorted(set([best_n - 12, best_n - 8, best_n - 6, best_n - 4, best_n - 2,
                         best_n, best_n + 2, best_n + 4, best_n + 6, best_n + 8,
                         70, 74, 78, 82, 86, 90, 94, 98, 102, 106, 110, 114, 120, 130]))
    n_list = [x for x in n_list if 50 <= x <= 150]
    # Sort by expected cost (smaller = cheaper, try those first)
    n_list.sort(key=lambda x: abs(x - best_n))

    time_per_res = max(2.0, (d3 - time.time()) / max(1, len(n_list)))
    for n_try in n_list:
        if time.time() > d3:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + time_per_res)
        h_r, c5_r = _lp_local(h_r, sub_dl, radius_init=0.12)
        if n_try <= 140 and time.time() < sub_dl - 2.0:
            sub_dl2 = min(sub_dl, time.time() + time_per_res * 0.4)
            h_r2, c5_r2 = _slsqp_solve(h_r, sub_dl2, maxiter=35)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if time.time() < sub_dl:
            sub_dl3 = min(sub_dl, time.time() + time_per_res * 0.3)
            h_r3, c5_r3 = _pair_refine(h_r, sub_dl3, max_passes=15)
            if c5_r3 < c5_r:
                h_r, c5_r = h_r3, c5_r3
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Intensive optimization at best resolution
    d4 = min(hard_deadline, t0 + budget_s * 0.82)
    # LP
    h4, c5_4 = _lp_local(best_h, d4, radius_init=0.1)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # SLSQP
    if best_n <= 150 and time.time() < d4:
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h4b, c5_4b = _slsqp_solve(best_h, sub_dl, maxiter=55)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()
    # Equalize
    if time.time() < d4:
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4c, c5_4c = _equalize_scores(best_h, sub_dl, n_iters=25)
        if c5_4c < best_c5:
            best_c5 = c5_4c
            best_h = h4c.copy()
    # Pair
    if time.time() < d4:
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4d, c5_4d = _pair_refine(best_h, sub_dl, max_passes=60)
        if c5_4d < best_c5:
            best_c5 = c5_4d
            best_h = h4d.copy()

    # Phase 5: Final polish
    d5 = hard_deadline
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.04)
        h5, c5_5 = _lp_local(best_h, sub_dl, radius_init=0.04)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    if time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.05)
        h5b, c5_5b = _pair_refine(best_h, sub_dl, max_passes=100)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()
    # Final SLSQP polish
    if best_n <= 140 and time.time() < d5:
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h5c, c5_5c = _slsqp_solve(best_h, sub_dl, maxiter=40)
        if c5_5c < best_c5:
            best_c5 = c5_5c
            best_h = h5c.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
