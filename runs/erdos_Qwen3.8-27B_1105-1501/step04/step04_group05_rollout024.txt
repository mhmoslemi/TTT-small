                h[j] = float(p - t)
                c = exact_c5(h)
                if c < best_c_t - tol:
                    best_c_t = c
                    best_t = float(t)
            if best_c_t < best_c - tol:
                h[i] = best_t
                h[j] = float(p - best_t)
                best_c = best_c_t
                sweep_improved = True
            else:
                h[i] = save_i
                h[j] = save_j
        if not sweep_improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h0 = np.array(initial_h_values, dtype=float)
    h0 = project(h0, target)
    best_h = h0.copy()
    best_c5 = exact_c5(h0)

    # Phase 1: Binary mask search (20% budget)
    d1 = t0 + budget_s * 0.20
    masks = []
    # Mask from thresholding
    masks.append(make_binary_from_h(h0, n))
    # Random masks
    for _ in range(4):
        perm = rng.permutation(n)
        m = np.zeros(n)
        m[perm[:n // 2]] = 1.0
        masks.append(m)
    # Periodic masks
    for period in [3, 5, 7]:
        m = np.zeros(n)
        for i in range(n):
            m[i] = 1.0 if (i % period < period - 1) and (i * (period - 1)) % n < n // 2 * period else 0.0
        # Fix count
        cnt = int(np.sum(m))
        if cnt > n // 2:
            idx = np.where(m > 0.5)[0]
            m[idx[cnt - n // 2:]] = 0.0
        elif cnt < n // 2:
            idx = np.where(m < 0.5)[0]
            m[idx[:n // 2 - cnt]] = 1.0
        if int(np.sum(m)) == n // 2:
            masks.append(m)

    best_mask = None
    best_mask_c = float('inf')
    for m in masks:
        if time.time() > d1 - 30:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.04)
        m_out, c_out = binary_descent(m, n, sub_dl)
        if c_out < best_mask_c:
            best_mask_c = c_out
            best_mask = m_out.copy()

    # Phase 2: Two-level optimization (5% budget)
    d2 = t0 + budget_s * 0.25
    h_candidate = None
    if best_mask is not None and time.time() < d2:
        d_opt, c_opt = optimize_d(best_mask, n)
        h_candidate = two_level_from_mask(best_mask, d_opt)
        c_candidate = exact_c5(h_candidate)
        if c_candidate < best_c5:
            best_c5 = c_candidate
            best_h = h_candidate.copy()

    # Phase 3: Pair line search - main (40% budget)
    d3 = t0 + budget_s * 0.65
    # Start from best continuous
    h_work = best_h.copy()
    if time.time() < d3:
        sub_dl = min(d3, time.time() + budget_s * 0.20)
        h_work, c_work = pair_line_search(h_work, sub_dl, n_grid=17, max_sweeps=3000)
        if c_work < best_c5:
            best_c5 = c_work
            best_h = h_work.copy()
    # Try from two-level if available
    if h_candidate is not None and time.time() < d3:
        h_work2 = h_candidate.copy()
        sub_dl2 = min(d3, time.time() + budget_s * 0.15)
        h_work2, c_work2 = pair_line_search(h_work2, sub_dl2, n_grid=17, max_sweeps=2000)
        if c_work2 < best_c5:
            best_c5 = c_work2
            best_h = h_work2.copy()

    # Phase 4: Resolution search (15% budget)
    d4 = t0 + budget_s * 0.80
    for dn in [2, -2, 4, -4, 6, -6]:
        if time.time() > d4:
            break
        n_try = n + dn
        if n_try < 16 or n_try > 128 or n_try % 2 != 0:
            continue
        h_r = resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_r, c_r = pair_line_search(h_r, sub_dl, n_grid=13, max_sweeps=500)
        if c_r < best_c5 - 1e-8:
            best_c5 = c_r
            best_h = h_r.copy()
            n = n_try
            target = n / 2.0

    # Phase 5: Final intensive polish with finer grid (20% budget)
    d5 = deadline - 3
    if time.time() < d5:
        h_final = best_h.copy()
        sub_dl = min(d5, time.time() + budget_s * 0.12)
        h_final, c_final = pair_line_search_quad(h_final, sub_dl, n_grid=41, max_sweeps=2000)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    # Phase 6: Random perturbation + reoptimize (remaining time)
    if time.time() < deadline - 5:
        for attempt in range(100):
            if time.time() > deadline - 5:
                break
            scale = 0.02
            h_pert = best_h + rng.normal(0, scale, len(best_h))
            h_pert = project(h_pert, target)
            sub_dl = min(deadline - 5, time.time() + budget_s * 0.01)
            h_try, c_try = pair_line_search(h_pert, sub_dl, n_grid=17, max_sweeps=300)
            if c_try < best_c5 - 1e-10:
                best_c5 = c_try
                best_h = h_try.copy()

    # Final projection and validation
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = exact_c5(best_h)
    return best_h, best_c5, len(best_h)
