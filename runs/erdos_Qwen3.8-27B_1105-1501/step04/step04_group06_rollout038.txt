Key observation: at the optimum, the "active" correlations (those equal to the max) form a set. The KKT conditions say that the gradient of the active correlations, weighted by the Lagrange multipliers, is in the span of the constraint gradients (the sum constraint and the active bound constraints).

For the sum constraint: the gradient is (1, 1, ..., 1).
For bound constraints: the gradient is e_i (standard basis vector).

So at the optimum, there exist multipliers λ_0 ≥ 0 for the sum and μ_i ≥ 0 for bounds such that:
Σ_r λ_r * ∇corr_r = λ_0 * (1,...,1) + Σ_i μ_i * e_i

This is a complex condition that's hard to exploit directly, but it tells us that at the optimum, the "average" gradient of active correlations should be close to a constant vector (plus adjustments for bounded variables).

This suggests that a good refinement strategy is to:
1. Identify active correlations
2. Check if their average gradient is close to constant
3. If not, the solution isn't optimal

But for practical optimization, the coordinate descent should handle this naturally.

```python
import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-14:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-14:
                    break
    return x


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _exact_c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _pair_coeff(h, i, n, s_range, nc):
    idx1 = i - s_range
    idx2 = i + s_range
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(nc)
    if np.any(m1):
        a[m1] = 1.0 - h[idx1[m1]]
    if np.any(m2):
        a[m2] -= h[idx2[m2]]
    return a


def _bisect_delta(corr, d, lo, hi):
    for _ in range(60):
        if hi - lo < 1e-15:
            break
        mid = (lo + hi) / 2.0
        vals = corr + mid * d
        mv = np.max(vals)
        active = vals >= mv - 1e-10
        if not np.any(active):
            active = np.zeros(len(vals), dtype=bool)
            active[np.argmax(vals)] = True
        da = d[active]
        if np.min(da) > 1e-15:
            hi = mid
        elif np.max(da) < -1e-15:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0


def _cd_sweep(h, rng, mode, s_range, nc, n):
    corr = _corr(h)
    order = rng.permutation(n)
    for i in order:
        if mode == 0:
            j = rng.integers(0, n)
            if j == i:
                j = (i + 1) % n
        elif mode == 1:
            j = (i + 1) % n
        else:
            j = (i + n // 2) % n
        a_i = _pair_coeff(h, i, n, s_range, nc)
        a_j = _pair_coeff(h, j, n, s_range, nc)
        d = a_i - a_j
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi - 1e-14:
            continue
        delta = _bisect_delta(corr, d, lo, hi)
        if abs(delta) > 1e-14:
            h[i] += delta
            h[j] -= delta
            corr += delta * d
    return h, _exact_c5(h)


def _cd_run(h, deadline, seed, max_sweeps=500000, stall_limit=300):
    n = len(h)
    nc = 2 * n - 1
    s_range = np.arange(nc) - (n - 1)
    rng = np.random.default_rng(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    pert_count = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        mode = sweep % 3
        h, c5 = _cd_sweep(h, rng, mode, s_range, nc, n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                pert_count += 1
                scale = max(0.001, 0.03 * (0.7 ** pert_count))
                h = _project(h + rng.normal(0, scale, n), n * 0.5)
                stall = 0
                if pert_count > 20:

    return best_h, best_c5


def _slsqp_f(z, n):
    return z[n]


def _slsqp_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _slsqp_eq_f(z, n):
    return np.sum(z[:n]) - n * 0.5


def _slsqp_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _slsqp_ineq_f(z, n):
    h, t = z[:n], z[n]
    c = np.correlate(h, 1.0 - h, mode='full')
    return t - c


def _slsqp_ineq_g(z, n):
    h = z[:n]
    nc = 2 * n - 1
    s_range = np.arange(nc) - (n - 1)
    J = np.zeros((nc, n + 1))
    for r in range(nc):
        J[r, :n] = _pair_coeff(h, 0, n, s_range, nc) * 0.0
        # Build row r of Jacobian
        js = np.arange(n)
        s = s_range[r]
        pd = np.zeros(n)
        gs = js - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = js + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        J[r, :n] = pd
        J[r, n] = 1.0
    return J


def _slsqp_solve(h_init, n, maxiter=300, ftol=1e-13):
    corr = _corr(h_init)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h_init, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _slsqp_ineq_f, 'jac': _slsqp_ineq_g, 'args': (n,)}
    ]
    try:
        res = minimize(_slsqp_f, z0, jac=_slsqp_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h_init.copy(), _exact_c5(h_init)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _make_fourier(n, seed):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) / n
    h = np.ones(n) * 0.5
    for k in range(1, 6):
        amp = rng.uniform(0.05, 0.25) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * x + phase)
    return _project(h, n * 0.5)


def _make_binary(n, seed):

def _lp_direction(h, n, radius, deadline):
    nc = 2 * n - 1
    s_range = np.arange(nc) - (n - 1)
    corr = _corr(h)
    top_k = min(20, nc)
    top_idx = np.argsort(-corr)[:top_k]
    J = np.zeros((top_k, n))
    b = np.zeros(top_k)
    for ii, r in enumerate(top_idx):
        J[ii, :] = _pair_coeff(h, 0, n, s_range, nc)[r] * 0.0
        s = s_range[r]
        js = np.arange(n)
        pd = np.zeros(n)
        gs = js - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = js + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        J[ii, :] = pd
        b[ii] = -corr[r]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = J
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for jj in range(n):
        lo_b = max(-radius, -h[jj])
        hi_b = min(radius, 1.0 - h[jj])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:

    except Exception:

    return None
```
