   - If all active slopes > 0: go left (t_hi = t_mid)
   - If all active slopes < 0: go right (t_lo = t_mid)
   - Otherwise: t_mid is (approximately) the minimum.

This gives O(log(1/eps) * n) work per pair, with eps being the desired accuracy in t. With 30 iterations of bisection, eps ~ t_max / 2^30, which is more than enough.

But actually, for a piecewise linear function, the bisection might not converge to the exact breakpoint in finite iterations (it converges to a subgradient containing 0, which is a flat region or a single point). In practice, 30 iterations gives accuracy of t_max / 2^30, which is way more than needed.

def find_optimal_t(base, slopes, t_max, n_evals=40):

        return 0.0

    f0 = float(np.max(base))
    # Check direction
    active0 = np.where(base >= f0 - 1e-14)[0]
    if len(active0) > 0:
        min_slope = float(np.min(slopes[active0]))
        if min_slope > 1e-12:
            return 0.0  # increasing at 0
    
    # Check t_max
    f_end = float(np.max(base + t_max * slopes))
    if f_end < f0:
        # Function is decreasing throughout (or minimum is at t_max)
        # Check slope at t_max
        f_end_val = base + t_max * slopes
        active_end = np.where(f_end_val >= f_end - 1e-14)[0]
        if len(active_end) > 0:
            max_slope_end = float(np.max(slopes[active_end]))
            if max_slope_end < -1e-12:
                return t_max  # still decreasing at t_max
    
    # Bisection for interior minimum
    lo, hi = 0.0, t_max
    for _ in range(n_evals):
        mid = (lo + hi) / 2.0
        f_mid = base + mid * slopes
        f_mid_val = float(np.max(f_mid))
        active = np.where(f_mid >= f_mid_val - 1e-14)[0]
        if len(active) == 0:
            break
        min_s = float(np.min(slopes[active]))
        max_s = float(np.max(slopes[active]))
        if min_s > 1e-12:
            # f is increasing at mid, minimum is to the left
            hi = mid
        elif max_s < -1e-12:
            # f is decreasing at mid, minimum is to the right
            lo = mid
        else:
            # 0 is in the subgradient, we've found the minimum (or a flat region)
            break
    
    t_opt = (lo + hi) / 2.0
    return t_opt

def anneal_pairs(h, deadline, rng, t_init=0.005, t_final=0.0001, max_iter=50000):
    """Simulated annealing with pair moves."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    base = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cur_c5 = float(np.max(base))
    best_h = h.copy()
    best_c5 = cur_c5
    t = t_init

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        if it % 100 == 0:
            t = t_init * (t_final / t_init) ** (it / max_iter)

        # Pick random pair
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        # Random direction and magnitude
        if rng.random() < 0.5:
            t_max = min(1.0 - h[a], h[b])
            delta_dir = lambda tt: (a, b, tt)
        else:
            t_max = min(1.0 - h[b], h[a])
            delta_dir = lambda tt: (b, a, tt)

        if t_max <= 1e-10:
            continue

        # Pick t as fraction of t_max
        frac = rng.random()
        t_step = frac * t_max
        if t_step < 1e-8:
            t_step = rng.uniform(1e-6, t_max)

        aa, bb, tt = delta_dir(t_step)
        if tt < 1e-10 or tt > min(1.0 - h[aa], h[bb]) + 1e-10:
            continue

        slopes = np.zeros(2 * n - 1)
        slopes[aa:aa + n] += 1.0 - 2.0 * h
        slopes[bb:bb + n] -= 1.0 - 2.0 * h

        new_base = base + tt * slopes
        new_base[2 * aa] -= tt * tt
        new_base[2 * bb] -= tt * tt
        new_base[aa + bb] += 2.0 * tt * tt
        new_c5 = float(np.max(new_base))
        delta = new_c5 - cur_c5

        accept = False
        if delta < -1e-13:
            accept = True
        elif t > 1e-8:
            accept = rng.random() < np.exp(-delta / t)

        if accept:
            h[aa] += tt
            h[bb] -= tt
            base = new_base
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    h = project_h(h, target)
    base_final = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / n)
    final_c5 = float(np.max(base_final))
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, base_final, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
