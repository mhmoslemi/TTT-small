For a pair exchange: increase h[i] by δ and decrease h[j] by δ (preserving sum and box). The change in the r-th correlation value is:
- For lag d ≥ 0: corr_r = Σ h[k-d](1-h[k]) for k=d..n-1
  - If i ≥ d and j ≥ d: change = (1-h[j])δ - h[i](1-h[j-i+d])... this gets complex
  - Actually, the gradient approach is simpler: grad_r[i] and grad_r[j] give the exact change per unit

For the r-th lag with offset d = r - (n-1):
- If d ≥ 0, m = n - d:
  - grad_r[i] for i in [d, n): = 1 - h[i-d]
  - grad_r[i] for i in [0, d): = -h[i+d]
- If d < 0, s = -d, m = n - s:
  - grad_r[i] for i in [0, m): = 1 - h[i+s]
  - grad_r[i] for i in [m, n): = -h[i-s]

So for a pair move (i gets +δ, j gets -δ):
Δcorr_r = δ * (grad_r[i] - grad_r[j])

For the max objective C = max_r corr_r * dx:
If we only care about the active set A (lags where corr_r is near max), the change is:
C_new ≈ max(r in A) (corr_r + δ*(grad_r[i] - grad_r[j])) * dx

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : 0 <= x_i <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _all_grads(h, top_k=10):
    """Compute gradients for top-k scoring lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scores = corr
    max_s = float(np.max(scores))
    # Get top-k active lags
    active_idx = np.argsort(scores)[::-1][:top_k]
    # Also include near-active
    near_mask = scores >= max_s - dx * 0.01
    active_set = set(active_idx.tolist())
    for i in np.where(near_mask)[0]:
        active_set.add(int(i))
    active_set = sorted(active_set)
    
    grads = []
    for r in active_set:
        d = r - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            m = n - d
            grad[d:] += 1.0 - h[:m]
            grad[:m] -= h[d:]
        else:
            s = -d
            m = n - s
            grad[:m] += 1.0 - h[s:]
            grad[s:] -= h[:m]
        grads.append((r, grad))
    return scores, max_s, grads

def _line_search(h, p, c5_cur, alpha_max, n_grid=20):
    """Line search along direction p."""
    if alpha_max < 1e-14:
        return 0.0, c5_cur
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    # Golden section refine
    lo_t = ts[max(0, idx - 1)]
    hi_t = ts[min(n_grid - 1, idx + 1)]
    if hi_t <= lo_t + 1e-12:
        lo_t, hi_t = max(0.0, best_t - ts[1]*0.5), min(alpha_max, best_t + ts[1]*0.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    f1, f2 = _f(c1), _f(c2)
    for _ in range(30):
        if hi_t - lo_t < 1e-12:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_direction(h, scores, radius):
    """LP direction to minimize active scores."""
    n = len(h)
    max_s = float(np.max(scores))
    # Focus on top active constraints
    active = np.where(scores >= max_s - max_s * 0.02)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    
    G = np.zeros((len(active), n))
    for ii, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[ii, d:] += 1.0
            G[ii, d:] -= h[:m]
            G[ii, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            G[ii, :m] += 1.0
            G[ii, :m] -= h[g:]
            G[ii, g:] -= h[:m]
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_optimize(h, deadline, tau_init=15.0, lr_init=0.02, mom_beta=0.9):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        dx = 2.0 / n
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        # Weighted gradient
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            wr = w[r]
            if wr < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                grad[d:] += wr * (1.0 - h[:m])
                grad[:m] -= wr * h[d:]
            else:
                g = -d
                m = n - g
                grad[:m] += wr * (1.0 - h[g:])
                grad[g:] -= wr * h[:m]
        grad *= dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.03, 0.15)
            tau = min(tau * 1.02, 200.0)
        else:
            stall += 1
            lr *= 0.92
            if stall > 60:
                break
            if stall % 25 == 0:
                tau = max(tau_init, tau * 0.8)
    return best_h, best_c5

def _anneal_search(h_best, deadline, seed=42, T_init=0.002, T_min=1e-5, cool=0.998):
    """Stochastic annealing with mass-preserving pair moves."""
    n = len(h_best)
    target = n / 2.0
    h = h_best.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    epoch = 0
    no_improve = 0
    
    while time.time() < deadline:
        epoch += 1
        # Recompute gradients periodically
        if epoch % 50 == 1:
            scores, max_s, grad_list = _all_grads(h, top_k=12)
            # Build aggregated active gradient
            A = np.zeros(n)
            for r, g in grad_list:
                w = max(0, scores[r] - max_s + max_s * 0.01)
                A += w * g
            A -= np.mean(A)
            A_scale = float(np.max(np.abs(A)))
            if A_scale < 1e-10:
                A_scale = 1.0
        else:
            A_scale = 1.0
        
        # Propose a pair move
        for trial in range(20):
            # Choose i (to increase) and j (to decrease)
            # Biased: increase where A is negative, decrease where A is positive
            if rng.random() < 0.3:
                # Uniform random pair
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            else:
                # Biased pair
                # i: prefer negative A
                probs_i = np.exp(-A / (A_scale * 0.5 + 1e-10))
                probs_i[h >= 1.0 - 1e-10] = 1e-10  # can't increase these
                probs_i = np.maximum(probs_i, 1e-10)
                probs_i /= probs_i.sum()
                i = rng.choice(n, p=probs_i)
                
                # j: prefer positive A
                probs_j = np.exp(A / (A_scale * 0.5 + 1e-10))
                probs_j[h <= 1e-10] = 1e-10  # can't decrease these
                probs_j = np.maximum(probs_j, 1e-10)
                probs_j /= probs_j.sum()
                j = rng.choice(n, p=probs_j)
            
            if i == j:
                continue
            
            # Step size
            max_inc = 1.0 - h[i]
            max_dec = h[j]
            if max_inc < 1e-12 or max_dec < 1e-12:
                continue
            delta = min(max_inc, max_dec) * rng.uniform(0.1, 1.0)
            if delta < 1e-10:
                continue
            
            # Apply move
            h[i] += delta
            h[j] -= delta
            
            # Check feasibility (should be fine by construction)
            if h[i] < -1e-12 or h[i] > 1.0 + 1e-12 or h[j] < -1e-12 or h[j] > 1.0 + 1e-12:
                h[i] -= delta
                h[j] += delta
                continue
            
            new_c5 = _exact_c5(h)
            delta_c = new_c5 - cur_c5
            
            if delta_c < 0:
                # Accept (improvement)
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
            else:
                # Metropolis acceptance
                if T > T_min and delta_c < best_c5 - cur_c5 + 0.003:
                    accept_prob = np.exp(-delta_c / T)
                    if rng.random() < accept_prob:
                        cur_c5 = new_c5
                        no_improve += 1
                        break
                # Reject
                h[i] -= delta
                h[j] += delta
        
        # Cool
        T = max(T_min, T * cool)
        
        # If stuck for too long, kick
        if no_improve > 2000:
            # Add small random perturbation
            kick = rng.normal(0, 0.01, n)
            kick -= np.mean(kick)
            h_kick = h + kick
            h_kick = _project(h_kick, target)
            c5_kick = _exact_c5(h_kick)
            if c5_kick < cur_c5 + 0.005:
                h = h_kick
                cur_c5 = c5_kick
            no_improve = 0
            T = max(T * 2, T_init * 0.5)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _row_grad(h, r):
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        g[d:] += 1.0 - h[:m]
        g[:m] -= h[d:]
    else:
        s = -d
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    return g

def _line_search(h, p, c5_cur, alpha_max, n_grid=16):
    if alpha_max < 1e-14:
        return 0.0, c5_cur
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 1)]
    hi_t = ts[min(n_grid - 1, idx + 1)]
    if hi_t <= lo_t + 1e-12:
        step = alpha_max / n_grid
        lo_t = max(0.0, best_t - step)
        hi_t = min(alpha_max, best_t + step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    def _eval(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    f1, f2 = _eval(c1), _eval(c2)
    for _ in range(25):
        if hi_t - lo_t < 1e-13:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _eval(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _eval(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _eval(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_dir(h, scores, radius, n_active=5):
    n = len(h)
    max_s = float(np.max(scores))
    active = np.argsort(scores)[::-1][:n_active]
    G = np.zeros((n_active, n))
    for ii, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            G[ii, d:] += 1.0
            G[ii, d:] -= h[:m]
            G[ii, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            G[ii, :m] += 1.0
            G[ii, :m] -= h[g:]
            G[ii, g:] -= h[:m]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_local(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 50:

        scores = _scores(h)
        p = _lp_dir(h, scores, radius, n_active=5)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        alpha_max = 1e10
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(min(alpha_max, 2.0), 1e-10)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-13:

            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.4)

def _grad_opt(h, deadline, tau=15.0, lr=0.02, beta=0.88):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        scores = _scores(h)
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] < 1e-15:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                grad[d:] += w[r] * (1.0 - h[:m])
                grad[:m] -= w[r] * h[d:]
            else:
                g = -d
                m = n - g
                grad[:m] += w[r] * (1.0 - h[g:])
                grad[g:] -= w[r] * h[:m]
        grad *= 2.0 / n
        mom = beta * mom + (1.0 - beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
            tau = min(tau * 1.01, 150.0)
        else:
            stall += 1
            lr *= 0.93
            if stall > 50:
                break
    return best_h, best_c5

def _anneal(h_init, deadline, seed=42, T_init=0.004, T_min=5e-6, cool=0.997):

    target = n / 2.0
    h = h_init.copy()

    no_imp = 0
    epoch = 0

        epoch += 1
        if epoch % 30 == 1:
            scores = _scores(h)
            max_s = float(np.max(scores))
            active = np.where(scores >= max_s - max_s * 0.015)[0]
            if len(active) == 0:

            if len(active) > 10:
                active = np.argsort(scores)[::-1][:10]
            A = np.zeros(n)
            for r in active:
                A += _row_grad(h, int(r))
            A -= np.mean(A)
            A_scale = float(np.max(np.abs(A)))
            if A_scale < 1e-10:
                A_scale = 1.0
        else:
            A_scale = 1.0
        for _ in range(15):
            if rng.random() < 0.25:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            else:
                pi = np.exp(-A / (A_scale * 0.4 + 1e-10))
                pi[h > 0.999] = 1e-12
                pi = np.maximum(pi, 1e-12)
                pi /= pi.sum()
                i = int(rng.choice(n, p=pi))
                pj = np.exp(A / (A_scale * 0.4 + 1e-10))
                pj[h < 0.001] = 1e-12
                pj = np.maximum(pj, 1e-12)
                pj /= pj.sum()
                j = int(rng.choice(n, p=pj))
            if i == j:
                continue
            max_inc = 1.0 - h[i]
            max_dec = h[j]
            if max_inc < 1e-12 or max_dec < 1e-12:
                continue
            scale = rng.uniform(0.05, 1.0)
            delta = min(max_inc, max_dec) * scale
            if delta < 1e-10:

            new_c5 = _exact_c5(h)

            ck = _exact_c5(h_k)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_start(n, seed):
    rng = np.random.default_rng(seed)
    candidates = []
    # Alternating pattern
    h_alt = np.tile([0.0, 1.0], n // 2 + 1)[:n]
    if n % 2 == 0:
        h_alt = np.tile([0.5, 0.5], n // 2)
        h_alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    candidates.append(_project(h_alt.copy(), n / 2.0))
    # Block pattern
    h_blk = np.zeros(n)
    h_blk[:n // 2] = 1.0
    candidates.append(_project(h_blk.copy(), n / 2.0))
    # Three-level
    h_3 = np.zeros(n)
    h_3[::3] = 1.0
    h_3[1::3] = 0.5
    candidates.append(_project(h_3.copy(), n / 2.0))
    # Random
    h_r = rng.random(n)
    candidates.append(_project(h_r, n / 2.0))
    # Piecewise sinusoidal
    x = np.linspace(0, 2, n)
    h_sin = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    candidates.append(_project(h_sin, n / 2.0))
    best_h = None
    best_c = 1.0
    for h_c in candidates:
        c = _exact_c5(h_c)
        if c < best_c:
            best_c = c
            best_h = h_c.copy()
    return best_h, best_c

    # Phase 6: Second annealing pass (10%)
    d6 = t0 + budget_s * 0.84
    h, c5 = _anneal(best_h, d6, seed=seed + 9000, T_init=0.002, T_min=2e-6, cool=0.9985)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 7: Pair polish + gradient (8%)
    d7 = t0 + budget_s * 0.92
    h, c5 = _pair_polish(best_h, d7, max_passes=30)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d7:
        h, c5 = _grad_opt(best_h, d7, tau=40.0, lr=0.008, beta=0.92)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 8: Final LP (7%)
    d8 = t0 + budget_s * 0.99
    h, c5 = _lp_local(best_h, d8, radius_init=0.04)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d8:
        h, c5 = _lp_local(best_h, d8, radius_init=0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final feasibility check
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
