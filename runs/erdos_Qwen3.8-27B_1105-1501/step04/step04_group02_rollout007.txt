    return np.correlate(h, 1.0 - h, mode='full')

def _gradient_matrix(h):
    """Compute gradient of all lag scores w.r.t. h. Shape (2n-1, n)."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))

            G[r, d:] += 1.0 - h[:cnt]
        else:
            s = -d
            cnt = n - s
            G[r, :cnt] += 1.0 - h[s:]
            G[r, s:] += -h[:cnt]
    return G

    G = _gradient_matrix(h)
    jac = np.empty((2 * n - 1, n + 1))
    jac[:, :n] = -G
    jac[:, n] = 1.0
    return jac

    return x[:n].sum() - n / 2.0

def _area_resample(h_source, n_new):
    """Area-preserving resampling."""
    n_old = len(h_source)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # Each source bin i covers [i*dx_old, (i+1)*dx_old]
    # Each target bin j covers [j*dx_new, (j+1)*dx_new]
    h_new = np.zeros(n_new)
    for j in range(n_new):
        j_start = j * dx_new
        j_end = (j + 1) * dx_new
        # Find overlapping source bins
        i_start = int(np.floor(j_start / dx_old))
        i_end = int(np.ceil(j_end / dx_old))
        for i in range(max(0, i_start), min(n_old, i_end + 1)):
            i_s = i * dx_old
            i_e = (i + 1) * dx_old
            overlap = min(j_end, i_e) - max(j_start, i_s)
            if overlap > 0:
                h_new[j] += h_source[i] * overlap / dx_new
    return _project(np.clip(h_new, 0, 1), n_new / 2.0)

    hi = float(v.max()) + 2.0

        else:

            c = n - s
