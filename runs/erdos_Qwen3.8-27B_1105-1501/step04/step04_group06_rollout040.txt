def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:

    n = len(h)

        km = k - s
        valid_km = (km >= 0) & (km < n)
        G[r, valid_km] -= h[km[valid_km]]

            f1 = _exact_c5(_project(h + c1 * p, n * 0.5))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, n * 0.5))
    a = (lo + hi) / 2.0
    c5_t = _exact_c5(_project(h + a * p, n * 0.5))
    if c5_t < best_c5:
        best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, n, deadline, radius_init=0.15, stall_limit=60):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    radius = radius_init
    stall = 0
    for _ in range(20000):
        if time.time() > deadline:
            break
        if stall > stall_limit:
            break
        scores = _corr(h)
        p, t_val = _lp_direction(h, scores, radius)
        if p is None:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _lp_line_search(h, p, n, cur_c5)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, n * 0.5)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5


def _lse_step(h, n, alpha, maxiter=100):
    def _obj(x):
        hh = np.asarray(x, dtype=float)
        corr = _corr(hh)
        ms = float(np.max(corr))
        vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
        exp_v = np.exp(vals)
        f = ms + (1.0 / alpha) * np.log(float(np.sum(exp_v)))
        return f

    def _grad(x):
        hh = np.asarray(x, dtype=float)
        corr = _corr(hh)
        ms = float(np.max(corr))
        vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
        exp_v = np.exp(vals)
        w = exp_v / float(np.sum(exp_v))
        G = _grad_matrix(hh)
        grad = G.T @ w
        grad -= np.mean(grad)
        return grad

    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, args=(),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 50, 'iprint': 0})
        x = _project(res.x, target)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _slsqp_solve(h_init, n, maxiter=300):
    def _epi_f(z, nn):
        return z[nn]

    def _epi_g(z, nn):
        g = np.zeros(nn + 1)
        g[nn] = 1.0
        return g

    def _eq_f(z, nn):
        return np.sum(z[:nn]) - nn * 0.5

    def _eq_g(z, nn):
        g = np.ones(nn + 1)
        g[nn] = 0.0
        return g

    def _ineq_f(z, nn):
        hh = z[:nn]
        return z[nn] - np.max(_corr(hh))

    corr0 = _corr(h_init)
    t0 = float(np.max(corr0)) + 1e-4
    z0 = np.concatenate([h_init, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 50.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_f, 'jac': _eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq_f, 'args': (n,)}
    ]
    try:
        res = minimize(_epi_f, z0, jac=_epi_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h_init.copy(), _exact_c5(h_init)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Binary greedy search (25% budget)
    d1 = t0 + budget_s * 0.25
    h_bin, c5_bin = _binary_greedy(h, n0, d1, K=6, max_iters=60000, restarts=8)
    if c5_bin < best_c5:
        best_c5, best_h, best_n = c5_bin, h_bin.copy(), n0

    # Phase 2: LSE smoothing on best found (10% budget)
    d2 = t0 + budget_s * 0.35
    h_cur = _project(best_h, best_n * 0.5)
    for alpha in [30.0, 60.0, 120.0]:
        if time.time() > d2:
            break
        h_cur, c5_lse = _lse_step(h_cur, best_n, alpha, maxiter=60)
        if c5_lse < best_c5:
            best_c5, best_h = c5_lse, h_cur.copy()

    # Phase 3: LP trust region (25% budget)
    d3 = t0 + budget_s * 0.60
    h_lp, c5_lp = _lp_loop(best_h, best_n, d3, radius_init=0.15, stall_limit=50)
    if c5_lp < best_c5:
        best_c5, best_h = c5_lp, h_lp.copy()

    # Phase 4: Multi-resolution search (25% budget)
    d4 = t0 + budget_s * 0.85
    n_candidates = [n0, n0 + 16, n0 - 16, n0 + 32, n0 - 32, 96, 128, 48]
    n_candidates = sorted(set([x for x in n_candidates if 32 <= x <= 256 and x % 2 == 0]))
    for n_try in n_candidates:
        if time.time() > d4:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        # Quick binary search
        sub_d1 = min(d4, time.time() + budget_s * 0.02)
        h_br, c5_br = _binary_greedy(h_r, n_try, sub_d1, K=5, max_iters=20000, restarts=3)
        if c5_br < _exact_c5(h_r):
            h_r = _project(h_br, n_try * 0.5)
        # LSE
        sub_d2 = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _lse_step(h_r, n_try, 50.0, maxiter=40)
        # LP
        sub_d3 = min(d4, time.time() + budget_s * 0.025)
        h_lp2, c5_lp2 = _lp_loop(h_r, n_try, sub_d3, radius_init=0.1, stall_limit=30)
        if c5_lp2 < c5_r:
            h_r, c5_r = h_lp2, c5_r
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 5: Final polish (15% budget)
    d5 = t0 + budget_s * 0.98
    # SLSQP
    h_s, c5_s = _slsqp_solve(best_h, best_n, maxiter=200)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    # LP final
    h_lp3, c5_lp3 = _lp_loop(best_h, best_n, d5, radius_init=0.05, stall_limit=80)
    if c5_lp3 < best_c5:
        best_c5, best_h = c5_lp3, h_lp3.copy()

    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
