import numpy as np

def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def scores_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_rows(h, active_idx):
    n = len(h)
    dx = 2.0 / n
    m = len(active_idx)
    G = np.zeros((m, n))
    for k in range(m):
        r = int(active_idx[k])
        d = r - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[k, d:d+length] += dx * (1.0 - h[:length])
                G[k, :length] -= dx * h[d:d+length]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[k, :length] += dx * (1.0 - h[g:])
                G[k, g:] -= dx * h[:length]
    return G

def roughness_matrix(n, rho):
    if rho <= 0:
        return None
    D = np.zeros(n)
    Dm = np.zeros(n)
    if n < 2:
        return rho * np.eye(n)
    D[0] = -D[0] + D[1]
    D[-1] = -D[-1] - D[-2]
    if n > 2:
        D[1:-1] = D[1:-1] - 2.0 * D[1:-1] + D[2:] - D[:-2] + D[1:-1] - 2.0 * D[1:-1] + D[1:-1]
    # Simpler: use (I - S) as roughness where S is shift
    # Actually use D^T D where D is first difference
    Dmat = np.zeros((n, n))
    for i in range(n - 1):
        Dmat[i, i] = 1.0
        Dmat[i, i+1] = -1.0
    return rho * (Dmat.T @ Dmat)

def active_ls_direction(h, scores, active_idx, q, target_level, ridge, rough):
    n = len(h)
    m = len(active_idx)
    G = build_grad_rows(h, active_idx)
    if not np.all(np.isfinite(G)):
        return None
    r = target_level - scores[active_idx]
    # H = G^T diag(q) G + ridge*I
    H = (G.T * q) @ G
    H += ridge * np.eye(n)
    rr = roughness_matrix(n, rough)
    if rr is not None:
        H += rr
    # KKT: [H  1][δ] = [G^T diag(q) r]
    #       [1^T 0][μ]   [0            ]
    KKT = np.zeros((n+1, n+1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n + 1)
    rhs[:n] = (G.T * q) @ r
    try:
        sol = np.linalg.solve(KKT, rhs)
        delta = sol[:n]
    except np.linalg.LinAlgError:
        try:
            sol, _, _, _ = np.linalg.lstsq(KKT, rhs, rcond=1e-12)
            delta = sol[:n]
        except Exception:
            return None
    if not np.all(np.isfinite(delta)):
        return None
    delta -= delta.mean()
    if np.max(np.abs(delta)) < 1e-12:
        return None
    return delta

def weighted_grad_direction(h, scores, active_idx, q):
    n = len(h)
    G = build_grad_rows(h, active_idx)
    p = G.T @ q
    p -= p.mean()
    if np.max(np.abs(p)) < 1e-12:
        return None
    return p

def line_search(h, p, trust=0.35):
    n = len(h)
    pmax = float(np.max(np.abs(p)))
    if pmax < 1e-14:
        return 0.0, exact_c5(h)
    p_unit = p / pmax
    alpha_max = trust
    for j in range(n):
        if p_unit[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p_unit[j])
        elif p_unit[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p_unit[j]))
    alpha_max = max(alpha_max, 1e-12)
    # Grid search
    n_samp = 12
    alphas = np.linspace(0, alpha_max, n_samp)
    cur_c5 = exact_c5(h)
    vals = np.array([exact_c5(h + a * p_unit) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a * pmax, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * p_unit)
    f2 = exact_c5(h + c2 * p_unit)
    for _ in range(20):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2; f2 = f1; c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * p_unit)
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * p_unit)
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * p_unit)
    if opt_c5 < best_c5:
        return opt_a * pmax, opt_c5
    return best_a * pmax, best_c5

def lp_direction(h, scores, top_k=None, radius=0.15):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
    else:
        idx = np.arange(n_lags)
        scores_k = scores
    G = build_grad_rows(h, idx.astype(int))
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def smooth_perturb(h, scale, rng, n_modes=4):
    n = len(h)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1) * scale / k
        pert += amp * np.cos(2 * np.pi * k * x + phase)
    h_pert = h + pert
    h_pert -= h_pert.mean()
    h_pert += h.mean()
    return project(h_pert, n / 2.0)

def quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return project(h, n / 2.0)

def binary_sa(n, deadline, rng, n_iters=30000):
    target = n // 2
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:target]] = 1.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.04
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(n_iters):
        if time.time() > deadline:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / T):
            cur_c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5

            h_div = project(h_div, n_div / 2.0)
            
            sub_dl = min(dl2, time.time() + budget_s * 0.025)
            # Alternate between variants
            var = ['ls_wide', 'lp', 'ls_tight'][attempt % 3]
            h_d, c5_d = local_optimize(h_div, sub_dl, variant=var, max_iters=1000)
            if c5_d < best_c5:
                best_c5 = c5_d
                best_h = h_d.copy()
                best_n = n_div

    # Phase 3: Binary SA at select resolutions
    dl3 = min(t0 + budget_s * 0.78, hard_deadline)
    for n_bin in [96, 128]:
        if time.time() > dl3:
            break
        sub_dl = min(dl3, time.time() + budget_s * 0.03)
        h_b, c5_b = binary_sa(n_bin, sub_dl, rng, n_iters=20000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_bin
            # Polish with active-LS
            if time.time() < dl3:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.02)
                h_bp, c5_bp = local_optimize(h_b, sub_dl2, variant='ls_wide', max_iters=600)
                if c5_bp < best_c5:
                    best_c5 = c5_bp
                    best_h = h_bp.copy()

    # Phase 4: Tight micro-optimization
    dl4 = min(t0 + budget_s * 0.90, hard_deadline)
    for var in ['ls_tight', 'ls_micro', 'ls_tight', 'ls_micro']:
        if time.time() > dl4:
            break
        sub_dl = min(dl4, time.time() + budget_s * 0.02)
        h_m, c5_m = local_optimize(best_h, sub_dl, variant=var, max_iters=800)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    # Phase 5: Final perturbation polish
    dl5 = min(hard_deadline, t0 + budget_s * 0.96)
    scale = 0.01
    for attempt in range(80):
        if time.time() > dl5:
            break
        h_pert = smooth_perturb(best_h, scale, rng, n_modes=3)
        sub_dl = min(dl5, time.time() + budget_s * 0.01)
        h_try, c5_try = local_optimize(h_pert, sub_dl, variant='ls_micro', max_iters=300)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.15)

    # Final projection and evaluation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
