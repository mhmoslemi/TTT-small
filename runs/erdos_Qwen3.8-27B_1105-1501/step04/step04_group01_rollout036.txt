Let me also think about the coordinate-wise optimization more carefully.

If I fix all variables except h[i], the objective max_r f_r(h) as a function of h[i] is:
max_r (a_r * h[i] + b_r)

where a_r = G[r, i] (the i-th component of the gradient of f_r) and b_r = f_r(h) - a_r * h[i].

This is the maximum of linear functions of h[i], which is a convex piecewise linear function. Its minimum on [0, 1] (or whatever the feasible range is) occurs at a breakpoint or at the boundary.

The breakpoints occur where two of the linear functions are equal: a_r * h[i] + b_r = a_s * h[i] + b_s, i.e., h[i] = (b_s - b_r) / (a_r - a_s) for a_r ≠ a_s.

So the exact coordinate-wise optimum can be found by:
1. Computing all breakpoints in [0, 1]
2. Evaluating the objective at each breakpoint and at 0 and 1
3. Taking the best one

But we also need to satisfy the sum constraint. If we change h[i] from its current value to h[i]_new, we need to adjust another variable. So pure coordinate descent doesn't work directly.

However, we can do **pairwise** optimization: fix all except h[i] and h[j], and optimize over the 2D space (h[i], h[j]) subject to h[i] + h[j] = h[i]_old + h[j]_old (to maintain the sum). This reduces to optimizing over a single parameter.

For a pair (i, j), let h[i] = h[i]_0 + t and h[j] = h[j]_0 - t (to maintain sum). Then the objective is:
max_r (a_r * (h[i]_0 + t) + c_r * (h[j]_0 - t) + b_r)
= max_r ((a_r - c_r) * t + (a_r * h[i]_0 + c_r * h[j]_0 + b_r))

where a_r = G[r,i], c_r = G[r,j].

This is again a convex piecewise linear function of t, with breakpoints at t = (d2 - d1) / (w1 - w2) where w1 = a_r - c_r, d1 = a_r*h[i]_0 + c_r*h[j]_0 + b_r.

def _exact_pair_optimize(h, i, j, dx):
    """Find the exact optimal t for h[i]+=t, h[j]-=t maintaining the sum."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    # For each lag r, f_r = corr[r] * dx
    # After change: f_r' = corr[r]*ax - dx * t * G[r,i] + dx * t * G[r,j]
    # = (corr[r] - t*G[r,i] + t*G[r,j]) * dx
    # = (corr[r] + t*(G[r,j] - G[r,i])) * dx
    # So a_r = G[r,j] - G[r,i], and the objective is max_r (corr[r] + t*a_r) * dx
    
    G = _grad_matrix(h)  # This is expensive, compute once
    w = G[:, j] - G[:, i]  # coefficient of t for each lag
    b = corr  # base value for each lag
    
    # Objective: max_r (b[r] + t * w[r]) * dx
    # This is convex in t. Minimum is where the subgradient contains 0.
    # The derivative of the max is w[argmax_r (b[r] + t*w[r])]
    # At the optimum, either w[argmax] = 0 or we're at a breakpoint.
    
    # Feasible range of t:
    t_min = max(-h[i], h[j] - 1.0)  # h[i]+t >= 0 and h[j]-t <= 1
    t_max = min(1.0 - h[i], h[j])    # h[i]+t <= 1 and h[j]-t >= 0
    
    if t_min >= t_max:
        return 0.0
    
    # Find breakpoints: t where b[r] + t*w[r] = b[s] + t*w[s]
    # t = (b[s] - b[r]) / (w[r] - w[s])
    # Collect all breakpoints in [t_min, t_max]
    breakpoints = [t_min, t_max]
    nr = 2*n - 1
    for r in range(nr):
        for s in range(r+1, nr):
            if w[r] != w[s]:
                t_bp = (b[s] - b[r]) / (w[r] - w[s])
                if t_min <= t_bp <= t_max:
                    breakpoints.append(t_bp)
    
    breakpoints = sorted(set([round(bp, 15) for bp in breakpoints]))
    
    # Evaluate at all breakpoints
    best_t = t_min
    best_val = np.inf
    for t in breakpoints:
        val = float(np.max(b + t * w)) * dx
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

import numpy as np
from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G


def _line_search(h, d, dx, trust):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust)
    t_hi = min(t_hi, trust)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    nv = 40
    grid = np.linspace(t_lo, t_hi, nv)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < nv - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(30):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0

    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _lp_direction(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_optimize(h, deadline, radius_init=0.08, min_radius=1e-5):

    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()

    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break

        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:

            continue

def _softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = float(np.max(corr))
    exp_s = np.exp(tau * (corr - max_s))
    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    grad = w @ G
    grad -= np.mean(grad)
    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _shift_perturb(h, shift_idx, rng):
    n = len(h)
    h_new = h.copy()
    s = int(shift_idx)
    h_new = np.roll(h, s)
    h_new = _project(h_new, n / 2.0)
    noise = rng.normal(0, 0.05, n)
    h_new = _project(h_new + noise, n / 2.0)
    return h_new


def _segment_flip(h, rng):
    n = len(h)
    h_new = h.copy()
    a = int(rng.integers(0, n // 2))
    length = int(rng.integers(n // 8, n // 3))
    b = min(a + length, n)
    seg_sum = float(np.sum(h_new[a:b]))
    seg_len = b - a
    if abs(seg_sum - seg_len / 2.0) < 0.3 * seg_len:
        h_new[a:b] = 1.0 - h_new[a:b]
        h_new = _project(h_new, n / 2.0)
    else:
        h_new = h.copy()
        idx = rng.integers(0, n)
        h_new[idx] = 1.0 - h_new[idx]
        h_new = _project(h_new, n / 2.0)
    return h_new
