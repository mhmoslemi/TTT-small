import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _grad_row(h, d):
    """Compute gradient of score at lag d with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[:m] = (1.0 - h[d:d+m])
            g[d:] = -h[:m]
    else:
        g2 = -d
        m = n - g2
        if m > 0:
            g[:m] = -h[g2:g2+m]
            g[g2:] = (1.0 - h[:m])
    return g

def _top_m_gradients(h, m):
    """Compute gradient as weighted sum of top-m active lag gradients."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    # Get top-m indices
    top_idx = np.argsort(scores)[-m:]
    g = np.zeros(n)
    for r in top_idx:
        d = r - (n - 1)
        g += _grad_row(h, d)
    g /= m
    return g, scores

def _subgradient_method(h, deadline, m_init=5, m_final=20, 
                        lr_init=0.04, mom_decay=0.9, max_iter=5000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    total_iters = 0
    
    while time.time() < deadline and total_iters < max_iter:
        total_iters += 1
        # Adaptive m
        m = m_init
        if cur_c5 < best_c5 + 0.001:
            m = min(m + 1, m_final)
        
        g, scores = _top_m_gradients(h, m)
        # Remove mean to satisfy sum constraint
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Momentum
        mom = mom_decay * mom + (1.0 - mom_decay) * (g / gn)
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            break
        mom = mom / mn
        
        # Line search
        d = lr * mom
        alpha, new_c5 = _line_search(h, d, dx, lr * 2.0)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.3)
        else:
            stall += 1
            lr *= 0.8
            if stall > 50:
                break
    
    return best_h, best_c5

def _lp_direction_restricted(h, top_indices, radius):
    """LP direction that only restricts top lags to decrease."""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    
    # Build constraint matrix for top lags only
    m = len(top_indices)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for ii, r in enumerate(top_indices):
        d_lag = r - (n - 1)
        g = _grad_row(h, d_lag)
        A_ub[ii, :n] = g
        A_ub[ii, n] = -1.0
        b_ub[ii] = -scores[r]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.ones((1, n + 1))
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop_restricted(h, deadline, m=5, radius_init=0.15):
    """LP loop with restricted constraints (only top-m lags)."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

            radius *= 0.7

            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)

            h = _project(h, target)

                best_h = h.copy()
            stall = 0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Only try interior points
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        for i in interior:
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if h[j] <= 1e-8 or h[j] >= 1.0 - 1e-8:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _random_restart(h_best, deadline, n_restarts=50, scale_init=0.04):

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
