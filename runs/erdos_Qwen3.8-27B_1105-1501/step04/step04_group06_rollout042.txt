```python
def compute_corr_grad(h, n):
    """Compute gradient of each corr[r] with respect to h.
    Returns array of shape (2n-1, n) where G[r, j] = dcorr[r]/dh[j].
    """
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        # First term: (1-h[j+s]) for valid j, j+s
        valid1 = (j >= 0) & (j < n) & (j + s >= 0) & (j + s < n)
        G[r, valid1] += (1.0 - h[j[valid1] + s])
        # Second term: -h[j-s] for valid j-s, j
        valid2 = (j >= 0) & (j < n) & (j - s >= 0) & (j - s < n)
        G[r, valid2] -= h[j[valid2] - s]
    return G
```

For each coordinate j, given the current h, the contribution to C₅ from changing h[j] is:
- h[j] appears in corr[r] for all r where j is in the valid range
- Specifically, for shift s, h[j] appears if j is valid and j+s is valid

So increasing h[j] increases corr[r] by (1-h[j+s]) - h[j-s] for each r where j appears.

If we want to decrease the max correlation, we should:
- If the max is at shift s*, and the gradient at that shift w.r.t. h[j] is positive, decrease h[j]
- If negative, increase h[j]

But we also need to maintain the sum constraint. So we can do:
- Pick the "worst" shift s*
- For each j in the valid range of s*, compute the marginal change
- Adjust h[j] to reduce corr[s*] while maintaining the sum

This is essentially what subgradient descent does, but coordinate descent is more targeted.

```python
    """One pass of coordinate descent to reduce max correlation."""
    h = h.copy()
    target = n * 0.5
    for sweep in range(max_sweeps):
        corr = corr_vector(h)
        r_max = int(np.argmax(corr))
        s = r_max - (n - 1)
        # For each j in valid range, try adjusting h[j]
        for j in range(n):
            # Check if j is in valid range for this shift
            if s >= 0:
                if not (0 <= j < n - s and 0 <= j + s < n):
                    continue
            else:
                if not (0 < -s + j <= n - 1 and 0 <= j + s < n):
                    continue
            # Actually, j appears in corr[r_max] if both j and j+s are in [0,n-1]
            if not (0 <= j + s < n):
                continue
            if s >= 0 and j >= n - s:
                continue
            if s < 0 and j < -s:
                continue
            
            # Current contribution of h[j] to corr[r_max]
            # dcorr/dh[j] = (1-h[j+s]) - h[j-s] (if j-s valid)
            g = (1.0 - h[j + s])
            if j - s >= 0:
                g -= h[j - s]
            
            # Try to move h[j] in direction -g (to reduce corr[r_max])
            # But maintain sum constraint: if we increase h[j], we must decrease some other h[j']
            # Simple approach: move h[j] and adjust the rest proportionally
            # Even simpler: try a small step and project
            step = 0.01 * np.sign(-g)
            h_new = h.copy()
            h_new[j] += step
            h_new = project_mass(h_new, target)

            c5_cur = float(np.max(corr) * 2.0 / n)
            if c5_new < c5_cur:
                h = h_new
                corr = corr_vector(h)
        # End for j
    return h
```

```python
import numpy as np

def project_mass(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] < 1.0:
                    a = min(1.0 - x[i], diff)
                    x[i] += a
                    diff -= a
                    if diff <= 1e-9: break
        else:
            for i in np.argsort(-x):
                if x[i] > 0.0:
                    a = min(x[i], -diff)
                    x[i] -= a
                    diff += a
                    if diff >= -1e-9: break
    return x

            tm[i] = 1 - tm[i // 2]
        else:
            tm[i] = tm[i // 2]
    return tm[:n]
```

```python
def subgrad_step(h, n, step_size):
    """One subgradient step with backtracking."""
    corr = corr_vec(h)
    r_max = int(np.argmax(corr))
    s = r_max - (n - 1)
    j = np.arange(n)
    v1 = (j + s >= 0) & (j + s < n)
    v2 = (j - s >= 0) & (j - s < n)
    grad = np.zeros(n)
    grad[v1] += (1.0 - h[j[v1] + s])
    grad[v2] -= h[j[v2] - s]
    grad -= np.mean(grad)  # Project out the sum-constraint direction
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return h.copy()
    grad = grad / norm
    h_new = project_mass(h - step_size * grad, n * 0.5)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def project_mass(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] < 1.0:
                    a = min(1.0 - x[i], diff)
                    x[i] += a
                    diff -= a
                    if diff <= 1e-9:
                        break
        else:
            for i in np.argsort(-x):
                if x[i] > 0.0:
                    a = min(x[i], -diff)
                    x[i] -= a
                    diff += a
                    if diff >= -1e-9:
                        break
    return x


def exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def tm_sequence(n):
    length = 1
    while length < n:
        length *= 2
    tm = np.zeros(length, dtype=np.float64)
    for i in range(1, length):
        if i % 2 == 1:
            tm[i] = 1.0 - tm[i // 2]
        else:
            tm[i] = tm[i // 2]
    return tm[:n]


def tm_start(n):
    h = tm_sequence(n)
    return project_mass(h, n * 0.5)


def wave_start(n, freq):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
    return project_mass(h, n * 0.5)


def random_start(n, seed_val):
    rng = np.random.default_rng(seed_val)
    h = rng.random(n)
    return project_mass(h, n * 0.5)


def mixed_start(n, seed_val):
    rng = np.random.default_rng(seed_val)
    h = np.zeros(n)
    block = max(1, n // 8)
    for i in range(0, n, block):
        val = rng.random()
        h[i:i + block] = val
    return project_mass(h, n * 0.5)


def _lse_obj(h, alpha, n, penalty):
    corr = corr_vec(h)
    m = float(np.max(corr))
    vals = np.clip(alpha * (corr - m), -400.0, 400.0)
    exp_v = np.exp(vals)
    lse = m + (1.0 / alpha) * np.log(float(np.sum(exp_v)))
    s = float(np.sum(h) - n * 0.5)
    return lse + penalty * s * s


def _lse_grad(h, alpha, n, penalty):
    corr = corr_vec(h)
    m = float(np.max(corr))
    vals = np.clip(alpha * (corr - m), -400.0, 400.0)
    exp_v = np.exp(vals)
    w = exp_v / float(np.sum(exp_v))
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        v1 = (j + s >= 0) & (j + s < n)
        grad[v1] += w[r] * (1.0 - h[j[v1] + s])
        v2 = (j - s >= 0) & (j - s < n)
        grad[v2] -= w[r] * h[j[v2] - s]
    s_val = float(np.sum(h) - n * 0.5)
    grad += 2.0 * penalty * s_val
    return grad


def lse_lbfgs(h_init, n, alpha, maxiter, penalty):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h_init, jac=_lse_grad,
                       args=(alpha, n, penalty), bounds=bounds,
                       method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 40, 'iprint': 0})
        h = project_mass(res.x, n * 0.5)
        return h, exact_c5(h)
    except Exception:
        return h_init.copy(), exact_c5(h_init)


def _sobj_f(z, n):
    return z[n]


def _sobj_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _seq_f(z, n):
    return float(np.sum(z[:n]) - n * 0.5)


def _seq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _sineq_f(z, n, active_idx):
    h = z[:n]
    t = z[n]
    c = corr_vec(h)
    return t - c[active_idx]


def _sineq_g(z, n, active_idx):
    h = z[:n]
    j = np.arange(n)
    K = len(active_idx)
    J = np.zeros((K, n + 1))
    for k in range(K):
        r = active_idx[k]
        s = r - (n - 1)
        v1 = (j + s >= 0) & (j + s < n)
        row = np.zeros(n)
        row[v1] = -(1.0 - h[j[v1] + s])
        v2 = (j - s >= 0) & (j - s < n)
        row[v2] += h[j[v2] - s]
        J[k, :n] = row
        J[k, n] = 1.0
    return J


def active_slsqp(h_init, n, active_idx, maxiter, ftol):
    corr = corr_vec(h_init)
    t0 = float(np.max(corr)) + 1e-4
    z0 = np.concatenate([h_init, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, n)]
    cons = [
        {'type': 'eq', 'fun': _seq_f, 'jac': _seq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _sineq_f, 'jac': _sineq_g, 'args': (n, active_idx)},
    ]
    try:
        res = minimize(_sobj_f, z0, jac=_sobj_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        h = project_mass(res.x[:n], n * 0.5)
        return h, exact_c5(h)
    except Exception:
        return h_init.copy(), exact_c5(h_init)


def active_set_loop(h_init, n, deadline, K, max_rounds):
    h = h_init.copy()
    cur_c5 = exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    no_improve = 0
    for _round in range(max_rounds):
        if time.time() > deadline:
            break
        corr = corr_vec(h)
        active_idx = np.argsort(-corr)[:K]
        h_new, c5_new = active_slsqp(h, n, active_idx, maxiter=150, ftol=1e-14)
        if c5_new < best_c5 - 1e-11:
            best_c5, best_h = c5_new, h_new.copy()
            h = h_new
            cur_c5 = c5_new
            no_improve = 0
        elif c5_new < cur_c5 - 1e-11:
            h = h_new
            cur_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            K = min(2 * n - 1, K + 8)
            if no_improve > 8:
                break
    return best_h, best_c5


def subgrad_step(h, n, step_size):
    corr = corr_vec(h)
    r_max = int(np.argmax(corr))
    s = r_max - (n - 1)
    j = np.arange(n)
    v1 = (j + s >= 0) & (j + s < n)
    v2 = (j - s >= 0) & (j - s < n)
    grad = np.zeros(n)
    grad[v1] += (1.0 - h[j[v1] + s])
    grad[v2] -= h[j[v2] - s]
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return h.copy()
    h_new = project_mass(h - step_size * grad / norm, n * 0.5)
    return h_new


def subgrad_polish(h, n, deadline, max_iters):
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    step = 0.05
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        h_new = subgrad_step(h, n, step)
        c5_new = exact_c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h.copy()
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.7
            if step < 0.001:
                break
    return best_h, best_c5


def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_mass(h_new, n_new * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n = len(h0)
    target = n * 0.5
    h = project_mass(h0, target)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n

    # Phase 1: LSE optimization with increasing alpha (15% budget)
    d1 = t0 + budget_s * 0.15
    alphas = [8.0, 15.0, 30.0, 60.0, 120.0]
    for alpha in alphas:
        if time.time() > d1:
            break
        h_new, c5_new = lse_lbfgs(h, n, alpha, maxiter=120, penalty=200.0)
        if c5_new < best_c5 - 1e-12:
            best_c5, best_h = c5_new, h_new.copy()
            h = h_new

    # Phase 2: Active SLSQP refinement (20% budget)
    d2 = t0 + budget_s * 0.35
    h_as, c5_as = active_set_loop(best_h, n, d2, K=20, max_rounds=80)
    if c5_as < best_c5:
        best_c5, best_h = c5_as, h_as.copy()

    # Phase 3: Diverse starting points (20% budget)
    d3 = t0 + budget_s * 0.55
    starts = []
    starts.append(tm_start(n))
    for f in [1, 2, 3, 4, 5, 7]:
        starts.append(wave_start(n, f))
    for sv in [11, 22, 33]:
        starts.append(random_start(n, sv))
    for sv in [44, 55]:
        starts.append(mixed_start(n, sv))

    for h_s in starts:
        if time.time() > d3:
            break
        h_t, c5_t = lse_lbfgs(h_s, n, 40.0, maxiter=60, penalty=200.0)
        if c5_t < best_c5 - 1e-8:
            h_t2, c5_t2 = active_set_loop(h_t, n, min(d3, time.time() + 15), K=20, max_rounds=20)
            if c5_t2 < best_c5:
                best_c5, best_h = c5_t2, h_t2.copy()
            elif c5_t < best_c5:
                best_c5, best_h = c5_t, h_t.copy()

    # Subgradient polish on best so far
    if time.time() < d3 + budget_s * 0.02:
        d_sg = min(d3 + budget_s * 0.02, t0 + budget_s * 0.60)
        h_sg, c5_sg = subgrad_polish(best_h, n, d_sg, max_iters=3000)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 4: Higher resolution (25% budget)
    d4 = t0 + budget_s * 0.85
    n_hi = n * 2
    h_hi = resample(best_h, n_hi)
    if time.time() < d4 - 60:
        for alpha in [20.0, 50.0, 100.0]:
            if time.time() > d4 - 60:
                break
            h_hi_t, c5_hi_t = lse_lbfgs(h_hi, n_hi, alpha, maxiter=80, penalty=200.0)
            if c5_hi_t < exact_c5(h_hi):
                h_hi = h_hi_t
    if time.time() < d4 - 10:
        h_hi_s, c5_hi_s = active_set_loop(h_hi, n_hi, d4, K=30, max_rounds=60)
        if c5_hi_s < exact_c5(h_hi):
            h_hi = h_hi_s
    c5_hi_final = exact_c5(h_hi)
    if c5_hi_final < best_c5:
        best_c5, best_h, best_n = c5_hi_final, h_hi.copy(), n_hi

    # Try n=96 as well if time permits
    if time.time() < d4 and n * 3 % 2 == 0:
        n_mid = n * 3
        h_mid = resample(best_h, n_mid)
        h_mid_t, c5_mid_t = lse_lbfgs(h_mid, n_mid, 50.0, maxiter=60, penalty=200.0)
        if time.time() < d4:
            h_mid_s, c5_mid_s = active_set_loop(h_mid_t, n_mid, d4, K=25, max_rounds=30)
            if c5_mid_s < best_c5:
                best_c5, best_h, best_n = c5_mid_s, h_mid_s.copy(), n_mid
            elif c5_mid_t < best_c5:
                best_c5, best_h, best_n = c5_mid_t, h_mid_t.copy(), n_mid

    # Phase 5: Final polish (12% budget)
    d5 = t0 + budget_s * 0.97
    h_f, c5_f = active_set_loop(best_h, best_n, d5, K=30, max_rounds=100)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h_f.copy()

    # Multi-start perturbation on best
    if time.time() < d5:
        for scale in [0.04, 0.02, 0.01]:
            if time.time() > d5:
                break
            h_p = best_h + rng.normal(0, scale, best_n)
            h_p = project_mass(h_p, best_n * 0.5)
            h_p2, c5_p2 = lse_lbfgs(h_p, best_n, 80.0, maxiter=50, penalty=200.0)
            if c5_p2 < best_c5 - 1e-10:
                h_p3, c5_p3 = active_set_loop(h_p2, best_n, min(d5, time.time() + 20), K=25, max_rounds=30)
                if c5_p3 < best_c5:
                    best_c5, best_h = c5_p3, h_p3.copy()
                elif c5_p2 < best_c5:
                    best_c5, best_h = c5_p2, h_p2.copy()
            elif c5_p2 < best_c5:
                best_c5, best_h = c5_p2, h_p2.copy()

    # Final subgradient polish
    if time.time() < t0 + budget_s * 0.99:
        h_fp, c5_fp = subgrad_polish(best_h, best_n, t0 + budget_s * 0.99, max_iters=2000)
        if c5_fp < best_c5:
            best_c5, best_h = c5_fp, h_fp.copy()

    best_h = project_mass(best_h, best_n * 0.5)
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, best_n
```
