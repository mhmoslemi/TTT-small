    # h[i] appears in lag r if:

    # For each lag r, the score is: sum_j h[j+a(j,d)] * (1-h[j+b(j,d)])

    # For d >= 0: (1-h[i-d]) * 1[h[i] is the first factor] + (-h[i+d]) * 1[h[i] is the second factor]

    #     h[i] as first factor (h[j+d]=h[i]): j = i-d, valid if 0<=i-d<=n-1-d, i.e., d<=i<=n-1
    #       contribution: h[i]*(1-h[i-d])
    #     h[i] as second factor (1-h[j]=1-h[i]): j = i, valid if 0<=i<=n-1-d
    #       contribution: h[i+d]*(1-h[i])
    #   if d < 0, let g=-d: raw_r = sum_{j=g}^{n-1} h[j](1-h[j-g])
    #     h[i] as first factor: j=i, valid if g<=i<=n-1

    #     h[i] as second factor (1-h[j-g]=1-h[i]): j-g=i, j=i+g, valid if g<=i+g<=n-1, i.e., 0<=i<=n-1-g

        t_opt = 0.0
    elif f_hi <= f_mid and f_hi <= f_lo:
        t_opt = 1.0

    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max, n_grid=20):
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 1)]
    hi_t = ts[min(n_grid - 1, idx + 1)]
    if hi_t <= lo_t:
        lo_t, hi_t = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)

    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)

    f1, f2 = _f(c1), _f(c2)
    for _ in range(40):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_optimize(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _raw_scores(h)
        p = _lp_dir(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-15:
            stall += 1
            radius *= 0.6
            if radius < 1e-8:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-15:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-15:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.5)
        else:
            stall += 1
            radius *= 0.65
    return best_h, best_c5

def _active_lp_dir(h, scores, radius, top_k=None):
    """LP direction using only top-k active lags for speed."""
    n = len(h)
    n_lags = 2 * n - 1
    if top_k is None or top_k >= n_lags:
        top_k = n_lags
    idx = np.argsort(scores)[::-1][:top_k]
    G = np.zeros((top_k, n))
    for ii, r in enumerate(idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            G[ii, d:] += 1.0
            G[ii, d:] -= h[:m]
            G[ii, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m <= 0:
                continue
            G[ii, :m] += 1.0
            G[ii, :m] -= h[g:]
            G[ii, g:] -= h[:m]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize_active(h, deadline, radius_init=0.15, top_k=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = _raw_scores(h)
        p = _active_lp_dir(h, scores, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-15:
            stall += 1
            radius *= 0.6
            if radius < 1e-8:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-15:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-15:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-12)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-12:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.12, 0.5)
        else:
            stall += 1
            radius *= 0.6
    return best_h, best_c5

def _binary_push(h, target, deadline, max_rounds=20):
    """Try to push near-extreme coordinates to 0 or 1 while preserving sum."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        near_zero = np.where(h < 0.05)[0]
        near_one = np.where(h > 0.95)[0]
        for i in near_zero:
            if time.time() > deadline:
                break
            delta = h[i]
            if delta < 1e-6:
                continue
            h_temp = h.copy()
            h_temp[i] = 0.0
            remaining = target - float(np.sum(h_temp))
            if remaining < -1e-10:
                continue
            mask = (h_temp > 0.01) & (h_temp < 0.99)
            if not mask.any():
                mask = h_temp < 1.0 - 1e-10
            cnt = int(np.sum(mask))
            if cnt > 0:
                h_temp[mask] += remaining / cnt
                h_temp = np.clip(h_temp, 0.0, 1.0)
                if abs(float(np.sum(h_temp)) - target) < 1e-8:
                    c5_t = _exact_c5(h_temp)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_temp.copy()
                        h = h_temp.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_optimize(h, deadline, tau_init=15.0, lr_init=0.02, mom_beta=0.9, seed=42):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    rng = np.random.default_rng(seed)
    while time.time() < deadline:
        if stall > 100:
            break
        dx = 2.0 / n
        scores = _raw_scores(h) * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(np.clip(shifted, -50, 0))
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(2 * n - 1):
            wr = w[r]
            if wr < 1e-14:
                continue
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                grad[d:] += wr
                grad[d:] -= wr * h[:m]
                grad[:m] -= wr * h[d:]
            else:
                g = -d
                m = n - g
                if m <= 0:
                    continue
                grad[:m] += wr
                grad[:m] -= wr * h[g:]
                grad[g:] -= wr * h[:m]
        grad *= dx
        mom = mom_beta * mom + (1.0 - mom_beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            lr = min(lr * 1.03, 0.15)
            tau = min(tau * 1.02, 300.0)
        else:
            stall += 1
            lr *= 0.9
            if stall % 40 == 0:
                tau = max(tau_init, tau * 0.7)
    return best_h, best_c5

def _pair_move(h, target, deadline, seed=0):
    """Try pairwise adjustments that preserve sum."""
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    attempts = 0
    while time.time() < deadline and attempts < 5000:
        attempts += 1
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        s = h[i] + h[j]
        if s < 1e-8 or s > 2.0 - 1e-8:
            h[i], h[j] = 0.0, s
            if h[j] > 1.0:
                h[i], h[j] = s - 1.0, 1.0
            c5_t = _exact_c5(h)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h.copy()
            else:
                h[i], h[j] = s - h[i], h[j]
                h[i] = s - h[j]
        else:
            # Try moving mass between i and j
            for frac in [0.3, 0.5, 0.7, 1.0]:
                h_temp = h.copy()
                h_temp[i] = min(1.0, h[i] + frac * 0.1)
                h_temp[j] = h[j] - (h_temp[i] - h[i])
                if h_temp[j] < 0 or h_temp[j] > 1:
                    h_temp[i] = max(0.0, h[i] - frac * 0.1)
                    h_temp[j] = h[j] - (h_temp[i] - h[i])
                    if h_temp[j] < 0 or h_temp[j] > 1:
                        continue
                c5_t = _exact_c5(h_temp)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_h = h_temp.copy()
                    h = h_temp.copy()
                    break
    return best_h, best_c5

def _multi_start_pool(h_parent, n_parent, seed=42):
    """Generate a diverse pool of starting candidates."""
    rng = np.random.default_rng(seed)
    pool = []
    target = n_parent / 2.0

    # Parent
    h_p = _project(h_parent.copy(), target)
    pool.append(h_p)

    # Reversed
    pool.append(_project(h_p[::-1].copy(), target))

    # Complemented (1-h)
    pool.append(_project(1.0 - h_p, target))

    # Sharpened
    for alpha in [0.3, 0.6, 1.0]:
        h_sh = h_p + alpha * (h_p - 0.5)
        h_sh = _project(h_sh, target)
        pool.append(h_sh)

    # Relaxed
    for alpha in [0.3, 0.6]:
        h_rel = h_p - alpha * (h_p - 0.5)
        h_rel = _project(h_rel, target)
        pool.append(h_rel)

    # Small noise
    for scale in [0.02, 0.05, 0.1]:
        h_noise = h_p + rng.normal(0, scale, n_parent)
        h_noise = _project(h_noise, target)
        pool.append(h_noise)

    # Different resolutions
    for dn in [-10, -5, 5, 10]:
        n_new = n_parent + dn
        if 40 <= n_new <= 150:
            h_r = _resample(h_p, n_new)
            pool.append(h_r)

    return pool

def _perturb_refine(h_best, deadline, budget_s, seed=42, max_attempts=100):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = 0.03
    time_per = max(2.0, budget_s * 0.015)
    for attempt in range(max_attempts):
        if time.time() > deadline:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + time_per)
        h_try, c5_try = _coord_descent(h_pert, target, sub_dl, max_passes=50, seed=seed + attempt)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.2, scale * 1.3)
    return best_h, best_c5

def _direction_search(h_init, deadline, n_dirs=8, radius=0.25, seed=0):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    while time.time() < deadline and stall < 30:
        improved = False
        scores = _raw_scores(h)
        # Top gradient directions
        top_idx = np.argsort(scores)[::-1][:4]
        dirs = []
        for r in top_idx:
            d = r - (n - 1)
            p = np.zeros(n)
            if d >= 0:
                m = n - d
                if m > 0:
                    p[d:] += 1.0
                    p[d:] -= h[:m]
                    p[:m] -= h[d:]
            else:
                g = -d
                m = n - g
                if m > 0:
                    p[:m] += 1.0
                    p[:m] -= h[g:]
                    p[g:] -= h[:m]
            p -= np.mean(p)
            norm = np.max(np.abs(p))
            if norm > 1e-10:
                p = p / norm * radius
                dirs.append(p)
        # Random directions
        for _ in range(n_dirs - len(dirs)):
            p = rng.normal(0, 1, n)
            p -= np.mean(p)
            norm = np.max(np.abs(p))
            if norm > 1e-10:
                p = p / norm * radius
                dirs.append(p)

        for p in dirs:
            if time.time() > deadline:
                break
            alpha_max = radius
            for j in range(n):
                if p[j] > 1e-15:
                    alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                elif p[j] < -1e-15:
                    alpha_max = min(alpha_max, h[j] / (-p[j]))
            if alpha_max < 1e-12:
                continue
            alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
            if new_c5 < cur_c5 - 1e-12:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        if not improved:
            stall += 1
            radius *= 0.75
            if radius < 0.01:
                break
        else:
            stall = 0
            radius = min(radius * 1.1, 0.4)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h0 = np.ones(90) * 0.5

    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = _project(h0, target0)

    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Phase 1: Gradient warmup (5%)
    d1 = t0 + budget_s * 0.05
    h, c5 = _grad_optimize(h0, d1, tau_init=12.0, lr_init=0.02, mom_beta=0.88, seed=seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Active LP local search (15%)
    d2 = t0 + budget_s * 0.20
    h, c5 = _lp_optimize_active(best_h, d2, radius_init=0.15, top_k=40)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Coordinate descent (10%)
    d3 = t0 + budget_s * 0.30
    h, c5 = _coord_descent(best_h, best_n / 2.0, d3, max_passes=200, seed=seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Direction search (10%)
    d4 = t0 + budget_s * 0.40
    h, c5 = _direction_search(best_h, d4, n_dirs=10, radius=0.3, seed=seed + 10)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Coordinate descent again (10%)
    d5 = t0 + budget_s * 0.50
    h, c5 = _coord_descent(best_h, best_n / 2.0, d5, max_passes=300, seed=seed + 20)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Intensive LP (15%)
    d6 = t0 + budget_s * 0.65
    h, c5 = _lp_optimize_active(best_h, d6, radius_init=0.1, top_k=30)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 7: Coordinate descent (10%)
    d7 = t0 + budget_s * 0.75
    h, c5 = _coord_descent(best_h, best_n / 2.0, d7, max_passes=400, seed=seed + 30)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 8: Multi-resolution exploration (10%)
    d8 = t0 + budget_s * 0.85
    n_base = best_n
    time_per = max(3.0, budget_s * 0.02)
    for dn in [-4, -2, 2, 4, -6, 6, -8, 8]:
        if time.time() > d8:
            break
        n_try = n_base + dn
        if n_try < 40 or n_try > 150:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d8, time.time() + time_per)
        h_r, c5_r = _coord_descent(h_r, n_try / 2.0, sub_dl, max_passes=50, seed=seed + dn)
        if time.time() < d8 and (d8 - time.time()) > 2.0:
            sub_dl2 = min(d8, time.time() + time_per * 0.5)
            h_r, c5_r = _lp_optimize_active(h_r, sub_dl2, radius_init=0.08, top_k=30)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 9: Perturbation restarts (8%)
    d9 = t0 + budget_s * 0.93
    best_h, best_c5 = _perturb_refine(best_h, d9, budget_s, seed=seed + 100)

    # Phase 10: Final coordinate descent polish (5%)
    d10 = t0 + budget_s * 0.98
    h, c5 = _coord_descent(best_h, best_n / 2.0, d10, max_passes=500, seed=seed + 200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Final LP polish
    if time.time() < t0 + budget_s * 0.99:
        h, c5 = _lp_optimize_active(best_h, t0 + budget_s * 0.995, radius_init=0.03, top_k=20)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
