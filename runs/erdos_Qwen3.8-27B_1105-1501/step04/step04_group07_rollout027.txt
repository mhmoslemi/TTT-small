    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _quad_search(h, V, s0):
    dx = 2.0 / len(h)
    a = dx * (np.correlate(V, 1.0 - h, 'full') - np.correlate(h, V, 'full'))
    b = -dx * np.correlate(V, V, 'full')
    amin = 0.0
    amax = 0.0
    pos = V > 1e-14
    neg = V < -1e-14
    if np.any(pos):
        amax = float(np.min((1.0 - h[pos]) / V[pos]))
    if np.any(neg):
        amin = float(np.max((-h[neg]) / V[neg]))
    if amax < amin:
        amax, amin = amin, amax
    if amax - amin < 1e-14:
        return 0.0, float(np.max(s0))
    def f(al):
        return float(np.max(s0 + a * al + b * al * al))
    gr = 0.6180339887498949
    lo, hi = amin, amax
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(80):
        if hi - lo < 1e-13:
            break
        if fc <= fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    alpha_star = (lo + hi) / 2.0
    best_a, best_f = alpha_star, f(alpha_star)
    for cand in (amin, amax):
        cf = f(cand)
        if cf < best_f:
            best_a, best_f = cand, cf
    return best_a, best_f

def _grad_vec(h, w):
    """Weighted gradient: sum_r w_r * ds_r/dh. Returns vector of length n."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        r = n - 1 + d
        wr = w[r]
        if abs(wr) < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                g[:m] += wr * (1.0 - h[d:d + m])
                g[d:d + m] += -wr * h[:m]
        else:
            gv = -d
            m = n - gv
            if m > 0:
                g[gv:gv + m] += wr * (1.0 - h[:m])
                g[:m] += -wr * h[gv:gv + m]
    return g * dx

def _binary_pair_search(h_bin, deadline, max_passes=50000):
    n = len(h_bin)
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = _exact_c5(h_bin)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            ones = np.append(ones[ones != i], j)
            ones = ones[ones != i]
            if i in zeros:
                zeros[zeros == i] = -1
            zeros = np.append(zeros, i)
            zeros = zeros[zeros != j]
            j_old = j
            zeros = np.delete(zeros, np.where(zeros == j_old)[0])
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            stall = 0
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            stall += 1
    return best_h, best_c5

def _continuous_optimize(h, deadline, p_start=8.0, p_end=4096.0, n_steps=5000, use_pairs=True):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    s0 = _scores(h)
    rng = np.random.default_rng(42)
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        s_max = float(s0.max())
        if s_max < 1e-15:
            break
        ratio = s0 / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(s0))] = 1.0
            sw = 1.0
        w = w / sw
        g = _grad_vec(h, w)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            stall += 1
            if stall > 100:
                break
            continue
        V = -g / gn
        # Try gradient direction
        alpha, new_c5 = _quad_search(h, V, s0)
        improved = False
        if new_c5 < cur_c5 - 1e-14 and alpha != 0.0:
            h = h + alpha * V
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            s0 = _scores(h)
            stall = 0
            improved = True
        # Try also a subgradient (random active lag)
        if not improved and use_pairs:
            n_active = max(1, int(0.1 * len(s0)))
            top_idx = np.argsort(s0)[-n_active:]
            r_star = int(top_idx[rng.integers(n_active)])
            w2 = np.zeros(2 * n - 1)
            w2[r_star] = 1.0
            g2 = _grad_vec(h, w2)
            g2 -= np.mean(g2)
            gn2 = np.linalg.norm(g2)
            if gn2 > 1e-14:
                V2 = -g2 / gn2
                alpha2, new_c5_2 = _quad_search(h, V2, s0)
                if new_c5_2 < cur_c5 - 1e-14 and alpha2 != 0.0:
                    h = h + alpha2 * V2
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    s0 = _scores(h)
                    stall = 0
                    improved = True
        if not improved:
            # Try a pair direction
            if use_pairs and n > 4:
                for _try in range(3):
                    p_idx = int(rng.integers(n))
                    q_idx = int(rng.integers(n))
                    if p_idx == q_idx:
                        continue
                    Vp = np.zeros(n)
                    Vp[p_idx] = 1.0
                    Vp[q_idx] = -1.0
                    alpha_p, new_c5_p = _quad_search(h, Vp, s0)
                    if new_c5_p < cur_c5 - 1e-14 and alpha_p != 0.0:
                        h = h + alpha_p * Vp
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        s0 = _scores(h)
                        stall = 0
                        improved = True
                        break
            if not improved:
                stall += 1
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=200):
    """Try all pairs systematically."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    s0 = _scores(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-6:
                    continue
                for sign in (1, -1):
                    V = np.zeros(n)
                    V[i] = sign
                    V[j] = -sign
                    # Check feasibility
                    if sign > 0:
                        if h[i] >= 1.0 - 1e-12 or h[j] <= 1e-12:
                            continue
                    else:
                        if h[i] <= 1e-12 or h[j] >= 1.0 - 1e-12:
                            continue
                    alpha, new_c5 = _quad_search(h, V, s0)
                    if new_c5 < cur_c5 - 1e-13 and abs(alpha) > 1e-12:
                        h = h + alpha * V
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        s0 = _scores(h)
                        stall = 0
                        improved = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_optimize(h_best, deadline, n_pert, scale_init, rng):
    n = len(h_best)
    scale = scale_init
    for _ in range(n_pert):
        if time.time() > deadline:
            break
        h_pert = h_best + rng.normal(0, scale, n)
        h_pert = _project(h_pert, n / 2.0)
        c5_pert = _exact_c5(h_pert)
        sub_dl = min(deadline, time.time() + 0.5)
        h_opt, c5_opt = _continuous_optimize(h_pert, sub_dl, p_start=16.0, p_end=2048.0, n_steps=300, use_pairs=True)
        if c5_opt < c5_pert > _exact_c5(h_best):
            # Perturbation helped get to a better basin
            pass
        return h_opt, c5_opt
    return h_best, _exact_c5(h_best)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Binary local search for good starting point (10%)
    dl1 = t0 + budget_s * 0.10
    h_bin = _project(h, n0 / 2.0)
    # Round to binary
    h_bin_2 = (h >= 0.5).astype(float)
    ones_cnt = int(h_bin_2.sum())
    if ones_cnt > n0 // 2:
        idx = np.where(h_bin_2 == 1.0)[0]
        np.random.seed(1)
        np.random.shuffle(idx)
        h_bin_2[idx[ones_cnt - n0 // 2:]] = 0.0
    elif ones_cnt < n0 // 2:
        idx = np.where(h_bin_2 == 0.0)[0]
        np.random.seed(1)
        np.random.shuffle(idx)
        h_bin_2[idx[:n0 // 2 - ones_cnt]] = 1.0
    h_bin_2, c5_bin = _binary_pair_search(h_bin_2, dl1, max_passes=80000)
    # Also try from the initial solution's rounding
    h_bin_orig = (h0 >= 0.5).astype(float)
    ones_orig = int(h_bin_orig.sum())
    if ones_orig > n0 // 2:
        idx = np.where(h_bin_orig == 1.0)[0]
        np.random.seed(2)
        np.random.shuffle(idx)
        h_bin_orig[idx[ones_orig - n0 // 2:]] = 0.0
    elif ones_orig < n0 // 2:
        idx = np.where(h_bin_orig == 0.0)[0]
        np.random.seed(2)
        np.random.shuffle(idx)
        h_bin_orig[idx[:n0 // 2 - ones_orig]] = 1.0
    h_bin_orig, c5_bin2 = _binary_pair_search(h_bin_orig, dl1, max_passes=30000)
    if c5_bin2 < best_c5:
        # Use the better binary as a starting point for continuous
        pass
    # Use the better of original and binary as starting point
    if c5_bin < _exact_c5(h):
        start_h = _project(h_bin_2, n0 / 2.0)
    else:
        start_h = h.copy()
    if c5_bin2 < _exact_c5(start_h):
        start_h = _project(h_bin_orig, n0 / 2.0)

    # Phase 2: Main continuous optimization (35%)
    dl2 = t0 + budget_s * 0.45
    h_opt, c5_opt = _continuous_optimize(start_h, dl2, p_start=6.0, p_end=4096.0, n_steps=8000, use_pairs=True)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 3: Pair sweep for fine-tuning (12%)
    dl3 = t0 + budget_s * 0.57
    h_ps, c5_ps = _pair_sweep(best_h, dl3, max_rounds=100)
    if c5_ps < best_c5:
        best_c5 = c5_ps
        best_h = h_ps.copy()

    # Phase 4: Higher p-continued optimization (13%)
    dl4 = t0 + budget_s * 0.70
    h_opt2, c5_opt2 = _continuous_optimize(best_h, dl4, p_start=256.0, p_end=65536.0, n_steps=3000, use_pairs=True)
    if c5_opt2 < best_c5:
        best_c5 = c5_opt2
        best_h = h_opt2.copy()

    # Phase 5: Equilibration pass - target active lags specifically (10%)
    dl5 = t0 + budget_s * 0.80
    n = len(best_h)
    s0 = _scores(best_h)
    s_max = float(s0.max())
    cur_h = best_h.copy()
    cur_c5 = best_c5
    for _ in range(500):
        if time.time() > dl5:
            break
        s0 = _scores(cur_h)
        s_max = float(s0.max())
        thresh = s_max * 0.95
        active = np.where(s0 >= thresh)[0]
        if len(active) == 0:
            break
        # Weight active lags: prioritize reducing the max
        active_scores = s0[active]
        w_act = (active_scores / active_scores.max()) ** 16.0
        w_act /= w_act.sum()
        w_full = np.zeros(2 * n - 1)
        for idx in active:
            w_full[idx] = w_act[list(active).index(idx)]
        g = _grad_vec(cur_h, w_full)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        V = -g / gn
        alpha, new_c5 = _quad_search(cur_h, V, _scores(cur_h))
        if new_c5 < cur_c5 - 1e-14 and alpha != 0.0:
            cur_h = cur_h + alpha * V
            cur_h = _project(cur_h, n / 2.0)
            cur_c5 = _exact_c5(cur_h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            # Try pair from active lag gradient
            r_star = int(np.argmax(s0))
            w_sub = np.zeros(2 * n - 1)
            w_sub[r_star] = 1.0
            g_sub = _grad_vec(cur_h, w_sub)
            g_sub -= np.mean(g_sub)
            gn_sub = np.linalg.norm(g_sub)
            if gn_sub > 1e-14:
                V_sub = -g_sub / gn_sub
                alpha_s, new_c5_s = _quad_search(cur_h, V_sub, _scores(cur_h))
                if new_c5_s < cur_c5 - 1e-14 and alpha_s != 0.0:
                    cur_h = cur_h + alpha_s * V_sub
                    cur_h = _project(cur_h, n / 2.0)
                    cur_c5 = _exact_c5(cur_h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur_h.copy()
                else:
                    break

    # Phase 6: Multi-start perturbation (8%)
    dl6 = t0 + budget_s * 0.88
    scale = 0.02
    for attempt in range(60):
        if time.time() > dl6:
            break
        if best_c5 < 0.3806:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.012)
        h_try, c5_try = _continuous_optimize(h_p, sub_dl, p_start=16.0, p_end=1024.0, n_steps=200, use_pairs=True)
        if c5_try < best_c5 - 1e-11:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.08, scale * 1.05)

    # Phase 7: Resolution sweep (4%)
    dl7 = t0 + budget_s * 0.92
    for dn in (-8, -4, -2, -1, 1, 2, 4, 8, 12, 16):
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + budget_s * 0.010)
        h_r, c5_r = _continuous_optimize(h_r, sub_dl, p_start=16.0, p_end=2048.0, n_steps=300, use_pairs=True)
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final pair polish + continued optimization (8%)
    dl8 = t0 + budget_s * 0.995
    h_final, c5_final = _pair_sweep(best_h, dl8, max_rounds=50)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    # One more continuous pass
    sub_dl = min(dl8, time.time() + 2.0)
    h_fin2, c5_fin2 = _continuous_optimize(best_h, sub_dl, p_start=512.0, p_end=65536.0, n_steps=500, use_pairs=True)
    if c5_fin2 < best_c5:
        best_c5 = c5_fin2
        best_h = h_fin2.copy()

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
