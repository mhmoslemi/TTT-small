    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _resample_area(h_old, n_new):
    n = len(h_old)
    if n_new == n:
        return h_old.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return _project(h_new, n_new / 2.0)

def _full_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            cnt = n - d
            if cnt > 0:
                G[r, :cnt] += -h[d:d+cnt]
                G[r, d:d+cnt] += 1.0 - h[:cnt]
        else:
            s = -d
            cnt = n - s
            if cnt > 0:
                G[r, :cnt] += 1.0 - h[s:s+cnt]
                G[r, s:s+cnt] += -h[:cnt]
    return G

class _EpiProb:
    def __init__(self, n, active_idx=None):
        self.n = n
        self.active_idx = active_idx
    
    def obj(self, x):
        return x[self.n]
    
    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    
    def ineq(self, x):
        h = x[:self.n]
        T = x[self.n]
        scores = np.correlate(h, 1.0 - h, mode='full')
        if self.active_idx is not None:
            return T - scores[self.active_idx]
        return T - scores
    
    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        G = _full_grad_matrix(h)
        if self.active_idx is not None:
            G = G[self.active_idx]
        m = G.shape[0]
        jac = np.zeros((m, n + 1))
        jac[:, :n] = -G
        jac[:, n] = 1.0
        return jac
    
    def eq(self, x):
        return np.array([np.sum(x[:self.n]) - self.n / 2.0])
    
    def eq_jac(self, x):
        J = np.zeros((1, self.n + 1))
        J[0, :self.n] = 1.0
        return J

def _slsqp(h0, deadline, k_active=None, maxiter=None):
    n = len(h0)
    if n < 10 or n > 250:
        return h0
    if maxiter is None:
        maxiter = max(15, min(60, int(6000 / max(n, 10))))
    
    if k_active is not None:
        scores = _raw_scores(h0)
        order = np.argsort(scores)[::-1]
        top_k = order[:min(k_active, 2*n-1)]
        prob = _EpiProb(n, active_idx=top_k)
    else:
        prob = _EpiProb(n)
    
    T0 = float(np.max(_raw_scores(h0))) * 1.01 + 0.01
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = T0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'ineq', 'fun': prob.ineq, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq, 'jac': prob.eq_jac}
    ]
    
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    try:
        t_start = time.time()
        res = minimize(
            prob.obj, x0, jac=prob.obj_grad,
            bounds=bounds, constraints=cons,
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        t_elapsed = time.time() - t_start
        
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], n / 2.0)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass
    
    return best_h

def _slsqp_active_set(h0, deadline, k_init=50, max_rounds=10):
    """Iterative active set SLSQP."""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    k = k_init
    
    for round_i in range(max_rounds):
        if time.time() > deadline:
            break
        # Determine active set from current best
        scores = _raw_scores(best_h)
        max_score = float(np.max(scores))
        # Active: scores within 2% of max, or top-k
        threshold = max_score * 0.98
        active_mask = scores >= threshold
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            order = np.argsort(scores)[::-1]
            active_idx = order[:k]
        elif len(active_idx) > 2*n - 1:
            order = np.argsort(scores)[::-1]
            active_idx = order[:k]
        
        maxiter = max(10, min(40, int(4000 / max(n, 10))))
        h_new = _slsqp(best_h, deadline, k_active=len(active_idx), maxiter=maxiter)
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
        else:
            # Try with slightly larger active set
            k = int(k * 1.3)
            if k > 2 * n - 1:
                break
    
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        
        if n <= 50:
            pairs = []
            for i in range(n):
                for j in range(i + 1, n):
                    pairs.append((i, j))
            rng.shuffle(pairs)
        else:
            samples = rng.integers(0, n, size=min(2*n, 200))
            pairs = []
            seen = set()
            for ii in range(len(samples)):
                for jj in range(ii + 1, len(samples)):
                    i, j = int(samples[ii]), int(samples[jj])
                    if i == j:
                        continue
                    key = (min(i,j), max(i,j))
                    if key not in seen:
                        seen.add(key)
                        pairs.append(key)
            rng.shuffle(pairs)
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            for (di, dj) in [(1.0, -1.0), (-1.0, 1.0)]:
                if di > 0:
                    am_max = min(1.0 - best_h[i], best_h[j])
                else:
                    am_max = min(best_h[i], 1.0 - best_h[j])
                if am_max < 1e-10:
                    continue
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    d = di * am_max * frac
                    if abs(d) < 1e-12:
                        continue
                    h_t = best_h.copy()
                    h_t[i] += d
                    h_t[j] -= d
                    if h_t[i] < -1e-10 or h_t[i] > 1.0 + 1e-10:
                        continue
                    if h_t[j] < -1e-10 or h_t[j] > 1.0 + 1e-10:
                        continue
                    h_t = np.clip(h_t, 0, 1)
                    # Re-project to maintain sum exactly
                    h_t = _project(h_t, n / 2.0)
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

def _gradient_step(h, scores, lr, n_top=30):
    """One gradient descent step on the softmax-approximated max."""
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:n_top]
    top_scores = scores[top_idx]
    tau = 5.0
    w = np.exp((top_scores - top_scores[0]) / tau)
    w /= np.sum(w)
    
    G = _full_grad_matrix(h)
    G_top = G[top_idx]
    grad = G_top.T @ w  # shape (n,)
    
    h_new = h - lr * grad
    h_new = _project(h_new, n / 2.0)
    return h_new

def _grad_loop(h, deadline, lr_init=0.05, max_iter=2000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        scores = _raw_scores(h)
        h_new = _gradient_step(h, scores, lr)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr *= 0.85
            if stall > 100:
                break
    
    return best_h, best_c5
