    n = len(h)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - v.sum()
        if abs(diff) < 1e-15: break
        mask = v < 1.0 if diff > 0 else v > 0.0
        if not mask.any(): break
        v[mask] += diff / mask.sum()
        np.clip(v, 0.0, 1.0, out=v)
    return v

def _grad_for_lag(h, d):
    """Gradient of F_d w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    lo = max(0, -d)
    hi = min(n, n - d)
    if lo >= hi:
        return np.zeros(n)
    g = np.zeros(n)
    i = np.arange(lo, hi)
    g[i + d] += dx * (1.0 - h[i])
    g[i] += -dx * h[i + d]
    return g

def _lp_step(h):

    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx

    c[n] = 1.0

        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue

        A_ub[r, i + d] += dx
        A_ub[r, i + d] -= dx * h[i]
        A_ub[r, i] -= dx * h[i + d]
        A_ub[r, n] = -1.0
    b_ub = F - (A_ub[:, :n] @ h)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success and res.x is not None:
            x_new = _project(res.x[:n], n / 2.0)
            # Line search from h to x_new
            v = x_new - h
            t, c_new = _lsearch(h, v, nc=30, nf=20)
            if t > 0 and c_new < _exact_c5(h):
                return _project(h + t * v, n / 2.0), c_new
    except Exception:
        pass
    return h, _exact_c5(h)

def _coord_descent(h, deadline, seed):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = _exact_c5(h)
    rng = np.random.default_rng(seed)
    idx = np.arange(n)
    rng.shuffle(idx)
    stall = 0
    
    while time.monotonic() < deadline - 1.0 and stall < 500:
        i = idx[rng.integers(n)]
        if h[i] < 1e-10 or h[i] > 1.0 - 1e-10:
            stall += 1
            continue
        # Try to adjust h[i] and compensate with another index
        # Use line search in the e_i - e_j direction
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        d = 1.0
        v = np.zeros(n)
        v[i] = d
        v[j] = -d
        t_lo, t_hi = _frange(h, v)
        if t_hi - t_lo < 1e-12:
            continue
        # Quick eval
        best_t = 0.0
        best_c_local = best_c
        for t in np.linspace(t_lo, t_hi, 16):
            if abs(t) < 1e-14:
                continue
            ht = h + t * v
            if ht.min() < -1e-12 or ht.max() > 1 + 1e-12:
                continue
            c = _exact_c5(ht)
            if c < best_c_local:
                best_c_local = c
                best_t = t
        if best_t != 0.0 and best_c_local < best_c - 1e-13:
            h = _project(h + best_t * v, n / 2.0)
            c_new = _exact_c5(h)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s
    reserve = max(10.0, 0.05 * budget_s)
    op_deadline = deadline - reserve
    
    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError, ValueError):
        h0 = np.zeros(96)
        h0[::2] = 1.0
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    
    best_h = h0.copy()
    best_c = _exact_c5(h0)
    best_n = n0
    
    # Phase 1: Main optimization on original resolution (55% of time)
    dl1 = t0 + budget_s * 0.55
    h1, c1 = _main_loop(best_h, dl1, seed)
    if c1 < best_c:
        best_c = c1
        best_h = h1
    best_n = len(best_h)
    
    # Phase 2: Coordinate descent (10%)
    dl2 = t0 + budget_s * 0.65
    h2, c2 = _coord_descent(best_h, dl2, seed + 1)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    best_n = len(best_h)
    
    # Phase 3: Multi-resolution (20%)
    dl3 = t0 + budget_s * 0.85
    n_candidates = sorted(set([
        max(32, best_n - 32), max(32, best_n - 16), max(32, best_n - 8),
        best_n,
        best_n + 8, best_n + 16, best_n + 32,
        48, 56, 64, 80, 96, 112, 128, 144, 160
    ]))
    n_candidates = [x for x in n_candidates if 32 <= x <= 200]
    # Sort by proximity to best_n, prefer smaller on tie
    n_candidates.sort(key=lambda x: (abs(x - best_n), x))
    
    per_n_time = (dl3 - time.monotonic()) / max(1, len(n_candidates))
    per_n_time = max(8.0, min(60.0, per_n_time))
    
    for n_try in n_candidates:
        if time.monotonic() > dl3 - 5.0:
            break
        if n_try == best_n:
            continue
        h_r = _resample(best_h, n_try)
        dl_r = min(dl3 - 2.0, time.monotonic() + per_n_time)
        if dl_r - time.monotonic() < 3.0:
            continue
        h_rq, c_rq = _main_loop(h_r, dl_r, seed + n_try)
        if c_rq < best_c - 1e-10:
            best_c = c_rq
            best_h = h_rq
            best_n = n_try
    
    # Phase 4: Final polish (remaining time before reserve)
    dl4 = deadline - reserve
    if time.monotonic() < dl4 - 5.0:
        h4, c4 = _main_loop(best_h, dl4 - 3.0, seed + 100)
        if c4 < best_c:
            best_c = c4
            best_h = h4
        best_n = len(best_h)
    
    if time.monotonic() < dl4 - 2.0:
        h5, c5 = _coord_descent(best_h, dl4 - 1.0, seed + 200)
        if c5 < best_c:
            best_c = c5
            best_h = h5
        best_n = len(best_h)
    
    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c = _exact_c5(best_h)
    if final_c <= best_c + 1e-12:
        best_c = final_c
    else:
        best_c = final_c
    
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project(best_h, best_n / 2.0)
        best_c = _exact_c5(best_h)
    
    return best_h, best_c, best_n
