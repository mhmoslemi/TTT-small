    best_c5 = cur_c5

- v = 1
- A breakpoint: where A_r + B_r * v = A_s + B_s * v for some pair (r,s), i.e., v = (A_s - A_r) / (B_r - B_s) (if B_r ≠ B_s)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()

        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

    exp_F = np.exp(shifted)
    exp_sum = np.sum(exp_F)

                break
            orig = cur[i]
            if orig <= 1e-8 or orig >= 1.0 - 1e-8:
                continue
            # Try a few values around current
            for dv in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02, 0.15, -0.15]:
                v = orig + dv
                if v < 0 or v > 1:
                    continue
                cur[i] = v
                cur = _project_box_sum(cur, target)
                c = _exact_c5(cur)
                if c < cur_c5 - 1e-14:
                    cur_c5 = c
                    improved = True
                    if c < best_c5:
                        best_c5 = c
                        best = cur.copy()
                    break
                elif c < cur_c5:
                    # still try other values
                    pass
                else:
                    cur[i] = orig
                    cur = _project_box_sum(cur, target)
            else:
                # no improvement found
                cur[i] = orig
                cur = _project_box_sum(cur, target)
                stall += 1
        if not improved:
            break
    
    return best, best_c5
