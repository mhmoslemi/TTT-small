import numpy as np
import time
from scipy.optimize import linprog, differential_evolution

def _c5_exact(h):
    """Compute the exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}."""
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    # Final correction
    s = float(np.sum(v))
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            v[mask] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

    """Build gradient rows for active lags."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(active_indices), n))

def _harmonic_params_to_h(params, n, freqs):
    """Convert harmonic parameters to h vector."""
    M = len(freqs)
    b = params[0]
    x = (np.arange(n) + 0.5) / n  # normalized position in [0,1)
    z = np.full(n, b)
    for j in range(M):
        f = freqs[j]
        a = params[1 + 2*j]
        c = params[2 + 2*j]
        z += a * np.cos(2.0 * np.pi * f * x)
        z += c * np.sin(2.0 * np.pi * f * x)
    z = np.clip(z, -12.0, 12.0)
    h_raw = 1.0 / (1.0 + np.exp(-z))
    return _project_box_sum(h_raw, n / 2.0)

def _freqs_integer(n, M):
    """Integer frequencies 1, 2, ..., M."""
    return np.arange(1, M + 1, dtype=float)

def _freqs_golden(n, M):
    """Golden ratio based frequencies."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    freqs = []
    for k in range(1, M + 1):
        f = n * ((k * phi) % 1.0)
        f = max(1, int(round(f)))
        if f not in freqs:
            freqs.append(f)
        else:
            f = max(1, f + 1)
            while f in freqs:

            freqs.append(f)
    return np.array(freqs[:M], dtype=float)

def _freqs_random(n, M, rng):

    while len(freqs) < M:

        freqs.add(int(f))

    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(F)

def _c5_exact(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    if n == 0:
        return np.array([])
    if target <= 1e-15:
        return np.zeros(n)
    if target >= n - 1e-15:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0

        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:

        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _active_grads(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    K = len(active_indices)
    G = np.zeros((K, n))
    for idx, r in enumerate(active_indices):
        d = r - (n - 1)
        low = max(0, -d)
        high = min(n, n - d)
        for i in range(low, high):
            j = i + d
            G[idx, i] += dx * (1.0 - h[j])
            if j != i:
                G[idx, j] += -dx * h[i]
    return G

def _harmonic_to_h(params, n, freqs):
    M = len(freqs)
    x = (np.arange(n) + 0.5) / n
    z = np.full(n, params[0])
    for j in range(M):
        f = freqs[j]
        a = params[1 + 2 * j]
        c = params[2 + 2 * j]
        z += a * np.cos(2.0 * np.pi * f * x)
        z += c * np.sin(2.0 * np.pi * f * x)
    z = np.clip(z, -15.0, 15.0)

            stall += 1
            radius *= 0.72
    return best_h, best_c5

def _pair_line_search(h_orig, i, j, n_grid=25):
    n = len(h_orig)
    hi = h_orig[i]
    hj = h_orig[j]
    t_min = max(-hi, hj - 1.0)
    t_max = min(1.0 - hi, hj)
    if t_max - t_min < 1e-10:
        return h_orig.copy()
    h = h_orig.copy()
    best_t = 0.0
    best_val = _c5_exact(h_orig)
    t_vals = np.linspace(t_min, t_max, n_grid)
    for t in t_vals:
        h[i] = hi + t
        h[j] = hj - t
        val = _c5_exact(h)
        if val < best_val:
            best_val = val
            best_t = t
    if best_t != 0.0:
        h[i] = hi + best_t
        h[j] = hj - best_t
    else:
        h = h_orig.copy()
    return h

def _pair_refine(h, deadline, rng, max_iters=50000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    cur_c5 = best_c5
    no_improve = 0
    idx = np.arange(n)
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        i = int(idx[rng.integers(n)])
        j = int(idx[rng.integers(n)])
        if i == j:
            continue
        h_new = _pair_line_search(h, i, j, n_grid=15)
        if h_new is not h:
            c5_new = _c5_exact(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5

def _grad_pair_refine(h, deadline, rng, max_iters=50000):
    """Pair search prioritized by gradient differences."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    cur_c5 = best_c5
    no_improve = 0
    
    while time.time() < deadline and no_improve < 5000:
        F = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_F = float(np.max(F))
        tol = 0.02 * max_F + 1e-5
        active = np.where(F >= max_F - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(F)])
        G = _active_grads(h, active)
        grad_sum = np.sum(G, axis=0)
        
        for _ in range(200):
            if time.time() > deadline or no_improve > 5000:
                break
            if cur_c5 < best_c5 - 5e-6:
                break
            # Pick pair with large gradient difference
            if rng.random() < 0.7:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
            else:
                gi = int(np.argmax(np.abs(grad_sum)))
                gj = int(np.argmin(grad_sum[gi]) if False else np.argmin(np.abs(grad_sum - grad_sum[gi])))
                if gj == gi:
                    gj = (gi + 1) % n
                i, j = gi, gj
            if i == j:
                continue
            hi = h[i]
            hj = h[j]
            if hi <= 1e-9 or hi >= 1.0 - 1e-9 or hj <= 1e-9 or hj >= 1.0 - 1e-9:
                no_improve += 1
                continue
            t_min = max(-hi, hj - 1.0)
            t_max = min(1.0 - hi, hj)
            if t_max - t_min < 1e-10:
                no_improve += 1
                continue
            n_grid = 10
            t_vals = np.linspace(t_min, t_max, n_grid)
            best_t = 0.0
            best_val = cur_c5
            for t in t_vals:
                h[i] = hi + t
                h[j] = hj - t
                val = _c5_exact(h)
                if val < best_val:
                    best_val = val
                    best_t = t
            if best_t != 0.0:
                h[i] = hi + best_t
                h[j] = hj - best_t
                cur_c5 = best_val
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_improve = 0
            else:
                h[i] = hi
                h[j] = hj
                no_improve += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _random_feasible(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.permutation(n)[:n_ones]
    h[idx] = 1.0
    return h

def _random_feasible_soft(n, rng, sigma=0.3):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.permutation(n)[:n_ones]
    h[idx] = 1.0
    h = h + sigma * rng.standard_normal(n)
    return _project_box_sum(h, n / 2.0)

def _subgrad_refine(h, deadline, n_beta=10):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    cur_c5 = best_c5
    betas = np.geomspace(5.0, 200.0, n_beta)
    dx = 2.0 / n
    
    for beta in betas:
        if time.time() > deadline:
            break
        stall = 0
        step = 0.05
        for _ in range(3000):
            if time.time() > deadline:
                break
            if stall > 40:
                break
            scores = np.correlate(h, 1.0 - h, mode='full') * dx
            max_s = float(np.max(scores))
            exp_v = np.exp(beta * (scores - max_s))
            w = exp_v / (np.sum(exp_v) + 1e-300)
            # Compute gradient via active rows
            tol = 0.01 * max_s + 1e-5
            active = np.where(scores >= max_s - tol)[0]
            G = _active_grads(h, active)
            g = G.T @ w[:len(active)]
            g -= np.mean(g)
            gn = float(np.max(np.abs(g)))
            if gn < 1e-15:
                break
            d = -g / gn * step
            best_a = 0.0
            best_tc = cur_c5
            for a in [1.0, 0.6, 0.3, 0.15]:
                trial = _project_box_sum(h + a * d, n / 2.0)
                tc = _c5_exact(trial)
                if tc < best_tc - 1e-13:
                    best_tc = tc
                    best_a = a
            if best_a > 0:
                h = _project_box_sum(h + best_a * d, n / 2.0)
                cur_c5 = _c5_exact(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                step = min(step * 1.05, 0.2)
            else:
                stall += 1
                step *= 0.7
    return best_h, best_c5
