```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
```

Let me define: for k = 0, ..., 2n-2, let d = k - (n-1).

Now, if I change h[i] by +δ and h[j] by -δ:

The terms that change are those where m = i or m+d = i or m = j or m+d = j.

1. m = i: term is h[i] * (1 - h[i+d]). Change: +δ * (1 - h[i+d]) if 0 ≤ i+d < n, else 0 (if i+d out of range, the term doesn't exist).

2. m + d = i, i.e., m = i - d: term is h[i-d] * (1 - h[i]). Change: h[i-d] * (-δ) if 0 ≤ i-d < n.

3. m = j: term is h[j] * (1 - h[j+d]). Change: -δ * (1 - h[j+d]) if 0 ≤ j+d < n.

4. m + d = j, i.e., m = j - d: term is h[j-d] * (1 - h[j]). Change: h[j-d] * (+δ) if 0 ≤ j-d < n.

```python
def _compute_w(h, i, j):
    n = len(h)
    center = n - 1
    n_corr = 2 * n - 1
    w = np.zeros(n_corr)
    
    # For each correlation index k, d = k - center
    # w[k] = (1-h[i+d]) if 0<=i+d<n else 0
    #       - h[i-d] if 0<=i-d<n else 0
    #       - (1-h[j+d]) if 0<=j+d<n else 0
    #       + h[j-d] if 0<=j-d<n else 0
    
    d_vals = np.arange(n_corr) - center  # -center, -center+1, ..., center
    
    # (1-h[i+d]) for valid i+d
    idx = i + d_vals
    mask = (idx >= 0) & (idx < n)
    w[mask] += (1.0 - h[idx[mask]])
    
    # -h[i-d] for valid i-d
    idx = i - d_vals
    mask = (idx >= 0) & (idx < n)
    w[mask] -= h[idx[mask]]
    
    # -(1-h[j+d]) for valid j+d
    idx = j + d_vals
    mask = (idx >= 0) & (idx < n)
    w[mask] -= (1.0 - h[idx[mask]])
    
    # +h[j-d] for valid j-d
    idx = j - d_vals
    mask = (idx >= 0) & (idx < n)
    w[mask] += h[idx[mask]]
    
    return w
```
