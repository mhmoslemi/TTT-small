            if m > 0:
                grad[:m] += (1.0 - h[d:d+m])
                grad[d:d+m] -= h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] -= h[g:g+m]
                grad[g:g+m] += (1.0 - h[:m])
    return grad

def _softmax_grad(h, beta, top_k=None):
    """Compute softmax-weighted gradient of max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = corr_raw * dx
    # Softmax weights
    shifted = beta * (scores - scores.max())
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    if top_k is not None:
        top_idx = np.argsort(w)[-top_k:]
        w = np.zeros_like(w)
        w[top_idx] = 1.0 / top_k
    # Gradient
    grad = np.zeros(n)
    nonzero = w > 1e-10
    shifts = np.where(nonzero)[0]
    grad = _grad_at_shifts(h, shifts)
    # Scale by weights
    # Actually _grad_at_shifts just sums, we need weighted sum
    grad = np.zeros(n)
    for r in shifts:
        if w[r] < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:d+m] -= w[r] * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] -= w[r] * h[g:g+m]
                grad[g:g+m] += w[r] * (1.0 - h[:m])
    return grad, scores

def _line_search_quad(h, d, dx, max_step):
    """Line search using quadratic model of correlation."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    # d is the direction, we look at h + t*d
    # corr(t) = S0 + t * T - t^2 * C where:
    # T = corr_raw(d, 1-h) - corr_raw(h, d) ... 
    # Actually: (h+td)_i * (1-(h+td)_i+td) ... this is getting complex.
    # Let me use the expansion:
    # f(t) = sum_i (h[i]+t*d[i]) * (1-h[i]-t*d[i] + t*d[i+shift])
    # For shift d_k: f_k(t) = sum_i (h[i]+t*d[i]) * (1-h[i+k]-t*d[i+k])
    # = sum_i h[i](1-h[i+k]) + t*sum_i d[i](1-h[i+k]) - t*sum_i h[i]d[i+k] - t^2*sum_i d[i]d[i+k]
    # = S0_k + t*(U_k - V_k) - t^2 * W_k
    # where U_k = sum_i d[i](1-h[i+k]), V_k = sum_i h[i]d[i+k], W_k = sum_i d[i]d[i+k]
    
    # U_k = sum_i d[i] - sum_i d[i]h[i+k]
    # V_k = sum_i h[i]d[i+k]
    # Note: U_k - V_k = sum_i d[i](1-h[i+k]) - sum_i h[i]d[i+k]
    #      = sum_i d[i] - sum_i d[i]h[i+k] - sum_i h[i]d[i+k]
    # Let me just compute T_k = U_k - V_k and W_k directly.
    
    # T_k = sum_i d[i](1-h[i+k]) - sum_i h[i]d[i+k]
    # For shift k (where k = r - (n-1)):
    # If k > 0: sum over i=0 to n-1-k
    #   d[i](1-h[i+k]) - h[i]d[i+k]
    # If k < 0: let g=-k, sum over i=g to n-1
    #   d[i](1-h[i-g]) - h[i-g]d[i]
    #   = sum over j=0 to n-1-g [d[i](1-h[i-g]) - h[i-g]d[i]] where i=j+g
    #   = sum over j=0 to n-1-g [d[j+g](1-h[j]) - h[j]d[j+g]]
    
    # This is O(n^2) to compute for all shifts. Let me use a different approach.
    # corr_raw(d, 1-h) gives U_k for all k
    # corr_raw(h, d) gives V_k for all k  (but need to be careful with indices)
    # corr_raw(d, d) gives W_k for all k
    
    U = np.correlate(d, 1.0 - h, mode='full')  # sum_i d[i] * (1-h)[i + k - (n-1)]
    V = np.correlate(h, d, mode='full')  # sum_i h[i] * d[i + k - (n-1)]
    W = np.correlate(d, d, mode='full')  # sum_i d[i] * d[i + k - (n-1)]
    
    T = U - V
    C = W  # coefficient of -t^2
    
    # f_k(t) = S0[k] + t*T[k] - t^2*C[k]
    # We want to minimize max_k f_k(t) * dx over t in [t_lo, t_hi]
    
    # Find feasible range
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo, t_hi = 0.0, max_step
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))  # h[neg] / (-d[neg]) ... wait
    # If d[i] < 0, we need h[i] + t*d[i] >= 0, so t <= h[i]/(-d[i])
    # So t_lo should be... actually t_lo = 0 (we don't want to go negative t)
    # Wait, we allow t in [t_neg, t_pos] where t_neg could be negative
    # But typically we want t >= 0 (move in direction d)
    # Let me just use t in [0, t_hi]
    
    t_lo = 0.0
    t_hi = min(max_step, float(np.min((1.0 - h[pos]) / d[pos]))) if np.any(pos) else max_step
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi < 1e-14:
        return 0.0, _c5(h)
    
    # Grid search + golden section
    n_grid = 30
    grid = np.linspace(0, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None]**2) * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_obj = obj[best_i]
    
    # Refine with golden section
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    for _ in range(30):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_obj = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    if opt_obj < best_obj:
        best_t, best_obj = opt_t, opt_obj
    
    return best_t, best_obj
