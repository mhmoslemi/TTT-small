                h = ensure_feasible(h, target)
                current_c5 = exact_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            elif delta < 0.005 and rng.random() < np.exp(-delta / T):
                h[i] += a
                h[j] -= a
                h = ensure_feasible(h, target)
                current_c5 = exact_c5(h)
        elif mode < 0.85:
            a1 = int(rng.integers(n))
            a2 = int(rng.integers(min(a1 + 5, n)))
            if a2 <= a1:
                continue
            h_new = segment_reverse(h, a1, a2)
            h_new = ensure_feasible(h_new, target)
            nc = exact_c5(h_new)
            delta = nc - current_c5
            if delta < -1e-13:
                h = h_new
                current_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
            elif delta < 0.005 and rng.random() < np.exp(-delta / T):
                h = h_new
                current_c5 = nc
        else:
            a1 = int(rng.integers(n))
            a2 = int(rng.integers(min(a1 + 8, n)))
            if a2 <= a1 + 1:
                continue
            shift = int(rng.integers(1, a2 - a1))
            h_new = segment_rotate(h, a1, a2, shift)
            h_new = ensure_feasible(h_new, target)
            nc = exact_c5(h_new)
            delta = nc - current_c5
            if delta < -1e-13:
                h = h_new
                current_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
            elif delta < 0.005 and rng.random() < np.exp(-delta / T):
                h = h_new
                current_c5 = nc
    return best_h, best_c5

def resolution_search(h_best, c5_best, deadline, rng):
    n_best = len(h_best)
    best_h = h_best.copy()
    best_c5 = c5_best
    best_n = n_best
    offsets = [0, -2, -4, 2, 4, -6, 6, -8, 8, -10, 10, -12, 12, -15, 15]
    tried = set()
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 200 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample_h(h_best, n_try)
        target = n_try / 2.0
        h_r, c5_r = active_lag_descent(h_r, target, min(deadline, time.time() + 15.0), n_pairs=15, tau_init=0.015, rng=rng)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            h_best = h_r.copy()
    return best_h, best_c5, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.95
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = ensure_feasible(h0, target0)
    best_c5 = exact_c5(h)
    best_h = h.copy()
    best_n = n0
    
    # Phase 1: Active-lag descent from parent
    d1 = t0 + budget_s * 0.20
    h, c5 = active_lag_descent(h, target0, d1, n_pairs=30, tau_init=0.015, rng=rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 2: Metropolis exploration
    d2 = t0 + budget_s * 0.45
    h_m, c5_m = metropolis_phase(best_h, target0, d2, rng, T_init=0.003)
    if c5_m < best_c5:
        best_c5 = c5_m
        best_h = h_m.copy()
    
    # Phase 3: Active-lag descent after exploration
    d3 = t0 + budget_s * 0.65
    h, c5 = active_lag_descent(best_h, target0, d3, n_pairs=30, tau_init=0.008, rng=rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 4: Resolution search
    d4 = t0 + budget_s * 0.80
    best_h, best_c5, best_n = resolution_search(best_h, best_c5, d4, rng)
    target_best = best_n / 2.0
    
    # Phase 5: Another exploration round
    d5 = t0 + budget_s * 0.90
    h_m2, c5_m2 = metropolis_phase(best_h, target_best, d5, rng, T_init=0.001)
    if c5_m2 < best_c5:
        best_c5 = c5_m2
        best_h = h_m2.copy()
        target_best = best_n / 2.0
    
    # Phase 6: Final active-lag descent
    d6 = t0 + budget_s * 0.94
    h_fp, c5_fp = active_lag_descent(best_h, target_best, d6, n_pairs=20, tau_init=0.005, rng=rng)
    if c5_fp < best_c5:
        best_c5 = c5_fp
        best_h = h_fp.copy()
        target_best = best_n / 2.0
    
    # Phase 7: Full pair sweep final polish
    if time.time() < hard_deadline:
        h_fp2, c5_fp2 = full_pair_sweep(best_h, target_best, max_passes=2, deadline=hard_deadline)
        if c5_fp2 < best_c5:
            best_c5 = c5_fp2
            best_h = h_fp2.copy()
    
    # Final validation
    best_h = ensure_feasible(best_h, target_best)
    final_c5 = exact_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        final_c5 = best_c5
    
    # Safety: never return worse than parent
    parent_c5 = exact_c5(ensure_feasible(h0, target0))
    if final_c5 > parent_c5:
        best_h = ensure_feasible(h0, target0)
        final_c5 = parent_c5
        best_n = n0
    
    return best_h, final_c5, best_n
