def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, tr):

    S0 = np.correlate(h, 1.0-h, mode='full')
    T = np.correlate(d, 1.0-h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo, t_hi = -tr, tr
    if np.any(pos): t_hi = min(t_hi, float(np.min((1.0-h[pos])/d[pos])))
    if np.any(neg): t_lo = max(t_lo, float(np.max(h[neg]/d[neg])))
    if t_hi - t_lo < 1e-14: return 0.0, _c5_of(h)
    # Grid + golden section

    vals = S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:]
    obj = np.max(vals, axis=1) * dx
    bi = int(np.argmin(obj))
    bt, bv = grid[bi], obj[bi]
    a = grid[max(0,bi-1)]; b = grid[min(len(grid)-1,bi+1)]
    gr = (np.sqrt(5.0)-1.0)/2.0

            f2 = float(np.max(S0+c2*T-c2*c2*C))*dx
    ot = (a+b)/2.0
    ov = float(np.max(S0+ot*T-ot*ot*C))*dx
    if ov < bv: bt, bv = ot, ov
    return bt, bv

def _lp_dir(h, scores, tr):
    n = len(h)
    # Build gradient matrix for active constraints
    # Only use top active constraints for efficiency
    active = np.argsort(scores)[::-1][:min(40, len(scores))]
    G = np.zeros((len(active), n))
    for idx, r in enumerate(active):
        d = r - (n-1)
        if d >= 0:
            m = n - d
            G[idx, d:] += (1.0 - h[:m])
            G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            G[idx, :m] += (1.0 - h[g:])
            G[idx, g:] += -h[:m]
    c = np.zeros(n+1); c[-1] = 1.0
    A_ub = np.zeros((len(active), n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n+1)); A_eq[0,:n] = 1.0
    bounds = [(max(-tr,-h[j]), min(tr,1.0-h[j])) for j in range(n)] + [(None,None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success: return res.x[:n]
    except: pass
    return None

    best_h, best_c5 = h.copy(), _c5_of(h)

        else:

    return best_h, best_c5
