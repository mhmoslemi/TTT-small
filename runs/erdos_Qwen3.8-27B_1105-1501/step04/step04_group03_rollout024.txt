def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
            # Final fix
            residual2 = target - np.sum(x)
            if abs(residual2) > 1e-14:
                if residual2 > 0:
                    idx = int(np.argmax(1.0 - x))
                else:
                    idx = int(np.argmin(x))
                x[idx] += residual2
                x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_matrix(h):
    """Build gradient matrix G where G[r, j] = d(score_r)/d(h[j])"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] -= h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] -= h[:m]
    return G

def project(v, target):

            lo = mid
        else:
            hi = mid

    if abs(residual) > 1e-12:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))

def build_grad_matrix(h):

        else:

            m = n - g

        h_t[i_dec] -= amax
        c5_t = exact_c5(h_t)
        if c5_t >= current_c5 - 1e-14:
            return None, h, current_c5
        lo_t, hi_t = 0.0, amax
        for _ in range(10):
            mid_t = (lo_t + hi_t) / 2.0
            h_mid = h.copy()
            h_mid[i_inc] += mid_t
            h_mid[i_dec] -= mid_t
            if exact_c5(h_mid) < current_c5:
                lo_t = mid_t
            else:
                hi_t = mid_t
        t_opt = lo_t
        if t_opt < 1e-10:
            return None, h, current_c5
        h[i_inc] += t_opt
        h[i_dec] -= t_opt
        h = project(h, target)
        cur = exact_c5(h)
        return cur, h, cur

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        cur_c5 = exact_c5(h)
        scores_raw = np.correlate(h, 1.0 - h, mode='full')
        scores = scores_raw * (2.0 / n)
        top_idx = np.argsort(scores)[-top_lags:]
        improved = False
        for lag_idx in top_idx:
            if time.time() > deadline:
                break
            d = lag_idx - (n - 1)
            if d == 0:
                continue
            if d >= 0:
                m = n - d
                if m <= 0:
                    continue
                contrib = h[d:d+m] * (1.0 - h[:m])
                order = np.argsort(contrib)[::-1]
                for ci in range(min(top_contribs, m)):
                    ic = int(order[ci])
                    if contrib[ic] < 1e-12:
                        break
                    i_inc = ic
                    i_dec = ic + d
                    if i_inc == i_dec:
                        continue
                    res, h, cur_c5 = try_pair_move2(h, i_inc, i_dec, cur_c5)
                    if res is not None and cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        improved = True
            else:
                g = -d
                m = n - g
                if m <= 0:
                    continue
                contrib = h[:m] * (1.0 - h[g:g+m])
                order = np.argsort(contrib)[::-1]
                for ci in range(min(top_contribs, m)):
                    ic = int(order[ci])
                    if contrib[ic] < 1e-12:
                        break
                    i_dec = ic
                    i_inc = ic + g
                    if i_dec == i_inc:
                        continue
                    res, h, cur_c5 = try_pair_move(h, i_dec, i_inc, cur_c5)
                    if res is not None and cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        improved = True
        if not improved:
            new_top = top_lags + 2
            if new_top > 20:
                break
            top_lags = new_top
            if rnd > 50:
                break
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def neighborhood_search(h, deadline, n_trials=200):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(int(time.time()) % 10000 + n)
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        amax = min(h[i], 1.0 - h[j])
        if amax < 1e-10:
            amax = min(1.0 - h[i], h[j])
            if amax < 1e-10:
                continue
            sign = 1
        else:
            sign = -1
        h_t = h.copy()
        if sign < 0:
            h_t[i] -= amax
            h_t[j] += amax
        else:
            h_t[i] += amax
            h_t[j] -= amax
        c5_t = exact_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            lo_t, hi_t = 0.0, amax
            for _ in range(8):
                mid_t = (lo_t + hi_t) / 2.0
                h_mid = h.copy()
                if sign < 0:
                    h_mid[i] -= mid_t
                    h_mid[j] += mid_t
                else:
                    h_mid[i] += mid_t
                    h_mid[j] -= mid_t
                if exact_c5(h_mid) < best_c5:
                    lo_t = mid_t
                else:
                    hi_t = mid_t
            t_opt = lo_t
            if sign < 0:
                h[i] -= t_opt
                h[j] += t_opt
            else:
                h[i] += t_opt
                h[j] -= t_opt
            h = project(h, target)
            best_c5 = exact_c5(h)
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.97

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: LP local search
    d1 = t0 + budget_s * 0.12
    h_lp, c5_lp = lp_loop(h, d1, radius_init=0.15, k_active=30)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: Targeted pair moves
    d2 = t0 + budget_s * 0.25
    h_tp, c5_tp = targeted_pairs(best_h, d2, max_rounds=500, top_lags=5, top_contribs=4)
    if c5_tp < best_c5:
        best_c5 = c5_tp
        best_h = h_tp.copy()

    # Phase 3: LP refinement
    d3 = t0 + budget_s * 0.35
    h_lp2, c5_lp2 = lp_loop(best_h, d3, radius_init=0.08, k_active=20)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 4: Targeted pairs again
    d4 = t0 + budget_s * 0.45
    h_tp2, c5_tp2 = targeted_pairs(best_h, d4, max_rounds=400, top_lags=8, top_contribs=5)
    if c5_tp2 < best_c5:
        best_c5 = c5_tp2
        best_h = h_tp2.copy()

    # Phase 5: Multi-resolution
    d5 = t0 + budget_s * 0.70
    n_candidates = []
    for dn in [10, 20, 30, 40, 50, 60, 70, -10, -20, -30, 15, 25, 35, 45]:
        nt = best_n + dn
        if 60 <= nt <= 200:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))
    tried_n = set([best_n])
    for nt in n_candidates:
        if time.time() > d5:
            break
        if nt in tried_n:
            continue
        tried_n.add(nt)
        h_r = resample_h(best_h, nt)
        sub_dl = min(d5, time.time() + budget_s * 0.025)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.12, k_active=25)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt
            # Short targeted pairs on new best
            if time.time() < d5:
                sub_dl2 = min(d5, time.time() + budget_s * 0.01)
                h_r2, c5_r2 = targeted_pairs(h_r, sub_dl2, max_rounds=100, top_lags=5, top_contribs=3)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase 6: Perturbation restarts
    d6 = t0 + budget_s * 0.85
    scales = [0.008, 0.015, 0.025, 0.04, 0.06]
    for ns in scales:
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        c5_pert = exact_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
        sub_dl = min(d6, time.time() + budget_s * 0.02)
        h_p2, c5_p2 = lp_loop(h_pert, sub_dl, radius_init=0.08, k_active=20)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()
        if time.time() < d6:
            sub_dl2 = min(d6, time.time() + budget_s * 0.012)
            h_p3, c5_p3 = targeted_pairs(h_p2, sub_dl2, max_rounds=150, top_lags=5, top_contribs=3)
            if c5_p3 < best_c5:
                best_c5 = c5_p3
                best_h = h_p3.copy()

    # Phase 7: Final polish - LP + targeted
    if time.time() < hard_deadline:
        d7 = hard_deadline - 5.0
        h_f, c5_f = lp_loop(best_h, d7, radius_init=0.04, k_active=15)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if time.time() < hard_deadline:
        h_t3, c5_t3 = targeted_pairs(best_h, hard_deadline - 2.0, max_rounds=300, top_lags=6, top_contribs=5)
        if c5_t3 < best_c5:
            best_c5 = c5_t3
            best_h = h_t3.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
