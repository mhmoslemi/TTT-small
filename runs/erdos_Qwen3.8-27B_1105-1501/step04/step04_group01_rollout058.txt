def _coord_descent(h, deadline, max_sweeps=10000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # Compute a_r and b_r for coordinate j
            a = np.zeros(2*n-1)
            for r in range(2*n-1):
                d = r - (n-1)
                jd = j + d
                jmd = j - d
                coeff = 0.0
                if 0 <= jd < n:
                    coeff += (1.0 - h[jd])
                if 0 <= jmd < n:
                    coeff -= h[jmd]
                a[r] = coeff
            b = corr - h[j] * a
            
            # Find v* minimizing max_r(a_r * v + b_r) over v in [0, 1]
            v_star = _min_convex_pl(a, b, 0.0, 1.0)
            
            if abs(v_star - h[j]) > 1e-12:

                # Project to maintain sum constraint and box
                h = _project(h, target)
                # Recompute corr (expensive but correct)
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _min_convex_pl(a, b, lo, hi):
    """Find v in [lo, hi] minimizing max_r(a_r * v + b_r)"""
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        min_slope = a[active].min()
        max_slope = a[active].max()
        if min_slope > 1e-14:
            hi = mid
        elif max_slope < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _min_convex_pl(a, b, lo, hi):
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        mx = vals.max()
        active = vals >= mx - 1e-11
        ms = a[active].min()
        xs = a[active].max()
        if ms > 1e-13:
            hi = mid
        elif xs < -1e-13:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def _coord_descent(h, deadline, max_sweeps=500000):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    d_arr = np.arange(-(n - 1), n)
    ns = 2 * n - 1
    stale = 0
    resync_period = 60

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_imp = False
        for j in range(n):
            if time.time() > deadline:
                break
            jd = j + d_arr
            jmd = j - d_arr
            a = np.zeros(ns)
            m1 = (jd >= 0) & (jd < n)
            m2 = (jmd >= 0) & (jmd < n)
            if m1.any():
                a[m1] = 1.0 - h[jd[m1]]
            if m2.any():
                a[m2] -= h[jmd[m2]]
            b = corr - h[j] * a
            vs = _min_convex_pl(a, b, 0.0, 1.0)
            delta = vs - h[j]
            if abs(delta) > 1e-11:
                h[j] += delta
                if n > 1:
                    inv = 1.0 / (n - 1)
                    h[:j] -= delta * inv
                    if j + 1 < n:
                        h[j + 1:] -= delta * inv
                np.clip(h, 0.0, 1.0, out=h)
                corr += delta * a
                any_imp = True

        if not any_imp:
            stale += 1
            if stale > 30:
                break
        else:
            stale = 0

        if sweep % resync_period == 0:
            h = _project(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()

    h = _project(h, target)
    cur = _exact_c5(h)
    if cur < best_c5:
        best_c5 = cur
        best_h = h.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)

def _perturb_optimize(h, budget, rng, scale, deadline):
    n = len(h)
    target = n * 0.5
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    sub_deadline = min(deadline, time.time() + budget)
    h_p = h + rng.normal(0, scale, n)
    h_p = _project(h_p, target)
    h_r, c5_r = _coord_descent(h_p, sub_deadline, max_sweeps=30000)
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()
    return best_h, best_c5
