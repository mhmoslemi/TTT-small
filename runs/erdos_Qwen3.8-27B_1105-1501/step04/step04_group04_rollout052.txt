        r = pos - (n - 1)  # actual lag
        w_r = w[pos]
        if w_r < 1e-15:
            continue
        # contribution from (1 - h_{j+r}) term: for j in [0, n), j+r in [0,n)
        # i.e., j in [-r, n-1-r) intersect [0, n)

def _exact_c5(h):

def _project_box_sum(v, target):

def _grad_smooth_max(h, beta, top_k=None):
    """Vectorized gradient of smooth max of correlation terms."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Numerically stable softmax
    F_max = F.max()
    w = np.exp(beta * (F - F_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        # All equal, uniform weights
        w = np.ones(L) / L
    else:
        w /= w_sum
    
    # Select top-k for efficiency
    if top_k is not None and top_k < L:
        k = min(top_k, L)
        idx = np.argpartition(w, -k)[-k:]
        w_sel = np.zeros(L)
        w_sel[idx] = w[idx]
        w_sel /= w_sel.sum()
        w = w_sel
    
    # Compute gradient
    # For each lag r (where r = pos - (n-1)):
    # grad += w_r * dx * [contribution from (1-h_{j+r}) - contribution from h_{j-r}]
    grad = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    
    one_minus_h = 1.0 - h
    
    for pos in active:
        r = pos - (n - 1)
        w_r = w[pos]
        if w_r * dx < 1e-16:
            continue
        
        # (1 - h_{j+r}) term: valid when 0 <= j < n and 0 <= j+r < n
        # j in [max(0,-r), min(n, n-r))
        lo1 = max(0, -r)
        hi1 = min(n, n - r)
        if lo1 < hi1:
            grad[lo1:hi1] += w_r * dx * one_minus_h[lo1 + r:hi1 + r]
        
        # -h_{j-r} term: valid when 0 <= j < n and 0 <= j-r < n
        # j in [max(0,r), min(n, n+r))
        lo2 = max(0, r)
        hi2 = min(n, n + r)
        if lo2 < hi2:
            grad[lo2:hi2] -= w_r * dx * h[lo2 - r:hi2 - r]
    
    return grad, F, w

def _smooth_descend(h, deadline, beta_start=5, beta_end=3000, n_beta=25,
                     top_k=50, use_momentum=True):

            break
        
        tk = min(top_k, 2 * n - 1)

        step_size = 0.04

                break
            if stall > max_stall:

            no_improve += 1
            continue
        if h[i2] <= 1e-10 or h[i2] >= 1.0 - 1e-10:
            no_improve += 1
            continue
        
        # Try moving mass from i1 to i2
        # h[i1] decreases by d, h[i2] increases by d
        # Feasible range: d in (-(h[i1]), h[i2]-h[i1]+... ) 
        # Actually: h[i1]-d >= 0 and h[i2]+d <= 1
        # So d in [-h[i2]... wait
        # Let's say: new_h[i1] = h[i1] - d, new_h[i2] = h[i2] + d
        # Constraints: 0 <= h[i1]-d <= 1 and 0 <= h[i2]+d <= 1
        # d <= h[i1], d >= h[i1]-1 (redundant), d >= -h[i2], d <= 1-h[i2]
        d_max = min(h[i1], 1.0 - h[i2])
        d_min = max(-h[i2], h[i1] - 1.0)  # = -h[i2] since h[i1] <= 1
        if d_max <= d_min:
            no_improve += 1
            continue
        
        # Sample a few values of d
        d_vals = [d_max * 0.3, d_max * 0.7, d_max, -d_max * 0.3, -d_max * 0.7, -d_max]
        rng.shuffle(d_vals)
        d_vals = [d for d in d_vals if d_min < d < d_max]
        
        found = False
        for d in d_vals:
            if d == 0:

            h[i1] -= d
            h[i2] += d

                no_improve = 0
                found = True
                # Continue trying to improve further
            else:
                h[i1] += d
                h[i2] -= d
                break
        
        if not found:
            no_improve += 1
    
    return best_h, best_c5

def _pair_exact_min(h, i, j, n_evaluate=30):
    """Find optimal d for pair transfer h_i += d, h_j -= d."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    
    # Current F values
    F0 = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Compute c_r for each lag r
    c = np.zeros(L)
    r = np.arange(-(n-1), n)
    
    for pos in range(L):
        r_pos = pos - (n - 1)
        # c_r = (h_{j+r} - h_{i+r}) + (h_{j-r} - h_{i-r})
        val = 0.0
        jr = j + r_pos
        ir = i + r_pos
        jm = j - r_pos
        im = i - r_pos
        if 0 <= jr < n:
            val += h[jr]
        if 0 <= ir < n:
            val -= h[ir]
        if 0 <= jm < n:
            val += h[jm]
        if 0 <= im < n:
            val -= h[im]
        c[pos] = val
    
    # Feasible range for d
    d_max = min(h[i], 1.0 - h[j])
    d_min = max(-h[j], h[i] - 1.0)  # = -h[j] since h[i] <= 1
    # Wait: h_i + d >= 0 => d >= -h_i; h_j - d >= 0 => d <= h_j
    # h_i + d <= 1 => d <= 1 - h_i; h_j - d <= 1 => d >= h_j - 1
    d_max = min(1.0 - h[i], h[j])
    d_min = max(-h[i], h[j] - 1.0)
    
    if d_max <= d_min:
        return 0.0
    
    # Evaluate f(d) = max_r (F0_r + dx * d * c_r)
    def f(d):
        return float(np.max(F0 + dx * d * c))
    
    # Ternary search (convex function)
    a, b = d_min, d_max
    for _ in range(n_evaluate):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    
    d_opt = (a + b) / 2.0
    return d_opt

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).astype(float).copy()
    for _ in range(300):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-15
        else:
            mask = v > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    # Final exact correction
    diff = target - float(np.sum(v))
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmin(1.0 - v)
        else:
            idx = np.argmax(v)
        v[idx] = v[idx] + diff
        v = np.clip(v, 0.0, 1.0)
    return v

def _grad_smooth_max(h, beta, top_k=None):

    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    F_max = F.max()
    exponents = beta * (F - F_max)
    exponents = np.clip(exponents, -700, 700)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.full(L, 1.0 / L)
    else:
        w /= w_sum
    if top_k is not None and top_k < L:
        k = min(top_k, L)
        idx = np.argpartition(w, -k)[-k:]
        w_new = np.zeros(L)
        w_new[idx] = w[idx]
        s2 = w_new.sum()
        if s2 > 1e-300:
            w_new /= s2
            w = w_new
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    active = np.where(w > 1e-14)[0]
    for pos in active:
        r = pos - (n - 1)
        w_r = w[pos]
        wd = w_r * dx
        if wd < 1e-16:
            continue
        lo1 = max(0, -r)
        hi1 = min(n, n - r)
        if lo1 < hi1:

        lo2 = max(0, r)
        hi2 = min(n, n + r)
        if lo2 < hi2:

                    top_k=40, max_iters=50000):

    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    betas = np.geomspace(beta_start, beta_end, n_beta)
    for beta in betas:

        tk = min(top_k, 2 * n - 1)
        momentum = np.zeros(n)
        step_size = 0.04

            grad -= np.mean(grad)

            direction = -grad / gn
            momentum = 0.85 * momentum + 0.15 * direction
            best_a = 0.0
            best_tc5 = cur_c5

        m2 = b - (b - a) / 3.0
        f1 = float(np.max(F0 + dx * m1 * c))
        f2 = float(np.max(F0 + dx * m2 * c))
        if f1 < f2:
            b = m2
        else:
            a = m1
    d_opt = (a + b) / 2.0
    f_opt = float(np.max(F0 + dx * d_opt * c))
    if f_opt < f_0 - 1e-14:
        return d_opt
    return 0.0

def _pair_descent(h, deadline, rng, max_iters=200000):
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    no_improve = 0
    interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if len(interior) == 0:
        return h, cur_c5
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 3000:
            break
        if len(interior) < 2:
            break
        i = int(interior[rng.integers(len(interior))])
        j = int(interior[rng.integers(len(interior))])
        if i == j:
            continue
        d = _pair_minimize(h, i, j, n_steps=20)
        if d != 0.0:
            h[i] += d
            h[j] -= d
            h = np.clip(h, 0.0, 1.0)
            h = _project_box_sum(h, n / 2.0)
            tc5 = _exact_c5(h)
            if tc5 < cur_c5 - 1e-14:
                cur_c5 = tc5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                no_improve = 0
                interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
            else:
                h[i] -= d
                h[j] += d
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _active_set_descend(h, deadline, max_iters=50000):
    """Descend using only the argmax lag's gradient (faster per step)."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    step_size = 0.03
    stall = 0
    momentum = np.zeros(n)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        F = np.correlate(cur_h, 1.0 - cur_h, mode='full') * (2.0 / n)
        r_max = int(np.argmax(F))
        r = r_max - (n - 1)
        dx = 2.0 / n
        one_minus_h = 1.0 - cur_h
        grad = np.zeros(n)
        lo1 = max(0, -r)
        hi1 = min(n, n - r)
        if lo1 < hi1:
            grad[lo1:hi1] += dx * one_minus_h[lo1 + r:hi1 + r]
        lo2 = max(0, r)
        hi2 = min(n, n + r)
        if lo2 < hi2:
            grad[lo2:hi2] -= dx * cur_h[lo2 - r:hi2 - r]
        grad -= np.mean(grad)
        gn = np.sqrt(np.sum(grad * grad))
        if gn < 1e-15:
            break
        direction = -grad / gn
        momentum = 0.9 * momentum + 0.1 * direction
        best_a = 0.0
        best_tc5 = cur_c5
        for a in [step_size, step_size * 0.5, step_size * 0.25]:
            trial = _project_box_sum(cur_h + a * momentum, target)
            tc5 = _exact_c5(trial)
            if tc5 < best_tc5 - 1e-14:
                best_tc5 = tc5
                best_a = a
        if best_a > 0:
            cur_h = _project_box_sum(cur_h + best_a * momentum, target)
            cur_c5 = _exact_c5(cur_h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur_h.copy()
            stall = 0
            step_size = min(step_size * 1.04, 0.1)
        else:
            stall += 1
            step_size *= 0.9
            momentum *= 0.95
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        n0 = 128
        h0 = np.full(n0, 0.5)

    n0 = len(h0)
    h = _project_box_sum(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Smooth gradient continuation (main fast driver)
    dl1 = t0 + budget_s * 0.30
    h1, c5_1 = _smooth_descend(h, dl1, beta_start=5, beta_end=3000, n_beta=18, top_k=35)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Active set descent (focus on dominant lag)
    dl2 = t0 + budget_s * 0.42
    h2, c5_2 = _active_set_descend(best_h, dl2, max_iters=80000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Exact pair minimization
    dl3 = t0 + budget_s * 0.58
    h3, c5_3 = _pair_descent(best_h, dl3, rng, max_iters=150000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Another smooth pass from improved point
    dl4 = t0 + budget_s * 0.70
    h4, c5_4 = _smooth_descend(best_h, dl4, beta_start=30, beta_end=8000, n_beta=12, top_k=25)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution exploration
    dl5 = t0 + budget_s * 0.88
    n_tries = sorted(set([48, 64, 80, 96, 112, 128, 144, 160, 192]))
    n_tries = [x for x in n_tries if x != best_n]
    remaining = dl5 - time.time()
    per_n = remaining / max(len(n_tries), 1) * 0.9

    for n_try in n_tries:
        if time.time() > dl5:
            break
        if (dl5 - time.time()) < 8:
            break
        h_r = _resample(best_h, n_try)
        dl_r1 = min(dl5, time.time() + per_n * 0.6)
        h_r, c5_r = _smooth_descend(h_r, dl_r1, beta_start=10, beta_end=3000, n_beta=10, top_k=30)
        if time.time() < dl5 - 3:
            dl_r2 = min(dl5, time.time() + per_n * 0.3)
            h_r, c5_r2 = _pair_descent(h_r, dl_r2, rng, max_iters=30000)
            if c5_r2 < c5_r:
                c5_r = c5_r2
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Final polish
    if time.time() < deadline - 15:
        h_f, c5_f = _smooth_descend(best_h, deadline - 10, beta_start=50, beta_end=15000, n_beta=8, top_k=20)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < deadline - 5:
        h_p, c5_p = _pair_descent(best_h, deadline - 2, rng, max_iters=50000)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final projection and validation
    best_h = _project_box_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, best_c5, best_n
