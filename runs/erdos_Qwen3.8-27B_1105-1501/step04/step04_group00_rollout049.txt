    """Log-sum-exp smoothing of the max overlap."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_s = np.max(corr)
    shifted = corr - max_s
    # Use log-sum-exp trick for numerical stability
    # f = max_s + (1/beta) * log(sum(exp(beta * shifted)))
    # = max_s + (1/beta) * (max_shifted + log(sum(exp(shifted - max_shifted))))
    # Since max_shifted = 0:
    # = max_s + (1/beta) * log(sum(exp(shifted)))
    log_sum = np.log(np.sum(np.exp(shifted)))
    return max_s + log_sum / beta

def _smooth_grad(h, beta):
    """Gradient of the smooth objective w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

            m = n - g

                grad[:m] += w_r * (1.0 - h[g:])
                grad[g:] += w_r * (-h[:m])
    grad *= dx
    # Remove mean to orthogonalize against the sum constraint

def _grad_dot(h, w):
    """Compute G.T @ w efficiently without forming G."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(-(n-1), n):

        w_r = w[r]

            continue

            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w_r * (1.0 - h[g:])
                grad[g:] += w_r * (-h[:m])
    return grad * dx

def _lp_direction(h, scores, radius, top_k=None):
    """Find a direction p and step t using LP."""
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
    else:
        idx = np.arange(n_lags)
        scores_k = scores
    
    # Build constraint matrix rows for selected lags
    # For each selected lag r, we need the row of G
    # G[r, i] is the derivative of S_r w.r.t. h[i]
    # We want: G_r @ p + t <= 0 for all selected r (to not increase S_r)
    # And: 1 @ p = 0 (maintain sum)
    # Bounds: -radius <= p[i] <= radius, also 0-h[i] <= p[i] <= 1-h[i]
    
    A_ub = np.zeros((len(idx), n + 1))
    for row, r in enumerate(idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                A_ub[row, d:] += (1.0 - h[:m])
                A_ub[row, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                A_ub[row, :m] += (1.0 - h[g:])
                A_ub[row, g:] += -h[:m]
        A_ub[row, n] = -1.0
    A_ub *= (2.0 / n)
    b_ub = -scores_k
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lb = max(-radius, -h[j] + 1e-10)
        ub = min(radius, 1.0 - h[j] - 1e-10)
        if lb < ub:
            bounds.append((lb, ub))
        else:
            bounds.append((0.0, 0.0))
    bounds.append((0.0, None))  # t is free (unconstrained above, >= 0 not needed)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # maximize t
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))

            f2 = f1

        else:

            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
