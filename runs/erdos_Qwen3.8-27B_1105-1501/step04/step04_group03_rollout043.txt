def compute_A(h):
    n = len(h)
    m = 2 * n - 1
    A = np.zeros((n, m))
    for k in range(m):

        # First term: (1-h[i+d]) for max(0,-d) <= i < min(n, n-d)
        lo1 = max(0, -d)

        hi2 = min(n, n + d)

def line_search_pair(f, c, alpha_max):
    """
    Minimize g(alpha) = max_k(f[k] + alpha * c[k]) over alpha in [0, alpha_max].
    f: current scores (shape m)
    c: direction for scores (shape m), g(alpha) = max_k(f[k] + alpha*c[k])
    Returns (best_alpha, best_value)
    """
    if alpha_max <= 0:
        return 0.0, float(np.max(f))
    
    # Check if alpha=0 is already optimal
    k_star = np.argmax(f)
    if c[k_star] >= -1e-15:
        return 0.0, float(f[k_star])
    
    # Coarse grid
    n_grid = 20
    alphas = np.linspace(0, alpha_max, n_grid + 1)
    # g at each alpha: max_k(f_k + alpha*c_k)
    # Vectorized: (n_grid+1, m) matrix
    G = f[None, :] + alphas[:, None] * c[None, :]
    g_vals = G.max(axis=1)
    
    best_idx = np.argmin(g_vals)
    best_alpha = alphas[best_idx]
    best_val = g_vals[best_idx]
    
    # Refine with golden section if not at boundary
    if 0 < best_idx < n_grid:
        lo = alphas[best_idx - 1]
        hi = alphas[best_idx + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = np.max(f + c1 * c)
        f2 = np.max(f + c2 * c)
        for _ in range(25):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = np.max(f + c1 * c)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = np.max(f + c2 * c)
        best_alpha = (lo + hi) / 2.0
        best_val = np.max(f + best_alpha * c)
    
    if best_val > f[k_star]:
        return 0.0, float(f[k_star])
    
    return best_alpha, float(best_val)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def all_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        mask = (x < 1.0 - 1e-10) if residual > 0 else (x > 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        idx = int(np.argmax(1.0 - x)) if residual2 > 0 else int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_A(h):
    n = len(h)
    m = 2 * n - 1
    A = np.zeros((n, m))
    for k in range(m):
        d = k - (n - 1)
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            A[lo1:hi1, k] += (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if lo2 < hi2:
            A[lo2:hi2, k] -= h[lo2 - d:hi2 - d]
    return A

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = all_scores(h)
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    A = compute_A(h)
    grad_vec = A.T @ w  # shape (n,)
    return val, dx * grad_vec

def slsqp_smooth(h0, tau, maxiter, n):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v, _ = soft_value_grad(x, tau)
        return v
    def grad(x):
        _, g = soft_value_grad(x, tau)
        return g
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints={'type': 'eq', 'fun': cons_fun, 'jac': cons_jac},
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def line_search_pair(f, c, alpha_max):

        return 0.0, float(np.max(f))
    k_star = int(np.argmax(f))

    G = f[None, :] + alphas[:, None] * c[None, :]

    best_idx = int(np.argmin(g_vals))
    best_alpha = alphas[best_idx]
    best_val = g_vals[best_idx]

                f2 = float(np.max(f + c2 * c))
        best_alpha = (lo + hi) * 0.5
        best_val = float(np.max(f + best_alpha * c))
    if best_val >= float(f[k_star]):
        return 0.0, float(f[k_star])
    return best_alpha, best_val

def lp_step(h, radius):
    n = len(h)
    m = 2 * n - 1
    scores = all_scores(h)
    A = compute_A(h)
    # LP: find p minimizing t s.t. A.T @ p - t <= -scores, sum(p)=0, |p|<=radius
    # Variables: p[0..n-1], t (last)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A.T
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:

    h_new = np.interp(x_new, x_old, h_old)
