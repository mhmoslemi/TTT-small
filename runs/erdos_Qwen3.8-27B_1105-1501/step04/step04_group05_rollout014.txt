    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    # LSE objective
    z_max = z.max()
    z_shifted = z - z_max
    exp_z = np.exp(z_shifted / tau)
    p = exp_z / exp_z.sum()
    obj = tau * (z_max / tau + np.log(exp_z.sum()))  # = z_max + tau * log(sum(exp(z_shifted/tau)))
    
    grad = np.zeros(n)

        d = i - (n - 1)
        pi = p[i] * dx
        if d >= 0:

            if n - m > 0:

                grad[m:] += -pi * h[:n-m]
    
    return obj, grad
