def _slsqp_ineq_g(z, n, idx):
    h = z[:n]
    nc = 2 * n - 1
    j = np.arange(n)
    J = np.zeros((len(idx), n + 1))
    for ii, r in enumerate(idx):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        J[ii, :n] = -pd

    return J

def _slsqp_solve(h_init, n, maxiter=200, K=30):
    """SLSQP with only top-K correlation constraints."""
    h = np.asarray(h_init, dtype=float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:K]
    t0 = float(np.max(corr[top_idx])) + 1e-4
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_g, 'args': (n,)},
    ]
    for idx in top_idx:
        cons.append({
            'type': 'ineq',
            'fun': _slsqp_ineq_f,
            'jac': _slsqp_ineq_g,
            'args': (n, np.array([idx]))
        })
    
    try:
        res = minimize(_slsqp_epi_f, z0, jac=_slsqp_epi_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj_grad, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 40, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _pen_obj_grad(x, n, t_val, mu):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    excess = np.maximum(0.0, corr - t_val)
    f = mu * np.sum(excess * excess)
    j = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if excess[r] > 0:
            sl = r - (n - 1)
            pd = np.zeros(n)
            gs = j - sl
            m1 = (gs >= 0) & (gs < n)
            pd[m1] = 1.0 - h[gs[m1]]
            ps = j + sl
            m2 = (ps >= 0) & (ps < n)
            pd[m2] -= h[ps[m2]]
            grad += 2.0 * mu * excess[r] * pd
    grad -= np.mean(grad)
    return f, grad


def _penalty_step(h, n, t_val, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pen_obj_grad, h, args=(n, t_val, mu), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'maxls': 40, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h.copy(), _exact_c5(h)


def _penalty_loop(h, n, deadline, max_rounds=30):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    t_val = float(np.max(_correlations(h)))
    mu = 1.0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        h, c5_new = _penalty_step(h, n, t_val, mu, maxiter=60)
        new_t = float(np.max(_correlations(h)))
        if c5_new < cur_c5 - 1e-10:
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h.copy()
            t_val = new_t
            mu *= 0.9
        else:
            mu *= 1.2
            t_val = new_t
            if mu > 1e6:
                break
    return best_h, best_c5


def _slsqp_f(z, n):
    return float(z[n])


def _slsqp_g(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _slsqp_eq_f(z, n):
    return float(np.sum(z[:n]) - n * 0.5)


def _slsqp_eq_g(z, n):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def _slsqp_ineq_f(z, n, idx_arr):
    h = z[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return z[n] - float(np.min(corr[idx_arr]))


def _slsqp_ineq_g(z, n, idx_arr):
    h = z[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    min_val = float(np.min(corr[idx_arr]))
    min_pos = idx_arr[np.argmin(corr[idx_arr])]
    sl = min_pos - (n - 1)
    j = np.arange(n)
    pd = np.zeros(n)
    gs = j - sl
    m1 = (gs >= 0) & (gs < n)
    pd[m1] = 1.0 - h[gs[m1]]
    ps = j + sl
    m2 = (ps >= 0) & (ps < n)
    pd[m2] -= h[ps[m2]]
    g = np.zeros(n + 1)
    g[:n] = -pd
    g[n] = 1.0
    return g


def _slsqp_solve(h_init, n, maxiter, K):
    h = np.asarray(h_init, dtype=float).copy()
    corr = _correlations(h)
    top_idx = np.argsort(-corr)[:K]
    t0 = float(np.max(corr[top_idx])) + 1e-3
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_f, 'jac': _slsqp_eq_g, 'args': (n,)},
        {'type': 'ineq', 'fun': _slsqp_ineq_f, 'jac': _slsqp_ineq_g,
         'args': (n, top_idx)}
    ]
    try:
        res = minimize(_slsqp_f, z0, jac=_slsqp_g, args=(n,),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        x = _project(res.x[:n], n * 0.5)
        return x, _exact_c5(x)
    except Exception:
        return h, _exact_c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _refine_candidate(h_cand, nc, deadline, budget_frac):
    """Full refinement pipeline for one candidate."""
    h = _project(h_cand, nc * 0.5)
    cur_c5 = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    local_budget = budget_frac

    # LSE with increasing alpha
    alphas = [10.0, 40.0, 150.0, 400.0]
    for alpha in alphas:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + budget_frac * local_budget * 0.15)
        h, c5 = _lse_step(h, nc, alpha, maxiter=80)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Penalty method
    if time.time() < deadline:
        sub_dl = min(deadline, time.time() + budget_frac * local_budget * 0.20)
        h_p, c5_p = _penalty_loop(h, nc, sub_dl, max_rounds=20)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # SLSQP
    if time.time() < deadline:
        sub_dl = min(deadline, time.time() + budget_frac * local_budget * 0.15)
        for K_try in [20, 30, 40]:
            if time.time() > sub_dl:
                break
            h_s, c5_s = _slsqp_solve(best_h, nc, maxiter=200, K=K_try)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 * 0.5)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Binary search (25% budget)
    d1 = t0 + budget_s * 0.25
    n_list = sorted(set([n0, n0 - 8, n0 + 8, 48, 56, 72, 80, 96]))
    n_list = [n for n in n_list if 32 <= n <= 128]

    best_binary = {}

    for nc in n_list:
        phase_dl = min(d1, time.time() + budget_s * 0.03)
        for sa_i in range(4):
            if time.time() > phase_dl:
                break
            h_b, c5_b = _binary_sa(nc, 4.0, 0.05,
                                   max_iters=int(max(200, (phase_dl - time.time()) / max(budget_s * 0.008, 1e-6))),
                                   seed=seed + sa_i * 7919 + nc)
            if nc not in best_binary or c5_b < best_binary[nc][1]:
                best_binary[nc] = (h_b, c5_b)
        # Also try rounded initial
        h_r = _resample(h0, nc)
        h_br = (h_r >= 0.5).astype(float)
        h_br = _project(h_br, nc * 0.5)
        c5_br = _exact_c5(h_br)
        if nc not in best_binary or c5_br < best_binary[nc][1]:
            best_binary[nc] = (h_br, c5_br)

    # Phase 2: Refine best candidates (45% budget)
    d2 = t0 + budget_s * 0.70
    sorted_b = sorted(best_binary.items(), key=lambda x: x[1][1])[:5]
    candidates = [(n0, h0, best_c5)]
    for nc, (hb, cb) in sorted_b:
        candidates.append((nc, hb, cb))
    # Remove duplicates by n
    seen_n = set()
    unique_cands = []
    for nc, hc, cc in candidates:
        if nc not in seen_n:
            seen_n.add(nc)
            unique_cands.append((nc, hc, cc))
    candidates = unique_cands

    for nc, h_cand, c5_cand in candidates:
        if time.time() > d2:
            break
        if c5_cand > best_c5 + 0.02:
            continue
        remaining = max(0.0, d2 - time.time())
        per_cand = min(0.06, remaining / max(len(candidates), 1))
        h_ref, c5_ref = _refine_candidate(h_cand, nc, d2, per_cand)
        if c5_ref < best_c5:
            best_c5, best_h, best_n = c5_ref, h_ref.copy(), nc

    # Phase 3: Polishing (20% budget)
    d3 = t0 + budget_s * 0.90
    for attempt in range(8):
        if time.time() > d3:
            break
        if best_c5 <= 0.3800:
            break
        h_s, c5_s = _slsqp_solve(best_h, best_n, maxiter=300, K=35)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
        else:
            scale = 0.01 + 0.02 * (attempt % 3)
            h_p = _project(best_h + rng.normal(0, scale, best_n), best_n * 0.5)
            h_s, c5_s = _slsqp_solve(h_p, best_n, maxiter=200, K=30)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()

    # LSE final sharpening
    if time.time() < d3:
        for alpha in [500.0, 1000.0, 2000.0]:
            if time.time() > d3:
                break
            h_l, c5_l = _lse_step(best_h, best_n, alpha, maxiter=100)
            if c5_l < best_c5:
                best_c5, best_h = c5_l, h_l.copy()

    # Penalty final
    if time.time() < d3:
        h_p, c5_p = _penalty_loop(best_h, best_n,
                                  min(d3, time.time() + budget_s * 0.04),
                                  max_rounds=15)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 4: Try nearby resolutions (5% budget)
    d4 = t0 + budget_s * 0.97
    for dn in [2, -2, 4, -4, 6, -6]:
        if time.time() > d4:
            break
        nc = best_n + dn
        if nc < 32 or nc > 128:
            continue
        h_r = _resample(best_h, nc)
        h_rf, c5_rf = _refine_candidate(h_r, nc, d4, 0.02)
        if c5_rf < best_c5:
            best_c5, best_h, best_n = c5_rf, h_rf.copy(), nc

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
