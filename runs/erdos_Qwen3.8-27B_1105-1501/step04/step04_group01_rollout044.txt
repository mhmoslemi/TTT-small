def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius, n_active=60):
    n = len(h)
    na = min(n_active, 2 * n - 1)
    order = np.argsort(-scores)[:na]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = G[order, :].copy()
    A_ub = np.column_stack([A_ub, -np.ones(na)])
    b_ub = -scores[order]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_direction_full(h, scores, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([G, -np.ones(2 * n - 1)])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=-scores, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_polish(h, deadline, radius_init=0.12, n_active=60):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 40:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, n_active=n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            p = _lp_direction_full(h, scores, G, radius)
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 1
                radius *= 0.7
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.30)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.015:
                radius = 0.05
    return best_h, best_c5

def _simulated_annealing(h, deadline, rng, n_active=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.00005
    cool_rate = 0.995
    radius = 0.10
    stall = 0
    step_count = 0
    while time.time() < deadline:
        step_count += 1
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, n_active=n_active)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            if step_count % 3 == 0:
                idx = rng.integers(0, 2 * n)
                p = np.zeros(n)
                p[idx % n] = 1.0
                p[(idx + 1) % n] = -1.0
            else:
                p = rng.normal(0, 1, n)
                p -= p.mean()
                pn = np.linalg.norm(p)
                if pn < 1e-10:
                    T *= cool_rate
                    continue
                p = p / pn
        alpha, new_c5 = _line_search(h, p, dx, radius)
        delta = new_c5 - cur_c5
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / max(T, 1e-10))
        if accept:
            h_new = h + alpha * p
            h_new = _project(h_new, target)
            cur_c5 = _exact_c5(h_new)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_new.copy()
                stall = 0
                radius = min(radius * 1.05, 0.25)
            else:
                stall += 1
                radius *= 0.95
            h = h_new
        else:
            stall += 1
            radius *= 0.93
        if stall > 30:
            T *= 0.9
            stall = 0
        T = max(T * cool_rate, T_min)
        if T <= T_min * 1.01 and stall > 50:
            break
        if T <= T_min:
            h_l, c5_l = _lp_polish(h, min(deadline, time.time() + 2.0), radius_init=0.05, n_active=n_active)
            if c5_l < best_c5:
                best_c5 = c5_l
                best_h = h_l.copy()
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hopping(h, deadline, rng, n_active=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T_sa = 0.003
    T_min = 0.0001
    kick_scale = 0.06
    max_kicks = 30
    kicks = 0
    while time.time() < deadline and kicks < max_kicks:
        # Kick
        h_kick = h + rng.normal(0, kick_scale, n)
        h_kick = _project(h_kick, target)
        # Local optimize
        dl = min(deadline, time.time() + budget_s * 0.015)
        h_loc, c5_loc = _lp_polish(h_kick, dl, radius_init=0.08, n_active=n_active)
        delta = c5_loc - cur_c5
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / max(T_sa, 1e-10))
        if accept:
            h = h_loc
            cur_c5 = c5_loc
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            kicks = 0
        else:
            kicks += 1
        T_sa = max(T_sa * 0.97, T_min)
        kick_scale = max(0.02, kick_scale * 0.95)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=30):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _make_step_h(n, n_segments=4, rng=None):
    """Create a piecewise constant function with given number of segments."""
    if rng is None:
        rng = np.random.default_rng(0)
    n_seg = n_segments
    seg_len = n // n_seg
    h = np.zeros(n)
    for s in range(n_seg):
        start = s * seg_len
        end = start + seg_len if s < n_seg - 1 else n
        h[start:end] = rng.uniform(0, 1)
    return _project(h, n / 2.0)

def _make_plateau_h(n, n_plateaus=3, rng=None):
    """Create a function with distinct plateau regions."""
    if rng is None:
        rng = np.random.default_rng(0)
    h = np.zeros(n)
    seg_len = n // n_plateaus
    vals = np.linspace(0.2, 0.8, n_plateaus)
    rng.shuffle(vals)
    for s in range(n_plateaus):
        start = s * seg_len
        end = start + seg_len if s < n_plateaus - 1 else n
        h[start:end] = vals[s]
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick LP polish (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_polish(h, d1, radius_init=0.12, n_active=60)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-resolution exploration (18%)
    d2 = t0 + budget_s * 0.26
    base_n = len(best_h)
    n_offsets = [-8, 8, -12, 12, -16, 16, -4, 4, -20, 20, -2, 2, -24, 24, -6, 6]
    tried_ns = set()
    for dn in n_offsets:
        if time.time() > d2:
            break
        n_try = base_n + dn
        if n_try < 30 or n_try > 160:
            continue
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_polish(h_r, sub_dl, radius_init=0.10, n_active=50)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Simulated annealing (35%)
    d3 = t0 + budget_s * 0.61
    h3, c5_3 = _simulated_annealing(best_h, d3, rng, n_active=50)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Basin hopping (20%)
    d4 = t0 + budget_s * 0.81
    h4, c5_4 = _basin_hopping(best_h, d4, rng, n_active=50)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Pair refinement + final LP (10%)
    d5 = t0 + budget_s * 0.91
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=25)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    d6 = t0 + budget_s * 0.99
    h6, c5_6 = _lp_polish(best_h, d6, radius_init=0.04, n_active=40)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
