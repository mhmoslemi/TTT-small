def subgrad_step(h, n, step_size=0.05):
    corr = corr_full(h)
    r_max = int(np.argmax(corr))
    s = r_max - (n - 1)
    J = grad_corr_full(h)
    grad = J[r_max, :]
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return h.copy(), exact_c5(h)
    grad = grad / norm * step_size
    h_new = project_box_sum(h - grad, n * 0.5)
    c5_new = exact_c5(h_new)
    c5_cur = exact_c5(h)
    if c5_new < c5_cur:
        return h_new, c5_new

    return project_box_sum(h_new, n_new * 0.5)

def fourier_perturb(h, n, rng, scale=0.05):
    """Perturb h with random Fourier modes."""
    h_p = h.copy()
    n_modes = rng.integers(2, 6)
    for _ in range(n_modes):
        freq = rng.integers(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0, scale)
        mode = amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        h_p += mode
    return project_box_sum(h_p, n * 0.5)

def block_perturb(h, n, rng, scale=0.1):
    """Perturb h by shifting blocks."""
    h_p = h.copy()
    for _ in range(rng.integers(1, 4)):
        start = rng.integers(0, n - 4)
        length = rng.integers(2, min(8, n - start))
        shift = rng.uniform(-scale, scale)
        h_p[start:start+length] += shift
    return project_box_sum(h_p, n * 0.5)
