import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _softmax_vals(scores, tau):
    M = np.max(scores)
    shifted = np.clip((scores - M) / tau, -50, 50)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    F = M + tau * np.log(Z)
    w = exps / Z
    return F, w

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    F, w = _softmax_vals(scores, tau)
    grad = np.zeros(n)

        wr = w[r]

        m = n - d

            f1 = f2

    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:

    return best_a, best_c5

            iters_in_tau += 1

                A[ii, :m] = dx*(1.0-h[g:])
                A[ii, g:] = -dx*h[:m]
            b[ii] = tgt - sc[r]
        A[k_act] = 1.0
        b[k_act] = 0.0
        delta = np.linalg.lstsq(A, b, rcond=None)[0]
        am = 0.5
        pos = delta > 1e-14
        neg = delta < -1e-14
        if np.any(pos): am = min(am, np.min((1.0-h[pos])/delta[pos]))
        if np.any(neg): am = min(am, np.min(h[neg]/(-delta[neg])))
        am = max(am, 1e-12)
        improved = False
        for a in [am, 0.5*am, 0.25*am]:
            ht = _project(h + a*delta, target)
            ct = _exact_c5(ht)
            if ct < best_c5 - 1e-14:
                best_c5 = ct; best_h = ht.copy(); h = ht.copy(); improved = True
        if not improved: break
    return best_h, best_c5

def _pair_swap_polish(h, deadline, n_pairs=2000):
    n = len(h)
    h = h.copy()
    best_h = h.copy(); best_c5 = _exact_c5(h)
    rng = np.random.default_rng(0)
    counts = 0
    for _ in range(n_pairs):
        if time.time() > deadline: break
        j = rng.integers(n); k = rng.integers(n)
        if j == k: continue
        if h[j] < 1e-10 or h[j] > 1-1e-10: continue
        if h[k] < 1e-10 or h[k] > 1-1e-10: continue
        for sign in [1, -1]:
            if sign > 0:
                am = min(1.0-h[j], h[k])
            else:
                am = min(h[j], 1.0-h[k])
            if am < 1e-12: continue
            lo_a, hi_a = 0.0, am
            for _ in range(10):
                m1 = lo_a + (hi_a-lo_a)/3.0
                m2 = hi_a - (hi_a-lo_a)/3.0
                h1 = h.copy(); h1[j] += sign*m1; h1[k] -= sign*m1
                h2 = h.copy(); h2[j] += sign*m2; h2[k] -= sign*m2
                f1 = _exact_c5(h1); f2 = _exact_c5(h2)
                if f1 < f2: hi_a = m2
                else: lo_a = m1
            a = (lo_a+hi_a)/2.0
            ht = h.copy(); ht[j] += sign*a; ht[k] -= sign*a
            ht = np.clip(ht, 0, 1)
            ct = _exact_c5(ht)
            if ct < best_c5 - 1e-14:
                best_c5 = ct; best_h = ht.copy(); h = ht.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    xo = (np.arange(n)+0.5)/n
    xn = (np.arange(n_new)+0.5)/n_new
    hn = np.interp(xn, xo, h)
    return _project(hn, n_new/2.0)
