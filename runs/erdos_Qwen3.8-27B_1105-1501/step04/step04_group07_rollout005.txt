def _full_lag_sqp(h, deadline, rho_init=0.3, max_iters=500):
    """Full-lag SQP: minimize max of all lag correlations using linearized model."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rho = rho_init
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        G = _grad_matrix(h)
        scores = _scores(h)  # dx * corr values
        L = 2 * n - 1
        
        # Build LP:
        # Variables: delta (n vars) + tau (1 var) = n+1 vars
        # Minimize: tau
        # Subject to:
        #   scores[l] + G[l] @ delta <= tau  for all l
        #   sum(delta) = 0
        #   -min(h[j], rho) <= delta[j] <= min(1-h[j], rho)
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize tau
        
        # Inequality constraints: G[l] @ delta - tau <= -scores[l]
        # i.e., G[l] @ delta + (-1)*tau <= -scores[l]
        A_ub = np.zeros((L, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        
        # Bounds
        bounds = []
        for j in range(n):
            lo = max(-rho, -h[j])
            hi = min(rho, 1.0 - h[j])
            if lo > hi:
                lo = hi = 0.0
            bounds.append((lo, hi))
        bounds.append((-1.0, None))  # tau can be anything (but LP will keep it tight)
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                         bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                rho *= 0.7
                if rho < 1e-6:
                    break
                continue
            delta = res.x[:n]
        except Exception:
            stall += 1
            rho *= 0.7
            if rho < 1e-6:
                break
            continue
        
        if np.max(np.abs(delta)) < 1e-13:
            stall += 1
            rho *= 0.7
            if rho < 1e-7:
                break
            continue
        
        # Line search along delta
        alpha, new_c5 = _line_search(h, delta, cur_c5)
        
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * delta
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            rho = min(rho * 1.05, 0.8)  # expand slightly on success
        else:
            stall += 1
            rho *= 0.75
            if rho < 1e-7:
                break
    
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, 0:m] += (1.0 - h[d:d+m])
            G[r, d:n] += (-h[0:m])
        else:
            g = -d
            m = n - g
            G[r, g:n] += (1.0 - h[0:m])
            G[r, 0:m] += (-h[g:n])
    return G * dx

def _line_search(h, direction, cur_c5, max_alpha=1.0):
