def _build_G(h):
    """Build the gradient matrix: G[r, i] = partial score[r] / partial h[i]."""
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    d_rng = np.arange(-(n-1), n)
    for r, dd in enumerate(d_rng):
        for i in range(n):
            # score[r] = sum_k h[k] * (1-h[k+dd])
            # d(score[r])/d(h[i]) = (1-h[i+dd]) if i+dd valid
            #                     - h[i-dd] if i-dd valid
            s = 0.0
            if 0 <= i + dd < n:
                s += (1.0 - h[i + dd])
            if 0 <= i - dd < n:
                s -= h[i - dd]
            G[r, i] = s
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
