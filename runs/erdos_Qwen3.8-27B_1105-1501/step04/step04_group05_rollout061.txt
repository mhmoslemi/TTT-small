def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        else:
            hi = mid

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_periodic_mask(n, period, fill):
    """Create periodic binary mask with given period and fill ratio."""
    if n % period != 0:
        return None
    m = n // 2
    blocks = n // period
    on_per_block = int(round(fill * period))
    if on_per_block * blocks != m:
        return None
    mask = np.zeros(n)
    for b in range(blocks):
        for i in range(on_per_block):
            mask[b * period + i] = 1.0
    return mask
