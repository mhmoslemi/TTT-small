    # g_k with h[j] removed: A_k
    # g_k = A_k + c_k * h[j]
    # So c_k = (g_k(h) - A_k) / h[j] ... but we need A_k directly
    
    # Actually, let's compute c_k directly
    # For lag d (from -(n-1) to n-1):
    #   If j is a valid first index: contributes h[j]*(1-h[j+d])
    #   If j is a valid second index: contributes -h[j-d]*h[j]
    # So c_k = (1-h[j+d]) if j+d valid, plus -h[j-d] if j-d valid
    
    c = np.zeros(2*n-1)
    # j is first index for lag d if 0 <= j < n and 0 <= j+d < n, i.e., -j <= d < n-j
    # j is second index for lag d if 0 <= j-d < n and 0 <= j < n, i.e., j-n < d <= j
    
    # For lag d: c[d_offset] = (1-h[j+d])*[valid first] + (-h[j-d])*[valid second]
    for d in range(-(n-1), n):
        r = n - 1 + d
        contrib = 0.0
        if 0 <= j + d < n:  # j is valid first index
            contrib += (1.0 - h[j+d])
        if 0 <= j - d < n:  # j is valid second index  
            contrib += -h[j-d]
        c[r] = contrib
    
    # Now g_k = A_k + c_k * t
    # A_k = scores_k - c_k * h[j] (since current g_k = A_k + c_k * h[j])
    A = scores - c * h[j]
    
    # Minimize max_k (A_k + c_k * t) over t in [0, 1]
    # f(t) = max_k (A_k + c_k * t) is convex
    # Use ternary search or bisection on subgradient
    
    # The subgradient of f at t is the convex hull of {c_k : A_k + c_k*t = f(t)}
    # f is minimized where 0 is in the subgradient
    # i.e., where there exist active k with c_k > 0 and active k with c_k < 0
    
    # Simple approach: ternary search
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = float(np.max(A + c * m1))
        f2 = float(np.max(A + c * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2
    return t_opt, float(np.max(A + c * t_opt))

**New idea: Frank-Wolfe (conditional gradient) method**

For the problem min f(h) s.t. h ∈ C where C = {h ∈ [0,1]^n : sum(h) = n/2}:

1. Compute subgradient g = ∇f(h) (or a subgradient of the max)
2. Solve the linear oracle: s = argmin_{y ∈ C} g^T y
3. Take a step: h_new = (1-γ)h + γ*s
4. Update γ (line search or diminishing)

The linear oracle for our constraint set C:
minimize sum_j g_j * y_j s.t. 0 <= y_j <= 1, sum y_j = n/2

This is a simple assignment problem: sort g_j, set y_j = 1 for the n/2 smallest g_j, y_j = 0 for the rest (with possible fractional at the boundary).

This is very efficient! O(n log n) per iteration.

The Frank-Wolfe method has a 1/k convergence rate, which is slow, but it has the advantage of always staying in the feasible set (no projection needed).

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * (2.0 / n)

def _line_search(h, p, cur_c5, max_alpha=1.0):
    n = len(h)
    alpha_max = max_alpha
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    n_s = 16
    alphas = np.linspace(0, alpha_max, n_s)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = float(vals[best_i])
    best_a = float(alphas[best_i])
    if best_i == 0:
        return 0.0, cur_c5
    if best_i > 0 and best_i < n_s - 1:
        lo = float(alphas[best_i - 1])
        hi = float(alphas[best_i + 1])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _exact_c5(h + c1 * p)
        f2 = _exact_c5(h + c2 * p)
        for _ in range(20):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi = c2
                f2 = f1
                c2 = c1
                c1 = hi - gr * (hi - lo)
                f1 = _exact_c5(h + c1 * p)
            else:
                lo = c1
                f1 = f2
                c1 = c2
                c2 = lo + gr * (hi - lo)
                f2 = _exact_c5(h + c2 * p)
        oa = (lo + hi) / 2.0
        oc = _exact_c5(h + oa * p)
        if oc < best_c5:
            best_a, best_c5 = oa, oc
    return best_a, best_c5

def _pnorm_descent(h, deadline, p_start=8.0, p_end=4096.0, n_steps=1000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        p = p_start * (p_end / p_start) ** frac
        scores = _scores(h)
        s_max = float(scores.max())
        if s_max < 1e-15:
            break
        ratio = scores / s_max
        w = ratio ** p
        sw = float(w.sum())
        if sw < 1e-300:
            w = np.zeros(n_lags)
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        grad = G.T @ w
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 60:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.15, 0.10, 0.05, 0.02, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:

        else:

    stall = 0
    r = radius

            break

            break

        if top_k is not None and top_k < n_lags:

            sk = scores
            Gk = G
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0

            r *= 0.7

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_move(h, deadline, max_passes=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    rng = np.random.default_rng(77)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if stall > 5000:
            break
        improved = False
        n_trials = min(200, n * 2)
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            j = int(rng.integers(n))
            k = int(rng.integers(n))
            if j == k:
                continue
            if h[j] < 1e-10 or h[j] > 1.0 - 1e-10:
                continue
            if h[k] < 1e-10 or h[k] > 1.0 - 1e-10:
                continue
            for sign in [1, -1]:
                if sign > 0:
                    a_max = min(1.0 - h[j], h[k])
                else:
                    a_max = min(h[j], 1.0 - h[k])
                if a_max < 1e-10:
                    continue
                lo2, hi2 = 0.0, a_max
                for _ in range(12):
                    m1 = lo2 + (hi2 - lo2) / 3.0
                    m2 = hi2 - (hi2 - lo2) / 3.0
                    ht1 = h.copy()
                    ht1[j] += sign * m1
                    ht1[k] -= sign * m1
                    f1 = _exact_c5(ht1)
                    ht2 = h.copy()
                    ht2[j] += sign * m2
                    ht2[k] -= sign * m2
                    f2 = _exact_c5(ht2)
                    if f1 < f2:
                        hi2 = m2
                    else:
                        lo2 = m1
                a_opt = (lo2 + hi2) / 2.0
                if a_opt < 1e-12:
                    continue
                ht = h.copy()
                ht[j] += sign * a_opt
                ht[k] -= sign * a_opt
                ht = np.clip(ht, 0.0, 1.0)
                c5_try = _exact_c5(ht)
                if c5_try < cur_c5 - 1e-14:
                    h = ht
                    cur_c5 = c5_try
                    improved = True
                    if cur_c5 < best_c5 - 1e-14:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _active_equalize(h, deadline, n_steps=800):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    G = _grad_matrix(h)
    n_lags = 2 * n - 1
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        scores = _scores(h)
        s_max = float(scores.max())
        thresh = s_max * 0.97
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        weights = np.zeros(n_lags)
        active_scores = scores[active]
        target_s = float(np.mean(active_scores))
        for idx in active:
            if scores[idx] > target_s:
                weights[idx] = 1.0
        if weights.sum() < 1e-10:
            stall += 1
            if stall > 80:
                break
            continue
        weights /= weights.sum()
        grad = G.T @ weights
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            stall += 1
            if stall > 80:
                break
            continue
        d = -grad / gn
        found = False
        for scale in [0.12, 0.06, 0.03, 0.01]:
            alpha, new_c5 = _line_search(h, d * scale, cur_c5, max_alpha=1.0)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d * scale
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                G = _grad_matrix(h)
                stall = 0
                found = True
                break
        if not found:
            stall += 1
            if stall > 150:
                break
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: P-norm descent (12%)
    dl1 = t0 + budget_s * 0.12
    h, c5 = _pnorm_descent(h, dl1, p_start=6.0, p_end=1024.0, n_steps=800)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Frank-Wolfe (15%)
    dl2 = t0 + budget_s * 0.27
    h, c5 = _frank_wolfe(best_h, dl2, max_iters=3000, active_frac=0.04)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: Active equalization (13%)
    dl3 = t0 + budget_s * 0.40
    h, c5 = _active_equalize(best_h, dl3, n_steps=600)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: LP refinement (18%)
    dl4 = t0 + budget_s * 0.58
    h, c5 = _lp_refine(best_h, dl4, top_k=10, radius=0.08, max_iters=5000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: P-norm high p (10%)
    dl5 = t0 + budget_s * 0.68
    h, c5 = _pnorm_descent(best_h, dl5, p_start=64.0, p_end=8192.0, n_steps=600)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 6: Multi-start perturbation (15%)
    dl6 = t0 + budget_s * 0.83
    scale = 0.02
    for attempt in range(500):
        if time.time() > dl6:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.015)
        h_try, c5_try = _lp_refine(h_pert, sub_dl, top_k=8, radius=0.06, max_iters=200)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.06)

    # Phase 7: Resolution sweep (8%)
    dl7 = t0 + budget_s * 0.91
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, 16, -16, 12, -12]:
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 128:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + budget_s * 0.010)
        h_r, c5_r = _lp_refine(h_r, sub_dl, top_k=8, radius=0.1, max_iters=200)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Coordinate descent polish (5%)
    dl8 = t0 + budget_s * 0.96
    h, c5 = _coord_descent(best_h, dl8, max_passes=500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 9: Pair move polish (4%)
    if time.time() < t0 + budget_s * 0.99:
        hp, c5p = _pair_move(best_h, t0 + budget_s * 0.995, max_passes=200)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Final: one more LP pass on best
    if time.time() < t0 + budget_s * 0.997:
        hf, c5f = _lp_refine(best_h, t0 + budget_s * 0.997, top_k=4, radius=0.01, max_iters=2000)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
