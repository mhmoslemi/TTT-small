def pair_optimize(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = exact_c5(h)
    best_h = h.copy()
    
    # Precompute gradient matrix
    G = compute_grad_matrix(h)  # shape (2n-1, n)
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        scores = np.correlate(h, 1-h, mode='full')
        
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                # Compute g_k = G[:, i] - G[:, j]
                g = G[:, i] - G[:, j]
                
                # Find feasible range of delta
                # h_i + delta in [0,1], h_j - delta in [0,1]
                delta_lo = max(-h[i], -(1-h[j]))  # h[i]+delta >= 0, h[j]-delta <= 1
                delta_hi = min(1-h[i], h[j])      # h[i]+delta <= 1, h[j]-delta >= 0
                
                if delta_hi - delta_lo < 1e-14:
                    continue
                
                # Find top-K constraints
                top_k = np.argsort(scores)[::-1][:30]
                
                # Candidate deltas: 0, delta_lo, delta_hi, and intersections
                candidates = [0.0, delta_lo, delta_hi]
                for a in range(len(top_k)):
                    for b in range(a+1, len(top_k)):
                        ka, kb = top_k[a], top_k[b]
                        denom = g[ka] - g[kb]
                        if abs(denom) > 1e-14:
                            d = (scores[kb] - scores[ka]) / denom
                            if delta_lo < d < delta_hi:
                                candidates.append(d)
                
                # Evaluate at each candidate
                best_delta = 0
                best_val = best_c5
                for d in candidates:
                    new_scores = scores + d * g
                    val = np.max(new_scores) * dx
                    if val < best_val:
                        best_val = val
                        best_delta = d

                    h[i] += best_delta
                    h[j] -= best_delta
                    h = np.clip(h, 0, 1)
                    scores = np.correlate(h, 1-h, mode='full')
                    best_c5 = np.max(scores) * dx
                    best_h = h.copy()
                    G = compute_grad_matrix(h)
                    improved = True
    
    return best_h, best_c5

- Cross term at k = j-i exists if i is in the valid range for k = j-i.

For the implementation, I'll just check these conditions.

Now, for the piecewise-linear/quadratic minimax:
If all q_k = 0 (which is the case for k not in {0, i-j, j-i}), the problem is:
min_δ max_k (c_k(0) + δ * g_k) s.t. δ_lo ≤ δ ≤ δ_hi

This is a convex piecewise-linear minimization. The minimum occurs at a vertex where two (or more) lines intersect, or at an endpoint.

If some q_k ≠ 0, the problem is:
min_δ max_k (c_k(0) + δ * g_k + δ² * q_k)

This is no longer piecewise-linear. But since q_k ∈ {-2, 0, 1} and there are at most 3 such k, the function is still the max of a piecewise-linear function and a few quadratics. The minimum can be found by:
1. Finding all "critical" δ values where the identity of the maximum changes
2. Checking those critical values plus the endpoints

For the piecewise-linear part, critical values are intersections. For the quadratic part, critical values are where the quadratic equals the piecewise-linear maximum.

This is getting complex. For simplicity, I'll use a dense grid of δ values (say 200 points in [δ_lo, δ_hi]) and find the minimum on the grid, then refine with a few iterations of the exact piecewise-linear/quadratic minimization.

If the exact evaluation shows improvement, accept; otherwise, reject. This is similar to the LP approach (use a model for direction, verify exactly).

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-15:
        return 0.0, _exact_c5(h)
    n_grid = 60
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G

def _lp_direction(h, scores, G, radius, top_k=40):
    n = len(h)
    n_rows = 2 * n - 1
    k = min(top_k, n_rows)
    top_idx = np.argsort(scores)[::-1][:k]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G[top_idx, :]
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.15, top_k=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _grad_matrix(h)
        p = _lp_direction(h, scores, G, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.4)
        else:
            stall += 1
            radius *= 0.72
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _pair_sweep(h, deadline, top_k=25, max_passes=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    G = _grad_matrix(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        max_s = float(np.max(scores))
        improved = False
        top_idx = np.argsort(scores)[::-1][:top_k]
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            for j in range(i + 1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                    continue
                g = G[:, i] - G[:, j]
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                s_top = scores[top_idx]
                g_top = g[top_idx]
                candidates = np.array([0.0, delta_lo, delta_hi])
                denom = np.outer(g_top, -g_top) + np.outer(g_top, g_top)
                # Find pairwise intersections
                for a in range(len(top_idx)):
                    for b in range(a + 1, len(top_idx)):
                        d = g_top[b] - g_top[a]
                        if abs(d) > 1e-14:
                            delta = (s_top[a] - s_top[b]) / d
                            if delta_lo + 1e-13 < delta < delta_hi - 1e-13:
                                candidates = np.append(candidates, delta)
                best_delta = 0.0
                best_val = float('inf')
                for d in candidates:
                    new_scores = scores + d * g
                    val = float(np.max(new_scores)) * dx
                    if val < best_val:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0 and best_val < best_c5 - 1e-14:
                    h[i] += best_delta
                    h[j] -= best_delta
                    h = np.clip(h, 0.0, 1.0)
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    G = _grad_matrix(h)
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _softmax_direction(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = float(np.max(scores))
    exp_s = np.exp(tau * (scores - max_s))
    w = exp_s / float(np.sum(exp_s))
    G = _grad_matrix(h)
    grad = w @ G
    grad -= np.mean(grad)
    return grad

def _perturb_structured(h, rng, scale):
    n = len(h)
    h_new = h.copy()
    method = rng.integers(0, 4)
    if method == 0:
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif method == 1:
        block_size = int(rng.integers(n // 6, n // 3))
        i = int(rng.integers(0, n - block_size))
        h_new[i:i+block_size], h_new[0:block_size] = h_new[0:block_size].copy(), h_new[i:i+block_size].copy()
    elif method == 2:
        h_new = h[::-1].copy()
        h_new += rng.normal(0, scale * 0.3, n)
    else:
        h_new += rng.normal(0, scale, n)
    return _project(h_new, n / 2.0)

def _anneal(h, deadline, initial_temp=0.04):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    temp = initial_temp
    stall = 0
    total_iter = 0
    while time.time() < deadline:
        if stall > 50:
            break
        h_pert = _perturb_structured(best_h, rng, temp)
        t_sub = min(deadline, time.time() + 8.0)
        h_opt, c5_opt = _lp_loop(h_pert, t_sub, radius_init=0.08, top_k=30)
        cur_c5 = _exact_c5(h_opt)
        delta = cur_c5 - best_c5
        if delta < -1e-14:
            best_c5 = cur_c5
            best_h = h_opt.copy()
            temp = max(0.005, temp * 0.92)
            stall = 0
        else:
            accept_prob = np.exp(-max(0, delta) / max(temp * 0.01, 1e-6))
            if rng.random() < accept_prob:
                best_h = h_opt.copy()
                best_c5 = cur_c5
            stall += 1
            temp = min(0.15, temp * 1.06)
        total_iter += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _grad_descent(h, deadline, tau_list=(40, 100, 250, 500), lr_init=0.03):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    ti = 0
    while time.time() < deadline:
        if stall > 80:
            break
        tau = tau_list[ti % len(tau_list)]
        d = _softmax_direction(h, dx, tau)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            ti += 1
            continue
        d = d / dn
        mom = 0.85 * mom + 0.15 * d
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 1.5)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * lr * mom
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.04, 0.45)
        else:
            stall += 1
            lr *= 0.83
            if stall % 30 == 0:
                ti += 1
                mom = np.zeros(n)
    return best_h, best_c5
