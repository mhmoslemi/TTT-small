        j = i + d
        G[idx, i] += dx * (1.0 - h[j])
        G[idx, j] += -dx * h[i]
    
    # LP: minimize t s.t. F_r + G_r * delta <= t, sum(delta) = 0, |delta|_inf <= radius
    # Variables: [delta_0, ..., delta_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = F[active] - F[active]  # F_r + G_r*delta <= F_r means G_r*delta <= 0
    # Wait, we want F_r + G_r*(x-h) <= t where t is the new max
    # So: G_r*delta <= t - F_r
    # With t as variable: G_r*delta - t <= -F_r
    b_ub = -F[active]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []
    for i in range(n):
        lo_i = max(-radius, -h[i])  # h[i] + delta >= 0
        hi_i = min(radius, 1.0 - h[i])  # h[i] + delta <= 1
        bounds.append((lo_i, hi_i))
    bounds.append((None, None))  # t free
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success and res.x is not None:
            delta = res.x[:n]
            return h + delta
    except Exception:
        pass
    return None

def lp_loop(h, deadline, radius_start=0.08, max_iters=5000):
    n = len(h)
    best_h = h.copy()
    best_c = c5_exact(h)
    cur_c = best_c
    radius = radius_start
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        
        x_new = lp_local_step(h, radius)
        if x_new is None:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        x_new = project_box_sum(x_new, n / 2.0)
        
        # Line search
        best_step = 0.0
        best_tc = cur_c
        for factor in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            trial = project_box_sum(h + factor * (x_new - h), n / 2.0)
            tc = c5_exact(trial)
            if tc < best_tc - 1e-13:
                best_tc = tc
                best_step = factor
        
        if best_step > 0:
            h = project_box_sum(h + best_step * (x_new - h), n / 2.0)
            cur_c = c5_exact(h)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.2)
        else:
            stall += 1
            radius *= 0.7
    
    return best_h, best_c

def harmonic_global_search(n, freqs, deadline, rng):
    """Differential evolution on harmonic parameters."""
    M = len(freqs)
    P = 2 * M + 1  # bias + M cos amplitudes + M sin amplitudes
    
    bounds = [(-2.0, 2.0)] + [(-3.0, 3.0)] * (2 * M)
    
    def obj(params):
        h = harmonic_to_h(params, n, freqs)
        return c5_exact(h)
    
    # Create initial population with some structure
    pop0 = np.zeros((48, P))
    # First few: small amplitudes (smooth functions)
    for i in range(8):
        pop0[i] = rng.uniform(-0.5, 0.5, P)
    # Next: medium amplitudes
    for i in range(8, 24):
        pop0[i] = rng.uniform(-1.5, 1.5, P)
    # Next: large amplitudes (near-binary)
    for i in range(24, 40):
        pop0[i] = rng.uniform(-3.0, 3.0, P)
    # Last: random across all
    for i in range(40, 48):
        pop0[i] = rng.uniform(-3.0, 3.0, P)
    
    stop_count = [0]
    def callback(xk, convergence):
        if time.time() > deadline:
            stop_count[0] += 1
            return stop_count[0] > 3
        return False
    
    try:
        result = differential_evolution(obj, bounds, seed=int(rng.integers(100000)),
                                         maxiter=100, popsize=48, tol=1e-5,
                                         mutation=(0.5, 0.9), recombination=0.8,
                                         init='pop0' if hasattr(differential_evolution, 'pop0') else 'latinhypercube',
                                         polish=False, callback=callback,
                                         updating='deferred', workers=1)
        if result.x is not None:
            h = harmonic_to_h(result.x, n, freqs)
            return h, c5_exact(h)
    except Exception:
        pass
    
    return None, float('inf')
