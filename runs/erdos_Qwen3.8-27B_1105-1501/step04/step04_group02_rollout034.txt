def _pair_optimal_delta(scores, coef, delta_lo, delta_hi):
    """Find delta in [delta_lo, delta_hi] minimizing max_r (scores[r] + coef[r]*delta)."""
    m = len(scores)
    # Binary search on value level
    # At value V, feasible iff:
    #   For b>0: delta <= (V - a)/b  => upper bound
    #   For b<0: delta >= (V - a)/b  => lower bound (note: dividing by negative flips)
    #   For b==0: need V >= a
    
    b_pos = coef > 1e-14
    b_neg = coef < -1e-14
    b_zero = ~b_pos & ~b_neg
    
    if np.any(b_zero):
        v_zero = np.max(scores[b_zero])
    else:
        v_zero = -np.inf
    
    # Lower bound on delta from negative slopes: delta >= (V - a[r]) / b[r] for b[r] < 0
    # Note: (V - a[r]) / b[r] where b[r] < 0
    # So this is a lower bound: delta >= max of these
    
    # Upper bound from positive slopes: delta <= (V - a[r]) / b[r] for b[r] > 0
    
    def feasible(V):
        """Check if there exists delta in [delta_lo, delta_hi] with all scores+coef*delta <= V."""
        if V < v_zero - 1e-12:
            return False
        lo_b = delta_lo
        hi_b = delta_hi
        if np.any(b_neg):
            # delta >= (V - scores[r]) / coef[r] for r in b_neg
            vals = (V - scores[b_neg]) / coef[b_neg]
            lo_b = max(lo_b, np.max(vals))
        if np.any(b_pos):
            vals = (V - scores[b_pos]) / coef[b_pos]
            hi_b = min(hi_b, np.min(vals))
        return lo_b <= hi_b + 1e-12
    
    # Find min V such that feasible(V)
    V_lo = np.max(scores) + min(0, np.min(coef)) * delta_hi  # rough lower bound
    # Actually the minimum possible max is at most the current max
    V_hi = np.max(scores + coef * delta_lo)  # value at delta_lo
    V_hi = max(V_hi, np.max(scores + coef * delta_hi))
    V_lo = min(V_lo, V_hi)
    # Better lower bound: min over r of... actually just use the minimum possible
    # The max of (a + b*t) is at least max(a) when t=0, but can be lower
    V_lo = min(np.max(scores), np.max(scores + coef * delta_lo), np.max(scores + coef * delta_hi))
    # Actually a better lower bound: for each r, a[r] + b[r]*t >= min(a[r] + b[r]*delta_lo, a[r] + b[r]*delta_hi)
    # So max_r >= max_r min(...) which is a valid lower bound
    V_lo = np.max(np.minimum(scores + coef * delta_lo, scores + coef * delta_hi))
    
    for _ in range(60):
        V_mid = (V_lo + V_hi) / 2
        if feasible(V_mid):
            V_hi = V_mid
        else:
            V_lo = V_mid
    
    V_opt = V_hi
    # Now find the actual delta: given V_opt, find any feasible delta
    lo_b = delta_lo
    hi_b = delta_hi
    if np.any(b_neg):
        vals = (V_opt - scores[b_neg]) / coef[b_neg]
        lo_b = max(lo_b, np.max(vals))
    if np.any(b_pos):
        vals = (V_opt - scores[b_pos]) / coef[b_pos]
        hi_b = min(hi_b, np.min(vals))
    
    delta_opt = (lo_b + hi_b) / 2
    # Clamp
    delta_opt = np.clip(delta_opt, delta_lo, delta_hi)

def _grad_matrix(h):
    """Compute G[r, i] = d(c_raw[r])/dh[i] for all r, i. Shape (2n-1, n)."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))

        if d >= 0:
            m_overlap = n - d

            # d/dh[k] = 1-h[k+d]
            G[r, :m_overlap] += (1.0 - h[d:d+m_overlap])
            # d/dh[k+d] = -h[k]
            G[r, d:] -= h[:m_overlap]
        else:
            s = -d
            m_overlap = n - s
            # For k=s..n-1: c_raw[r] includes h[k]*(1-h[k-s])
            # Let k' = k-s, k'=0..n-1-s: h[k'+s]*(1-h[k'])
            # d/dh[k'+s] = 1-h[k']
            G[r, s:] += (1.0 - h[:m_overlap])
            # d/dh[k'] = -h[k'+s]
            G[r, :m_overlap] -= h[s:]
    return G

```python
def _solve_1d_minimax(a, b, lo, hi):
    """
    Find delta in [lo, hi] minimizing max_r (a[r] + b[r]*delta).
    Returns (delta_opt, min_value).
    """
    m = len(a)
    
    # Identify positive, negative, zero slope components
    b_pos_idx = np.where(b > 1e-14)[0]
    b_neg_idx = np.where(b < -1e-14)[0]
    b_zero_idx = np.where(np.abs(b) <= 1e-14)[0]
    
    # Lower bound on optimal value
    if len(b_zero_idx) > 0:
        v_zero = float(np.max(a[b_zero_idx]))
    else:
        v_zero = -np.inf
    
    # Binary search on V
    # Feasible(V): exists delta in [lo, hi] s.t. a[r] + b[r]*delta <= V for all r
    #   For b[r] > 0: delta <= (V - a[r]) / b[r]
    #   For b[r] < 0: delta >= (V - a[r]) / b[r] (note: dividing by negative flips inequality)
    #   For b[r] = 0: need V >= a[r]
    
    # Lower bound for V: the minimum possible max
    # At delta=lo: max(a + b*lo)
    # At delta=hi: max(a + b*hi)
    # The true minimum is <= min of these
    val_lo = float(np.max(a + b * lo))
    val_hi = float(np.max(a + b * hi))
    V_upper = min(val_lo, val_hi)  # we know the answer is <= this
    V_lower = max(v_zero, float(np.max(np.minimum(a + b * lo, a + b * hi))))
    
    # Binary search
    for _ in range(50):
        V_mid = (V_lower + V_upper) / 2.0
        # Check feasibility
        if V_mid < v_zero - 1e-12:
            V_lower = V_mid
            continue
        # Lower bound on delta from negative slopes
        delta_lb = lo
        if len(b_neg_idx) > 0:
            # delta >= (V_mid - a[r]) / b[r] for r in b_neg_idx
            # Note b[r] < 0, so (V_mid - a[r]) / b[r] is a lower bound
            vals = (V_mid - a[b_neg_idx]) / b[b_neg_idx]
            delta_lb = max(delta_lb, float(np.max(vals)))
        # Upper bound on delta from positive slopes
        delta_ub = hi
        if len(b_pos_idx) > 0:
            vals = (V_mid - a[b_pos_idx]) / b[b_pos_idx]
            delta_ub = min(delta_ub, float(np.min(vals)))
        
        if delta_lb <= delta_ub + 1e-12:
            V_upper = V_mid
        else:
            V_lower = V_mid
    
    V_opt = V_upper
    
    # Find a feasible delta
    delta_lb = lo
    delta_ub = hi
    if len(b_neg_idx) > 0:
        vals = (V_opt - a[b_neg_idx]) / b[b_neg_idx]
        delta_lb = max(delta_lb, float(np.max(vals)))
    if len(b_pos_idx) > 0:
        vals = (V_opt - a[b_pos_idx]) / b[b_pos_idx]
        delta_ub = min(delta_ub, float(np.min(vals)))
    
    if delta_lb > delta_ub:
        # Numerical issue, clamp
        delta_opt = (lo + hi) / 2.0
    else:
        delta_opt = (delta_lb + delta_ub) / 2.0
    
    return delta_opt, V_opt
```

        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            mo = n - d
            G[r, :mo] += (1.0 - h[d:d + mo])
            G[r, d:] -= h[:mo]
        else:
            s = -d
            mo = n - s
            G[r, s:] += (1.0 - h[:mo])
            G[r, :mo] -= h[s:]
    return G

def _solve_1d_minimax(a, b, lo, hi):
    m = len(a)
    b_pos = b > 1e-14
    b_neg = b < -1e-14
    b_zero = ~b_pos & ~b_neg
    if np.any(b_zero):
        v_zero = float(np.max(a[b_zero]))
    else:
        v_zero = -np.inf
    val_lo = float(np.max(a + b * lo))
    val_hi = float(np.max(a + b * hi))
    V_upper = min(val_lo, val_hi)
    V_lower = max(v_zero, float(np.max(np.minimum(a + b * lo, a + b * hi))))
    if V_upper - V_lower < 1e-14:
        mid = (lo + hi) / 2.0
        return mid, V_upper
    bp_idx = np.where(b_pos)[0]
    bn_idx = np.where(b_neg)[0]
    for _ in range(55):
        V_mid = (V_lower + V_upper) / 2.0
        if V_mid < v_zero - 1e-13:
            V_lower = V_mid
            continue
        dl = lo
        if len(bn_idx) > 0:
            vals = (V_mid - a[bn_idx]) / b[bn_idx]
            dl = max(dl, float(np.max(vals)))
        du = hi
        if len(bp_idx) > 0:
            vals = (V_mid - a[bp_idx]) / b[bp_idx]
            du = min(du, float(np.min(vals)))
        if dl <= du + 1e-13:
            V_upper = V_mid
        else:
            V_lower = V_mid
    V_opt = V_upper
    dl = lo
    du = hi
    if len(bn_idx) > 0:
        vals = (V_opt - a[bn_idx]) / b[bn_idx]
        dl = max(dl, float(np.max(vals)))
    if len(bp_idx) > 0:
        vals = (V_opt - a[bp_idx]) / b[bp_idx]
        du = min(du, float(np.min(vals)))
    if dl > du:
        delta_opt = (lo + hi) / 2.0
    else:
        delta_opt = (dl + du) / 2.0
    return delta_opt, V_opt

def _pair_descent(h, deadline, max_passes=200, seed=0, min_n_pairs=None):
    n = len(h)
    m = 2 * n - 1
    target = n / 2.0
    h = h.copy()
    scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = float(np.max(scores))
    best_h = h.copy()
    best_max = cur_max
    G = _grad_matrix(h)
    rng = np.random.default_rng(seed)
    stall = 0
    pass_num = 0
    recompute_counter = 0
    while time.time() < deadline and pass_num < max_passes:
        pass_num += 1
        if min_n_pairs:
            n_pairs = min(n_pairs, n * (n - 1) // 2)
        else:
            n_pairs = n * (n - 1) // 2
        idx_i = np.random.default_rng(pass_num * 31 + seed).permutation(n)
        pass_improved = 0
        for pi in range(min(n_pairs, len(idx_i) * (len(idx_i) - 1) // 2)):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            coef = G[:, i] - G[:, j]
            delta_lo = max(-h[i], h[j] - 1.0)
            delta_hi = min(1.0 - h[i], h[j])
            if delta_lo >= delta_hi - 1e-14:
                continue
            delta_opt, val_opt = _solve_1d_minimax(scores, coef, delta_lo, delta_hi)
            if val_opt < cur_max - 1e-11:
                h[i] += delta_opt
                h[j] -= delta_opt
                scores += coef * delta_opt
                actual_max = float(np.max(scores))
                if actual_max < cur_max - 1e-14:
                    cur_max = actual_max
                    pass_improved += 1
                    recompute_counter += 1
                    if recompute_counter >= 5:
                        G = _grad_matrix(h)
                        recompute_counter = 0
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                else:
                    h[i] -= delta_opt
                    h[j] += delta_opt
                    scores -= coef * delta_opt
                    G = _grad_matrix(h)
                    recompute_counter = 0
        if pass_improved == 0:
            stall += 1
            if stall >= 8:
                break
        else:
            stall = 0
    return best_h, best_max

def _lp_local(h, deadline, radius_init=0.12, top_k=None):
    n = len(h)
    m = 2 * n - 1
    target = n / 2.0
    h = h.copy()
    scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c5 = float(np.max(scores) * 2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    G = _grad_matrix(h)
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 60:
            break
        scores = np.correlate(h, 1.0 - h, mode='full').astype(float)
        cur_c5 = float(np.max(scores) * 2.0 / n)
        if top_k is not None and top_k < m:
            top_idx = np.argsort(scores)[::-1][:top_k]
        else:
            top_idx = np.arange(m)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(top_idx), n + 1))
        for xi, r in enumerate(top_idx):
            A_ub[xi, :n] = G[r]
            A_ub[xi, n] = -1.0
        b_ub = -scores[top_idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                radius *= 0.7
                if radius < 1e-7:
                    break
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            radius *= 0.7
            continue
        if float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        best_alpha = 0.0
        best_val = cur_c5
        n_grid = 20
        for t in np.linspace(0.0, alpha_max, n_grid):
            h_t = h + t * p
            if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
                v = float(np.max(np.correlate(h_t, 1.0 - h_t, mode='full')) * 2.0 / n)
                if v < best_val:
                    best_val = v
                    best_alpha = t
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t = best_alpha * 0.5
        hi_t = best_alpha * 1.5
        hi_t = min(hi_t, alpha_max)
        if hi_t <= lo_t:
            lo_t = 0.0
            hi_t = alpha_max
        f_c1 = best_val
        f_c2 = best_val
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        hc1 = h + c1 * p
        if hc1.min() >= -1e-12 and hc1.max() <= 1.0 + 1e-12:
            f_c1 = float(np.max(np.correlate(hc1, 1.0 - hc1, mode='full')) * 2.0 / n)
        hc2 = h + c2 * p
        if hc2.min() >= -1e-12 and hc2.max() <= 1.0 + 1e-12:
            f_c2 = float(np.max(np.correlate(hc2, 1.0 - hc2, mode='full')) * 2.0 / n)
        for _ in range(30):
            if hi_t - lo_t < 1e-12:
                break
            if f_c1 < f_c2:
                hi_t = c2
                c2, f_c2 = c1, f_c1
                c1 = hi_t - gr * (hi_t - lo_t)
                ht = h + c1 * p
                if ht.min() >= -1e-12 and ht.max() <= 1.0 + 1e-12:
                    f_c1 = float(np.max(np.correlate(ht, 1.0 - ht, mode='full')) * 2.0 / n)
                else:
                    f_c1 = best_val + 1.0
            else:
                lo_t = c1
                c1, f_c1 = c2, f_c2
                c2 = lo_t + gr * (hi_t - lo_t)
                ht = h + c2 * p
                if ht.min() >= -1e-12 and ht.max() <= 1.0 + 1e-12:
                    f_c2 = float(np.max(np.correlate(ht, 1.0 - ht, mode='full')) * 2.0 / n)
                else:
                    f_c2 = best_val + 1.0
        opt_t = (lo_t + hi_t) / 2.0
        ho = h + opt_t * p
        if ho.min() >= -1e-12 and ho.max() <= 1.0 + 1e-12:
            fo = float(np.max(np.correlate(ho, 1.0 - ho, mode='full')) * 2.0 / n)
            if fo < best_val:
                best_val = fo
                best_alpha = opt_t
        if best_val < cur_c5 - 1e-13:
            h = h + best_alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.2, 0.4)
            G = _grad_matrix(h)
        else:
            stall += 1
            radius *= 0.65
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    x_old = (np.arange(n) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb_optimize(h_best, deadline, budget_per, seed, scale_init=0.05):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = scale_init
    attempt = 0
    while time.time() < deadline and attempt < 500:
        attempt += 1
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _project(h_p, target)
        c5_p = _exact_c5(h_p)
        sub_dl = min(deadline, time.time() + budget_per)
        h_try, c5_raw = _pair_descent(h_p, sub_dl, max_passes=30, seed=seed + attempt)
        if c5_raw * 2.0 / n < best_c5 - 1e-12:
            best_h = h_try.copy()
            best_c5 = c5_raw * 2.0 / n
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.3, scale * 1.3)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safety_end = t0 + budget_s * 0.97

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 90
        rng0 = np.random.default_rng(seed)
        h0 = _project(rng0.uniform(0.2, 0.8, n0), n0 / 2.0)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Pair descent on parent (30% budget)
    d1 = t0 + budget_s * 0.30
    h_try, c5_raw = _pair_descent(best_h, d1, max_passes=500, seed=seed)
    c5_try = c5_raw * 2.0 / len(h_try)
    if c5_try < best_c5 - 1e-12:
        best_c5 = c5_try
        best_h = h_try.copy()
        best_n = len(h_try)

    # Phase 2: LP local search (15% budget)
    d2 = t0 + budget_s * 0.45
    h_try, c5_try = _lp_local(best_h, d2, radius_init=0.10, top_k=20)
    if c5_try < best_c5 - 1e-12:
        best_c5 = c5_try
        best_h = h_try.copy()
        best_n = len(h_try)

    # Phase 3: More pair descent (15% budget)
    d3 = t0 + budget_s * 0.60
    h_try, c5_raw = _pair_descent(best_h, d3, max_passes=500, seed=seed + 1000)
    c5_try = c5_raw * 2.0 / len(h_try)
    if c5_try < best_c5 - 1e-12:
        best_c5 = c5_try
        best_h = h_try.copy()
        best_n = len(h_try)

    # Phase 4: Multi-resolution (20% budget)
    d4 = t0 + budget_s * 0.80
    offsets = [-5, -4, -3, -2, -1, 1, 2, 3, 4, 5, -8, 8, -10, 10]
    time_per_res = max(2.0, (d4 - time.time()) / len(offsets) * 0.7)
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 160:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(d4, time.time() + time_per_res)
        h_r2, c5_raw2 = _pair_descent(h_r, sub_dl, max_passes=100, seed=seed + n_try)
        c5_r2 = c5_raw2 * 2.0 / n_try
        if c5_r2 < best_c5 - 1e-12:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = n_try

    # Phase 5: Perturbation + re-optimize (15% budget)
    d5 = t0 + budget_s * 0.95
    budget_per_pert = max(1.0, (d5 - time.time()) / 100)
    best_h, best_c5 = _perturb_optimize(best_h, d5, budget_per_pert, seed + 5000, scale_init=0.04)

    # Final: quick pair polish
    d6 = t0 + budget_s * 0.99
    if time.time() < d6:
        h_try, c5_raw = _pair_descent(best_h, d6, max_passes=100, seed=seed + 9999)
        c5_try = c5_raw * 2.0 / len(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)

    # Ensure feasibility
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
