    n = len(h)
    c = h[j] + h[k]
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    if t_hi - t_lo < 1e-12:
        return h, exact_c5(h)
    
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')

    # coeff[idx][l] for l in 0..2n-2, lag d = l-(n-1):
    #   (1-h[idx+d]) * [0<=idx+d<n] - h[idx-d] * [0<=idx-d<n]

        if 0 <= j+d < n:
            cj += 1.0 - h[j+d]
        if 0 <= j-d < n:
            cj -= h[j-d]
        ck = 0.0
        if 0 <= k+d < n:
            ck += 1.0 - h[k+d]
        if 0 <= k-d < n:

    delta_hi = t_hi - h[j]

    a, b = delta_lo, delta_hi
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)

def pair_descent_incremental(h, deadline, max_sweeps=500):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full')
    c5 = float(np.max(corr)) * dx
    
    # Precompute the "coefficient matrix" for each coordinate
    # coeff[i][l] = derivative of corr[l] w.r.t. h[i]
    # This is a (2n-1) x n matrix
    # coeff[i][n-1+d] = (1-h[i+d])*[valid] - h[i-d]*[valid]
    # But this changes as h changes! So we can't precompute it.
    # Instead, we compute g = coeff_j - coeff_k on the fly for each pair.
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j+1, n):
                c = h[j] + h[k]
                t_lo = max(0.0, c - 1.0)
                t_hi = min(1.0, c)
                if t_hi - t_lo < 1e-12:
                    continue
                
                # Compute g = coeff_j - coeff_k
                g = np.zeros(2*n-1)
                for d in range(-(n-1), n):
                    l = n-1+d
                    cj = 0.0
                    if 0 <= j+d < n:
                        cj += 1.0 - h[j+d]
                    if 0 <= j-d < n:
                        cj -= h[j-d]
                    ck = 0.0
                    if 0 <= k+d < n:
                        ck += 1.0 - h[k+d]
                    if 0 <= k-d < n:
                        ck -= h[k-d]
                    g[l] = cj - ck
                
                # Check if there's any potential for improvement
                # The current max is corr.max(). If we move in direction g,
                # the new max is max(corr + delta*g). For this to be < corr.max(),
                # we need some active components to decrease.
                active = corr >= np.max(corr) - 1e-10
                if not np.any(g[active] < -1e-10):
                    continue  # No improvement possible (all active components increase)
                
                # Golden section search
                delta_lo = t_lo - h[j]
                delta_hi = t_hi - h[j]
                if delta_lo == delta_hi:
                    continue
                
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = delta_lo, delta_hi
                c1 = b - gr*(b-a)
                c2 = a + gr*(b-a)
                f1 = np.max(corr + c1*g)
                f2 = np.max(corr + c2*g)
                for _ in range(30):
                    if b - a < 1e-13:
                        break
                    if f1 <= f2:
                        b = c2
                        c2 = c1
                        f2 = f1
                        c1 = b - gr*(b-a)
                        f1 = np.max(corr + c1*g)
                    else:
                        a = c1
                        c1 = c2
                        f1 = f2
                        c2 = a + gr*(b-a)
                        f2 = np.max(corr + c2*g)
                
                opt_d = (a+b) / 2.0
                new_max = np.max(corr + opt_d * g)
                if new_max < np.max(corr) - 1e-12:
                    h[j] += opt_d
                    h[k] -= opt_d
                    corr += opt_d * g
                    c5 = float(np.max(corr)) * dx
                    sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return h, c5, corr

    d = np.arange(-(n-1), n)  # lag values

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    residual2 = target - np.sum(x)
    if abs(residual2) > 1e-12:
        if residual2 > 0:
            idx = int(np.argmax(1.0 - x))
        else:
            idx = int(np.argmin(x))
        x[idx] += residual2
        x = np.clip(x, 0.0, 1.0)
    return x

def pair_descent(h, deadline, max_sweeps=500):

    h = h.copy()
    m = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)

    d_arr = np.arange(-(n-1), n)
    g = np.zeros(m)
    gr = 0.6180339887498949
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h, cur_max * dx, corr
            for k in range(j + 1, n):
                c_val = h[j] + h[k]
                t_lo = max(0.0, c_val - 1.0)
                t_hi = min(1.0, c_val)
                if t_hi - t_lo < 1e-12:
                    continue
                
                g.fill(0.0)
                jd = j + d_arr
                mk1 = (jd >= 0) & (jd < n)
                if np.any(mk1):
                    g[mk1] += 1.0 - h[jd[mk1]]
                jd2 = j - d_arr
                mk2 = (jd2 >= 0) & (jd2 < n)
                if np.any(mk2):
                    g[mk2] -= h[jd2[mk2]]
                kd = k + d_arr
                mk3 = (kd >= 0) & (kd < n)
                if np.any(mk3):
                    g[mk3] -= (1.0 - h[kd[mk3]])
                kd2 = k - d_arr
                mk4 = (kd2 >= 0) & (kd2 < n)
                if np.any(mk4):
                    g[mk4] += h[kd2[mk4]]
                
                am = corr >= cur_max - 1e-9
                if not np.any(g[am] < -1e-10):
                    continue
                
                delta_lo = t_lo - h[j]
                delta_hi = t_hi - h[j]
                a, b = delta_lo, delta_hi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                f1 = np.max(corr + c1 * g)
                f2 = np.max(corr + c2 * g)
                for _ in range(35):

                        f2 = f1

                        f1 = np.max(corr + c1 * g)

                        a = c1
                        f1 = f2
                        c1 = c2

                    h[k] -= opt_d
                    corr += opt_d * g

        else:

                break
    
    return h, cur_max * dx, corr

def build_corr_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g2 = -d
            m = n - g2
            if m > 0:
                G[r, :m] += (1.0 - h[g2:])
                G[r, g2:] += -h[:m]
    return G

def lp_loop(h, deadline, radius_init=0.1):

        else:

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)
