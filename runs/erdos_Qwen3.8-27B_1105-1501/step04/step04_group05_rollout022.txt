```python
def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _c5_full(h):
    """Return the full correlation array scaled by dx, and the max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx, float(np.max(corr) * dx)

def _two_level_counts(mask, n):
    """For a binary mask, compute A(k), B(k), C(k), D(k) for all offsets k.
    
    A(k) = #{i: mask[i]=1, mask[i+k]=1, both valid}
    B(k) = #{i: mask[i]=1, mask[i+k]=0, both valid}
    C(k) = #{i: mask[i]=0, mask[i+k]=1, both valid}
    D(k) = #{i: mask[i]=0, mask[i+k]=0, both valid}
    """
    # offset k goes from -(n-1) to n-1, stored at index k + (n-1)
    offsets = np.arange(-(n-1), n)
    A = np.zeros(2*n-1, dtype=float)
    B = np.zeros(2*n-1, dtype=float)
    C = np.zeros(2*n-1, dtype=float)
    D = np.zeros(2*n-1, dtype=float)
    
    for idx, k in enumerate(offsets):
        if k >= 0:
            i_range = range(0, n - k)
        else:
            i_range = range(-k, n)
        for i in i_range:
            j = i + k
            if mask[i] and mask[j]:
                A[idx] += 1
            elif mask[i] and not mask[j]:
                B[idx] += 1
            elif not mask[i] and mask[j]:
                C[idx] += 1
            else:
                D[idx] += 1
    return A, B, C, D
```

```python
def _optimal_d_for_counts(A, B, C, D, n, d_grid=None):
    """Find d that minimizes max_k [(A+D)*d*(1-d) + B*(1-d)^2 + C*d^2] * 2/n."""
    if d_grid is None:
        d_grid = np.linspace(0, 0.5, 200)
    # f_k(d) = (A+D)*d*(1-d) + B*(1-d)^2 + C*d^2 for each k
    # = (A+D)*d - (A+D)*d^2 + B*(1-2d+d^2) + C*d^2
    # = B + (A+D-2B)*d + (-A-D+B+C)*d^2
    a = A + D  # shape (2n-1,)
    b = B       # shape (2n-1,)
    c = C       # shape (2n-1,)
    # f_k(d) = b + (a - 2b)*d + (-a + b + c)*d^2
    # For all d in grid and all k:
    # f[k, j] = b[k] + (a[k] - 2*b[k]) * d_grid[j] + (-a[k] + b[k] + c[k]) * d_grid[j]^2
    d2 = d_grid ** 2
    d1 = d_grid
    f = b[:, None] + (a - 2*b)[:, None] * d1[None, :] + (-a + b + c)[:, None] * d2[None, :]
    # C5(d) = max_k f[k, j] * 2/n
    c5_at_d = f.max(axis=0) * (2.0 / n)
    best_idx = c5_at_d.argmin()
    return d_grid[best_idx], float(c5_at_d[best_idx])
```

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _mask_counts(mask, n):
    mf = mask.astype(float)
    oc = 1.0 - mf
    A = np.correlate(mf, mf, mode='full')
    B = np.correlate(mf, oc, mode='full')
    C = np.correlate(oc, mf, mode='full')
    D = np.correlate(oc, oc, mode='full')
    return A, B, C, D

def _two_level_c5_at_d(d, A, B, C, D, n):
    f = (A + D) * d * (1.0 - d) + B * (1.0 - d) * (1.0 - d) + C * d * d
    return float(np.max(f) * 2.0 / n)
