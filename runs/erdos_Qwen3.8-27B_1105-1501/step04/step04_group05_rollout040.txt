def _min_max_lines(slopes, intercepts, lo, hi):
    """Minimize max_i (slopes[i]*x + intercepts[i]) over x in [lo, hi]."""
    n = len(slopes)
    # Sort by slope
    order = np.argsort(slopes)
    s = slopes[order]
    b = intercepts[order]
    
    # Build upper envelope (convex hull)
    # For max of lines sorted by slope, the upper envelope is the "upper convex hull"
    # Use a stack-based approach
    hull = []  # list of (slope, intercept, x_start) where x_start is where this line becomes active
    
    for i in range(n):
        line = (s[i], b[i])
        while hull:
            # Check if the new line makes the last hull line obsolete
            # Intersection of last hull line and new line:
            # s_last * x + b_last = s_new * x + b_new
            # x = (b_new - b_last) / (s_last - s_new)
            if s[i] == hull[-1][0]:
                # Parallel: keep the one with higher intercept
                if b[i] > hull[-1][1]:
                    hull.pop()
                else:
                    continue  # new line is never active
                continue
            x_int = (b[i] - hull[-1][1]) / (hull[-1][0] - s[i])
            if x_int <= hull[-1][2]:
                # The new line takes over before the last hull line starts
                hull.pop()
            else:
                break
        if not hull:
            hull.append((s[i], b[i], -np.inf))
        else:
            x_int = (b[i] - hull[-1][1]) / (hull[-1][0] - s[i])
            hull.append((s[i], b[i], x_int))
    
    # Now find the minimum of the upper envelope over [lo, hi]
    # The envelope is convex (slope is non-decreasing)
    # Find where the slope changes from <= 0 to >= 0
    
    # Evaluate at boundaries
    def envelope_val(x):
        # Find the active line at x and evaluate
        # Binary search on hull
        lo_h, hi_h = 0, len(hull) - 1
        while lo_h < hi_h:

            if hull[mid][2] <= x:
                lo_h = mid
            else:
                hi_h = mid - 1
        return hull[lo_h][0] * x + hull[lo_h][1]
    
    best_val = min(envelope_val(lo), envelope_val(hi))
    best_x = lo if envelope_val(lo) <= envelope_val(hi) else hi
