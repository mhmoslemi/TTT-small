def _lse_gradient(h, corr, beta):
    n = len(h)
    lags = np.arange(2*n-1) - (n-1)
    
    # Compute softmax weights
    shifted = corr - np.max(corr)
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # Compute gradient: g[i] = sum_d w_d * [ (1-h[i+lag_d])*I(valid) - h[i-lag_d]*I(valid) ]
    g = np.zeros(n)
    for d in range(2*n-1):
        if w[d] < 1e-15:
            continue
        lag = lags[d]
        # Term 1: (1-h[i+lag]) for valid i+lag
        # i ranges from max(0, -lag) to min(n-1, n-1-lag)
        # i+lag ranges from max(0, -lag)+lag to min(n-1, n-1-lag)+lag
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            # (1-h[i+lag]) for i in [lo1, hi1)
            vals = 1.0 - h[lo1+lag : hi1+lag]
            g[lo1:hi1] += w[d] * vals
        
        # Term 2: -h[i-lag] for valid i-lag
        # i-lag ranges from 0 to n-1, so i ranges from lag to lag+n-1
        # i must be in [0, n-1], so i in [max(0, lag), min(n-1, lag+n-1)]
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            vals = h[lo2-lag : hi2-lag]
            g[lo2:hi2] += -w[d] * vals
    
    return g

     τ_min ≤ τ ≤ τ_max

2. Start from the top lag (max c_d)
3. Determine the slope
4. If slope > 0, we want to decrease τ (move left). The next constraint to hit is the one that catches up first.
5. If slope < 0, we want to increase τ (move right).

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_box_sum(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0).copy()
    for _ in range(200):
        s = float(np.sum(v))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _pair_slopes(h, i, j, n, lags):
    idx_ip = i + lags
    idx_im = i - lags
    idx_jp = j + lags
    idx_jm = j - lags
    v_ip = (idx_ip >= 0) & (idx_ip < n)
    v_im = (idx_im >= 0) & (idx_im < n)
    v_jp = (idx_jp >= 0) & (idx_jp < n)
    v_jm = (idx_jm >= 0) & (idx_jm < n)
    c_ip = np.clip(idx_ip, 0, n - 1)
    c_im = np.clip(idx_im, 0, n - 1)
    c_jp = np.clip(idx_jp, 0, n - 1)
    c_jm = np.clip(idx_jm, 0, n - 1)
    slope = np.zeros(2 * n - 1)
    slope += np.where(v_ip, 1.0 - h[c_ip], 0.0)
    slope -= np.where(v_im, h[c_im], 0.0)
    slope -= np.where(v_jp, 1.0 - h[c_jp], 0.0)
    slope += np.where(v_jm, h[c_jm], 0.0)
    return slope

def _pair_optimal(h, i, j, corr, n, lags):
    tau_min = max(-h[i], h[j] - 1.0)
    tau_max = min(1.0 - h[i], h[j])
    if tau_max - tau_min < 1e-14:
        return 0.0, corr, float(np.max(corr))
    slope = _pair_slopes(h, i, j, n, lags)
    a, b = tau_min, tau_max
    cur_max = float(np.max(corr))
    for _ in range(50):
        m1 = a + (b - a) * 0.333333
        m2 = a + (b - a) * 0.666667
        f1 = float(np.max(corr + m1 * slope))
        f2 = float(np.max(corr + m2 * slope))
        if f1 < f2:
            b = m2
        else:
            a = m1
    tau_mid = 0.5 * (a + b)
    candidates = [tau_min, tau_max, tau_mid]
    best_tau = tau_min
    best_val = np.inf
    for t in candidates:
        val = float(np.max(corr + t * slope))
        if val < best_val:
            best_val = val
            best_tau = t
    if best_val < cur_max - 1e-13:
        new_corr = corr + best_tau * slope
        return best_tau, new_corr, best_val
    return 0.0, corr, cur_max

def _two_coord_loop(h, deadline, rng, max_iters=800000, batch_size=20):
    n = len(h)
    dx = 2.0 / n
    lags = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx
    no_improve = 0
    n_accepted = 0
    recompute_interval = 500

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 8000:
            break
        if n_accepted > 0 and n_accepted % recompute_interval == 0:
            corr = np.correlate(h, 1.0 - h, mode='full').copy()
            cur_max = float(np.max(corr))

        tried = 0
        accepted = False
        while tried < batch_size:
            if time.time() > deadline:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            tried += 1
            if i == j:
                continue
            if h[i] <= 1e-10 or h[i] >= 1.0 - 1e-10:
                continue
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            tau, new_corr, new_max = _pair_optimal(h, i, j, corr, n, lags)
            if abs(tau) > 1e-13 and new_max < cur_max - 1e-13:
                h[i] = max(0.0, min(1.0, h[i] + tau))
                h[j] = max(0.0, min(1.0, h[j] - tau))
                corr = new_corr
                cur_max = new_max
                n_accepted += 1
                no_improve = 0
                if cur_max * dx < best_c5:
                    best_c5 = cur_max * dx
                    best_h = h.copy()
                accepted = True
                break
        if not accepted:
            no_improve += 1

    return best_h, best_c5

def _lse_grad(h, corr, beta):
    n = len(h)
    lags = np.arange(2 * n - 1) - (n - 1)
    shifted = corr - float(np.max(corr))
    exp_v = np.exp(beta * shifted)
    w = exp_v / (float(np.sum(exp_v)) + 1e-300)
    g = np.zeros(n)
    for d in range(2 * n - 1):
        if w[d] < 1e-12:
            continue
        lag = lags[d]
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            g[lo1:hi1] += w[d] * (1.0 - h[lo1 + lag:hi1 + lag])
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            g[lo2:hi2] += -w[d] * h[lo2 - lag:hi2 - lag]
    return g

def _sgd_loop(h, deadline, beta_start=10.0, beta_end=200.0, n_beta=10):

    dx = 2.0 / n
    best_h = h.copy()

    cur_c5 = best_c5
    beta_list = np.geomspace(beta_start, beta_end, n_beta)

            break

            g = _lse_grad(h, corr, beta)
            g -= np.mean(g)
            gn = float(np.max(np.abs(g)))
            if gn < 1e-14:

            d = -g / gn * step

                stall = 0

    return best_h, best_c5

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        low = max(0, -d)
        high = min(n, n - d)
        if low >= high:
            continue
        i = np.arange(low, high)
        j = i + d
        G[r, j] += dx * (1.0 - h[i])
        G[r, i] += -dx * h[j]
    return G

def _lp_trust_step(h, radius):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    G = _build_G(h)
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - F
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000, 'presolve': True})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_start=0.08, max_iters=500):

    best_h = h.copy()
    best_c5 = _exact_c5(h)

            break
        if stall > 30:
            break
        x_new = _lp_trust_step(h, radius)

            stall += 1
            radius *= 0.7

                break
            continue
        x_new = _project_box_sum(x_new, n / 2.0)
        best_step = 0.0
        best_tc5 = cur_c5
        for factor in factors:
            trial = _project_box_sum(h + factor * (x_new - h), n / 2.0)
            tc5 = _exact_c5(trial)
            if tc5 < best_tc5 - 1e-13:

                best_step = factor

            h = _project_box_sum(h + best_step * (x_new - h), n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.2)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5
