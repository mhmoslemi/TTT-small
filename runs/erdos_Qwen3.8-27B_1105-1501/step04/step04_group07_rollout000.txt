def _sqp_step(h, rho, deadline):
    """Full-lag trust region step."""
    n = len(h)
    scores = _scores(h)
    G = _grad_matrix(h)
    L = 2 * n - 1
    
    # LP: minimize tau
    # s.t. G @ delta - tau <= -scores  (L constraints)
    #      sum(delta) = 0
    #      max(-h[i], -rho) <= delta[i] <= min(1-h[i], rho)
    #      -2 <= tau <= 2
    
    # Variables: [delta_0, ..., delta_{n-1}, tau]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: G @ delta - tau <= -scores
    # In standard form: A_ub @ x <= b_ub
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds
    lb = np.maximum(-h, -rho)
    ub = np.minimum(1.0 - h, rho)
    bounds = list(zip(lb, ub))
    bounds.append((-2.0, 2.0))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-13:
        return None
    
    return delta

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    lo = float(v.min()) - 2.0

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d

                G[r, d:] += (1.0 - h[:m])

            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]

def _sqp_full_lag(h, rho, top_k=None):
    """Full-lag (or top-k) trust region LP step."""
    n = len(h)
    scores = _scores(h)
    G = _grad_matrix(h)
    L = 2 * n - 1
    
    if top_k is not None and top_k < L:
        idx = np.argsort(scores)[-top_k:]
        s_k = scores[idx]
        G_k = G[idx]
    else:
        s_k = scores
        G_k = G

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -s_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    lb = np.maximum(-h, -rho)
    ub = np.minimum(1.0 - h, rho)
    bounds = list(zip(lb, ub)) + [(-2.0, 2.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')

        return None
    if not res.success:
        return None
    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-13:
        return None
    return delta
