import numpy as np

def _project(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    x = np.clip(v + lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0 - 1e-12:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 1e-12:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        js = j - s
        m1 = (js >= 0) & (js < n)
        G[k, m1] += (1.0 - h[js[m1]])
        js2 = j + s
        m2 = (js2 >= 0) & (js2 < n)
        G[k, m2] -= h[js2[m2]]
    return G


def _active_slsqp(h, active_idx, n, maxiter=300, ftol=1e-13):
    """Minimize t s.t. corr_r(h) <= t for r in active_idx, 0<=h<=1, sum(h)=n/2."""
    corr = _corr(h)
    t0 = float(np.max(corr[active_idx])) + 0.05
    z0 = np.concatenate([h, [t0]])

    def obj(z):
        return z[n]

    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_f(z):
        hz = z[:n]
        tz = z[n]
        c = _corr(hz)
        return tz - c[active_idx]

    def ineq_jac(z):
        hz = z[:n]
        G = _grad_matrix(hz, n)
        K = len(active_idx)
        J = np.zeros((K, n + 1))
        J[:, :n] = -G[active_idx, :]
        J[:, n] = 1.0
        return J

    def eq_f(z):
        return np.sum(z[:n]) - n * 0.5

    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, n * 2.0)]
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': ineq_f, 'jac': ineq_jac},
                                    {'type': 'eq', 'fun': eq_f, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        h_new = _project(res.x[:n], n * 0.5)
        return h_new
    except Exception:
        return h.copy()


def _active_set_loop(h, n, deadline, K_init=8, K_max=40, margin_frac=0.05):
    """Iteratively grow active set and solve SLSQP."""
    h = h.copy()
    cur_c5 = _c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    corr = _corr(h)
    max_corr = float(np.max(corr))
    
    for it in range(500):
        if time.time() > deadline:
            break
        # Select active set: top K plus those within margin
        margin = max(1e-8, margin_frac * max_corr)
        active_scores = corr >= (max_corr - margin)
        top_k = np.argsort(-corr)[:K_max]
        active = np.where(active_scores)[0]
        if len(active) > K_max:
            active = np.argsort(-corr[active])[:K_max]
            active = np.sort(active)
        else:
            active = np.sort(active)
        # Ensure at least K_init

            active = np.sort(top_k[:K_init])
        
        h_new = _active_slsqp(h, active, n, maxiter=200, ftol=1e-13)
        new_c5 = _c5(h_new)
        
        if new_c5 < cur_c5 - 1e-10:
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            corr = _corr(h)
            max_corr = float(np.max(corr))
        else:

def _lp_step(h, n, radius, scores):
    """Find LP direction to reduce max correlation."""
    G = _grad_matrix(h, n)
    nc = 2 * n - 1
    A_ub = np.zeros((nc, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    bounds = []
    for i in range(n):
        lo_b = max(-radius, -h[i])
        hi_b = min(radius, 1.0 - h[i])
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search(h, p, c5_cur, n, alpha_max_hint=None):
    """Find best step size along direction p."""
    if alpha_max_hint is None:
        alpha_max = 0.5
        for i in range(n):
            if p[i] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
            elif p[i] < -1e-14:
                alpha_max = min(alpha_max, h[i] / (-p[i]))
        alpha_max = max(alpha_max, 1e-8)
    else:
        alpha_max = alpha_max_hint
    
    best_a, best_c5 = 0.0, c5_cur
    # Coarse grid
    for t in np.linspace(0.0, 1.0, 12):
        a = t * alpha_max
        if a < 1e-10:
            continue
        h_t = _project(h + a * p, n * 0.5)
        c5_t = _c5(h_t)
        if c5_t < best_c5:
            best_c5, best_a = c5_t, a
    # Golden section refinement
    if best_a > 0:
        lo = max(0.0, best_a - alpha_max * 0.15)
        hi = best_a + alpha_max * 0.15
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = _c5(_project(h + c1 * p, n * 0.5))
        f2 = _c5(_project(h + c2 * p, n * 0.5))
        for _ in range(12):
            if hi - lo < 1e-12:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = _c5(_project(h + c1 * p, n * 0.5))
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = _c5(_project(h + c2 * p, n * 0.5))
        a = (lo + hi) / 2.0
        c5_t = _c5(_project(h + a * p, n * 0.5))
        if c5_t < best_c5:
            best_a, best_c5 = a, c5_t
    return best_a, best_c5


def _lp_loop(h, n, deadline, radius_init=0.08, max_iters=3000, stall_limit=60):

                break
            continue
        alpha, new_c5 = _line_search(h, p, cur_c5, n)
        if new_c5 < cur_c5 - 1e-12:
            h = _project(h + alpha * p, n * 0.5)
            cur_c5 = _c5(h)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.25)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5


def _lse_obj_grad(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = _corr(h)
    ms = float(np.max(corr))
    vals = np.clip(alpha * (corr - ms), -500.0, 500.0)
    exp_v = np.exp(vals)
    s = float(np.sum(exp_v))
    f = ms + (1.0 / alpha) * np.log(s)
    w = exp_v / s
    G = _grad_matrix(h, n)
    grad = G.T @ w
    grad -= np.mean(grad)
    return f, grad


def _lse_step(h, n, alpha, maxiter=150):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj_grad, h, args=(n, alpha), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 40, 'iprint': 0})
        x = _project(res.x, n * 0.5)
        return x, _c5(x)
    except Exception:
        return h.copy(), _c5(h)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _subgrad_descent(h, n, deadline, step0=0.05, decay=0.999):
    """Simple subgradient descent with projection."""
    h = h.copy()
    cur_c5 = _c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    step = step0
    stall = 0
    while time.time() < deadline:
        if stall > 200:
            break
        corr = _corr(h)
        r_max = int(np.argmax(corr))
        G = _grad_matrix(h, n)
        g = G[r_max, :]
        g -= np.mean(g)
        norm = np.linalg.norm(g)
        if norm < 1e-15:
            break
        g = g / norm * step
        h_new = _project(h - g, n * 0.5)
        c5_new = _c5(h_new)
        if c5_new < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()
            step = min(step * 1.05, 0.3)
            stall = 0
        else:
            step *= decay
            stall += 1
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.asarray(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 * 0.5
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _c5(h)
    best_n = n0

    # Phase 1: LSE warm-up (5% budget)
    d1 = t0 + budget_s * 0.05
    for alpha in [15.0, 30.0, 60.0]:
        if time.time() > d1:
            break
        h, c5 = _lse_step(h, n0, alpha, maxiter=80)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Active-set SLSQP (25% budget)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _active_set_loop(best_h, n0, d2, K_init=6, K_max=25, margin_frac=0.08)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: LP trust region polish (20% budget)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _lp_loop(best_h, n0, d3, radius_init=0.06, max_iters=5000, stall_limit=80)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Subgradient + LP interleave (10% budget)
    d4 = t0 + budget_s * 0.60
    h_sub, c5_sub = _subgrad_descent(best_h, n0, min(d4, time.time() + budget_s * 0.03), step0=0.03)
    if c5_sub < best_c5:
        best_c5, best_h = c5_sub, h_sub.copy()
    h4, c5_4 = _lp_loop(best_h, n0, d4, radius_init=0.04, max_iters=3000, stall_limit=60)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Resolution sweep (20% budget)
    d5 = t0 + budget_s * 0.80
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, -8, 12]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 32 or n_try > 160:
            continue
        if n_try % 2 != 0:
            continue
        if n_try == best_n and dn != 0:
            continue
        h_r = _resample(best_h, n_try)
        # Quick LSE
        sub_d = min(d5, time.time() + budget_s * 0.012)
        for alpha in [20.0, 40.0]:
            if time.time() > sub_d:
                break
            h_r, c5_r = _lse_step(h_r, n_try, alpha, maxiter=60)
        # Active set SLSQP
        sub_d2 = min(d5, time.time() + budget_s * 0.018)
        h_r, c5_r = _active_set_loop(h_r, n_try, sub_d2, K_init=6, K_max=20, margin_frac=0.1)
        # LP polish
        sub_d3 = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_loop(h_r, n_try, sub_d3, radius_init=0.05, max_iters=2000, stall_limit=50)
        if c5_r < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Phase 6: Random restarts (10% budget)
    d6 = t0 + budget_s * 0.90
    scale = 0.04
    for attempt in range(300):
        if time.time() > d6:
            break
        h_p = best_h + rng.normal(0, scale, best_n)
        h_p = _project(h_p, best_n * 0.5)
        # Quick LSE
        h_p, c5_p = _lse_step(h_p, best_n, 30.0, maxiter=50)
        # LP
        sub_d = min(d6, time.time() + budget_s * 0.008)
        h_p, c5_p = _lp_loop(h_p, best_n, sub_d, radius_init=0.04, max_iters=1000, stall_limit=40)
        if c5_p < best_c5 - 1e-10:
            best_c5, best_h = c5_p, h_p.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.12, scale * 1.1)

    # Phase 7: Final polish (10% budget)
    d7 = t0 + budget_s * 0.97
    h7, c5_7 = _active_set_loop(best_h, best_n, min(d7, time.time() + budget_s * 0.03),
                                 K_init=5, K_max=15, margin_frac=0.03)
    if c5_7 < best_c5:
        best_c5, best_h = c5_7, h7.copy()
    h7b, c5_7b = _lp_loop(best_h, best_n, d7, radius_init=0.02, max_iters=5000, stall_limit=100)
    if c5_7b < best_c5:
        best_c5, best_h = c5_7b, h7b.copy()

    # Final validation
    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _c5(best_h)
    return best_h, final_c5, best_n
