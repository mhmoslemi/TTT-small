def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    x = np.clip(v - mu, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(h, r):
    """Gradient of raw score r w.r.t. h (without dx factor)."""
    n = len(h)
    g = np.zeros(n)
    d = r - (n - 1)
    if d >= 0:
        m = n - d
        g[d:] += 1.0 - h[:m]
        g[:m] -= h[d:]
    else:
        s = -d
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    return g

def _lp_dir(h, scores, active_idx, radius):
    """LP direction to decrease selected scores."""
    n = len(h)
    m = len(active_idx)
    # Build constraint matrix for active rows only
    A = np.zeros((m, n))
    for i, r in enumerate(active_idx):
        A[i] = _row_grad(h, r)
    # Variables: p (n) + t (1), minimize t
    # Constraint: A @ p - t <= -(scores[active] - scores_max + delta)
    # Actually: A @ p <= scores[active] - delta (we want to decrease by delta)
    # Better: minimize -t subject to A@p >= t (t is the max decrease)
    # Or: minimize t subject to A@p + t >= 0 (t >= -A@p for all active)
    # Let's do: minimize t s.t. A@p - t <= 0 (t >= A@p for all active rows)
    # Wait, I want to minimize the maximum predicted score.
    # Predicted score r = scores[r] + A[r] @ p
    # Want: scores[r] + A[r] @ p <= t for all active r
    # Minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)  # scores[r] + A[r]@p - t <= 0 => A[r]@p - t <= -scores[r]
    # Wait: scores[r] + A[r]@p <= t => A[r]@p - t <= -scores[r]
    b_ub = -scores[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0  # sum(p) = 0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max):

    f_opt = _f(opt_t)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    # Area-preserving resampling
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    x_old = (np.arange(n) + 0.5) * dx_old
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sim_anneal(h, deadline, T_init=0.05, T_min=0.001, alpha=0.995):

    target = n / 2.0
    h = h.copy()

    best_h = h.copy()
    best_c = cur
    T = T_init
    
    while time.time() < deadline and T > T_min:

        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j]) if np.random.random() < 0.5 else min(1.0 - h[i], h[j])
        if max_delta < 1e-8:
            T *= alpha
            continue
        delta = np.random.uniform(0.1, 1.0) * max_delta
        h[i] += delta
        h[j] -= delta
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= delta
            h[j] += delta
            T *= alpha
            continue
        h = np.clip(h, 0, 1)
        h = _project(h, target)
        c_new = _exact_c5(h)
        delta_c = c_new - cur
        if delta_c < 0 or np.random.random() < np.exp(-delta_c / T):

                best_h = h.copy()
        else:
            h[i] -= delta  # undo (approximate)
            h[j] += delta
            h = np.clip(h, 0, 1)
            h = _project(h, target)
        T *= alpha
    return best_h, best_c

            best_c = c_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Gradient + LP polish (10%)
    d6 = t0 + budget_s * 0.90
    h, c = _grad_refine(best_h, min(d6, time.time() + budget_s * 0.04), tau=50, lr=0.005, beta=0.92)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    if time.time() < d6:
        h, c = _lp_refine(best_h, d6, top_ks=[2, 3], radii=[0.001, 0.003])
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 7: Perturbation escape (5%)
    d7 = t0 + budget_s * 0.95
    h, c = _perturb_loop(best_h, d7, budget_s, seed=seed + 7)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 8: Final pair polish (5%)
    d8 = t0 + budget_s * 0.99
    h, c = _full_pass_pair(best_h, d8, min_frac=0.001)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
