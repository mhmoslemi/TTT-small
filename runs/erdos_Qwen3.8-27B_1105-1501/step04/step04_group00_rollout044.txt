def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    return G * dx

def _lp_direction(h, scores, G, radius, active_rows):
    n = len(h)
    scores_k = scores[active_rows]
    G_k = G[active_rows]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active_rows), n + 1))
    A_ub[:, :n] = G_k
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _line_search(h, p, n_samp=20, refine=22):
    n = len(h)
    alpha_max = 0.5
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    alphas = np.linspace(0, alpha_max, n_samp)
    vals = np.array([_exact_c5(h + a * p) for a in alphas])
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_samp - 1):
        return best_a, best_c5
    lo = alphas[max(0, best_i - 1)]
    hi = alphas[min(n_samp - 1, best_i + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(h + c1 * p)
    f2 = _exact_c5(h + c2 * p)
    for _ in range(refine):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(n, deadline, rng, n_iters=50000):

def _fourier_evolve(h, deadline, n_pop=30, K=12, n_gen=300, mutation_scale=0.10, rng=None):
    n = len(h)
    x = np.arange(n) / n
    B = np.zeros((n, 2 * K))
    for k in range(1, K + 1):
        B[:, 2*(k-1)] = np.cos(2 * np.pi * k * x)
        B[:, 2*(k-1)+1] = np.sin(2 * np.pi * k * x)
    B -= B.mean(axis=0)
    target = n / 2.0
    a_init = np.linalg.lstsq(B, h - 0.5, rcond=None)[0]
    pop_a = np.tile(a_init, (n_pop, 1)) + mutation_scale * rng.standard_normal((n_pop, 2*K))
    pop_c5 = np.zeros(n_pop)
    for i in range(n_pop):
        h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
        h_try = _project(h_try, target)
        pop_c5[i] = _exact_c5(h_try)
    best_i = int(np.argmin(pop_c5))
    best_c5 = pop_c5[best_i]
    best_h = np.clip(0.5 + B @ pop_a[best_i], 0, 1)
    best_h = _project(best_h, target)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        order = np.argsort(pop_c5)
        n_keep = max(2, n_pop // 2)
        pop_a[:n_keep] = pop_a[order[:n_keep]]
        pop_c5[:n_keep] = pop_c5[order[:n_keep]]
        for i in range(n_keep, n_pop):
            p1 = order[rng.integers(0, n_keep)]
            p2 = order[rng.integers(0, n_keep)]
            mix = rng.random()
            pop_a[i] = mix * pop_a[order[p1]] + (1.0 - mix) * pop_a[order[p2]]
            pop_a[i] += mutation_scale * rng.standard_normal(2*K)
            h_try = np.clip(0.5 + B @ pop_a[i], 0, 1)
            h_try = _project(h_try, target)
            pop_c5[i] = _exact_c5(h_try)
        mutation_scale *= 0.995
        cur_best = int(np.argmin(pop_c5))
        if pop_c5[cur_best] < best_c5:
            best_c5 = pop_c5[cur_best]
            best_h = np.clip(0.5 + B @ pop_a[cur_best], 0, 1)
            best_h = _project(best_h, target)
    return best_h, best_c5

def _balancing_step(h, deadline, n_iters=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()

    for it in range(n_iters):
        if time.time() > deadline:
            break

        max_s = scores.max()
        # Find constraints that are close to max
        thresh = max_s - 0.005
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        # Compute gradient only from active constraints
        G = _grad_matrix(h)
        # Weight active constraints inversely to how close they are to max
        gaps = max_s - scores[active]
        w = 1.0 / (gaps + 0.001)
        w /= w.sum()
        grad = G[active].T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.7
            if alpha < 1e-11:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0.copy(), n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: Intensive local optimization from parent at n=128
    dl1 = t0 + budget_s * 0.10
    h1, c5_1 = _adaptive_local(h_start, dl1, radius_init=0.2, max_iters=5000,
                               min_k=3, max_k=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Weighted gradient refinement
    if elapsed() < budget_s * 0.15:
        dl2 = t0 + budget_s * 0.18
        h2, c5_2 = _weighted_grad_opt(best_h, dl2, n_steps=400, gamma_start=15.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Balancing step
    if elapsed() < budget_s * 0.20:
        dl3 = t0 + budget_s * 0.22
        h3, c5_3 = _balancing_step(best_h, dl3, n_iters=200)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        # Quick LP polish after balancing
        dl3b = t0 + budget_s * 0.24
        h3b, c5_3b = _adaptive_local(best_h, dl3b, radius_init=0.08, max_iters=2000,
                                     min_k=3, max_k=15)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration
    dl4 = t0 + budget_s * 0.55
    resolutions = [96, 112, 144, 160, 192, 256, 128, 144, 192]
    alphas = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113, 0.5,
              0.6180339887, 0.2541010752, 0.5, 0.6180339887]

    for ri, n_try in enumerate(resolutions):
        if elapsed() > dl4 - budget_s * 0.02:
            break
        if best_c5 < 0.3805:
            break
        # Two starting points per resolution
        for s in range(2):
            if elapsed() > dl4 - budget_s * 0.01:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas[ri % len(alphas)]
                h_init = _quasi_periodic_init(n_try, alpha)
                h_init = h_init + rng.normal(0, 0.06, n_try)
                h_init = _project(h_init, n_try / 2.0)

            sub_dl = min(dl4, elapsed() + budget_s * 0.04)
            h_try, c5_try = _adaptive_local(h_init, t0 + sub_dl,
                                             radius_init=0.18, max_iters=3000,
                                             min_k=3, max_k=20)
            # Follow up with weighted gradient
            if elapsed() < sub_dl:
                sub_dl2 = min(t0 + sub_dl, elapsed() + budget_s * 0.015)
                h_wg, c5_wg = _weighted_grad_opt(h_try, t0 + sub_dl2, n_steps=150)
                if c5_wg < c5_try:
                    c5_try = c5_wg
                    h_try = h_wg.copy()
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try

    # Phase 5: Fourier evolution at best and at 128
    if elapsed() < budget_s * 0.62:
        dl5 = t0 + budget_s * 0.70
        h_f, c5_f = _fourier_evolve(best_h, dl5, n_pop=30, K=12,
                                     n_gen=400, mutation_scale=0.08, rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # LP polish on Fourier result
        if elapsed() < dl5:
            sub_dl = min(dl5, elapsed() + budget_s * 0.03)
            h_fp, c5_fp = _adaptive_local(best_h, t0 + sub_dl,
                                           radius_init=0.1, max_iters=2000,
                                           min_k=3, max_k=15)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 6: Binary SA at a few resolutions
    if elapsed() < budget_s * 0.72:
        dl6 = t0 + budget_s * 0.80
        for n_bin in [96, 128, 160]:
            if elapsed() > dl6:
                break
            sub_dl = min(dl6, elapsed() + budget_s * 0.04)
            h_b, c5_b = _binary_sa(n_bin, t0 + sub_dl, rng, n_iters=60000)
            # Refine binary result with LP
            if elapsed() < dl6:
                sub_dl2 = min(dl6, elapsed() + budget_s * 0.025)
                h_br, c5_br = _adaptive_local(h_b, t0 + sub_dl2,
                                               radius_init=0.12, max_iters=2000,
                                               min_k=3, max_k=12)
                if c5_br < c5_b:
                    c5_b = c5_br
                    h_b = h_br.copy()
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin

    # Phase 7: Perturbation restarts from best
    dl7 = t0 + budget_s * 0.90
    scale = 0.025
    for attempt in range(200):
        if elapsed() > dl7:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl7, elapsed() + budget_s * 0.012)
        h_try, c5_try = _adaptive_local(h_pert, t0 + sub_dl,
                                         radius_init=0.06, max_iters=300,
                                         min_k=3, max_k=10)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.82)
        else:
            scale = min(0.12, scale * 1.10)

    # Phase 8: Pair polish
    if elapsed() < budget_s * 0.94:
        dl8 = t0 + budget_s * 0.97
        hp, c5p = _pair_polish(best_h, dl8, max_passes=6)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 9: Final micro-optimization
    if elapsed() < budget_s * 0.985:
        dl9 = t0 + budget_s * 0.995
        h_fin, c5_fin = _adaptive_local(best_h, dl9,
                                         radius_init=0.008, max_iters=1000,
                                         min_k=2, max_k=8)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
