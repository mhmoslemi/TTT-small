import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def soft_value_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    scores_raw = np.correlate(h, omh, mode='full')
    scores = scores_raw * dx
    max_s = float(np.max(scores))
    exp_s = np.exp((scores - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            grad[js1:je1 + 1] += wk * (1.0 - h[js1 + d:je1 + 1 + d])
        js2 = max(0, d)
        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wk * h[js2 - d:je2 + 1 - d]
    return val, dx * grad

def nesterov_projected_grad(h0, tau, deadline, lr_init=0.01, momentum=0.9):
    n = len(h0)
    target = n / 2.0
    x = project(h0.copy(), target)
    v = np.zeros(n)
    best_x = x.copy()
    best_val = exact_c5(x)
    lr = lr_init
    stall = 0
    mu = momentum
    
    while time.time() < deadline:
        # Nesterov step
        x_prev = x.copy()
        x_nc = x + mu * v
        x_nc = project(x_nc, target)
        val, grad = soft_value_grad(x_nc, tau)
        # Gradient step
        x_new = x_nc - lr * grad
        x_new = project(x_new, target)
        # Update momentum
        v = x_new - x
        x = x_new
        cur_c5 = exact_c5(x)
        if cur_c5 < best_val:
            best_val = cur_c5
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                lr *= 0.8
                mu = min(mu + 0.01, 0.99)
            if stall > 200:
                break
    return best_x, best_val

        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= wk * h[js2 - d:je2 + 1 - d]
    return val, dx * grad

def coord_step(h, j):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    num_shifts = 2 * n - 1
    a = np.zeros(num_shifts)
    for k in range(num_shifts):
        d = k - (n - 1)
        i1 = (0 <= j - d < n)
        i2 = (0 <= j + d < n)
        hd = h[j - d] if i1 else 0.0
        hjd = h[j + d] if i2 else 0.0
        a[k] = (1.0 - hd) * (1.0 if i1 else 0.0) - hjd * (1.0 if i2 else 0.0)
    b = scores - a * h[j]
    # Ternary search for min of max(a*t + b) on [0,1] (convex PL function)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-12:
            break
    return (lo + hi) / 2.0

def coord_descent(h, deadline, max_sweeps=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        max_change = 0.0
        for j in range(n):
            t_opt = coord_step(h, j)
            delta = t_opt - h[j]
            if abs(delta) > 1e-11:
                h[j] = t_opt
                h = project(h, target)
                max_change = max(max_change, abs(delta))
        c5 = exact_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        if max_change < 1e-10:
            break
    return best_h, best_c5

def pair_swap_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 2):
            if time.time() > deadline:
                break
            k = (j + 1) % n
            if k == j:
                continue
            h_cur = best_h[j]
            h_k = best_h[k]
            # Try transferring from j to k and vice versa
            for sign in [1.0, -1.0]:
                amax = min(h_cur, 1.0 - h_k) if sign > 0 else min(1.0 - h_cur, h_k)
                if amax < 1e-12:
                    continue
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    a = amax * frac
                    h_t = best_h.copy()
                    h_t[j] += sign * a
                    h_t[k] -= sign * a
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def subgradient_method(h0, deadline, lr_init=0.05, max_iter=5000):
    n = len(h0)
    target = n / 2.0
    x = project(h0.copy(), target)
    best_x = x.copy()
    best_c5 = exact_c5(x)
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        scores = np.correlate(x, 1.0 - x, mode='full')
        k_star = int(np.argmax(scores))
        d = k_star - (n - 1)
        grad = np.zeros(n)
        js1 = max(0, -d)
        je1 = min(n - 1, n - 1 - d)
        if js1 <= je1:
            grad[js1:je1 + 1] += (1.0 - x[js1 + d:je1 + 1 + d])
        js2 = max(0, d)
        je2 = min(n - 1, n + d - 1)
        if js2 <= je2:
            grad[js2:je2 + 1] -= x[js2 - d:je2 + 1 - d]
        grad *= (2.0 / n)
        x_new = x - lr * grad
        x_new = project(x_new, target)
        c5_new = exact_c5(x_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_x = x_new.copy()
            x = x_new
            stall = 0
        else:
            x = x_new
            stall += 1
            if stall % 50 == 0:
                lr *= 0.85
        if lr < 1e-8:
            break
    return best_x, best_c5

def slsqp_smooth(h0, tau, maxiter):
    n = len(h0)
    target = n / 2.0
    def obj_func(x):
        v, _ = soft_value_grad(x, tau)
        return v
    def grad_func(x):
        _, g = soft_value_grad(x, tau)
        return g
    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h0, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Exact coordinate descent (main workhorse)
    d1 = t0 + budget_s * 0.25
    h_cd, c5_cd = coord_descent(h, d1, max_sweeps=300)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 2: SLSQP with adaptive tau
    d2 = t0 + budget_s * 0.42
    tau_list = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
    iter_list = [80, 80, 70, 60, 50, 40, 30]
    h_s = best_h.copy()
    for i in range(len(tau_list)):
        if time.time() > d2:
            break
        h_s = slsqp_smooth(h_s, tau_list[i], iter_list[i])
        h_s = project(h_s, best_n / 2.0)
        c5_s = exact_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Coordinate descent after SLSQP
    d3 = t0 + budget_s * 0.52
    h_cd2, c5_cd2 = coord_descent(best_h, d3, max_sweeps=200)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    # Phase 4: Pair swap polish
    d4 = t0 + budget_s * 0.58
    h_ps, c5_ps = pair_swap_polish(best_h, d4, max_passes=60)
    if c5_ps < best_c5:
        best_c5 = c5_ps
        best_h = h_ps.copy()

    # Phase 5: Diversified restarts with multiple methods
    d5 = t0 + budget_s * 0.78
    scales = [0.02, 0.04, 0.06, 0.10, 0.15, 0.20, 0.30]
    for ns in scales:
        if time.time() > d5:
            break
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        # Try coordinate descent
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_try, c5_try = coord_descent(h_pert, sub_dl, max_sweeps=80)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Try subgradient
        if time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.02)
            h_sg, c5_sg = subgradient_method(h_pert, sub_dl2, lr_init=0.03, max_iter=2000)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
        # Try SLSQP from perturbed
        if time.time() < d5:
            h_sl = slsqp_smooth(h_pert, 0.1, 50)
            h_sl = project(h_sl, best_n / 2.0)
            c5_sl = exact_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

    # Phase 6: Multi-resolution search
    d6 = t0 + budget_s * 0.88
    offsets = [0, -6, 6, -12, 12, -16, 16, -24, 24, -32, 32, -40, 40, -48, 48]
    tried_n = set([best_n])
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 160:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        # Coordinate descent
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = coord_descent(h_r, sub_dl, max_sweeps=100)
        # Quick SLSQP
        if time.time() < d6:
            h_r = slsqp_smooth(h_r, 0.1, 40)
            h_r = project(h_r, n_try / 2.0)
            c5_r = exact_c5(h_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final coordinate descent + pair polish
    d7 = hard_deadline
    h_fin, c5_fin = coord_descent(best_h, d7, max_sweeps=200)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()
    if time.time() < hard_deadline:
        h_pp, c5_pp = pair_swap_polish(best_h, hard_deadline, max_passes=30)
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
