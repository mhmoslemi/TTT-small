import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_single_lag(h, d, n):
    """Gradient of g_d(h) = sum_i h[i]*(1-h[i+d])*dx with respect to h"""
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif d > 0:
        # g_d = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d]) * dx
        # d/dh[i] = (1-h[i+d])*dx for i < n-d (from h[i] term)
        #           -h[i-d]*dx for i >= d (from -h[i]*h[i-d] term, wait...)
        # Actually: g_d = sum_{i=0}^{n-1-d} h[i] - h[i]*h[i+d] * dx
        # d/dh[j]: 
        #   if j < n-d: contributes (1 - h[j+d]) * dx (from h[j] term and -h[j]*h[j+d] term)
        #   if j >= d: contributes -h[j-d] * dx (from -h[j-d]*h[j] term in the sum)
        m = n - d
        g[:m] += (1.0 - h[d:]) * dx
        g[d:] -= h[:m] * dx
    else:
        d_pos = -d
        m = n - d_pos
        g[d_pos:] += (1.0 - h[:m]) * dx
        g[:m] -= h[d_pos:] * dx
    return g

def soft_subgradient(h, tau):
    """Compute soft-max weighted subgradient"""
    n = len(h)
    dx = 2.0 / n
    scores = compute_scores(h)
    max_s = np.max(scores)
    exp_s = np.exp((scores - max_s) / tau)
    Z = np.sum(exp_s)
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    
    # Compute weighted gradient
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        g += wk * grad_single_lag(h, d, n)
    return val, g

def active_lp_step(h, scores, K, radius):
    """LP step using only top K constraints"""
    n = len(h)
    # Get top K lags
    top_idx = np.argsort(scores)[-K:][::-1]  # descending
    
    # Build G matrix for only these K lags
    G = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        d = k - (n - 1)
        G[i, :] = grad_single_lag(h, d, n)
    
    # LP: minimize t s.t. G*x <= t - scores[top_idx], sum(x)=0, bounds
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[top_idx]
    A_eq = np.ones((1, n + 1))
    b_eq = 0.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def coord_descent(h, deadline, max_sweeps=50):
    """Coordinate descent: try adjusting one element at a time"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            # Try setting h[j] to various values, adjusting one other to maintain constraint
            if h[j] <= 0 or h[j] >= 1:
                continue
            # Find a partner to adjust
            # Try small perturbations
            for delta in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
                new_j = h[j] + delta
                if new_j < -1e-10 or new_j > 1.0 + 1e-10:
                    continue
                # Adjust one other element to maintain the sum
                # Pick the element that's closest to 0.5 (most flexible)
                candidates = np.where((h > 0.1) & (h < 0.9))[0]
                candidates = candidates[candidates != j]
                if len(candidates) == 0:
                    continue
                # Pick the one that gives best improvement
                for k_idx in candidates[:5]:  # try a few
                    new_k = h[k_idx] - delta
                    if new_k < -1e-10 or new_k > 1.0 + 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[j] = new_j
                    h_try[k_idx] = new_k
                    c5_try = exact_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    
    return best_h, best_c5
