def project(v, target=None):

    if target >= n:

def build_active_grads(h, active_idx):

    m = len(active_idx)
    G = np.zeros((m, n))
    for k in range(m):

def roughness_hessian(n, rho):
    if rho <= 0:
        return np.zeros((n, n))
    H = np.zeros((n, n))
    for i in range(n):
        if i == 0:
            H[i, i] = rho
        elif i == n - 1:
            H[i, i] = rho
        else:
            H[i, i] = 2.0 * rho
        if i > 0:
            H[i, i-1] = -rho
            H[i-1, i] = -rho
    return H

def solve_direction(h, scores, n, active_idx, priority_power, margin_factor,
                    margin_floor, ridge, rough, trust):
    m = len(active_idx)
    if m == 0:
        return None
    G = build_active_grads(h, active_idx)
    s_act = scores[active_idx]
    c_max = float(np.max(s_act))
    c_min = float(np.min(s_act))
    spread = c_max - c_min
    margin = max(margin_floor, margin_factor * (spread + margin_floor))
    target = c_min - margin
    residual = target - s_act
    gaps = c_max - s_act
    max_gap = float(np.max(gaps))
    if max_gap < 1e-15:
        q = np.ones(m)
    else:
        q = 1.0 + priority_power * (gaps / max_gap) ** 2
    q = np.maximum(q, 1e-6)
    W = np.diag(q)
    H = G.T @ (q[:, None] * G)
    H += ridge * np.eye(n)
    if rough > 0:
        H += roughness_hessian(n, rough)
    rhs_vec = G.T @ (q * residual)
    KKT = np.zeros((n + 1, n + 1))
    KKT[:n, :n] = H
    KKT[:n, n] = 1.0
    KKT[n, :n] = 1.0
    rhs = np.zeros(n + 1)
    rhs[:n] = rhs_vec
    try:
        sol = np.linalg.solve(KKT, rhs)
        delta = sol[:n]
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(KKT, rhs, rcond=1e-14)[0]
            delta = sol[:n]
        except Exception:
            delta = -G.T @ q
            delta -= delta.mean()
    delta -= delta.mean()
    max_abs = float(np.max(np.abs(delta)))
    if max_abs < 1e-12:
        return None
    p = delta / max_abs
    return p

def line_search(h, p, trust, cur_c5):
    n = len(h)
    p = p - p.mean()
    max_abs = float(np.max(np.abs(p)))
    if max_abs < 1e-14:
        return 0.0, cur_c5
    p_unit = p / max_abs
    alpha_max = trust
    for j in range(n):
        if p_unit[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p_unit[j] * max_abs)
        elif p_unit[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p_unit[j]) * max_abs)
    alpha_max = max(alpha_max, 1e-12)
    alpha_max = min(alpha_max, trust)
    n_samp = 12
    alphas = np.linspace(0.0, alpha_max, n_samp)
    vals = np.empty(n_samp)
    for i in range(n_samp):
        vals[i] = exact_c5(h + alphas[i] * p)
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i == 0 or best_i == n_samp - 1:
        return best_a, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = c2
            f2 = f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = exact_c5(h + c1 * p)
        else:
            lo = c1
            f1 = f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo + hi) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def quasi_periodic_init(n, alpha):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0) < 0.5
    h = h.astype(float)
    return project(h, n / 2.0)

def smooth_perturb(h, scale, rng):
    n = len(h)
    noise = rng.normal(0, scale, n)
    if n > 4:
        kernel = np.array([1.0, 2.0, 3.0, 2.0, 1.0]) / 9.0
        noise = np.convolve(noise, kernel, mode='same')
    noise -= noise.mean()
    h_new = h + noise
    return project(h_new, n / 2.0)

def fourier_init(n, n_harm, phases, amps):
    x = np.arange(n) / n
    h = 0.5 * np.ones(n)
    for k, ph, a in zip(n_harm, phases, amps):
        h += a * np.cos(2 * np.pi * k * x + ph)
    h = project(h, n / 2.0)
    return h
