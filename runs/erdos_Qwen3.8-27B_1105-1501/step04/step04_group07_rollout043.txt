def _project_simplex(w):
    """Project w onto the probability simplex."""
    n = len(w)
    w = w.copy().astype(float)
    if n == 0:
        return w
    # Sort in decreasing order
    u = np.sort(w)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(w - theta, 0.0)

                        lo_a, hi_a = alphas[-2], a_max
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi_a - gr * (hi_a - lo_a)
                    c2 = lo_a + gr * (hi_a - lo_a)
                    f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
                    f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
                    for _gs in range(35):
                        if hi_a - lo_a < 1e-14:
                            break
                        if f1 < f2:
                            hi_a, f2, c2 = c2, f1, c1
                            c1 = hi_a - gr * (hi_a - lo_a)
                            f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
                        else:
                            lo_a, f1, c1 = c1, f2, c2
                            c2 = lo_a + gr * (hi_a - lo_a)
                            f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt > 1e-12:
                        h_try = h.copy()
                        h_try[i] += sign * a_opt
                        h_try[j] -= sign * a_opt
                        h_try = _project(np.clip(h_try, 0, 1), target)
                        nc = _exact_c5(h_try)
                        if nc < best_c5 - 1e-14:
                            best_c5 = nc
                            best_h = h_try.copy()
                            h = h_try.copy()
                            improved = True
        if not improved:
            break
    return best_h, best_c5

def _triple_move_polish(h, deadline, max_passes=5, seed=99):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    rng = np.random.default_rng(seed)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_attempts = 500
        for _at in range(n_attempts):
            if time.time() > deadline:
                break
            if h < 1e-8:
                continue
            interior = np.where((h > 1e-5) & (h < 1.0 - 1e-5))[0]
            if len(interior) < 3:
                continue
            idx = rng.choice(interior, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            r = rng.uniform(0.2, 0.8)
            for sign in (1, -1):
                if sign > 0:
                    a_max = min((1.0 - h[i]) / 1.0, (1.0 - h[j]) / r, h[k])
                else:
                    a_max = min(h[i], h[j] / r, (1.0 - h[k]))
                if a_max < 1e-12:
                    continue
                d = np.zeros(n)
                if sign > 0:
                    d[i] = 1.0
                    d[j] = r
                    d[k] = -(1.0 + r)
                else:
                    d[i] = -1.0
                    d[j] = -r
                    d[k] = (1.0 + r)
                raw = _scores_raw(h)
                L = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                Q = -np.correlate(d, d, 'full')
                n_g = 16
                alphas = np.linspace(0.0, a_max, n_g)
                vals = np.array([np.max(raw + a * L + a * a * Q) for a in alphas])
                bi = int(np.argmin(vals))
                if bi > 0 and bi < n_g - 1:
                    lo_a, hi_a = alphas[bi-1], alphas[bi+1]
                elif bi == 0:
                    lo_a, hi_a = 0.0, alphas[1]
                else:
                    lo_a, hi_a = alphas[-2], a_max
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi_a - gr * (hi_a - lo_a)
                c2 = lo_a + gr * (hi_a - lo_a)
                f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
                f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
                for _gs in range(25):
                    if hi_a - lo_a < 1e-14:
                        break
                    if f1 < f2:
                        hi_a, f2, c2 = c2, f1, c1
                        c1 = hi_a - gr * (hi_a - lo_a)
                        f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
                    else:
                        lo_a, f1, c1 = c1, f2, c2
                        c2 = lo_a + gr * (hi_a - lo_a)
                        f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
                a_opt = (lo_a + hi_a) / 2.0
                if a_opt > 1e-12:
                    h_try = _project(h + a_opt * d, target)
                    nc = _exact_c5(h_try)
                    if nc < best_c5 - 1e-14:
                        best_c5 = nc
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _active_descent(h, deadline, max_iters=3000, seed=55):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    rng = np.random.default_rng(seed)
    stall = 0
    for _it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        raw = _scores_raw(h)
        s_max = float(raw.max())
        tol = max(1e-7 * s_max, 1e-10)
        active = np.where(raw >= s_max - tol)[0]
        m_a = len(active)
        # Compute gradient as weighted sum of active lag gradients
        grad = np.zeros(n)
        w = np.ones(m_a) / m_a
        for idx_a, r in enumerate(active):
            m = r - (n - 1)
            src = np.zeros(n)
            tgt = np.zeros(n)
            if m >= 0:
                src[:n - m] = 1.0 - h[m:]
                tgt[m:] = -h[:n - m]
            else:
                gm = -m
                src[gm:] = 1.0 - h[:n - gm]
                tgt[:n - gm] = -h[gm:]
            grad += w[idx_a] * (src + tgt)
        grad -= np.mean(grad)
        d = -grad
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = rng.normal(0, 1, n)
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                stall += 1
                continue
            d /= dn
        else:
            d /= dn
        found = False
        for scale in (0.5, 0.25, 0.1, 0.05, 0.02):
            d_try = d * scale
            lo_a, hi_a = 0.0, 0.0
            for jj in range(n):
                if d_try[jj] > 1e-14:
                    hi_a = min(hi_a, (1.0 - h[jj]) / d_try[jj])
                elif d_try[jj] < -1e-14:
                    lo_a = max(lo_a, -h[jj] / d_try[jj])
            if hi_a - lo_a < 1e-14:
                continue
            L = np.correlate(d_try, 1.0 - h, 'full') - np.correlate(h, d_try, 'full')
            Q = -np.correlate(d_try, d_try, 'full')
            n_g = 24
            alphas = np.linspace(lo_a, hi_a, n_g)
            vals = np.array([np.max(raw + a * L + a * a * Q) for a in alphas])
            bi = int(np.argmin(vals))
            if bi > 0 and bi < n_g - 1:
                lo_a2, hi_a2 = alphas[bi-1], alphas[bi+1]
            elif bi == 0:
                lo_a2, hi_a2 = lo_a, alphas[1]
            else:
                lo_a2, hi_a2 = alphas[-2], hi_a
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_a2 - gr * (hi_a2 - lo_a2)
            c2 = lo_a2 + gr * (hi_a2 - lo_a2)
            f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
            f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
            for _ in range(30):
                if hi_a2 - lo_a2 < 1e-14:
                    break
                if f1 < f2:
                    hi_a2, f2, c2 = c2, f1, c1
                    c1 = hi_a2 - gr * (hi_a2 - lo_a2)
                    f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
                else:
                    lo_a2, f1, c1 = c1, f2, c2
                    c2 = lo_a2 + gr * (hi_a2 - lo_a2)
                    f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
            a_opt = (lo_a2 + hi_a2) / 2.0
            h_try = _project(h + a_opt * d_try, target)
            nc = _exact_c5(h_try)
            if nc < best_c5 - 1e-14:
                h = h_try
                best_c5 = nc
                best_h = h.copy()
                stall = 0
                found = True
                break
        if not found:
            stall += 1
    return best_h, best_c5

def _lp_direction(h, deadline, top_k=None, radius=0.05, max_iters=2000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    stall = 0
    r = radius
    n_lags = 2 * n - 1
    for _it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        raw = _scores_raw(h)
        scores = raw * (2.0 / n)
        if top_k is not None and top_k < n_lags:
            idx = np.argsort(scores)[-top_k:]
            sk = raw[idx]
        else:
            idx = np.arange(n_lags)
            sk = raw
        # Build LP: minimize t s.t. G@delta + sk <= t, sum(delta)=0, |delta|<=r
        n_active = len(idx)
        G = np.zeros((n_active, n))
        for ii, r_idx in enumerate(idx):
            m = r_idx - (n - 1)
            if m >= 0:
                G[ii, :n-m] += (1.0 - h[m:])
                G[ii, m:] += (-h[:n-m])
            else:
                gm = -m
                G[ii, gm:] += (1.0 - h[:n-gm])
                G[ii, :n-gm] += (-h[gm:])
        A_ub = np.zeros((n_active, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[j]), min(r, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                          b_eq=np.zeros(1), bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                r *= 0.7
                continue
            p = res.x[:n]
        except Exception:
            stall += 1
            r *= 0.7
            continue
        if np.max(np.abs(p)) < 1e-13:
            stall += 1
            r *= 0.7
            if r < 1e-8:
                break
            continue
        # Line search along p
        lo_a, hi_a = 0.0, 0.0
        for jj in range(n):
            if p[jj] > 1e-14:
                hi_a = min(hi_a, (1.0 - h[jj]) / p[jj])
            elif p[jj] < -1e-14:
                lo_a = max(lo_a, -h[jj] / p[jj])
        if hi_a - lo_a > 1e-14:
            L = np.correlate(p, 1.0 - h, 'full') - np.correlate(h, p, 'full')
            Q = -np.correlate(p, p, 'full')
            n_g = 20
            alphas = np.linspace(lo_a, hi_a, n_g)
            vals = np.array([np.max(raw + a * L + a * a * Q) for a in alphas])
            bi = int(np.argmin(vals))
            if bi > 0 and bi < n_g - 1:
                lo2, hi2 = alphas[bi-1], alphas[bi+1]
            elif bi == 0:
                lo2, hi2 = lo_a, alphas[1]
            else:
                lo2, hi2 = alphas[-2], hi_a
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi2 - gr * (hi2 - lo2)
            c2 = lo2 + gr * (hi2 - lo2)
            f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
            f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
            for _ in range(25):
                if hi2 - lo2 < 1e-14:
                    break
                if f1 < f2:
                    hi2, f2, c2 = c2, f1, c1
                    c1 = hi2 - gr * (hi2 - lo2)
                    f1 = float(np.max(raw + c1 * L + c1 * c1 * Q))
                else:
                    lo2, f1, c1 = c1, f2, c2
                    c2 = lo2 + gr * (hi2 - lo2)
                    f2 = float(np.max(raw + c2 * L + c2 * c2 * Q))
            a_opt = (lo2 + hi2) / 2.0
            h_try = _project(h + a_opt * p, target)
            nc = _exact_c5(h_try)
            if nc < best_c5 - 1e-14:
                h = h_try
                best_c5 = nc
                best_h = h.copy()
                stall = 0
                r = min(r * 1.05, 0.3)
            else:
                stall += 1
                r *= 0.8
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h, best_c5, best_n = h.copy(), _exact_c5(h), n0

    # Phase 1: Fienup binary init (8%)
    dl1 = t0 + budget_s * 0.08
    hb, c5b = _fienup_init(n0, seed, n_restarts=6, n_iters=300)
    if c5b < best_c5:
        best_c5, best_h, best_n = c5b, _project(hb, n0/2.0), n0

    # Phase 2: Binary local search on best binary (10%)
    dl2 = t0 + budget_s * 0.18
    h_bin = (best_h >= 0.5).astype(float)
    if abs(h_bin.sum() - n0/2) > 0.1:
        h_bin = (np.array(initial_h_values) >= 0.5).astype(float)
    hb2, c5b2 = _binary_swap_search(h_bin, dl2, n_iter=100000, seed=123)
    if c5b2 < best_c5:
        best_c5 = c5b2
        best_h = _project(hb2, n0/2.0)

    # Phase 3: LP direction refinement (15%)
    dl3 = t0 + budget_s * 0.33
    h3, c53 = _lp_direction(best_h, dl3, top_k=16, radius=0.15, max_iters=2000)
    if c53 < best_c5:
        best_c5, best_h = c53, h3

    # Phase 4: Active descent (17%)
    dl4 = t0 + budget_s * 0.50
    h4, c54 = _active_descent(best_h, dl4, max_iters=3000, seed=55)
    if c54 < best_c5:
        best_c5, best_h = c54, h4

    # Phase 5: Pair polish (18%)
    dl5 = t0 + budget_s * 0.68
    h5, c55 = _pair_polish(best_h, dl5, max_passes=15, seed=7)
    if c55 < best_c5:
        best_c5, best_h = c55, h5

    # Phase 6: Triple moves (8%)
    dl6 = t0 + budget_s * 0.76
    h6, c56 = _triple_move_polish(best_h, dl6, max_passes=5, seed=99)
    if c56 < best_c5:
        best_c5, best_h = c56, h6

    # Phase 7: Multi-resolution sweep (12%)
    dl7 = t0 + budget_s * 0.88
    for dn in (2, 4, -2, -4, 6, -6, 8, 12, -8):
        if time.time() > dl7:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 160 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(dl7, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_direction(h_r, sub_dl, top_k=12, radius=0.12, max_iters=400)
        if c5_r < best_c5 - 1e-12:
            sub_dl2 = min(dl7, time.time() + budget_s * 0.010)
            h_r, c5_r = _pair_polish(h_r, sub_dl2, max_passes=5, seed=3)
            if c5_r < best_c5 - 1e-12:
                best_c5, best_h, best_n = c5_r, h_r, n_try

    # Phase 8: Final active descent + pair polish (9%)
    dl8 = t0 + budget_s * 0.97
    h8, c58 = _active_descent(best_h, min(dl8, t0 + budget_s * 0.05), max_iters=2000, seed=77)
    if c58 < best_c5:
        best_c5, best_h = c58, h8
    h9, c59 = _pair_polish(best_h, dl8, max_passes=10, seed=11)
    if c59 < best_c5:
        best_c5, best_h = c59, h9

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
