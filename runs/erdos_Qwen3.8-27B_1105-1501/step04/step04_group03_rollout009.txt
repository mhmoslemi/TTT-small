import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    return np.clip(x, 0.0, 1.0)

def build_active_grad(h, m=20):
    """Build gradient matrix for top-m active constraints"""
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    # Get top m indices
    top_idx = np.argsort(scores)[::-1][:m]
    G = np.zeros((m, n))
    for j, k in enumerate(top_idx):
        d = k - (n - 1)
        if d >= 0:
            m_val = n - d
            if m_val > 0:
                G[j, d:] += (1.0 - h[:m_val])
                G[j, :m_val] += -h[d:]
        else:
            g = -d
            m_val = n - g
            if m_val > 0:
                G[j, :m_val] += (1.0 - h[g:])
                G[j, g:] += -h[:m_val]
    return G, scores[top_idx]

def lp_direction_active(h, m, radius):
    """LP direction using only top-m active constraints"""
    n = len(h)
    G, active_scores = build_active_grad(h, m)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # maximize t
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -active_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))  # t >= 0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    # Coarse grid
    for i in range(20):
        a = alpha_max * (i + 1) / 20.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    # Golden section
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(20):
        if hi_a - lo_a < 1e-14: break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lp_loop(h, deadline, radius_init=0.1, m_active=20):
    n = len(h)

    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 100: break
        p = lp_direction_active(h, m_active, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7

            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.75
            if stall % 20 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def soft_c5_grad(h, tau, top_m=None):
    """Compute smooth approximation of max and its gradient."""
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_m is not None and top_m < len(scores):
        idx = np.argsort(scores)[::-1][:top_m]

        exp_s = np.exp((scores_t - max_s) / tau)

        w = exp_s / Z
        val = max_s + tau * np.log(Z)
        grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += residual / cnt
    return np.clip(x, 0.0, 1.0)

def compute_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_grad_row(h, d):
    """Gradient of c_k with respect to h, where k corresponds to offset d."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        g_val = -d
        m = n - g_val
        if m > 0:
            g[:m] += (1.0 - h[g_val:])
            g[g_val:] += -h[:m]
    return g

def build_active_grad(h, m=15):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(scores)[::-1][:m]
    G = np.zeros((m, n))
    for j, k in enumerate(top_idx):
        d = k - (n - 1)
        G[j] = build_grad_row(h, d)
    return G, scores[top_idx]

def lp_dir_active(h, m, radius):
    n = len(h)
    G, active_scores = build_active_grad(h, m)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -active_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_dir_full(h, radius):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        G[n - 1 + d] = build_grad_row(h, d)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] > 1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, p, c5_cur):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-10)
    best_a = 0.0
    best_c5 = c5_cur
    for i in range(22):
        a = alpha_max * (i + 1) / 22.0
        c5_t = exact_c5(h + a * p)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, c5_cur
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.5)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(h + c1 * p)
    f2 = exact_c5(h + c2 * p)
    for _ in range(22):
        if hi_a - lo_a < 1e-14:
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(h + c1 * p)
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(h + c2 * p)
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(h + opt_a * p)
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5

def lp_loop(h, deadline, radius_init=0.1, m_active=15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    use_full = False
    while time.time() < deadline:
        if stall > 100:
            break
        if use_full:
            p = lp_dir_full(h, radius)
        else:
            p = lp_dir_active(h, m_active, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-8:
                break
            if stall % 8 == 0:
                use_full = not use_full
            continue
        alpha, new_c5 = line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            use_full = False
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.72
            if stall % 12 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def smooth_obj_grad(h, tau, top_m):
    n = len(h)
    dx = 2.0 / n
    scores_raw = np.correlate(h, 1.0 - h, mode='full')
    scores = scores_raw * dx
    nl = len(scores)
    if top_m < nl:
        idx = np.argsort(scores)[::-1][:top_m]
    else:
        idx = np.arange(nl)
    st = scores[idx]
    max_s = float(np.max(st))
    exp_s = np.exp((st - max_s) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    val = max_s + tau * np.log(Z)
    grad = np.zeros(n)
    for jj in range(len(idx)):
        wj = w[jj]
        if wj < 1e-15:
            continue
        d = idx[jj] - (n - 1)
        row = build_grad_row(h, d)
        grad += wj * row
    return val, grad * dx

def slsqp_smooth(h0, tau, maxiter, top_m=20):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, _ = smooth_obj_grad(x, tau, top_m)
        return v

    def grad(x):
        _, g = smooth_obj_grad(x, tau, top_m)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def subgradient_descent(h, deadline, tau_init=0.3):
    """Subgradient method on the smooth approximation with decreasing step."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.01
    tau = tau_init
    for it in range(500):
        if time.time() > deadline:
            break
        val, grad = smooth_obj_grad(h, tau, 20)
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad_normed = grad / gn
        h_new = project(h - step * grad_normed, target)
        new_c5 = exact_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        step *= 0.98
        tau *= 0.97
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def pair_polish(h, deadline, max_passes=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for k in range(j + 1, n):
                for sign in [1.0, -1.0]:
                    if sign > 0:
                        amax = min(best_h[j], 1.0 - best_h[k])
                    else:
                        amax = min(1.0 - best_h[j], best_h[k])
                    if amax < 1e-10:
                        continue
                    h_t = best_h.copy()
                    h_t[j] += sign * amax
                    h_t[k] -= sign * amax
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        improved = True
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def local_pair_search(h, deadline):
    """More aggressive local pair search with fractional moves."""
    n = len(h)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(12345)
    it = 0
    while time.time() < deadline:
        it += 1
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue
        direction = 1.0 if best_h[j] > best_h[k] else -1.0
        if direction > 0:
            amax = min(best_h[j], 1.0 - best_h[k])
        else:
            amax = min(1.0 - best_h[j], best_h[k])
        if amax < 1e-12:
            continue
        # Try several fractions
        for frac in [0.3, 0.5, 0.7, 1.0]:
            a = amax * frac
            h_t = best_h.copy()
            h_t[j] += direction * a
            h_t[k] -= direction * a
            c5_t = exact_c5(h_t)
            if c5_t < best_c5 - 1e-13:
                best_c5 = c5_t
                best_h = h_t.copy()
                break
    return best_h, best_c5
