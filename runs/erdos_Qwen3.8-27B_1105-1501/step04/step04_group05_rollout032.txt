                lo_t = max(0.0, p - 1.0)
                hi_t = min(1.0, p)
                if hi_t - lo_t < 1e-12:
                    continue
                a, b = lo_t, hi_t
                c = b - gr * (b - a)
                dd = a + gr * (b - a)
                h[i] = c
                h[j] = p - c
                fc = _exact_c5(h)
                h[i] = dd
                h[j] = p - dd
                fd = _exact_c5(h)
                for _ in range(25):
                    if b - a < 1e-11:
                        break
                    if fc < fd:
                        b = dd
                        dd = c
                        fd = fc
                        c = b - gr * (b - a)
                        h[i] = c
                        h[j] = p - c
                        fc = _exact_c5(h)
                    else:
                        a = c
                        c = dd
                        fc = fd
                        dd = a + gr * (b - a)
                        h[i] = dd
                        h[j] = p - dd
                        fd = _exact_c5(h)
                t_opt = (a + b) / 2.0
                h[i] = t_opt
                h[j] = p - t_opt
                c5_new = _exact_c5(h)
                if c5_new < best_c5 - tol:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    return h, best_c5

def _slsqp_refine(h, deadline, maxiter=300):
    n = len(h)
    target = n / 2.0
    def obj(x):
        c = _corr_vec(x)
        return float(np.max(c))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target),
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h.copy(), bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x, target)
            return h_new, _exact_c5(h_new)
    except Exception:
        pass
    return h, _exact_c5(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 16 or n_new > 256:
        return None
    x_old = (np.arange(n, dtype=np.float64) + 0.5) / n
    x_new = (np.arange(n_new, dtype=np.float64) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start_slq(h, n, deadline, rng, scale=0.03, max_attempts=100):
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    target = n / 2.0
    for _ in range(max_attempts):
        if time.time() > deadline:
            break
        h_p = best_h + rng.normal(0.0, scale, n)
        h_p = _project(h_p, target)
        h_try = _project(minimize(lambda x: float(np.max(_corr_vec(x))), x0=h_p,
                                  bounds=[(0.0, 1.0)] * n, method='L-BFGS-B',
                                  options={'maxiter': 50, 'ftol': 1e-12}).x, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.3)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=np.float64)
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Binary mask SA + two-level (25% budget)
    d1 = t0 + budget_s * 0.25
    mask_seeds = []
    # Threshold of parent
    sorted_vals = np.sort(h)[::-1]
    thresh = sorted_vals[int(target) - 1]
    mask1 = (h >= thresh).astype(np.float64)
    if np.sum(mask1) != int(target):
        idx = np.where(h >= thresh)[0]
        rem = int(np.sum(mask1)) - int(target)
        for _ in range(rem):
            mask1[idx[rng.integers(0, len(idx))]] = 0.0
        mask1 = (np.sum(mask1) == int(target)) * mask1
    mask_seeds.append(mask1)
    # Complement
    mask_seeds.append(1.0 - mask1)
    # Random seeds
    for _ in range(6):
        m = np.zeros(n)
        m[rng.choice(n, size=int(target), replace=False)] = 1.0
        mask_seeds.append(m)
    # Arithmetic
    for st in [3, 5, 7, 9]:
        if np.gcd(st, n) == 1:
            pos = sorted([(k * st) % n for k in range(int(target))])
            m = np.zeros(n)
            m[pos] = 1.0
            mask_seeds.append(m)

    per_seed = max(1.0, (d1 - time.time()) / max(1, len(mask_seeds)))
    for ms in mask_seeds:
        if time.time() > d1:
            break
        pd = time.time() + min(per_seed, 15.0)
        bm, bs = _sa_binary(n, pd, rng, mask_init=ms)
        h2 = _two_level_optimize(bm)
        c5_2 = _exact_c5(h2)
        if c5_2 < best_c5 - 1e-14:
            best_c5 = c5_2
            best_h = h2.copy()
        # Also try the binary itself as a starting point for pair descent
        c5_bin = _binary_score(bm, n)
        if c5_bin < best_c5 - 1e-14:
            best_c5 = c5_bin
            best_h = bm.astype(np.float64).copy()

    # Phase 2: Coarse pair descent (25% budget)
    d2 = t0 + budget_s * 0.50
    hp, c5p = _pair_descent(best_h, d2, max_passes=200, tol=1e-13)
    if c5p < best_c5:
        best_c5 = c5p
        best_h = hp.copy()

    # Phase 3: Multi-start with L-BFGS (15% budget)
    d3 = t0 + budget_s * 0.65
    if time.time() < d3:
        hm, c5m = _multi_start_slq(best_h, n, d3, rng, scale=0.04, max_attempts=80)
        if c5m < best_c5:
            best_c5 = c5m
            best_h = hm.copy()

    # Phase 4: Fine pair descent with golden section (20% budget)
    d4 = t0 + budget_s * 0.85
    hf, c5f = _pair_descent_fine(best_h, d4, max_passes=200, tol=1e-14)
    if c5f < best_c5:
        best_c5 = c5f
        best_h = hf.copy()

    # Phase 5: Resolution search (10% budget)
    d5 = t0 + budget_s * 0.95
    for dn in [-2, -4, -6, 2, 4, 6, -8, 8, -10, 10]:
        if time.time() > d5:
            break
        nt = n + dn
        if nt < 20 or nt > 200 or nt % 2 != 0:
            continue
        h_r = _resample(best_h, nt)
        if h_r is None:
            continue
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        hr, c5r = _pair_descent(h_r, sub_dl, max_passes=80, tol=1e-12)
        if c5r < best_c5 - 1e-10:
            best_c5 = c5r
            best_h = hr.copy()
            n = nt

    # Phase 6: Final SLSQP + fine polish (5% budget)
    d6 = deadline
    if time.time() < d6:
        hsl, c5sl = _slsqp_refine(best_h, d6, maxiter=200)
        if c5sl < best_c5:
            best_c5 = c5sl
            best_h = hsl.copy()
        if time.time() < d6:
            hpf, c5pf = _pair_descent_fine(best_h, d6, max_passes=100, tol=1e-15)
            if c5pf < best_c5:
                best_c5 = c5pf
                best_h = hpf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > 0.38085:
        final_c5 = _exact_c5(_project(h0, len(h0) / 2.0))
        best_h = _project(h0, len(h0) / 2.0)
    return best_h, final_c5, len(best_h)
