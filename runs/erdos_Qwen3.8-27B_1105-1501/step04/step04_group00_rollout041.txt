import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project(v, target):
    # Water-filling projection onto {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    # Fix any numerical drift
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
    return np.clip(x, 0, 1)

def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                # For j in [0, m-1]: contribution is (1-h[j+d])
                G[r, :m] += (1.0 - h[d:d+m])
                # For j in [d, n-1]: contribution is -h[j-d]
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                # For j in [g, n-1]: contribution is (1-h[j-g])
                G[r, g:] += (1.0 - h[:m])
                # For j in [0, m-1]: contribution is -h[j+g]
                G[r, :m] += -h[g:]
    return G * dx

def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        sk = scores[idx]
        Gk = G[idx]
    else:
        sk = scores
        Gk = G
    # LP: min t
    # s.t. Gk @ p - t <= -sk
    #      sum(p) = 0
    #      bounds on p
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.zeros((len(sk), n+1))
    A_ub[:, :n] = Gk
    A_ub[:, n] = -1.0
    b_ub = -sk
    A_eq = np.array([[1.0]*n + [0.0]])
    b_eq = np.array([0.0])
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))

    except Exception:

    return None, None

def _softmax_gradient_step(h, deadline, beta_start=10.0, beta_end=200.0, n_steps=300, alpha_init=0.1):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _kkt_direction(h, scores, G, active_indices):
    """Find optimal direction using KKT analysis."""
    n = len(h)
    # Free variables: 0 < h[i] < 1
    free = (h > 1e-6) & (h < 1.0 - 1e-6)
    n_free = np.sum(free)
    if n_free < 2:
        return None
    
    # Solve QP: min ||G[A, free]^T @ lambda||^2 / 2
    # s.t. sum(lambda) = 1, lambda >= 0
    # This gives the optimal weight on active constraints
    
    G_active = G[active_indices, free]  # shape (k, n_free)
    k = len(active_indices)
    
    # We want to minimize ||G_active^T @ lambda + c||^2 where c accounts for bounded vars
    # Simplified: just minimize ||G_active^T @ lambda||^2
    # This is min lambda^T G G^T lambda s.t. sum(lambda)=1, lambda>=0
    
    H = G_active @ G_active.T  # shape (k, k)
    # min lambda^T H lambda s.t. 1^T lambda = 1, lambda >= 0
    
    # Simple approach: use the closed form for equality constraint only
    # lambda* = H^{-1} 1 / (1^T H^{-1} 1)
    try:
        H_reg = H + 1e-8 * np.eye(k)
        inv_H = np.linalg.inv(H_reg)
        ones_k = np.ones(k)
        lam = inv_H @ ones_k
        lam = lam / np.sum(lam)
        # Project to non-negative
        lam = np.maximum(lam, 0)
        lam = lam / np.sum(lam)
    except Exception:
        lam = np.ones(k) / k
    
    # Direction: -G_active^T @ lambda, projected to free vars and sum=0
    grad = np.zeros(n)
    grad[free] = -(G_active.T @ lam)
    # Also handle bounded vars (they don't contribute to the gradient in the projected sense)
    grad -= grad.mean()
    
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return None
    return grad / gn

from scipy.optimize import linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _scores(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)


def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))

        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G * dx


def _lp_direction(h, scores, G, radius, top_k=None):
    n = len(h)
    n_lags = len(scores)
    if top_k is not None and top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        sk = scores[idx]
        Gk = G[idx]
    else:
        sk = scores
        Gk = G
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(sk), n + 1))
    A_ub[:, :n] = Gk
    A_ub[:, n] = -1.0
    b_ub = -sk
    A_eq = np.array([[1.0] * n + [0.0]])
    b_eq = np.array([0.0])
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        if lo_b >= hi_b:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None


def _line_search(h, p, n_coarse=12, n_fine=22):
    n = len(h)
    alpha_max = 1.0
    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    alpha_max = min(alpha_max, 0.8)
    alphas = np.linspace(0, alpha_max, n_coarse)
    vals = np.empty(n_coarse)
    for i, a in enumerate(alphas):
        vals[i] = _exact_c5(np.clip(h + a * p, 0, 1))
    best_i = int(np.argmin(vals))
    best_c5 = vals[best_i]
    best_a = alphas[best_i]
    if best_i in (0, n_coarse - 1):
        return best_a, best_c5
    lo = alphas[best_i - 1]
    hi = alphas[best_i + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
    f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    for _ in range(n_fine):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi, f2 = c2, f1
            c2 = c1
            c1 = hi - gr * (hi - lo)
            f1 = _exact_c5(np.clip(h + c1 * p, 0, 1))
        else:
            lo, f1 = c1, f2
            c1 = c2
            c2 = lo + gr * (hi - lo)
            f2 = _exact_c5(np.clip(h + c2 * p, 0, 1))
    opt_a = (lo + hi) / 2.0
    opt_c5 = _exact_c5(np.clip(h + opt_a * p, 0, 1))
    if opt_c5 < best_c5:
        return opt_a, opt_c5
    return best_a, best_c5


def _local_opt(h, deadline, top_k_list=None, radius_init=0.15, max_iters=5000):
    if top_k_list is None:
        top_k_list = [None, 30, 15, 8]
    n = len(h)
    target = n / 2.0
    radius = radius_init
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    ti = 0
    n_tk = len(top_k_list)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        tk = top_k_list[ti % n_tk]
        scores = _scores(h)
        G = _grad_matrix(h)
        p, _ = _lp_direction(h, scores, G, radius, top_k=tk)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.65
            ti += 1
            if radius < 1e-7:
                break
            continue
        p -= p.mean()
        alpha, new_c5 = _line_search(h, p)
        if new_c5 < cur_c5 - 1e-13:
            h = np.clip(h + alpha * p, 0, 1)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.06, 0.35)
        else:
            stall += 1
            radius *= 0.72
            ti += 1
            if stall > 10 and radius < 0.03:
                radius = 0.08
    return best_h, best_c5


def _softmax_grad_step(h, deadline, beta_start=8.0, beta_end=150.0, n_steps=300, alpha_init=0.12):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        scores = _scores(h)
        max_s = np.max(scores)
        w = np.exp(beta * (scores - max_s))
        sw = np.sum(w)
        if sw < 1e-300:
            w = np.zeros(len(scores))
            w[int(np.argmax(scores))] = 1.0
        else:
            w /= sw
        G = _grad_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        alpha = alpha_init * (1.0 - 0.4 * frac)
        improved = False
        for _ in range(45):
            h_new = _project(h - alpha * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            alpha *= 0.68
            if alpha < 1e-12:
                break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quasi_periodic(n, alpha, rng=None):
    x = np.arange(n) / n
    h = ((x * alpha) % 1.0).astype(float)
    h = (h < 0.5).astype(float)
    return _project(h, n / 2.0)


def _binary_sa(n, deadline, rng, n_iters=40000, T_start=0.04, T_end=0.0008):
    target = n // 2
    positions = rng.permutation(n)
    h = np.zeros(n)
    h[positions[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_start
    cool = (T_end / T_start) ** (1.0 / max(n_iters, 1))
    for it in range(n_iters):
        if time.time() > deadline:
            break
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        i = int(ones_idx[rng.integers(len(ones_idx))])
        j = int(zeros_idx[rng.integers(len(zeros_idx))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or (T > T_end and rng.random() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return best_h, best_c5


def _kkt_local_opt(h, deadline, max_iters=2000, n_active=25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        scores = _scores(h)
        max_s = np.max(scores)
        tol = 0.005
        active_mask = scores > max_s * (1.0 - tol)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 3:
            active_idx = np.argsort(scores)[-n_active:]
        if len(active_idx) > n_active:
            active_idx = np.argsort(scores)[-n_active:]
        G = _grad_matrix(h)
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        n_free = int(np.sum(free))
        if n_free < 3:
            stall += 5
            continue
        G_active = G[active_idx, free]
        k = len(active_idx)
        H = G_active @ G_active.T + 1e-9 * np.eye(k)
        try:
            inv_H = np.linalg.inv(H)
            ones_k = np.ones(k)
            lam = inv_H @ ones_k
            lam = np.maximum(lam, 0.0)
            lam_sum = np.sum(lam)
            if lam_sum < 1e-12:
                lam = np.ones(k) / k
            else:
                lam /= lam_sum
        except Exception:
            lam = np.ones(k) / k
        grad = np.zeros(n)
        grad[free] = -(G_active.T @ lam)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            stall += 1
            continue
        grad /= gn
        alpha, new_c5 = _line_search(h, grad, n_coarse=10, n_fine=18)
        if new_c5 < cur_c5 - 1e-13:
            h = np.clip(h + alpha * grad, 0, 1)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def _pair_polish(h, deadline, max_passes=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        step = max(1, n // 20)
        for j in range(n):
            if time.time() > deadline:
                break
            if h[j] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            for k in range(0, n, step):
                if k == j:
                    continue
                if h[k] < 1e-8 or h[k] > 1.0 - 1e-8:
                    continue
                for sign in [1, -1]:
                    if sign > 0:
                        a_max = min(1.0 - h[j], h[k])
                    else:
                        a_max = min(h[j], 1.0 - h[k])
                    if a_max < 1e-10:
                        continue
                    lo_a, hi_a = 0.0, a_max
                    for _ in range(9):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy()
                        h1[j] += sign * m1
                        h1[k] -= sign * m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy()
                        h2[j] += sign * m2
                        h2[k] -= sign * m2
                        f2 = _exact_c5(h2)
                        if f1 < f2:
                            hi_a = m2
                        else:
                            lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    if a_opt < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def _block_swap_polish(h, deadline, rng, n_blocks=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    block_sizes = [2, 3, 4, 5, 6]
    for _ in range(n_blocks):
        if time.time() > deadline:
            break
        bs = int(block_sizes[rng.integers(len(block_sizes))])
        if bs >= n:
            continue
        i = int(rng.integers(0, n - bs))
        j = int(rng.integers(0, n - bs))
        if abs(i - j) < bs:
            continue
        h_try = h.copy()
        h_try[i:i+bs], h_try[j:j+bs] = h[j:j+bs].copy(), h[i:i+bs].copy()
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    def elapsed():
        return time.time() - t0

    # Phase 1: LP from parent (0-7%)
    dl = t0 + budget_s * 0.07
    h1, c51 = _local_opt(h_start, dl, top_k_list=[None, 25, 12],
                         radius_init=0.2, max_iters=2000)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Softmax gradient (7-13%)
    if elapsed() < budget_s * 0.13:
        dl = t0 + budget_s * 0.13
        h2, c52 = _softmax_grad_step(best_h, dl, beta_start=5, beta_end=120,
                                     n_steps=200, alpha_init=0.15)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()

    # Phase 3: Multi-resolution diverse search (13-45%)
    dl3 = t0 + budget_s * 0.45
    resolutions = [96, 112, 128, 144, 128, 144, 160, 128, 144, 160, 192]
    alphas_qp = [0.6180339887, 0.4142135624, 0.7071067812, 0.3819660113,
                 0.534784, 0.6823278, 0.5572809]
    ri = 0
    for n_try in resolutions:
        if time.time() > dl3:
            break
        if best_c5 < 0.3802:
            break
        for s in range(2):
            if time.time() > dl3:
                break
            if s == 0:
                h_init = _resample(best_h, n_try)
            else:
                alpha = alphas_qp[ri % len(alphas_qp)]
                h_init = _quasi_periodic(n_try, alpha)
                h_init += rng.normal(0, 0.06, n_try)
                h_init = _project(h_init, n_try / 2.0)
            sub_dl = min(dl3, time.time() + budget_s * 0.03)
            h_try, c5_try = _local_opt(h_init, sub_dl,
                                        top_k_list=[None, 20, 10],
                                        radius_init=0.15, max_iters=2500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            if time.time() < dl3:
                sub_dl2 = min(dl3, time.time() + budget_s * 0.012)
                h_wg, c5_wg = _softmax_grad_step(h_try, sub_dl2,
                                                  beta_start=8, beta_end=60,
                                                  n_steps=80, alpha_init=0.08)
                if c5_wg < best_c5:
                    best_c5 = c5_wg
                    best_h = h_wg.copy()
                    best_n = n_try
            ri += 1

    # Phase 4: KKT optimization (45-58%)
    if elapsed() < budget_s * 0.58:
        dl = t0 + budget_s * 0.58
        h3, c53 = _kkt_local_opt(best_h, dl, max_iters=2000, n_active=25)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()
        if elapsed() < budget_s * 0.57 and best_n < 160:
            n_hi = min(160, best_n + 32)
            if n_hi % 2 != 0:
                n_hi -= 1
            h_hi = _resample(best_h, n_hi)
            sub_dl = min(dl, time.time() + budget_s * 0.025)
            h_hi, c5_hi = _local_opt(h_hi, sub_dl, top_k_list=[None, 15],
                                      radius_init=0.12, max_iters=1500)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi
                if time.time() < dl:
                    sub_dl2 = min(dl, time.time() + budget_s * 0.01)
                    h_hi2, c5_hi2 = _kkt_local_opt(h_hi, sub_dl2, max_iters=800, n_active=20)
                    if c5_hi2 < best_c5:
                        best_c5 = c5_hi2
                        best_h = h_hi2.copy()

    # Phase 5: Binary SA (58-72%)
    if elapsed() < budget_s * 0.72:
        dl = t0 + budget_s * 0.72
        n_opts = [best_n, best_n - 16, best_n + 16, 128]
        seen = set()
        for n_bin in n_opts:
            if time.time() > dl:
                break
            if n_bin < 32 or n_bin in seen:
                continue
            seen.add(n_bin)
            if n_bin % 2 != 0:
                n_bin += 1
            sub_dl = min(dl, time.time() + budget_s * 0.02)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=30000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_bin
                if time.time() < dl:
                    sub_dl2 = min(dl, time.time() + budget_s * 0.012)
                    h_bp, c5_bp = _local_opt(h_b, sub_dl2, top_k_list=[None, 10],
                                              radius_init=0.08, max_iters=800)
                    if c5_bp < best_c5:
                        best_c5 = c5_bp
                        best_h = h_bp.copy()

    # Phase 6: Perturbation restarts (72-88%)
    dl6 = t0 + budget_s * 0.88
    scale = 0.04
    for attempt in range(250):
        if time.time() > dl6:
            break
        h_pert = best_h + rng.normal(0, scale, best_n)
        h_pert = _project(h_pert, best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.007)
        h_try, c5_try = _local_opt(h_pert, sub_dl,
                                    top_k_list=[10, 5, 3],
                                    radius_init=0.05, max_iters=250)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.85)
        else:
            scale = min(0.08, scale * 1.08)

    # Phase 7: Block swap + pair polish (88-96%)
    if elapsed() < budget_s * 0.96:
        dl = t0 + budget_s * 0.94
        hb, c5b = _block_swap_polish(best_h, dl, rng, n_blocks=300)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb.copy()
        if time.time() < dl + budget_s * 0.015:
            hp, c5p = _pair_polish(best_h, t0 + budget_s * 0.96, max_passes=6)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = hp.copy()

    # Phase 8: Final tiny LP refinement (96-99.5%)
    if elapsed() < budget_s * 0.995:
        h_fin, c5_fin = _local_opt(best_h, t0 + budget_s * 0.997,
                                    top_k_list=[5, 3, 8, 4],
                                    radius_init=0.01, max_iters=500)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
