                        break
                    i, j = cands[ii], cands[jj]
                    result = _pair_swap_opt(h, i, j, cur_c5)
                    if result is not None:
                        h, cur_c5 = result
                        found_pdf = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        break

        if found_pdf:
            stall = 0
            continue

        # Also try some random pairs among all entries
        if iter_count % 3 == 0 and len(h) > 4:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                result = _pair_swap_opt(h, i, j, cur_c5)
                if result is not None:
                    h, cur_c5 = result
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    continue

        # Perturbation to escape
        stall += 1
        scale = 0.005 + 0.01 * np.sqrt(stall / 10.0)
        scale = min(scale, 0.15)
        h_pert = h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        c5_pert = _c5_exact(h_pert)
        if c5_pert < cur_c5:
            h = h_pert
            cur_c5 = c5_pert

    return best_h, best_c5

def _cook_intensive(h, deadline, seed=0):
    """Longer cooking with adaptive perturbation."""
    n = len(h)
    target = n / 2.0
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    phase = 0

    while time.time() < deadline:
        if stall > 150:
            # Try larger escape
            phase = (phase + 1) % 4
            scales = [0.02, 0.05, 0.1, 0.2]
            h_pert = h + rng.normal(0, scales[phase], n)
            h_pert = _project(h_pert, target)
            c5_pert = _c5_exact(h_pert)
            if c5_pert < cur_c5:
                h = h_pert
                cur_c5 = c5_pert
                stall = 0
                continue
            stall = 0
            # Try complement shift
            if phase == 2:
                h_shift = np.roll(h, 1)
                h_shift = _project(h_shift, target)
                c5_sh = _c5_exact(h_shift)
                if c5_sh < cur_c5:
                    h = h_shift
                    cur_c5 = c5_sh
                    best_c5 = min(best_c5, c5_sh)
                    best_h = h.copy()
            continue

        scores = _scores(h)
        max_score = float(np.max(scores))
        tol = max(1e-7, max_score * 5e-6)
        active_idx_all = np.where(scores >= max_score - tol)[0]
        n_act = len(active_idx_all)

        found = False
        k_opts = sorted(set([max(1, min(n_act, k)) for k in [2, 3, 4, 6, 10, 15]]))
        k_opts.append(n_act)
        k_opts = sorted(set(k_opts), reverse=True)

        for k in k_opts:
            if found or time.time() >= deadline:
                break
            top_k = active_idx_all[:k] if n_act >= k else active_idx_all
            # Sort by score descending
            top_k = top_k[np.argsort(scores[top_k])[::-1][:k]]
            for radius in [0.015, 0.04, 0.07, 0.1, 0.15, 0.25]:
                if time.time() >= deadline:
                    break
                p = _lp_dir(h, top_k, radius)
                if p is None:
                    continue
                p -= np.mean(p)
                if np.max(np.abs(p)) < 1e-12:
                    continue
                alpha_max = _max_alpha(h, p)
                if alpha_max < 1e-12:
                    continue
                alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
                if new_c5 < cur_c5 - 1e-14:
                    h = _project(h + alpha * p, target)
                    cur_c5 = _c5_exact(h)
                    found = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break

        if found:
            stall = 0
            continue

        # Pair swaps - more aggressive
        frac = np.where((h > 5e-7) & (h < 1.0 - 5e-7))[0]
        found_pair = False
        if len(frac) >= 2:
            fo = np.argsort(np.abs(h[frac] - 0.5))
            cands = frac[fo[:25]]
            for ii in range(len(cands)):
                if found_pair or time.time() >= deadline:
                    break
                for jj in range(ii + 1, min(ii + 12, len(cands))):
                    if time.time() >= deadline:
                        break
                    result = _pair_swap_opt(h, cands[ii], cands[jj], cur_c5)
                    if result is not None:
                        h, cur_c5 = result
                        found_pair = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        break

        if found_pair:
            stall = 0
            continue

        stall += 1
        # Small perturbation
        if stall % 5 == 0:
            sc = 0.003 + 0.005 * (stall % 20) / 20.0
            h_p = h + rng.normal(0, sc, n)
            h_p = _project(h_p, target)
            cp = _c5_exact(h_p)
            if cp < cur_c5:
                h = h_p
                cur_c5 = cp

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    elapsed = lambda: time.time() - t0
    remaining = lambda: budget_s - elapsed()

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    best_n = n0

    # Phase 1: Initial cooking at parent resolution (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _cook(h, d1, seed=seed)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1.copy(), n0

    # Phase 2: Multi-resolution search (35%)
    d2 = t0 + budget_s * 0.60
    n_candidates = sorted(set([best_n + d for d in range(-12, 13)] +
                               [60, 64, 70, 72, 76, 80, 84, 88, 90, 92, 96, 100, 104, 108, 112, 120]))
    n_candidates = [n for n in n_candidates if 50 <= n <= 130 and n != best_n]
    n_candidates.sort()

    # Allocate time per resolution
    n_res = len(n_candidates)
    if n_res > 0:
        time_per = max(5.0, (d2 - time.time()) / n_res)
    else:
        time_per = 0.0

    for n_try in n_candidates:
        if time.time() >= d2 - 5.0:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2 - 2.0, time.time() + time_per)
        h_r, c5_r = _cook(h_r, sub_dl, seed=seed + n_try)
        if c5_r < best_c5 - 1e-12:
            best_c5, best_h = c5_r, h_r.copy()
            best_n = n_try

    # Phase 3: Intensive cooking at best resolution (30%)
    d3 = t0 + budget_s * 0.90
    h3, c5_3 = _cook_intensive(best_h, d3, seed=seed + 1000)
    if c5_3 < best_c5:
        best_c5, best_h, best_n = c5_3, h3.copy(), len(h3)

    # Phase 4: Final polish (remaining time, at least 5s reserved)
    d4 = t0 + budget_s * 0.99
    if time.time() < d4:
        h4, c5_4 = _cook(best_h, d4, seed=seed + 2000, max_stall=200)
        if c5_4 < best_c5:
            best_c5, best_h, best_n = c5_4, h4.copy(), len(h4)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5_exact(best_h)
    # If projection worsened things slightly, check
    if final_c5 > best_c5 + 1e-9:
        # Re-evaluate without projection (shouldn't happen)
        pass

    return best_h, final_c5, best_n
