def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0

        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:

                a = min(1.0 - x[i], diff)
                x[i] += a

    return x

def _lp_step(h, n, K=None, radius=0.1):
    """Solve LP to find descent direction."""
    corr = _corr(h)
    if K is None:
        K = 2*n - 1  # use all
    top_idx = np.argsort(-corr)[:K]
    G = _grad_matrix(h, n)
    G_active = G[top_idx, :]
    scores = corr[top_idx]
    
    # Variables: p (n), t (1) -> total n+1
    # Objective: min t
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Inequality: G_active @ p - t <= -scores  (i.e., G_active @ p + scores <= t)
    # Rearranged: G_active @ p - t <= -scores
    A_ub = np.hstack([G_active, -np.ones((K, 1))])
    b_ub = -scores
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: for p_i: max(-radius, -h[i]) to min(radius, 1-h[i])
    # For t: [0, None] or [-large, None]
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i])
        hi = min(radius, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0 (correlations are non-negative... actually not necessarily)
    
    # Actually, t can be negative in the LP (it's the predicted max correlation)
    # But we're looking for t < current max, so let's allow negative
    bounds[-1] = (None, None)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x[:n]

def _line_search(h, p, n):
    """Find best alpha for h + alpha*p."""
    # Find max alpha where h + alpha*p stays in [0,1]
    alpha_max = 10.0
    for i in range(n):
        if p[i] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[i]) / p[i])
        elif p[i] < -1e-14:
            alpha_max = min(alpha_max, h[i] / (-p[i]))
    alpha_max = max(alpha_max, 1e-6)
    
    cur_c5 = _c5_val(h)
    best_a = 0.0
    best_c5 = cur_c5
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, alpha_max
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = _c5_val(_project(h + c1 * p, n * 0.5))
    f2 = _c5_val(_project(h + c2 * p, n * 0.5))
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = _c5_val(_project(h + c1 * p, n * 0.5))
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = _c5_val(_project(h + c2 * p, n * 0.5))
    
    a = (lo + hi) / 2.0

    c5_new = _c5_val(h_new)

        best_h, best_c5 = h1, c5_1

    # Phase 2: Subgradient warm-up then LP (12% budget)
    d2 = t0 + budget_s * 0.24
    h2, c5_2 = _subgrad_loop(best_h, best_n, min(d2, time.time() + budget_s * 0.04), step_init=0.03)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2
    h2b, c5_2b = _lp_loop(best_h, best_n, d2, K=None, radius_init=0.08, max_iters=3000, stall_limit=25)
    if c5_2b < best_c5:
        best_h, best_c5 = h2b, c5_2b

    # Phase 3: Multi-K LP refinement (18% budget)
    d3 = t0 + budget_s * 0.42
    for K_val in [None, 50, 30, 15, 10, 5]:
        if time.time() > d3:
            break
        sub_d = min(d3, time.time() + budget_s * 0.03)
        h_k, c5_k = _lp_loop(best_h, best_n, sub_d, K=K_val, radius_init=0.06,
                             max_iters=2000, stall_limit=20)
        if c5_k < best_c5:
            best_h, best_c5 = h_k, c5_k

    # Phase 4: Resolution search (30% budget)
    d4 = t0 + budget_s * 0.72
    n_candidates = [best_n, best_n + 4, best_n - 4, best_n + 8, best_n - 8,
                    best_n + 16, best_n - 16, 48, 56, 72, 80, 96, 112, 128, 160, 192]
    n_candidates = sorted(set(n for n in n_candidates if 32 <= n <= 256 and n % 2 == 0))
    for n_try in n_candidates:
        if time.time() > d4:
            break
        h_r = _resample(best_h, n_try)
        sub_d = min(d4, time.time() + budget_s * 0.022)
        h_rpc, c5_rpc = _lp_loop(h_r, n_try, sub_d, K=None, radius_init=0.1,
                                 max_iters=2000, stall_limit=25)
        if c5_rpc < best_c5 - 1e-12:
            best_c5, best_h, best_n = c5_rpc, h_rpc.copy(), n_try
        # Second round with different K
        if time.time() < d4:
            sub_d2 = min(d4, time.time() + budget_s * 0.012)
            h_rpc2, c5_rpc2 = _lp_loop(best_h, best_n, sub_d2, K=20, radius_init=0.04,
                                       max_iters=1000, stall_limit=15)
            if c5_rpc2 < best_c5:
                best_c5, best_h = c5_rpc2, h_rpc2.copy()

    # Phase 5: Perturbation + LP (16% budget)
    d5 = t0 + budget_s * 0.88
    h_p, c5_p = _lp_perturb_loop(best_h, best_n, d5, rng, max_attempts=300)
    if c5_p < best_c5:
        best_h, best_c5 = h_p, c5_p

    # Phase 6: Final polish (12% budget)
    d6 = t0 + budget_s * 0.99
    h_f, c5_f = _lp_loop(best_h, best_n, d6, K=None, radius_init=0.04,
                         max_iters=5000, stall_limit=80)
    if c5_f < best_c5:
        best_h, best_c5 = h_f, c5_f
    # One more subgradient polish
    if time.time() < d6:
        h_s, c5_s = _subgrad_loop(best_h, best_n, d6, step_init=0.02, max_iters=2000)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    best_h = _project(best_h, best_n * 0.5)
    final_c5 = _c5_val(best_h)
    if 'evaluate_erdos_solution' in dir():
        try:
            ev = evaluate_erdos_solution(best_h, best_n)
            if ev < final_c5:
                final_c5 = ev
        except Exception:
            pass
    return best_h, final_c5, best_n
