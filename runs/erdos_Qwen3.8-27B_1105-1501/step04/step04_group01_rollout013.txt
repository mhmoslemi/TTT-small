def _lp_direction(h, scores, radius):
    """Find LP direction to reduce max overlap."""
    n = len(h)
    G = _all_grads(h)
    # c[r] = sum_i h[i]*(1-h[i+s]) -> dc[r]/dh[k] = G[k, r]
    # We want to find d with sum(d)=0, |d_i| <= radius, such that
    # for all r: sum_k G[k,r]*d[k] <= 0 (i.e., all c[r] are non-increasing)
    # Actually, we want to minimize max_r (c[r] + t * dc[r]/dt) for small t
    # which means minimize max_r dc[r] (the directional derivative)
    # LP: min max_r (G @ d)[r] s.t. sum(d)=0, -radius <= d_i <= radius, and bounds
    
    # More precisely: min t s.t. (G @ d)[r] <= t for all r, sum(d)=0, bounds
    # But we also want d to be "effective" - let's add a constraint that at least one active lag decreases.
    
    # Simpler: minimize sum of positive parts, or just minimize max_r (G@d)[r]
    # Using LP: min t s.t. G@d - t*1 <= 0, sum(d)=0, bounds

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t

    A_ub[:, n] = -1.0  # -t
    b_ub = np.zeros(2*n-1)  # G@d - t <= 0
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0  # sum(d) = 0
    b_eq = np.zeros(1)

    for k in range(n):
        lo_b = max(-radius, -h[k])
        hi_b = min(radius, 1.0 - h[k])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))  # t is free
    
    try:

                      bounds=bounds, method='highs')
        if res.success:

            if float(np.max(np.abs(d))) > 1e-12:

        pass

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx
    for _ in range(40):
        if b-a < 1e-13: break
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx
    opt_t = (a+b)/2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*Q)) * dx
    return opt_t, opt_val
