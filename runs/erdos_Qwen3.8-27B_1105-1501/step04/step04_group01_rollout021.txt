            continue
        stall = 0
        lr = min(lr * 1.02, 0.15)
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
    return best_h, best_c5

def _lp_dir(h, scores, G, rad):
    """LP direction for reducing max overlap."""
    n = len(h)
    c = np.zeros(n+1); c[-1] = 1.0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:,:n] = G
    A_ub[:,n] = -1.0
    b_ub = -scores
    A_eq = np.ones((1,n+1)); A_eq[0,n] = 0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-rad, -h[j])
        hi = min(rad, 1.0-h[j])
        bounds.append((lo,hi))
    bounds.append((None,None))
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs', options={'time_limit': 10})
        if r.success and r.x is not None:
            return r.x[:n]
    except:
        pass
    return None

def _grad_mat(h):
    n = len(h)
    dx = 2.0/n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        r = n-1+d
        if d >= 0:
            m = n-d
            if m > 0:
                G[r,:m] += dx*(1.0-h[d:d+m])
                G[r,d:] += dx*(-h[:m])
        else:
            g = -d
            m = n-g
            if m > 0:
                G[r,:m] += dx*(-h[g:g+m])
                G[r,g:] += dx*(1.0-h[:m])
    return G

def _fw_dir(h, g):
    """Frank-Wolfe direction."""
    n = len(h)
    nw = int(n/2)
    idx = np.argsort(g)
    v = np.zeros(n)
    v[idx[:nw]] = 1.0
    return v - h

def _lp_loop(h, deadline, rad0=0.12):
    n = len(h)
    target = n/2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    rad = rad0
    stall = 0
    while time.time() < deadline:
        if stall > 60: break
        scores = np.correlate(h, 1.0-h, mode='full')
        G = _grad_mat(h)
        p = _lp_dir(h, scores, G, rad)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            rad *= 0.7
            if rad < 1e-6: break
            continue
        a, nc5 = _quad_ls(h, p, rad)
        if nc5 < best_c5 - 1e-13:
            h = _project(h + a*p, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad*1.08, 0.35)
        else:
            # Try FW direction
            g = _sm_grad(h, 200)
            d_fw = _fw_dir(h, g)
            dn = np.linalg.norm(d_fw)
            if dn > 1e-12:
                d_fw = d_fw/dn * rad
                a2, nc52 = _quad_ls(h, d_fw, rad)
                if nc52 < best_c5 - 1e-13:
                    h = _project(h + a2*d_fw, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    stall = 0
                    continue
            stall += 1
            rad *= 0.75
            if stall % 20 == 0 and rad < 0.02:
                rad = 0.05
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n)+0.5)/n
    x_new = (np.arange(n_new)+0.5)/n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new/2.0)

def _pair_refine(h, deadline, max_pass=30):
    n = len(h)
    dx = 2.0/n
    target = n/2.0
    h = h.copy()
    best_h, best_c5 = h.copy(), _exact_c5(h)
    for _p in range(max_pass):
        if time.time() > deadline: break
        imp = False
        for i in range(n):
            if time.time() > deadline: break
            if h[i] <= 1e-10 or h[i] >= 1.0-1e-10: continue
            for j in range(i+1, n):
                if h[j] <= 1e-10 or h[j] >= 1.0-1e-10: continue
                d = np.zeros(n)
                d[i] = -1.0; d[j] = 1.0
                a, c5t = _quad_ls(h, d, 0.5)
                if c5t < best_c5 - 1e-14:
                    h = _project(h + a*d, target)
                    if _exact_c5(h) < best_c5:
                        best_c5 = _exact_c5(h)
                        best_h = h.copy()
                        imp = True
        if not imp: break
    return best_h, best_c5

def _thue_morse(n):
    """Thue-Morse sequence of length n as {0,1}."""
    t = np.zeros(n, dtype=int)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Initial solutions
    candidates = []
    if 'initial_h_values' in dir() or True:
        try:
            h_init = np.array(initial_h_values, dtype=float)
            candidates.append(_project(h_init, len(h_init)/2.0))
        except:
            pass
    
    # Thue-Morse initializations
    for n_tm in [64, 96, 112, 128]:
        tm = _thue_morse(n_tm)
        # Smooth it a bit
        h_tm = 0.4 + 0.2*tm  # values in [0.4, 0.6]
        candidates.append(_project(h_tm, n_tm/2.0))
    
    # Sine-based
    for n_s in [64, 96, 112, 128]:
        x = np.arange(n_s)
        for freq in [1, 2, 3]:
            h_s = 0.5 + 0.3*np.sin(2*np.pi*freq*x/n_s)
            candidates.append(_project(h_s, n_s/2.0))
    
    # Random
    for n_r in [64, 112, 128]:
        h_r = rng.random(n_r)
        candidates.append(_project(h_r, n_r/2.0))
    
    # Quick evaluation of all candidates
    scored = []
    for h_c in candidates:
        c5_c = _exact_c5(h_c)
        scored.append((c5_c, h_c))
    scored.sort(key=lambda x: x[0])
    best_h, best_c5 = scored[0][1].copy(), scored[0][0]
    
    # Phase 1: Adam smooth optimization (15%)
    d1 = t0 + budget_s * 0.15
    h_a, c5_a = _adam(best_h, d1, m_start=20, m_end=5000)
    if c5_a < best_c5:
        best_c5, best_h = c5_a, h_a
    
    # Phase 2: LP local search (25%)
    d2 = t0 + budget_s * 0.40
    h_l, c5_l = _lp_loop(best_h, d2, rad0=0.12)
    if c5_l < best_c5:
        best_c5, best_h = c5_l, h_l
    
    # Phase 3: Multi-resolution (25%)
    d3 = t0 + budget_s * 0.65
    bn = len(best_h)
    for dn in [-16,-12,-8,-6,-4,-3,-2,-1,1,2,3,4,6,8,12,16,20,24,32,-32]:
        if time.time() > d3: break
        n_try = bn + dn
        if n_try < 32 or n_try > 200: continue
        if n_try % 2 != 0: continue  # need even for sum = n/2
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s*0.012)
        h_r, c5_r = _lp_loop(h_r, sub_dl, rad
