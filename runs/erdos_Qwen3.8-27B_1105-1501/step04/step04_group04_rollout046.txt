def _grad_lse(h, F, beta):
    """Gradient of log-sum-exp smoothed max of F."""
    n = len(h)
    dx = 2.0 / n
    fmax = np.max(F)
    shifted = beta * (F - fmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    # Vectorized gradient computation
    # For each lag r, d = r - (n-1)
    # Gradient contribution to g:
    #   For d >= 0: g[:n-d] += w[r]*dx*(1-h[d:]), g[d:] += w[r]*dx*(-h[:n-d])
    #   For d < 0 (let dd=-d): g[dd:] += w[r]*dx*(1-h[:-dd]), g[:n-dd] += w[r]*dx*(-h[dd:])
    for r in range(len(F)):
        wr = w[r]
        if wr < 1e-10:
            continue
        d = r - (n - 1)
        if d >= 0:
            g[:n-d] += wr * dx * (1.0 - h[d:])
            g[d:] += wr * dx * (-h[:n-d])
        else:
            dd = -d
            g[dd:] += wr * dx * (1.0 - h[:-dd])
            g[:n-dd] += wr * dx * (-h[dd:])
    g -= np.mean(g)
    return g

def _smoothed_descent(h, deadline, beta_start=5.0, beta_end=5000.0, n_beta=20):
    n = len(h)
    target = n / 2.0
    h = _project_box_sum(h, target)

    cur_c5 = best_c5
    F = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    beta_list = np.geomspace(beta_start, beta_end, n_beta)
    step_size = 0.05
    for beta in beta_list:
        stalled = 0
        for it in range(50000):
            if time.time() > deadline:
                break
            if stalled > 100:
                break
            g = _grad_lse(h, F, beta)
            gn = np.max(np.abs(g))
            if gn < 1e-15:
                break
            g_norm = g / gn
            # Try different step sizes
            best_step = 0.0
            best_trial = cur_c5
            for factor in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                trial = _project_box_sum(h - factor * step_size * g_norm, target)
                tc5 = _exact_c5(trial)
                if tc5 < best_trial - 1e-14:
                    best_trial = tc5
                    best_step = factor
            if best_step > 0:
                h = _project_box_sum(h - best_step * step_size * g_norm, target)
                F = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stalled = 0
                step_size = min(step_size * 1.02, 0.3)
            else:
                stalled += 1
                step_size *= 0.8
                if step_size < 1e-6:
                    break
    return best_h, best_c5

    target = n / 2.0

    best_h = h.copy()

        if no_improve > 8000:
            break
        fmax = np.max(F)

                g[:n - dd] += dx * (-h[dd:])

            no_improve += 1
            continue

        interior = (h > 0.01) & (h < 0.99) & descent_mask
        if np.sum(interior) > 0:
            candidates = np.where(interior)[0]
        else:
            candidates = np.where(descent_mask)[0]

        # Find compensating coordinate j (opposite gradient or random)

        if len(j_candidates) == 0:
            no_improve += 1
            continue
        j = j_candidates[rng.integers(len(j_candidates))]
        delta = rng.uniform(0.01, 0.08)
        new_i = h[i] + delta
        new_j = h[j] - delta
        if new_i > 1.0 or new_j < 0.0:
            no_improve += 1

        h[j] = new_j

                best_h = h.copy()
            no_improve = 0
        else:

            h[j] = h[j] + delta
            no_improve += 1

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, n_new / 2.0)

def _anneal_descent(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project_box_sum(h, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    temp = 0.02
    temp_min = 1e-8
    step_size = 0.04
    while time.time() < deadline:
        F = np.correlate(h, 1.0 - h, mode='full') * dx
        fmax = np.max(F)
        tol = max(1e-8, 0.005 * fmax)
        active = np.where(F >= fmax - tol)[0]
        if len(active) > 50:
            active = np.argsort(F)[-50:]
        g = np.zeros(n)
        for r in active:

            if d >= 0:

                g[d:] += dx * (-h[:n - d])
            else:
                dd = -d
                g[dd:] += dx * (1.0 - h[:-dd])
                g[:n - dd] += dx * (-h[dd:])
        g /= len(active)
        g -= np.mean(g)
        gn = float(np.max(np.abs(g)))
        if gn < 1e-16:
            break
        direction = -g / gn
        noise = rng.normal(0, 0.3, n)

        delta_c = c5_trial - cur_c5

            cur_c5 = c5_trial
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()

def _binary_mix(h, rng):
    n = len(h)
    target = n / 2.0

    perm = rng.permutation(n)
    binary = np.zeros(n)

    mixed = alpha * h + (1 - alpha) * binary
    return _project_box_sum(mixed, target)

    best_h = _project_box_sum(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        # Revert to last valid
        best_h = _project_box_sum(best_h, best_n / 2.0)
        final_c5 = _exact_c5(best_h)
    best_c5 = final_c5

    # Safety
    if not np.all(best_h >= -1e-10) or not np.all(best_h <= 1.0 + 1e-10):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_c5 = _exact_c5(best_h)

    return best_h, best_c5, best_n
