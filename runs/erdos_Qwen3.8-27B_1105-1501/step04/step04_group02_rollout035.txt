import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _raw_scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _penalty_and_grad(h, theta_raw):
    """Compute squared-hinge penalty and its gradient for raw scores exceeding theta_raw."""
    n = len(h)
    raw = _raw_scores(h)
    excess = np.maximum(raw - theta_raw, 0.0)
    penalty = float(np.sum(excess * excess))
    
    # Gradient
    grad = np.zeros(n)
    active = excess > 1e-14
    if not np.any(active):
        return penalty, grad
    
    for r in np.where(active)[0]:
        w = 2.0 * excess[r]
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            grad[:m] += w * (1.0 - h[d:])
            grad[d:] += w * (-h[:m])
        else:
            s = -d
            m = n - s
            grad[s:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[s:])
    
    return penalty, grad

def _slsqp_target(h, target_c5, max_iter=100, ftol=1e-10):
    """Use SLSQP to minimize penalty for scores exceeding target."""
    n = len(h)
    dx = 2.0 / n
    theta_raw = target_c5 / dx
    target_sum = n / 2.0
    h0 = h.copy()
    
    def obj(x):
        p, _ = _penalty_and_grad(x, theta_raw)
        return p
    
    def obj_grad(x):
        _, g = _penalty_and_grad(x, theta_raw)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        h_new = res.x
        h_new = _project(h_new, target_sum)
        c5_new = _exact_c5(h_new)
        if c5_new <= max(target_c5 * 1.001, _exact_c5(h) + 0.001):
            return h_new, c5_new
    except Exception:
        pass
    return h, _exact_c5(h)

def _lp_direction(h, scores, radius, top_k=None):
    """Find LP direction to decrease active scores."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            mk = n - d
            G[r, :mk] += 1.0
            G[r, :mk] -= h[d:]
            G[r, d:] -= h[:mk]
        else:
            s = -d
            mk = n - s
            G[r, s:] += 1.0
            G[r, s:] -= h[:mk]
            G[r, :mk] -= h[s:]
    
    # Select top-k active rows
    if top_k is not None and top_k < m:
        idx = np.argsort(scores)[::-1][:top_k]
    else:
        idx = np.arange(m)
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G[idx]
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max):
    """Golden section line search."""
    n = len(h)
    best_t = 0.0
    best_val = c5_cur
    n_grid = 25
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    
    lo_t = ts[max(0, idx - 2)]
    hi_t = ts[min(n_grid - 1, idx + 2)]
    if hi_t <= lo_t:
        lo_t, hi_t = 0.0, alpha_max
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    
    f1, f2 = _f(c1), _f(c2)
    for _ in range(30):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_local(h, deadline, radius_init=0.15, top_k=None):
    """LP-based local search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    
    while time.time() < deadline and stall < 60:
        dx = 2.0 / n
        scores = _raw_scores(h) * dx
        p = _lp_direction(h, scores, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.7
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.08
    
    return best_h, best_c5

                m = n - s
                grad[s:] += wr * (1.0 - h[:m])
                grad[:m] += wr * (-h[s:])

                if c5_check < best_c5 - 1e-12:
                    best_c5 = c5_check
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()
        if not improved:
            break
    return best_h, best_c5

def _pair_coord(h, i, j, deadline=None):
    n = len(h)
    m = 2 * n - 1
    raw = _raw_scores(h)
    B_i = np.zeros(m)
    B_j = np.zeros(m)
    for r in range(m):
        d = r - (n - 1)
        for var, Bv in [(i, B_i), (j, B_j)]:
            if d >= 0:
                if var < n - d:
                    Bv[r] += (1.0 - h[var + d])
                if var >= d:
                    Bv[r] += (-h[var - d])
            else:
                s = -d
                if var >= s:
                    Bv[r] += (1.0 - h[var - s])
                if var < n - s:
                    Bv[r] += (-h[var + s])
    C = raw - B_i * h[i] - B_j * h[j]
    lo_x, hi_x = 0.0, 1.0
    for _ in range(50):
        mid = (lo_x + hi_x) / 2.0
        lo_y = max(0.0, min(1.0, np.max((C + B_i * 0 / 1) - B_j * mid) if False else 0.0))
        # For given h[i]=mid, find best h[j]
        # max_r (C_r + B_j[r]*h[j]) minimized over h[j] in [0,1]
        # = max( max_r(C_r), max_r(C_r + B_j[r]) ) if we take h[j]=0 or 1
        # Actually need proper 1D minimax
        # For fixed x1=mid: minimize over x2 of max_r(C_r + B_j[r]*x2)
        # Binary search on value
        lo_v = float(np.max(C))
        hi_v = float(np.max(C + B_j))
        for _b in range(40):
            mid_v = (lo_v + hi_v) / 2.0
            # Check feasible: need x2 in [0,1] s.t. C_r + B_j[r]*x2 <= mid_v for all r
            lbs = 0.0
            ubs = 1.0
            feasible = True
            for r in range(m):
                if B_j[r] > 1e-14:
                    ub = (mid_v - C[r]) / B_j[r]
                    if ub < ubs:
                        ubs = ub
                elif B_j[r] < -1e-14:
                    lb = (mid_v - C[r]) / B_j[r]
                    if lb > lbs:
                        lbs = lb
                else:
                    if C[r] > mid_v:
                        feasible = False
                        break
            if feasible and lbs <= ubs:
                hi_v = mid_v
            else:
                lo_v = mid_v
        if hi_v < _raw_scores(h).max() - 1e-10:
            lo_x = mid
        else:
            hi_x = mid
    return None

def _slsqp_target(h, target_c5, max_iter=100, ftol=1e-9):
    n = len(h)
    dx = 2.0 / n
    theta_raw = target_c5 / dx
    target_sum = n / 2.0
    h0 = h.copy()

    def _obj(x):
        raw = _raw_scores(x)
        excess = np.maximum(raw - theta_raw, 0.0)
        return float(np.sum(excess * excess))

    def _obj_grad(x):
        raw = _raw_scores(x)
        excess = np.maximum(raw - theta_raw, 0.0)
        grad = np.zeros(len(x))
        for r in np.where(excess > 1e-14)[0]:
            w = 2.0 * excess[r]
            d = r - (n - 1)
            if d >= 0:
                mk = n - d
                grad[:mk] += w * (1.0 - x[d:])
                grad[d:] += w * (-x[:mk])
            else:
                s = -d
                mk = n - s
                grad[s:] += w * (1.0 - x[:mk])
                grad[:mk] += w * (-x[s:])
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        h_new = _project(res.x, target_sum)
        c5_new = _exact_c5(h_new)
        cur_c5 = _exact_c5(h)
        if c5_new < cur_c5:
            return h_new, c5_new
    except Exception:
        pass
    return h, _exact_c5(h)

def _slsqp_drive(h, deadline):
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    target = cur_c5 - 0.0001
    stall = 0
    while time.time() < deadline and stall < 25:
        if best_c5 <= target + 1e-10:
            target = best_c5 - max(0.00004, (best_c5 - 0.3800) * 0.25)
            stall = 0
            continue
        h_new, c5_new = _slsqp_target(h, target, max_iter=60, ftol=1e-8)
        if c5_new < best_c5 - 1e-11:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            target += max(0.00005, (cur_c5 - best_c5) * 0.3)
            stall += 1
    return best_h, best_c5

def _lp_direction(h, scores, radius, top_k=None):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            mk = n - d
            G[r, :mk] += 1.0
            G[r, :mk] -= h[d:]
            G[r, d:] -= h[:mk]
        else:
            s = -d
            mk = n - s
            G[r, s:] += 1.0
            G[r, s:] -= h[:mk]
            G[r, :mk] -= h[s:]
    if top_k is not None and top_k < m:
        idx = np.argsort(scores)[::-1][:top_k]
    else:
        idx = np.arange(m)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(idx), n + 1))
    A_ub[:, :n] = G[idx]
    A_ub[:, n] = -1.0
    b_ub = -scores[idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search(h, p, c5_cur, alpha_max):
    best_t = 0.0
    best_val = c5_cur
    n_grid = 25
    ts = np.linspace(0.0, alpha_max, n_grid)
    vals = np.full(n_grid, c5_cur + 1.0)
    for i, t in enumerate(ts):
        h_t = h + t * p
        if h_t.min() >= -1e-12 and h_t.max() <= 1.0 + 1e-12:
            vals[i] = _exact_c5(h_t)
    idx = int(np.argmin(vals))
    best_t, best_val = ts[idx], vals[idx]
    lo_t = ts[max(0, idx - 2)]
    hi_t = ts[min(n_grid - 1, idx + 2)]
    if hi_t <= lo_t:
        lo_t, hi_t = 0.0, alpha_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    f_cur = c5_cur + 1.0
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0
        return _exact_c5(h_t)
    f1, f2 = _f(c1), _f(c2)
    for _ in range(28):
        if hi_t - lo_t < 1e-14:
            break
        if f1 < f2:
            hi_t, c2, f2 = c2, c1, f1
            c1 = hi_t - gr * (hi_t - lo_t)
            f1 = _f(c1)
        else:
            lo_t, c1, f1 = c1, c2, f2
            c2 = lo_t + gr * (hi_t - lo_t)
            f2 = _f(c2)
    opt_t = (lo_t + hi_t) / 2.0
    f_opt = _f(opt_t)
    if f_opt < best_val:
        best_t, best_val = opt_t, f_opt
    return best_t, best_val

def _lp_local(h, deadline, radius_init=0.15, top_k=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline and stall < 60:
        dx = 2.0 / n
        scores = _raw_scores(h) * dx
        p = _lp_direction(h, scores, radius, top_k=top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.6
            if radius < 1e-7:
                break
            continue
        alpha_max = 1.0
        for j in range(n):
            if p[j] > 1e-14:
                alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
            elif p[j] < -1e-14:
                alpha_max = min(alpha_max, h[j] / (-p[j]))
        alpha_max = max(alpha_max, 1e-10)
        alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            radius = min(radius * 1.1, 0.4)
        else:
            stall += 1
            radius *= 0.65
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.08
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=30):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        rng = np.random.default_rng(_pass * 104729 + 7)
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        if len(interior) > 16:
            idx_j = rng.choice(interior, size=8, replace=False)
            idx_k = rng.choice(interior, size=8, replace=False)
        else:
            idx_j = interior
            idx_k = interior
        pair_list = [(int(min(a, b)), int(max(a, b)))
                     for a in idx_j for b in idx_k if a != b]
        pair_list = list(set(pair_list))
        rng.shuffle(pair_list)
        for (j, k) in pair_list:
            if time.time() > deadline:
                break
            if h[j] <= 1e-6 or h[j] >= 1.0 - 1e-6 or h[k] <= 1e-6 or h[k] >= 1.0 - 1e-6:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if dj > 0:
                    am = (1.0 - h[j]) / dj
                else:
                    am = h[j] / (-dj)
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-12:
                    continue
                lo_d, hi_d = 0.0, am
                best_frac = 0.0
                best_val = best_c5
                for _gs in range(14):
                    if hi_d - lo_d < am * 1e-7:
                        break
                    m1 = hi_d - 0.618 * (hi_d - lo_d)
                    m2 = lo_d + 0.618 * (hi_d - lo_d)
                    h1 = h.copy()
                    h1[j] += dj * m1
                    h1[k] += dk * m1
                    h2 = h.copy()
                    h2[j] += dj * m2
                    h2[k] += dk * m2
                    f1 = _exact_c5(h1)
                    f2 = _exact_c5(h2)
                    if f1 < f2:
                        hi_d = m2
                    else:
                        lo_d = m1
                    if f1 < best_val:
                        best_val = f1
                        best_frac = m1
                    if f2 < best_val:
                        best_val = f2
                        best_frac = m2
                if best_frac > 0 and best_val < best_c5 - 1e-12:
                    h[j] += dj * best_frac
                    h[k] += dk * best_frac
                    h = _project(h, target)
                    best_c5 = _exact_c5(h)
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gradient_descent(h, deadline, tau=15.0, lr=0.02, beta=0.88):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    stall = 0
    while time.time() < deadline and stall < 50:
        dx = 2.0 / n
        scores = _raw_scores(h) * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        grad = np.zeros(n)
        for r in range(len(scores)):
            if w[r] < 1e-10:
                continue
            wr = w[r]
            d = r - (n - 1)
            if d >= 0:
                mk = n - d
                grad[:mk] += wr * (1.0 - h[d:])
                grad[d:] += wr * (-h[:mk])
            else:
                s = -d
                mk = n - s
                grad[s:] += wr * (1.0 - h[:mk])
                grad[:mk] += wr * (-h[s:])
        grad *= dx
        mom = beta * mom + (1.0 - beta) * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target_sum)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
        else:
            stall += 1
            lr *= 0.92
    return best_h, best_c5

def _perturb_refine(h_best, deadline, budget_s, seed=42):
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = _exact_c5(h_best)
    scale = 0.03
    time_per = budget_s * 0.025
    attempt = 0
    while time.time() < deadline and attempt < 80:
        attempt += 1
        h_pert = best_h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        sub_dl = min(deadline, time.time() + time_per)
        h_try, c5_try = _lp_local(h_pert, sub_dl, radius_init=0.1, top_k=12)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.18, scale * 1.25)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Gradient warmup (5%)
    d1 = t0 + budget_s * 0.05
    h, c5 = _gradient_descent(h, d1, tau=12.0, lr=0.018, beta=0.88)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP target-driven (22%)
    d2 = t0 + budget_s * 0.27
    h, c5 = _slsqp_drive(best_h, d2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: LP local search (13%)
    d3 = t0 + budget_s * 0.40
    h, c5 = _lp_local(best_h, d3, radius_init=0.15, top_k=15)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Coordinate descent (15%)
    d4 = t0 + budget_s * 0.55
    h, c5 = _coord_descent(best_h, d4, max_passes=150)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Multi-resolution (15%)
    d5 = t0 + budget_s * 0.70
    offsets = [-2, -1, 1, 2, -4, 4, -6, 6, -8, 8, 10, -10]
    time_per_mr = budget_s * 0.012
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 150:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + time_per_mr)
        h_r, c5_r = _lp_local(h_r, sub_dl, radius_init=0.12, top_k=12)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: SLSQP again (10%)
    d6 = t0 + budget_s * 0.80
    h, c5 = _slsqp_drive(best_h, d6)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 7: Perturbation + refine (10%)
    d7 = t0 + budget_s * 0.90
    best_h, best_c5 = _perturb_refine(best_h, d7, budget_s, seed=seed + 200)

    # Phase 8: Final polishing (7%)
    d8 = deadline
    h, c5 = _lp_local(best_h, min(d8, time.time() + budget_s * 0.02), radius_init=0.05, top_k=10)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d8:
        h, c5 = _coord_descent(best_h, min(d8, time.time() + budget_s * 0.02), max_passes=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    if time.time() < d8:
        h, c5 = _pair_refine(best_h, d8, max_passes=20)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
