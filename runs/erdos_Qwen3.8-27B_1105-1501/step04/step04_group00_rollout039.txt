```python
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(0, np.minimum(1, v - mid))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    # Fix rounding
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0, 1)
    return x

def _active_grad(h, lag_indices, weights=None):
    """Compute gradient of weighted sum of scores at given lags."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if weights is None:
        weights = np.ones(len(lag_indices))
    for i, r in enumerate(lag_indices):
        d = r - (n - 1)
        w = weights[i] * dx
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w * (1.0 - h[d:])
                grad[d:] += -w * h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w * (1.0 - h[g:])
                grad[g:] += -w * h[:m]
    return grad

def _adam_step(h, grad, momentum,adam_state, lr, target):
    """Adam step with projection."""
    n = len(h)
    m = momentum['m']
    v = momentum['v']
    t = momentum['t']
    b1, b2 = 0.9, 0.999
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad**2
    t += 1
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    h_new = _project(h_new, target)
    momentum['m'] = m
    momentum['v'] = v
    momentum['t'] = t
    return h_new

def _adam_optimize(h, deadline, target, init_lr=0.05, top_k=10, patience=200):
    """Optimize using Adam with active set gradient."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    momentum = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
    stall = 0
    lr = init_lr
    while time.time() < deadline:
        scores = _scores(h)
        # Select top-k lags
        if top_k < len(scores):
            top_idx = np.argsort(scores)[-top_k:]
        else:
            top_idx = np.arange(len(scores))
        top_scores = scores[top_idx]
        max_s = np.max(top_scores)
        # Softmax weights
        beta = 50.0
        w = np.exp(beta * (top_scores - max_s))
        w /= np.sum(w)
        # Compute gradient
        grad = _active_grad(h, top_idx, w)
        # Remove mean (project onto sum=0)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        # Adam step
        h_new = _adam_step(h, grad, momentum, lr, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Increase lr slightly
            lr = min(lr * 1.01, 0.2)
        else:
            h = h_new  # Keep the step for exploration
            stall += 1
            lr *= 0.95
            if stall > patience:
                break
    return best_h, best_c5
```

```python
def _lp_direction_fast(h, scores, radius, top_k=10):
    """Find best LP direction using only top-k constraints."""
    n = len(h)
    n_lags = len(scores)
    if top_k < n_lags:
        idx = np.argsort(scores)[-top_k:]
        scores_k = scores[idx]
    else:
        idx = np.arange(n_lags)
        scores_k = scores
    k = len(idx)
    
    # Build A_ub: for each constraint, the gradient row
    A = np.zeros((k, n))
    for i, r in enumerate(idx):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                A[i, :m] = (1.0 - h[d:])
                A[i, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                A[i, :m] = (1.0 - h[g:])
                A[i, g:] += -h[:m]
    A *= 2.0 / n  # dx
    
    # LP: minimize t subject to A @ p <= -scores_k + t*max(scores_k), ||p||_inf <= radius, sum(p) = 0
    # Actually, the constraint is scores_k + A @ p <= t (new max)
    # Minimize t
    # Variable: (p_1, ..., p_n, t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ [p; t] <= b_ub
    # A @ p - 1*t <= -scores_k  =>  scores_k + A@p <= t
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -scores_k
    
    # sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -radius <= p_i <= radius, also 0 <= p_i <= 1-h_i and -h_i <= p_i <= 1-h_i
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except:
        pass
    return None, None
```

```python
                if m2 > 0:
                    g[:m2] += wi * (1.0 - h[gg:])
                    g[gg:] += -wi * h[:m2]
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14: break
        g /= gn
        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = _project(h - lr * mh / (np.sqrt(vh) + 1e-8), n / 2.0)
        c5_new = _exact_c5(h)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > patience: break
    return best_h, best_c5

def _line_search(h, p, target):
    n = len(h)
    amax = 1.0
    for j in range(n):
        if p[j] > 1e-14: amax = min(amax, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14: amax = min(amax, h[j] / (-p[j]))
    amax = min(amax, 1.0)
    if amax < 1e-14: return 0.0, _exact_c5(h)
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi_ = 0.0, amax
    c1 = hi_ - gr * (hi_ - lo)
    c2 = lo + gr * (hi_ - lo)
    f1 = _exact_c5(_project(h + c1 * p, target))
    f2 = _exact_c5(_project(h + c2 * p, target))
    for _ in range(20):
        if hi_ - lo < 1e-12: break
        if f1 < f2:
            hi_ = c2; f2 = f1; c2 = c1
            c1 = hi_ - gr * (hi_ - lo)
            f1 = _exact_c5(_project(h + c1 * p, target))
        else:
            lo = c1; f1 = f2; c1 = c2
            c2 = lo + gr * (hi_ - lo)
            f2 = _exact_c5(_project(h + c2 * p, target))
    a = (lo + hi_) / 2.0
    return a, _exact_c5(_project(h + a * p, target))

def _lp_opt(h, deadline, top_k_list=None, radius_init=0.15, max_iter=5000):
    if top_k_list is None:
        top_k_list = [6, None, 4, 8, 5]
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    radius = radius_init
    stall = 0
    ti = 0
    ntl = len(top_k_list)
    for it in range(max_iter):
        if time.time() > deadline: break
        if stall > 80: break
        tk = top_k_list[ti % ntl]
        sc = _scores(h)
        nl = len(sc)
        if tk is not None and tk < nl:
            idx = np.argsort(sc)[-tk:]
        else:
            idx = np.arange(nl)
        sk = sc[idx]
        k = len(idx)
        # Build gradient rows
        A = np.zeros((k, n))
        for i, r in enumerate(idx):
            d = int(r) - (n - 1)
            dy = 2.0 / n
            if d >= 0:
                m2 = n - d
                if m2 > 0:
                    A[i, :m2] += dy * (1.0 - h[d:])
                    A[i, d:] += -dy * h[:m2]
            else:
                gg = -d
                m2 = n - gg
                if m2 > 0:
                    A[i, :m2] += dy * (1.0 - h[gg:])
                    A[i, gg:] += -dy * h[:m2]
        c = np.zeros(n + 1); c[-1] = 1.0
        A_ub = np.zeros((k, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = -sk
        A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((0.0, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
                pnorm = np.max(np.abs(p))
                if pnorm > 1e-13:
                    a, c5_new = _line_search(h, p, target)
                    if c5_new < cur_c5 - 1e-13:
                        h = _project(h + a * p, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = 0
                        radius = min(radius * 1.03, 0.4)
                    else:
                        stall += 1
                        radius *= 0.88
                        ti += 1
                else:
                    stall += 1
                    radius *= 0.85
                    ti += 1
            else:
                stall += 1
                radius *= 0.88
                ti += 1
        except Exception:
            stall += 1
            radius *= 0.85
        if radius < 1e-7:
            radius = 0.05
            stall = 0
    return best_h, best_c5

def _binary_sa(n, deadline, rng, n_iters=30000):
    target = n // 2
    h = np.zeros(n)
    pos = rng.permutation(n)
    h[pos[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.03
    T_min = 0.0003
    cool = (T_min / T) ** (1.0 / max(n_iters, 1))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    for it in range(n_iters):
        if time.time() > deadline: break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0; h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0; h[j] = 0.0
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
        T *= cool
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=4):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _p in range(max_passes):
        if time.time() > deadline: break
        improved = False
        for j in range(n):
            if time.time() > deadline: break
            if h[j] < 1e-9 or h[j] > 1.0 - 1e-9: continue
            step = max(1, n // 40)
            for k in range(j + step, n, step):
                if h[k] < 1e-9 or h[k] > 1.0 - 1e-9: continue
                for sign in [1, -1]:
                    if sign > 0:
                        amax = min(1.0 - h[j], h[k])
                    else:
                        amax = min(h[j], 1.0 - h[k])
                    if amax < 1e-11: continue
                    lo_a, hi_a = 0.0, amax
                    for _ in range(7):
                        m1 = lo_a + (hi_a - lo_a) / 3.0
                        m2 = hi_a - (hi_a - lo_a) / 3.0
                        h1 = h.copy(); h1[j] += sign*m1; h1[k] -= sign*m1
                        f1 = _exact_c5(h1)
                        h2 = h.copy(); h2[j] += sign*m2; h2[k] -= sign*m2
                        f2 = _exact_c5(h2)
                        if f1 < f2: hi_a = m2
                        else: lo_a = m1
                    a_opt = (lo_a + hi_a) / 2.0
                    h_try = h.copy()
                    h_try[j] += sign * a_opt
                    h_try[k] -= sign * a_opt
                    h_try = np.clip(h_try, 0, 1)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True
        if not improved: break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    # Load initial values
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_start = _project(h0, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)
    best_n = n0

    def update_candidate(h_cand, c5_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_cand

    # Phase 1: Quick Adam warmup from initial (5% budget)
    dl1 = t0 + budget_s * 0.05
    h1, c5_1 = _adam_opt(h_start, dl1, lr=0.03, top_k=8, patience=100)
    update_candidate(h1, c5_1, n0)

    # Phase 2: LP from initial (15% budget)
    dl2 = t0 + budget_s * 0.20
    h2, c5_2 = _lp_opt(h_start, dl2, top_k_list=[6, None, 4, 8, 5], radius_init=0.2, max_iter=5000)
    update_candidate(h2, c5_2, n0)
    # LP from best so far
    if time.time() < dl2:
        h2b, c5_2b = _lp_opt(best_h, dl2, top_k_list=[5, 4, 6, None, 3], radius_init=0.1, max_iter=3000)
        update_candidate(h2b, c5_2b, best_n)

    # Phase 3: Multi-resolution diverse search (40% budget)
    dl3 = t0 + budget_s * 0.60
    alphas = [0.618034, 0.414214, 0.707107, 0.381966, 0.5, 0.682328, 0.347296]
    resolutions = [96, 112, 128, 128, 128, 128]
    for ri, n_try in enumerate(resolutions):
        if time.time() > dl3: break
        for s in range(3):
            if time.time() > dl3: break
            if s == 0:
                h_init = _resample(best_h, n_try)
            elif s == 1:
                alpha = alphas[(ri + s) % len(alphas)]
                h_init = _quasi_init(n_try, alpha, rng, noise=0.04)
            else:
                h_init = _quasi_init(n_try, alphas[ri % len(alphas)], rng, noise=0.12)
                h_init += rng.normal(0, 0.03, n_try)
                h_init = _project(h_init, n_try / 2.0)
            # Quick Adam
            sub_dl = min(dl3, time.time() + budget_s * 0.015)
            h_a, c5_a = _adam_opt(h_init, sub_dl, lr=0.03, top_k=6, patience=80)
            # LP polish
            sub_dl2 = min(dl3, time.time() + budget_s * 0.03)
            h_l, c5_l = _lp_opt(h_a, sub_dl2, top_k_list=[5, 4, 6, 3, None], radius_init=0.12, max_iter=3000)
            update_candidate(h_l, c5_l, n_try)
            if c5_a < c5_l:
                update_candidate(h_a, c5_a, n_try)

    # Phase 4: LP deep optimization on best (20% budget)
    if time.time() < t0 + budget_s * 0.80:
        dl4 = t0 + budget_s * 0.80
        h4, c5_4 = _lp_opt(best_h, dl4, top_k_list=[4, 5, 6, 3, 8, None], radius_init=0.08, max_iter=8000)
        update_candidate(h4, c5_4, best_n)
        # Try from a perturbed version
        if time.time() < dl4:
            h_pert = _project(best_h + rng.normal(0, 0.02, best_n), best_n / 2.0)
            sub_dl = min(dl4, time.time() + budget_s * 0.04)
            h4b, c5_4b = _lp_opt(h_pert, sub_dl, top_k_list=[5, 4, 6, 3], radius_init=0.1, max_iter=3000)
            update_candidate(h4b, c5_4b, best_n)

    # Phase 5: Binary SA + LP (10% budget)
    if time.time() < t0 + budget_s * 0.90:
        dl5 = t0 + budget_s * 0.90
        for n_bin in [96, 128]:
            if time.time() > dl5: break
            sub_dl = min(dl5, time.time() + budget_s * 0.02)
            h_b, c5_b = _binary_sa(n_bin, sub_dl, rng, n_iters=20000)
            # LP polish on binary result
            sub_dl2 = min(dl5, time.time() + budget_s * 0.02)
            h_bl, c5_bl = _lp_opt(h_b, sub_dl2, top_k_list=[5, 4, 6], radius_init=0.15, max_iter=2000)
            update_candidate(h_bl, c5_bl, n_bin)
            if c5_b < c5_bl:
                update_candidate(h_b, c5_b, n_bin)

    # Phase 6: Perturbation restarts (5% budget)
    dl6 = t0 + budget_s * 0.95
    scale = 0.02
    for attempt in range(200):
        if time.time() > dl6: break
        h_p = _project(best_h + rng.normal(0, scale, best_n), best_n / 2.0)
        sub_dl = min(dl6, time.time() + budget_s * 0.008)
        h_t, c5_t = _lp_opt(h_p, sub_dl, top_k_list=[4, 5, 3, 6], radius_init=0.05, max_iter=500)
        if c5_t < best_c5 - 1e-13:
            update_candidate(h_t, c5_t, best_n)
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.15)

    # Phase 7: Pair polish (3% budget)
    if time.time() < t0 + budget_s * 0.98:
        dl7 = t0 + budget_s * 0.98
        hp, c5p = _pair_polish(best_h, dl7, max_passes=4)
        update_candidate(hp, c5p, best_n)

    # Phase 8: Final tiny LP (2% budget)
    if time.time() < t0 + budget_s * 0.995:
        h_fin, c5_fin = _lp_opt(best_h, t0 + budget_s * 0.995,
                                top_k_list=[3, 4, 5], radius_init=0.01, max_iter=1000)
        update_candidate(h_fin, c5_fin, best_n)

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
```
