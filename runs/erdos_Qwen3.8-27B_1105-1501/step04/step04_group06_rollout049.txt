def project_mass(v, target):
    n = len(v)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(x)
        else:
            idx = np.argsort(-x)
        for i in idx:
            if diff > 0:
                a = min(1.0 - x[i], diff)
            else:
                a = min(x[i], -diff)
                a = -a
            x[i] += a
            diff -= a
            if abs(diff) <= 1e-12:
                break
    return x


def exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def corr_grad(h, n):
    nc = 2 * n - 1
    G = np.zeros((nc, n))
    j = np.arange(n)
    for r in range(nc):
        s = r - (n - 1)
        pd = np.zeros(n)
        gs = j - s
        m1 = (gs >= 0) & (gs < n)
        pd[m1] = 1.0 - h[gs[m1]]
        ps = j + s
        m2 = (ps >= 0) & (ps < n)
        pd[m2] -= h[ps[m2]]
        G[r, :] = pd
    return G

    return z[n]

    g = np.zeros(n + 1)

    J = np.zeros((m, n + 1))

        J[i, :n] = -G[active[i], :]
        J[i, n] = 1.0
    return J
