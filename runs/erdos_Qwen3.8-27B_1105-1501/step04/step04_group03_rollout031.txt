import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v to [0,1]^n with sum = target using bisection on clip."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - x)[::-1]
        else:
            idx = np.argsort(x)
        for i in range(n):
            if residual > 0:
                room = 1.0 - x[idx[i]]
                add = min(room, residual)
                x[idx[i]] += add
                residual -= add
            else:
                room = x[idx[i]]
                sub = min(room, -residual)
                x[idx[i]] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return np.clip(x, 0.0, 1.0)

            if m > 0:
                # S = sum_{i=0}^{m-1} h[i+d]*(1-h[i])

                #   from h[i+d] term: j = i+d, so i = j-d, contributes (1-h[i]) = (1-h[j-d]) for j in [d, n-1]
                #   from (1-h[i]) term: j = i, contributes -h[i+d] = -h[j+d] for j in [0, m-1]
                G[r, d:] += (1.0 - h[:m])       # j in [d, n-1], i=j-d in [0, m-1]

                G[r, q:] += -h[:m]                # j in [q, n-1], j-q in [0, m-1]

        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])

    for j in range(n):
        if p[j] > 1e-14:
            alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
        elif p[j] < -1e-14:
            alpha_max = min(alpha_max, h[j] / (-p[j]))
    alpha_max = max(alpha_max, 1e-12)
    best_a = 0.0
    best_c5 = cur_c5
    n_coarse = 25
    for i in range(1, n_coarse + 1):
        a = alpha_max * i / (n_coarse + 1)
        h_t = project(h + a * p, target)
        c5_t = exact_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_a = a
    if best_a == 0.0:
        return 0.0, cur_c5, h
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo_a = max(0.0, best_a * 0.5)
    hi_a = min(alpha_max, best_a * 1.6)
    f_lo = exact_c5(project(h + lo_a * p, target))
    f_hi = exact_c5(project(h + hi_a * p, target))
    c1 = hi_a - gr * (hi_a - lo_a)
    c2 = lo_a + gr * (hi_a - lo_a)
    f1 = exact_c5(project(h + c1 * p, target))
    f2 = exact_c5(project(h + c2 * p, target))
    for _ in range(35):
        if hi_a - lo_a < max(1e-14, alpha_max * 1e-13):
            break
        if f1 < f2:
            hi_a, c2, f2 = c2, c1, f1
            c1 = hi_a - gr * (hi_a - lo_a)
            f1 = exact_c5(project(h + c1 * p, target))
        else:
            lo_a, c1, f1 = c1, c2, f2
            c2 = lo_a + gr * (hi_a - lo_a)
            f2 = exact_c5(project(h + c2 * p, target))
    opt_a = (lo_a + hi_a) / 2.0
    opt_c5 = exact_c5(project(h + opt_a * p, target))
    if opt_c5 < best_c5:
        best_a, best_c5 = opt_a, opt_c5
    return best_a, best_c5, project(h + best_a * p, target)

def soft_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    scores = raw * dx
    max_s = np.max(scores)
    exp_s = np.exp(beta * (scores - max_s))
    Z = np.sum(exp_s)
    w = exp_s / Z
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        d = k - center
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += wk * (1.0 - h[:m])
                grad[:m] += -wk * h[d:d + m]
        else:
            q = -d
            m = n - q
            if m > 0:
                grad[:m] += wk * (1.0 - h[q:q + m])
                grad[q:] += -wk * h[:m]
    return grad

def grad_optimize(h, beta, deadline, lr_init=0.005):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        grad = soft_grad(h, beta)
        g = grad - np.mean(grad)
        if np.max(np.abs(g)) < 1e-14:
            break
        best_a = 0.0
        best_val = cur_c5
        a = lr
        for _ in range(20):
            h_t = project(h - a * g, target)
            v = exact_c5(h_t)
            if v < best_val:
                best_val = v
                best_a = a
            a *= 0.65
            if a < 1e-12:
                break
        if best_a > 0 and best_val < cur_c5 - 1e-14:
            h = project(h - best_a * g, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.3)
        else:
            stall += 1
            lr *= 0.5
            if stall > 60 or lr < 1e-8:
                break
    return best_h, best_c5

def pair_polish(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        scores = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_lag = int(np.argmax(scores))
        d = max_lag - (n - 1)
        if d >= 0:
            m = n - d
            gi = np.zeros(n)
            gi[d:] += (1.0 - best_h[:m])
            gi[:m] += -best_h[d:d + m]
        else:
            q = -d
            m = n - q
            gi[:m] += (1.0 - best_h[q:q + m])
            gi[q:] += -best_h[:m]
        up_candidates = np.argsort(gi)[::-1][:30]
        down_candidates = np.argsort(gi)[:30]
        tried = set()
        for ui in up_candidates:
            if time.time() > deadline:
                break
            for di in down_candidates:
                if ui == di or (ui, di) in tried:
                    continue
                tried.add((ui, di))
                best_h[np.array([ui])] += 0  # placeholder to keep flow
                amax = min(best_h[ui] if False else (1.0 - best_h[ui]), best_h[di])
                # Move mass from di to ui (increase ui, decrease di)
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    a = amax * frac
                    h_t = best_h.copy()
                    h_t[ui] = min(1.0, best_h[ui] + a)
                    actual = h_t[ui] - best_h[ui]
                    h_t[di] = max(0.0, best_h[di] - actual)
                    c5_t = exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t
                        improved = True
                        break
                    elif c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h_t
                        improved = True
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try reverse direction
            for ui in down_candidates:
                if time.time() > deadline:
                    break
                for di in up_candidates:
                    if ui == di:
                        continue
                    amax = min(1.0 - best_h[ui], best_h[di])
                    for frac in [0.25, 0.5, 0.75, 1.0]:
                        a = amax * frac
                        h_t = best_h.copy()
                        h_t[ui] = min(1.0, best_h[ui] + a)
                        actual = h_t[ui] - best_h[ui]
                        h_t[di] = max(0.0, best_h[di] - actual)
                        c5_t = exact_c5(h_t)
                        if c5_t < best_c5 - 1e-13:
                            best_c5 = c5_t
                            best_h = h_t
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    return best_h, best_c5

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new / 2.0)

def thue_morse(n):
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        seq[i] = c & 1
    return seq

def lp_loop(h, deadline, radius_init=0.15, max_iter=300):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        scores = np.correlate(h, 1.0 - h, mode='full') * dx_local = (2.0 / n)
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_val = np.max(scores)
        # Get active lags: those within 1% of max
        tol = 0.005 * max_val
        active_mask = scores >= max_val - tol
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(scores)])
        # Try top-1 first
        top1 = np.array([np.argmax(scores)])
        p = lp_direction(h, top1, radius)
        if p is None and len(active_lags) > 1:
            p = lp_direction(h, active_lags[:3], radius)
        if p is None:
            if len(active_lags) > 1:
                p = lp_direction(h, active_lags, radius)
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.6
            if radius < 1e-8 or stall > 40:
                break
            continue
        alpha, new_c5, h_new = thorough_line_search(h, p, cur_c5)
        if new_c5 < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.15, 0.4)
        else:
            stall += 1
            radius *= 0.6
            if stall % 20 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    best_n = n0

    # Phase 1: Aggressive LP with top-1 active lag (15% time)
    d1 = t0 + budget_s * 0.15
    h_lp, c5_lp = lp_loop(h, d1, radius_init=0.2, max_iter=500)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 2: Gradient-based optimization with various beta (20% time)
    d2 = t0 + budget_s * 0.35
    betas = [50, 30, 20, 15, 10, 7, 5, 3, 2]
    h_grad = best_h.copy()
    for beta in betas:
        if time.time() > d2:
            break
        remaining = d2 - time.time()
        if remaining < budget_s * 0.01:
            break
        share = remaining * 0.8
        sub_dl = time.time() + share
        h_grad, c5_g = grad_optimize(h_grad, beta, sub_dl, lr_init=0.003)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_grad.copy()

    # Phase 3: LP refinement after gradient (15% time)
    d3 = t0 + budget_s * 0.50
    h_lp2, c5_lp2 = lp_loop(best_h, d3, radius_init=0.1, max_iter=400)
    if c5_lp2 < best_c5:
        best_c5 = c5_lp2
        best_h = h_lp2.copy()

    # Phase 4: Diversification - perturb + re-optimize (25% time)
    d4 = t0 + budget_s * 0.75
    scales = [0.02, 0.035, 0.06, 0.09, 0.13, 0.18, 0.25]
    for ns in scales:
        if time.time() > d4:
            break
        sub_budget = budget_s * 0.02
        h_pert = best_h + rng.normal(0, ns, best_n)
        h_pert = project(h_pert, best_n / 2.0)
        # Quick LP
        sub_dl = min(d4, time.time() + sub_budget)
        h_try, c5_try = lp_loop(h_pert, sub_dl, radius_init=0.12, max_iter=100)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Grad-based
        if time.time() < d4:
            sub_dl2 = min(d4, time.time() + sub_budget * 0.5)
            h_g, c5_g = grad_optimize(h_pert, 15, sub_dl2, lr_init=0.004)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Phase 5: Resolution search (10% time)
    d5 = t0 + budget_s * 0.85
    tried_n = set([best_n])
    for dn in [0, -4, 4, -8, 8, -12, 12, -20, 20, -28, 28, -40, 40, -50, 50]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.01)
        h_r, c5_r = lp_loop(h_r, sub_dl, radius_init=0.1, max_iter=80)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Pair polish (5% time)
    d6 = t0 + budget_s * 0.93
    h_pol, c5_pol = pair_polish(best_h, d6, max_passes=60)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 7: Final micro-LP
    if time.time() < hard_deadline:
        h_f, c5_f = lp_loop(best_h, hard_deadline, radius_init=0.03, max_iter=200)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety: never return worse than parent
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-14:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
