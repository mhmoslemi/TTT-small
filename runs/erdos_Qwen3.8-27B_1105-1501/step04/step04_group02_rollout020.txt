import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_gradient(h, r, n):
    """Gradient of raw[r] w.r.t. h, where raw = correlate(h, 1-h, 'full')."""
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0:
            return g
        g[d:] += 1.0
        g[d:] -= h[:m]
        g[:m] -= h[d:]
    else:
        s = -d
        m = n - s
        if m <= 0:
            return g
        g[:m] += 1.0
        g[:m] -= h[s:]
        g[s:] -= h[:m]
    return g

def _all_row_gradients(h, rows, n):
    """Stack row gradients for given rows."""
    G = np.zeros((len(rows), n))
    for i, r in enumerate(rows):
        G[i] = _row_gradient(h, r, n)
    return G

def _line_search(h, p, c5_cur, alpha_max, n_grid=20):

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_t - gr * (hi_t - lo_t)
    c2 = lo_t + gr * (hi_t - lo_t)
    
    def _f(t):
        h_t = h + t * p
        if h_t.min() < -1e-12 or h_t.max() > 1.0 + 1e-12:
            return c5_cur + 1.0

def _svd_direction(h, n, k_active, margin_frac):
    """SVD-based descent direction from active constraints."""
    raw = _scores_raw(h)
    c5 = raw.max() * (2.0 / n)
    max_raw = raw.max()
    margin = margin_frac * (max_raw - np.median(raw))
    active_mask = raw >= (max_raw - margin)
    rows = np.where(active_mask)[0]
    if len(rows) == 0:
        return None
    # Limit to top k_active
    if len(rows) > k_active:
        top_idx = np.argsort(raw[rows])[::-1][:k_active]
        rows = rows[top_idx]
    G = _all_row_gradients(h, rows, n)
    # Remove mean from each row to respect sum constraint
    row_means = G.mean(axis=1, keepdims=True)
    G = G - row_means
    # Also project out the all-ones direction
    ones = np.ones(n)
    ones = ones - ones.mean()
    norm_ones = np.linalg.norm(ones)
    if norm_ones > 1e-12:
        G = G - G @ (ones / norm_ones)[:, None] * (ones / norm_ones)[None, :]
    # SVD
    try:
        U, S, Vt = np.linalg.svd(G, full_matrices=False)
        if len(S) > 0 and S[0] > 1e-10:
            p = Vt[0]
            p = p - p.mean()
            norm = np.max(np.abs(p))
            if norm > 1e-12:
                return p / norm
    except np.linalg.LinAlgError:
        pass
    # Fallback: negative weighted gradient
    p = -G.T @ np.ones(len(rows)) / len(rows)
    p = p - p.mean()
    norm = np.max(np.abs(p))
    if norm > 1e-12:
        return p / norm
    return None

def _roll_direction(h, n, shift):
    """Roll-based direction: roll(h, shift) - h, mean-zero."""
    p = np.roll(h, shift) - h
    p = p - p.mean()
    norm = np.max(np.abs(p))
    if norm > 1e-12:
        return p / norm
    return None

def _lp_direction(h, scores, radius):
    """Find LP direction minimizing max (score + G @ p)."""
    n = len(h)
    n_rows = 2 * n - 1
    G = np.zeros((n_rows, n))
    for r in range(n_rows):
        G[r] = _row_gradient(h, r, n)
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((n_rows, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _grad_optimize(h, deadline, tau_init=20.0, lr_init=0.015):
    """Gradient descent with softmax."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    mom = np.zeros(n)
    tau = tau_init
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        dx = 2.0 / n
        scores = _scores_raw(h) * dx
        max_s = float(np.max(scores))
        shifted = tau * (scores - max_s)
        exp_s = np.exp(shifted)
        w = exp_s / float(np.sum(exp_s))
        # Weighted gradient
        grad = np.zeros(n)
        n_rows = 2 * n - 1
        for r in range(n_rows):
            wr = w[r]
            if wr < 1e-14:
                continue
            grad += wr * _row_gradient(h, r, n)
        grad *= dx
        mom = 0.9 * mom + 0.1 * grad
        h_new = h - lr * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:

            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

            tau = min(tau * 1.01, 300.0)
        else:
            stall += 1
            lr *= 0.93

                break
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=40):
    """Pair-wise local refinement."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Sample pairs
        n_pairs = min(n * 2, 200)
        idx_j = rng.integers(0, n, size=n_pairs)
        idx_k = rng.integers(0, n, size=n_pairs)
        for jj, kk in zip(idx_j, idx_k):
            if jj == kk:
                continue
            j, k = int(jj), int(kk)
            if h[j] < 1e-6 or h[j] > 1 - 1e-6 or h[k] < 1e-6 or h[k] > 1 - 1e-6:
                continue
            for (dj, dk) in [(1.0, -1.0), (-1.0, 1.0)]:
                if dj > 0:
                    am = (1.0 - h[j]) / dj
                else:
                    am = h[j] / (-dj)
                if dk > 0:
                    am = min(am, (1.0 - h[k]) / dk)
                else:
                    am = min(am, h[k] / (-dk))
                if am < 1e-12:
                    continue
                for frac in [0.5, 1.0]:
                    h_t = h.copy()
                    h_t[j] += dj * am * frac
                    h_t[k] += dk * am * frac
                    if h_t[j] < -1e-10 or h_t[j] > 1 + 1e-10 or h_t[k] < -1e-10 or h_t[k] > 1 + 1e-10:
                        continue
                    c5_t = _exact_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        h = h_t.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

            h_opt, c5_opt = _svd_local_search(h_sh, sub_dl, k_active=15, margin_frac=0.3, radius_init=0.06)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.965

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    h = _project(h0, target0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Quick gradient warmup
    d1 = t0 + budget_s * 0.03
    h, c5 = _grad_optimize(h, d1, tau_init=12.0, lr_init=0.012)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SVD local search (main engine)
    d2 = t0 + budget_s * 0.22
    h, c5 = _svd_local_search(best_h, d2, k_active=25, margin_frac=0.35, radius_init=0.18, seed=seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 3: LP local search
    d3 = t0 + budget_s * 0.40
    h, c5 = _lp_local(best_h, d3, radius_init=0.12)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 4: Multi-resolution exploration
    d4 = t0 + budget_s * 0.55
    offsets = [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12]
    time_per_res = budget_s * 0.012
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try % 2 != 0:
            n_try += 1
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + time_per_res)
        h_r, c5_r = _svd_local_search(h_r, sub_dl, k_active=15, margin_frac=0.3, radius_init=0.1, seed=seed + n_try)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Sharpening exploration
    d5 = t0 + budget_s * 0.62
    if (deadline - time.time()) > 5.0:
        h, c5 = _sharpen_optimize(best_h, d5, budget_s * 0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Intensive refinement at best resolution
    d6 = t0 + budget_s * 0.78
    h, c5 = _svd_local_search(best_h, d6, k_active=30, margin_frac=0.5, radius_init=0.12, seed=seed + 500)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    if time.time() < d6:
        h, c5 = _lp_local(best_h, d6, radius_init=0.08)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 7: Perturbation + refine
    d7 = t0 + budget_s * 0.88
    best_h, best_c5 = _perturb_refine(best_h, d7, budget_s, seed=seed + 300)

    # Phase 8: Final SVD polish
    d8 = min(deadline, t0 + budget_s * 0.94)
    h, c5 = _svd_local_search(best_h, d8, k_active=20, margin_frac=0.4, radius_init=0.05, seed=seed + 800)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 9: Pair refinement
    if time.time() < deadline:
        h, c5 = _pair_refine(best_h, min(deadline, time.time() + budget_s * 0.02), max_passes=40, seed=seed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final LP micro-polish
    if time.time() < deadline:
        h, c5 = _lp_local(best_h, deadline, radius_init=0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, best_n
