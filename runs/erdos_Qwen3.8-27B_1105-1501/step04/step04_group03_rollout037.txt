def project(v, target):

            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            h = best_h.copy()

    # Phase 5: Anneal (10% of time)
    d5 = t0 + budget_s * 0.83
    if time.time() < d5:
        h_try, c5_try = anneal_pairs(h, d5, rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = best_h.copy()

    # Phase 6: Final intensive optimization (10% of time)
    d6 = hard_deadline
    if time.time() < d6:
        h_try, c5_try = subgrad_optimizer(h, min(d6, time.time() + budget_s * 0.04), K=8, tau0=0.005)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    if time.time() < d6:
        h_try, c5_try = pair_optimizer(best_h, min(d6, time.time() + budget_s * 0.03))
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    if time.time() < d6:
        h_try, c5_try = coordinate_optimizer(best_h, d6)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Final validation
    best_h = project(best_h, best_n / 2.0)
    final_c5 = exact_c5(best_h)

    # Safety fallback
    parent_h = project(h0, n0 / 2.0)
    parent_c5 = exact_c5(parent_h)
    if final_c5 > parent_c5 + 1e-12:
        best_h = parent_h
        final_c5 = parent_c5
        best_n = n0

    return best_h, final_c5, best_n
