import numpy as np
from scipy.optimize import linprog
import time

def _waterfill(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _correlate_vectors(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _line_search(h, d, dx, trust_radius):
    """Exact line search along direction d from h within trust region."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = np.min((1.0 - h[pos]) / d[pos])
    if np.any(neg):
        t_lo = np.max(h[neg] / d[neg])
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search on obj(t) = max(S0 + t*T - t^2*C) * dx
    def obj(t):
        return float(np.max(S0 + t * T - t*t * C) * dx)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = obj(c1), obj(c2)
    for _ in range(50):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = obj(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = obj(c2)
    t_opt = (a + b) / 2.0
    if t_opt < t_lo: t_opt = t_lo
    if t_opt > t_hi: t_opt = t_hi
    f_opt = obj(t_opt)
    
    # Also check t=0
    f0 = obj(0.0)
    if f0 < f_opt:
        t_opt = 0.0
        f_opt = f0
    
    return t_opt, f_opt

def _softmax_grad(h, dx, beta):
    """Compute softmax-weighted gradient of the max overlap."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scores = corr * dx
    max_s = np.max(scores)
    w = np.exp(-beta * (max_s - scores))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[:m] += w[r] * (1.0 - h[d:d+m])
                grad[d:] += w[r] * (-h[:m])
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += w[r] * (-h[g:g+m])
                grad[g:] += w[r] * (1.0 - h[g:])
    grad -= np.mean(grad)
    return grad

def _subgradient_loop(h, deadline, betas=(30, 80, 200, 500), lr_init=0.03, momentum=0.8):
    """Fast subgradient descent with adaptive step and momentum."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    bi = 0
    
    while time.time() < deadline:
        if stall > 80:
            break
        beta = betas[bi % len(betas)]
        d = _softmax_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            bi += 1
            stall += 1
            continue
        d = d / dn
        mom = momentum * mom + (1 - momentum) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            bi += 1
            continue
        mom = mom / mn
        
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.0)
        if new_c5 < cur - 1e-14:
            h = h + alpha * lr * mom
            h = _waterfill(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1

                bi += 1
                mom = np.zeros(n)
    return best_h, best_c5

def _lp_direction(h, scores, radius):
    """Solve LP to find best direction within trust region."""
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n + 1))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                A_ub[r, d:d+m] += (1.0 - h[:m])
                A_ub[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                A_ub[r, :m] += -(h[g:])
                A_ub[r, g:] += (1.0 - h[:m])
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.ones((1, n))
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _waterfill(h_new, n_new / 2.0)

def _random_binary(n, rng):
    """Random binary vector with exactly n/2 ones."""
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h

def _fourier_profile(n, rng):
    """Generate a profile using Fourier modes."""
    x = np.linspace(0, 1, n, endpoint=False)
    h = np.zeros(n)
    n_modes = rng.integers(2, 6)
    for m in range(1, n_modes + 1):
        freq = rng.integers(1, 4) * m
        amp = rng.uniform(-0.5, 0.5)
        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.cos(2*np.pi*freq*x + phase)
    h = _waterfill(h + 0.5, n/2)
    return h

            if m > 0:
                grad[:m] += w[r] * (-h[g:])
                grad[g:] += w[r] * (1.0 - h[g:g + m])
    grad -= np.mean(grad)
    return grad


def _subgrad_loop(h, deadline, betas=(40, 100, 300, 800), lr_init=0.03, mom_decay=0.82):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    lr = lr_init
    mom = np.zeros(n)
    stall = 0
    bi = 0
    while time.time() < deadline:
        if stall > 100:
            break
        beta = betas[bi % len(betas)]
        d = _softmax_grad(h, dx, beta)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            bi += 1
            stall += 2
            continue
        d = d / dn
        mom = mom_decay * mom + (1.0 - mom_decay) * d
        mn = np.linalg.norm(mom)
        if mn < 1e-13:
            bi += 1
            continue
        mom = mom / mn
        alpha, new_c5 = _line_search(h, lr * mom, dx, lr * 2.5)
        if new_c5 < cur - 1e-14:
            h = h + alpha * lr * mom
            h = _waterfill(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.6)
        else:
            stall += 1
            lr *= 0.82
            if stall % 15 == 0:
                bi += 1
            if stall % 40 == 0:
                mom = np.zeros(n)
    return best_h, best_c5


def _lp_dir(h, scores, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                A_ub[r, d:d + m] += (1.0 - h[:m])
                A_ub[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                A_ub[r, :m] += -h[g:]
                A_ub[r, g:] += (1.0 - h[:m])
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _lp_loop(h, deadline, radius_init=0.15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_dir(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.65
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur - 1e-13:
            h = h + alpha * p
            h = _waterfill(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.08, 0.45)
        else:
            stall += 1
            radius *= 0.7
            if stall % 25 == 0 and radius < 0.015:
                radius = 0.1
    return best_h, best_c5


def _pair_opt(h, deadline, rng, max_rounds=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        n_try = min(n * (n - 1) // 2, 4000)
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if h[i] <= 1e-9 or h[i] >= 1.0 - 1e-9:
                continue
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _waterfill(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _waterfill(h_new, n_new / 2.0)


def _block_move(h, rng, deadline, max_blocks=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _ in range(max_blocks):
        if time.time() > deadline:
            break
        L = int(rng.integers(2, max(3, n // 8) + 1))
        a = int(rng.integers(0, n - L))
        b = int(rng.integers(0, n - L))
        if b <= a + L and b >= a - L:
            continue
        d = np.zeros(n)
        d[a:a + L] = -1.0
        d[b:b + L] = 1.0
        d2 = np.zeros(n)
        d2[a:a + L] = 1.0
        d2[b:b + L] = -1.0
        for dd in [d, d2]:
            alpha, c5_try = _line_search(best_h, dd, dx, 0.3)
            if c5_try < best_c5 - 1e-14:
                best_h = best_h + alpha * dd
                best_h = _waterfill(best_h, target)
                best_c5 = _exact_c5(best_h)
    return best_h, best_c5


def _anneal(h_best, c5_best, deadline, rng):
    n = len(h_best)
    best_h = h_best.copy()
    best_c5 = c5_best
    scale = 0.04
    stall = 0
    while time.time() < deadline and stall < 120:
        h_p = best_h + rng.normal(0, scale, n)
        h_p = _waterfill(h_p, n / 2)
        sub_dl = min(deadline, time.time() + max(1.5, (deadline - time.time()) * 0.03))
        h_t, c5_t = _subgrad_loop(h_p, sub_dl, betas=(50, 200), lr_init=0.025)
        if time.time() < deadline:
            sub_dl2 = min(deadline, time.time() + max(1.0, (deadline - time.time()) * 0.02))
            h_t, c5_t = _lp_loop(h_t, sub_dl2, radius_init=0.08)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.008, scale * 0.88)
            stall = 0
        else:
            stall += 1
            scale = min(0.18, scale * 1.12)
    return best_h, best_c5


def _fourier_start(n, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    h = np.zeros(n)
    nm = int(rng.integers(2, 6))
    for m in range(1, nm + 1):
        freq = int(rng.integers(1, 5)) * m
        amp = rng.uniform(-0.4, 0.4)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x + ph)
    return _waterfill(h + 0.5, n / 2)


def _cosine_start(n, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    center = rng.uniform(0.2, 0.8)
    width = rng.uniform(0.1, 0.4)
    h = np.cos(np.pi * (x - center) / width) ** 2
    h = np.clip(h, 0, 1)
    return _waterfill(h, n / 2)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_dl = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _waterfill(h0, n0 / 2)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Quick subgradient warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _subgrad_loop(h_init, d1, betas=(30, 80, 200, 600), lr_init=0.025, mom_decay=0.8)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP local search (18%)
    d2 = t0 + budget_s * 0.26
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.18)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair + block refinement (7%)
    d3 = t0 + budget_s * 0.33
    h3, c5_3 = _pair_opt(best_h, d3, rng, max_rounds=50)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3:
        h3b, c5_3b = _block_move(best_h, rng, d3)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution exploration (22%)
    d4 = t0 + budget_s * 0.55
    n_cur = len(best_h)
    offsets = sorted(set(list(range(-24, 25, 2)) + list(range(-36, 37, 6)) + [0]))
    tried = set()
    rng4 = np.random.default_rng(seed + 100)
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_cur + dn
        if n_try < 36 or n_try > 240 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.013)
        h_o, c5_o = _subgrad_loop(h_r, sub_dl, betas=(40, 120, 400), lr_init=0.025)
        rem = d4 - time.time()
        if rem > 3 and c5_o < best_c5 + 0.005:
            sub_dl2 = min(d4, time.time() + max(2, budget_s * 0.008))
            h_o, c5_o = _lp_loop(h_o, sub_dl2, radius_init=0.1)
        if c5_o < best_c5 - 1e-12:
            best_c5 = c5_o
            best_h = h_o.copy()
            n_cur = n_try

    # Phase 5: Diverse starting points (15%)
    d5 = t0 + budget_s * 0.70
    n_best = len(best_h)
    diverse_starts = []
    h_sym = _waterfill((best_h + best_h[::-1]) / 2.0, n_best / 2)
    diverse_starts.append(h_sym)
    for _ in range(4):
        diverse_starts.append(_fourier_start(n_best, rng4))
    for _ in range(3):
        diverse_starts.append(_cosine_start(n_best, rng4))
    for s in diverse_starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_s, c5_s = _subgrad_loop(s, sub_dl, betas=(30, 100, 400), lr_init=0.03)
        if time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.006)
            h_s, c5_s = _lp_loop(h_s, sub_dl2, radius_init=0.1)
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Annealing / random restarts (12%)
    d6 = t0 + budget_s * 0.82
    rng6 = np.random.default_rng(seed + 200)
    h6, c5_6 = _anneal(best_h, best_c5, d6, rng6)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final LP polish (10%)
    d7 = t0 + budget_s * 0.92
    h7, c5_7 = _lp_loop(best_h, d7, radius_init=0.08)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Pair + subgradient final squeeze (8%)
    d8 = hard_dl
    rng8 = np.random.default_rng(seed + 300)
    h8, c5_8 = _pair_opt(best_h, min(d8, t0 + budget_s * 0.96), rng8, max_rounds=40)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8:
        h8b, c5_8b = _subgrad_loop(best_h, d8, betas=(200, 600, 1500), lr_init=0.015, mom_decay=0.85)
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()

    # Final validation
    best_h = _waterfill(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
