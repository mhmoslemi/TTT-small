def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores_raw(h):

    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -5.0, 5.0)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):

        if s > target:

        else:

    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))

def _grad_row(h, r, n):
    """Gradient of raw score r w.r.t. h."""
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        # g[i] for i=0..m-1: contribution from (1-h[i+d])*[i is first] + (-h[i-d])*[i-d is valid]
        # First index contribution: for i < m, g[i] += (1 - h[i+d])
        g[:m] += (1.0 - h[d:])
        # Second index contribution: for i >= d, g[i] += -h[i-d]
        g[d:] += -h[:m]
    else:
        s = -d
        m = n - s
        # score = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
        # First index: for i >= s, g[i] += (1 - h[i-s])
        g[s:] += (1.0 - h[:m])
        # Second index: for i+s < n, i.e., i < n-s = m, g[i] += -h[i+s]
        g[:m] += -h[s:]

        return -float(np.sum(p[:m] * p[d:]))
    else:
        s = -d
        m = n - s
        return -float(np.sum(p[s:] * p[:m]))

def _line_search_analytic(h, p, scores, r_top, alpha_max):

def _line_search(h, p, c5_cur, alpha_max):

    n = len(h)

                return c5_cur + 1.0

    """LP: find direction p minimizing max of active scores (linearized)."""
    n = len(h)
    m_active = len(active_indices)
    
    # Build gradient rows for active indices
    G_active = np.zeros((m_active, n))
    for i, r in enumerate(active_indices):
        G_active[i] = _grad_row(h, r, n)
    
    # LP: minimize t s.t. G_active @ p + scores_active <= t, |p| <= radius, sum(p)=0
    # Variables: p (n vars) + t (1 var) = n+1 vars
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0  # minimize t
    
    # Constraints: G_active @ p - t <= -scores_active
    A_ub = np.zeros((m_active, n + 1))
    A_ub[:, :n] = G_active
    A_ub[:, n] = -1.0
    b_ub = -np.array([scores[r] for r in active_indices])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def _try_shifts(h):
    """Try all cyclic shifts and return the best one."""
    n = len(h)
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    for s in range(1, n):
        h_s = np.roll(h, s)
        c5 = _exact_c5(h_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s.copy()
    return best_h, best_c5

def _lp_optimize(h, deadline, k_values=None, radius_init=0.1):
    """LP-based optimization with multiple active set sizes."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = radius_init
    
    if k_values is None:
        k_values = [1, 2, 3, 5, 8, 12, 20, 30]
    
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _scores_raw(h)
        max_score = float(np.max(scores))
        
        # Try different active set sizes
        improved = False
        for k in k_values:
            if k >= 2 * n - 1:
                k = 2 * n - 2
            top_idx = np.argsort(scores)[::-1][:k]
            # Ensure we include the actual max
            if 0 not in top_idx:  # always include index 0 (or argmax)
                top_idx = np.append(top_idx, np.argmax(scores))
            
            p, t_pred = _lp_direction(h, scores, top_idx, radius)
            if p is None:
                continue
            
            # Check if the LP says we can improve
            if t_pred >= max_score:
                continue
            
            # Compute alpha_max
            alpha_max = 1.0
            for j in range(n):
                if p[j] > 1e-14:
                    alpha_max = min(alpha_max, (1.0 - h[j]) / p[j])
                elif p[j] < -1e-14:
                    alpha_max = min(alpha_max, h[j] / (-p[j]))
            if alpha_max < 1e-12:
                continue
            
            alpha, new_c5 = _line_search(h, p, cur_c5, alpha_max)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                radius = min(radius * 1.1, 0.5)
                break  # restart with new scores
        
        if not improved:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
    
    return best_h, best_c5

def _grad_step(h, tau, lr):
    """One gradient step with softmax weights."""
    n = len(h)
    target = n / 2.0
    scores = _scores_raw(h)
    dx = 2.0 / n
    scores_phys = scores * dx
    max_s = float(np.max(scores_phys))
    shifted = tau * (scores_phys - max_s)
    exp_s = np.exp(shifted)
    w = exp_s / float(np.sum(exp_s))
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        grad += w[r] * _grad_row(h, r, n)
    
    h_new = h - lr * grad
    h_new = _project(h_new, target)
    return h_new

def _grad_optimize(h, deadline, tau_init=20.0, lr_init=0.015):
    """Gradient descent with adaptive parameters."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    tau = tau_init
    lr = lr_init
    stall = 0
    mom = np.zeros(n)
    
    while time.time() < deadline:
        if stall > 60:
            break
        h_new = _grad_step(h, tau, lr)
        # Add momentum
        h_new = h + 0.3 * (h_new - h) + 0.7 * mom
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-14:
            mom = h_new - h
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            lr = min(lr * 1.02, 0.1)
            tau = min(tau * 1.01, 150.0)
        else:
            stall += 1
            lr *= 0.9
            mom *= 0.5
            if stall % 20 == 0:
                tau = max(tau_init, tau * 0.7)
    
    return best_h, best_c5
