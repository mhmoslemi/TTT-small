    lo = v.min() - 1.0

def _project(v, target):
    n = len(v)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5*(lo+hi), 0.0, 1.0)

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
