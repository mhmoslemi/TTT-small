import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} via bisection on clipping threshold."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _lp_vertex(w, target):
    """Solve min w·s s.t. 0<=s<=1, sum(s)=target. Closed form."""
    n = len(w)
    s = np.zeros(n)
    idx = np.argsort(w)  # ascending
    k = int(target)
    if k > 0 and k < n:
        s[idx[:k]] = 1.0
        # If target not integer, handle fractional part
        rem = target - k
        if rem > 1e-12 and k < n:
            s[idx[k]] = rem
    elif k >= n:
        s[:] = 1.0
    return s

def _grad_q(h, q, dx):
    """Gradient of the q-th correlation component w.r.t. h."""
    n = len(h)
    d = q - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        # C_q = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        if d < n:
            for i in range(n - d):
                grad[i + d] += dx * (1.0 - h[i])
                grad[i] -= dx * h[i + d]
    else:
        k = -d
        # C_q = dx * sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])
        for i in range(n - k):
            grad[i] += dx * (1.0 - h[i + k])
            grad[i + k] -= dx * h[i]
    return grad

def _fw_step(h, target, num_active, temperature, max_alphas):
    """One Frank-Wolfe step: compute subgradient, find LP vertex, line search."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = float(np.max(corr))
    
    if max_corr <= 1e-15:
        return h.copy(), 0.0
    
    # Softmax weights over correlation components
    # Focus on the top ones
    sorted_idx = np.argsort(corr)[::-1]
    top_idx = sorted_idx[:min(num_active, len(corr))]
    
    # Temperature-scaled softmax
    top_corr = corr[top_idx]
    scaled = (top_corr - max_corr) / max(temperature * max_corr, 1e-10)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # Accumulate gradient
    grad = np.zeros(n)
    for w, qi in zip(weights, top_idx):
        grad += w * _grad_q(h, qi, dx)
    
    # LP vertex minimizing grad·s
    s = _lp_vertex(grad, target)
    
    # Line search: h_alpha = (1-alpha)*h + alpha*s
    best_alpha = 0.0
    best_val = _exact_c5(h)
    
    alphas = np.linspace(0, 1, max_alphas)
    vals = np.zeros(len(alphas))
    for i, a in enumerate(alphas):
        h_t = (1 - a) * h + a * s
        vals[i] = _exact_c5(h_t)
    
    min_idx = int(np.argmin(vals))
    if vals[min_idx] < best_val - 1e-14:
        best_alpha = alphas[min_idx]
        best_val = vals[min_idx]
        
        # Refine around best
        if 0 < min_idx < len(alphas) - 1:
            a_lo = alphas[max(0, min_idx - 2)]
            a_hi = alphas[min(len(alphas) - 1, min_idx + 2)]
            fine_alphas = np.linspace(a_lo, a_hi, 11)
            for a in fine_alphas:
                h_t = (1 - a) * h + a * s
                v = _exact_c5(h_t)
                if v < best_val:
                    best_val = v
                    best_alpha = a
    
    h_new = (1 - best_alpha) * h + best_alpha * s
    # Ensure feasibility
    h_new = _project(h_new, target)
    return h_new, best_val

        # d(h[i]+)/d(h[j]): -dx*h[j+d] for j in [0, n-d-1]

        # C_q = dx * sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])
        grad[:-k] += dx * (1.0 - h[k:])
        grad[k:] -= dx * h[:-k]
    return grad

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:

    lo = float(v.min()) - 2.0

def _lp_vertex(w, target):
    n = len(w)

def _grad_q(h, q, n, dx):
    d = q - (n - 1)
