```python
from scipy.optimize import linprog

def compute_minimax_direction(h, active_indices, corr_values):
    """Find the LP vertex s that minimizes the maximum active constraint increase."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    # Compute gradients for active constraints
    grads = []
    for q in active_indices:
        d = q - (n - 1)
        g = np.zeros(n)
        if d == 0:
            g[:] = dx * (1 - 2*h)
        elif d > 0:
            idx1 = np.arange(n - d)
            g[idx1] += dx * (1 - h[idx1 + d])
            g[idx1 + d] += -dx * h[idx1]
        else:
            p = -d
            idx1 = np.arange(n - p)
            g[idx1] += dx * (1 - h[idx1 + p])
            g[idx1 + p] += -dx * h[idx1]
        grads.append(g)
    
    grads = np.array(grads)  # shape (K, n)
    
    # LP: min t s.t. grads[i] @ (s - h) <= t for all i
    #     0 <= s <= 1, sum(s) = target
    # Variables: [s_0, ..., s_{n-1}, t]
    # Objective: minimize t → c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: grads[i] @ s - grads[i] @ h <= t
    # i.e., grads[i] @ s - t <= grads[i] @ h
    A_ub = np.zeros((len(grads), n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = grads @ h
    
    # Bounds: 0 <= s_i <= 1, t free (but bounded below by min of grads@h)
    bounds = [(0, 1)] * n + [(-1, 1)]  # t is free-ish
    
    # Equality: sum(s) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        s = res.x[:n]
        return s
    return None
```

2. Sample this range and find the minimum of max_q C_q
3. Refine with golden section or parabolic interpolation

Actually, for the max of quadratics, the minimum is either at a boundary or at a point where two of the quadratics are equal and all others are below. This is a 1D minimax problem that can be solved efficiently.

But for simplicity, I'll use a fine grid + refinement approach.

import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_to_box_sum(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} using bisection on the mean."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient_q(h, q):
    """Compute gradient of C_q with respect to h."""
    n = len(h)
    dx = 2.0 / n
    d = q - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        idx = np.arange(n - d)
        g[idx] += dx * (1.0 - h[idx + d])
        g[idx + d] += -dx * h[idx]
    else:
        p = -d
        idx = np.arange(n - p)
        g[idx] += dx * (1.0 - h[idx + p])
        g[idx + p] += -dx * h[idx]
    return g

def compute_minimax_vertex(h, active_indices):
    """Solve LP to find vertex s minimizing max_{q in active} grad_q . (s-h)."""
    n = len(h)
    target = n / 2.0
    K = len(active_indices)
    if K == 0:
        return None
    
    grads = np.array([compute_gradient_q(h, q) for q in active_indices])
    
    # LP: min t s.t. grads[i] @ s - grads[i] @ h <= t
    #     0 <= s_i <= 1, sum(s) = target
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = grads @ h
    
    bounds = [(0, 1)] * n + [(-10.0, 10.0)]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:
            s = res.x[:n]
            # Verify feasibility
            if np.all(s >= -1e-9) and np.all(s <= 1 + 1e-9):
                s = np.clip(s, 0, 1)
                s = project_to_box_sum(s, target)
                return s
    except Exception:
        pass
    return None

def line_search(h, s, target, max_evals=60):
    """Find best alpha in [0,1] for h_new = (1-alpha)*h + alpha*s."""
    best_alpha = 0.0
    best_val = exact_c5(h)
    
    # Coarse grid
    alphas = np.linspace(0, 1, 31)
    vals = np.array([exact_c5((1-a)*h + a*s) for a in alphas])
    i_min = np.argmin(vals)
    
    if vals[i_min] < best_val:
        best_val = vals[i_min]
        best_alpha = alphas[i_min]
    
    # Refine around best
    if 0 < i_min < 30:
        lo = alphas[max(0, i_min - 1)]
        hi = alphas[min(30, i_min + 1)]
        for _ in range(15):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            v1 = exact_c5((1-m1)*h + m1*s)
            v2 = exact_c5((1-m2)*h + m2*s)

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_to_box_sum(h_new, n_new / 2.0)
