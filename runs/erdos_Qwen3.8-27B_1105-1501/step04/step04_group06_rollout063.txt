```python
def _corr_grad(h, n, r):
    """Gradient of corr[r] w.r.t. h."""
    s = r - (n - 1)
    j = np.arange(n)
    gd = np.zeros(n)
    # First term: (1 - h[j+s]) where j+s is in [0, n)
    js = j + s
    m1 = (js >= 0) & (js < n)
    gd[j[m1]] += (1 - h[js[m1]])
    # Second term: -h[j-s] where j-s is in [0, n)
    jm = j - s
    m2 = (jm >= 0) & (jm < n)
    gd[j[m2]] -= h[jm[m2]]
    return gd
```

def _project(v, target):
    """Project v onto {x: 0≤x_i≤1, sum(x)=target}."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    target = float(target)

    if target >= n:
        return np.ones(n)

            lo = mid
        else:
            hi = mid

                diff -= a

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            for i in np.argsort(x):
                if x[i] >= 1.0:
                    continue
                a = min(1.0 - x[i], diff)
                x[i] += a
                diff -= a
                if diff <= 1e-12:
                    break
        else:
            for i in np.argsort(-x):
                if x[i] <= 0.0:
                    continue
                a = min(x[i], -diff)
                x[i] -= a
                diff += a
                if diff >= -1e-12:
                    break
    return x


def _exact_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)


def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _pair_coeff(h, n, j, k, s_arr):
    nc = 2 * n - 1
    a = np.zeros(nc)
    js = j + s_arr
    jm = j - s_arr
    m1 = (js >= 0) & (js < n)
    m2 = (jm >= 0) & (jm < n)
    a[m1] -= (1.0 - h[js[m1]])
    a[m2] += h[jm[m2]]
    ks = k + s_arr
    km = k - s_arr
    m3 = (ks >= 0) & (ks < n)
    m4 = (km >= 0) & (km < n)
    a[m3] += (1.0 - h[ks[m3]])
    a[m4] -= h[km[m4]]
    return a


def _golden_delta(corr, a, lo, hi):

    a_lo, a_hi = lo, hi
    c1 = a_hi - gr * (a_hi - a_lo)
    c2 = a_lo + gr * (a_hi - a_lo)

    f2 = float(np.max(corr + a * c2))

            c2 = c1
            f2 = f1
            c1 = a_hi - gr * (a_hi - a_lo)
            f1 = float(np.max(corr + a * c1))
        else:

            c1 = c2
            f1 = f2
            c2 = a_lo + gr * (a_hi - a_lo)
            f2 = float(np.max(corr + a * c2))
    return (a_lo + a_hi) / 2.0

    best_h = h.copy()

    stall = 0

        sweep += 1
        if stall > 30:
            break
        corr = _corr_vec(h)
        improved = 0
        order = rng.permutation(n)
        for j in order:

            # For efficiency, precompute once per sweep

                    h[k] -= delta
                    # Don't update corr since h is back
        if improved == 0:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _lse_obj(x, n, alpha):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    ms = float(np.max(corr))
    shifted = alpha * (corr - ms)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_v = np.exp(shifted)
    f = ms + (1.0 / alpha) * math.log(float(np.sum(exp_v)))
    w = exp_v / float(np.sum(exp_v))
    # Gradient
    j_arr = np.arange(n)
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        s = r - (n - 1)
        js = j_arr + s
        jm = j_arr - s
        g = np.zeros(n)
        m1 = (js >= 0) & (js < n)
        g[j_arr[m1]] += (1.0 - h[js[m1]])
        m2 = (jm >= 0) & (jm < n)
        g[j_arr[m2]] -= h[jm[m2]]
        grad += w[r] * g
    return f, grad


def _lse_refine(h, n, deadline):
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    for alpha in [10.0, 30.0, 100.0, 300.0]:
        if time.time() > deadline:
            break
        try:
            res = minimize(_lse_obj, h, args=(n, alpha), jac=True,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-15, 'maxls': 50, 'iprint': 0})
            x = _project(res.x, target)
            c5 = _exact_c5(x)
            if c5 < best_c5:
                best_c5, best_h, h = c5, x.copy(), x
        except Exception:
            pass
    return best_h, best_c5
