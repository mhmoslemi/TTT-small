```python
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _lse_obj_and_grad(h, tau):

    c = _corr_scaled(h)
    m = c.max()
    exp_c = np.exp((c - m) / tau)
    z = exp_c.sum()
    lse_val = m + tau * np.log(z)
    p = exp_c / z
    dx = 2.0 / n
    grad = np.zeros(n)

    return lse_val, grad

        pass

def _subgradient_step(h, step_size, n_top=3):
    n = len(h)
    c = _corr_scaled(h)
    top_idx = np.argsort(c)[-n_top:]
    p = np.ones(len(top_idx)) / len(top_idx)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i, q in enumerate(top_idx):
        w = p[i] * dx
        s = q - (n - 1)
        if s >= 0:

        else:
            d = -s
            grad[d:] += w * (1.0 - h[:n-d])
            grad[:n-d] -= w * h[d:]
    grad -= grad.mean()  # maintain sum constraint
    return _project(h - step_size * grad, n / 2.0)

        c5_cur = _exact_c5(h_cur)

            h_cur = h_new

        else:

            lr = max(lr * 0.85, min_lr)

                break
    return best_h, best_c5

def _resample(h, n_new):

    if n_new == n:

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _constraint_grad(h, q):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    s = q - (n - 1)
    if s >= 0:
        grad[:n - s] += dx * (1.0 - h[s:])
        grad[s:] -= dx * h[:n - s]
    else:
        d = -s
        grad[d:] += dx * (1.0 - h[:n - d])
        grad[:n - d] -= dx * h[d:]
    return grad


def _lse_obj_grad(h, tau):
    c = _corr_scaled(h)
    m = c.max()
    exp_c = np.exp((c - m) / tau)
    z = exp_c.sum()
    lse_val = m + tau * np.log(z)
    p = exp_c / z
    grad = np.zeros(len(h))
    for q in range(len(c)):
        w = p[q]
        if w < 1e-13:
            continue
        n = len(h)
        s = q - (n - 1)
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            d = -s
            grad[d:] += w * (1.0 - h[:n - d])
            grad[:n - d] -= w * h[d:]
    grad *= (2.0 / len(h))
    return lse_val, grad


def _eq_fun(x):
    return x.sum() - len(x) / 2.0


def _eq_jac(x):
    return np.ones(len(x))


def _lse_slsqp(h0, tau, maxiter=300):
    n = len(h0)

    def obj(x):
        val, _ = _lse_obj_grad(x, tau)
        return val

    def grad(x):
        _, g = _lse_obj_grad(x, tau)
        return g

    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return _project(h0, n / 2.0)


def _active_jac(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    J = np.zeros((len(active_indices), n))
    for i, q in enumerate(active_indices):
        s = q - (n - 1)
        if s >= 0:
            J[i, :n - s] += dx * (1.0 - h[s:])
            J[i, s:] -= dx * h[:n - s]
        else:
            d = -s
            J[i, d:] += dx * (1.0 - h[:n - d])
            J[i, :n - d] -= dx * h[d:]
    return J


def _active_slsqp(h, active_indices, maxiter=300, ftol=1e-13):
    n = len(h)
    c = _corr_scaled(h)
    max_c = float(c.max())
    x0 = np.concatenate([[max_c + 0.015], h])

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def ineq(x):
        hx = np.clip(x[1:], 0.0, 1.0)
        cx = _corr_scaled(hx)
        return x[0] - cx[active_indices]

    def ineq_jac(x):
        hx = np.clip(x[1:], 0.0, 1.0)
        Jg = _active_jac(hx, active_indices)
        J = np.zeros((len(active_indices), n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -Jg
        return J

    def eq(x):
        return x[1:].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    bounds = [(-50.0, 50.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _project(np.clip(res.x[1:], 0.0, 1.0), n / 2.0)
    except Exception:
        pass
    return h


def _get_active(c, frac=0.5, min_count=10):
    mx = c.max()
    if mx <= 0:
        return np.arange(len(c))
    thr = mx * frac
    act = np.where(c >= thr)[0]
    if len(act) < min_count:
        k = min(min_count * 2, len(c))
        act = np.argsort(c)[-k:]
    return np.sort(act)


def _subgrad_step(h, lr, n_top=3):
    n = len(h)
    c = _corr_scaled(h)
    top = np.argsort(c)[-n_top:]
    grad = np.zeros(n)
    for q in top:
        grad += _constraint_grad(h, q)
    grad /= n_top
    grad -= grad.mean()
    return _project(h - lr * grad, n / 2.0)


def _subgrad_loop(h, deadline, lr=0.08, min_lr=1e-5, patience=50):

    best_c5 = _exact_c5(h)
    h_cur = h.copy()
    cur_lr = lr
    stall = 0
    for _ in range(20000):

        if stall > patience:
            break
        h_new = _subgrad_step(h_cur, cur_lr)
        c5_new = _exact_c5(h_new)
        c5_cur = _exact_c5(h_cur)
        if c5_new < best_c5 - 1e-14:

            best_h = h_new.copy()
            h_cur = h_new
            stall = 0

            stall = 0
        else:

            cur_lr = max(cur_lr * 0.85, min_lr)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _dual_refine(h, deadline, n_iter=200):
    """Mirror-descent style dual update for the minimax structure."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    h_cur = h.copy()
    eta = 0.3
    for iteration in range(n_iter):
        if time.time() > deadline:
            break
        c = _corr_scaled(h_cur)
        m = c.max()
        w = np.exp((c - m) / eta)
        w /= w.sum()
        grad = np.zeros(n)
        for q in range(2 * n - 1):
            if w[q] < 1e-12:
                continue
            grad += w[q] * _constraint_grad(h_cur, q)
        grad -= grad.mean()
        # Adaptive step size
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005]:
            h_try = _project(h_cur - alpha * grad, n / 2.0)
            c5_try = _exact_c5(h_try)
            if c5_try < _exact_c5(h_cur) - 1e-13:
                h_cur = h_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                break
        eta *= 0.97
    return best_h, best_c5
```

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0

        else:

def _constraint_grad(h, q):

    else:

    return grad
