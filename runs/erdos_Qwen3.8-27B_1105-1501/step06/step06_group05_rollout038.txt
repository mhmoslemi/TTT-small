def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _grad_matrix(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))

def _line_search(h, d, dx):
    n = len(h)
    pos = d > 1e-14
    neg = d < -1e-14
    L = 0.0
    U = 1e6
    if np.any(pos):
        L = max(L, float(np.max(-h[pos] / d[pos])))
        U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        L = max(L, float(np.max((1.0 - h[neg]) / d[neg])))
        U = min(U, float(np.min(-h[neg] / d[neg])))
    if U - L < 1e-14:
        return 0.0, _exact_c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    nq = 80
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx

        else:

            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)

def _slp_direction(h, corr_raw, G):
    n = len(h)
    M = 2 * n - 1
    target = n / 2.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.einsum('ij,i->j', G, h) - corr_raw
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 8.0})
        if res.success:
            d = res.x[:n] - h
            if np.linalg.norm(d) > 1e-10:
                return d
    except Exception:
        pass
    return None


def _smooth_direction(h, corr_raw, G, tau):
    m = np.max(corr_raw)
    w = np.exp((corr_raw - m) / tau)
    zz = np.sum(w)
    if zz < 1e-300:
        return None
    w = w / zz
    grad = G.T @ w
    d = -grad
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm


def _pair_line_search(h, i, j, dx):

    s_lo = max(-h[i], h[j] - 1.0)
    s_hi = min(1.0 - h[i], h[j])

    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0

        t = s_hi
    if t < s_lo:
        t = s_lo
    hn = h + t * d
    f = _exact_c5(hn)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n, dtype=float)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 5 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def _ramp_init(n, slope, offset, rng):
    x = np.arange(n, dtype=float) / n
    base = np.clip(slope * x + offset, 0.0, 1.0)
    base = base + rng.normal(0, 0.01, n)
    return _project(base, n / 2.0)


def _triangle_init(n, freq, phase, rng):
    x = np.arange(n, dtype=float) / n
    base = 0.5 + 0.5 * np.abs(2.0 * ((freq * x + phase) % 1.0) - 1.0)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)

    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None

    if base is not None and base.size >= 8:
        n0 = len(base)
        h0 = _project(base, n0 / 2.0)
    else:
        n0 = 122
        h0 = _sine_init(n0, 1, 0.0, rng)

    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: Multi-start (12% of budget)
    d1 = t0 + budget_s * 0.12
    starts = [h0.copy()]
    freqs = [(1, 0.0), (1, 0.7), (2, 0.0), (2, 1.3), (3, 0.5), (4, 0.0), (5, 0.8)]
    for fq, ph in freqs:
        starts.append(_sine_init(n0, fq, ph, rng))
    for _ in range(3):
        starts.append(_block_init(n0, rng))
    for _ in range(2):
        starts.append(_triangle_init(n0, 1, rng.random(), rng))
    for sl in [1.5, 2.5, 4.0]:
        starts.append(_ramp_init(n0, sl, -0.3, rng))

    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + max(1.5, budget_s * 0.015))
        hh, ff = _optimize(s, sub_dl, rng, slp_frac=0.35, smooth_frac=0.35, pair_frac=0.20, rand_frac=0.10)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Phase 2: Main deep optimization (50% of budget)
    d2 = t0 + budget_s * 0.62
    hh, ff = _optimize(best_h, d2, rng, slp_frac=0.42, smooth_frac=0.28, pair_frac=0.22, rand_frac=0.08)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: Multi-resolution (18% of budget)
    d3 = t0 + budget_s * 0.80
    bn = len(best_h)
    tried = set()
    for dn in [-2, 2, -4, 4, -3, 3, -6, 6, -8, 8]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 20 or n_try > 300:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + max(3.0, budget_s * 0.02))
        hr2, fr2 = _optimize(hr, sub_dl, rng, slp_frac=0.40, smooth_frac=0.30, pair_frac=0.22, rand_frac=0.08)
        if fr2 < best_f - 1e-9:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 4: Final polish (20% of budget)
    d4 = deadline
    hh, ff = _optimize(best_h, d4, rng, slp_frac=0.50, smooth_frac=0.20, pair_frac=0.25, rand_frac=0.05)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
