Let me think about the mathematical structure more carefully:

For h of length n, the correlation at lag k (where k ranges from -(n-1) to n-1) is:
- For k ≥ 0: sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
- For k < 0: sum_{i=0}^{n+k-1} h[i+k] * (1-h[i])

The C₅ value is dx * max_k of these.

The derivative w.r.t. h[j] for lag k ≥ 0:
- If j < n-k: contribution from h[j] is (1-h[j+k]), and from (1-h[j+k]) is -h[j]
- If j >= k and j+k < n: contribution from (1-h[j+k]) when the index i = j-k gives -h[j-k]...

For a = h, b = 1-h:
np.correlate(h, 1-h, 'full')[r] where r = n-1+k (k is the lag)

For lag k ≥ 0: sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
For lag k < 0 (let s = -k > 0): sum_{i=0}^{n-s-1} h[i+s] * (1-h[i])

The gradient of the objective (max of these) w.r.t. h[j]:
Using soft-max with temperature beta:
f_smooth = (1/beta) * log(sum_k exp(beta * s_k))

∂f_smooth/∂h[j] = sum_k w_k * (∂corr_k/∂h[j]) * dx / n * ... 

Actually let me just directly compute the gradient of each correlation w.r.t. h[j]:

For lag k ≥ 0: corr_k = sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
∂corr_k/∂h[j] = (contribution from first factor: 1 if j < n-k, times (1-h[j+k]))
              + (contribution from second factor: -1 if j >= k and j-k < n-k, i.e., k <= j < n, times h[j-k])

So: ∂corr_k/∂h[j] = [j < n-k] * (1-h[j+k]) - [k <= j < n] * h[j-k]

For lag k < 0 (s = -k): corr_k = sum_{i=0}^{n-s-1} h[i+s] * (1-h[i])
∂corr_k/∂h[j] = [j >= s and j < n] * (1-h[j-s]) - [j < n-s] * h[j+s]

This matches what the previous code's `_grad_matrix` does.

The previous program uses LP to find a direction that decreases all active constraints simultaneously. This is good but expensive.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_lags(h):
    """Compute all correlation values (unnormalized)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_row(h, lag):
    """
    Compute gradient of correlation at given lag w.r.t. h.
    lag is the actual lag (can be negative).
    corr_lag = sum_{i} appropriate products
    """
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return g
        # corr_lag = sum_{i=0}^{m-1} h[i] * (1-h[i+lag])
        # d/dh[j]: if j < m, contributes (1-h[j+lag])
        #          if j >= lag, contributes -h[j-lag] (from the (1-h[j]) term when i = j-lag)
        g[:m] += (1.0 - h[lag:n])
        g[lag:] -= h[:m]
    else:
        s = -lag
        m = n - s
        if m <= 0:
            return g
        # corr_lag = sum_{i=0}^{m-1} h[i+s] * (1-h[i])
        # d/dh[j]: if j >= s and j < n, contributes (1-h[j-s]) [first factor]
        #          if j < m, contributes -h[j+s] [second factor, (1-h[j])]
        g[s:n] += (1.0 - h[:m])
        g[:m] -= h[s:n]
    return g

def _smooth_gradient(h, raw, beta):
    """
    Compute the gradient of the soft-max surrogate.
    raw: array of correlation values (unnormalized), length 2n-1
    beta: temperature parameter (larger = sharper)
    """
    n = len(h)
    # Soft-max weights
    # Shift to avoid overflow
    r = raw - raw.max()
    ex = np.exp(beta * r)
    w = ex / (ex.sum() + 1e-300)
    
    g = np.zeros(n)
    for lag in range(-(n-1), n):
        if w[n-1+lag] > 1e-15:
            g += w[n-1+lag] * _grad_row(h, lag)
    return g

def _line_search(h, d, dx, radius):
    """
    Exact quadratic line search.
    Along h + alpha*d, each correlation is a quadratic in alpha.
    Find alpha in [0, radius] (positive direction) that minimizes max of quadratics.
    Also check [-radius, 0] for negative direction.
    """
    n = len(h)
    S0 = _corr_lags(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Evaluate at a grid of points
    n_grid = 40
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    best_val = float(np.max(S0) * dx)
    best_t = 0.0
    
    for idx in range(n_grid):
        t = grid_t[idx]
        val = float(np.max(S0 + t * T - t * t * C)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_t = t
    
    # Refine around best
    if best_t == 0.0:
        return 0.0, best_val
    
    lo_b = grid_t[max(0, int(np.argmin([float(np.max(S0 + t * T - t*t*C)) for t in grid_t])) - 2)]
    hi_b = grid_t[min(n_grid - 1, int(np.argmin([float(np.max(S0 + t * T - t*t*C)) for t in grid_t])) + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if hi_b - lo_b < 1e-13:
            break
        if f1 < f2:
            hi_b = c2
            c2 = c1
            f2 = f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            lo_b = c1
            c1 = c2
            f1 = f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    if opt_val < best_val:
        best_val = opt_val
        best_t = opt_t
    
    return best_t, best_val

def _make_directions(h, g, rng, n_max_dirs=6):
    """Generate candidate descent directions from gradient g."""
    n = len(h)
    dirs = []
    
    # 1. Negative gradient (centered)
    d = -g - np.mean(g)
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm > 1e-12:
        dirs.append(d / nrm)
    
    # 2. Block-averaged gradient directions
    for blk in [8, 16]:
        if n % blk != 0:
            continue
        ng = g.copy()
        ng -= ng.mean()
        # Average in blocks
        gb = ng.reshape(-1, blk).mean(axis=1)
        d = -np.tile(gb, (1, n // blk)).reshape(n)
        d -= d.mean()
        nrm = np.linalg.norm(d)
        if nrm > 1e-12:
            dirs.append(d / nrm)
    
    # 3. Pair directions from gradient extremes
    # Find positions where we want to increase (low g) and decrease (high g)
    n_pairs = min(3, n_max_dirs - len(dirs))
    if n_pairs > 0:
        # Sort by gradient
        order = np.argsort(g)
        # Take pairs from bottom and top
        for pi in range(n_pairs):
            i = order[pi]  # low gradient -> want to increase
            j = order[n - 1 - pi]  # high gradient -> want to decrease
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            d -= d.mean()
            nrm = np.linalg.norm(d)
            if nrm > 1e-12:
                dirs.append(d / nrm)
    
    # 4. Low-frequency cosine direction
    x = np.arange(n, dtype=float)
    d = np.cos(2.0 * np.pi * rng.uniform(1.0, 4.0) * x / n + rng.uniform(0, 2*np.pi))
    d -= d.mean()
    nrm = np.linalg.norm(d)
    if nrm > 1e-12:
        dirs.append(d / nrm)
    
    return dirs[:n_max_dirs]

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _symmetry_avg(h):
    """Average h with its reflection and complement-reflection."""
    n = len(h)
    h_ref = np.flip(h)
    # h and 1-h have same C5, so average with complement-reflection
    h_comp_ref = 1.0 - np.flip(h)
    # But we need to maintain sum = n/2
    # h_ref has same sum as h
    # h_comp_ref has sum = n - sum(h) = n - n/2 = n/2
    avg = _project(0.5 * (h + h_ref), n / 2.0)
    return avg

The vertex of this parabola is at α_v = T[k] / (2·C[k]) (if C[k] > 0).

        if time.time() > d2 - 3:
            break
        n_try = n0 + dn
        if n_try < 64 or n_try > 240 or n_try in tried_ns:
            continue
        if n_try % 2 != 0:
            n_try += 1
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        alloc = min(budget_s * 0.018, (d2 - time.time()) * 0.4)
        sub_dl = min(d2, time.time() + alloc)
        h_r, c5_r = _local_optimize(h_r, sub_dl, beta_init=10.0, radius_init=0.10, max_stall=35, n_dirs=5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    d3 = t0 + budget_s * 0.72
    n_final = len(best_h)
    beta_list = [25.0, 18.0, 35.0, 15.0]
    for bt in beta_list:
        if time.time() > d3 - 8:
            break
        alloc = min(budget_s * 0.045, (d3 - time.time()) * 0.35)
        sub_dl = min(d3, time.time() + alloc)
        h_try = _project(best_h.copy(), n_final / 2.0)
        h_try, c5_try = _local_optimize(h_try, sub_dl, beta_init=bt, radius_init=0.05, max_stall=60, n_dirs=6)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    if time.time() < d3 - 5:
        alloc = min(budget_s * 0.03, (d3 - time.time()) * 0.4)
        sub_dl = min(d3, time.time() + alloc)
        h_lp = _project(best_h.copy(), n_final / 2.0)
        h_lp, c5_lp = _lp_polish(h_lp, sub_dl)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    if time.time() < d3 - 3:
        n_final = len(best_h)
        h_pair = _project(best_h.copy(), n_final / 2.0)
        dx = 2.0 / n_final
        cur_pc5 = _exact_c5(h_pair)
        stall_p = 0
        alloc = min(budget_s * 0.025, (d3 - time.time()) * 0.5)
        sub_dl = min(d3, time.time() + alloc)
        for _ in range(50000):
            if time.time() > sub_dl or stall_p > 300:
                break
            i = int(rng.integers(0, n_final))
            j = int(rng.integers(0, n_final))
            if i == j:
                continue
            d = np.zeros(n_final)
            d[i] = 1.0
            d[j] = -1.0
            alpha, c5_try = _line_search_exact(h_pair, d, dx, 0.25, top_k=6)
            if c5_try < cur_pc5 - 1e-14:
                h_new = h_pair + alpha * d
                h_new = _project(h_new, n_final / 2.0)
                actual = _exact_c5(h_new)
                if actual < cur_pc5 - 1e-14:
                    h_pair = h_new
                    cur_pc5 = actual
                    if cur_pc5 < best_c5:
                        best_c5 = cur_pc5
                        best_h = h_pair.copy()
                    stall_p = 0
                else:
                    stall_p += 1
            else:
                stall_p += 1
        if cur_pc5 < best_c5:
            best_c5 = cur_pc5
            best_h = h_pair.copy()

    d4 = t0 + budget_s * 0.87
    n_final = len(best_h)
    scale = 0.015
    n_restarts = 0
    while time.time() < d4 and n_restarts < 300:
        n_restarts += 1
        x = np.arange(n_final, dtype=float)
        perturb = np.zeros(n_final)
        for _ in range(3):
            freq = rng.uniform(0.5, 10.0)
            phase = rng.uniform(0, 2 * np.pi)
            ampl = rng.uniform(0.4, 1.5)
            perturb += ampl * np.sin(2.0 * np.pi * freq * x / n_final + phase)
        perturb += rng.normal(0, 0.3 * scale, n_final)
        perturb *= scale
        h_pert = _project(best_h + perturb, n_final / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.003)
        h_t, c5_t = _local_optimize(h_pert, sub_dl, beta_init=15.0, radius_init=0.05, max_stall=20, n_dirs=3)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.001, scale * 0.85)
        else:
            scale = min(0.05, scale * 1.06)

    d5 = deadline
    n_best = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d5 - 8:
            break
        n_try = n_best + dn
        if n_try < 64 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        alloc = min(budget_s * 0.012, max(2, (d5 - time.time()) * 0.3))
        sub_dl = min(d5, time.time() + alloc)
        h_r, c5_r = _local_optimize(h_r, sub_dl, beta_init=20.0, radius_init=0.04, max_stall=25, n_dirs=4)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    if time.time() < deadline - 3:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl = min(deadline - 1, time.time() + budget_s * 0.03)
        h_f, c5_f = _local_optimize(h_f, sub_dl, beta_init=30.0, radius_init=0.02, max_stall=60, n_dirs=4)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if deadline - time.time() > 4:
        h_f2 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl2 = min(deadline - 1, time.time() + 3.0)
        h_f2, c5_f2 = _lp_polish(h_f2, sub_dl2)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
