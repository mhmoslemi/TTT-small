import numpy as np
import time
from scipy.optimize import linprog

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_row_grad(h, k):
    """Compute gradient of row k (0-indexed in full correlation) w.r.t. h."""
    n = len(h)
    s = k - (n - 1)
    m = n - abs(s)
    g = np.zeros(n)
    if m <= 0:
        return g
    if s >= 0:
        for j in range(s, n):
            g[j] += 1.0 - h[j - s]
        for j in range(m):
            g[j] += -h[j + s]
    else:
        r = -s
        for j in range(m):
            g[j] += 1.0 - h[j + r]
        for j in range(r, n):
            g[j] += -h[j - r]
    return g

def line_search_exactly(h, d, dx, R0):
    """Exact 1D minimization of max(R0 + a*T - a^2*C)*dx along direction d."""
    n = len(h)
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Determine valid alpha range
    lo, hi = -1.0, 1.0
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    
    def f(a):
        return float(np.max(R0 + a * T - a * a * C)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    
    opt = (a + b) / 2.0
    val = f(opt)
    # Check endpoints
    f_lo, f_hi = f(lo), f(hi)
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def lp_direction(h, active_rows, n_active_max):
    """Find direction via LP. Returns direction d (zero-sum) or None."""
    n = len(h)
    total_lags = 2 * n - 1
    R = np.correlate(h, 1.0 - h, 'full')
    
    # Select active rows (top n_active_max by score)
    scores = R * (2.0 / n)
    max_score = float(np.max(scores))
    threshold = max_score - 0.01 * max_score
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-scores)[:5]
    active = active[:n_active_max]
    
    if len(active) == 0:
        return None, R, scores
    
    # Compute gradients for active rows
    grads = np.zeros((len(active), n))
    for i, k in enumerate(active):
        grads[i, :] = compute_row_grad(h, k)
    
    R_active = R[active]
    
    # LP: min t
    # s.t. grads[i] · d - t <= -R_active[i]  for each i
    #      sum(d) = 0
    #      -h[j] <= d[j] <= 1-h[j]
    
    # Variables: [d[0],...,d[n-1], t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for i in range(len(active)):
        A_ub[i, :n] = grads[i, :]
        A_ub[i, n] = -1.0
        b_ub[i] = -R_active[i]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)]
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, R, scores
    
    d = res.x[:n]
    d -= np.sum(d) / n  # ensure exactly zero sum (numerical)
    
    # Check if direction is meaningful
    if np.linalg.norm(d) < 1e-10:
        return None, R, scores
    
    return d, R, scores

def _golden_pair(h, i, j, dx, S0):
    """Exact line search for direction e_j - e_i."""
    n = len(h)
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _lp_optimize(h, deadline, rng, n_active_max=20):
    """Main LP-based optimization loop."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, 'full')
    stall = 0
    
    while time.time() < deadline and stall < 100:
        d, R0, scores = lp_direction(h, None, n_active_max)
        if d is None:
            stall += 1
            if stall > 30:
                break
            # Try random pair
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                alpha, new_c5 = _golden_pair(h, i, j, dx, R0)
                if alpha != 0 and new_c5 < cur_c5 - 1e-14:
                    h[i] -= alpha
                    h[j] += alpha
                    h = project(h, target)
                    cur_c5 = c5_exact(h)
                    R0 = np.correlate(h, 1.0 - h, 'full')
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
            continue
        
        alpha, new_c5 = line_search_exactly(h, d, dx, R0)
        if alpha != 0 and new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R0 = np.correlate(h, 1.0 - h, 'full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20 and rng.random() < 0.3:
                idx = rng.integers(n, size=min(6, n))
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.03, 0.03)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R0 = np.correlate(h, 1.0 - h, 'full')
                stall = 0
    
    return best_h, best_c5

    d2 = t0 + budget_s * 0.55
    h_p2, c5_p2 = _pair_refine(best_h, d2, rng, n_pairs=20)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()
    
    # Phase 3: More LP with different active set size (10%)
    d3 = t0 + budget_s * 0.65
    h_p3, c5_p3 = _lp_main(best_h, d3, rng, n_active_max=25)
    if c5_p3 < best_c5:
        best_c5 = c5_p3
        best_h = h_p3.copy()
    
    # Phase 4: Try different n values (15%)
    d4 = t0 + budget_s * 0.80
    for dn in [-6, -4, -2, 2, 4, 6, -8, 8, -10, 10]:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 300 or n_try % 2 != 0:
            continue
        h_r = resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.025)
        h_r, c5_r = _lp_main(h_r, sub_dl, rng, n_active_max=12)
        if c5_r < best_c5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.01)
            h_r, c5_r2 = _pair_refine(h_r, sub_dl2, rng, n_pairs=12)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r.copy()
                best_n = n_try
    
    # Phase 5: 4-coordinate search (8%)
    d5 = t0 + budget_s * 0.88
    h_p5, c5_p5 = _four_coord(best_h, d5, rng)
    if c5_p5 < best_c5:
        best_c5 = c5_p5
        best_h = h_p5.copy()
    
    # Phase 6: Final LP polish (8%)
    d6 = deadline
    h_p6, c5_p6 = _lp_main(best_h, d6, rng, n_active_max=20)
    if c5_p6 < best_c5:
        best_c5 = c5_p6
        best_h = h_p6.copy()
    
    # Final project and evaluate
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    
    return best_h, final_c5, len(best_h)
