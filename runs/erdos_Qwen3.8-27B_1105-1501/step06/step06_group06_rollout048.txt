    """Find optimal δ for pair swap (i,j)."""
    n = len(h)
    dx = 2.0 / n
    # Feasible range for δ
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    if delta_hi - delta_lo < 1e-15:
        return 0.0, _exact_c5(h)
    
    # Compute a_k = d corr_k / dδ = d corr_k / dh[i] - d corr_k / dh[j]
    # and b_k = corr_k at δ=0
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Compute partial derivatives
    def partial(j_pos):
        """Compute d corr_k / dh[j_pos] for all k."""
        d = np.zeros(2*n-1)
        for k in range(2*n-1):
            s = k - (n-1)
            if s >= 0:
                if j_pos < n - s:
                    d[k] += (1.0 - h[j_pos + s]) * dx
                if j_pos >= s:
                    d[k] += (-h[j_pos - s]) * dx
            else:
                g = -s
                if j_pos >= g:
                    d[k] += (1.0 - h[j_pos - g]) * dx
                if j_pos < n - g:
                    d[k] += (-h[j_pos + g]) * dx
        return d
    
    a = partial(i) - partial(j)
    b = corr.copy()  # corr at δ=0
    
    # Now minimize max_k (a[k]*δ + b[k]) over [delta_lo, delta_hi]
    # Binary search on the subgradient
    lo = delta_lo
    hi = delta_hi
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        max_val = vals.max()
        active = np.where(np.abs(vals - max_val) < 1e-12)[0]
        min_a = a[active].min()
        max_a = a[active].max()
        if min_a > 1e-12:
            hi = mid  # all active slopes positive, need to go left
        elif max_a < -1e-12:
            lo = mid  # all active slopes negative, need to go right
        else:
            break  # 0 is in subgradient
    best_delta = (lo + hi) / 2.0
    # Evaluate at best_delta and boundaries
    candidates = [delta_lo, delta_hi, best_delta]
    best_c5 = float('inf')
    for d in candidates:
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        c5 = _exact_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_delta = d
    return best_delta, best_c5

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lsep_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - corr.max()
    log_sum = float(np.log(np.exp(beta * shifted).sum()) + corr.max() * beta) / beta
    return log_sum

def _lsep_grad(h, beta):

    dx = 2.0 / n

def _corr_gradients(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            G[k, :m] += (1.0 - h[s:s + m])
            G[k, s:s + m] += -h[:m]
        else:
            g = -s
            G[k, g:g + m] += (1.0 - h[:m])
            G[k, :m] += -h[g:g + m]
    G *= dx
    return G

def _hessian_quad(h, d):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            c[k] = -np.sum(d[:m] * d[s:s + m])
        else:
            g = -s
            c[k] = -np.sum(d[g:g + m] * d[:m])
    c *= dx
    return c

def _exact_line_search(h, d, S0, T, C, radius):
    n = len(h)
    dx = 2.0 / n
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] + grid[:, None] ** 2 * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = grid[max(0, best_i - 1)]
    b = grid[min(n_grid - 1, best_i + 1)]
    f1 = float(np.max(S0 + a * T + a * a * C)) * dx
    f2 = float(np.max(S0 + b * T + b * b * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, f2 = a, f1
            a = b - gr * (b - a)
            f1 = float(np.max(S0 + a * T + a * a * C)) * dx
        else:
            a, f1 = b, f2
            b = a + gr * (b - a)
            f2 = float(np.max(S0 + b * T + b * b * C)) * dx
    mid = (a + b) / 2.0
    mid_val = float(np.max(S0 + mid * T + mid * mid * C)) * dx
    if mid_val < best_val:
        best_t, best_val = mid, mid_val
    return best_t, best_val

def _block_minimize(h, deadline, block_size=8, max_sweeps=50):

            start = b * block_size

            bsize = end - start
            fixed_sum = float(np.sum(h) - np.sum(h[start:end]))
            block_target = target - fixed_sum
            if block_target < -1e-10 or block_target > bsize + 1e-10:

            h_block = h[start:end].copy()
            h_rest = np.delete(h, np.arange(start, end))
            
            def block_c5(xb):
                h_full = np.concatenate([h_rest[:start], xb, h_rest[start:]])
                return _exact_c5(h_full)
            
            x0 = h_block.copy()
            best_block = h_block.copy()
            best_block_c5 = block_c5(x0)
            bounds = [(0.0, 1.0)] * bsize

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_polish(h, deadline, max_outer=12):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        K = min(35, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        t_init = cur_c5 + 3e-5
        x0 = np.concatenate([h, [t_init]])
        
        def obj_fn(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]
        
        def ineq_jac(x):
            hx = x[:n]
            G = _corr_gradients(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J
        
        def eq_fn(x):
            return np.sum(x[:n]) - target
        
        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-13})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:

            break
    return best_h, best_c5

        best_h = h2.copy()
        best_n = len(h2)
    
    # Phase 3: SLSQP polish (10% time)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _slsqp_polish(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)
    
    # Phase 4: Multi-resolution sweep (20% time)
    d4 = t0 + budget_s * 0.85
    n_base = best_n
    tried = set([n_base])
    offsets = []
    for dn in range(-20, 21, 2):
        if dn != 0:
            offsets.append(dn)
    rng.shuffle(offsets)
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_base + dn
        if n_try < 60 or n_try > 180:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _lsep_optimize(h_r, sub_dl, max_total_iters=300)
        if time.time() < d4 - 5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.012)
            h_r, c5_r = _active_descent_v2(h_r, sub_dl2, max_iter=200)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            n_base = n_try
    
    # Phase 5: Half-period symmetry (7% time)
    d5 = t0 + budget_s * 0.92
    if best_n % 2 == 0:
        h5, c5_5 = _half_period_optimize(best_h, d5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            best_n = len(h5)
    
    # Phase 6: Final polish (8% time)
    d6 = t0 + budget_s * 0.96
    h6, c5_6 = _slsqp_polish(best_h, min(d6, time.time() + budget_s * 0.04), max_outer=10)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)
    if time.time() < deadline - 5:
        h6b, c5_6b = _active_descent_v2(best_h, deadline, max_iter=300)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
            best_n = len(h6b)
    
    # Block minimize on best
    if time.time() < deadline - 3:
        h7, c5_7 = _block_minimize(best_h, deadline, block_size=6, max_sweeps=20)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
            best_n = len(h7)
    
    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    return best_h, best_c5, len(best_h)
