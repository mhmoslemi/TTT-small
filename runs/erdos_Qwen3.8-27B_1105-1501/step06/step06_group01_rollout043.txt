import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    if np.all((v >= 0) & (v <= 1)) and abs(float(v.sum()) - target) < 1e-10:
        return v.copy()
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _pair_line_search(h, i, j, dx, S0):
    """Line search for direction d = e_i - e_j (move mass from j to i).
    Returns (best_alpha, best_c5, best_h) where h_new = h + alpha*(e_i - e_j)."""
    n = len(h)
    # Feasible range for alpha: h[i] + alpha >= 0, h[i] + alpha <= 1, h[j] - alpha >= 0, h[j] - alpha <= 1
    alpha_lo = max(-h[i], h[j] - 1.0)
    alpha_hi = min(1.0 - h[i], h[j])
    if alpha_hi - alpha_lo < 1e-15:
        return 0.0, float(np.max(S0)), h
    
    # Compute T = G @ d where d = e_i - e_j
    k_vals = np.arange(2 * n - 1) - (n - 1)
    T = np.zeros(2 * n - 1)
    for m, k in enumerate(k_vals):
        # d(corr[m])/d(h[i]) - d(corr[m])/d(h[j])
        # = [(1-h[i+k])*1_{0<=i+k<n} - h[i-k]*1_{0<=i-k<n}] - [(1-h[j+k])*1_{0<=j+k<n} - h[j-k]*1_{0<=j-k<n}]
        gi = 0.0
        gj = 0.0
        if 0 <= i + k < n:
            gi += (1.0 - h[i + k])
        if 0 <= i - k < n:
            gi -= h[i - k]
        if 0 <= j + k < n:
            gj += (1.0 - h[j + k])
        if 0 <= j - k < n:
            gj -= h[j - k]
        T[m] = dx * (gi - gj)
    
    # C[m] = dx * correlate(d, d, 'full')[m] where d = e_i - e_j
    # = dx * [2*delta_{m,n-1} - delta_{m, n-1+i-j} - delta_{m, n-1+j-i}]
    C = np.zeros(2 * n - 1)
    C[n-1] = 2.0 * dx
    pos1 = n - 1 + (i - j)
    pos2 = n - 1 + (j - i)
    if 0 <= pos1 < 2*n - 1:
        C[pos1] -= dx
    if 0 <= pos2 < 2*n - 1:
        C[pos2] -= dx
    
    # f(alpha) = max(S0 + alpha*T - alpha^2*C)
    # Grid search
    n_grid = 60
    grid = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1)
    best_i = int(np.argmin(obj_grid))
    best_alpha = grid[best_i]
    best_val = obj_grid[best_i]
    
    # Refine with golden section
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = alpha_lo, alpha_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    opt_alpha = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_alpha * T - opt_alpha * opt_alpha * C))
    if opt_val < best_val:
        best_alpha, best_val = opt_alpha, opt_val
    
    return best_alpha, best_val, h

    # Vectorized approach:
    # T[m] = dx * [(1-h[i+k_m])*valid_i_plus - h[i-k_m]*valid_i_minus - (1-h[j+k_m])*valid_j_plus + h[j-k_m]*valid_j_minus]

    ik_valid = (ik >= 0) & (ik < n)
    ik_minus = i - k_vals  # i - k for each lag
    ik_minus_valid = (ik_minus >= 0) & (ik_minus < n)
    jk = j + k_vals
    jk_valid = (jk >= 0) & (jk < n)
    jk_minus = j - k_vals
    jk_minus_valid = (jk_minus >= 0) & (jk_minus < n)

    T_i_minus = np.where(ik_minus_valid, h[np.clip(ik_minus, 0, n-1)], 0.0)
    T_j_plus = np.where(jk_valid, 1.0 - h[np.clip(jk, 0, n-1)], 0.0)
    T_j_minus = np.where(jk_minus_valid, h[np.clip(jk_minus, 0, n-1)], 0.0)

    if 0 <= pos2 < 2*n-1:
        C[pos2] -= dx

    n_grid = 50

    best_alpha = grid[best_i]
    best_val = obj_grid[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = alpha_lo, alpha_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C))

            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C))

            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    opt_alpha = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_alpha * T - opt_alpha*opt_alpha * C))
    if opt_val < best_val:
        best_alpha, best_val = opt_alpha, opt_val
    
    return best_alpha, best_val

        C = dx * np.correlate(d, d, mode='full')
        S0_base = S0
        # Feasible range
        pos = d > 1e-14
        neg = d < -1e-14
        t_lo = -radius
        t_hi = radius
        if np.any(pos):
            t_lo = max(t_lo, float(np.min(-h[pos]/d[pos])))
            t_hi = min(t_hi, float(np.min((1.0-h[pos])/d[pos])))
        if np.any(neg):
            t_lo = max(t_lo, float(np.min((1.0-h[neg])/d[neg])))
            t_hi = min(t_hi, float(np.min(-h[neg]/d[neg])))
        if t_hi - t_lo < 1e-14:
            radius *= 0.7
            stall += 1
            continue
        ng = 50
        grid = np.linspace(t_lo, t_hi, ng)
        vals = S0_base[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:]
        obj = np.max(vals, axis=1)
        bi = int(np.argmin(obj))
        if obj[bi] < best_c5 - 1e-14:
            best_t = grid[bi]
            h = h + best_t * d
            h = _project(h, target)
            S0 = _corr_vals(h)
            cur = float(S0.max())
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            predicted = float(np.max(S0_base + best_t*T - best_t**2*C))
            ratio = cur / max(predicted, 1e-15)
            if ratio < 0.7:
                radius = min(radius*1.1, 0.35)
            elif ratio > 1.5:
                radius *= 0.8
        else:
            radius *= 0.75
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_start(n, rng, method=0):
    if method == 0:
        f1 = rng.integers(1, 7)
        f2 = rng.integers(1, 7)
        p1 = rng.uniform(0, 2*np.pi)
        p2 = rng.uniform(0, 2*np.pi)
        x = np.arange(n)
        h = 0.5 + 0.35*np.sin(2*np.pi*f1*x/n + p1) + 0.25*np.sin(2*np.pi*f2*x/n + p2)
        h = h + rng.normal(0, 0.05, n)
    elif method == 1:
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, rng.integers(1, max(2, n//6)))
            h[pos:pos+bl] = 1.0 if rng.random() < 0.6 else 0.0
            pos += bl
        h = h + rng.normal(0, 0.1, n)
    else:
        h = np.zeros(n)
        for _ in range(5):
            start = rng.integers(0, n)
            length = rng.integers(n//4, n//2)
            h[start:start+length] += 1.0
        h = h + rng.normal(0, 0.15, n)
    return _project(h, n/2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    
    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.35 * np.sin(2*np.pi*3*x/n0 + 0.5) + 0.2*np.sin(2*np.pi*5*x/n0 + 1.2)
    
    h0 = _project(h0, n0/2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    # Phase 1: Intensive pair descent on parent (35% budget)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _pair_descent(best_h, d1, max_iters=300000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: LP optimization (15% budget)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _lp_optimize(best_h, d2, start_radius=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: More pair descent after LP (15% budget)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _pair_descent(best_h, d3, max_iters=300000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: Diverse starts with optimization (15% budget)
    d4 = t0 + budget_s * 0.80
    n_starts = 4
    remaining = max(d4 - time.time(), 1.0)
    per_start = remaining / n_starts
    for si in range(n_starts):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + per_start * 0.8)
        h_s = _diverse_start(n0, rng, method=si % 3)
        h_s, c5_s = _pair_descent(h_s, sub_dl, max_iters=50000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 5: Multi-n exploration (8% budget)
    d5 = t0 + budget_s * 0.88
    tried_n = {len(best_h)}
    offsets = [-4, 4, -6, 6, -8, 8, -12, 12, -16, 16]
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = len(best_h) + dn
        if n_try < 30 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.01)
        h_r, c5_r = _pair_descent(h_r, sub_dl, max_iters=30000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 6: Final polish (remaining)
    h6, c5_6 = _pair_descent(best_h, deadline, max_iters=200000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h)/2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
