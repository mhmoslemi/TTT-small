def _project(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

Actually, I want to be more careful about the smooth minimax. The issue is that as μ → 0, the softmax weights concentrate on the single most active constraint, and the method reduces to optimizing just that one constraint. This can cause the method to "jump" between constraints and oscillate.

A better approach: use a **hybrid** method:
1. Identify the active set (constraints within ε of the max)
2. Average the gradients over the active set (uniform weights, not softmax)
3. This is equivalent to the subgradient of the max function

This is the classical approach for maximization problems. The "subgradient" of max_k c_k(h) at a point where multiple constraints are active is any convex combination of their gradients.

Using the average of active gradients is a natural choice and is equivalent to the softmax approach with μ → 0 (for the active set).

Let me use this approach:
1. Compute all c_k
2. Find the max M
3. Active set = {k : c_k > M - ε} for some small ε (e.g., 1e-3)
4. Direction = -mean of gradients over active set (projected to feasible directions)
5. Line search

This is simpler and more robust than the softmax approach. Let me use this.

The key parameter is ε (the tolerance for "active"). If ε is too small, we might only use 1 constraint and get a step that satisfies it but worsens others. If ε is too large, we average over too many constraints and get a weak direction.

I'll use an adaptive ε: start with ε = 0.005 and decrease it over time.

Let me also add a **momentum** term: instead of using just the current gradient, use an exponential moving average of past gradients. This can help smooth out oscillations.

def _slsqp_local(h0, deadline, max_outer=12):

            K = len(active)
        top_k = np.argsort(-corr)[:K]
        t_init = cur + 5e-5
        x0 = np.concatenate([h, [t_init]])

        def obj(x):
            return x[n]
        def obj_g(x):
            g = np.zeros(n+1)
            g[n] = 1.0
            return g
        def ineq(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0-hx, mode='full') * dx
            return x[n] - c[top_k]
        def ineq_j(x):
            hx = x[:n]
            G = np.zeros((K, n))
            for idx, kk in enumerate(top_k):
                s = int(kk) - (n-1)
                m = n - abs(s)
                if s >= 0:
                    G[idx, :m] += (1.0 - hx[s:s+m])
                    G[idx, s:s+m] += (-hx[:m])
                else:
                    gg = -s
                    G[idx, gg:gg+m] += (1.0 - hx[:m])
                    G[idx, :m] += (-hx[gg:gg+m])
            G *= dx
            J = np.zeros((K, n+1))
            J[:, :n] = -G
            J[:, n] = 1.0
            return J
        def eq(x):
            return np.sum(x[:n]) - target
        def eq_j(x):
            g = np.zeros(n+1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)]*n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq, 'jac': ineq_j},
            {'type': 'eq', 'fun': eq, 'jac': eq_j},
        ]
        res = minimize(obj, x0, method='SLSQP', jac=obj_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-14})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur - 1e-14:
                h = h_new
                cur = c5_new
                if cur < best_c5:
                    best_c5 = cur

    if time.time() < d7 - 2:
        h7b, c5_7b = _active_descent(best_h, d7, tol=0.002, max_iter=500)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()
            best_n = len(h7b)

    # Final SLSQP + project
    h_f, c5_f = _slsqp_local(best_h, deadline, max_outer=10)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(h_f)

    best_h = _project(best_h, len(best_h)/2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    # Verify
    assert len(best_h) % 2 == 0
    assert abs(np.sum(best_h) - len(best_h)/2.0) < 1e-6
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1+1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, len(best_h)/2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
