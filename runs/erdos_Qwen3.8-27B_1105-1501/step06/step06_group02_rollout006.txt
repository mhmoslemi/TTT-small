import numpy as np

from scipy.optimize import linprog

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_G(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            G[k, :m] += (1.0 - h[d:d + m])
            G[k, d:d + m] += -h[:m]
        else:
            q = -d
            m = n - q
            if m <= 0:
                continue
            G[k, q:] += (1.0 - h[:m])
            G[k, :m] += -h[q:q + m]
    return G

def _line_search(h, d, R0, dx):
    n = len(h)
    # Feasible interval
    pos = d > 1e-15
    neg = d < -1e-15
    hi = 1.0
    lo = -1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.min(h[neg] / d[neg])))

    lo = max(lo, -1.0)
    if hi - lo < 1e-14:
        return 0.0, float(np.max(R0) * dx)
    
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    def _f(alpha):
        return float(np.max(R0 + alpha * T - alpha * alpha * C)) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = _f(c1)
    f2 = _f(c2)
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _f(c2)
    opt = (a + b) / 2.0
    val = _f(opt)
    # Check endpoints

    f_hi = _f(hi)
    f_0 = _f(0.0)  # current value
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:

    return opt, val

def _lp_step(h, G, R, radius):
    n = len(h)
    L = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize t
    
    # Inequality: G[i,:] @ v - t <= -c_i where c_i = R[i] - G[i,:] @ h
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = G @ h - R  # = -c_i
    
    # Equality: sum(v) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lb = np.ones(n + 1) * (-np.inf)
    ub = np.ones(n + 1) * (np.inf)
    lb[1:] = np.maximum(0.0, h - radius)
    ub[1:] = np.minimum(1.0, h + radius)
    # t is bounded below by something reasonable
    lb[0] = 0.0  # t >= 0 since R >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lb, ub)), method='highs')
    
    if res.success:
        v = res.x[1:]
        d = v - h
        # Ensure zero sum (should be from equality constraint, but numerical)
        d -= np.mean(d)
        return d
    return None

def _resample(h, n_new):
    n = len(h)

    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def project(v, target):

        if s > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

            x = np.clip(x, 0.0, 1.0)

def _compute_G(h):
    n = len(h)

    G = np.zeros((L, n))
