def _project(v, target):

        return np.ones(n)

def _softmax_grad(h, S, n, tau, top_k):
    M = float(S.max())
    shifted = (S - M) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, idx, n)
    grad = w[idx] @ G
    grad -= grad.mean()
    if np.abs(grad).max() < 1e-12:
        return None
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return None
    return -grad / norm

                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _fast_sa(h_init, deadline, rng, T_init=0.004, T_min=0.000003):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    cooldown = 0.99995
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    S_curr = np.correlate(h, 1.0 - h, mode='full')
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        n_inner = 300
        for _ in range(n_inner):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            S_new = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(np.max(S_new)) * dx
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                S_curr = S_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _fourier_init(n, n_harmonics, rng):
    x = np.arange(n)
    t = 2 * np.pi * x / n
    h = np.full(n, 0.5)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.3) / k
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(k * t + ph)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _multi_pair_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(444 + n)
    stall = 0
    while time.time() < deadline and stall < 60:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 4:
            break
        idx = rng.choice(len(interior), size=min(6, len(interior)), replace=False)
        found = False
        for _tr in range(20):
            i = interior[idx[rng.integers(len(idx))]]
            j = interior[idx[rng.integers(len(idx))]]
            k2 = interior[idx[rng.integers(len(idx))]]
            l = interior[idx[rng.integers(len(idx))]]
            if len({i, j, k2, l}) < 4:
                continue
            signs = [(1, 1, -1, -1), (-1, -1, 1, 1)]
            for s in signs:
                d = np.zeros(n)
                d[i] = s[0]
                d[j] = s[1]
                d[k2] = s[2]
                d[l] = s[3]
                alpha, c5_try = _line_search(h, d, dx, 0.35)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        stall = 0
                        found = True
                    break
            if found:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (8% time)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        for ph in [0.0, 0.8, 1.6, 2.5, 3.8]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(12):
        starts.append(_fourier_init(n0, rng.integers(2, 8), rng))
    for _ in range(10):
        starts.append(_block_init(n0, rng, rng.integers(3, 18)))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main LP optimization (22% time)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.14, K=18, use_smooth=False)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Gradient flow (12% time)
    d3 = t0 + budget_s * 0.42
    h3, c5_3 = _grad_flow(best_h, d3, tau_init=0.006, step_init=0.004)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP with smooth backup (10% time)
    d4 = t0 + budget_s * 0.52
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.07, K=25, use_smooth=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary SA + LP polish (12% time)
    d5 = t0 + budget_s * 0.64
    h5a, c5_5a = _fast_sa(best_h, t0 + budget_s * 0.56, rng)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5:
        h5b, c5_5b = _lp_loop(best_h, d5, radius_init=0.06, K=15)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution (20% time)
    d6 = t0 + budget_s * 0.84
    base_n = len(best_h)
    n_candidates = []
    for dn in range(-20, 25, 2):
        nn = base_n + dn
        if 40 <= nn <= 260 and nn % 2 == 0:
            n_candidates.append(nn)
    # Also try some structurally different sizes
    for nn in [64, 80, 96, 128, 144, 160, 192, 200, 220, 240]:
        if nn % 2 == 0 and 40 <= nn <= 260 and nn not in n_candidates:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    # Sort by proximity to base_n first
    n_candidates.sort(key=lambda x: abs(x - base_n))
    for n_try in n_candidates:
        if time.time() > d6:
            break
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d6, time.time() + budget_s * 0.025)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, K=14)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.02:
                sub_dl2 = min(d6, time.time() + budget_s * 0.01)
                h_r2, c5_r2 = _fast_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                    if time.time() < d6 - budget_s * 0.01:
                        h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + budget_s * 0.008),
                                             radius_init=0.05, K=12)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                base_n = n_try

    # Phase 7: Aggressive perturbation (7% time)
    d7 = t0 + budget_s * 0.91
    scale = 0.025
    for attempt in range(400):
        if time.time() > d7:
            break
        if best_c5 < 0.3804:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.06, K=14)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 8: Pair + multi-pair polish (5% time)
    d8 = t0 + budget_s * 0.96
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=200)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline:
        h9, c5_9 = _multi_pair_refine(best_h, deadline)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final: gradient flow + LP polish
    if time.time() < deadline:
        h_gf, c5_gf = _grad_flow(best_h, min(deadline, time.time() + budget_s * 0.005),
                                 tau_init=0.003, step_init=0.002)
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.03, K=12)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
