    h = (h0 > 0.5).astype(float)
    target = n / 2.0
    # Adjust to get exactly n/2 ones
    n_ones = int(h.sum())
    if n_ones > int(target):
        # Flip some 1s to 0s
        one_idx = np.where(h == 1)[0]
        rng = np.random.default_rng(42)
        perm = rng.permutation(len(one_idx))
        for idx in perm[:n_ones - int(target)]:
            h[one_idx[idx]] = 0
    elif n_ones < int(target):
        zero_idx = np.where(h == 0)[0]
        rng = np.random.default_rng(42)
        perm = rng.permutation(len(zero_idx))
        for idx in perm[:int(target) - n_ones]:
            h[zero_idx[idx]] = 1
    
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    rng = np.random.default_rng(999)
    temp = 0.005
    cooling = 0.9995
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Random swap
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        # Do the swap
        h[i], h[j] = h[j], h[i]
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
        temp *= cooling
    
    return best_h, best_c5

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    return best_h, best_c5

def _active_minimax(h0, deadline, max_iter=1000):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    c5_cur = best_c5
    radius = 0.08
    stall = 0
    rng = np.random.default_rng(555)

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            if radius > 0.02:
                radius *= 0.5
            else:
                h = h + rng.normal(0, 0.01, n)
                h = _project(h, target)
                c5_cur = _exact_c5(h)
                radius = 0.08
                stall = 0
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                continue
            if stall > 120:
                break

        corr = _corr_values(h)
        M = c5_cur
        tol = max(1e-5, M * 0.002)
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:10]
        elif len(active) > 50:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:50]]

        G = _corr_grad_matrix(h)
        G_act = G[active]
        scores_act = corr[active]

        # LP: min t s.t. G_act @ d - t <= -scores_act, sum(d)=0, |d_i| <= radius
        K = len(active)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((K, n + 1))
        A_ub[:, :n] = G_act
        A_ub[:, n] = -1.0
        b_ub = -scores_act
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-radius, -h[j] - 1e-9), min(radius, 1.0 - h[j] + 1e-9)) for j in range(n)]
        bounds.append((None, None))

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n]
            else:
                d = None
        except Exception:
            d = None

        if d is None or np.max(np.abs(d)) < 1e-13:
            # Fallback: negative sum of active gradients
            grad_sum = G_act.sum(axis=0)
            grad_sum -= grad_sum.mean()
            gn = np.linalg.norm(grad_sum)
            if gn < 1e-12:
                stall += 10
                radius *= 0.7
                continue
            d = -grad_sum / gn * radius

        # Exact line search with quadratic model
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C_quad = _hessian_diag(h, d) / dx

        # Determine valid range
        t_lo, t_hi = -radius, radius
        pos = d > 1e-14
        neg = d < -1e-14
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
        t_lo = max(t_lo, -radius)
        t_hi = min(t_hi, radius)

        if t_hi - t_lo < 1e-14:
            stall += 1
            radius *= 0.8
            continue

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = t_lo, t_hi
        c = b - gr * (b - a)
        dpt = a + gr * (b - a)
        f_c = float(np.max(S0 + c * T + c * c * C_quad)) * dx
        f_d = float(np.max(S0 + dpt * T + dpt * dpt * C_quad)) * dx

        for _ in range(50):
            if b - a < 1e-13:
                break
            if f_c < f_d:
                b, f_d = dpt, f_d
                dpt = b - gr * (b - a)
                f_d = float(np.max(S0 + dpt * T + dpt * dpt * C_quad)) * dx
                c = b - gr * (b - a)
                f_c = float(np.max(S0 + c * T + c * c * C_quad)) * dx
            else:
                a, f_c = c, f_c
                c = a + gr * (b - a)
                f_c = float(np.max(S0 + c * T + c * c * C_quad)) * dx
                dpt = b - gr * (b - a)
                f_d = float(np.max(S0 + dpt * T + dpt * dpt * C_quad)) * dx

        best_t = (a + b) / 2.0
        h_new = h + best_t * d
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)

        if c5_new < c5_cur - 1e-14:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            stall = 0
            if abs(best_t) > 0.5 * radius:
                radius = min(radius * 1.05, 0.25)
            else:
                radius = min(radius * 1.02, 0.20)
        else:
            stall += 1
            radius *= 0.7

    return best_h, best_c5

def _binary_sa(h0, deadline, n_iter=100000, block_len=None):
    n = len(h0)
    target = int(n / 2)
    h = (h0 > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones != target:
        if n_ones > target:
            one_idx = np.where(h == 1.0)[0]
            order = np.argsort(-h0[one_idx])
            for idx in order[:n_ones - target]:
                h[one_idx[idx]] = 0.0
        else:
            zero_idx = np.where(h == 0.0)[0]
            order = np.argsort(h0[zero_idx])
            for idx in order[:target - n_ones]:
                h[zero_idx[idx]] = 1.0

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5

    rng = np.random.default_rng(7777)
    temp = 0.003
    cooling = 0.9997
    use_block = block_len is not None

    for it in range(n_iter):
        if time.time() > deadline:
            break
        if use_block:
            L = block_len
            i = rng.integers(0, n - L + 1)
            j = rng.integers(0, n - L + 1)
            if abs(i - j) < L:
                j = (j + L + rng.integers(0, n - 2 * L)) % (n - L + 1)
                if abs(i - j) < L:
                    j = (i + L + rng.integers(0, max(1, n - 2 * L))) % (n - L + 1)
                    if abs(i - j) < L:
                        continue
            h[i:i + L], h[j:j + L] = h[j:j + L].copy(), h[i:i + L].copy()
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]

        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / temp)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            if use_block:
                h[i:i + L], h[j:j + L] = h[j:j + L].copy(), h[i:i + L].copy()
            else:
                h[i], h[j] = h[j], h[i]
        temp *= cooling

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_polish(h0, deadline, max_outer=12):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _corr_values(h)
        K = min(20, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]

        t_init = cur_c5 + 1e-5
        x0 = np.concatenate([h, [t_init]])

        def _obj_fn(x):
            return x[n]

        def _obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def _ineq_fn(x):
            hx = x[:n]
            c = _corr_values(hx)
            return x[n] - c[top_k]

        def _ineq_jac(x):
            hx = x[:n]
            G = _corr_grad_matrix(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J

        def _eq_fn(x):
            return np.sum(x[:n]) - target

        def _eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _ineq_fn, 'jac': _ineq_jac},
            {'type': 'eq', 'fun': _eq_fn, 'jac': _eq_jac},
        ]

        res = minimize(_obj_fn, x0, method='SLSQP', jac=_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-14})

        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            break

    return best_h, best_c5

def _make_initial(n, seed=42):
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * 2.0 * x / n) + 0.15 * np.sin(2 * np.pi * 5.0 * x / n + 0.7)
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _make_initial(n0, seed)

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: LSE descent (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lse_descent(h, d1, tau0=0.06, tau1=0.0005, n_iter=300)
    if c5_1 < best_c5:
        best_c5, best_h, best_n = c5_1, h1, len(h1)

    # Phase 2: Active minimax (20% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _active_minimax(best_h, d2, max_iter=800)
    if c5_2 < best_c5:
        best_c5, best_h, best_n = c5_2, h2, len(h2)

    # Phase 3: SLSQP polish (10% time)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _slsqp_polish(best_h, d3, max_outer=10)
    if c5_3 < best_c5:
        best_c5, best_h, best_n = c5_3, h3, len(h3)

    # Phase 4: Binary SA with single-bit swaps (20% time)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _binary_sa(best_h, d4, n_iter=200000, block_len=None)
    if c5_4 < best_c5:
        best_c5, best_h, best_n = c5_4, h4.copy(), len(h4)

    # Phase 5: Binary SA with block swaps (10% time)
    d5 = t0 + budget_s * 0.80
    h5, c5_5 = _binary_sa(best_h, d5, n_iter=80000, block_len=4)
    if c5_5 < best_c5:
        best_c5, best_h, best_n = c5_5, h5.copy(), len(h5)

    # Phase 6: Multi-resolution search (12% time)
    d6 = t0 + budget_s * 0.92
    offsets = [-6, 6, -4, 4, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = _lse_descent(h_r, sub_dl, tau0=0.04, tau1=0.001, n_iter=80)
        sub_dl2 = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _active_minimax(h_r, sub_dl2, max_iter=300)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final polish (8% time)
    d7 = deadline - 2.0
    h7, c5_7 = _slsqp_polish(best_h, min(d7, time.time() + budget_s * 0.03), max_outer=8)
    if c5_7 < best_c5:
        best_c5, best_h, best_n = c5_7, h7, len(h7)
    if time.time() < d7:
        h7b, c5_7b = _active_minimax(best_h, d7, max_iter=500)
        if c5_7b < best_c5:
            best_c5, best_h, best_n = c5_7b, h7b, len(h7b)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
