def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h, n):
    """Compute the full (2n-1) x n gradient matrix."""
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        k = r - (n - 1)
        g = G[r]
        if k >= 0:
            m = n - k
            g[k:] += (1.0 - h[:m])
            g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            g[:m] += (1.0 - h[d:d+m])
            g[d:] += -h[:m]
    return G

    n = len(h)

    S = np.correlate(h, 1.0 - h, mode='full') * dx

    w = exp_s / exp_s.sum()
    G = _grad_matrix(h, n)
    grad = (w @ G) * dx
    obj = M + np.log(exp_s.sum()) / beta

        obj, grad, w = _smooth_grad(y, beta)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0

For a binary sequence with n/2 ones, the total number of (1,0) pairs at shift k plus the number of (0,1) pairs at shift k equals the number of "transitions" at that shift. And |{i: h[i]=1, h[i+k]=0}| + |{i: h[i]=0, h[i+k]=1}| = number of disagreeing pairs.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 60
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    # Golden section refinement
    a, b = grid[max(0, best_i-1)], grid[min(n_grid-1, best_i+1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    tol = max(0.03 * M, 1e-5)
    near_idx = np.where(S >= M - tol)[0]
    if len(near_idx) < K:
        sorted_idx = np.argsort(-S)
        near_idx = np.concatenate([near_idx, sorted_idx[:K]])
    combined = np.unique(near_idx)
    K_max = 40
    if len(combined) > K_max:
        # Use the top K_max by S value
        order = np.argsort(-S[combined])
        combined = combined[order[:K_max]]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_loop(h, deadline, beta_init=30.0, step_init=0.03):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = step_init
    beta = beta_init
    stall = 0
    v = np.zeros(n)  # momentum
    for it in range(200000):
        if time.time() > deadline:
            break
        S = _corr_scaled(h)
        M = S.max()
        # Softmax weights
        shifted = beta * (S - M)
        shifted = np.clip(shifted, -700, 0)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        # Compute weighted gradient
        top_k = min(50, 2*n-1)
        top_idx = np.argsort(-w)[:top_k]
        G = _grad_rows(h, top_idx, n)
        grad = (w[top_idx] @ G) * dx
        # Subtract mean to maintain sum constraint in the direction
        grad -= grad.mean()
        # Momentum step
        v = 0.9 * v + 0.1 * grad
        d = -v
        # Project onto box + sum constraint
        h_new = h + step * d
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            step = min(step * 1.02, 0.12)
        elif c5_new < _exact_c5(h) + 1e-14:
            h = h_new
            stall = 0
        else:
            stall += 1
            step *= 0.7
            v *= 0.8
            if stall > 50:
                # Reset

                step = 0.03
                stall = 0
            if step < 1e-7:
                # Try increasing beta
                beta = min(beta * 1.5, 500.0)
                step = 0.02
                if beta > 400:
                    break
    return best_h, best_c5

        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    tol = max(0.025 * M, 1e-5)
    near_idx = np.where(S >= M - tol)[0]
    if len(near_idx) < K:
        sorted_idx = np.argsort(-S)
        extra = sorted_idx[:K]
        near_idx = np.unique(np.concatenate([near_idx, extra]))
    combined = np.unique(near_idx)
    K_max = 45
    if len(combined) > K_max:
        order = np.argsort(-S[combined])
        combined = combined[order[:K_max]]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = _corr_scaled(h)
    M = S.max()
    shifted = beta * (S - M)
    shifted = np.clip(shifted, -700, 0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    top_k = min(60, 2 * n - 1)
    top_idx = np.argsort(-w)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad = (w[top_idx] @ G) * dx
    grad -= grad.mean()
    return grad

def _smooth_loop(h, deadline, beta_init=40.0, step_init=0.04):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = step_init
    beta = beta_init
    stall = 0
    v = np.zeros(n)
    for it in range(300000):
        if time.time() > deadline:
            break
        grad = _smooth_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            stall += 5
        else:
            v = 0.85 * v + 0.15 * (grad / gnorm)
            d = -v
            d_norm = np.linalg.norm(d)
            if d_norm > 1e-10:
                d = d / d_norm
                h_new = h + step * d
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                    step = min(step * 1.03, 0.15)
                    beta = min(beta * 1.01, 300.0)
                elif c5_new < _exact_c5(h) + 1e-13:
                    h = h_new
                    stall = 0
                else:
                    stall += 1
                    step *= 0.75
                    v *= 0.8
            else:
                stall += 1
        if stall > 80:
            v = np.zeros(n)
            step = max(0.01, step * 0.5)
            beta = min(beta * 1.8, 400.0)
            stall = 0
            if beta > 350 and step < 0.015:
                break
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.12, K=20, use_smooth=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777 + n)
    while time.time() < deadline:
        if stall > 70:
            if restart_ctr < 15:
                scale = 0.01 + 0.008 * restart_ctr
                h = h + rng.normal(0, scale, n)
                h = _project(h, n / 2.0)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            if use_smooth:
                grad = _smooth_grad(h, 50.0)
                gn = np.linalg.norm(grad)
                if gn > 1e-10:
                    p = -grad / gn * radius
                else:
                    p = None
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.05, 0.30)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.95, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.012:
                radius = 0.05
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    cooldown = 0.9999
    patience = 0
    max_patience = 20000
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 2:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        ones_list = np.where(h > 0.5)[0]
        zeros_list = np.where(h <= 0.5)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        n_trials = min(300, max(50, int(1e7 / (n * n))))
        for _ in range(n_trials):
            if time.time() > deadline + 1:
                break
            i = int(ones_list[rng.integers(len(ones_list))])
            j = int(zeros_list[rng.integers(len(zeros_list))])
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        G = _grad_rows(h, np.arange(2 * n - 1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_cand = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_cand]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks < 2:
        n_blocks = 2
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks - 1, n - 1), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(len(boundaries) - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        tm[i] = 1.0 if c % 2 == 0 else 0.0
    return _project(tm, n / 2.0)

def _rudin_shapiro_init(n):
    rs = np.zeros(n)
    a = 0
    b = 1
    seq = [0.0]
    for i in range(1, n):
        if i % 2 == 0:
            val = seq[i // 2]
        else:
            val = 1.0 - seq[i // 2]
        seq.append(val)
        rs[i] = val
    return _project(rs, n / 2.0)

def _golden_init(n, rng, freqs):
    x = np.arange(n) / n
    h = np.zeros(n)
    for f, ph in freqs:
        h += 0.5 + 0.5 * np.sin(2 * np.pi * f * x + ph)
    h /= len(freqs)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.98

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Diverse multi-start (12% time)
    d1 = t0 + budget_s * 0.12
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
        for ph in [0.0, 0.8, 1.6, 2.4]:
            starts.append(_sine_init(n0, freq, ph))
    starts.append(_thue_morse_init(n0))
    starts.append(_rudin_shapiro_init(n0))
    gr_freqs = [(1, 0.3), (2, 1.1), (3, 2.0)]
    starts.append(_golden_init(n0, rng, gr_freqs))
    gr_freqs2 = [(2, 0.5), (3, 1.7), (5, 2.9)]
    starts.append(_golden_init(n0, rng, gr_freqs2))
    for _ in range(10):
        starts.append(_block_init(n0, rng, int(rng.integers(3, 20))))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.12, K=15)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Smooth gradient optimization (20% time)
    d2 = t0 + budget_s * 0.32
    h2, c5_2 = _smooth_loop(best_h, d2, beta_init=50.0, step_init=0.04)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Main LP optimization (18% time)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.14, K=25)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP with smooth fallback (10% time)
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.08, K=30, use_smooth=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary SA + LP polish (10% time)
    d5 = t0 + budget_s * 0.70
    h5a, c5_5a = _binary_sa(best_h, t0 + budget_s * 0.64, rng)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5:
        h5b, c5_5b = _lp_loop(best_h, d5, radius_init=0.06, K=20)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Pair refinement (8% time)
    d6 = t0 + budget_s * 0.78
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=300)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Multi-resolution search (12% time)
    d7 = t0 + budget_s * 0.90
    base_n = len(best_h)
    tried = set()
    offset_range = list(range(-24, 25, 2))
    for dn in offset_range:
        if time.time() > d7:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d7, time.time() + budget_s * 0.012)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=15)
            if c5_r < best_c5 + 0.001 and time.time() < d7 - budget_s * 0.01:
                sub_dl2 = min(d7, time.time() + budget_s * 0.008)
                h_r, c5_r = _smooth_loop(h_r, sub_dl2, beta_init=40.0, step_init=0.03)
            if c5_r < best_c5 and time.time() < d7 - budget_s * 0.01:
                sub_dl3 = min(d7, time.time() + budget_s * 0.01)
                h_r, c5_r = _lp_loop(h_r, sub_dl3, radius_init=0.05, K=15)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Iterated local search with perturbation (5% time)
    d8 = t0 + budget_s * 0.955
    scale = 0.02
    for attempt in range(500):
        if time.time() > d8:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d8, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=15)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 9: Final polish (2.5% time)
    d9 = deadline
    h9, c5_9 = _pair_refine(best_h, d9, max_passes=80)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()
    if time.time() < deadline:
        hf, cf = _lp_loop(best_h, deadline, radius_init=0.03, K=12)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
