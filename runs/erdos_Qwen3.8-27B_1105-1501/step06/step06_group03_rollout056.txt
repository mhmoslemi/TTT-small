- For m < 0 (let d = -m): vector with entries (1-h[0:n-d]) in positions [d, n) and -h[d:n] in positions [0, n-d)

def _softmax_gradient_fast(h, tau):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    S_shifted = (S - S.max()) / tau
    w = np.exp(S_shifted)
    w_sum = w.sum()
    p = w / w_sum
    grad = np.zeros(n)
    # m = 0
    grad += p[n-1] * (1.0 - 2.0 * h)
    # m > 0
    for m in range(1, n):
        pm = p[n-1+m]
        grad[:n-m] += pm * (1.0 - h[m:])
        grad[m:] += -pm * h[:n-m]
    # m < 0 (d = -m > 0)
    for d in range(1, n):
        pd = p[n-1-d]
        grad[d:] += pd * (1.0 - h[:n-d])
        grad[:n-d] += -pd * h[d:]
    obj = tau * np.log(w_sum) + S.max()
    return grad, obj

import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _softmax_grad(h, tau):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    S_shifted = (S - S.max()) / tau
    w = np.exp(S_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n), 0.0
    p = w / w_sum
    grad = np.zeros(n)
    grad += p[n-1] * (1.0 - 2.0 * h)
    for m in range(1, n):
        pm = p[n-1+m]
        grad[:n-m] += pm * (1.0 - h[m:])
        grad[m:] += -pm * h[:n-m]
    for d in range(1, n):
        pd = p[n-1-d]
        grad[d:] += pd * (1.0 - h[:n-d])
        grad[:n-d] += -pd * h[d:]
    obj = tau * np.log(w_sum) + S.max()
    return grad, obj

def _softmax_gd(h, deadline, tau_init=0.03, tau_end=0.002, momentum=0.9, step_init=0.02):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_obj = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_obj
    v = np.zeros(n)
    step = step_init
    tau = tau_init
    tau_steps_total = 0
    step_reduce = 0.8
    while time.time() < deadline:
        tau_steps_total += 1
        if tau_steps_total % 30 == 0 and tau > tau_end * 2:
            tau = max(tau_end, tau * 0.92)
        for _inner in range(50):
            if time.time() > deadline:
                break
            g, _ = _softmax_grad(h, tau)
            g = g * dx
            v = momentum * v - step * g
            h_new = h + v
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_obj - 1e-14:
                h = h_new
                cur_obj = new_c5
                if cur_obj < best_c5:
                    best_c5 = cur_obj
                    best_h = h.copy()
            else:
                step *= step_reduce
                v = np.zeros(n)
                if step < 1e-6:
                    break
        if step < 1e-7:
            step = step_init * 0.5
    return best_h, best_c5

def _lbfgs_polish(h, deadline, tau=0.001):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    
    def objective(x):
        S = np.correlate(x, 1.0 - x, mode='full')
        S_shifted = (S - S.max()) / tau
        w = np.exp(S_shifted)
        return (tau * np.log(w.sum()) + S.max()) * dx + 100.0 * (np.sum(x) - target)**2
    
    def gradient(x):
        g, _ = _softmax_grad(x, tau)
        g = g * dx
        g += 200.0 * (np.sum(x) - target)
        return g
    
    try:
        res = minimize(objective, h.copy(), jac=gradient, method='L-BFGS-B',
                       bounds=[(0,1)]*n, options={'maxiter': 200, 'gtol': 1e-8,
                       'max_time': max(1, deadline - time.time())})
        h_new = _project(res.x, target)
        c5_new = _exact_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, _exact_c5(h)

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.05 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, K=15, use_smooth=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    tau = 0.02 * cur_c5
    while time.time() < deadline:
        if stall > 50:
            if restart_ctr < 8:
                scale = 0.01 + 0.008 * restart_ctr
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                tau = 0.02 * cur_c5
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if (p is None or float(np.max(np.abs(p))) < 1e-13) and use_smooth:
            M = float(S.max())
            shifted = (S - M) / tau
            shifted -= shifted.max()
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum > 1e-300:
                w = w / w_sum
                top_k = min(20, 2 * n - 1)
                top_idx = np.argsort(-S)[:top_k]
                G = _grad_rows(h, top_idx, n)
                grad_smooth = w[top_idx] @ G
                grad_smooth -= grad_smooth.mean()
                norm = np.linalg.norm(grad_smooth)
                if norm > 1e-12:
                    p = -grad_smooth / norm * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if use_smooth and stall % 5 == 0:
                tau *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.08, 0.30)
            elif ratio > 0.3:
                radius = min(radius * 1.03, 0.25)
            else:
                radius = max(radius * 0.93, 0.015)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(888)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 80)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _fw_away(h, deadline, tau=0.01, max_iter=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g, _ = _softmax_grad(h, tau)
        g = g * (2.0 / n)
        # FW direction: min g^T v s.t. v in hypersimplex
        order = np.argsort(g)
        v_fw = np.zeros(n)
        v_fw[order[:n//2]] = 1.0
        # Away direction: vertex most "in the way"
        # Use the vertex that agrees with h on positions where h is most extreme
        h_rounded = (h >= 0.5).astype(float)
        if h_rounded.sum() != n // 2:
            indices = np.argsort(-np.abs(h - 0.5))
            if h_rounded.sum() > n // 2:
                h_rounded[indices[:int(h_rounded.sum() - n//2)]] = 0
            else:
                h_rounded[indices[:int(n//2 - h_rounded.sum())]] = 1
        d_fw = v_fw - h
        d_away = h - h_rounded
        # Try FW step
        step = 2.0 / (it + 2)
        h_new = (1 - step) * h + step * v_fw
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        # Try away step
        h_away = h + step * d_away
        h_away = _project(h_away, target)
        c5_away = _exact_c5(h_away)
        if c5_away < c5_new and c5_away < cur_c5:
            h_new, c5_new = h_away, c5_away
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

            if time.time() > deadline:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                h[i], h[j] = h[j], h[i]
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    perspective = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)
    
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    # Phase 1: Soft-max GD (15% time)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _softmax_gd(best_h, d1, tau_init=0.03, tau_end=0.003, momentum=0.85, step_init=0.015)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: LP refinement (25% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.14, K=18, use_smooth=True)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Soft-max with small tau + L-BFGS (15% time)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _softmax_gd(best_h, min(d3, t0 + budget_s * 0.50), tau_init=0.005, tau_end=0.001, momentum=0.9, step_init=0.008)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3:
        h3b, c5_3b = _lbfgs_polish(best_h, d3, tau=0.001)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()
    
    # Phase 4: FW with away steps (10% time)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _fw_away(best_h, d4, tau=0.005)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Multi-start exploration (12% time)
    d5 = t0 + budget_s * 0.77
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
        for ph in [0.0, 0.5, 1.0, 1.5, 2.0]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(12):
        starts.append(_block_init(n0, rng, rng.integers(3, 20)))
    # Add perturbed versions of best
    for sc in [0.02, 0.04, 0.06, 0.08, 0.12, 0.15]:
        starts.append(_project(best_h + rng.normal(0, sc, n0), n0/2.0))
    
    for s in starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        hs, c5s = _softmax_gd(s, sub_dl, tau_init=0.02, tau_end=0.003, momentum=0.85, step_init=0.01)
        if c5s < best_c5 - 1e-6 and time.time() < d5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.015)
            hs, c5s = _lp_loop(hs, sub_dl2, radius_init=0.08, K=15)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = hs.copy()
    
    # Phase 6: Binary SA + LP (8% time)
    d6 = t0 + budget_s * 0.85
    h6a, c5_6a = _binary_sa(best_h, min(d6, t0 + budget_s * 0.79), rng)
    if c5_6a < best_c5:
        best_c5 = c5_6a
        best_h = h6a.copy()
    # Unfreeze: take binary solution, add noise, re-optimize
    if time.time() < d6:
        h_unfreeze = best_h + rng.normal(0, 0.15, n0)
        h_unfreeze = _project(h_unfreeze, n0/2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.04)
        h6b, c5_6b = _lp_loop(h_unfreeze, sub_dl, radius_init=0.12, K=15, use_smooth=True)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
    
    # Phase 7: Multi-resolution (8% time)
    d7 = t0 + budget_s * 0.93
    base_n = len(best_h)
    tried = set()
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18, -24, 24]:
        if time.time() > d7:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.003:
            sub_dl = min(d7, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=15)
            if c5_r < best_c5 and time.time() < d7 - budget_s * 0.01:
                sub_dl2 = min(d7, time.time() + budget_s * 0.01)
                h_r2, c5_r2 = _softmax_gd(h_r, sub_dl2, tau_init=0.003, tau_end=0.001, step_init=0.005)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 8: Final LP + pair polish (7% time)
    d8 = t0 + budget_s * 0.975
    h8, c5_8 = _lp_loop(best_h, min(d8, t0 + budget_s * 0.94), radius_init=0.06, K=20)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8:
        h8b, c5_8b = _pair_refine(best_h, d8, max_passes=80)
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()
    
    # Final L-BFGS polish
    if time.time() < perspective:
        hf, c5f = _lbfgs_polish(best_h, perspective, tau=0.0005)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
