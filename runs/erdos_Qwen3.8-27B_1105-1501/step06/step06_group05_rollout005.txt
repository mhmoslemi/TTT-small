    raw_max = float(np.max(corr))

        n_active = n_lags

    # For lag k (output index r = n-1+k), correlation = sum h[i+k]*(1-h[i]) for valid i
    # Gradient wrt h[j]:
    #   ∂/∂h[j] of sum_i h[i+k]*(1-h[i]) = contribution when j = i+k (coefficient 1-h[i]) + contribution when j = i (coefficient -h[i+k])
    
    # Build gradient matrix G of shape (n_lags, n)
    G = np.zeros((n_lags, n))
    for r in range(n_lags):
        k = r - (n - 1)
        if k >= 0:
            nn = range(0, n - k)
        else:
            nn = range(-k, n)
        for i in nn:
            a = i + k  # index in h that gets multiplied by (1-h[i])
            b = i      # index in h that gets 1-h
            # term is h[a] * (1-h[b])
            # d/dh[a]: (1-h[b])
            # d/dh[b]: -h[a]
            G[r, a] += (1.0 - h[b])
            G[r, b] += (-h[a])

    idx = np.arange(n_lags)

    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Constraints:
    #   G[idx, j] * delta_j - t <= -corr[idx] for each active lag
    #   -h[i] <= delta[i] <= 1-h[i]
    #   sum(delta) = 0
    #   |delta[i]| <= trust_radius (optional for stability)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag: G[r,:] @ delta - t <= -corr[r]
    A_rows = []
    b_rows = []
    for r in idx:
        row = np.zeros(n + 1)
        row[:n] = G[r, :]
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-corr[r])
    
    # Box on delta: delta[i] >= -h[i] -> -delta[i] <= h[i]
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(h[i])
        
        row2 = np.zeros(n + 1)
        row2[i] = 1.0
        A_rows.append(row2)
        b_rows.append(1.0 - h[i])
    
    # Trust radius
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(trust_radius)
        
        row2 = np.zeros(n + 1)
        row2[i] = 1.0
        A_rows.append(row2)
        b_rows.append(trust_radius)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * n + [(0.0, raw_max + 1.0)]
    # Actually t should be >= 0 and we're minimizing it
    bounds[-1] = (0.0, None)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, raw_max
    
    delta = res.x[:n]
    t_new = res.x[-1]
    
    return delta, raw_max

    n = len(v)
    if target <= 0:
        return np.zeros(n)

def _build_grad_matrix(h):
    """Build the full gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):

            for i in range(0, n - k):
                a = i + k
                b = i
                G[r, a] += (1.0 - h[b])
                G[r, b] += (-h[a])
        else:
            for i in range(-k, n):
                a = i + k
                b = i
                G[r, a] += (1.0 - h[b])
                G[r, b] += (-h[a])
    return G

    n = len(h)
    n_lags = 2 * n - 1

    # Select active lags

        sorted_idx = np.argsort(-corr)
        active = sorted_idx[:k_active]
    else:
        active = np.arange(n_lags)
    
    # Build LP
    nv = n + 1  # delta[0..n-1], t
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0
    
    A_rows = []
    b_rows = []
    
    for r in active:
        row = np.zeros(nv)
        row[:n] = G[r, :]
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corr[r])
    
    for i in range(n):
        # -delta[i] <= h[i]
        row = np.zeros(nv)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(h[i])
        # delta[i] <= 1 - h[i]
        row2 = np.zeros(nv)
        row2[i] = 1.0
        A_rows.append(row2)
        b_rows.append(1.0 - h[i])

    b_ub = np.array(b_rows)

import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    for _ in range(20):
        res = target - float(np.sum(x))
        if abs(res) < 1e-14:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _build_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    c = n - 1
    G[c, :] = 1.0 - 2.0 * h
    for k in range(1, n):
        m = n - k
        G[c + k, k:] += (1.0 - h[:m])
        G[c + k, :m] += (-h[k:])
        G[c - k, :m] += (1.0 - h[k:])
        G[c - k, k:] += (-h[:m])
    return G

def _slp_once(h, k_top, use_all):
    n = len(h)
    n_lags = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    raw_max = float(np.max(corr))
    
    if use_all:
        active = np.arange(n_lags)
    else:
        active = np.argsort(-corr)[:k_top]
    
    G = _build_grad(h)
    G_act = G[active, :]
    
    # LP: min t s.t. G_act @ delta - t <= -corr[active]
    #      -h[i] <= delta[i] <= 1-h[i]
    #      sum(delta) = 0
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    
    # Build A_ub, b_ub
    n_act = len(active)
    A_parts = []
    b_parts = []
    
    # Lag constraints: G_act[r,:] @ delta - t <= -corr[active[r]]
    A_lag = np.hstack([G_act, -np.ones((n_act, 1))])
    A_parts.append(A_lag)
    b_parts.append(-corr[active])
    
    # Box: -delta[i] <= h[i]
    I = np.eye(n)
    A_parts.append(-I)
    b_parts.append(h)
    # delta[i] <= 1-h[i]
    A_parts.append(I)
    b_parts.append(1.0 - h)
    
    A_ub = np.vstack(A_parts)
    b_ub = np.concatenate(b_parts)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i], 1.0 - h[i])] * n + [(0.0, raw_max + 1.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, raw_max, corr
    return res.x[:n].copy(), raw_max, corr

def _slp_optimize(h, deadline, rng, k_top=80, use_all=False, max_iter=50000):

    target = n / 2.0

    cur_f = _exact_c5(h)

    best_f = cur_f
    stall = 0
    it = 0

            if stall > 30:

                stall = 0
            continue

        if norm < 1e-12:
            stall += 1
            if stall > 30:
                h = _project(best_h + rng.normal(0, 0.1, n), target)
                cur_f = _exact_c5(h)
                stall = 0
            continue
        
        # Line search: try different step sizes
        best_alpha = 0
        best_f_try = cur_f
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_try = h + alpha * delta
            if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:

            f_try = _exact_c5(h_try)
            if f_try < best_f_try:

                best_alpha = alpha

                stall = 0

        else:

            if abs(s - target) > 1e-10:
                h = _project(h, target)
        
        cur_f = _exact_c5(h)
        if cur_f < best_f - 1e-13:
            best_f = cur_f
            best_h = h.copy()
        stall = 0
    
    return best_h, best_f

def _pair_polish(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()

        j = int(rng.integers(n))

        if amt < 1e-10:

        h_try = h.copy()
        h_try[i] += amt
        h_try[j] -= amt
        f_try = _exact_c5(h_try)

            cur_f = f_try
            stall = 0
            if f_try < best_f:

                best_h = h.copy()

                if f_try2 < best_f:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
