def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

    return x

def _grad_row(h, k, n):
    """Gradient of q_k(h) = dx * sum h[i+k]*(1-h[i]) for k>=0, or h[i]*(1-h[i+d]) for k<0."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += (1.0 - h[:m]) * dx
            g[:m] += -h[k:] * dx
    else:
        d = -k
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d+m]) * dx
            g[d:] += -h[:m] * dx
    return g

def _soft_grad(h, n, mu):
    """Compute soft-max weighted gradient."""
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft-max weights
    S_shifted = S - S.max()
    w = np.exp(S_shifted / mu)
    w_sum = w.sum()

        return None
    w = w / w_sum
    # Only compute gradients for significantly weighted lags

            continue
        g += w[idx] * _grad_row(h, idx - (n-1), n)
    g -= g.mean()  # maintain zero-sum (preserves integral constraint direction)
    return g

def _line_search(h, d, dx, trust_radius):

    neg = d < -1e-15

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:k+m] += (1.0 - h[:m]) * dx
        g[:m] += -h[k:k+m] * dx
    else:
        d = -k
        m = n - d
        if m <= 0:
            return g
        g[:m] += (1.0 - h[d:d+m]) * dx
        g[d:d+m] += -h[:m] * dx
    return g

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo, t_hi = 0.0, 0.0
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _soft_grad(h, n, mu):
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    S_shifted = S - S.max()
    w = np.exp(S_shifted / mu)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_k = min(35, 2 * n - 1)
    top_idx = np.argsort(-w)[:top_k]
    g = np.zeros(n)
    for idx in top_idx:
        if w[idx] < 1e-8:
            continue
        g += w[idx] * _grad_row(h, idx - (n - 1), n)
    g -= g.mean()
    return g

def _lp_direction(h, S_scaled, n, K, radius):
    M = float(S_scaled.max())
    sorted_idx = np.argsort(-S_scaled)
    active = sorted_idx[:K]
    tol = max(0.02 * M, 1e-6)
    near_idx = np.where(S_scaled >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 22:
        combined = sorted_idx[:22]
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    for idx, r in enumerate(combined):
        k = r - (n - 1)
        g = _grad_row(h, k, n)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
    b_ub = np.zeros(n_act)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j])
        hi_b = min(radius, 1.0 - h[j])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and res.x[:n].max() > 1e-14:
            return res.x[:n]
    except Exception:
        pass
    return None

def _optimize(h, deadline, mu_init=0.02, mu_min=0.001, radius_init=0.12):

    h = h.copy()

    radius = radius_init
    stall = 0
    mu = mu_init
    rng = np.random.default_rng(999)
    restarts = 0
    while time.time() < deadline:
        S_scaled = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        p = _lp_direction(h, S_scaled, n, K=16, radius=radius)
        if p is None or np.max(np.abs(p)) < 1e-14:

            if g is not None and np.linalg.norm(g) > 1e-12:

                    cur_c5 = _exact_c5(h)
                    radius = 0.10
                    mu = mu_init
                    stall = 0
                    restarts += 1
                    continue
                else:
                    break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)

            h = _project(h, target)

            stall = 0

            else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        all_idx = np.arange(2 * n - 1)
        G = np.zeros((2 * n - 1, n))
        for r in all_idx:
            G[r] = _grad_row(h, r - (n - 1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 55)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, min(ii + 18, len(candidates))):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)

                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()

    return best_h, best_c5

def _binary_hill(h_init, deadline, seed):

    target = n // 2
    sorted_idx = np.argsort(-h_init)

    h[sorted_idx[:target]] = 1.0

    best_h = h.copy()
    best_c5 = cur_c5

    improved = True
    while improved and time.time() < deadline:
