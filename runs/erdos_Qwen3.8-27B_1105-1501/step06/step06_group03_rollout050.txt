    S = _corr_scaled(h)
    # log-sum-exp for numerical stability
    M = S.max()
    shifted = (S - M) / tau
    shifted = shifted - shifted.max()  # additional stability
    w = np.exp(shifted)

                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.03 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 35:
        combined = sorted_idx[:35]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_grad_direction(h, n, tau):
    S = _corr_scaled(h)
    M = float(S.max())
    shifted = (S - M) / tau
    shifted = shifted - shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    all_idx = np.arange(2 * n - 1)
    G = _grad_rows(h, all_idx, n)
    grad = w @ G
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return None
    return -grad / norm

def _lp_loop(h, deadline, radius_init=0.12, K=15, use_smooth=False, tau_init=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    if tau_init is not None:
        tau = tau_init
    else:
        tau = 0.015 * cur_c5
    while time.time() < deadline:
        if stall > 80:
            if restart_ctr < 15:
                scale = 0.008 + 0.005 * restart_ctr
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.06
                stall = 0
                restart_ctr += 1
                tau = 0.015 * cur_c5
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            if use_smooth:
                p = _smooth_grad_direction(h, n, tau)
                if p is not None:
                    p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.70
            if use_smooth and stall % 4 == 0:
                tau *= 0.85
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.8:
                radius = min(radius * 1.08, 0.30)
            elif ratio > 0.4:
                radius = min(radius * 1.03, 0.25)
            else:
                radius = max(radius * 0.93, 0.01)
        else:
            stall += 1
            radius *= 0.72
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 80)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _quadruple_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(444)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 40)
        pos_cand = np.where(total_grad < -1e-10)[0]
        neg_cand = np.where(total_grad > 1e-10)[0]
        if len(pos_cand) < 2 or len(neg_cand) < 2:
            break
        pos_cand = pos_cand[np.argsort(total_grad[pos_cand])[:10]]
        neg_cand = neg_cand[np.argsort(-total_grad[neg_cand])[:10]]
        for ii in range(min(6, len(pos_cand))):
            if time.time() > deadline:
                break
            i = pos_cand[ii]
            for jj in range(min(6, len(neg_cand))):
                j = neg_cand[jj]
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.3)
                if c5_try < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.000003
    cooldown = 0.99992
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        for _ in range(300):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _greedy_binary(n, rng):
    h = np.zeros(n)
    target = n // 2
    placed = 0
    dx = 2.0 / n
    for _ in range(target):
        best_pos = -1
        best_c5 = 1e10
        candidates = np.where(h == 0)[0]
        if len(candidates) == 0:
            break
        try_n = min(20, len(candidates))
        test_indices = candidates[rng.choice(len(candidates), size=try_n, replace=False)]
        for pos in test_indices:
            h[pos] = 1.0
            c5_val = _exact_c5(h)
            h[pos] = 0.0
            if c5_val < best_c5:
                best_c5 = c5_val
                best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
            placed += 1
    return h

def _structured_init(n, period, offset=0):
    h = np.zeros(n)
    block_len = max(1, period // 2)
    for i in range(n):
        if ((i + offset) % period) < block_len:
            h[i] = 1.0
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start exploration (10% time)
    d1 = t0 + budget_s * 0.10
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
        for ph in [0.0, 0.7, 1.5, 2.3, 3.1]:
            starts.append(_sine_init(n0, freq, ph))
    for period in [4, 6, 8, 10, 12]:
        for off in [0, 1, 2]:
            starts.append(_structured_init(n0, period, off))
    for _ in range(10):
        starts.append(_block_init(n0, rng, rng.integers(3, 20)))
    starts.append(_greedy_binary(n0, rng))

    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.007)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=14)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main LP optimization with large K (22% time)
    d2 = t0 + budget_s * 0.32
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.15, K=25, use_smooth=False)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP with smooth gradient backup (13% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.10, K=30, use_smooth=True, tau_init=0.012 * best_c5)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP polish (12% time)
    d4 = t0 + budget_s * 0.57
    h4a, c5_4a = _binary_sa(best_h, t0 + budget_s * 0.50, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, radius_init=0.08, K=20)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Gradient-guided pair refinement (8% time)
    d5 = t0 + budget_s * 0.65
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 5b: Quadruple refinement
    if time.time() < d5 + budget_s * 0.03:
        d5b = t0 + budget_s * 0.68
        h5b, c5_5b = _quadruple_refine(best_h, d5b, max_passes=100)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution (18% time)
    d6 = t0 + budget_s * 0.86
    base_n = len(best_h)
    tried = set()
    offsets = list(range(-20, 25, 2))
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d6, time.time() + budget_s * 0.025)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, K=18)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.02:
                sub_dl2 = min(d6, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d6:
                    h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + budget_s * 0.012),
                                         radius_init=0.06, K=15)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Aggressive perturbation + re-opt (7% time)
    d7 = t0 + budget_s * 0.93
    scale = 0.025
    for attempt in range(500):
        if time.time() > d7:
            break
        if best_c5 < 0.3800:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=15)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 8: Final intensive polish (5% time)
    d8 = t0 + budget_s * 0.97
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=500)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final LP polish with very small steps
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.02, K=30, use_smooth=True,
                              tau_init=0.008 * best_c5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
