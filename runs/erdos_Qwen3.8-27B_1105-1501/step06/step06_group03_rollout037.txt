    if target >= n: return np.ones(n)

        if s > target: lo = mid
        else: hi = mid

    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0: mask = x < 0.999
        else: mask = x > 0.001
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

    """Compute gradients for given correlation indices (0 to 2n-2)."""

                g[d:n] += (1.0 - h[0:m])

    best_t = grid[best_i]
    best_val = obj[best_i]

def _block_descent(h, deadline, block_size=6, max_sweeps=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        # Shuffle block order
        np.random.seed(sweep * 1000 + 42)
        order = np.random.permutation(n // block_size)
        
        for block_idx in order:
            if time.time() > deadline:
                break
            # Get block indices
            start = block_idx * block_size
            end = min(start + block_size, n)
            # Skip if block is at the edge and too small
            if end - start < 2:
                continue
            
            block_vals = h[start:end].copy()
            block_sum = float(np.sum(block_vals))
            
            # Define objective for block optimization
            def block_obj(u):
                h_trial = h.copy()
                h_trial[start:end] = u
                # Enforce sum constraint within block
                diff = block_sum - float(np.sum(u))
                if abs(diff) > 1e-14:
                    u_adj = u + diff / (end - start)
                    h_trial[start:end] = u_adj
                # Enforce bounds
                h_trial[start:end] = np.clip(h_trial[start:end], 0, 1)
                # Final projection to ensure exact sum
                h_trial = _project(h_trial, target)
                return _exact_c5(h_trial)
            
            # Optimize
            res = minimize(block_obj, block_vals, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8,
                                   'adaptive': True})
            
            if res.fun < best_c5 - 1e-12:
                h_new = h.copy()
                h_new[start:end] = res.x
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

    M = float(S.max())
    top_K = np.argsort(-S)[:K]
    tol = 0.15 * (M - float(S.min()))
    near = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([top_K, near]))
    max_rows = min(30, len(combined))
    if len(combined) > max_rows:
        combined = np.argsort(-S[combined])
        combined = combined[combined[:max_rows]]
    G = _grad_batch(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined] + 1e-10
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-9)
        hi_b = min(radius, 1.0 - h[j] - 1e-9)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-12:
                return p
    except Exception:
        pass
    return None

def _softmax_dir(h, S, n, beta, step_scale):
    M = float(S.max())
    shifted = beta * (S - M)
    shifted = np.clip(shifted, -400, 400)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    active = np.where(w > 0.005)[0]
    if len(active) < 3:
        active = np.argsort(-S)[:10]
    G = _grad_batch(h, active, n)
    g = w[active] @ G
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm * step_scale

def _local_opt(h, deadline, radius_init=0.10, K=15, use_softmax=True, rng=None):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    stall = 0
    beta = 40.0
    v_mom = np.zeros(n)
    momentum = 0.6
    if rng is None:
        rng = np.random.default_rng(12345)
    while time.time() < deadline:
        S = _corr_scaled(h)
        M = float(S.max())
        directions = []
        p_lp = _lp_direction(h, S, n, K, radius, rng)
        if p_lp is not None:
            directions.append(p_lp)
        if use_softmax:
            p_sm = _softmax_dir(h, S, n, beta, radius)
            if p_sm is not None:
                directions.append(p_sm)
        if stall > 10:
            p_rand = rng.standard_normal(n) * radius
            p_rand -= p_rand.mean()
            directions.append(p_rand)
        best_step_c5 = best_c5
        best_step_h = h.copy()
        best_step_alpha = 0.0
        best_step_d = None
        for d in directions:
            v_mom = momentum * v_mom + (1 - momentum) * d
            v_mom = v_mom / max(np.linalg.norm(v_mom), 1e-14) * radius
            alpha, c5_try = _line_search(h, v_mom, dx, radius * 2.0)
            if c5_try < best_step_c5:
                best_step_c5 = c5_try
                best_step_alpha = alpha
                best_step_d = v_mom.copy()
                h_trial = h + alpha * v_mom
                h_trial = _project(h_trial, target)
                best_step_h = h_trial
        if best_step_d is not None and best_step_c5 < best_c5 - 1e-14:
            h = best_step_h
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            if abs(best_step_alpha) > radius * 0.7:
                radius = min(radius * 1.05, 0.22)
            beta = min(beta * 1.02, 80.0)
        else:
            stall += 1
            radius *= 0.82
            beta *= 0.85
            v_mom *= 0.5
            if radius < 0.005:
                if stall > 80:
                    break
                h_pert = h + rng.normal(0, 0.02, n)
                h = _project(h_pert, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5 + 0.002:
                    radius = 0.06
                    stall = 0
                    beta = 30.0
                else:
                    h = best_h.copy()
                    if stall > 120:
                        break
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_batch(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_cand = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_cand]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.35)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _triple_refine(h, deadline, max_iters=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 3:
            break
        idx = rng.choice(len(interior), size=3, replace=False)
        i, j, k = interior[idx]
        for signs in [(1, 1, -2), (1, -1, 0), (-1, -1, 2), (2, -1, -1)]:
            if signs[2] == 0:
                continue
            d = np.zeros(n)
            d[i] = signs[0]
            d[j] = signs[1]
            d[k] = signs[2]
            if abs(float(d.sum())) > 1e-10:
                continue
            alpha, c5_try = _line_search(h, d, dx, 0.2)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                break
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n or n_new < 4:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks <= 1:
        h[:] = 1.0 / 2.0
        return _project(h, n / 2.0)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.99

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    if n0 < 4:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (8%)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        for ph in [0.0, 1.0, 2.0, 3.0]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(10):
        starts.append(_block_init(n0, rng, int(rng.integers(3, 20))))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.007)
        h_s, c5_s = _local_opt(s, sub_dl, radius_init=0.10, K=12,
                               use_softmax=True, rng=rng)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main optimization - mixed directions (30%)
    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _local_opt(best_h, d2, radius_init=0.12, K=18,
                          use_softmax=True, rng=rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Focused optimization with tighter params (18%)
    d3 = t0 + budget_s * 0.56
    h3, c5_3 = _local_opt(best_h, d3, radius_init=0.07, K=25,
                          use_softmax=True, rng=rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair refinement (12%)
    d4 = t0 + budget_s * 0.68
    h4, c5_4 = _pair_refine(best_h, d4, max_passes=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Triple refine + LP polish (8%)
    d5 = t0 + budget_s * 0.76
    h5a, c5_5a = _triple_refine(best_h, min(d5, t0 + budget_s * 0.72), max_iters=500)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5:
        h5b, c5_5b = _local_opt(best_h, d5, radius_init=0.04, K=20,
                                use_softmax=False, rng=rng)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Resolution sweep (15%)
    d6 = t0 + budget_s * 0.91
    base_n = len(best_h)
    tried = set()
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, 24, -24, 28, -28]:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d6, time.time() + budget_s * 0.018)
            h_r, c5_r = _local_opt(h_r, sub_dl, radius_init=0.08, K=15,
                                   use_softmax=True, rng=rng)
            if c5_r < best_c5 - 1e-12:
                if time.time() < d6 - budget_s * 0.015:
                    h_r, c5_r2 = _pair_refine(h_r, min(d6, time.time() + budget_s * 0.01), max_passes=50)
                    if c5_r2 < c5_r:
                        c5_r = c5_r2
                        h_r = h_r.copy()
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 7: Aggressive perturbation restart (5%)
    d7 = t0 + budget_s * 0.96
    scale = 0.02
    for attempt in range(400):
        if time.time() > d7:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _local_opt(h_pert, sub_dl, radius_init=0.06, K=15,
                               use_softmax=True, rng=rng)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.7)
        else:
            scale = min(0.10, scale * 1.3)

    # Phase 8: Final polish (4%)
    d8 = deadline
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=100)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline - 1:
        h9, c5_9 = _local_opt(best_h, deadline, radius_init=0.03, K=25,
                              use_softmax=False, rng=rng)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
