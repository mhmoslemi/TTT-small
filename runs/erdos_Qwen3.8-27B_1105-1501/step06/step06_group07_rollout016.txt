def _sigmoid(x):

def _solve_lambda(u, target):
    lo, hi = -60.0, 60.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = _sigmoid(u + mid).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0


def _u_to_h(u, target):

    return _sigmoid(u + lam)


def _h_to_u(h, target):

    u = np.log(h_safe / (1.0 - h_safe))
    lam = _solve_lambda(u, target)
    return u - lam


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        if s > target:

        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

def _exact_c5(h):

def _corr_gradient(h, weights):
    n = len(h)
    G = np.zeros(n)
    m = 2 * n - 1
    for r in range(m):
        if abs(weights[r]) < 1e-16:
            continue
        k = r - (n - 1)
        w = weights[r]
        if k >= 0:
            G[k:] += w * (1.0 - h[:n - k])
            G[:n - k] -= w * h[k:]
        else:
            q = -k
            G[:n - q] += w * (1.0 - h[q:])
            G[q:] -= w * h[:n - q]
    return G


def _smooth_obj_grad(u, target, eps):
    h = _u_to_h(u, target)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mval = float(np.max(corr))
    Val = eps * float(np.log(np.sum(np.exp((corr - mval) / eps)))) + mval
    w = np.exp((corr - mval) / eps)
    w /= w.sum()
    G_h = _corr_gradient(h, w) * dx
    s = h * (1.0 - h)
    S = float(s.sum())
    if S < 1e-20:
        G_u = np.zeros(n)
    else:
        G_u = s * (G_h - (G_h * s).sum() / S)
    return Val, G_u

def _quad_line_search(h, d, target, max_t=1.0):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C_arr = np.correlate(d, d, mode='full')

    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)

    n_grid = 40
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T_arr[None, :] - (grid[:, None] ** 2) * C_arr[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = float(obj_vals[best_i])

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
    f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_arr - opt_t * opt_t * C_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val

    if best_t < 0:
        best_t = 0.0
        best_val = _exact_c5(h)
    return best_t, best_val


def _active_descent(h, deadline, K=30, max_iters=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0

    for _it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        alpha = 1.0 / (0.002 * n)
        w = np.exp(alpha * (top_vals - M))
        w_sum = float(w.sum())
        if w_sum < 1e-30:
            w = np.ones(K) / K
        else:
            w /= w_sum

        weights_full = np.zeros(2 * n - 1)
        for wi, idx in zip(w, top_idx):
            weights_full[int(idx)] = wi

        grad = _corr_gradient(h, weights_full) * dx
        grad -= np.mean(grad)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-13:
            break

        direction = -grad / gn
        t, f_try = _quad_line_search(h, direction, target, max_t=step)
        if t < 1e-13:
            step *= 0.7
            if step < 0.005:
                break
            continue

        h_new = h + t * direction
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)

        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.8 * step:
                step = min(step * 1.05, 1.5)
            else:
                step = max(step * 0.92, 0.01)
        else:
            step *= 0.75
            if step < 0.005:
                break

    return best_h, best_c5


def _binary_sa(h_init, deadline, rng, max_iters=100000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    ones_set = set(int(i) for i in np.where(h > 0.5)[0])
    zeros_set = set(int(i) for i in np.where(h <= 0.5)[0])
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)

    T = 0.015
    T_min = 1e-7
    cooling = 0.99995

    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = int(ones_list[rng.integers(len(ones_list))])
        j = int(zeros_list[rng.integers(len(zeros_list))])
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _slsqp_epigraph(h, deadline, lambda_reg=0.01, n_active=30, max_calls=4):

            return g

        constraints = [{'type': 'eq',
                        'fun': lambda x, t=target: float(np.sum(x[:n]) - t),
                        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]

        for r in top_idx:
            kk = int(r) - (n - 1)

            def _cfun(x, k=kk):
                hv = x[:n]
                if k >= 0:
                    m = n - k
                    val = float(np.dot(hv[k:], 1.0 - hv[:m]))
                else:
                    ga = -k
                    m = n - ga
                    val = float(np.dot(hv[:m], 1.0 - hv[ga:]))
                return x[n] - val * dx

            def _cjac(x, k=kk):
                g = _corr_gradient(x[:n], np.zeros(2 * len(x[:n]) - 1))
                n_local = len(x[:n])
                wg = np.zeros(2 * n_local - 1)
                wg[k + n_local - 1] = 1.0
                g = _corr_gradient(x[:n], wg) * dx
                j = -g.copy()
                j[n_local] = 1.0
                return j

            constraints.append({'type': 'ineq', 'fun': _cfun, 'jac': _cjac})

        bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.05)]
        x0 = np.concatenate([h, [t0]])

        try:
            res = minimize(_obj_func, x0, method='SLSQP', jac=_obj_jac,
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.x is not None:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    break
        except Exception:
            break

    return best_h, best_c5


def _perturbation_search(h, deadline, rng, n_trials=200):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.04

    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        h_try, c5_try = _active_descent(h_pert, min(deadline, time.time() + 0.5), K=20, max_iters=30)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.15, scale * 1.1)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * x / n0) + 0.1 * np.sin(2 * np.pi * 7 * x / n0)
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: L-BFGS on log-sum-exp (40% time)
    d1 = t0 + budget_s * 0.40
    h1, c5_1 = _lbfgs_smooth(h, d1, max_iter=400, m_mem=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Active descent (20% time)
    d2 = t0 + budget_s * 0.60
    h2, c5_2 = _active_descent(best_h, d2, K=35, max_iters=200)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: SLSQP epigraph (15% time)
    d3 = t0 + budget_s * 0.75
    for lam in [0.005, 0.02, 0.08]:
        if time.time() > d3:
            break
        h_s, c5_s = _slsqp_epigraph(best_h, min(d3, time.time() + budget_s * 0.04),
                                    lambda_reg=lam, n_active=30, max_calls=3)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 4: Multi-resolution (15% time)
    d4 = t0 + budget_s * 0.90
    best_n = len(best_h)
    offsets = [-4, 4, -6, 6, -8, 8, -12, 12, -16, 16]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 256:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _active_descent(h_r, sub_dl, K=25, max_iters=50)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Binary SA + polish (10% time)
    d5 = deadline - 2.0
    h_bin, c5_bin = _binary_sa(best_h, min(d5, time.time() + budget_s * 0.08), rng, max_iters=50000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Polish binary with continuous optimization
    if time.time() < d5:
        h_pol, c5_pol = _active_descent(best_h, d5, K=20, max_iters=100)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final: perturbation + polish
    if time.time() < deadline - 1.0:
        h_p, c5_p = _perturbation_search(best_h, deadline - 1.0, rng, n_trials=100)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final polish
    if time.time() < deadline - 0.5:
        h_f, c5_f = _active_descent(best_h, deadline - 0.1, K=15, max_iters=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
