import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_M(v, n):
    """Build correlation matrix M where M[q,i] = dx*v[i+q-n+1] if valid."""
    dx = 2.0 / n
    M = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        k = q - n + 1
        if k >= 0:
            m = n - k
            M[q, :m] = dx * v[k:k+m]
        else:
            d = -k
            m = n - d
            M[q, d:d+m] = dx * v[:m]
    return M

def _solve_lp(M, n, target):
    """Solve min tau s.t. M@h - tau <= 0, sum(h)=target, 0<=h<=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Constraints: M@h - tau <= 0 → [M, -1] @ [h; tau] <= 0
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_on_segment(u, w, n):
    """Find best alpha in [0,1] minimizing C((1-alpha)*u + alpha*w)."""
    dx = 2.0 / n
    # Coarse grid
    best_alpha = 0.5
    best_val = float('inf')
    n_grid = 31
    for i in range(n_grid):
        alpha = i / (n_grid - 1)
        h = (1 - alpha) * u + alpha * w
        val = _exact_c5(h)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    # Find bracket around best_alpha
    if best_alpha > 0 and best_alpha < 1:
        a = max(0, best_alpha - 1.0/(n_grid-1))
        b = min(1, best_alpha + 1.0/(n_grid-1))
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = _exact_c5((1-c1)*u + c1*w)
    f2 = _exact_c5((1-c2)*u + c2*w)
    for _ in range(40):
        if b - a < 1e-12:

        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _exact_c5((1-c1)*u + c1*w)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _exact_c5((1-c2)*u + c2*w)
    opt_alpha = (a + b) / 2.0
    opt_val = _exact_c5((1-opt_alpha)*u + opt_alpha*w)
    if opt_val < best_val:
        best_val = opt_val
        best_alpha = opt_alpha
    return best_alpha, best_val

    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Step 2: Line search between h_new and 1-v = h (original)
        if use_line:
            u = h_new  # one endpoint
            w = h      # other endpoint (which is 1-v)
            # Actually both h_new and h are feasible
            # Try segment between them
            alpha, val = _line_search_on_segment(h, h_new, n)
            if val < best_c5 - 1e-12:
                h_seg = (1-alpha) * h + alpha * h_new
                h_seg = _project(h_seg, target)
                val2 = _exact_c5(h_seg)
                if val2 < best_c5:
                    best_c5 = val2
                    best_h = h_seg.copy()
        
        # Update h for next iteration
        h = h_new.copy()
        
        # Check convergence
        if _exact_c5(h) - best_c5 > 0.01:
            # Diverging, reset to best
            h = best_h.copy()
    
    return best_h, best_c5

def _alt_lp_with_complement(h, deadline, max_iters=3000):
    """Alternating LP with complement pair."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Phase A: Fix b = 1-h, solve for a
        b = 1.0 - h
        M_a = _build_M(b, n)
        a = _solve_lp(M_a, n, target)
        if a is None:
            break
        a = _project(a, target)
        
        # Phase B: Fix a, solve for b
        M_b = _build_M(a, n)
        b_new = _solve_lp(M_b, n, target)
        if b_new is None:
            break
        b_new = _project(b_new, target)
        
        # Candidates: a, 1-b_new, and segment
        u = _project(1.0 - b_new, target)  # complement of b_new
        w = a
        
        # Evaluate endpoints
        c5_u = _exact_c5(u)
        c5_w = _exact_c5(w)
        
        # Line search
        alpha, val = _line_search_on_segment(u, w, n)
        h_seg = (1-alpha) * u + alpha * w
        h_seg = _project(h_seg, target)
        c5_seg = _exact_c5(h_seg)
        
        # Pick best
        candidates = [(c5_u, u), (c5_w, w), (c5_seg, h_seg), (best_c5, h)]
        best_cand = min(candidates, key=lambda x: x[0])
        if best_cand[0] < best_c5 - 1e-12:
            best_c5 = best_cand[0]
            best_h = best_cand[1].copy()
            h = best_cand[1].copy()
        else:
            # No improvement, update h to the LP solution
            h = a.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_start(n, rng):
    """Generate diverse random initial conditions."""
    strategy = rng.integers(0, 5)
    if strategy == 0:
        # Sinusoidal
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2*np.pi)
        x = np.arange(n)
        h = 0.5 + 0.4 * np.sin(2*np.pi*freq*x/n + phase) + rng.normal(0, 0.1, n)
    elif strategy == 1:
        # Blocks
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, int(rng.integers(2, max(3, n//6))))
            if rng.random() < 0.5:
                h[pos:pos+bl] = 1.0
            pos += bl
    elif strategy == 2:
        # Separable Fourier
        h = np.zeros(n)
        x = np.arange(n) / n
        for _ in range(rng.integers(2, 5)):
            f = rng.integers(1, 8)
            ph = rng.uniform(0, 2*np.pi)
            amp = rng.uniform(0.1, 0.3)
            h += amp * np.sin(2*np.pi*f*x + ph)
        h = 0.5 + h
    elif strategy == 3:
        # Piecewise constant with random boundaries
        n_blocks = rng.integers(n//10, n//4)
        h = np.zeros(n)
        boundaries = np.sort(rng.choice(np.arange(1, n), size=n_blocks-1, replace=False))
        boundaries = np.concatenate([[0], boundaries, [n]])
        for j in range(n_blocks):
            h[boundaries[j]:boundaries[j+1]] = rng.uniform(0, 1)
    else:
        # Smooth random
        h = 0.5 + rng.normal(0, 0.3, n)
        # Smooth with moving average
        k = max(1, n // 20)
        kernel = np.ones(k) / k
        h = np.convolve(h, kernel, mode='same')
    
    return _project(h, n/2.0)

            for i in range(m):

                    grad[i+k] += -h[i]

                        else:
                            hi_t = mid_t

                    val = _exact_c5(h_try)
                    # For minimization, we want to check if this direction helps
                    # Use goldens section
                    pass

                    h_try = h + frac * d
                    if np.min(h_try) < -1e-9 or np.max(h_try) > 1 + 1e-9:
                        continue
                    h_try = _project(h_try, target)
                    val = _exact_c5(h_try)
                    if val < best_c5 - 1e-12:

Hmm, the subgradient approach is getting complicated because the objective is bilinear, not linear. Let me focus on what I know works well:

1. Alternating LP (the main workhorse)

Let me also think about a key improvement: instead of just using v = 1-h as the fixed partner, I can use v = 1-h + noise, or a completely random balanced vector. This can help escape local optima.

Let me also think about the "dual" view. The LP:
min τ s.t. M@h ≤ τ*1, sum(h) = n/2, 0 ≤ h ≤ 1

The dual variables for the constraints M@h ≤ τ*1 give us a probability distribution over the lags. At the optimum, the active lags (where M@h = τ) are the "critical" lags. The dual solution tells us which lags are tight.

This is useful: if only 1-2 lags are tight, we might be able to do a more targeted optimization.

Let me also think about a key insight for the problem structure:

For the specific case where h is binary (0 or 1), the objective simplifies:
C(h) = max_k dx * (# of 1s in h that don't overlap with 1s in h shifted by k)

For a binary h with exactly n/2 ones:
- S_0 = dx * (n/2) * (1 - 0) = (2/n)*(n/2) = 1... wait that's not right.

Actually for binary h: h[i]*(1-h[i]) = 0 for all i. So S_0 = 0!

For k ≠ 0: S_k = dx * sum_{i in overlap} h[i] * (1-h[i+k])
This counts the number of positions where h has a 1 and h shifted by k has a 0, times dx.

So for binary h, C_5 = max_{k≠0} dx * (number of "1s in h that don't align with 1s in h shifted by k")

This is related to the "overlap" problem. We want a binary sequence of length n with n/2 ones such that for every shift k≠0, the number of non-overlapping 1s is minimized.

This is essentially a "minimum dispersion" or "balanced autocorrelation" problem!

If h is binary, then:
- Total 1s = n/2
- For shift k: number of pairs (i, i+k) where both are 1 = A_k (autocorrelation)
- Number of pairs where h[i]=1 and h[i+k]=0 = (n-k) - A_k (for k>0, assuming the overlap has n-k positions)

Wait, more carefully: for k > 0, the overlap has n-k positions. In these positions, some have h[i]=1 and h[i+k]=1 (count A_k), and the rest have at least one being 0.

S_k = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

For binary h: this equals dx * (number of i in [0,n-1-k] where h[i]=1 and h[i+k]=0)
= dx * ((number of i where h[i]=1) in [0,n-1-k]) - A_k
= dx * (P_k - A_k) where P_k = number of 1s in h[0..n-1-k]

Hmm, this depends on the distribution of 1s in the prefix, not just the total count.

For a "uniformly distributed" binary sequence (1s evenly spaced), P_k ≈ (n/2) * (n-k)/n = (n-k)/2 (approximately).
Then S_k ≈ dx * ((n-k)/2 - A_k)

For the "best" case where A_k is maximized (all 1s align), S_k would be 0. But we can't have all lags simultaneously.

This confirms that binary solutions are interesting to explore. Let me include binary refinement in the algorithm.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_M(v, n):
    dx = 2.0 / n
    M = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        k = q - n + 1
        if k >= 0:
            m = n - k
            M[q, :m] = dx * v[k:k+m]
        else:
            d = -k
            m = n - d
            M[q, d:d+m] = dx * v[:m]
    return M

def _solve_lp(M, n, target):
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _seg_search(u, w, n, n_grid=31):
    """Find best alpha in [0,1] minimizing C((1-alpha)*u + alpha*w)."""
    best_alpha = 0.5
    best_val = float('inf')
    for i in range(n_grid):
        alpha = i / (n_grid - 1)
        h = (1 - alpha) * u + alpha * w
        val = _exact_c5(h)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    step = 1.0 / (n_grid - 1)
    a = max(0.0, best_alpha - step)
    b = min(1.0, best_alpha + step)
    if b - a < 1e-10:
        return best_alpha, best_val
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = _exact_c5((1-c1)*u + c1*w)
    f2 = _exact_c5((1-c2)*u + c2*w)
    for _ in range(35):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _exact_c5((1-c1)*u + c1*w)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _exact_c5((1-c2)*u +
