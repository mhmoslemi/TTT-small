from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Euclidean projection onto {x : 0 <= x_i <= 1, sum(x) = target}
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    # Fix remaining error
    diff = target - np.sum(x)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0, 1)
    return x

def _sigmoid(x):
    # Numerically stable sigmoid
    out = np.zeros_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def _solve_lambda(u, target):
    # Find lambda such that sum(sigmoid(u + lambda)) = target
    lo, hi = -50.0, 50.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(_sigmoid(u + mid))
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2

def _h_from_u(u, target):
    lam = _solve_lambda(u, target)
    return _sigmoid(u + lam)

def _to_logit(h):
    eps = 1e-10
    h = np.clip(h, eps, 1 - eps)
    return np.log(h / (1 - h))

def _lse_obj_grad(u, target, eps):
    """Compute LSE smoothed objective and gradient."""
    n = len(u)
    lam = _solve_lambda(u, target)
    h = _sigmoid(u + lam)
    dx = 2.0 / n
    
    # Compute all correlation values
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = len(corr)  # 2n-1
    
    # Log-sum-exp
    corr_max = np.max(corr)
    exp_vals = np.exp((corr - corr_max) / eps)
    lse = corr_max + eps * np.log(np.sum(exp_vals))
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient w.r.t. h
    G_h = np.zeros(n)
    for r in range(m):
        if p[r] < 1e-15:
            continue
        k = r - (n - 1)
        # d(corr_k)/dh
        if k >= 0:
            # (1-h[j-k]) for j in [k, n-1]
            G_h[k:] += p[r] * (1.0 - h[:n-k])
            # -h[j+k] for j in [0, n-k-1]
            G_h[:n-k] -= p[r] * h[k:]
        else:
            q = -k
            # (1-h[j+q]) for j in [0, n-q-1]
            G_h[:n-q] += p[r] * (1.0 - h[q:])
            # -h[j-q] for j in [q, n-1]
            G_h[q:] -= p[r] * h[:n-q]
    G_h *= dx
    
    # Transform to u-space
    s = h * (1.0 - h)
    s_sum = np.sum(s)
    if s_sum < 1e-15:
        grad_u = np.zeros(n)
    else:
        grad_u = s * (G_h - np.sum(G_h * s) / s_sum)
    
    return lse, grad_u

def _optimize_lse(u_init, target, eps, deadline, maxiter=300):

import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _sigmoid(x):
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def _solve_lambda(u, target):
    lo, hi = -60.0, 60.0
    for _ in range(150):
        mid = (lo + hi) / 2.0
        s = float(np.sum(_sigmoid(u + mid)))
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0


def _h_from_u(u, target):
    lam = _solve_lambda(u, target)
    return _sigmoid(u + lam)


def _to_logit(h):
    eps = 1e-10
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))


def _lse_obj_grad(u, target, eps):
    n = len(u)
    lam = _solve_lambda(u, target)
    h = _sigmoid(u + lam)
    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = len(corr)

    corr_max = float(np.max(corr))
    shifted = (corr - corr_max) / eps
    exp_vals = np.exp(shifted)
    exp_sum = float(np.sum(exp_vals))
    if exp_sum < 1e-300:
        exp_vals = np.zeros(m)
        exp_vals[np.argmax(corr)] = 1.0
        exp_sum = 1.0
    lse = corr_max + eps * np.log(exp_sum)
    p = exp_vals / exp_sum

    G_h = np.zeros(n)
    active = p > 1e-12
    active_idx = np.where(active)[0]
    for r in active_idx:
        pr = p[r]
        k = int(r) - (n - 1)
        if k >= 0:
            G_h[k:] += pr * (1.0 - h[:n - k])
            G_h[:n - k] -= pr * h[k:]
        else:
            q = -k
            G_h[:n - q] += pr * (1.0 - h[q:])
            G_h[q:] -= pr * h[:n - q]
    G_h *= dx

    s = h * (1.0 - h)
    s_sum = float(np.sum(s))
    if s_sum < 1e-15:
        grad_u = np.zeros(n)
    else:
        grad_u = s * (G_h - float(np.dot(G_h, s)) / s_sum)

    return lse, grad_u


def _optimize_lse(u_init, target, eps, deadline, maxiter=250):
    n = len(u_init)
    u = np.clip(u_init, -30.0, 30.0).copy()

    val0, _ = _lse_obj_grad(u, target, eps)

    def _fun(x):
        if time.time() > deadline:
            return 1e10
        v, _ = _lse_obj_grad(x, target, eps)
        return v

    def _jac(x):
        if time.time() > deadline:
            return np.zeros(n)
        _, g = _lse_obj_grad(x, target, eps)
        return g

    bounds_u = [(-30.0, 30.0)] * n
    try:
        res = minimize(_fun, u, jac=_jac, method='L-BFGS-B',
                       bounds=bounds_u,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-12, 'maxfun': maxiter * 4})
        u_new = np.clip(res.x, -30.0, 30.0)
    except Exception:
        u_new = u.copy()

    val_new, _ = _lse_obj_grad(u_new, target, eps)
    if val_new < val0:
        return u_new
    return u


def _exact_max_descent(h, deadline, max_iters=100):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        r_star = int(np.argmax(corr))
        k = r_star - (n - 1)

        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
        else:
            q = -k
            grad[:n - q] += (1.0 - h[q:])
            grad[q:] -= h[:n - q]
        grad *= dx
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        direction = -grad / gn

        # Line search
        best_t = 0.0
        best_f = _exact_c5(h)
        t = 0.3
        for _ls in range(30):
            h_try = _project(h + t * direction, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
            if f_try <= best_c5 + 1e-5 * t * (-np.dot(direction, grad * dx)):
                break
            t *= 0.6
            if t < 1e-10:
                break

        if best_t > 1e-12 and best_f < best_c5 - 1e-14:
            h = _project(h + best_t * direction, target)
            if best_f < best_c5:
                best_c5 = best_f
                best_h = h.copy()
        else:
            break

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
