def _project(v, target):

        else:

def _grad_rows(h, indices, n):

        else:
            d = -k
            m = n - d
            if m > 0:
                g[d:] += dx * (1.0 - h[:m])
                g[:m] += -dx * h[d:]
    return G

def _lp_direction(h, S, n, K, radius):

    sorted_idx = np.argsort(-S)
    tol = max(0.08 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([sorted_idx[:K], near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b = hi_b = 0.0
        bounds.append((lo_b, hi_b))

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        pass
    return None

def _smooth_dir(h, S, n, tau):
    M = float(S.max())
    shifted = (S - M) / tau
    shifted -= shifted.max()
    w = np.exp(np.clip(shifted, -50, 0))
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_k = min(25, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    w_t = w[top_idx]
    if w_t.sum() < 1e-300:
        return None
    w_t = w_t / w_t.sum()
    G = _grad_rows(h, top_idx, n)
    g = w_t @ G
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-12:
        return None
    return -g / norm

def _binary_local_search(h, deadline, rng):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        S = _corr_arr(h)
        M = float(S.max())
        top_lags = np.argsort(-S)[:20]
        for lag in top_lags:
            if time.time() > deadline:
                break
            k = lag - (n - 1)
            if k >= 0:
                m = n - k
                if m <= 0:
                    continue
                contrib = h[:m] * (1.0 - h[k:k+m])
                for i in range(min(5, m)):
                    if contrib[i] > 1e-10:
                        o = np.random.default_rng(i).choice(len(ones_list))
                        z = np.random.default_rng(100+i).choice(len(zeros_list))
                        oi = ones_list[o]
                        zi = zeros_list[z]
                        if oi == i or oi == i + k:
                            h[zi] = 1.0
                            h[oi] = 0.0
                            new_c5 = _exact_c5(h)
                            if new_c5 < best_c5 - 1e-14:
                                best_c5 = new_c5
                                best_h = h.copy()
                                improved = True
                                ones_list = np.where(h > 0.5)[0]
                                zeros_list = np.where(h <= 0.5)[0]
                            else:
                                h[oi] = 1.0
                                h[zi] = 0.0
            else:
                d = -k
                m = n - d
                if m <= 0:
                    continue
                contrib = h[d:] * (1.0 - h[:m])
                for i in range(min(5, m)):
                    if contrib[i] > 1e-10:
                        o = np.random.default_rng(i + 50).choice(len(ones_list))
                        z = np.random.default_rng(150 + i).choice(len(zeros_list))
                        oi = ones_list[o]
                        zi = zeros_list[z]
                        if oi == i + d or oi == i:
                            h[zi] = 1.0
                            h[oi] = 0.0
                            new_c5 = _exact_c5(h)
                            if new_c5 < best_c5 - 1e-14:
                                best_c5 = new_c5
                                best_h = h.copy()
                                improved = True
                                ones_list = np.where(h > 0.5)[0]
                                zeros_list = np.where(h <= 0.5)[0]
                            else:
                                h[oi] = 1.0
                                h[zi] = 0.0
    return best_h, best_c5
