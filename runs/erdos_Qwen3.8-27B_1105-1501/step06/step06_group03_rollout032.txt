            stall += 1
            radius *= 0.72
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def _pair_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    S = _corr_scaled(h)
    G = _grad_rows(h, np.arange(2*n-1), n)
    total_grad = S @ G
    total_grad -= total_grad * (1.0/n)
    pend = 0
    while time.time() < deadline and pend < 60:
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        improved = False
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii+1, len(candidates)):
                if time.time() > deadline:
                    break
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i], d[j] = 1.0, -1.0
                else:
                    d[j], d[i] = 1.0, -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = _project(h + alpha * d, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        S = _corr_scaled(h)
                        total_grad = S @ G
                        total_grad -= total_grad * (1.0/n)
                        break
            if improved:
                break
        if not improved:
            pend += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2*np.pi*freq*x/n + phase)
    return _project(h, n/2.0)

def _run(start_h, budget_s, seed=42):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    best_h = start_h.astype(float).copy()
    best_c5 = _exact_c5(best_h)
    n0 = len(best_h)

    # Phase 1: Quick multi-start (5%)
    d1 = t0 + budget_s * 0.05
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 5]:
        for ph in [0.0, 1.0, 2.0]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(5):
        h_r = np.zeros(n0)
        nb = rng.integers(3, 10)
        bounds = sorted(rng.choice(np.arange(1, n0), size=nb-1, replace=False))
        bounds = [0] + list(bounds) + [n0]
        for b in range(nb-1):
            if rng.random() < 0.5:
                h_r[bounds[b]:bounds[b+1]] = 1.0
        starts.append(_project(h_r, n0/2.0))
    for s in starts:
        if time.time() > d1:
            break
        hd = min(d1, time.time() + budget_s*0.006)
        hs, cs = _lp_optimize(s, hd, radius_init=0.08, K=10)
        if cs < best_c5:
            best_c5, best_h = cs, hs.copy()

    # Phase 2: Main LP (30%)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _lp_optimize(best_h, d2, radius_init=0.14, K=20, use_subgrad=False)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: LP with subgradient backup (12%)
    d3 = t0 + budget_s * 0.47
    h3, c5_3 = _lp_optimize(best_h, d3, radius_init=0.08, K=25, use_subgrad=True)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Pair refinement (10%)
    d4 = t0 + budget_s * 0.57
    h4, c5_4 = _pair_refine(best_h, d4)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Resolution sweep (15%)
    d5 = t0 + budget_s * 0.72
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        nt = base_n + dn
        if nt < 60 or nt > 220 or nt % 2 != 0 or nt in tried:
            continue
        tried.add(nt)
        hr = _resample(best_h, nt)
        cr = _exact_c5(hr)
        if cr < best_c5 + 0.003:
            sd1 = min(d5, time.time() + budget_s*0.015)
            hr, cr = _lp_optimize(hr, sd1, radius_init=0.10, K=15)
            if cr < best_c5 - 1e-12:
                best_c5, best_h = cr, hr.copy()
            if time.time() < d5 - budget_s*0.01:
                sd2 = min(d5, time.time() + budget_s*0.008)
                hr2, cr2 = _pair_refine(hr, sd2)
                if cr2 < best_c5 - 1e-12:
                    best_c5, best_h = cr2, hr2.copy()

    # Phase 6: Perturbation + re-opt (10%)
    d6 = t0 + budget_s * 0.82
    scale = 0.02
    for _ in range(400):
        if time.time() > d6:
            break
        hp = _project(best_h + rng.normal(0, scale, len(best_h)), len(best_h)/2.0)
        hd = min(d6, time.time() + budget_s*0.004)
        ht, ct = _lp_optimize(hp, hd, radius_init=0.04, K=12)
        if ct < best_c5 - 1e-13:
            best_c5, best_h = ct, ht.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.1, scale * 1.15)

    # Phase 7: Ultra-fine polish (15%)
    d7 = t0 + budget_s * 0.97
    np_ = len(best_h)
    target = np_ / 2.0
    hpol = best_h.copy()
    cpol = _exact_c5(hpol)
    radius = 0.005
    stall = 0
    while time.time() < d7:
        if stall > 100:
            break
        Spol = _corr_scaled(hpol)
        M = float(Spol.max())
        tol = max(0.02 * M, 1e-7)
        active = np.where(Spol >= M - tol)[0]
        if len(active) > 50:
            active = np.argsort(-Spol)[:50]
        G = _grad_rows(hpol, active, np_)
        c_vec = np.zeros(np_ + 1)
        c_vec[np_] = -1.0
        A_ub = np.zeros((len(active), np_ + 1))
        A_ub[:, :np_] = G
        A_ub[:, np_] = 1.0
        b_ub = M - Spol[active]
        A_eq = np.zeros((1, np_ + 1))
        A_eq[0, :np_] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(np_):
            lo_b = max(-radius, -hpol[j] + 1e-10)
            hi_b = min(radius, 1.0 - hpol[j] - 1e-10)
            if lo_b >= hi_b:
                lo_b, hi_b = 0.0, 0.0
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        p = None
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x is not None:
                p = res.x[:np_]
        except Exception:
            pass
        if p is None or np.max(np.abs(p)) < 1e-14:
            d = -G @ (np.ones(len(active)) / len(active))
            d -= d * (1.0/np_)
            nm = np.linalg.norm(d)
            if nm > 1e-14:
                p = -d / nm * radius
            else:
                stall += 1
                radius *= 0.6
                if radius < 1e-8:
                    break
                continue
        alpha, new_c5 = _line_search(hpol, p, 2.0/np_, radius)
        if new_c5 < cpol - 1e-15:
            hpol = _project(hpol + alpha * p, target)
            cpol = _exact_c5(hpol)
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.05, 0.02)
            else:
                radius = max(radius * 0.9, 0.001)
        else:
            stall += 1
            radius *= 0.65
            if radius < 1e-8:
                break
    if cpol < best_c5:
        best_c5, best_h = cpol, hpol.copy()

    # Final: pair polish
    if time.time() < deadline - 1:
        hf, cf = _pair_refine(best_h, deadline - 0.5)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Validate
    best_h = _project(best_h, len(best_h)/2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
    except NameError:
        h0 = _sine_init(122, 2, 0.5)

    h0 = _project(h0, len(h0)/2.0)
    c5_0 = _exact_c5(h0)

    best_h, best_c5, best_n = _run(h0, min(deadline - t0, budget_s*0.92), seed=seed)

    # If we have time left, try a slightly different resolution
    remaining = deadline - time.time()
    if remaining > 30:
        n_try = best_n + (2 if best_n % 4 == 2 else -2)
        if 60 <= n_try <= 220 and n_try % 2 == 0 and n_try != best_n:
            ht = _resample(best_h, n_try)
            ct = _exact_c5(ht)
            if ct < best_c5 + 0.002:
                ht, ct, nt = _run(ht, remaining * 0.7, seed=seed+1)
                if ct < best_c5:
                    best_h, best_c5, best_n = ht, ct, nt

    best_h = _project(best_h, len(best_h)/2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
