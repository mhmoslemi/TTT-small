    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _pair_search(h, deadline, n_pairs, rng):
    """Directed pair search targeting active peaks."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    
    while time.time() < deadline and stall < 60:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.005 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]
        
        # Compute weighted gradient
        G = _grad_matrix(h)
        grad = G[active_idx].mean(axis=0)
        
        # Find best pair move
        # Donor: where grad is most positive (decreasing h here helps most)
        # Receiver: where grad is most negative (increasing h here helps most)
        donor_score = grad  # high grad → decrease h
        recv_score = -grad   # low grad → increase h
        
        # Get top candidates
        n_cand = min(n_pairs, n // 4)
        donors = np.argsort(-donor_score)[:n_cand]
        receivers = np.argsort(recv_score)[:n_cand]
        
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        
        for i in donors:
            for j in receivers:
                if i == j:
                    continue
                # Golden section search for alpha
                # h[i] -= alpha, h[j] += alpha
                lo = -h[j]  # h[j] + alpha >= 0 → alpha >= -h[j]
                hi = min(h[i], 1.0 - h[j])
                if hi - lo < 1e-14:
                    continue
                # Parametric: S0 + alpha * T - alpha² * C
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                C = np.correlate(d, d, 'full')
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
                f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
                for _ in range(35):
                    if b - a < 1e-13:
                        break
                    if f1 < f2:
                        b, c2, f2 = c2, c1, f1
                        c1 = b - gr * (b - a)
                        f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
                    else:
                        a, c1, f1 = c1, c2, f2
                        c2 = a + gr * (b - a)
                        f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
                opt = (a + b) / 2.0
                val = float(np.max(S0 + opt * T - opt*opt * C)) * dx
                # Check endpoints
                f_lo = float(np.max(S0 + lo * T - lo*lo * C)) * dx
                f_hi = float(np.max(S0 + hi * T - hi*hi * C)) * dx
                if f_lo < val:
                    opt, val = lo, f_lo
                if f_hi < val:
                    opt, val = hi, f_hi
                
                if opt > 1e-10 and val < best_new_c5 - 1e-14:
                    best_new_c5 = val
                    best_alpha = opt
                    best_pair = (i, j)
        
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.4:
                # Perturbation
                idx = rng.integers(n, size=6)
                h[idx] += rng.uniform(-0.08, 0.08, size=6)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    
    return best_h, best_c5

def _gradient_step(h, deadline):
    """Gradient-based step using active row gradient."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    
    while time.time() < deadline and stall < 20:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.002 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:5]
        
        G = _grad_matrix(h)
        g = G[active_idx].mean(axis=0)
        g -= np.mean(g)  # ensure sum = 0
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        d = -g / gn
        
        # Line search
        lo, hi = 0.0, 1.0
        pos = d > 1e-10
        neg = d < -1e-10
        if np.any(pos):
            hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            lo = max(lo, float(np.max(h[neg] / (-d[neg]))))
        
        if hi - lo < 1e-12:
            break
        
        T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
        C = np.correlate(d, d, 'full')

        a, b = lo, hi
        c1 = b - gr * (b - a)
