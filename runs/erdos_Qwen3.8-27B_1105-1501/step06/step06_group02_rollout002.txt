import numpy as np
import time
import cvxpy as cp

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_row_grad(h, lag):
    """Compute gradient of correlation row at given centered lag."""
    n = len(h)
    g = np.zeros(n)
    s = lag  # centered lag: -(n-1) to (n-1)
    if s >= 0:
        m = n - s
        if m <= 0:
            return g
        # R_s = sum_{t=0}^{m-1} h[t+s]*(1-h[t])
        # dR/dh[t+s] = 1 - h[t], dR/dh[t] = -h[t+s]
        g[s:s+m] += (1.0 - h[:m])
        g[:m] += -h[s:s+m]
    else:
        r = -s
        m = n - r
        if m <= 0:
            return g
        # R_{-r} = sum_{t=0}^{m-1} h[t]*(1-h[t+r])
        # dR/dh[t] = 1 - h[t+r], dR/dh[t+r] = -h[t]
        g[:m] += (1.0 - h[r:r+m])
        g[r:r+m] += -h[:m]
    return g

def _lp_direction(h, R0, active_idx, l1_bound):
    """Find direction via LP that minimizes max active peak (first order)."""
    n = len(h)
    M = len(active_idx)
    if M == 0:
        return None
    
    # Compute gradients for active rows
    # active_idx are indices into R0, which corresponds to centered lag s = k - (n-1)
    grads = np.zeros((M, n))
    for k, idx in enumerate(active_idx):
        s = idx - (n - 1)
        grads[k] = _compute_row_grad(h, s)
    
    # LP: minimize t
    # subject to: R0[active] + grads @ d <= t
    #             sum(d) = 0
    #             -lb_i <= d_i <= ub_i  (from box constraints)
    #             sum(|d|) <= l1_bound
    
    d = cp.Variable(n)
    t = cp.Variable()
    s_abs = cp.Variable(n)  # for L1
    
    constraints = [
        t >= R0[active_idx] + grads @ d,
        cp.sum(d) == 0,
        d >= -min(h, 1.0 - h) * 0.5,  # don't go more than half the distance to boundary
        d <= min(h, 1.0 - h) * 0.5,
        s_abs >= d,
        s_abs >= -d,
        cp.sum(s_abs) <= l1_bound,
    ]
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None
    
    if d.value is None:
        return None
    
    d_val = d.value.copy()
    # Clean up
    d_val[np.abs(d_val) < 1e-10] = 0.0
    return d_val

def _exact_line_search(h, d, R0, dx):
    """Exact 1D minimization along direction d from base h."""
    n = len(h)
    # Compute T and C once
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Find valid range for alpha
    pos = d > 1e-10
    neg = d < -1e-10
    lo = -1.0
    hi = 1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    if hi - lo < 1e-12:
        return 0.0, float(np.max(R0)) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(R0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _lp_optimize(h, deadline, max_iters=200):
    """Main optimization loop using LP directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        scores = R0 * dx
        max_score = float(np.max(scores))
        
        # Select active rows: those within a small fraction of max
        threshold = max_score - 0.02 * max_score
        active_idx = np.where(scores >= threshold)[0]
        
        # If too many, take top 20
        if len(active_idx) > 25:
            # Weighted selection
            order = np.argsort(-scores[active_idx])
            active_idx = active_idx[order[:25]]
        if len(active_idx) < 3:
            active_idx = np.argsort(-scores)[:5]
        
        # Try LP direction
        d = _lp_direction(h, R0, active_idx, l1_bound=2.0)
        
        if d is None or np.all(np.abs(d) < 1e-10):
            # Fallback: weighted gradient
            grads = np.zeros(n)
            for idx in active_idx:
                s = idx - (n - 1)
                g = _compute_row_grad(h, s)
                w = np.exp((scores[idx] - max_score) / (0.01 * max_score + 1e-15))
                grads += w * g
            d = -grads
            d -= np.mean(d)
            norm = np.linalg.norm(d)
            if norm < 1e-10:
                stall += 1
                if stall > 20:
                    break
                continue
            d = d / norm * 0.5
        
        # Exact line search
        alpha, new_c5 = _exact_line_search(h, d, R0, dx)
        
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_h, best_c5

def _pair_search(h, dx, target, deadline, n_pairs=20, seed=777):
    """Pair move search (fallback/diversification)."""
    n = len(h)
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.time() < deadline and stall < 100:
        scores = R0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.03 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) < 3:
            active_idx = np.argsort(-scores)[:8]
        
        # Compute weighted gradient
        wgrad = np.zeros(n)
        for idx in active_idx:
            s = idx - (n - 1)
            g = _compute_row_grad(h, s)
            w = np.exp((scores[idx] - max_score) / (0.01 * max_score + 1e-15))
            wgrad += w * g
        
        # Best pair: maximize decrease
        # For pair (i,j): move alpha from i to j
        # Approximate change: alpha * (wgrad[j] - wgrad[i])
        # Find i with highest wgrad, j with lowest
        donor = np.argsort(-wgrad)[:n_pairs]
        recv = np.argsort(wgrad)[:n_pairs]
        
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        
        for i in donor:
            if time.time() > deadline:
                break
            for j in recv:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, new_c5 = _exact_line_search(h, d, R0, dx)
                if alpha > 1e-10 and new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb
                idx = rng.integers(n, size=3)
                for k in idx:
                    h[k] += rng.uniform(-0.05, 0.05)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    
    return best_h, best_c5

import numpy as np
import time
import cvxpy as cp

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(h, lag):
    n = len(h)
    g = np.zeros(n)
    s = lag
    if s >= 0:
        m = n - s
        if m <= 0:
            return g
        g[s:s+m] += (1.0 - h[:m])
        g[:m] += -h[s:s+m]
    else:
        r = -s
        m = n - r
        if m <= 0:
            return g
        g[:m] += (1.0 - h[r:r+m])
        g[r:r+m] += -h[:m]
    return g

def _exact_line(h, d, R0, dx):
    n = len(h)
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    pos = d > 1e-10
    neg = d < -1e-10
    lo, hi = -1.0, 1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-12:
        return 0.0, float(np.max(R0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(R0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def _lp_dir(h, R0, active_idx, max_step=0.4):
    n = len(h)
    M = len(active_idx)
    if M == 0:
        return None
    grads = np.zeros((M, n))
    for k, idx in enumerate(active_idx):
        s = idx - (n - 1)
        grads[k] = _row_grad(h, s)
    d = cp.Variable(n)
    t = cp.Variable()
    cons = [
        t >= R0[active_idx] + grads @ d,
        cp.sum(d) == 0,
        d >= -max_step,
        d <= max_step,
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except Exception:
            return None
    if d.value is None:
        return None
    dv = d.value.copy()
    dv[np.abs(dv) < 1e-10] = 0.0
    if np.max(np.abs(dv)) < 1e-8:
        return None
    return dv

def _grad_dir(h, R0, active_idx):
    n = len(h)
    scores = R0 * (2.0 / n)
    max_s = float(np.max(scores))
    g = np.zeros(n)
    for idx in active_idx:
        s = idx - (n - 1)
        gr = _row_grad(h, s)
        w = np.exp((scores[idx] - max_s) / (0.008 * max_s + 1e-15))
        g += w * gr
    d = -g
    d -= np.mean(d)
    nm = np.linalg.norm(d)
    if nm < 1e-10:
        return None
    return d / nm

def _lp_loop(h, deadline, max_iters=500, use_lp=True):

    target = n / 2.0

        elif len(active_idx) < 4:
            active_idx = np.argsort(-scores)[:8]

        d = None
        if use_lp:
            d = _lp_dir(h, R0, active_idx, max_step=0.3)
        if d is None:
            d = _grad_dir(h, R0, active_idx)
        if d is None:
            stall += 1
            if stall > 30:
                break
            continue

        # Check for cycling

            recent = np.array(last_dirs[-5:])
            norms = np.linalg.norm(recent, axis=1)
            norms[norms < 1e-10] = 1.0

            if np.mean(dot) > 0.95:

    return project(h_new, n_new / 2.0)
