            stall += 1
            radius *= 0.85
            if radius < 1e-8:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.9:
                radius = min(radius * 1.05, 0.06)
            else:
                radius = max(radius * 0.92, 0.002)
        else:
            stall += 1
            radius *= 0.80
    return best_h, best_c5

def _pair_refine_grad(h, deadline, max_passes=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(888)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 80)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.000001
    patience = 0
    max_patience = 20000
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])
    it = 0
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        batch_size = 300
        for _ in range(batch_size):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            it += 1
            move_type = rng.random()
            if move_type < 0.7:
                i = ones_list[rng.integers(len(ones_list))]
                j = zeros_list[rng.integers(len(zeros_list))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
            else:
                i1 = ones_list[rng.integers(len(ones_list))]
                i2 = ones_list[rng.integers(len(ones_list))]
                j1 = zeros_list[rng.integers(len(zeros_list))]
                j2 = zeros_list[rng.integers(len(zeros_list))]
                if i1 == i2 or j1 == j2 or i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                    continue
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = list(np.where(h > 0.5)[0])
                zeros_list = list(np.where(h <= 0.5)[0])
            else:
                if move_type < 0.7:
                    h[i] = 1.0
                    h[j] = 0.0
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
        T = max(T_min, T * 0.9998)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _structured_perturb(h, rng, scale):
    """Structured perturbation that respects correlation structure."""
    n = len(h)
    pert = np.zeros(n)
    n_pert = rng.integers(n // 4, n // 2)
    idx = rng.choice(n, size=n_pert, replace=False)
    perturb[idx] = rng.normal(0, scale, n_pert)
    # Add a low-frequency component
    freq = rng.uniform(0.5, 3.0)
    pert += scale * 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
    return h + pert

def _multi_start_loop(h, deadline, rng, n):
    """Try structured perturbations and quick LP to escape local minima."""
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.02
    for attempt in range(200):
        if time.time() > deadline:
            break
        h_pert = _structured_perturb(best_h, rng, scale)
        h_pert = _project(h_pert, n / 2.0)
        c5_init = _exact_c5(h_pert)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_pert.copy()
            scale = max(0.002, scale * 0.85)
            continue
        sub_dl = min(deadline, time.time() + 0.15)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.04, K=15, adaptive_k=True)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.15)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.99

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick multi-start (8%)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        for ph in [0.0, 0.8, 1.6, 2.4]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(12):
        starts.append(_block_init(n0, rng, rng.integers(3, 18)))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.08, K=12, adaptive_k=True)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Intensive LP with adaptive K (35%)
    d2 = t0 + budget_s * 0.43
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.15, K=20, use_smooth=False, adaptive_k=True)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP with smooth backup (12%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.06, K=25, use_smooth=True, adaptive_k=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP polish (13%)
    d4a = t0 + budget_s * 0.63
    h4a, c5_4a = _binary_sa(best_h, d4a, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    d4 = t0 + budget_s * 0.68
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, radius_init=0.05, K=15, adaptive_k=True)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Pair refinement (8%)
    d5 = t0 + budget_s * 0.76
    h5, c5_5 = _pair_refine_grad(best_h, d5, max_passes=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (12%)
    d6 = t0 + budget_s * 0.88
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18, -20, 20, 24, -24, 28, -28, 32, -32]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d6, time.time() + budget_s * 0.018)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.08, K=15, adaptive_k=True)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.015:
                sub_dl2 = min(d6, time.time() + budget_s * 0.012)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d6:
                    h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + budget_s * 0.008),
                                         radius_init=0.04, K=12, adaptive_k=True)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Structured perturbation escape (7%)
    d7 = t0 + budget_s * 0.95
    h7, c5_7 = _multi_start_loop(best_h, d7, rng, len(best_h))
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Ultra-fine polish (5%)
    d8 = deadline
    h8, c5_8 = _polish_loop(best_h, d8, max_iters=80000)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
