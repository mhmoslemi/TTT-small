```python
def _compute_B_vec(h, i, j, n):
    u = 1.0 - h
    # For each shift d, B_{n-1+d} = u[i+d] - h[i-d] - u[j+d] + h[j-d]
    # We can compute this by constructing the full vectors
    # u_shifted[d] = u[i+d] for valid i+d, 0 otherwise
    # h_shifted[d] = h[i-d] for valid i-d, 0 otherwise
    
    B = np.zeros(2*n-1)
    # For i: u[i+d] where d ranges so that i+d is in [0,n-1], i.e., d in [-i, n-1-i]
    # h[i-d] where i-d is in [0,n-1], i.e., d in [i-(n-1), i]
    # Combined with the - sign for the j terms
    
    # Let's just do it directly with advanced indexing
    d = np.arange(-(n-1), n)  # all shifts
    # u[i+d] where valid
    idx_u_i = i + d
    mask = (idx_u_i >= 0) & (idx_u_i < n)
    B[mask] += u[idx_u_i[mask]]
    # -h[i-d] where valid
    idx_h_i = i - d
    mask2 = (idx_h_i >= 0) & (idx_h_i < n)
    B[mask2] -= h[idx_h_i[mask2]]

    idx_h_j = j - d
    mask4 = (idx_h_j >= 0) & (idx_h_j < n)
    B[mask4] += h[idx_h_j[mask4]]
```

```python
    """Find optimal alpha for direction e_i - e_j."""
    dx = 2.0 / n
    # Feasible range for alpha
    alpha_min = max(-h[i], h[j] - 1.0)
    alpha_max = min(1.0 - h[i], h[j])
    if alpha_max - alpha_min < 1e-15:
        return 0.0, float(np.max(S) * dx)
    
    # Compute B and C
    B = _compute_B_vec(h, i, j, n)
    C = np.zeros(2*n-1)
    C[n-1] = 2.0
    C[n-1 + j - i] = -1.0
    C[n-1 + i - j] = -1.0
    
    # Evaluate f(alpha) = dx * max_k (S[k] + alpha*B[k] - alpha^2*C[k])
    def f(alpha):
        vals = S + alpha * B - alpha * alpha * C
        return float(np.max(vals) * dx)
    
    # Grid search
    n_grid = 30
    alphas = np.linspace(alpha_min, alpha_max, n_grid)
    f_vals = np.array([f(a) for a in alphas])
    best_idx = int(np.argmin(f_vals))
    
    # Golden section refinement
    lo = alphas[max(0, best_idx-2)]
    hi = alphas[min(n_grid-1, best_idx+2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)

    opt_val = f(opt_alpha)

    candidates = [(0.0, float(np.max(S) * dx)), (opt_alpha, opt_val),

    best = min(candidates, key=lambda x: x[1])
```

```python
def _compute_grad(h, S, gamma):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    # Soft-max weights
    S_dx = S * dx
    S_max = np.max(S_dx)
    w = np.exp(gamma * (S_dx - S_max))
    w /= np.sum(w)
    
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-18:
            continue
        d = k - (n-1)
        wt = w[k] * dx
        # (1-h[t+d]) for valid t: t+d in [0,n-1] means t in [max(0,-d), min(n-1, n-1-d)]
        # Contribution: u[t+d] shifted
        if d >= 0:
            # t+d valid: t in [0, n-1-d]
            # u[t+d] = u[d : n]
            g[:n-d] += wt * u[d:]
            # h[t-d] valid: t-d in [0,n-1] means t in [d, n-1]
            # h[t-d] = h[0 : n-d]
            g[d:] -= wt * h[:n-d]
        else:
            p = -d
            # t+d = t-p valid: t in [p, n-1]
            # u[t-p] = u[0 : n-p]
            g[p:] += wt * u[:n-p]
            # h[t-d] = h[t+p] valid: t in [0, n-1-p]
            # h[t+p] = h[p : n]
            g[:n-p] -= wt * h[p:]
    return g
```

```python
def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
def _grad(h, S, T):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    S_dx = S * dx
    S_max = np.max(S_dx)
    w = np.exp((S_dx - S_max) / T)
    w /= np.sum(w)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        wt = w[k]
        if d >= 0:
            m = n - d
            if m > 0:

                g[d:] -= wt * h[:m]
        else:
            p = -d
            m = n - p
            g[p:] += wt * u[:m]
            g[:m] -= wt * h[p:]
    return g * dx
```

```python
def _sa_kick(h, deadline, T_init=0.002, max_iter=5000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    S = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        # Random pair swap
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        # Determine max step
        a_max = min(1.0 - h[i], h[j])
        if a_max < 1e-12:
            continue
        a = a_max * rng.uniform(0.3, 1.0)
        h[i] += a
        h[j] -= a
        S_new = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S_new) * dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            S = S_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] -= a
            h[j] += a
        T *= 0.9995
    
    return best_h, best_c5
```

```python
def _resample(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)
```

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _pair_line(S, h, i, j, n, dx):

def _grad(h, S, T):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    S_dx = S * dx
    S_max = np.max(S_dx)
    w = np.exp((S_dx - S_max) / T)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n)

    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue

        else:
            p = -d
            m = n - p
            g[p:] += wt * u[:m]
