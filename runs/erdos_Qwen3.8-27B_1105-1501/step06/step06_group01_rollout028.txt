def _lp_dir(h, G, S, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def _line_search_dir(h, d, dx, radius):
    n = len(h)
    target = n / 2.0
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    grid = np.linspace(t_lo, t_hi, 30)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1)
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]

    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _lp_optimize(h, deadline, start_radius=0.12):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = start_radius
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 60:
            break
        S = _corr_vals(h)
        G = _build_G(h, dx)
        p = _lp_dir(h, G, S, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.7
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search_dir(h, p, dx, radius)
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-14:
            h = _project(h + alpha * p, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            predicted = float(np.max(S + G @ p * alpha))
            ratio = cur_c5 / max(predicted, 1e-15)
            if ratio < 0.6:
                radius = min(radius * 1.2, 0.4)
            elif ratio > 1.8:
                radius *= 0.75
            else:
                radius = min(radius * 1.03, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if radius < 0.005:
                break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _diverse_start(n, rng):
    x = np.arange(n) / n
    freq1 = rng.integers(1, 6)
    freq2 = rng.integers(1, 6)
    if freq2 == freq1:
        freq2 = freq1 + 1
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a1 = rng.uniform(0.2, 0.45)
    a2 = rng.uniform(0.1, 0.3)
    h = 0.5 + a1 * np.sin(2 * np.pi * freq1 * x + p1) + a2 * np.sin(2 * np.pi * freq2 * x + p2)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def _binary_start(n, rng):
    target = n // 2
    strategy = rng.integers(0, 4)
    if strategy == 0:
        h = np.zeros(n)
        positions = np.sort(rng.choice(n, size=target, replace=False))
        h[positions] = 1.0
    elif strategy == 1:
        h = np.zeros(n)
        block = n // target
        for i in range(target):
            h[i * block] = 1.0
    elif strategy == 2:
        h = np.zeros(n)
        shift = rng.integers(0, n)
        for i in range(target):
            h[(i * 2 + shift) % n] = 1.0
    else:
        h = np.zeros(n)
        h[::2][:target] = 1.0
        perm = rng.permutation(n)
        h = h[perm]
    return _project(h + rng.normal(0, 0.08, n), n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0)
        h0 = _project(0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / n0), n0 / 2.0)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Smooth descent to find good basin (30% budget)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _smooth_descend(best_h, d1, tau_init=0.012, tau_min=0.003)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refinement (20% budget)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _lp_optimize(best_h, d2, start_radius=0.15)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: More smooth descent from LP result (12% budget)
    d3 = t0 + budget_s * 0.62
    h3, c5_3 = _smooth_descend(best_h, d3, tau_init=0.006, tau_min=0.002)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Diverse starts (15% budget)
    d4 = t0 + budget_s * 0.77
    n_starts = 4
    for si in range(n_starts):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        if si % 2 == 0:
            h_s = _diverse_start(n0, rng)
        else:
            h_s = _binary_start(n0, rng)
        h_s, c5_s = _smooth_descend(h_s, sub_dl, tau_init=0.010, tau_min=0.003)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            if time.time() < d4:
                sub_dl2 = min(d4, time.time() + budget_s * 0.02)
                h_s2, c5_s2 = _lp_optimize(best_h, sub_dl2, start_radius=0.10)
                if c5_s2 < best_c5:
                    best_c5 = c5_s2
                    best_h = h_s2.copy()

    # Phase 5: Multi-resolution (10% budget)
    d5 = t0 + budget_s * 0.87
    tried_n = set()
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16]
    n_cur = len(best_h)
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_cur + dn
        if n_try < 30 or n_try > 180 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _smooth_descend(h_r, sub_dl, tau_init=0.008, tau_min=0.003)
        if c5_r < best_c5 - 1e-6:
            if time.time() < d5:
                sub_dl2 = min(d5, time.time() + budget_s * 0.008)
                h_r2, c5_r2 = _lp_optimize(h_r, sub_dl2, start_radius=0.08)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Final LP polish (6% budget)
    d6 = deadline
    h6, c5_6 = _lp_optimize(best_h, d6, start_radius=0.06)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
