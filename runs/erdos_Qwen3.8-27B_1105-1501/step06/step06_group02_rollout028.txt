    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

    Z = float(np.sum(exp_s))
    w = exp_s / Z  # softmax weights
    f = tau * (s_max / tau + np.log(Z))  # = s_max + tau*log(Z)

import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * dx
    s_max = float(np.max(s))
    exp_s = np.exp((s - s_max) / tau)
    Z = float(np.sum(exp_s))
    w = exp_s / Z
    f = s_max + tau * np.log(Z)
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - center
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            grad[d:d+m] += wk * (1.0 - h[:m])
            grad[:m] += -wk * h[d:d+m]
        else:
            r = -d
            m = n - r
            if m <= 0:
                continue
            grad[:m] += wk * (1.0 - h[r:r+m])
            grad[r:r+m] += -wk * h[:m]
    grad *= dx / tau
    return f, grad

    return float(np.sum(h)) - len(h) / 2.0

    return np.ones(len(h))

            active_idx = np.argsort(-scores)[:8]
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        tried = 0
        for i in range(n):
            if tried > max_pairs * 10:
                break
            for j in range(n):
                if i == j:
                    continue
                tried += 1
                lo = -min(h[j], 1.0 - h[i])
                hi = min(h[i], 1.0 - h[j])
                if hi - lo < 1e-14:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                C = np.correlate(d, d, 'full')
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
                f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
                for _ in range(25):
                    if b - a < 1e-13:
                        break
                    if f1 < f2:
                        b, c2, f2 = c2, c1, f1
                        c1 = b - gr * (b - a)
                        f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
                    else:
                        a, c1, f1 = c1, c2, f2
                        c2 = a + gr * (b - a)
                        f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
                opt = (a + b) / 2.0
                val = float(np.max(S0 + opt * T - opt*opt * C)) * dx
                f_lo_v = float(np.max(S0 + lo * T - lo*lo * C)) * dx
                f_hi_v = float(np.max(S0 + hi * T - hi*hi * C)) * dx
                if f_lo_v < val:
                    opt, val = lo, f_lo_v
                if f_hi_v < val:
                    opt, val = fi, f_hi_v
                if abs(opt) > 1e-10 and val < best_new_c5 - 1e-14:
                    best_new_c5 = val
                    best_alpha = opt
                    best_pair = (i, j)
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30 and rng.random() < 0.3:
                idx = rng.integers(n, size=3)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.03, 0.03)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    return best_h, best_c5
