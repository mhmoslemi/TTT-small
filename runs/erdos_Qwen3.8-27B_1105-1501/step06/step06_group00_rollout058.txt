import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:d + m] += u[:m]
                G[r, :m] += -h[d:d + m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += u[g:]
                G[r, g:g + m] += -h[:m]
    return G

def _smooth_max_grad(h, scores, G, beta):
    n = G.shape[1]
    y = scores * (2.0 / n)
    M = float(np.max(y))
    exps = np.exp(np.clip(beta * (y - M), -50.0, 0.0))
    s = float(np.sum(exps))
    if s < 1e-30:
        w = np.zeros(len(y))
        top = np.where(y >= M - 1e-15)[0]
        if len(top) == 0:
            top = np.array([np.argmax(y)])
        w[top] = 1.0 / len(top)
    else:
        w = exps / s
    g = G.T @ w
    return g

def _line_search_pair(h, i, j, radius, scores, G):
    n = len(h)
    dx = 2.0 / n
    a_lo = max(-radius, -h[i], h[j] - 1.0)
    a_hi = min(radius, 1.0 - h[i], h[j])
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(scores) * dx)
    T = G[:, i] - G[:, j]
    C = np.zeros(2 * n - 1)
    C[n - 1] = 2.0
    delta = i - j
    if delta != 0:
        idx1 = n - 1 + delta
        idx2 = n - 1 - delta
        if 0 <= idx1 < len(C):
            C[idx1] = -1.0
        if 0 <= idx2 < len(C):
            C[idx2] = -1.0

    def obj(alpha):
        return float(np.max(scores + alpha * T - alpha * alpha * C)) * dx

    n_grid = 60
    grid = np.linspace(a_lo, a_hi, n_grid)
    best_v = float('inf')
    best_a = 0.0
    for k in range(n_grid):
        v = obj(grid[k])
        if v < best_v:
            best_v = v
            best_a = grid[k]
    idx = int(round((best_a - a_lo) / max(a_hi - a_lo, 1e-15) * (n_grid - 1)))
    lo = grid[max(0, idx - 2)]
    hi = grid[min(n_grid - 1, idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = obj(a1)
    f2 = obj(a2)
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1
            f1 = obj(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = obj(a2)
    opt_a = (lo + hi) / 2.0
    opt_v = obj(opt_a)
    if opt_v < best_v:
        best_v = opt_v
        best_a = opt_a
    v0 = obj(0.0)
    if v0 < best_v:
        best_v = v0
        best_a = 0.0

def _pair_descent(h, deadline, beta_init=20.0, radius_init=0.12, max_iter=8000):

    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_init
    radius = radius_init
    stall = 0
    stall_hard = 0
    rng = np.random.default_rng(12345)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall_hard > 300:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _build_G(h)
        g = _smooth_max_grad(h, scores, G, beta)
        order = np.argsort(g)
        n_cand = min(30, n // 3)
        sinks = order[:n_cand]
        sources = order[-n_cand:]
        best_new_c5 = cur_c5
        best_a = 0.0
        best_i = -1
        best_j = -1
        pairs_tried = 0
        for si in range(n_cand):
            i = sources[n_cand - 1 - si]
            if h[i] < 1e-9 or h[i] > 1.0 - 1e-9:
                continue
            for kj in range(min(8, n_cand)):
                j = sinks[kj]
                if h[j] < 1e-9:
                    continue
                a, cv = _line_search_pair(h, i, j, radius, scores, G)
                pairs_tried += 1
                if cv < best_new_c5 - 1e-14:
                    best_new_c5 = cv
                    best_a = a
                    best_i = i
                    best_j = j
            if pairs_tried >= 60:
                break
        if best_i < 0:
            for _ in range(8):
                if time.time() > deadline:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or h[i] < 1e-9 or h[i] > 1 - 1e-9 or h[j] < 1e-9:
                    continue
                a, cv = _line_search_pair(h, i, j, radius, scores, G)
                if cv < best_new_c5 - 1e-14:
                    best_new_c5 = cv
                    best_a = a
                    best_i = i
                    best_j = j
                    break
        if best_i >= 0:
            h[best_i] += best_a
            h[best_j] -= best_a
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            stall_hard = 0
            radius = min(radius * 1.04, 0.28)
            beta = max(beta * 0.98, 6.0)
        else:

            stall_hard += 1
            radius *= 0.72
            if stall % 20 == 0:

                    radius = 0.06
            if radius < 1e-5:
                break
    return best_h, best_c5

def _lp_refine(h, deadline, radius=0.08, max_iter=40):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        G = _build_G(h)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -scores
        A_eq = np.ones(1, n + 1)
        A_eq[0, n] = 0.0
        b_eq = np.zeros(1)
        bounds = []
        for jj in range(n):
            lo = max(-radius, -h[jj])
            hi = min(radius, 1.0 - h[jj])
            bounds.append((lo, hi))
        bounds.append((None, None))
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                break
            d = res.x[:n]
        except Exception:
            break
        if float(np.max(np.abs(d))) < 1e-13:
            break
        a_lo = 0.0
        a_hi = 1.0
        pos = d > 1e-14
        neg = d < -1e-14
        if np.any(pos):
            a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
        if a_hi <= a_lo:
            break
        T = G @ d
        C = np.correlate(d, d, mode='full')
        def f(alpha):
            return float(np.max(scores + alpha * T - alpha * alpha * C)) * (2.0 / n)
        best_a = 0.0
        best_v = f(0.0)
        for k in range(30):
            alpha = a_lo + (a_hi - a_lo) * k / 29.0
            v = f(alpha)
            if v < best_v:
                best_v = v
                best_a = alpha
        if best_v < cur_c5 - 1e-13:
            h = h + best_a * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=15000):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j or cur[i] == cur[j]:
            continue
        h_try = cur.copy()
        h_try[i], h_try[j] = 0.0, 1.0
        c_try = _exact_c5(h_try)
        if c_try < cur_c5 - 1e-14:
            cur = h_try
            cur_c5 = c_try
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amp
    p += amp * 0.1 * rng.standard_normal(n)
    return _project(h + p, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Pair descent (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _pair_descent(h_init, d1, beta_init=20.0, radius_init=0.12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary + LP polish (15%)
    d2 = t0 + budget_s * 0.35
    hb = _to_binary(best_h, len(best_h))
    sd = min(d2, time.time() + budget_s * 0.06)
    hb, cb = _binary_local(hb, sd)
    if cb < best_c5:
        best_c5 = cb
        best_h = hb.copy()
    sd2 = min(d2, time.time() + budget_s * 0.05)
    hp = _project(hb.copy(), len(hb) / 2.0)
    hp, cp = _lp_refine(hp, sd2, radius=0.06)
    if cp < best_c5:
        best_c5 = cp
        best_h = hp.copy()
    sd3 = min(d2, time.time() + budget_s * 0.04)
    hp2, cp2 = _pair_descent(_project(best_h.copy(), len(best_h) / 2.0), sd3,
                             beta_init=15.0, radius_init=0.08)
    if cp2 < best_c5:
        best_c5 = cp2
        best_h = hp2.copy()

    # Phase 3: Multi-n exploration (30%)
    d3 = t0 + budget_s * 0.65
    nc = len(best_h)
    offsets = [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, -16]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        nt = nc + dn
        if nt < 72 or nt > 240 or nt in tried:
            continue
        tried.add(nt)
        hr = _resample(best_h, nt)
        sd = min(d3, time.time() + budget_s * 0.025)
        hr, cr = _pair_descent(hr, sd, beta_init=18.0, radius_init=0.10)
        if time.time() < d3 - 2:
            sd2 = min(d3, time.time() + budget_s * 0.012)
            hr2, cr2 = _lp_refine(_project(hr.copy(), nt / 2.0), sd2, radius=0.05)
            if cr2 < cr:
                cr = cr2
                hr = hr2
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            nc = nt

    # Phase 4: Intensive refinement (25%)
    d4 = t0 + budget_s * 0.90
    nb = len(best_h)
    for trial in range(6):
        sd = min(d4, time.time() + budget_s * 0.035)
        if time.time() > sd:
            break
        h4 = _project(best_h.copy(), nb / 2.0)
        rad = 0.08 / (trial + 1)
        h4, c4 = _pair_descent(h4, sd, beta_init=16.0 + trial * 2, radius_init=rad)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()

    # Phase 5: Basin hopping + final LP (10%)
    d5 = deadline - 3.0
    scale = 0.02
    for _ in range(80):
        if time.time() > d5:
            break
        hp = _perturb(best_h, scale, rng)
        sd = min(d5, time.time() + budget_s * 0.007)
        hp, cp = _pair_descent(hp, sd, beta_init=20.0, radius_init=0.07)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
            scale = max(0.002, scale * 0.8)
        else:
            scale = min(0.05, scale * 1.08)

    # Final LP polish
    sd = deadline - 0.5
    hf = _project(best_h.copy(), len(best_h) / 2.0)
    hf, cf = _lp_refine(hf, sd, radius=0.03, max_iter=20)
    if cf < best_c5:
        best_c5 = cf
        best_h = hf.copy()

    # Final pair descent
    if deadline - time.time() > 2:
        hf2, cf2 = _pair_descent(_project(best_h.copy(), len(best_h) / 2.0),
                                 deadline - 0.1, beta_init=25.0, radius_init=0.02)
        if cf2 < best_c5:
            best_c5 = cf2
            best_h = hf2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
