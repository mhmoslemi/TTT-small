from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if res > 1e-12:
        mask = x < 1.0
    elif res < -1e-12:
        mask = x > 0.0
    else:
        return x
    cnt = np.sum(mask)
    if cnt > 0:
        x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, dx

def _smooth_grad(h, tau, c, dx):
    n = len(h)
    # Numerically stable softmax
    tc = tau * c
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    p = exp_tc / np.sum(exp_tc)
    # Gradient
    v = 1.0 - 2.0 * h
    grad = dx * np.correlate(p, v, 'valid')
    return grad

def _smooth_obj(h, tau, c):
    tc = tau * c
    return np.max(tc) / tau + np.log(np.sum(np.exp(tc - np.max(tc)))) / tau

            c_try, _ = _corr_vals(h_try)

def _exact_c5(h):

def _project(v, target):

        return np.zeros(n)
