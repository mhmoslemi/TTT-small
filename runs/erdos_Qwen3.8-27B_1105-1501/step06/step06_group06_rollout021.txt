            if actual_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                if abs(alpha) > 0.6 * radius:
                    radius = min(radius * 1.3, 0.35)
                else:
                    radius = min(radius * 1.03, 0.25)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break

    return best_h, best_c5

def _slsqp_optimize(h0, deadline, max_outer=20):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _all_corrs(h)
        K = min(20, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        t_init = cur_c5 + 1e-5
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]

        def obj_fn(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]

        def ineq_jac(x):
            hx = x[:n]
            G = _gradients(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J

        def eq_fn(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 60, 'ftol': 1e-15})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_swaps(h, deadline, n_trials=5000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    rng = np.random.default_rng(777)
    h_best = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    trials_per_iter = 200
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(-0.1, 0.1)
        lo_i = max(-h[i], -delta)
        hi_i = min(1.0 - h[i], -delta)
        if delta > 0:
            lo_i = max(h[i] * 0, 0.0)
            lo_i = max(0.0, -h[i])
            hi_i = min(1.0 - h[i], delta)
        else:
            lo_i = max(delta, -h[i])
            hi_i = min(0.0, 1.0 - h[i])
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        t_min, t_max = _feasible_interval(h, d, 0.5)
        if t_max - t_min < 1e-12:
            continue
        alpha, new_c5 = _exact_line_search(h, d, t_min, t_max)
        if new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    h_best = h.copy()
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = np.ones(n0) * 0.5

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Active descent (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _active_descent(h, d1, max_iter=3000, rng_seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: SLSQP (10% time)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _slsqp_optimize(best_h, d2, max_outer=15)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: Pair swaps (10% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _pair_swaps(best_h, d3, n_trials=8000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Multi-resolution (30% time)
    d4 = t0 + budget_s * 0.75
    offsets = []
    for dn in range(-16, 17, 2):
        if dn != 0:
            offsets.append(dn)
    offsets.sort(key=lambda x: abs(x))
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 70 or n_try > 170 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _active_descent(h_r, sub_dl, max_iter=500, rng_seed=100 + n_try)
        if time.time() < d4 - 3:
            sub_dl2 = min(d4, time.time() + budget_s * 0.012)
            h_r, c5_r2 = _slsqp_optimize(h_r, sub_dl2, max_outer=8)
            if c5_r2 < c5_r:
                c5_r = c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Multi-start (10% time)
    d5 = t0 + budget_s * 0.85
    starts = []
    for freq in [1, 2, 3, 5]:
        for ph in [0.0, 1.0, 2.0]:
            x = np.arange(best_n)
            hs = 0.5 + 0.38 * np.sin(2.0 * np.pi * freq * x / best_n + ph)
            hs = hs + rng.normal(0, 0.04, best_n)
            starts.append(_project(hs, best_n / 2.0))
    for s in starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_s, c5_s = _active_descent(s, sub_dl, max_iter=300, rng_seed=200)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 6: Final active descent + SLSQP polish (10% time)
    d6 = deadline - 3
    h6, c5_6 = _active_descent(best_h, min(d6, time.time() + budget_s * 0.04), max_iter=500, rng_seed=300)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)
    if time.time() < d6:
        h6b, c5_6b = _slsqp_optimize(best_h, d6, max_outer=10)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
            best_n = len(h6b)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
