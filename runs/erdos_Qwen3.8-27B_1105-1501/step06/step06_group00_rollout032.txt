def _project(v, target):

    n = len(v)

def _smooth_max_gradient(h, S, gamma=50.0):
    """
    Compute gradient of a smooth approximation to max(S) using log-sum-exp.
    Returns the gradient vector (length n).
    """
    n = len(h)
    # Smooth max: F ≈ (1/gamma) * log(sum(exp(gamma*S)))
    # Gradient: ∇F = (exp(gamma*S) / sum(exp(gamma*S))) · ∇S
    # So weights = softmax(gamma * S)
    
    Smax = np.max(S)
    z = np.exp(gamma * (S - Smax))
    w = z / np.sum(z)
    
    # Now compute gradient: ∑_r w[r] * ∂S[r]/∂h[k] for each k
    # ∂S[r]/∂h[k] = (1-h[k+D_r])*[valid] - h[k-D_r]*[valid]
    # where D_r = r - (n-1)
    
    # Vectorized: for each k, sum over r of w[r] * [(1-h[k+D_r])*[0<=k+D_r<n] - h[k-D_r]*[0<=k-D_r<n]]
    # = sum_r w[r]*(1-h[k+D_r])*[0<=k+D_r<n] - sum_r w[r]*h[k-D_r]*[0<=k-D_r<n]
    
    # Let's define W_plus[k] = sum_r w[r] * [0<=k+D_r<n] ... this isn't quite right
    # because h[k+D_r] depends on both k and r.
    
    # Alternative: build the gradient matrix G (2n-1 x n) and compute w @ G
    # G[r, k] = (1-h[k+D_r])*[0<=k+D_r<n] - h[k-D_r]*[0<=k-D_r<n]
    
    # This is O(n^2) to build but that's fine for n <= 256.
    
    D = np.arange(-(n-1), n)  # D[r] = r - (n-1)
    
    G = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        d = D[r]
        if d >= 0:
            m = n - d
            # x ranges 0 to m-1
            G[r, :m] += (1.0 - h[d:d+m])  # (1-h[x+d]) for x=0..m-1
            G[r, d:n] += -h[:m]  # -h[x] for x=d..n-1 (from -h[x-d] with x-d=0..m-1)
        else:
            p = -d
            m = n - p
            # x ranges p to n-1
            G[r, p:n] += (1.0 - h[:m])  # (1-h[x-p]) for x=p..n-1
            G[r, :m] += -h[p:p+m]  # -h[x+p] for x=0..m-1 (from -h[x+p] with x+p=p..n-1)
    
    grad = w @ G
    # Center the gradient (subtract mean) to preserve sum constraint direction
    grad -= np.mean(grad)
    return grad

def _pair_descent(h, deadline, gamma=50.0, radius=0.3, K=30, max_iters=500):
    """
    Core descent loop using pair-mass transfer with smooth-max gradient guidance.
    """
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(777)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        
        S = _compute_S(h)
        grad = _smooth_max_gradient(h, S, gamma)
        
        # Rank pair directions: score(i,j) = grad[j] - grad[i] (we want to move mass from high grad to low grad)
        # Actually: directional derivative of F along e_i - e_j is grad[i] - grad[j]
        # We want negative directional derivative, so grad[i] - grad[j] < 0, i.e., grad[i] < grad[j]
        # Score = grad[j] - grad[i], pick largest scores
        
        # For efficiency, pick top K pairs by score
        # Full ranking is O(n²), which is fine for n <= 256
        # scores[i,j] = grad[j] - grad[i]
        # Best pairs have high grad[j] and low grad[i]
        
        # Sort indices by grad
        idx_sorted = np.argsort(grad)  # ascending
        # Best pairs: i from low grad end, j from high grad end
        n_low = min(K//2 + 2, n//2)
        n_high = min(K//2 + 2, n//2)
        low_candidates = idx_sorted[:n_low]
        high_candidates = idx_sorted[-n_high:]
        
        # Generate candidate pairs
        candidates = []
        for i in low_candidates:
            for j in high_candidates:
                if i != j:
                    candidates.append((i, j))
        
        # Shuffle for diversity
        rng.shuffle(candidates)
        candidates = candidates[:K]
        
        improved = False
        for (i, j) in candidates:
            if time.time() > deadline:
                break
            a, v = _pair_line_search(h, i, j, dx, S, radius)
            if v < cur_c5 - 1e-13:
                h[i] += a
                h[j] -= a
                # Numerical safety
                h = np.clip(h, 0.0, 1.0)
                # Verify sum constraint
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                break  # Restart with new S and grad
        
        if not improved:
            stall += 1
            radius *= 0.85
            if radius < 0.005:
                break
            # Try random pairs as escape
            if stall % 10 == 5:
                for _ in range(5):
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j: continue
                    a, v = _pair_line_search(h, i, j, dx, S, radius*2)
                    if v < cur_c5 - 1e-13:
                        h[i] += a
                        h[j] -= a
                        h = np.clip(h, 0.0, 1.0)
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        improved = True
                        break
    
    return best_h, best_c5

def _binary_descent(h_bin, deadline, max_iters=20000):
    """Descent on binary (0/1) vectors by pair swaps."""
    n = len(h_bin)
    h = h_bin.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(42)
    stall = 0
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200 or len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            ones = list(np.where(h > 0.5)[0])
            zeros = list(np.where(h <= 0.5)[0])
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            h[oi] = 1.0
            h[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    f1 = rng.uniform(0.5, 4.0)
    f2 = rng.uniform(2.0, 8.0)
    p1 = rng.uniform(0, 2*np.pi)
    p2 = rng.uniform(0, 2*np.pi)
    a1 = rng.uniform(0.5, 1.0)
    a2 = rng.uniform(0.3, 0.7)
    p = a1*np.sin(2*np.pi*f1*x/n + p1) + a2*np.sin(2*np.pi*f2*x/n + p2)
    p += 0.1*amp*rng.normal(0, 1, n)
    p *= amp
    return _project(h + p, n/2.0)
