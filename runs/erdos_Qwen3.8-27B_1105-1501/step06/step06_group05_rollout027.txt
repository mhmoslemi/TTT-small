```python
        g[n] = 1.0

        z = x[n]
```

```python
def _compute_jacobian(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        ell = r - (n - 1)
        if ell >= 0:
            # positions m where m >= ell and m < n: contribute (1-h[m-ell])
            # positions m where m >= 0 and m < n-ell: contribute -h[m+ell]
            m1 = np.arange(ell, n)  # m >= ell
            m2 = np.arange(0, n - ell)  # m < n - ell
            J[r, m1] += dx * (1.0 - h[m1 - ell])
            J[r, m2] += -dx * h[m2 + ell]
        else:
            q = -ell
            m1 = np.arange(0, n - q)  # m < n - q
            m2 = np.arange(q, n)  # m >= q
            J[r, m1] += dx * (1.0 - h[m1 + q])
            J[r, m2] += -dx * h[m2 - q]
    return J
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_N = 0


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _obj(x):
    return x[_N]


def _obj_jac(x):
    g = np.zeros(_N + 1)
    g[_N] = 1.0
    return g


def _ineq(x):
    n = _N
    h = x[:n]
    z = x[n]
    s = _corr(h) * (2.0 / n)
    return z - s


def _ineq_jac(x):
    n = _N
    h = x[:n]
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n + 1))
    J[:, n] = 1.0
    for r in range(2 * n - 1):
        ell = r - (n - 1)
        if ell >= 0:
            J[r, ell:] -= dx * (1.0 - h[:n - ell])
            J[r, :n - ell] -= dx * h[ell:]
        else:
            q = -ell
            J[r, :n - q] -= dx * (1.0 - h[q:])
            J[r, q:] -= dx * h[:n - q]
    return J


def _eq(x):
    return np.sum(x[:_N]) - _N / 2.0


def _eq_jac(x):
    J = np.zeros(_N + 1)
    J[:_N] = 1.0
    return J


def _slsqp(h_init, maxiter=250):
    global _N
    n = len(h_init)
    _N = n
    z0 = _exact_c5(h_init) + 0.04
    x0 = np.concatenate([h_init, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = (
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
    )
    try:
        res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_sol = _project(res.x[:n], n / 2.0)
        f_sol = _exact_c5(h_sol)
        return h_sol, f_sol
    except Exception:
        return None, None


def _row_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nn = np.arange(0, n - k)
    else:
        nn = np.arange(-k, n)
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):

            break

        else:

            c2 = a + gr * (b - a)

    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:

    return t, tv


def _active_direction(h, corr, k_active, tau, rng):
    n = len(h)
    scores = corr
    m = float(np.max(scores))
    w = np.exp((scores - m) / tau)
    idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = r - (n - 1)
        G += w[r] * _row_grad(h, k)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm


def _multi_active_direction(h, corr, k_active, rng):

    m = float(np.max(corr))
    threshold = m - 0.02
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:

    nrm = float(np.linalg.norm(d))

        d -= d.mean()
        nrm = float(np.linalg.norm(d))

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 5 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)
```
