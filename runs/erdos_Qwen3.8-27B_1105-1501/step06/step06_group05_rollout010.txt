    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _reverse_complement(h):
    return 1.0 - h[::-1].copy()


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        x = np.arange(n)
        base = 0.5 + 0.3 * np.sin(2.0 * np.pi * x / n) + 0.1 * np.sin(4.0 * np.pi * x / n + 0.7)
        base = base + rng.normal(0, 0.02, n)
    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: SLSQP iterations from initial point (25% of budget)
    d1 = t0 + budget_s * 0.25
    hh, ff = _slsqp_iterate(h0, d1, max_rounds=12, top_m=25)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Try reversed complement as additional start
    if time.time() < d1 - 5:
        h_rev = _project(_reverse_complement(best_h), n0 / 2.0)
        if _exact_c5(h_rev) < best_f:
            best_f = _exact_c5(h_rev)
            best_h = h_rev.copy()
        hh2, ff2 = _slsqp_iterate(h_rev, min(d1, time.time() + 30), max_rounds=5, top_m=20)
        if ff2 < best_f:
            best_f = ff2
            best_h = hh2.copy()

    # Phase 2: Direction-based optimization (30% of budget)
    d2 = t0 + budget_s * 0.55
    hh, ff = _direction_optimize(best_h, d2, rng, pair_frac=0.22, rand_frac=0.06,
                                 k_active=10, tau=0.015)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: SLSQP again after direction optimization (20% of budget)
    d3 = t0 + budget_s * 0.75
    hh, ff = _slsqp_iterate(best_h, d3, max_rounds=12, top_m=20)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 4: Multi-resolution exploration (15% of budget)
    d4 = t0 + budget_s * 0.90
    bn = len(best_h)
    tried = set()
    for dn in [-4, 4, -6, 6, -8, 8, -2, 2, -10, 10]:
        if time.time() > d4:
            break
        n_try = bn + dn
        if n_try < 20 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        hr2, fr2 = _direction_optimize(hr, min(d4, time.time() + 20), rng,
                                       pair_frac=0.25, rand_frac=0.08, k_active=8, tau=0.02)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try
        hr3, fr3 = _slsqp_iterate(hr2, min(d4, time.time() + 15), max_rounds=5, top_m=20)
        if fr3 < best_f:
            best_f = fr3
            best_h = hr3.copy()
            bn = n_try

    # Phase 5: Final combined polish (10% of budget)
    hh, ff = _slsqp_iterate(best_h, deadline - 10, max_rounds=8, top_m=30)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    if time.time() < deadline - 5:
        hh2, ff2 = _direction_optimize(best_h, deadline - 2, rng,
                                       pair_frac=0.40, rand_frac=0.04, k_active=12, tau=0.010)
        if ff2 < best_f:
            best_f = ff2
            best_h = hh2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
