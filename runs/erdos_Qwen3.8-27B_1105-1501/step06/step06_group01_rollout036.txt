import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2*n-1, n))
    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        G[mid+r, :m] += dx * (1.0 - h[r:r+m])
        G[mid+r, r:] += -dx * h[:m]
        G[mid-r, r:] += dx * (1.0 - h[:m])
        G[mid-r, :m] += -dx * h[r:r+m]
    return G

def _pair_line_search(S0, T, Q_nonzero, dx, t_lo, t_hi):
    """Line search with Q having only a few non-zero entries.
    Q_nonzero: list of (index, value) pairs."""
    # Build Q as full vector for the max computation
    # Actually, for speed, compute max analytically:
    # max_q (S0[q] + alpha*T[q] + alpha^2*Q[q])
    # For q not in Q_nonzero: S0[q] + alpha*T[q] (linear)
    # For q in Q_nonzero: S0[q] + alpha*T[q] + alpha^2*Q_val (quadratic)
    
    # Grid search
    n_grid = 30
    grid = np.linspace(t_lo, t_hi, n_grid)
    # Base: S0 + alpha * T (for all q)
    base = S0 + grid[:, None] * T[None, :]
    # Add quadratic terms
    for q_idx, q_val in Q_nonzero:
        base[:, q_idx] += grid**2 * q_val
    obj = np.max(base, axis=1)
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    
    if best_i > 0 and best_i < n_grid - 1:
        a = grid[best_i - 1]
        b = grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    
    def _eval(alpha):
        v = S0 + alpha * T
        for q_idx, q_val in Q_nonzero:
            v[q_idx] += alpha * alpha * q_val
        return float(np.max(v))
    
    f1 = _eval(c1)
    f2 = _eval(c2)
    for _ in range(35):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _eval(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _eval(c2)
    
    opt_t = (a + b) / 2.0
    opt_val = _eval(opt_t)
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    
    return float(best_t), float(best_val)

def _local_optimize(h, deadline, max_iters=10000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    mid = n - 1
    
    tau = 0.003  # LSE temperature
    lr = 0.01    # Learning rate for LSE
    K = 150      # Number of pairs to try
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        S = _corr_vals(h)
        G = _build_G(h, dx)
        
        # --- LSE gradient step ---
        S_max = float(np.max(S))
        w = np.exp((S - S_max) / tau)
        w /= float(np.sum(w))
        grad = G.T @ w  # shape (n,)
        # Normalize gradient to unit L2
        gnorm = float(np.linalg.norm(grad))
        if gnorm > 1e-12:
            grad /= gnorm
            # Try step
            h_new = h - lr * grad
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
                lr = min(lr * 1.03, 0.05)
            else:
                lr *= 0.95
                stall += 1
        
        # --- Pair moves ---
        # Rank pairs by gradient: high grad[i], low grad[j] → good for d = e_j - e_i
        # We want to reduce grad, so move from high-grad to low-grad
        order = np.argsort(-grad)  # highest gradient first
        n_tried = 0
        improved_pair = False
        for ii in range(min(50, n)):
            if n_tried >= K:
                break
            i = order[ii]
            for jj in range(min(50, n)):
                if n_tried >= K:
                    break
                j = order[jj]
                if i == j:
                    continue
                if h[i] < 1e-6 or h[j] > 1 - 1e-6:
                    continue
                
                # Direction: d[i] = -1, d[j] = +1
                T = G[:, j] - G[:, i]
                # Q has non-zero at mid, mid+(j-i), mid+(i-j)
                dlag = j - i

                    q_entries.append((mid + dlag, dx))
                if 0 <= mid - dlag < 2*n - 1:
                    q_entries.append((mid - dlag, dx))
                
                # Feasible range
                t_lo = max(h[i] - 1.0, -h[j])
                t_hi = min(h[i], 1.0 - h[j])
                if t_hi - t_lo < 1e-12:

                alpha, new_c5 = _pair_line_search(S, T, q_entries, dx, t_lo, t_hi)
                if new_c5 < best_c5 - 1e-14:

                    h[j] += alpha
                    h = _project(h, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        improved_pair = True
                        stall = 0

                    G = _build_G(h, dx)
                    # Update grad

import numpy as np
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        G[mid + r, :m] += dx * (1.0 - h[r:r + m])
        G[mid + r, r:] += -dx * h[:m]
        G[mid - r, r:] += dx * (1.0 - h[:m])
        G[mid - r, :m] += -dx * h[r:r + m]
    return G

def _pair_search(S0, T, qe, dx, t_lo, t_hi):
    n = len(S0)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    ng = 35
    grid = np.linspace(t_lo, t_hi, ng)
    base = S0[None, :] + grid[:, None] * T[None, :]
    for qi, qv in qe:
        base[:, qi] += grid * grid * qv
    obj = np.max(base, axis=1)
    bi = int(np.argmin(obj))
    bt, bv = grid[bi], obj[bi]
    a = grid[max(bi - 1, 0)] if bi > 0 else t_lo
    b = grid[min(bi + 1, ng - 1)] if bi < ng - 1 else t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)

    def _ev(al):
        v = S0 + al * T
        for qi, qv in qe:
            v[qi] += al * al * qv
        return float(np.max(v))

    f1, f2 = _ev(c1), _ev(c2)
    for _ in range(38):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _ev(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _ev(c2)
    ot = (a + b) / 2.0
    ov = _ev(ot)
    if ov < bv:
        bt, bv = ot, ov
    return float(bt), float(bv)

def _resample(h, n_new):

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_start(n, rng, method):

    elif method == 1:

        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
        h += rng.normal(0, 0.05, n)
        return _project(h, target)
    elif method == 2:

        while pos < n:
