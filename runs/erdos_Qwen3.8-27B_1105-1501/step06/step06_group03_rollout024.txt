def _thue_morse(n):

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks < 2: return _project(h + 0.5, n/2.0)
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks-1, n-1), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    val = 0.0
    for b in range(len(boundaries)-1):
        h[boundaries[b]:boundaries[b+1]] = 1.0 - val
        val = 1.0 - val
    return _project(h, n / 2.0)

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_smooth(h, S, tau):

    dx = 2.0 / n
    M = S.max()

    ws = w.sum()

    w = w / ws
    g = np.zeros(n)
    for k in range(-(n-1), n):
        r = k + (n-1)
        wk = w[r] * dx
        if wk < 1e-14: continue
        if k >= 0:
            m = n - k
            if m <= 0: continue
            g[:m] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:m]
        else:
            d = -k
            m = n - d
            if m <= 0: continue
            g[d:] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[d:]
    return g

def _line_search_quad(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = -radius, radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos): t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg): t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14: return 0.0, float(np.max(S0) * dx)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t, best_val = grid[best_i], obj[best_i]
    if best_i > 0 and best_i < n_grid-1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0)-1.0)/2.0
    c1, c2 = b - gr*(b-a), a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(40):
        if b-a < 1e-13: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt_t = (a+b)/2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C)) * dx
    if opt_val < best_val: best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _smooth_desc(h, deadline, tau_init=0.01, radius_init=0.12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_init
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(31337)
    while time.time() < deadline:
        if stall > 80:
            if best_c5 < 0.3805: break
            h = _project(h + rng.normal(0, 0.02, n), target)
            best_h = h.copy()
            best_c5 = _exact_c5(h)
            stall = 0
            tau = max(tau_init * 0.3, tau * 0.5)
            radius = 0.08
            continue
        S = _corr_scaled(h)
        g = _grad_smooth(h, S, tau)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            stall += 1; tau *= 0.9
            continue
        d = -g / gn
        alpha, new_c5 = _line_search_quad(h, d, dx, radius)
        if new_c5 < best_c5 - 1e-15:
            h = _project(h + alpha * d, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7: radius = min(radius * 1.05, 0.25)
            else: radius = max(radius * 0.92, 0.03)
        else:
            stall += 1
            radius *= 0.75
            tau *= 0.92
            if radius < 0.01:
                radius = 0.06
                stall = max(0, stall - 20)
    return best_h, best_c5
