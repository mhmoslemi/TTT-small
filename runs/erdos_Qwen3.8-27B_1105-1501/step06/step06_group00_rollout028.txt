    return np.clip(x, 0.0, 1.0)

def _build_G(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    u = 1.0 - h
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            G[r, :m] += u[d:]
            G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            G[r, g:] += u[:m]
            G[r, :m] += -h[g:]
    return G

def _eval_f(raw, T, Q, alphas):
    """Evaluate f(alpha) = max(raw + alpha*T - alpha^2*Q) for array of alphas."""
    new_raw = raw[:, None] + alphas[None, :] * T[:, None] - (alphas[None, :] ** 2) * Q[:, None]
    return np.max(new_raw, axis=0)

def _find_best_pair(h, G, raw, radius, n_src, n_snk, n_rand, n_samples, rng):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    M = float(np.max(raw))
    e = np.exp(np.clip(22.0 * (raw - M), -60.0, 0.0))
    w = e / (float(e.sum()) + 1e-30)
    grad = G.T @ w
    grad -= float(np.mean(grad))

    top_sinks = np.argsort(grad)[:n_snk]
    top_sources = np.argsort(grad)[-n_src:]

    pairs = []
    for i in top_sinks:
        for j in top_sources:
            if i != j:
                pairs.append((i, j))
    for _ in range(n_rand):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j:
            pairs.append((i, j))

    best_c5 = float('inf')
    best_pair = None
    best_alpha = 0.0

    seen = set()
    for (i, j) in pairs:
        if (i, j) in seen:
            continue
        seen.add((i, j))
        a_min = max(-h[i], h[j] - 1.0, -radius)
        a_max = min(1.0 - h[i], h[j], radius)
        if a_max - a_min < 1e-12:
            continue
        T = G[:, i] - G[:, j]
        Q = np.zeros(L)
        Q[n - 1] = 2.0
        if i != j:
            Q[n - 1 + (j - i)] = -1.0
            Q[n - 1 + (i - j)] = -1.0
        alphas = np.linspace(a_min, a_max, n_samples)
        f_vals = _eval_f(raw, T, Q, alphas) * dx
        bi = int(np.argmin(f_vals))
        bc = float(f_vals[bi])
        ba = float(alphas[bi])
        if bi > 0 and bi < n_samples - 1:
            lo = alphas[max(0, bi - 3)]
            hi = alphas[min(n_samples - 1, bi + 3)]
            if hi - lo > 1e-12:
                fine = np.linspace(lo, hi, 60)
                ff = _eval_f(raw, T, Q, fine) * dx
                fi = int(np.argmin(ff))
                if ff[fi] < bc:
                    bc = float(ff[fi])
                    ba = float(fine[fi])
        if bc < best_c5:
            best_c5 = bc
            best_pair = (i, j)
            best_alpha = ba
    return best_pair, best_alpha, best_c5

def _local_opt(h, deadline, rng, init_radius=0.25, max_radius=0.40, n_src=14, n_snk=14, n_rand=12, n_samples=150, stall_limit=80):
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    radius = init_radius
    stall = 0
    beta = 22.0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        G = _build_G(h)
        raw = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        pair, alpha, c5_try = _find_best_pair(h, G, raw, radius, n_src, n_snk, n_rand, n_samples, rng)
        if pair is None or c5_try >= cur - 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 0.003:
                break
            continue
        h[pair[0]] += alpha
        h[pair[1]] -= alpha
        h = np.clip(h, 0.0, 1.0)
        cur = _exact_c5(h)
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
        stall = 0
        radius = min(radius * 1.04, max_radius)
        beta = min(beta * 1.01, 45.0)
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_pair_sweep(h, deadline, rng, max_rounds=20):
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                h_t = h.copy()
                h_t[oi] = 0.0
                h_t[oz] = 1.0
                c_t = _exact_c5(h_t)
                if c_t < cur - 1e-14:
                    h = h_t
                    cur = c_t
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    ones = list(np.where(h > 0.5)[0])
                    zeros = list(np.where(h <= 0.5)[0])
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def _binary_sa(h, deadline, rng, max_iter=30000, T_init=0.003):
    n = len(h)
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_t = h.copy()
        h_t[oi] = 0.0
        h_t[oz] = 1.0
        c_t = _exact_c5(h_t)
        delta = c_t - cur
        if delta < -1e-14 or (T > 1e-8 and rng.random() < np.exp(delta / T)):
            h = h_t
            cur = c_t
            ones = list(np.where(h > 0.5)[0])
            zeros = list(np.where(h <= 0.5)[0])
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
    return best_h, best_c

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    p = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 8.0)
        ph = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amp
    p += amp * 0.3 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

        h0[:half] = 1.0

    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Primary local optimization (45% time)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _local_opt(h_init, d1, rng, init_radius=0.25, max_radius=0.40,
                          n_src=14, n_snk=14, n_rand=14, n_samples=150, stall_limit=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Basin hopping with perturbations (15% time)
    d2 = t0 + budget_s * 0.60

    n_restarts = 0
    while time.time() < d2 and n_restarts < 200:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d2, time.time() + budget_s * 0.006)
