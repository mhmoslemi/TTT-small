```python
def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))

        else:
            hi = mid
    theta = (lo + hi) / 2.0

    if abs(res) > 1e-12:

        else:

        cnt = int(np.sum(mask))

                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0: x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_mat(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, r in enumerate(lags):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def _ls(h, d, max_step):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi, t_lo = max_step, -max_step
    if np.any(pos): t_hi = min(t_hi, float(np.min((1.0-h[pos])/d[pos])))
    if np.any(neg): t_lo = max(t_lo, float(np.max(h[neg]/d[neg])))
    if t_hi - t_lo < 1e-14: return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0)-1.0)/2.0
    a, b = t_lo, t_hi
    c1 = b - gr*(b-a); c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C))*dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C))*dx
    for _ in range(50):
        if b-a < 1e-12: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))*dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C))*dx
    t = (a+b)/2.0
    return t, float(np.max(S0 + t*T - t*t*C))*dx

def _lp_dir(h, scores, radius, top_k):
    n = len(h)
    m = 2*n-1
    si = np.argsort(-scores)[:min(top_k, m)]
    G = _grad_mat(h, si)
    c_vec = np.zeros(n+1); c_vec[-1] = 1.0
    A_ub = np.zeros((len(si), n+1))
    A_ub[:, :n] = G; A_ub[:, n] = -1.0
    b_ub = -scores[si]
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except: pass
    return None

def _cont_opt(h, deadline, radius=0.10, top_k=20):
    n = len(h)
    target = n/2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius_cur = radius
    stall = 0
    rng = np.random.default_rng(12345)
    while time.time() < deadline:
        if stall > 60:
            h = h + rng.normal(0, 0.02, n)
            h = _proj(h, target)
            best_c5 = min(best_c5, _exact_c5(h))
            radius_cur = 0.08
            stall = 0
        scores = _corr_values(h)
        d = _lp_dir(h, scores, radius_cur, top_k)
        if d is None or np.max(np.abs(d)) < 1e-13:
            # Fallback: negative gradient of max
            idx = np.argmax(scores)
            G = _grad_mat(h, [idx])
            d = -G[0].copy()
            d -= d.mean()
            norm = np.linalg.norm(d)
            if norm < 1e-14:
                stall += 1
                radius_cur *= 0.9
                continue
            d = d / norm * radius_cur
        # Clip to feasible
        pos = d > 0; neg = d < 0
        if np.any(pos):
            scale = np.min((1.0-h[pos])/d[pos])
            d[pos] *= min(1.0, scale)
        if np.any(neg):
            scale = np.min(h[neg]/(-d[neg]))
            d[neg] *= -min(1.0, scale)
        t, new_c5 = _ls(h, d, radius_cur*1.5)
        if new_c5 < _exact_c5(h) - 1e-14:
            h = h + t*d
            h = _proj(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            radius_cur = min(radius_cur*1.03, 0.3)
        else:
            stall += 1
            radius_cur *= 0.85
            if radius_cur < 0.005: radius_cur = 0.06
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):

    target = n // 2
    dx = 2.0 / n
    # Initialize from h_init
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.05
    T_min = 0.001
    cool = 0.9998
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    patience = 0
    max_pat = 5000
    while time.time() < deadline:
        if patience > max_pat: break
        if len(ones) == 0 or len(zeros) == 0: break
        # Choose move: swap a 1 and a 0
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0; h[j] = 1.0

            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0; h[j] = 0.0
        T *= cool
    return best_h, best_c5

        else:

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n)+0.5)/n
    x_new = (np.arange(n_new)+0.5)/n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new/2.0)

    x = np.arange(n)
    h = 0.5 + 0.5*np.sin(2*np.pi*freq*x/n + phase)
    return _proj(h, n/2.0)
```

```python
    best_c5 = _exact_c5(h)
    for _ in range(max_passes):
        improved = False
        best_delta = 0
        best_move = None
        for i in ones:
            for j in zeros:
                # Compute new c5 if we move 1 from i to j
                h[i] = 0.0; h[j] = 1.0
                c5_try = _exact_c5(h)
                if c5_try < best_c5 - 1e-14:
                    best_delta = c5_try - best_c5
                    best_move = (i, j, c5_try)
                h[i] = 1.0; h[j] = 0.0
        if best_move is not None:
            i, j, new_c5 = best_move
            h[i] = 0.0; h[j] = 1.0
            best_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            improved = True
        if not improved:
            break
    return h, best_c5
```

But this is O(n^2 * n^2) = O(n^4) per pass which for n=122 is about 220 million operations per pass. That's too slow in Python.

```python
def _binary_one_step(h, i, j):
    """Compute new C5 after moving 1 from i to j, using incremental update."""
    n = len(h)
    # Current correlations
    # Apply the change: h[i]: 1->0, h[j]: 0->1
    # Delta for each shift k: h[i+k] + h[i-k] - h[j+k] - h[j-k] (with bounds)
    # ... this is complex with the np.correlate indexing
    # Just recompute
    h[i] = 0.0; h[j] = 1.0
    c5 = _exact_c5(h)
    h[i] = 1.0; h[j] = 0.0
    return c5
```

So the key question is: can we achieve max correlation ≤ 23 with a binary sequence of length 122 with 61 ones?

Let me think about this combinatorially. We need a {0,1} sequence of length 122 with 61 ones such that for every shift k (from -(121) to 121), the correlation is ≤ 23.

For k=0: always 0 for binary.
For k=1: c[1] = #{n: h[n]=0, h[n+1]=1} = number of 0→1 transitions
         c[-1] = #{n: h[n]=1, h[n+1]=0} = number of 1→0 transitions

The number of 1→0 transitions equals the number of blocks of consecutive 1s (call it B).
The number of 0→1 transitions is either B (if the sequence starts and ends with 0) or B-1 (if it starts with 1 and ends with 0) or B+1 (if it starts with 0 and ends with 1) or B (if it starts and ends with 1).

In any case, both are approximately B. So we need B ≤ 23 (roughly).

With 61 ones in at most 23 blocks, the average block size is 61/23 ≈ 2.65. So blocks of size 2 or 3 mostly.

For larger shifts k, the constraint is that the "partial overlap" between the 1s and the zeros (shifted by k) is at most 23.

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0: x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_mat(h, lags):
    n = len(h)
    G = np.zeros((len(lags), n))
    for idx, r in enumerate(lags):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def _ls(h, d, max_step):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi, t_lo = max_step, -max_step
    if np.any(pos): t_hi = min(t_hi, float(np.min((1.0-h[pos])/d[pos])))
    if np.any(neg): t_lo = max(t_lo, float(np.max(h[neg]/d[neg])))
    if t_hi - t_lo < 1e-14: return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0)-1.0)/2.0
    a, b = t_lo, t_hi
    c1 = b - gr*(b-a); c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C))*dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C))*dx
    for _ in range(50):
        if b-a < 1e-12: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))*dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C))*dx
    t = (a+b)/2.0
    return t, float(np.max(S0 + t*T - t*t*C))*dx

def _lp_dir(h, scores, radius, top_k):
    n = len(h)
    m = 2*n-1
    si = np.argsort(-scores)[:min(top_k, m)]
    G = _grad_mat(h, si)
    c_vec = np.zeros(n+1); c_vec[-1] = 1.0
    A_ub = np.zeros((len(si), n+1))
    A_ub[:, :n] = G; A_ub[:, n] = -1.0
    b_ub = -scores[si]
    A_eq = np.zeros((1, n+1)); A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0-h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _cont_opt(h, deadline, radius=0.10, top_k=25):
    n = len(h)
    target = n/2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius_cur = radius
    stall = 0
    rng = np.random.default_rng(12345)
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 50:
            h = h + rng.normal(0, 0.02, n)
            h = _proj(h, target)
            c_try = _exact_c5(h)
            if c_try < best_c5:
                best_c5 = c_try; best_h = h.copy()
            radius_cur = 0.08
            stall = 0
        scores = _corr_values(h)
        d = _lp_dir(h, scores, radius_cur, top_k)
        if d is None or np.max(np.abs(d)) < 1e-13:
            active = np.where(scores >= scores.max() - 0.05*scores.max())[0]
            if len(active) == 0:
                active = np.array([np.argmax(scores)])
            G = _grad_mat(h, active)
            d = -G.mean(axis=0).copy()
            d -= d.mean()
            norm = np.linalg.norm(d)
            if norm < 1e-14:
                stall += 1; radius_cur *= 0.9
                continue
            d = d / norm * radius_cur
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        if np.any(pos_mask):
            sc = np.min((1.0-h[pos_mask])/d[pos_mask])
            if sc < 1.0: d[pos_mask] *= sc
        if np.any(neg_mask):
            sc = np.min(h[neg_mask]/(-d[neg_mask]))
            if sc < 1.0: d[neg_mask] *= -sc
        t, new_c5 = _ls(h, d, radius_cur*1.5)
        cur_c5 = _exact_c5(h)
        if new_c5 < cur_c5 - 1e-14:
            h = h + t*d
            h = _proj(h, target)
            c_now = _exact_c5(h)
            if c_now < best_c5:
                best_c5 = c_now; best_h = h.copy(); stall = 0
            else: stall += 1
            radius_cur = min(radius_cur*1.03, 0.35)
        else:
            stall += 1
            radius_cur *= 0.85
            if radius_cur < 0.005: radius_cur = 0.06
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng, T0=0.08):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T0
    T_min = 0.0005
    cool = 0.99985
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    patience = 0
    max_pat = 8000
    while time.time() < deadline:
        if patience > max_pat: break
        if len(ones) == 0 or len(zeros) == 0: break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0; h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta/max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5; best_h = h.copy(); patience = 0
            else: patience += 1
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0; h[j] = 0.0
        T = max(T_min, T * cool)
    return best_h, best_c5

def _binary_ls(h, deadline, max_rounds=200):
    n = len(h)
    h = h.copy()
    best_c5 = _exact_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline: break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_move = None
        best_new_c5 = best_c5
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                c_try = _exact_c5(h)
                if c_try < best_new_c5 - 1e-15:
                    best_new_c5 = c_try
                    best_move = (i, j)
                h[i] = 1.0; h[j] = 0.0
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0; h[j] = 1.0
            best_c5 = best_new_c5
        else:
            break
    return h, best_c5

def _pair_refine(h, deadline, max_iter=50000):
    n = len(h)
    dx = 2.0 / n
    target = n/2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(77)
    stall = 0
    it = 0
    while time.time() < deadline and stall < 80 and it < max_iter:
        it += 1
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2: break
        i = interior[rng.integers(len(interior))]
        j = interior[rng.integers(len(interior))]
        if i == j: continue
        improved = False
        for sgn in [1.0, -1.0]:
            d = np.zeros(n)
            d[i] = -sgn; d[j] = sgn
            t, c5_try = _ls(h, d, 0.5)
            if c5_try < best_c5 - 1e-14:
                h = h + t*d
                h = _proj(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur; best_h = h.copy(); improved = True
                break
        if improved: stall = 0
        else: stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n)+0.5)/n
    x_new = (np.arange(n_new)+0.5)/n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new/2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5*np.sin(2*np.pi*freq*x/n + phase)
    return _proj(h, n/2.0)

def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)

    h0 = _proj(h0, n0/2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)

    # Phase 1: Quick continuous optimization (10% time)
    d1 = t0 + budget_s * 0.10
    h1, c1 = _cont_opt(best_h, d1, radius=0.10, top_k=25)
    if c1 < best_c5:
        best_c5 = c1; best_h = h1.copy()

    # Phase 2: Extended Binary SA with multiple restarts (35% time)
    d2 = t0 + budget_s * 0.45
    n = len(best_h)
    n_restarts = 10
    for r in range(n_restarts):
        if time.time() > d2: break
        remaining = d2 - time.time()
        per_time = remaining / max(1, n_restarts - r)
        sub_dl = min(d2, time.time() + per_time)
        if r == 0:
            h_start = best_h.copy()
        else:
            h_start = _random_binary(n, rng)
        h_b, c_b = _binary_sa(h_start, sub_dl, rng, T0=0.06+r*0.005)
        if c_b < best_c5:
            best_c5 = c_b; best_h = h_b.copy()

    # Phase 3: Binary local search polish (10% time)
    d3 = t0 + budget_s * 0.55
    if np.max(np.abs(best_h - np.round(best_h))) < 0.01:
        h3, c3 = _binary_ls(best_h, d3, max_rounds=50)
        if c3 < best_c5:

    # Phase 4: Continuous optimization from best (15% time)
    d4 = t0 + budget_s * 0.70
    h4, c4 = _cont_opt(best_h, d4, radius=0.08, top_k=30)
    if c4 < best_c5:

    # Phase 5: Multi-resolution search (15% time)
    d5 = t0 + budget_s * 0.85

    offsets = [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12, 14, -14, 16, 18, 20, 24, 28, 32, -16, -18, -20]

        if c_rc < best_c5 - 1e-12:

    h6, c6 = _pair_refine(best_h, d6, max_iter=80000)
```
