def _project(v, target):
    # Project v onto {x : 0 <= x <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):

        if res > 0:

        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _grad_S(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        g[0:n-k] += (1.0 - h[k:n])
        g[k:n] += (-h[0:n-k])
    else:
        j = -k
        g[j:n] += (1.0 - h[0:n-j])
        g[0:n-j] += (-h[j:n])
    return g

def _quad_dir_deriv2(h, d, k):
    """Compute d^T H_k d where H_k is the Hessian of S_k."""
    n = len(h)
    if k == 0:
        return -4.0 * np.sum(d**2)  # H_0 = -2I, so d^T H_0 d = -2*||d||²... 
        # Actually S_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # H_0 = -2I (diagonal -2)
        # d^T H_0 d = -2 * ||d||^2
        # Wait: d^T (-2I) d = -2 * d^T d = -2 * ||d||^2
        return -2.0 * float(np.dot(d, d))
    if k > 0:
        # H_k has -1 at (i, i+k) and (i+k, i) for i = 0..n-1-k
        # d^T H_k d = -2 * sum_{i=0}^{n-1-k} d[i]*d[i+k]
        return -2.0 * float(np.dot(d[0:n-k], d[k:n]))
    else:
        j = -k
        # Same as for k=j > 0 but with different indexing

        return -2.0 * float(np.dot(d[j:n], d[0:n-j]))

def _solve_qp_direction(active_grads, active_vals, lam=1e-2, n=0):
    """
    Find direction d that minimizes max_i (active_vals[i] + active_grads[i] @ d)
    subject to 1^T d = 0.
    Returns the direction (unnormalized).
    """
    m = len(active_vals)
    G = active_grads  # m x n
    # Centralize gradients
    Gc = G - G.mean(axis=1, keepdims=True)[:, 0]  # wrong, need to center over columns
    # Actually, we want to solve: min_d max_i (S_i + g_i^T d) s.t. 1^T d = 0
    # KKT gives: d = -mu * sum_i lambda_i * (g_i - mean(g_i))
    # where lambda solves: (Gc^T Gc) lambda = (t - S) / mu (on active set)
    # For simplicity, use: d = -Gc^T @ ones / (Gc^T Gc diagonal) ... no
    # Simplest: d = -Gc^T @ w where w = softmax(S)
    # Or: solve (Gc @ Gc^T + lam*I) @ alpha = S (to equalize)
    # Then d = Gc^T @ alpha, projected to zero mean
    
    # Better approach: solve (Gc @ Gc^T + lam*I) @ alpha = S
    # Then d = -Gc^T @ alpha (this direction decreases all S_i)
    # Project to zero mean: d = d - d.mean()
    
    A = Gc @ Gc.T + lam * np.eye(m)
    try:
        alpha = np.linalg.solve(A, active_vals)
    except np.linalg.LinAlgError:
        alpha = np.linalg.lstsq(A, active_vals, rcond=None)[0]
    d_raw = Gc.T @ alpha
    d = d_raw - d_raw.mean()
    return d

def _line_search_quadratic(h, d, S_vals, active_k, t_max, n_eval_max=200):
    """
    Line search along h + t*d, using exact quadratic model for active constraints.
    S_k(t) = S_k + t * a_k + 0.5 * t^2 * b_k
    where a_k = g_k^T d, b_k = d^T H_k d
    Find t in [0, t_max] that minimizes max_k S_k(t).
    """
    n = len(h)
    # Compute a_k and b_k for active constraints
    m = len(active_k)
    a = np.array([float(np.dot(_grad_S(h, k), d)) for k in active_k])
    b = np.array([_quad_dir_deriv2(h, d, k) for k in active_k])
    
    # Evaluate at t=0
    f0 = float(np.max(S_vals))
    
    # Find candidate t values where S_i(t) = S_j(t) for pairs
    candidates = [0.0, t_max]
    
    # Also check where each S_k(t) individual minimum is (if b_k < 0)
    for i in range(m):
        if b[i] < -1e-15:
            t_min = -a[i] / b[i]
            if 0 < t_min < t_max:
                candidates.append(t_min)
    
    for i in range(m):
        for j in range(i+1, m):
            # Solve S_i(t) = S_j(t)
            # S_i + a_i*t + 0.5*b_i*t^2 = S_j + a_j*t + 0.5*b_j*t^2
            # 0.5*(b_i-b_j)*t^2 + (a_i-a_j)*t + (S_i-S_j) = 0
            A = 0.5 * (b[i] - b[j])
            B = a[i] - a_j
            C = S_vals[active_k[i]] - S_vals[active_k[j]]
            if abs(A) < 1e-15:
                if abs(B) > 1e-15:
                    t = -C / B
                    if 0 < t < t_max:
                        candidates.append(t)
            else:
                disc = B*B - 4*A*C
                if disc >= 0:
                    sq = np.sqrt(disc)
                    t1 = (-B + sq) / (2*A)
                    t2 = (-B - sq) / (2*A)
                    for t in [t1, t2]:
                        if 0 < t < t_max:
                            candidates.append(t)
    
    # Also add some grid points
    if t_max > 0:
        candidates.extend(np.linspace(0, t_max, 20).tolist())
    
    best_t = 0.0
    best_f = f0
    
    for t in candidates:
        if t < 0 or t > t_max:
            continue
        # Evaluate max S_k at this t using the quadratic model
        vals = S_vals[active_k] + t * a + 0.5 * t * t * b
        f_model = float(np.max(vals))
        if f_model < best_f - 1e-10:
            best_f = f_model
            best_t = t
    
    return best_t, best_f
