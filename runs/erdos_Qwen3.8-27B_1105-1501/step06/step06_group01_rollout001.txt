    S0 = _corr_vals(h)
    # S(h + a*d) = S0 + a*T - a^2*C where T and C are computed from correlations
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    
    # Find feasible alpha range
    a_lo, a_hi = -alpha_max, alpha_max
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        a_lo = max(a_lo, np.min(-h[pos] / d[pos]))
        a_hi = min(a_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        a_lo = max(a_lo, np.min((1.0 - h[neg]) / d[neg]))
        a_hi = min(a_hi, np.min(-h[neg] / d[neg]))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0))
    
    # Grid search + golden section
    ng = 30
    grid = np.linspace(a_lo, a_hi, ng)
    vals = np.max(S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:], axis=1)
    best_i = np.argmin(vals)
    best_a, best_v = grid[best_i], vals[best_i]
    
    # Golden section refinement
    if 0 < best_i < ng-1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = a_lo, a_hi
    gr = (np.sqrt(5.0)-1.0)/2.0
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C))
    f2 = float(np.max(S0 + c2*T - c2*c2*C))
    for _ in range(30):
        if b-a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C))
    opt_a = (a+b)/2.0
    opt_v = float(np.max(S0 + opt_a*T - opt_a*opt_a*C))
    if opt_v < best_v:
        best_a, best_v = opt_a, opt_v
    
    return best_a, best_v

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        k = q - (n-1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[q, k:] += dx * (1.0 - h[:m])
                G[q, :m] += -dx * h[k:]
        else:
            da = -k
            m = n - da
            if m > 0:
                G[q, :m] += dx * (1.0 - h[da:])
                G[q, da:] += -dx * h[:m]
    return G

def _sa_optimize(h, deadline, T_init=0.002, T_min=1e-7, alpha_max=0.5):
    """Simulated annealing with exact quadratic line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    cur_T = T_init
    rng = np.random.default_rng(12345)
    cool_rate = 0.998
    it = 0
    patience = 0
    
    while time.time() < deadline:
        it += 1
        # Generate a random direction
        # Mix of: single coord, pair, small block
        move_type = rng.random()
        if move_type < 0.4:
            # Single coordinate perturbation
            i = rng.integers(n)
            d = np.zeros(n)
            d[i] = rng.choice([-1.0, 1.0])
            # Need to maintain sum constraint: adjust another coordinate
            j = rng.integers(n)
            if j != i:
                d[j] = -d[i]
            else:
                # Skip if we can't maintain balance
                continue
        elif move_type < 0.7:
            # Pair move
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        elif move_type < 0.85:
            # Block move: shift a small block
            blen = rng.integers(2, 5)
            start = rng.integers(0, n - blen)
            d = np.zeros(n)
            d[start] = -1.0
            d[start + blen] = 1.0
        else:
            # Random smooth perturbation
            d = rng.normal(0, 1, n)
            d -= d.mean()  # Maintain sum
        
        alpha, new_c5 = _quadratic_line_search(h, d, dx, alpha_max)
        if new_c5 == cur_c5 and abs(alpha) < 1e-12:
            # No movement possible in this direction
            continue
        
        delta = new_c5 - cur_c5
        if delta < 0 or (cur_T > T_min and rng.random() < np.exp(-delta / cur_T)):
            # Accept
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            new_c5_actual = _exact_c5(h_new)
            if new_c5_actual < best_c5:
                best_c5 = new_c5_actual
                best_h = h_new.copy()
                patience = 0
            cur_c5 = new_c5_actual
            h = h_new
        # Cool down
        if it % 10 == 0:
            cur_T *= cool_rate
    
    return best_h, best_c5

def _lp_optimize(h, deadline, start_radius=0.12):
    """LP-based trust region optimization with exact line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = start_radius
    stall = 0
    perturb_ctr = 0
    rng = np.random.default_rng(777)
    
    while time.time() < deadline:
        if stall > 60:
            if perturb_ctr < 10:
                h = h + rng.normal(0, 0.05, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.10
                stall = 0
                perturb_ctr += 1
                continue
            else:
                break
        
        S = _corr_vals(h)
        # Find active lags
        S_max = float(np.max(S))
        active = np.where(S > S_max - 0.01)[0]
        
        # Build gradient from active lags
        G = _build_G(h, dx)
        if len(active) > 0:
            # Weighted gradient (emphasize most active)
            w = (S[active] - (S_max - 0.01)) ** 2
            w = w / w.sum()
            grad = G[active].T @ w
        else:
            grad = G[np.argmax(S)]
        
        # Negative gradient direction
        d = -grad
        d -= d.mean()  # Maintain sum
        
        # LP for better direction
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
        b_ub = -S
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for i in range(n):
            lo = max(-radius, -h[i] + 1e-12)
            hi = min(radius, 1.0 - h[i] - 1e-12)
            if lo >= hi:
                bounds.append((0.0, 0.0))
            else:
                bounds.append((lo, hi))
        bounds.append((None, None))
        
        lp_dir = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                lp_dir = res.x[:n]
        except Exception:
            pass
        
        if lp_dir is not None and np.max(np.abs(lp_dir)) > 1e-12:
            d_use = lp_dir
        else:
            # Scale gradient direction to fit in radius
            d_max = np.max(np.abs(d))
            if d_max > radius:
                d *= radius / d_max
            d_use = d
        
        alpha, new_c5 = _quadratic_line_search(h, d_use, dx, radius)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d_use
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Adaptive radius
            S_new = _corr_vals(h)
            predicted = float(np.max(S + G @ d_use))
            ratio = cur_c5 / max(predicted, 1e-15)
            if ratio < 0.5:
                radius = min(radius * 1.2, 0.4)
            elif ratio > 2.0:
                radius *= 0.7
        else:
            stall += 1
            radius *= 0.8
            if radius < 0.01:
                radius = 0.05  # Reset
    
    return best_h, best_c5
