import numpy as np

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    s = raw * dx
    max_s = s.max()
    shifted = beta * (s - max_s)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        wr = w[r] * dx
        if wr < 1e-14:
            continue
        # Term 1: (1-h[j+d]) for j in [max(0,-d), min(n, n-d))
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wr * (1.0 - h[lo1 + d:hi1 + d])
        # Term 2: (-h[j-d]) for j in [max(0,d), min(n, n+d))
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if lo2 < hi2:
            grad[lo2:hi2] += wr * (-h[lo2 - d:hi2 - d])
    return grad

def _lse_optimize(h, deadline, beta_start=30.0, beta_end=300.0,

    for beta in betas:
        if time.time() > deadline:
            break
        cur_c5 = _exact_c5(h)
        grad = _lse_gradient(h, beta)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Initial step size
        step = min(1.0 / (beta * gnorm + 1e-10), 0.1)
        
        for _ in range(max_steps):
            if time.time() > deadline:
                break
            h_trial = h - step * grad
            h_trial = _project(h_trial, target)
            c5_trial = _exact_c5(h_trial)
            
            if c5_trial < cur_c5 - 1e-15:
                h = h_trial
                cur_c5 = c5_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h.copy()
                # Recompute gradient

                grad -= grad.mean()
                gnorm = np.linalg.norm(grad)

                    break
                step = min(step * 1.1, 1.0 / (beta * gnorm + 1e-10), 0.1)

    return best_h, best_c5

def _corr_grad_matrix(h, lag_indices):
    """Compute gradient of corr values at given lags w.r.t. h."""
    n = len(h)
    m = len(lag_indices)
    G = np.zeros((m, n))
    for idx, r in enumerate(lag_indices):
        d = r - (n - 1)
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            G[idx, lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if lo2 < hi2:
            G[idx, lo2:hi2] += (-h[lo2 - d:hi2 - d])
    return G * (2.0 / n)

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_direction(h, scores, radius, top_k=20):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = _corr_grad_matrix(h, active)
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)

    for dn in [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20]:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _lse_optimize(h_r, sub_dl, beta_start=30, beta_end=200,
                                   n_betas=8, max_steps=150)
        if c5_r < best_c5:
            sub_dl2 = min(d3, time.time() + budget_s * 0.01)
            h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.06, top_k=10)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 4: Pair refinement + final polish (15% time)
    h4, c5_4 = _pair_refine(best_h, min(deadline - 5, t0 + budget_s * 0.95))
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Final LSE polish at very high beta
    if time.time() < deadline - 10:
        h5, c5_5 = _lse_optimize(best_h, deadline - 5, beta_start=200, 
                                  beta_end=1000, n_betas=5, max_steps=100)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
