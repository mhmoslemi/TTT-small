    n = len(h)
    diff = abs(i - j)

    if i > j:
        s_min = max(-h[i], h[j] - 1.0)
        s_max = min(1.0 - h[i], h[j])
    else:
        s_min = max(-h[j], h[i] - 1.0)
        s_max = min(1.0 - h[j], h[i])
    
    if s_min >= s_max - 1e-15:
        return None
    
    # Compute T vectors
    lag = np.arange(n)
    
    # For pair (i, j) with i > j (WLOG, we'll handle both):
    # A[lag] = T_right: (1-h[i-lag])*[i-lag in [0,n-1]] - (1-h[j-lag])*[j-lag in [0,n-1]] 
    #           - h[i+lag]*[i+lag in [0,n-1]] + h[j+lag]*[j+lag in [0,n-1]]
    # B[lag] = T_left: (1-h[i+lag])*[i+lag in [0,n-1]] - (1-h[j+lag])*[j+lag in [0,n-1]]
    #           - h[i-lag]*[i-lag in [0,n-1]] + h[j-lag]*[j-lag in [0,n-1]]
    
    A = np.zeros(n)
    B = np.zeros(n)
    
    # i-lag valid: 0 <= i-lag < n → 0 <= lag <= i
    m = lag <= i
    A[m] += 1.0 - h[i - lag[m]]
    B[m] -= h[i - lag[m]]
    
    # j-lag valid: 0 <= j-lag < n → 0 <= lag <= j
    m = lag <= j
    A[m] -= 1.0 - h[j - lag[m]]
    B[m] += h[j - lag[m]]
    
    # i+lag valid: 0 <= i+lag < n → 0 <= lag < n-i
    m = lag < (n - i)
    A[m] -= h[i + lag[m]]
    B[m] += 1.0 - h[i + lag[m]]
    
    # j+lag valid: 0 <= j+lag < n → 0 <= lag < n-j
    m = lag < (n - j)
    A[m] += h[j + lag[m]]
    B[m] -= 1.0 - h[j + lag[m]]
    
    # Q
    Q = np.zeros(n)
    Q[0] = 2.0
    Q[diff] = -1.0
    
    # Grid search
    n_grid = 25
    s_vals = np.linspace(s_min, s_max, n_grid)

    # Shape: (n_grid, n) for each
    sq = s_vals ** 2
    obj_r = (S_r[None, :] + s_vals[:, None] * A[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
    obj_l = (S_l[None, :] + s_vals[:, None] * B[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
    obj = np.maximum(obj_r, obj_l)

    best_s = s_vals[best_k]
    best_val = obj[best_k]
    
    # Current value
    cur_val = max(S_r.max(), S_l.max())
    
    if best_val < cur_val - 1e-14:
        return best_s
    return None

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

    res = target - float(np.sum(x))

            A[m] += h[idx]
            B[m] -= 1.0 - h[idx]
        Q = np.zeros(n)
        Q[0] = 2.0
        Q[diff] = -1.0
        s_vals = np.linspace(s_min, s_max, n_grid)
        sq = s_vals * s_vals
        obj_r = (S_r[None, :] + s_vals[:, None] * A[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
        obj_l = (S_l[None, :] + s_vals[:, None] * B[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
        obj = np.maximum(obj_r, obj_l)
        best_k = int(obj.argmin())
        best_s = s_vals[best_k]
        best_val = obj[best_k]
        if best_val < cur_max - 1e-13 and abs(best_s) > 1e-13:
            h[i] += best_s
            h[j] -= best_s
            h = np.clip(h, 0.0, 1.0)
            S_r, S_l = _compute_corr_sets(h)
            cur_max = max(S_r.max(), S_l.max())
            c5_now = cur_max * dx
            if c5_now < best_c5:
                best_c5 = c5_now
            no_improve = 0
        else:
            no_improve += 1
    return h, best_c5


def _pair_pass_ordered(h, deadline, n_grid=35):
    n = len(h)
    dx = 2.0 / n
    best_c5 = _exact_c5(h)
    h = h.copy()
    S_r, S_l = _compute_corr_sets(h)
    cur_max = max(float(S_r.max()), float(S_l.max()))
    lag = np.arange(n)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    np.random.shuffle(pairs)
    improve_count = 0
    for i, j in pairs:
        if time.time() > deadline:
            break
        hi_v, hj_v = h[i], h[j]
        for direction in (1, -1):

                a, b = i, j
            else:
                a, b = j, i
            ha, hb = h[a], h[b]
            s_min = max(-ha, hb - 1.0)
            s_max = min(1.0 - ha, hb)

            B = np.zeros(n)
            m = lag <= a
            if np.any(m):
                A[m] += 1.0 - h[a - lag[m]]
                B[m] -= h[a - lag[m]]
            m = lag <= b
            if np.any(m):
                A[m] -= 1.0 - h[b - lag[m]]
                B[m] += h[b - lag[m]]
            m = lag < (n - a)
            if np.any(m):
                A[m] -= h[a + lag[m]]
                B[m] += 1.0 - h[a + lag[m]]
            m = lag < (n - b)
            if np.any(m):
                A[m] += h[b + lag[m]]
                B[m] -= 1.0 - h[b + lag[m]]
            Q = np.zeros(n)
            Q[0] = 2.0
            Q[diff] = -1.0

                S_r, S_l = _compute_corr_sets(h)
                cur_max = max(float(S_r.max()), float(S_l.max()))
                c5_now = cur_max * dx
                if c5_now < best_c5:
                    best_c5 = c5_now
                improve_count += 1
                break
    return h, best_c5


def _gradient_descent(h, deadline, K=30, alpha_init=15.0, alpha_mult=1.5,
                      max_iters=300, n_grid=30):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_init

    for it in range(max_iters):
        if time.time() > deadline:
            break
        S_r, S_l = _compute_corr_sets(h)
        all_S = np.concatenate([S_r, S_l])
        M = float(all_S.max())
        top_k = min(K, n)
        sorted_idx = np.argsort(-all_S)[:top_k]
        top_vals = all_S[sorted_idx]
        spread = max(float(top_vals[0] - top_vals[-1]), 1e-10)
        w = np.exp(alpha * (top_vals - top_vals[0]) / spread)
        w /= w.sum()
        grad = np.zeros(2 * n)
        lag = np.arange(n)
        for wi, idx in zip(w, sorted_idx):
            li = int(idx)
            if li < n:
                ll = li
                gA = np.zeros(n)
                m = lag <= (np.arange(n))
                for k in range(n):
                    val = 0.0
                    if k - ll >= 0 and k - ll < n:
                        val += 1.0 - h[k - ll]
                    if k + ll < n:
                        val -= h[k + ll]
                    gA[k] = val
                grad[:n] += wi * gA
                gB = np.zeros(n)
                for k in range(n):
                    val = 0.0
                    if k + ll < n:
                        val += 1.0 - h[k + ll]
                    if k - ll >= 0 and k - ll < n:
                        val -= h[k - ll]
                    gB[k] = val
                grad[n:] += wi * gB
            else:
                ll = li - n
                gA = np.zeros(n)
                for k in range(n):
                    val = 0.0
                    if k + ll < n:
                        val += 1.0 - h[k + ll]
                    if k - ll >= 0 and k - ll < n:
                        val -= h[k - ll]
                    gA[k] = val
                grad[:n] += wi * gA
                gB = np.zeros(n)
                for k in range(n):
                    val = 0.0
                    if k - ll >= 0 and k - ll < n:
                        val += 1.0 - h[k - ll]
                    if k + ll < n:
                        val -= h[k + ll]
                    gB[k] = val
                grad[n:] += wi * gB
        hg = np.zeros(n)
        hg += grad[:n]
        hg += grad[n:]
        hg -= hg.mean()
        gn = np.linalg.norm(hg)
        if gn < 1e-14:
            break
        direction = -hg / gn
        S_full = np.concatenate([S_r, S_l])
        T_full = np.concatenate([
            np.zeros(n), np.zeros(n)
        ])
        T_full = np.zeros(2 * n)
        for k in range(2 * n):
            # compute T at lag k for direction
            alpha_val = int(k)
            if alpha_val < n:
                ll = alpha_val
            else:
                ll = alpha_val - n
            tv = 0.0
            for m_idx in range(n):
                # This is getting too slow for inner loop. Use the precomputed approach.
                pass
        # Use simpler approach: evaluate f(s) directly
        s_vals = np.linspace(0.0, 0.3, n_grid)
        best_s = 0.0
        best_f = best_c5
        for sv in s_vals:
            h_try = h + sv * direction
            h_try = _project(h_try, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_s = sv
        if best_s > 1e-10 and best_f < best_c5 - 1e-13:
            h = _project(h + best_s * direction, target)
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            alpha = min(alpha * alpha_mult, 200.0)
        else:
            alpha = min(alpha * 1.2, 200.0)
            if alpha > 100:
                break
    return best_h, best_c5


def _gradient_descent_fast(h, deadline, K=25, max_iters=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = 20.0
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        full = np.correlate(h, 1.0 - h, mode='full')
        M = float(full.max())
        sorted_idx = np.argsort(-full)[:K]
        top_vals = full[sorted_idx]
        spread = max(float(top_vals[0] - top_vals[-1]), 1e-10)
        w = np.exp(alpha * (top_vals - top_vals[0]) / spread)
        w /= w.sum()

        grad = np.zeros(n)
        for wi, idx in zip(w, sorted_idx):
            li = int(idx) - (n - 1)
            if li >= 0:
                m = n - li
                for k in range(n):
                    g = 0.0
                    if k >= li and k < n:
                        g += 1.0 - h[k - li]
                    if k + li < n:
                        g -= h[k + li]
                    grad[k] += wi * g
            else:
                li2 = -li
                for k in range(n):
                    g = 0.0
                    if k >= 0 and k + li2 < n:
                        g += 1.0 - h[k + li2]
                    if k - li2 >= 0 and k - li2 < n:
                        g -= h[k - li2]
                    grad[k] += wi * g
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        direction = -grad / gn
        # Line search
        best_s = 0.0
        best_f = best_c5
        step = 0.15
        for trial in range(20):
            sv = step * (0.5 ** trial)
            h_try = _project(h + sv * direction, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_s = sv
            if sv < 1e-4:
                break
        if best_s > 1e-10 and best_f < best_c5 - 1e-12:
            h = _project(h + best_s * direction, target)
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            alpha = min(alpha * 1.3, 300.0)
            if stall > 20:
                break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _block_pass(h, deadline, block_sizes=(2, 3, 4, 5)):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for bs in block_sizes:
        if time.time() > deadline:
            break
        for start_a in range(0, n - bs + 1, max(1, n // 40)):
            if time.time() > deadline:
                break
            for start_b in range(start_a + bs, n - bs + 1, max(1, n // 40)):
                if time.time() > deadline:
                    break
                a_idx = np.arange(start_a, start_a + bs)
                b_idx = np.arange(start_b, start_b + bs)
                mean_a = h[a_idx].mean()
                mean_b = h[b_idx].mean()
                if abs(mean_a - mean_b) < 1e-8:
                    continue
                # Try moving mass from higher block to lower
                for frac in (0.1, 0.25, 0.5, 0.75, 0.9):
                    if mean_a > mean_b:
                        delta = (mean_a - mean_b) * frac
                        h[a_idx] -= delta
                        h[b_idx] += delta
                    else:
                        delta = (mean_b - mean_a) * frac
                        h[a_idx] += delta
                        h[b_idx] -= delta
                    if np.any(h < -1e-10) or np.any(h > 1 + 1e-10):
                        h[a_idx] += delta if mean_a > mean_b else -delta
                        h[b_idx] -= delta if mean_a > mean_b else delta
                        continue
                    h = _project(h, target)
                    c5_try = _exact_c5(h)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h.copy()
                    else:
                        h = best_h.copy()
    return best_h, best_c5


def _micro_optimize(h, deadline, n_pairs_per_iter=200, max_iters=5000):
    """Very fine pair optimization focusing on small moves."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    S_r, S_l = _compute_corr_sets(h)
    cur_max = max(float(S_r.max()), float(S_l.max()))
    lag = np.arange(n)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        for _ in range(n_pairs_per_iter):
            if time.time() > deadline:
                break
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i == j:
                continue
            hi_v, hj_v = h[i], h[j]
            s_min = max(-hi_v, hj_v - 1.0)
            s_max = min(1.0 - hi_v, hj_v)
            if s_min >= s_max - 1e-14:
                continue
            diff = abs(i - j)
            a, b = (i, j) if i > j else (j, i)
            A = np.zeros(n)
            B = np.zeros(n)
            m = lag <= a
            if np.any(m):
                A[m] += 1.0 - h[a - lag[m]]
                B[m] -= h[a - lag[m]]
            m = lag <= b
            if np.any(m):
                A[m] -= 1.0 - h[b - lag[m]]
                B[m] += h[b - lag[m]]
            m = lag < (n - a)
            if np.any(m):
                A[m] -= h[a + lag[m]]
                B[m] += 1.0 - h[a + lag[m]]
            m = lag < (n - b)
            if np.any(m):
                A[m] += h[b + lag[m]]
                B[m] -= 1.0 - h[b + lag[m]]
            Q = np.zeros(n)
            Q[0] = 2.0
            Q[diff] = -1.0
            # Fine grid near 0
            span = min(0.05, (s_max - s_min) * 0.1)
            s_vals = np.linspace(max(s_min, -span), min(s_max, span), 15)
            if len(s_vals) < 2:
                s_vals = np.array([s_min + 1e-6, s_max - 1e-6])
            sq = s_vals * s_vals
            obj_r = (S_r[None, :] + s_vals[:, None] * A[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
            obj_l = (S_l[None, :] + s_vals[:, None] * B[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
            obj = np.maximum(obj_r, obj_l)
            best_k = int(obj.argmin())
            best_s = s_vals[best_k]
            best_val = obj[best_k]
            if best_val < cur_max - 1e-14 and abs(best_s) > 1e-12:
                h[i] += best_s
                h[j] -= best_s
                h = np.clip(h, 0.0, 1.0)
                S_r, S_l = _compute_corr_sets(h)
                cur_max = max(float(S_r.max()), float(S_l.max()))
                c5_now = cur_max * dx
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
    return best_h, best_c5


def _try_resolutions(h_best, c5_best, deadline, rng):
    n_best = len(h_best)
    cands = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16, -12, -10, 18, 20]:
        n_try = n_best + dn
        if n_try < 40 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(h_best, n_try)
        c5_r = _exact_c5(h_r)
        cands.append((c5_r, n_try, h_r))
    cands.sort(key=lambda x: x[0])
    best_h = h_best.copy()
    best_c5 = c5_best
    for c5_r, n_try, h_r in cands[:6]:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 4.0)
        h_opt, c5_opt = _pair_pass(h_r, sub_dl, n_grid=35)
        sub_dl2 = min(deadline, time.time() + 3.0)
        h_opt, c5_opt2 = _gradient_descent_fast(h_opt, sub_dl2, K=20, max_iters=100)
        if c5_opt2 < c5_opt:
            c5_opt = c5_opt2
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5


def _raven_pair_pass(h, deadline, n_grid=50):
    """Pair pass with finer grid and exhaustive inner loop."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _exact_c5(h)
    S_r, S_l = _compute_corr_sets(h)
    cur_max = max(float(S_r.max()), float(S_l.max()))
    lag = np.arange(n)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        for i, j in pairs:
            if time.time() > deadline:
                break
            hi_v, hj_v = h[i], h[j]
            for swap_dir in (0, 1):
                if swap_dir == 0:
                    a, b = i, j
                else:
                    a, b = j, i
                ha, hb = h[a], h[b]
                s_min = max(-ha, hb - 1.0)
                s_max = min(1.0 - ha, hb)
                if s_min >= s_max - 1e-14:
                    continue
                diff = abs(a - b)
                A = np.zeros(n)
                B = np.zeros(n)
                m = lag <= a
                if np.any(m):
                    A[m] += 1.0 - h[a - lag[m]]
                    B[m] -= h[a - lag[m]]
                m = lag <= b
                if np.any(m):
                    A[m] -= 1.0 - h[b - lag[m]]
                    B[m] += h[b - lag[m]]
                m = lag < (n - a)
                if np.any(m):
                    A[m] -= h[a + lag[m]]
                    B[m] += 1.0 - h[a + lag[m]]
                m = lag < (n - b)
                if np.any(m):
                    A[m] += h[b + lag[m]]
                    B[m] -= 1.0 - h[b + lag[m]]
                Q = np.zeros(n)
                Q[0] = 2.0
                Q[diff] = -1.0
                s_vals = np.linspace(s_min, s_max, n_grid)
                sq = s_vals * s_vals
                obj_r = (S_r[None, :] + s_vals[:, None] * A[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
                obj_l = (S_l[None, :] + s_vals[:, None] * B[None, :] - sq[:, None] * Q[None, :]).max(axis=1)
                obj = np.maximum(obj_r, obj_l)
                best_k = int(obj.argmin())
                best_s = s_vals[best_k]
                best_val = obj[best_k]
                if best_val < cur_max - 1e-13 and abs(best_s) > 1e-12:
                    h[a] += best_s
                    h[b] -= best_s
                    h = np.clip(h, 0.0, 1.0)
                    S_r, S_l = _compute_corr_sets(h)
                    cur_max = max(float(S_r.max()), float(S_l.max()))
                    c5_now = cur_max * dx
                    if c5_now < best_c5:
                        best_c5 = c5_now
                    improved_any = True
                    break
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / n0) + 0.1 * np.cos(4 * np.pi * x / n0)
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Initial gradient descent (10% time)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _gradient_descent_fast(h, d1, K=25, max_iters=200)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Thorough pair-exchange local search (30% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _pair_pass(best_h, min(d2, time.time() + budget_s * 0.10), n_grid=35)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    h2b, c5_2b = _raven_pair_pass(best_h, d2, n_grid=40)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()

    # Phase 3: Gradient + pair alternating (20% time)
    d3 = t0 + budget_s * 0.60
    for _ in range(3):
        if time.time() > d3:
            break
        sub1 = min(d3, time.time() + budget_s * 0.04)
        h3, c5_3 = _gradient_descent_fast(best_h, sub1, K=20, max_iters=80)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        sub2 = min(d3, time.time() + budget_s * 0.04)
        h3b, c5_3b = _pair_pass(best_h, sub2, n_grid=40)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution (20% time)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = _try_resolutions(best_h, best_c5, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Fine micro-optimization (15% time)
    d5 = t0 + budget_s * 0.95
    h5, c5_5 = _micro_optimize(best_h, d5, n_pairs_per_iter=100, max_iters=4000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final polish
    d6 = deadline - 2.0
    h6, c5_6 = _raven_pair_pass(best_h, d6, n_grid=50)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final validation and return
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
