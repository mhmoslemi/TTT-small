        c2 = -np.correlate(d, d, mode='full')
        a_best, v_best = _minimize_line(corr_full, c1, c2, dx, a_lo, a_hi, n_grid=40)
        if abs(a_best) > 1e-10:
            h_new = _project(h + a_best * d, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur - 5e-14:
                h = h_new
                cur = c5_new
                corr_full = np.correlate(h, 1.0 - h, mode='full')
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_h(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _random_start(rng, n, freq, phase):
    x = np.arange(n)
    v = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / n + phase)
    v = v + rng.normal(0, 0.08, n)
    return _project(v, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _random_start(rng, n0, 1, 0)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0
    # Phase 1: soft-max descent (40%)
    d1 = t0 + budget_s * 0.40
    h1, c1 = _soft_descent(h, d1, n_gammas=8, iters_per_gamma=500)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()
        best_n = len(h1)
    # Phase 2: pair swap (15%)
    d2 = t0 + budget_s * 0.55
    h2, c2 = _pair_search(best_h, d2, max_iter=80000)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()
        best_n = len(h2)
    # Phase 3: try nearby n values (25%)
    d3 = t0 + budget_s * 0.80
    tried = set()
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20]:
        if time.time() > d3:
            break
        nt = best_n + dn
        if nt < 60 or nt > 220 or nt % 2 != 0:
            continue
        if nt in tried:
            continue
        tried.add(nt)
        hr = _resample(best_h, nt)
        sub1 = min(d3, time.time() + 12)
        hr, cr = _soft_descent(hr, sub1, n_gammas=5, iters_per_gamma=300)
        if time.time() < d3:
            sub2 = min(d3, time.time() + 12)
            hr, cr2 = _pair_search(hr, sub2, max_iter=30000)
            if cr2 < cr:
                cr = cr2
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            best_n = nt
    # Phase 4: varied starts (8%)
    d4 = t0 + budget_s * 0.88
    starts = [_binary_h(best_h), _random_start(rng, best_n, 2, 0.5),
              _random_start(rng, best_n, 3, 1.0), _random_start(rng, best_n, 1, np.pi)]
    for s in starts:
        if time.time() > d4:
            break
        sub = min(d4, time.time() + 8)
        hs, cs = _soft_descent(s, sub, n_gammas=5, iters_per_gamma=250)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()
            best_n = len(hs)
    # Phase 5: final polish (7%)
    d5 = deadline
    h5, c5 = _soft_descent(best_h, min(d5, time.time() + 25), n_gammas=4, iters_per_gamma=400)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()
    if time.time() < deadline - 3:
        h6, c6 = _pair_search(best_h, deadline, max_iter=40000)
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6.copy()
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
