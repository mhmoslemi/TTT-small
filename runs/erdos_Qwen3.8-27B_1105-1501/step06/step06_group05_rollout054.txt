def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))

def _softmax_direction(h, K=20, beta=50.0):
    """Get a descent direction using softmax-weighted gradients of top K overlaps"""
    n = len(h)
    overlaps = _all_overlaps(h)
    # Weight by softmax of overlaps
    scores = overlaps
    m = float(np.max(scores))
    w = np.exp((scores - m) * beta)
    w /= float(np.sum(w))
    # Only use top K
    top_K = np.argsort(-scores)[:K]
    w[:len(scores)] = 0  # reset
    w[top_K] = w[top_K]
    w = w[top_K] / float(np.sum(w[top_K]))
    
    # Compute weighted gradient
    G = np.zeros(n)
    for idx, l in zip(top_K, range(K)):
        lag = int(idx - (n - 1))
        G += w[idx] * _overlap_grad(h, lag)
    
    # Make sum-zero
    G -= G.mean()
    # Negate for descent
    d = -G
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        return None
    return d / nrm

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _all_overlaps(h):
    n = len(h)
    overlaps = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        l = idx - (n - 1)
        i_min = max(0, -l)
        i_max = min(n, n - l)
        if i_min >= i_max:
            continue
        i = np.arange(i_min, i_max)
        overlaps[idx] = float(np.sum(h[i] * (1.0 - h[i + l])))
    return overlaps

def _overlap_grad(h, l):
    n = len(h)
    g = np.zeros(n)
    if l >= 0:
        if l < n:
            jr = np.arange(0, n - l)
            g[jr] += (1.0 - h[jr + l])
            jr2 = np.arange(l, n)
            g[jr2] += (-h[jr2 - l])
    else:
        m = -l
        if m < n:
            jr = np.arange(m, n)
            g[jr] += (1.0 - h[jr - m])
            jr2 = np.arange(0, n - m)
            g[jr2] += (-h[jr2 + m])
    return g

def _slsqp_step(h, K=15, maxiter=40):
    global _slags
    n = len(h)
    overlaps = _all_overlaps(h)
    top_idx = np.argsort(-overlaps)[:K]
    lags = [int(idx - (n - 1)) for idx in top_idx]
    _slags = lags
    z0 = float(np.max(overlaps)) * 1.002 + 0.001
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(-1.0, 5.0)]
    cons = [
        {'type': 'ineq', 'fun': _con_vec_fun, 'jac': _con_vec_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]
    try:
        res = minimize(_obj_fun, x0, method='SLSQP', jac=_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = np.clip(res.x[:-1], 0.0, 1.0)
        h_new = _project(h_new, 0.5 * n)
        f_new = _exact_c5(h_new)
        f_cur = _exact_c5(h)
        if f_new < f_cur - 1e-10:
            return h_new
    except Exception:
        pass
    return None

def _quad_line_search(h, d):
    n = len(h)
    dx = 2.0 / n
    A = _all_overlaps(h)
    corr_hd = np.correlate(h, d, mode='full')
    corr_dh = np.correlate(d, h, mode='full')
    corr_dd = np.correlate(d, d, mode='full')
    d_prefix = np.concatenate([[0.0], np.cumsum(d)])
    B = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        l = idx - (n - 1)
        i_min = max(0, -l)
        i_max = min(n, n - l)
        sum_d = d_prefix[i_max] - d_prefix[i_min]
        B[idx] = sum_d - corr_hd[idx] - corr_dh[idx]
    C = -corr_dd
    def obj_at(t):
        return float(np.max(A + t * B + t * t * C)) * dx
    pos = d > 1e-15
    neg = d < -1e-15
    lo_list = [0.0]
    up_list = [1e10]
    if np.any(pos):
        lo_list.append(float(np.min(-h[pos] / d[pos])))
        up_list.append(float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo_list.append(float(np.min((1.0 - h[neg]) / d[neg])))
        up_list.append(float(np.min(-h[neg] / d[neg])))
    L = max(lo_list)
    U = min(up_list)
    if U - L < 1e-12:
        return 0.0, obj_at(0.0)
    if L >= 0.0:
        L = 0.0
    if U <= 0.0:
        U = 0.0
    if U <= L:
        return 0.0, obj_at(0.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = obj_at(c1), obj_at(c2)
    f0 = obj_at(0.0)
    best_t, best_f = 0.0, f0
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            if f2 < best_f:
                best_t, best_f = c2, f2
            c1 = b - gr * (b - a)
            f1 = obj_at(c1)
            if f1 < best_f:
                best_t, best_f = c1, f1
        else:
            a, c1, f1 = c1, c2, f2
            if f1 < best_f:
                best_t, best_f = c1, f1
            c2 = a + gr * (b - a)
            f2 = obj_at(c2)
            if f2 < best_f:
                best_t, best_f = c2, f2
    tm = 0.5 * (a + b)
    fm = obj_at(tm)
    if fm < best_f:
        best_t, best_f = tm, fm
    if best_t == 0.0 and f0 < best_f:
        best_t, best_f = 0.0, f0
    return best_t, best_f

def _softmax_direction(h, K=20, beta=40.0, rng=None):
    n = len(h)
    overlaps = _all_overlaps(h)
    scores = overlaps
    m = float(np.max(scores))
    w = np.exp((scores - m) * beta)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros_like(scores)
        w[int(np.argmax(scores))] = 1.0
    else:
        w /= w_sum
    G = np.zeros(n)
    top_K = min(K, 2 * n - 1)
    idxs = np.argsort(-scores)[:top_K]
    for idx in idxs:
        l = int(idx - (n - 1))
        G += w[idx] * _overlap_grad(h, l)
    G -= G.mean()
    d = -G
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        if rng is not None:
            d = rng.standard_normal(n)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                return None
    else:
        d = d / nrm
    return d

def _pair_direction(h, i, j):
    n = len(h)
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, 0.5 * n_new)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, 0.5 * n)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 5))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, 0.5 * n)

def _optimize_slashed(h, deadline, rng, use_slsqp=True, slsqp_K=15,
                      slsqp_maxiter=35, softmax_K=20, beta=40.0,
                      slsqp_frac=0.3, pair_frac=0.15, rand_frac=0.08,
                      sigma=0.04, uphill=3e-4):
    global _slags
    n = len(h)
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    stall = 0
    restarts = 0
    while time.time() < deadline:
        r = rng.random()
        if use_slsqp and r < slsqp_frac:
            h_new = _slsqp_step(h, K=slsqp_K, maxiter=slsqp_maxiter)
            if h_new is not None:
                nf = _exact_c5(h_new)
                if nf < best_f - 1e-12:
                    best_f = nf
                    best_h = h_new.copy()
                if nf < cur_f + uphill:
                    h = h_new
                    cur_f = nf
                    stall = 0
                else:
                    h = best_h.copy()
                    cur_f = best_f
                    stall += 1
            else:
                d = _softmax_direction(h, K=softmax_K, beta=beta, rng=rng)
                if d is None:
                    stall += 1
                    continue
                t, newf = _quad_line_search(h, d)
                hn = _project(h + t * d, 0.5 * n)
                nf = _exact_c5(hn)
                if nf < best_f - 1e-12:
                    best_f = nf
                    best_h = hn.copy()
                if nf < cur_f + uphill:
                    h = hn
                    cur_f = nf
                    stall = 0
                else:
                    h = best_h.copy()
                    cur_f = best_f
                    stall += 1
        elif r < slsqp_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = _pair_direction(h, i, j)
            t, newf = _quad_line_search(h, d)
            if abs(t) < 1e-12:
                stall += 1
                continue
            hn = _project(h + t * d, 0.5 * n)
            nf = _exact_c5(hn)
            if nf < best_f - 1e-12:
                best_f = nf
                best_h = hn.copy()
            if nf < cur_f + uphill:
                h = hn
                cur_f = nf
                stall = 0
            else:
                h = best_h.copy()
                cur_f = best_f
                stall += 1
        elif r < slsqp_frac + pair_frac + rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d /= nrm
            t, newf = _quad_line_search(h, d)
            if abs(t) < 1e-12:
                stall += 1
                continue
            hn = _project(h + t * d, 0.5 * n)
            nf = _exact_c5(hn)
            if nf < best_f - 1e-12:
                best_f = nf
                best_h = hn.copy()
            if nf < cur_f + uphill:
                h = hn
                cur_f = nf
                stall = 0
            else:
                h = best_h.copy()
                cur_f = best_f
                stall += 1
        else:
            d = _softmax_direction(h, K=softmax_K, beta=beta, rng=rng)
            if d is None:
                stall += 1
                continue
            t, newf = _quad_line_search(h, d)
            if abs(t) < 1e-12:
                stall += 1
                continue
            hn = _project(h + t * d, 0.5 * n)
            nf = _exact_c5(hn)
            if nf < best_f - 1e-12:
                best_f = nf
                best_h = hn.copy()
            if nf < cur_f + uphill:
                h = hn
                cur_f = nf
                stall = 0
            else:
                h = best_h.copy()
                cur_f = best_f
                stall += 1
        if stall > 120:
            restarts += 1
            if restarts > 25:
                h = _project(best_h + rng.normal(0, max(0.03, sigma * 3), n), 0.5 * n)
            else:
                h = _project(best_h + rng.normal(0, sigma, n), 0.5 * n)
            cur_f = _exact_c5(h)
            stall = 0
            sigma = max(0.003, sigma * 0.85)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        base = _sine_init(n, 1, 0.0, rng)
    n0 = len(base)
    h0 = _project(base, 0.5 * n0)

    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: Multi-start (12% of budget)
    d1 = t0 + budget_s * 0.12
    starts = [h0.copy()]
    for f_, p_ in [(1, 0.0), (1, 1.0), (2, 0.5), (3, 0.0), (2, 1.0)]:
        starts.append(_sine_init(n0, f_, p_, rng))
    for _ in range(3):
        starts.append(_block_init(n0, rng))
    # Also try complement and reverse of h0
    starts.append(_project(1.0 - h0, 0.5 * n0))
    starts.append(_project(h0[::-1].copy(), 0.5 * n0))

    for s in starts:
        if time.time() > d1:
            break
        sub = min(d1, time.time() + max(3.0, budget_s * 0.015))
        hh, ff = _optimize_slashed(s, sub, rng, use_slsqp=True,
                                   slsqp_K=12, slsqp_maxiter=25,
                                   softmax_K=15, beta=30.0,
                                   slsqp_frac=0.4, pair_frac=0.2,
                                   rand_frac=0.1, sigma=0.03)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Phase 2: Main deep optimization (55% of budget)
    d2 = t0 + budget_s * 0.72
    hh, ff = _optimize_slashed(best_h, d2, rng, use_slsqp=True,
                               slsqp_K=18, slsqp_maxiter=40,
                               softmax_K=22, beta=50.0,
                               slsqp_frac=0.35, pair_frac=0.18,
                               rand_frac=0.06, sigma=0.035,
                               uphill=2e-4)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: Multi-resolution (15% of budget)
    d3 = t0 + budget_s * 0.88
    bn = len(best_h)
    tried = set()
    for dn in [-2, 2, -4, 4, -6, 6, 8, -8, -10, 10, -14, 14]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 30 or n_try > 260:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub = min(d3, time.time() + max(3.0, budget_s * 0.018))
        hr2, fr2 = _optimize_slashed(hr, sub, rng, use_slsqp=True,
                                     slsqp_K=12, slsqp_maxiter=25,
                                     softmax_K=15, beta=35.0,
                                     slsqp_frac=0.35, pair_frac=0.2,
                                     rand_frac=0.08, sigma=0.03)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 4: Final polish (8% of budget)
    hh, ff = _optimize_slashed(best_h, deadline, rng, use_slsqp=True,
                               slsqp_K=20, slsqp_maxiter=45,
                               softmax_K=25, beta=60.0,
                               slsqp_frac=0.5, pair_frac=0.25,
                               rand_frac=0.03, sigma=0.015,
                               uphill=1e-4)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    best_h = _project(best_h, 0.5 * len(best_h))
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
