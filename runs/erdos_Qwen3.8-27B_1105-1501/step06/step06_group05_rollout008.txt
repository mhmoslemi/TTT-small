import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) < 1e-14:
            break
        if res > 0:
            mask = x < 1.0 - 1e-13
        else:
            mask = x > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _smooth_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp((corr - mx) / tau)
    ws = float(w.sum())
    if ws < 1e-30:
        r = int(np.argmax(corr))
        k = r - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            if k == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[k:] = 1.0 - h[:n - k]
                g[:n - k] -= h[k:]
        else:
            j = -k
            g[:n - j] = 1.0 - h[j:]
            g[j:] -= h[:n - j]
        return g
    w = w / ws
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        k = r - (n - 1)
        wk = w[r]
        if k >= 0:
            if k == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                g[k:] += wk * (1.0 - h[:n - k])
                g[:n - k] -= wk * h[k:]
        else:
            j = -k
            g[:n - j] += wk * (1.0 - h[j:])
            g[j:] -= wk * h[:n - j]
    return g


def _line_search(corr, T, Q, dx, L, U):
    if U - L < 1e-14:
        t = 0.5 * (L + U)
        return t, float(np.max(corr + t * T - t * t * Q)) * dx
    nq = 60
    grid = np.linspace(L, U, nq)
    vals = corr[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(corr + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t = grid[i]
        tv = obj[i]
    return t, tv


def _pair_direction(h, grad, rng, n_pairs=3):
    n = len(h)
    pos_idx = np.argsort(grad)[-n_pairs:]
    neg_idx = np.argsort(grad)[:n_pairs]
    i = int(pos_idx[rng.integers(n_pairs)])
    j = int(neg_idx[rng.integers(n_pairs)])
    if i == j:
        return None
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    return d


def _optimize(h, deadline, rng, tau_start=0.04, tau_end=0.003,
              pair_frac=0.15, rand_frac=0.08, momentum=0.6):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    vel = np.zeros(n)
    t_start = time.time()
    total_time = max(1.0, deadline - t_start)
    
    while time.time() < deadline:
        elapsed = time.time() - t_start
        frac = min(1.0, elapsed / total_time)
        tau = tau_start * (tau_end / tau_start) ** frac
        if tau < tau_end:
            tau = tau_end
        
        r = rng.random()
        d = None
        if r < rand_frac:
            d = rng.standard_normal(n)
        elif r < rand_frac + pair_frac:
            g_tmp = _smooth_grad(h, max(tau, 0.01))
            d = _pair_direction(h, g_tmp, rng, n_pairs=4)
            if d is None:
                d = rng.standard_normal(n)
        else:
            d = _smooth_grad(h, tau)
        
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        
        vel = momentum * vel + (1.0 - momentum) * d
        vel -= vel.mean()
        vnorm = float(np.linalg.norm(vel))
        if vnorm > 1e-14:
            d = vel / vnorm
        
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-12:
            stall += 1
            continue
        
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, 2.0 / n, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue
        
        hn = h + t * d
        if hn.min() < -1e-12 or hn.max() > 1.0 + 1e-12:
            hn = _project(hn, target)
        else:
            hn = np.clip(hn, 0.0, 1.0)
            hn = _project(hn, target)
        
        nf = _exact_c5(hn)
        
        anneal_temp = 0.0005 * (1.0 - frac) + 0.00005
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + anneal_temp or rng.random() < anneal_temp:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
        
        if stall > 100:
            sigma = 0.04 * (1.0 - frac) + 0.005
            h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            vel = np.zeros(n)
            stall = 0
    
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 5 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def _try_quantize(h, max_denom=16):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_f = _exact_c5(h)
    for denom in [2, 3, 4, 5, 6, 8, 10, 12, 16]:
        if max_denom is not None and denom > max_denom:
            break
        hq = np.round(h * denom) / denom
        hq = _project(hq, target)
        hq = np.clip(hq, 0.0, 1.0)
        fq = _exact_c5(hq)
        if fq < best_f:
            best_f = fq
            best_h = hq.copy()
    return best_h, best_f

        hr = _resample(best_h, n_try)

        hr2, fr2 = _optimize(hr, sub, rng, tau_start=0.03, tau_end=0.003,
                             pair_frac=0.2, rand_frac=0.08, momentum=0.55)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try
    
    # Phase 4: Intensive final polish (10% of budget)
    hh, ff = _optimize(best_h, deadline, rng, tau_start=0.015, tau_end=0.001,
                       pair_frac=0.35, rand_frac=0.04, momentum=0.4)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    
    # Final: try quantization
    best_h, qf = _try_quantize(best_h, max_denom=8)
    if qf < best_f:
        best_f = qf
    
    # Ensure feasibility
    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    
    if time.time() > deadline - 2:
        return best_h, final_f, len(best_h)
    
    return best_h, final_f, len(best_h)
