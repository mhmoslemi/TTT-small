```python
import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13 and abs(res) > 1e-15:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lsoftmax_val(h, beta):
    c = _corr_all(h)
    cmax = np.max(c)
    y = beta * (c - cmax)
    y -= y.max()
    emax = np.exp(y).max()
    if emax > 0:
        y -= np.log(emax)
    return cmax + np.log(np.exp(y).sum()) / beta

def _lsoftmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = _corr_all(h)
    cmax = np.max(c)
    y = beta * (c - cmax)
    y -= y.max()
    w = np.exp(y)
    wsum = w.sum()
    if wsum < 1e-300:
        # Degenerate: all weight on the max
        idx = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
    else:
        w /= wsum
    
    grad = np.zeros(n)
    for j in range(n):
        # c_k(x) for x = h[j]: c_k = c_low[k] + x * slope[k]
        # grad[j] = dx * sum_k w[k] * slope[k]
        # slope[k] = dc_k/dh[j]
        # = (1-h[j+s_k]) - h[j-s_k] (where valid)
        # s_k = k - n + 1
        
        # For term (1-h[j+s]): valid when 0 <= j+s < n, i.e., -j <= s < n-j
        # For term -h[j-s]: valid when 0 <= j-s < n, i.e., j-n+1 <= s <= j
        
        # Combined: sum over s where (j+s valid) of (1-h[j+s]) * w[n-1+s]
        #          - sum over s where (j-s valid) of h[j-s] * w[n-1+s]
        
        s1_lo = -j  # from j+s >= 0
        s1_hi = n - 1 - j  # from j+s <= n-1
        if s1_hi >= s1_lo:
            s_range = np.arange(max(s1_lo, -(n-1)), min(s1_hi, n-1) + 1)
            grad[j] += np.sum(w[n-1+s_range] * (1.0 - h[j+s_range]))
        
        s2_lo = j - (n-1)  # from j-s >= 0 => s <= j
        s2_hi = j  # from j-s <= n-1 => s >= j-(n-1)
        if s2_hi >= s2_lo:
            s_range2 = np.arange(max(s2_lo, -(n-1)), min(s2_hi, n-1) + 1)
            grad[j] -= np.sum(w[n-1+s_range2] * h[j-s_range2])
    
    grad *= dx
    return grad

def _smooth_optimize(h0, deadline, n_betas=12, steps_per_beta=150):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    betas = np.geomspace(2.0, 300.0, n_betas)
    
    for beta in betas:
        if time.time() > deadline:
            break
        # Adaptive learning rate
        lr = 1.0 / (1.0 + 0.5 * beta)
        for step in range(steps_per_beta):
            if time.time() > deadline:
                break
            g = _lsoftmax_grad(h, beta)
            h_new = _project(h - lr * g, target)
            if np.max(np.abs(h_new - h)) < 1e-16:
                break
            h = h_new
            c5 = _exact_c5(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
            elif step > 20:
                # No improvement for a while, reduce lr
                lr *= 0.5
                if lr < 1e-6:
                    break
    
    return best_h, best_c5

def _coord_descent(h0, deadline, max_sweeps=300):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stall_count = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            # Compute c at h[j]=0 and h[j]=1
            old_j = h[j]
            h[j] = 0.0
            c0 = _corr_all(h)
            h[j] = 1.0
            c1 = _corr_all(h)
            h[j] = old_j
            
            slope = c1 - c0
            # Minimize max_k (c0[k] + x*slope[k]) for x in [0,1]
            # Golden section search
            a, b = 0.0, 1.0
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            fc = np.max(c0 + c_pt * slope)
            fd = np.max(c0 + d_pt * slope)
            for _ in range(40):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    fc = np.max(c0 + c_pt * slope)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    fd = np.max(c0 + d_pt * slope)
            x_opt = (a + b) / 2.0
            
            if abs(x_opt - old_j) > 1e-10:
                h[j] = x_opt
                improved = True
        
        # Project to maintain sum
        h = _project(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 30:
            break
    
    return best_h, best_c5

def _slsqp_refine(h0, deadline, max_outer=15):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _corr_all(h)
        K = min(30, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        
        t_init = cur_c5 + 1e-5
        
        def obj_fn(xv):
            return xv[n]
        
        def obj_grad(xv):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def ineq_fn(xv):
            hx = xv[:n]
            c = _corr_all(hx)
            return xv[n] - c[top_k]
        
        def ineq_jac(xv):
            hx = xv[:n]
            # Compute gradients of c_k w.r.t. h for k in top_k
            J = np.zeros((K, n + 1))
            for i, k_idx in enumerate(top_k):
                s = k_idx - n + 1  # lag
                # dc_k/dh[j] = (1-h[j+s])*1_{valid} - h[j-s]*1_{valid}
                for j in range(n):
                    val = 0.0
                    if 0 <= j + s < n:
                        val += (1.0 - h[j + s])
                    if 0 <= j - s < n:
                        val -= h[j - s]
                    J[i, j] = val * dx
                J[i, n] = 1.0
            return J
        
        def eq_fn(xv):
            return np.sum(xv[:n]) - target
        
        def eq_jac(xv):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        
        x0 = np.concatenate([h, [t_init]])
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-12})
        
        if res.success and res.x[n] < cur_c5 - 1e-12:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
```
