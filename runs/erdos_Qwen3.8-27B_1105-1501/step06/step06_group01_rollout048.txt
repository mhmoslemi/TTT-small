import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Exact correction
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(fixed, dx, n):
    """Build correlation coefficient matrix.
    A[k,i] = coefficient of variable[i] in correlation[lag k] when fixed vector is the other operand.
    corr(a, fixed)[k] = sum_i a[i] * A[k,i] (before multiplying by dx, but we include dx here)
    """
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m = k - n + 1  # effective lag
        i_start = max(0, m)
        i_end = min(n, n + m)
        if i_end > i_start:
            A[k, i_start:i_end] = fixed[i_start-m:i_end-m] * dx
    return A

def _solve_minimax_lp(A, n):
    """Solve: min tau s.t. A@x - tau <= 0, sum(x) = n/2, 0<=x<=1, tau>=0"""
    nvar = n + 1  # n variables + tau
    c = np.zeros(nvar)
    c[-1] = 1.0  # minimize tau
    
    n_cons = A.shape[0]
    A_ub = np.hstack([A, -np.ones((n_cons, 1))])
    b_ub = np.zeros(n_cons)
    
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[:n], res.x[-1]
    return None, None

def _bridge_search(h1, h2, n, max_evals=50):
    """Search along line h1 to h2 for minimum C5."""
    target = n / 2.0
    best_h = h1.copy()
    best_c5 = _exact_c5(h1)
    
    # Evaluate at several points
    alphas = np.linspace(0, 1, max_evals)
    for alpha in alphas:
        h_try = (1-alpha) * h1 + alpha * h2
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_alternating(h, deadline, max_iter=500, n_bridge_evals=30):
    """Alternating LP optimization with bridge search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0

            break
        
        # Step 1: Fix b = 1-h, find optimal a
        b = 1.0 - h
        A1 = _build_A(b, dx, n)
        a_opt, tau1 = _solve_minimax_lp(A1, n)
        
        if a_opt is not None:
            a_opt = _project(a_opt, target)
            c5_a = _exact_c5(a_opt)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a_opt.copy()
        
        # Step 2: Fix a = a_opt, find optimal b
        if a_opt is not None:
            A2 = _build_A(a_opt, dx, n)
            b_opt, tau2 = _solve_minimax_lp(A2, n)
            
            if b_opt is not None:
                b_opt = _project(b_opt, target)
                h_from_b = 1.0 - b_opt
                h_from_b = _project(h_from_b, target)
                c5_b = _exact_c5(h_from_b)
                if c5_b < best_c5:

                h_bridge, c5_bridge = _bridge_search(a_opt, h_from_b, n, n_bridge_evals)
                if c5_bridge < best_c5:
                    best_c5 = c5_bridge
                    best_h = h_bridge.copy()

            stall += 1
        else:

    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A1(fixed, dx, n):
    """A1[i,j]: coeff of var[j] in corr(var, fixed)[i], i.e. A1[i,j] = fixed[j-i+n-1]*dx"""
    A = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        p = i - n + 1
        j_lo = max(0, p)
        j_hi = min(n, n + p)
        if j_hi > j_lo:
            A[i, j_lo:j_hi] = fixed[j_lo - p:j_hi - p] * dx
    return A

def _build_A2(fixed, dx, n):
    """A2[i,j]: coeff of var[j] in corr(fixed, var)[i], i.e. A2[i,j] = fixed[j+i-n+1]*dx"""
    A = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        p = i - n + 1
        j_lo = max(0, -p)
        j_hi = min(n, n - p)
        if j_hi > j_lo:
            A[i, j_lo:j_hi] = fixed[j_lo + p:j_hi + p] * dx
    return A

def _solve_minimax(A, n):
    nvar = n + 1
    c = np.zeros(nvar)
    c[-1] = 1.0
    ncons = A.shape[0]
    A_ub = np.hstack([A, -np.ones((ncons, 1))])
    b_ub = np.zeros(ncons)
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.array(res.x[:n], dtype=float), float(res.x[-1])
    except Exception:
        pass
    return None, None

def _bridge_search(h1, h2, n, n_evals=40):
    target = n / 2.0
    best_h = h1.copy()
    best_c5 = _exact_c5(h1)
    c5_at_1 = _exact_c5(h2)
    if c5_at_1 < best_c5:

        best_h = h2.copy()

        h_try = _project(h_try, target)

        if c5_try < best_c5:
