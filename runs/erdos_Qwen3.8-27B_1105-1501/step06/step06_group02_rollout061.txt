def solve_target_qp(h, R, G, M, rho, lambda_z=0.85, lambda_norm=0.15):
    """Solve the target QP. Returns direction d (length n) or None."""
    n = len(h)
    C = len(R)  # 2n-1
    
    # Variable ordering: [d_0,...,d_{n-1}, z_0,...,z_{C-1}]
    dim = n + C
    
    # Bounds
    lb_d = np.maximum(-h, -rho)
    ub_d = np.minimum(1.0 - h, rho)
    z_max = float(np.max(R)) - M + 1.0  # reasonable cap
    z_max = max(z_max, 0.1)
    lb_z = np.zeros(C)
    ub_z = np.full(C, z_max)
    
    bounds = list(zip(lb_d, ub_d)) + list(zip(lb_z, ub_z))
    
    # Initial guess
    d0 = np.zeros(n)
    z0 = np.maximum(R - M, 0.0)
    x0 = np.concatenate([d0, z0])
    
    # Scale factors
    scale_z = max(float(np.mean(z0)), 1e-6)
    norm_scale = rho * rho + 1e-12
    
    def objective(x):
        d = x[:n]
        z = x[n:]
        return lambda_z * float(np.sum(z)) / (C * scale_z) + lambda_norm * float(np.sum(d * d)) / (n * norm_scale)
    
    def gradient(x):
        d = x[:n]
        grad = np.zeros(dim)
        grad[:n] = 2.0 * lambda_norm * d / (n * norm_scale)
        grad[n:] = lambda_z / (C * scale_z)
        return grad
    
    # Constraints
    G_d = G  # (C, n)
    
    def constr_func(x):
        d = x[:n]
        z = x[n:]
        return z + M - G @ d - R  # should be >= 0
    
    # Equality: sum(d) = 0
    eq_vec = np.zeros(dim)
    eq_vec[:n] = 1.0
    
    constraints = [
        {'type': 'ineq', 'fun': constr_func},
        {'type': 'eq', 'fun': lambda x: float(np.sum(x[:n])), 'jac': lambda x: eq_vec}
    ]
    
    try:
        result = minimize(objective, x0, jac=gradient, bounds=bounds,
                         constraints=constraints, method='SLSQP',
                         options={'maxiter': 200, 'ftol': 1e-10})
    except Exception:
        return None
    
    if not np.all(np.isfinite(result.x)):
        return None
    
    d = result.x[:n]
    d -= float(np.mean(d))  # enforce zero sum
    
    if np.max(np.abs(d)) < 1e-12:
        return None
    
    return d

import numpy as np
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_row_grads(h, row_indices):
    n = len(h)
    grads = np.zeros((len(row_indices), n))
    for k_idx, k in enumerate(row_indices):
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            grads[k_idx, :m] += -h[d:d + m]
            grads[k_idx, d:d + m] += (1.0 - h[:m])
        else:
            r = -d
            m = n - r
            if m <= 0:
                continue
            grads[k_idx, :m] += (1.0 - h[r:r + m])
            grads[k_idx, r:r + m] += -h[:m]
    return grads

def exact_line_search(h, p, R0, dx):
    n = len(h)
    D = np.correlate(p, 1.0 - h, 'full') - np.correlate(h, p, 'full')
    Q = np.correlate(p, p, 'full')
    pos = p > 1e-14
    neg = p < -1e-14
    alpha_hi = 1.0
    if np.any(pos):
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    alpha_lo = -1.0
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max(h[neg] / p[neg])))
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, c5_exact(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = alpha_lo, alpha_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * D - c1 * c1 * Q)) * dx
    f2 = float(np.max(R0 + c2 * D - c2 * c2 * Q)) * dx
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * D - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * D - c2 * c2 * Q)) * dx
    opt = (a + b) / 2.0
    best = float(np.max(R0 + opt * D - opt * opt * Q)) * dx
    for alpha in [alpha_lo, alpha_hi]:
        v = float(np.max(R0 + alpha * D - alpha * alpha * Q)) * dx
        if v < best:
            best = v
            opt = alpha
    return opt, best

def gradient_step(h, R0, dx, n_active, step_scale):
    n = len(h)
    scores = R0 * dx
    max_score = float(np.max(scores))
    threshold = max_score - 0.005 * max_score
    candidates = np.where(scores >= threshold)[0]
    if len(candidates) < n_active:
        candidates = np.argsort(-scores)[:n_active]
    active = candidates[:min(len(candidates), n_active)]
    grads = compute_row_grads(h, active)
    w = np.exp((scores[active] - max_score) / (0.008 * max_score + 1e-15))
    w /= float(np.sum(w))
    g = grads.T @ w
    g -= float(np.mean(g))
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, c5_exact(h), R0
    d = -g / gn * step_scale
    opt, best_val = exact_line_search(h, d, R0, dx)
    if opt < 1e-10:
        return h, c5_exact(h), R0
    h_new = h + opt * d
    h_new = project(h_new, n / 2.0)
    new_R0 = np.correlate(h_new, 1.0 - h_new, 'full')
    return h_new, best_val, new_R0

def gradient_polish(h, deadline, n_active=25):
    n = len(h)
    dx = 2.0 / n
    R0 = np.correlate(h, 1.0 - h, 'full')
    cur = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur
    stall = 0
    scales = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002]
    while time.time() < deadline and stall < 150:
        sc = scales[min(stall // 30, len(scales) - 1)]
        h, c5, R0 = gradient_step(h, R0, dx, n_active, sc)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def binary_anneal(h_init, deadline, rng, T_init=0.004, T_min=0.00008):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur = c5_exact(h)
    best = cur
    best_h = h.copy()
    ones = [int(x) for x in np.where(h > 0.5)[0]]
    zeros = [int(x) for x in np.where(h <= 0.5)[0]]
    T = T_init
    patience = 0
    reheat_count = 0
    while time.time() < deadline:
        if patience > 4000:
            if reheat_count < 5:
                T = max(T, T_init * 0.5)
                reheat_count += 1
                patience = 0
            elif T <= T_min:
                break
            else:
                T = T_min
                patience = 0
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        do_4swap = rng.random() < 0.3
        if do_4swap and n_ones >= 2 and n_zeros >= 2:
            i1 = ones[rng.integers(n_ones)]
            i2 = ones[rng.integers(n_ones)]
            if i1 == i2:
                continue
            j1 = zeros[rng.integers(n_zeros)]
            j2 = zeros[rng.integers(n_zeros)]
            if j1 == j2:
                continue
            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c = c5_exact(h)
            delta = new_c - cur
            accepted = delta < 0
            if not accepted and T > T_min:
                log_r = np.log(rng.random() + 1e-30)
                if log_r > -delta / T:
                    accepted = True
            if accepted:
                cur = new_c
                if cur < best:
                    best = cur
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones.remove(i1)
                ones.remove(i2)
                zeros.remove(j1)
                zeros.remove(j2)
                ones.extend([j1, j2])
                zeros.extend([i1, i2])
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                patience += 1
        else:
            i = ones[rng.integers(n_ones)]
            j = zeros[rng.integers(n_zeros)]
            h[i] = 0.0
            h[j] = 1.0
            new_c = c5_exact(h)
            delta = new_c - cur
            accepted = delta < 0
            if not accepted and T > T_min:
                log_r = np.log(rng.random() + 1e-30)
                if log_r > -delta / T:
                    accepted = True
            if accepted:
                cur = new_c
                if cur < best:
                    best = cur
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones.remove(i)
                zeros.remove(j)
                zeros.append(i)
                ones.append(j)
            else:
                h[i] = 1.0
                h[j] = 0.0
                patience += 1
        T = max(T_min, T * 0.99992)
    return best_h, best

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def make_binary(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[sorted_idx[:target]] = 1.0
    return hb
