def threshold_exact(h, rng):
    """Convert h to a binary vector with exactly n/2 ones."""
    n = len(h)
    target = n // 2
    # Rank by value, pick top target as 1s
    order = rng.permutation(n)
    # Actually, deterministic: pick top h values
    order = np.argsort(-h)
    bits = np.zeros(n)
    bits[order[:target]] = 1.0
    return bits

def transport_polish(h, deadline, rng):

    n = len(h)

    target = n / 2.0

    best_h = h.copy()
    best_c5 = cur
    stall = 0

            break
        if stall > 300:
            break
        
        # Pick random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            d = np.zeros(n)
            d[i] = 1.0

            else:

        else:

import time


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def soft_max_direction(h, gamma):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    M = float(np.max(corr))
    w = np.exp(gamma * (corr - M))
    sw = float(np.sum(w))
    if sw < 1e-30:
        return None
    p = w / sw
    direction = np.zeros(n)
    nlag = 2 * n - 1
    for k in range(nlag):
        sk = k - (n - 1)
        pk = p[k]
        if pk < 1e-15:
            continue
        if sk >= 0:
            direction[:n - sk] += -dx * pk * (1.0 - h[sk:n])
            direction[sk:] += dx * pk * h[:n - sk]
        else:
            g = -sk
            direction[g:] += -dx * pk * (1.0 - h[:n - g])
            direction[:n - g] += dx * pk * h[g:]
    direction -= direction.mean()
    norm = float(np.sum(np.abs(direction)))
    if norm < 1e-14:
        return None
    return direction / norm


def line_minimize(h, d, dx):
    n = len(h)
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    alpha_lo = -1.0
    alpha_hi = 1.0
    if np.any(pos):
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max(-h[neg] / d[neg])))
    alpha_lo = max(alpha_lo, -1.0)
    alpha_hi = min(alpha_hi, 1.0)
    if 0.0 > alpha_lo:
        alpha_lo = 0.0
    if 0.0 < alpha_hi:
        alpha_hi = 0.0
    if alpha_hi - alpha_lo < 1e-12:
        return 0.0, float(np.max(c0) * dx)
    n_grid = 80
    grid = np.linspace(alpha_lo, alpha_hi, n_grid)
    if alpha_lo < 0.0 < alpha_hi:
        grid = np.sort(np.append(grid, 0.0))
    vals = c0[None, :] + grid[:, None] * L[None, :] + (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_alpha = grid[best_i]
    best_val = float(obj[best_i])
    if 0 < best_i < len(grid) - 1:
        a = grid[best_i - 1]
        b = grid[best_i + 1]
    else:
        a, b = alpha_lo, alpha_hi
    if b - a < 1e-13:
        return best_alpha, best_val
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    f1 = float(np.max(c0 + a * L + a * a * Q) * dx)
    f2 = float(np.max(c0 + b * L + b * b * Q) * dx)
    for _ in range(60):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = a + gr * (b - a)
            f2 = float(np.max(c0 + b * L + b * b * Q) * dx)
        else:
            a = b - gr * (b - a)
            f1 = float(np.max(c0 + a * L + a * a * Q) * dx)
    mid = (a + b) / 2.0
    mid_val = float(np.max(c0 + mid * L + mid * mid * Q) * dx)
    if mid_val < best_val:
        best_alpha, best_val = mid, mid_val
    return best_alpha, best_val


def soft_descent(h, deadline, gammas, max_iters_per_gamma):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    for gamma in gammas:
        stall = 0
        for it in range(max_iters_per_gamma):
            if time.time() > deadline:
                break
            if stall > max_iters_per_gamma // 2:
                break
            d = soft_max_direction(h, gamma)
            if d is None:
                break
            alpha, new_c5 = line_minimize(h, d, dx)
            if alpha != 0.0:
                h_new = h + alpha * d
                h_new = project(h_new, target)
                actual_c5 = exact_c5(h_new)
                if actual_c5 < cur - 1e-14:
                    h = h_new
                    cur = actual_c5
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 5
    return best_h, best_c5


def transport_polish(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    stall = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            stall += 1
            continue
        if abs(h[i] - h[j]) < 1e-6:
            stall += 1
            continue
        d = np.zeros(n)
        if h[i] > h[j]:
            d[i] = -1.0
            d[j] = 1.0
        else:
            d[i] = 1.0
            d[j] = -1.0
        alpha, new_c5 = line_minimize(h, d, dx)
        if alpha != 0.0:
            h_new = h + alpha * d
            h_new = project(h_new, target)
            actual_c5 = exact_c5(h_new)
            if actual_c5 < cur - 1e-14:
                h = h_new
                cur = actual_c5
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def grad_pair_polish(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    stall = 0
    for it in range(10000):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        d_soft = soft_max_direction(h, 80.0)
        if d_soft is None:
            break
        i_pos = int(np.argmax(d_soft))
        i_neg = int(np.argmin(d_soft))
        if h[i_pos] == h[i_neg]:
            stall += 1
            continue
        d = np.zeros(n)
        d[i_pos] = 1.0
        d[i_neg] = -1.0
        alpha, new_c5 = line_minimize(h, d, dx)
        if alpha != 0.0:
            h_new = h + alpha * d
            h_new = project(h_new, target)
            actual_c5 = exact_c5(h_new)
            if actual_c5 < cur - 1e-14:
                h = h_new
                cur = actual_c5
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 50 and stall % 50 == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and abs(h[i] - h[j]) > 1e-4:
                d2 = np.zeros(n)
                if h[i] > h[j]:
                    d2[i] = -1.0
                    d2[j] = 1.0
                else:
                    d2[i] = 1.0
                    d2[j] = -1.0
                a2, c2 = line_minimize(h, d2, dx)
                if a2 != 0.0:
                    h2_new = h + a2 * d2
                    h2_new = project(h2_new, target)
                    c2_act = exact_c5(h2_new)
                    if c2_act < cur - 1e-14:
                        h = h2_new
                        cur = c2_act
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                        stall = 0
    return best_h, best_c5


def binarize_push(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    stall = 0
    for it in range(10000):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        interior = (h > 0.02) & (h < 0.98)
        cnt = int(np.sum(interior))
        if cnt == 0:
            break
        idx = np.where(interior)[0]
        i = int(idx[int(rng.integers(cnt))])
        j = int(rng.integers(n))
        trial = 0
        while j == i and trial < 10:
            j = int(rng.integers(n))
            trial += 1
        if j == i:
            continue
        d = np.zeros(n)
        if h[i] > 0.5:
            d[i] = 1.0
            d[j] = -1.0
        else:
            d[i] = -1.0
            d[j] = 1.0
        alpha, new_c5 = line_minimize(h, d, dx)
        if alpha != 0.0:
            h_new = h + alpha * d
            h_new = project(h_new, target)
            actual_c5 = exact_c5(h_new)
            if actual_c5 < cur - 1e-14:
                h = h_new
                cur = actual_c5
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)


def slsqp_polish(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    curr_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    K = min(20, 2 * n - 1)
    top_k = np.argsort(-corr)[:K]
    t_init = curr_c5 + 5e-5

    def obj_fn(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fn(x):
        hx = x[:n]
        c = np.correlate(hx, 1.0 - hx, mode='full') * dx
        return x[n] - c[top_k]

    def ineq_jac(x):
        hx = x[:n]
        G = np.zeros((K, n))
        for ii, kk in enumerate(top_k):
            s = kk - (n - 1)
            if s >= 0:
                G[ii, s:] = (1.0 - hx[:n - s])
                G[ii, :n - s] += (-hx[s:])
            else:
                g = -s
                G[ii, g:] = (1.0 - hx[:n - g])
                G[ii, :n - g] += (-hx[g:])
        G *= dx
        J = np.zeros((K, n + 1))
        J[:, :n] = -G
        J[:, n] = 1.0
        return J

    def eq_fn(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    x0 = np.concatenate([h, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
    ]
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-12})
        if res.success:
            h_new = project(res.x[:n], target)
            c5_new = exact_c5(h_new)
            if c5_new < best_c5:
                return h_new, c5_new
    except Exception:
        pass
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 3.0 * np.arange(n0) / n0)

    h0 = project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = exact_c5(h0)
    best_n = n0

    # Phase 1: SLSQP initial polish (10% time)
    t1 = t0 + budget_s * 0.10
    h1, c5_1 = slsqp_polish(best_h, t1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Soft-max descent (20% time)
    t2 = t0 + budget_s * 0.30
    gammas = [10, 30, 100, 300, 1000, 3000, 10000]
    h2, c5_2 = soft_descent(best_h, t2, gammas, max_iters_per_gamma=60)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: Transport polish (10% time)
    t3 = t0 + budget_s * 0.40
    h3, c5_3 = transport_polish(best_h, t3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Gradient pair polish (10% time)
    t4 = t0 + budget_s * 0.50
    h4, c5_4 = grad_pair_polish(best_h, t4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        best_n = len(h4)

    # Phase 5: Binarization push (8% time)
    t5 = t0 + budget_s * 0.58
    h5, c5_5 = binarize_push(best_h, t5, rng)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
        best_n = len(h5)

    # Phase 6: Another round of soft descent with higher gammas (12% time)
    t6 = t0 + budget_s * 0.70
    gammas2 = [500, 2000, 5000, 10000, 30000]
    h6, c5_6 = soft_descent(best_h, t6, gammas2, max_iters_per_gamma=50)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)

    # Phase 7: Multi-resolution search (12% time)
    t7 = t0 + budget_s * 0.82
    for dn in [-6, -4, -2, 2, 4, 6, -8, 8]:
        if time.time() > t7:
            break
        n_try = best_n + dn
        if n_try < 60 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = resample(best_h, n_try)
        sub_dl = min(t7, time.time() + budget_s * 0.012)
        h_r, c5_r = soft_descent(h_r, sub_dl, [100, 500, 2000], max_iters_per_gamma=25)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final SLSQP + transport polish (8% time)
    t8 = t0 + budget_s * 0.90
    h8, c5_8 = slsqp_polish(best_h, min(t8, time.time() + budget_s * 0.04))
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
        best_n = len(h8)

    # Phase 9: Final soft descent polish (8% time)
    t9 = deadline - 5.0
    h9, c5_9 = soft_descent(best_h, t9, [2000, 5000, 20000], max_iters_per_gamma=40)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()
        best_n = len(h9)

    # Final transport polish
    if time.time() < deadline - 3.0:
        h10, c5_10 = transport_polish(best_h, deadline - 1.0, rng)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()
            best_n = len(h10)

    # Ensure validity
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
