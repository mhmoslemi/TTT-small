    n = len(v)
    if target <= 0:

    if target >= n:

            x = np.clip(x, 0.0, 1.0)

    """Compute gradient of the full softmax-smoothed max correlation.
    Returns (gradient, smooth_value, raw_correlations)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx  # scaled values
    # Stable softmax
    vmax = np.max(v)
    exp_v = np.exp(alpha * (v - vmax))
    p = exp_v / np.sum(exp_v)
    smooth_val = vmax + np.log(np.sum(exp_v)) / alpha
    # Gradient: sum_r p_r * dx * grad_C_r
    grad = np.zeros(n)
    # For efficiency, iterate over significant p_r

    n_corr = len(corr)  # 2n-1
    for r in range(n_corr):
        if p[r] < 1e-15:

import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_single_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += (1.0 - h[:m])
            g[:m] -= h[k:]
    else:
        q = -k
        m = n - q
        if m > 0:
            g[:m] += (1.0 - h[q:])
            g[q:] -= h[:m]
    return g

def _full_softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    vmax = float(np.max(v))
    logits = alpha * (v - vmax)
    # Clip for numerical stability
    logits = np.clip(logits, -700, 700)
    exp_l = np.exp(logits)
    p = exp_l / float(np.sum(exp_l))
    smooth_val = vmax + float(np.log(float(np.sum(exp_l)))) / alpha
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for r in range(n_corr):
        pr = p[r]
        if pr < 1e-14:
            continue
        k = r - (n - 1)
        g_k = _grad_single_lag(h, k)
        grad += pr * g_k
    grad *= dx
    return grad, smooth_val, v, p

def _topK_grad(h, alpha, K=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    top_idx = np.argsort(-v)[:K]
    top_vals = v[top_idx]
    logits = alpha * (top_vals - top_vals[0])
    logits = np.clip(logits, -700, 700)
    exp_l = np.exp(logits)
    w = exp_l / float(np.sum(exp_l))
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _grad_single_lag(h, k)
    grad *= dx
    return grad, v

def _line_search_exact(h, direction, target, f0, max_t=1.0):
    """Find the best t in [0, max_t] that minimizes C5(h + t*d) after projection."""
    n = len(h)

    pos = direction > 1e-14
    neg = direction < -1e-14
    t_hi = max_t
    if np.any(pos):

    if np.any(neg):

        f_try = _exact_c5(h_try)
        if f_try < best_f:
            best_f = f_try
            best_t = t

    if best_t > 0 and best_t < t_hi:
        idx = int(round(best_t / t_hi * (n_grid - 1)))
        a = grid[max(0, idx - 1)]
        b = grid[min(n_grid - 1, idx + 1)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = _exact_c5(_project(h + c1 * direction, target))

            else:

                c2 = a + gr * (b - a)
                f2 = _exact_c5(_project(h + c2 * direction, target))
        opt_t = (a + b) / 2.0
        opt_f = _exact_c5(_project(h + opt_t * direction, target))
        if opt_f < best_f:
            best_t, best_f = opt_t, opt_f
    
    return best_t, best_f
