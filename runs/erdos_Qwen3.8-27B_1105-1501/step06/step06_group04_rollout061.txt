Now for the SLSQP approach:

**Direct Minimax:**
Minimize M
Subject to:
- M - C_r(h) >= 0 for r = 0, ..., 2n-2 (i.e., M >= C_r(h))
- sum(h) = n/2
- 0 <= h_i <= 1

Variables: (M, h_0, ..., h_{n-1}) of length n+1

Objective: y[0]
Inequality constraints: y[0] - C_r(y[1:]) >= 0 (scipy uses >= 0 for 'ineq')
Equality constraint: sum(y[1:]) - n/2 = 0

For SLSQP, I need:
- fun(y) = y[0]
- jac(y) = [1, 0, ..., 0]
- cons: list of dicts with 'fun' and 'jac'

The inequality constraint function returns a vector of length 2n-1:
ineq(y) = y[0] - dx * correlate(y[1:], 1-y[1:], 'full')

The Jacobian of ineq:
Row r: [1, -dx * dC_r/dh[0], -dx * dC_r/dh[1], ..., -dx * dC_r/dh[n-1]]

The equality constraint:
eq(y) = sum(y[1:]) - n/2
eq_jac(y) = [0, 1, 1, ..., 1]

**Log-Sum-Exp:**
Minimize (1/beta) * log(sum_r exp(beta * C_r(h)))
Subject to:
- sum(h) = n/2
- 0 <= h_i <= 1

Variables: (h_0, ..., h_{n-1}) of length n
Objective: (1/beta) * log(sum_r exp(beta * dx * corr_r))
Gradient: dx * sum_r p_r * dC_r/dh where p_r = exp(beta * C_r) / sum(exp(beta * C_r))

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_row(h, d):
    """Gradient of C_d (unscaled) w.r.t. h, where d is the offset."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        g = -d
        m = n - g
        if m > 0:
            g[:m] += (1.0 - h[g:])
            g[g:] += -h[:m]
    return g

def _all_grads(h):
    """All gradient rows, shape (2n-1, n), unscaled."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for idx in range(2*n-1):
        d = idx - (n-1)
        G[idx] = _grad_row(h, d)
    return G

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _sse_slsqp_solve(h, beta, target, maxiter=200):
    """SLSQP with log-sum-exp objective."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    
    def fun(x):
        hx = _project(x, target)
        C = _corr_scaled(hx)
        # Numerically stable LSE
        Cmax = C.max()
        val = (Cmax + np.log(np.sum(np.exp(beta * (C - Cmax))))) / beta
        return val
    
    def jac(x):
        hx = _project(x, target)
        C = _corr_scaled(hx)
        Cmax = C.max()
        z = beta * (C - Cmax)
        p = np.exp(z)
        p /= p.sum()
        G = _all_grads(hx)
        grad = dx * (p @ G)
        # Wait, G is already unscaled, and C = dx * corr, so dC/dh = dx * dcorr/dh
        # d(fun)/dh = (1/beta) * sum_r beta * p_r * dC_r/dh = sum_r p_r * dC_r/dh
        # But C_r = dx * corr_r, so dC_r/dh = dx * dcorr_r/dh
        # d(fun)/dh = sum_r p_r * dx * dcorr_r/dh = dx * sum_r p_r * G_r
        return grad
    
    def eq_fun(x):
        return np.array([np.sum(x) - target])
    
    def eq_jac(x):
        return np.ones((1, n))
    
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(fun, x0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, _compute_c5(h)

def _minimax_slsqp_solve(h, maxiter=100):
    """SLSQP with direct minimax formulation."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    target = n / 2.0
    c5_init = _compute_c5(h)
    
    def fun(y):
        return y[0]
    
    def jac(y):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def ineq(y):
        hx = y[1:]
        C = np.correlate(hx, 1.0 - hx, mode='full') * dx
        return y[0] - C
    
    def ineq_jac(y):
        hx = y[1:]
        G = _all_grads(hx)
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * G
        return J
    
    def eq_fun(y):
        return np.array([np.sum(y[1:]) - target])
    
    def eq_jac(y):
        J = np.zeros((1, n + 1))
        J[0, 1:] = 1.0
        return J
    
    y0 = np.zeros(n + 1)
    y0[0] = c5_init + 0.05  # Start with some slack
    y0[1:] = h
    
    bounds = [(0.0, c5_init + 0.1)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    try:
        res = minimize(fun, y0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = _project(res.x[1:], target)
            c5_new = _compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, _compute_c5(h)

def _clip_dir(h, d, radius):
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        max_t = np.min((1.0 - h[pos]) / d[pos])
        d[pos] = np.minimum(d[pos], max_t * d[pos])
    if np.any(neg):
        max_t = np.min(h[neg] / (-d[neg]))
        d[neg] = np.maximum(d[neg], -max_t * (-d[neg]))
    max_abs = np.max(np.abs(d))
    if max_abs > radius:
        d *= radius / max_abs
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, rng, max_iters=3000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)

    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.0001
    cooling = 0.9997
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    patience = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 2000:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            ones_list = np.where(h > 0.5)[0]
            zeros_list = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, _compute_c5(best_h)

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_row(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        gg = -d
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] += -h[:m]
    return g

def _all_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        d = idx - (n - 1)
        G[idx] = _grad_row(h, d)
    return G

def _weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        d = idx - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += w * (1.0 - h[:m])
                grad[:m] += -w * h[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                grad[:m] += w * (1.0 - h[gg:])
                grad[gg:] += -w * h[:m]
    return grad * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _clip_dir(h, d, radius):
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        max_t = np.min((1.0 - h[pos]) / d[pos])
        d[pos] = np.minimum(d[pos], max_t * d[pos])
    if np.any(neg):
        max_t = np.min(h[neg] / (-d[neg]))
        d[neg] = np.maximum(d[neg], -max_t * (-d[neg]))
    max_abs = np.max(np.abs(d))
    if max_abs > radius:
        d *= radius / max_abs
    return d

def _lse_slsqp(h, beta, target, maxiter=200):
    n = len(h)
    dx = 2.0 / n

    def fun(x):
        hx = _project(x, target)
        C = _corr_scaled(hx)
        Cmax = float(C.max())
        z = beta * (C - Cmax)
        val = Cmax + np.log(np.sum(np.exp(z))) / beta
        return val

    def jac(x):
        hx = _project(x, target)
        C = _corr_scaled(hx)
        Cmax = float(C.max())
        z = beta * (C - Cmax)
        p = np.exp(z)
        p_sum = float(p.sum())
        if p_sum < 1e-300:
            p = np.zeros(len(C))
            p[int(np.argmax(C))] = 1.0
        else:
            p /= p_sum
        return _weighted_grad(hx, p)

    def eq_fun(x):
        return np.array([np.sum(x) - target])

    def eq_jac(x):
        return np.ones((1, n))

    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(fun, x0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            h_new = _project(res.x, target)
            return h_new, _compute_c5(h_new)
    except Exception:
        pass
    return h, _compute_c5(h)

def _resample(h, n_new):

    if n_new == n:
        return h.copy()

        best_h = h8.copy()
    if time.time() < d8:
        h8b, c5_8b = _line_polish(best_h, d8, top_k=6, max_rounds=200)
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()
    if time.time() < d8:
        h8c, c5_8c = _lse_slsqp(best_h, 3000, len(best_h) / 2.0, maxiter=100)
        if c5_8c < best_c5:
            best_c5 = c5_8c
            best_h = h8c.copy()
        if time.time() < d8:
            h8d, c5_8d = _line_polish(best_h, d8, top_k=6, max_rounds=100)
            if c5_8d < best_c5:
                best_c5 = c5_8d
                best_h = h8d.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0_fb = np.array(initial_h_values, dtype=float)
            h0_fb = _project(h0_fb, len(h0_fb) / 2.0)
            return h0_fb, _compute_c5(h0_fb), len(h0_fb)
        except NameError:
            h_fb = _project(0.5 * np.ones(122), 61.0)
            return h_fb, _compute_c5(h_fb), 122
    return best_h, final_c5, len(best_h)
