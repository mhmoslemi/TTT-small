def _c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

def _line_minimize(h, d, dx):
    n = len(h)
    # Compute L and Q
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    c0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Feasible range
    pos = d > 1e-14
    neg = d < -1e-14
    a_lo = -1e15
    a_hi = 1e15
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.max(h[neg] / (-d[neg]))))
    
    # Clip to reasonable range
    a_lo = max(a_lo, -2.0)
    a_hi = min(a_hi, 2.0)
    if a_hi - a_lo < 1e-14:
        return 0.0, _c5(h)
    
    # Include 0 in the range
    if a_lo > 0: a_lo = 0.0
    if a_hi < 0: a_hi = 0.0
    
    # Grid evaluation
    n_grid = 64
    grid = np.linspace(a_lo, a_hi, n_grid)
    # Add 0 if not already there
    if 0.0 < a_lo or 0.0 > a_hi:
        grid = np.concatenate([grid, [0.0]])
    
    # Evaluate phi at all grid points
    # phi(alpha) = dx * max_k (c0[k] + alpha*L[k] + alpha^2*Q[k])
    # Vectorized: for each alpha, compute all lags
    # Shape: (n_grid, 2n-1)
    vals = c0[None, :] + grid[:, None] * L[None, :] + grid[None, :]**2 * Q[None, :]
    phi = np.max(vals, axis=1) * dx
    
    best_i = int(np.argmin(phi))
    best_alpha = grid[best_i]
    best_val = phi[best_i]
    
    # Refine with golden section in the best interval
    if 0 < best_i < n_grid - 1:
        lo = grid[best_i - 1]
        hi = grid[best_i + 1]
    else:
        lo, hi = a_lo, a_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    dpt = lo + gr * (hi - lo)
    f_c = float(np.max(c0 + c*L + c*c*Q)) * dx
    f_d = float(np.max(c0 + dpt*L + dpt*dpt*Q)) * dx
    
    for _ in range(60):
        if hi - lo < 1e-13:
            break
        if f_c < f_d:
            hi = dpt
            dpt = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            f_c = float(np.max(c0 + c*L + c*c*Q)) * dx
        else:
            lo = c
            c = dpt
            f_c = f_d
            dpt = lo + gr * (hi - lo)
            f_d = float(np.max(c0 + dpt*L + dpt*dpt*Q)) * dx
    
    mid = (lo + hi) / 2.0
    f_mid = float(np.max(c0 + mid*L + mid*mid*Q)) * dx
    if f_mid < best_val:
        best_alpha, best_val = mid, f_mid
    
    return best_alpha, best_val

        else:

            d[j] = 1.0
            alpha, new_c5 = _line_minimize(h, d, dx)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_threshold(h, rng):
    n = len(h)
    target = n / 2
    v = h.copy()
    order = np.argsort(-v)
    h_b = np.zeros(n)
    for i in range(target):
        h_b[order[i]] = 1.0
    out = h_b.copy()
    inter = rng.random(n)
    order2 = np.argsort(inter)
    for i in range(target):
        out[order2[i]] = 1.0
    best = h_b
    bc = _c5(h_b)
    oc = _c5(out)
    if oc < bc:
        best = out
    return best

def _quick_optimize(h, deadline, rng):
    n = len(h)
    h = h.copy()
    cur_c5 = _c5(h)
    for g in [50, 500, 5000]:
        if time.time() > deadline:
            break
        h, cur_c5 = _soft_anneal(h, g, max_iters=30, deadline=deadline, rng=rng)
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except NameError:
        n = 122
        x = np.arange(n)
        h0 = 0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * x / n) + 0.1 * np.sin(2.0 * np.pi * 7.0 * x / n)
    
    h = _project(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = _c5(h)

    d1 = t0 + budget_s * 0.45
    gammas_phase1 = [8, 20, 50, 150, 500, 2000, 8000, 30000]
    for g in gammas_phase1:
        if time.time() > d1:
            break
        h_c, c5_c = _soft_anneal(best_h, g, max_iters=60, deadline=d1, rng=rng)
        if c5_c < best_c5 - 1e-13:
            best_c5 = c5_c
            best_h = h_c.copy()

    d2 = t0 + budget_s * 0.65
    h_t, c5_t = _transport_polish(best_h, d2, rng, max_iter=10000)
    if c5_t < best_c5 - 1e-13:
        best_c5 = c5_t
        best_h = h_t.copy()

    d3 = t0 + budget_s * 0.80
    h_c2, c5_c2 = _soft_anneal(best_h, 1000, max_iters=80, deadline=d3, rng=rng)
    if c5_c2 < best_c5 - 1e-13:
        best_c5 = c5_c2
        best_h = h_c2.copy()
    h_t2, c5_t2 = _transport_polish(best_h, d3, rng, max_iter=5000)
    if c5_t2 < best_c5 - 1e-13:
        best_c5 = c5_t2
        best_h = h_t2.copy()

    d4 = t0 + budget_s * 0.90
    offsets = [0, -2, 2, -4, 4, -6, 6]
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n + dn
        if n_try < 40 or n_try > 300 or n_try % 2 != 0:
            continue
        if n_try == n:
            continue
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _quick_optimize(h_r, d4, rng)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            n = n_try

    d5 = t0 + budget_s * 0.95
    h_bin = _binary_threshold(best_h, rng)
    c5_bin = _c5(h_bin)
    if c5_bin < best_c5:
        h_bin_opt, c5_bin_opt = _soft_anneal(h_bin, 100, max_iters=40, deadline=d5, rng=rng)
        if c5_bin_opt < best_c5:
            best_c5 = c5_bin_opt
            best_h = h_bin_opt.copy()

    d6 = deadline - 5.0
    for g in [200, 2000, 20000, 100000]:
        if time.time() > d6:
            break
        h_f, c5_f = _soft_anneal(best_h, g, max_iters=40, deadline=d6, rng=rng)
        if c5_f < best_c5 - 1e-13:
            best_c5 = c5_f
            best_h = h_f.copy()

    if time.time() < deadline - 10.0:
        h_tp, c5_tp = _transport_polish(best_h, deadline - 3.0, rng, max_iter=5000)
        if c5_tp < best_c5 - 1e-13:
            best_c5 = c5_tp
            best_h = h_tp.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
