Thus c = -corr(d,d): c = -2 at lag 0, +1 at lags (j-i) and (i-j), 0 elsewhere.

So I can compute c without np.correlate! And I can compute g more efficiently.

def _pair_line_search_opt(h, i, j, dx, target):

    nm1 = n - 1  # offset for full correlation
    
    S = np.correlate(h, 1.0 - h, mode='full')

    best_c5 = float(np.max(S)) * dx

            t_max = min(1.0 - h[i], h[j])
        else:
            t_max = min(h[i], 1.0 - h[j])
        if t_max < 1e-14:
            continue
        
        # Compute g = corr(d, 1-h) - corr(h, d) where d[i]=sign, d[j]=-sign
        # corr(d,1-h)[r] = sign*(1-h[i+k]) - sign*(1-h[j+k]) where k=r-nm1
        # = sign * (h[j+k] - h[i+k])
        # corr(h,d)[r] = sign*h[i-k] - sign*h[j-k]
        # g[r] = sign * (h[j+k] - h[i+k] - h[i-k] + h[j-k])

        # h[j+k] is valid when 0 <= j+k < n, i.e., -j <= k < n-j
        # h[i+k] is valid when 0 <= i+k < n, i.e., -i <= k < n-i
        # h[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 < k <= i  (i.e., -(n-1-i) < k <= i)
        # h[j-k] is valid when 0 <= j-k < n, i.e., j-n+1 < k <= j
        
        k = np.arange(len(S)) - nm1  # lags from -(n-1) to (n-1)
        
        # h[j+k]: valid where 0 <= j+k < n
        valid_jk = (k >= -j) & (k < n - j)
        h_jk = np.zeros(len(S))
        h_jk[valid_jk] = h[j + k[valid_jk]]
        
        # h[i+k]: valid where 0 <= i+k < n
        valid_ik = (k >= -i) & (k < n - i)
        h_ik = np.zeros(len(S))
        h_ik[valid_ik] = h[i + k[valid_ik]]
        
        # h[i-k]: valid where 0 <= i-k < n
        valid_ik2 = (k >= i - (n-1)) & (k <= i)
        h_ik2 = np.zeros(len(S))
        h_ik2[valid_ik2] = h[i - k[valid_ik2]]
        
        # h[j-k]: valid where 0 <= j-k < n
        valid_jk2 = (k >= j - (n-1)) & (k <= j)
        h_jk2 = np.zeros(len(S))
        h_jk2[valid_jk2] = h[j - k[valid_jk2]]
        
        g = sign * (h_jk - h_ik - h_ik2 + h_jk2)

        c = np.zeros(len(S))
        c[nm1] = -2.0
        r_ji = nm1 + (j - i)
        r_ij = nm1 + (i - j)
        if 0 <= r_ji < len(S):
            c[r_ji] += 1.0
        if 0 <= r_ij < len(S):
            c[r_ij] += 1.0
        
        # Line search
        n_grid = 50
        grid = np.linspace(0.0, t_max, n_grid)
        # Objective: max(S + t*g + t²*c) * dx
        vals = np.max(S[None,:] + grid[:,None]*g[None,:] + (grid[:,None]**2)*c[None,:], axis=1) * dx
        best_i = int(np.argmin(vals))
        best_t = grid[best_i]
        
        # Golden section
        lo = grid[max(0, best_i-1)]
        hi = grid[min(n_grid-1, best_i+1)]
        if hi - lo > 1e-12:
            gra = (np.sqrt(5)-1)/2
            c1 = hi - gra*(hi-lo)
            c2 = lo + gra*(hi-lo)
            f1 = float(np.max(S + c1*g + c1*c1*c)) * dx
            f2 = float(np.max(S + c2*g + c2*c2*c)) * dx
            for _ in range(35):
                if hi - lo < 1e-12:
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gra*(hi-lo)
                    f1 = float(np.max(S + c1*g + c1*c1*c)) * dx
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gra*(hi-lo)
                    f2 = float(np.max(S + c2*g + c2*c2*c)) * dx
            best_t = (lo+hi)/2
            best_t = max(0.0, min(best_t, t_max))
        
        # Evaluate
        h_try = h.copy()
        h_try[i] += sign * best_t
        h_try[j] -= sign * best_t
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return best_h, best_c5

S_k = sum_p h[p] * (1-h[p+k]) (over valid p)

∂S_k/∂h[q] = sum_p [δ_{p,q}(1-h[p+k]) + h[p] * (-δ_{p+k,q})]

So:

- -h[q-k] is the contribution from h[q] appearing as the second factor (h[p+k] where p+k=q, i.e., p=q-k)
