import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.maximum(np.correlate(h, 1.0 - h, mode='full') * dx, 1e-15)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lse_obj(h, tau):
    S = _corr_vals(h)
    # log-sum-exp for numerical stability
    S_max = np.max(S)
    return tau * (S_max + np.log(np.sum(np.exp((S - S_max) / tau))))

def _lse_grad(h, tau):
    S = _corr_vals(h)
    S_max = np.max(S)
    logits = (S - S_max) / tau
    exp_logits = np.exp(logits)
    Z = float(np.sum(exp_logits))
    w = exp_logits / Z  # softmax weights
    return _grad_S_weighted(h, w)

def _grad_S_weighted(h, w):
    """Compute Σ_q w_q * ∇S_q / ∇h"""
    n = len(h)
    dx = 2.0 / n
    oh = 1.0 - h
    grad = np.zeros(n)
    
    # s = 0: q = n-1
    w0 = w[n-1] * dx
    grad += w0 * (1.0 - 2.0 * h)
    
    for s in range(1, n):
        m = n - s
        ws_pos = w[n-1+s] * dx
        if ws_pos > 0:
            grad[s:] += ws_pos * oh[:m]
            grad[:m] -= ws_pos * h[s:]
        
        ws_neg = w[n-1-s] * dx
        if ws_neg > 0:
            grad[:m] += ws_neg * oh[s:]
            grad[s:] -= ws_neg * h[:m]
    
    return grad

def _slsqp_lse(h, tau, deadline, maxiter=200):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return _lse_obj(x, tau)
    
    def jac(x):
        return _lse_grad(x, tau)

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success:
            return _project(res.x, target)
    except Exception:
        pass
    return h

def _slsqp_pnorm(h, p, deadline, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        S = _corr_vals(x)
        return float(np.sum(S ** p)) ** (1.0 / p)
    
    def jac(x):
        S = _corr_vals(x)
        denom = float(np.sum(S ** p))
        if denom < 1e-30:
            return np.zeros(n)
        coef = (1.0 / p) * denom ** (1.0 / p - 1.0) * p
        # = denom^(1/p - 1)
        w = S ** (p-1) / (denom ** (1 - 1.0/p))
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            return np.zeros(n)
        w_norm = w / w_sum
        # Actually let me just compute properly
        sp = S ** (p-1)
        total_sp = float(np.sum(S ** p))
        coef2 = sp / (total_sp ** (1.0 - 1.0/p))
        w_mat = coef2 / float(np.sum(coef2))
        return _grad_S_weighted(x, w_mat)
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success:
            return _project(res.x, target)
    except Exception:
        pass
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_from_h(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def _binary_local_search(h_bin, deadline, max_iter=30000, seed=0):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    best_c5 = _exact_c5(h)
    current_c5 = best_c5
    rng = np.random.default_rng(seed)
    patience = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = _exact_c5(h_new)
        
        if c5_new < current_c5 - 1e-14:
            h = h_new
            current_c5 = c5_new
            patience = 0
            if c5_new < best_c5:
                best_c5 = c5_new
        else:
            patience += 1
            # Accept with some probability (simulated annealing)
            if patience > 2000:
                break
            delta = c5_new - current_c5
            if rng.random() < np.exp(-max(0, delta) / 0.001):
                h = h_new
                current_c5 = c5_new
    
    return h, best_c5

def _diverse_start(n, rng):
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    # Add second harmonic
    freq2 = rng.integers(2, 12)
    phase2 = rng.uniform(0, 2 * np.pi)
    h += 0.15 * np.sin(2 * np.pi * freq2 * x / n + phase2)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)
