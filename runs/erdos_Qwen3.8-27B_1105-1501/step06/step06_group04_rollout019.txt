def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

```python
def _lse_gradient_fast(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    S_max = S.max()
    log_w = np.clip(beta * (S - S_max), -50, 50)
    w = np.exp(log_w)
    w /= w.sum()
    
    # Build the gradient using matrix multiplication
    # dS_r/dh[j] = dx * [(1-h[j-lag_r]) * 1_{0≤j-lag_r<n} - h[j+lag_r] * 1_{0≤j+lag_r<n}]
    
    # Create matrix M of shape (2n-1, n) where M[r, j] = dS_r/dh[j] / dx
    lags = np.arange(-(n-1), n)
    j = np.arange(n)
    
    # For term 1: (1-h[j-lag_r]) * valid(j-lag_r)
    # j - lag_r = j - (r - (n-1)) = j - r + (n-1)
    # Create 2D array: rows are r (0 to 2n-2), cols are j (0 to n-1)
    k1 = j[None, :] - lags[:, None]  # shape (2n-1, n), k1[r,j] = j - lag_r
    mask1 = (k1 >= 0) & (k1 < n)
    term1 = np.where(mask1, 1.0 - h[k1], 0.0)
    
    # For term 2: -h[j+lag_r] * valid(j+lag_r)
    k2 = j[None, :] + lags[:, None]  # shape (2n-1, n), k2[r,j] = j + lag_r
    mask2 = (k2 >= 0) & (k2 < n)
    term2 = np.where(mask2, h[k2], 0.0)
    
    # grad[j] = dx * sum_r w[r] * (term1[r,j] - term2[r,j])
    grad = dx * (w @ (term1 - term2))
    return grad
```

def _lse_optimize(h, deadline, beta_start=5, beta_end=200, n_anneal=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    
    # Anneal beta
    for stage in range(n_anneal):
        if time.time() > deadline:
            break
        frac = stage / max(1, n_anneal - 1)
        beta = beta_start + (beta_end - beta_start) * frac * frac  # quadratic annealing
        
        # Inner loop at this beta
        lr = 0.02
        for inner in range(200):
            if time.time() > deadline:
                break
            grad = _lse_gradient_fast(h, beta)
            # Project gradient to sum=0
            grad -= grad.mean()
            
            # Try step
            h_new = h - lr * grad
            h_new = _project(h_new, target)
            
            new_c5 = _exact_c5(h_new)
            if new_c5 < best_c5 - 1e-15:
                h = h_new
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
                lr = min(lr * 1.1, 0.5)

    j = np.arange(n)

            if time.time() > deadline:
                break
            grad = _lse_step(h, beta, dx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            d = -grad / gnorm
            # Scale to max_r
            max_r = min(lr, 0.3)
            # Clip direction to stay in bounds
            pos = d > 0
            neg = d < 0
            if np.any(pos):
                scale = np.min((1.0 - h[pos]) / d[pos])
                max_r = min(max_r, scale * 0.99)
            if np.any(neg):
                scale = np.min(h[neg] / (-d[neg]))
                max_r = min(max_r, scale * 0.99)
            max_r = max(max_r, 1e-8)
            
            t, new_c5 = _line_search(h, d, dx, max_r)
            if new_c5 < best_c5 - 1e-16:
                h = h + t * d
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
                stall = 0
                lr = min(lr * 1.1, 1.0)
            elif new_c5 < _exact_c5(h) - 1e-16:
                h = h + t * d
                h = _project(h, target)
                stall = 0
            else:
                lr *= 0.5
                stall += 1
                if stall > 30:
                    break
    
    return best_h, best_c5

def _topk_anneal(h, deadline, k_range=(10, 50), temp_range=(5, 50)):
    """Top-k softmax gradient descent."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Try different k and temperature values
    params = []
    for k in k_range:
        for temp in temp_range:
            params.append((k, temp))
    
    rng = np.random.default_rng(123)
    rng.shuffle(params)
    
    for k, temp in params:
        if time.time() > deadline:
            break
        lr = 0.3
        stall = 0
        for _ in range(200):
            if time.time() > deadline:
                break
            grad = _topk_step(h, k, temp, dx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            d = -grad / gnorm
            max_r = lr
            pos = d > 0
            neg = d < 0
            if np.any(pos):
                max_r = min(max_r, float(np.min((1.0 - h[pos]) / d[pos])) * 0.99)
            if np.any(neg):
                max_r = min(max_r, float(np.min(h[neg] / (-d[neg]))) * 0.99)
            max_r = max(max_r, 1e-8)
            
            t, new_c5 = _line_search(h, d, dx, max_r)
            if new_c5 < best_c5 - 1e-16:
                h = h + t * d
                h = _project(h, target)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
                stall = 0
                lr = min(lr * 1.1, 0.8)
            else:
                if new_c5 < _exact_c5(h) - 1e-16:
                    h = h + t * d
                    h = _project(h, target)
                    stall = 0
                else:
                    lr *= 0.5
                    stall += 1
                    if stall > 20:
                        break
    
    return best_h, best_c5
