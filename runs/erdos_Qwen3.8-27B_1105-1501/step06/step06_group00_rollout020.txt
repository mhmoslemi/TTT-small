The efficient approach:

2. The objective at α is: f(α) = dx * max_r (raw0[r] + α*T[r] - α²*Q[r])
3. To find the minimum, I can:
   a. Find the feasible interval [α_min, α_max]
   b. Sample the interval on a grid (e.g., 50 points)
   c. Find the grid point with minimum f
   d. Refine around that point using the active set (lags that achieve the max)
   e. For the active set, find the exact minimum by considering vertices and intersections

For step (e), if the active set has size K, the minimum of max of K quadratics occurs at:
- A vertex of one of the quadratics (α = T[r]/(2*Q[r]))
- An intersection of two of the quadratics

So I need to find all vertices and pairwise intersections of the active quadratics, filter to the feasible interval, and pick the one with minimum f.

In practice, the active set is small (usually 1-5 lags), so this is fast.

                # Basin hop: smooth perturbation
                amp = 0.02 + 0.01 * rng.random()
                x = np.arange(n, dtype=float)
                pert = np.zeros(n)
                for f in [1.0, 2.0, 3.0, 5.0]:
                    pert += rng.uniform(0.3, 1.0) * np.sin(2*np.pi*f*x/n + rng.uniform(0, 2*np.pi))
                pert *= amp
                pert += amp * 0.1 * rng.normal(0, 1, n)
                h_try = _project(h + pert, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 + 0.002:
                    h = h_try
                    cur_c5 = c5_try
                stall = 0
                trust = 0.10
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=20000):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(77)
    T = 0.003
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < math.exp(-max(delta, 0.0) / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            T *= 0.999
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Optimize at n0 (30% budget)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _local_solve(h_init, d1, beta0=20.0, trust0=0.12, max_stall=100)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Binary refinement at n0
    if time.time() < d1 - 5:
        d1b = min(d1, time.time() + budget_s * 0.05)
        h_bin = _to_binary(best_h)
        h_br, c5_br = _binary_local(h_bin, d1b, max_iter=25000)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
        # Polish binary result with smooth method
        if time.time() < d1 - 2:
            h_pol = _project(h_br.copy(), n0 / 2.0)
            d1c = min(d1, time.time() + budget_s * 0.03)
            h_pol, c5_pol = _local_solve(h_pol, d1c, beta0=25.0, trust0=0.08, max_stall=40)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 2: Explore other n values (40% budget)
    d2 = t0 + budget_s * 0.70
    n_best = len(best_h)
    offsets = [-6, -4, -2, 2, 4, 6, -10, 10, 8, -8, 12, -12, 14, -14, 16, 20]
    tried_ns = set([n0])
    for dn in offsets:
        if time.time() > d2 - 3:
            break
        n_try = n_best + dn
        if n_try < 80 or n_try > 260 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.015)
        h_r, c5_r = _local_solve(h_r, sub_dl, beta0=20.0, trust0=0.10, max_stall=50)
        # Try binary at this n
        if time.time() < d2 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d2, time.time() + budget_s * 0.008)
            h_b, c5_b = _binary_local(h_b, sub_dl2, max_iter=10000)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        # Polish
        if c5_r < best_c5 + 0.002 and time.time() < d2 - 2:
            h_p = _project(h_r.copy(), n_try / 2.0)
            sub_dl3 = min(d2, time.time() + budget_s * 0.008)
            h_p, c5_p = _local_solve(h_p, sub_dl3, beta0=25.0, trust0=0.06, max_stall=30)
            if c5_p < c5_r:
                c5_r = c5_p
                h_r = h_p
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 3: Basin hopping around best (15% budget)
    d3 = t0 + budget_s * 0.85
    n_cr = len(best_h)
    scale = 0.02
    n_restarts = 0
    while time.time() < d3 and n_restarts < 80:
        n_restarts += 1
        x = np.arange(n_cr, dtype=float)
        pert = np.zeros(n_cr)
        for f in [1.0, 2.0, 3.0, 4.0, 6.0]:
            pert += rng.uniform(0.2, 1.0) * np.sin(2*np.pi*f*x/n_cr + rng.uniform(0, 2*np.pi))
        pert *= scale
        pert += scale * 0.15 * rng.normal(0, 1, n_cr)
        h_pert = _project(best_h + pert, n_cr / 2.0)
        sub_dl = min(d3, time.time() + budget_s * 0.006)
        h_t, c5_t = _local_solve(h_pert, sub_dl, beta0=20.0, trust0=0.08, max_stall=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.1)

    # Phase 4: Final polish (remaining time)
    d4 = deadline - 2.0
    n_final = len(best_h)
    # Try a few nearby n
    for dn in [-2, 2, -4, 4]:
        if time.time() > d4 - 5:
            break
        n_try = n_final + dn
        if n_try < 80 or n_try > 260:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.010)
        h_r, c5_r = _local_solve(h_r, sub_dl, beta0=30.0, trust0=0.06, max_stall=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_final = n_try
    # Final continuous optimization
    if time.time() < d4 - 1:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _local_solve(h_f, d4, beta0=30.0, trust0=0.05, max_stall=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    # Final binary try
    if time.time() < d4 - 1:
        h_b = _to_binary(best_h)
        h_b, c5_b = _binary_local(h_b, d4, max_iter=15000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
