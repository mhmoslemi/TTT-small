import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads(h):
    """Compute the (2n-1, n) gradient matrix for R = correlate(h, 1-h, 'full')."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        if d >= 0:
            m = n - d
            # R[k] = sum_{i=0}^{m-1} h[i+d] * (1-h[i])
            # dR/dh[j] for j in [d, m-1+d]: += (1-h[j-d])
            # dR/dh[j] for j in [0, m-1]: += -h[j+d]
            if d == 0:
                G[k, :] = 1.0 - 2.0 * h
            else:
                G[k, d:d+m] += (1.0 - h[:m])
                G[k, :m] += -h[d:d+m]
        else:
            r = -d
            m = n - r
            # R[k] = sum_{i=0}^{m-1} h[i] * (1-h[i+r])
            # dR/dh[j] for j in [0, m-1]: += (1-h[j+r])
            # dR/dh[j] for j in [r, m-1+r]: += -h[j-r]
            G[k, :m] += (1.0 - h[r:r+m])
            G[k, r:r+m] += -h[:m]
    return G

import numpy as np
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    # Project v onto {x: 0<=x_i<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads(h):
    """Compute (2n-1, n) gradient matrix for R[k] = correlate(h, 1-h, 'full')[k]."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        if d >= 0:
            m = n - d
            if d == 0:
                G[k, :] = 1.0 - 2.0 * h
            else:
                G[k, d:d+m] += (1.0 - h[:m])
                G[k, :m] += -h[d:d+m]
        else:
            r = -d
            m = n - r
            G[k, :m] += (1.0 - h[r:r+m])
            G[k, r:r+m] += -h[:m]
    return G

def line_search(h, d, R0, dx, n_samples=40):
    """Exact line search along direction d from base h.
    Returns (best_alpha, best_c5)."""
    n = len(h)
    # Compute T and Q
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    
    # Feasible range
    pos = d > 1e-15
    neg = d < -1e-15
    alpha_max = 1e10
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(h[neg] / (-d[neg]))))
    alpha_max = max(alpha_max, 0.0)
    if alpha_max < 1e-14:
        return 0.0, c5_exact(h)
    
    # Sample
    alphas = np.linspace(0, alpha_max, n_samples)
    best_alpha = 0.0
    best_val = float('inf')
    for a in alphas:
        val = float(np.max(R0 + a*T - a*a*Q)) * dx
        if val < best_val:
            best_val = val
            best_alpha = a
    
    # Golden section refinement around best
    # Find bracket
    for i in range(len(alphas)-1):
        if alphas[i] <= best_alpha <= alphas[i+1]:
            a_lo = alphas[max(0, i-1)]
            a_hi = alphas[min(len(alphas)-1, i+1)]
            break
    else:
        a_lo, a_hi = 0.0, alpha_max
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = a_lo, a_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1*T - c1*c1*Q)) * dx
    f2 = float(np.max(R0 + c2*T - c2*c2*Q)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1*T - c1*c1*Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2*T - c2*c2*Q)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt*T - opt*opt*Q)) * dx
    
    # Check endpoints
    f0 = float(np.max(R0)) * dx  # alpha=0
    fmax = float(np.max(R0 + alpha_max*T - alpha_max*alpha_max*Q)) * dx
    if f0 < val:
        opt, val = 0.0, f0
    if fmax < val:
        opt, val = alpha_max, fmax
    
    return opt, val
