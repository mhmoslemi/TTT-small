    # i+k < n => k < n-i => idx < n-i + n-1 = 2n-1-i, i.e., idx <= 2n-2-i
    # So for term 2: valid idx range is [n-1-i, 2n-2-i]
    # And i+k = i + (idx - n + 1) = idx + i - n + 1
    
    for i in range(n):
        # Term 1: idx from i to i+n-1
        idx1 = np.arange(i, i + n)
        positions1 = i + n - 1 - idx1  # = i - k
        grad[i, idx1] = 1.0 - h[positions1]
        
        # Term 2: idx from n-1-i to 2n-2-i
        idx2 = np.arange(n - 1 - i, 2 * n - 1 - i)
        positions2 = idx2 + i - n + 1  # = i + k
        grad[i, idx2] -= h[positions2]
    
    return grad

def _line_search_pair(c, D, dx, lo, hi):
    """
    Minimize max_idx (c[idx] + t * D[idx]) * dx over t in [lo, hi].
    c: current correlation (length 2n-1)
    D: direction in correlation space (length 2n-1)
    Returns (best_t, best_value)
    """
    # Convex piecewise linear in t
    # Use golden section search
    if hi - lo < 1e-14:
        t = 0.5 * (lo + hi)
        return t, float(np.max(c + t * D)) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(c + c1 * D)) * dx
    f2 = float(np.max(c + c2 * D)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(c + c1 * D)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(c + c2 * D)) * dx
    t = 0.5 * (a + b)
    f = float(np.max(c + t * D)) * dx
    # Also check midpoint and endpoints
    for tcand in [lo, hi, 0.0]:
        if lo <= tcand <= hi:
            fcand = float(np.max(c + tcand * D)) * dx
            if fcand < f:
                t, f = tcand, fcand
    return t, f

    target = n / 2.0
    dx = 2.0 / n
    m = 2 * n - 1
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    grad = _compute_grad(h)
    cur_f = float(np.max(c)) * dx
    best_h = h.copy()
    best_f = cur_f
    stall = 0
    iters = 0
    
    while time.time() < deadline:
        # Compute scores
        top_k = np.argsort(-c)[:K]
        w = np.zeros(m)
        w[top_k] = 1.0 / K
        s = grad @ w  # s[i] = sum of grad[i, top_k] / K
        
        # Find best pair
        # We want s[i] - s[j] < 0 (i.e., s[i] < s[j])
        # h[i] should be < 1 (room to grow) and h[j] should be > 0 (room to shrink)

        # Recipients: low s, h < 1
        recip_mask = h < 0.999
        donor_mask = h > 0.001

        donor_idx = np.where(donor_mask)[0]
        
        if len(recip_idx) < M or len(donor_idx) < M:
            # Fallback: random perturbation
            h = _project(h + rng.normal(0, 0.02, n), target)

            cur_f = float(np.max(c)) * dx
            stall += 1
            if stall > 50:

        donor_sorted = donor_idx[np.argsort(-s[donor_idx])][:M]

                best_h = h.copy()

                if stall > 80:
                    phase = 'pair'
                continue
            h_new = h + t * d
            if h_new.min() < -1e-10 or h_new.max() > 1.0 + 1e-10:
                h_new = _project(h_new, target)
            h = h_new
            c = np.correlate(h, 1.0 - h, mode='full')
            cur_f = float(np.max(c)) * dx
            if cur_f < best_f - 1e-13:
                best_f = cur_f
                best_h = h.copy()
                stall = 0
            elif cur_f > best_f * 1.002 + 1e-5:
                h = best_h.copy()
                c = np.correlate(h, 1.0 - h, mode='full')
                cur_f = best_f
                stall += 1
            else:
                stall += 1
            if stall > 80:
                phase = 'pair'
            beta = min(beta * 1.06, beta_max)
            grad = _compute_grad(h)

        else:
            top_k = np.argsort(-c)[:K]
            w = np.zeros(m)
            w[top_k] = 1.0 / K
            s = grad @ w

            recip_mask = h < 0.999
            donor_mask = h > 0.001
            recip_idx = np.where(recip_mask)[0]
            donor_idx = np.where(donor_mask)[0]

            if len(recip_idx) < 2 or len(donor_idx) < 2:
                h = _project(h + rng.normal(0, 0.02, n), target)
                c = np.correlate(h, 1.0 - h, mode='full')
                grad = _compute_grad(h)
                cur_f = float(np.max(c)) * dx
                stall += 1
                if stall > 60:
                    phase = 'flow'
                    beta = beta0
                    stall = 0
                continue

            M_eff = min(M, len(recip_idx), len(donor_idx))
            recip_sorted = recip_idx[np.argsort(s[recip_idx])][:M_eff]
            donor_sorted = donor_idx[np.argsort(-s[donor_idx])][:M_eff]

            pairs = []
            D_list = []
            hi_list = []
            for i in recip_sorted:
                for j in donor_sorted:
                    if i == j:
                        continue
                    hi_t = min(1.0 - h[i], h[j])
                    if hi_t <= 1e-12:
                        continue
                    pairs.append((i, j))
                    D_list.append(grad[i] - grad[j])
                    hi_list.append(hi_t)

            if not pairs:
                stall += 1
                if stall > 60:
                    h = _project(h + rng.normal(0, 0.015, n), target)
                    c = np.correlate(h, 1.0 - h, mode='full')
                    grad = _compute_grad(h)
                    cur_f = float(np.max(c)) * dx
                    stall = 0
                    phase = 'flow'
                    beta = beta0
                continue

            D_mat = np.array(D_list)
            hi_arr = np.array(hi_list)
            bp, bt, bv = _batch_line_search(c, D_mat, hi_arr, dx, n_frac=20)

            if bp < 0:
                stall += 1
                if stall > 80:
                    sigma = 0.01 + 0.0001 * stall
                    h = _project(h + rng.normal(0, min(sigma, 0.08), n), target)
                    c = np.correlate(h, 1.0 - h, mode='full')
                    grad = _compute_grad(h)
                    cur_f = float(np.max(c)) * dx
                    stall = 0
                    phase = 'flow'
                    beta = beta0
                continue

            i, j = pairs[bp]
            D_best = D_mat[bp]
            hi_t_best = hi_arr[bp]
            t, val = _line_search(c, D_best, dx, 0.0, hi_t_best)
            if val < bv:
                bv = val

            h[i] += t
            h[j] -= t
            c += t * D_best
            for a in range(n):
                idx1 = a - i + n - 1
                if 0 <= idx1 < m:
                    grad[a, idx1] -= t
                idx2 = i - a + n - 1
                if 0 <= idx2 < m:
                    grad[a, idx2] -= t
                idx3 = a - j + n - 1
                if 0 <= idx3 < m:
                    grad[a, idx3] += t
                idx4 = j - a + n - 1
                if 0 <= idx4 < m:
                    grad[a, idx4] += t

            cur_f = float(np.max(c)) * dx
            if cur_f < best_f:
                best_f = cur_f
                best_h = h.copy()
            stall = 0
            if stall == 0 and int(stall * 0) == 0:
                pass
            # Occasionally switch back to flow for perturbation
            if stall % 200 == 199:
                if rng.random() < 0.3:
                    phase = 'flow'
                    beta = beta0 * 2

    return best_h, best_f

def _optimize_refined(h, deadline, rng, K=40, M=35):
    """Refined pair-only optimizer with larger search sets."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    m = 2 * n - 1
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    grad = _compute_grad(h)
    cur_f = float(np.max(c)) * dx
    best_h = h.copy()
    best_f = cur_f
    stall = 0

    while time.time() < deadline:
        top_k = np.argsort(-c)[:K]
        w = np.zeros(m)
        w[top_k] = 1.0 / K
        s = grad @ w

        recip_mask = h < 0.999
        donor_mask = h > 0.001
        recip_idx = np.where(recip_mask)[0]
        donor_idx = np.where(donor_mask)[0]

        if len(recip_idx) < 2 or len(donor_idx) < 2:
            h = _project(h + rng.normal(0, 0.01, n), target)
            c = np.correlate(h, 1.0 - h, mode='full')
            grad = _compute_grad(h)
            cur_f = float(np.max(c)) * dx
            stall += 1
            if stall > 100:
                break
            continue

        M_eff = min(M, len(recip_idx), len(donor_idx))
        recip_sorted = recip_idx[np.argsort(s[recip_idx])][:M_eff]
        donor_sorted = donor_idx[np.argsort(-s[donor_idx])][:M_eff]

        pairs = []
        D_list = []
        hi_list = []
        for i in recip_sorted:
            for j in donor_sorted:
                if i == j:
                    continue
                hi_t = min(1.0 - h[i], h[j])
                if hi_t <= 1e-12:
                    continue
                pairs.append((i, j))
                D_list.append(grad[i] - grad[j])
                hi_list.append(hi_t)

        if not pairs:
            stall += 1
            if stall > 80:
                h = _project(h + rng.normal(0, 0.01, n), target)
                c = np.correlate(h, 1.0 - h, mode='full')
                grad = _compute_grad(h)
                cur_f = float(np.max(c)) * dx
                stall = 0
            continue

        D_mat = np.array(D_list)
        hi_arr = np.array(hi_list)
        bp, bt, bv = _batch_line_search(c, D_mat, hi_arr, dx, n_frac=20)

        if bp < 0:
            stall += 1
            if stall > 100:
                sigma = min(0.05, 0.005 + 0.001 * (stall - 100))
                h = _project(h + rng.normal(0, sigma, n), target)
                c = np.correlate(h, 1.0 - h, mode='full')
                grad = _compute_grad(h)
                cur_f = float(np.max(c)) * dx
                stall = 0
            continue

        i, j = pairs[bp]
        D_best = D_mat[bp]
        t, val = _line_search(c, D_best, dx, 0.0, hi_arr[bp])
        if val > bv:
            t, val = bt, bv

        h[i] += t
        h[j] -= t
        c += t * D_best
        for a in range(n):
            idx1 = a - i + n - 1
            if 0 <= idx1 < m:
                grad[a, idx1] -= t
            idx2 = i - a + n - 1
            if 0 <= idx2 < m:
                grad[a, idx2] -= t
            idx3 = a - j + n - 1
            if 0 <= idx3 < m:
                grad[a, idx3] += t
            idx4 = j - a + n - 1
            if 0 <= idx4 < m:
                grad[a, idx4] += t

        cur_f = float(np.max(c)) * dx
        if cur_f < best_f:
            best_f = cur_f
            best_h = h.copy()
        stall = 0

    return best_h, best_f

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)
