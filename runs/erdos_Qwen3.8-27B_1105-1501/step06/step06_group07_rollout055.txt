def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _project_sum(v, target, lo=0.0, hi=1.0):
    """Project v to [lo,hi]^n with sum = target."""
    n = len(v)
    if target <= lo * n:
        return np.full(n, lo)
    if target >= hi * n:
        return np.full(n, hi)
    # Binary search for threshold
    a = float(v.min()) - 1.0
    b = float(v.max()) + 1.0
    for _ in range(60):
        mid = (a + b) / 2.0
        s = np.sum(np.clip(v - mid, lo, hi))
        if s > target:
            a = mid
        else:
            b = mid
    theta = (a + b) / 2.0
    x = np.clip(v - theta, lo, hi)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < hi - 1e-12
        else:
            mask = x > lo + 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, lo, hi)
    return x

def _lag_gradient(h, lag):
    """Gradient of correlation at a specific lag with respect to h."""
    n = len(h)
    gh = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return gh
        # corr[lag] = sum_{i=0}^{m-1} h[i+lag] * (1-h[i])
        # d/dh[i+lag] = (1-h[i]) for i=0..m-1
        # d/dh[i] = -h[i+lag] for i=0..m-1
        gh[lag:] += (1.0 - h[:m])
        gh[:m] -= h[lag:]
    else:
        ga = -lag
        m = n - ga
        if m <= 0:
            return gh
        # corr[2n-1+lag] = corr at negative lag -ga
        # = sum_{i=0}^{m-1} h[i] * (1-h[i+ga])
        gh[:m] += (1.0 - h[ga:])
        gh[ga:] -= h[:m]
    return gh

def _weighted_gradient(h, corr, K=30, alpha=50.0):
    """Weighted sum of gradients of top K lags."""
    n = len(h)
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(K) / K
    else:
        w /= w_sum
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _lag_gradient(h, k)
    return grad

def _line_search_quadratic(h, d, dx, max_t=1.0, n_iter=80):
    """
    Line search along h + t*d for the objective max(corr) * dx.
    corr(t) = corr0 + t*linear - t^2*quadratic (exact for the product h*(1-h))
    We find t in [0, max_t] (adjusted for box constraints) that minimizes max(corr(t)).
    """
    n = len(h)
    # Compute the three correlation arrays
    # h(t) = h + t*d
    # 1-h(t) = (1-h) - t*d
    # corr(t) = corr(h+t*d, 1-h-t*d) = corr(h,1-h) - t*corr(h,d) - t*corr(d,1-h) + t^2*corr(d,d)
    #         = corr0 + t*(-corr(h,d) - corr(d,1-h)) + t^2*corr(d,d)
    
    c0 = np.correlate(h, 1.0 - h, mode='full')
    c1 = -(np.correlate(h, d, mode='full') + np.correlate(d, 1.0 - h, mode='full'))
    c2 = np.correlate(d, d, mode='full')
    
    # Box constraint: 0 <= h + t*d <= 1
    # For d[i] > 0: t <= (1-h[i])/d[i]
    # For d[i] < 0: t >= -h[i]/d[i] = h[i]/(-d[i])
    pos = d > 1e-14
    neg = d < -1e-14
    t_max = max_t
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    t_min = 0.0
    if np.any(neg):
        t_min = max(t_min, float(np.min(h[neg] / (-d[neg]))))
        # Actually t_min should stay 0 since we're doing descent
        t_min = 0.0
    
    if t_max < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Objective: f(t) = max(c0 + t*c1 + t^2*c2) * dx
    # This is the max of quadratics, which is convex in t.
    # Find minimum via ternary search.
    
    a = t_min
    b = t_max
    
    # Quick grid to narrow down
    grid = np.linspace(a, b, 40)
    obj = np.zeros(len(grid))
    for i, t in enumerate(grid):
        obj[i] = np.max(c0 + t * c1 + t * t * c2)
    best_i = int(np.argmin(obj))
    
    # Narrow to neighborhood
    if best_i > 0:
        a = grid[max(0, best_i - 1)]
    if best_i < len(grid) - 1:
        b = grid[min(len(grid)-1, best_i + 1)]
    
    # Ternary search
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c0 + m1 * c1 + m1 * m1 * c2)
        f2 = np.max(c0 + m2 * c1 + m2 * m2 * c2)
        if f1 < f2:
            b = m2
        else:
            a = m1
        if b - a < 1e-12:
            break
    
    t_opt = (a + b) / 2.0
    f_opt = float(np.max(c0 + t_opt * c1 + t_opt * t_opt * c2)) * dx
    return t_opt, f_opt

def _binary_to_continuous(h_bin, n_frac=4):

            delta = new_c5 - cur_c5
            
            if delta < best_delta:

                best_i, best_j = i, j
            
            # Undo
            h[i] = 1.0
            h[j] = 0.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return _project_sum(h_new, n_new / 2.0)

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_arr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    a = float(v.min()) - 2.0
    b = float(v.max()) + 2.0
    for _ in range(60):
        mid = (a + b) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            a = mid
        else:
            b = mid
    theta = (a + b) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _lag_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        if m <= 0:
            return g
        g[L:] += (1.0 - h[:m])
        g[:m] -= h[L:]
    else:
        q = -L
        m = n - q
        if m <= 0:
            return g
        g[:m] += (1.0 - h[q:])
        g[q:] -= h[:m]
    return g


def _smooth_grad(h, corr, K, alpha):
    n = len(h)
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.ones(K) / K
    else:
        w /= wsum
    grad = np.zeros(n)
    for ii in range(K):
        k = int(top_idx[ii]) - (n - 1)
        grad += w[ii] * _lag_grad(h, k)
    return grad


def _quad_line_search(h, d, dx, max_t=1.0, n_ternary=60):
    n = len(h)
    c0 = np.correlate(h, 1.0 - h, mode='full')
    c1 = -(np.correlate(h, d, mode='full') + np.correlate(d, 1.0 - h, mode='full'))
    c2 = np.correlate(d, d, mode='full')
    pos = d > 1e-14
    t_max = max_t
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if t_max < 1e-14:
        return 0.0, _exact_c5(h)
    a, b = 0.0, t_max
    grid = np.linspace(a, b, 30)
    obj = np.zeros(30)
    for i in range(30):
        obj[i] = np.max(c0 + grid[i] * c1 + grid[i] * grid[i] * c2)
    bi = int(np.argmin(obj))
    if bi > 0:
        a = grid[bi - 1]
    if bi < 29:
        b = grid[bi + 1]
    for _ in range(n_ternary):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c0 + m1 * c1 + m1 * m1 * c2)
        f2 = np.max(c0 + m2 * c1 + m2 * m2 * c2)
        if f1 < f2:
            b = m2
        else:
            a = m1
        if b - a < 1e-13:
            break
    t_opt = (a + b) / 2.0
    f_opt = float(np.max(c0 + t_opt * c1 + t_opt * t_opt * c2)) * dx
    return t_opt, f_opt

            break
        d = -grad / gn
        t, f_try = _quad_line_search(h, d, dx, max_t=step_scale)
        if t > 1e-13:
            h_new = h + t * d
            h_new = _project_sum(h_new, target)
            f_new = _exact_c5(h_new)
            if f_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = f_new
                best_h = h.copy()
                stall = 0
                if t > 0.4 * step_scale:
                    step_scale = min(step_scale * 1.08, 1.5)
                else:
                    step_scale = max(step_scale * 0.88, 0.03)
            else:
                stall += 1
                step_scale *= 0.75
        else:
            stall += 1
            step_scale *= 0.7
        if it % 30 == 29:
            alpha = min(alpha * 1.2, 400.0)
        if stall > 40:
            if step_scale < 0.01:
                break
            step_scale = max(step_scale * 0.5, 0.005)
            alpha = min(alpha * 1.5, 400.0)
            stall = 0
    return best_h, best_c5


def _binary_local_search(h, deadline, rng, max_rounds=30000, batch_size=80):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    order = np.argsort(-h)
    h = np.zeros(n)
    h[order[:int(target)]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_delta = 0.0
        best_i, best_j = -1, -1
        for _ in range(batch_size):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
            h[i] = 1.0
            h[j] = 0.0
        if best_i >= 0 and best_delta < -1e-14:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_c5 = cur_c5 + best_delta
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break
    return best_h, best_c5


def _guided_binary_search(h, deadline, rng, max_rounds=50000, batch_size=100):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = _corr_arr(h)
    order = np.argsort(-h)
    h = np.zeros(n)
    h[order[:int(target)]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        refresh = (rnd % 200 == 0)
        if refresh:
            corr = _corr_arr(h)
            grad = _smooth_grad(h, corr, 40, 50.0)
        hot_score = np.zeros(n)
        cold_score = np.zeros(n)
        if refresh or rnd == 0:
            for idx in np.argsort(-corr)[:30]:
                k = int(idx) - (n - 1)
                g = _lag_grad(h, k)
                hot_score += np.abs(g)
            grad_now = _smooth_grad(h, _corr_arr(h), 40, 50.0)
            hot_score[:]=-np.concatenate
            # Positions where grad is most positive = want to increase = cold (value 0)
            # Positions where grad is most negative = want to decrease = hot (value 1)
            hot_score = np.zeros(n)
            cold_score = np.zeros(n)
            for idx in np.argsort(-_corr_arr(h))[:25]:
                k = int(idx) - (n - 1)
                g = _lag_grad(h, k)
                hot_score += -np.maximum(g, 0)
                cold_score += np.maximum(-g, 0)
        # Add noise
        hot_score += rng.uniform(0, 0.1, n)
        cold_score += rng.uniform(0, 0.1, n)
        best_delta = 0.0
        best_i, best_j = -1, -1
        for _ in range(batch_size):
            pi = np.maximum(hot_score[ones], 0)
            pi = pi / (pi.sum() + 1e-30)
            pj = np.maximum(cold_score[zeros], 0)
            pj = pj / (pj.sum() + 1e-30)
            i = ones[rng.choice(len(ones), p=pi)]
            j = zeros[rng.choice(len(zeros), p=pj)]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
            h[i] = 1.0
            h[j] = 0.0
        if best_i >= 0 and best_delta < -1e-14:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            ones.remove(best_i)
            ones.append(best_j)
            zeros.remove(best_j)
            zeros.append(best_i)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 8000:
                break
    return best_h, best_c5


def _binary_to_cont(h_bin, n_frac):
    n = len(h_bin)
    h = h_bin.astype(float)
    ones = np.where(h > 0.5)[0]
    if len(ones) >= n_frac:
        h[ones[:n_frac]] = 0.5
    zeros = np.where(h <= 0.5)[0]
    if len(zeros) >= n_frac:
        h[zeros[:n_frac]] = 0.5
    return _project_sum(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return _project_sum(h_new, n_new / 2.0)


def _adam_smooth_max(h, deadline, K=30, alpha_init=30.0, lr=0.012, max_iters=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = _corr_arr(h)
        grad = _smooth_grad(h, corr, K, alpha)
        grad -= np.mean(grad)
        t = it + 1
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad * grad)
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _project_sum(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        if it % 100 == 99:
            alpha = min(alpha * 1.12, 350.0)
            if alpha > 200:
                lr = max(lr * 0.85, 0.001)
    return best_h, best_c5


def _multi_start_polish(h_init, deadline, rng, n_starts=5, per_start_time=15.0):
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    for s in range(n_starts):
        sub_dl = min(deadline, time.time() + per_start_time)
        if time.time() > deadline:
            break
        # Create perturbed start
        scale = 0.15 + 0.05 * s
        h_pert = h_init + rng.normal(0, scale, n)
        h_pert = _project_sum(h_pert, target)
        # Quick continuous descent
        h_try, c5_try = _continuous_descent(h_pert, sub_dl, K=25, alpha_init=40.0, max_iters=100)
        # Then binary
        sub_dl2 = min(deadline, time.time() + per_start_time * 0.7)
        h_bin, c5_bin = _binary_local_search(h_try, sub_dl2, rng, max_rounds=10000, batch_size=60)
        if c5_bin < best_c5 - 1e-13:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        elif c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    # Get initial construction
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 118
        x = np.linspace(0.5 / n0, 1.0 - 0.5 / n0, n0)
        h0 = 0.5 + 0.2 * np.sin(2 * np.pi * 4 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        h0 = np.clip(h0, 0, 1)
        h0 = _project_sum(h0, n0 / 2.0)

    h = _project_sum(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Adam smooth-max from parent (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _adam_smooth_max(h, d1, K=35, alpha_init=25.0, lr=0.01, max_iters=4000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Quadratic line search descent (15%)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _continuous_descent(best_h, d2, K=30, alpha_init=50.0, max_iters=300)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary local search (20%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _binary_local_search(best_h, d3, rng, max_rounds=50000, batch_size=80)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Guided binary search (15%)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _guided_binary_search(best_h, d4, rng, max_rounds=50000, batch_size=100)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Continuous polish of best binary result (10%)
    d5 = t0 + budget_s * 0.80
    h5c = _binary_to_cont(best_h, n_frac=6)
    h5, c5_5 = _continuous_descent(h5c, d5, K=25, alpha_init=60.0, max_iters=200)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Also try pure continuous from the continuous version
    h5b, c5_5b = _adam_smooth_max(h5c, d5, K=25, alpha_init=50.0, lr=0.005, max_iters=2000)
    if c5_5b < best_c5:
        best_c5 = c5_5b
        best_h = h5b.copy()

    # Phase 6: Multi-resolution search (15%)
    d6 = t0 + budget_s * 0.95
    tried_n = set()
    offsets = [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12, -16, 16, -20]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = len(best_h) + dn
        if n_try < 50 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = _adam_smooth_max(h_r, sub_dl, K=25, alpha_init=35.0, lr=0.008, max_iters=1500)
        if time.time() < d6 - 10:
            sub_dl2 = min(d6, time.time() + budget_s * 0.01)
            h_r2, c5_r2 = _binary_local_search(h_r, sub_dl2, rng, max_rounds=8000, batch_size=50)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if time.time() < deadline - 3.0:
        h_f, c5_f = _continuous_descent(best_h, deadline - 1.0, K=20, alpha_init=80.0, max_iters=80)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure validity
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
