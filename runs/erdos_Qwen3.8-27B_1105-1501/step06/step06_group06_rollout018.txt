    candidates.add(t_lo)
    candidates.add(t_hi)

    for k in range(n):
        if U[k] < -1e-15:
            tc = -T[k] / (2.0 * U[k])
            if t_lo < tc < t_hi:
                candidates.add(tc)
    
    # Pairwise intersections
    for i in range(n):
        for j in range(i+1, n):
            a = U[i] - U[j]
            b = T[i] - T[j]
            c = S[i] - S[j]
            if abs(a) < 1e-15:
                if abs(b) > 1e-15:
                    tc = -c / b
                    if t_lo < tc < t_hi:

def _generate_directions(h, active_lags, n_dirs, rng):
    """Generate diverse search directions, all sum-zero."""
    n = len(h)
    dirs = []
    one = np.ones(n)
    
    # 1. Steepest descent on active set
    G = np.zeros((len(active_lags), n))
    for idx, lag in enumerate(active_lags):
        s = int(lag) - (n - 1)
        m = n - abs(s)
        if s >= 0:
            G[idx, :m] += (1.0 - h[s:s+m])
            G[idx, s:s+m] += (-h[:m])
        else:
            g = -s
            G[idx, g:g+m] += (1.0 - h[:m])
            G[idx, :m] += (-h[g:g+m])
    
    # Project gradient to sum-zero
    grad = G.sum(axis=0)
    grad -= grad.mean()
    d = -grad  # descending
    d -= d.mean()  # ensure sum zero (should already be, but just in case)
    if np.linalg.norm(d) > 1e-14:
        dirs.append(d / np.linalg.norm(d))
    
    # 2. Null-space directions (random in null-space of active constraints)
    if len(active_lags) < n - 1:
        # Random vectors orthogonal to active gradients and to 1
        for _ in range(min(5, n_dirs)):
            v = rng.normal(size=n)
            # Project out active gradients
            for row in G:
                proj = np.dot(v, row) / (np.dot(row, row) + 1e-15)
                v -= proj * row
            # Project out 1
            v -= v.mean() * one
            if np.linalg.norm(v) > 1e-10:
                dirs.append(v / np.linalg.norm(v))
    
    # 3. Fourier-based smooth directions
    n_fourier = min(8, n_dirs)
    for freq in range(1, n_fourier + 1):
        phase = rng.uniform(0, 2 * np.pi)
        f = rng.integers(1, max(2, n // 4))
        d = np.cos(2 * np.pi * f * np.arange(n) / n + phase)
        d -= d.mean()  # sum zero
        if np.linalg.norm(d) > 1e-10:
            dirs.append(d / np.linalg.norm(d))
    
    # 4. Paired coordinate directions
    n_pairs = min(10, n_dirs)

            dirs.append(d / np.linalg.norm(d))

    return dirs

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_optimize(h0, deadline, max_outer=10):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        K = min(20, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]

        t_init = cur_c5 + 1e-5
        x0 = np.concatenate([h, [t_init]])

        def _obj_fn(x):
            return x[n]

        def _obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def _ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]

        def _ineq_jac(x):
            hx = x[:n]
            G = _corr_grads(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J

        def _eq_fn(x):
            return np.sum(x[:n]) - target

        def _eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _ineq_fn, 'jac': _ineq_jac},
            {'type': 'eq', 'fun': _eq_fn, 'jac': _eq_jac},
        ]

        res = minimize(_obj_fn, x0, method='SLSQP', jac=_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-14})

        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            break

    return best_h, best_c5

def _structured_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 3 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 122
        h = _structured_init(n, rng)
    n = len(h)
    h = _project(h, n / 2.0)

    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: SLSQP quick start (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _slsqp_optimize(h, d1, max_outer=8)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Exact active descent (50%)
    d2 = t0 + budget_s * 0.60
    h2, c5_2 = _active_descent_exact(best_h, d2, max_iter=5000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: SLSQP after descent (10%)
    d3 = t0 + budget_s * 0.70
    h3, c5_3 = _slsqp_optimize(best_h, d3, max_outer=10)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Multi-resolution (15%)
    d4 = t0 + budget_s * 0.85
    offsets = [-6, 6, -4, 4, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _active_descent_exact(h_r, sub_dl, max_iter=800)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Multi-start (5%)
    d5 = t0 + budget_s * 0.90
    for _ in range(5):
        if time.time() > d5:
            break
        h_rs = best_h + rng.normal(0, 0.03, best_n)
        h_rs = _project(h_rs, best_n / 2.0)
        sub_dl = min(d5, time.time() + budget_s * 0.008)
        h_rs, c5_rs = _active_descent_exact(h_rs, sub_dl, max_iter=500)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
            best_n = len(h_rs)

    # Phase 6: Final exact descent polish (10%)
    d6 = t0 + budget_s * 0.98
    h6, c5_6 = _active_descent_exact(best_h, d6, max_iter=10000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
