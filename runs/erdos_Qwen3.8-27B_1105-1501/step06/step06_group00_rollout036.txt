    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid

def _pair_line_search(S0, T, C, t_lo, t_hi, n_grid=100):
    """Find t in [t_lo, t_hi] minimizing max_k (S0[k] + t*T[k] - t^2*C[k])"""
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S0))
    
    # Dense grid
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None,:] + ts[:,None]*T[None,:] - ts[:,None]**2*C[None,:], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Refine with golden section
    lo = ts[max(0, best_idx-2)]
    hi = ts[min(n_grid-1, best_idx+2)]
    
    def f(t):
        return float(np.max(S0 + t*T - t*t*C))
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr*(hi-lo)
    c2 = lo + gr*(hi-lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if hi - lo < 1e-12:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr*(hi-lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr*(hi-lo)
            f2 = f(c2)
    
    opt_t = (lo+hi)/2.0
    opt_val = f(opt_t)
    
    # Also check endpoints and grid best
    candidates = [(opt_t, opt_val), (t_lo, f(t_lo)), (t_hi, f(t_hi)), (ts[best_idx], float(vals[best_idx]))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _smooth_max_gradient(S0, beta=20.0):
    """Compute gradient of smooth max of S0 using softmax weighting."""
    M = float(np.max(S0))
    if M < 1e-15:
        return np.zeros(len(S0))
    w = np.exp(beta * (S0 - M))
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        # Fallback: uniform over top elements
        top_k = max(1, int(0.1 * len(S0)))
        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= w_sum
    return w

            #   If j < k: j appears as second factor at i... wait

            #     First factor: delta_{j, i+k} * (1-h[i]) where i = j-k, valid if 0 <= j-k <= n-k-1
            #     Second factor: h[i+k] * (-delta_{j,i}) where i = j, valid if 0 <= j <= n-k-1
            #   So: dS_k/dh[j] = (1-h[0]) * delta_{j,k} ... no wait
            #   Let me re-index: S_k = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])
            #   dS_k/dh[j] = sum_{i=0}^{n-k-1} [delta_{j,i+k}*(1-h[i]) + h[i+k]*(-delta_{j,i})]
            #   = (1-h[j-k]) if k <= j <= k+(n-k-1) = n-1, i.e., j >= k
            #     minus h[j-k] ... wait
            #   First term: j = i+k means i = j-k, valid if 0 <= j-k <= n-k-1, i.e., k <= j <= n-1
            #     contributes (1-h[j-k])
            #   Second term: j = i, valid if 0 <= j <= n-k-1
            #     contributes -h[j+k]
            
            m = n - k
            # First factor contribution: j from k to n-1, value (1-h[j-k])
            G[row, k:] += (1.0 - h[:m])
            # Second factor contribution: j from 0 to m-1, value -h[j+k]  
            G[row, :m] += -h[k:]
        else:
            g = -k  # g > 0
            m = n - g
            # S_{-g} = sum_{i=g}^{n-1} h[i]*(1-h[i-g])
            # dS/dh[j]:
            #   First factor: delta_{j,i}*(1-h[i-g]) valid if g <= j <= n-1
            #     contributes (1-h[j-g])
            #   Second factor: h[i]*(-delta_{j,i-g}) = -h[i-g]*delta_{j,i-g}, valid if j+g <= n-1, i.e., j <= n-1-g
            #     contributes -h[j+g]
            
            G[row, g:] += (1.0 - h[:m])
            G[row, :m] += -h[g:]
    return G

def _select_pairs(h, S0, G, w, n_pairs=30, rng=None):
    """Select candidate pairs (i,j) for transfer based on gradient."""
    n = len(h)
    # Smooth max gradient for each coordinate
    coord_grad = G.T @ w  # shape (n,)
    # We want to move mass FROM high gradient TO low gradient
    # (increasing h at low-grad positions, decreasing at high-grad positions)
    # Transfer d = e_sink - e_source reduces objective if grad[sink] < grad[source]
    
    if rng is not None:
        # Mix of greedy and random pairs
        n_greedy = max(5, n_pairs // 2)
        n_random = n_pairs - n_greedy
        
        sorted_idx = np.argsort(coord_grad)
        sources = sorted_idx[-n_greedy:]  # highest gradient (want to decrease)
        sinks = sorted_idx[:n_greedy]      # lowest gradient (want to increase)
        
        pairs = []
        for s_idx in range(n_greedy):
            for kn in range(3):  # try a few sinks per source
                if kn < len(sinks):
                    pairs.append((int(sinks[kn]), int(sources[s_idx])))
        
        # Add random pairs
        for _ in range(n_random):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((i, j))
        
        # Shuffle
        rng.shuffle(pairs)
        return pairs[:n_pairs]
    else:
        # Pure greedy
        sorted_idx = np.argsort(coord_grad)
        sources = sorted_idx[-n_pairs:]
        sinks = sorted_idx[:n_pairs]
        pairs = []
        for s_idx in range(n_pairs):
            pairs.append((int(sinks[s_idx]), int(sources[s_idx])))
        return pairs

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _pair_line_search(S0, T, C, t_lo, t_hi):

    # Golden section refinement
    lo = ts[max(0, best_idx-2)]

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(S0, T, C, t_lo, t_hi):
    """Minimize max_k(S0[k] + t*T[k] - t^2*C[k]) over t in [t_lo, t_hi]."""

        return t_lo, float(np.max(S0))
    n_grid = 70
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S0[None, :] + ts[:, None] * T[None, :] - (ts[:, None]) ** 2 * C[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    lo = ts[max(0, best_idx - 2)]
    hi = ts[min(n_grid - 1, best_idx + 2)]
    if hi - lo < 1e-12:
        return float(ts[best_idx]), float(vals[best_idx])
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c2 = lo + gr * (hi - lo)

    M = float(np.max(S0))
    if M < 1e-15:
        return np.ones(len(S0)) / len(S0)
    w = np.exp(beta * (S0 - M))
    s = float(np.sum(w))

        idx = np.argsort(S0)[-top_k:]
        w = np.zeros(len(S0))
        w[idx] = 1.0 / top_k
    else:
        w /= s
    return w

def _grad_matrix(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            m = n - k
            G[row, k:] += (1.0 - h[:m])
            G[row, :m] -= h[k:]
        else:
            p = -k
            m = n - p
            G[row, p:] += (1.0 - h[:m])
            G[row, :m] -= h[p:]
    return G
