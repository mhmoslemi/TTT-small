import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for clipping threshold
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_jacobian(h, dx):
    """Build the Jacobian matrix G where G[q, i] = dS_q/dh_i"""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k == 0:
            # S_q = dx * sum h_i*(1-h_i)
            # dS_q/dh_i = dx*(1 - 2*h_i)
            G[q, :] = dx * (1.0 - 2.0 * h)
        elif k > 0:
            # S_q = dx * sum_{i=0}^{n-k-1} h_{i+k}*(1-h_i)
            m = n - k
            # d/dh_i: for i in 0..m-1: -dx*h_i (from 1-h_i term)
            #           for i in k..n-1: dx*(1-h_{i-k}) (from h_{i+k} term)
            G[q, :m] += -dx * h[:m]
            G[q, k:] += dx * (1.0 - h[:m])
        else:
            # S_q = dx * sum_{i=0}^{n-|k|-1} h_{i-|k|}*(1-h_i)
            r = -k
            m = n - r
            G[q, r:] += -dx * h[:m]
            G[q, :m] += dx * (1.0 - h[r:])
    return G

def _pnorm_obj_grad(h_flat, p, dx):
    """Compute the p-norm objective and its gradient."""
    n = h_flat.shape[0]
    target = n / 2.0
    h = _project(h_flat, target)
    S = _corr_vals(h)
    S_max = float(np.max(S))
    if S_max < 1e-15:
        return 0.0, np.zeros(n)
    r = S / S_max
    U = float(np.sum(r ** p))
    phi = S_max * (U ** (1.0 / p))
    G = _build_jacobian(h, dx)
    # grad phi = S_max * U^(1/p - 1) * sum r_q^(p-1) * G_q
    weights = r ** (p - 1)
    grad = S_max * (U ** (1.0 / p - 1.0)) * (G.T @ weights)
    return phi, grad

def _pnorm_optimize(h, deadline, p_list=None):
    """Optimize using p-norm surrogate with progressive p increase."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    if p_list is None:
        p_list = [2, 3, 5, 8, 13, 21, 34]
    
    for p in p_list:
        if time.time() > deadline:
            break
        # Optimize at this p level
        sub_deadline = min(deadline, time.time() + max(2.0, (deadline - time.time()) / (len(p_list) - p_list.index(p))))
        current_h = h.copy()
        
        # Use SLSQP with p-norm objective
        def objective(x):
            phi, _ = _pnorm_obj_grad(x, p, dx)
            return phi
        
        def gradient(x):
            _, g = _pnorm_obj_grad(x, p, dx)
            return g
        
        # Constraint: sum(x) = target
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
        
        bounds = [(0, 1)] * n
        
        # Multiple restarts within this p level
        n_restarts = 3
        for rs in range(n_restarts):
            if time.time() > sub_deadline:
                break
            x0 = current_h.copy()
            if rs > 0:
                rng = np.random.default_rng(1000 + rs * 7 + p)
                x0 = _project(x0 + rng.normal(0, 0.05 * (p / 13.0), n), target)
            
            try:
                res = minimize(objective, x0, jac=gradient, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
                if res.success or res.fun < 1e10:
                    h_new = _project(res.x, target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        current_h = h_new.copy()
            except Exception:
                pass
        
        h = current_h
    
    return best_h, best_c5

        if not improved:

            if stall % 5 == 0 and len(ones) >= 2 and len(zeros) >= 2:
                for _ in range(20):
                    if time.time() > deadline:
                        break
                    idx_o = rng.choice(len(ones), size=2, replace=False)
                    idx_z = rng.choice(len(zeros), size=2, replace=False)
                    i1, i2 = ones[idx_o[0]], ones[idx_o[1]]
                    j1, j2 = zeros[idx_z[0]], zeros[idx_z[1]]
                    if len({i1, i2, j1, j2}) < 4:
                        continue
                    h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
                    c5_try = _exact_c5(h)
                    evals += 1
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h.copy()
                        improved = True
                        ones = np.where(h == 1.0)[0]
                        zeros = np.where(h == 0.0)[0]
                        stall = 0
                        break
                    else:
                        h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
    return best_h, best_c5

def _pair_optimize(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_vals(h)
        max_idx = np.argmax(S)
        G = _build_jacobian(h, dx)
        grad = G[max_idx]
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(len(order), 25)):
            if time.time() > deadline:
                break
            i = order[ii]
            for jj in range(min(len(order), 25)):
                if tried > 50:
                    break
                j = order[jj]
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search_exact(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                tried += 1
        if not improved:
            for _ in range(20):
                if time.time() > deadline:
                    break
                if len(interior) < 2:
                    break
                idx = rng.choice(len(interior), size=2, replace=False)
                i, j = interior[idx[0]], interior[idx[1]]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search_exact(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
            if not improved:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_round(h):
    n = len(h)
    target = int(n / 2)
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def _diverse_start(n, rng, seed_type=0):
    x = np.arange(n)
    if seed_type == 0:
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / n + phase)
        h += rng.normal(0, 0.1, n)
    elif seed_type == 1:
        f1, f2 = rng.integers(1, 5), rng.integers(5, 12)
        p1, p2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.25 * np.sin(2*np.pi*f1*x/n + p1) + 0.25 * np.sin(2*np.pi*f2*x/n + p2)
    elif seed_type == 2:
        block_size = max(2, n // rng.integers(4, 12))
        h = np.zeros(n)
        pos = 0
        while pos < n:
            if rng.random() < 0.5:
                h[pos:pos+block_size] = 1.0
            pos += block_size
        h += rng.normal(0, 0.05, n)
    else:
        h = np.zeros(n)
        indices = rng.choice(n, size=n//2, replace=False)
        h[indices] = 1.0
        h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _multi_start_optimize(h_best, c5_best, deadline, n, rng, budget_per_start):
    """Try diverse starts with p-norm + LP."""
    for si in range(6):
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + budget_per_start)
        h_s = _diverse_start(n, rng, seed_type=si % 4)
        # P-norm phase
        d_p = time.time() + budget_per_start * 0.5
        for p in [4, 16]:
            if time.time() > d_p:
                break
            h_s, c5_s = _pnorm_pgd(h_s, p, min(d_p, time.time() + budget_per_start * 0.25))
        # LP phase
        d_lp = min(sub_dl, time.time() + budget_per_start * 0.4)
        h_s, c5_s = _active_lp(h_s, d_lp, start_radius=0.10)
        if c5_s < c5_best:
            c5_best = c5_s
            h_best = h_s.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x / n0)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: p-norm PGD progressive (18%)
    d1 = t0 + budget_s * 0.18
    for p in [2, 4, 8, 16, 32, 64]:
        if time.time() > d1:
            break
        frac = 1.0 / (6 - [2,4,8,16,32,64].index(p))
        sub_dl = min(d1, time.time() + max(1.5, (d1 - time.time()) * frac * 1.5))
        h1, c5_1 = _pnorm_pgd(best_h, p, sub_dl, lr_init=0.03)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Active LP (20%)
    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _active_lp(best_h, d2, start_radius=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Pair optimization (8%)
    d3 = t0 + budget_s * 0.46
    h3, c5_3 = _pair_optimize(best_h, d3, max_passes=40)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary local search (12%)
    d4 = t0 + budget_s * 0.58
    h_bin = _binary_round(best_h)
    h_bin, c5_bin = _binary_local_search(h_bin, d4, max_evals=15000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Also try binary from p-norm result
    h_bin2 = _binary_round(_pnorm_pgd(h0, 16, t0 + 2.0)[0])
    if time.time() < d4:
        h_bin2, c5_bin2 = _binary_local_search(h_bin2, min(d4, time.time() + budget_s * 0.03))
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Phase 5: More LP + pairs on best (12%)
    d5 = t0 + budget_s * 0.70
    h5, c5_5 = _active_lp(best_h, d5, start_radius=0.08)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < d5:
        h5b, c5_5b = _pair_optimize(best_h, min(d5, time.time() + budget_s * 0.03), max_passes=20)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-start diverse (15%)
    d6 = t0 + budget_s * 0.85
    per_start = max(2.0, (d6 - time.time()) / 5)
    best_h, best_c5 = _multi_start_optimize(best_h, best_c5, d6, n0, rng, per_start)

    # Phase 7: Resolution transfer (10%)
    d7 = t0 + budget_s * 0.95
    tried_n = set([n0])
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10]
    for dn in offsets:
        if time.time() > d7:
            break
        n_try = n0 + dn
        if n_try < 30 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.012)
        h_r, c5_r = _pnorm_pgd(h_r, 8, min(sub_dl, time.time() + budget_s * 0.006))
        h_r, c5_r = _active_lp(h_r, sub_dl, start_radius=0.08)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n0 = n_try

    # Phase 8: Final polish (5%)
    d8 = deadline
    h8, c5_8 = _active_lp(best_h, d8, start_radius=0.04)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8:
        h8b, c5_8b = _pair_optimize(best_h, min(d8, time.time() + 5.0), max_passes=15)
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
