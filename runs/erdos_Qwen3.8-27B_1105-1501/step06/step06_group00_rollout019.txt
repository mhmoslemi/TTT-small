Given h and direction d (with sum(d) ≈ 0):
- S0 = correlate(h, 1-h) [length 2n-1]
- For the line h(α) = h + αd:
  - 1 - h(α) = (1-h) - αd
  - correlate(h(α), 1-h(α)) = correlate(h+αd, (1-h)-αd)
  - = correlate(h, 1-h) + α[correlate(d, 1-h) - correlate(h, d)] - α² correlate(d, d)
  - = S0 + α*T - α²*Q

where T = correlate(d, 1-h) - correlate(h, d) and Q = correlate(d, d).

The objective at α is: f(α) = dx * max(S0 + α*T - α²*Q)

This is the max of (2n-1) quadratic functions of α, each of the form:
  r_i(α) = S0[i] + α*T[i] - α²*Q[i]

To find min f(α) over feasible α:
- The feasible interval is determined by 0 ≤ h_i + α*d_i ≤ 1 for all i
- Within that interval, we want to minimize the upper envelope of quadratics

Strategy for finding the minimum:
1. Find vertices of each active quadratic: α_i = T[i] / (2*Q[i]) (if Q[i] > 0)
2. Find pairwise intersections of active quadratics
3. Include endpoints of feasible interval
4. Evaluate f at all candidate points, pick the best

For the gradient, using soft-max:
  w_i = exp(β * (S0[i]*dx - max_val)) / sum(exp(β * (S0[j]*dx - max_val)))
  
Then the gradient of the smooth objective w.r.t. h is:
  ∂/∂h_k [sum_i w_i * S0[i]]

For a fixed lag λ (where row index r = n-1+λ):

  - If k appears as (1-h[p+λ]) term: contributes -h[p] for valid p

This gives us the gradient. Then direction = -gradient, projected to zero-sum.

    if a_hi - a_lo > 1e-14:

        candidates.extend(grid.tolist())

        if val < best_val:
            best_val = val
            best_alpha = float(a)
    
    return best_alpha, best_val

                if d3 is not None:
                    directions.append(d3)

        i_max = int(np.argmax(corr))
        i_min = int(np.argmin(corr[corr > 0])) if np.any(corr > 0) else 0
        if i_max != i_min and g.std() > 1e-10:
            idx_hi = int(np.argmax(g))
            idx_lo = int(np.argmin(g))
            if idx_hi != idx_lo:
                d4 = np.zeros(n)
                d4[idx_hi] = -1.0
                d4[idx_lo] = 1.0
                d4n = _normalize_dir(d4)
                if d4n is not None:
                    directions.append(d4n)

        if use_lp and it % lp_every == 0:
            scores = corr * dx
            d_lp = _lp_direction(h, scores, radius)
            if d_lp is not None:
                d_lp_n = _normalize_dir(d_lp)
                if d_lp_n is not None:
                    directions.insert(0, d_lp_n)

        improved = False
        for d in directions:
            if time.time() >= deadline:
                break
            alpha, c5_try = _line_search(h, d, dx, radius)
            if c5_try < cur_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                c5_actual = _exact_c5(h_new)
                if c5_actual < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_actual
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    radius = min(radius * 1.04, 0.8)
                    beta = min(beta * 1.02, 80.0)
                    break
        if not improved:
            stall += 1
            radius *= 0.82
            if stall % 8 == 0:
                beta *= 1.4
            if radius < 0.03:
                break

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.3, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    perturb *= amplitude
    return _project(h + perturb, n / 2.0)

def _binary_refine(h, deadline, max_iter=20000):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(4242)
    T = 0.003
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        if stall > 400:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        h_new = cur.copy()
        h_new[ones[i]] = 0.0
        h_new[zeros[j]] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=5):
    n = len(h)
    target = n / 2.0
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for _ in range(max_rounds):
        if time.time() >= deadline:
            break
        found = False
        idx_hi = np.where(cur > 0.5)[0]
        idx_lo = np.where(cur < 0.5)[0]
        if len(idx_hi) == 0 or len(idx_lo) == 0:
            break
        for i in idx_hi:
            if time.time() >= deadline:
                break
            d = np.zeros(n)
            d[i] = -1.0
            for j in idx_lo:
                d[j] = 1.0
                dn = _normalize_dir(d)
                if dn is None:
                    continue
                alpha, c5_try = _line_search(cur, dn, 2.0/n, 0.5)
                if c5_try < cur_c5 - 1e-13:
                    h_new = cur + alpha * dn
                    h_new = _project(h_new, target)
                    c5_act = _exact_c5(h_new)
                    if c5_act < cur_c5 - 1e-14:
                        cur = h_new
                        cur_c5 = c5_act
                        best_h = cur.copy()
                        best_c5 = c5_act
                        found = True
                        break
                d[j] = 0.0
        if not found:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Primary local optimization (35%)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _local_opt(h_init, d1, beta0=25.0, radius0=0.3, max_iter=10000, use_lp=True, lp_every=15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Resolution sweep (20%)
    d2 = t0 + budget_s * 0.55
    n_curr = len(best_h)
    offsets = [-4, 4, -8, 8, -12, 12, -6, 6, -16, 16, -20, 20, -10, 10, 24, -24, 14, -14, 18, -18, 28, -28, 32, -32]
    tried = set()
    for dn in offsets:
        if time.time() >= d2:
            break
        n_try = n0 + dn
        if n_try < 48 or n_try > 260 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.012)
        h_r, c5_r = _local_opt(h_r, sub_dl, beta0=20.0, radius0=0.25, max_iter=4000, use_lp=True, lp_every=30)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
            n_curr = n_try

    # Phase 3: Binary refinement (10%)
    d3 = t0 + budget_s * 0.65
    nb = len(best_h)
    h_b = _resample(best_h, nb)
    idx_b = np.argsort(h_b)[-nb // 2:]
    h_bin = np.zeros(nb)
    h_bin[idx_b] = 1.0
    sub_dl = min(d3, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_refine(h_bin, sub_dl, max_iter=30000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin
    # Polish binary result
    sub_dl2 = min(d3, time.time() + budget_s * 0.04)
    h_pol = _project(best_h.copy(), len(best_h) / 2.0)
    h_pol, c5_pol = _local_opt(h_pol, sub_dl2, beta0=30.0, radius0=0.15, max_iter=3000, use_lp=True, lp_every=20)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol

    # Phase 4: Basin hopping (25%)
    d4 = t0 + budget_s * 0.90
    scale = 0.025
    n_restarts = 0
    while time.time() < d4 and n_restarts < 300:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.006)
        h_t, c5_t = _local_opt(h_pert, sub_dl, beta0=20.0, radius0=0.2, max_iter=2000, use_lp=True, lp_every=40)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.1, scale * 1.07)

    # Phase 5: Final polish (10%)
    d5 = deadline - 3
    h5 = _project(best_h.copy(), len(best_h) / 2.0)
    h5, c5_5 = _local_opt(h5, min(d5, time.time() + budget_s * 0.06), beta0=40.0, radius0=0.08, max_iter=5000, use_lp=True, lp_every=10)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5

    # Pair sweep on best
    if deadline - time.time() > 5:
        h_ps = _project(best_h.copy(), len(best_h) / 2.0)
        h_ps, c5_ps = _pair_sweep(h_ps, deadline - 2, max_rounds=3)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
