import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt:
            x = np.clip(x + res / cnt * mask, 0.0, 1.0)
    return x


def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += 1.0 - h[:m]
        g[:m] -= h[k:]
    else:
        a = -k
        m = n - a
        if m <= 0:
            return g
        g[:m] += 1.0 - h[a:]
        g[a:] -= h[:m]
    return g * dx


def _t_max_box(h, d, cap):
    pos, neg = d > 1e-15, d < -1e-15
    tm = cap
    if np.any(pos):
        tm = min(tm, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        tm = min(tm, float(np.min(h[neg] / (-d[neg]))))
    return max(tm, 0.0)


def _line_min(h, d, dx, cap=1.5):
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    tm = _t_max_box(h, d, cap)
    if tm < 1e-14:
        return 0.0, float(np.max(S)) * dx
    ng = 60
    grid = np.linspace(0.0, tm, ng)
    bv, bt = np.inf, 0.0
    for i in range(ng):
        v = float(np.max(S + grid[i] * T - grid[i] * grid[i] * Q))
        if v < bv:
            bv, bt = v, grid[i]
    gr = 0.61803398875
    idx = int(bt / tm * (ng - 1)) if tm > 0 else 0
    a = grid[max(0, idx - 1)]
    b = grid[min(ng - 1, idx + 1)]
    if b - a < 1e-14:
        return float(bt), bv * dx

    def fa(t):
        return float(np.max(S + t * T - t * t * Q))

    c1, c2 = b - gr * (b - a), a + gr * (b - a)
    f1, f2 = fa(c1), fa(c2)
    for _ in range(55):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = fa(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = fa(c2)
    ot = (a + b) * 0.5
    ov = fa(ot)
    if ov < bv:
        bv, bt = ov, ot
    return float(bt), bv * dx


def _descend(h, deadline, K0=8, K1=45, a0=15.0, a1=600.0, niter=500, step0=0.3):
    n = len(h)
    target, dx = n / 2.0, 2.0 / n
    h = h.copy()
    bh, bc = h.copy(), _exact_c5(h)
    step, stall = step0, 0
    for i in range(niter):
        if time.time() > deadline:
            break
        frac = i / max(1, niter - 1)
        K = max(3, int(K0 + (K1 - K0) * frac))
        alpha = a0 + (a1 - a0) * frac
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(-corr)[:K]
        M = corr[idx[0]]
        w = np.exp(-alpha * (M - corr[idx]))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, r in zip(w, idx):
            grad += wi * _grad_lag(h, int(r) - (n - 1), dx)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        d = -grad / gn
        t, fv = _line_min(h, d, dx, cap=step)
        if t < 1e-12:
            step *= 0.7
            stall += 1
            if stall > 50:
                break
            continue
        hnew = _project(h + t * d, target)
        fc = _exact_c5(hnew)
        if fc < bc - 1e-14:
            h, bc, bh = hnew, fc, hnew.copy()
            stall = 0
            if t > 0.8 * step:
                step = min(step * 1.04, 2.0)
            else:
                step = max(step * 0.93, 0.03)
        else:
            stall += 1
            step *= 0.75
            if stall > 50:
                break
    return bh, bc


def _pair_kick(h, rng, n_pairs=300, max_move=0.15):
    n = len(h)
    target, dx = n / 2.0, 2.0 / n
    h = h.copy()
    for _ in range(n_pairs):
        i, j = int(rng.integers(n)), int(rng.integers(n))
        if i == j:
            continue
        S = np.correlate(h, 1.0 - h, mode='full')
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        thi = min(1.0 - h[i], h[j], max_move)
        if thi < 1e-12:
            continue
        ng = 30
        grid = np.linspace(0.0, thi, ng)
        bv, bt = np.inf, 0.0
        for t in grid:
            v = float(np.max(S + t * T - t * t * Q))
            if v < bv:
                bv, bt = v, t
        gr = 0.618
        idx = int(bt / thi * (ng - 1)) if thi > 0 else 0
        a = grid[max(0, idx - 1)]
        b = grid[min(ng - 1, idx + 1)]
        if b - a < 1e-14:
            continue
        fa = lambda t: float(np.max(S + t * T - t * t * Q))
        c1, c2 = b - gr * (b - a), a + gr * (b - a)
        f1, f2 = fa(c1), fa(c2)
        for _ in range(35):
            if b - a < 1e-13:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = fa(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = fa(c2)
        ot = (a + b) * 0.5
        ov = fa(ot)
        if ov < bv:
            bv, bt = ov, ot
        h[i] += bt
        h[j] -= bt
        h = np.clip(h, 0.0, 1.0)
    return _project(h, target)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _multi_alpha_descent(h, deadline):
    n = len(h)
    bh, bc = h.copy(), _exact_c5(h)
    configs = [
        (5, 15, 10.0, 200.0, 200),
        (10, 30, 20.0, 400.0, 200),
        (15, 50, 30.0, 800.0, 200),
        (20, 60, 50.0, 1200.0, 200),
        (8, 25, 15.0, 300.0, 200),
        (12, 35, 25.0, 500.0, 200),
    ]
    for K0, K1, a0, a1, niter in configs:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + 8.0)
        h2, c2 = _descend(bh, sub_dl, K0=K0, K1=K1, a0=a0, a1=a1, niter=niter)
        if c2 < bc:
            bh, bc = h2, c2
    return bh, bc


def _sa_binary(h, deadline, rng, n_iters=40000):
    n = len(h)
    target = n // 2
    order = np.argsort(-h)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    cur = _exact_c5(h)
    bh, bc = h.copy(), cur
    T = 0.008
    T_min = 2e-6
    cool = 0.99994
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h <= 0.5)[0].tolist())
    ones_list = list(ones)
    zeros_list = list(zeros)
    for _ in range(n_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones_list[int(rng.integers(len(ones_list)))]
        j = zeros_list[int(rng.integers(len(zeros_list)))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = new_c5
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if cur < bc:
                bc = cur
                bh = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cool)
    return bh, bc
