The previous program tried n offsets of ±2, ±4, etc. from 122. So it tried 118, 120, 124, 126, 130, 134, 138, 142, 146, 150. But it only spent 2% of the time on each. Maybe I should spend more time on the most promising resolutions.

Let me also consider: what if the optimal n is NOT close to 122? What if n=100 or n=150 gives a better result?

Actually, for the discrete problem, the objective is:
C5 = dx * max_r corr_r = (2/n) * max_r corr_r

The correlation values scale with n (each term in the sum involves n terms of order 0.25 on average). So max_r corr_r ~ n * 0.25, and C5 ~ 2/n * n * 0.25 = 0.5. But the actual value is ~0.38, which means the max is below the "random" level.

The key insight is that C5 depends on the specific structure of h, not just n. Changing n changes both the resolution AND the set of feasible structures.

Let me try a different strategy: **genetic algorithm / population-based search**.

Maintain a population of, say, 10 candidates. Each generation:
1. Evaluate all candidates
2. Keep the top 5
3. Generate 5 new candidates by:
   - Mutation: add small noise to a top candidate
   - Crossover: blend two top candidates
   - Resampling: resample a top candidate to a different n
4. Optimize each new candidate briefly

This might help escape local optima.

import numpy as np
from scipy.optimize import linprog
import time

# --- Core evaluation ---

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

# --- Projection ---

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

# --- Gradients ---

def _grad_lag(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:k+m])
            g[k:k+m] += (-h[:m])
    else:
        d = -k
        m = n - d
        if m > 0:
            g[d:d+m] += (1.0 - h[:m])
            g[:m] += (-h[d:d+m])
    return g

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        G[idx] = _grad_lag(h, k, n)
    return G

# --- Line search ---

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid + golden section
    n_grid = 32
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

# --- LP direction ---

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    # Use both top-K and near-max
    tol = max(0.03 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([sorted_idx[:K], near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

# --- Log-sum-exp direction ---

def _lse_direction(h, S, n, tau, radius):
    M = float(np.max(S))
    w = np.exp((S - M) / tau)
    sw = w.sum()
    if sw < 1e-300:
        return None
    w = w / sw
    active = np.where(w > 1e-5)[0]
    G = _grad_rows(h, active, n)
    grad = w[active] @ G
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return None
    d = -grad / norm
    # Scale to trust radius while respecting box constraints
    pos = d > 1e-15
    neg = d < -1e-15
    t_max = radius
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_max = min(t_max, float(np.max(-h[neg] / d[neg])))
    t_max = max(t_max, 1e-10)
    if t_max >= radius:
        return d * radius
    return d * t_max

# --- Main LP loop ---

def _lp_loop(h, deadline, radius_init=0.12, K=15, use_lse=False, tau=0.02):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.time() < deadline:
        if stall > 50:
            # Perturb and restart
            scale = 0.01 + 0.005 * (stall % 10)
            h = _project(h + rng.normal(0, scale, n), target)
            cur_c5 = _exact_c5(h)
            radius = max(0.03, radius * 0.5)
            stall = 20
            if stall > 150:
                break
        
        S = _corr_scaled(h)
        p = None
        if use_lse:
            p = _lse_direction(h, S, n, tau, radius)
            tau *= 0.98  # Gradually cool
            if tau < 0.001:
                tau = 0.001
        if p is None or np.max(np.abs(p)) < 1e-13:
            p = _lp_direction(h, S, n, K, radius)
        
        if p is None or np.max(np.abs(p)) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.08, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.03, 0.25)
            else:
                radius = max(radius * 0.93, 0.02)
        else:
            stall += 1
            radius *= 0.78
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.04
    
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        M = np.max(S)
        active = np.where(S >= M - 0.05)[0]
        G = _grad_rows(h, active, n)
        total_grad = np.sum(G, axis=0) / len(active)
        total_grad -= total_grad.mean()
        
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):

    dx = 2.0 / n

                else:

                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

# --- Resampling ---

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

# --- Sine initialization ---

def _sine_init(n, freq, phase):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

# --- Structured perturbation ---

def _sine_perturb(h, rng, n_modes=5):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.15)
        pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h + pert, n / 2.0)

# --- Multi-start helper ---

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks < 2:
        return _project(h + 0.5 + 0.5*rng.standard_normal(n), n/2.0)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)
