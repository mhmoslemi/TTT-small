def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    # Project v onto {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _line_search(S0, T, Q, dx, L, U):
    # Minimize max_k (S0[k] + t*T[k] - t^2*Q[k]) * dx over t in [L, U]
    if U - L < 1e-14:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t*T - t*t*Q)) * dx
    
    # Grid search + golden section
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*Q[None,:]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i-1, 0)]
    b = grid[min(i+1, nq-1)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr*(b - a)
    c2 = a + gr*(b - a)
    f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b - a)
            f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b - a)
            f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx
    t = 0.5*(a + b)
    tv = float(np.max(S0 + t*T - t*t*Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_round(h, rng):

    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b
