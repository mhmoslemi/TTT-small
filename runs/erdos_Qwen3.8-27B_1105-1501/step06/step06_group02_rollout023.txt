def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
import time
from scipy.optimize import minimize, differential_evolution

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _rows_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _soft_obj_grad(x, T, tau):
    n = len(x)
    rows = _rows_of(x)
    z = (rows - T) / tau
    z_max = float(np.max(z))
    exp_z = np.exp(z - z_max)
    s_exp = float(np.sum(exp_z))
    obj = tau * (z_max + np.log(s_exp))
    w = exp_z / s_exp
    grad = np.zeros(n)
    n_half = n - 1
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - n_half
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            grad[:m] += wk * (1.0 - x[d:d + m])
            grad[d:] += -wk * x[:m]
        else:
            r = -d
            grad[r:] += wk * (1.0 - x[:m])
            grad[:m] += -wk * x[r:r + m]
    grad *= dx
    return obj, grad

def _sum_eq(x):
    return float(np.sum(x)) - len(x) / 2.0

def _sum_eq_jac(x):
    return np.ones(len(x))

def _sine_init(n, freq, phase, amp):
    x = (np.arange(n) + 0.5) / n
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    return project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _golden_line_search(h, p, dx, S0):
    n = len(h)
    pos = p > 1e-12
    neg = p < -1e-12
    hi = 0.5
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / p[pos])))
    lo = 0.0
    if np.any(neg):
        lo = max(lo, -float(np.min(h[neg] / p[neg])))
    if hi - lo < 1e-13:
        return 0.0, float(np.max(S0)) * dx
    A = np.correlate(p, 1.0 - h, mode='full')
    B = np.correlate(h, p, mode='full')
    C = np.correlate(p, p, mode='full')
    T = A - B
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx

        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:

    if f_hi < val:

    return opt, val

def _fourier_from_coeffs(n, coeffs):
    x = (np.arange(n) + 0.5) / n
    h = np.zeros(n)
    h += 0.5
    K = len(coeffs) // 2
    for k in range(K):
        a = coeffs[2 * k]
        b = coeffs[2 * k + 1]
        h += a * np.cos(np.pi * (k + 1) * 2.0 * x)
        h += b * np.sin(np.pi * (k + 1) * 2.0 * x)
    return project(h, n / 2.0)

def _fourier_eval(z, n, K):
    coeffs = z[:2 * K]
    h = _fourier_from_coeffs(n, coeffs)
    return c5_exact(h)

def _fourier_de(n, K, deadline, seed):
    D = 2 * K
    bounds = [(-0.6, 0.6)] * D
    popsize = max(10, min(30, 2 * D + 4))
    maxiter = max(50, min(200, int((deadline - time.time()) / 0.5)))
    res = differential_evolution(
        _fourier_eval, bounds, args=(n, K),
        seed=seed, popsize=popsize, maxiter=maxiter,
        tol=1e-8, polish=True, init='halton'
    )
    if res.fun < 0.5:
        h = _fourier_from_coeffs(n, res.x[:2 * K])
        return h, float(res.fun)
    return None, float('inf')

    for ci in range(min(3, len(candidates))):
        if time.time() > d2:
            break
        h_c, c5_c, n_c = candidates[ci]
        if n_c != best_n:
            h_c = _resample(h_c, best_n)
            c5_c = c5_exact(h_c)
        sub_deadline = min(d2, time.time() + budget_s * 0.15)
        h_sm, c5_sm = _smooth_solve(h_c, c5_c, sub_deadline)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()
    
    # Phase 3: Directional polish (12% time)
    d3 = t0 + budget_s * 0.64
    h_dp, c5_dp = _directional_polish(best_h, d3)
    if c5_dp < best_c5:
        best_c5 = c5_dp
        best_h = h_dp.copy()
    
    # Phase 4: Pair polish (12% time)
    d4 = t0 + budget_s * 0.76
    h_pp, c5_pp = _pair_polish(best_h, d4)
    if c5_pp < best_c5:
        best_c5 = c5_pp
        best_h = h_pp.copy()
    
    # Phase 5: Multi-resolution (14% time)
    d5 = t0 + budget_s * 0.90
    tried_n = {best_n}
    for dn in [-8, 8, -6, 6, -12, 12, -4, 4]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = c5_exact(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.03)
        h_sm_r, c5_sm_r = _smooth_solve(h_r, c5_r, sub_dl)
        if c5_sm_r < best_c5:
            sub_dl2 = min(d5, time.time() + budget_s * 0.02)
            h_sm_r, c5_sm_r = _directional_polish(h_sm_r, sub_dl2)
            if c5_sm_r < best_c5:
                best_c5 = c5_sm_r
                best_h = h_sm_r.copy()
                best_n = n_try
    
    # Phase 6: Final smooth + polish (4% time)
    d6 = deadline
    h_final = best_h.copy()
    sub_dl = min(d6, time.time() + budget_s * 0.02)
    h_sf, c5_sf = _smooth_solve(h_final, best_c5, sub_dl, max_rounds=3)
    if c5_sf < best_c5:
        best_c5 = c5_sf
        best_h = h_sf.copy()
    
    sub_dl2 = min(d6, time.time() + budget_s * 0.015)
    h_df, c5_df = _directional_polish(best_h, sub_dl2, n_max_iter=30)
    if c5_df < best_c5:
        best_c5 = c5_df
        best_h = h_df.copy()
    
    # Ensure validity
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    
    return best_h, final_c5, len(best_h)
