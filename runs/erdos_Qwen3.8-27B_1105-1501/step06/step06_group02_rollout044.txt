Let me also think about the QP approach. The idea is:
- Compute all 2n-1 correlation values
- Want to minimize the maximum
- This is equivalent to: find d such that max_k (R_k + d^T G_k) is minimized, subject to box constraints and sum(d)=0

This can be formulated as: minimize t subject to G_k^T d + R_k <= t for all k, ||d||_2 <= rho (trust region), box constraints, sum(d)=0.

This is a QP that can be solved with scipy.

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    # Project v onto {0 <= x <= 1, sum(x) = target}
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads_full(h):
    """Compute gradient matrix for all 2n-1 correlation rows.
    Returns array of shape (2n-1, n)."""
    n = len(h)
    m_total = 2 * n - 1
    G = np.zeros((m_total, n))
    for k in range(m_total):
        d = k - (n - 1)  # shift
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            if d == 0:
                G[k, :m] = 1.0 - 2.0 * h[:m]
            else:
                G[k, d:d+m] += (1.0 - h[:m])
                G[k, :m] += -h[d:d+m]
        else:
            g = -d
            G[k, g:] += (1.0 - h[:m])
            G[k, :m] += -h[g:g+m]
    return G

def qp_tangent_step(h, rho, target_c5):
    """Solve QP tangent step to reduce max correlation below target_c5.
    Returns direction d (zero-sum, box-feasible) or None."""
    n = len(h)
    m_total = 2 * n - 1
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')  # raw correlation values
    target_raw = target_c5 / dx  # target in raw units
    
    G = compute_grads_full(h)
    
    # QP: minimize 0.5 * d^T d + mu * max_k (R_k + G_k^T d - target_raw, 0)
    # SLP approximation: minimize 0.5*d^T d + mu * sum_k z_k
    # s.t. G_k^T d + R_k - target_raw - z_k <= 0, z_k >= 0, sum(d)=0, box(d)
    
    # Use scipy minimize with SLSQP
    # Variables: d (n-vector), z (m_total-vector)
    
    # Box bounds for d
    d_lo = -np.minimum(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    
    z_lo = np.zeros(m_total)
    z_hi = np.full(m_total, max(R.max() - target_raw, 0) + rho + 1.0)
    
    x0 = np.concatenate([np.zeros(n), np.maximum(R - target_raw, 0)])
    
    mu = 1.0
    
    def objective(x):
        d = x[:n]
        z = x[n:]
        return 0.5 * np.dot(d, d) + mu * np.sum(z)
    
    def gradient(x):
        d = x[:n]
        z = x[n:]
        grad_d = d.copy()
        grad_z = np.full(m_total, mu)
        return np.concatenate([grad_d, grad_z])
    
    constraints = []
    # sum(d) = 0
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]),
        'jac': lambda x: np.concatenate([np.ones(n), np.zeros(m_total)])
    })
    # G_k^T d + R_k - target_raw - z_k <= 0  =>  -(G_k^T d + R_k - target_raw - z_k) >= 0
    for k in range(m_total):
        row_G = G[k].copy()
        rhs = target_raw - R[k]
        idx_z = n + k
        def conj(x, k=k, row_G=row_G, rhs=rhs, idx_z=idx_z):
            return -(row_G @ x[:n] + R[k] - target_raw - x[idx_z])
        def conj_jac(x, k=k, row_G=row_G, idx_z=idx_z):
            j = np.zeros(n + m_total)
            j[:n] = -row_G
            j[idx_z] = 1.0
            return j
        constraints.append({'type': 'ineq', 'fun': conj, 'jac': conj_jac})
    
    bounds = list(zip(d_lo, d_hi)) + list(zip(z_lo, z_hi))
    
    res = minimize(objective, x0, jac=gradient, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 300, 'ftol': 1e-12})
    
    if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
        d = res.x[:n].copy()
        d -= np.mean(d)  # ensure zero sum
        # Check box feasibility
        d = np.clip(d, d_lo + 1e-10, d_hi - 1e-10)
        d -= np.mean(d)
        if np.max(np.abs(d)) > 1e-12:
            return d
    return None
