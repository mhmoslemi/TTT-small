def _exact_c5(h):

def _project(v, target):

    if target >= n:

        if s > target:

        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):

        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)

    return x

def _row_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:

    else:
        nn = np.arange(-k, n)
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g

    return g

        if h_new.min() < -1e-10 or h_new.max() > 1.0 + 1e-10:
            h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < best_f - 1e-14:
            best_f = f_new
            best_h = h_new.copy()
        if f_new < best_f + 5e-4:
            h = h_new
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
        if stall > 120:
            h = _project(best_h + rng.normal(0, sigma, n), target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            best_f = _exact_c5(h)
            best_h = h.copy()
            stall = 0
            sigma = max(0.005, sigma * 0.85)
    return best_h, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 5 + 1))))
        bl = min(bl, n - pos)
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n0 = 122
        x = np.arange(n0)
        base = 0.5 + 0.25 * np.sin(2.0 * np.pi * x / n0) + rng.normal(0, 0.03, n0)

    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Try complement
    h_comp = _project(1.0 - h0, n0 / 2.0)
    f_comp = _exact_c5(h_comp)
    if f_comp < best_f:
        best_f = f_comp
        best_h = h_comp.copy()

    # Phase 1: SLSQP epigraph (28% of budget)
    d1 = t0 + budget_s * 0.28
    slsqp_configs = [
        (0.5, 200), (0.6, 200), (1.0, 300), (0.4, 200), (0.7, 250)
    ]
    for frac, mit in slsqp_configs:
        if time.time() > d1:
            break
        h_try, f_try = _slsqp_solve(best_h, frac, mit)
        if f_try < best_f:
            best_f = f_try
            best_h = h_try.copy()
        if time.time() > d1:
            break
        h_try2, f_try2 = _slsqp_solve(_project(1.0 - best_h, n0 / 2.0), frac, mit)
        if f_try2 < best_f:
            best_f = f_try2
            best_h = h_try2.copy()

    # Phase 2: Smooth max optimization (22% of budget)
    d2 = t0 + budget_s * 0.50
    if time.time() < d2:
        h2, f2 = _smooth_optimize(best_h, d2, alpha_start=30.0, alpha_end=180.0)
        if f2 < best_f:
            best_f = f2
            best_h = h2.copy()

    # Phase 3: Local optimization with active directions (28% of budget)
    d3 = t0 + budget_s * 0.78
    if time.time() < d3:
        h3, f3 = _local_optimize(best_h, d3, rng, k_active=10, tau=0.014, pair_frac=0.25, rand_frac=0.08)
        if f3 < best_f:
            best_f = f3
            best_h = h3.copy()

    # Phase 4: Multi-resolution (14% of budget)
    d4 = t0 + budget_s * 0.92
    bn = len(best_h)
    tried_ns = set()
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, 12, -12]:
        if time.time() > d4:
            break
        n_try = bn + dn
        if n_try < 60 or n_try > 240 or n_try % 2 != 0:
            continue
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        hr = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + max(3.0, budget_s * 0.02))
        hr2, fr2 = _local_optimize(hr, sub_dl, rng, k_active=8, tau=0.018, pair_frac=0.35, rand_frac=0.1)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 5: Final polish (8% of budget)
    h_fin, f_fin = _local_optimize(best_h, deadline, rng, k_active=12, tau=0.010, pair_frac=0.45, rand_frac=0.05)
    if f_fin < best_f:
        best_f = f_fin
        best_h = h_fin.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
