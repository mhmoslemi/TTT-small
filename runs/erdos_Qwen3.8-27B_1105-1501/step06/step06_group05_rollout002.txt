def _resample(h, n_new):

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _block_init(n, rng):
    # Create a blocky pattern
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(2, n // 6 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _binary_init(n, rng):
    """Random binary initialization."""
    m = n // 2
    h = np.zeros(n)
    ones = rng.choice(n, size=m, replace=False)
    h[ones] = 1.0
    return h
