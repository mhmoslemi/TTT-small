import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        # Lag +r: S[q] = dx * sum_{t=0}^{m-1} h[t] * (1 - h[t+r])
        # dS/dh[t] for t in [0, m-1]: dx * (1 - h[t+r])
        # dS/dh[t+r] for t in [0, m-1]: -dx * h[t]
        q_pos = mid + r
        G[q_pos, :m] += dx * (1.0 - h[r:r+m])
        G[q_pos, r:r+m] += -dx * h[:m]
        # Lag -r: S[q] = dx * sum_{t=0}^{m-1} h[t+r] * (1 - h[t])
        q_neg = mid - r
        G[q_neg, r:r+m] += dx * (1.0 - h[:m])
        G[q_neg, :m] += -dx * h[r:r+m]
    return G

def _pair_line_search(h, i, j, dx):

        active_thresh = S_max - 0.005
        active_lags = np.where(S > active_thresh)[0]

        # For each active lag, find best source/sink

            # Source: high positive gradient (decreasing helps), with h > 0
            # Sink: high negative gradient (increasing helps), with h < 1

        if not improved:
            # Try random pairs as fallback
            for _ in range(30):
                ii = rng.integers(n)
                jj = rng.integers(n)
                if ii == jj or (ii, jj) in tried:
                    continue
                tried.add((ii, jj))
                for (src, snk) in [(ii, jj), (jj, ii)]:
                    if h[src] < 0.01 or h[snk] > 0.99:
                        continue
                    a, c5_try = _pair_line_search(h, snk, src, dx)
                    if c5_try < best_c5 - 1e-14:
                        h[snk] += a
                        h[src] -= a
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved = True
                            break
                if improved:
                    break
        if improved:
            stall = 0
        else:
            stall += 1
        if stall % 30 == 0:
            # Perturb to escape
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            stall = max(0, stall - 10)
    return best_h, best_c5

def _lp_dir(h, G, S, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize(h, deadline, start_radius=0.10, max_iters=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = start_radius
    stall = 0
    rng = np.random.default_rng(999)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        S = _corr_vals(h)
        G = _build_G(h, dx)
        p = _lp_dir(h, G, S, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            if stall > 60:
                break
            continue
        # Line search along p
        d = p / max(float(np.max(np.abs(p))), 1e-15)
        a, new_c5 = _pair_line_search(h, 0, 0, dx)  # placeholder, do full search
        # Full line search for general direction
        S0 = _corr_vals(h)
        T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
        C = np.correlate(d, d, mode='full') * dx
        # Feasible range
        pos = d > 1e-14
        neg = d < -1e-14
        t_lo, t_hi = -radius, radius
        if np.any(pos):
            t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
            t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
        if t_hi - t_lo < 1e-14:
            stall += 1
            radius *= 0.7
            continue
        # Golden section on [t_lo, t_hi]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_g, b_g = t_lo, t_hi
        c1_g = b_g - gr * (b_g - a_g)
        c2_g = a_g + gr * (b_g - a_g)
        f1 = float(np.max(S0 + c1_g * T - c1_g*c1_g * C))
        f2 = float(np.max(S0 + c2_g * T - c2_g*c2_g * C))
        for _ in range(30):
            if b_g - a_g < 1e-13:
                break
            if f1 < f2:
                b_g, c2_g, f2 = c2_g, c1_g, f1
                c1_g = b_g - gr * (b_g - a_g)
                f1 = float(np.max(S0 + c1_g * T - c1_g*c1_g * C))
            else:
                a_g, c1_g, f1 = c1_g, c2_g, f2
                c2_g = a_g + gr * (b_g - a_g)
                f2 = float(np.max(S0 + c2_g * T - c2_g*c2_g * C))
        opt_t = (a_g + b_g) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C))
        if opt_val < cur_c5 - 1e-14:
            h = h + opt_t * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Adapt radius
            predicted = float(np.max(S + G @ d * opt_t))
            ratio = cur_c5 / max(predicted, 1e-15)
            if ratio < 0.7:
                radius = min(radius * 1.2, 0.30)
            elif ratio > 1.6:
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.8
            if stall % 40 == 0 and radius < 0.03:
                h = h + rng.normal(0, 0.015, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
        if stall > 80:
            break
    return best_h, best_c5

def _anneal(h, dx, target, deadline, rng, T_init=0.01, T_end=0.0001):
    """Simulated annealing with pair moves."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    T = T_init
    n_iter = 0
    while time.time() < deadline:
        n_iter += 1
        T = T_init * (T_end / T_init) ** (min(n_iter, 10000) / 10000.0)
        # Pick random pair
        ii = rng.integers(n)
        jj = rng.integers(n)
        if ii == jj:
            continue
        # Random transfer
        max_transfer = min(h[ii], 1.0 - h[jj]) - 0.01
        if max_transfer < 0.005:
            continue
        a = rng.uniform(-max_transfer, max_transfer)
        if abs(a) < 1e-4:
            continue
        h_new = h.copy()
        h_new[ii] += a
        h_new[jj] -= a
        if h_new[ii] < -1e-10 or h_new[ii] > 1.0 + 1e-10:
            continue
        if h_new[jj] < -1e-10 or h_new[jj] > 1.0 + 1e-10:
            continue
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def _smooth_lse_descent(h, deadline, tau_init=0.005, max_iters=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_init
    lr = 0.003
    for it in range(max_iters):
        if time.time() > deadline:
            break
        S = _corr_vals(h)
        S_max = float(np.max(S))
        w = np.exp((S - S_max) / tau)
        w_sum = float(np.sum(w))
        if w_sum <= 0:
            break
        w /= w_sum
        G = _build_G(h, dx)
        grad = G.T @ w
        # Project gradient to sum=zero direction
        grad -= float(np.mean(grad))
        h = h - lr * grad
        h = _project(h, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            lr = min(lr * 1.03, 0.015)
        else:
            lr *= 0.92
        if it % 30 == 0:
            tau *= 0.95
            lr = max(lr, 0.001)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_init(n, rng, mode=0):
    """Generate diverse initial conditions."""
    if mode == 0:
        freq = rng.integers(1, 7)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 1:
        # Binary random
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    elif mode == 2:
        # Blocks
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, int(rng.integers(1, n // 4 + 1)))
            if rng.random() < 0.5:
                h[pos:pos+bl] = 1.0
            pos += bl
    elif mode == 3:
        # Two-frequency
        f1, f2 = rng.integers(1, 5), rng.integers(2, 8)
        x = np.arange(n)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * f1 * x / n) + 0.2 * np.cos(2 * np.pi * f2 * x / n)
    else:
        h = 0.5 + rng.normal(0, 0.2, n)
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / n0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Phase 1: Quick coordinate descent polish (15% budget)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _greedy_pair_cd(best_h, 2.0/best_n, best_n/2.0, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP optimization (25% budget)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _lp_optimize(best_h, d2, start_radius=0.12)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Follow up with CD
    if time.time() < d2 + budget_s * 0.03:
        h2b, c5_2b = _greedy_pair_cd(best_h, 2.0/best_n, best_n/2.0, d2 + budget_s*0.03, rng)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Smooth LSE descent (10% budget)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _smooth_lse_descent(best_h, d3, tau_init=0.004)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Polish with CD
    if time.time() < d3 + budget_s * 0.02:
        h3b, c5_3b = _greedy_pair_cd(best_h, 2.0/best_n, best_n/2.0, d3 + budget_s*0.02, rng)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Simulated annealing (15% budget)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _anneal(best_h, 2.0/best_n, best_n/2.0, d4, rng, T_init=0.008, T_end=0.00005)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Polish after annealing
    if time.time() < d4 + budget_s * 0.03:
        h4b, c5_4b = _greedy_pair_cd(best_h, 2.0/best_n, best_n/2.0, d4 + budget_s*0.03, rng)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-resolution exploration (15% budget)
    d5 = t0 + budget_s * 0.80
    n_best = best_n
    tried_n = {n_best}
    offsets = [-6, 6, -8, 8, -10, 10, -12, 12, 16, -16, 20, -20, -14, 14]
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, start_radius=0.10)
        if time.time() < d5:
            h_r, c5_r2 = _greedy_pair_cd(h_r, 2.0/n_try, n_try/2.0, min(d5, time.time()+budget_s*0.008), rng)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r.copy()
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 6: Diverse starts with full pipeline (12% budget)
    d6 = deadline
    n_starts = 4
    for si in range(n_starts):
        if time.time() > d6:
            break
        sub_budget = (d6 - time.time()) / max(n_starts - si, 1)
        if sub_budget < 10:
            break
        h_s = _diverse_init(n_best, rng, mode=si % 4)
        sub_dl = time.time() + sub_budget * 0.4
        h_s, c5_s = _lp_optimize(h_s, sub_dl, start_radius=0.10)
        if time.time() < d6:
            h_s, c5_s = _greedy_pair_cd(h_s, 2.0/n_best, n_best/2.0, min(d6, time.time()+sub_budget*0.3), rng)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Final polish
    if time.time() < deadline:
        h_f, c5_f = _greedy_pair_cd(best_h, 2.0/best_n, best_n/2.0, deadline, rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
