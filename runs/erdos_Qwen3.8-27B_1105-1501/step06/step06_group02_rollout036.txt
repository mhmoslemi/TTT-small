import time
from scipy.optimize import linprog

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

        else:

def exact_line_search(h, d, S0, dx, max_iter=60):
    """Exact line search along direction d. Returns (best_alpha, best_c5)."""
    n = len(h)
    # Compute T and C
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Find feasible alpha range
    lo = -1.0
    hi = 1.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    
    # Grid search
    n_grid = 60
    grid_alphas = np.linspace(lo, hi, n_grid)
    best_a = lo
    best_val = np.inf
    for a in grid_alphas:
        val = float(np.max(S0 + a * T - a * a * C)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Golden section refinement
    # Find bracket around best_a
    idx = int(np.argmin([float(np.max(S0 + a*T - a*a*C)) for a in grid_alphas]))
    if idx > 0 and idx < n_grid - 1:
        a_lo = grid_alphas[idx - 1]
        a_hi = grid_alphas[idx + 1]
    else:
        a_lo = lo
        a_hi = hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = a_hi - gr * (a_hi - a_lo)
    a2 = a_lo + gr * (a_hi - a_lo)
    f1 = float(np.max(S0 + a1 * T - a1 * a1 * C)) * dx
    f2 = float(np.max(S0 + a2 * T - a2 * a2 * C)) * dx
    
    for _ in range(40):
        if a_hi - a_lo < 1e-12:
            break
        if f1 < f2:
            a_hi = a2
            a2 = a1
            f2 = f1
            a1 = a_hi - gr * (a_hi - a_lo)
            f1 = float(np.max(S0 + a1 * T - a1 * a1 * C)) * dx
        else:
            a_lo = a1
            a1 = a2
            f1 = f2
            a2 = a_lo + gr * (a_hi - a_lo)
            f2 = float(np.max(S0 + a2 * T - a2 * a2 * C)) * dx
    
    opt = (a_lo + a_hi) / 2.0
    opt_val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    
    # Check endpoints
    for a in [lo, hi]:
        val = float(np.max(S0 + a * T - a * a * C)) * dx
        if val < opt_val:
            opt, opt_val = a, val
    
    if opt_val < best_val:
        best_a, best_val = opt, opt_val
    
    return best_a, best_val

    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = G @ h - R
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    low = np.maximum(0.0, h - radius)
    high = np.minimum(1.0, h + radius)
    
    bounds = [(None, None)] + list(zip(low, high))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    v = res.x[1:]
    d = v - h
    # Ensure zero sum (should be from equality constraint)
    d -= np.mean(d)
    
    if np.linalg.norm(d) < 1e-12:
        return None
    
    return d

def gradient_direction(h, G, R):
    """Generate a gradient-based direction as fallback."""
    n = len(h)
    max_R = float(np.max(R))
    threshold = max_R - 0.01 * max_R
    active = R >= threshold
    if not np.any(active):
        active = np.zeros(len(R), dtype=bool)
        active[np.argsort(-R)[:10]] = True
    
    # Weighted gradient of active rows
    weights = np.zeros(len(R))
    weights[active] = 1.0
    weights /= np.sum(weights)
    
    g = weights @ G  # shape (n,)
    g -= np.mean(g)  # zero sum
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _gradient_polish(h, deadline):
    """Polish using gradient direction with line search."""
    n = len(h)
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, 'full')
    G = build_G(h)
    R = S0
    stall = 0
    
    while time.time() < deadline and stall < 30:
        d = gradient_direction(h, G, R)
        if d is None:
            break
        
        alpha, new_c5 = exact_line_search(h, d, S0, dx)
        
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-14:

            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, 'full')
            G = build_G(h)
            R = S0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _pair_search(h, deadline, n_pairs=20):
    """Directed pair search from parent (as fallback)."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, 'full')
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.time() < deadline and stall < 60:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.005 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]
        
        # Compute weighted gradient for active rows
        G_act = build_G(h)[active_idx]
        w = np.exp((scores[active_idx] - max_score) / (0.01 * max_score + 1e-15))
        w /= np.sum(w)
        wgrad = G_act.T @ w
        
        donor_pool = np.argsort(-wgrad)[:n_pairs]
        recv_pool = np.argsort(wgrad)[:n_pairs]
        
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        
        for i in donor_pool:
            if time.time() > deadline:
                break
            for j in recv_pool:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, new_c5 = exact_line_search(h, d, S0, dx)
                if new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, 'full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.3:
                idx = rng.integers(n, size=4)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.05, 0.05)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, 'full')
                stall = 0
    
    return best_h, best_c5

def _lp_descent(h, deadline, radii=None):
    """Main LP trust-region descent."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()

                h = project(h + alpha * d, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, 'full')
                G = build_G(h)
                R = S0.copy()
                if cur_c5 < best_c5:

                    best_h = h.copy()
                improved_this_round = True
                stall = 0
                total_stall = 0

            total_stall += 1
            if total_stall > 20:

                G = build_G(h)
                R = S0.copy()

def c5_exact(h):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:

def build_G(h):

            G[i, :m] += 1.0 - h[d:d + m]
            G[i, d:d + m] += -h[:m]
        else:
            q = -d
            m = n - q
            G[i, q:] += 1.0 - h[:m]

    g = w @ G
    g -= np.mean(g)
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm


def resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)


def _lp_descent(h, deadline, radii=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5

    if radii is None:
        radii = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]

    S0 = np.correlate(h, 1.0 - h, 'full')
    G = build_G(h)
    R = S0.copy()
    total_stall = 0
    rng = np.random.default_rng(42)

    while time.time() < deadline:
        improved = False
        for radius in radii:
            if time.time() > deadline:
                break
            d = lp_step(h, G, R, radius)
            if d is None:
                d = gradient_direction(h, G, R)
                if d is None:
                    continue
                dn = np.linalg.norm(d)
                if dn > 0:
                    d = d * radius / dn

            alpha, pred_c5 = exact_line_search(h, d, S0, dx)
            if alpha < 1e-10:
                continue

            h_cand = project(h + alpha * d, target)
            actual_c5 = c5_exact(h_cand)

            if actual_c5 < cur_c5 - 1e-15:
                h = h_cand
                cur_c5 = actual_c5
                S0 = np.correlate(h, 1.0 - h, 'full')
                G = build_G(h)
                R = S0.copy()
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                total_stall = 0
                break

        if not improved:
            total_stall += 1
            if total_stall > 12:
                n_pert = max(4, n // 15)
                idx = rng.integers(0, n, size=n_pert)
                for k in idx:
                    h[k] += rng.uniform(-0.10, 0.10)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, 'full')
                G = build_G(h)
                R = S0.copy()
                total_stall = 0
            if total_stall > 40:
                break

    return best_h, best_c5


def _gradient_polish(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, 'full')
    G = build_G(h)
    R = S0.copy()
    stall = 0

    while time.time() < deadline and stall < 40:
        d = gradient_direction(h, G, R)
        if d is None:
            break
        alpha, pred = exact_line_search(h, d, S0, dx)
        if alpha > 1e-10:
            h_cand = project(h + alpha * d, target)
            actual = c5_exact(h_cand)
            if actual < cur_c5 - 1e-15:
                h = h_cand
                cur_c5 = actual
                S0 = np.correlate(h, 1.0 - h, 'full')
                G = build_G(h)
                R = S0.copy()
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                continue
        stall += 1

    return best_h, best_c5


def _pair_search(h, deadline, n_pairs=15):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, 'full')
    stall = 0
    rng = np.random.default_rng(123)

    while time.time() < deadline and stall < 80:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.005 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]

        G_all = build_G(h)
        G_act = G_all[active_idx]
        w = np.exp((scores[active_idx] - max_score) / (0.01 * max_score + 1e-15))
        w /= np.sum(w)
        wgrad = G_act.T @ w

        donor = np.argsort(-wgrad)[:n_pairs]
        recv = np.argsort(wgrad)[:n_pairs]

        best_alpha = 0.0
        best_new = cur_c5
        best_pair = None

        for i in donor:
            if time.time() > deadline:
                break
            for j in recv:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, nc5 = exact_line_search(h, d, S0, dx)
                if alpha > 1e-10:
                    h_cand = project(h + alpha * d, target)
                    actual = c5_exact(h_cand)
                    if actual < best_new - 1e-14:
                        best_new = actual
                        best_alpha = alpha
                        best_pair = (i, j)

        if best_pair is not None and best_new < cur_c5 - 1e-14:
            i, j = best_pair
            h = project(h + best_alpha * (np.zeros(n) - 1.0 * np.eye(n)[i] + 1.0 * np.eye(n)[j]), target) if False else h
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, 'full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20 and rng.random() < 0.4:
                idx = rng.integers(n, size=5)
                for k in idx:
                    h[k] += rng.uniform(-0.06, 0.06)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, 'full')
                stall = 0

    return best_h, best_c5


def _binary_optimize(h, deadline, rng):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    cur_c5 = c5_exact(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin <= 0.5)[0]
    patience = 0

    while time.time() < deadline and patience < 4000:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        nc5 = c5_exact(h_bin)
        if nc5 < cur_c5 - 1e-15:
            cur_c5 = nc5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
                patience = 0
            else:
                patience += 1
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin <= 0.5)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            patience += 1

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.45 * np.sin(2 * np.pi * 2 * x / n0 + 0.5)

    n = len(h0)
    h = project(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n
    rng = np.random.default_rng(seed)

    # Phase 1: LP descent (45%)
    d1 = t0 + budget_s * 0.45
    radii1 = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    h1, c5_1 = _lp_descent(h, d1, radii=radii1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Gradient polish (8%)
    d2 = t0 + budget_s * 0.53
    h2, c5_2 = _gradient_polish(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Second LP with finer radii (15%)
    d3 = t0 + budget_s * 0.68
    radii2 = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4]
    h3, c5_3 = _lp_descent(best_h, d3, radii=radii2)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (12%)
    d4 = t0 + budget_s * 0.80
    tried_n = {best_n}
    for dn in [0, -4, 4, -6, 6, -8, 8, -2, 2]:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_h(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _lp_descent(h_r, sub_dl, radii=[0.15, 0.06, 0.02, 0.008])
        if c5_r < best_c5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r, c5_r = _gradient_polish(h_r, sub_dl2)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 5: Pair search (7%)
    d5 = t0 + budget_s * 0.87
    h5, c5_5 = _pair_search(best_h, d5, n_pairs=12)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Binary optimization (4%)
    d6 = t0 + budget_s * 0.91
    h6, c5_6 = _binary_optimize(best_h, d6, rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)

    # Phase 7: Final LP polish (4%)
    d7 = deadline
    h7, c5_7 = _lp_descent(best_h, d7, radii=[0.05, 0.02, 0.01, 0.004, 0.001])
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final projection and exact evaluation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)

    return best_h, final_c5, len(best_h)
