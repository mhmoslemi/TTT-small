def _binary_sa(h_init, deadline, rng, n_swaps_per_iter=50):

    T = 0.004
    T_min = 0.000003
    cooldown = 0.99995
    patience = 0
    max_patience = 30000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]

            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        for _ in range(n_swaps_per_iter):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _sine2_init(n, f1, f2, p1, p2, amp):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2 * np.pi * f1 * x / n + p1) + (1-amp) * np.sin(2 * np.pi * f2 * x / n + p2)
    h = 0.5 + 0.3 * h  # rescale to be in reasonable range
    return _project(h, n / 2.0)

def _periodic_block_init(n, period, duty):
    """Create a periodic block pattern."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        if phase < duty:
            h[i] = 1.0
    return _project(h, n / 2.0)

def _frank_wolfe(h, deadline, max_iters=500):
    """Frank-Wolfe style: find vertex of feasible set that minimizes linearization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

            break
        if stall > 30:
            break
        S = _corr_scaled(h)
        M = float(S.max())
        # Soft weights
        tau = 0.01 * M
        shifted = (S - M) / tau
        shifted -= shifted.max()
        w = np.exp(shifted)
        w = w / w.sum()
        
        # Gradient of soft-max
        top_k = min(30, 2*n-1)
        top_idx = np.argsort(-w)[:top_k]
        G = _grad_rows(h, top_idx, n)
        grad = w[top_idx] @ G
        grad -= grad.mean()
        
        # Frank-Wolfe: minimize grad^T * v over v in [0,1]^n, sum(v) = target
        # This is: pick top target entries of -grad
        neg_grad = -grad
        fw_indices = np.argsort(-neg_grad)[:int(target)]
        v = np.zeros(n)
        v[fw_indices] = 1.0
        
        # Line search: h_new = (1-t)*h + t*v
        # This automatically preserves sum and bounds

    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.14, K=15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline:
        if stall > 90:
            if stall > 150:
                break
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.08
            stall = 80
            continue
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            tau = 0.012 * max(cur_c5, 0.1)
            grad = _soft_grad(h, n, tau)
            grad -= grad.mean()
            norm = np.linalg.norm(grad)
            if norm < 1e-12:
                stall += 1
                radius *= 0.75
                if radius < 1e-6:
                    break
                continue
            p = -grad / norm * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.65:
                radius = min(radius * 1.05, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.93, 0.02)
        else:
            stall += 1
            radius *= 0.74
            if stall % 18 == 0 and radius < 0.02:
                radius = 0.07
    return best_h, best_c5

def _soft_loop(h, deadline, tau_init=0.012):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_init
    radius = 0.12
    stall = 0
    
    while time.time() < deadline and stall < 100:
        grad = _soft_grad(h, n, tau)
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        d = -grad / norm * radius
        alpha, new_c5 = _line_search(h, d, 2.0 / n, radius)
        cur = _exact_c5(h)
        if new_c5 < cur - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            tau = max(0.002, tau * 0.97)
            radius = min(radius * 1.04, 0.28)
        else:
            stall += 1
            radius *= 0.8
            if stall % 12 == 0:
                tau = min(0.04, tau * 1.15)
                radius = max(radius, 0.04)
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000002
    patience = 0
    max_patience = 40000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 4:
                break
            T = max(T_min, T * 0.8)
            patience = 0
        if T < T_min:
            break
        n_swaps = 80
        for _ in range(n_swaps):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * 0.99994)
    return best_h, best_c5

def _multi_swap_sa(h_init, deadline, rng, swap_size=3):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.01
    T_min = 0.0001
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 5:
                break
            T = max(T_min, T * 0.7)
            patience = 0
        if T < T_min:
            break
        for _ in range(40):
            if len(ones_list) < swap_size or len(zeros_list) < swap_size:
                break
            idx_ones = rng.choice(len(ones_list), size=swap_size, replace=False)
            idx_zeros = rng.choice(len(zeros_list), size=swap_size, replace=False)
            old_ones = ones_list[idx_ones]
            old_zeros = zeros_list[idx_zeros]
            h[old_ones] = 0.0
            h[old_zeros] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[old_ones] = 1.0
                h[old_zeros] = 0.0
        T = max(T_min, T * 0.9998)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=400):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        S = _corr_scaled(h)
        G = _grad_rows(h, np.arange(2 * n - 1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        
        n_try = min(25, n)
        pos_top = np.argsort(-total_grad)[:n_try]
        neg_top = np.argsort(total_grad)[:n_try]
        
        improved = False
        for i in pos_top:
            if stall == 0:
                break
            for j in neg_top:
                if i == j:
                    continue
                if h[i] <= 0.01 or h[j] >= 0.99:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        stall = 0
                        break
        if not improved:
            stall += 1
    return best_h, best_c5

def _frank_wolfe(h, deadline, max_iters=150):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 25:
            break
        S = _corr_scaled(h)
        tau = 0.01 * max(float(S.max()), 0.1)
        grad = _soft_grad(h, n, tau)
        grad -= grad.mean()
        
        neg_grad = -grad
        fw_idx = np.argsort(-neg_grad)[:int(target)]
        v = np.zeros(n)
        v[fw_idx] = 1.0
        d = v - h
        
        best_t = 0.0
        best_val = _exact_c5(h)
        for t_test in np.linspace(0, 1, 15):
            h_test = (1.0 - t_test) * h + t_test * v
            h_test = _project(h_test, target)
            val = _exact_c5(h_test)
            if val < best_val:
                best_val = val
                best_t = t_test
        if best_t > 0.01 and best_val < best_c5 - 1e-14:
            h = (1.0 - best_t) * h + best_t * v
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 120
        h0 = _sine_init(n0, 2, 0.5)
    
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    # Phase 1: Quick multi-start (6% time)
    d1 = t0 + budget_s * 0.06
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        for ph in [0.0, 1.0, 2.0]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(6):
        starts.append(_block_init(n0, rng, rng.integers(3, 12)))
    
    time_per = (d1 - time.time()) / max(1, len(starts))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + max(2, time_per * 0.85))
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 2: Main LP optimization (14% time)
    d2 = t0 + budget_s * 0.20
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.15, K=16)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Soft gradient loop (8% time)
    d3 = t0 + budget_s * 0.28
    h3, c5_3 = _soft_loop(best_h, d3, tau_init=0.01)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3:
        h3b, c5_3b = _lp_loop(best_h, d3, radius_init=0.08, K=14)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()
    
    # Phase 4: Multi-resolution search (30% time)
    d4 = t0 + budget_s * 0.58
    base_n = len(best_h)
    n_offsets = []
    for dn in range(-48, 49, 2):
        n_try = base_n + dn
        if 40 <= n_try <= 260 and n_try % 2 == 0:
            n_offsets.append(n_try)
    n_offsets.sort(key=lambda x: abs(x - base_n))
    
    total_time_avail = d4 - time.time()
    time_per_n = total_time_avail / max(1, len(n_offsets))
    
    for n_try in n_offsets:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + max(4, time_per_n * 0.92))
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        
        if c5_r < best_c5 + 0.004:
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, K=14)
            if c5_r < best_c5 and time.time() < d4 - 20:
                sa_dl = min(d4, time.time() + max(3, (d4 - time.time()) * 0.25))
                h_sa, c5_sa = _binary_sa(h_r, sa_dl, rng)
                if c5_sa < c5_r:
                    h_r, c5_r = h_sa, c5_sa
                if time.time() < d4 - 5:
                    h_r, c5_r = _lp_loop(h_r, min(d4, time.time() + 8), radius_init=0.05, K=12)
        
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 5: Binary SA single-swap (10% time)
    d5 = t0 + budget_s * 0.68
    h5a, c5_5a = _binary_sa(best_h, d5, rng)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5:
        h5b, c5_5b = _lp_loop(best_h, d5, radius_init=0.07, K=14)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()
    
    # Phase 6: Multi-swap SA + LP (8% time)
    d6 = t0 + budget_s * 0.76
    h6a, c5_6a = _multi_swap_sa(best_h, d6, rng, swap_size=3)
    if c5_6a < best_c5:
        best_c5 = c5_6a
        best_h = h6a.copy()
    if time.time() < d6:
        h6b, c5_6b = _lp_loop(best_h, d6, radius_init=0.06, K=14)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
    
    # Phase 7: Pair refinement (8% time)
    d7 = t0 + budget_s * 0.84
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=400)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    
    # Phase 8: Frank-Wolfe + LP (6% time)
    d8 = t0 + budget_s * 0.90
    h8a, c5_8a = _frank_wolfe(best_h, t0 + budget_s * 0.87, max_iters=120)
    if c5_8a < best_c5:
        best_c5 = c5_8a
        best_h = h8a.copy()
    if time.time() < d8:
        h8b, c5_8b = _lp_loop(best_h, d8, radius_init=0.05, K=14)
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()
    
    # Phase 9: Aggressive perturbation (5% time)
    d9 = t0 + budget_s * 0.95
    scale = 0.025
    for attempt in range(200):
        if time.time() > d9:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d9, time.time() + max(2, (d9 - time.time()) * 0.02))
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=12)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.1, scale * 1.15)
    
    # Phase 10: Final polish (3% time)
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.03, K=12)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Final pair polish
    if time.time() < deadline - 1:
        h_fp, c5_fp = _pair_refine(best_h, deadline, max_passes=80)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    
    return best_h, final_c5, n_final
