         + Σ_{k=0}^{j} W_neg[k]·(1-h[j-k]) - Σ_{k=0}^{n-1-j} W_neg[k]·h[j+k]

(Where we interpret the sums as 0 if the range is empty, and note k=0 gives W_pos[0]*(1-h[j]) which is the k=0 lag contribution where a=b=j, so both "a" and "b" terms apply: (1-h[j]) + (-h[j]) = 1-2h[j]. Indeed f_0 = Σ h[i](1-h[i]), and d/dh[j] = (1-h[j]) - h[j] = 1-2h[j]. ✓)

This is a **convolution** (or cross-correlation) of W with h! Specifically:
- Σ_{k=0}^{j} W[k]·(1-h[j-k]) = Σ_{k=0}^{j} W[k] - Σ_{k=0}^{j} W[k]·h[j-k] = S_W(j) - (W * h)[j]  (truncated convolution)
- Σ_{k=0}^{n-1-j} W[k]·h[j+k] = (W ^ h)[j]  (cross-correlation, truncated)

Where * is convolution and ^ is cross-correlation. But these are truncated (one-sided) sums, not full convolutions.

However, I can extend W to be zero outside [0, n-1] and use full convolution:
Full conv: (W * h)[j] = Σ_{k=0}^{n-1} W[k]·h[j-k] (only terms where 0 ≤ j-k ≤ n-1 contribute)

For j in 0..n-1:
- Corr part: Σ_{k=0}^{j} W[k]·h[j-k] + Σ_{k=j+1}^{n-1} W[k]·h[j-k]
  The second sum has j-k < 0, so h[j-k] = 0 (out of bounds). So full conv = Σ_{k=0}^{j} W[k]·h[j-k]. ✓ matches what we want for the "first factor" part.

- Cross-corr part: Σ_{k=0}^{n-1-j} W[k]·h[j+k]
  This is (h * W)[n-1-j] (convolution of h and W, evaluated at n-1-j):
  (h * W)[n-1-j] = Σ_{k=0}^{n-1} h[k]·W[n-1-j-k]
  = Σ_{m=0}^{n-1-j} h[n-1-j-m]·W[m]   (letting m = n-1-j-k, k = n-1-j-m)

If I zero-pad W and (1-h) to length 2n-1, compute the full linear convolution via FFT, and take the first n elements, I get exactly what I want.

def _smoothed_grad_fast(h, w, dx):
    """Compute gradient of smoothed objective using FFT
    h: (n,) array
    w: (2n-1,) array of softmax weights (already normalized)
    dx: step size
    Returns: (n,) gradient
    """
    n = len(h)
    # Split w into pos and neg parts
    # w[n-1+k] for k=0..n-1 are positive lags
    # w[n-1-k'] for k'=0..n-1 are negative lags (k'=0 is same as k=0)
    W_pos = w[n-1:]  # length n, W_pos[k] = w[n-1+k]
    W_neg = w[:n][::-1]  # length n, W_neg[k'] = w[n-1-k']
    
    one_minus_h = 1.0 - h
    # Size for FFT
    N = 1
    while N < 2 * n:
        N *= 2
    
    # First factor sums: sum_k W[k] * (1-h[j-k]) for k=0..j
    # This is the "valid" convolution of W and (1-h), first n elements
    # Full convolution: (W * (1-h))[j] = sum_{k=0}^{n-1} W[k] * (1-h[j-k]) for 0<=j-k<=n-1
    # For j in 0..n-1, terms with j-k < 0 or j-k >= n vanish (since (1-h) is 0 there)
    # So (W * (1-h))[j] = sum_{k=max(0,j-n+1)}^{min(j, n-1)} W[k] * (1-h[j-k])
    # For j in 0..n-1 and n-1 <= j (which is always): = sum_{k=0}^{j} W[k] * (1-h[j-k]) ✓
    
    Wp = np.zeros(N)
    Wp[:n] = W_pos
    oH = np.zeros(N)
    oH[:n] = one_minus_h
    conv1 = np.fft.irfft(np.fft.rfft(Wp, N) * np.fft.rfft(oH, N), N)[:n]
    
    # Second factor sums: sum_k W[k] * h[j+k] for k=0..n-1-j
    # This is cross-correlation: sum_k W[k] * h[j+k]
    # = (W ^ h)[j] where ^ is cross-correlation
    # Cross-corr (W ^ h)[j] = sum_k W[k+j] * h[k] = sum_m W[m] * h[m-j]... 
    # Let me think again. We want: sum_{k=0}^{n-1-j} W[k] * h[j+k]
    # Let m = k: sum_{m=0}^{n-1-j} W[m] * h[j+m]
    # This is the "anti-convolution" or flipped convolution.
    # If I reverse W: W_rev[m] = W[n-1-m]
    # Then (W_rev * h)[n-1-j] = sum_m W_rev[m] * h[n-1-j-m] = sum_m W[n-1-m] * h[n-1-j-m]
    # Let l = n-1-m: sum_l W[l] * h[j+l] ✓ (as l ranges appropriately)
    
    Wp_rev = np.zeros(N)
    Wp_rev[:n] = W_pos[::-1]
    h_pad = np.zeros(N)
    h_pad[:n] = h
    conv2 = np.fft.irfft(np.fft.rfft(Wp_rev, N) * np.fft.rfft(h_pad, N), N)[:n]
    # conv2[j] = (W_rev * h)[j] = sum_m W_rev[m]*h[j-m] = sum_m W[n-1-m]*h[j-m]
    # = sum_l W[l]*h[j-(n-1-l)] = sum_l W[l]*h[j-n+1+l]
    # For this to equal sum_{l=0}^{n-1-j} W[l]*h[j+l], we need j-n+1+l = j+l, which gives n-1=0. Wrong.
    
    # Let me try differently. sum_{k=0}^{n-1-j} W[k]*h[j+k]
    # = (correlate W with h) at lag -j? 
    # In numpy: np.correlate(W, h, 'full') gives output of length 2n-1
    # output[r] = sum_i W[i] * h[i - (r-(n-1))]  (from earlier derivation)
    # We want sum_{k=0}^{n-1-j} W[k]*h[j+k] = sum_k W[k]*h[k+j]
    # = sum_i W[i] * h[i+j] where i=k
    # This is a cross-correlation at lag -j (h shifted by -j relative to W):
    # = np.correlate(W, h, 'full')[n-1-j]  ? Let me check.
    
    # Actually, from my earlier derivation, np.correlate(a, b, 'full')[r] = sum_i a[i]*b[i - (r-(n-1))]
    # So np.correlate(W, h, 'full')[r] = sum_i W[i]*h[i-(r-n+1)]
    # For r = n-1+j: sum_i W[i]*h[i-j]. That's W cross-correlated with h at lag +j.
    # For r = n-1-j: sum_i W[i]*h[i+j]. That's what we want! (with valid range)
    
    # So the answer is np.correlate(W_pos, h, 'full')[n-1-j] for j = 0..n-1
    # = np.correlate(W_pos, h, 'full')[0:n]  (indices n-1, n-2, ..., 0)
    # Wait, for j=0: index n-1. For j=n-1: index 0.
    # So the array is np.correlate(W_pos, h, 'full')[n-1::-1]
    
    # Using FFT:  np.correlate(a,b) = convolve(a, b[::-1])
    # So np.correlate(W_pos, h, 'full') = np.convolve(W_pos, h[::-1], 'full')
    # And we want indices 0..n-1 (reversed): result[n-1::-1]
    
    # Let me just use np.correlate directly for clarity (it's O(n^2) via FFT internally? No, it's O(n^2) for same-length arrays probably)
    
    # For n=122, np.correlate is fast enough. Let me just use it.
    cross = np.correlate(W_pos, h, 'full')
    # cross[r] = sum_i W_pos[i]*h[i-(r-(n-1))] for valid i
    # We want for j=0..n-1: sum_k W_pos[k]*h[j+k] = cross[n-1-j]
    second_pos = cross[n-1::-1][:n]
    
    # Similarly for W_neg
    cross_neg = np.correlate(W_neg, h, 'full')
    second_neg = cross_neg[n-1::-1][:n]
    
    # First factor: we need sum_{k=0}^{j} W[k]*(1-h[j-k])
    # = (W * (1-h))[j] via FFT (computed above as conv1)
    # Actually let me just use np.convolve for clarity
    conv_pos = np.convolve(W_pos, one_minus_h, 'full')[:n]
    conv_neg = np.convolve(W_neg, one_minus_h, 'full')[:n]
    
    grad = conv_pos - second_pos + conv_neg - second_neg
    grad = (dx / float(np.sum(w))) * grad  # Wait, w should already be normalized by this point
    
    return grad

import numpy as np
import time
import math

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) < 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _row_grad(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        i = np.arange(0, n - k)
        a = i + k
        b = i
        g[b] += -h[a]
        g[a] += (1.0 - h[b])
    else:
        kk = -k
        i = np.arange(kk, n)
        a = i
        b = i - kk
        g[a] += (1.0 - h[b])
        g[b] += -h[a]
    return g

def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 120
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(50):
        if (b - a) < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv

def _smoothed_dir(h, tau, M, rng):
    n = len(h)
    corr = _corr(h)
    m = float(np.max(corr))
    w = np.exp((corr - m) / tau)
    wsum = float(np.sum(w))
    w = w / wsum
    d = np.zeros(n)
    idx = np.argsort(-corr)[:M]
    for r in idx:
        k = r - (n - 1)
        d += w[r] * _row_grad(h, k, n)
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return -d / nrm, corr

def _mseq_init(m, taps, length):
    state = (1 << m) - 1
    out = []
    for _ in range(length):
        out.append(state & 1)
        fb = 0
        for t in taps:
            fb ^= (state >> t) & 1
        state = (state >> 1) | (fb << (m - 1))
    return np.array(out, dtype=float)

def _sturmian_init(n, rng, slope=None):
    if slope is None:
        slope = (1.0 + math.sqrt(5.0)) / 2.0 - 1.0
    x = np.arange(n, dtype=float)
    h = np.zeros(n)
    prev = math.floor(0.0 * slope + 1e-15)
    for i in range(n):
        cur = math.floor((i + 1) * slope + 1e-15)
        h[i] = cur - prev
        prev = cur
    h = h + rng.normal(0, 0.02, n)
    return _project(h, n / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n, dtype=float)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize(h, deadline, rng, pair_frac=0.15, rand_frac=0.08, M=10, tau_init=0.025, tau_min=0.005):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = _corr(h)
    stall = 0
    restarts = 0
    sigma = 0.04
    uphill = 5e-4
    tau = tau_init
    momentum = 0.0
    prev_d = np.zeros(n)
    step_count = 0
    t_start = time.time()
    
    while time.time() < deadline:
        step_count += 1
        tau = max(tau_min, tau_init * (0.97 ** step_count))
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d = d / nrm
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d = d / nrm
        else:
            d_new, corr = _smoothed_dir(h, tau, M, rng)
            momentum = 0.9 * momentum
            d = momentum * prev_d + (1.0 - momentum) * d_new
            prev_d = d_new.copy()
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d = d / nrm
        
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue
        
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, dx, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        else:
            hn = np.clip(hn, 0.0, 1.0)
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = _corr(h)
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = _corr(h)
            stall += 1
        
        if stall > 100:
            restarts += 1
            sig = max(0.003, sigma * (0.9 ** min(restarts, 20)))
            if restarts > 40:
                sig *= 2.5
            h = _project(best_h + rng.normal(0, sig, n), target)
            cur_f = _exact_c5(h)
            corr = _corr(h)
            stall = 0
            momentum = 0.0
            prev_d = np.zeros(n)
    
    return best_h, best_f

def _cvxpolish(h, max_time=10.0):
    try:
        import cvxpy as cp
    except ImportError:
        return h, _exact_c5(h)
    
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    t0 = time.time()
    
    corr = _corr(h)
    active = np.argsort(-corr)[:min(20, 2 * n - 1)]
    
    x = cp.Variable(n)
    tau = cp.Variable()
    constraints = [sum(x) == target, x >= 0, x <= 1]
    
    grads = []
    for r in active:
        k = r - (n - 1)
        g = _row_grad(h, k, n)
        f_k = float(corr[r])
        lin_val = f_k + float(g @ (x - h))
        constraints.append(lin_val <= tau)
        grads.append(g)
    
    obj = cp.Minimize(tau)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=500, abstol=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=500)
        except Exception:
            return h, _exact_c5(h)
    
    if x.value is not None and time.time() - t0 < max_time:
        x_v = np.array(x.value).flatten()
        x_v = _project(x_v, target)
        f_new = _exact_c5(x_v)
        f_old = _exact_c5(h)
        if f_new < f_old:
            return x_v, f_new
    return h, _exact_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    
    if base is not None and base.size >= 8:
        n0 = len(base)
        h0 = _project(base, n0 / 2.0)
    else:
        n0 = 122
        h0 = _sine_init(n0, 1, 0.0, rng)
    
    best_h = h0.copy()
    best_f = _exact_c5(h0)
    
    # Phase 1: Multi-start (10% budget)
    d1 = t0 + budget_s * 0.10
    starts = [h0.copy()]
    
    # Sturmian init
    starts.append(_sturmian_init(n0, rng))
    starts.append(_sturmian_init(n0, rng, slope=0.618))
    # M-sequence based
    ms = _mseq_init(7, [3, 0], n0)
    ms = ms + rng.normal(0, 0.03, n0)
    starts.append(_project(ms, n0 / 2.0))
    # Sine inits
    for f_, p_ in [(1, 0.5), (2, 1.0), (3, 0.0), (5, 1.5)]:
        starts.append(_sine_init(n0, f_, p_, rng))
    
    for s in starts:
        if time.time() > d1:
            break
        sub = min(d1, time.time() + max(2.0, budget_s * 0.015))
        hh, ff = _optimize(s, sub, rng, M=8, tau_init=0.02, tau_min=0.01)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()
    
    # Phase 2: Main optimization (55% budget)
    d2 = t0 + budget_s * 0.65
    hh, ff = _optimize(best_h, d2, rng, pair_frac=0.18, rand_frac=0.06, M=12, tau_init=0.02, tau_min=0.004)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    
    # Phase 3: Multi-resolution (20% budget)
    d3 = t0 + budget_s * 0.85
    bn = len(best_h)
    tried = set()
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -1, 1, 14, -14, 16, -16]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 24 or n_try > 260:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub = min(d3, time.time() + max(2.5, budget_s * 0.018))
        hr2, fr2 = _optimize(hr, sub, rng, M=10, tau_init=0.018, tau_min=0.006)
        if fr2 < best_f - 1e-6:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try
    
    # Phase 4: CVXPY polish + final pair search
    d4 = t0 + budget_s * 0.93
    hh, ff = _cvxpolish(best_h, max_time=min(15.0, max(3.0, d4 - time.time())))
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    
    d5 = deadline
    hh, ff = _optimize(best_h, d5, rng, pair_frac=0.40, rand_frac=0.04, M=15, tau_init=0.008, tau_min=0.003)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    
    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    
    return best_h, final_f, len(best_h)
