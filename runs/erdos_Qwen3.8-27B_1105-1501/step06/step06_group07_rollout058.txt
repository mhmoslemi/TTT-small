import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _line_ub(h, d):
    t_hi = 1e10
    pos = d > 1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    neg = d < -1e-15
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    return max(0.0, t_hi)


def _ls_1d(h, d, dx, t_cap=None):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_max = _line_ub(h, d)
    if t_cap is not None:
        t_max = min(t_max, t_cap)
    if t_max < 1e-15:
        return 0.0, float(np.max(S)) * dx
    n_grid = 80
    ts = np.linspace(0.0, t_max, n_grid)
    best_t = 0.0
    best_val = float(np.max(S)) * dx
    best_i = 0
    for i in range(1, n_grid):
        t = ts[i]
        f = float(np.max(S + t * B - t * t * C)) * dx
        if f < best_val:
            best_val = f
            best_t = t
            best_i = i
    if 0 < best_i < n_grid - 1:
        a = max(0.0, ts[max(0, best_i - 2)])
        b = min(t_max, ts[min(n_grid - 1, best_i + 2)])
        if b - a > 1e-13:
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            f1 = float(np.max(S + c1 * B - c1 * c1 * C)) * dx
            f2 = float(np.max(S + c2 * B - c2 * c2 * C)) * dx
            for _ in range(45):
                if b - a < 1e-14:
                    break
                if f1 < f2:
                    b, c2, f2 = c2, c1, f1
                    c1 = b - gr * (b - a)
                    f1 = float(np.max(S + c1 * B - c1 * c1 * C)) * dx
                else:
                    a, c1, f1 = c1, c2, f2
                    c2 = a + gr * (b - a)
                    f2 = float(np.max(S + c2 * B - c2 * c2 * C)) * dx
            opt_t = (a + b) / 2.0
            opt_val = float(np.max(S + opt_t * B - opt_t * opt_t * C)) * dx
            if opt_val < best_val:
                best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _grad_weighted(h, K, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    M = float(top_vals[0])
    w = np.exp(alpha * (top_vals - M))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        if k >= 0:
            m = n - k
            if m <= 0:
                continue
            grad[:m] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:m]
        else:
            ga = -k
            m = n - ga
            if m <= 0:
                continue
            grad[ga:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[ga:]
    grad *= dx
    return grad, corr


def _symmetrize(h):
    n = len(h)
    h_s = 0.5 * (h + h[::-1])
    return _project(h_s, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_init
    K = K_init
    step_cap = 1.0
    stall = 0
    n_iter = 0
    pair_frac = 0.3

    for it in range(max_iters):
        if time.time() > deadline:
            break
        n_iter += 1

        if rng.random() < pair_frac and n > 4:
            # Pair swap move
            grad, corr = _grad_weighted(h, min(K, 15), alpha)
            dec_mask = grad > 1e-10
            inc_mask = grad < -1e-10
            if np.any(dec_mask) and np.any(inc_mask):
                i = int(rng.choice(np.where(dec_mask)[0], p=np.abs(grad[dec_mask]) / np.abs(grad[dec_mask]).sum()))
                j = int(rng.choice(np.where(inc_mask)[0], p=np.abs(grad[inc_mask]) / np.abs(grad[inc_mask]).sum()))
            else:
                i, j = rng.integers(0, n, 2)
                if i == j:
                    continue
            if h[i] <= h[j] + 1e-12:
                continue
            d = np.zeros(n)
            d[j] = 1.0
            d[i] = -1.0
            t_max = min(h[i], 1.0 - h[j])
            if t_max < 1e-12:
                continue
            t, f_try = _ls_1d(h, d, dx, t_cap=t_max)
            if t > 1e-12 and f_try < best_c5 - 1e-14:
                h_new = h + t * d
                h_new = _project(h_new, target)
                f_new = _exact_c5(h_new)
                if f_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = f_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Gradient step
            grad, corr = _grad_weighted(h, K, alpha)
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            direction = -grad / gn
            t, f_try = _ls_1d(h, direction, dx, t_cap=step_cap)
            if t > 1e-12 and f_try < _exact_c5(h) - 1e-14:
                h_new = h + t * direction
                h_new = _project(h_new, target)
                f_new = _exact_c5(h_new)
                if f_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = f_new
                    best_h = h.copy()
                    stall = 0
                    if t > 0.7 * step_cap:
                        step_cap = min(step_cap * 1.1, 2.0)
                    else:
                        step_cap = max(step_cap * 0.9, 0.05)
                else:
                    stall += 1
            else:
                stall += 1
                step_cap *= 0.8

        # Adaptive parameters
        if stall > 0:
            alpha *= 1.3
            if alpha > 500:
                alpha = 500.0
            K = min(K + 1, 60)
            pair_frac = min(0.5, pair_frac + 0.005)
        if stall > 100:
            # Perturbation restart
            h = h + rng.normal(0, 0.03, n)
            h = _project(h, target)
            pop = _exact_c5(h)
            if pop < best_c5:
                best_c5 = pop
                best_h = h.copy()
            stall = 0
            alpha = alpha_init
            K = K_init
            step_cap = 0.5
            pair_frac = 0.3
            h = best_h.copy()

    return best_h, best_c5


def _intensive_polish(h, deadline, rng, n_rounds=5):
    """Multiple rounds of alternating K and alpha."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    K_list = [15, 25, 40, 55, 70]
    alpha_list = [20.0, 50.0, 100.0, 200.0, 400.0]

    for r in range(n_rounds):
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / max(1, n_rounds - r))
        K = K_list[r % len(K_list)]
        alpha = alpha_list[r % len(alpha_list)]
        h_try, c5_try = _local_search(best_h, sub_dl, rng, max_iters=200000, K_init=K, alpha_init=alpha)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5


def _binary_sa(h_init, deadline, rng, max_iters=100000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    if n % 2 == 1:
        h[sorted_idx[target]] = 1.0
        h[np.argsort(h_init)[0]] = 0.0
        target = n // 2
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.01
    T_min = 1e-7
    cooling = 0.99995
    ones_set = [i for i in range(n) if h[i] > 0.5]
    zeros_set = [i for i in range(n) if h[i] <= 0.5]

    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones_set[int(rng.integers(len(ones_set)))]
        j = zeros_set[int(rng.integers(len(zeros_set)))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_set.remove(i)
            ones_set.append(j)
            zeros_set.remove(j)
            zeros_set.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s * 0.96
    rng = np.random.default_rng(seed)

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = None

    if h0 is not None and len(h0) > 10:
        n0 = len(h0)
        h_start = _project(h0.copy(), n0 / 2.0)
    else:
        n0 = 120
        x = np.arange(n0)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * x / n0)
        h_start = _project(h_start, n0 / 2.0)

    best_h = h_start.copy()
    best_c5 = _exact_c5(h_start)

    # Phase 1: Try diverse starting points (15% time)
    d1 = t_start + budget_s * 0.15
    n0 = len(h_start)
    candidates = [h_start.copy()]

    # Symmetric
    h_sym = _symmetrize(h_start)
    candidates.append(h_sym)

    # Resampled versions
    for dn in [-8, -4, 4, 8]:
        nn = n0 + dn
        if 30 <= nn <= 200 and nn % 2 == 0:
            candidates.append(_resample(h_start, nn))

    # Binary threshold
    h_bin = np.zeros(n0)
    h_bin[np.argsort(-h_start)[:n0 // 2]] = 1.0
    candidates.append(h_bin)

    # Low-freq sinusoidal
    for freq in [3, 4, 6, 7]:
        h_sine = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n0) / n0 + rng.uniform(0, 2 * np.pi))
        candidates.append(_project(h_sine, n0 / 2.0))

    for hc in candidates:
        if time.time() > d1:
            break
        hc = _project(hc, len(hc) / 2.0)
        c5_c = _exact_c5(hc)
        sub_dl = min(d1, time.time() + 8)
        h_opt, c5_opt = _local_search(hc, sub_dl, rng, max_iters=30000, K_init=20, alpha_init=30.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Intensive local search (50% time)
    d2 = t_start + budget_s * 0.65
    h2, c5_2 = _intensive_polish(best_h, d2, rng, n_rounds=8)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA + polish (15% time)
    d3 = t_start + budget_s * 0.80
    h3, c5_3 = _binary_sa(best_h, min(d3, time.time() + budget_s * 0.08), rng, max_iters=80000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Polish the binary result
    if time.time() < d3:
        h3b, c5_3b = _local_search(best_h, d3, rng, max_iters=100000, K_init=20, alpha_init=40.0)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution (15% time)
    d4 = t_start + budget_s * 0.95
    best_n = len(best_h)
    tried = set([best_n])
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, -10, 10, -12, 12, -16, 16, -20, 20]:
        if time.time() > d4:
            break
        nn = best_n + dn
        if nn < 40 or nn > 220 or nn % 2 != 0 or nn in tried:
            continue
        tried.add(nn)
        h_r = _resample(best_h, nn)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _local_search(h_r, sub_dl, rng, max_iters=50000, K_init=20, alpha_init=35.0)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Symmetrization check
    if time.time() < deadline - 2:
        h_s = _symmetrize(best_h)
        c5_s = _exact_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
            h_fin, c5_fin = _local_search(best_h, deadline - 1, rng, max_iters=50000, K_init=20, alpha_init=30.0)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
