def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _softobj(h, mu):
    corr = _corr_vals(h)
    m = corr.max()
    s = np.exp(mu * (corr - m))
    return m + np.log(s.sum()) / mu

def _softgrad(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = _corr_vals(h)
    m = corr.max()
    w = np.exp(mu * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[n-1+s]
        if ws < 1e-15: continue
        if s >= 0:
            grad[:n-s] += ws * (1.0 - h[s:n])
            grad[s:] += ws * (-h[:n-s])
        else:
            t = -s
            grad[t:] += ws * (1.0 - h[:n-t])
            grad[:n-t] += ws * (-h[t:n])
    return grad * dx

def _slsqp_active(h0, deadline, max_outer=30):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur = _exact_c5(h)
    best_h, best_c = h.copy(), cur
    for outer in range(max_outer):
        if time.time() > deadline: break
        corr = _corr_vals(h)
        K = min(50, 2*n-1)
        top = np.argsort(-corr)[:K]
        t0 = cur + 1e-5
        x0 = np.concatenate([h, [t0]])
        def obj(x): return x[n]
        def objg(x):
            g = np.zeros(n+1); g[n] = 1.0; return g
        def ineq(x):
            c = _corr_vals(x[:n])
            return x[n] - c[top]
        def ineqj(x):
            hx = x[:n]
            # Compute gradients for top lags
            G = np.zeros((len(top), n))
            for idx, k in enumerate(top):
                s = k - (n-1)
                if s >= 0:
                    G[idx, :n-s] += (1.0 - hx[s:n])
                    G[idx, s:] += (-hx[:n-s])
                else:
                    t = -s
                    G[idx, t:] += (1.0 - hx[:n-t])
                    G[idx, :n-t] += (-hx[t:n])
            G *= dx
            J = np.zeros((len(top), n+1))
            J[:, :n] = -G
            J[:, n] = 1.0
            return J
        def eq(x): return x[:n].sum() - target
        def eqj(x):
            g = np.zeros(n+1); g[:n] = 1.0; return g
        bounds = [(0,1)]*n + [(0,3)]
        cons = [{'type':'ineq','fun':ineq,'jac':ineqj},
                {'type':'eq','fun':eq,'jac':eqj}]

                       constraints=cons, options={'maxiter':50,'ftol':1e-12})

            else:

        else:
            break

def _subgradient(h0, deadline, max_iter=2000):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur = _exact_c5(h)
    best_h, best_c = h.copy(), cur
    # Collect iterates for averaging
    h_sum = np.zeros(n)
    h_count = 0
    for it in range(max_iter):
        if time.time() > deadline: break
        alpha = 0.1 / (1 + 0.01 * it) ** 0.5
        corr = _corr_vals(h)
        M = corr.max()
        # Subgradient: average of gradients of max-attaining lags
        active = np.where(corr > M - 0.001)[0]
        if len(active) == 0: active = [int(np.argmax(corr))]
        g = np.zeros(n)
        for k in active:
            s = k - (n-1)
            if s >= 0:
                g[:n-s] += (1.0 - h[s:n])
                g[s:] += (-h[:n-s])
            else:
                t = -s
                g[t:] += (1.0 - h[:n-t])
                g[:n-t] += (-h[t:n])

        h = _project(h - alpha * g, target)
        c = _exact_c5(h)
        h_sum += h
        h_count += 1
        if c < best_c - 1e-14:

    c_avg = _exact_c5(h_avg)
    if c_avg < best_c:

    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(x_new, x_old, h), n_new/2.0)

def _reflect_sym(h):
    # h'(x) = 1 - h(2-x), in discrete: h'[i] = 1 - h[n-1-i]
    n = len(h)
    return 1.0 - h[::-1]

def _half_period_sym(h):

    n = len(h)

    m = n // 2
    g = h[:m]  # values on [0,1]
    h_sym = np.concatenate([g, 1.0 - g])
    return h_sym  # automatically satisfies sum = n/2

        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lg_op(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] = (1.0 - h[s:n])
        g[s:] = g[s:] + (-h[:n - s])
    else:
        t = -s
        g[t:] = (1.0 - h[:n - t])
        g[:n - t] = g[:n - t] + (-h[t:n])
    return g

def _softobj_grad(h, mu):
    corr = _corr_vals(h)
    m = corr.max()
    shifted = corr - m
    exps = np.exp(mu * shifted)
    obj = m + np.log(exps.sum()) / mu
    w = exps / exps.sum()
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(w)):
        wi = w[i]
        if wi < 1e-14:
            continue
        grad += wi * _lg_op(h, i, n)
    grad *= dx
    return obj, grad

def _obj_a(x):
    return x[_STATE['n']]

def _obj_a_grad(x):
    n = _STATE['n']
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _ineq_a(x):
    n = _STATE['n']
    top = _STATE['top']
    c = _corr_vals(x[:n])
    return x[n] - c[top]

def _ineq_a_jac(x):
    n = _STATE['n']
    top = _STATE['top']
    dx = _STATE['dx']
    hx = x[:n]
    K = len(top)
    J = np.zeros((K, n + 1))
    for i in range(K):
        J[i, :n] = -_lg_op(hx, top[i], n) * dx
    J[:, n]
