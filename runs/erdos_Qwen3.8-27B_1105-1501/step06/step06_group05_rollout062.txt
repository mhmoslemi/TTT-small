import numpy as np
import time
from scipy.optimize import minimize

_g_alpha = [50.0]
_g_sigma = [500.0]


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _smooth_obj_and_grad(h, alpha, sigma):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * dx
    smax = float(np.max(s))
    sm = s - smax
    exps = np.exp(alpha * sm)
    sum_exps = float(np.sum(exps))
    psi = smax + np.log(sum_exps) / alpha
    penalty = 0.5 * sigma * (np.sum(h) - n * 0.5) ** 2
    obj = psi + penalty
    w = exps / sum_exps
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        l = r - (n - 1)
        wl = w[r]
        if wl < 1e-15:
            continue
        m1s = max(0, l)
        m1e = min(n, n + l)
        if m1s < m1e:
            grad[m1s:m1e] += wl * (1.0 - h[m1s - l:m1e - l])
        m2s = max(0, -l)
        m2e = min(n, n - l)
        if m2s < m2e:
            grad[m2s:m2e] -= wl * h[m2s + l:m2e + l]
    grad *= dx
    grad += sigma * (np.sum(h) - n * 0.5)
    return obj, grad


def _smooth_obj_func(x):
    val, _ = _smooth_obj_and_grad(x, _g_alpha[0], _g_sigma[0])
    return val


def _smooth_grad_func(x):
    _, g = _smooth_obj_and_grad(x, _g_alpha[0], _g_sigma[0])
    return g


def _lbfgs_step(h, alpha, deadline, maxiter=200):
    n = len(h)
    if time.time() > deadline:
        return h
    _g_alpha[0] = alpha
    _g_sigma[0] = 500.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj_func, h.copy(), jac=_smooth_grad_func,
                       bounds=bounds, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'gtol': 1e-9})
        h_new = np.clip(res.x, 0.0, 1.0)
    except Exception:
        h_new = h.copy()
    h_new = _project(h_new, n * 0.5)
    return h_new


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _line_search_polish(h, deadline, rng, pair_frac=0.20, rand_frac=0.08, k_active=10, tau=0.018):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0

    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            scores = corr * dx
            m = float(np.max(scores))
            w = np.exp((scores - m) / tau)
            idx = np.argsort(-scores)[:k_active]
            G = np.zeros(n)
            for ri in idx:
                k = ri - (n - 1)
                wl = w[ri]
                m1s = max(0, k)
                m1e = min(n, n + k)
                if m1s < m1e:

                m2s = max(0, -k)
                m2e = min(n, n - k)

                    G[m2s:m2e] -= wl * h[m2s + k:m2e + k]

            d -= d.mean()

            stall += 1
            continue
        d = d / nrm
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]

            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue

        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        nq = 100
        grid = np.linspace(L, U, nq)
        vals = corr[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
        obj = np.max(vals, axis=1) * dx
        i = int(np.argmin(obj))
        a = grid[max(i - 1, 0)]
        b = grid[min(i + 1, nq - 1)]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(30):
            if (b - a) < 1e-13:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
        t = 0.5 * (a + b)
        tv = float(np.max(corr + t * T - t * t * Q)) * dx
        if obj[i] < tv:
            t, tv = grid[i], obj[i]
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-12 or hn.max() > 1.0 + 1e-12:
            hn = _project(hn, target)
        hn = np.clip(hn, 0.0, 1.0)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-13:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + 1e-4:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
        if stall > 120:
            h = _project(best_h + rng.normal(0, 0.04, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
    return best_h, best_f


def _run_alpha_schedule(h, alphas, deadline, maxiter=200):
    n = len(h)
    best_h = h.copy()
    best_f = _exact_c5(h)
    h_cur = h.copy()
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        h_new = _lbfgs_step(h_cur, alpha, deadline - 2, maxiter=maxiter)
        f_new = _exact_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_f


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        x = np.arange(n)
        base = 0.5 + 0.3 * np.sin(2.0 * np.pi * x / n) + 0.15 * np.sin(4.0 * np.pi * x / n)

    n0 = len(base)
    h0 = _project(base, n0 * 0.5)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1a: Coarse resolution scan (15% budget)
    d1a = t0 + budget_s * 0.15
    coarse_res = []
    for nr in range(60, 201, 10):
        if nr % 2 == 0:
            coarse_res.append(nr)
    alphas_coarse = [100.0, 500.0, 2000.0]
    best_res = n0
    for nr in coarse_res:
        if time.time() > d1a - 5:
            break
        if nr == n0:
            h_try = h0.copy()
        else:
            h_try = _resample(h0, nr)
        hr, fr = _run_alpha_schedule(h_try, alphas_coarse, d1a - 5, maxiter=150)
        if fr < best_f:
            best_f = fr
            best_h = hr.copy()
            best_res = nr

    # Phase 1b: Fine resolution scan (20% budget)
    d1b = t0 + budget_s * 0.35
    fine_res = []
    for dn in range(-6, 7, 2):
        nr = best_res + dn
        if 36 <= nr <= 220 and nr % 2 == 0:
            fine_res.append(nr)
    fine_res = sorted(set(fine_res))
    alphas_fine = [100.0, 300.0, 800.0, 2000.0, 5000.0, 10000.0]
    for nr in fine_res:
        if time.time() > d1b - 5:
            break
        if nr == len(best_h):
            h_try = best_h.copy()
        else:
            h_try = _resample(best_h, nr)
        hr, fr = _run_alpha_schedule(h_try, alphas_fine, d1b - 5, maxiter=200)
        if fr < best_f:
            best_f = fr
            best_h = hr.copy()

    # Phase 2: Line search polish (25% budget)
    d2 = t0 + budget_s * 0.60
    hh, ff = _line_search_polish(best_h, d2, rng, pair_frac=0.20, rand_frac=0.08, k_active=10, tau=0.018)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: Smooth optimization from diverse starts (18% budget)
    d3 = t0 + budget_s * 0.78
    n_best = len(best_h)
    diverse_alphas = [500.0, 1500.0, 4000.0, 10000.0]
    for _ in range(8):
        if time.time() > d3 - 5:
            break
        # Different perturbation strategies
        strat = rng.random()
        if strat < 0.4:
            h_try = _project(best_h + rng.normal(0, 0.02, n_best), n_best * 0.5)
        elif strat < 0.6:
            h_try = best_h[::-1].copy()
            h_try = _project(h_try, n_best * 0.5)
        elif strat < 0.8:
            shift = int(rng.integers(1, n_best))
            h_try = np.roll(best_h, shift)
            h_try = _project(h_try, n_best * 0.5)
        else:
            # Smooth smaller perturbation
            h_try = _project(best_h + rng.normal(0, 0.005, n_best), n_best * 0.5)
        hr, fr = _run_alpha_schedule(h_try, diverse_alphas, d3 - 5, maxiter=150)
        if fr < best_f:
            best_f = fr
            best_h = hr.copy()

    # Phase 4: Final line search polish (12% budget)
    hh, ff = _line_search_polish(best_h, deadline, rng, pair_frac=0.40, rand_frac=0.05, k_active=12, tau=0.012)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) * 0.5)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
