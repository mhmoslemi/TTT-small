import numpy as np

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _build_G(h, dx):
    n = len(h)
    m = len(h)
    G = np.zeros((2 * m - 1, m))
    for k in range(2 * m - 1):
        lo1 = max(0, k - (m - 1))
        hi1 = min(m - 1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            G[k, idx] += dx * (1.0 - h[idx - k + (m - 1)])
        lo2 = max(0, (m - 1) - k)
        hi2 = min(m - 1, 2 * m - 2 - k)
        if lo2 <= hi2:
            jv = np.arange(lo2, hi2 + 1)
            G[k, jv] += -dx * h[jv + k - (m - 1)]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    # Linearized change: dS ≈ T * t - C * t^2
    # T[k] = G[k] @ d (gradient dot direction)
    # C[k] = dx * corr(d, d)[k] (quadratic term)
    G = _build_G(h, dx)
    T = G @ d
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1)
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(35):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_dir(h, G, S, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _build_lp_matrix(b, dx):
    """Build matrix A where A[k,j] = dx*(1-b)[j-k+(n-1)] for valid indices.
    This gives the linear objective dx*corr(a, 1-b)[k] = sum_j A[k,j]*a[j]."""
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A[k, idx] = dx * (1.0 - b[idx - k + (n - 1)])
    return A

def _lp_minimax(A, target):
    """Solve: min tau s.t. A@x <= tau, sum(x)=target, 0<=x<=1, tau>=0."""
    n = A.shape[1]
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _seg_search(u1, u2, n_grid=31):
    """Search along segment (1-alpha)*u1 + alpha*u2 for best C5."""
    best_val = float('inf')
    best_point = None
    for alpha in np.linspace(0.0, 1.0, n_grid):
        pt = (1.0 - alpha) * u1 + alpha * u2
        pt = _project(pt, len(pt) / 2.0)
        val = _exact_c5(pt)
        if val < best_val:
            best_val = val
            best_point = pt.copy()
    return best_point, best_val

def _relax_pair(h, deadline, n_rounds=30):
    """Relaxed pair: fix partner b, LP for a; fix a, LP for b; bridge segment."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for rnd in range(n_rounds):

            break
        # Choose partner b
        if rnd % 3 == 0:
            b = 1.0 - h
        elif rnd % 3 == 1:

        else:

            freq = rng.integers(1, 6)
            phase = rng.uniform(0, 2 * np.pi)
            b = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
            b = _project(b, target)
        b = _project(b, target)
        # Build LP matrix for a given b
        A = _build_lp_matrix(b, dx)
        a_star = _lp_minimax(A, target)
        if a_star is None:
            continue
        a_star = _project(a_star, target)
        # Evaluate endpoints
        c5_a = _exact_c5(a_star)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = a_star.copy()
        u2 = _project(1.0 - a_star, target)
        c5_u2 = _exact_c5(u2)
        if c5_u2 < best_c5:
            best_c5 = c5_u2
            best_h = u2.copy()
        # Solve dual: fix a_star, find b_star
        A2 = _build_lp_matrix(a_star, dx)
        b_star = _lp_minimax(A2, target)
        if b_star is not None:
            b_star = _project(b_star, target)
            u1 = _project(1.0 - b_star, target)
            c5_u1 = _exact_c5(u1)
            if c5_u1 < best_c5:

                best_c5 = seg_val2
                best_h = seg_pt2.copy()
            # Update h for next round
            h = best_h.copy()
    return best_h, best_c5

def _smooth_descent(h, deadline, tau_init=0.006):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_init
    lr = 0.004
    for it in range(3000):
        if time.time() > deadline:
            break
        S = _corr_vals(h)
        S_max = float(np.max(S))
        w = np.exp((S - S_max) / tau)
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            w = np.zeros(len(S))
            w[int(np.argmax(S))] = 1.0
        else:
            w /= w_sum
        G = _build_G(h, dx)
        grad = G.T @ w
        h = h - lr * grad
        h = _project(h, target)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            lr = min(lr * 1.05, 0.015)
        else:
            lr *= 0.92
        if it % 40 == 0:
            tau *= 0.96
            lr = max(lr, 0.001)
    return best_h, best_c5

    h3, c5_3 = _smooth_descent(best_h, d3, tau_init=0.006)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Diverse starts with LP (15%)
    d4 = t0 + budget_s * 0.70
    n_starts = 5
    per_start = (d4 - time.time()) / n_starts if d4 > time.time() else 1.0
    for si in range(n_starts):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + max(per_start, 5.0))
        h_s = _diverse_start(n0, rng)
        h_s, c5_s = _lp_optimize(h_s, sub_dl, start_radius=0.10)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Pair optimization (8%)
    d5 = t0 + budget_s * 0.78
    h5, c5_5 = _pair_optimize(best_h, d5, max_passes=80)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (10%)
    d6 = t0 + budget_s * 0.88
    n_best = len(best_h)
    tried_n = set()
    offsets = [4, -4, 6, -6, 8, -8, 10, -10, 12, -12, 16, -16, 20, -20]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = n_best + dn
        if n_try < 24 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + max(budget_s * 0.015, 3.0))
        h_r, c5_r = _lp_optimize(h_r, sub_dl, start_radius=0.10)
        if c5_r < best_c5:
            if time.time() < d6:
                h_r, c5_r = _pair_optimize(h_r, min(d6, time.time() + 4.0), max_passes=20)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_best = n_try

    # Phase 7: Final LP + smooth polish (5%)
    d7 = deadline
    h7, c5_7 = _lp_optimize(best_h, d7, start_radius=0.06)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final: quick smooth descent if time remains
    if time.time() < deadline - 2.0:
        h8, c5_8 = _smooth_descent(best_h, deadline, tau_init=0.003)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
