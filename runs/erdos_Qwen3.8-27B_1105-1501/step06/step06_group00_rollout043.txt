                    best_h = h.copy()

                best_h = h.copy()
                stall = 0
            else:
                stall += 1

                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(scores + c2 * T + c2 * c2 * C))
        t_opt = (lo_b + hi_b) / 2.0
        val_opt = float(np.max(scores + t_opt * T + t_opt * t_opt * C)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_t = t_opt
    return best_t, best_val

def _fw_loop(h, deadline, max_iter=300000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline:
        if stall > 400:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        m_active = int(np.argmax(scores))
        g = _subgrad_single(h, m_active)
        x_greedy = _fw_greedy(g, target)
        s = x_greedy - h
        if float(np.max(np.abs(s))) < 1e-12:
            stall += 5
            continue
        corr_ws = np.correlate(1.0 - h, s, mode='full')
        corr_hs = np.correlate(h, s, mode='full')
        T = corr_ws - corr_hs[::-1]
        C = -np.correlate(s, s, mode='full')
        pos = s > 1e-15
        neg = s < -1e-15
        t_max = 1.0
        if np.any(pos):
            t_max = min(t_max, float(np.min((1.0 - h[pos]) / s[pos])))
        if np.any(neg):
            t_max = min(t_max, float(np.min(h[neg] / (-s[neg]))))
        if t_max < 1e-14:
            stall += 1
            continue
        t_opt, c5_try = _line_search_quad(scores, T, C, t_max, dx)
        if c5_try < cur_c5 - 1e-14:
            h_new = h + t_opt * s
            h = _project(h_new, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _soft_grad(h, tau):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    shifted = scores - np.max(scores)
    weights = np.exp(tau * shifted)
    wsum = float(np.sum(weights))
    if wsum < 1e-30:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(scores)] = 1.0
    else:
        weights /= wsum
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-8:
            continue
        d = m - (n - 1)
        if d >= 0:
            ms = n - d
            if ms > 0:
                g[:ms] += w * (1.0 - h[d:])
                g[d:] += -w
