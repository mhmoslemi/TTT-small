def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min())-2, float(v.max())+2
    for _ in range(60):
        mid = (lo+hi)/2
        if float(np.sum(np.clip(v-mid, 0, 1))) > target: lo = mid
        else: hi = mid
    x = np.clip(v-(lo+hi)/2, 0, 1)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        m = (x<1) if res>0 else (x>0)
        c = int(m.sum())
        if c: x = np.clip(x + res/c, 0, 1)
    return x

def _grad_lag(h, k):
    n = len(h); g = np.zeros(n)
    if k >= 0:
        m = n-k
        if m > 0: g[k:] += 1-h[:m]; g[:m] -= h[k:]
    else:
        a = -k; m = n-a
        if m > 0: g[:m] += 1-h[a:]; g[a:] -= h[:m]
    return g

def _corr_and_grad(h, alpha, K):
    n = len(h); dx = 2.0/n
    corr = np.correlate(h, 1-h, 'full') * dx
    idx = np.argsort(-corr)[:K]
    vals = corr[idx]
    w = np.exp(alpha*(vals-vals[0])); w /= w.sum()

        g += wi * _grad_lag(h, int(r)-(n-1))
    return g * dx, corr

    n = len(h)
