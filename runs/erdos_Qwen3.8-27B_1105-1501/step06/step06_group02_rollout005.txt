import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_matrix(h):
    """Compute gradient matrix G of shape (2n-1, n) for all correlation rows."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for ell in range(-(n-1), n):
        q = ell + (n - 1)  # row index
        if ell >= 0:
            m = n - ell
            # G[q, i] += (1-h[i-ell]) for ell <= i <= n-1
            idx = np.arange(ell, n)
            G[q, idx] += (1.0 - h[idx - ell])
            # G[q, i] += -h[i+ell] for 0 <= i <= m-1
            idx2 = np.arange(0, m)
            G[q, idx2] += (-h[idx2 + ell])
        else:
            r = -ell
            m = n - r
            # G[q, i] += (1-h[i+r]) for 0 <= i <= m-1
            idx = np.arange(0, m)
            G[q, idx] += (1.0 - h[idx + r])
            # G[q, i] += -h[i-r] for r <= i <= n-1
            idx2 = np.arange(r, n)
            G[q, idx2] += (-h[idx2 - r])
    return G

def exact_line_search(h, p, R0=None, n_iter=50):
    """Exact line search along direction p (zero-sum) from base h.
    Returns (best_alpha, best_c5, best_h)."""
    n = len(h)
    dx = 2.0 / n
    p = p - np.mean(p)  # ensure zero sum
    
    # Compute D and Q
    D = np.correlate(p, 1.0 - h, 'full') - np.correlate(h, p, 'full')
    Q = np.correlate(p, p, 'full')
    if R0 is None:
        R0 = np.correlate(h, 1.0 - h, 'full')
    
    def f_at(alpha):
        rows = R0 + alpha * D - alpha * alpha * Q
        return float(np.max(rows) * dx)
    
    # Bounds
    pos = p > 1e-12
    neg = p < -1e-12
    lo = 0.0
    hi = 5.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if hi - lo < 1e-12:
        return 0.0, f_at(0.0), h.copy()
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_at(c1)
    f2 = f_at(c2)
    for _ in range(n_iter):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f_at(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_at(c2)
    opt = (a + b) / 2.0
    val = f_at(opt)
    
    # Check endpoints
    f_lo = f_at(lo)
    f_hi = f_at(hi)
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    
    new_h = h + opt * p
    new_h = project(new_h, n / 2.0)
    actual_c5 = c5_exact(new_h)
    if actual_c5 < val - 1e-12:
        val = actual_c5
        # Use the projected version
    return opt, val, new_h

import time
from scipy.optimize import minimize

        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for ell in range(-(n-1), n):
        q = ell + (n - 1)
        if ell >= 0:
            m = n - ell
            idx = np.arange(ell, n)
            G[q, idx] += (1.0 - h[idx - ell])
            idx2 = np.arange(0, m)
            G[q, idx2] += (-h[idx2 + ell])
        else:
            r = -ell
            m = n - r
            idx = np.arange(0, m)
            G[q, idx] += (1.0 - h[idx + r])
            idx2 = np.arange(r, n)
            G[q, idx2] += (-h[idx2 - r])
    return G

def _qp_objective(x, G, R, target, n, C, rho, lam_z, lam_n, z_scale):
    d = x[:n]
    z = x[n:]
    f = lam_z * np.sum(z) / (C * z_scale) + lam_n * np.sum(d * d) / (n * rho * rho)
    return f

def _qp_grad(x, G, R, target, n, C, rho, lam_z, lam_n, z_scale):
    d = x[:n]
    z = x[n:]
    gd = np.zeros_like(x)
    gd[:n] = 2.0 * lam_n * d / (n * rho * rho)
    gd[n:] = lam_z / (C * z_scale)
    return gd

def solve_target_qp(h, target_raw, rho, lam_z=0.8, lam_n=0.2, maxiter=200):
    """Solve the target QP. Returns direction d or None."""
    n = len(h)
    C = 2 * n - 1
    R = np.correlate(h, 1.0 - h, 'full')
    G = compute_grad_matrix(h)
    
    # Variable bounds
    d_lo = -np.minimum(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    z_lo = np.zeros(C)
    z_hi = np.maximum(R + np.max(np.abs(G)) * rho - target_raw + 1.0, 1.0)
    
    bounds = list(zip(d_lo, d_hi)) + list(zip(z_lo, z_hi))
    
    # Objective params
    z_scale = max(np.mean(np.maximum(R - target_raw, 0.0)), 1e-10)
    
    # Inequality constraints: G @ d + R[i] - target_raw - z[i] <= 0
    # i.e., z[i] >= G @ d + R[i] - target_raw
    # As inequality: G @ d + R[i] - target_raw - z[i] <= 0
    
    # We'll use scipy's constraint format: fun(x) >= 0
    # So: -(G @ d + R[i] - target_raw - z[i]) >= 0
    # i.e., -G @ d - R[i] + target_raw + z[i] >= 0
    
    def constraint_func(x):
        d = x[:n]
        z = x[n:]
        # For each row i: -G[i] @ d - R[i] + target_raw + z[i] >= 0
        return -G @ d + (target_raw - R) + z
    
    def constraint_jac(x):
        # Row i: -G[i,:] @ d + z[i]
        # d part: -G[i,:]
        # z part: identity
        jac = np.zeros((C, n + C))
        jac[:, :n] = -G.T  # Wait, this is wrong dimension
        # Actually G is (C, n), so G @ d is (C,)
        # d/dx_i of constraint_j = -G[j, i] for i < n
        # d/dx_{n+j} of constraint_j = 1
        jac[:, :n] = -G
        # Wait, G is already (C, n). So jac[:, :n] = -G gives (C, n). Good.
        for j in range(C):
            jac[j, n + j] = 1.0
        return jac
    
    # Equality constraint: sum(d) = 0
    def eq_func(x):
        return np.sum(x[:n])
    
    def eq_jac(x):
        j = np.zeros(n + C)
        j[:n] = 1.0
        return j
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_func, 'jac': constraint_jac},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac}
    ]
    
    # Initial point
    x0 = np.zeros(n + C)
    x0[n:] = np.maximum(R - target_raw, 0.0)  # z initial
    
    # Ensure feasibility
    x0[:n] = 0.0
    x0[n:] = np.maximum(G @ x0[:n] + R - target_raw, 0.0)
    # z >= G@d + R - target_raw, with d=0: z >= R - target_raw
    x0[n:] = np.maximum(R - target_raw, 0.0)
    
    z_scale = max(np.mean(x0[n:]), 1e-10)
    
    try:
        res = minimize(
            _qp_objective, x0, args=(G, R, target_raw, n, C, rho, lam_z, lam_n, z_scale),
            jac=_qp_grad, method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-8}
        )
        d = res.x[:n].copy()
        d -= np.mean(d)  # ensure zero sum
        if np.max(np.abs(d)) > 1e-14:
            return d
    except Exception:
        pass
    return None

        q = ell + (n - 1)
        if ell >= 0:
            m = n - ell
            idx = np.arange(ell, n)
            G[q, idx] += (1.0 - h[idx - ell])
            idx2 = np.arange(0, m)
            G[q, idx2] += (-h[idx2 + ell])
        else:
            r = -ell
            m = n - r
            idx = np.arange(0, m)
            G[q, idx] += (1.0 - h[idx + r])
            idx2 = np.arange(r, n)
            G[q, idx2] += (-h[idx2 - r])
    return G

# QP objective and gradient (top-level, no closures)
def _qp_obj(x, n, C, rho, lam_z, lam_n, z_scale):
    d = x[:n]
    z = x[n:]
    return lam_z * float(np.sum(z)) / (C * z_scale) + lam_n * float(np.sum(d * d)) / (n * rho * rho)

def _qp_grad(x, n, C, rho, lam_z, lam_n, z_scale):
    g = np.zeros(len(x))
    g[:n] = 2.0 * lam_n * x[:n] / (n * rho * rho)
    g[n:] = lam_z / (C * z_scale)
    return g

# Constraint functions (top-level)
def _qp_ineq(x, params):
    n, G, R, target = params
    d = x[:n]
    z = x[n:]
    return -G @ d + (target - R) + z

def _qp_ineq_jac(x, params):
    n, G, R, target = params
    C = len(R)
    jac = np.zeros((C, n + C))
    jac[:, :n] = -G
    idx = np.arange(C)
    jac[idx, n + idx] = 1.0
    return jac

def _qp_eq(x, n):
    return float(np.sum(x[:n]))

def _qp_eq_jac(x, n):
    j = np.zeros(n + (2 * n - 1))
    j[:n] = 1.0
    return j

def solve_target_qp(h, target_raw, rho, maxiter=150):
    n = len(h)
    C = 2 * n - 1
    R = np.correlate(h, 1.0 - h, 'full')
    G = compute_grad_matrix(h)
    
    lam_z = 0.8
    lam_n = 0.2
    z_scale = max(float(np.mean(np.maximum(R - target_raw, 0.0))), 1e-10)
    
    d_lo = -np.minimum(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    z_lo = np.zeros(C)
    z_hi = np.maximum(R - target_raw + np.max(np.abs(G)) * rho + 2.0, 1.0)
    
    bounds = list(zip(d_lo.tolist(), d_hi.tolist())) + list(zip(z_lo.tolist(), z_hi.tolist()))
    
    # Initial point: d=0, z = max(R - target, 0)
    x0 = np.zeros(n + C)
    x0[n:] = np.maximum(R - target_raw, 0.0)
    
    params = (n, G, R, target_raw)
    
    constraints = [
        {'type': 'ineq', 'fun': _qp_ineq, 'jac': _qp_ineq_jac, 'args': params},
        {'type': 'eq', 'fun': _qp_eq, 'jac': _qp_eq_jac, 'args': (n,)}
    ]
    
    obj_args = (n, C, rho, lam_z, lam_n, z_scale)
    grad_args = (n, C, rho, lam_z, lam_n, z_scale)
    
    try:
        res = minimize(_qp_obj, x0, args=obj_args,
                       jac=lambda x: _qp_grad(x, *grad_args),
                       method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-8, 'iprint': 0})
        d = res.x[:n].copy()
        d -= np.mean(d)
        if np.max(np.abs(d)) > 1e-14:
            return d
    except Exception:
        pass
    return None

Now let me think about the line search more carefully. After getting direction d from the QP, I need to do an exact line search. The formula is:

R(alpha) = R0 + alpha * D - alpha^2 * Q

where D = corr(p, 1-h) - corr(h, p) and Q = corr(p, p).

This is exact because:
corr(h + αp, 1 - h - αp) = corr(h,1-h) + α*corr(p,1-h) - α*corr(h,p) - α²*corr(p,p)

Wait, let me verify:
corr(h+αp, 1-h-αp) = sum (h_i + αp_i)(1 - h_i - αp_i)
= sum h_i(1-h_i) - α*h_i*p_i + α*p_i(1-h_i) - α²*p_i²
= corr(h,1-h) + α*(corr(p,1-h) - corr(h,p)) - α²*corr(p,p)

Yes! This is exact (for the raw correlation). So the line search is exact.

But I need to be careful: the formula gives the raw correlation at each lag. The C5 is max over all lags times dx. So:

C5(α) = max_lag [R0_lag + α*D_lag - α²*Q_lag] * dx

This is the maximum of a set of quadratic functions, which is convex in α (since each quadratic opens downward... wait, -α²*Q_lag. Q_lag = corr(p,p) at lag... actually Q is the autocorrelation of p, which is always >= 0 at lag 0, but can be negative at other lags.

Hmm wait. Q(ell) = sum p[t+ell] * p[t] (for ell >= 0). This is not necessarily non-negative for all ell. So the quadratic -α² * Q(ell) can be upward opening for some ell.

This means C5(α) as a function of α is NOT necessarily convex. It's the max of a mix of upward and downward parabolas. So the golden section search might not find the global minimum.

However, for small α (which is what we expect since rho limits the step), the linear term dominates and the function is roughly linear (decreasing if the direction is good). So the minimum should be at a small α, and golden section should work fine in practice.

def c5_exact(h):

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float)
    v_min = float(v.min()) - 2.0
    v_max = float(v.max()) + 2.0
    lo, hi = v_min, v_max
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        else:
            r = -ell
            m = n - r
            G[q, :m] += (1.0 - h[r:r + m])
            G[q, r:] += (-h[:m])
