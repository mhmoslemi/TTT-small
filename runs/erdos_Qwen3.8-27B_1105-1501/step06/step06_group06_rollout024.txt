def _project(v, target):

    n = len(v)

        return np.zeros(n)

        else:

def _correvals(h):

def _corr_grad_k(k, h, n):

    s = k - (n - 1)  # actual shift

    else:

        g[g2:g2+m] += (1.0 - h[:m])

def _smooth_max_grad(h, sigma, corr_vals=None, grads=None):

    shifted = sigma * (corr_vals - max_c)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    g = weights @ grads
    return g

def _resample(h, n_new):
    n = len(h)

        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                radius *= 0.7
                continue
            dnorm = np.linalg.norm(d2)
            if dnorm < 1e-12:
                stall += 5
                radius *= 0.7
                continue
            d = d2 / dnorm * radius
        alpha, new_c5 = _line_search_2nd(h, d, radius, dx)
        if new_c5 < best_c5 - 1e-14:
            h = _project(h + alpha * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            if abs(alpha) > 0.5 * radius:
                radius = min(radius * 1.15, 0.3)
            else:
                radius = min(radius * 1.03, 0.25)
            tol = min(tol * 1.3, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if radius < 1e-6:
                break
    return best_h, best_c5

def _smooth_pgd(h0, sigma, deadline, max_iter=3000):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = 0.5 / (sigma * 3.0)
    momentum = np.zeros(n)
    beta = 0.85
    for it in range(max_iter):
        if time.time() > deadline:
            break
        cv = _correvals(h)
        max_c = np.max(cv)
        sh = sigma * (cv - max_c)
        exps = np.exp(np.clip(sh, -50, 0))
        w = exps / (np.sum(exps) + 1e-30)
        G = _all_grads(h)
        g = (w * (G * dx).T).T.sum(axis=0)
        g -= g.mean()
        momentum = beta * momentum + (1.0 - beta) * g
        step = -lr * momentum
        sn = np.linalg.norm(step)
        if sn > 0.05:
            step = step * 0.05 / sn
        h_new = _project(h + step, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < _exact_c5(h) - 1e-12:
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.85
        h = h_new
    return best_h, best_c5

def _nm_reduced(h, active_idx, deadline, max_iter=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    m = min(len(active_idx), 25)
    if m == 0:
        return h.copy(), _exact_c5(h)
    G = _all_grads(h)[:m, :] * dx if active_idx[0] < 2 * n - 1 else _all_grads(h)
    G_act = _all_grads(h)[active_idx[:m]] * dx
    A = np.vstack([G_act, np.ones(n)])
    U, S, Vt = np.linalg.svd(A, full_matrices=True)
    rank = int(np.sum(S > 1e-7))
    d = n - rank
    if d <= 1:
        return h.copy(), _exact_c5(h)
    if d > 15:
        d = 15
    Z = Vt[rank:rank + d, :].T
    def _obj(theta):
        hh = _project(h + Z @ theta, target)
        return _exact_c5(hh)
    simplex = [np.zeros(d)]
    step0 = 0.03
    for i in range(d):
        pt = np.zeros(d)
        pt[i] = step0
        simplex.append(pt)
    ndim = d + 1
    vals = [_obj(simplex[i]) for i in range(ndim)]
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        order = np.argsort(vals)
        if vals[order[-1]] - vals[order[0]] < 1e-11:
            break
        centroid = np.mean(simplex[order[:ndim - 1]], axis=0)
        worst = simplex[order[-1]]
        xr = centroid + 1.0 * (centroid - worst)
        fr = _obj(xr)
        if fr < vals[order[0]]:
            xe = centroid + 2.0 * (centroid - worst)
            fe = _obj(xe)
            if fe < fr:
                simplex[order[-1]], vals[order[-1]] = xe, fe
            else:
                simplex[order[-1]], vals[order[-1]] = xr, fr
        elif fr < vals[order[-2]]:
            simplex[order[-1]], vals[order[-1]] = xr, fr
        else:
            xc = centroid + 0.5 * (worst - centroid)
            fc = _obj(xc)
            if fc < vals[order[-1]]:
                simplex[order[-1]], vals[order[-1]] = xc, fc
            else:
                for i in range(1, ndim):
                    simplex[i] = simplex[order[0]] + 0.5 * (simplex[i] - simplex[order[0]])
                    vals[i] = _obj(simplex[i])
    bi = int(np.argmin(vals))
    h_new = _project(h + Z @ simplex[bi], target)
    return h_new, _exact_c5(h_new)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_polish(h, deadline, max_outer=8):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _correvals(best_h) * dx
        M = np.max(corr)
        K = min(20, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        G_top = _all_grads(best_h)[top_k] * dx
        x0 = np.concatenate([best_h, [M + 1e-5]])
        def _obj(x):
            return x[n]
        def _og(x):
            gd = np.zeros(n + 1)
            gd[n] = 1.0
            return gd
        def _ineq(x):
            c = np.correlate(x[:n], 1.0 - x[:n], mode='full') * dx
            return x[n] - c[top_k]
        def _ijac(x):
            J = np.zeros((K, n + 1))
            J[:, :n] = -G_top
            J[:, n] = 1.0
            return J
        def _eq(x):
            return np.sum(x[:n]) - target
        def _eqj(x):
            ge = np.zeros(n + 1)
            ge[:n] = 1.0
            return ge
        res = minimize(_obj, x0, method='SLSQP', jac=_og,
                       bounds=[(0.0, 1.0)] * n + [(0.0, 2.0)],
                       constraints=[{'type': 'ineq', 'fun': _ineq, 'jac': _ijac},
                                    {'type': 'eq', 'fun': _eq, 'jac': _eqj}],
                       options={'maxiter': 30, 'ftol': 1e-13})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                break
        else:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * 3.0 * np.arange(n0) / n0), 0.0, 1.0)

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Active set refinement at parent resolution (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _active_refine(h, d1, tol=5e-5, max_rounds=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Resolution sweep (35%)
    d2 = t0 + budget_s * 0.50
    base_n = len(best_h)
    tried = set([base_n])
    # Systematic sweep of nearby resolutions
    for dn in list(range(-20, 21, 2)):
        if time.time() > d2:
            break
        n_try = base_n + dn
        if n_try < 70 or n_try > 170 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        alloc = min(budget_s * 0.012, 8.0)
        sub_dl = min(d2, time.time() + alloc)
        h_r, c5_r = _active_refine(h_r, sub_dl, tol=5e-5, max_rounds=150)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            base_n = n_try
            tried = set([base_n])

    # Phase 3: Smooth max PGD at best resolution (15%)
    d3 = t0 + budget_s * 0.65
    for sig in [40.0, 80.0, 150.0, 300.0]:
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + budget_s * 0.03)
        h3, c5_3 = _smooth_pgd(best_h, sig, sub_dl, max_iter=2000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Active set refinement (tighter) (15%)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = _active_refine(best_h, d4, tol=2e-5, max_rounds=400)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Null-space Nelder-Mead escape (10%)
    d5 = t0 + budget_s * 0.90
    if time.time() < d5:
        corr = _correvals(best_h) * (2.0 / len(best_h))
        M = np.max(corr)
        active = np.where(corr > M - 2e-4)[0]
        if 0 < len(active) < len(best_h) - 5:
            sub_dl = min(d5, time.time() + budget_s * 0.08)
            h5, c5_5 = _nm_reduced(best_h, active, sub_dl, max_iter=300)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()

    # Phase 6: Final resolution fine-tune + polish (10%)
    d6 = deadline
    # Try a few more resolutions around best
    base_n2 = len(best_h)
    for dn in [-4, 4, -6, 6, -8, 8, -10, 10]:
        if time.time() > d6 - 30:
            break
        n_try = base_n2 + dn
        if n_try < 70 or n_try > 170 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6 - 30, time.time() + budget_s * 0.01)
        h_r, c5_r = _active_refine(h_r, sub_dl, tol=3e-5, max_rounds=100)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # SLSQP final polish
    if time.time() < d6 - 5:
        sub_dl = min(d6, time.time() + budget_s * 0.04)
        h6, c5_6 = _slsqp_polish(best_h, sub_dl, max_outer=10)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final active refinement
    if time.time() < d6 - 2:
        h7, c5_7 = _active_refine(best_h, d6, tol=1e-5, max_rounds=200)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Validate
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, float(best_c5), len(best_h)
