def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):

            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

            mask = (x < 1.0) & (x > 0.0)  # middle elements

    return x

def _project(v, target):

        return np.zeros(n)

            lo = mid
        else:
            hi = mid

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

        t_hi = radius
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        if t_hi < 1e-14:
            stall += 1
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo_t, hi_t = 0.0, t_hi
        c1 = hi_t - gr * (hi_t - lo_t)
        c2 = lo_t + gr * (hi_t - lo_t)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(30):
            if hi_t - lo_t < 1e-11:
                break
            if f1 < f2:
                hi_t = c2
                c2 = c1
                f2 = f1
                c1 = hi_t - gr * (hi_t - lo_t)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_t = c1
                c1 = c2
                f1 = f2
                c2 = lo_t + gr * (hi_t - lo_t)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_t + hi_t) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
        f0 = float(np.max(S0)) * dx
        best_t = min([(0.0, f0), (opt_t, opt_val), (t_hi, float(np.max(S0 + t_hi*T - t_hi*t_hi*C))*dx)], key=lambda x: x[1])[0]
        best_v = min(f0, opt_val, float(np.max(S0 + t_hi*T - t_hi*t_hi*C))*dx)
        if best_v < cur_c5 - 1e-13:
            h = h + best_t * d
            h = _project(h, target)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = float(np.max(S0)) * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.3, 6.0, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amplitudes = rng.uniform(0.5, 1.0, size=3)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amplitudes):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP warmup (5%)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _lp_active(h_init, d1, radius_init=0.08, stall_limit=40, top_k=10)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent (40%) - main workhorse
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.20), max_sweeps=3000, seed=123)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Second round of coord descent with different seed
    h2b, c5_2b = _coord_descent(best_h, min(d2, time.time() + budget_s * 0.15), max_sweeps=2000, seed=456)
    if c5_2b < best_c5:
        best_c5 = c5_2b
        best_h = h2b.copy()

    # Phase 3: LP refinement (10%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _lp_active(best_h, d3, radius_init=0.06, stall_limit=80, top_k=15)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + pair sweep (15%)
    d4 = t0 + budget_s * 0.70
    h_bin = _to_binary(best_h, n0)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.003, seed=444)
    sub_dl2 = min(d4, time.time() + budget_s * 0.03)
    h_bin, c5_bin = _binary_pair_sweep(h_bin, sub_dl2)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # LP polish on binary
    sub_dl3 = min(d4, time.time() + budget_s * 0.03)
    h_pol = _project(h_bin.copy(), n0 / 2.0)
    h_pol, c5_pol = _lp_active(h_pol, sub_dl3, radius_init=0.05, stall_limit=40, top_k=12)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 5: Multi-n exploration (12%)
    d5 = t0 + budget_s * 0.82
    n_curr = len(best_h)
    tried_ns = set()
    offset_list = list(range(-12, 13, 2))
    for dn in offset_list:
        if time.time() > d5:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.008)
        h_r, c5_r = _coord_descent(h_r, sub_dl, max_sweeps=500, seed=700 + dn)
        if time.time() < d5 - 2:
            sub_dl2 = min(d5, time.time() + budget_s * 0.005)
            h_r, c5_r2 = _lp_active(h_r, sub_dl2, radius_init=0.05, stall_limit=25, top_k=10)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2 if False else h_r
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Intensive final optimization (13%)
    d6 = deadline
    n_final = len(best_h)
    # Coordinate descent
    sub_dl = min(d6, time.time() + budget_s * 0.04)
    h6, c5_6 = _coord_descent(best_h, sub_dl, max_sweeps=3000, seed=999)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # LP
    sub_dl = min(d6, time.time() + budget_s * 0.03)
    h6, c5_6 = _lp_active(best_h, sub_dl, radius_init=0.03, stall_limit=50, top_k=20)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Pair swap
    if time.time() < d6 - 3:
        h6, c5_6 = _pair_swap_loop(best_h, min(d6, time.time() + budget_s * 0.03), n_pairs=4000, radius=0.20, seed=555)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    # Basin hopping
    scale = 0.02
    n_restarts = 0
    while time.time() < d6 - 2 and n_restarts < 60:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6 - 2, time.time() + budget_s * 0.004)
        h_t, c5_t = _coord_descent(h_pert, sub_dl, max_sweeps=200, seed=2000 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Final projection and evaluation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
