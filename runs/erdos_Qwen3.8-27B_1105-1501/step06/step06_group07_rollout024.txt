def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    """Gradient of correlation at lag k with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        # corr[n-1+k] = sum_{j=0}^{m-1} h[j+k]*(1-h[j])
        # d/dh[i]: if i = j+k for some j in [0,m-1]: contributes (1-h[j]) = (1-h[i-k])
        #          if i = j for some j in [0,m-1]: contributes -h[j+k] = -h[i+k]
        idx_top = np.arange(k, n)  # i = j+k, j from 0 to m-1
        g[idx_top] += (1.0 - h[idx_top - k])
        idx_bot = np.arange(0, m)  # i = j, j from 0 to m-1
        g[idx_bot] -= h[idx_bot + k]
    else:

        # corr[n-1-ga] = sum_{j=0}^{m-1} h[j+ga]*(1-h[j])

        g[idx_top] += (1.0 - h[idx_top - ga])
        idx_bot = np.arange(0, m)
        g[idx_bot] -= h[idx_bot + ga]

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vector(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _coord_descend(h, deadline, n_inner=50, max_passes=30):
    """Coordinate descent: for each coord, find best value via grid search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h[i]
            # Compute A_k and B_k
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            m = len(corr0)  # 2n-1
            B = np.zeros(m)
            for k in range(-(n-1), n):
                idx = n - 1 + k
                B_k = 0.0
                # Position i as h[n+k]: n = i-k, valid if 0 <= i-k < n
                if 0 <= i - k < n:
                    B_k += (1.0 - h[i-k])
                # Position i as (1-h[n]): n = i, valid if 0 <= i < n and 0 <= i+k < n
                if 0 <= i + k < n:
                    B_k += (-h[i+k])
                B[idx] = B_k
            # Evaluate on grid
            t_grid = np.linspace(0.0, 1.0, n_inner)
            # C5(t) = max_k (corr0[k] + t*B[k]) * dx
            vals = np.max(corr0[None, :] + t_grid[:, None] * B[None, :], axis=1) * dx
            best_t = t_grid[int(np.argmin(vals))]
            h[i] = best_t
            # Quick check (every few coords or at the end)
            if (i + 1) % 20 == 0 or i == n - 1:
                c5_now = _exact_c5(h)
                if c5_now < best_c5 - 1e-12:
                    best_c5 = c5_now
                    best_h = h.copy()
                    improved = True
            if best_t != old_val:
                improved = True
        if not improved:
            break
    
    return best_h, best_c5

Specifically, if h[i] → h[i]+t and h[j] → h[j]-t, the change in corr[k] is:

The cross terms: when both i and j are modified, there's an interaction term. Specifically, if i+k = j or j+k = i (or i-k=j or j-k=i), there's a t² term.

In general:
corr_new[k] = corr_old[k] + t*(g_i[k] - g_j[k]) + t² * c_ij[k]

where c_ij[k] is 1 if (i+k=j and i-k valid and i+k valid) or (j+k=i and j-k valid and j+k valid), with appropriate signs, and 0 otherwise.

This is a quadratic in t. The max of quadratics is not necessarily convex, but we can still do a grid search.

For the pair direction, the grid search is over t in [t_min, t_max] where:
t_min = max(-h[i], h[j]-1) (ensuring h[i]+t >= 0 and h[j]-t <= 1)
t_max = min(1-h[i], h[j]) (ensuring h[i]+t <= 1 and h[j]-t >= 0)
