def _grad_weighted(h, alpha, K):

    corr = _corr_full(h) * dx
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]

    w_sum = np.sum(w)
    if w_sum < 1e-300:

def _line_search_exact(h, d, dx, max_t=None):
    """Exact line search along sum-zero direction d.
    Returns (best_t, best_objective)."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T1 = np.correlate(d, 1.0 - h, mode='full')
    T2 = np.correlate(h, d, mode='full')
    T = T1 - T2
    C = np.correlate(d, d, mode='full')
    
    # Bounds
    t_lo = 0.0
    t_hi = max_t if max_t is not None else 10.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(-h[neg] / d[neg])))
    
    t_hi = max(t_hi, t_lo)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S)) * dx
    
    # Grid search
    n_grid = min(60, max(20, int(2*(t_hi - t_lo) * 100)))
    n_grid = min(n_grid, 80)
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.empty(n_grid)
    for i, t in enumerate(grid):
        vals[i] = float(np.max(S + t * T - t * t * C)) * dx
    best_i = int(np.argmin(vals))
    best_t = grid[best_i]
    best_val = vals[best_i]
    
    # Golden section in neighborhood
    if best_i > 0 and best_i < n_grid - 1:
        a = grid[best_i - 1]
        b = grid[best_i + 1]
    elif best_i == 0:
        a = t_lo
        b = grid[1]
    else:
        a = grid[-2]
        b = t_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(40):
        if b - a < 1e-10:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S + c2 * T - c2 * c2 * C)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    
    return best_t, best_val

def _pair_swap_delta_binary(corr, h, i, j, n):
    """Compute change in correlation vector when swapping h[i](1→0) and h[j](0→1) for binary h."""
    # Returns delta array of length 2n-1
    delta = np.zeros(2*n-1)
    # h[i]: 1→0
    # As first element at position i (l+k=i → l=i-k):
    #   lag k: needs h[i-k] to be 0 for contribution. i-k valid: 0≤i-k<n → k ≤ i < n-k → k ∈ [i-n+1, i]
    #   But we also need 0≤l<n and 0≤l+k<n. l=i-k, so 0≤i-k<n and 0≤i<n (always true for k range)
    #   Actually for np.correlate(h, 1-h, 'full'), the index is j = n-1+k
    #   corr[j] = Σ_l h[l+k]*(1-h[l]) where k = j-(n-1)
    #   For h[i] changing 1→0, as the "h[l+k]" term (l+k=i, l=i-k):
    #     valid if 0≤l<n and 0≤l+k<n → 0≤i-k<n and 0≤i<n
    #     contribution was (1-h[i-k]) (since h[i]=1), now 0. Change: -(1-h[i-k])
    #     k range: 0≤i-k<n → i-n<k≤i, so k ∈ [i-n+1, i]
    #     j = n-1+k ∈ [n-1+i-n+1, n-1+i] = [i, n-1+i]
    #   As the "h[l]" term (l=i):
    #     valid if 0≤l<n and 0≤l+k<n → 0≤i<n and 0≤i+k<n → -i≤k<n-i
    #     contribution was h[i+k]*(1-h[i]) = h[i+k]*0 = 0, now h[i+k]*(1-0) = h[i+k]. Change: +h[i+k]
    #     k range: max(-i, -(n-1)) ≤ k ≤ min(n-i-1, n-1) → k ∈ [-i, n-i-1] (clamped to [-(n-1), n-1])
    #     j = n-1+k ∈ [n-1-i, n-1+n-i-1] = [n-1-i, 2n-i-2]
    
    # For binary h, h[i-k] is 0 or 1, so (1-h[i-k]) is 1 or 0.
    # Let me just compute it directly using h values.
    
    # For h[i] 1→0:
    # As first (l+k=i): for valid k, change = -(1-h[i-k]) = h[i-k]-1
    lo1 = max(-(n-1), i-(n-1))  # k range lower
    hi1 = min(n-1, i)  # k range upper
    for k in range(lo1, hi1+1):
        l = i - k
        if 0 <= l < n:
            delta[n-1+k] += h[l] - 1.0  # -(1-h[l]) = h[l]-1
    
    # As second (l=i): for valid k, change = +h[i+k]
    lo2 = max(-(n-1), -i)
    hi2 = min(n-1, n-1-i)
    for k in range(lo2, hi2+1):
        m = i + k
        if 0 <= m < n:
            delta[n-1+k] += h[m]
    
    # For h[j] 0→1:
    # As first (l+k=j): change = +(1-h[j-k]) = 1-h[j-k]  (was 0, now (1-h[j-k]))
    lo3 = max(-(n-1), j-(n-1))
    hi3 = min(n-1, j)
    for k in range(lo3, hi3+1):
        l = j - k
        if 0 <= l < n:
            delta[n-1+k] += 1.0 - h[l]
    
    # As second (l=j): change = -h[j+k]  (was h[j+k]*(1-0)=h[j+k], now 0)
    lo4 = max(-(n-1), -j)
    hi4 = min(n-1, n-1-j)
    for k in range(lo4, hi4+1):
        m = j + k
        if 0 <= m < n:
            delta[n-1+k] -= h[m]
    
    return delta

            h_try = h + rng.normal(0, 0.05, n)
        elif s == 1:
            h_try = h + rng.normal(0, 0.15, n)
        elif s == 2:
            # Binary threshold then smooth
            sorted_idx = np.argsort(-h)
            h_bin = np.zeros(n)
            h_bin[sorted_idx[:int(target)]] = 1.0
            h_try = _project(h_bin + rng.normal(0, 0.1, n), target)
        elif s == 3:
            # Random low-freq component
            x = np.arange(n)
            freq = int(rng.integers(1, 6))
            phase = rng.random() * 2.0 * np.pi
            pert = 0.1 * np.sin(2.0 * np.pi * freq * x / n + phase)
            h_try = _project(h + pert, target)
        else:
            # Mirror blend
            h_mirror = h[::-1].copy()
            w = np.random.default_rng(99 + s).random()
            h_try = _project(w * h + (1 - w) * h_mirror, target)

        sub_dl = min(deadline, time.time() + per_seed_time)
        h_pol, c5_pol = _continuous_polish(h_try, sub_dl, K=25, alpha_init=20.0,
                                           max_iters=150, seed_val=200 + s)
        if c5_pol < best_c5 - 1e-14:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        x = np.linspace(0.0, 2.0, n0, endpoint=False) + 1.0 / n0
        h0 = 0.5 + 0.3 * np.sin(2.0 * np.pi * 4.0 * x / 2.0)

    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Main continuous polish (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _continuous_polish(h, d1, K=30, alpha_init=25.0, max_iters=500, seed_val=77)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary SA + continuous polish (20% time)
    d2 = t0 + budget_s * 0.50
    sa_dl = min(d2, time.time() + budget_s * 0.10)
    h_bin, c5_bin = _binary_sa(best_h, sa_dl, rng, max_iters=200000)
    if c5_bin < best_c5:
        pol_dl = min(d2, time.time() + budget_s * 0.07)
        h_pol, c5_pol = _continuous_polish(h_bin, pol_dl, K=25, alpha_init=15.0,
                                           max_iters=200, seed_val=333)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        else:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Multi-resolution search (25% time)
    d3 = t0 + budget_s * 0.75
    best_n = len(best_h)
    offsets = [0, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14,
               -16, 16, -20, 20, -24, 24, -28, 28, -32, 32]
    tried = set([best_n])
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _continuous_polish(h_r, sub_dl, K=25, alpha_init=20.0,
                                       max_iters=60, seed_val=n_try)
        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Diverse restarts (15% time)
    d4 = t0 + budget_s * 0.90
    h_div, c5_div = _diverse_polish(best_h, d4, rng, n_seeds=6, polish_time_frac=0.7)
    if c5_div < best_c5:
        best_c5 = c5_div
        best_h = h_div.copy()

    # Phase 5: Second binary SA with different temperature (8% time)
    d5 = t0 + budget_s * 0.95
    if time.time() < d5 and len(best_h) >= 40:
        h_pert = _project(best_h + rng.normal(0, 0.2, len(best_h)), len(best_h) / 2.0)
        h_bin2, c5_bin2 = _binary_sa(h_pert, d5, rng, max_iters=100000)
        if c5_bin2 < best_c5:
            pol_dl = min(deadline, time.time() + 10.0)
            h_pol2, c5_pol2 = _continuous_polish(h_bin2, pol_dl, K=20, alpha_init=20.0,
                                                 max_iters=80, seed_val=555)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()
            else:
                best_c5 = c5_bin2
                best_h = h_bin2.copy()

    # Final polish
    if time.time() < deadline - 1.0:
        h_f, c5_f = _continuous_polish(best_h, deadline, K=20, alpha_init=35.0,
                                       max_iters=100, seed_val=999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
