from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += (1.0 - h[:m])
        g[:m] -= h[k:]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        g[:m] += (1.0 - h[ga:])
        g[ga:] -= h[:m]
    return g * dx

def _smooth_max_grad(h, alpha, K=30):
    corr = _corr_full(h)
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    s = w.sum()
    if s < 1e-300:
        w = np.ones(K) / K
    else:
        w /= s
    g = np.zeros(len(h))
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (len(h) - 1)
        g += wi * _grad_row(h, k)
    g -= g.mean()
    return g, corr

def _quad_line_search(h, d, max_t=1.0, n_grid=60):
    """Exact line search along direction d (assumed sum-zero)."""
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C_arr = np.correlate(d, d, mode='full')
    
    # Find valid t range
    t_hi = max_t
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
        t_lo = max(0.0, t_lo)
    else:
        t_lo = 0.0
    t_hi = max(t_hi, 0.0)
    if t_hi < 1e-14 or t_hi <= t_lo:
        return 0.0, _exact_c5(h)
    
    # Grid search
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T_arr[None, :] - (grid[:, None] ** 2) * C_arr[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    
    # Golden section refine
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_arr - c1**2 * C_arr)) * dx
    f2 = float(np.max(S0 + c2 * T_arr - c2**2 * C_arr)) * dx
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_arr - c1**2 * C_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_arr - c2**2 * C_arr)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_arr - opt_t**2 * C_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _vertex_descent(h, deadline, max_iters=500, K=25):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = _corr_full(h)
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        M = top_vals[0]
        gap = max(M - top_vals[-1], 1e-10)
        alpha = 50.0 / gap
        w = np.exp(alpha * (top_vals - M))
        w /= w.sum()
        
        g = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            g += wi * _grad_row(h, k)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Generate vertex direction: threshold -g
        # v[i] = 1 if i in top-m of (-g + noise)
        m = n // 2
        scores = -g / gn
        # Add small noise for diversity
        rng_local = np.random.default_rng(it)
        noise = rng_local.normal(0, 0.1, n)
        scores_noisy = scores + noise
        
        # Get binary vertex
        v = np.zeros(n)
        v[np.argsort(-scores_noisy)[:m]] = 1.0
        
        # Direction from h to v
        d = v - h
        d_sum = d.sum()
        if abs(d_sum) > 1e-10:
            d -= d_sum / n  # Make sum-zero (shouldn't be needed for binary->binary)
        
        # Line search
        t, f_try = _quad_line_search(h, d, max_t=1.0)
        if t > 1e-10:
            h_new = h + t * d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    return best_h, best_c5
