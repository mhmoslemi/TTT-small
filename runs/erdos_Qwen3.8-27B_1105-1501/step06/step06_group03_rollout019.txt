def _smoothed_grad(h, S, n, tau):
    """Compute gradient of log-sum-exp smoothing."""
    # softmax weights
    S_shifted = S - S.max()
    w = np.exp(S_shifted / tau)
    w_sum = w.sum()
    w = w / w_sum
    # Get top lags for efficiency
    n_top = min(50, 2*n-1)
    top_idx = np.argsort(-S)[:n_top]
    filter_w = w[top_idx] / w[top_idx].sum()  # renormalize
    G = _grad_rows(h, top_idx, n)
    grad = filter_w @ G
    grad -= grad.mean()  # remove mean (for the equality constraint)
    return grad

def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        g = G[idx]

            d = -k
            m = n - d

    n_act = len(combined)

    A_ub[:, n] = 1.0

def _smoothed_grad(h, S, n, tau):
    S_shifted = S - S.max()
    w = np.exp(S_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    n_top = min(60, 2 * n - 1)
    top_idx = np.argsort(-S)[:n_top]
    fw = w[top_idx]
    fw_sum = fw.sum()
    if fw_sum < 1e-300:
        return None
    fw = fw / fw_sum
    G = _grad_rows(h, top_idx, n)
    grad = fw @ G
    grad -= grad.mean()
    if np.linalg.norm(grad) < 1e-12:
        return None
    return grad

        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, radius, K)
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            tau = 0.02 * max(cur_c5, 0.01)
            p = _smoothed_grad(h, S, n, tau)
            if p is not None:
                p = p * radius
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if radius < 1e-7:
                break
            continue
        
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.06, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.93, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.05
    
    return best_h, best_c5

def _smoothed_optimize(h, deadline, tau_init=0.05, tau_end=0.003):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.02
    max_iter = 500
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        progress = min(it / max(max_iter - 1, 1), 1.0)
        tau = tau_init * (tau_end / tau_init) ** progress
        
        S = _corr_scaled(h)
        grad = _smoothed_grad(h, S, n, tau)
        if grad is None:
            break
        
        improved = False
        for _ in range(15):
            h_try = h - step * grad
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_h = h.copy()
                best_c5 = c5_try
                improved = True
                step = min(step * 1.15, 0.08)
                break
            step *= 0.7
        if not improved:
            step *= 0.6
            if step < 1e-9:
                break
    
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(333)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        G = _grad_rows(h, np.arange(2*n-1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        
        n_int = min(n, 40)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
