```python
    """Compute correlation change when h[p]:1->0 and h[q]:0->1."""
    delta = np.zeros(n)
    delta[p] = -1.0
    delta[q] = 1.0
    
    change = np.zeros(2*n-1)
    for pos in [p, q]:
        d = delta[pos]
        if d == 0:
            continue
        # Condition 1: k in [pos-(n-1), pos], change += d*(1-h[pos-k])
        # Condition 2: k in [-pos, n-1-pos], change += -d*h[pos+k]
        # Correction at k=0: -= d^2
        
        # Condition 1
        k_lo = pos - (n-1)
        k_hi = pos
        for k in range(k_lo, k_hi+1):
            n_idx = pos - k
            if 0 <= n_idx < n:
                change[k + (n-1)] += d * (1.0 - h[n_idx])
        
        # Condition 2
        k_lo2 = -pos
        k_hi2 = n-1-pos
        for k in range(k_lo2, k_hi2+1):
            n_idx = pos + k
            if 0 <= n_idx < n:
                change[k + (n-1)] += -d * h[n_idx]
        
        # Correction
        change[n-1] -= d * d
    
    return change
```

This is O(n) per swap but with Python loops. For n=122, that's about 244 iterations per swap, which is fine.

But we can vectorize it:

```python
def _swap_update_corr(h, corr, p, q, n):
    change = np.zeros(2*n-1)
    for pos, d in [(p, -1.0), (q, 1.0)]:
        # Condition 1: k in [pos-(n-1), pos]
        k_vals = np.arange(pos-(n-1), pos+1)
        n_vals = pos - k_vals  # = n-1, n-2, ..., 0
        # all n_vals are in [0, n-1] since pos in [0,n-1]
        change[k_vals + (n-1)] += d * (1.0 - h[n_vals])
        
        # Condition 2: k in [-pos, n-1-pos]
        k_vals2 = np.arange(-pos, n-pos)
        n_vals2 = pos + k_vals2  # = 0, 1, ..., n-1-pos
        # all n_vals2 are in [0, n-1]
        change[k_vals2 + (n-1)] += -d * h[n_vals2]
        
        # Correction at k=0
        change[n-1] -= d * d
    return change
```
