        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def _obj_fn(x):
    return x[-1]


def _obj_grad(x):
    n = len(x) - 1
    g = np.zeros(len(x))
    g[n] = 1.0
    return g


def _ineq_fun(x):
    n = _CURRENT_N
    h = x[:n]
    z = x[n]
    s = _corr_values(h)
    return z - s


def _ineq_jac(x):
    n = _CURRENT_N
    h = x[:n]
    nfull = 2 * n - 1
    dx = 2.0 / n
    jac = np.zeros((nfull, n + 1))
    for ell in range(-(n - 1), n):
        r = ell + (n - 1)
        g = np.zeros(n)
        if ell == 0:
            g[:] = dx * (1.0 - 2.0 * h)
        elif ell > 0:
            g[ell:] += dx * (1.0 - h[:n - ell])
            g[:n - ell] += dx * (-h[ell:])
        else:
            q = -ell
            g[:n - q] += dx * (1.0 - h[q:])
            g[q:] += dx * (-h[:n - q])
        jac[r, :n] = -g
        jac[r, n] = 1.0
    return jac


def _eq_fun(x):
    n = _CURRENT_N
    return float(x[:n].sum() - n / 2.0)


def _eq_jac(x):
    g = np.zeros(len(x))
    g[:_CURRENT_N] = 1.0
    return g


def _slsqp_solve(h, maxiter=250, ftol=1e-10):
    global _CURRENT_N
    n = len(h)
    _CURRENT_N = n
    h = _project(h, n / 2.0)
    z0 = _exact_c5(h) + 0.005
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]
    try:
        res = _minimize(_obj_fn, x0, method='SLSQP', jac=_obj_grad,
                        bounds=bounds, constraints=cons,
                        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], n / 2.0)
            f_new = _exact_c5(h_new)
            if f_new < _exact_c5(h):
                return h_new, f_new
    except Exception:
        pass
    return h, _exact_c5(h)


def _slsqp_tight(h, margin=0.002, maxiter=200):
    """SLSQP with z bounded above to force tighter solution."""
    global _CURRENT_N
    n = len(h)
    _CURRENT_N = n
    h = _project(h, n / 2.0)
    cur_f = _exact_c5(h)
    z0 = cur_f * (1.0 + margin)
    z_ub = cur_f * (1.0 + margin)
    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z_ub)]
    cons = [
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]
    try:
        res = _minimize(_obj_fn, x0, method='SLSQP', jac=_obj_grad,
                        bounds=bounds, constraints=cons,
                        options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n], n / 2.0)
            f_new = _exact_c5(h_new)
            if f_new < cur_f:
                return h_new, f_new
    except Exception:
        pass
    return h, cur_f


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 120
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv


def _row_grad_raw(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nn = np.arange(0, n - k)
    else:
        nn = np.arange(-k, n)
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g


def _active_dir(h, raw_corr, k_active, tau, rng):
    n = len(h)
    scores = raw_corr
    m = float(np.max(scores))
    w = np.exp((scores - m) / tau)
    idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = r - (n - 1)
        G += w[r] * _row_grad_raw(h, k)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm


def _optimize_ls(h, deadline, rng, pair_frac=0.18, rand_frac=0.10, k_active=8, tau=0.02):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    restarts = 0
    sigma = 0.04
    uphill = 5e-4
    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            d = _active_dir(h, raw_corr, k_active, tau, rng)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(raw_corr, T, Q, dx, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-12 or hn.max() > 1.0 + 1e-12:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-13:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            raw_corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            raw_corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
        if stall > 120:
            restarts += 1
            if restarts > 25:
                h = _project(best_h + rng.normal(0, max(0.04, sigma * 3), n), target)
            else:
                h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            raw_corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.003, sigma * 0.87)
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 5 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def _two_phase_slsqp(h, maxiter1=200, maxiter2=300):
    """Two-phase: first relax z, then tighten."""
    h1, f1 = _slsqp_solve(h, maxiter=maxiter1, ftol=1e-9)
    if f1 < _exact_c5(h):
        h2, f2 = _slsqp_tight(h1, margin=0.003, maxiter=maxiter2)
        return h2, f2
    return h1, f1


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        base = _sine_init(n, 1, 0.0, rng)
    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    f0 = _exact_c5(h0)

    best_h = h0.copy()
    best_f = f0

    # Phase 1: SLSQP epigraph optimization (35% budget)
    d1 = t0 + budget_s * 0.35
    h_sls, f_sls = _two_phase_slsqp(h0, maxiter1=200, maxiter2=300)
    if f_sls < best_f:
        best_f = f_sls
        best_h = h_sls.copy()

    # SLSQP from complement
    if time.time() < d1:
        h_comp = _project(1.0 - best_h, n0 / 2.0)
        h_sls, f_sls = _two_phase_slsqp(h_comp, maxiter1=150, maxiter2=200)
        if f_sls < best_f:
            best_f = f_sls
            best_h = h_sls.copy()

    # SLSQP from perturbations
    perturb_sizes = [0.025, 0.04, 0.06, 0.08, 0.035, 0.05, 0.07, 0.10]
    for ps in perturb_sizes:
        if time.time() > d1:
            break
        h_pert = _project(best_h + rng.normal(0, ps, n0), n0 / 2.0)
        h_sls, f_sls = _slsqp_solve(h_pert, maxiter=200, ftol=1e-10)
        if f_sls < best_f:
            best_f = f_sls
            best_h = h_sls.copy()

    # SLSQP from structured starts
    for f_, p_ in [(1, 0.3), (2, 1.0), (3, 0.5), (4, 0.0)]:
        if time.time() > d1:
            break
        h_struct = _sine_init(n0, f_, p_, rng)
        h_sls, f_sls = _slsqp_solve(h_struct, maxiter=150, ftol=1e-9)
        if f_sls < best_f:
            best_f = f_sls
            best_h = h_sls.copy()

    # Phase 2: Line-search active direction (30% budget)
    d2 = t0 + budget_s * 0.65
    hh, ff = _optimize_ls(best_h, d2, rng, pair_frac=0.15, rand_frac=0.08, k_active=10, tau=0.015)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: SLSQP refinement + multi-resolution (25% budget)
    d3 = t0 + budget_s * 0.90
    # One more SLSQP pass on current best
    if time.time() < d3:
        h_sls, f_sls = _two_phase_slsqp(best_h, maxiter1=200, maxiter2=250)
        if f_sls < best_f:
            best_f = f_sls
            best_h = h_sls.copy()

    # Multi-resolution SLSQP
    bn = len(best_h)
    tried = set()
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, 10, -10, 12, -12]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 24 or n_try > 200:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        h_sls, f_sls = _slsqp_solve(hr, maxiter=200, ftol=1e-10)
        if f_sls < best_f:
            best_f = f_sls
            best_h = h_sls.copy()
            bn = n_try

    # Phase 4: Final line-search polish (10% budget)
    hh, ff = _optimize_ls(best_h, deadline, rng, pair_frac=0.50, rand_frac=0.05, k_active=12, tau=0.010)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Final SLSQP tight pass if time remains
    if time.time() < deadline - 10:
        h_sls, f_sls = _slsqp_tight(best_h, margin=0.002, maxiter=200)
        if f_sls < best_f:
            best_f = f_sls
            best_h = h_sls.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
