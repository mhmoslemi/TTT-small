- 0 ≤ a ≤ 1, sum a = n/2
- 0 ≤ b ≤ 1, sum b = n/2

- u1 = a* is a valid original candidate (balanced, in [0,1])

The key insight: the relaxed problem has a lower optimal value than the original (since the feasible set is larger), but the solutions can still guide us to good original solutions.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_corr_matrix(fixed_vec):
    """Build matrix A where A[j,i] gives coefficient of fixed_vec in corr(e_i, fixed_vec)[j].
    
    For np.correlate(a, b, 'full'): output[j] = sum_i a[i] * b[i + n-1-j]
    So if b is fixed, A[j,i] = b[i+n-1-j] (if valid index, else 0)
    """
    n = len(fixed_vec)
    A = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        offset = n - 1 - j  # b index = i + offset
        # b[i + offset] valid when 0 <= i+offset < n
        i_start = max(0, -offset)
        i_end = min(n, n - offset)
        if i_start < i_end:
            A[j, i_start:i_end] = fixed_vec[i_start+offset:i_end+offset]
    return A

def _lp_minimax(A, target):
    """Solve: min tau s.t. A@x <= tau*1, sum(x)=target, 0<=x<=1"""
    n = A.shape[1]
    m = A.shape[0]
    # Variables: x[0:n], tau = x[n]
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize tau
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def _relaxed_pair_optimize(h_init, deadline, n_max_cy=30):
    """Alternating relaxed LP optimization."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    a = h_init.copy()
    b = 1.0 - h_init.copy()
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    
    for cyc in range(n_max_cy):
        if time.time() > deadline:
            break
        # LP in a with b fixed
        A_a = _build_corr_matrix(b)
        a_new, tau_a = _lp_minimax(A_a, target)
        if a_new is not None:
            a = _project(a_new, target)
            # Check a as candidate
            c5_a = _exact_c5(a)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a.copy()
        
        # LP in b with a fixed
        A_b = _build_corr_matrix(a)
        # For corr(a, b): output[j] = sum_i a[i+j-(n-1)] * b[i] ... 
        # Actually: output[j] = sum_i a[i] * b[i+n-1-j] when b varies
        # By symmetry, we need: output[j] = sum_i b[i] * a[j-i+(n-1)] = ...
        # Let's use: output[j] = sum_l b[l] * coeff[j,l]
        # From: output[j] = sum_i a[i] * b[i+n-1-j]
        # Let l = i+n-1-j, so i = l-j+n-1
        # output[j] = sum_l a[l-j+n-1] * b[l] (where valid)
        # So coefficient of b[l] in output[j] is a[l-j+n-1] if valid
        A_b2 = np.zeros((2*n-1, n))
        for j in range(2*n-1):
            # coeff of b[l] in output[j] is a[l-j+n-1] if 0<=l-j+n-1<n
            # l range: j-(n-1) <= l <= j
            l_start = max(0, j - (n-1))
            l_end = min(n, j + 1)
            if l_start < l_end:
                idx = np.arange(l_start, l_end)
                a_idx = idx - j + n - 1
                A_b2[j, l_start:l_end] = a[a_idx]
        b_new, tau_b = _lp_minimax(A_b2, target)
        if b_new is not None:
            b = _project(b_new, target)
            # Check 1-b as candidate
            cand = 1.0 - b
            cand = _project(cand, target)
            c5_b = _exact_c5(cand)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = cand.copy()
        
        # Check segment between a and 1-b
        u1 = a.copy()
        u2 = _project(1.0 - b, target)
        for t in [0.25, 0.5, 0.75]:
            seg = (1-t)*u2 + t*u1
            seg = _project(seg, target)
            c5_seg = _exact_c5(seg)
            if c5_seg < best_c5:
                best_c5 = c5_seg
                best_h = seg.copy()
    
    return best_h, best_c5

def _subgradient_descent(h_init, deadline, n_max=500):
    """Subgradient method with projection for the max-of-correlations objective."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = 0.01
    for it in range(n_max):
        if time.time() > deadline:
            break
        S = _corr_vals(h)
        S_max = float(np.max(S))
        # Subgradient of max_k S_k w.r.t. h
        # S_k = dx * sum_i h[i] * (1-h[i+k-offset])
        # dS_k/dh[j] = dx * (1-h[j+k-offset]) + dx * h[j] * (-1) * delta... 
        # Actually it's bilinear, so gradient is complex.
        # Use the active index subgradient
        active = np.argmax(S)
        # Build gradient for the active lag
        # S_active = dx * sum_i h[i] * (1-h[i+(n-1-active)])
        # Let offset = n-1-active
        # S_active = dx * sum_i h[i] * (1-h[i+offset]) (with bounds)
        offset = n - 1 - active
        grad = np.zeros(n)
        for i in range(n):
            # d/dh[j] of h[i]*(1-h[i+offset])
            # If j == i: (1-h[i+offset])
            # If j == i+offset: -h[i]
            grad[i] += (1.0 - (h[i+offset] if 0 <= i+offset < n else 0.0))
            if 0 <= i+offset < n:
                grad[i+offset] += -h[i]
        grad *= dx
        
        h = h - lr * grad
        h = _project(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.95
        if it % 20 == 0:
            lr = max(lr * 0.9, 0.001)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gen_start(n, rng, mode=0):
    """Generate diverse starting points."""
    x = np.arange(n)
    if mode == 0:
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.35*np.sin(2*np.pi*freq*x/n + phase)
    elif mode == 1:
        # Block pattern
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, int(rng.integers(1, n//6+1)))
            if rng.random() < 0.5:
                h[pos:pos+bl] = 1.0
            pos += bl
    elif mode == 2:
        # Two-frequency
        f1, f2 = rng.integers(1, 4), rng.integers(4, 8)
        p1, p2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.25*np.sin(2*np.pi*f1*x/n+p1) + 0.15*np.sin(2*np.pi*f2*x/n+p2)
    else:
        h = 0.5 + 0.1*rng.standard_normal(n)
    h += 0.05 * rng.standard_normal(n)
    return _project(h, n/2.0)

def _refine_pair(h, deadline):
    """After relaxed pair, do coordinate refinement."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    
    for it in range(200):
        if time.time() > deadline:
            break
        S = _corr_vals(h)
        max_idx = np.argmax(S)
        offset = n - 1 - max_idx
        
        # Compute gradient of active correlation
        grad = np.zeros(n)
        for i in range(n):
            j1 = i
            j2 = i + offset
            if 0 <= j1 < n:
                val2 = h[j2] if 0 <= j2 < n else 0.0
                grad[j1] += dx * (1.0 - val2)
            if 0 <= j2 < n:
                grad[j2] += dx * (-h[j1])
        
        # Pick coordinate with largest gradient and move it
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        
        # Find pair (i,j) to swap mass
        order = np.argsort(-np.abs(grad[interior]))
        improved = False
        for idx in order[:10]:
            i = interior[idx]
            # Find best partner
            others = np.setdiff1d(interior, [i])
            if len(others) == 0:
                continue
            for j in others[:5]:
                for sign in [1, -1]:
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    # Line search
                    # S_k(t) = S_k + t * T_k - t^2 * C_k (bilinear)
                    S0 = np.correlate(h, 1-h, mode='full') * dx
                    T = (np.correlate(d, 1-h, mode='full') - np.correlate(h, d, mode='full')) * dx
                    C = np.correlate(d, d, mode='full') * dx
                    # Minimize max_k (S0[k] + t*T[k] - t^2*C[k])
                    # Try a few t values
                    best_t, best_val = 0.0, float(np.max(S0))
                    for t_try in np.linspace(-0.3, 0.3, 20):
                        # Check bounds
                        h_try = h + t_try * d
                        if h_try.min() < -1e-10 or h_try.max() > 1.0+1e-10:
                            continue
                        S_try = S0 + t_try*T - t_try*t_try*C
                        val = float(np.max(S_try))
                        if val < best_val:
                            best_val = val
                            best_t = t_try
                    if best_val < best_c5 - 1e-14:
                        h = h + best_t * d
                        h = _project(h, target)
                        c5 = _exact_c5(h)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                            improved = True
                            break
                if improved:
                    break
        if not improved:
            # Random perturbation
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            if _exact_c5(h) >= best_c5:
                h = best_h.copy()
                if it > 100:
                    break
    return best_h, best_c5

               - h[j-o] * I(0<=j-o<n and 0<=j<n) * I(j-o is in valid range for i)]

The valid range for i is [max(0,-o), min(n, n-o)).
So j-o is valid if max(0,-o) <= j-o < min(n, n-o), i.e., max(o, 0) <= j < min(n, n-o)+o.

For 0 <= o < n (which is the case when k >= 0):
- First term: valid when 0 <= j < n and 0 <= j+o < n, i.e., 0 <= j < n-o
- Second term: valid when 0 <= j-o < n-o, i.e., o <= j < 2n-o. Combined with 0<=j<n: o <= j < n.

So for o in [0, n-1]:
dS_k/dh[j] = dx * [(1-h[j+o]) for j in [0, n-o)  -  h[j-o] for j in [o, n)]

For o < 0 (k > n-1), let o' = -o in [1, n-1]:
S_k = dx * sum_i h[i] * (1-h[i-o']) where 0 <= i-o' < n, i.e., o' <= i < n.
Valid range: i in [o', n).
dS_k/dh[j] = dx * [
  First term (j is the "first h"): (1-h[j-o']) for j in [o', n)
  Second term (j is in "1-h"): -h[j+o'] for j in [0, n-o')
]

So the gradient computation is O(n) per lag, or O(n^2) for all lags. With n=122, that's about 15000 operations per gradient evaluation, which is fast.

For the smoothed objective, I need the weighted sum of gradients across all lags, which is O(n^2) total. That's fine.

Let me now write a cleaner implementation:

def _smoothed_gradient(h, tau):
    """Compute gradient of smoothed max objective."""
    n = len(h)
    dx = 2.0 / n
    S = _corr_vals(h)
    S_max = float(np.max(S))
    weights = np.exp(tau * (S - S_max))
    weights /= float(np.sum(weights))
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        o = n - 1 - k  # offset
        if o >= 0:
            # dS_k/dh[j] = dx*(1-h[j+o]) for j in [0, n-o)
            #            - dx*h[j-o]   for j in [o, n)
            j1 = np.arange(0, n-o)
            grad[j1] += weights[k] * dx * (1.0 - h[j1+o])
            j2 = np.arange(o, n)
            grad[j2] += weights[k] * (-dx) * h[j2-o]
        else:
            o2 = -o  # positive
            # dS_k/dh[j] = dx*(1-h[j-o2]) for j in [o2, n)
            #            - dx*h[j+o2]     for j in [0, n-o2)
            j1 = np.arange(o2, n)
            grad[j1] += weights[k] * dx * (1.0 - h[j1-o2])
            j2 = np.arange(0, n-o2)
            grad[j2] += weights[k] * (-dx) * h[j2+o2]
    return grad

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_corr_matrix(fixed_vec):
    """A[j,i] = fixed_vec[i+n-1-j] if valid, else 0. For corr(a, fixed_vec) linear in a."""
    n = len(fixed_vec)
    A = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        offset = n - 1 - j
        i_start = max(0, -offset)
        i_end = min(n, n - offset)
        if i_start < i_end:
            A[j, i_start:i_end] = fixed_vec[i_start + offset:i_end + offset]
    return A

def _build_corr_matrix_b(fixed_a):
    """A[j,l] = fixed_a[l-n+1+j] if valid. For corr(fixed_a, b) linear in b."""
    n = len(fixed_a)
    A = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        l_start = max(0, n - 1 - j)
        l_end = min(n, 2 * n - 1 - j)
        if l_start < l_end:
            idx = np.arange(l_start, l_end)
            a_idx = idx - n + 1 + j
            A[j, l_start:l_end] = fixed_a[a_idx]
    return A

def _lp_minimax(A, target):
    """min tau s.t. A@x <= tau, sum(x)=target, 0<=x<=1. Returns (x, tau)."""
    n = A.shape[1]
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def _relaxed_pair_optimize(h_init, deadline, max_cy=40):
    """Alternating relaxed LP: fix b optimize a, fix a optimize b."""
    n = len(h_init)
    target = n / 2.0
    a = h_init.copy()
    b = (1.0 - h_init).copy()
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    for cyc in range(max_cy):
        if time.time() > deadline:
            break
        # LP in a with b fixed
        A_a = _build_corr_matrix(b)
        a_new, _ = _lp_minimax(A_a, target)
        if a_new is not None:
            a = _project(a_new, target)
            c5_a = _exact_c5(a)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a.copy()

        # LP in b with a fixed
        A_b = _build_corr_matrix_b(a)
        b_new, _ = _lp_minimax(A_b, target)
        if b_new is not None:
            b = _project(b_new, target)
            cand = _project(1.0 - b, target)
            c5_b = _exact_c5(cand)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = cand.copy()

        # Check segment between a and 1-b
        u2 = _project(1.0 - b, target)
        for t in (0.25, 0.5, 0.75):
            seg = (1.0 - t) * u2 + t * a
            seg = _project(seg, target)
            c5_seg = _exact_c5(seg)
            if c5_seg < best_c5:
                best_c5 = c5_seg
                best_h = seg.copy()

    return best_h, best_c5

def _smoothed_descend(h_init, deadline, tau_init=0.3):
    """Gradient descent on smoothed max-of-correlations."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_init
    lr = 0.008

    it = 0
    while time.time() < deadline:
        S = _corr_vals(h)
        S_max = float(np.max(S))
        w = np.exp(tau * (S - S_max))
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            break
        w /= w_sum

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            o = n - 1 - k
            if o >= 0:
                j1 = np.arange(0, n - o)
                grad[j1] += w[k] * dx * (1.0 - h[j1 + o])
                j2 = np.arange(o, n)
                grad[j2] += w[k] * (-dx) * h[j2 - o]
            else:
                o2 = -o
                j1 = np.arange(o2, n)
                grad[j1] += w[k] * dx * (1.0 - h[j1 - o2])
                j2 = np.arange(0, n - o2)
                grad[j2] += w[k] * (-dx) * h[j2 + o2]

        h_new = h - lr * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.92
            if lr < 0.0005:
                lr = 0.005
                tau *= 0.9
        if it % 30 == 0:
            tau = max(tau * 0.95, 0.1)
        it += 1

    return best_h, best_c5

def _line_search_c5(h, d, dx, max_step=0.5):
    """Line search: minimize max_k S_k(h + t*d) over valid t."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    # Bounds on t from 0<=h+t*d<=1
    t_lo, t_hi = -max_step, max_step
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    if np.any(pos_mask):
        t_lo = max(t_lo, float(np.min(-h[pos_mask] / d[pos_mask])) - 1e-12)
        t_hi = min(t_hi, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])) - 1e-12)
    if np.any(neg_mask):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg_mask]) / d[neg_mask])) - 1e-12)
        t_hi = min(t_hi, float(np.min(-h[neg_mask] / d[neg_mask])) - 1e-12)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    # Golden section on [t_lo, t_hi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt, b_pt = t_lo, t_hi
    c1 = b_pt - gr * (b_pt - a_pt)
    c2 = a_pt + gr * (b_pt - a_pt)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(40):
        if b_pt - a_pt < 1e-13:
            break
        if f1 < f2:
            b_pt, c2, f2 = c2, c1, f1
            c1 = b_pt - gr * (b_pt - a_pt)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a_pt, c1, f1 = c1, c2, f2
            c2 = a_pt + gr * (b_pt - a_pt)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a_pt + b_pt) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _coord_refine(h_init, deadline, max_iters=300):
    """Coordinate descent with pair moves on active lags."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(777)
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        S = _corr_vals(h)
        max_idx = int(np.argmax(S))
        offset = n - 1 - max_idx

        # Gradient of active lag
        grad = np.zeros(n)
        if offset >= 0:
            j1 = np.arange(0, n - offset)
            grad[j1] += dx * (1.0 - h[j1 + offset])
            j2 = np.arange(offset, n)
            grad[j2] += -dx * h[j2 - offset]
        else:
            o2 = -offset
            j1 = np.arange(o2, n)
            grad[j1] += dx * (1.0 - h[j1 - o2])
            j2 = np.arange(0, n - o2)
            grad[j2] += -dx * h[j2 + o2]

        # Try pair moves: move mass from high-grad to low-grad
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        order = np.argsort(grad[interior])
        improved = False
        tried = 0
        for ii in range(min(8, len(order))):
            for jj in range(min(8, len(order))):
                if tried > 40:
                    break
                i, j = interior[order[ii]], interior[order[jj]]
                if i == j:
                    continue
                for sign in (1, -1):
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    alpha, c5_try = _line_search_c5(h, d, dx, 0.4)
                    if c5_try < best_c5 - 1e-14:
                        h_new = h + alpha * d
                        h_new = _project(h_new, target)
                        c5_actual = _exact_c5(h_new)
                        if c5_actual < best_c5:
                            h = h_new
                            best_c5 = c5_actual
                            best_h = h.copy()
                            improved = True
                            break
                tried += 1
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 30:
                # Perturb
                h = h + rng.normal(0, 0.02, n)
                h = _project(h, target)
                stall = 0
                if _exact_c5(h) > best_c5 + 0.01:
                    h = best_h.copy()
                    break
        else:
            stall = 0

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = np.arange(n_new, dtype=float) + 0.5
    x_new = x_new / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _gen_start(n, rng, mode):
    x = np.arange(n, dtype=float)
    if mode == 0:
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 1:
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, int(rng.integers(1, n // 5 + 1)))
            if rng.random() < 0.5:
                h[pos:pos + bl] = 1.0
            pos += bl
    elif mode == 2:
        f1 = rng.integers(1, 4)
        f2 = rng.integers(4, 9)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * f1 * x / n + p1) + 0.15 * np.sin(2 * np.pi * f2 * x / n + p2)
    else:
        h = 0.5 + 0.08 * rng.standard_normal(n)
    h = h + 0.04 * rng.standard_normal(n)
    return _project(h, n / 2.0)

def _lp_directional_step(h, G_active, radius, target):
    """LP: find direction d that reduces active correlation, project back."""
    n = len(h)
    # Minimize tau s.t. G_active @ d + S >= -tau (want to reduce S)
    # Equivalently: minimize tau s.t. G_active @ d + S + tau <= 0
    # With bounds: |d| <= radius, sum(d) = 0
    m = G_active.shape[0]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([G_active, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_optimize_local(h_init, deadline, start_radius=0.1):
    """LP-based local optimization (similar to parent but tuned)."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = start_radius
    stall = 0
    rng = np.random.default_rng(31337)

    while time.time() < deadline:
        if stall > 60:
            if radius < 0.01:
                break
            h = h + rng.normal(0, 0.03, n)
            h = _project(h, target)
            best_h = h.copy()
            best_c5 = _exact_c5(h)
            stall = 0
            radius = 0.08
            continue

        S = _corr_vals(h)
        # Build full Jacobian
        G = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            o = n - 1 - k
            if o >= 0:
                j1 = np.arange(0, n - o)
                G[k, j1] = dx * (1.0 - h[j1 + o])
                j2 = np.arange(o, n)
                G[k, j2] -= dx * h[j2 - o]
            else:
                o2 = -o
                j1 = np.arange(o2, n)
                G[k, j1] = dx * (1.0 - h[j1 - o2])
                j2 = np.arange(0, n - o2)
                G[k, j2] -= dx * h[j2 + o2]

        # LP direction
        d = _lp_directional_step(h, G, radius, target)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            stall += 1
            radius *= 0.75
            if radius < 1e-5:
                break
            continue

        alpha, new_c5 = _line_search_c5(h, d, dx, radius)
        if new_c5 < best_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            c5_actual = _exact_c5(h_new)
            if c5_actual < best_c5:
                h = h_new
                best_c5 = c5_actual
                best_h = h.copy()
                stall = 0
                # Adapt radius
                predicted = float(np.max(S + G @ d))
                ratio = c5_actual / max(predicted, 1e-15)
                if ratio < 0.7:
                    radius = min(radius * 1.2, 0.3)
                elif ratio > 1.5:
                    radius *= 0.8
                else:
                    radius = min(radius * 1.05, 0.25)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.75

    return best_h, best_c5

    # Phase 6: Resolution changes (10%)
    d6 = t0 + budget_s * 0.90
    n_best = len(best_h)
    tried_n = set([n_best])
    for dn in (-4, 4, -6, 6, -2, 2, -8, 8, -10, 10):
        if time.time() > d6:
            break
        n_try = n_best + dn
        if n_try < 24 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _relaxed_pair_optimize(h_r, sub_dl, max_cy=15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 7: Final polish
    d7 = deadline
    if time.time() < d7:
        h7, c5_7 = _lp_optimize_local(best_h, d7, start_radius=0.06)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    if time.time() < deadline:
        h7b, c5_7b = _coord_refine(best_h, deadline, max_iters=50)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
