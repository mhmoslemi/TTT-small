import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_A(b, n):
    """Build matrix A where A @ a = dx * corr(a, b, 'full')"""
    dx = 2.0 / n
    A = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        shift = q - (n - 1)
        if shift >= 0:
            A[q, shift:] = b[:n - shift]
        else:
            A[q, :n + shift] = b[-shift:]
    return A * dx

def _solve_minimax_lp(A, n):
    """Solve: min tau s.t. A@x <= tau, sum(x)=n/2, 0<=x<=1, tau>=0"""
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def _alternating_lp(h_init, deadline, max_iter=30):
    """Alternating LP optimization."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Step 1: b = 1 - h, solve LP for a
        b = 1.0 - h
        A_a = _corr_A(b, n)
        a_star = _solve_minimax_lp(A_a, n)
        
        if a_star is not None:
            a_star = _project(a_star, target)
            c5_a = _exact_c5(a_star)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a_star.copy()
                h = a_star.copy()
            
            # Step 2: fix a = a_star, solve LP for b
            A_b = _corr_A(a_star, n)  # corr(x, a_star) for variable x
            # Wait, I need to be careful about the orientation
            # We want to find b that minimizes max_q corr(a_star, b)_q
            # corr(a_star, b)_q is linear in b
            # The matrix for this is different from _corr_A
            # _corr_A(b, n) gives A such that A @ a = corr(a, b)
            # We need: corr(a_star, b) as a function of b
            # corr(a_star, b)[q] = sum_i a_star[i] * b[i - (q-(n-1))]
            # So the coefficient of b[j] in corr(a_star, b)[q] is a_star[j + (q-(n-1))] if valid
            # This is actually _corr_A_flipped(a_star, n)
            
            # Let me define the flipped matrix
            A_b = _corr_A_flip(a_star, n)
            b_star = _solve_minimax_lp(A_b, n)
            
            if b_star is not None:
                b_star = _project(b_star, target)
                # 1 - b_star is a candidate
                cand = _project(1.0 - b_star, target)
                c5_cand = _exact_c5(cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = cand.copy()
                    h = cand.copy()
                
                # Bridge search between a_star and 1-b_star
                uL = a_star
                uR = _project(1.0 - b_star, target)
                bridge_h, bridge_c5 = _bridge_search(uL, uR, target)
                if bridge_c5 < best_c5:
                    best_c5 = bridge_c5
                    best_h = bridge_h.copy()
                    h = bridge_h.copy()
    
    return best_h, best_c5

But the actual objective is max_q corr(a*, 1-a*)_q. These are different!

However, if a* is close to h (which it should be if h is already good), then 1-a* ≈ 1-h = b, and the LP objective approximates the true objective.

The alternating scheme helps converge: after finding a*, we use 1-a* as the new b and re-solve. At a fixed point, we'd have b* = 1-a* and b = 1-a*, so the LP is solving the right problem.

This is essentially a block coordinate descent on the relaxed problem, followed by projection back to the original feasible set.

        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * C))
    return opt_t, opt_val

def _alternating_lp(h_init, deadline, max_iter=40):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rx = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Try the natural complement
        b = 1.0 - h
        A_a = _corr_A_right(b, n)
        a_star = _solve_minimax_lp(A_a, n)
        
        candidates = []
        if a_star is not None:
            a_star = _project(a_star, target)
            candidates.append(a_star)
        
        # Also try a perturbed complement
        b2 = b + rx.normal(0, 0.05, n)
        b2 = _project(b2, target)
        A_a2 = _corr_A_right(b2, n)
        a_star2 = _solve_minimax_lp(A_a2, n)
        if a_star2 is not None:
            a_star2 = _project(a_star2, target)
            candidates.append(a_star2)
        
        # Evaluate candidates and do second-stage LP
        for a_cand in candidates:
            c5_cand = _exact_c5(a_cand)
            if c5_cand < best_c5:
                # Second stage
                A_b = _corr_A_left(a_cand, n)
                b_star = _solve_minimax_lp(A_b, n)
                if b_star is not None:
                    b_star = _project(b_star, target)
                    cand2 = _project(1.0 - b_star, target)
                    c5_c2 = _exact_c5(cand2)
                    # Bridge search
                    for alpha in np.linspace(0.0, 1.0, 11):
                        h_try = (1.0 - alpha) * cand2 + alpha * a_cand
                        h_try = _project(h_try, target)
                        c5_try = _exact_c5(h_try)
                        if c5_try < best_c5:
                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h = h_try.copy()
                    if c5_c2 < best_c5:
                        best_c5 = c5_c2
                        best_h = cand2.copy()
                        h = cand2.copy()
                best_c5 = min(best_c5, c5_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = a_cand.copy()
                    h = a_cand.copy()
            elif it < 10:
                # Even if not improving, try second stage for exploration
                A_b = _corr_A_left(a_cand, n)
                b_star = _solve_minimax_lp(A_b, n)
                if b_star is not None:
                    b_star = _project(b_star, target)
                    cand2 = _project(1.0 - b_star, target)
                    c5_c2 = _exact_c5(cand2)
                    if c5_c2 < best_c5:
                        best_c5 = c5_c2
                        best_h = cand2.copy()
                        h = cand2.copy()
    
    return best_h, best_c5

def _pair_sweep(h, deadline, max_sweeps=20):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.005) & (h < 0.995))[0]
        if len(interior) < 2:
            break
        for i in range(len(interior)):
            if time.time() > deadline:
                break
            ii = interior[i]
            for j in range(i+1, len(interior)):
                jj = interior[j]
                # Move from ii to jj
                d = np.zeros(n)
                d[ii] = -1.0
                d[jj] = 1.0
                t, c5_try = _line_search_quad(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = _project(h + t * d, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        break
                # Move from jj to ii
                d[ii] = 1.0
                d[jj] = -1.0
                t, c5_try = _line_search_quad(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = _project(h + t * d, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
