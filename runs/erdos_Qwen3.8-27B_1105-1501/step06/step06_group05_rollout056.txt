import numpy as np
import time
from scipy.optimize import minimize


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _build_active(h, n_active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(-corr)[:n_active]
    a_list = []
    b_list = []
    for r in top:
        lag = int(r - (n - 1))
        a_start = max(0, lag)
        a_end = min(n, n + lag)
        ai = np.arange(a_start, a_end)
        bi = ai - lag
        a_list.append(ai)
        b_list.append(bi)
    return a_list, b_list


def _slq_ineq_fun(x, data):
    n, dx, a_list, b_list = data
    hx = x[:n]
    z = x[n]
    vals = np.empty(len(a_list))
    for c in range(len(a_list)):
        ai = a_list[c]
        bi = b_list[c]
        vals[c] = float(np.sum(hx[ai] * (1.0 - hx[bi]))) * dx - z
    return vals


def _slq_ineq_jac(x, data):
    n, dx, a_list, b_list = data
    hx = x[:n]
    nc = len(a_list)
    J = np.zeros((nc, n + 1))
    for c in range(nc):
        ai = a_list[c]
        bi = b_list[c]
        J[c, ai] = J[c, ai] + dx * (1.0 - hx[bi])
        J[c, bi] = J[c, bi] - dx * hx[ai]
        J[c, n] = -1.0
    return J


def _slq_eq_fun(x, target):
    n = x.size - 1
    return float(np.sum(x[:n]) - target)


def _slq_eq_jac(x, target):
    n = x.size - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def _slq_obj(x):
    return x[-1]


def _slq_obj_grad(x):
    g = np.zeros(x.size)
    g[-1] = 1.0
    return g


def _slsqp_round(h, n_active, max_iter, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    a_list, b_list = _build_active(h, n_active)
    if len(a_list) == 0:
        return h, _exact_c5(h)
    data = (n, dx, a_list, b_list)
    corr_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    z0 = float(np.max(corr_vals)) + 0.01
    x0 = np.empty(n + 1)
    x0[:n] = h
    x0[n] = z0
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': _slq_ineq_fun, 'jac': _slq_ineq_jac, 'args': (data,)},
        {'type': 'eq', 'fun': _slq_eq_fun, 'jac': _slq_eq_jac, 'args': (target,)}
    ]
    try:
        res = minimize(_slq_obj, x0, method='SLSQP', jac=_slq_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-9})
        if time.time() > deadline:
            return h, _exact_c5(h)
        h_new = _project(res.x[:n], target)
        f_new = _exact_c5(h_new)
        return h_new, f_new
    except Exception:
        return h, _exact_c5(h)

def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv


def _row_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nn = np.arange(0, n - k)
    else:
        nn = np.arange(-k, n)
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g


def _active_direction(h, corr, k_active, tau, rng):
    n = len(h)
    m = float(np.max(corr))
    w = np.exp((corr - m) / tau)
    idx = np.argsort(-corr)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = int(r - (n - 1))
        G += w[r] * _row_grad(h, k)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm


def _local_search(h, deadline, rng, pair_frac=0.18, rand_frac=0.10, k_active=8, tau=0.02, uphill=5e-4):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            d = _active_direction(h, corr, k_active, tau, rng)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, dx, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < 0.0 or hn.max() > 1.0:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-13:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
        if stall > 150:
            h = _project(best_h + rng.normal(0, 0.03, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
    return best_h, best_f

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
