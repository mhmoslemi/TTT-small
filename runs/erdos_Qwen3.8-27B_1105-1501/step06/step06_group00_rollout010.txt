def _project_sum(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    
    # Bisection for water-filling
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    residual = target - x.sum()
    if abs(residual) > 1e-14:

            mask = x < 1.0
        else:
            mask = x > 0.0

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, :m] += (1.0 - h[d:d + m])
                G[r, d:] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, g:] += (1.0 - h[:m])
                G[r, :m] += -h[g:]
    return G

def _lp_direction_focused(h, scores, G, radius, top_k):
    n = len(h)
    top_idx = np.argsort(scores)[::-1][:top_k]
    top_scores = scores[top_idx]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, :n] = G[top_idx]
    A_ub[:, n] = -1.0
    b_ub = -top_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:

    except Exception:
        pass
    return None

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = _scores(h)
    T = _scores(d) - _scores(h) * 0  # placeholder, compute properly
    # Compute T_k = sum G[k,:] dot d using score difference
    # score_k(t) = score_k(0) + t*T_k - t^2*C_k
    # T_k = G[k,:] dot d
    # C_k = sum of d[i]*d[i+l] for valid range (the -t^2 coefficient has -C_k)
    # Actually: score_k = sum h[i]*(1-h[i+l])
    # score_k(t) = sum (h[i]+t*d[i])*(1-h[i+l]-t*d[i+l])
    #            = sum h[i]*(1-h[i+l]) + t*sum(d[i]*(1-h[i+l]) - h[i]*d[i+l]) - t^2*sum(d[i]*d[i+l])
    # So T_k = G[k,:] dot d, C_k = sum d[i]*d[i+l] over valid range
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        r = n - 1 + lag
        if lag >= 0:
            m = n - lag
            if m > 0:
                T[r] = float(np.sum(d[:m] * (1.0 - h[lag:lag + m]) - h[:m] * d[lag:lag + m]))
                C[r] = float(np.sum(d[:m] * d[lag:lag + m]))
        else:
            g = -lag
            m = n - g
            if m > 0:
                T[r] = float(np.sum(d[g:] * (1.0 - h[:m]) - h[g:] * d[:m]))
                C[r] = float(np.sum(d[g:] * d[:m]))
    
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Sample the 1D function at several points
    n_grid = 40
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    
    # Golden section refinement around best
    lo_b = grid_t[max(0, best_idx - 3)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 3)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = grid_f[best_idx]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    candidates = [(opt_t, opt_val), (t_lo, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _focused_lp_loop(h, deadline, radius_init=0.12, top_k_init=4, max_top_k=12, stall_limit=80):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    top_k = top_k_init
    stall = 0
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        G = _grad_matrix(h)
        p = _lp_direction_focused(h, scores, G, radius, top_k)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            top_k = min(max_top_k, top_k + 2)
            radius *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-13:
            h = h + alpha * p
            h = _project_sum(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            top_k = max(3, top_k - 1)
            radius = min(radius * 1.04, 0.25)
        else:
            stall += 1
            top_k = min(max_top_k, top_k + 1)
            radius *= 0.75
            if stall % 30 == 0 and radius < 0.02:
                radius = 0.08
                top_k = top_k_init
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.004, seed=123):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 600:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9998
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h, deadline, max_rounds=25):
    n = len(h)
    cur = h.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = list(np.where(cur > 0.5)[0])
        zeros = list(np.where(cur <= 0.5)[0])
        for oi in ones:
            if time.time() > deadline:
                break
            for oz in zeros:
                cur[oi] = 0.0
                cur[oz] = 1.0
                c_try = _exact_c5(cur)
                if c_try < cur_c5 - 1e-14:
                    cur_c5 = c_try
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = cur.copy()
                    improved = True
                    break
                else:
                    cur[oi] = 1.0
                    cur[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, n_new / 2.0)

def _perturb(h, rng, amplitude):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 10.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert *= amplitude
    pert += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project_sum(h + pert, n / 2.0)

def _pair_swap_loop(h, deadline, n_pairs=5000, radius=0.25, seed=111):

        h4 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h4, c5_4 = _pair_swap_loop(h4, sub_dl, n_pairs=8000, radius=0.20, seed=222)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    # Binary refinement
    if time.time() < d4:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=40000, T_init=0.002, seed=777)
        sub_dl2 = min(d4, time.time() + budget_s * 0.025)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2, max_rounds=25)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    # Another LP pass with very tight focus
    if time.time() < d4:
        h4 = _project_sum(best_h.copy(), n_cur / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h4, c5_4 = _focused_lp_loop(h4, sub_dl, radius_init=0.03, top_k_init=2, max_top_k=5, stall_limit=60)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.90
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 and n_restarts < 200:
        n_restarts += 1
        h_pert = _perturb(best_h, rng, scale)
        sub_dl = min(d5, time.time() + budget_s * 0.004)
        h_t, c5_t = _focused_lp_loop(h_pert, sub_dl, radius_init=0.05, top_k_init=4, max_top_k=8, stall_limit=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.12, scale * 1.06)
    
    # Phase 6: Final polish (10%)
    d6 = deadline
    n_best = len(best_h)
    for dn in [-4, 4, -8, 8, -2, 2]:
        if time.time() > d6 - 5:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 250 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.008)
        h_r, c5_r = _focused_lp_loop(h_r, sub_dl, radius_init=0.05, top_k_init=3, max_top_k=7, stall_limit=30)
        if time.time() < d6 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d6, time.time() + budget_s * 0.005)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=12000, T_init=0.001, seed=444)
            if c5_b < c5_r:
                h_r, c5_r = h_b, c5_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    # Final focused LP
    h6 = _project_sum(best_h.copy(), len(best_h) / 2.0)
    h6, c5_6 = _focused_lp_loop(h6, d6, radius_init=0.02, top_k_init=3, max_top_k=6, stall_limit=50)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    # Final pair sweep
    if deadline - time.time() > 3:
        h_b = _to_binary(best_h)
        h_b, c5_b = _pair_sweep(h_b, d6, max_rounds=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
