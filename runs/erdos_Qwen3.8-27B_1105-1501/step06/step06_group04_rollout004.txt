def _project(h, target):
    """Project h to [0,1]^n with sum = target using bisection on threshold."""
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(h - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_constraint(h, r):
    """Gradient of corr(h, 1-h)[r] with respect to h."""
    n = len(h)
    d = r - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] += -h[d:]
    else:
        s = -d
        m = n - s
        if m > 0:
            grad[:m] += (1.0 - h[s:])
            grad[s:] += -h[:m]
    return grad

    n = len(h)

def _c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, target):
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(h - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0:
            mask = (x < 1.0) if res > 0 else (x > 0.0)
            cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _line_search(h, A, max_grid=40):

def _grad_constrained(h, r, n):
    d = r - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] += -h[d:]
    else:
        s = -d
        m = n - s
        if m > 0:
            grad[:m] += (1.0 - h[s:])
            grad[s:] += -h[:m]
    return grad

def _grad_direction(h, top_k=6):
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    active = np.argsort(-scores)[:top_k]
    total = np.zeros(n)
    for r in active:
        total += _grad_constrained(h, r, n)
    total -= total.mean()
    norm = np.linalg.norm(total)
    if norm < 1e-14:
        return None
    return -total / norm

    h_new = np.interp(x_new, x_old, h)
