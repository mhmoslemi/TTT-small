    """Gradient of correlation at lag offset d (where d = r - (n-1))."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        g[:m] += (1.0 - h[d:d+m])

    else:

        m = n - g

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(R) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.nan_to_num(v, nan=target / n, posinf=1.0, neginf=0.0)
    v = np.clip(v, -5.0, 5.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(20):
        if np.sum(np.clip(v - lo, 0, 1)) < target:
            lo -= 1.0
        else:
            break
    for _ in range(20):
        if np.sum(np.clip(v - hi, 0, 1)) > target:
            hi += 1.0
        else:
            break
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        d = r - (n - 1)
        if d == 0:
            G[r] = 1.0 - 2.0 * h
        elif d > 0:
            m = n - d
            G[r, :m] += (1.0 - h[d:d + m])
            G[r, d:] -= h[:m]
        else:
            g = -d
            m = n - g
            G[r, g:] += (1.0 - h[:m])
            G[r, :m] -= h[g:]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    h_new = h + opt_t * d
    opt_val = _compute_c5(h_new)
    return opt_t, opt_val

def _lse_slsqp(h_init, deadline, betas=None, maxiter=150):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    if betas is None:
        betas = [15, 30, 60, 120, 250, 500, 1000, 2000]
    eq_jac = np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target), 'jac': lambda x: eq_jac}]
    for beta in betas:
        if time.time() > deadline:
            break
        def lse_obj(x):
            R = np.correlate(x, 1.0 - x, mode='full') * dx
            mx = float(R.max())
            z = beta * (R - mx)
            z = np.clip(z, -60, 60)
            e = np.exp(z)
            return mx + float(np.log(e.sum())) / beta
        def lse_grad(x):
            R = np.correlate(x, 1.0 - x, mode='full') * dx
            mx = float(R.max())
            z = beta * (R - mx)
            z = np.clip(z, -60, 60)
            e = np.exp(z)
            p = e / float(e.sum())
            G = _grad_matrix(x)
            return dx * (p @ G)
        x0 = h.copy()
        res = minimize(lse_obj, x0, jac=lse_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and res.x is not None:
            rx = np.clip(res.x, 0.0, 1.0)
            rx = _project(rx, target)
            c5_new = _compute_c5(rx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = rx.copy()
            h = rx
        else:
            h = best_h.copy()
    return best_h, best_c5

def _direct_minimax(h_init, deadline, maxiter=100):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    M0 = _compute_c5(h) + 0.05
    y0 = np.zeros(n + 1)
    y0[0] = M0
    y0[1:] = h
    eq_jac = np.zeros(n + 1)
    eq_jac[1:] = 1.0
    bounds_mm = [(0.0, M0 + 0.5)] + [(0.0, 1.0)] * n
    nn = 2 * n - 1
    def mm_obj(y):
        return y[0]
    def mm_jac(y):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    def mm_ineq(y):
        h_y = y[1:]
        R = np.correlate(h_y, 1.0 - h_y, mode='full') * dx
        return y[0] - R
    def mm_ineq_jac(y):
        h_y = y[1:]
        G = _grad_matrix(h_y)
        J = np.zeros((nn, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * G
        return J
    cons = [
        {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
        {'type': 'eq', 'fun': lambda y: float(np.sum(y[1:]) - target), 'jac': lambda y: eq_jac}
    ]
    res = minimize(mm_obj, y0, jac=mm_jac, method='SLSQP',
                   bounds=bounds_mm, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if hasattr(res, 'x') and res.x is not None:
        h_new = np.clip(res.x[1:], 0.0, 1.0)
        h_new = _project(h_new, target)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def _active_set_polish(h, deadline, max_rounds=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    rng = np.random.default_rng(123)
    stall = 0
    it = 0
    while time.time() < deadline and stall < 40:
        it += 1
        C = _corr_scaled(h)
        max_c = float(C.max())
        # Try multiple threshold levels
        found_improve = False
        for frac in [0.0, 0.02, 0.05, 0.10, 0.20]:
            if time.time() > deadline:
                break
            thr = max_c - frac * max_c
            active = np.where(C >= thr)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(C))])
            G = _grad_matrix(h)
            sub = G[active, :]
            k = min(len(active), 12)
            # Try average of top-k
            idx_sorted = np.argsort(-C[active])
            for k_try in [1, 2, 3, k]:
                sel = idx_sorted[:k_try]
                g = sub[sel].mean(axis=0)
                g -= float(g.mean())
                norm = np.linalg.norm(g)
                if norm < 1e-14:
                    continue
                d = -g / norm
                # Clip to feasible directions
                pos = d > 0
                neg = d < 0
                if np.any(pos):
                    max_step = float(np.min((1.0 - h[pos]) / d[pos]))
                    d[pos] = np.minimum(d[pos], max_step * d[pos] - 1e-12)
                if np.any(neg):
                    max_step = float(np.min(h[neg] / (-d[neg])))
                    d[neg] = np.maximum(d[neg], -max_step * (-d[neg]) + 1e-12)
                if np.max(np.abs(d)) < 1e-14:
                    continue
                alpha, new_c5 = _line_search(h, d, dx, 1.0)
                if new_c5 < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _compute_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        found_improve = True
                        stall = 0
                        break
            if found_improve:
                break
        # Random perturbation to escape local minima
        if not found_improve:
            stall += 1
            if stall > 20:
                h = h + rng.normal(0, 0.01, n)
                h = _project(h, target)
                cur = _compute_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            if it % 50 == 0:
                # Random direction line search
                d = rng.normal(0, 1, n)
                d -= d.mean()
                pos = d > 0
                neg = d < 0
                if np.any(pos):
                    ms = float(np.min((1.0 - h[pos]) / d[pos]))
                    d[pos] *= ms
                if np.any(neg):
                    ms = float(np.min(h[neg] / (-d[neg])))
                    d[neg] *= -ms
                alpha, new_c5 = _line_search(h, d, dx, 0.5)
                if new_c5 < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _compute_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return _project(h, n / 2.0)

def _binary_quantize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    hq = np.zeros(n)
    hq[idx[:target]] = 1.0
    return hq

def _binary_sa(h_init, deadline, rng, max_iters=6000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.003
    T_min = 0.00005
    cooling = 0.9997
    patience = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 4000:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / max(T, 1e-12))):
            cur_c5 = new_c5
            patience += 1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
        else:
            h[i], h[j] = h[j], h[i]
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _refine_at_resolution(h_best, n_try, deadline, sub_time):
    h_r = _resample(h_best, n_try)
    dl = min(deadline, time.time() + sub_time)
    h_r, c5_r = _lse_slsqp(h_r, dl, betas=[30, 100, 300, 800], maxiter=80)
    dl2 = min(deadline, time.time() + sub_time * 0.4)
    h_r, c5_r = _active_set_polish(h_r, dl2, max_rounds=100)
    return h_r, c5_r

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s
    reserve = max(5.0, 0.02 * budget_s)
    hard_deadline = t0 + budget_s - reserve

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: LSE-SLSQP from parent (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lse_slsqp(best_h, d1, betas=[15, 30, 60, 120, 250, 500, 1000, 2000], maxiter=120)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Direct minimax (15% time)
    d2 = t0 + budget_s * 0.40
    if time.time() < d2:
        h2, c5_2 = _direct_minimax(best_h, d2, maxiter=80)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Active set polish (20% time)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _active_set_polish(best_h, d3, max_rounds=800)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (15% time)
    d4 = t0 + budget_s * 0.75
    n_cur = len(best_h)
    offsets = [-4, -2, 2, 4, -6, 6, -8, 8, -12, 12, -16, 16, -20, 20, 24, -24]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_cur + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        sub_time = min(15.0, (d4 - time.time()) / max(1, len(offsets) - len(tried)))
        hr, cr = _refine_at_resolution(best_h, n_try, d4, sub_time)
        if cr < best_c5 - 1e-12:
            best_c5 = cr
            best_h = hr.copy()

    # Phase 5: Binary SA + re-optimize (10% time)
    d5 = t0 + budget_s * 0.85
    if time.time() < d5:
        hb, cb = _binary_sa(best_h, min(d5, time.time() + 60), rng, max_iters=5000)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        if time.time() < d5:
            h5, c5_5 = _lse_slsqp(best_h, d5, betas=[50, 200, 500, 1500], maxiter=100)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()

    # Phase 6: Multi-start exploration (5% time)
    d6 = t0 + budget_s * 0.90
    if time.time() < d6:
        n = len(best_h)
        starts = []
        for freq in [1, 2, 3, 4, 5, 7]:
            starts.append(_sine_init(n, freq, 0.3 * freq))
        for p in [5, 6, 7, 8, 10, 12]:
            starts.append(_periodic_init(n, p, 0.45))
        for s in starts[:6]:
            if time.time() > d6:
                break
            sub_dl = min(d6, time.time() + 8.0)
            hs, cs = _lse_slsqp(s, sub_dl, betas=[50, 200, 600], maxiter=60)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()

    # Phase 7: Final polish (5% time)
    h_f, c5_f = _active_set_polish(best_h, hard_deadline, max_rounds=300)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h_fb = np.array(initial_h_values, dtype=float)
            h_fb = _project(h_fb, len(h_fb) / 2.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
        except NameError:
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _compute_c5(h_fb), 122
    return best_h, final_c5, len(best_h)
