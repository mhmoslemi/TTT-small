  So ∂corr[k]/∂h[i] = (1-h[i-k])·[i-k ∈ [0,n-1]] - h[i+k]·[i+k ∈ [0,n-1]]

The gradient of the max objective (subgradient) is the average of gradients over active shifts:
g[i] = (1/|K*|) * sum_{k ∈ K*} [(1-h[i-k])·[i-k valid] - h[i+k]·[i+k valid]]

This gives a descent direction for the continuous problem. Then I can project it onto the box [0,1]^n and the constraint sum(h) = n/2 to get a feasible direction.

This is essentially what the LP-based method does (the LP dual gives the weights for the active constraints). But I can compute it more directly and cheaply.

Let me implement this as a "fast LP" alternative:

```python
def _subgradient_direction(h, corr, dx):
    """Compute subgradient of max(corr*dx) at current point."""
    n = len(h)
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-12)[0]
    
    g = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        # ∂corr[m]/∂h[i] = (1-h[i-k]) - h[i+k] (with bounds)
        if k >= 0:
            # h[n+k] term: i from 0 to n-1-k, contributes (1-h[i]) * ... 
            # Wait, let me re-derive.
            # corr[m] = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n]) for k >= 0
            # ∂/∂h[i]:
            #   From h[n+k] where n+k=i: n=i-k, valid if 0<=i-k<=n-1-k, i.e., k<=i<=n-1
            #   Contribution: (1-h[i-k])
            #   From (1-h[n]) where n=i: valid if 0<=i<=n-1-k, i.e., 0<=i<=n-1-k
            #   Contribution: -h[i+k]
            for i in range(k, n):
                g[i] += (1 - h[i-k])
            for i in range(0, n-k):
                g[i] -= h[i+k]
        else:
            k_abs = -k
            # corr[m] = sum_{n=k_abs}^{n-1} h[n+k] * (1-h[n]) = sum_{n=k_abs}^{n-1} h[n-k_abs] * (1-h[n])
            # ∂/∂h[i]:
            #   From h[n-k_abs] where n-k_abs=i: n=i+k_abs, valid if k_abs<=i+k_abs<=n-1, i.e., 0<=i<=n-1-k_abs
            #   Contribution: (1-h[i+k_abs])
            #   From (1-h[n]) where n=i: valid if k_abs<=i<=n-1, i.e., k_abs<=i<=n-1
            #   Contribution: -h[i-k_abs]
            for i in range(0, n-k_abs):
                g[i] += (1 - h[i+k_abs])
            for i in range(k_abs, n):
                g[i] -= h[i-k_abs]
    
    g /= len(active)
    return g
```

This is O(n * |K*|) which is at most O(n²). For n=180, that's 32400 operations per gradient evaluation. Very fast.

Then the descent direction is: d = -g (negative gradient/subgradient). I project d onto the feasible set (box + sum constraint) and do a line search.

Actually, for the max of affine functions (which is what we have - each corr[k]*dx is affine in h), the subgradient is a convex combination of the individual gradients. import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _subgradient(h, corr):
    n = len(h)
    max_val = float(corr.max())
    active = np.where(corr >= max_val - 1e-10)[0]
    g = np.zeros(n)
    for m in active:
        k = int(m - (n - 1))
        if k >= 0:
            idx1 = np.arange(k, n)
            g[idx1] += (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            g[idx2] -= h[idx2 + k]
        else:
            ka = -k
            idx1 = np.arange(0, n - ka)
            g[idx1] += (1.0 - h[idx1 + ka])
            idx2 = np.arange(ka, n)
            g[idx2] -= h[idx2 - ka]
    g /= len(active)
    return g

def _line_search(h, d, dx, radius, n_grid=30):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    grid_t = np.linspace(0.0, t_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        t = grid_t[idx]
        grid_f[idx] = float(np.max(S0 + t * T - t * t * C)) * dx
    best_idx = int(np.argmin(grid_f))
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t = grid_t[best_idx]
        opt_val = float(grid_f[best_idx])
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    best_c = min([(opt_t, opt_val), (0.0, float(np.max(S0) * dx)),
                  (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx)],
                 key=lambda x: x[1])
    return best_c[0], best_c[1]

def _grad_direction_loop(h, deadline, radius_init=0.10, max_iter=5000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = _subgradient(h, corr)
        d = -g
        d = d - np.mean(d)
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-13:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.02, 0.30)
        else:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
    return best_h, best_c5

def _binary_c5_fast(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx), corr

def _binary_local_search(h_bin, deadline, max_iters=200000, temp_init=0.005, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_c5_fast(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    T = temp_init
    stall = 0
    best_stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if best_stall > 5000:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        trial = cur.copy()
        trial[oi] = 0.0
        trial[oz] = 1.0
        new_c5, _ = _binary_c5_fast(trial)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur = trial
            cur_c5 = new_c5
            cur_corr = None
            ones[i] = oz
            zeros[j] = oi
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
                best_stall = 0
            else:
                stall += 1
            T = max(0.0005, T * 0.9997)
        else:
            stall += 1
            best_stall += 1
    return best_h, best_c5

def _targeted_binary_search(h_bin, deadline, max_iters=50000, seed=99):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5, cur_corr = _binary_c5_fast(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        max_idx = int(np.argmax(cur_corr))
        best_delta = 0.0
        best_move = None
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):
            i = int(rng.integers(0, len(ones)))
            j = int(rng.integers(0, len(zeros)))
            oi = ones[i]
            oz = zeros[j]
            trial = cur.copy()
            trial[oi] = 0.0

def _block_swap_search(h_bin, deadline, max_iters=20000, seed=77):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 2000:
            break
        length = int(rng.integers(2, min(8, n // 4)))
        i = int(rng.integers(0, n - length))
        j = int(rng.integers(0, n - length))
        if abs(i - j) < length:
            stall += 1
            continue
        block1 = cur[i:i+length].copy()
        block2 = cur[j:j+length].copy()
        if np.sum(block1) != np.sum(block2):
            stall += 1
            continue
        trial = cur.copy()
        trial[i:i+length] = block2
        trial[j:j+length] = block1
        new_c5 = _exact_c5(trial)
        if new_c5 < cur_c5 - 1e-14:
            cur = trial
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _lp_direction(h, corr, radius):
    n = len(h)
    m = 2 * n - 1
    g = _subgradient(h, corr)
    d = -g
    d = d - np.mean(d)
    d = np.clip(d, -radius, radius)
    d = d - np.mean(d)
    return d

def _lp_polish(h, deadline, radius_init=0.05, stall_limit=60):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        d = _lp_direction(h, corr, radius)
        if float(np.max(np.abs(d))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, d, dx, radius)
        if new_c5 < cur - 1e-12:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.72
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.2, 5.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    perturb = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.3 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _pair_swap_loop(h, deadline, n_pairs=10000, radius=0.20, seed=111):

def _greedy_construction(n):
    h = np.zeros(n)
    ones_needed = n // 2
    for step in range(ones_needed):
        best_pos = -1
        best_c5 = float('inf')
        for pos in range(n):
            if h[pos] == 0:
                h[pos] = 1.0
                c5 = _exact_c5(h)
                h[pos] = 0.0
                if c5 < best_c5:
                    best_c5 = c5
                    best_pos = pos
        if best_pos >= 0:
            h[best_pos] = 1.0
    return h

            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Phase 4: Intensive refinement (25%)
    d4 = deadline
    n_best = len(best_h)

    # LP polish
    sub_dl = min(d4, time.time() + budget_s * 0.04)
    h4 = _project(best_h.copy(), n_best / 2.0)
    h4, c5_4 = _lp_polish(h4, sub_dl, radius_init=0.04, stall_limit=80)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Pair swap
    if time.time() < d4 - 20:
        h4 = _project(best_h.copy(), n_best / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_swap_loop(h4, sub_dl, n_pairs=8000, radius=0.20, seed=666)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Binary targeted + SA on best
    if time.time() < d4 - 30:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.06)
        h_b, c5_b = _targeted_binary_search(h_b, sub_dl, max_iters=80000, seed=777)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if time.time() < d4 - 20:
        h_b = _to_binary(best_h, n_best)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_local_search(h_b, sub_dl, max_iters=80000, temp_init=0.002, seed=888)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Basin hopping with smooth perturbations
    if time.time() < d4 - 15:
        scale = 0.02
        for _ in range(50):
            if time.time() > d4 - 5:
                break
            h_pert = _smooth_perturb(best_h, scale, rng)
            sub_dl = min(d4, time.time() + budget_s * 0.004)
            h_t, c5_t = _lp_polish(h_pert, sub_dl, radius_init=0.04, stall_limit=20)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.003, scale * 0.85)
            else:
                scale = min(0.06, scale * 1.08)

    # Final polish
    if time.time() < d4 - 3:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _lp_polish(h6, d4 - 2, radius_init=0.02, stall_limit=30)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final check with nearby n values
    n_fin = len(best_h)
    for dn in [-2, 2, -4, 4]:
        if time.time() > d4 - 8:
            break
        n_try = n_fin + dn
        if n_try < 60:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4 - 1, time.time() + 5)
        h_r, c5_r = _lp_polish(h_r, sub_dl, radius_init=0.03, stall_limit=20)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
