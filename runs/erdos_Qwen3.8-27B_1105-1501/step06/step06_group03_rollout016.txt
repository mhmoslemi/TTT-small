def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _softmax_weights(S, beta):
    M = float(S.max())
    exps = np.exp(beta * (S - M))
    s = exps.sum()
    if s < 1e-300:
        return np.zeros_like(S)
    return exps / s

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.04 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 40:
        combined = sorted_idx[:40]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_dir(h, S, n, tau):
    M = float(S.max())
    beta = 1.0 / max(tau, 1e-10)
    w = _softmax_weights(S, beta)
    top_k = min(30, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad_s = w[top_idx] @ G
    grad_s -= grad_s.mean()
    norm = np.linalg.norm(grad_s)
    if norm < 1e-12:
        return None
    return -grad_s / norm

def _lp_loop(h, deadline, radius_init=0.14, K=20, use_smooth=False, max_stall=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(12345)
    tau = 0.015 * cur_c5
    while time.time() < deadline:
        if stall > max_stall:
            if restart_ctr < 12:
                scale = 0.008 + 0.008 * restart_ctr * 0.4
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                tau = 0.015 * cur_c5
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if (p is None or float(np.max(np.abs(p))) < 1e-13) and use_smooth:
            p = _smooth_dir(h, S, n, tau)
            if p is not None:
                p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if use_smooth and stall % 5 == 0:
                tau *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.08, 0.30)
            elif ratio > 0.3:
                radius = min(radius * 1.03, 0.28)
            else:
                radius = max(radius * 0.95, 0.015)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.012:
                radius = 0.04
    return best_h, best_c5

def _binary_sa_soft(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    S = _corr_raw(h)
    cur_c5 = float(S.max()) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    cooldown = 0.99995
    max_patience = 20000
    patience = 0
    beta = 80.0
    beta_max = 200.0
    M = float(S.max())
    ones_list = np.arange(target)
    zeros_list = np.arange(target, n)
    step_count = 0
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
            M = float(S.max())
        if T < T_min:
            break
        batch = 300
        for _ in range(batch):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            if rng.random() < 0.25 and len(ones_list) >= 2 and len(zeros_list) >= 2:
                i1 = ones_list[rng.integers(len(ones_list))]
                i2 = ones_list[rng.integers(len(ones_list))]
                if i1 == i2:
                    continue
                j1 = zeros_list[rng.integers(len(zeros_list))]
                j2 = zeros_list[rng.integers(len(zeros_list))]
                if j1 == j2:
                    continue
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                S_new = _corr_raw(h)
                new_c5 = float(S_new.max()) * dx
                M = max(M * 0.999, float(S_new.max()))
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    S = S_new
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                    oi = list(np.where(h > 0.5)[0])
                    zi = list(np.where(h <= 0.5)[0])
                    ones_list = np.array(oi)
                    zeros_list = np.array(zi)
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
            else:
                i = ones_list[rng.integers(len(ones_list))]
                j = zeros_list[rng.integers(len(zeros_list))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                S_new = _corr_raw(h)
                new_c5 = float(S_new.max()) * dx
                M = max(M * 0.999, float(S_new.max()))
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    S = S_new
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                    ones_list = np.where(h > 0.5)[0]
                    zeros_list = np.where(h <= 0.5)[0]
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        step_count += batch
        if step_count % 5000 == 0:
            beta = min(beta_max, beta * 1.05)
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=150):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 80)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def _multi_pair_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    stall = 0
    while time.time() < deadline and stall < 60:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 4:
            break
        idx = rng.choice(len(interior), size=min(6, len(interior)), replace=False)
        found = False
        for a in range(len(idx)):
            if time.time() > deadline:
                break
            for b in range(a + 1, len(idx)):
                i, j = interior[idx[a]], interior[idx[b]]
                for s in (-1.0, 1.0):
                    d = np.zeros(n)
                    d[i] = s
                    d[j] = -s
                    alpha, c5_try = _line_search(h, d, dx, 0.4)
                    if c5_try < best_c5 - 1e-14:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            stall = 0
                            found = True
                        break
                if found:
                    break
            if found or time.time() > deadline:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _softmax_grad_descent(h, deadline, beta_start=30.0, beta_end=300.0, max_iter=2000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    n_total = 2 * n - 1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max_iter
        beta = beta_start + (beta_end - beta_start) * frac
        S = _corr_scaled(h)
        w = _softmax_weights(S, beta)
        all_idx = np.arange(n_total)
        G = _grad_rows(h, all_idx, n)
        grad = (w @ G)
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad = -grad / norm
        lr = 0.02 / (1.0 + 0.1 * beta)
        h_new = h + lr * grad
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start with diverse initializations
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        for ph in [0.0, 0.5, 1.2, 2.0, 3.0]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(12):
        starts.append(_block_init(n0, rng, int(rng.integers(3, 18))))
    # Sorted/shuffled patterns
    for _ in range(4):
        perm = rng.permutation(n0)
        h_perm = np.zeros(n0)
        h_perm[perm[:n0 // 2]] = 1.0
        starts.append(h_perm.copy())
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=15, max_stall=30)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main LP optimization with larger K
    d2 = t0 + budget_s * 0.32
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.16, K=25, use_smooth=False, max_stall=70)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Soft-max gradient descent (smooth landscape)
    d3 = t0 + budget_s * 0.42
    h3, c5_3 = _softmax_grad_descent(best_h, d3, beta_start=20.0, beta_end=200.0, max_iter=5000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP with smooth backup
    d4 = t0 + budget_s * 0.52
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.10, K=30, use_smooth=True, max_stall=80)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary SA with 2-swap moves + soft cost
    d5a = t0 + budget_s * 0.62
    h5a, c5_5a = _binary_sa_soft(best_h, d5a, rng)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()

    # LP polish after SA
    d5b = t0 + budget_s * 0.66
    if time.time() < d5b:
        h5b, c5_5b = _lp_loop(best_h, d5b, radius_init=0.08, K=20, max_stall=40)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Pair refinement
    d6 = t0 + budget_s * 0.74
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Multi-resolution
    d7 = t0 + budget_s * 0.87
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d7:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d7, time.time() + budget_s * 0.025)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=18, max_stall=40)
            if c5_r < best_c5 and time.time() < d7 - budget_s * 0.03:
                sub_dl2 = min(d7, time.time() + budget_s * 0.02)
                h_r2, c5_r2 = _binary_sa_soft(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d7:
                    h_r, c5_r = _lp_loop(h_r, min(d7, time.time() + budget_s * 0.012),
                                         radius_init=0.06, K=15, max_stall=30)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Aggressive perturbation
    d8 = t0 + budget_s * 0.94
    scale = 0.025
    for attempt in range(400):
        if time.time() > d8:
            break
        if best_c5 < 0.3804:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d8, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.06, K=15, max_stall=25)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.75)
        else:
            scale = min(0.15, scale * 1.25)

    # Phase 9: Final pair + multi-pair polish
    d9 = t0 + budget_s * 0.97
    h9, c5_9 = _pair_refine(best_h, d9, max_passes=80)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()
    if time.time() < deadline:
        h9b, c5_9b = _multi_pair_refine(best_h, deadline)
        if c5_9b < best_c5:
            best_c5 = c5_9b
            best_h = h9b.copy()

    # Final LP polish
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.04, K=15, max_stall=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
