            if fc < fd:
                b = c_pt
                d_pt = a + gr * (b - a)
                fd = _exact_c5(_project_full(h + d_pt * d, target))
            else:
                a = d_pt
                c_pt = b - gr * (b - a)
                fc = _exact_c5(_project_full(h + c_pt * d, target))
        
        t_best = (a + b) / 2.0
        h_new = _project_full(h + t_best * d, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
    
    return best_h, best_c5

def _make_sine_init(m, freq, phase):
    x = np.arange(m, dtype=float)
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / m + phase)
    return np.clip(h, 0.0, 1.0)

def _make_random_smooth_init(m, rng):
    h = np.zeros(m)
    for k in range(3):
        freq = rng.integers(1, 6) + k
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.2 / (k + 1)
        h += amp * np.sin(2.0 * np.pi * freq * np.arange(m) / m + phase)
    h = 0.5 + h
    return np.clip(h, 0.0, 1.0)

def _make_blocks_init(m, rng):
    h = np.zeros(m)
    i = 0
    while i < m:
        blen = int(rng.integers(2, max(3, m // 5 + 1)))
        if rng.random() < 0.5:
            h[i:i + blen] = 0.9
        i += blen
    return np.clip(h, 0.0, 1.0)
