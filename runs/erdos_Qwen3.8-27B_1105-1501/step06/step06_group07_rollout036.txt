import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_S(h):
    """Compute raw correlation S = correlate(h, 1-h, 'full')."""
    return np.correlate(h, 1.0 - h, mode='full')

    Returns (best_t, best_val) where best_val is the new max correlation (before *dx).

    t_hi = min(1.0 - h[i], h[j])

        return 0.0, float(np.max(S))

    # Grid search
    n_grid = 40
    t_grid = np.linspace(t_lo, t_hi, n_grid)

    #       - h[i+k] if 0<=i+k<n else 0

        T[ks] += (1.0 - h[i - ks])
    # corr(e_j, 1-h): k=0..j, T[k] -= (1-h)[j-k]
    if j >= 0:

        else:

            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = eval_f(c2)

    opt_val = eval_f(opt_t)
    if opt_val < best_val:

= corr(e_i,1-h) - corr(e_j,1-h) - corr(h,e_i) + corr(h,e_j)

T[k] = (1-h)[i-k] (if k≤i) - (1-h)[j-k] (if k≤j) - h[i+k] (if k≤n-1-i) + h[j+k] (if k≤n-1-j)

This matches what I had before. Good.

Now, the full objective along the line is:
C5(h + t*(e_i - e_j)) = dx * max_k (S[k] + t*T[k] - t²*Q[k])

where Q[0]=2, Q[|i-j|]=-1, Q[k]=0 otherwise.

And -t²*Q[k]:
- At k=0: -2t²
- At k=|i-j|: +t²
- Elsewhere: 0

So the term S[k] + t*T[k] - t²*Q[k] is:
- k=0: S[0] + t*T[0] - 2t²
- k=|i-j|: S[|i-j|] + t*T[|i-j|] + t²
- k≠0,|i-j|: S[k] + t*T[k]

The maximum of these over all k gives the new max correlation.

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_sweep(h, S, order, dx, max_improves=None):

    Returns updated h, S, and number of improvements.
    """

    m = len(S)  # 2n-1
    improvements = 0

        if max_improves is not None and improvements >= max_improves:
            break
        cur_max = float(np.max(S))
        
        # Feasible t range for h_i + t, h_j - t
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        if t_hi - t_lo < 1e-14:
            continue
        
        # Compute T and Q for this pair
        k_max = max(i, j, n - 1 - i, n - 1 - j)
        d_abs = abs(i - j)
        
        # Background max
        bg_max = 0.0
        if k_max + 1 < m:
            bg_max = float(np.max(S[k_max + 1:]))
        
        # Build T
        T = np.zeros(k_max + 1)
        # (1-h)[i-k] for k=0..min(i, k_max)
        ki = min(i, k_max)
        if ki >= 0:
            T[:ki+1] += (1.0 - h[i - np.arange(ki+1)])
        # -(1-h)[j-k] for k=0..min(j, k_max)
        kj = min(j, k_max)
        if kj >= 0:
            T[:kj+1] -= (1.0 - h[j - np.arange(kj+1)])
        # -h[i+k] for k=0..min(n-1-i, k_max)
        kii = min(n - 1 - i, k_max)
        if kii >= 0:
            T[:kii+1] -= h[i + np.arange(kii+1)]
        # +h[j+k] for k=0..min(n-1-j, k_max)
        kjj = min(n - 1 - j, k_max)
        if kjj >= 0:
            T[:kjj+1] += h[j + np.arange(kjj+1)]
        
        # Q values
        # Q[0] = 2, Q[d_abs] = -1
        # Effect on objective: -t^2 * Q
        # At k=0: -2t^2
        # At k=d_abs: +t^2
        
        S_sub = S[:k_max + 1]
        
        # Grid search
        n_grid = 30
        t_grid = np.linspace(t_lo, t_hi, n_grid)
        
        # values[k, m] = S[k] + t[T[k]] - t^2 * Q_val[k]
        # Q_val: 2 at position 0, -1 at position d_abs, 0 elsewhere
        Q_arr = np.zeros(k_max + 1)
        Q_arr[0] = 2.0
        if 0 < d_abs <= k_max:
            Q_arr[d_abs] = -1.0
        
        vals = S_sub[:, None] + t_grid[None, :] * T[:, None] - (t_grid[None, :]**2) * Q_arr[:, None]
        f_vals = np.maximum(np.max(vals, axis=0), bg_max)
        
        best_i = int(np.argmin(f_vals))
        best_t = t_grid[best_i]
        best_f = f_vals[best_i]
        
        # Check if improvement
        if best_f >= cur_max - 1e-14:
            continue
        
        # Golden section refinement
        if best_i > 0 and best_i < n_grid - 1:
            a = t_grid[best_i - 1]
            b = t_grid[best_i + 1]
        else:
            a = t_lo
            b = t_hi
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        
        def _eval_f(t, S_sub=S_sub, T=T, Q_arr=Q_arr, bg_max=bg_max):
            v = S_sub + t * T - t * t * Q_arr
            return max(bg_max, float(np.max(v)))
        
        f1 = _eval_f(c1)
        f2 = _eval_f(c2)
        for _ in range(40):
            if b - a < 1e-13:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = _eval_f(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = _eval_f(c2)
        
        opt_t = (a + b) / 2.0
        opt_f = _eval_f(opt_t)
        if opt_f < best_f:
            best_t, best_f = opt_t, opt_f
        
        # Final check
        if best_f >= cur_max - 1e-14:
            continue
        
        # Apply the move
        h[i] += best_t
        h[j] -= best_t
        
        # Update S: S_new[k] = S[k] + best_t * T[k] - best_t^2 * Q[k] for k in support
        S[:k_max+1] = S_sub + best_t * T - best_t*best_t * Q_arr
        # Positions beyond k_max are unchanged
        
        improvements += 1
    
    return h, S, improvements
