    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, r, n):
    g = np.zeros(n)
    k = r - (n - 1)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += (1.0 - h[:m])
            g[:m] += -h[k:]
    else:
        d = -k
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d+m])
            g[d:] += -h[:m]
    return g

def _line_search_quad(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_pos = d > 1e-15
    t_neg = d < -1e-15
    t_lo = -radius
    t_hi = radius
    if np.any(t_pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[t_pos]) / d[t_pos])))
    if np.any(t_neg):
        t_lo = max(t_lo, float(np.max(h[t_neg] / d[t_neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(55):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _four_tuple_move(h, i, j, k, l, dx, best_c5):
    """Try 4-point move: +1 at i,j and -1 at k,l (or sign flip)."""
    n = len(h)
    target = n / 2.0
    for signs in [(1, 1, -1, -1), (-1, -1, 1, 1)]:
        d = np.zeros(n)
        d[i] = signs[0]
        d[j] = signs[1]
        d[k] = signs[2]
        d[l] = signs[3]
        alpha, c5_try = _line_search_quad(h, d, dx, 0.4)
        if c5_try < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            best_c5 = _exact_c5(h)
    return h, best_c5

def _smooth_grad(h, n, beta, top_k):
    S = _corr_scaled(h)
    M = S.max()
    tol = max(0.03 * M, 1e-4)
    active = S >= M - tol
    idx = np.where(active)[0]
    if len(idx) > top_k:
        sorted_idx = np.argsort(-S)[:top_k]
        idx = sorted_idx
    if len(idx) == 0:
        return np.zeros(n), S
    sub_S = S[idx]
    e = np.exp((sub_S - sub_S.max()) / beta)
    total = e.sum()
    if total < 1e-300:
        w = np.ones(len(idx)) / len(idx)
    else:
        w = e / total
    grad = np.zeros(n)
    for ii, r in enumerate(idx):
        grad += w[ii] * _grad_row(h, r, n)
    grad -= grad.mean()
    return grad, S

def _main_optimize(h_init, deadline, rng):
    """Combined smooth gradient + pair + 4-tuple optimization."""
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    mu = 0.85
    beta = 0.04 * best_c5
    lr = 0.04
    stall = 0
    total_iter = 0
    
    while time.time() < deadline:
        total_iter += 1
        grad, S = _smooth_grad(h, n, beta, top_k=25)
        
        # Smooth gradient step with momentum
        v = mu * v + grad
        h_try = h - lr * v
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            stall = 0
            if total_iter % 30 == 0:
                beta = max(0.002 * best_c5, beta * 0.96)
                lr = min(0.12, lr * 1.03)
        else:
            stall += 1
            v *= 0.7
        
        # Periodically inject pair moves
        if total_iter % 10 == 0:
            pos_idx = np.argsort(-grad)[:8]
            neg_idx = np.argsort(grad)[:8]
            for ii in pos_idx[:6]:
                if time.time() > deadline:
                    break
                for jj in neg_idx[:6]:
                    if ii == jj:
                        continue
                    if h[int(ii)] >= 1.0 - 1e-8 or h[int(jj)] <= 1e-8:
                        continue
                    d = np.zeros(n)
                    d[int(ii)] = 1.0
                    d[int(jj)] = -1.0
                    alpha, c5_p = _line_search_quad(h, d, dx, 0.45)
                    if c5_p < best_c5 - 1e-14:
                        h = h + alpha * d
                        h = _project(h, target)
                        best_c5 = _exact_c5(h)
                        best_h = h.copy()
                        stall = 0
                        break
        
        # Occasionally try 4-tuple moves
        if total_iter % 15 == 0 and stall > 3:
            pos4 = np.argsort(-grad)[:10]
            neg4 = np.argsort(grad)[:10]
            for ii in pos4[:4]:
                if time.time() > deadline:
                    break
                for jj in pos4[:4]:
                    if ii == jj:
                        continue
                    for kk in neg4[:4]:
                        for ll in neg4[:4]:
                            if kk == ll or kk == ii or kk == jj or ll == ii or ll == jj:
                                continue
                            if h[int(ii)] >= 1.0 - 1e-8 or h[int(jj)] >= 1.0 - 1e-8:
                                continue
                            if h[int(kk)] <= 1e-8 or h[int(ll)] <= 1e-8:
                                continue
                            h, best_c5 = _four_tuple_move(h, int(ii), int(jj), int(kk), int(ll), dx, best_c5)
                            if best_c5 < _exact_c5(best_h) - 1e-14:
                                best_h = h.copy()
                                stall = 0
                            break
                        if time.time() > deadline:
                            break
        
        # Adaptive parameters
        if stall > 40:
            beta = max(0.001 * best_c5, beta * 0.8)
            lr = max(0.01, lr * 0.85)
            v *= 0.3
            if stall > 120:
                h = h + rng.normal(0, 0.015, n)
                h = _project(h, target)
                v = np.zeros(n)
                stall = 60
    
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    """Binary simulated annealing with adaptive acceptance."""
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.000003
    cooldown = 0.99995
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        for _ in range(300):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = int(ones_list[rng.integers(len(ones_list))])
            j = int(zeros_list[rng.integers(len(zeros_list))])
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _de_population(h_init, deadline, pop_size=14, max_gen=2000):
    """Differential evolution with binary constraint handling."""
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    rng = np.random.default_rng(77777)
    
    pop = [h_init.copy()]
    for i in range(pop_size - 1):
        h_i = h_init + rng.normal(0, 0.04, n)
        h_i = _project(h_i, target)
        pop.append(h_i)
    fitness = [_exact_c5(h) for h in pop]
    
    d_param = 0.6
    c_param = 0.8
    stall = 0
    gen = 0
    
    while time.time() < deadline and gen < max_gen:
        gen += 1
        order = np.argsort(fitness)
        pop = [pop[i] for i in order]
        fitness = [fitness[i] for i in order]
        
        if fitness[0] < best_c5 - 1e-15:
            best_c5 = fitness[0]
            best_h = pop[0].copy()
            stall = 0
            d_param = max(0.3, d_param * 0.99)
        else:
            stall += 1
            if stall > 80:
                d_param = min(1.0, d_param * 1.15)
                for i in range(max(2, pop_size // 4)):
                    pop[i] = best_h + rng.normal(0, 0.02 * (1 + stall / 200), n)
                    pop[i] = _project(pop[i], target)
                    fitness[i] = _exact_c5(pop[i])
                stall = 0
        
        for i in range(1, pop_size):
            if time.time() > deadline:
                break
            others = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(others, size=3, replace=False)
            mutant = pop[r1] + d_param * (pop[r2] - pop[r3])
            mutant = _project(mutant, target)
            trial = pop[i].copy()
            cross = rng.random(n) < c_param
            trial[cross] = mutant[cross]
            trial = _project(trial, target)
            fit_t = _exact_c5(trial)
            if fit_t <= fitness[i] + 1e-15:
                pop[i] = trial
                fitness[i] = fit_t
    
    return best_h, best_c5

def _intensive_pairs(h, deadline, max_passes=500):
    """Very intensive pair refinement."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        S = _corr_scaled(h)
        M = S.max()
        tol = max(0.02 * M, 1e-5)
        active = np.where(S >= M - tol)[0]
        
        grad = np.zeros(n)
        for r in active[:30]:
            grad += _grad_row(h, r, n)
        if len(active) > 0:
            grad /= min(len(active), 30)
        grad -= grad.mean()
        
        n_cand = min(n, 50)
        top_pos = np.argsort(-grad)[:n_cand]
        top_neg = np.argsort(grad)[:n_cand]
        
        improved = False
        for ii in top_pos[:20]:
            if time.time() > deadline:
                break
            for jj in top_neg[:20]:
                if ii == jj:
                    continue
                if h[int(ii)] >= 1.0 - 1e-9 or h[int(jj)] <= 1e-9:
                    continue
                d = np.zeros(n)
                d[int(ii)] = 1.0
                d[int(jj)] = -1.0
                alpha, c5_try = _line_search_quad(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            for _ in range(30):
                if time.time() > deadline:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or h[i] >= 1.0 - 1e-9 or h[j] <= 1e-9:
                    continue
                d = np.zeros(n)
                d[i] = 1.0
                d[j] = -1.0
                alpha, c5_try = _line_search_quad(h, d, dx, 0.45)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
            if not improved:
                # Try 4-tuples
                interior = np.where((h > 0.05) & (h < 0.95))[0]
                if len(interior) >= 4:
                    idx4 = rng.choice(len(interior), size=4, replace=False)
                    i, j, k, l = interior[idx4]
                    h, best_c5 = _four_tuple_move(h, int(i), int(j), int(k), int(l), dx, best_c5)
                    if best_c5 < _exact_c5(best_h) - 1e-14:
                        best_h = h.copy()
                        improved = True
            if not improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.985
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)
    
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    
    # Phase 1: Main combined optimization (30%)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _main_optimize(best_h, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Intensive pairs (12%)
    d2 = t0 + budget_s * 0.42
    h2, c5_2 = _intensive_pairs(best_h, d2, max_passes=400)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Binary SA (15%)
    d3 = t0 + budget_s * 0.57
    h3, c5_3 = _binary_sa(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Phase 4: DE population (15%)
    d4 = t0 + budget_s * 0.72
    h4, c5_4 = _de_population(best_h, d4, pop_size=14, max_gen=2000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Multi-resolution (20%)
    d5 = t0 + budget_s * 0.92
    base_n = len(best_h)
    offsets = sorted(set(range(-24, 25, 2)), key=lambda x: abs(x))
    rng5 = np.random.default_rng(seed + 200)
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d5, time.time() + budget_s * 0.018)
            h_r, c5_r = _main_optimize(h_r, sub_dl, rng)
            if c5_r < best_c5 + 0.004 and time.time() < d5:
                sub_dl2 = min(d5, time.time() + budget_s * 0.014)
                h_r, c5_r = _intensive_pairs(h_r, sub_dl2, max_passes=100)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
    
    # Phase 6: Final intensive polish (8%)
    if time.time() < deadline:
        h6, c5_6 = _intensive_pairs(best_h, deadline, max_passes=300)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
