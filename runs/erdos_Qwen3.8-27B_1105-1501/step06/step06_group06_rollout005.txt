import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad_single(h, k):
    n = len(h)
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    m = n - abs(s)
    if s >= 0:
        g[:m] += (1.0 - h[s:s + m])
        g[s:s + m] += -h[:m]
    else:
        gg = -s
        g[gg:gg + m] += (1.0 - h[:m])
        g[:m] += -h[gg:gg + m]
    return g * dx

        c = corr
    cm = c.max()
    shifted = tau * (c - cm)

def _smooth_obj_grad(x):

    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(c))
        w[0] = 1.0
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(len(idx)):

    return grad

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _perturb(h, sigma, rng):
    h_p = h + rng.normal(0, sigma, len(h))
    return _project(h_p, len(h) / 2.0)

def _block_perturb(h, rng):
    n = len(h)
    h_p = h.copy()
    for _ in range(rng.integers(1, 4)):

        length = int(rng.integers(2, max(3, n // 4)))
        length = min(length, n - start)
        delta = rng.normal(0, 0.1)
        h_p[start:start + length] += delta
    return _project(h_p, n / 2.0)
