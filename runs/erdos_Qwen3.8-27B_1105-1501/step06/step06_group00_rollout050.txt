def _compute_gradient(h, w):
    n = len(h)
    grad = np.zeros(n)
    # Lag 0
    grad += w[n-1] * (1 - 2*h)
    # Positive lags
    for g in range(1, n):
        a = w[n-1+g]
        b = w[n-1-g]
        # A[g] contribution: [g<=i]*(1-h[i-g]) - [i<=n-1-g]*h[i+g]
        if a != 0:
            # First part: i ranges from g to n-1
            grad[g:] += a * (1 - h[:n-g])
            # Second part: i ranges from 0 to n-1-g
            grad[:n-g] -= a * h[g:]
        # B[g] contribution: [i<=n-1-g]*(1-h[i+g]) - [g<=i]*h[i-g]
        if b != 0:
            grad[:n-g] += b * (1 - h[g:])
            grad[g:] -= b * h[:n-g]
    return grad

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Water-filling projection onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad(h, w):
    """Compute G^T @ w where G is the gradient matrix of c_k wrt h."""
    n = len(h)
    grad = np.zeros(n)
    # Lag 0: ∂c[0]/∂h[i] = 1 - 2*h[i]
    grad += w[n-1] * (1.0 - 2.0 * h)
    # Other lags
    for g in range(1, n):
        a = w[n-1+g]  # weight for +g lag
        b = w[n-1-g]  # weight for -g lag
        if a > 1e-15:
            grad[g:] += a * (1.0 - h[:-g])
            grad[:-g] -= a * h[g:]
        if b > 1e-15:
            grad[:-g] += b * (1.0 - h[g:])
            grad[g:] -= b * h[:-g]
    return grad

def _pair_direction(h, i, j, rho):
    """Feasible alpha range for transfer from j to i."""
    lo = max(-h[i], h[j] - 1.0, -rho)
    hi = min(1.0 - h[i], h[j], rho)
    if lo >= hi - 1e-14:
        return None
    return lo, hi

def _line_search(h, i, j, lo, hi, n_grid=25):
    """Find best alpha in [lo, hi] for pair transfer h + alpha*(e_i - e_j)."""
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    # C is sparse: C[0]=2, C[j-i]=-1, C[i-j]=-1 (at appropriate positions)
    C = np.zeros(2*n-1)
    C[n-1] = 2.0  # lag 0
    di = j - i
    dj = i - j
    if di != 0:
        C[n-1+di] = -1.0
    if dj != 0:
        C[n-1+dj] = -1.0
    
    alphas = np.linspace(lo, hi, n_grid)
    best_val = float(np.max(S0) * dx)
    best_alpha = 0.0
    for a in alphas:
        val = float(np.max(S0 + a * T - a * a * C)) * dx
        if val < best_val:
            best_val = val
            best_alpha = a
    return best_alpha, best_val

def _optimize_pairs(h, deadline, beta_init=20.0, m_init=8, rho_init=0.15, max_iter=5000):
    """Main optimization loop using pair transfers guided by smooth-max gradient."""
    n = len(h)
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = beta_init
    m = m_init
    rho = rho_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        # Compute scores and smooth-max weights
        S0 = np.correlate(h, 1.0 - h, mode='full')
        scores = S0 * (2.0 / n)
        M = np.max(scores)
        if M <= 0:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            e = np.exp(-beta * (M - scores))  # exp(beta*(scores-M)) clipped
            # Wait: we want to emphasize the MAX, so weight = exp(beta * scores) normalized
            # Better: w_k = exp(beta * (s_k - M)) / sum  (this puts weight on max)
            e = np.exp(beta * (scores - M))
            s = e.sum()
            if s < 1e-30:
                w = np.ones(2*n-1) / (2*n-1)
            else:
                w = e / s
        
        # Compute gradient
        grad = _compute_grad(h, w)
        
        # Select top-m sources and bottom-m sinks
        order = np.argsort(grad)
        sinks = order[:m]  # lowest gradient
        sources = order[-m:]  # highest gradient
        
        # Try all m*m pairs
        best_alpha = 0.0
        best_c5_try = cur_c5
        best_i, best_j = -1, -1
        
        for si in sources:
            for sj in sinks:
                if si == sj:
                    continue
                range_ = _pair_direction(h, si, sj, rho)
                if range_ is None:
                    continue
                lo, hi = range_
                alpha, c5_try = _line_search(h, si, sj, lo, hi, n_grid=20)
                if c5_try < best_c5_try:
                    best_c5_try = c5_try
                    best_alpha = alpha
                    best_i, best_j = si, sj
        
        if best_i >= 0 and best_c5_try < cur_c5 - 1e-14:
            h[best_i] += best_alpha
            h[best_j] -= best_alpha
            h = np.clip(h, 0.0, 1.0)
            # Re-project to maintain sum (should be exact but floating point)
            target = n / 2.0
            if abs(float(np.sum(h)) - target) > 1e-10:
                h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            rho = min(rho * 1.05, 0.35)
        else:
            stall += 1
            rho *= 0.85
            if stall % 50 == 0:
                beta = beta * 1.2  # sharpen focus
                m = max(4, m - 1)
                if rho < 0.01:
                    rho = 0.05
                    beta = max(10.0, beta * 0.7)
    
    return best_h, best_c5

```python
def _line_search_fast(h, i, j, lo, hi, n_grid=25):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    # T[k] for d = e_i - e_j:
    # T = correlate(e_i-e_j, 1-h) - correlate(h, e_i-e_j)
    # correlate(e_i, 1-h)[k] = (1-h[i-k]) if 0<=i-k<n else 0  [shifted (1-h)]
    # Actually, correlate(a,b)[k] = sum_n a[n+k]*b[n]
    # correlate(e_i, 1-h)[k] = (1-h) at position where i = n+k... 
    # Wait: correlate(e_i, b)[k] = sum_n e_i[n+k] * b[n] = b[n] where n+k=i, i.e., n=i-k
    # = b[i-k] if 0 <= i-k <= n-1, else 0
    # So correlate(e_i, 1-h)[k] = (1-h[i-k]) if 0<=i-k<=n-1
    # correlate(e_j, 1-h)[k] = (1-h[j-k]) if 0<=j-k<=n-1
    # correlate(h, e_i)[k] = sum_n h[n+k]*e_i[n] = h[n+k] where n=i, so h[i+k] if 0<=i+k<=n-1
    # correlate(h, e_j)[k] = h[j+k] if 0<=j+k<=n-1
    
    # So T[k] = (1-h[i-k])*(valid) - (1-h[j-k])*(valid) - h[i+k]*(valid) + h[j+k]*(valid)
    # This is nonzero only at k values where i-k, j-k, i+k, or j+k is in [0,n-1]
    # k ranges from -(n-1) to n-1
    
    # C[k] for d = e_i - e_j:
    # C[0] = 2, C[j-i] = -1, C[i-j] = -1 (in the full correlation indexing)
    # In numpy index: position p corresponds to lag p-(n-1)
    # C at lag 0: index n-1, value 2
    # C at lag (j-i): index n-1+(j-i), value -1
    # C at lag (i-j): index n-1+(i-j), value -1
    
    # For the line search, we need: score(alpha) = max_k (S0[k] + alpha*T[k] - alpha^2*C[k])
    # T and C are sparse, so we only need to evaluate at the nonzero positions of T and C
    # plus we need to check if the max is at one of those positions or elsewhere.
    
    # Actually, since T and C are sparse (at most 8 nonzero positions for T, 3 for C),
    # the only positions where the score can CHANGE with alpha are those nonzero positions.
    # At all other positions, the score is just S0[k] (constant in alpha).
    # So the max at any alpha is either:
    #   max of (S0[k] + alpha*T[k] - alpha^2*C[k]) for k in nonzero_positions
    #   or max of S0[k] for k not in nonzero_positions (which is constant)
    
    # Let's find the constant max (positions where T=0 and C=0)
    # And the variable positions
    
    # Actually, C is nonzero at positions n-1, n-1+(j-i), n-1+(i-j)
    # T is nonzero at positions where i-k, j-k, i+k, or j+k is in [0,n-1]
    # That's positions k = i - r for r in [0,n-1] (i.e., k in [i-(n-1), i])
    #            and k = j - r for r in [0,n-1] (i.e., k in [j-(n-1), j])
    #            and k = r - i for r in [0,n-1] (i.e., k in [-i, n-1-i])
    #            and k = r - j for r in [0,n-1] (i.e., k in [-j, n-1-j])
    
    # In full correlation indexing (0 to 2n-2), position p = k + (n-1):
    # T is potentially nonzero for p in:
    #   [i-(n-1)+(n-1), i+(n-1)] = [i, i+2n-2]  (from i-k valid)
    #   [j, j+2n-2]  (from j-k valid)
    #   [(n-1)-i, n-1-i+n-1] = [n-1-i, 2n-2-i]  (from i+k valid)
    #   [n-1-j, 2n-2-j]  (from j+k valid)
    
    # This is at most 4(n-1) positions, which is O(n). Not that sparse for n=200.
    # But we can compute T as a full vector quickly using the 4-lookup formula.
    
    # For efficiency, let's just build T as a full vector (it's O(n) work).
    T = np.zeros(2*n-1)
    # T[p] where p = k + (n-1), so k = p - (n-1)
    # T[k] = (1-h[i-k]) if 0<=i-k<=n-1 else 0
    #      - (1-h[j-k]) if 0<=j-k<=n-1 else 0
    #      - h[i+k] if 0<=i+k<=n-1 else 0
    #      + h[j+k] if 0<=j+k<=n-1 else 0
    # k ranges from -(n-1) to n-1, so p from 0 to 2n-2
    
    ks = np.arange(-(n-1), n)  # all lags
    # (1-h[i-k]) valid when 0 <= i-k <= n-1, i.e., i-(n-1) <= k <= i
    mask1 = (ks >= i-(n-1)) & (ks <= i)
    idx1 = i - ks[mask1]  # indices into h
    if np.any(mask1):
        T[mask1] += (1.0 - h[idx1])
    
    # -(1-h[j-k]) valid when 0 <= j-k <= n-1, i.e., j-(n-1) <= k <= j
    mask2 = (ks >= j-(n-1)) & (ks <= j)
    idx2 = j - ks[mask2]
    if np.any(mask2):
        T[mask2] -= (1.0 - h[idx2])
    
    # -h[i+k] valid when 0 <= i+k <= n-1, i.e., -i <= k <= n-1-i
    mask3 = (ks >= -i) & (ks <= n-1-i)
    idx3 = i + ks[mask3]
    if np.any(mask3):
