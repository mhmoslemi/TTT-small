def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection onto {0 <= v <= 1, sum(v) = target}
    n = len(v)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, k):
    """Gradient of corr[k] with respect to h, times dx."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k >= n:
            return g
        # corr[k] = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
        # d/dh[i] = (1-h[i-k]) for k<=i<n, minus h[i+k] for 0<=i<n-k
        idx1 = np.arange(k, n)
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        g[idx2] -= h[idx2 + k]
    else:

            best_t = t

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        
        # Select active set with softmax weights
        alpha = alpha_base + it * 0.5
        w = np.exp(alpha * (corr - M))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros_like(w)
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        # Keep only top K

        top_idx = np.argsort(-corr)[:top_k]

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_vec(h):

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k >= n:
            return g
        idx1 = np.arange(k, n)
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        g[idx2] -= h[idx2 + k]
    else:
        m = -k
        if m >= n:
            return g
        idx1 = np.arange(0, n - m)
        g[idx1] += (1.0 - h[idx1 + m])
        idx2 = np.arange(m, n)

    Uses exact quadratic model f(t) = max_r (S_r + t*T_r + t^2*Q_r)*dx"""

def _active_descent(h, deadline, K=20, alpha_init=25.0, max_iters=1000):

    best_c5 = _exact_c5(h)
    alpha = alpha_init
    stall = 0
    
    for it in range(max_iters):

            break
        
        corr = _corr_vec(h) * dx
        M = float(np.max(corr))
        
        # Softmax weights for active constraints
        w = np.exp(alpha * (corr - M))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx_max = np.argmax(corr)
            w = np.zeros_like(w)
            w[idx_max] = 1.0
        else:
            w /= w_sum
        
        # Weighted gradient
        grad = np.zeros(n)
        # Only compute for significant weights
        active = w > 1e-6
        for idx in np.where(active)[0]:
            k = int(idx) - (n - 1)
            grad += w[idx] * _grad_lag(h, k)
