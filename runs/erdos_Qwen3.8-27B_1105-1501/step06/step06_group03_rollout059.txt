```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    # Project v onto {x : 0≤x≤1, sum(x)=target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

2. Checking the minimum at each breakpoint and at the vertex of each active parabola

```python
def _exact_1d_min(S0, T, C, t_lo, t_hi, K=20):
    """Find exact minimum of max_k (S0[k] + t*T[k] - t²*C[k]) over t in [t_lo, t_hi]."""
    # Take top K by value at t=0
    vals0 = S0.copy()
    top_k = np.argsort(-vals0)[:K]
    
    candidates = [t_lo, t_hi]
    
    # Add vertices of top K parabolas
    for k in top_k:
        if C[k] > 1e-15:
            t_v = T[k] / (2.0 * C[k])
            if t_lo <= t_v <= t_hi:
                candidates.append(t_v)
    
    # Add pairwise intersections of top K parabolas
    for i in range(len(top_k)):
        for j in range(i+1, len(top_k)):
            ki, kj = top_k[i], top_k[j]
            # S0[i] + t*T[i] - t²*C[i] = S0[j] + t*T[j] - t²*C[j]
            # (C[j]-C[i])t² + (T[i]-T[j])t + (S0[i]-S0[j]) = 0
            a = C[kj] - C[ki]
            b = T[ki] - T[kj]
            c = S0[ki] - S0[kj]
            if abs(a) < 1e-15:
                if abs(b) > 1e-15:
                    t_int = -c / b
                    if t_lo <= t_int <= t_hi:
                        candidates.append(t_int)
            else:
                disc = b*b - 4*a*c
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    for t_int in [(-b + sqrt_disc) / (2*a), (-b - sqrt_disc) / (2*a)]:
                        if t_lo <= t_int <= t_hi:
                            candidates.append(t_int)
    
    # Evaluate at all candidates
    best_t = t_lo
    best_val = float('inf')
    for t in candidates:
        vals = S0 + t * T - t*t * C
        val = float(np.max(vals))
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_gradient(h, w, dx):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    idx_p = np.arange(n)[:, None] + lags[None, :]
    valid_p = (idx_p >= 0) & (idx_p < n)
    idx_p_c = np.clip(idx_p, 0, n - 1)
    h_p = h[idx_p_c]
    P = (w[None, :] * (1.0 - h_p) * valid_p.astype(float)).sum(axis=1)
    idx_m = np.arange(n)[:, None] - lags[None, :]
    valid_m = (idx_m >= 0) & (idx_m < n)
    idx_m_c = np.clip(idx_m, 0, n - 1)
    h_m = h[idx_m_c]
    Q = (w[None, :] * h_m * valid_m.astype(float)).sum(axis=1)
    return dx * (P - Q)

def _smoothed_descent(h_init, tau, deadline, n_steps=3000):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()

    best_c5 = _exact_c5(h)
    alpha = 0.05
    momentum = np.zeros(n)
    beta = 0.85
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        S = _corr_raw(h)
        M = float(S.max())
        w = np.exp((S - M) / tau)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum
        grad = _smooth_gradient(h, w, dx)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        momentum = beta * momentum - alpha * grad
        momentum = momentum - np.mean(momentum)
        h_try = h + momentum
        h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()

        else:

                h = h_try
            else:
                momentum = momentum + alpha * grad
                alpha *= 0.7
            if alpha < 1e-10:

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d + m])
                g[d:] += -h[:m]
    return G

def _exact_1d_min(S0, T, C, t_lo, t_hi, K=25):
    vals0 = S0 + 0.0 * T - 0.0 * C
    top_k = np.argsort(-vals0)[:K]
    M = float(vals0.max())
    tol = max(0.03 * M, 1e-6)
    near = np.where(vals0 >= M - tol)[0]
    combined = np.unique(np.concatenate([top_k, near]))
    if len(combined) > K:
        combined = top_k
    candidates = [t_lo, t_hi]
    for ki in combined:
        if C[ki] > 1e-15:
            t_v = T[ki] / (2.0 * C[ki])
            if t_lo - 1e-12 <= t_v <= t_hi + 1e-12:
                candidates.append(max(t_lo, min(t_hi, t_v)))
    n_c = len(combined)
    for ii in range(n_c):
        for jj in range(ii + 1, n_c):

            a = C[kj] - C[ki]
            b = T[ki] - T[kj]
            c = S0[ki] - S0[kj]
            if abs(a) < 1e-15:
                if abs(b) > 1e-15:

                    if t_lo - 1e-12 <= t_int <= t_hi + 1e-12:
                        candidates.append(max(t_lo, min(t_hi, t_int)))
            else:

                if disc >= 0:

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    best_t, best_val = _exact_1d_min(S0, T, C, t_lo, t_hi, K=30)
    return best_t, best_val * dx

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.04 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_dir(h, S, n, tau):
    M = float(S.max())
    shifted = (S - M) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_k = min(25, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    g = w[top_idx] @ G
    g -= g.mean()
    if np.abs(g).max() < 1e-12:
        return None
    nrm = np.linalg.norm(g)
    if nrm < 1e-12:
        return None
    return -g / nrm

def _lp_loop(h, deadline, radius_init=0.12, K=15, use_smooth=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    tau = 0.02 * cur_c5
    while time.time() < deadline:
        if stall > 50:
            if restart_ctr < 12:
                scale = 0.008 + 0.006 * restart_ctr
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.06
                stall = 0
                restart_ctr += 1
                tau = 0.02 * cur_c5
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if (p is None or float(np.max(np.abs(p))) < 1e-13) and use_smooth:
            p = _smooth_dir(h, S, n, tau)
            if p is not None:
                p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if use_smooth and stall % 5 == 0:
                tau *= 0.8
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.08, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.93, 0.02)
        else:
            stall += 1
            radius *= 0.72
            if stall % 10 == 0 and radius < 0.012:
                radius = 0.04
    return best_h, best_c5

def _double_swap_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    cooldown = 0.99995
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        n_iter = 300
        for _ in range(n_iter):
            if len(ones_list) < 2 or len(zeros_list) < 2:
                break
            i1 = ones_list[rng.integers(len(ones_list))]
            i2 = ones_list[rng.integers(len(ones_list))]
            j1 = zeros_list[rng.integers(len(zeros_list))]
            j2 = zeros_list[rng.integers(len(zeros_list))]
            if i1 == i2 or j1 == j2:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _pair_refine(h, deadline, max_passes=150):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _multi_pair_refine(h, deadline):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(444)
    stall = 0
    while time.time() < deadline and stall < 50:
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 4:
            break
        idx = rng.choice(len(interior), size=min(4, len(interior)), replace=False)
        i, j, k, l = interior[idx]
        found = False
        for signs in [(-1, -1, 1, 1), (1, 1, -1, -1)]:
            d = np.zeros(n)
            d[i] = signs[0]
            d[j] = signs[1]
            d[k] = signs[2]
            d[l] = signs[3]
            alpha, c5_try = _line_search(h, d, dx, 0.3)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                    found = True
                break
        if not found:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start LP (7% time)
    d1 = t0 + budget_s * 0.07
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 9]:
        for ph in [0.0, 0.7, 1.5, 2.3, 3.1]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(10):
        starts.append(_block_init(n0, rng, int(rng.integers(3, 17))))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Smoothed gradient descent (16% time)
    d2 = t0 + budget_s * 0.23
    tau_list = [0.06, 0.04, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
    sub_budget = (d2 - time.time()) / len(tau_list)
    for tau in tau_list:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + max(sub_budget, budget_s * 0.008))
        h_sm, c5_sm = _smoothed_descent(best_h, tau, sub_dl, n_steps=2000)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()

    # Phase 3: Main LP optimization (22% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.14, K=18, use_smooth=False)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP with smooth backup (12% time)
    d4 = t0 + budget_s * 0.57
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.08, K=22, use_smooth=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Double-swap SA + LP polish (13% time)
    d5 = t0 + budget_s * 0.70
    h5a, c5_5a = _double_swap_sa(best_h, t0 + budget_s * 0.63, rng)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5:
        h5b, c5_5b = _lp_loop(best_h, d5, radius_init=0.06, K=15)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution (12% time)
    d6 = t0 + budget_s * 0.82
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -32, 32]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.006:
            sub_dl = min(d6, time.time() + budget_s * 0.018)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=14)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.015:
                sub_dl2 = min(d6 - budget_s * 0.02, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = _double_swap_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d6:
                    h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + budget_s * 0.01),
                                         radius_init=0.05, K=14)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Aggressive perturbation + re-opt (5% time)
    d7 = t0 + budget_s * 0.87
    scale = 0.025
    for attempt in range(400):
        if time.time() > d7:
            break
        if best_c5 < 0.3803:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=12)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.75)
        else:
            scale = min(0.15, scale * 1.25)

    # Phase 8: Final pair + multi-pair polish (6% time)
    d8 = t0 + budget_s * 0.93
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=100)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline:
        h9, c5_9 = _multi_pair_refine(best_h, deadline)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final LP polish
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.03, K=10)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
```
