import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_gradient(h, gamma):
    """Compute the soft-max weighted gradient of the C5 objective."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = c.max()
    # Numerical stability: subtract M before exp
    w = np.exp(gamma * (c - M))
    total = w.sum()
    if total < 1e-300:
        # Fallback: uniform weights
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = w / total
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)  # lag
        # First term: (1-h)[i+s] for valid i
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += pk * one_minus_h[i_lo+s:i_hi+s+1]
        # Second term: -h[i-s] for valid i
        i_lo2 = max(0, s)
        i_hi2 = min(n-1, n-1+s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= pk * h[i_lo2-s:i_hi2-s+1]
    grad *= dx
    return grad

def _line_search(h, d):
    """Exact 1D line search along direction d (sum(d)=0).
    Returns (best_alpha, best_c5_value)."""
    n = len(h)
    dx = 2.0 / n

    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    
    # Feasible interval for alpha

    pos = d > eps
    neg = d < -eps
    
    alpha_lo = 0.0
    alpha_hi = 0.0

def _project(v, target):

        return np.zeros(n)

        if s > target:

        else:

def _soft_gradient(h, gamma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = c.max()
    w = np.exp(gamma * (c - M))

    if total < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = w / total
    grad = np.zeros(n)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_gradient(h, gamma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    M = c.max()
    w = np.exp(gamma * (c - M))
    total = w.sum()
    if total < 1e-300:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = w / total
    grad = np.zeros(n)
    omlh = 1.0 - h
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += pk * omlh[i_lo + s:i_hi + s + 1]
        i_lo2 = max(0, s)
        i_hi2 = min(n - 1, n - 1 + s)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= pk * h[i_lo2 - s:i_hi2 - s + 1]
    grad *= dx
    return grad

def _line_search(h, d):
    n = len(h)
    dx = 2.0 / n
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    eps = 1e-14
    pos = d > eps
    neg = d < -eps
    alpha_lo = 0.0
    alpha_hi = 0.0
    if np.any(pos):
        alpha_lo = max(alpha_lo, float(np.max(-h[pos] / d[pos])))
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        alpha_hi = min(alpha_hi, float(np.min(-h[neg] / d[neg])))
    alpha_lo = min(alpha_lo, 0.0)
    alpha_hi = max(alpha_hi, 0.0)
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, float(np.max(c0) * dx)
    n_grid = 80
    grid = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = c0[None, :] + grid[:, None] * L[None, :] + (grid[:, None]) ** 2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_alpha = grid[best_i]
    best_val = obj[best_i]
    span = max(1e-10, (alpha_hi - alpha_lo) / n_grid)
    fine = np.linspace(best_alpha - 6 * span, best_alpha + 6 * span, 300)
    fine = fine[(fine >= alpha_lo) & (fine <= alpha_hi)]
    if len(fine) > 1:
        vals2 = c0[None, :] + fine[:, None] * L[None, :] + (fine[:, None]) ** 2 * Q[None, :]
        obj2 = np.max(vals2, axis=1) * dx
        fi = int(np.argmin(obj2))
        if obj2[fi] < best_val:
            best_alpha = fine[fi]
            best_val = obj2[fi]
    val0 = float(np.max(c0) * dx)
    if best_val >= val0:
        return 0.0, val0
    return float(best_alpha), float(best_val)

def _soft_descent_phase(h, deadline, gamma_schedule, max_iter_per_gamma=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(99999)
    stall = 0
    for gamma in gamma_schedule:
        for _ in range(max_iter_per_gamma):
            if time.time() > deadline:
                return best_h, best_c5
            grad = _soft_gradient(h, gamma)
            grad = grad - grad.mean()
            norm = np.sum(np.abs(grad))
            if norm - 0 < 1e-14:
                break
            d = -grad
            alpha, new_c5 = _line_search(h, d)
            if alpha != 0.0 and new_c5 < cur_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                actual = _exact_c5(h_new)
                if actual < cur_c5 - 1e-15:
                    h = h_new
                    cur_c5 = actual
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                else:
                    break
            else:
                # Try random transport
                for _try in range(3):
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i == j:
                        continue
                    d_t = np.zeros(n)
                    d_t[i] = 1.0
                    d_t[j] = -1.0
                    alpha_t, new_c5_t = _line_search(h, d_t)
                    if alpha_t != 0.0 and new_c5_t < cur_c5 - 1e-13:
                        h_new = h + alpha_t * d_t
                        h_new = _project(h_new, target)
                        actual = _exact_c5(h_new)
                        if actual < cur_c5 - 1e-15:
                            h = h_new
                            cur_c5 = actual
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            stall = 0
                            break
                else:
                    stall += 1
                    if stall > 50:
                        # Perturb and reset
                        h = h + rng.normal(0, 0.02, n)
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        stall = 0
    return best_h, best_c5

def _transport_phase(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline:
        if stall > 500:
            break
        order = np.argsort(h)
        candidates = []
        for _ in range(15):
            i = order[rng.integers(n // 4, max(n // 4 + 1, 3 * n // 4))]
            j = order[rng.integers(0, max(1, n // 2))]
            if i != j and h[i] > h[j] + 0.01:
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                candidates.append(d)
        for _ in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                d = np.zeros(n)
                if h[i] >= h[j]:
                    d[i] = -1.0
                    d[j] = 1.0
                else:
                    d[i] = 1.0
                    d[j] = -1.0
                candidates.append(d)
        improved = False
        for d in candidates:
            if time.time() > deadline:
                break
            alpha, new_c5 = _line_search(h, d)
            if alpha != 0.0 and new_c5 < cur_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                actual = _exact_c5(h_new)
                if actual < cur_c5 - 1e-15:
                    h = h_new
                    cur_c5 = actual
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    improved = True
                    break
        if not improved:
            stall += 1
            if stall % 100 == 0:
                h = h + rng.normal(0, 0.01, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_init(n, rng):
    target = n // 2
    idx = rng.permutation(n)[:target]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def _sine_init(n, freq, phase, amplitude):
    x = np.arange(n, dtype=float)
    h = 0.5 + amplitude * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, block_len, rng):
    h = np.zeros(n)
    pos = 0
    target = n // 2
    assigned = 0
    while pos < n and assigned < target:
        blen = min(block_len, n - pos, target - assigned)
        if blen > 0:
            h[pos:pos + blen] = 1.0
            assigned += blen
        pos += blen
        gap = max(1, int(rng.integers(1, block_len + 1)))
        pos += gap
    return _project(h, n / 2.0)
