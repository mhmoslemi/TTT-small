import numpy as np

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.5, float(v.max()) + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads(h, active_idx):
    n = len(h)
    G = np.zeros((len(active_idx), n))
    for r, p in enumerate(active_idx):
        k = p - (n - 1)
        if k >= 0:
            m = n - k
            i1 = np.arange(k, n)
            G[r, i1] += (1.0 - h[i1 - k])
            i2 = np.arange(0, m)
            G[r, i2] -= h[i2 + k]
        else:
            r2 = -k
            m = n - r2
            i1 = np.arange(0, m)
            G[r, i1] += (1.0 - h[i1 + r2])
            i2 = np.arange(r2, n)
            G[r, i2] -= h[i2 - r2]
    return G

def exact_line_search(h, d, S0, rho_limit=None, dx=0.01639):
    n = len(h)
    dx = 2.0 / n
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    pos = d > 1e-14
    neg = d < -1e-14
    t_max = 1e6
    if pos.any():
        t_max = float(np.min((1.0 - h[pos]) / d[pos]))
    if rho_limit is not None:
        t_max = min(t_max, rho_limit)
    t_min = 0.0
    if neg.any():
        t_min = float(-np.min(h[neg] / d[neg]))
        t_min = max(t_min, 0.0)
    if t_max - t_min < 1e-14:
        return 0.0, c5_exact(h), h.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_min, t_max
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * D - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * D - c2 * c2 * Q)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * D - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * D - c2 * c2 * Q)) * dx
    t_opt = (a + b) / 2.0
    f_opt = float(np.max(S0 + t_opt * D - t_opt * t_opt * Q)) * dx
    f_lo = float(np.max(S0 + t_min * D - t_min * t_min * Q)) * dx
    f_hi = float(np.max(S0 + t_max * D - t_max * t_max * Q)) * dx
    if f_lo < f_opt:
        t_opt, f_opt = t_min, f_lo
    if f_hi < f_opt:
        t_opt, f_opt = t_max, f_hi
    h_new = h + t_opt * d
    h_new = project(h_new, n / 2.0)
    return t_opt, c5_exact(h_new), h_new

def lse_gradient_step(h, S0, temperature, rho_limit, deadline):
    n = len(h)
    dx = 2.0 / n
    scores = S0 * dx
    s_max = scores.max()
    log_w = (scores - s_max) / temperature
    log_w = np.clip(log_w, -40.0, 0.0)
    weights = np.exp(log_w)
    weights /= weights.sum()
    thresh_w = weights.max() * 0.005
    active_idx = np.where(weights >= thresh_w)[0]
    G = compute_grads(h, active_idx)
    w_a = weights[active_idx]
    w_a /= w_a.sum()
    d = -(G.T @ w_a)
    d -= d.mean()
    dn = np.linalg.norm(d)
    if dn < 1e-15:
        return h, c5_exact(h)
    d = d / dn
    t, c5_new, h_new = exact_line_search(h, d, S0, rho_limit, dx)
    return h_new, c5_new

def qp_direction(h, S0, dx, M=10.0):
    n = len(h)
    scores = S0 * dx
    s_max = scores.max()
    log_w = (scores - s_max) / (0.005 * s_max + 1e-12)
    log_w = np.clip(log_w, -40.0, 0.0)
    weights = np.exp(log_w)
    weights /= weights.sum()
    thresh_w = weights.max() * 0.01
    active_idx = np.where(weights >= thresh_w)[0]
    if len(active_idx) == 0:
        active_idx = np.argsort(-scores)[:10]
    G = compute_grads(h, active_idx)
    C = len(active_idx)
    x0 = np.concatenate([np.zeros(n), [0.0]])
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(0.0, 10.0)]
    
    def obj(x):
        d = x[:n]
        t = x[n]
        return 0.5 * np.dot(d, d) + M * t
    
    def obj_grad(x):
        d = x[:n]
        t = x[n]
        g = np.zeros(n + 1)
        g[:n] = d
        g[n] = M
        return g
    
    ineq_cons = []
    for i in range(C):
        def ineq(x, i=i):
            return -G[i, :] @ x[:n] - x[n]
        def ineq_jac(x, i=i):
            j = np.zeros(n + 1)
            j[:n] = -G[i, :]
            j[n] = -1.0
            return j
        ineq_cons.append({'type': 'ineq', 'fun': ineq, 'jac': ineq_jac})
    
    eq_cons = [{'type': 'eq',
                'fun': lambda x: np.sum(x[:n]),
                'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=ineq_cons + eq_cons,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            d = res.x[:n].copy()
            d -= d.mean()
            if np.linalg.norm(d) > 1e-12:
                return d
    except Exception:
        pass
    return None

def directed_pair_step(h, n_pairs, deadline):

    n = len(h)

    d[i] = -1.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def binary_local_search(h, deadline, max_iter=20000):
