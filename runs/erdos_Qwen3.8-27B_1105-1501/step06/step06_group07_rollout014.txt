    dx = 2.0 / n

    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible t range
    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.max(-h[neg] / d[neg])))
    
    if t_hi < 1e-14:

    # Shape: grid (n_grid,), S,T,Q (m,)

    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = float(obj[best_i])
    
    # Golden section refine
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    elif best_i == 0:
        a, b = 0.0, grid[1]
    else:
        a, b = grid[-2], t_hi
    
    def f_at(t):
        return float(np.max(S + t * T - t * t * Q)) * dx
    
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_at(c1)
    f2 = f_at(c2)

- Sort g in ascending order
- Set y[i] = 1 for the bottom n/2 (where g is smallest), y[i] = 0 for rest
