```python
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_AB(h):
    """Compute A[d] and B[d] arrays."""
    n = len(h)
    A = np.zeros(n)
    B = np.zeros(n)
    for d in range(1, n):
        m = n - d
        A[d] = float(np.dot(h[:m], 1.0 - h[d:]))
        B[d] = float(np.dot(h[d:], 1.0 - h[:m]))
    A[0] = float(np.dot(h, 1.0 - h))
```

Actually, let me think about a simpler continuous optimization: **subgradient method with projection**.

The objective is f(h) = max_k c_k(h) * dx where c_k are the correlation values.
A subgradient of f at h is: g = dx * ∇c_k*(h) where k* is an argmax.

Or more smoothly: g = dx * Σ_k p_k * ∇c_k where p_k is the softmax over the top c_k values.

The update is: h_new = project(h - step * g, target)

This is simple to implement. Let me code it:

```python
def _subgrad_optimize(h, deadline, step_init=0.01, beta=50.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    
    while time.time() < deadline:
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
        c_vals = corr * dx  # these are the "costs" for each lag
        
        # Softmax weights
        c_max = c_vals.max()
        exp_c = np.exp(beta * (c_vals - c_max))
        p = exp_c / exp_c.sum()  # weights for each lag
        
        # Compute gradient: g[i] = dx * sum_r p[r] * ∂corr[r]/∂h[i]
        # ∂corr[r]/∂h[i] depends on the convention. Let me compute it directly.
        #
        # For the full correlation corr = np.correlate(h, 1-h, 'full'):
        # I'll compute the gradient by noting that corr[r] is a function of h.
        # Each corr[r] = sum of products involving h and (1-h).
        # ∂corr[r]/∂h[i] = sum of terms where h[i] appears.
        #
        # This is complex. Let me use a different approach: compute the gradient
        # of c_k(h) directly.
        #
        # c_k(h) * dx = corr[k] * dx
        # corr[k] = (h ⊛ (1-h))[k] where ⊛ is cross-correlation
        # 
        # For the cross-correlation, the gradient w.r.t. h[i] is:
        # ∂corr[k]/∂h[i] = ∂/∂h[i] sum_m h[m] * (1-h[m-k']) for valid m
        # where k' is determined by k and the convention.
        
        # Let me just compute the gradient numerically (sparse, O(n) per coordinate)
        # Actually, let me compute it analytically.
        
        # Define: for each lag d (d = -(n-1), ..., 0, ..., n-1):
        # corr[n-1+d] involves pairs (i, i+d) or (i+d, i)
        # 
        # Let me define it as: corr_val[d] for d = 0, ..., n-1 covers both A[d] and B[d]
        # and the full correlation has both A and B values.
        
        # I'll use the A[d], B[d] formulation:
        # A[d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d]) for d >= 0
        # B[d] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i]) for d >= 0
        # A[0] = sum h[i]*(1-h[i])
        
        # The full correlation array has 2n-1 elements. I need to map from
        # the full array index to A[d] or B[d].
        
        # Let me just compute A and B directly and figure out the mapping.
        
        # Actually, for the gradient, I don't need the mapping. I can compute:
        # g[i] = dx * (sum_d p_A[d] * ∂A[d]/∂h[i] + sum_d p_B[d] * ∂B[d]/∂h[i])
        # where p_A and p_B are the weights (from the softmax) for each A[d] and B[d].
        
        # But I need to know which lag in the full correlation corresponds to which A[d] or B[d].
```
