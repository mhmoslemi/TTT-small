import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_grad(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        if s >= 0:
            m = n - s
            G[k, s:] += (1.0 - h[:m])
            G[k, :m] += (-h[s:])
        else:
            g = -s
            m = n - g
            G[k, :m] += (1.0 - h[g:])
            G[k, g:] += (-h[:m])
    G *= dx
    return G

def _quad_term(d):
    n = len(d)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            c[k] = -np.sum(d[s:s + m] * d[:m])
        else:
            g = -s
            c[k] = -np.sum(d[:m] * d[g:g + m])
    c *= dx
    return c

def _linear_term(h, d):
    n = len(h)
    dx = 2.0 / n
    c = np.zeros(2 * n - 1)
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            c[k] = np.sum(d[:m] * (1.0 - h[s:s + m])) - np.sum(h[:m] * d[s:s + m])
        else:
            g = -s
            c[k] = np.sum(d[g:g + m] * (1.0 - h[:m])) - np.sum(h[g:g + m] * d[:m])
    c *= dx
    return c

def _line_search(h, d, corr_base, radius):
    n = len(h)
    dx = 2.0 / n
    lin = _linear_term(h, d)
    quad = _quad_term(d)
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -radius
    t_hi = radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 80
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = corr_base[None, :] + grid[:, None] * lin[None, :] + grid[:, None] ** 2 * quad[None, :]
    obj = np.max(vals, axis=1)
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if 0 < best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    f1 = float(np.max(corr_base + a * lin + a * a * quad))
    f2 = float(np.max(corr_base + b * lin + b * b * quad))
    for _ in range(50):
        if b - a < 1e-13:
            break
        m1 = a + gr * (b - a)
        m2 = b - gr * (b - a)
        fm1 = float(np.max(corr_base + m1 * lin + m1 * m1 * quad))
        fm2 = float(np.max(corr_base + m2 * lin + m2 * m2 * quad))
        if fm1 < fm2:
            b, f2 = m2, fm2
        else:
            a, f1 = m1, fm1
    mid = (a + b) / 2.0
    mid_val = float(np.max(corr_base + mid * lin + mid * mid * quad))
    if mid_val < best_val:
        best_t, best_val = mid, mid_val
    return best_t, best_val

def _null_space_dir(active_grads, n, rng):
    A = np.vstack([active_grads, np.ones(n)])
    try:
        _, _, Vt = np.linalg.svd(A, full_matrices=True)
        null_dim = Vt.shape[0] - A.shape[0]
        if null_dim <= 0:
            return None
        d = Vt[A.shape[0]:].T
        idx = rng.integers(0, null_dim)
        return d[:, idx].copy()
    except Exception:
        return None

def _continuous_descent(h_init, deadline, max_iter=500):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    corr = _corr(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.15
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:

                radius = max(0.03, radius * 0.5)

                corr = _corr(h)
                radius = 0.1
                stall = 0
                continue
            if stall > 100:
                break

        M = cur_c5
        tol = max(1e-5, M * 0.01)
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:10]
        elif len(active) > 30:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:30]]

        directions = []
        G_act = _corr_grad(h)[active]
        grad_sum = G_act.sum(axis=0)
        grad_sum -= grad_sum.mean()
        gn = np.linalg.norm(grad_sum)
        if gn > 1e-12:
            directions.append(-grad_sum / gn)

        d_null = _null_space_dir(G_act, n, rng)
        if d_null is not None:
            d_null -= d_null.mean()
            dn = np.linalg.norm(d_null)
            if dn > 1e-12:
                directions.append(d_null / dn)

        directions.append(rng.normal(0, 1, n))
        directions[-1] -= directions[-1].mean()
        dn = np.linalg.norm(directions[-1])
        if dn > 1e-12:
            directions[-1] /= dn

        for d in directions:
            if time.time() > deadline:
                break
            alpha, new_c5 = _line_search(h, d, corr, radius)
            if new_c5 < cur_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                corr = _corr(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.3)
                break
        else:
            stall += 1
            radius *= 0.7

    return best_h, best_c5

def _slsqp_polish(h0, deadline, max_outer=10):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _corr(h)
        K = min(20, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        t_init = cur_c5 + 5e-5
        x0 = np.concatenate([h, [t_init]])

        def obj_fn(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x):
            hx = x[:n]
            c = _corr(hx)
            return x[n] - c[top_k]

        def ineq_jac(x):
            hx = x[:n]
            G = _corr_grad(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J

        def eq_fn(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-12})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_01(h):
    n = len(h)
    target = int(n / 2)
    h_01 = np.zeros(n)
    top = np.argsort(-h)[:target]
    h_01[top] = 1.0
    h_01 = h_01
    if float(np.sum(h_01)) != float(target):

    return h_01
