import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    """Compute gradient matrix G where G[r,i] = dC_r/dh_i (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d == 0:
            G[r] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            L = n - d
            G[r, d:] += dx * (1.0 - h[:L])
            G[r, :L] += -dx * h[d:]
        else:
            g = -d
            L = n - g
            G[r, :L] += dx * (1.0 - h[g:])
            G[r, g:] += -dx * h[:L]
    return G

def _lse_objective_and_grad(h, beta):
    n = len(h)
    C = _corr_scaled(h)
    max_C = np.max(C)
    z = beta * (C - max_C)
    exp_z = np.exp(z)
    sum_exp = np.sum(exp_z)
    p = exp_z / sum_exp
    Q = max_C + np.log(sum_exp) / beta
    G = _grad_matrix(h)
    grad = G.T @ p
    return Q, grad

def _slsqp_lse(h, beta, deadline, maxiter=200):
    """Run SLSQP on LSE objective from starting point h."""
    n = len(h)
    target = n / 2.0
    h0 = h.copy()
    
    def fun(x):
        h_x = _project(x, target)
        Q, _ = _lse_objective_and_grad(h_x, beta)
        return Q
    
    def jac(x):
        h_x = _project(x, target)
        _, grad = _lse_objective_and_grad(h_x, beta)
        return grad
    
    def eq_constraint(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    x0 = h0.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac}]
    
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)
    
    try:
        res = minimize(fun, x0, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except Exception:
        pass
    
    return best_h, best_c5

    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)


def _grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d == 0:
            G[r] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            L = n - d
            if L > 0:
                G[r, d:] += dx * (1.0 - h[:L])
                G[r, :L] += -dx * h[d:]
        else:
            g = -d
            L = n - g
            if L > 0:
                G[r, :L] += dx * (1.0 - h[g:])
                G[r, g:] += -dx * h[:L]
    return G


class LSEProblem:
    def __init__(self, beta, n, target):
        self.beta = beta
        self.n = n
        self.target = target
        self.best_h = None
        self.best_c5 = np.inf

    def _eval(self, x):
        h = np.clip(x, 0.0, 1.0)
        s = float(np.sum(h))
        if abs(s - self.target) > 0.5:
            h = _project(h, self.target)
        C = _corr_scaled(h)
        max_C = float(np.max(C))
        z = self.beta * (C - max_C)
        exp_z = np.exp(z)
        sum_exp = float(np.sum(exp_z))
        p = exp_z / sum_exp
        Q = max_C + np.log(sum_exp) / self.beta
        G = _grad_matrix(h)
        grad = G.T @ p
        c5 = max_C
        if c5 < self.best_c5:
            self.best_c5 = c5
            self.best_h = h.copy()
        return Q, grad

    def fun(self, x):
        Q, _ = self._eval(x)
        return Q

    def jac(self, x):
        _, grad = self._eval(x)
        return grad

    def eq_fun(self, x):
        return float(np.sum(x)) - self.target

    def eq_jac(self, x):
        return np.ones(self.n)


class MinimaxReduced:
    def __init__(self, top_k, n, target):
        self.top_k = top_k
        self.n = n
        self.target = target
        self.best_h = None
        self.best_c5 = np.inf

    def fun(self, y):
        M = y[0]
        h = np.clip(y[1:], 0.0, 1.0)
        C = _corr_scaled(h)
        c5 = float(np.max(C))
        if c5 < self.best_c5:
            self.best_c5 = c5
            self.best_h = h.copy()
        return M

    def jac(self, y):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g

    def ineq_fun(self, y):
        h = np.clip(y[1:], 0.0, 1.0)
        C = _corr_scaled(h)
        sorted_idx = np.argsort(-C)
        top = sorted_idx[:self.top_k]
        return y[0] - C[top]

    def ineq_jac(self, y):
        h = np.clip(y[1:], 0.0, 1.0)
        C = _corr_scaled(h)
        sorted_idx = np.argsort(-C)
        top = sorted_idx[:self.top_k]
        G = _grad_matrix(h)
        Gt = G[top]
        J = np.zeros((self.top_k, self.n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -Gt
        return J

    def eq_fun(self, y):
        return float(np.sum(y[1:])) - self.target

    def eq_jac(self, y):
        g = np.zeros(self.n + 1)
        g[1:] = 1.0
        return g


def _slsqp_lse(h, beta, target, deadline, maxiter=150):
    n = len(h)
    prob = LSEProblem(beta, n, target)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}]
    x0 = h.copy()
    try:
        res = minimize(prob.fun, x0, jac=prob.jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        res = None
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    if res is not None and res.x is not None:
        h_new = _project(res.x, target)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    if prob.best_h is not None and prob.best_c5 < best_c5:
        best_h = prob.best_h
        best_c5 = prob.best_c5
    return best_h, best_c5


def _slsqp_minimax_reduced(h, top_k, target, deadline, maxiter=100):
    n = len(h)
    C = _corr_scaled(h)
    M0 = float(np.max(C)) + 0.01
    prob = MinimaxReduced(top_k, n, target)
    x0 = np.zeros(n + 1)
    x0[0] = M0
    x0[1:] = h
    bounds = [(0.0, M0 + 0.1)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
    ]
    try:
        res = minimize(prob.fun, x0, jac=prob.jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        res = None
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    if res is not None and res.x is not None:
        h_new = _project(res.x[1:], target)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    if prob.best_h is not None and prob.best_c5 < best_c5:
        best_h = prob.best_h
        best_c5 = prob.best_c5
    return best_h, best_c5


def _line_search(h, d):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = 0.0
    t_hi = 0.0
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    R0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    n_grid = 40
    ts = np.linspace(t_lo, t_hi, n_grid)
    best_val = np.inf
    best_t = 0.0
    for t in ts:
        vals = R0 + t * T - t * t * Q
        v = float(np.max(vals) * dx)
        if v < best_val:
            best_val = v
            best_t = t
    # Golden section around best_t
    span = (t_hi - t_lo) / n_grid
    a = max(t_lo, best_t - 2 * span)
    b = min(t_hi, best_t + 2 * span)
    if b - a < 1e-14:
        return best_t, best_val
    g = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max((R0 + c1 * T - c1 * c1 * Q)) * dx)
    f2 = float(np.max((R0 + c2 * T - c2 * c2 * Q)) * dx)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max((R0 + c1 * T - c1 * c1 * Q)) * dx)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max((R0 + c2 * T - c2 * c2 * Q)) * dx)
    opt_t = (a + b) / 2.0
    opt_val = float(np.max((R0 + opt_t * T - opt_t * opt_t * Q)) * dx)
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _gen_topk_dir(h, scores, k, target):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    top = sorted_idx[:min(k, m)]
    G = _grad_matrix(h)
    avg_grad = G[top].mean(axis=0)
    avg_grad -= avg_grad.mean()
    norm = np.linalg.norm(avg_grad)
    if norm < 1e-14:
        return None
    d = -avg_grad / norm
    return d


def _gen_single_dir(h, scores, r, target):
    n = len(h)
    G = _grad_matrix(h)
    g = G[r].copy()
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm


def _gen_pair_dir(h, i, j, target):
    n = len(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    return d


def _gen_random_dir(h, rng, k, target):
    n = len(h)
    idx = rng.choice(n, size=k, replace=False)
    d = np.zeros(n)
    half = k // 2
    d[idx[:half]] = -1.0
    d[idx[half:]] = 1.0
    if k % 2 == 1:
        d[idx[-1]] = 0.0
    return d


def _gen_gaussian_dir(h, rng, target):
    n = len(h)
    d = rng.normal(0, 1, n)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm


def _direction_feasible(h, d, max_step=1.0):
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        max_pos = float(np.min((1.0 - h[pos]) / d[pos]))
    else:
        max_pos = max_step
    if np.any(neg):
        max_neg = float(np.min(h[neg] / (-d[neg])))
    else:
        max_neg = max_step
    return min(max_pos, max_neg, max_step)


def _line_polish(h, target, deadline, rng, max_iters=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    stall = 0
    it = 0
    while time.time() < deadline and stall < 200:
        it += 1
        scores = _corr_scaled(best_h)
        max_score = float(np.max(scores))
        dirs = []
        # Top-K averaged directions
        for k in [5, 10, 20, 30]:
            d = _gen_topk_dir(best_h, scores, k, target)
            if d is not None:
                dirs.append(d)
        # Individual top constraint directions
        sorted_idx = np.argsort(-scores)
        for r in sorted_idx[:5]:
            d = _gen_single_dir(best_h, scores, r, target)
            if d is not None:
                dirs.append(d)
        # Pair moves on interior points
        interior = np.where((best_h > 0.05) & (best_h < 0.95))[0]
        if len(interior) >= 2:
            for _ in range(8):
                i = interior[rng.integers(len(interior))]
                j = interior[rng.integers(len(interior))]
                if i != j:
                    dirs.append(_gen_pair_dir(best_h, i, j, target))
        # Random sparse
        for k in [4, 8, 16]:
            d = _gen_random_dir(best_h, rng, k, target)
            dirs.append(d)
        # Gaussian
        d = _gen_gaussian_dir(best_h, rng, target)
        if d is not None:
            dirs.append(d)

        improved = False
        for d in dirs:
            if time.time() >= deadline:
                break
            max_step = _direction_feasible(best_h, d, 1.0)
            if max_step < 1e-12:
                continue
            d = d * max_step
            alpha, new_c5 = _line_search(best_h, d)
            if alpha != 0.0 and new_c5 < best_c5 - 1e-14:
                h_new = best_h + alpha * d
                h_new = _project(h_new, target)
                actual_c5 = _compute_c5(h_new)
                if actual_c5 < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = actual_c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            if stall % 50 == 0:
                best_h = best_h + rng.normal(0, 0.02, n)
                best_h = _project(best_h, target)
                best_c5 = _compute_c5(best_h)
                stall = 0
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)


def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    pattern = np.zeros(period)
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)


def _run_lse_sequence(h, target, deadline, betas=None):
    if betas is None:
        betas = [10, 30, 80, 200, 500]
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    for beta in betas:
        if time.time() >= deadline:
            break
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (len(betas)))
        h_new, c5_new = _slsqp_lse(best_h, beta, target, sub_dl, maxiter=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5


def _run_minimax_sequence(h, target, deadline, top_ks=None):
    if top_ks is None:
        top_ks = [10, 20, 40]
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    for k in top_ks:
        if time.time() >= deadline:
            break
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (len(top_ks)))
        h_new, c5_new = _slsqp_minimax_reduced(best_h, k, target, sub_dl, maxiter=80)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    reserve = max(5.0, 0.02 * budget_s)
    hard_deadline = t0 + budget_s - reserve

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except (NameError, AttributeError):
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)

    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = _project(h0, target0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: LSE-SLSQP sequence (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _run_lse_sequence(best_h, target0, d1, betas=[10, 40, 100, 300])
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Reduced minimax (10% time)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _run_minimax_sequence(best_h, target0, d2, top_ks=[15, 30, 50])
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Line polish (15% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _line_polish(best_h, target0, d3, rng, max_iters=8000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-start (15% time)
    d4 = t0 + budget_s * 0.60
    n = len(best_h)
    starts = []
    for freq in [1, 2, 3, 4, 5, 7]:
        for ph in [0.0, 0.7, 1.5, 2.5]:
            starts.append(_sine_init(n, freq, ph))
    for p in [4, 5, 6, 7, 8, 10, 12, 15]:
        for duty in [0.35, 0.5, 0.65]:
            starts.append(_periodic_init(n, p, duty))
    for _ in range(5):
        vals = rng.random(n)
        starts.append(_project(vals, n / 2.0))
    starts.append(best_h.copy())

    for s in starts:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.010)
        h_s, c5_s = _run_lse_sequence(s, n / 2.0, sub_dl, betas=[30, 100])
        if c5_s < best_c5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_s, c5_s = _line_polish(h_s, n / 2.0, sub_dl2, rng, max_iters=2000)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 5: Further LSE from best (10% time)
    d5 = t0 + budget_s * 0.70
    h5, c5_5 = _run_lse_sequence(best_h, target0, d5, betas=[50, 150, 400, 800])
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (12% time)
    d6 = t0 + budget_s * 0.82
    n_best = len(best_h)
    offsets = [-4, -2, 2, 4, -6, 6, -8, 8, -12, 12, -16, 16]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 300:
            continue
        if n_try % 2 != 0:
            n_try += 1
        if n_try in tried or n_try < 40:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        target_r = n_try / 2.0
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _run_lse_sequence(h_r, target_r, sub_dl, betas=[30, 100, 300])
        if c5_r < best_c5 - 1e-12:
            sub_dl2 = min(d6, time.time() + budget_s * 0.006)
            h_r, c5_r = _line_polish(h_r, target_r, sub_dl2, rng, max_iters=2000)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_best = n_try

    # Phase 7: Aggressive line polish (10% time)
    d7 = t0 + budget_s * 0.92
    h7, c5_7 = _line_polish(best_h, len(best_h) / 2.0, d7, rng, max_iters=10000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final SLSQP polish (5% time)
    h8, c5_8 = _slsqp_lse(best_h, 500, len(best_h) / 2.0, hard_deadline, maxiter=80)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    n_final = len(best_h)
    best_h = _project(best_h, n_final / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0_fb = np.array(initial_h_values, dtype=float)
            h0_fb = _project(h0_fb, len(h0_fb) / 2.0)
            return h0_fb, _compute_c5(h0_fb), len(h0_fb)
        except (NameError, AttributeError):
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
