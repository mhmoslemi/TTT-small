def _line_search(h, d):

    S0 = np.correlate(h, 1.0 - h, mode='full')

def _gen_direction_lse(h, beta, top_k=20):
    n = len(h)
    _, grad, p = _lse_eval(h, beta)
    grad -= grad.mean()  # Project to sum=0
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    d = -grad / norm
    # Also try top-k active gradient
    active = np.argsort(-p)[-top_k:]
    p_top = p[active].copy()
    p_top /= p_top.sum()
    G = _grad_all(h)
    g_top = G[active].T @ p_top
    g_top -= g_top.mean()
    norm2 = np.linalg.norm(g_top)
    if norm2 > 1e-14:
        d_top = -g_top / norm2
        return d_top  # Return the more focused one
    return d

def _gen_direction_avg_active(h, frac=0.1):

    scores = _corr_scaled(h)
    max_s = scores.max()
    thr = max_s - frac * (max_s - scores.min())
    active = np.where(scores >= thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    G = _grad_all(h)
    d = G[active].mean(axis=0)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return -d / norm

def _line_polish(h, deadline, n_stages=50, beta=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    rng = np.random.default_rng(12345)
    stage = 0
    while time.time() < deadline and stage < n_stages:
        stage += 1
        improved = False
        # Try LSE gradient direction
        d = _gen_direction_lse(h, beta)
        if d is not None:
            d = d * 0.5  # Scale
            t, val = _line_search(h, d)
            if val < best_c5 - 1e-14 and t > 1e-10:
                h_new = h + t * d
                h_new = _project(h_new, target)
                c5_new = _c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
        # Try active average direction
        if not improved:
            d = _gen_direction_avg_active(h, 0.05)
            if d is not None:
                for scale in [0.3, 0.5, 0.8, 1.0]:
                    t, val = _line_search(h, d * scale)
                    if val < best_c5 - 1e-14 and t > 1e-10:
                        h_new = _project(h + t * d * scale, target)
                        c5_new = _c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new
                            improved = True
                            break
        # Try random direction
        if not improved:
            d = rng.normal(0, 1, n)
            d -= d.mean()
            norm = np.linalg.norm(d)
            if norm > 0:
                d /= norm
                t, val = _line_search(h, d * 0.3)
                if val < best_c5 - 1e-14 and t > 1e-10:
                    h_new = _project(h + t * d * 0.3, target)
                    c5_new = _c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new
                        improved = True
        # Try pair move
        if not improved:
            interior = np.where((h > 0.05) & (h < 0.95))[0]
            if len(interior) >= 2:
                i = interior[rng.integers(len(interior))]
                j = interior[rng.integers(len(interior))]
                if i != j:
                    for sign in [1.0, -1.0]:
                        d = np.zeros(n)
                        d[i] = -sign
                        d[j] = sign
                        t, val = _line_search(h, d)
                        if val < best_c5 - 1e-14 and abs(t) > 1e-10:
                            h_new = _project(h + t * d, target)
                            c5_new = _c5(h_new)
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h_new.copy()
                                h = h_new
                                improved = True
                                break
        if not improved:
            # Small perturbation to escape
            h = _project(h + rng.normal(0, 0.01, n), target)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, max_iters=5000):
    n = len(h_init)
    target = n // 2
    rng = np.random.default_rng(42)
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.003
    T_min = 0.00005
    cooling = 0.9998
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])
    patience = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 5000 or len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _c5(h)
        delta = new_c5 - cur_c5
        if delta < -1e-15 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
