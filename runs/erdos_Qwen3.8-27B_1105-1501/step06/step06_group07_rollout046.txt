import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection onto {0 <= v <= 1, sum(v) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += (1.0 - h[:m])
        g[:m] -= h[k:]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        g[:m] += (1.0 - h[ga:])
        g[ga:] -= h[:m]
    return g * dx

def _line_search_quadratic(h, d):
    """Exact line search along direction d using quadratic model.
    Returns (best_t, best_val)."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find valid t range from box constraints
    pos = d > 1e-12
    neg = d < -1e-12
    t_hi = 0.5
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min((-h[neg]) / d[neg])))
    t_hi = min(t_hi, 1.0)
    if t_hi < 1e-14:
        return 0.0, float(np.max(S) * dx)
    
    # Grid search
    n_grid = 80
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = np.max(S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :], axis=1) * dx
    best_i = int(np.argmin(vals))
    best_t = grid[best_i]
    best_val = vals[best_i]
    
    # Golden section refine
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _smooth_max_grad(h, alpha, K):
    """Compute weighted gradient of top K correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(K) / K
    else:
        w /= w_sum
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _grad_row(h, k)
    return grad

def _lp_subproblem(h, n_active, deadline):
    """Solve LP subproblem that linearizes active constraints."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr_values = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(corr_values))
    
    # Select active constraints
    top_count = min(n_active, n + (n - 1))  # total lags = 2n-1
    top_idx = np.argsort(-corr_values)[:top_count]
    
    # Build LP: minimize t
    # Variables: h[0], ..., h[n-1], t  -> n+1 variables
    # Constraints:
    #   For each active lag k: grad_k . h <= t + b_k  (linearized)
    #   sum(h) = n/2
    #   0 <= h[i] <= 1
    #   t is free (or >= 0)
    
    # Linearized constraint for lag k:
    # c_k(actual) + grad_k . (h - h_current) <= t
    # => grad_k . h <= t + c_k - grad_k . h_current
    # => grad_k . h - t <= c_k - grad_k . h_current
    
    # In standard form for linprog (minimize c'x s.t. A_ub x <= b_ub):
    # grad_k . h - t <= c_k - grad_k . h_current
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    
    for idx in top_idx:
        k = int(idx) - (n - 1)
        g = _grad_row(h, k)  # this includes dx factor
        # c_k = corr_values[idx]
        g_dot_h = float(np.dot(g, h))
        b = corr_values[idx] - g_dot_h
        # Constraint: g . h - t <= b
        row = np.concatenate([g, [-1.0]])
        A_ub.append(row)
        b_ub.append(b)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_new = res.x[:n]
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    
    return None, None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _directed_random_dir(h, rng, K=20, n_mixed=5):
    """Generate a random direction biased toward reducing top correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_idx = np.argsort(-corr)[:K]
    
    # Random subset of active gradients
    subset_size = max(2, n_mixed + rng.integers(0, 4))
    subset = rng.choice(len(top_idx), size=min(subset_size, len(top_idx)), replace=False)
    
    grad = np.zeros(n)
    for s in subset:
        k = int(top_idx[s]) - (n - 1)
        grad += rng.uniform(0.5, 2.0) * _grad_row(h, k)
    
    # Add random component
    grad += rng.normal(0, 0.1, n)
    
    # Project to sum-zero
    grad -= np.mean(grad)
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return None
    return -grad / gn  # negative gradient direction
