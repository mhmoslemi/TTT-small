```python
def _lse_grad(h, beta):
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    S_max = S.max()
    exp_S = np.exp(beta * (S - S_max))
    w = exp_S / exp_S.sum()
    
    grad = np.zeros(n)
    for c in range(m):
        wc = w[c]
        if wc < 1e-16:
            continue
        lag = c - (n - 1)
        if lag >= 0:
            d = lag
            if d < n:
                nd = n - d
                grad[:nd] += wc * (1.0 - h[d:])
                grad[d:] += wc * (-h[:nd])
        else:
            g = -lag
            ng = n - g
            if g < n:
                grad[g:] += wc * (1.0 - h[:ng])
                grad[:ng] += wc * (-h[g:])
    
    return grad
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full') * dx
    S_max = float(S.max())
    exp_S = np.exp(beta * (S - S_max))
    exp_sum = float(exp_S.sum())
    w = exp_S / exp_sum
    obj = S_max + np.log(exp_sum) / beta
    m = 2 * n - 1
    grad = np.zeros(n)
    for c in range(m):
        wc = w[c]
        if wc < 1e-16:
            continue
        lag = c - (n - 1)
        if lag >= 0:
            d = lag
            if d < n:
                nd = n - d
                grad[:nd] += wc * (1.0 - h[d:])
                grad[d:] += wc * (-h[:nd])
        else:
            g = -lag
            if g < n:
                ng = n - g
                grad[g:] += wc * (1.0 - h[:ng])
                grad[:ng] += wc * (-h[g:])
    return float(obj), grad

def _lse_anneal(h0, deadline, betas=None, lr_start=0.06, momentum=0.85):

def _slsqp_polish(h0, beta, deadline, maxiter=100):
    n = len(h0)
    target = n / 2.0
    if time.time() > deadline:
        return h0, _exact_c5(h0)
    def obj(x):
        o, _ = _lse_obj_grad(x, beta)
        return o
    def obj_grad(x):
        _, g = _lse_obj_grad(x, beta)
        return g
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success:
            h_res = _project(res.x, target)
            return h_res, _exact_c5(h_res)
    except Exception:
        pass
    return h0, _exact_c5(h0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
```
