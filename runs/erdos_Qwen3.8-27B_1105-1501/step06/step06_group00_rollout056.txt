def evaluate_swap(h, corr, i, j):
    # h[i] = 1, h[j] = 0
    # Compute corr + delta for all shifts
    n = len(h)
    delta = np.zeros(2*n-1)
    
    # For each shift, compute the change
    for lag in range(-(n-1), n):
        s = lag + (n-1)  # shift index
        d = 0
        # i as source (r=i): change = -(1-h[i+lag]) if valid
        if 0 <= i + lag < n:
            d -= (1 - h[i+lag])
        # i as target (r+lag=i, r=i-lag): change = +h[i-lag] if valid
        if 0 <= i - lag < n:
            d += h[i-lag]
        # j as source (r=j): change = +(1-h[j+lag]) if valid
        if 0 <= j + lag < n:
            d += (1 - h[j+lag])
        # j as target (r+lag=j, r=j-lag): change = -h[j-lag] if valid
        if 0 <= j - lag < n:
            d -= h[j-lag]
        delta[s] = d
    
    new_corr = corr + delta
    return np.max(new_corr)
