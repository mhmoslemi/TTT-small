def _exact_c5(h):

def _corr_scaled(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):

def _smooth_direction(h, S, n, tau):
    M = float(S.max())
    shifted = (S - M) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_k = min(25, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad_smooth = w[top_idx] @ G
    grad_smooth -= grad_smooth.mean()
    if np.abs(grad_smooth).max() < 1e-12:
        return None
    norm = np.linalg.norm(grad_smooth)
    if norm < 1e-12:
        return None
    return -grad_smooth / norm

    T_min = 0.000003
    cooldown = 0.99995
    patience = 0
    max_patience = 20000
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 2:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        for _ in range(300):
            if time.time() > deadline:
                break
            move_type = rng.integers(0, 3)
            if move_type == 0:
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
                if len(ones_list) == 0 or len(zeros_list) == 0:
                    continue
                i = ones_list[rng.integers(len(ones_list))]
                j = zeros_list[rng.integers(len(zeros_list))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = _exact_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            elif move_type == 1:
                block_size = min(rng.integers(2, 8), target)
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
                if len(ones_list) < block_size or len(zeros_list) < block_size:
                    continue
                i_start = ones_list[rng.integers(len(ones_list))]
                pos = np.searchsorted(np.sort(ones_list), i_start)
                if pos + block_size > len(ones_list):
                    pos = len(ones_list) - block_size
                block_ones = np.sort(ones_list)[pos:pos + block_size]
                j_start = zeros_list[rng.integers(len(zeros_list))]
                pos2 = np.searchsorted(np.sort(zeros_list), j_start)
                if pos2 + block_size > len(zeros_list):
                    pos2 = len(zeros_list) - block_size
                block_zeros = np.sort(zeros_list)[pos2:pos2 + block_size]
                h[block_ones] = 0.0
                h[block_zeros] = 1.0
                new_c5 = _exact_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                else:
                    h[block_ones] = 1.0
                    h[block_zeros] = 0.0
            else:
                w = rng.integers(1, n // 3)
                if w * 2 >= n:
                    continue
                shift = rng.integers(1, n - w)
                h_old = h.copy()
                seg = h[shift:shift + w].copy()
                h[shift:shift + w] = h[(shift + w) % n:]
                h[:w] = seg
                h_new = h
                if abs(float(np.sum(h_new)) - target) > 1e-6:
                    h = h_old
                    continue
                new_c5 = _exact_c5(h_new)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                else:
                    h[:] = h_old
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=150):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _cosine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _two_harmonic_init(n, f1, f2, phase1, phase2, amp2):
    x = np.arange(n)
    h = 0.5 + 0.5 * (np.sin(2 * np.pi * f1 * x / n + phase1) +
                     amp2 * np.sin(2 * np.pi * f2 * x / n + phase2))
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    s = float(np.sum(h))
    if s == 0:
        h[:n // 2] = 1.0
    return _project(h, n / 2.0)

def _contiguous_init(n, offset):
    h = np.zeros(n)
    half = n // 2
    start = offset % n
    for i in range(half):
        h[(start + i) % n] = 1.0
    return h

def _fourier_optimize(h, deadline, n_coeffs=8):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    x = np.arange(n, dtype=float)
    rng = np.random.default_rng(123)
    for _iter in range(2000):
        if time.time() > deadline:
            break
        S = _corr_scaled(h)
        M = S.max()
        active = np.argsort(-S)[:12]
        G = _grad_rows(h, active, n)
        grad = S[active] @ G
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        grad = grad / norm
        d = -grad * 0.1
        alpha, new_c5 = _line_search(h, d, 2.0 / n, 0.5)
        if new_c5 < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Diverse multi-start (10% time)
    d1 = t0 + budget_s * 0.10
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 9]:
        for ph in [0.0, 0.7, 1.5, 2.3, 3.1]:
            starts.append(_sine_init(n0, freq, ph))
            starts.append(_cosine_init(n0, freq, ph + 0.4))
    for f1 in [2, 3, 5]:
        for f2 in [4, 7, 11]:
            for a2 in [0.3, 0.5, 0.7]:
                starts.append(_two_harmonic_init(n0, f1, f2, 0.5, 1.2, a2))
    for _ in range(12):
        starts.append(_block_init(n0, rng, rng.integers(3, 18)))
    for off in [0, n0 // 4, n0 // 3, n0 // 2]:
        starts.append(_contiguous_init(n0, off))
    starts = starts[:60]
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main LP optimization (22% time)
    d2 = t0 + budget_s * 0.32
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.14, K=18, use_smooth=False)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP with balanced constraints (12% time)
    d3 = t0 + budget_s * 0.44
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.10, K=22, use_smooth=True, use_balance=True)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: SA with block moves + LP polish (15% time)
    d4 = t0 + budget_s * 0.59
    h4a, c5_4a = _sa_block_moves(best_h, t0 + budget_s * 0.52, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, radius_init=0.07, K=18, use_smooth=True)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-resolution (15% time)
    d5 = t0 + budget_s * 0.74
    base_n = len(best_h)
    tried = set()
    n_candidates = list(range(max(60, base_n - 30), min(220, base_n + 50), 2))
    rng.shuffle(n_candidates)
    n_candidates.sort(key=lambda x: abs(x - base_n))
    for n_try in n_candidates:
        if time.time() > d5:
            break
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d5, time.time() + budget_s * 0.018)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=15)
            if c5_r < best_c5 and time.time() < d5 - budget_s * 0.02:
                sub_dl2 = min(d5, time.time() + budget_s * 0.014)
                h_r2, c5_r2 = _sa_block_moves(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d5:
                    h_r, c5_r = _lp_loop(h_r, min(d5, time.time() + budget_s * 0.01),
                                         radius_init=0.05, K=14)
        if c5_r < best_c5 - 1e-12:

            best_h = h_r.copy()

        if best_c5 < 0.3804:

    d7 = t0 + budget_s * 0.89
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=200)
    if c5_7 < best_c5:

    # Phase 8: Fourier-guided optimization (4% time)
    d8 = t0 + budget_s * 0.93
    h8, c5_8 = _fourier_optimize(best_h, d8)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Phase 9: Final LP + pair polish
    d9 = t0 + budget_s * 0.96
    h9, c5_9 = _lp_loop(best_h, d9, radius_init=0.04, K=15, use_smooth=True)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    if time.time() < deadline:
        h10, c5_10 = _pair_refine(best_h, deadline, max_passes=80)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
