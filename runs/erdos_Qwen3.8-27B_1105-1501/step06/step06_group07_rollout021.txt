def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target} via bisection on threshold."""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _line_search_1d(S, T, C, t_lo, t_hi, n_grid=40, n_golden=50):
    """Minimize F(t) = max_r(S[r] + t*T[r] - t^2*C[r]) for t in [t_lo, t_hi]."""
    if t_hi - t_lo < 1e-14:
        mid = (t_lo + t_hi) / 2.0
        return mid, float(np.max(S + mid * T - mid * mid * C))
    
    def _F(t):
        return float(np.max(S + t * T - t * t * C))
    
    # Grid search
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([_F(t) for t in grid])
    best_i = int(np.argmin(vals))
    best_val = vals[best_i]
    
    # Golden section in bracket around best
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    
    f_a = _F(a)
    f_b = _F(b)

        f_a, f_b = f_b, f_a

    c2 = a + gr * (b - a)

    f2 = _F(c2)

            f1 = _F(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)

def _grad_for_lag(h, k):

        #         -h_{j-k} appears from term i=j (h_{j-k}*(1-h_j))

        #      g[j] -= h[j-k] for j in [k, n-1] (from h_{j-k}*(1-h_j) term)
        for j in range(m):

            g[j] += (1.0 - h[j + ga])

            g[j] -= h[j - ga]

def _weighted_gradient(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(K) / K
    else:
        w /= w_sum
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _grad_for_lag(h, k)
    return grad, corr
