def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

    S = _corr_scaled(h)

    z_max = z.max()
    z_shifted = z - z_max

            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n].copy()
            lam = res.x[n]
            if lam > 1e-8:
                return d
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, K=15, use_smooth=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    tau = 0.015 * cur_c5
    while time.time() < deadline:
        if stall > 50:
            if restart_ctr < 15:
                scale = 0.008 + 0.005 * restart_ctr
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                tau = 0.015 * cur_c5
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None and use_smooth:
            p = _lse_grad(h, 2.0 / max(cur_c5, 0.1))
            if p is not None and np.linalg.norm(p) > 1e-12:
                p = -p / np.linalg.norm(p) * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if use_smooth and stall % 8 == 0:
                tau *= 0.85
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.05, 0.30)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.96, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.012:
                radius = 0.05
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    cooldown = 0.99995
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 2:
                break
            T = max(T_min, T * 0.9)
            patience = 0
        if T < T_min:
            break
        for _ in range(300):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _stride_swap_search(h, deadline, rng, max_strides=30):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    while time.time() < deadline and stall < 50:
        improved = False
        # Compute gradient to guide stride choice
        S = _corr_scaled(h)
        G = _grad_rows(h, np.arange(2 * n - 1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        # Try different strides
        for k in range(1, min(max_strides, n)):
            if time.time() > deadline:
                break
            # For stride k, try swapping h[i] and h[i+k]
            # The effect on the objective depends on the gradient
            # Choose i that maximizes the improvement
            if k < n:
                m = n - k
                # Direction: +1 at i, -1 at i+k (or vice versa)
                # Try both directions
                for direction in [1, -1]:
                    d = np.zeros(n)
                    # Pick i where the directional derivative is most negative
                    # d[i] = direction, d[i+k] = -direction
                    # grad · d = direction * (total_grad[i] - total_grad[i+k])
                    if direction == 1:
                        gains = total_grad[:m] - total_grad[k:]
                    else:
                        gains = total_grad[k:] - total_grad[:m]
                    best_i = int(np.argmax(gains))
                    if gains[best_i] <= 0:
                        continue
                    d = np.zeros(n)
                    if direction == 1:
                        d[best_i] = 1.0
                        d[best_i + k] = -1.0
                    else:
                        d[best_i + k] = 1.0
                        d[best_i] = -1.0
                    alpha, c5_try = _line_search(h, d, dx, 0.3)
                    if c5_try < best_c5 - 1e-14:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                            improved = True
                            break
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _quad_residue_init(n, rng):
    # Use quadratic residues for a structured initial pattern
    h = np.zeros(n)
    # Try to use a prime p near n/2
    p = n // 2
    # Find nearest prime
    def is_prime(x):
        if x < 2:
            return False
        for i in range(2, int(x**0.5) + 1):
            if x % i == 0:
                return False
        return True
    while not is_prime(p) and p > 3:
        p -= 1
    if p > 3:
        residues = set()
        for a in range(1, p):
            residues.add((a * a) % p)
        # Map residues to positions
        for i in range(n):
            if i % p in residues:
                h[i] = 1.0
    else:
        h = _sine_init(n, 2, rng.random() * 6.28)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks > 1:
        boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    else:
        boundaries = []
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.
