def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

        return float(np.max(R0 + alpha * T - alpha * alpha * C)) * dx2

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _build_G(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        d = k - (n - 1)
        m = n - abs(d)
        if m <= 0:

        if d >= 0:
            G[k, :m] += 1.0 - h[d:n]
            G[k, d:] += -h[:n-d]
        else:

            G[k, q:] += 1.0 - h[:m]

def _lp_step(h, r, active_rows=None):
    """Try one LP trust-region step. Returns (new_h, new_c5) or (None, None)."""
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    G = _build_G(h)
    
    if active_rows is not None:
        G_sel = G[active_rows]
        R_sel = R[active_rows]
    else:
        G_sel = G
        R_sel = R
    
    L_sel = len(G_sel)
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    A_ub = np.hstack((-np.ones((L_sel, 1)), G_sel))
    b_ub = G_sel @ h - R_sel
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(-1e8, 1e8)]
    v_lo = np.maximum(0.0, h - r)
    v_hi = np.minimum(1.0, h + r)
    bounds += list(zip(v_lo.tolist(), v_hi.tolist()))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if res.status != 0 or not np.all(np.isfinite(res.x)):
        return None, None
    
    v = res.x[1:]
    d = v - h
    d -= np.mean(d)  # ensure zero sum
    
    if np.linalg.norm(d) < 1e-12:
        return None, None
    
    h_new, c5_new = _line_search(h, d, R, n)
    return h_new, c5_new

def _golden_pair(h, i, j, dx, S0):
    n = len(h)
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0

    C = np.correlate(d, d, 'full')

def _compute_grads(h, active_rows):
    n = len(h)
    grads = np.zeros((len(active_rows), n))
    for idx, r in enumerate(active_rows):

            continue

        else:

            grads[idx, :m] += -h[g:g + m]
