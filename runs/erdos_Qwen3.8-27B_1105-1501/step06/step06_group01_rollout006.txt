import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    n_grid = 30
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1)
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(20):
        if b - a < 1e-11:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _local_descent(h, deadline, seed=42, max_iters=5000):
    """Local search: only accept improvements. Uses pair and gradient moves."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rng = np.random.default_rng(seed)
    stall = 0
    
    # Precompute G for gradient moves
    G = _build_G(h, dx)
    g_interval = 15  # rebuild G every 15 iterations
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        r = rng.random()
        
        if r < 0.6:
            # Pair move: transfer mass from i to j
            interior = np.where((h > 0.01) & (h < 0.99))[0]
            if len(interior) < 2:
                continue
            if r < 0.3:
                # Guided by gradient of active lag
                if it % g_interval == 0 or it == 0:
                    S = _corr_vals(h)
                    G = _build_G(h, dx)
                active = int(np.argmax(S))
                grad = G[active]
                # Move from high-grad to low-grad
                sources = interior[np.argsort(grad[interior])[-5:]]  # top 5 sources
                sinks = interior[np.argsort(grad[interior])[:5]]  # top 5 sinks
                i = sources[rng.integers(len(sources))]
                j = sinks[rng.integers(len(sinks))]
            else:
                i = interior[rng.integers(len(interior))]
                j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        elif r < 0.85:
            # Small smooth perturbation
            d = rng.normal(0, 0.05, n)
            d -= d.mean()
            norm = np.max(np.abs(d))
            if norm > 0:
                d *= 0.15 / norm
            else:
                continue
        else:
            # Global gradient of active lag
            if it % g_interval == 0 or it == 0:
                S = _corr_vals(h)
                G = _build_G(h, dx)
            active = int(np.argmax(S))
            d = -G[active]
            d -= d.mean()
            norm = np.max(np.abs(d))
            if norm > 1e-10:
                d *= 0.15 / norm
            else:
                continue
        
        alpha, c5_try = _line_search(h, d, dx, 0.25)
        if c5_try < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb to escape
                h = h + rng.normal(0, 0.03, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                stall = 0
    
    return best_h, best_c5

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[q, k:] += dx * (1.0 - h[:m])
                G[q, :m] += -dx * h[k:]
        else:
            d_abs = -k
            m = n - d_abs
            if m > 0:
                G[q, :m] += dx * (1.0 - h[d_abs:])
                G[q, d_abs:] += -dx * h[:m]
    return G

def _sa_optimize(h, deadline, seed=42, n_iters=5000):
    """Simulated annealing with pair moves and gradient moves."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    rng = np.random.default
