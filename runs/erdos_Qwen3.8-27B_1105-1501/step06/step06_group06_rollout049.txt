                h = _project(h + rng.normal(0, 0.02, n), target)
                cur = _c5(h)
                radius = 0.15
                stall = 0
                if cur < best:
                    best = cur
                    best_h = h.copy()
                continue
            if stall > 500:
                break
        corr = _corr_vals(h)
        M = cur
        tol = max(2e-5, 0.015 * M)
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:8]
        elif len(active) > 35:
            oa = np.argsort(-corr[active])
            active = active[oa[:35]]
        # Compute negative gradient of active corrs projected to sum-zero
        G_sum = np.zeros(n)
        for k in active:
            G_sum += _corr_grad_row(h, k)
        G_sum -= G_sum.mean()
        pnorm = np.linalg.norm(G_sum)
        if pnorm < 1e-14:
            stall += 5
            radius *= 0.8
            continue
        d = -G_sum / pnorm * radius
        # Line search
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.zeros(2 * n - 1)
        for s in range(-(n - 1), n):
            k = n - 1 + s
            m = n - abs(s)
            if s >= 0:
                C[k] = -np.sum(d[:m] * d[s:s + m])
            else:
                g = -s
                C[k] = -np.sum(d[g:g + m] * d[:m])
        C *= dx
        S0 *= dx
        T *= dx
        # Find valid alpha range
        pos = d > 1e-14
        neg = d < -1e-14
        a_lo, a_hi = -radius, radius
        if np.any(pos):
            a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            a_lo = max(a_lo, float(np.max(h[neg] / d[neg])))
        if a_hi - a_lo < 1e-14:
            stall += 1
            continue
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = a_lo, a_hi
        f1 = float(np.max(S0 + a * T + a * a * C))
        f2 = float(np.max(S0 + b * T + b * b * C))
        for _ in range(50):
            if b - a < 1e-12:
                break
            if f1 < f2:
                b, f2 = a, f1
                a = b - gr * (b - a)
                f1 = float(np.max(S0 + a * T + a * a * C))
            else:
                a, f1 = b, f2
                b = a + gr * (b - a)
                f2 = float(np.max(S0 + b * T + b * b * C))
        alpha = (a + b) / 2.0
        h_new = _project(h + alpha * d, target)
        c_new = _c5(h_new)
        if c_new < cur - 1e-15:
            h = h_new
            cur = c_new
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
            if abs(alpha) > 0.4 * radius:
                radius = min(radius * 1.08, 0.4)
        else:
            stall += 1
            radius *= 0.8
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = np.full(n0, 0.5)
        rng0 = np.random.default_rng(seed)
        flip = rng0.integers(0, n0, n0 // 2)
        h0[flip] = 1.0

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _c5(h0)
    best_n = n0

    def _update(hc, c5c, nc):
        nonlocal best_h, best_c5, best_n
        if c5c < best_c5 - 1e-15:
            best_h = hc.copy()
            best_c5 = c5c
            best_n = nc

    # Phase 1: SLSQP on initial (12%)
    d1 = t0 + budget_s * 0.12
    h1, c1 = _slsqp(best_h, d1, max_outer=100)
    _update(h1, c1, len(h1))

    # Phase 2: Active descent (8%)
    d2 = t0 + budget_s * 0.20
    h2, c2 = _active_descent(best_h, d2)
    _update(h2, c2, len(h2))

    # Phase 3: Binary local search + relax + polish (25%)
    d3 = t0 + budget_s * 0.45
    h_bin, c_bin = _binary_ls(best_h, min(d3, time.time() + budget_s * 0.12))
    # Try multiple smoothing widths
    for w in [3, 5, 7, 9]:
        if time.time() > d3:
            break
        h_s = _smooth_relax(h_bin, best_n, w)
        h_s, c_s = _slsqp(h_s, min(d3, time.time() + budget_s * 0.03), max_outer=80)
        _update(h_s, c_s, len(h_s))
        if time.time() < d3 - 20:
            h_s2, c_s2 = _active_descent(h_s, min(d3, time.time() + budget_s * 0.03))
            _update(h_s2, c_s2, len(h_s2))

    # Phase 4: Multi-resolution (30%)
    d4 = t0 + budget_s * 0.75
    offsets = [0, -2, 2, -6, 6, -10, 10, -14, 14, -20, 20, -24, 24, -30, 30, -40, 40, 50, -50]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 280 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        # Quick SLSQP
        sub_d = min(d4, time.time() + budget_s * 0.025)
        h_r, c_r = _slsqp(h_r, sub_d, max_outer=50)
        # Binary LS on this resolution
        if time.time() < d4 - 15:
            sub_d2 = min(d4 - 5, time.time() + budget_s * 0.02)
            h_br, c_br = _binary_ls(h_r, sub_d2)
            h_sr = _smooth_relax(h_br, n_try, 5)
            sub_d3 = min(d4 - 3, time.time() + budget_s * 0.02)
            h_sr, c_sr = _slsqp(h_sr, sub_d3, max_outer=40)
            _update(h_sr, c_sr, n_try)
        _update(h_r, c_r, n_try)

    # Phase 5: Multi-start diversification (12%)
    d5 = t0 + budget_s * 0.87
    rng_ms = np.random.default_rng(seed + 1)
    n_start = best_n
    for trial in range(6):
        if time.time() > d5:
            break
        # Random perturbation
        h_p = best_h + rng_ms.normal(0, 0.08, n_start)
        h_p = _project(h_p, n_start / 2.0)
        sub_d = min(d5, time.time() + budget_s * 0.015)
        h_p, c_p = _slsqp(h_p, sub_d, max_outer=40)
        _update(h_p, c_p, n_start)

    # Phase 6: Final polish (10%)
    d6 = deadline - 5.0
    h6, c6 = _slsqp(best_h, min(d6, time.time() + budget_s * 0.05), max_outer=100)
    _update(h6, c6, len(h6))
    if time.time() < d6 - 10:
        h6b, c6b = _active_descent(best_h, d6)
        _update(h6b, c6b, len(h6b))
    h6c, c6c = _slsqp(best_h, deadline - 2.0, max_outer=50)
    _update(h6c, c6c, len(h6c))

    # Final verification
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
