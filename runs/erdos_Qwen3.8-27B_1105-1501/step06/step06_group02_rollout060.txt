def solve_lp_direction(h, R, G, radius):
    n = len(h)
    # Variables: [d_0, ..., d_{n-1}, t]
    # Objective: minimize t
    # Constraints:
    #   G @ d - t <= -R  (i.e., R + G@d <= t)
    #   sum(d) = 0
    #   -min(h_i, radius) <= d_i <= min(1-h_i, radius)
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = -R
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = [(-min(h[i], radius), min(1-h[i], radius)) for i in range(n)]
    bounds.append((0, None))  # t is unbounded above (or set a reasonable upper bound)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    else:
        # Try to get the best we can
        if hasattr(result, 'x') and result.x is not None:
            d = result.x[:n]
            d -= np.mean(d)  # ensure zero sum
            return d
        return None

from scipy.optimize import linprog

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    v = np.clip(v, -2.0, 3.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_matrix(h):
    """Compute gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for lag in range(-(n-1), n):
        row = lag + (n - 1)
        if lag == 0:
            G[row, :] = 1.0 - 2.0 * h
        elif lag > 0:
            # R_lag = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # dR/dh_j = (1-h_{j-lag}) for j in [lag, n-1]
            #         - h_{j+lag} for j in [0, n-1-lag]
            m = n - lag
            G[row, lag:] = (1.0 - h[:m])  # j from lag to n-1, h[j-lag] = h[0..m-1]
            G[row, :m] += -h[lag:]  # j from 0 to n-1-lag, h[j+lag] = h[lag..n-1]
        else:

def solve_lp_direction(h, R, G, radius):
    """Solve LP to find direction that minimizes linearized max overlap."""
    n = len(h)
    # Variables: [d_0, ..., d_{n-1}, t]
    # Minimize: t
    # Subject to: R_i + G_i·d <= t  =>  G_i·d - t <= -R_i
    #             sum(d) = 0
    #             -min(h_i, radius) <= d_i <= min(1-h_i, radius)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = -R.copy()
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = -min(h[i], radius)
        hi = min(1.0 - h[i], radius)
        bounds.append((lo, hi))
    bounds.append((0.0, 10.0))  # t is unbounded above, but set a large cap
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success or (result.x is not None and np.all(np.isfinite(result.x))):
            d = result.x[:n].copy()
            d -= np.mean(d)  # enforce zero sum exactly
            return d
    except Exception:
        pass
    return None

import numpy as np
import time
from scipy.optimize import linprog

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    v = np.clip(v, -2.0, 3.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for lag in range(-(n - 1), n):
        row = lag + (n - 1)
        if lag == 0:
            G[row, :] = 1.0 - 2.0 * h
        elif lag > 0:
            m = n - lag
            G[row, lag:] = (1.0 - h[:m])
            G[row, :m] += -h[lag:]
        else:
            r = -lag
            m = n - r
            G[row, :m] = (1.0 - h[r:])
            G[row, r:] += -h[:m]
    return G

def solve_lp_direction(h, R, G, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -R.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = -min(h[i], radius)
        hi = min(1.0 - h[i], radius)
        bounds.append((lo, hi))
    bounds.append((0.0, 10.0))
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.x is not None and np.all(np.isfinite(result.x)):
            d = result.x[:n].copy()
            d -= np.mean(d)
            return d
    except Exception:
        pass
    return None

def subgradient_direction(h, R, G, radius):
    n = len(h)
    max_R = float(np.max(R))
    threshold = max_R - 0.02 * max_R
    active = np.where(R >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-R)[:5]
    weights = np.exp((R[active] - max_R) / (0.005 * max_R + 1e-15))
    weights /= float(np.sum(weights))
    g = G[active].T @ weights
    d = -g
    d -= np.mean(d)
    d = np.clip(d, -radius, radius)
    inner = (h > 1e-6) & (h < 1.0 - 1e-6)
    d[~inner] = 0.0
    d -= np.mean(d)
    return d

def exact_line_search(h, d, R, dx):
    n = len(h)
    if np.max(np.abs(d)) < 1e-14:
        return 0.0, c5_exact(h)
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    pos = d > 1e-14
    neg = d < -1e-14
    hi = 1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    lo = -1.0
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    lo = max(lo, -10.0)
    hi = min(hi, 10.0)
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)

    def f(alpha):
        return float(np.max(R + alpha * D - alpha * alpha * Q)) * dx

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if b - a < 1e-13 * max(1.0, abs(a), abs(b)):
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    opt = (a + b) / 2.0
    val = f(opt)
    for endpoint in [lo, hi, 0.0]:
        ve = f(endpoint)
        if ve < val:
            val, opt = ve, endpoint
    return opt, val

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)

def _lp_optimize(h, deadline, rng, max_iter=500):

    while time.time() < deadline and it < max_iter:

def _pair_polish(h, deadline, rng, n_pairs=20):

    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0

    while time.time() < deadline and stall < 60:
        scores = S0 * dx
        max_s = float(np.max(scores))
        thr = max_s - 0.01 * max_s
        active = np.where(scores >= thr)[0]
        if len(active) == 0:
            active = np.argsort(-scores)[:10]
        G = compute_grad_matrix(h)
        active_r = G[active]
        W = np.exp((scores[active] - max_s) / (0.01 * max_s + 1e-15))
        W /= float(np.sum(W))
        wg = active_r.T @ W
        donors = np.argsort(-wg)[:n_pairs]
        recvs = np.argsort(wg)[:n_pairs]
        best_alpha = 0.0
        best_new = cur_c5
        best_pair = None
        for i in donors:
            if time.time() > deadline:
                break
            for j in recvs:
                if i == j:
                    continue
                alpha, nc = exact_line_search(h, _pair_dir(n, i, j), S0, dx)
                if abs(alpha) > 1e-10 and nc < best_new - 1e-14:
                    best_new = nc
                    best_alpha = alpha
                    best_pair = (i, j)
        if best_pair is not None and best_new < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')

            stall = 0
        else:
            stall += 1

                h[idx] += rng.uniform(-0.04, 0.04, size=4)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def _pair_dir(n, i, j):
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    return d
