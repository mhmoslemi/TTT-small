def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:

        else:

    G = _all_gradients(h) * dx  # (2n-1, n)
    
    # Variables: x (n vars) + t (1 var) = n+1 vars
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: G[k]·(x-h) - t <= corr[k]
    # i.e., G[k]·x - t <= corr[k] + G[k]·h
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = corr + G.dot(h)
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def _slsqp_full(h0, deadline, max_outer=30):

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

            return x[n]

            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    """Compute G of shape (2n-1, n): G[k,i] = d(corr[k])/d(h[i]) (without dx factor)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            G[k, :m] += (1.0 - h[s:s + m])
            G[k, s:s + m] += (-h[:m])
        else:
            g = -s
            G[k, g:g + m] += (1.0 - h[:m])
            G[k, :m] += (-h[g:g + m])
    return G

def _lp_step(h, budget_t=None):
    """One LP step: solve min t s.t. linearized constraints.
    Returns (x_new, t_pred) or (None, None)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # (2n-1,)
    G = _grad_matrix(h) * dx  # (2n-1, n)
    n_lags = 2 * n - 1
    
    # Variables: [x_0, ..., x_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((n_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G.dot(h) - corr  # = ∇f_k·h - f_k(h)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None

def _line_search(h, d, t_max=1.0):
    """Exact line search along h + alpha*d. Returns (best_alpha, best_c5)."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    # Find valid range of alpha
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = 0.0
    a_hi = t_max
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.max(h[neg] / (-d[neg]))))  # h + alpha*d >= 0
    
    # Also need sum(h + alpha*d) = n/2, which is automatic if sum(d)=0
    # (which it should be from the LP since the equality constraint is sum(x)=n/2)
    # But after projection, sum might not be exactly n/2, so let's check
    
    if a_hi - a_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Bisection / golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = a_lo, a_hi
    fa = _exact_c5(_project(h + a * d, target))
    fb = _exact_c5(_project(h + b * d, target))
    
    for _ in range(60):
        if b - a < 1e-12:
            break
        m1 = a + gr * (b - a)
        m2 = b - gr * (b - a)
        fm1 = _exact_c5(_project(h + m1 * d, target))
        fm2 = _exact_c5(_project(h + m2 * d, target))
        if fm1 < fm2:
            b = m2
            fb = fm2
        else:
            a = m1
            fa = fm1
        if m1 - a < 1e-13:
            break
        if b - m2 < 1e-13:
            break
    
    best_a = (a + b) / 2.0
    best_h = _project(h + best_a * d, target)
    best_c5 = _exact_c5(best_h)
    
    # Try a few more points
    for trial_a in [best_a * 0.5, best_a * 1.5, a_lo + 0.1 * (a_hi - a_lo)]:
        if a_lo <= trial_a <= a_hi:
            th = _project(h + trial_a * d, target)
            tc = _exact_c5(th)
            if tc < best_c5:
                best_c5 = tc
                best_a = trial_a
    
    return best_a, best_c5

def _lp_descent(h_init, deadline, max_iter=1000):
    """LP-based descent with exact line search."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            # Perturb
            h = h + rng.normal(0, 0.01, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            stall = 0
            continue
        if stall > 500:
            break
        
        x_new, t_pred = _lp_step(h)
        if x_new is None:
            stall += 5
            continue
        
        d = x_new - h
        # Ensure sum(d) = 0 (should be from LP, but just in case)
        d -= d.mean()
        
        if np.linalg.norm(d) < 1e-14:
            stall += 10
            continue
        
        # Line search
        a, c5_new = _line_search(h, d, t_max=1.0)
        
        if c5_new < cur_c5 - 1e-15:
            h = _project(h + a * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            # Try smaller step
            if a > 0.01:
                for frac in [0.5, 0.25, 0.1]:
                    a2 = a * frac
                    h2 = _project(h + a2 * d, target)
                    c5_2 = _exact_c5(h2)
                    if c5_2 < cur_c5 - 1e-15:
                        h = h2
                        cur_c5 = c5_2
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stall = min(stall, 3)
                        break
    
    return best_h, best_c5

def _slsqp_opt(h0, deadline, max_outer=30):
    """SLSQP with adaptive active set."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = cur_c5
        # Active set: all lags within 1% of max, or top 60
        tol = max(M * 0.01, 1e-6)
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:30]
        elif len(active) > 60:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:60]]
        
        t_init = M + 1e-6
        x0 = np.concatenate([h, [t_init]])
        
        def obj_fn(x):
            return x[n]
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[active]
        def ineq_jac(x):
            hx = x[:n]
            G = _grad_matrix(hx)
            J = np.zeros((len(active), n + 1))
            J[:, :n] = -G[active, :] * dx
            J[:, n] = 1.0
            return J
        def eq_fn(x):
            return np.sum(x[:n]) - target
        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 80, 'ftol': 1e-15})
        
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = np.arange(n_new) + 0.5
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_sine(n, freq, phase):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _make_rand(n, rng):
    h = rng.uniform(0.2, 0.8, n)
    return _project(h, n / 2.0)

The total should be well within budget. The main concern is whether the algorithm actually makes progress. Let me think about this...

The LP descent should work well when the current point is not at a local optimum. The key question is: is the parent at a local optimum? If yes, the LP descent will stall immediately, and we need the multi-start or perturbation to escape.

Given that the previous code achieved 0.380891 and we need 0.3808, the improvement needed is very small. This suggests that:
1. Either the parent is very close to a local optimum and we need a tiny perturbation + polish
2. Or there's a slightly better local optimum nearby that's accessible with a different optimization path

My multi-start and perturbation strategies should handle both cases.
