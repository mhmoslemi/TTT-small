def _subgrad_direction(h, S, n):

    k_best = int(np.argmax(S))
    k = k_best - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += (1.0 - h[:m])
            g[:m] += -h[k:]
    else:
        d = -k
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d+m])
            g[d:] += -h[:m]
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-12:
        return None
    return -g / norm

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:min(K, len(S))]
    tol = max(0.06 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 55:
        combined = sorted_idx[:55]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgrad_direction(h, S, n, top_k=5):
    idx_top = np.argsort(-S)[:top_k]
    G = _grad_rows(h, idx_top, n)
    g = G.sum(axis=0)
    # Weight by how close to max
    w = S[idx_top] - S[idx_top].min() + 1.0
    g = (w[:, None] * G).sum(axis=0)
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-12:
        return None
    return -g / norm

def _lp_loop(h, deadline, radius_init=0.12, K=25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_stall = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > 100:
            scale = 0.015 + 0.008 * min(total_stall // 100, 6)
            h = h + rng.normal(0, scale, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.1
            stall = 0
            total_stall += 100
            if total_stall > 3000:
                break
            continue
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            p = _subgrad_direction(h, S, n)
            if p is not None:
                p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            total_stall += 1
            radius *= 0.82
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-15:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.65:
                radius = min(radius * 1.06, 0.35)
            elif ratio > 0.25:
                radius = min(radius * 1.02, 0.30)
            else:
                radius = max(radius * 0.88, 0.015)
        else:
            stall += 1
            total_stall += 1
            radius *= 0.76
            if stall % 18 == 0 and radius < 0.01:
                radius = 0.07
    return best_h, best_c5

def _coord_descent(h, deadline, max_rounds=200):
    """Coordinate descent: optimize one variable at a time using 1D convex solve."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            # Compute a_k for all k (2n-1 shifts)
            a = np.ones(2 * n - 1)
            for r_idx in range(2 * n - 1):
                k = r_idx - (n - 1)
                if 0 <= i + k < n:
                    a[r_idx] -= h[i + k]
                if 0 <= i - k < n:
                    a[r_idx] -= h[i - k]
            # C_k = f_k - h[i]*a_k (value when h[i]=0)
            S = np.correlate(h, 1.0 - h, mode='full')
            C = S - h[i] * a
            # Minimize max_k (C[k] + x * a[k]) for x in [0,1]
            x_lo, x_hi = 0.0, 1.0
            for _ in range(50):
                x_mid = (x_lo + x_hi) / 2.0
                vals = C + x_mid * a
                mx = float(np.max(vals))
                active_mask = np.abs(vals - mx) < 1e-10
                a_act = a[active_mask]
                if np.all(a_act >= -1e-12):
                    x_hi = x_mid
                elif np.all(a_act <= 1e-12):
                    x_lo = x_mid
                else:
                    x_lo = max(0, x_mid - 1e-9)
                    x_hi = min(1, x_mid + 1e-9)
                    break
            x_opt = (x_lo + x_hi) / 2.0
            if abs(x_opt - h[i]) > 1e-6:
                h[i] = x_opt
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5 - 1e-15:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _fractional_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.000002
    cooldown = 0.9997
    patience = 0
    max_patience = 8000
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 4:
                break
            T = max(T_min, T * 2.0)
            patience = 0
        if T < T_min:
            break
        n_inner = 150
        for _ in range(n_inner):
            if time.time() > deadline:
                break
            scale = 0.04 * (1.0 + 2.0 * T / 0.004)
            d = rng.normal(0, scale, n)
            d -= d.mean()
            h_new = h + d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            delta = new_c5 - cur_c5
            if delta < -1e-15 or rng.random() < np.exp(-delta / T):
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    cooldown = 0.9999
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.9)
            patience = 0
        if T < T_min:
            break
        for _ in range(300):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _pair_refine(h, deadline, max_iter=5000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    stall = 0
    for _it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c5_try = _line_search(h, d, dx, 0.5)
        if c5_try < best_c5 - 1e-15:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.5)
            if c5_try < best_c5 - 1e-15:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick LP from initial (5%)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _lp_loop(h0, d1, radius_init=0.12, K=25)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-start (10%)
    d2 = t0 + budget_s * 0.15
    starts = []
    for freq in [1, 2, 3, 4, 5, 6, 7, 9]:
        for ph in [0.0, 0.6, 1.2, 2.0, 2.8, 3.9]:
            starts.append(_sine_init(n0, freq, ph))
    for nb in [4, 6, 8, 10, 14]:
        h_b = np.zeros(n0)
        bnd = np.sort(rng.choice(np.arange(1, n0), size=nb - 1, replace=False))
        bnd = np.concatenate([[0], bnd, [n0]])
        for b in range(nb - 1):
            if b % 2 == 0:
                h_b[int(bnd[b]):int(bnd[b+1])] = 1.0
        starts.append(_project(h_b, n0 / 2.0))
    for s in starts:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.009)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=18)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Deep LP (20%)
    d3 = t0 + budget_s * 0.35
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.15, K=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Fractional SA + LP polish (12%)
    d4 = t0 + budget_s * 0.47
    h4, c5_4 = _fractional_sa(best_h, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < t0 + budget_s * 0.50:
        h4b, c5_4b = _lp_loop(best_h, t0 + budget_s * 0.50, radius_init=0.08, K=25)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Coordinate descent (8%)
    d5 = t0 + budget_s * 0.58
    h5, c5_5 = _coord_descent(best_h, d5, max_rounds=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < t0 + budget_s * 0.61:
        h5b, c5_5b = _lp_loop(best_h, t0 + budget_s * 0.61, radius_init=0.08, K=25)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution search (20%)
    d6 = t0 + budget_s * 0.81
    base_n = len(best_h)
    n_candidates = sorted(set(
        [base_n + dn for dn in
         [-12, -10, -8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16, 20, 24, 28, 32, 36, 40]
         if 60 <= base_n + dn <= 320 and (base_n + dn) % 2 == 0]
    ))
    for n_try in n_candidates:
        if time.time() > d6:
            break
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.012:
            sub_dl = min(d6, time.time() + budget_s * 0.022)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, K=20)
            if c5_r < best_c5 + 0.005 and time.time() < d6 - budget_s * 0.02:
                sub_dl2 = min(d6, time.time() + budget_s * 0.018)
                h_r2, c5_r2 = _fractional_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d6 - budget_s * 0.012:
                    h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + budget_s * 0.015),
                                         radius_init=0.06, K=20)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Binary SA on best (8%)
    d7 = t0 + budget_s * 0.89
    h7, c5_7 = _binary_sa(best_h, d7, rng)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < t0 + budget_s * 0.92:
        h7b, c5_7b = _lp_loop(best_h, t0 + budget_s * 0.92, radius_init=0.06, K=22)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Phase 8: Pair refinement + final polish (6%)
    d8 = t0 + budget_s * 0.96
    h8, c5_8 = _pair_refine(best_h, d8, max_iter=4000)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final LP polish
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.04, K=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Coord descent final touch
    if time.time() < deadline - 2.0:
        h_cd, c5_cd = _coord_descent(best_h, deadline, max_rounds=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
