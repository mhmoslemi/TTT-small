    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2, 0, 1)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))

    else:
