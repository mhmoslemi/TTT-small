def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grads(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            G[k, :m] += (1.0 - h[s:s + m])
            G[k, s:s + m] += (-h[:m])
        else:
            g = -s
            G[k, g:g + m] += (1.0 - h[:m])
            G[k, :m] += (-h[g:g + m])
    G *= dx
    return G

def _line_search_quad(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full').astype(float)
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')).astype(float)
    Q = np.correlate(d, d, mode='full').astype(float)
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-15:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 60
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None]) ** 2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if 0 < best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    f1 = float(np.max(S0 + a * T - a * a * Q)) * dx
    f2 = float(np.max(S0 + b * T - b * b * Q)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, f2 = a, f1
            a = b - gr * (b - a)
            f1 = float(np.max(S0 + a * T - a * a * Q)) * dx
        else:
            a, f1 = b, f2
            b = a + gr * (b - a)
            f2 = float(np.max(S0 + b * T - b * b * Q)) * dx
    mid = (a + b) / 2.0
    mid_val = float(np.max(S0 + mid * T - mid * mid * Q)) * dx
    if mid_val < best_val:
        best_t, best_val = mid, mid_val
    return best_t, best_val

def _subgradient_step(h, lr, active_tol, max_active=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(corr))
    active = np.where(corr > M - active_tol)[0]
    if len(active) == 0:
        active = np.argsort(-corr)[:5]
    elif len(active) > max_active:
        order = np.argsort(-corr[active])
        active = active[order[:max_active]]
    G = _corr_grads(h)
    sg = G[active].mean(axis=0)
    h_new = _project(h - lr * sg, n / 2.0)
    return h_new

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_optimize(h0, deadline, max_outer=12):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        K = min(20, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        t_init = cur_c5 + 1e-5
        x0 = np.concatenate([h, [t_init]])

        def obj_fn(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]

        def ineq_jac(x):
            hx = x[:n]
            G = _corr_grads(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J

        def eq_fn(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]

        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-12})

        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break

    return best_h, best_c5

def _smooth_restart(h, rng, sigma=2.0):
    n = len(h)
    kernel_size = max(3, int(4 * sigma))
    if kernel_size % 2 == 0:
        kernel_size += 1
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    h_smooth = np.convolve(h, kernel, mode='same')
    h_perturbed = h_smooth + rng.normal(0, 0.03, n)
    return _project(h_perturbed, n / 2.0)

def _structured_init(n, freq, amp, rng):

    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)
