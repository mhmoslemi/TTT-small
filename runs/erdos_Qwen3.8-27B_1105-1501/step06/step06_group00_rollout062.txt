def _compute_gradient(h, w, dx):
    n = len(h)
    # Build the W kernel
    w_pos = w[n-1:]       # length n, indices 0..n-1 → lags 0..n-1
    w_neg = w[:n-1][::-1] # length n-1, indices 0..n-2 → lags -1..-(n-1)
    
    W = np.zeros(n)
    W[0] = w_pos[0]
    for r in range(1, n):
        W[r] = w_pos[r] + w_neg[r-1]
    
    # const[i] = sum_{p=0}^{i} w_pos[p] + sum_{q=0}^{n-2-i} w_neg[q]
    cumsum_wpos = np.cumsum(w_pos)  # cumsum_wpos[i] = sum_{p=0}^{i} w_pos[p]
    cumsum_wneg = np.cumsum(w_neg)  # cumsum_wneg[i] = sum_{q=0}^{i} w_neg[q]
    # sum_{q=0}^{n-2-i} w_neg[q] = cumsum_wneg[n-2-i] for n-2-i >= 0, else 0
    const = np.zeros(n)
    const += cumsum_wpos  # sum_{p=0}^{i} w_pos[p] for each i
    for i in range(n):
        if n-2-i >= 0:
            const[i] += cumsum_wneg[n-2-i]
        # else: sum is 0
    
    # h-convolution term: sum_{q=-i}^{n-1-i} W[|q|] * h[i+q]
    # = sum_{q=0}^{i} W[q]*h[i-q] + sum_{q=1}^{n-1-i} W[q]*h[i+q]
    # (q=0 appears in both sums, giving W[0]*h[i] twice)
    
    # Using Toeplitz matrix:
    # M[i,k] = W[|i-k|] if |i-k| < n and (i+k) < 2n-1... 
    # Actually: the first sum is q=0..i, position i-q (decreasing)
    #            the second sum is q=1..n-1-i, position i+q (increasing)
    # Combined: sum_{k=0}^{n-1} W_anti[i,k] * h[k]
    # where W_anti[i,k] = W[|i-k|] if i>=k (anti-cause) and W[i-k]... 
    # Let me just build the matrix directly:
    
    idx = np.arange(n)
    i_idx = idx[:, None]  # (n,1)
    k_idx = idx[None, :]  # (1,n)
    distances = np.abs(i_idx - k_idx)  # (n,n)
    # For position i and k:
    # If k <= i: contributes W[i-k] (from anti-cause sum, q = i-k, valid when i-k <= i i.e. always, and i-k < n)
    # If k > i: contributes W[k-i] (from cause sum, q = k-i, valid when k-i < n-i i.e. k < n, always true)
    # So in all cases: M[i,k] = W[|i-k|]
    
    M = W[distances]  # (n,n)
    h_conv = M @ h  # (n,)
    
    g = (const - h_conv) * dx
    return g

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_weights(c, beta):
    """Compute smooth-max weights on correlation values c."""
    M = float(np.max(c))
    e = np.exp(np.clip(beta * (c - M), -50.0, 0.0))
    s = float(np.sum(e))
    if s < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return e / s

def _compute_gradient(h, w, dx):
    """Compute g = dx * G^T @ w where G is the gradient matrix of correlation."""
    n = len(h)
    # w has length 2n-1, indexed by lag: w[0] = lag -(n-1), ..., w[n-1] = lag 0, ..., w[2n-2] = lag n-1
    w_pos = w[n-1:]       # length n: w_pos[p] = weight at lag p, p=0..n-1
    w_neg = w[:n-1][::-1] # length n-1: w_neg[q] = weight at lag -(q+1), q=0..n-2
    
    # Build kernel W of length n
    W = np.empty(n)
    W[0] = w_pos[0]
    if n > 1:
        W[1:] = w_pos[1:] + w_neg[:n-1]
    
    # Build const vector
    cumsum_wp = np.cumsum(w_pos)  # cumsum_wp[i] = sum_{p=0}^{i} w_pos[p]
    if n > 1:
        cumsum_wn = np.cumsum(w_neg)
        # const[i] = cumsum_wp[i] + cumsum_wn[n-2-i] for i <= n-2, else cumsum_wp[i]
        const = cumsum_wp.copy()
        const[:n-1] += cumsum_wn[::-1]
    else:
        const = cumsum_wp.copy()
    
    # Compute M @ h where M[i,k] = W[|i-k|]
    idx = np.arange(n)
    dist = np.abs(idx[:, None] - idx[None, :])
    M = W[dist]
    h_conv = M @ h
    
    g = (const - h_conv) * dx
    return g

def _pair_T(h, i, j):
    """Compute T[m] = G[m,i] - G[m,j] for all m (length 2n-1)."""
    n = len(h)
    T = np.zeros(2*n - 1)
    # G[m,k] for lag ℓ = m-(n-1):
    #   G = (1-h[k-ℓ])*[k-ℓ in [0,n) and lag convention] - h[k+ℓ]*[k+ℓ in [0,n) and ...]
    # Simplified: G[m,k] = (1-h[k-ℓ]) * [0<=k-ℓ<n, valid lag range] - h[k+ℓ] * [0<=k+ℓ<n, valid]
    # For ℓ >= 0 (m >= n-1):
    #   First: valid when k >= ℓ (k-ℓ >= 0) and k-ℓ < n (k < n+ℓ, always true for k<n, ℓ<n)
    #          → k >= ℓ
    #   Second: valid when k+ℓ < n → k < n-ℓ
    # For ℓ < 0 (let p = -ℓ > 0, m < n-1):
    #   First: valid when k+p < n → k < n-p = n+ℓ
    #   Second: valid when k-p >= 0 → k >= p = -ℓ
    
    lags = np.arange(-(n-1), n)  # length 2n-1
    for k in (i, j):
        for m_idx, ℓ in enumerate(lags):
            if ℓ >= 0:
                # First term: (1-h[k-ℓ]) if k >= ℓ and k-ℓ < n
                if k >= ℓ and k - ℓ < n:
                    T[m_idx] += (1.0 - h[k - ℓ]) if k == i else -(1.0 - h[k - ℓ])
                # Second term: -h[k+ℓ] if k+ℓ < n
                if k + ℓ < n:
                    T[m_idx] += -h[k + ℓ] if k == i else h[k + ℓ]
            else:
                p = -ℓ
                # First term: (1-h[k+p]) if k+p < n
                if k + p < n:
                    T[m_idx] += (1.0 - h[k + p]) if k == i else -(1.0 - h[k + p])
                # Second term: -h[k-p] if k >= p
                if k >= p:
                    T[m_idx] += -h[k - p] if k == i else h[k - p]
    return T
