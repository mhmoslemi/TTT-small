def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))

def _build_corr_mat(fixed, dx):

    M = np.zeros((2 * n - 1, n))
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        M[:, i] = np.correlate(e, fixed, 'full') * dx
    return M

def _solve_lp_minmax(M, n, target):

    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, 'full') * dx

        sub_dl = min(d4, time.time() + max(3.0, (d4 - time.time()) / max(1, n_starts - si) * 0.5))
        h_s, c5_s = _lp_local(h_s, sub_dl, start_radius=0.08)
        rem = d4 - time.time()
        if rem > 3.0:
            h_s, c5_s2 = _relaxed_pair(h_s, min(d4, time.time() + rem * 0.4), max_rounds=50)
            if c5_s2 < c5_s:
                c5_s = c5_s2
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Binary optimization
    d5 = t0 + budget_s * 0.82
    h5, c5_5 = _binary_local(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Smooth descent
    d6 = t0 + budget_s * 0.90
    h6, c5_6 = _smooth_descent(best_h, d6)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Multi-resolution
    d7 = t0 + budget_s * 0.95
    n_best = len(best_h)
    tried_n = set()
    for dn in (-4, 4, -6, 6, -8, 8, 12, -12, 16, -16, 20, -20):
        if time.time() > d7:
            break
        n_try = n_best + dn
        if n_try < 24 or n_try > 250 or n_try % 2 != 0 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + max(2.0, budget_s * 0.012))
        h_r, c5_r = _lp_local(h_r, sub_dl, start_radius=0.08)
        rem2 = d7 - time.time()
        if rem2 > 2.0:
            h_r, c5_r2 = _relaxed_pair(h_r, min(d7, time.time() + rem2 * 0.3), max_rounds=30)
            if c5_r2 < c5_r:
                c5_r = c5_r2
        if c5_r < best_c5 - 1e-11:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 8: Final pair coord + LP polish
    h8, c5_8 = _pair_coord(best_h, deadline, max_passes=30)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    rem_final = deadline - time.time()
    if rem_final > 2.0:
        h9, c5_9 = _lp_local(best_h, deadline, start_radius=0.05)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
