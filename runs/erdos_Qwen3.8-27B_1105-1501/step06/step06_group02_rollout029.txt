def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

            lo = mid
        else:

def _compute_grads(h, active_rows):
    """Compute gradient of each active row with respect to h. Returns (n_active, n) array."""
    n = len(h)
    grads = np.zeros((len(active_rows), n))
    for k, r in enumerate(active_rows):
        d = r - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            if d == 0:
                grads[k, :m] = 1.0 - 2.0 * h[:m]
            else:
                grads[k, :m] += (1.0 - h[d:d + m])
                grads[k, d:d + m] += -h[:m]

            else:
                grads[k, g:] += (1.0 - h[:m])
                grads[k, :m] += -h[g:g + m]
    return grads

def _lp_direction(h, S0, dx, n_active, delta):
    """Find LP-based minimax direction. Returns direction vector d."""
    n = len(h)
    scores = S0 * dx
    max_score = float(np.max(scores))
    
    # Select active peaks: top n_active or those within 5% of max
    threshold = max_score - 0.02 * max_score
    active_idx = np.where(scores >= threshold)[0]
    if len(active_idx) > n_active:
        # Keep only top n_active
        active_idx = active_idx[np.argsort(-scores[active_idx])[:n_active]]
    elif len(active_idx) < 3:
        active_idx = np.argsort(-scores)[:n_active]
    
    # Compute gradients
    grads = _compute_grads(h, active_idx)  # shape (M, n)
    M = len(active_idx)
    
    # LP: minimize t
    # s.t. g_k · d <= t for k in active
    #      sum(d) = 0
    #      -delta <= d_i <= delta
    #      no lower bound on t (but implicitly t >= min(g_k · d))
    
    # Variables: d[0..n-1], t  -> total n+1 variables
    # Objective: minimize t -> c = [0,...,0, 1]
    
    # A_ub * x <= b_ub:
    #   For each active k: g[k,:] · d - t <= 0  ->  [g[k,0],...,g[k,n-1], -1] · x <= 0
    #   Box: d_i <= delta, -d_i <= delta
    # A_eq * x = b_eq:
    #   sum(d) = 0 -> [1,...,1, 0] · x = 0
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints
    A_ub_rows = []
    b_ub_vals = []
    
    # For each active row: g_k · d - t <= 0
    for k in range(M):
        row = np.zeros(n + 1)
        row[:n] = grads[k]
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(0.0)
    
    # Box constraints on d
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_vals.append(delta)
        
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(delta)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: t is unbounded
    bounds = [(None, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    d = result.x[:n]
    # Ensure zero sum
    d -= np.mean(d)
    
    # Check if direction is useful (t < 0 means all active peaks decrease)
    t_val = result.x[n]
    if t_val >= -1e-10:
        return None  # No improvement possible
    
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _random_restarts(h, deadline, rng, n_starts=3):
    """Try small perturbations and optimize each."""
    n = len(h)
    target = n / 2.0
    best_c5 = c5_exact(h)
    best_h = h.copy()
    
    for s in range(n_starts):
        if time.time() > deadline:
            break
        h_pert = h + rng.uniform(-0.02, 0.02, size=n)
        h_pert = project(h_pert, target)
        sub_dl = min(deadline, time.time() + 30)
        h_opt, c5_opt = _subgradient_polish(h_pert, sub_dl, n_active=8)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

            active_idx = np.argsort(-scores)[:8]
        grads = _compute_grads(h, active_idx)
        weight = np.exp((scores[active_idx] - max_score) / (0.01 * max_score + 1e-15))
        weight /= float(np.sum(weight))
        wgrad = grads.T @ weight
        # Best donor (highest wgrad) and two best receivers (lowest wgrad)
        i = np.argmax(wgrad)
        recv_candidates = np.argsort(wgrad)[:min(20, n)]
        recv_candidates = [r for r in recv_candidates if r != i]
        if len(recv_candidates) < 2:
            stall += 5
            continue
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        for j in recv_candidates[:10]:
            if time.time() > deadline:
                break
            for k in recv_candidates[:10]:
                if k == j or k == i or j == i:
                    continue
                if j > k:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 0.5
                d[k] = 0.5
                opt, new_c5 = _exact_line_search(h, d, dx, S0, max_alpha=0.3)
                if opt > 1e-10 and new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = opt
                    best_pair = (i, j, k)
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j, k = best_pair
            h[i] -= best_alpha
            h[j] += 0.5 * best_alpha
            h[k] += 0.5 * best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _multi_level_optimize(h, deadline):
    """Optimize at coarser resolution, interpolate back, refine."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = c5_exact(h)
    
    # Try coarser resolutions
    for factor in [4, 3]:
        if time.time() > deadline:
            break
        n_coarse = n // factor
        if n_coarse < 20 or n_coarse % 2 != 0:
            continue
        # Downsample by averaging
        h_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            start = i * factor
            end = min(start + factor, n)
            h_coarse[i] = np.mean(h[start:end])
        h_coarse = project(h_coarse, n_coarse / 2.0)
        
        # Optimize at coarse level
        sub_dl = min(deadline, time.time() + budget_s * 0.02)
        h_coarse, c5_coarse = _subgradient_polish(h_coarse, sub_dl, n_active=6)
        
        # Interpolate back to fine level
        h_fine = _resample(h_coarse, n)
        c5_fine = c5_exact(h_fine)
        
        # Refine
        if c5_fine < best_c5 + 0.01:
            sub_dl2 = min(deadline, time.time() + budget_s * 0.015)
            h_fine, c5_fine = _subgradient_polish(h_fine, sub_dl2, n_active=8)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s_global
    budget_s_global = budget_s
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97
    
    # Initialize from parent if available
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.25 * np.sin(2 * np.pi * 2 * x / n0) + 0.1 * np.sin(2 * np.pi * 5 * x / n0 + 1.0)
    
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0
    
    # Phase 1: LP direction search (20% time)
    d1 = t0 + budget_s * 0.20
    h_p1, c5_p1 = _lp_direction_search(h, d1, n_active=12, delta=0.04)
    if c5_p1 < best_c5:
        best_c5 = c5_p1
        best_h = h_p1.copy()
    
    # Phase 2: Subgradient polish with momentum (15% time)
    d2 = t0 + budget_s * 0.35
    h_p2, c5_p2 = _subgradient_polish(best_h, d2, n_active=10)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()
    
    # Phase 3: Directed pair search (15% time)
    d3 = t0 + budget_s * 0.50
    h_p3, c5_p3 = _directed_pair_search(best_h, 20, d3)
    if c5_p3 < best_c5:
        best_c5 = c5_p3
        best_h = h_p3.copy()
    
    # Phase 4: LP direction search round 2 (12% time)
    d4 = t0 + budget_s * 0.62
    h_p4, c5_p4 = _lp_direction_search(best_h, d4, n_active=8, delta=0.02)
    if c5_p4 < best_c5:
        best_c5 = c5_p4
        best_h = h_p4.copy()
    
    # Phase 5: Triplet search (8% time)
    d5 = t0 + budget_s * 0.70
    h_p5, c5_p5 = _triplet_search(best_h, d5, rng)
    if c5_p5 < best_c5:
        best_c5 = c5_p5
        best_h = h_p5.copy()
    
    # Phase 6: Multi-resolution (10% time)
    d6 = t0 + budget_s * 0.80
    h_p6, c5_p6 = _multi_level_optimize(best_h, d6)
    if c5_p6 < best_c5:
        best_c5 = c5_p6
        best_h = h_p6.copy()
    
    # Phase 7: Try different grid sizes (8% time)
    d7 = t0 + budget_s * 0.88
    offsets = [4, -4, 6, -6, 8, -8, 2, -2, 12, -12, 16, -16]
    tried_n = {best_n}
    for dn in offsets:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = c5_exact(h_r)
        if c5_r < best_c5 + 0.003:
            sub_dl = min(d7, time.time() + budget_s * 0.012)
            h_r, c5_r = _subgradient_polish(h_r, sub_dl, n_active=8)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    
    # Phase 8: Final LP + subgradient (6% time)
    d8 = t0 + budget_s * 0.94
    h_p8, c5_p8 = _lp_direction_search(best_h, d8, n_active=10, delta=0.015)
    if c5_p8 < best_c5:
        best_c5 = c5_p8
        best_h = h_p8.copy()
    
    # Phase 9: Final directed pair polish (3% time)
    d9 = t0 + budget_s * 0.97
    h_p9, c5_p9 = _directed_pair_search(best_h, 30, d9)
    if c5_p9 < best_c5:
        best_c5 = c5_p9
        best_h = h_p9.copy()
    
    # Final subgradient with remaining
    h_f, c5_f = _subgradient_polish(best_h, deadline, n_active=8)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    # Ensure validity
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    
    return best_h, final_c5, len(best_h)
