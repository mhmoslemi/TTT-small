import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_row_weights(h, target, tau):
    """Compute softmax weights for each correlation row."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has length 2n-1, indexed k=0..2n-2
    # d = k - (n-1)
    z = (corr - target) / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / np.sum(exp_z)
    return w, corr, dx

def smooth_obj_grad(h, target, tau):
    """Compute smooth objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = (corr - target) / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / np.sum(exp_z)
    obj = tau * (np.log(np.sum(exp_z)) + z_max)
    
    # Compute gradient
    grad = np.zeros(n)
    # For each lag d = k - (n-1), add w[k] * grad_R_d
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        if d >= 0:
            # R_d = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # grad[j] += wk * (1-h[j-d]) for j in [d, n-1]
            # grad[j] += wk * (-h[j+d]) for j in [0, n-1-d]
            grad[d:] += wk * (1.0 - h[:n-d])

            grad[:n-r] += wk * (1.0 - h[r:])
            grad[r:] += wk * (-h[:n-r])
    
    return obj, grad

def make_fourier_start(n, freqs, amps, phases):

    T = np.correlate(direction, 1.0 - h, mode='full') - np.correlate(h, direction, mode='full')

    pos = direction > 1e-12
    neg = direction < -1e-12
    hi = np.inf
    lo = 0.0
    if np.any(pos):

    if np.any(neg):

    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    
    def f(alpha):
        return float(np.max(S0 + alpha * T - alpha * alpha * C) * dx)

    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)

            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    opt = (a + b) / 2.0
    val = f(opt)
    # Check endpoints
    f_lo = f(lo)
    f_hi = f(hi)
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
