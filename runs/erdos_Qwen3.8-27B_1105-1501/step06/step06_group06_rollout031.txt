import numpy as np
from scipy.optimize import minimize
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def soft_max_direction(h, gamma):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    # Soft-max weights
    log_w = gamma * (corr - M)
    log_w -= np.max(log_w)  # numerical stability
    w = np.exp(log_w)
    p = w / np.sum(w)
    
    # Compute direction: d[j] = -sum_s p[s] * dx * dc_s/dh[j]
    # dc_s/dh[j] = (1-h[j+s]) if valid - h[j-s] if valid
    # where s = index - (n-1)
    d = np.zeros(n)
    for idx in range(2 * n - 1):
        if p[idx] < 1e-12:
            continue
        s = idx - (n - 1)
        w_p = p[idx] * dx
        # For each j, dc/dh[j] = (1-h[j+s]) if 0<=j+s<n minus h[j-s] if 0<=j-s<n
        # First term: (1-h[j+s]) for j where 0<=j+s<n
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_lo < j_hi:
            d[j_lo:j_hi] += w_p * (1.0 - h[j_lo + s:j_hi + s])
        # Second term: -h[j-s] for j where 0<=j-s<n
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_lo2 < j_hi2:
            d[j_lo2:j_hi2] -= w_p * h[j_lo2 - s:j_hi2 - s]
    
    d = -d  # descent direction
    d -= np.mean(d)  # make sum zero
    return d

def line_search(h, d):
    n = len(h)
    dx = 2.0 / n
    # Feasible alpha range (assuming alpha >= 0)
    pos = d > 1e-14
    neg = d < -1e-14
    alpha_max = np.inf
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(-h[neg] / d[neg])))
    alpha_max = max(alpha_max, 1e-14)
    
    # c(alpha) = c0 + alpha*L + alpha^2*Q (all raw, multiply by dx at end)
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    
    # Grid search
    n_grid = 80
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = c0[None, :] + alphas[:, None] * L[None, :] + alphas[:, None]**2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_alpha = alphas[best_i]
    best_val = obj[best_i]
    
    # Golden section refinement
    if 0 < best_i < n_grid - 1:
        a = alphas[best_i - 1]
        b = alphas[best_i + 1]
    elif best_i == 0:
        a = 0.0
        b = alphas[1]
    else:
        a = alphas[-2]
        b = alphas[-1]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    f_a = float(np.max(c0 + a * L + a*a * Q)) * dx
    f_b = float(np.max(c0 + b * L + b*b * Q)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f_a < f_b:
            b = a + gr * (b - a)
            f_b = float(np.max(c0 + b * L + b*b * Q)) * dx
        else:
            a = b - gr * (b - a)
            f_a = float(np.max(c0 + a * L + a*a * Q)) * dx
    mid = (a + b) / 2.0
    f_mid = float(np.max(c0 + mid * L + mid*mid * Q)) * dx
    if f_mid < best_val:
        best_alpha = mid
        best_val = f_mid
    
    return best_alpha, best_val

def random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h

def threshold_binary(h, n):
    # Convert to binary preserving exactly n/2 ones
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:n//2]] = 1.0
    return h_b

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def soft_descent(h, deadline, gamma_start=10.0, gamma_end=5000.0, n_gamma_steps=12, max_iters_per_gamma=60):
    n = len(h)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    gammas = np.geomspace(gamma_start, gamma_end, n_gamma_steps)
    
    for g in gammas:
        if time.time() > deadline:
            break
        for it in range(max_iters_per_gamma):
            if time.time() > deadline:
                break
            d = soft_max_direction(h, float(g))
            if np.max(np.abs(d)) < 1e-14:
                stall += 10
                continue
            d_norm = np.sum(np.abs(d))
            if d_norm < 1e-14:
                stall += 10
                continue
            alpha, new_val = line_search(h, d)
            if new_val < cur_c5 - 1e-14:
                h_new = h + alpha * d
                h_new = project(h_new, n / 2.0)
                actual_val = exact_c5(h_new)
                if actual_val < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = actual_val
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            if stall > 15:
                break
        stall = 0  # reset at each gamma
    
    return best_h, best_c5

def pair_direction_polish(h, deadline, max_pairs=2000):
    n = len(h)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    stall = 0
    
    for _ in range(max_pairs):
        if time.time() > deadline:
            break
        # Pick a random pair (i, j) with i != j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Direction: transfer from i to j (if h[i] > h[j]) or vice versa
        if h[i] >= h[j]:
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        
        alpha, new_val = line_search(h, d)
        if new_val < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = project(h_new, n / 2.0)
            actual_val = exact_c5(h_new)
            if actual_val < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual_val
                if cur_c5 < best_c5:

                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

def coordinate_descent(h, deadline, max_sweeps=20):
    n = len(h)
    dx = 2.0 / n
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            # Compute slopes for each lag
            c0 = np.correlate(h, 1.0 - h, mode='full')
            # Slope of c[idx] with respect to h[j]
            slopes = np.zeros(2*n-1)
            for idx in range(2*n-1):
                s = idx - (n-1)
                slope = 0.0

                # Valid first index range: max(0,-s) to min(n-1, n-1-s)
                if max(0, -s) <= j <= min(n-1, n-1-s):

            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_dir(h, gamma):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    log_w = gamma * (corr - M)
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    p = w / np.sum(w)
    d = np.zeros(n)
    for idx in range(2 * n - 1):
        if p[idx] < 1e-10:
            continue
        s = idx - (n - 1)
        w_p = p[idx] * dx
        j1 = max(0, -s)
        j2 = min(n, n - s)
        if j1 < j2:
            d[j1:j2] += w_p * (1.0 - h[j1+s:j2+s])
        j3 = max(0, s)
        j4 = min(n, n + s)
        if j3 < j4:
            d[j3:j4] -= w_p * h[j3-s:j4-s]
    d = -d
    d -= np.mean(d)
    return d

def _line_search(h, d):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-14
    neg = d < -1e-14
    alpha_max = 1.0
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min(-h[neg] / d[neg])))
    alpha_max = max(alpha_max, 1e-15)
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    n_grid = 64
    alphas = np.linspace(0.0, alpha_max, n_grid)
    vals = c0[None, :] + alphas[:, None] * L[None, :] + alphas[:, None]**2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_alpha = float(alphas[best_i])
    best_val = float(obj[best_i])
    if 0 < best_i < n_grid - 1:
        a = float(alphas[best_i - 1])
        b = float(alphas[best_i + 1])
    elif best_i == 0:
        a = 0.0
        b = float(alphas[1])
    else:
        a = float(alphas[-2])
        b = float(alphas[-1])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    fa = float(np.max(c0 + a * L + a * a * Q)) * dx
    fb = float(np.max(c0 + b * L + b * b * Q)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if fa < fb:
            b = a + gr * (b - a)
            fb = float(np.max(c0 + b * L + b * b * Q)) * dx
        else:
            a = b - gr * (b - a)
            fa = float(np.max(c0 + a * L + a * a * Q)) * dx
    mid = (a + b) / 2.0
    fm = float(np.max(c0 + mid * L + mid * mid * Q)) * dx
    if fm < best_val:
        best_alpha = mid
        best_val = fm
    return best_alpha, best_val

def _soft_descent(h, deadline, g_lo=5.0, g_hi=5000.0, n_g=10, iters=50):
    n = len(h)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    gammas = np.geomspace(g_lo, g_hi, n_g)
    for g in gammas:
        stall = 0
        for _ in range(iters):
            if time.time() > deadline:
                break
            d = _soft_dir(h, float(g))
            dn = np.max(np.abs(d))
            if dn < 1e-14:
                stall += 10
                break
            alpha, nv = _line_search(h, d)
            if nv < cur - 1e-14:
                h_new = h + alpha * d
                h_new = _project(h_new, n / 2.0)
                av = _exact_c5(h_new)
                if av < cur - 1e-14:
                    h = h_new
                    cur = av
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            if stall > 12:
                break
    return best_h, best_c5

def _coord_descent(h, deadline, max_sweeps=15):
    n = len(h)
    dx = 2.0 / n
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        c0 = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            slopes = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                s = idx - (n - 1)
                sl = 0.0
                if 0 <= j + s < n and max(0, -s) <= j <= min(n - 1, n - 1 - s):
                    sl += (1.0 - h[j + s])
                if 0 <= j - s < n and max(0, -s) <= (j - s) <= min(n - 1, n - 1 - s):
                    sl -= h[j - s]
                slopes[idx] = sl
            t_vals = np.linspace(0.0, 1.0, 40)
            vals = c0[None, :] + t_vals[:, None] * slopes[None, :]
            obj = np.max(vals, axis=1) * dx
            bi = int(np.argmin(obj))
            if 0 < bi < 39:
                a = float(t_vals[bi - 1])
                b = float(t_vals[bi + 1])
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                fa = float(np.max(c0 + a * slopes)) * dx
                fb = float(np.max(c0 + b * slopes)) * dx
                for _ in range(30):
                    if b - a < 1e-13:
                        break
                    if fa < fb:
                        b = a + gr * (b - a)
                        fb = float(np.max(c0 + b * slopes)) * dx
                    else:
                        a = b - gr * (b - a)
                        fa = float(np.max(c0 + a * slopes)) * dx
                tb = (a + b) / 2.0
            else:
                tb = float(t_vals[bi])
            nv = float(np.max(c0 + tb * slopes)) * dx
            if nv < cur - 1e-14:
                h = h.copy()
                h[j] = tb
                cur = nv
                improved = True
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                c0 = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    return best_h, best_c5

def _binary_local_search(h, deadline, max_iters=80000):
    n = len(h)
    order = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[order[:n // 2]] = 1.0
    cur = _exact_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur
    rng = np.random.default_rng(77)
    ones = np.where(h_b == 1.0)[0]
    zeros = np.where(h_b == 0.0)[0]
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = int(ones[rng.integers(0, len(ones))])
        j = int(zeros[rng.integers(0, len(zeros))])
        if i == j:
            continue
        h_b[i] = 0.0
        h_b[j] = 1.0
        nv = _exact_c5(h_b)
        if nv < cur - 1e-14:
            cur = nv
            ones = np.where(h_b == 1.0)[0]
            zeros = np.where(h_b == 0.0)[0]
            if cur < best_c5:
                best_c5 = cur
                best_h = h_b.copy()
            stall = 0
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
            stall += 1
        if stall > 300:
            stall = 0
    return best_h, best_c5

def _pair_polish(h, deadline, max_iters=5000):
    n = len(h)
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    rng = np.random.default_rng(2024)
    stall = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        if h[i] >= h[j]:
            d[i] = -1.0
            d[j] = 1.0
        else:
            d[i] = 1.0
            d[j] = -1.0
        alpha, nv = _line_search(h, d)
        if nv < cur - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, n / 2.0)
            av = _exact_c5(h_new)
            if av < cur - 1e-14:
                h = h_new
                cur = av
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 300:
            break
    return best_h, best_c5

def _resample(h, nn):
    n = len(h)
    if nn == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(nn) + 0.5) / nn
    hn = np.interp(xn, xo, h)
    return _project(hn, nn / 2.0)

def _rand_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def _structured_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 4 + 1))))
        if rng.random() < 0.55:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _rand_binary(n0, rng)

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Binary local search from thresholded parent (15%)
    d1 = t0 + budget_s * 0.15
    hb, c5b = _binary_local_search(h, d1, max_iters=40000)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()

    # Phase 2: Soft descent from best (20%)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _soft_descent(best_h, d2, g_lo=5, g_hi=3000, n_g=8, iters=45)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: Coordinate descent polish (10%)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _coord_descent(best_h, d3, max_sweeps=12)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair polish (10%)
    d4 = t0 + budget_s * 0.55
    h4, c5_4 = _pair_polish(best_h, d4, max_iters=4000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (20%)
    d5 = t0 + budget_s * 0.75
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        nt = best_n + dn
        if nt < 60 or nt > 200 or nt % 2 != 0:
            continue
        if nt in tried:
            continue
        tried.add(nt)
        hr = _resample(best_h, nt)
        # Soft descent
        sd = min(d5, time.time() + budget_s * 0.012)
        hr, cr = _soft_descent(hr, sd, g_lo=10, g_hi=2000, n_g=6, iters=30)
        # Coord descent
        if time.time() < d5 - 3:
            cd = min(d5, time.time() + budget_s * 0.008)
            hr, cr = _coord_descent(hr, cd, max_sweeps=6)
        # Pair polish
        if time.time() < d5 - 2:
            pd = min(d5, time.time() + budget_s * 0.005)
            hr, cr = _pair_polish(hr, pd, max_iters=1000)
        if cr < best_c5 - 1e-13:
            best_c5 = cr
            best_h = hr.copy()
            best_n = nt

    # Phase 6: Diverse starts (10%)
    d6 = t0 + budget_s * 0.85
    starts = []
    for _ in range(3):
        starts.append(_rand_binary(best_n, rng))
    for _ in range(2):
        starts.append(_structured_init(best_n, rng))
    order_idx = np.argsort(-best_h)
    hb_init = np.zeros(best_n)
    hb_init[order_idx[:best_n // 2]] = 1.0
    starts.append(hb_init)
    for s in starts:
        if time.time() > d6:
            break
        sd = min(d6, time.time() + budget_s * 0.012)
        hs, cs = _soft_descent(s, sd, g_lo=5, g_hi=2000, n_g=6, iters=30)
        if time.time() < d6 - 2:
            cd = min(d6, time.time() + budget_s * 0.006)
            hs, cs = _coord_descent(hs, cd, max_sweeps=5)
        if cs < best_c5 - 1e-13:
            best_c5 = cs
            best_h = hs.copy()
            best_n = len(hs)

    # Phase 7: Final polish with high gamma (15%)
    d7 = deadline - 5
    h7, c5_7 = _soft_descent(best_h, min(d7, time.time() + budget_s * 0.06), g_lo=100, g_hi=50000, n_g=6, iters=30)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7 - 3:
        h7b, c5_7b = _coord_descent(best_h, min(d7, time.time() + budget_s * 0.04), max_sweeps=8)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()
    if time.time() < d7 - 2:
        h7c, c5_7c = _pair_polish(best_h, d7, max_iters=2000)
        if c5_7c < best_c5:
            best_c5 = c5_7c
            best_h = h7c.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
