    If constraint_idx is None, use all constraints.
    Otherwise, use only constraints at the given indices.
    """
    target = n / 2.0
    nc = len(constraint_idx) if constraint_idx is not None else (2 * n - 1)
    
    if constraint_idx is None:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.array(constraint_idx)
    
    def obj(x):
        return x[-1]

        h = x[:-1]
        z = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return z - corr[idx] * dx

        h = x[:-1]
        G = _all_grads(h, n)
        J = -dx * G[idx, :]
        J = np.hstack([J, np.ones((nc, 1))])
        return J

        return np.array([np.sum(x[:-1]) - target])
    
    def eq_jac(x):
        row = np.zeros(n + 1)
        row[:n] = 1.0
        return row.reshape(1, n + 1)

    bounds = [(0.0, 1.0)] * n + [(z0 * 0.9, 1.0)]

        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},

                          constraints=constraints,

        f_sol = _exact_c5(h_sol)
        if f_sol <= z0 + 1e-6:
            return h_sol, f_sol
    return h0, _exact_c5(h0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(6):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _all_grads(h, n):
    grads = np.zeros((2 * n - 1, n))
    for k in range(n):
        r = n - 1 + k
        g = np.zeros(n)
        g[k:] = 1.0 - h[:n - k]
        g[:n - k] += -h[k:]
        grads[r] = g
    for q in range(1, n):
        r = n - 1 - q
        g = np.zeros(n)
        g[:n - q] = 1.0 - h[q:]
        g[q:] += -h[:n - q]
        grads[r] = g
    return grads


def _slsqp_solve(h0, n, dx, z0, constraint_idx=None, maxiter=300):
    target = n / 2.0
    if constraint_idx is None:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.asarray(constraint_idx, dtype=int)
    nc = len(idx)

    def obj(x):
        return x[-1]

    def ineq_fun(x):
        h = x[:-1]
        z = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return z - corr[idx] * dx

    def ineq_jac(x):
        h = x[:-1]
        G = _all_grads(h, n)
        J = -dx * G[idx, :]
        return np.hstack([J, np.ones((nc, 1))])

    def eq_fun(x):
        return np.array([np.sum(x[:-1]) - target])

    def eq_jac(x):
        row = np.zeros(n + 1)
        row[:n] = 1.0
        return row.reshape(1, n + 1)

    x0 = np.concatenate([h0, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z0 + 0.05)]

    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]

    try:
        result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,

        return h0, _exact_c5(h0)

        if f_sol <= z0 + 1e-8:

    return h0, _exact_c5(h0)


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(40):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv


def _active_direction(h, corr, k_active, tau, rng):
    n = len(h)
    scores = corr * (2.0 / n)
    m = float(np.max(scores))
    w = np.exp((scores - m) / tau)
    idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = r - (n - 1)
        if k >= 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] += -h[k:]
        else:
            q = -k
            g = np.zeros(n)
            g[:n - q] += (1.0 - h[q:])
            g[q:] += -h[:n - q]
        G += w[r] * g
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm
