import numpy as np
from scipy.optimize import linprog
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_matrix(b, dx):
    """Build A where A[q, i] = dx * corr(e_i, b, full)[q]"""
    n = len(b)
    m = 2 * n - 1
    A = np.zeros((m, n))
    # A[q, i] = dx * b[i + q - (n-1)] if 0 <= i + q - (n-1) < n
    for i in range(n):
        # For what q is i + q - (n-1) in [0, n-1]?
        # 0 <= i + q - (n-1) <= n-1
        # n-1-i <= q <= 2n-2-i
        q_lo = max(0, n - 1 - i)
        q_hi = min(2 * n - 2, 2 * n - 2 - i)
        for q in range(q_lo, q_hi + 1):
            j = i + q - (n - 1)
            if 0 <= j < n:
                A[q, i] = dx * b[j]
    return A

```python
def _solve_lp(A, n, target):
    """min tau s.t. A@x - tau <= 0, sum(x) = target, 0<=x<=1, tau>=0"""
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and np.all(np.isfinite(res.x)):
        return res.x[:n]
    return None
```

import numpy as np
from scipy.optimize import linprog
import time

def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A1(b, dx):
    n = len(b)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for q in range(m):
        i_lo = max(0, n - 1 - q)
        i_hi = min(n - 1, 2 * n - 2 - q)
        if i_lo <= i_hi:
            j_lo = i_lo + q - (n - 1)
            A[q, i_lo:i_hi + 1] = dx * b[j_lo:j_lo + (i_hi - i_lo + 1)]
    return A

def _build_A2(a, dx):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for q in range(m):
        j_lo = max(0, q - (n - 1))
        j_hi = min(n - 1, q)
        if j_lo <= j_hi:
            a_lo = j_lo - q + (n - 1)
            A[q, j_lo:j_hi + 1] = dx * a[a_lo:a_lo + (j_hi - j_lo + 1)]
    return A

def _solve_lp(A, n, target):
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            x = res.x[:n]
            return _project(x, target)
    except Exception:
        pass
    return None

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    n_grid = 36
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1)
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a_g, b_g = grid[best_i - 1], grid[best_i + 1]
    else:
        a_g, b_g = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b_g - gr * (b_g - a_g)
    c2 = a_g + gr * (b_g - a_g)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(30):
        if b_g - a_g < 1e-13:
            break
        if f1 < f2:
            b_g, c2, f2 = c2, c1, f1
            c1 = b_g - gr * (b_g - a_g)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a_g, c1, f1 = c1, c2, f2
            c2 = a_g + gr * (b_g - a_g)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a_g + b_g) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

                                     start_radius=0.08, max_iter=30)

                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try
    
    # Phase 4: Pair moves on best (10% budget)
    d4 = t0 + budget_s * 0.85
    h4, c5_4 = _pair_moves(best_h, d4, max_rounds=40)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Final local polish (remaining)
    d5 = deadline
    h5, c5_5 = _local_lp(best_h, d5, start_radius=0.05, max_iter=60)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _score(best_h)
    
    return best_h, final_c5, len(best_h)
