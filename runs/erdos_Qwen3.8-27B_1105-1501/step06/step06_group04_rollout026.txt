def _compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_row(h, r):
    """Compute gradient of corr_r(h) with respect to h."""
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        g_abs = -d
        m = n - g_abs
        if m > 0:
            g[:m] += (1.0 - h[g_abs:])
            g[g_abs:] += -h[:m]
    return g

    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1

    return G

def _lse_slsqp(h0, beta, maxiter=200, deadline=None):
    n = len(h0)
    target = n / 2.0
    m = 2 * n - 1
    
    def fun(y):
        h = np.clip(y, 0, 1)
        scores = _corr_scores(h)
        # Numerically stable LSE
        smax = float(scores.max())
        z = beta * (scores - smax)
        # Clip for numerical safety
        z = np.clip(z, -50, 50)
        logsum = smax + np.log(np.sum(np.exp(z))) / beta
        return float(logsum)
    
    def grad(y):
        h = np.clip(y, 0, 1)
        scores = _corr_scores(h)
        smax = float(scores.max())
        z = beta * (scores - smax)
        z = np.clip(z, -50, 50)
        p = np.exp(z)
        p /= p.sum()
        G = _all_grads_scaled(h)
        return p @ G
    
    # Bounds
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda y: np.sum(y) - target,
                   'jac': lambda: np.ones(n)}]
    
    try:
        res = minimize(fun, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,

        if not used:
            stall += 1
            d = _zero_sum(rng.normal(0, 0.1, n))
            d = _normalize_dir(d, 0.15)
            alpha, val = _line_search(h, d, 0.2)
            if val < best_c5 - 1e-14:
                h = h + alpha * d
                h = _proj(h, target)
                cur = _c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
    return best_h, best_c5

def _pair_refine(h, deadline, max_rounds=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    rng = np.random.default_rng(314)
    stall = 0
    interior = np.where((h > 0.01) & (h < 0.99))[0]
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 40 or len(interior) < 2:
            break
        improved = False
        for _ in range(100):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, val = _line_search(h, d, 0.4)
            if val < best_c5 - 1e-14:
                h = h + alpha * d
                h = _proj(h, target)
                cur = _c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    interior = np.where((h > 0.01) & (h < 0.99))[0]
                    if len(interior) < 2:
                        break
                    break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _proj(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return _proj(h, n / 2.0)

def _binary_quantize(h):
    n = len(h)
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def _proj_grad_step(h, grad, step_size):
    n = len(h)
    target = n / 2.0
    h_new = h - step_size * grad
    return _proj(h_new, target)

def _prograd_lse(h, beta, n_steps=200, deadline=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    lr = 0.05
    for i in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        s = _scores(h)
        smax = float(s.max())
        z = np.clip(beta * (s - smax), -40.0, 40.0)
        p = np.exp(z)
        p /= p.sum()
        G = _gradmat(h)
        grad = p @ G
        h_new = _proj(h - lr * grad, target)
        c5_new = _c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < _c5(h):
            h = h_new.copy()
            lr = min(lr * 1.05, 0.3)
        else:
            lr = max(lr * 0.7, 0.005)
    return best_h, best_c5

def _anneal_lse(h0, betas, maxiter_each=100, deadline=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    for beta in betas:
        if deadline is not None and time.time() > deadline:
            break
        h_new, c5_new = _lse_solve(h, beta, maxiter=maxiter_each)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new.copy()
    return best_h, best_c5

def _multi_start_optimize(starter, deadline, betas, maxiter_each):
    n = len(starter)
    best_h = starter.copy()
    best_c5 = _c5(starter)
    for beta in betas:
        if time.time() > deadline:
            break
        h_new, c5_new = _lse_solve(best_h, beta, maxiter=maxiter_each)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    reserve = max(3.0, 0.02 * budget_s)
    hard_deadline = deadline - reserve
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = _sine_init(120, 3, 0.5)

    n0 = len(h0)
    h0 = _proj(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _c5(best_h)

    # Phase 1: Quick LSE with moderate beta (10% time)
    d1 = t0 + budget_s * 0.08
    betas_fast = [20, 50, 100]
    h1, c5_1 = _anneal_lse(best_h, betas_fast, maxiter_each=80, deadline=d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Full beta annealing (25% time)
    d2 = t0 + budget_s * 0.33
    betas_full = [15, 40, 80, 150, 300, 500, 800, 1200]
    h2, c5_2 = _anneal_lse(best_h, betas_full, maxiter_each=120, deadline=d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Line polish (15% time)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _line_polish(best_h, d3, max_rounds=800)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Minimax SLSQP (10% time)
    d4 = t0 + budget_s * 0.58
    h4, c5_4 = _minimax_solve(best_h, maxiter=120)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4:
        h4b, c5_4b = _lse_solve(best_h, 500, maxiter=100)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-start with diverse initializations (15% time)
    d5 = t0 + budget_s * 0.73
    n = len(best_h)
    starters = []
    for freq in [2, 3, 4, 5, 7]:
        for ph in [0.0, 0.7, 1.5, 2.5]:
            starters.append(_sine_init(n, freq, ph))
    for p in [5, 6, 7, 8, 11, 13]:
        for duty in [0.35, 0.45, 0.5, 0.55, 0.65]:
            starters.append(_periodic_init(n, p, duty))
    starters.append(_binary_quantize(best_h))
    starters.append(best_h + rng.normal(0, 0.02, n))
    starters = [s for s in starters if len(s) == n]
    betas_quick = [50, 200, 600]
    per_starter = max(2, int((d5 - time.time()) / max(len(starters), 1)))
    for st in starters:
        if time.time() > d5:
            break
        st = _proj(st, n / 2.0)
        c5_st = _c5(st)
        if c5_st > best_c5 + 0.05:
            continue
        sub_dl = min(d5, time.time() + per_starter)
        h_s, c5_s = _multi_start_optimize(st, sub_dl, betas_quick, maxiter_each=50)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Multi-resolution (12% time)
    d6 = t0 + budget_s * 0.85
    cur_n = len(best_h)
    offsets = [-6, 6, -4, 4, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, -30, 30, -40, 40]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = cur_n + dn
        if n_try < 40 or n_try > 300:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + max(1.0, budget_s * 0.012))
        h_r, c5_r = _multi_start_optimize(h_r, sub_dl, [50, 200], maxiter_each=60)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            cur_n = n_try

    # Phase 7: Pair refine + final line polish (9% time)
    d7 = t0 + budget_s * 0.94
    h7, c5_7 = _pair_refine(best_h, min(d7, time.time() + budget_s * 0.04), max_rounds=300)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7:
        h7b, c5_7b = _line_polish(best_h, d7, max_rounds=400)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Phase 8: Final SLSQP polish (6% time)
    h_f, c5_f = _lse_solve(best_h, 800, maxiter=100)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < hard_deadline:
        h_f2, c5_f2 = _minimax_solve(best_h, maxiter=80)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final validation
    best_h = _proj(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            hf = np.array(initial_h_values, dtype=float).copy()
            hf = _proj(hf, len(hf) / 2.0)
            return hf, _c5(hf), len(hf)
        except NameError:
            hf = _sine_init(120, 3, 0.0)
            return hf, _c5(hf), len(hf)
    return best_h, final_c5, len(best_h)
