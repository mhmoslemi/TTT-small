def softmax_descent(h, deadline, max_iters=200):
    """Run soft-max annealing descent."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    gamma_schedule = [5, 15, 50, 200, 1000, 5000, 30000]
    
    for gamma in gamma_schedule:
        if time.time() > deadline:
            break
        stall = 0
        for it in range(max_iters):
            if time.time() > deadline:
                break
            if stall > 30:
                break
            
            # Compute direction
            d = compute_direction(h, gamma)
            if d is None:
                break
            
            # Compute line search data
            c0 = np.correlate(h, 1.0 - h, mode='full')
            L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
            Q = -np.correlate(d, d, mode='full')
            
            alpha, new_val = exact_line_search(h, d, c0, L, Q, dx)
            
            if alpha != 0:
                h_new = h + alpha * d
                h_new = project_to_box_sum(h_new, target)
                new_c5 = exact_c5(h_new)
                if new_c5 < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = new_c5
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                else:
                    stall += 1
            else:
                stall += 1
            
            # Occasionally try random transport
            if stall > 5 and it % 3 == 0:
                rng_local = np.random.default_rng(hash((gamma, it)) & 0xFFFFFFFF)
                p = int(rng_local.integers(n))
                q = int(rng_local.integers(n))
                if p != q and h[q] > h[p]:
                    d_rand = np.zeros(n)
                    d_rand[p] = 1.0
                    d_rand[q] = -1.0
                    c0_r = np.correlate(h, 1.0 - h, mode='full')
                    L_r = np.correlate(d_rand, 1.0 - h, mode='full') - np.correlate(h, d_rand, mode='full')
                    Q_r = -np.correlate(d_rand, d_rand, mode='full')
                    alpha_r, val_r = exact_line_search(h, d_rand, c0_r, L_r, Q_r, dx)
                    if alpha_r != 0:
                        h_r = h + alpha_r * d_rand
                        h_r = project_to_box_sum(h_r, target)
                        c5_r = exact_c5(h_r)
                        if c5_r < cur_c5 - 1e-14:
                            h = h_r
                            cur_c5 = c5_r
                            stall = 0
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
    
    return best_h, best_c5

def transport_polish(h, deadline, max_iters=5000):
    """Polish using pair transport directions guided by soft-max potential."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(12345)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        # Compute soft-max potential for pair selection
        gamma_sel = 50
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        w = compute_softmax_weights(corr, gamma_sel)
        
        # Compute potential V[i] = sum_k w[k] * grad_k[i] * dx
        # This is the negative of what compute_direction does (before negation and normalization)
        offsets = np.arange(2*n-1) - (n-1)
        V = np.zeros(n)
        for idx in range(2*n-1):
            wk = w[idx]
            if wk < 1e-15:
                continue
            s = offsets[idx]
            lo1 = max(0, s)
            hi1 = min(n-1, s + n - 1)
            if lo1 <= hi1:
                V[lo1:hi1+1] += wk * (1.0 - h[np.arange(lo1,hi1+1) - s])
            lo2 = max(0, -s)
            hi2 = min(n-1, n-1-s)
            if lo2 <= hi2:
                V[lo2:hi2+1] -= wk * h[np.arange(lo2,hi2+1) + s]
        V *= dx
        
        # Select pairs: want V[p] - V[q] very negative means moving from q to p helps
        # i.e., we want large V[q] and small V[p] for direction e_p - e_q
        # Randomize: pick q from top-V, p from bottom-V
        
        n_cand = min(10, n // 2)
        top_q = np.argpartition(V, -n_cand)[-n_cand:]  # highest V
        bot_p = np.argpartition(V, n_cand - 1)[:n_cand - 1]  # lowest V
        
        # Pick one pair at random from candidates
        q_idx = top_q[int(rng.integers(n_cand))]
        p_idx = bot_p[int(rng.integers(n_cand))]
        
        # Also sometimes use purely random pair
        if rng.random() < 0.3:
            p_idx = int(rng.integers(n))
            q_idx = int(rng.integers(n))
        
        if p_idx == q_idx:
            stall += 1
            continue
        
        d = np.zeros(n)
        d[p_idx] = 1.0
        d[q_idx] = -1.0
        
        c0 = np.correlate(h, 1.0 - h, mode='full')
        L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = -np.correlate(d, d, mode='full')
        
        alpha, new_val = exact_line_search(h, d, c0, L, Q, dx)
        
        if alpha != 0:
            h_new = h + alpha * d
            h_new = project_to_box_sum(h_new, target)
            new_c5 = exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-15:
                h = h_new
                cur_c5 = new_c5
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
