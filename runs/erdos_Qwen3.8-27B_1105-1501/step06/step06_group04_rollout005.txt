def _smoothed_grad(h, beta, top_k=None):
    n = len(h)
    S = _corr_values(h)
    m = len(S)
    if top_k is not None and top_k < m:
        idx = np.argpartition(-S, top_k)[:top_k]
        idx2 = idx[S[idx] > S.min()]
        if len(idx2) == 0:
            idx2 = idx
        S_sub = S[idx2]
        Smax = np.max(S_sub)
        exp_S = np.exp(beta * (S_sub - Smax))
        Z = np.sum(exp_S)
        weights = exp_S / Z
        G = _grad_all(h)
        grad = G[idx2].T @ weights
        val = (Smax + np.log(Z)) / beta
    else:
        Smax = np.max(S)
        exp_S = np.exp(beta * (S - Smax))
        Z = np.sum(exp_S)
        weights = exp_S / Z
        G = _grad_all(h)
        grad = G.T @ weights
        val = (Smax + np.log(Z)) / beta
    return val, grad

def _anneal_optimize(h, target, deadline, beta_start=3.0, beta_end=300.0, n_betas=30, top_k=None):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    momentum = 0.0
    v = np.zeros(n)
    for bi, beta in enumerate(betas):
        if time.time() > deadline:
            break
        momentum = min(0.92, 0.7 + 0.2 * bi / n_betas)
        lr = 0.03
        sub_iter = 0
        sub_dl = min(deadline, time.time() + max(1.0, (deadline - time.time()) / (n_betas - bi)))
        stall = 0
        while sub_iter < 400 and time.time() < sub_dl:
            sub_iter += 1
            val, grad = _smoothed_grad(h, beta, top_k=top_k)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            v = momentum * v + grad
            h_new = h - lr * v
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                stall = 0
                lr = min(lr * 1.02, 0.15)
            else:
                h = h_new
                stall += 1
                if stall > 30:
                    lr = lr * 0.7
                    v = v * 0.5
                    stall = 0
                if lr < 0.001:
                    break
        if best_c5 < 0.36:
            break
    return best_h, best_c5

def _topk_focused_step(h, target, radius, k=8):
    n = len(h)
    S = _corr_values(h)
    m = len(S)
    idx = np.argpartition(-S, min(k, m) - 1)[:min(k, m)]
    G = _grad_all(h)
    G_sub = G[idx]
    S_sub = S[idx]
    # Find descent direction from average of top-k gradients
    avg_g = G_sub.mean(axis=0)
    avg_g -= avg_g.mean()
    norm = np.linalg.norm(avg_g)
    if norm < 1e-14:
        return None, _exact_c5(h)
    d = -avg_g / norm
    # Clip to feasible directions
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        t_max = np.min((1.0 - h[pos]) / d[pos])
    else:
        t_max = radius
    if np.any(neg):
        t_min = max(np.max(h[neg] / (-d[neg])), -radius)
    else:
        t_min = -radius
    t_min = max(t_min, -radius)
    t_max = min(t_max, radius)
    if t_max - t_min < 1e-14:
        return None, _exact_c5(h)
    # Line search
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (G_sub.T @ np.ones(len(idx))) * 0  # placeholder
    # Recompute properly
    T_full = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C_full = np.correlate(d, d, mode='full')
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_min, t_max
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T_full - c1 * c1 * C_full)) * dx
    f2 = float(np.max(S0 + c2 * T_full - c2 * c2 * C_full)) * dx
    for _ in range(50):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T_full - c1 * c1 * C_full)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T_full - c2 * c2 * C_full)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_full - opt_t * opt_t * C_full)) * dx
    if opt_val < _exact_c5(h) - 1e-14:
        return d, opt_val
    return None, _exact_c5(h)

def _lp_direction(h, scores, radius, top_k=12):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = np.zeros((K, n))
    for idx_pos, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            mlen = n - d
            if mlen > 0:
                G[idx_pos, d:] += (1.0 - h[:mlen])
                G[idx_pos, :mlen] += -h[d:]
        else:
            g = -d
            mlen = n - g
            if mlen > 0:
                G[idx_pos, :mlen] += (1.0 - h[g:])
                G[idx_pos, g:] += -h[:mlen]
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_dir(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_loop(h, deadline, radius_init=0.1, top_k=12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(314)
    while time.time() < deadline:
        scores = _corr_values(h)
        d = _lp_direction(h, scores, radius, top_k=top_k)
        if d is not None and np.max(np.abs(d)) > 1e-13:
            alpha, new_c5 = _line_search_dir(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                radius = min(radius * 1.04, 0.3)
                continue
        # Top-k focused step
        d2, val2 = _topk_focused_step(h, target, radius, k=10)
        if d2 is not None and val2 < cur_c5 - 1e-14:
            alpha, new_c5 = _line_search_dir(h, d2, dx, radius)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d2
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                continue
        stall += 1
        radius *= 0.88
        if radius < 0.004:
            radius = 0.08
            h = h + rng.normal(0, 0.01, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
        if stall > 200:
            break
    return best_h, best_c5

def _binary_local_search(h, deadline, max_rounds=200):
    n = len(h)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(271)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 3000:
            break
        improved = False
        # Try random swaps
        n_try = 500
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            if new_c5 < cur_c5 - 1e-15:
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        if improved:
            continue
        stall += 1
        # Periodic perturbation
        if stall % 10 == 0 and len(ones) > 2 and len(zeros) > 2:
            for _ in range(3):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            cur_c5 = _exact_c5(h)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    stall = 0
    while time.time() < deadline and stall < 40:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        improved = False
        n_try = 300
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            for sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                alpha, c5_try = _line_search_dir(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        interior = np.where((h > 0.01) & (h < 0.99))[0]
                        if len(interior) < 2:
                            break
                        break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / n0) + 0.15 * np.sin(2 * np.pi * 7 * x / n0)
        h0 = _project(h0, n0 / 2.0)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)

    # Phase 1: Annealed smoothed gradient (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _anneal_optimize(best_h, n0 / 2.0, d1, beta_start=3.0, beta_end=200.0, n_betas=25, top_k=None)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Top-k focused annealing (15% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _anneal_optimize(best_h, n0 / 2.0, d2, beta_start=10.0, beta_end=400.0, n_betas=20, top_k=15)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP-based optimization (20% time)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.08, top_k=12)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary local search (15% time)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = _binary_local_search(best_h, d4, max_rounds=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Polish binary with continuous
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, min(d4, time.time() + budget_s * 0.03), radius_init=0.05, top_k=10)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-resolution (15% time)
    d5 = t0 + budget_s * 0.90
    n_cur = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_cur + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_ra, c5_ra = _anneal_optimize(h_r, n_try / 2.0, sub_dl, beta_start=10.0, beta_end=150.0, n_betas=15, top_k=12)
        if c5_ra < best_c5 - 1e-12:
            sub_dl2 = min(d5, time.time() + budget_s * 0.01)
            h_rb, c5_rb = _lp_loop(h_ra, sub_dl2, radius_init=0.05, top_k=10)
            if c5_rb < c5_ra:
                h_ra, c5_ra = h_rb, c5_rb
        if c5_ra < best_c5 - 1e-12:
            best_c5 = c5_ra
            best_h = h_ra.copy()

    # Phase 6: Final polish (10% time)
    d6 = deadline - max(2.0, budget_s * 0.02)
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=150)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6:
        h6b, c5_6b = _lp_loop(best_h, d6, radius_init=0.03, top_k=8)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0_fb = np.array(initial_h_values, dtype=float)
            h0_fb = _project(h0_fb, len(h0_fb) / 2.0)
            return h0_fb, _exact_c5(h0_fb), len(h0_fb)
        except NameError:
            n_fb = 122
            x_fb = np.arange(n_fb)
            h_fb = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x_fb / n_fb)
            h_fb = _project(h_fb, n_fb / 2.0)
            return h_fb, _exact_c5(h_fb), n_fb
    return best_h, final_c5, len(best_h)
