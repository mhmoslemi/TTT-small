import numpy as np
import time
from scipy.optimize import linprog

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_G(h):
    """Compute gradient matrix G of shape (2n-1, n) for all overlap rows."""
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for i in range(2*n - 1):
        d = i - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            # For k in [d, n-1]: G[i, k] += 1 - h[k-d]
            G[i, d:d+m] += 1.0 - h[:m]
            # For k in [0, m-1]: G[i, k] += -h[k+d]
            G[i, :m] += -h[d:d+m]

            m = n - q
            if m <= 0:

            # For k in [q, n-1]: G[i, k] += -h[k-q]
            G[i, q:q+m] += -h[:m]

def lp_descent(h, radius, row_filter=None):

    n = len(h)
    target = n / 2.0
    R = np.correlate(h, 1.0 - h, mode='full')
    G = compute_G(h)
    
    # For each row i: R_i(v) ≈ G[i,:] @ v + c[i] where c[i] = R[i] - G[i,:] @ h
    c = R - G @ h
    
    # LP: min t s.t. G[i,:] @ v - t <= -c[i], 0 <= v <= 1, sum(v) = n/2
    # Variable: x = [t, v_0, ..., v_{n-1}]
    
    # Row filtering: only include rows with R[i] >= threshold
    if row_filter is not None:
        max_R = np.max(R)
        thr = max_R - row_filter * max_R
        mask = R >= thr
        if np.sum(mask) < 5:
            mask = np.ones(len(R), dtype=bool)
        G_sel = G[mask]
        c_sel = c[mask]
    else:
        G_sel = G
        c_sel = c
    
    L = len(G_sel)
    # A_ub: L x (n+1), b_ub: L
    A_ub = np.zeros((L, n + 1))
    A_ub[:, :n] = G_sel
    A_ub[:, n] = -1.0
    b_ub = -c_sel  # = G[i,:]@h - R[i]
    
    # c_obj
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    # Equality: sum(v) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:n+1] = 1.0
    b_eq = np.array([target])
    
    # Bounds: t is free (or [some lower bound, inf]), v in [max(0, h-radius), min(1, h+radius)]
    v_lo = np.maximum(0.0, h - radius)
    v_hi = np.minimum(1.0, h + radius)
    
    bounds = [(None, None)] + list(zip(v_lo, v_hi))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:

        # Ensure feasibility

    return None

def gradient_direction(h, G, R, top_k=50):
    """Generate a weighted gradient direction from top active rows."""
    n = len(h)
    max_R = np.max(R)
    thr = max_R - 0.02 * max_R
    active = np.where(R >= thr)[0]
    if len(active) < top_k:
        active = np.argsort(-R)[:top_k]
    
    weight = np.exp((R[active] - max_R) / (0.01 * max_R + 1e-15))
    weight /= np.sum(weight)
    
    g = G[active].T @ weight
    g -= np.mean(g)  # zero sum for sum constraint
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return None
    return -g / gn

    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

            v = lp_solve(h_r, radius)
            if v is None:
                continue
            d = v - h_r
            d -= np.mean(d)
            if np.linalg.norm(d) < 1e-12:
                continue
            R0 = np.correlate(h_r, 1.0 - h_r, 'full')
            opt, val = line_search_exact(h_r, d, R0)
            if opt > 1e-10:
                h_r = h_r + opt * d
                h_r = project(h_r, n_try / 2.0)
                cur_r = c5_exact(h_r)
                if cur_r < best_c5 - 1e-12:
                    best_c5 = cur_r
                    best_h = h_r.copy()
                    best_n = n_try

    # Phase 4: Final fine polish (15% time)
    p4_deadline = deadline
    h = best_h.copy()
    n = len(h)
    target = n / 2.0
    stall = 0
    while time.time() < p4_deadline and stall < 20:
        improved = False
        for radius in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() >= p4_deadline:
                break
            v = lp_solve(h, radius, row_slack=0.005)
            if v is None:
                v = lp_solve(h, radius)
            if v is None:
                continue
            d = v - h
            d -= np.mean(d)
            if np.linalg.norm(d) < 1e-12:
                continue
            R0 = np.correlate(h, 1.0 - h, 'full')
            opt, val = line_search_exact(h, d, R0)
            if opt > 1e-10:
                h_new = h + opt * d
                h_new = project(h_new, target)
                val_new = c5_exact(h_new)
                if val_new < c5_exact(h) - 1e-12:
                    h = h_new
                    if val_new < best_c5 - 1e-12:
                        best_c5 = val_new
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            G = compute_G(h)
            R = np.correlate(h, 1.0 - h, 'full')
            d = gradient_dir(h, G, R)
            if d is not None:
                R0 = np.correlate(h, 1.0 - h, 'full')
                opt, val = line_search_exact(h, d, R0)
                if opt > 1e-10:
                    h_new = h + opt * d
                    h_new = project(h_new, target)
                    val_new = c5_exact(h_new)
                    if val_new < best_c5 - 1e-12:
                        h = h_new
                        best_c5 = val_new
                        best_h = h.copy()
                        improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break

    # Final pair polish with remaining time
    if time.time() < deadline - 10:
        h_pp, c5_pp = _pair_refine(best_h.copy(), deadline - 5, rng)
        if c5_pp < best_c5 - 1e-12:
            best_c5 = c5_pp
            best_h = h_pp.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
