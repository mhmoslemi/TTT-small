    """One step of dual ascent: compute gradient, line search, update weights."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    # Compute gradient of weighted sum
    g = _weighted_grad(h, w)
    
    # Project to sum-zero
    g -= g.mean()
    
    # Clip to [0,1] box (adjust direction)
    d = -g
    d -= d.mean()
    
    # Find feasible range
    pos = d > 1e-12
    neg = d < -1e-12
    L = 0.0
    U = np.inf

    eta = 0.02

    w_new /= w_new.sum()

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-15:
            continue
        k = r - (n - 1)

            g[:n - k] += wr * (1.0 - h[k:])
            g[k:] -= wr * h[:n - k]
        else:
            kk = -k
            g[kk:] += wr * (1.0 - h[:n - kk])
            g[:n - kk] -= wr * h[kk:]

    obj = np.max(vals, axis=1) * dx

        if f_new < best_f:

            best_h = h_new.copy()

        if f_new < cur_f + 0.0002:

            cur_f = f_new

            # Smooth with old weights
