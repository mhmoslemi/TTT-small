def _full_binary_sweep(h, corr):
    """Try all pair swaps and return the best one."""
    n = len(h)
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    
    n_ones = len(ones_idx)
    n_zeros = len(zeros_idx)
    
    # Compute effects
    s = np.arange(-(n-1), n)  # all shifts, shape (2n-1,)
    
    # For each one position i, effect of changing to 0:
    # effect[s] = h[i+s] (if valid) - (1-h[i-s]) (if valid)
    ones_effects = np.zeros((n_ones, 2*n-1))
    for a, i in enumerate(ones_idx):
        is_val = i + s
        is_neg = i - s
        valid_p = (is_val >= 0) & (is_val < n)
        valid_n = (is_neg >= 0) & (is_neg < n)
        ones_effects[a, valid_p] = h[is_val[valid_p]]
        ones_effects[a, valid_n] -= (1.0 - h[is_neg[valid_n]])
    
    # For each zero position j, effect of changing to 1:
    zeros_effects = np.zeros((n_zeros, 2*n-1))
    for b, j in enumerate(zeros_idx):
        js_val = j + s
        js_neg = j - s

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
    return h_b

def _binary_sweep(h, corr):
    """Full sweep: find best single swap. Returns (best_i, best_j, best_max_value) 
    or (None, None, cur_max) if no improvement."""
    n = len(h)
    dx = 2.0 / n
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]
    n_ones = len(ones_idx)
    n_zeros = len(zeros_idx)
    
    if n_ones == 0 or n_zeros == 0:
        return None, None, float(np.max(corr))
    
    s = np.arange(-(n-1), n)
    
    # Compute effects for all ones and zeros
    ones_eff = np.zeros((n_ones, 2*n-1))
    for a in range(n_ones):
        i = ones_idx[a]
        ip = i + s
        im = i - s
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        ones_eff[a, vp] = h[ip[vp]]
        ones_eff[a, vm] -= (1.0 - h[im[vm]])
    
    zeros_eff = np.zeros((n_zeros, 2*n-1))
    for b in range(n_zeros):
        j = zeros_idx[b]
        jp = j + s
        jm = j - s
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        zeros_eff[b, vp] = -h[jp[vp]]
        zeros_eff[b, vm] += (1.0 - h[jm[vm]])
    
    # Find best swap
    new_max_2d = (corr[None, None, :] + ones_eff[:, None, :] + zeros_eff[None, :, :]).max(axis=2)
    best_val = float(new_max_2d.min())
    best_pos = np.unravel_index(np.argmin(new_max_2d), new_max_2d.shape)
    
    return ones_idx[best_pos[0]], zeros_idx[best_pos[1]], best_val

def _binary_local_search(h_init, deadline, rng, max_iters=50000):
    """Binary steepest descent with iterated local search."""
    n = len(h_init)
    dx = 2.0 / n
    
    h = _to_binary(h_init)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx
    
    stuck_counter = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        bi, bj, best_val = _binary_sweep(h, corr)
        
        if bi is None or best_val >= cur_max - 1e-14:
            # No improving swap
            stuck_counter += 1
            if stuck_counter > 10:
                # Kick: random 3-swap to escape
                ones_idx = np.where(h > 0.5)[0]
                zeros_idx = np.where(h < 0.5)[0]
                if len(ones_idx) >= 3 and len(zeros_idx) >= 3:
                    kick_ones = rng.choice(ones_idx, size=3, replace=False)
                    kick_zeros = rng.choice(zeros_idx, size=3, replace=False)
                    h[kick_ones] = 0.0
                    h[kick_zeros] = 1.0
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    cur_max = float(np.max(corr))
                    stuck_counter = 0
                else:
                    break
            elif stuck_counter > 3:
                # Try a random single swap (SA-like)
                ones_idx = np.where(h > 0.5)[0]
                zeros_idx = np.where(h < 0.5)[0]
                i = ones_idx[rng.integers(len(ones_idx))]
                j = zeros_idx[rng.integers(len(zeros_idx))]
                # Check if this swap is not too bad
                s = np.arange(-(n-1), n)
                # Compute delta for this specific swap
                ip, im = i+s, i-s
                jp, jm = j+s, j-s
                delta = np.zeros(2*n-1)
                vp = (ip >= 0) & (ip < n)
                vm = (im >= 0) & (im < n)
                delta[vp] += h[ip[vp]]
                delta[vm] -= (1.0 - h[im[vm]])
                vp = (jp >= 0) & (jp < n)
                vm = (jm >= 0) & (jm < n)
                delta[vp] -= h[jp[vp]]
                delta[vm] += (1.0 - h[jm[vm]])
                new_max = float(np.max(corr + delta))
                # Accept with probability based on how bad it is
                penalty = (new_max - cur_max) * dx
                if rng.random() < np.exp(-penalty / 0.003):
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = corr + delta
                    cur_max = new_max
                stuck_counter = 0
        else:
            # Apply best swap
            h[bi] = 0.0
            h[bj] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_max = best_val
            stuck_counter = 0
            
            if cur_max * dx < best_c5:
                best_c5 = cur_max * dx
                best_h = h.copy()
    
    return best_h, best_c5
