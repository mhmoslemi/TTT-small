def _project(v, target):
    # Project v onto {x : 0≤x≤1, sum(x)=target}
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

def _solve_minimax_lp(A, n):

    cvec[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
