```python
def _softmax_obj_grad(x, T, dx, target, rho):
    """
    Compute f(x) and its gradient.
    x is the h vector (n values).
    f = (1/T)*log(sum exp(T*c_k)) + (rho/2)*(sum(x)-target)^2
    """
    n = len(x)
    h = x  # Already bounded by L-BFGS-B
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Softmax
    Tc = T * c
    Tc_shifted = Tc - np.max(Tc)  # For numerical stability
    log_sum = np.max(Tc) + np.log(np.sum(np.exp(Tc_shifted)))
    f_soft = log_sum / T
    
    # Weights
    w = np.exp(Tc_shifted) / np.sum(np.exp(Tc_shifted))
    
    # Gradient of soft part
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        A = max(0, lag)
        B = min(n, n + lag)
        # First term: dx*(1-h[j-lag]) for j in [A, B)
        idx = np.arange(A, B)
        grad[idx] += w[k] * dx * (1.0 - h[idx - lag])
        # Second term: -dx*h[j+lag] for j+lag in [A, B), i.e., j in [A-lag, B-lag)
        idx2_start = max(0, A - lag)
        idx2_end = min(n, B - lag)
        if idx2_start < idx2_end:
            idx2 = np.arange(idx2_start, idx2_end)
            grad[idx2] -= w[k] * dx * h[idx2 + lag]
    
    # Gradient of penalty
    grad += rho * (np.sum(x) - target)
    
    f_total = f_soft + 0.5 * rho * (np.sum(x) - target)**2
    return f_total, grad
```

```python
    n = len(h)
    grad = np.zeros(n)
    
    # Lag 0
    w0 = w[n-1]
    grad += w0 * dx * (1.0 - 2.0 * h)
    
    # Positive and negative lags
    for q in range(1, n):
        w_pos = w[n-1+q]  # lag +q
        w_neg = w[n-1-q]  # lag -q
        
        if w_pos > 1e-15:
            # First term: i in [q, n-1], +w*dx*(1-h[i-q])
            grad[q:] += w_pos * dx * (1.0 - h[:-q])
            # Second term: i in [0, n-q-1], -w*dx*h[i+q]
            grad[:-q] -= w_pos * dx * h[q:]
        
        if w_neg > 1e-15:
            # First term: i in [0, n-q-1], +w*dx*(1-h[i+q])
            grad[:-q] += w_neg * dx * (1.0 - h[q:])
            # Second term: i in [q, n-1], -w*dx*h[i-q]
            grad[q:] -= w_neg * dx * h[:-q]
    
    return grad
```

```python
def _softmax_lbfgs(h0, time_limit=30.0, T_start=10.0, T_end=1000.0):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = _project(h0, target)
    
    best_f = _exact_c5(h0)
    best_h = h0.copy()
    t_start = time.time()
    
    # Annealing schedule
    n_iters = 20
    T = T_start
    rho = 1.0
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        # Increase T and rho
        progress = (it + 1) / n_iters
        T = T_start * (T_end / T_start) ** progress
        rho = 1.0 + 100.0 * progress ** 2
        
        def fun_and_grad(x):
            h = np.clip(x, 0.0, 1.0)  # Ensure bounds
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            Tc = T * c
            Tc_max = np.max(Tc)
            exps = np.exp(Tc - Tc_max)
            log_sum = Tc_max + np.log(np.sum(exps))
            f_soft = log_sum / T
            w = exps / np.sum(exps)
            
            # Gradient
            n_local = len(h)
            grad = np.zeros(n_local)
            w0 = w[n_local - 1]
            grad += w0 * dx * (1.0 - 2.0 * h)
            for q in range(1, n_local):
                wp = w[n_local - 1 + q]
                wn = w[n_local - 1 - q]
                if wp > 1e-18:
                    grad[q:] += wp * dx * (1.0 - h[:-q])
                    grad[:-q] -= wp * dx * h[q:]
                if wn > 1e-18:
                    grad[:-q] += wn * dx * (1.0 - h[q:])
                    grad[q:] -= wn * dx * h[:-q]
            
            # Add penalty gradient
            s = np.sum(x) - target
            grad += rho * s
            f = f_soft + 0.5 * rho * s * s
            
            # If x is outside bounds, push gradient outward
            # (L-BFGS-B handles bounds, but the gradient at bounds needs care)
            # Actually, L-BFGS-B with bounds should handle this correctly.
            # We just need to make sure our function is evaluated correctly.
            return f, grad
        
        res = minimize(fun_and_grad, h0, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-8,
                               'maxls': 50})
        
        h_new = _project(res.x, target)
        f_new = _exact_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
            h0 = h_new.copy()
    
    return best_h, best_f
```

```python
def _slsqp_eq(x):
    return float(np.sum(x[:-1])) - _g_target

def _softmax_lbfgs(h0, time_limit, T_start=5.0, T_end=300.0, n_iters=20):
    """Run soft-max L-BFGS-B with annealing."""
    global _g_n, _g_dx, _g_target, _g_T, _g_rho
    n = len(h0)
    _g_n = n
    _g_dx = 2.0 / n
    _g_target = n / 2.0
    
    h0 = _project(h0, n / 2.0)
    best_f = _exact_c5(h0)
    best_h = h0.copy()
    t_start = time.time()
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        progress = (it + 1) / n_iters
        _g_T = T_start * (T_end / T_start) ** (2.0 * progress)
        _g_rho = 1.0 + 50.0 * progress ** 2
        
        res = minimize(_lbfgs_fun, h0, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': 80, 'ftol': 1e-15, 'gtol': 1e-10,
                               'maxls': 40, 'maxfun': 200})
        
        h_new = _project(res.x, n / 2.0)
        f_new = _exact_c5(h_new)
        if f_new < best_f - 1e-14:
            best_f = f_new
            best_h = h_new.copy()
            h0 = h_new.copy()
    
    return best_h, best_f

def _slsqp_cg(h0, time_limit, n_active_init=30, max_outer=20):
    """SLSQP with constraint generation."""
    global _g_n, _g_dx, _g_target, _g_active_mask
    n = len(h0)
    _g_n = n
    _g_dx = 2.0 / n
    _g_target = n / 2.0
    
    h0 = _project(h0, n / 2.0)
    best_f = _exact_c5(h0)
    best_h = h0.copy()
    t_start = time.time()
    
    corr = _corr_vals(h0)
    n_corr = len(corr)
    
    for outer in range(max_outer):
        if time.time() - t_start > time_limit:
            break
        
        # Determine active set
        n_active = min(n_active_init + outer * 5, n_corr)
        active_idx = np.argsort(-corr)[:n_active]
        _g_active_mask = active_idx
        
        # Initial point
        z0 = float(np.max(corr)) + 0.02
        x0 = np.concatenate([h0, [z0]])
        bounds = [(0.0, 1.0)] * n + [(-0.1, 1.0)]
        
        constraints = [
            {'type': 'ineq', 'fun': _slsqp_ineq},
            {'type': 'eq', 'fun': _slsqp_eq},
        ]
        
        t_sub_start = time.time()
        sub_limit = min(30.0, time_limit - (t_sub_start - t_start))
        
        res = minimize(_slsqp_obj, x0, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 80, 'ftol': 1e-12})
        
        if res.x is not None and np.isfinite(res.x).all():
            h_new = _project(res.x[:-1], n / 2.0)
            f_new = _exact_c5(h_new)
            if f_new < best_f - 1e-14:
                best_f = f_new
                best_h = h_new.copy()
                h0 = h_new.copy()
                corr = _corr_vals(h_new)
        
        # Check time
        if time.time() - t_start > time_limit * 0.85:
            break
    
    return best_h, best_f

def _pair_optimize(h, deadline, rng, pair_frac=0.25, rand_frac=0.08, k_active=10, tau=0.015):
    """Lightweight pairwise optimizer for polish."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h, target)
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    sigma = 0.04
    
    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
        elif r < rand_frac + pair_frac:
            d = np.zeros(n)
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d[i] = -1.0
            d[j] = 1.0
        else:
            # Active direction
            m = float(np.max(corr))
            w = np.exp((corr - m) / tau)
            idx = np.argsort(-corr)[:k_active]
            G = np.zeros(n)
            for r_ in idx:
                k = r_ - (n - 1)
                if k >= 0:
                    nn = np.arange(0, n - k)
                else:
                    nn = np.arange(-k, n)
                a = nn + k
                b = nn
                G[b] += -w[r_] * h[a]
                G[a] += w[r_] * (1.0 - h[b])
            d = -G
        
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        
        # Line search
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
           
            
            continue
        
        # Golden section search
        S0 = corr
        T_vec = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q_vec = np.correlate(d, d, mode='full')
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a = L
        b = U
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(S0 + c1 * T_vec - c1 * c1 * Q_vec)) * dx
        f2 = float(np.max(S0 + c2 * T_vec - c2 * c2 * Q_vec)) * dx
        
        for _ in range(40):
            if b - a < 1e-13:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = float(np.max(S0 + c1 * T_vec - c1 * c1 * Q_vec)) * dx
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(S0 + c2 * T_vec - c2 * c2 * Q_vec)) * dx
        
        t = 0.5 * (a + b)
        hn = h + t * d
        if hn.min() < 0.0 or hn.max() > 1.0:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + 5e-4:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
        
        if stall > 100:
            h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.003, sigma * 0.85)
    
    return best_h, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)
```

```python
        return np.zeros(n)

        return np.ones(n)

            f2 = float(np.max(S0 + c2 * T_vec - c2 * c2 * Q_vec)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T_vec - t * t * Q_vec)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv


def _pair_optimize(h, deadline, rng, pair_frac=0.25, rand_frac=0.08, k_active=10, tau=0.015):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h, target)
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    sigma = 0.04
    uphill = 5e-4
    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
        elif r < rand_frac + pair_frac:
            d = np.zeros(n)
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d[i] = -1.0
            d[j] = 1.0
        else:
            m = float(np.max(corr))
            w = np.exp((corr - m) / tau)
            w /= float(np.sum(w))
            idx = np.argsort(-corr)[:k_active]
            G = _row_grad(h, 0, dx) * w[n - 1]
            for r_ in idx:
                if r_ == n - 1:
                    continue
                k = r_ - (n - 1)
                G += w[r_] * _row_grad(h, k, dx)
            d = -G
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue
        T_vec = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q_vec = np.correlate(d, d, mode='full')
        t, _ = _line_search(corr, T_vec, Q_vec, dx, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
        if stall > 120:
            h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.003, sigma * 0.85)
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.04, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 4))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.06, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        base = _sine_init(100, 1, 0.0, rng)

    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: Soft-max L-BFGS-B on parent n (25% budget)
    d1 = t0 + budget_s * 0.25
    if time.time() < d1:
        t_sub = d1 - time.time()
        hh, ff = _softmax_lbfgs(h0, min(t_sub, 80.0), T_start=5.0, T_end=250.0, n_iters=18)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Try complement
    h_comp = _project(1.0 - h0, n0 / 2.0)
    if time.time() < d1:
        t_sub = min(d1 - time.time(), 40.0)
        hh, ff = _softmax_lbfgs(h_comp, t_sub, T_start=5.0, T_end=200.0, n_iters=12)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Phase 2: Multi-resolution (35% budget)
    d2 = t0 + budget_s * 0.60
    n_list = [80, 60, 50, 100, 120]
    if n0 in n_list:
        n_list.remove(n0)
    n_list = [x for x in n_list if x >= 40 and x <= 140]
    for n_try in n_list:
        if time.time() > d2:
            break
        hr = _resample(best_h, n_try)
        hr2 = _sine_init(n_try, 1, rng.random() * 6.28, rng)
        hr3 = _block_init(n_try, rng)
        for start_h in [hr, hr2, hr3]:
            if time.time() > d2:
                break
            t_sub = min(d2 - time.time(), 35.0)
            if t_sub < 5.0:
                break
            hh, ff = _softmax_lbfgs(start_h, t_sub, T_start=5.0, T_end=200.0, n_iters=12)
            if ff < best_f:
                best_f = ff
                best_h = hh.copy()

    # Phase 3: SLSQP constraint generation (20% budget)
    d3 = t0 + budget_s * 0.80
    if time.time() < d3:
        t_sub = min(d3 - time.time(), 180.0)
        hh, ff = _slsqp_cg(best_h, t_sub, n_active_init=15, max_outer=20)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Also try SLSQP on a resampled version if time permits
    if time.time() < d3:
        n_alt = len(best_h) - 2
        if n_alt >= 40 and n_alt % 2 == 0:
            hr = _resample(best_h, n_alt)
            t_sub = min(d3 - time.time(), 60.0)
            if t_sub > 10.0:
                hh, ff = _slsqp_cg(hr, t_sub, n_active_init=15, max_outer=10)
                if ff < best_f:
                    best_f = ff
                    best_h = hh.copy()

    # Phase 4: Soft-max polish on best (10% budget)
    d4 = t0 + budget_s * 0.90
    if time.time() < d4:
        t_sub = min(d4 - time.time(), 60.0)
        hh, ff = _softmax_lbfgs(best_h, t_sub, T_start=50.0, T_end=500.0, n_iters=10)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Phase 5: Pairwise polish (remaining budget)
    t_sub = deadline - time.time()
    if t_sub > 5.0:
        hh, ff = _pair_optimize(best_h, deadline, rng, pair_frac=0.30, rand_frac=0.06, k_active=12, tau=0.012)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
```
