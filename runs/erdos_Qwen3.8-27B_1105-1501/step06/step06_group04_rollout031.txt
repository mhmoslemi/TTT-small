```python
def _lse_objective_and_grad(h, beta, precomputed_corr=None):
    """Compute LSE objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    z = beta * (corr - max_c)
    exp_z = np.exp(z)
    sum_exp = np.sum(exp_z)
    log_sum = np.log(sum_exp)
    val = (max_c + log_sum) / beta
    p = exp_z / sum_exp
    # Gradient
    grad = np.zeros(n)
    N = len(corr)
    for r in range(N):
        if p[r] < 1e-15:
            continue
        d = r - (n - 1)
        m = n - abs(d)
        if d >= 0:
            if m > 0:
                grad[d:] += p[r] * (1.0 - h[:m])
                grad[:m] += p[r] * (-h[d:])
        else:
            g = -d
            if m > 0:
                grad[:m] += p[r] * (1.0 - h[g:])
                grad[g:] += p[r] * (-h[:m])
    grad *= dx
    return val, grad
```

```python
def _project(v, target):

    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0, 1)
    lo, hi = -1.0, 2.0

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if stagnation > 300:
            # Escape: random perturbation
            for _ in range(5):
                i = int(ones[rng.integers(len(ones))])
                j = int(zeros[rng.integers(len(zeros))])
                h[i], h[j] = h[j], h[i]
            cur = _c5(h)
            c5arr = _corr_all(h)
            stagnation = 0
            if cur < best:
                best = cur
                best_h = h.copy()
            continue

        # Full sweep: try all pairs, find best improvement
        best_delta = -1e15
        best_i, best_j = -1, -1
        n_ones = len(ones)
        n_zeros = len(zeros)
        # Limit pairs for speed on large n
        if n_ones * n_zeros > 20000:
            # Subsample
            idx_o = rng.choice(n_ones, size=min(n_ones, 200), replace=False)
            idx_z = rng.choice(n_zeros, size=min(n_zeros, 200), replace=False)
            ones_sub = ones[idx_o]
            zeros_sub = zeros[idx_z]
        else:
            ones_sub = ones
            zeros_sub = zeros

        for i in ones_sub:
            for j in zeros_sub:
                h[i], h[j] = h[j], h[i]
                new_c5arr = _corr_all(h)
                new_c5 = float(np.max(new_c5arr))
                h[i], h[j] = h[j], h[i]
                delta = cur - new_c5
                if delta > best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j

        if best_i >= 0 and best_delta > 1e-15:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            cur += best_delta
            if cur < best:
                best = cur
                best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

    return best_h, best


def _binary_sa(h, deadline, rng, T_init=0.003, T_min=0.00005, cooling=0.9998):
    """Binary simulated annealing."""
    n = len(h)
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best = cur
    T = T_init
    patience = 0

    while time.time() < deadline:
        if patience > 5000:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        h[i], h[j] = h[j], h[i]
        new_c5 = _c5(h)
        delta = new_c5 - cur
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur = new_c5
            if cur < best:
                best = cur
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            h[i], h[j] = h[j], h[i]
            patience += 1
        T = max(T_min, T * cooling)

    return best_h, best


def _lse_grad(h, beta, dx, c5arr):
    """Compute gradient of LSE objective."""
    n = len(h)
    N = len(c5arr)
    max_c = np.max(c5arr)
    z = beta * (c5arr - max_c)
    z -= z.max()  # for numerical stability
    exp_z = np.exp(z)
    p = exp_z / np.sum(exp_z)
    grad = np.zeros(n)
    for r in range(N):
        if p[r] < 1e-12:
            continue
        d = r - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        w = p[r]
        if d >= 0:
            grad[d:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[d:])
        else:
            g = -d
            grad[:m] += w * (1.0 - h[g:])
            grad[g:] += w * (-h[:m])
    return grad * dx


def _slsqp_lse(h0, beta, maxiter=300, maxfun=800):
    """SLSQP with LSE smoothed objective."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    def fun(x):
        h = np.clip(x, 0, 1)
        c5arr = _corr_all(h)
        max_c = np.max(c5arr)
        z = beta * (c5arr - max_c)
        z -= z.max()
        exp_z = np.exp(z)
        return (max_c + np.log(np.sum(exp_z))) / beta

    def jac(x):
        h = np.clip(x, 0, 1)
        c5arr = _corr_all(h)
        return _lse_grad(h, beta, dx, c5arr)

    def eq_fun(x):
        return np.array([np.sum(x) - target])

    def eq_jac(x):
        return np.ones((1, n))

    res = minimize(fun, h0, jac=jac, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                   options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-14})
    if res.success:
        return np.clip(res.x, 0, 1)
    return h0


def _slsqp_minimax(h0, maxiter=200, maxfun=400):
    """Direct minimax SLSQP: minimize M s.t. M >= C_r(h)."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    N = 2 * n - 1
    c5arr0 = _corr_all(h0)
    M0 = float(np.max(c5arr0)) + 0.05

    def fun(y):
        return y[0]

    def fun_grad(y):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def ineq(y):
        h = y[1:]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return y[0] - corr

    def ineq_jac(y):
        h = y[1:]
        n2 = len(h)
        J = np.zeros((N, n + 1))
        J[:, 0] = 1.0
        for r in range(N):
            d = r - (n2 - 1)
            m = n2 - abs(d)
            if m <= 0:
                continue
            if d >= 0:
                J[r, 1 + d:1 + n2] += -(1.0 - h[:m]) * dx
                J[r, 1:1 + m] += -h[d:] * dx
            else:
                g = -d
                J[r, 1:1 + m] += -(1.0 - h[g:]) * dx
                J[r, 1 + g:1 + n2] += -h[:m] * dx
        return J

    def eq_fun(y):
        return np.array([np.sum(y[1:]) - target])

    def eq_jac(y):
        J = np.zeros((1, n + 1))
        J[0, 1:] = 1.0
        return J

    y0 = np.concatenate(([M0], h0))
    bounds = [(0, M0 + 0.3)] + [(0, 1)] * n

    try:
        res = minimize(fun, y0, jac=fun_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
                                    {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-12})
        if res.x[1:].min() >= -0.01 and res.x[1:].max() <= 1.01:
            return _project(res.x[1:], target)
    except Exception:
        pass
    return h0


def _line_search_exact(h, d, dx, radius):
    """Exact line search along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = radius
    t_lo = -radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val


def _pair_refine(h, deadline, max_passes=100):
    """2-coordinate refinement."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best = _c5(h)
    stall = 0
    rng = np.random.default_rng(123)

    while time.time() < deadline and stall < 30:
        interior = np.where((h > 0.03) & (h < 0.97))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(300):
            i = int(interior[rng.integers(len(interior))])
            j = int(interior[rng.integers(len(interior))])
            if i == j:
                continue
            for sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                alpha, c5_try = _line_search_exact(h, d, dx, 0.5)
                if c5_try < best - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _c5(h)
                    if cur < best:
                        best = cur
                        best_h = h.copy()
                        improved = True
                        break
        if improved:
            stall = 0
        else:
            stall += 1

    return best_h, best


def _coordinate_refine(h, deadline):
    """Single-coordinate refinement."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best = _c5(h)
    rng = np.random.default_rng(77)
    stall = 0

    while time.time() < deadline and stall < 20:
        improved = False
        for _ in range(200):
            i = int(rng.integers(n))
            c5arr = _corr_all(h)
            # Compute gradient of max constraint at i
            # Try small perturbations
            for eps in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
                h[i] = np.clip(h[i] + eps, 0, 1)
                new_c5 = _c5(h)
                h[i] = np.clip(h[i] - eps, 0, 1)
                # Project and check
                h2 = _project(h + np.where(np.arange(n) == i, eps, 0), target)
                new_c5 = _c5(h2)
                if new_c5 < best - 1e-14:
                    h = h2
                    best = new_c5
                    best_h = h.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1

    return best_h, best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Get initial construction
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_parent = _sine_init(120, 3, 0.5)

    n_parent = len(h_parent)
    h_parent = _project(h_parent, n_parent / 2.0)
    parent_c5 = _c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = parent_c5
    best_n = n_parent

    # Phase 1: Binary best-improvement at parent n (30% time)
    d1 = min(deadline, t0 + budget_s * 0.30)
    h_bin = _binarize(h_parent, n_parent)
    h_b, c5_b = _binary_best_improvement(h_bin, d1, rng, max_sweeps=3000)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
        best_n = n_parent

    # Phase 2: Binary SA for exploration (10% time)
    d2 = min(deadline, t0 + budget_s * 0.40)
    h_sa, c5_sa = _binary_sa(h_b, d2, rng, T_init=0.004, T_min=0.00003, cooling=0.9997)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
        best_n = n_parent

    # Phase 3: Multi-n binary search (20% time)
    d3 = min(deadline, t0 + budget_s * 0.60)
    n_list = sorted(set([n_parent - 2, n_parent, n_parent + 2, n_parent + 4,
                         n_parent - 4, n_parent + 6, n_parent - 6,
                         n_parent + 8, n_parent - 8]))
    n_list = [x for x in n_list if x >= 60 and x <= 250 and x % 2 == 0]
    n_list = sorted(n_list)

    for n_try in n_list:
        if time.time() > d3:
            break
        if n_try == n_parent:
            continue
        h_r = _resample(h_parent, n_try)
        h_bin_r = _binarize(h_r, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.04)
        h_br, c5_br = _binary_best_improvement(h_bin_r, sub_dl, rng, max_sweeps=1500)
        if c5_br < best_c5 - 1e-12:
            best_c5 = c5_br
            best_h = h_br.copy()
            best_n = n_try

    # Phase 4: LSE-SLSQP continuous refinement (15% time)
    d4 = min(deadline, t0 + budget_s * 0.75)
    h_cont = _project(best_h.astype(float), best_n / 2.0)
    for beta in [40, 80, 150, 300, 600]:
        if time.time() > d4:
            break
        h_new = _slsqp_lse(h_cont, beta, maxiter=200, maxfun=500)
        h_new = _project(h_new, best_n / 2.0)
        c5_new = _c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()

    # Phase 5: Direct minimax SLSQP (10% time)
    d5 = min(deadline, t0 + budget_s * 0.85)
    if time.time() < d5:
        h_mm = _slsqp_minimax(h_cont, maxiter=150, maxfun=300)
        h_mm = _project(h_mm, best_n / 2.0)
        c5_mm = _c5(h_mm)
        if c5_mm < best_c5 - 1e-12:
            best_c5 = c5_mm
            best_h = h_mm.copy()
            h_cont = h_mm.copy()

    # Phase 6: Pair + coordinate refinement (8% time)
    d6 = min(deadline, t0 + budget_s * 0.93)
    h_pr, c5_pr = _pair_refine(best_h, min(d6, t0 + budget_s * 0.88), max_passes=80)
    if c5_pr < best_c5:
        best_c5 = c5_pr
        best_h = h_pr.copy()
    if time.time() < d6:
        h_cr, c5_cr = _coordinate_refine(best_h, d6)
        if c5_cr < best_c5:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Phase 7: Multi-n continuous refinement on best (7% time)
    d7 = deadline
    for dn in [-4, -2, 2, 4, 6, -6]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 300 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_r = _slsqp_lse(h_r, 100, maxiter=100, maxfun=200)
        h_r = _project(h_r, n_try / 2.0)
        c5_r = _c5(h_r)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Final validation
    best_h = _project(best_h, best_n / 2.0)
    final_c5 = _c5(best_h)

    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h_fb = np.array(initial_h_values, dtype=float).copy()
            h_fb = _project(h_fb, len(h_fb) / 2.0)
            return h_fb, _c5(h_fb), len(h_fb)
        except NameError:
            h_fb = _sine_init(120, 3, 0.0)
            return h_fb, _c5(h_fb), 120

    return best_h, final_c5, best_n
```
