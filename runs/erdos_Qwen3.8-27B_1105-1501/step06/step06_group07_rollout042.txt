def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        if float(np.sum(np.clip(v-mid, 0, 1))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2, 0, 1)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        m = x < 1 if res > 0 else x > 0
        c = int(np.sum(m))
        if c > 0:
            x[m] += res/c
            x = np.clip(x, 0, 1)
    return x

def _compute_B(h):
    n = len(h)
    B = np.zeros((n, 2*n-1))
    for k in range(2*n-1):
        s = k - (n-1)
        col = np.zeros(n)

        i_lo = max(0, s)
        i_hi = min(n-1, n-1+s)
        if i_lo <= i_hi:
            col[i_lo:i_hi+1] += 1.0 - h[i_lo-s:i_hi+1-s]
        # Term 2: -h[i+s] for valid i

        i_hi2 = min(n-1, n-1-s)
        if i_lo2 <= i_hi2:
            col[i_lo2:i_hi2+1] -= h[i_lo2+s:i_hi2+1+s]
        B[:, k] = col
    return B

def _pair_min_1d(c, d, delta_lo, delta_hi):
    """Minimize max_k (c[k] + delta * d[k]) for delta in [delta_lo, delta_hi]."""
    if delta_lo >= delta_hi:
        return 0.0
    # Grid
    n_grid = 40
    grid = np.linspace(delta_lo, delta_hi, n_grid)
    vals = np.max(c[None,:] + grid[:,None] * d[None,:], axis=1)
    best_i = int(np.argmin(vals))
    best_val = vals[best_i]
    best_delta = grid[best_i]
    # Golden section refine
    if best_i == 0:
        a, b = delta_lo, grid[1]
    elif best_i == n_grid-1:
        a, b = grid[n_grid-2], delta_hi
    else:
        a, b = grid[best_i-1], grid[best_i+1]
    gr = (np.sqrt(5)-1)/2
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(c + c1*d))
    f2 = float(np.max(c + c2*d))
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(c + c1*d))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(c + c2*d))
    opt = (a+b)/2
    fopt = float(np.max(c + opt*d))
    if fopt < best_val:
        best_delta, best_val = opt, fopt
    return best_delta

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        m = x < 1.0 if res > 0 else x > 0.0
        c = int(np.sum(m))
        if c > 0:
            x[m] += res / c
            x = np.clip(x, 0.0, 1.0)
    return x


def _compute_B(h):
    n = len(h)
    B = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        col = np.zeros(n)
        i_lo = max(0, s)
        i_hi = min(n - 1, n - 1 + s)
        if i_lo <= i_hi:
            col[i_lo:i_hi + 1] += 1.0 - h[i_lo - s:i_hi + 1 - s]
        i_lo2 = max(0, -s)
        i_hi2 = min(n - 1, n - 1 - s)
        if i_lo2 <= i_hi2:
            col[i_lo2:i_hi2 + 1] -= h[i_lo2 + s:i_hi2 + 1 + s]
        B[:, k] = col
    return B


def _pair_min_1d(c, d, dlo, dhi):
    if dlo >= dhi - 1e-14:
        return 0.0
    n_grid = 35
    grid = np.linspace(dlo, dhi, n_grid)
    vals = np.max(c[None, :] + grid[:, None] * d[None, :], axis=1)
    best_i = int(np.argmin(vals))
    best_val = vals[best_i]
    best_d = grid[best_i]
    if best_i == 0:
        a, b = dlo, grid[1]
    elif best_i == n_grid - 1:
        a, b = grid[n_grid - 2], dhi
    else:
        a, b = grid[best_i - 1], grid[best_i + 1]

    opt = (a + b) / 2.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
