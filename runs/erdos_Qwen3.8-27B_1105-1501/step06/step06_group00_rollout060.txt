import numpy as np
from scipy.optimize import minimize_scalar, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Water-filling projection to sum=target, 0<=v<=1"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_t(h, i, j, n):
    """Compute t[l] for pair (i,j) where direction is e_j - e_i (decrease i, increase j)
    t[l] = h[i-l] - (1-h[i+l]) + (1-h[j+l]) - h[j-l]
    l ranges from -(n-1) to (n-1), stored at index l+(n-1)
    """
    t = np.zeros(2*n-1)
    offset = n - 1
    h1 = 1.0 - h
    for l in range(-(n-1), n):
        t[l + offset] = 0.0
        # h[i-l]: valid if 0 <= i-l < n
        if 0 <= i - l < n:
            t[l + offset] += h[i - l]
        # -(1-h[i+l]): valid if 0 <= i+l < n
        if 0 <= i + l < n:
            t[l + offset] -= h1[i + l]
        # +(1-h[j+l]): valid if 0 <= j+l < n
        if 0 <= j + l < n:
            t[l + offset] += h1[j + l]
        # -h[j-l]: valid if 0 <= j-l < n
        if 0 <= j - l < n:
            t[l + offset] -= h[j - l]
    return t

    A[i, p] = h[i-l] if valid, where l = p-(n-1). 0 otherwise.
    B[i, p] = (1-h[i+l]) if valid, where l = p-(n-1). 0 otherwise.
    """

    A = np.zeros((n, L))
    B = np.zeros((n, L))
    h1 = 1.0 - h

        # A[i, p] = h[i + n - 1 - p] if 0 <= i+n-1-p < n
