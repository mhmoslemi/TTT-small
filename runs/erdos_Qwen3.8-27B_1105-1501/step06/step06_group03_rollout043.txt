    n = len(h)

    for k in range(n):
        if k == 0:
            G[0, :] = one_minus_h - h  # = 1 - 2h
        else:
            m = n - k
            G[k, k:] = one_minus_h[:m]  # (1-h[i-k]) for i=k,...,n-1
            G[k, :m] -= h[k:]  # -h[i+k] for i=0,...,m-1
    return G

        G[k, :m] -= h[k:]
    return G

def _minimax_1d(c, b, lo, hi):
    if hi - lo < 1e-15:
        return lo, float(np.max(c + lo * b))
    f_lo = float(np.max(c + lo * b))
    f_hi = float(np.max(c + hi * b))
    V_lo = min(f_lo, f_hi) - 1.0
    V_hi = max(f_lo, f_hi) + 1e-12
    for _ in range(50):
        if V_hi - V_lo < 1e-15:
            break
        V_mid = (V_lo + V_hi) * 0.5
        pos = b > 1e-15
        neg = b < -1e-15
        d_hi = hi
        d_lo = lo
        if np.any(pos):
            d_hi = min(d_hi, float(np.min((V_mid - c[pos]) / b[pos])))
        if np.any(neg):
            d_lo = max(d_lo, float(np.max((V_mid - c[neg]) / b[neg])))
        if d_lo <= d_hi:
            z = ~(pos | neg)
            if not (np.any(z) and np.any(c[z] > V_mid + 1e-15)):
                V_hi = V_mid
            else:
                V_lo = V_mid
        else:
            V_lo = V_mid
    V = V_hi
    pos = b > 1e-15
    neg = b < -1e-15
    d_hi = hi
    d_lo = lo
    if np.any(pos):
        d_hi = min(d_hi, float(np.min((V - c[pos]) / b[pos])))
    if np.any(neg):
        d_lo = max(d_lo, float(np.max((V - c[neg]) / b[neg])))
    if d_lo > d_hi:
        d_lo = d_hi = (lo + hi) * 0.5
    return (d_lo + d_hi) * 0.5, V_hi

def _pair_minimax(h, C, G, i, j):
    b = G[:, i] - G[:, j]
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, float(np.max(C))
    return _minimax_1d(C, b, lo, hi)

def _4pt_minimax(h, C, G, i1, i2, j1, j2):
    b = G[:, i1] + G[:, i2] - G[:, j1] - G[:, j2]
    lo = max(-h[i1], -h[i2], h[j1] - 1.0, h[j2] - 1.0)
    hi = min(1.0 - h[i1], 1.0 - h[i2], h[j1], h[j2])
    if hi - lo < 1e-14:
        return 0.0, float(np.max(C))
    return _minimax_1d(C, b, lo, hi)

def _dir_minimax(h, C, G, d):
    b = G @ d
    pos = d > 1e-15
    neg = d < -1e-15
    lo = 0.0
    hi = np.inf
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(-h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, float(np.max(C))
    return _minimax_1d(C, b, lo, hi)

def _lp_direction(h, C, n, K, radius):
    M = float(C.max())
    sorted_idx = np.argsort(-C)
    active = sorted_idx[:K]
    tol = max(0.03 * M, 1e-6)
    near_idx = np.where(C >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _compute_G(h)
    G_sub = G[combined, :]
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = 1.0
    b_ub = M - C[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-13:
                return d
    except Exception:
        pass
    return None

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _minimax_loop(h, deadline, use_lp=True, lp_interval=50, tau_scale=0.06):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    C = _corr_all(h)
    G = _compute_G(h)
    rng = np.random.default_rng(555)
    stall = 0
    iter_count = 0
    n_pair_tries = 8
    n_4pt_tries = 3

    while time.time() < deadline:
        iter_count += 1
        cur_max = float(np.max(C))
        if cur_max < best_c5 - 1e-14:
            best_c5 = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 8000:
                h = h + rng.normal(0, 0.03, n)
                h = _project(h, target)
                C = _corr_all(h)
                G = _compute_G(h)
                stall = 0
                continue

        improved = False

        # LP direction (periodic)
        if use_lp and iter_count % lp_interval == 0 and stall > 50:
            rad = 0.05 + 0.15 * min(1.0, stall / 500.0)
            d = _lp_direction(h, C, n, 12, rad)
            if d is not None:
                t, new_val = _dir_minimax(h, C, G, d)
                if new_val < cur_max - 1e-12 and t != 0:
                    h = h + t * d
                    h = np.clip(h, 0.0, 1.0)
                    s = float(np.sum(h))
                    if abs(s - target) > 1e-10:
                        h = _project(h, target)
                    C = _corr_all(h)
                    G = _compute_G(h)
                    improved = True

        if improved:
            continue

        # Compute weighted gradient
        tau = max(tau_scale * cur_max, 0.005)
        w = np.exp(C / tau - np.max(C / tau))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(C))
            w[np.argmax(C)] = 1.0
        else:
            w /= w_sum
        Gtot = G.T @ w

        # Try top pairs
        top_pos = np.argsort(-Gtot)[:n_pair_tries]
        top_neg = np.argsort(Gtot)[:n_pair_tries]

        for ii in top_pos:
            if improved:
                break
            for jj in top_neg:
                if ii == jj:
                    continue
                delta, new_val = _pair_minimax(h, C, G, ii, jj)
                if new_val < cur_max - 1e-12:
                    h[ii] += delta
                    h[jj] -= delta
                    h = np.clip(h, 0.0, 1.0)
                    C = _corr_all(h)
                    G = _compute_G(h)
                    improved = True
                    break

        if improved:
            continue

        # Try 4-point moves
        tp2 = np.argsort(-Gtot)[:n_4pt_tries * 2]
        tn2 = np.argsort(Gtot)[:n_4pt_tries * 2]
        for a in range(min(n_4pt_tries, len(tp2) // 2)):
            if improved:
                break
            i1, i2 = tp2[2 * a], tp2[2 * a + 1]
            j1, j2 = tn2[2 * a], tn2[2 * a + 1]
            delta, new_val = _4pt_minimax(h, C, G, i1, i2, j1, j2)
            if new_val < cur_max - 1e-12:
                h[i1] += delta
                h[i2] += delta
                h[j1] -= delta
                h[j2] -= delta
                h = np.clip(h, 0.0, 1.0)
                C = _corr_all(h)
                G = _compute_G(h)
                improved = True

        if improved:
            continue

        # Random pairs
        for _ in range(6):
            ii = rng.integers(n)
            jj = rng.integers(n)
            if ii == jj:
                continue
            delta, new_val = _pair_minimax(h, C, G, ii, jj)
            if new_val < cur_max - 1e-12:
                h[ii] += delta
                h[jj] -= delta
                h = np.clip(h, 0.0, 1.0)
                C = _corr_all(h)
                G = _compute_G(h)
                improved = True
                break

        if not improved and iter_count % 200 == 0:
            # Slightly dampen: move toward 0.5 to escape flat regions
            mask = (h > 0.05) & (h < 0.95)
            if np.sum(mask) > 0:
                h[mask] += (0.5 - h[mask]) * 0.1
                h = _project(h, target)
                C = _corr_all(h)
                G = _compute_G(h)

    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    cooldown = 0.9999
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]

    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        n_trials = 300
        for _ in range(n_trials):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta_c = new_c5 - cur_c5
            if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _lp_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    C = _corr_all(h)
    G = _compute_G(h)
    stall = 0
    radius = 0.06

    while time.time() < deadline:
        if stall > 200:
            break
        d = _lp_direction(h, C, n, 15, radius)
        if d is None:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        t, new_val = _dir_minimax(h, C, G, d)
        if new_val < best_c5 - 1e-13 and t != 0:
            h = h + t * d
            h = np.clip(h, 0.0, 1.0)
            if abs(float(np.sum(h)) - target) > 1e-10:
                h = _project(h, target)
            C = _corr_all(h)
            G = _compute_G(h)
            cur = float(np.max(C))
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            ratio = abs(t) / max(np.max(np.abs(d)), 1e-15)
            radius = min(radius * 1.05, 0.2)
        else:
            stall += 1
            radius *= 0.9
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.98

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (10% time)
    d1 = t0 + budget_s * 0.10
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        for ph in [0.0, np.pi / 3, 2 * np.pi / 3]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(10):
        starts.append(_block_init(n0, rng, rng.integers(3, 12)))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _minimax_loop(s, sub_dl, use_lp=False, lp_interval=999, tau_scale=0.08)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main minimax optimization (45% time)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _minimax_loop(best_h, d2, use_lp=True, lp_interval=40, tau_scale=0.05)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution (20% time)
    d3 = t0 + budget_s * 0.75
    base_n = len(best_h)
    tried_n = set()
    offsets = [0, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, -24, 24, -30, 30,
               126, 124, 128, 130, 120, 132, 118, 134]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = base_n + dn if dn < 100 else dn
        if n_try < 50 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        sub_dl = min(d3, time.time() + budget_s * 0.018)
        h_r, c5_r = _minimax_loop(h_r, sub_dl, use_lp=True, lp_interval=30, tau_scale=0.05)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            base_n = n_try

    # Phase 4: Binary SA + LP polish (12% time)
    d4 = t0 + budget_s * 0.87
    h4a, c5_4a = _binary_sa(best_h, min(d4, t0 + budget_s * 0.80), rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_polish(best_h, d4)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Final minimax refinement (7% time)
    d5 = deadline
    h5, c5_5 = _minimax_loop(best_h, d5, use_lp=True, lp_interval=20, tau_scale=0.03)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final LP polish
    if time.time() < deadline - 5:
        h_f, c5_f = _lp_polish(best_h, deadline)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
