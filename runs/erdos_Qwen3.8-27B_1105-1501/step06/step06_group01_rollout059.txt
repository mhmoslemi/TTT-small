def _grad_of_lag(h, r):
    """Gradient of S_r with respect to h, where r is the lag offset (r = k - (n-1))."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        s = j - r
        if 0 <= s < n:
            val += (1.0 - h[s])
        t = j + r
        if 0 <= t < n:
            val -= h[t]
        grad[j] = dx * val
    return grad

Wait, this is O(n^2 * n) = O(n^3) per pass because for each swap I'm computing the full correlation. For n=122, that's 122^3 ≈ 1.8M operations. In Python with numpy, each _corr_vals call is maybe 5 μs, so 3721 * 5μs = 18ms per pass. That's 50 passes per second. Over 100 seconds, that's 5000 passes. Should be enough.

But I can do better. Let me precompute the effect of each position on each lag, then compute the effect of a swap in O(1) per lag (or even O(1) total by tracking only the affected lags).

Actually, let me think about this more carefully. For a binary h:
S_k = dx * sum_i h[i] * (1-h[i-r_k]) where r_k = k-(n-1)

When I change h[a] from 1 to 0:
- For each k where a-r_k is in [0,n-1]: S_k decreases by dx*(1-h[a-r_k]) = dx*0 (since h is binary, 1-h[a-r_k] is 0 or 1, and it's 1 only if h[a-r_k]=0)
  Wait: 1-h[a-r_k] where h[a-r_k] is 0 or 1. If h[a-r_k]=0, then 1-h[a-r_k]=1. If h[a-r_k]=1, then 1-h[a-r_k]=0.
  So S_k decreases by dx*1 if h[a-r_k]=0, or dx*0 if h[a-r_k]=1.

- For each k where a+r_k is in [0,n-1]: The term h[a+r_k]*(1-h[a]) was 0 (since h[a]=1), now becomes h[a+r_k]*(1-0) = h[a+r_k]. So S_k increases by dx*h[a+r_k] = dx*1 if h[a+r_k]=1, or 0 otherwise.

Similarly for h[b] from 0 to 1:
- For each k where b-r_k in [0,n-1]: S_k increases by dx*(1-h[b-r_k]) = dx*1 if h[b-r_k]=0, else 0.
- For each k where b+r_k in [0,n-1]: S_k decreases by dx*h[b+r_k] = dx*1 if h[b+r_k]=1, else 0.

So the total change for swap (a,b) is:
delta_S[k] = dx * [
  -1{h[a-r_k]=0, a-r_k valid} + 1{h[a+r_k]=1, a+r_k valid}
  + 1{h[b-r_k]=0, b-r_k valid} - 1{h[b+r_k]=1, b+r_k valid}
]

This is at most 4*dx per lag, and only O(n) lags are non-zero (those where a or b participate).

I can precompute for each position p:
- L_p = set of k where p-r_k is valid (p is "left" element) = {k: 0 <= p-(k-(n-1)) < n} = {k: p-n+1 <= k <= p+n-1} ∩ [0, 2n-2]
- R_p = set of k where p+r_k is valid (p is "right" element) = {k: 0 <= p+(k-(n-1)) < n} = {k: n-1-p <= k <= n-1+p} ∩ [0, 2n-2]

For a swap (a,b), the affected lags are L_a ∪ R_a ∪ L_b ∪ R_b.

The number of affected lags is at most |L_a| + |R_a| + |L_b| + |R_b| ≈ 4n (with overlaps).

For each affected lag, I can compute the new S_k in O(1) (just add the delta). Then the new max is max(new values of affected lags, max of unaffected lags).

To get the max of unaffected lags quickly, I can precompute a sorted list of S values. The max of unaffected = the largest S_k where k is not in the affected set. This can be done in O(log n) with a sorted array.

But this is getting complex. Let me just use the brute-force approach (recompute full correlation for each trial swap) since it's fast enough for n=122.

Actually, let me benchmark mentally:
- np.correlate on two length-122 arrays: ~242 multiply-adds ≈ 500 clock cycles ≈ 0.15 μs on a 3GHz CPU
- Plus overhead: maybe 5-10 μs total in numpy
- 3721 calls: 3721 * 10μs ≈ 37ms per full steepest-descent pass
- In 100 seconds: ~2700 passes

That's more than enough. Let me go with the simple approach.

But wait, I realize there's an even faster approach: use the incremental update.

For a binary h, when I swap a (1→0) and b (0→1), I can update S in O(n) time (only the affected lags change, and there are O(n) of them). This is the same as computing the full correlation, so no speedup.

The real speedup comes from NOT trying all swaps. Instead:
- Try a random subset of swaps (e.g., 100 per pass)
- This gives 100 * 10μs = 1ms per pass
- In 100 seconds: 100K passes

Or use a smarter heuristic:
- Only try swaps that are "promising" (e.g., a is in a position where many lags are high, b is in a position where it would reduce them)

For now, let me use the steepest-descent approach (try all swaps, pick best) since it's fast enough.

Let me also add 2-exchange moves (swap two 1s with two 0s). This can escape local minima. The number of 2-exchanges is C(61,2) * C(61,2) ≈ 1.8M, which is too many. I'll sample randomly.

1. **Gradient descent** (30%): Smooth log-sum-exp projected gradient
2. **Binary steepest descent** (30%): Try all single swaps, apply best
3. **Binary 2-exchange** (15%): Random 2-exchange moves
4. **Multi-resolution** (15%): Try different n values
5. **Final polish** (10%): More gradient + binary

```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - float(np.sum(x))
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += resid / cnt
    return np.clip(x, 0.0, 1.0)

def _binary_round(h):
    n = len(h)
    t = int(n // 2)
    idx = np.argsort(-h, kind='stable')
    out = np.zeros(n)
    out[idx[:t]] = 1.0
    return out

def _grad_lag(h, r):
    """Gradient of S_r w.r.t. h, where r = k - (n-1)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if r >= 0:
        g[r:] += (1.0 - h[:n-r])
        g[:n-r] -= h[r:]
    else:
        rp = -r
        g[:n-rp] += (1.0 - h[rp:])
        g[rp:] -= h[:n-rp]
    return g * dx

def _smooth_grad(h):
    """Compute gradient of smooth approximation C_smooth(h) = tau*log(sum exp(S_k/tau))."""
    n = len(h)
    S = _corr_vals(h)
    S_max = S.max()
    tau = 0.005
    w = np.exp((S - S_max) / tau)
    wsum = float(np.sum(w))
    if wsum < 1e-300:
        w = np.zeros_like(S)
        w[int(np.argmax(S))] = 1.0
    else:
        w /= wsum
    # Only use lags with significant weight
    active = np.where(w > 0.01)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(S))])
    g = np.zeros(n)
    for k in active:
        r = k - (n - 1)
        g += w[k] * _grad_lag(h, r)
    return g

def _gradient_descent(h, deadline, lr_init=0.02, tau=0.005):
    """Projected gradient with adaptive step."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    stall = 0
    rng = np.random.default_rng(123)
    
    while time.time() < deadline:
        g = _smooth_grad(h)
        h_new = h - lr * g
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, 0.08)
        else:
            stall += 1
            if stall % 20 == 0:
                # Try binary round
                h_b = _binary_round(h)
                c5_b = _exact_c5(h_b)
                if c5_b < best_c5 - 1e-13:
                    h = h_b
                    best_c5 = c5_b
                    best_h = h.copy()
                    stall = 0
                    lr = 0.02
                else:
                    lr = max(lr * 0.85, 0.002)
            elif stall > 200:
                break
    
    return best_h, best_c5

def _binary_steepest(h, deadline, max_passes=5000):

    best_h = h.copy()

    improves = 0

        S_max = float(S.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improve = 0.0
        best_pair = None
        best_new_S_max = S_max

            for b in zeros:
                # Compute change in S from swapping a->0, b->1
                # Use incremental update
                trial = h.copy()
                trial[a] = 0

                    best_improve = S_max - new_max
                    best_pair = (a, b)

                trial[b] = 0

        h[a] = 0
        h[b] = 1

            best_h = h.copy()
        improves += 1
    
    return best_h, best_c5

def _binary_steepest_fast(h, deadline, max_passes=5000):
    n = len(h)
    h = _binary_round(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        S = _corr_vals(h)
        S_max = float(S.max())
        
        best_improve = 0.0
        best_a = -1
        best_b = -1
        
        for a in ones:
            h[a] = 0
            for b in zeros:
                h[b] = 1
                new_S = _corr_vals(h)
                new_max = float(new_S.max())
                if new_max < S_max - 1e-12:
                    imp = S_max - new_max
                    if imp > best_improve:
                        best_improve = imp
                        best_a = a
                        best_b = b
                h[b] = 0
            h[a] = 1
        
        if best_a < 0:
            break
        
        # Apply best swap
        h[best_a] = 0
        h[best_b] = 1
        ones.remove(best_a)
        zeros.remove(best_b)
        ones.append(best_b)
        zeros.append(best_a)
        
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```
