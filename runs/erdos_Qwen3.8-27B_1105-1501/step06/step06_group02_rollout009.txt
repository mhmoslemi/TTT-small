def c5_exact(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def smooth_obj_grad(h, target_T, tau):
    """Compute smooth log-sum-exp objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    z_max = np.max(z)
    # Stable log-sum-exp
    exp_z = np.exp(z - z_max)
    sum_exp = np.sum(exp_z)
    obj = tau * (z_max + np.log(sum_exp))
    
    # Softmax weights
    w = exp_z / sum_exp  # shape (2n-1,)
    
    # Gradient: sum_k w_k * dS_k/dh
    # S_k = dx * R_k where R_k uses lag d_k = k - (n-1)
    grad = np.zeros(n)

        d = k - (n - 1)
        # dR_d/dh[m] = (1-h[m-d]) * [0<=m-d<n and 0<=m<n] + (-h[m+d]) * [0<=m<n and 0<=m+d<n]

            # m ranges from r to n-1, contributing -h[m+d] = -h[m-r]
            grad[r:] += w[k] * dx * (-h[:n-r])
    
    return obj, grad

def _line_search_dir(h, direction, S0, dx, max_iter=50):

    min_idx = int(np.argmin(vals))

    else:
        a_br = alphas[max(0, min_idx - 1)]
        b_br = alphas[min(n_eval - 1, min_idx + 1)]
    
    # Golden section

    a, b = a_br, b_br
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_vec - c1 * c1 * C_vec)) * dx
    f2 = float(np.max(S0 + c2 * T_vec - c2 * c2 * C_vec)) * dx

            break

            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_vec - c1 * c1 * C_vec)) * dx
        else:

            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_vec - c2 * c2 * C_vec)) * dx
    
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T_vec - opt * opt * C_vec)) * dx
    # Check endpoints
    f_lo = float(np.max(S0 + lo * T_vec - lo * lo * C_vec)) * dx
    f_hi = float(np.max(S0 + hi * T_vec - hi * hi * C_vec)) * dx
    f_0 = float(np.max(S0)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    if f_0 < val:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _paired_line_search(h, n_iterations, deadline):

    n = len(h)

import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj_grad(h, target_T, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    S = corr * dx
    z = (S - target_T) / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    sum_exp = np.sum(exp_z)
    obj = tau * (z_max + np.log(sum_exp))
    w = exp_z / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            grad[d:] += wk * dx * (1.0 - h[:m])
            grad[:m] += wk * dx * (-h[d:])
        else:
            r = -d
            m = n - r
            grad[:m] += wk * dx * (1.0 - h[r:])
            grad[r:] += wk * dx * (-h[:m])
    return obj, grad

def _sum_constraint(h):
    return float(np.sum(h)) - len(h) / 2.0

def _slsqp_one(h_start, target_T, tau, maxiter, deadline):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_constraint}]
    try:
        result = minimize(
            fun=smooth_obj_grad,
            x0=h_start,
            args=(target_T, tau),
            method='SLSQP',
            jac=True,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return result.x
    except Exception:
        pass
    return h_start.copy()

def _line_search_dir(h, direction, S0, dx):
    n = len(h)
    direction = direction - np.mean(direction)
    pos = direction > 1e-12
    neg = direction < -1e-12
    hi = 1e10
    lo = -1e10
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        lo = max(lo, float(-np.min(h[neg] / direction[neg])))
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if hi - lo < 1e-12:

    T_vec = np.correlate(direction, 1.0 - h, 'full') - np.correlate(h, direction, 'full')
    C_vec = np.correlate(direction, direction, 'full')
    # Coarse bracket

    alphas = np.linspace(lo, hi, n_eval)

    if min_idx == 0:

    elif min_idx == n_eval - 1:

    f1 = float(np.max(S0 + c1 * T_vec - c1 * c1 * C_vec)) * dx

def _gradient_polish(h, deadline, n_iters=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    tau = 0.01
    for _ in range(n_iters):
        if time.time() > deadline or stall > 30:
            break
        obj, grad = smooth_obj_grad(h, cur_c5 - 0.001, tau)
        grad_dir = -(grad - np.mean(grad))
        if np.linalg.norm(grad_dir) < 1e-14:
            break
        alpha, val = _line_search_dir(h, grad_dir, S0, dx)
        if alpha > 1e-10 and val < cur_c5 - 1e-12:
            h = h + alpha * grad_dir
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            tau *= 0.7
            if tau < 1e-6:
                break
    return best_h, best_c5

    target = n / 2.0
