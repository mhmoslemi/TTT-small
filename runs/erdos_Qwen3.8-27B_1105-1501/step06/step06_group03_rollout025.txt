def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _exact_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, r, n, dx):
    k = r - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:k+m]) * dx
            if k > 0:
                g[k:k+m] += -h[:m] * dx
    else:
        d = -k
        m = n - d
        if m > 0:
            g[d:d+m] += (1.0 - h[:m]) * dx
            g[:m] += -h[d:d+m] * dx
    return g

def _soft_dir(h, S, n, theta, step):
    M = float(S.max())
    sh = (S - M) / theta
    sh -= sh.max()
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-300:
        return None
    w = w / ws
    tk = min(30, 2 * n - 1)
    ti = np.argsort(-S)[:tk]
    wt = w[ti]
    dx = 2.0 / n
    G = np.array([_grad_row(h, r, n, dx) for r in ti])
    gs = wt @ G
    gs -= gs.mean()
    nm = np.linalg.norm(gs)
    if nm < 1e-12:
        return None
    p = -gs / nm
    pm = np.abs(p).max()
    if pm > step:
        p *= step / pm
    return p

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(xn, xo, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, nb):
    h = np.zeros(n)
    if nb < 2:
        nb = 2
    bounds = sorted(rng.choice(np.arange(1, n), size=min(nb - 1, n - 1), replace=False))
    bounds = [0] + list(bounds) + [n]
    for b in range(len(bounds) - 1):
        if rng.random() < 0.5:
            h[bounds[b]:bounds[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _nesterov_smooth(h, deadline, alpha0=0.003):
    """Nesterov accelerated gradient on smoothed max."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h, target)
    y = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha0
    k = 1
    lr0 = 0.8
    while time.time() < deadline:
        S = _exact_scores(y)
        M = float(S.max())
        sh = (S - M) / alpha
        sh -= sh.max()
        w = np.exp(sh)
        ws = w.sum()
        if ws < 1e-300:
            break
        w = w / ws
        tk = min(40, 2 * n - 1)
        ti = np.argsort(-S)[:tk]
        wt = w[ti]
        G = np.array([_grad_row(y, r, n, dx) for r in ti])
        grad = wt @ G
        grad -= grad.mean()
        lr = lr0 / (k + 1)
        y_new = y - lr * grad
        t_new = 1 + (t if False else 1.0) / 2 * (1 + np.sqrt(4 + 1.0))
        t_val = 1 + (1.0 + np.sqrt(1 + 4 * k * k)) / 2
        beta = (t_val - 1) / t_new if t_new > 0 else 0
        beta = (t_val - 1) / ((1 + np.sqrt(1 + 4 * (k + 1) * (k + 1))) / 2)
        y = _project(y_new + beta * (y_new - y), target)
        c5_y = _exact_c5(y)
        if c5_y < best_c5:
            best_c5 = c5_y
            best_h = y.copy()
        k += 1
        if k % 50 == 0:
            alpha *= 0.95
        if alpha < 0.0002:
            alpha = 0.0002
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (8%)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7]:
        for ph in [0.0, 0.7, 1.5, 2.3, 3.1]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(10):
        starts.append(_block_init(n0, rng, int(rng.integers(3, 16))))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        hs, cs = _local_opt(s, sub_dl, rad0=0.08, K=12)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()

    # Phase 2: Main local opt (30%)
    d2 = t0 + budget_s * 0.38
    h2, c2 = _local_opt(best_h, d2, rad0=0.12, K=15, use_sm=True)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2.copy()

    # Phase 3: Nesterov smooth (10%)
    d3 = t0 + budget_s * 0.48
    h3, c3 = _nesterov_smooth(best_h, d3, alpha0=0.003)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3.copy()

    # Phase 4: Binary SA + polish (15%)
    d4 = t0 + budget_s * 0.63
    sa_dl = min(d4, t0 + budget_s * 0.56)
    h4, c4 = _binary_sa(best_h, sa_dl, rng)
    if c4 < best_c5:
        best_c5 = c4
        best_h = h4.copy()
    if time.time() < d4:
        h4b, c4b = _local_opt(best_h, d4, rad0=0.05, K=15)
        if c4b < best_c5:
            best_c5 = c4b
            best_h = h4b.copy()

    # Phase 5: Systematic pair refinement (12%)
    d5 = t0 + budget_s * 0.75
    h5, c5 = _pair_refine(best_h, d5, max_passes=400)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5.copy()

    # Phase 6: Multi-pair (5%)
    d6 = t0 + budget_s * 0.80
    h6, c6 = _multi_pair(best_h, d6)
    if c6 < best_c5:
        best_c5 = c6
        best_h = h6.copy()

    # Phase 7: Resolution sweep (15%)
    d7 = t0 + budget_s * 0.95
    bn = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, -24, 24, -28, 28]
    tried = set()
    for dn in offsets:
        if time.time() > d7:
            break
        nt = bn + dn
        if nt < 40 or nt > 260 or nt % 2 != 0:
            continue
        if nt in tried:
            continue
        tried.add(nt)
        hr = _resample(best_h, nt)
        cr = _exact_c5(hr)
        if cr < best_c5 + 0.004:
            sub_dl = min(d7, time.time() + budget_s * 0.015)
            hr, cr = _local_opt(hr, sub_dl, rad0=0.08, K=12)
            if time.time() < d7 - 1.0:
                sub_dl2 = min(d7 - 1.0, time.time() + budget_s * 0.01)
                hr2, cr2 = _pair_refine(hr, sub_dl2, max_passes=50)
                if cr2 < cr:
                    hr, cr = hr2, cr2
            if cr < best_c5 - 1e-13:
                best_c5 = cr
                best_h = hr.copy()

    # Phase 8: Perturbation + re-opt (5%)
    d8 = deadline
    scale = 0.02
    for _ in range(500):
        if time.time() > d8:
            break
        if best_c5 < 0.3807:
            break
        hp = best_h + rng.normal(0, scale, len(best_h))
        hp = _project(hp, len(best_h) / 2.0)
        sub_dl = min(d8, time.time() + budget_s * 0.004)
        ht, ct = _local_opt(hp, sub_dl, rad0=0.04, K=10)
        if ct < best_c5 - 1e-13:
            best_c5 = ct
            best_h = ht.copy()
            scale = max(0.002, scale * 0.75)
        else:
            scale = min(0.1, scale * 1.3)

    # Final pair polish
    if time.time() < deadline:
        hf, cf = _pair_refine(best_h, deadline, max_passes=200)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Final LP polish
    if time.time() < deadline:
        hf, cf = _local_opt(best_h, deadline, rad0=0.03, K=12)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
