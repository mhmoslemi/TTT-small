def _resample(h, n_new):

    if n_new == n:

        M = cur_c5

        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:10]
        
        # Compute gradient of sum of active correlations
        K = len(active)
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                if s == 0:
                    grad[:m] += (1.0 - 2.0 * h[:m])
                else:
                    grad[:m] += (1.0 - h[s:s+m])
                    grad[s:s+m] += (-h[:m])
            else:
                g = -s
                m = n - g
                for j in range(n):
                    gval = 0.0
                    if j >= g:
                        gval += 1.0
                    if 0 <= j - g < m:
                        gval -= h[j - g]
                    if 0 <= j + g < n and j < m:
                        gval -= h[j + g]
                    grad[j] += gval
        
        # Project to sum-zero
        grad -= grad.mean()
        
        # Negative gradient direction
        d = -grad
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            stall += 5
            continue
        d = d / d_norm
        
        alpha, val = _exact_1d_minimax_quadratic(h, d, n_corr_top=K + 10)
        
        if val < cur_c5 - 1e-15:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                # Try half step
                h_new = h + 0.5 * alpha * d
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c5_new
                    stall = 0
                    if cur_c5 < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                else:
                    stall += 1
                    step_scale *= 0.8
            step_scale = min(step_scale * 1.05, 0.2)
        else:
            stall += 1
            step_scale *= 0.7
            # Occasionally try random direction
            if stall % 5 == 0:
                r = rng.normal(0, 1, n)
                r -= r.mean()
                r /= np.linalg.norm(r)
                alpha_r, val_r = _exact_1d_minimax_quadratic(h, r, n_corr_top=20)
                if val_r < cur_c5 - 1e-14:
                    h_new = _project(h + alpha_r * r, target)
                    c5_new = _exact_c5(h_new)
                    if c5_new < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = c5_new
                        stall = 0
                        if cur_c5 < best_c5:
                            best_c5 = c5_new
                            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        # Generate a structured initial
        h0 = np.zeros(n0)
        pos = 0
        while pos < n0:
            blen = max(1, int(rng.integers(2, max(3, n0 // 4 + 1))))
            if rng.random() < 0.5:
                h0[pos:pos + blen] = 1.0
            pos += blen
        h0 = _project(h0, n0 / 2.0)
    
    h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(h)
    best_h = h.copy()
    best_n = n0
    
    # Phase 1: Gradient polish on initial (10% time)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _grad_based_polish(h, d1, max_iter=300, rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)
    
    # Phase 2: Bundle search (15% time)
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _bundle_search(best_h, d2, n_dir=15, max_rounds=150, rng=rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)
    
    # Phase 3: SLSQP polish (8% time)
    d3 = t0 + budget_s * 0.33
    h3, c5_3 = _slsqp_polish(best_h, d3, max_outer=10)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)
    
    # Phase 4: Try different resolutions (27% time)
    d4 = t0 + budget_s * 0.60
    tried_n = set()
    # Try a range of n values around current best
    n_candidates = []
    for dn in range(-20, 21, 2):
        nt = best_n + dn
        if 60 <= nt <= 180 and nt % 2 == 0 and nt not in tried_n:
            n_candidates.append(nt)
    # Also try some specific values
    for nt in [80, 90, 100, 110, 120, 124, 126, 128, 130, 132, 134, 136, 140, 144, 148, 150, 152, 156, 160]:
        if nt % 2 == 0 and nt not in tried_n and 60 <= nt <= 180:
            n_candidates.append(nt)
    
    # Sort by distance from current best n
    n_candidates.sort(key=lambda x: abs(x - best_n))
    
    for nt in n_candidates:
        if time.time() > d4:
            break
        if nt in tried_n:
            continue
        tried_n.add(nt)
        
        h_r = _resample(best_h, nt)
        # Quick gradient polish
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_r, c5_r = _grad_based_polish(h_r, sub_dl, max_iter=100, rng=rng)
        
        # If promising, do more
        if c5_r < best_c5 + 0.001 and time.time() < d4 - budget_s * 0.02:
            sub_dl2 = min(d4 - budget_s * 0.01, time.time() + budget_s * 0.03)
            h_r, c5_r = _bundle_search(h_r, sub_dl2, n_dir=10, max_rounds=80, rng=rng)
        
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt
    
    # Phase 5: SLSQP on best (7% time)
    d5 = t0 + budget_s * 0.67
    h5, c5_5 = _slsqp_polish(best_h, d5, max_outer=10)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
        best_n = len(h5)
    
    # Phase 6: Aggressive bundle search on best (15% time)
    d6 = t0 + budget_s * 0.82
    h6, c5_6 = _bundle_search(best_h, d6, n_dir=20, max_rounds=300, rng=rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)
    
    # Phase 7: Fine resolution sweep around best (8% time)
    d7 = t0 + budget_s * 0.90
    for dn in [-4, -2, 2, 4, -6, 6, -8, 8]:
        if time.time() > d7:
            break
        nt = best_n + dn
        if nt < 60 or nt > 200 or nt % 2 != 0:
            continue
        if nt in tried_n:
            continue
        tried_n.add(nt)
        h_r = _resample(best_h, nt)
        sub_dl = min(d7, time.time() + budget_s * 0.015)
        h_r, c5_r = _grad_based_polish(h_r, sub_dl, max_iter=80, rng=rng)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = nt
    
    # Phase 8: Final polish (10% time)
    d8 = deadline
    h8, c5_8 = _slsqp_polish(best_h, min(d8, time.time() + budget_s * 0.04), max_outer=15)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
        best_n = len(h8)
    
    if time.time() < d8 - 1:
        h8b, c5_8b = _grad_based_polish(best_h, d8, max_iter=200, rng=rng)
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()
            best_n = len(h8b)
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
