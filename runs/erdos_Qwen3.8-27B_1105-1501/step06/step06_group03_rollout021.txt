def _exact_c5(h):

    dx = 2.0 / n

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx

            c1 = b - gr * (b - a)

def _lp_direction_exp(h, S, n, radius):
    M = float(S.max())
    tau = 0.03 * M + 1e-8
    weights = np.exp((S - M) / tau)
    weights /= weights.sum()
    threshold = 0.01 * weights.max()
    active = np.where(weights >= threshold)[0]
    if len(active) > 30:
        active = np.argsort(-S)[:30]
    if len(active) == 0:
        active = np.argsort(-S)[:10]
    G = _grad_rows(h, active, n)
    n_act = len(active)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _smooth_direction(S, h, n, beta):
    shifted = beta * S - beta * S.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(S)) / len(S)
    else:
        w = w / w_sum
    top_k = min(40, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad_smooth = w[top_idx] @ G
    grad_smooth -= grad_smooth.mean()
    norm = np.linalg.norm(grad_smooth)
    if norm < 1e-12:
        return None
    return -grad_smooth / norm

                h = _project(h, target)

            else:
                break

def _smooth_descent(h, deadline):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _spectral_init(n, rng):
    """Create initial h with approximately flat spectrum."""
    m = n // 2
    target = n / 2.0
    phases = rng.uniform(0, 2 * np.pi, size=m)
    h = np.zeros(n)
    for k in range(1, m + 1):
        amp = rng.uniform(0.3, 1.0)
        h += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phases[k - 1])
    h = _project(h, target)
    return h

def _pair_refine(h, deadline, max_passes=150):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    S = _corr_scaled(h)
    G_all = _grad_rows(h, np.arange(2 * n - 1), n)
    total_grad = S @ G_all
    total_grad -= total_grad.mean()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.35)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        S = _corr_scaled(h)
                        G_all = _grad_rows(h, np.arange(2 * n - 1), n)
                        total_grad = S @ G_all
                        total_grad -= total_grad.mean()
        if not improved:
            break
    return best_h, best_c5

        best_h = h3.copy()

    # Phase 4: LP with smooth backup (12% time)
    d4 = t0 + budget_s * 0.51
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.08, use_smooth=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Smoothed gradient descent (10% time)
    d5 = t0 + budget_s * 0.61
    h5, c5_5 = _smooth_descent(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Binary SA with block moves (12% time)
    d6 = t0 + budget_s * 0.73
    h6a, c5_6a = _binary_sa_block(best_h, t0 + budget_s * 0.68, rng)
    if c5_6a < best_c5:
        best_c5 = c5_6a
        best_h = h6a.copy()
    if time.time() < d6:
        h6b, c5_6b = _lp_loop(best_h, d6, radius_init=0.06, use_smooth=False)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Pair refinement (8% time)
    d7 = t0 + budget_s * 0.81
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=300)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Multi-resolution refinement (10% time)
    d8 = t0 + budget_s * 0.91
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, 8, -8, 10, -12, 12, 14, -16, 16, 20, -20, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d8:
            break
        n_try = base_n + dn
        if n_try < 60 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.003:
            sub_dl = min(d8, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, use_smooth=True)
            if c5_r < best_c5 and time.time() < d8 - budget_s * 0.015:
                sub_dl2 = min(d8, time.time() + budget_s * 0.01)
                h_r2, c5_r2 = _binary_sa_block(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d8:
                    h_r, c5_r = _lp_loop(h_r, min(d8, time.time() + budget_s * 0.008),
                                         radius_init=0.05, use_smooth=False)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 9: Aggressive perturbation (5% time)
    d9 = t0 + budget_s * 0.96
    scale = 0.02
    for attempt in range(500):
        if time.time() > d9:
            break
        if best_c5 < 0.3807:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d9, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, use_smooth=True)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 10: Final polish (4% time)
    if time.time() < deadline:
        h_f, c5_f = _pair_refine(best_h, deadline, max_passes=100)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    if time.time() < deadline - 0.5:
        h_f2, c5_f2 = _lp_loop(best_h, deadline, radius_init=0.03, use_smooth=True)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    n_final = len(best_h)
    return best_h, final_c5, n_final
