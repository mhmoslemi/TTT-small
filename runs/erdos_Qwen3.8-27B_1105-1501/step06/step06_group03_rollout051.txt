import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 5e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _active_indices(S, M, tol_frac=0.06, max_count=30):
    tol = max(tol_frac * M, 1e-6)
    near = np.where(S >= M - tol)[0]

    combined = np.unique(np.concatenate([near, sorted_idx]))

    return combined

def _lp_direction(h, S, n, K, radius):

    c_vec[n] = -1.0

    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)

    return None

def _smooth_direction(h, S, n, tau):
    M = float(S.max())
    shifted = (S - M) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_k = min(25, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad_smooth = w[top_idx] @ G
    grad_smooth -= grad_smooth.mean()
    if np.abs(grad_smooth).max() < 1e-12:
        return None
    norm = np.linalg.norm(grad_smooth)
    if norm < 1e-12:
        return None
    return -grad_smooth / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks < 2:
        n_blocks = 2
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks - 1, n - 1), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(len(boundaries) - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _pair_refine_grad(h, deadline, max_passes=150):

    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 5e-15:

                    h = _project(h, target)
                    cur = _exact_c5(h)

                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Binary SA (10% time)
    d6 = t0 + budget_s * 0.74
    h6, c5_6 = _binary_sa(best_h, d6, rng)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6 + budget_s * 0.03:
        h6b, c5_6b = _lp_loop(best_h, d6 + budget_s * 0.03, radius_init=0.05, K=16)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Multi-resolution (12% time)
    d7 = t0 + budget_s * 0.86
    base_n = len(best_h)
    offsets = [0, 2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12, 14, -14, 16, -16, 20, -20, 24, -24, 28, -28, 32, -32]
    tried = set()
    for dn in sorted(offsets, key=abs):
        if time.time() > d7:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d7, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=14)
            if c5_r < best_c5:
                sub_dl2 = min(d7, time.time() + budget_s * 0.008)
                h_r2, c5_r2 = _continuous_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d7 - 1:
                    h_r, c5_r = _lp_loop(h_r, min(d7, time.time() + budget_s * 0.005),
                                         radius_init=0.04, K=14)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Target-driven refinement (6% time)
    d8 = t0 + budget_s * 0.92
    target = best_c5 - 0.0002
    h8, c5_8 = _target_driven(best_h, d8, target, rng)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Phase 9: Pair + multi-pair polish (5% time)
    d9 = t0 + budget_s * 0.965
    h9, c5_9 = _pair_refine_grad(best_h, d9, max_passes=200)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()
    if time.time() < d9 + budget_s * 0.01:
        h9b, c5_9b = _multi_pair_refine(best_h, d9 + budget_s * 0.01)
        if c5_9b < best_c5:
            best_c5 = c5_9b
            best_h = h9b.copy()

    # Phase 10: Final LP polish
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.03, K=20, use_smooth=True)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
