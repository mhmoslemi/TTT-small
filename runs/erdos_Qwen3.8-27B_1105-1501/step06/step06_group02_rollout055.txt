import numpy as np
import time
from scipy.optimize import minimize

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    for d in range(-(n-1), n):
        q = d + (n - 1)
        if d >= 0:
            m = n - d
            # G[q, d..n-1] += (1 - h[0..m-1])
            # G[q, 0..m-1] += -h[d..n-1]
            if m > 0:

                G[q, :m] += -h[d:]
        else:
            r = -d
            m = n - r
            if m > 0:
                G[q, r:] += (1.0 - h[:m])
                G[q, :m] += -h[r:]
    return G

                    continue
                tried.add(key)
                alpha, new_c5 = _golden_pair(h, i, j, dx, S0)
                if new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.4:
                idx = rng.integers(n, size=6)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.08, 0.08)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)
    
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0
    
    target_c5 = 0.3808
    
    # Phase 1: QP-based optimization (main phase, 50% time)
    d1 = t0 + budget_s * 0.50
    cur_h = best_h.copy()
    cur_c5 = best_c5
    cur_n = best_n
    stall_global = 0
    
    rho_schedule = [0.20, 0.15, 0.12, 0.09, 0.07, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01]
    rho_idx = 0
    lam_z = 0.8
    lam_d = 0.2
    maxiter_qp = 200
    
    while time.time() < d1 and stall_global < 100:
        cur_n = len(cur_h)
        G = compute_grad_matrix(cur_h)
        R = raw_corr(cur_h)
        dx = 2.0 / cur_n
        cur_max_raw = float(np.max(R))
        cur_c5_val = cur_max_raw * dx
        
        # Target: slightly below current, approaching target_c5
        target_raw_desired = target_c5 * cur_n / 2.0
        gap = cur_max_raw - target_raw_desired
        if gap < 0.01:
            target_raw = cur_max_raw - 0.005 * cur_max_raw * dx * cur_n / 2.0
            # Actually work in raw units
            target_raw = cur_max_raw * (1 - 0.005)
        else:
            # Interpolate: start close, tighten gradually
            frac = min(1.0, 0.1 + 0.9 * stall_global / 50.0)
            target_raw = cur_max_raw - frac * gap * 0.3
        
        # Ensure target is below current max
        target_raw = min(target_raw, cur_max_raw * 0.999)
        # Ensure target is above 0
        target_raw = max(target_raw, 0.0)
        
        rho = rho_schedule[min(rho_idx, len(rho_schedule) - 1)]
        
        d_dir = solve_qp_direction(cur_h, G, R, target_raw, rho, lam_z, lam_d, maxiter_qp)
        
        if d_dir is not None:
            d_dir -= np.mean(d_dir)
            alpha, h_new, c5_new = line_search_exact(cur_h, d_dir, R)
            if c5_new < cur_c5_val - 1e-12:
                h_new = project(h_new, cur_n / 2.0)
                c5_new_check = c5_exact(h_new)
                if c5_new_check < cur_c5 - 1e-12:
                    cur_h = h_new
                    cur_c5 = c5_new_check
                    stall_global = 0
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur_h.copy()
                        best_n = cur_n
                    # Tighten rho after success
                    if rho_idx < len(rho_schedule) - 1 and stall_global % 5 == 0:
                        rho_idx += 1
                    continue
        
        # QP failed, try pairwise as fallback
        if rng.random() < 0.5 or stall_global > 20:
            sub_dl = min(d1, time.time() + budget_s * 0.02)
            ph, pc5 = _pairwise_search(cur_h, 15, sub_dl, rng)
            if pc5 < cur_c5 - 1e-12:
                cur_h = ph
                cur_c5 = pc5
                stall_global = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()
                    best_n = cur_n
            else:
                stall_global += 1
        
        # Random perturbation to escape
        if stall_global > 30:
            n_cur = len(cur_h)
            idx = rng.integers(n_cur, size=8)
            for ik in idx:
                cur_h[ik] += rng.uniform(-0.1, 0.1)
            cur_h = project(cur_h, n_cur / 2.0)
            cur_c5 = c5_exact(cur_h)
            stall_global = 0
        
        # Adjust hyperparameters
        if stall_global > 15:
            lam_z = min(0.95, lam_z + 0.05)
            lam_d = max(0.05, lam_d - 0.05)
            if rho_idx < len(rho_schedule) - 1:
                rho_idx += 1
    
    # Phase 2: Pairwise polish (15% time)
    d2 = t0 + budget_s * 0.65
    ph2, pc5_2 = _pairwise_search(best_h, 25, d2, rng)
    if pc5_2 < best_c5:
        best_c5 = pc5_2
        best_h = ph2.copy()
        best_n = len(best_h)
    
    # Phase 3: Multi-resolution (20% time)
    d3 = t0 + budget_s * 0.85
    tried_n = {best_n}
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, -10, 12, -12]:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = c5_exact(h_r)
        
        # Quick QP on this resolution
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        G_r = compute_grad_matrix(h_r)
        R_r = raw_corr(h_r)
        dx_r = 2.0 / n_try
        target_raw_r = min(c5_r * (1 - 0.01), target_c5 * n_try / 2.0 + 0.5 * (c5_r - target_c5 * n_try / 2.0))
        target_raw_r = max(target_raw_r, 0.1)
        
        for rho_try in [0.15, 0.1, 0.06, 0.04]:
            if time.time() > sub_dl:
                break
            d_dir = solve_qp_direction(h_r, G_r, R_r, target_raw_r, rho_try, 0.8, 0.2, 150)
            if d_dir is not None:
                d_dir -= np.mean(d_dir)
                alpha, h_new, c5_new = line_search_exact(h_r, d_dir, R_r)
                if c5_new < c5_r - 1e-12:
                    h_new = project(h_new, n_try / 2.0)
                    c5_new = c5_exact(h_new)
                    h_r = h_new
                    c5_r = c5_new
                    G_r = compute_grad_matrix(h_r)
                    R_r = raw_corr(h_r)
        
        # Quick pairwise
        sub_dl2 = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _pairwise_search(h_r, 20, sub_dl2, rng)
        
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 4: Final QP polish (8% time)
    d4 = t0 + budget_s * 0.93
    cur_h = best_h.copy()
    cur_c5 = best_c5
    for rho_try in [0.03, 0.02, 0.015, 0.01, 0.008, 0.005]:
        if time.time() > d4:
            break
        n_cur = len(cur_h)
        G = compute_grad_matrix(cur_h)
        R = raw_corr(cur_h)
        dx = 2.0 / n_cur
        target_raw = min(cur_c5 * n_cur / 2.0 * 0.999, target_c5 * n_cur / 2.0 * 1.01)
        target_raw = max(target_raw, 0.01)
        
        d_dir = solve_qp_direction(cur_h, G, R, target_raw, rho_try, 0.9, 0.1, 300)
        if d_dir is not None:
            d_dir -= np.mean(d_dir)
            alpha, h_new, c5_new = line_search_exact(cur_h, d_dir, R)
            if c5_new < cur_c5 - 1e-13:
                h_new = project(h_new, n_cur / 2.0)
                c5_new = c5_exact(h_new)
                if c5_new < cur_c5:
                    cur_h = h_new
                    cur_c5 = c5_new
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur_h.copy()
    
    # Phase 5: Final pairwise micro-polish (7% time)
    d5 = deadline
    ph5, pc5_5 = _pairwise_search(best_h, 30, d5, rng)
    if pc5_5 < best_c5:
        best_c5 = pc5_5
        best_h = ph5.copy()
        best_n = len(best_h)
    
    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    
    if not np.all(np.isfinite(best_h)) or np.max(best_h) > 1.0 + 1e-10 or np.min(best_h) < -1e-10:
        best_h = project(np.clip(best_h, 0, 1), len(best_h) / 2.0)
        final_c5 = c5_exact(best_h)
    
    return best_h, final_c5, len(best_h)
