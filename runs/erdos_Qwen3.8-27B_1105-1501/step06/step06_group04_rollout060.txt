import numpy as np
from scipy.optimize import minimize, linprog
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -10, 10)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 0.999
        else:
            mask = x > 0.001
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _row_grad(h, d):
    """Gradient of C_{n-1+d}(h) with respect to h, unscaled."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:]
    else:
        m = n + d  # d is negative, so m = n - |d|
        if m > 0:
            g[:m] += (1.0 - h[-d:])
            g[-d:] -= h[:m]
    return g

def _all_row_grads(h):
    """Build gradient matrix for all lags."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        G[r] = _row_grad(h, d)
    return G

def _lse_objective(h, beta, target):
    """LSE objective value and gradient."""
    n = len(h)
    dx = 2.0 / n
    scores = _corr_scaled(h)
    max_s = float(scores.max())
    z = beta * (scores - max_s)
    # Clip for numerical stability
    z = np.clip(z, -500, 500)
    exp_z = np.exp(z)
    sum_exp = float(exp_z.sum())
    if sum_exp < 1e-300:
        return max_s, np.zeros(n)
    p = exp_z / sum_exp
    obj = (max_s + np.log(sum_exp) / beta)
    # Gradient
    G = _all_row_grads(h)
    grad = dx * (p @ G)
    return float(obj), grad

def _solve_lse(h_init, beta, dx, maxiter=200, tol=1e-10):
    """Solve LSE minimization using SLSQP."""
    n = len(h_init)
    target = n / 2.0
    
    def fun(x):
        return _lse_objective(x, beta, target)[0]
    
    def grad(x):
        return _lse_objective(x, beta, target)[1]
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
         'jac': lambda x: np.ones(n)},
    ]
    bounds = [(0.0, 1.0)] * n
    x0 = h_init.copy()
    
    result = minimize(fun, x0, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': tol})
    if result.success or result.fun < _compute_c5(h_init):
        return _project(result.x, target)
    return h_init.copy()

def _line_search(h, d, dx):
    """Exact line search along direction d (assumed sum-zero)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    
    # Feasible interval
    lo, hi = 0.0, 0.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        lo = float(np.max(h[neg] / d[neg]))
    hi = min(hi, 1.0)
    lo = max(lo, -1.0)
    
    if hi - lo < 1e-14:
        return 0.0, _compute_c5(h)
    
    # Golden section search
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * Q)) * dx
    return opt_t, opt_val

def _active_grad_direction(h, scores, top_frac=0.15):
    """Direction from weighted gradient of top active constraints."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    max_s = float(scores.max())
    thr = max_s - top_frac * max_s
    active = np.where(scores >= thr)[0]
    if len(active) < 1:
        active = np.array([np.argmax(scores)])
    # Softmax weights over active
    sa = scores[active]
    z = 20.0 * (sa - sa.max())
    w = np.exp(z) / np.exp(z).sum()
    G = np.zeros((len(active), n))
    for idx, r in enumerate(active):
        d = r - (n - 1)
        G[idx] = _row_grad(h, d)
    grad = dx * (w @ G)
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return -grad / norm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return _project(h, n / 2.0)

def _binary_quantize(h):
    """Quantize h to binary by selecting top n/2 values."""
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def _direct_minimax(h_init, deadline):
    """Direct minimax with SLSQP."""
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    m = 2 * n - 1
    
    # Initial M
    scores = _corr_scaled(h_init)
    M0 = float(scores.max()) + 0.01
    
    def objective(y):
        return y[0]
    
    def obj_grad(y):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def ineq_cons(y):
        h = y[1:]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return y[0] - dx * corr
    
    def ineq_jac(y):
        h = y[1:]
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        for r in range(m):
            d = r - (n - 1)
            J[r, 1:] = -dx * _row_grad(h, d)
        return J
    
    def eq_cons(y):
        return np.sum(y[1:]) - target
    
    def eq_jac(y):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    
    bounds = [(0.0, M0 + 0.1)] + [(0.0, 1.0)] * n
    y0 = np.concatenate([[M0], h_init])
    
    constraints = [
        {'type': 'ineq', 'fun': ineq_cons, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_cons, 'jac': eq_jac},
    ]
    
    try:
        result = minimize(objective, y0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 80, 'ftol': 1e-12})
        if result.fun < M0:
            h_new = _project(result.x[1:], target)
            return h_new, _compute_c5(h_new)
    except Exception:
        pass
    return h_init.copy(), _compute_c5(h_init)

def _lp_direction(h, scores, radius, top_k=12):

        sweeps += 1
        improved = False
        S0 = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if (h[i] <= 0.001 and h[j] >= 0.999) or (h[i] >= 0.999 and h[j] <= 0.001):
                    continue
                for sign in [1.0, -1.0]:
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
                    Q = -np.correlate(d, d, mode='full')
                    lo, hi = 0.0, 0.0
                    pos = d > 1e-14
                    neg = d < -1e-14
                    if np.any(pos):
                        hi = float(np.min((1.0 - h[pos]) / d[pos]))
                    if np.any(neg):
                        lo = float(np.max(h[neg] / d[neg]))
                    hi = min(hi, 1.0)
                    lo = max(lo, -1.0)
                    if hi - lo < 1e-14:
                        continue
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    a, b = lo, hi
                    c1 = b - gr * (b - a)
                    c2 = a + gr * (b - a)
                    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * (2.0 / n)
                    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * (2.0 / n)
                    for _ in range(30):
                        if b - a < 1e-12:
                            break
                        if f1 < f2:
                            b, c2, f2 = c2, c1, f1
                            c1 = b - gr * (b - a)
                            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * (2.0 / n)
                        else:
                            a, c1, f1 = c1, c2, f2
                            c2 = a + gr * (b - a)
                            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * (2.0 / n)
                    opt_t = (a + b) / 2.0
                    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * Q)) * (2.0 / n)
                    if opt_val < best_c5 - 1e-14:
                        h = h + opt_t * d
                        h = _project(h, target)
                        S0 = np.correlate(h, 1.0 - h, mode='full')
                        cur = _compute_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved = True
        if not improved:
            break
    return best_h, best_c5


def _linear_search_loop(h, deadline):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    radius = 0.10
    stall = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline and stall < 50:
        scores = _corr_scaled(h)
        d = _active_grad_dir(h, top_frac=0.12, sharpness=20.0)
        if d is None:
            d = _lp_dir(h, scores, radius, top_k=12)
        if d is None:
            d = rng.normal(0, 1, n)
            d -= d.mean()
            d = d / (np.max(np.abs(d)) + 1e-15) * radius
        d = d * radius
        pos = d > 0
        neg = d < 0
        if np.any(pos):
            tm = np.min((1.0 - h[pos]) / d[pos])
            if tm < 1.0:
                d = d * min(tm, 0.99)
        if np.any(neg):
            tm = np.min(h[neg] / (-d[neg]))
            if tm < 1.0:
                d = d * min(tm, 0.99)
        if np.max(np.abs(d)) > 1e-14:
            alpha, c5_try = _line_search(h, d)
            if c5_try < best_c5 - 1e-14:
                h = _project(h + alpha * d, target)
                cur = _compute_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                radius = min(radius * 1.04, 0.4)
                continue
        # Try random direction
        d = rng.normal(0, 1, n)
        d -= d.mean()
        d = d / (np.max(np.abs(d)) + 1e-15) * radius * 0.5
        alpha, c5_try = _line_search(h, d)
        if c5_try < best_c5 - 1e-14:
            h = _project(h + alpha * d, target)
            cur = _compute_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
                continue
        stall += 1
        radius *= 0.9
        if radius < 0.01:
            radius = 0.08
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
    return best_h, best_c5


def _lse_anneal(h, deadline, betas=(10, 25, 50, 100, 200, 400, 800)):
    n = len(h)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    for beta in betas:
        if time.time() >= deadline:
            break
        sub_time = min(20.0, max(5.0, (deadline - time.time()) / len(betas)))
        sub_dl = time.time() + sub_time
        maxiter = max(30, int(sub_time * 5))
        h_new = _solve_lse(best_h, beta, maxiter=maxiter)
        h_new = _project(h_new, n / 2.0)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)


def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return _project(h, n / 2.0)


def _binary_quantize(h, rng=None):
    n = len(h)
    target = n // 2
    if rng is not None:
        noise = rng.normal(0, 0.01, n)
        h = h + noise
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def _binary_sa(h_start, deadline, rng, max_iters=20000):
    n = len(h_start)
    dx = 2.0 / n
    target = n // 2
    sorted_idx = np.argsort(-h_start)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.003
    T_min = 0.00005
    cooling = 0.9997
    patience = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 5000:
            break
        i0 = rng.integers(target)
        i1 = target + rng.integers(target)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 5.0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: LSE annealing (20%)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _lse_anneal(best_h, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Linear search loop (15%)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _linear_search_loop(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Full pair sweep (15%)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _full_pair_sweep(best_h, d3, max_sweeps=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Direct minimax (10%)
    d4 = t0 + budget_s * 0.60
    h4 = _solve_minimax(best_h, maxiter=80)
    c5_4 = _compute_c5(h4)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4:
        h4b = _solve_minimax(best_h + rng.normal(0, 0.03, len(best_h)), maxiter=60)
        c5_4b = _compute_c5(h4b)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()
    # More LSE after minimax
    if time.time() < d4:
        h4c = _solve_lse(best_h, 300, maxiter=100)
        c5_4c = _compute_c5(h4c)
        if c5_4c < best_c5:
            best_c5 = c5_4c
            best_h = h4c.copy()

    # Phase 5: Multi-start LSE (12%)
    d5 = t0 + budget_s * 0.72
    n = len(best_h)
    starts = []
    for freq in [1, 2, 3, 4, 5, 7]:
        for ph in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0, np.pi]:
            starts.append(_sine_init(n, freq, ph))
    for p in [3, 4, 5, 6, 7, 8, 10, 12, 16]:
        for duty in [0.3, 0.4, 0.5, 0.6]:
            starts.append(_periodic_init(n, p, duty))
    starts.append(_binary_quantize(best_h))
    for s in starts:
        if time.time() >= d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.006)
        h_s, c5_s = _lse_anneal(s, sub_dl, betas=(30, 80, 200))
        h_s2, c5_s2 = _full_pair_sweep(h_s, min(d5, time.time() + budget_s * 0.003), max_sweeps=5)
        if c5_s2 < c5_s:
            h_s, c5_s = h_s2, c5_s2
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Binary SA (8%)
    d6 = t0 + budget_s * 0.80
    h6, c5_6 = _binary_sa(best_h, d6, rng, max_iters=15000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6:
        h6b, c5_6b = _lse_anneal(best_h, d6, betas=(50, 150, 400))
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Multi-resolution (8%)
    d7 = t0 + budget_s * 0.88
    best_n = len(best_h)
    tried = set()
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, 14, -4]:
        if time.time() >= d7:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 300 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.005)
        h_r, c5_r = _lse_anneal(h_r, sub_dl, betas=(30, 100, 300))
        h_r2, c5_r2 = _full_pair_sweep(h_r, min(d7, time.time() + budget_s * 0.003), max_sweeps=5)
        if c5_r2 < c5_r:
            h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 8: Final aggressive polish (12%)
    d8 = deadline
    h8, c5_8 = _full_pair_sweep(best_h, min(d8, t0 + budget_s * 0.96), max_sweeps=20)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8 - 5:
        h8b, c5_8b = _lse_anneal(best_h, d8 - 5, betas=(100, 300, 600))
        if c5_8b < best_c5:
            best_c5 = c5_8b
            best_h = h8b.copy()
        if time.time() < d8:
            h8c, c5_8c = _full_pair_sweep(best_h, d8, max_sweeps=10)
            if c5_8c < best_c5:
                best_c5 = c5_8c
                best_h = h8c.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        h_fb = _project(h0, n0 / 2.0)
        return h_fb, _compute_c5(h_fb), n0
    return best_h, final_c5, len(best_h)
