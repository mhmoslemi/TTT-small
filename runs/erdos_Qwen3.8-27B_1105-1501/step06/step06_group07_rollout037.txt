def _accel_pg(h0, deadline, eps_init=0.05, eps_min=0.001, eps_decay=0.95):
    """Accelerated projected gradient with LSE smoothing."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    y = h0.copy()  # For Nesterov momentum
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    eps = eps_init
    L = 1.0  # Step size (inverse Lipschitz constant estimate)
    t = 1.0  # Momentum parameter
    
    while time.time() < deadline:
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h + beta * (h - y)
        
        val, grad = _lse_value_grad(y, eps)
        
        # Gradient step
        y_new = y - grad / L
        h_new = _project(y_new, target)
        
        # Check convergence / improvement
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new

            L *= 1.5
            t = t_new  # Reset momentum
            continue
        
        h = h_new
        t = t_new
        eps = max(eps_min, eps * eps_decay)
    
    return best_h, best_c5

def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_val_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    s = (corr - cmax) / eps
    ex = np.exp(s)
    exsum = np.sum(ex)
    if exsum < 1e-300:
        exsum = 1e-300
    val = cmax + eps * np.log(exsum)
    p = ex / exsum
    grad = np.zeros(n)
    for k in range(n):
        w = p[n-1+k] * dx
        if abs(w) < 1e-16: continue
        m = n - k
        grad[k:k+m] += w * (1.0 - h[:m])
        grad[:m] -= w * h[k:k+m]
    for q in range(1, n):
        w = p[n-1-q] * dx
        if abs(w) < 1e-16: continue
        m = n - q
        grad[:m] += w * (1.0 - h[q:q+m])
        grad[q:q+m] -= w * h[:m]
    return val, grad

def _resample(h, n_new):

            h = _project(h + best_t * d, target)
            best_c5 = best_f
            best_h = h.copy()
            if best_t > 0.5 * step:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _binary_local_search(h0, deadline, max_iters=50000):
    """Local search on binary solutions."""
    n = len(h0)
    if n % 2 != 0:
        return h0.copy(), _exact_c5(h0)
    target = n // 2
    sorted_idx = np.argsort(-h0)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    rng = np.random.default_rng(12345)

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
