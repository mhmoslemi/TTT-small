def _soft_smooth_direction(h, S, n, beta):
    """Direction based on soft-max gradient with temperature 1/beta."""
    M = float(S.max())
    # Shift for numerical stability
    shifted = beta * (S - M)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    # Use top-K rows
    top_k = min(30, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad_smooth = w[top_idx] @ G
    grad_smooth -= grad_smooth.mean()
    norm = np.linalg.norm(grad_smooth)
    if norm < 1e-14:
        return None
    return -grad_smooth / norm

def _block_sa(h_init, deadline, rng):
    """Binary SA with block moves and 2+2 moves."""
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    # Start from binary projection

    zeros = list(np.where(h <= 0.5)[0])
    
    def _try_move(h, i, j, cur_c5):
        """Single swap."""
        h[i] = 0.0
        h[j] = 1.0

        return new_c5

                j = zeros[rng.integers(len(zeros))]
                if i == j:
                    continue

                h[i] = 0.0
                h[j] = 1.0

                i2 = ones[rng.integers(len(ones))]

                        patience = 0
                    else:
                        patience += 1

                    zeros = list(np.where(h <= 0.5)[0])
                else:
                    h[i1] = vals[0]
                    h[i2] = vals[1]
                    h[j1] = vals[2]
                    h[j2] = vals[3]

                block_0 = h[found_j:found_j + block_len].copy()

                h[found_j:found_j + block_len] = block_1

                    h[found_j:found_j + block_len] = block_0

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()

    rng = np.random.default_rng(888)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 60)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0

                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _strategic_init(n, rng):
    """Create structured initializations based on difference set ideas."""
    # Try to create sequences with good autocorrelation properties
    # Basis: periodic patterns with varying period
    period = int(rng.integers(3, 7))
    pattern = [1.0] * int(period * 0.5) + [0.0] * int(period - period * 0.5)
    h = np.tile(pattern, n // period + 1)[:n]
    # Add some random perturbation
    mask = rng.random(n) < 0.1
    h[mask] = 1.0 - h[mask]
    return _project(h, n / 2.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)

            m = n - k

            m = n - d

    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _soft_dir(h, S, n, beta):
    M = float(S.max())
    shifted = np.clip(beta * (S - M), -400, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_k = min(40, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad = w[top_idx] @ G
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return -grad / norm

def _lp_dir(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.05 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 25:
        combined = sorted_idx[:25]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _cont_opt(h_init, deadline, radius_init=0.12, K=15, beta_init=50):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    beta = beta_init
    rng = np.random.default_rng(31337)
    while time.time() < deadline:
        if stall > 70:
            scale = 0.02 + 0.015 * min(stall / 50.0, 3.0)
            h = h + rng.normal(0, scale, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.08
            stall = 0
            beta = beta_init
            continue
        S = _corr_scaled(h)
        p = _lp_dir(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            p = _soft_dir(h, S, n, beta)
            if p is not None:
                p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            beta = max(20, beta * 0.97)
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.05, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.95, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0:
                beta = min(600, beta * 1.12)
            if stall % 15 == 0 and radius < 0.012:
                radius = 0.04
    return best_h, best_c5

def _block_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 1e-6
    cooldown = 0.99996
    patience = 0
    max_patience = 25000
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 4:
                break
            T = max(T_min, T * 0.6)
            patience = 0
        if T < T_min:
            break
        for _ in range(400):
            if time.time() > deadline:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            r = rng.random()
            if r < 0.55:
                i = ones[int(rng.integers(len(ones)))]
                j = zeros[int(rng.integers(len(zeros)))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = _exact_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                    ones = list(np.where(h > 0.5)[0])
                    zeros = list(np.where(h <= 0.5)[0])
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            elif r < 0.82 and len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[int(rng.integers(len(ones)))]
                i2 = ones[int(rng.integers(len(ones)))]
                j1 = zeros[int(rng.integers(len(zeros)))]
                j2 = zeros[int(rng.integers(len(zeros)))]
                if len(set([i1, i2, j1, j2])) < 4:
                    continue
                sv = [h[i1], h[i2], h[j1], h[j2]]
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                new_c5 = _exact_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                    ones = list(np.where(h > 0.5)[0])
                    zeros = list(np.where(h <= 0.5)[0])
                else:
                    h[i1] = sv[0]
                    h[i2] = sv[1]
                    h[j1] = sv[2]
                    h[j2] = sv[3]
            else:
                bl = int(rng.integers(1, min(5, min(len(ones), len(zeros)) + 1)))
                fi = None
                for _ in range(12):
                    st = int(rng.integers(0, n - bl + 1))
                    if all(h[st + b] > 0.5 for b in range(bl)):
                        fi = st
                        break
                if fi is None:
                    continue
                fj = None
                for _ in range(12):
                    st = int(rng.integers(0, n - bl + 1))
                    if st < fi + bl and st + bl > fi:
                        continue
                    if all(h[st + b] <= 0.5 for b in range(bl)):
                        fj = st
                        break
                if fj is None:
                    continue
                b1 = h[fi:fi + bl].copy()
                b0 = h[fj:fj + bl].copy()
                h[fi:fi + bl] = b0
                h[fj:fj + bl] = b1
                new_c5 = _exact_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        patience = 0
                    else:
                        patience += 1
                    ones = list(np.where(h > 0.5)[0])
                    zeros = list(np.where(h <= 0.5)[0])
                else:
                    h[fi:fi + bl] = b1
                    h[fj:fj + bl] = b0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 60)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _triplet_refine(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(555)
    stall = 0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 50)
        cands = np.argsort(-np.abs(total_grad))[:n_int]
        improved = False
        for _ in range(30):
            if time.time() > deadline:
                break
            trio = rng.choice(len(cands), size=3, replace=False)
            i, j, k = cands[trio]
            sg = np.sign([total_grad[i], total_grad[j], total_grad[k]])
            if sg[0] + sg[1] + sg[2] != 0:
                continue
            d = np.zeros(n)
            d[i] = sg[0]
            d[j] = sg[1]
            d[k] = sg[2]
            alpha, c5_try = _line_search(h, d, dx, 0.3)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
        if stall > 20:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _pattern_init(n, period, duty, rng):
    pat = np.zeros(period)
    n_ones = max(1, int(round(period * duty)))
    pat[:n_ones] = 1.0
    h = np.tile(pat, n // period + 1)[:n]
    mask = rng.random(n) < 0.08
    h[mask] = 1.0 - h[mask]
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (6%)
    d1 = t0 + budget_s * 0.06
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7]:
        for ph in [0.0, 1.0, 2.0, 3.0]:
            starts.append(_sine_init(n0, freq, ph))
    for per in [3, 4, 5, 6, 7, 8]:
        for dut in [0.4, 0.45, 0.5, 0.55]:
            starts.append(_pattern_init(n0, per, dut, rng))
    for st in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.004)
        h_s, c5_s = _cont_opt(st, sub_dl, radius_init=0.10, K=12, beta_init=30)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main continuous optimization (24%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _cont_opt(best_h, d2, radius_init=0.14, K=18, beta_init=50)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA (18%)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _block_sa(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Polish with continuous
    if time.time() < t0 + budget_s * 0.54:
        d3b = min(t0 + budget_s * 0.54, deadline)
        h3b, c5_3b = _cont_opt(best_h, d3b, radius_init=0.06, K=15, beta_init=80)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: ILS (28%)
    d4 = t0 + budget_s * 0.82
    ills_iter = 0
    no_improve = 0
    while time.time() < d4 and ills_iter < 400:
        ills_iter += 1
        scale = 0.012 + 0.025 * max(0.0, 1.0 - ills_iter / 300.0)
        if ills_iter % 7 == 0:
            scale *= 1.8
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_c = min(d4, time.time() + budget_s * 0.007)
        h_ls, c5_ls = _cont_opt(h_pert, sub_c, radius_init=0.08, K=14, beta_init=40)
        if time.time() < d4 - budget_s * 0.03 and c5_ls < best_c5 + 0.003:
            sub_b = min(d4, time.time() + budget_s * 0.005)
            h_bn, c5_bn = _block_sa(h_ls, sub_b, rng)
            if c5_bn < c5_ls:
                h_ls, c5_ls = h_bn, c5_bn
            if time.time() < d4 - budget_s * 0.01:
                h_pol, c5_pol = _cont_opt(h_ls, min(d4, time.time() + budget_s * 0.004),
                                          radius_init=0.04, K=12, beta_init=60)
                if c5_pol < c5_ls:
                    h_ls, c5_ls = h_pol, c5_pol
        if c5_ls < best_c5 - 1e-12:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                scale = max(0.04, scale * 1.5)

    # Phase 5: Multi-resolution (10%)
    d5 = t0 + budget_s * 0.92
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -18, 18]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d5, time.time() + budget_s * 0.012)
            h_r, c5_r = _cont_opt(h_r, sub_dl, radius_init=0.10, K=14)
            if time.time() < d5 - budget_s * 0.02:
                sub_dl2 = min(d5, time.time() + budget_s * 0.008)
                h_r2, c5_r2 = _block_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d5 - budget_s * 0.005:
                    h_r, c5_r = _cont_opt(h_r, min(d5, time.time() + budget_s * 0.004),
                                          radius_init=0.04, K=12)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Final refinement (5%)
    d6 = deadline
    h6, c5_6 = _triplet_refine(best_h, min(d6, t0 + budget_s * 0.955), max_passes=40)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    h7, c5_7 = _pair_refine(best_h, min(d6, t0 + budget_s * 0.97), max_passes=60)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < deadline:
        h8, c5_8 = _cont_opt(best_h, deadline, radius_init=0.025, K=10, beta_init=120)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
