    """Find delta in [L, U] minimizing max_r(corr[r] + delta*T[r])"""
    lo, hi = L, U
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * T
        imax = np.argmax(vals)
        vmax = vals[imax]
        # Check if 0 is in subgradient
        # Find all indices achieving the max (within tolerance)
        active = np.where(vals >= vmax - 1e-12)[0]

            lo = hi = mid
            break
