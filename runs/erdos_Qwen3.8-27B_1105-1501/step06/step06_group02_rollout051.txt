def _proj(v, tgt):
    n = len(v)
    if tgt <= 0: return np.zeros(n)
    if tgt >= n: return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(64):
        mid = (lo+hi)/2
        if float(np.sum(np.clip(v-mid, 0, 1))) > tgt:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2, 0, 1)

        c = int(m.sum())
