def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    stall = 0
    while time.time() < deadline and stall < 30:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(300):
            if time.time() > deadline:
                break
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    interior = np.where((h > 0.01) & (h < 0.99))[0]
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        if (i % period) < fill:
            h[i] = 1.0
    return _project(h, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 4))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)

    # Phase 1: LSE annealing (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lse_optimize(best_h, d1, beta_start=3.0, beta_end=300.0, n_steps=200)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP refinement (25% time)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.10, top_k=18)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Another LSE pass with higher beta (10% time)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _lse_optimize(best_h, d3, beta_start=50.0, beta_end=1000.0, n_steps=150)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (20% time)
    d4 = t0 + budget_s * 0.80
    best_n = len(best_h)
    n_offsets = [-4, -2, 2, 4, -6, 6, -8, 8, -10, 10, 12, -12, 14, -14, 16, -16, 20, -20, 24, -24]
    tried = set()
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.08, top_k=14)
        if c5_r < best_c5 - 1e-12:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _pair_refine(h_r, sub_dl2)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 5: Multi-start (8% time)
    d5 = t0 + budget_s * 0.88
    n = len(best_h)
    starts = []
    for freq in [2, 3, 5, 7]:
        for ph in [0.0, 1.0, 2.0]:
            starts.append(_sine_init(n, freq, ph))
    for p in [5, 6, 7, 8, 11]:
        for duty in [0.35, 0.5, 0.65]:
            starts.append(_periodic_init(n, p, duty))
    for _ in range(3):
        starts.append(_block_init(n, rng))
    for s in starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.06, top_k=10)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Final polish - LSE + LP + pair (7% time)
    d6 = deadline - 5
    h6, c5_6 = _lse_optimize(best_h, min(d6, t0 + budget_s * 0.90), beta_start=100.0, beta_end=2000.0, n_steps=100)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6 - 10:
        h6b, c5_6b = _lp_loop(best_h, min(d6, time.time() + budget_s * 0.03), radius_init=0.04, top_k=12)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
    if time.time() < d6 - 5:
        h6c, c5_6c = _pair_refine(best_h, d6)
        if c5_6c < best_c5:
            best_c5 = c5_6c
            best_h = h6c.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0_fallback = np.array(initial_h_values, dtype=float)
            h0_fallback = _project(h0_fallback, len(h0_fallback) / 2.0)
            return h0_fallback, _exact_c5(h0_fallback), len(h0_fallback)
        except NameError:
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _exact_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
