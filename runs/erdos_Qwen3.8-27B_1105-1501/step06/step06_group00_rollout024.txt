def _project(v, target):
    n = len(v)
    # Project onto {h : 0<=h[i]<=1, sum(h)=target}
    # This is a standard projection onto the intersection of a box and a hyperplane
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
import time

def _scores(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_scores(h)) * (2.0 / n))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _topk_subgrad(h, scores, k=5, temperature=0.01):
    """Compute subgradient weighted by softmax over top-k scores."""
    n = len(h)
    # Softmax over all scores (with temperature control)
    max_s = float(np.max(scores))
    # Use only top k for focus
    top_idx = np.argsort(scores)[-k:]
    top_vals = scores[top_idx]
    w = np.exp((top_vals - top_vals[-1]) / temperature)
    w /= np.sum(w)
    
    g = np.zeros(n)
    for t, idx in enumerate(top_idx):
        d = int(idx) - (n - 1)
        wt = float(w[t])
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            # g[:m] += wt * (1 - h[d:])
            # g[d:d+m] += wt * (-h[:m])
            g[:m] += wt * (1.0 - h[d:d+m])
            g[d:d+m] += wt * (-h[:m])
        else:
            d2 = -d
            g[:m] += wt * (1.0 - h[d2:d2+m])
            g[d2:d2+m] += wt * (-h[:m])
    return g

def _l2_subgrad(h, scores):
    """Subgradient of sum of squares of scores."""
    n = len(h)
    g = np.zeros(n)
    for idx in range(len(scores)):
        s = scores[idx]
        if abs(s) < 1e-10:
            continue
        d = int(idx) - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            g[:m] += 2.0 * s * (1.0 - h[d:d+m])
            g[d:d+m] += 2.0 * s * (-h[:m])
        else:
            d2 = -d
            g[:m] += 2.0 * s * (1.0 - h[d2:d2+m])
            g[d2:d2+m] += 2.0 * s * (-h[:m])
    return g

def _fw_min(g, target):
    """Linear minimizer for FW: argmin g·s s.t. s∈[0,1], sum(s)=target."""
    n = len(g)
    order = np.argsort(g)
    s = np.zeros(n)
    k = int(target)
    s[order[:k]] = 1.0
    frac = target - k
    if 0 < frac < 1 and k < n:
        s[order[k]] = frac
    return s

def _line_search(h, s, dx, n_grid=20):
    """Line search along h → s using quadratic model."""
    n = len(h)
    a = s - h
    S0 = _scores(h)
    # scores(t) = S0 + t*T - t^2*C where T and C are precomputed
    T = np.correlate(a, 1.0 - h, mode='full') - np.correlate(h, a, mode='full')
    C = np.correlate(a, a, mode='full')
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(40):
        if hi - lo < 1e-11:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    t = (lo + hi) / 2.0
    f = float(np.max(S0 + t*T - t*t*C)) * dx
    # Check endpoints
    f0 = float(np.max(S0)) * dx
    f1v = float(np.max(S0 + T - C)) * dx
    if f1v < f:
        t, f = 1.0, f1v
    if f0 < f:
        t, f = 0.0, f0
    return t, f

def _fw_loop(h, deadline, top_k=5, momentum=0.7, stall_limit=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best = cur
    avg_g = np.zeros(n)
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        g = _topk_subgrad(h, scores, k=top_k)
        avg_g = momentum * avg_g + (1.0 - momentum) * g
        s = _fw_min(avg_g, target)
        t, new_c5 = _line_search(h, s, dx)
        if t > 1e-10 and new_c5 < cur - 1e-11:
            h = (1.0 - t) * h + t * s
            h = _project(h, target)
            cur = _c5(h)
            if cur < best:
                best = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 15 == 0:
                avg_g = np.zeros(n)
    return best_h, best

def _l2_loop(h, deadline, lr_init=0.02):
    """Gradient descent on sum of squares of scores."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best = cur
    lr = lr_init
    stall = 0
    while time.time() < deadline:
        if stall > 100:
            break
        scores = _scores(h)
        g = _l2_subgrad(h, scores)
        # Gradient step
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-12:
            break
        g = g / g_norm
        # Try step
        h_new = _project(h - lr * g, target)
        new_c5 = _c5(h_new)
        if new_c5 < cur - 1e-12:
            h = h_new
            cur = new_c5
            if cur < best:
                best = cur
                best_h = h.copy()
            lr = min(lr * 1.05, 0.1)
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if lr < 1e-6:
                lr = lr_init
                stall = 0
    return best_h, best

def _binary_search(h_bin, deadline, max_passes=500):
    """Greedy swap local search for binary h."""
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _c5(cur)
    best_h = cur.copy()
    best = cur_c5
    scores = _scores(cur)
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        best_delta = 0  # We want to maximize the improvement (reduce C5)
        best_i, best_j = -1, -1
        best_new_scores = None
        
        for i in zeros:
            for j in ones:
                # Compute delta in scores
                # For each k: Δscore_k = (1-h[i+k]-h[i-k]) - (1-h[j+k]-h[j-k])
                # where out-of-range terms are 0
                delta = np.zeros(len(scores))
                d_i = _position_contribution(cur, i)
                d_j = _position_contribution(cur, j)
                delta = d_i - d_j
                new_scores = scores + delta
                new_max = float(np.max(new_scores)) * dx
                if new_max < cur_c5 - 1e-14:
                    # This swap improves
                    improvement = cur_c5 - new_max
                    if improvement > best_delta:
                        best_delta = improvement
                        best_i, best_j = i, j
                        best_new_scores = new_scores
        
        if best_i >= 0:
            cur[best_i] = 1.0
            cur[best_j] = 0.0
            scores = best_new_scores
            cur_c5 = float(np.max(scores)) * dx
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best:
                best = cur_c5
                best_h = cur.copy()
            improved = True
        if not improved:
            break
    return best_h, best

def _position_contribution(h, pos):
    """Compute the contribution vector of position pos to all scores.
    This is the vector v where v[k] = ∂score_k/∂h[pos] (the marginal effect)."""
    n = len(h)
    v = np.zeros(2 * n - 1)
    for k_idx in range(2 * n - 1):
        d = k_idx - (n - 1)
        # h[pos] appears as h[j] where j=pos, contributing (1-h[pos+d])
        # h[pos] appears as h[j+d] where j=pos-d, contributing -h[pos-d]
        contrib = 0.0
        if 0 <= pos + d < n:
            contrib += (1.0 - h[pos + d])
        if 0 <= pos - d < n:
            contrib += -h[pos - d]
        v[k_idx] = contrib
    return v
