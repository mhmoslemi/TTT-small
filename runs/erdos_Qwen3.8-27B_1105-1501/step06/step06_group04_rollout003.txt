import numpy as np
from scipy.optimize import minimize, linprog
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.asarray(v, dtype=float).copy()
    v[~np.isfinite(v)] = 0.5
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_row(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:]
    else:
        a = -d
        m = n - a
        if m > 0:
            g[:m] += (1.0 - h[a:])
            g[a:] -= h[:m]
    return g

def _all_grads(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        G[r] = _grad_row(h, d)
    return G

def _line_search_exact(h, d, dx, t_max):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi_min = t_max
    if np.any(pos):
        t_hi_min = min(t_hi_min, float(np.min((1.0 - h[pos]) / d[pos])))
    t_lo_max = -t_max
    if np.any(neg):
        t_lo_max = max(t_lo_max, float(np.max(h[neg] / d[neg])))
    if t_hi_min - t_lo_max < 1e-14:
        return 0.0, _compute_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo_max, t_hi_min
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

class _LSE:
    def __init__(self, beta, dx):
        self.beta = beta
        self.dx = dx
    def _eval(self, h):
        scores = np.correlate(h, 1.0 - h, mode='full') * self.dx
        max_s = float(np.max(scores))
        z = self.beta * (scores - max_s)
        z = np.clip(z, -500.0, 500.0)
        ez = np.exp(z)
        se = float(np.sum(ez))
        obj = (max_s * self.beta + np.log(se)) / self.beta
        p = ez / se
        n = len(h)
        grad = np.zeros(n)
        for r in range(len(scores)):
            if p[r] > 1e-11:
                d_off = r - (n - 1)
                grad += p[r] * self.dx * _grad_row(h, d_off)
        return obj, grad
    def fun(self, x):
        obj, _ = self._eval(x)
        return obj
    def jac(self, x):
        _, grad = self._eval(x)
        return grad
    def eq_fun(self, x):
        return float(np.sum(x)) - self.dx * len(x) / 2.0 * (len(x) / 2.0) / self.dx
    def eq_jac(self, x):
        return np.ones(len(x))

def _lse_solver(h_init, beta, target, maxiter=300, ftol=1e-11):
    n = len(h_init)
    dx = 2.0 / n
    lse = _LSE(beta, dx)
    # Fix eq_fun to use target
    def eq_fun_inner(x):
        return float(np.sum(x)) - target
    def eq_jac_inner(x):
        return np.ones(len(x))
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun_inner, 'jac': eq_jac_inner}]
    try:
        res = minimize(lse.fun, h_init.copy(), jac=lse.jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = _project(res.x, target)
            return h_new, _compute_c5(h_new)
    except Exception:
        pass
    return h_init, _compute_c5(h_init)

def _lse_projected_grad(h_init, beta, target, max_iter=500, deadline=time.time()):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    lse = _LSE(beta, dx)
    alpha = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad = lse._eval(h)
        grad -= grad.mean()
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-10:
            break
        d = -grad / norm_g
        # Backtracking line search
        for _ in range(40):
            h_try = _project(h + alpha * d, target)
            c5_try = _compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                break
            alpha *= 0.7
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                break
    return best_h, best_c5

def _active_grad_direction(h, scores, radius, frac=0.15):
    n = len(h)
    max_s = float(np.max(scores))
    thr = max_s - frac * max_s
    active = np.where(scores >= thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    G = _all_grads(h)
    avg = G[active].mean(axis=0)
    avg -= avg.mean()
    norm = np.linalg.norm(avg)
    if norm < 1e-14:
        return None
    d = -avg / norm * radius
    return d

def _random_direction(h, rng, radius, sparsity=0.3):
    n = len(h)
    d = np.zeros(n)
    k = max(2, int(n * sparsity))
    idx = rng.choice(n, size=k, replace=False)
    d[idx] = rng.normal(0, 1, k)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    d = d / norm * radius
    # Clip to feasible
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        scale = np.min((1.0 - h[pos]) / d[pos])
        d[pos] = np.minimum(d[pos], scale * d[pos] + 1e-15)
    if np.any(neg):
        scale = np.min(h[neg] / (-d[neg]))
        d[neg] = np.maximum(d[neg], -scale * (-d[neg]) - 1e-15)
    return d

def _line_polish(h, deadline, rng, max_rounds=200, radius=0.15):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    stall = 0
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 60:
            h = _project(h + rng.normal(0, 0.02, n), target)
            best_c5 = _compute_c5(h)
            if best_c5 < _compute_c5(best_h):
                best_c5 = _compute_c5(h)
                best_h = h.copy()
            stall = 0
        scores = _corr_scaled(h)
        # Try active gradient direction
        d = _active_grad_direction(h, scores, radius, frac=0.12)
        if d is not None:
            t, val = _line_search_exact(h, d, dx, radius * 2)
            if val < best_c5 - 1e-14:
                h = _project(h + t * d, target)
                cur = _compute_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                    continue
        # Try narrower active set
        d2 = _active_grad_direction(h, scores, radius, frac=0.05)
        if d2 is not None:
            t2, val2 = _line_search_exact(h, d2, dx, radius * 2)
            if val2 < best_c5 - 1e-14:
                h = _project(h + t2 * d2, target)
                cur = _compute_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                    continue
        # Random direction
        d3 = _random_direction(h, rng, radius * 0.8, sparsity=0.25)
        if d3 is not None and np.max(np.abs(d3)) > 1e-12:
            t3, val3 = _line_search_exact(h, d3, dx, radius)
            if val3 < best_c5 - 1e-14:
                h = _project(h + t3 * d3, target)
                cur = _compute_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                    continue
        # Pair move
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        if len(interior) >= 2:
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i != j:
                dp = np.zeros(n)
                dp[i] = -1.0
                dp[j] = 1.0
                tp, valp = _line_search_exact(h, dp, dx, 0.5)
                if valp < best_c5 - 1e-14:
                    h = _project(h + tp * dp, target)
                    cur = _compute_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        stall = 0
                        continue
        stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _binary_quantize(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, rng, max_iters=15000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.00005
    cooling = 0.9997
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    patience = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 4000:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            ones_list = np.where(h > 0.5)[0]
            zeros_list = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _lp_direction(h, scores, radius, top_k=12):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = np.zeros((K, n))
    for idx, r in enumerate(active):
        d = r - (n - 1)
        G[idx] = _grad_row(h, d)
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.10, top_k=12):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > 100:
            h = _project(h + rng.normal(0, 0.02, n), target)
            cur_c5 = _compute_c5(h)
            radius = 0.10
            stall = 0
        scores = _corr_scaled(h)
        d = _lp_direction(h, scores, radius, top_k=top_k)
        if d is not None and np.max(np.abs(d)) > 1e-13:
            alpha, new_c5 = _line_search_exact(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-14:
                h = _project(h + alpha * d, target)
                cur_c5 = _compute_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                radius = min(radius * 1.05, 0.30)
                continue
        # Active gradient fallback
        d2 = _active_grad_direction(h, scores, radius, frac=0.10)
        if d2 is not None:
            a2, v2 = _line_search_exact(h, d2, dx, radius * 1.5)
            if v2 < cur_c5 - 1e-14:
                h = _project(h + a2 * d2, target)
                cur_c5 = _compute_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                continue
        stall += 1
        radius *= 0.9
        if radius < 0.005:
            radius = 0.08
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    rng = np.random.default_rng(42)
    stall = 0
    while time.time() < deadline and stall < 40:
        interior = np.where((h > 0.03) & (h < 0.97))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(300):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            for sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                alpha, c5_try = _line_search_exact(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = _project(h + alpha * d, target)
                    cur = _compute_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        interior = np.where((h > 0.03) & (h < 0.97))[0]
                        break
            if improved:
                break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s
    reserve = max(5.0, 0.02 * budget_s)
    hard_deadline = t0 + budget_s - reserve

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: LSE SLSQP with continuation (20% time)
    d1 = t0 + budget_s * 0.20
    target = n0 / 2.0
    for beta in [15, 40, 100, 250, 600, 1200]:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.025)
        h_lse, c5_lse = _lse_solver(best_h, beta, target, maxiter=250)
        if c5_lse < best_c5:
            best_c5 = c5_lse
            best_h = h_lse.copy()
        # Also try projected gradient with same beta
        if time.time() < d1 - 5:
            h_pg, c5_pg = _lse_projected_grad(best_h, beta, target, max_iter=200,
                                              deadline=t0 + budget_s * 0.02)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    # Phase 2: LP loop (15% time)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.12, top_k=14)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Line polish (15% time)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _line_polish(best_h, d3, rng, max_rounds=500, radius=0.12)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-start (15% time)
    d4 = t0 + budget_s * 0.65
    n = len(best_h)
    starts = []
    for freq in [1, 2, 3, 5, 7]:
        for ph in [0.0, 1.0, 2.0, 3.5]:
            starts.append(_sine_init(n, freq, ph))
    for p in [4, 5, 6, 7, 9, 11, 13]:
        for duty in [0.3, 0.4, 0.5, 0.6, 0.7]:
            starts.append(_periodic_init(n, p, duty))
    # Binary quantize best
    starts.append(_binary_quantize(best_h))
    # Random
    for _ in range(3):
        starts.append(_project(rng.random(n), n / 2.0))
    for s in starts:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.008)
        h_s, c5_s = _line_polish(s, sub_dl, rng, max_rounds=80, radius=0.10)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Binary SA (12% time)
    d5 = t0 + budget_s * 0.77
    h_bin, c5_bin = _binary_sa(best_h, min(d5, t0 + budget_s * 0.72), rng, max_iters=20000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Polish the binary result
    if time.time() < d5:
        h_bp, c5_bp = _line_polish(best_h, d5, rng, max_rounds=100, radius=0.08)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()

    # Phase 6: Multi-resolution (10% time)
    d6 = t0 + budget_s * 0.87
    best_n = len(best_h)
    tried = set()
    for dn in [-4, 4, -8, 8, -12, 12, -16, 16, -20, 20, 24, -24]:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _line_polish(h_r, sub_dl, rng, max_rounds=100, radius=0.10)
        if c5_r < best_c5 - 1e-12 and time.time() < d6 - 3:
            h_r2, c5_r2 = _pair_refine(h_r, min(d6, time.time() + budget_s * 0.006))
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final LSE + pair polish (7% time)
    d7 = hard_deadline
    if time.time() < d7 - 10:
        for beta in [200, 500]:
            if time.time() > d7 - 5:
                break
            sub_dl = min(d7 - 5, time.time() + budget_s * 0.01)
            h_f, c5_f = _lse_solver(best_h, beta, len(best_h) / 2.0, maxiter=150)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
    if time.time() < d7 - 3:
        h_p, c5_p = _pair_refine(best_h, d7, max_passes=60)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        h0_fb = _project(h0, n0 / 2.0)
        return h0_fb, _compute_c5(h0_fb), n0
    return best_h, final_c5, len(best_h)
