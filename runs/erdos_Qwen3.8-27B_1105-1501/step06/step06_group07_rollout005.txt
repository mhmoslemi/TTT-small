def _project(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection on threshold
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

        best_idx = int(np.argmin(grid_vals))
        if best_idx > 0 and best_idx < n_grid - 1:
            a = grid[best_idx - 1]
            b = grid[best_idx + 1]
        else:
            a, b = s_lo, s_hi
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        
        def _obj_1d(s):
            vals = S + s * T
            for pos, qv in zip(q_positions, q_values):
                if 0 <= pos < len(vals):
                    vals[pos] -= s * s * qv
            return float(np.max(vals)) * dx
        
        f1 = _obj_1d(c1)
        f2 = _obj_1d(c2)
        for _ in range(50):
            if b - a < 1e-10:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = _obj_1d(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = _obj_1d(c2)
        opt_s = (a + b) / 2.0
        opt_val = _obj_1d(opt_s)
        if opt_val < best_val:
            best_val = opt_val
            best_s = opt_s
    
    return best_s, best_val

def _active_gradient(h, S, sens, K=30, alpha=50.0):
    """Compute weighted gradient from top K active correlations."""
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    corr_vals = S * dx  # scaled
    top_idx = np.argsort(-corr_vals)[:K]
    top_vals = corr_vals[top_idx]
    spread = max(top_vals[0] - top_vals[-1], 1e-15)
    w = np.exp(-alpha * (top_vals[0] - top_vals) / spread)
    w /= np.sum(w)
    # Gradient = sum w_r * sens for active positions
    # Wait, I need the gradient of the correlation with respect to h.
    # corr(h, 1-h)[r] = sum_i h[i] * (1-h)[i + n - 1 - r] (with bounds)
    # d/dh[j] of this = (1-h[j+n-1-r]) - h[j] * δ(j, j+n-1-r)
    # This is exactly what sens[j] stores (but with opposite sign convention?)
    # Let me verify: sens[j] = corr(e_j, 1-h, 'full') - corr(h, e_j, 'full')
    # corr(e_j, 1-h, 'full')[r] = (1-h)[j + n - 1 - r]
    # corr(h, e_j, 'full')[r] = h[n + n - 1 - r - j]... 
    
    # Actually, let me just use the definition directly.
    # The gradient of corr(h, 1-h, 'full')[r] with respect to h[j] is:
    # Let corr(h, 1-h, 'full')[r] = sum_i h[i] * (1-h)[i + offset_r]
    # where offset_r = n - 1 - r (with appropriate bounds checking)
    # d/dh[j] = (1-h[j + offset_r]) - h[j] * δ(j, j + offset_r)
    #          = (1-h[j + offset_r]) - (1 if offset_r == 0 else 0) * h[j]
    # Hmm, this is getting complicated. Let me just use sens directly.
    
    # sens[j] should be the gradient of the correlation vector with respect to h[j].
    # Let me verify: the correlation vector C = corr(h, 1-h, 'full')
    # C[r] = sum_i h[i]*(1-h[i+n-1-r]) (approximately, with bounds)
    # dC[r]/dh[j] = (1-h[j+n-1-r]) + (-1)*h[j]*d/dh[j](h[j+n-1-r])
    #             = (1-h[j+n-1-r]) - h[j]*δ(j, j+n-1-r)
    #             = (1-h[j+n-1-r]) - (h[j] if n-1-r==0 else 0)
    #             = (1-h[j+n-1-r]) - (h[j] if r==n-1 else 0)
    
    # And sens[j] = corr(e_j, 1-h, 'full') - corr(h, e_j, 'full')
    # corr(e_j, 1-h, 'full')[r] = (1-h)[j+n-1-r] (if valid index)
    # corr(h, e_j, 'full')[r] = h[n+n-1-r-j]... 
    
    # I think there might be a sign or index issue. Let me just verify numerically
    # in the code by checking that the directional derivative matches.
    
    # For now, assume sens[j] gives the correct gradient direction.
    # The gradient of the objective w.r.t. h is:
    # grad = sum_r w_r * dC[r]/dh  (for the max, we use active set weights)
    # = sum_r w_r * (something involving sens)
    
    # Actually, from the definition:
    # C[r] = corr(h, 1-h, 'full')[r]
    # dC/ds for h -> h + s*d:
    # = corr(d, 1-h, 'full')[r] - corr(h, d, 'full')[r]
    # = d @ (something) ... 
    # = sum_p d[p] * (corr(e_p, 1-h, 'full')[r] - corr(h, e_p, 'full')[r])
    # = sum_p d[p] * sens[p][r]
    # = (sens.T @ d)[r]
    
    # So the gradient of C[r] w.r.t. h is the r-th row of sens.T, i.e., sens[:, r].
    # The gradient of the objective (smoothed max) is:
    # grad[h] = sum_r w_r * sens[:, r] = sens @ w
    
    grad = sens.dot(w)
    return grad

def _gradient_descent_step(h, S, sens, target, dx, K=30, alpha=50.0, max_line_iters=80):
    """One gradient descent step with exact line search."""
    n = len(h)
    grad = _active_gradient(h, S, sens, K, alpha)
    grad -= np.mean(grad)  # project to sum-zero
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, _exact_c5(h)
    direction = -grad / gn
    
    # Find feasible range for s
    pos = direction > 1e-14
    neg = direction < -1e-14
    s_hi = 1.0
    s_lo = -1.0
    if np.any(pos):
        s_hi = min(s_hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        s_lo = max(s_lo, float(np.max(-h[neg] / direction[neg])))
        # Wait: h[neg] + s*direction[neg] >= 0 → s <= h[neg]/(-direction[neg]) = -h[neg]/direction[neg]
        # Since direction[neg] < 0, -h[neg]/direction[neg] > 0
        s_hi = min(s_hi, float(np.max(-h[neg] / direction[neg])))
    # Also h[pos] + s*direction[pos] <= 1 → s <= (1-h[pos])/direction[pos] (direction[pos] > 0)
    # h[pos] + s*direction[pos] >= 0 → s >= -h[pos]/direction[pos] (negative, so s_lo stays at -1 or whatever)
    
    # Recompute T and Q for this direction
    T = sens.T.dot(direction)  # T[r] = sum_p direction[p] * sens[p][r]
    Q = np.correlate(direction, direction, mode='full')
    
    # Line search
    s_grid = np.linspace(max(s_lo, 0.0), min(s_hi, 1.0), 40)
    if len(s_grid) < 2 or s_grid[-1] <= s_grid[0]:
        s_grid = np.array([0.0, min(s_hi, 0.5)])
    
    best_s = 0.0
    best_val = float(np.max(S)) * dx
    for s in s_grid:
        vals = S + s * T - s * s * Q
        v = float(np.max(vals)) * dx
        if v < best_val:
            best_val = v
            best_s = s
    
    # Golden section refine
    if best_s > s_grid[0] + 1e-14 and best_s < s_grid[-1] - 1e-14:
        idx = int(np.argmin([float(np.max(S + s * T - s*s * Q)) * dx for s in [0.0]]))
        # Find bracket
        a = max(s_grid[0], best_s - (s_grid[1] - s_grid[0]))
        b = min(s_grid[-1], best_s + (s_grid[1] - s_grid[0]))
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        def _obj(s):
            return float(np.max(S + s * T - s * s * Q)) * dx
        f1 = _obj(c1)
        f2 = _obj(c2)
        for _ in range(40):
            if b - a < 1e-12:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = _obj(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = _obj(c2)
        opt_s = (a + b) / 2.0
        opt_val = _obj(opt_s)
        if opt_val < best_val:
            best_val = opt_val
            best_s = opt_s
    
    if best_s < 1e-14:
        return h, best_val
    
    h_new = h + best_s * direction
    h_new = _project(h_new, target)
    c5_new = _exact_c5(h_new)
    if c5_new < best_val + 1e-10:  # Allow small numerical error
        return h_new, c5_new
    return h, _exact_c5(h)

def _opt_pair(h, S, sens, i, j, target, dx):
    """Optimize the pair (i, j): find best s for h_i += s, h_j -= s."""
    n = len(h)
    d = abs(i - j)
    if d == 0:
        return h, _exact_c5(h)
    
    # Feasible range for s
    s_lo = max(-h[i], h[j] - 1.0)
    s_hi = min(1.0 - h[i], h[j])
    if s_hi <= s_lo:
        return h, _exact_c5(h)
    
    # T = sens[i] - sens[j]
    T = sens[i] - sens[j]
    
    # Q non-zero positions
    mid = n - 1
    q_positions = [mid, mid + d, mid - d]
    q_values = [2.0, -1.0, -1.0]
    # Filter valid positions
    valid = [(p, q) for p, q in zip(q_positions, q_values) if 0 <= p < len(S)]
    
    # Line search
    s_grid = np.linspace(s_lo, s_hi, 30)
    best_s = 0.0
    best_val = float(np.max(S)) * dx  # current value (s=0)
    
    for s in s_grid:
        if s == 0:
            continue
        vals = S + s * T
        for p, qv in valid:
            vals[p] -= s * s * qv
        v = float(np.max(vals)) * dx
        if v < best_val:
            best_val = v
            best_s = s
    
    # Golden section
    if best_s > s_lo + 1e-12 and best_s < s_hi - 1e-12:
        # Find bracket from grid
        grid_vals = np.empty(len(s_grid))
        for idx_s, s in enumerate(s_grid):
            vals = S + s * T
            for p, qv in valid:
                vals[p] -= s * s * qv
            grid_vals[idx_s] = float(np.max(vals)) * dx
        best_idx = int(np.argmin(grid_vals))
        if best_idx > 0 and best_idx < len(s_grid) - 1:
            a = s_grid[best_idx - 1]
            b = s_grid[best_idx + 1]
        else:
            a, b = s_lo, s_hi
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        def _obj(s):
            vals = S + s * T
            for p, qv in valid:
                vals[p] -= s * s * qv
            return float(np.max(vals)) * dx
        f1 = _obj(c1)
        f2 = _obj(c2)
        for _ in range(50):
            if b - a < 1e-12:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = _obj(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = _obj(c2)
        opt_s = (a + b) / 2.0
        opt_val = _obj(opt_s)
        if opt_val < best_val:
            best_val = opt_val
            best_s = opt_s
    
    if best_s < 1e-14:
        return h, best_val
    
    h_new = h.copy()
    h_new[i] += best_s
    h_new[j] -= best_s
    h_new = _project(h_new, target)
    c5_new = _exact_c5(h_new)
    if c5_new < best_val + 1e-10:
        return h_new, c5_new
    return h, _exact_c5(h)

def _pair_pass(h, S, sens, target, dx, deadline, max_pairs=2000):
    """Do a pass of pair-exchange optimization."""
    n = len(h)
    c5_cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    # Generate pairs: focus on pairs with large gradient difference
    all_pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            all_pairs.append((i, j))
    np.random.shuffle(all_pairs)  # Randomize order
    all_pairs = all_pairs[:max_pairs]
    
    improvements = 0
    for i, j in all_pairs:
        if time.time() > deadline:
            break
        h_new, c5_new = _opt_pair(h, S, sens, i, j, target, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new
            h = h_new
            # Recompute S and sens (expensive but necessary for correctness)
            # Actually, for a single pair change, we can update S incrementally
            # But it's simpler to just recompute periodically
            improvements += 1
            if improvements % 10 == 0:
                S, sens = _precompute(h)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _quick_optimize(h, deadline, n_iterations=50):
    """Quick local optimization using gradient + pair steps."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        
        # Gradient step
        S, sens = _precompute(h)
        h_new, c5_new = _gradient_descent_step(h, S, sens, target, dx)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        
        # A few pair steps
        S, sens = _precompute(h)
        n_pairs = min(200, n * (n - 1) // 2)
        rng_local = np.random.default_rng(it * 7 + 3)
        for _ in range(n_pairs // 50):  # A few random pairs
            if time.time() > deadline:
                break
            i = rng_local.integers(n)
            j = rng_local.integers(n)
            if i == j:
                continue
            h_new2, c5_new2 = _opt_pair(h, S, sens, i, j, target, dx)
            if c5_new2 < best_c5 - 1e-13:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                h = h_new2
                S, sens = _precompute(h)
    
    return best_h, best_c5
