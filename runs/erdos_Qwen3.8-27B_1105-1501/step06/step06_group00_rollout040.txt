    n = len(h)
    G = np.zeros((2*n-1, n))
    # For shift d >= 0: G[n-1+d, :n-d] += (1-h[d:]), G[n-1+d, d:] += -h[:n-d]
    # For shift d < 0: G[n-1+d, -d:] += (1-h[:n+d]), G[n-1+d, :n+d] += -h[-d:]
    # Can we vectorize this? It's a banded structure.
    # Actually, for n=180, the Python loop is fine (359 iterations).
    for d in range(-(n-1), n):
        k = n-1+d
        if d >= 0:
            m = n-d
            G[k, :m] += (1.0 - h[d:n])
            G[k, d:n] += -h[:m]
        else:
            g = -d
            m = n-g
            G[k, g:n] += (1.0 - h[:m])
            G[k, :m] += -h[g:n]
    return G

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        k = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[k, :m] += (1.0 - h[d:n])
                G[k, d:n] += -h[:m]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[k, g:n] += (1.0 - h[:m])
                G[k, :m] += -h[g:n]
    return G

def _line_search(h, d, n):
    dx = 2.0 / n
    target = n / 2.0
    f0 = _exact_c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = 1.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-14:
        return 0.0, f0
    grid = 30
    gt = np.linspace(t_lo, t_hi, grid)
    gf = np.empty(grid)
    for i in range(grid):
        t = gt[i]
        gf[i] = float(np.max(S0 + t * T - t * t * C)) * dx
    bi = int(np.argmin(gf))
    lo_b = gt[max(0, bi - 2)]
    hi_b = gt[min(grid - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if hi_b - lo_b < 1e-10:
        opt_t, opt_v = gt[bi], gf[bi]
    else:
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(40):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0

    cands = [(opt_t, opt_v), (0.0, f0), (t_hi, float(np.max(S0 + t_hi * T - t_hi * t_hi * C)) * dx), (gt[bi], gf[bi])]
    best = min(cands, key=lambda x: x[1])

def _qp_loop(h, deadline, radius=0.12, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    r = radius
    stall = 0
    while time.time() < deadline and stall < stall_limit:
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        G = _grad_matrix(h)
        reduction = max(0.0003, 0.015 * cur)
        c_tgt = cur - reduction
        d_var = cp.Variable(n)
        cons = [
            G @ d_var <= (c_tgt - scores),
            cp.sum(d_var) == 0,
            d_var <= r,
            -d_var <= r,
            h + d_var >= 0,
            h + d_var <= 1,
        ]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(d_var)), cons)
        d = None
        try:
            prob.solve(solver=cp.CLARABEL)
            if prob.status in ('optimal', 'optimal_inaccurate') and d_var.value is not None:
                d = np.array(d_var.value).flatten()
        except Exception:
            pass
        if d is None:
            try:
                prob.solve(solver=cp.ECOS, max_iters=3000)
                if d_var.value is not None:
                    d = np.array(d_var.value).flatten()
            except Exception:
                pass
        if d is None or np.max(np.abs(d)) < 1e-12:
            stall += 1
            r *= 0.8
            if r < 0.003:
                break
            continue
        t, new_c5 = _line_search(h, d, n)
        if t > 0 and new_c5 < cur - 1e-13:
            h_new = h + t * d
            h_new = _project(h_new, target)
            cur2 = _exact_c5(h_new)
            h = h_new
            cur = cur2
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            r = min(r * 1.04, 0.22)
        else:
            stall += 1
            r *= 0.82
    return best_h, best_c5

def _subgrad_loop(h, deadline, eps_init=0.002, lr_init=0.004, stall_limit=60):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    v = np.zeros(n)
    beta = 0.85
    lr = lr_init
    eps = eps_init
    stall = 0
    while time.time() < deadline and stall < stall_limit:
        scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        s_max = scores.max()
        w = np.exp((scores - s_max) / eps)
        w /= w.sum()
        G = _grad_matrix(h)
        grad = G.T @ w
        v = beta * v + (1 - beta) * grad
        h_new = h - lr * v
        h_new = _project(h_new, target)
        c_new = _exact_c5(h_new)
        if c_new < cur - 1e-14:
            h = h_new
            cur = c_new
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.015, 0.015)
        else:
            stall += 1
            lr *= 0.92
            eps = min(eps * 1.15, 0.04)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    n_freqs = 5
    freqs = rng.uniform(0.3, 10.0, n_freqs)
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    amps = rng.uniform(0.4, 1.0, n_freqs)
    p = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        p += a * np.sin(2 * np.pi * f * x / n + ph)
    p *= amp
    p += amp * 0.25 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def _pair_swap(h, deadline, n_tries=4000, seed=77):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _exact_c5(h)
    best_h, best_c5 = h.copy(), cur
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        if stall > 120:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        amt = min(h[j], 1.0 - h[i]) * rng.uniform(0.2, 1.0)
        if amt < 1e-5:
            stall += 1
            continue
        h_try = h.copy()
        h_try[i] += amt
        h_try[j] -= amt
        h_try = _project(h_try, target)
        c_try = _exact_c5(h_try)
        if c_try < cur - 1e-14:
            h = h_try
            cur = c_try
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: LP warmup (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(h_init, d1, radius_init=0.10, stall_limit=40)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: QP refinement (15%)
    d2 = t0 + budget_s * 0.23
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c5_2 = _qp_loop(h2, d2, radius=0.10, stall_limit=60)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Subgradient polish (7%)
    d3 = t0 + budget_s * 0.30
    h3 = _project(best_h.copy(), len(best_h) / 2.0)
    h3, c5_3 = _subgrad_loop(h3, d3, eps_init=0.0015, lr_init=0.003)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: QP again tighter (12%)
    d4 = t0 + budget_s * 0.42
    h4 = _project(best_h.copy(), len(best_h) / 2.0)
    h4, c5_4 = _qp_loop(h4, d4, radius=0.06, stall_limit=80)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Multi-n exploration (28%) - fine sweep around n0
    d5 = t0 + budget_s * 0.70
    n_curr = len(best_h)
    tried = set([n_curr])
    # Fine sweep: n0-8 to n0+8
    for dn in range(-8, 9):
        if time.time() > d5:
            break
        n_try = n0 + dn
        if n_try < 120 or n_try > 240 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _qp_loop(h_r, sub_dl, radius=0.09, stall_limit=25)
        if c5_r < best_c5 + 0.002 and time.time() < d5 - 2:
            sub_dl2 = min(d5, time.time() + budget_s * 0.006)
            h_r2 = _project(h_r.copy(), n_try / 2.0)
            h_r2, c5_r2 = _lp_loop(h_r2, sub_dl2, radius_init=0.05, stall_limit=20)
            if c5_r2 < c5_r:
                c5_r, h_r = c5_r2, h_r2
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()
            n_curr = n_try

    # Coarse sweep for diversity
    for dn in [-24, -18, -12, 12, 18, 24, 30, -30]:
        if time.time() > d5:
            break
        n_try = n0 + dn
        if n_try < 100 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.010)
        h_r, c5_r = _qp_loop(h_r, sub_dl, radius=0.10, stall_limit=20)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()
            n_curr = n_try

    # Phase 6: Basin hopping (12%)
    d6 = t0 + budget_s * 0.82
    scale = 0.012
    for _ in range(300):
        if time.time() > d6:
            break
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d6, time.time() + budget_s * 0.0035)
        h_t, c5_t = _qp_loop(h_p, sub_dl, radius=0.07, stall_limit=15)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t.copy()
            scale = max(0.002, scale * 0.88)
        else:
            scale = min(0.05, scale * 1.04)

    # Phase 7: Pair swap (5%)
    d7 = t0 + budget_s * 0.87
    h7 = _project(best_h.copy(), len(best_h) / 2.0)
    h7, c5_7 = _pair_swap(h7, d7, n_tries=5000, seed=42)
    if c5_7 < best_c5:
        best_c5, best_h = c5_7, h7.copy()

    # Phase 8: Final QP + LP polish (8%)
    d8 = deadline
    h8 = _project(best_h.copy(), len(best_h) / 2.0)
    h8, c5_8 = _qp_loop(h8, d8, radius=0.03, stall_limit=60)
    if c5_8 < best_c5:
        best_c5, best_h = c5_8, h8.copy()
    if deadline - time.time() > 5:
        h8b = _project(best_h.copy(), len(best_h) / 2.0)
        h8b, c5_8b = _lp_loop(h8b, deadline, radius_init=0.02, stall_limit=30)
        if c5_8b < best_c5:
            best_c5, best_h = c5_8b, h8b.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
