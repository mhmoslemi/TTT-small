```python
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 <= x[i] <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)
    return x

    dx = 2.0 / n

def _softmax_flow(h0, deadline, beta_start=10, beta_end=200, n_steps=None):
    """Gradient flow with softmax approximation of max."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    if n_steps is None:
        n_steps = 500
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        t_frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac
        
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Softmax weights (numerically stable)
        corr_shifted = corr - corr.max()
        exp_corr = np.exp(beta * corr_shifted)
        weights = exp_corr / exp_corr.sum()
        
        # Gradient: weighted average of corr gradients
        G = _corr_grads(h)  # already includes dx
        grad = weights @ G
        
        # Project gradient to sum-zero (for the equality constraint)
        grad -= grad.mean()
        
        # Adaptive step size

        if gnorm < 1e-14:
            break
        lr = 0.1 / (gnorm + 1e-8)

        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
```

Let me use a **trust-region reflective method** with exact second-order information:

At point h, for a direction d, the change in corr[k] is:
Δcorr[k] = G[k] @ d + d^T H_k d

where H_k is the Hessian of corr[k] (which is a rank-1 matrix for bilinear forms).

Specifically, for corr[k] = sum_i h[i]*(1-h[i+k]) (simplified, ignoring boundary effects for a moment):
∂²corr[k]/∂h[i]∂h[j] = -[i+k=j] - [j+k=i] + 2*[i=j and i+k=j] (the last term is zero unless k=0)

So for k ≠ 0, H_k has entries -1 at positions (i, i+k) and (i+k, i).
For k = 0, H_0 has entries -2 at (i, i) for each i.

This means d^T H_k d = -2 * sum_i d[i] * d[i+k] (for k > 0)
or = -2 * sum_i d[i]^2 (for k = 0)

So the trust region subproblem is:
minimize max_k (corr[k] + G[k] @ d + d^T H_k d)
subject to ||d||² ≤ Δ², sum(d) = 0, (h_j - d_j) ∈ [0,1]

This is a nonconvex trust region subproblem (because of the max of quadratics). But we can solve it approximately.

minimize t
s.t. f_k(h) := corr[k] * dx - t ≤ 0 for all k

The KKT conditions involve Lagrange multipliers for each constraint. At the optimum, only the active constraints (f_k(h) = 0) have non-zero multipliers.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _corr_grads(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            G[k, :m] += (1.0 - h[s:s + m])
            G[k, s:s + m] += (-h[:m])
        else:
            g = -s
            G[k, g:g + m] += (1.0 - h[:m])
            G[k, :m] += (-h[g:g + m])
    G *= dx
    return G


def _quad_line_search(h, d, t_hi_limit):
    """Exact line search along direction d using 2nd-order model."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    # Precompute the quadratic model coefficients
    S0 = np.correlate(h, 1.0 - h, mode='full')
    # T[k] = d(corr[k])/dt at t=0 = sum of linear terms
    T = np.zeros(2 * n - 1)
    C = np.zeros(2 * n - 1)
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            T[k] = np.sum(d[:m] * (1.0 - h[s:s + m])) + np.sum((1.0 - d[s:s + m]) * (-h[:m]))
            C[k] = -np.sum(d[:m] * d[s:s + m])
        else:
            g = -s
            T[k] = np.sum(d[g:g + m] * (1.0 - h[:m])) + np.sum((1.0 - d[:m]) * (-h[g:g + m]))
            C[k] = -np.sum(d[g:g + m] * d[:m])

    # Determine valid t range
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -t_hi_limit
    t_hi = t_hi_limit
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -t_hi_limit)
    t_hi = min(t_hi, t_hi_limit)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)

    # Grid + golden section
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] + grid[:, None] ** 2 * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]

    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if 0 < best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    f1 = float(np.max(S0 + a * T + a * a * C)) * dx
    f2 = float(np.max(S0 + b * T + b * b * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, f2 = a, f1
            a = b - gr * (b - a)
            f1 = float(np.max(S0 + a * T + a * a * C)) * dx
        else:
            a, f1 = b, f2
            b = a + gr * (b - a)
            f2 = float(np.max(S0 + b * T + b * b * C)) * dx
    mid = (a + b) / 2.0
    mid_val = float(np.max(S0 + mid * T + mid * mid * C)) * dx
    if mid_val < best_val:
        best_t, best_val = mid, mid_val
    return best_t, best_val


def _slsqp_optimize(h0, deadline, max_outer=20):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        K = min(25, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]

        t_init = cur_c5 + 2e-5
        x0 = np.concatenate([h, [t_init]])

        def obj_fn(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]

        def ineq_jac(x):
            hx = x[:n]
            G = _corr_grads(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J

        def eq_fn(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]

        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-12})

        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break
    return best_h, best_c5


def _active_descent(h_init, deadline, max_iter=1000):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.15
    stall = 0
    rng = np.random.default_rng(777)

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            if radius > 0.05:
                radius = max(0.05, radius * 0.5)
            else:
                h = h + rng.normal(0, 0.02, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.15
                stall = 0
                continue
            if stall > 120:
                break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = cur_c5
        active = np.where(corr > M - 1e-4)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:10]
        elif len(active) > 25:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:25]]

        # Find LP direction
        G = _corr_grads(h)
        G_act = G[active]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((len(active), n + 1))
        A_ub[:, :n] = G_act
        A_ub[:, n] = -1.0
        b_ub = -(corr[active] - 1e-10)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        p = None
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            pass

        if p is None or np.max(np.abs(p)) < 1e-14:
            # Fallback: weighted subgradient
            weights = np.exp((corr[active] - M) / 0.001)
            weights /= weights.sum()
            p = -(weights @ G_act)
            p -= p.mean()
            pn = np.linalg.norm(p)
            if pn < 1e-14:
                stall += 10
                radius *= 0.7
                continue
            p = p / pn * radius

        alpha, new_c5 = _quad_line_search(h, p, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            if abs(alpha) > 0.4 * radius:
                radius = min(radius * 1.15, 0.4)
            else:
                radius = min(radius * 1.02, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if radius < 1e-5:
                break
    return best_h, best_c5


def _softmax_polish(h0, deadline, n_iter=3000):
    """Polish using softmax-approximated gradient with adaptive step."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = 50.0
    gamma = 1.0 + 1.0 / n_iter * 3.0

    for it in range(n_iter):
        if time.time() > deadline:
            break
        beta = min(beta * gamma, 500.0)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        C5 = corr.max()

        # Softmax weights on all lags (focused on top)
        shifted = corr - C5
        exp_vals = np.exp(beta * shifted)
        weights = exp_vals / exp_vals.sum()

        G = _corr_grads(h)
        grad = weights @ G
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 0.3 / (gnorm * n)
        improved = False
        for _ in range(30):
            h_new = _project(h - step * grad, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h_new.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return best_h, best_c5


def _coordinate_polish(h0, deadline, n_sweeps=50):
    """Coordinate-wise 1D optimization with sum compensation."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # Try adjusting h[j] and compensating uniformly in others
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_new = h.copy()
                h_new[j] += delta
                comp = -delta / (n - 1)
                h_new[list(range(n))[:j]] += comp
                h_new[list(range(n))[j+1:]] += comp
                if h_new.min() < -1e-10 or h_new.max() > 1.0 + 1e-10:
                    continue
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    # Phase 7: Final active descent + coordinate polish (5% time)
    d7 = deadline - 2.0
    h7, c5_7 = _active_descent(best_h, d7, max_iter=300)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    if time.time() < deadline - 5:
        h7b, c5_7b = _coordinate_polish(best_h, deadline - 2, n_sweeps=30)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
