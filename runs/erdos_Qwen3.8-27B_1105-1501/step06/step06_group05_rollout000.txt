def _project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)

        else:

    x = np.clip(v - theta, 0.0, 1.0)

            break
        x[mask] += res / cnt

    return x

        # Let's handle this carefully

        row[b] -= h[a]

    n = len(h)
    M = 2 * n - 1
    
    # Variables: [h_0, ..., h_{n-1}, tau]
    # Minimize tau
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize tau
    
    # Constraints: G_k · h - tau <= G_k · h_current - O_k for each k
    # i.e., G_k · h - tau <= b_k where b_k = G_k · h_current - O_k
    # In scipy linprog form: A_ub @ x <= b_ub
    
    b_ub = G @ h - O  # right-hand side for each constraint
    
    # A_ub: M rows, (n+1) columns
    # Row k: G[k] as first n entries, -1 as last entry

    A_ub[:, :n] = G
    A_ub[:, n] = -1.0

And correlate(h, d, 'full')[n-1+k] = sum_i h[i] * d[i-k]
  = sum_{i=k}^{n-1} h[i] * d[i-k]
  = sum_{j=0}^{n-k-1} h[j+k] * d[j]

So T = correlate(d, 1-h) - correlate(h, d) at position n-1+k:
  = sum_j d[j+k](1-h[j]) - h[j+k]*d[j]
  = sum_j [d[j+k](1-h[j]) - h[j+k]*d[j]] ✓

And correlate(d, d, 'full')[n-1+k] = sum_j d[j+k]*d[j] = Q_k ✓

So O_k(t) = O_k(0) + t*T_k - t^2*Q_k where T and Q are as computed.

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-13
        else:
            mask = x > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _grad_rows(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(-(n - 1), n):
        row = np.zeros(n)
        idx = k + (n - 1)
        if k >= 0:
            nn = np.arange(0, n - k)
        else:
            nn = np.arange(-k, n)
        a = nn + k
        b = nn
        row[a] += (1.0 - h[b])
        row[b] -= h[a]
        G[idx] = row
    return G


def _slp_step(h, G, O):
    n = len(h)
    M = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = G @ h - O
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.status != 0:
            return None
        return result.x[:n]
    except Exception:
        return None


def _line_search_opt(S0, T, Q, dx, h, d, max_t):

    target = n / 2.0
    if max_t < 1e-14:
        return 0.0, _exact_c5(h)
    nq = 80
    grid = np.linspace(0.0, max_t, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(30):

            c1 = b - gr * (b - a)

        else:

    t = 0.5 * (a + b)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 4 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _mixed_init(n, rng):
    x = np.arange(n)
    base = 0.5 + 0.3 * np.sin(2.0 * np.pi * x / n) + 0.2 * np.sin(4.0 * np.pi * x / n + 0.5)
    base = base + 0.1 * np.sin(6.0 * np.pi * x / n + 1.0) + rng.normal(0, 0.04, n)
    return _project(base, n / 2.0)
