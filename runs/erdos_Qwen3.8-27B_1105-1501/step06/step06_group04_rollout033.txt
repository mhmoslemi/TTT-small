def _slsqp_solve(h, active_r, n, target, M_bound, maxiter=100):
    """SLSQP with M-trick for active constraints."""
    class Prob:
        def __init__(self, h0, act, n, tgt, dx, M_ub):
            self.n = n
            self.tgt = tgt
            self.dx = dx
            self.act = act
            self.M_ub = M_ub
        
        def obj(self, x):
            return float(x[0])
        
        def obj_grad(self, x):
            g = np.zeros(self.n + 1)
            g[0] = 1.0
            return g
        
        def ineq(self, x):
            M = x[0]
            h = x[1:]
            corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
            return np.array([M - corrs[r] for r in self.act])
        
        def ineq_jac(self, x):
            M = x[0]
            h = x[1:]
            n = self.n
            dx = self.dx
            jac = np.zeros((len(self.act), n + 1))
            for i, r in enumerate(self.act):
                jac[i, 0] = 1.0
                g = _grad_corr(h, r) * dx
                jac[i, 1:] = -g
            return jac
        
        def eq(self, x):
            return float(np.sum(x[1:]) - self.tgt)
        
        def eq_jac(self, x):
            j = np.zeros(self.n + 1)
            j[1:] = 1.0
            return j
    
    prob = Prob(h, active_r, n, target, 2.0/n, M_bound)
    x0 = np.zeros(n + 1)
    x0[0] = M_bound * 0.9
    x0[1:] = h
    
    bounds = [(0.0, M_bound)] + [(0.0, 1.0)] * n
    cons = [
        {'type': 'ineq', 'fun': prob.ineq, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq, 'jac': prob.eq_jac}
    ]
    
    try:
        res = minimize(prob.obj, x0, jac=prob.obj_grad, bounds=bounds,
                      constraints=cons, method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            h_new = _project(res.x[1:], target)
            c5_new = _compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, _compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _all_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.nan_to_num(v.copy(), nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_corr(h, r):
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:]
    else:
        m = n + d
        if m > 0:
            g[:m] += (1.0 - h[-d:])
            g[-d:] -= h[:m]
    return g

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    gphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gphi * (b - a)
    c2 = a + gphi * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gphi * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gphi * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_t = max(t_lo, min(t_hi, opt_t))
    opt_val = _compute_c5(h + opt_t * d)
    return opt_t, opt_val

class _MinimaxProb:
    def __init__(self, active_r, n, target):
        self.active_r = active_r
        self.n = n
        self.target = target
        self.dx = 2.0 / n

    def obj(self, x):
        return float(x[0])

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g

    def ineq(self, x):
        M = x[0]
        h = x[1:]
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return np.array([M - corrs[r] for r in self.active_r])

    def ineq_jac(self, x):
        h = x[1:]
        n = self.n
        dx = self.dx
        jac = np.zeros((len(self.active_r), n + 1))
        for i, r in enumerate(self.active_r):
            jac[i, 0] = 1.0
            g = _grad_corr(h, r) * dx
            jac[i, 1:] = -g
        return jac

    def eq(self, x):
        return float(np.sum(x[1:]) - self.target)

    def eq_jac(self, x):
        j = np.zeros(self.n + 1)
        j[1:] = 1.0
        return j

def _slsqp_optimize(h, deadline, K=None, maxiter=80):
    n = len(h)
    target = n / 2.0
    h_work = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h_work)
    for _ in range(10):
        if time.time() > deadline:
            break
        corrs = _all_corr_scaled(h_work)
        cur_max = float(np.max(corrs))
        if K is None:
            active_r = np.arange(len(corrs))
        else:
            top_k = min(K, len(corrs))
            active_r = np.argsort(-corrs)[:top_k]
        M_lb = max(0.0, cur_max - 0.01)
        M_ub = cur_max + 0.05
        prob = _MinimaxProb(active_r, n, target)
        x0 = np.zeros(n + 1)
        x0[0] = cur_max * 0.95
        x0[1:] = h_work
        bounds = [(M_lb, M_ub)] + [(0.0, 1.0)] * n
        cons = [
            {'type': 'ineq', 'fun': prob.ineq, 'jac': prob.ineq_jac},
            {'type': 'eq', 'fun': prob.eq, 'jac': prob.eq_jac}
        ]
        try:
            res = minimize(prob.obj, x0, jac=prob.obj_grad, bounds=bounds,
                          constraints=cons, method='SLSQP',
                          options={'maxiter': maxiter, 'ftol': 1e-10})
            if res.x is not None:
                h_new = _project(res.x[1:], target)
                c5_new = _compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_work = h_new.copy()
                elif c5_new < cur_max - 1e-8:
                    h_work = h_new.copy()
        except Exception:
            pass
        if time.time() > deadline:
            break
    return best_h, best_c5

def _lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = _all_corr_scaled(h)
    c_max = float(np.max(corrs))
    shifted = beta * (corrs - c_max)
    max_shifted = float(np.max(shifted))
    exps = np.exp(shifted - max_shifted)
    weights = exps / float(np.sum(exps))
    grad = np.zeros(n)
    for r in range(len(corrs)):
        if weights[r] > 1e-12:
            grad += weights[r] * _grad_corr(h, r) * dx
    return grad, corrs

def _lse_optimize(h, deadline, betas=(30, 60, 120, 250, 500)):
    n = len(h)
    target = n / 2.0
    h_work = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h_work)
    alpha = 1.0
    for beta in betas:
        if time.time() > deadline:
            break
        stagnant = 0
        for step in range(200):
            if time.time() > deadline:
                break
            grad, corrs = _lse_gradient(h_work, beta)
            grad_tan = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_tan)
            if gnorm < 1e-14:
                break
            improved = False
            for trail in range(15):
                step_size = alpha / (trail + 1)
                h_try = _project(h_work - step_size * grad_tan, target)
                c5_try = _compute_c5(h_try)
                if c5_try < _compute_c5(h_work) - 1e-13:
                    h_work = h_try
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_work.copy()
                        stagnant = 0
                    else:
                        stagnant += 1
                    improved = True
                    break
            if not improved:
                alpha *= 0.5
                stagnant += 1
            else:
                alpha = min(alpha * 1.1, 5.0)
            if stagnant > 30:
                break
    return best_h, best_c5

def _weighted_grad_direction(h, scores, eps=0.02):
    n = len(h)
    dx = 2.0 / n
    c_max = float(np.max(scores))
    active = scores >= (c_max - eps * c_max)
    if np.sum(active) == 0:
        active = np.zeros(len(scores), dtype=bool)
        active[np.argmax(scores)] = True
    K = np.sum(active)
    if K == 0:
        return None
    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for r in active_idx:
        grad += _grad_corr(h, r) * dx
    grad /= K
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return -grad / norm

def _top_diff_direction(h, scores, top_k=5):
    n = len(h)
    dx = 2.0 / n
    sorted_idx = np.argsort(-scores)[:top_k]
    if len(sorted_idx) < 2:
        return None
    grad1 = _grad_corr(h, sorted_idx[0]) * dx
    grad2 = _grad_corr(h, sorted_idx[1]) * dx
    d = grad2 - grad1
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return -d / norm

def _random_direction(n, rng, sparsity=0.3):
    d = rng.normal(0, 1, n)
    mask = rng.random(n) > sparsity
    d[mask] = 0.0
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm
