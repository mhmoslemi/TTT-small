    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))

        else:

    theta = 0.5 * (lo + hi)

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)


def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _lag_grad(h, m):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if m >= 0:
        if m < n:
            g[m:] += (1.0 - h[:n - m]) * dx
            g[:n - m] += -h[m:] * dx
    else:
        q = -m
        if q < n:
            g[:n - q] += (1.0 - h[q:]) * dx
            g[q:] += -h[:n - q] * dx
    return g


def _active_lags(corr_vals, n, k=12):
    idx = np.argsort(-corr_vals)[:k]
    return (idx - (n - 1)).tolist()


def _slsqp_solve(h, active_lags_list, max_iter=150, ftol=1e-12):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cv = _corr_vals(h)
    z0 = float(np.max(cv)) + 2e-5

    cons = []
    for m in active_lags_list:
        obj = _C5Ineq(n, m, dx)
        cons.append({'type': 'ineq', 'fun': obj.fun, 'jac': obj.jac})
    eq = _C5Eq(n, target)
    cons.append({'type': 'eq', 'fun': eq.fun, 'jac': eq.jac})
    oj = _C5Objn(n)

    bounds = [(0.0, 1.0)] * (n + 1)
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = z0

    try:
        res = minimize(oj.fun, x0, method='SLSQP', jac=oj.jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
    except Exception:
        return h.copy(), _exact_c5(h)

    if len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
        h_new = _project(res.x[:n], target)
        return h_new, _exact_c5(h_new)
    return h.copy(), _exact_c5(h)


def _line_search(h, d, corr_raw):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    low = [0.0]
    up = [1e30]
    if np.any(pos):
        low.append(float(np.min(-h[pos] / d[pos])))
        up.append(float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        low.append(float(np.min((1.0 - h[neg]) / d[neg])))
        up.append(float(np.min(-h[neg] / d[neg])))
    L = max(low)
    U = min(up)
    if U - L < 1e-12:
        return 0.0, _exact_c5(h)
    S0 = corr_raw
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    return t, float(np.max(S0 + t * T - t * t * Q)) * dx


def _active_direction(h, corr_vals, n, k, tau, rng):
    active = _active_lags(corr_vals, n, k=k)
    m = float(np.max(corr_vals))
    G = np.zeros(n)
    for lag in active:
        idx = lag + (n - 1)
        w = np.exp((corr_vals[idx] - m) / tau)
        G += w * _lag_grad(h, lag)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm


def _min_norm_dir(h, active_lags_list, n, rng):
    A = np.array([_lag_grad(h, m) for m in active_lags_list])
    A -= A.mean(axis=1, keepdims=True)
    k = A.shape[0]
    try:
        A_pinv = np.linalg.pinv(A)
        d = -A_ppin.T if False else -A_to_T
    except Exception:
        d = _active_direction(h, _corr_vals(h), n, len(active_lags_list), 0.02, rng)
    return d


def _optimize(h, deadline, rng, slsqp_prob=0.45, k_active=12, tau=0.02, max_slsqp_iter=120):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    sigma = 0.04
    slsqp_count = 0

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 1.0:
            break
        r = rng.random()
        if r < slsqp_prob:
            cv = corr_raw * dx
            al = _active_lags(cv, n, k=k_active)
            slsqp_count += 1
            iters = max(50, max_slsqp_iter - slsqp_count // 2)
            h_try, f_try = _slsqp_solve(h, al, max_iter=iters)
            if f_try < best_f - 1e-13:
                best_f = f_try
                best_h = h_try.copy()
            if f_try < cur_f + 5e-4:
                h = h_try
                cur_f = f_try
                corr_raw = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
            else:
                stall += 1
        else:
            r2 = rng.random()
            if r2 < 0.20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                d = np.zeros(n)
                d[j] = 1.0
                d[i] = -1.0
            elif r2 < 0.30:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                d = np.zeros(n)
                d[j] = 2.0
                d[i] = -1.0
                h_half = np.zeros(n)
                h_half[j] = 1.0
                h_half[i] = -0.5
                d = np.zeros(n)
                d[j] = 1.0
                d[i] = -0.5
            elif r2 < 0.40:
                d = rng.standard_normal(n)
            else:
                d = _active_direction(h, corr_raw * dx, n, k_active, tau, rng)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d = d / nrm
            t, newf = _line_search(h, d, corr_raw)
            if abs(t) < 1e-14:
                stall += 1
                continue
            hn = h + t * d
            if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
                hn = _project(hn, target)
            nf = _exact_c5(hn)
            if nf < best_f - 1e-13:
                best_f = nf
                best_h = hn.copy()
            if nf < cur_f + 5e-4:
                h = hn
                cur_f = nf
                corr_raw = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
            else:
                h = best_h.copy()
                cur_f = best_f
                corr_raw = np.correlate(h, 1.0 - h, mode='full')
                stall += 1
        if stall > 80:
            h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr_raw = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.003, sigma * 0.85)
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n, dtype=float)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        base = _sine_init(n, 1, 0.0, rng)

    n0 = len(base)
    h0 = _project(base, n0 / 2.0)

    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: quick SLSQP on initial point and complement
    d1 = t0 + budget_s * 0.08
    cv0 = _corr_vals(h0)
    al0 = _active_lags(cv0, n0, k=14)
    h1, f1 = _slsqp_solve(h0, al0, max_iter=200)
    if f1 < best_f:
        best_f = f1
        best_h = h1.copy()

    # Try complement
    h_comp = _project(1.0 - h0, n0 / 2.0)
    fc = _exact_c5(h_comp)
    if fc < best_f:
        best_f = fc
        best_h = h_comp.copy()
    cvc = _corr_vals(h_comp)
    alc = _active_lags(cvc, n0, k=14)
    hc, fcc = _slsqp_solve(h_comp, alc, max_iter=200)
    if fcc < best_f:
        best_f = fcc
        best_h = hc.copy()

    # Phase 2: main optimization
    d2 = t0 + budget_s * 0.75
    hh, ff = _optimize(best_h, d2, rng, slsqp_prob=0.40, k_active=12, tau=0.02, max_slsqp_iter=130)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Try complement of current best
    h_comp2 = _project(1.0 - best_h, len(best_h) / 2.0)
    fc2 = _exact_c5(h_comp2)
    if fc2 < best_f:
        best_f = fc2
        best_h = h_comp2.copy()

    if time.time() < d2:
        hh2, ff2 = _optimize(best_h, d2, rng, slsqp_prob=0.50, k_active=15, tau=0.015, max_slsqp_iter=100)
        if ff2 < best_f:
            best_f = ff2
            best_h = hh2.copy()

    # Phase 3: multi-resolution
    d3 = t0 + budget_s * 0.90
    bn = len(best_h)
    tried = set()
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, -10, 10, 12, -12]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub = min(d3, time.time() + max(2.0, budget_s * 0.02))
        hr2, fr2 = _optimize(hr, sub, rng, slsqp_prob=0.45, k_active=10, tau=0.02, max_slsqp_iter=100)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 4: final deep polish
    hh, ff = _optimize(best_h, deadline, rng, slsqp_prob=0.55, k_active=16, tau=0.012, max_slsqp_iter=200)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Final complement check
    h_fin_comp = _project(1.0 - best_h, len(best_h) / 2.0)
    f_fin_c = _exact_c5(h_fin_comp)
    if f_fin_c < best_f:
        best_f = f_fin_c
        best_h = h_fin_comp.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
