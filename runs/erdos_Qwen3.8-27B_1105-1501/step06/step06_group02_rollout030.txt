    R = np.correlate(x, 1.0 - x, mode='full')
    z = (R * dx - target) / tau
    z_max = np.max(z)
    if z_max > 500:
        z_max = 500
    w = np.exp(z - z_max)
    log_sum = z_max + np.log(float(np.sum(w)))
    return tau * log_sum

def _smooth_grad(x, target, tau):
    """Gradient of log-sum-exp objective."""
    n = len(x)
    dx = 2.0 / n
    R = np.correlate(x, 1.0 - x, mode='full')
    z = (R * dx - target) / tau
    z_max = np.max(z)
    w = np.exp(z - z_max)
    W = float(np.sum(w))
    w = w / W
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w_k = w[idx]
        if w_k < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            grad[k:k + m] += w_k * (1.0 - x[:m])
            grad[:m] += w_k * (-x[k:k + m])
        else:
            g = -k
            m = n - g
            grad[:m] += w_k * (1.0 - x[g:g + m])
            grad[g:g + m] += w_k * (-x[:m])
    
    return grad * (dx / tau)

def _obj_and_grad(x, target, tau):
    """Combined objective and gradient for minimize()."""
    obj = _smooth_obj(x, target, tau)
    g = _smooth_grad(x, target, tau)
    return obj, g

def _eq_constraint(x):
    """Equality constraint: sum(x) = n/2."""
    return float(np.sum(x)) - len(x) / 2.0

def _eq_constraint_jac(x):
    n = len(x)
    return np.ones(n)

def _run_slsqp(h0, target, tau, deadline, maxiter=300):
    """Run SLSQP with smooth log-sum-exp objective."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_constraint_jac}]
    
    t_start = time.time()
    # Dynamic maxiter based on time
    if n <= 130:
        effective_maxiter = min(maxiter, 500)
    else:
        effective_maxiter = min(maxiter, 300)
    
    res = minimize(
        _obj_and_grad, h0, args=(target, tau),
        method='SLSQP', jac=True,
        bounds=bounds, constraints=constraints,
        options={'maxiter': effective_maxiter, 'ftol': 1e-10, 'iprint': 0}
    )
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = joint_project(res.x, n / 2.0)
        c5 = c5_exact(h_new)
        return h_new, c5
    return h0.copy(), c5_exact(h0)

def _exact_dir_line_search(h, direction, dx, max_iter=50):
    """Exact line search along a feasible direction."""
    n = len(h)
    # Zero-mean the direction
    direction = direction - np.mean(direction)
    # Clamp to feasible interval
    lo, hi = 0.0, 1.0
    pos = direction > 1e-12
    neg = direction < -1e-12
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        lo = max(lo, float(np.min(-h[neg] / direction[neg])))
    if hi - lo < 1e-12:
        return h.copy(), c5_exact(h)
    
    R0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(direction, 1.0 - h, mode='full') - np.correlate(h, direction, mode='full')
    C = np.correlate(direction, direction, mode='full')
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
    # Check endpoints
    f_lo = float(np.max(R0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    
    if opt > 1e-12:
        h_new = joint_project(h + opt * direction, n / 2.0)
        return h_new, val
    return h.copy(), c5_exact(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return joint_project(h_new, n_new / 2.0)
