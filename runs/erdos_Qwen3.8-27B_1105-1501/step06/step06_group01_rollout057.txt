import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_corr_mat(c, n, dx):
    m = 2 * n - 1
    M = np.zeros((m, n))
    for k in range(n):
        M[k, :k+1] = dx * c[:k+1][::-1]
    for k in range(n, m):
        start = k - n + 1
        M[k, start:] = dx * c[start:n][::-1]
    return M

def _lp_minimax(c, n):
    dx = 2.0 / n
    m = 2 * n - 1
    M = _build_corr_mat(c, n, dx)
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.hstack([M, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_c5(a, b, n_eval=60):
    best_t = 0.0
    best_val = float('inf')
    best_h = a.copy()
    ts = np.linspace(0.0, 1.0, n_eval)
    for t in ts:
        h = (1.0 - t) * a + t * b
        v = _exact_c5(h)
        if v < best_val:
            best_val = v
            best_t = t
            best_h = h.copy()
    if 0 < best_t < 1:
        lo_t = ts[max(0, int(np.argmin(np.abs(ts - best_t))) - 1)]
        hi_t = ts[min(n_eval - 1, int(np.argmin(np.abs(ts - best_t))) + 1)]
        for t2 in np.linspace(lo_t, hi_t, 20):
            h = (1.0 - t2) * a + t2 * b
            v = _exact_c5(h)
            if v < best_val:
                best_val = v
                best_t = t2
                best_h = h.copy()
    return best_h, best_val, best_t

            d[move_j] = -1.0
            best_alpha = 0.0
            best_new = best_c5
            for alpha in np.linspace(-0.3, 0.3, 30):
                h_test = h + alpha * d
                if h_test.min() < -0.01 or h_test.max() > 1.01:
                    continue
                h_test = np.clip(h_test, 0, 1)
                if abs(float(np.sum(h_test)) - n / 2.0) > 0.5:
                    h_test = _project(h_test, n / 2.0)
                c5_test = _exact_c5(h_test)
                if c5_test < best_new:
                    best_new = c5_test
                    best_alpha = alpha
            if best_alpha != 0.0 and best_new < best_c5 - 1e-12:
                h = h + best_alpha * d
                h = _project(h, n / 2.0)
                best_c5 = _exact_c5(h)
                best_h = h.copy()
            else:
                h = h + np.random.default_rng(it * 7).normal(0, 0.01, n)
                h = _project(h, n / 2.0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        h0 = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * np.arange(n0) / n0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Alternate LP optimization from parent (45% budget)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = _alt_optimize(best_h, n0, d1, rng, max_iter=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Local refine (10% budget)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = _local_refine(best_h, n0, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution search (25% budget)
    d3 = t0 + budget_s * 0.80
    n_best = len(best_h)
    tried_n = set([n_best])
    offsets = sorted([2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 16, -16, 20, -20, 30, -30])
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 16 or n_try > 256 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _alt_optimize(h_r, n_try, sub_dl, rng, max_iter=40)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try
            # Quick local refine at new resolution
            if time.time() < d3:
                sub_dl2 = min(d3, time.time() + budget_s * 0.008)
                h_r2, c5_r2 = _local_refine(best_h, n_best, sub_dl2)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase 4: Diverse starts (10% budget)
    d4 = deadline
    n_starts = 4
    for si in range(n_starts):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + (d4 - time.time()) / max(1, n_starts - si))
        h_s = _random_partner(n_best, rng)
        h_s, c5_s = _alt_optimize(h_s, n_best, sub_dl, rng, max_iter=30)
        if c5_s < best_c5 - 1e-10:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Final: local refine with remaining time
    if time.time() < deadline:
        h_f, c5_f = _local_refine(best_h, len(best_h), deadline)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
