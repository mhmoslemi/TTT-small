        return np.zeros(n)

            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * C))
    opt_a = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_a * T + opt_a * opt_a * C))
    fa = float(np.max(S0 + a * T + a * a * C))
    fb = float(np.max(S0 + b * T + b * b * C))
    if fa < opt_val:
        opt_a, opt_val = a, fa
    if fb < opt_val:
        opt_a, opt_val = b, fb
    return opt_a, opt_val

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k >= 0:
            m = n - k
            G[q, k:] += dx * (1.0 - h[:m])
            G[q, :m] += -dx * h[k:]
        else:
            d_abs = -k
            m = n - d_abs
            G[q, :m] += dx * (1.0 - h[d_abs:])
            G[q, d_abs:] += -dx * h[:m]
    return G

def _lp_dir(h, G_sub, S_sub, radius):
    n = len(h)
    n_rows = G_sub.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G_sub, -np.ones((n_rows, 1))])
    b_ub = -S_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _local_search(h, deadline, max_iter=8000, start_radius=0.1, rng_seed=42):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = start_radius
    stall = 0
    perturb_count = 0
    rng = np.random.default_rng(rng_seed)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        S = _corr_vals(h)
        cur_max = float(np.max(S))

        # Build gradient matrix
        G = _build_G(h, dx)

        # Strategy 1: LP on top active lags
        top_frac = 0.92
        active = np.where(S >= cur_max * top_frac)[0]
        if len(active) == 0:
            active = np.array([np.argmax(S)])
        if len(active) > 60:
            active = active[:60]

        p = _lp_dir(h, G[active], S[active], radius)
        if p is not None and float(np.max(np.abs(p))) > 1e-12:
            alpha, new_c5 = _line_search(h, p, dx, radius)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * p
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                predicted = float(np.max(S + G[active] @ p))
                ratio = cur_c5 / max(predicted, 1e-15)
                if ratio < 0.5:
                    radius = min(radius * 1.12, 0.30)
                elif ratio > 1.8:
                    radius *= 0.82
                else:
                    radius = min(radius * 1.01, 0.25)
                continue

        max_idx = int(np.argmax(S))
        grad = G[max_idx]
        pos_mask = (grad > 0.003) & (h > 0.02)
        neg_mask = (grad < -0.003) & (h < 0.98)
        found = False
        if np.any(pos_mask) and np.any(neg_mask):

            neg_j = np.where(neg_mask)[0]

            nj_top = neg_j[np.argsort(grad[neg_j])[:6]]

                    d[j] = 1.0

                        cur_c5 = _exact_c5(h)
