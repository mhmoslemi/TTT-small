    grid = np.linspace(a_lo, a_hi, n_grid)
    vals = np.max(S0[None, :] + grid[:, None] * T[None, :] + (grid[:, None] ** 2) * Q[None, :], axis=1)
    best_i = int(np.argmin(vals))
    best_a = grid[best_i]
    best_val = vals[best_i]
    
    # Golden section refinement
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = a_lo, a_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    
    def f_alf(al):
        return float(np.max(S0 + al * T + al * al * Q))
    
    f1, f2 = f_alf(c1), f_alf(c2)
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f_alf(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_alf(c2)
    
    opt_a = (a + b) / 2.0
    opt_val = f_alf(opt_a)
    if opt_val < best_val:
        best_a, best_val = opt_a, opt_val
    
    return best_a, best_val

def _softmax_descent(h, deadline, tau_init=0.015, tau_min=0.002, 
                     max_iter=5000, lr_init=1.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_init
    lr = lr_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        S = _corr_vals(h)
        G = _build_G(h, dx)
        
        # Soft-max weights
        S_shifted = (S - np.max(S)) / tau
        w = np.exp(S_shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            break
        w /= w_sum
        
        # Gradient direction (projected to sum=0)
        grad = G.T @ w
        d = -(grad - np.mean(grad))
        
        # Scale direction
        d_norm = float(np.max(np.abs(d)))
        if d_norm < 1e-15:
            break
        d = (d / d_norm) * lr
        
        # Line search
        alpha, new_val = _line_search(h, d, dx, alpha_max=1.0)
        
        if new_val < best_c5 - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < best_c5 - 1e-14:
                best_c5 = actual_c5
                best_h = h_new.copy()
                h = h_new
                stall = 0
                lr = min(lr * 1.03, 2.0)
            else:
                h = h_new
                stall += 1
                lr *= 0.95
        else:
            stall += 1
            lr *= 0.9
            
        if stall > 50:
            tau *= 0.8
            lr = max(lr, 0.3)
            stall = 0
            if tau < tau_min:
                break
        
        # Anneal tau periodically
        if it % 100 == 0 and tau > tau_min:
            tau *= 0.95
    
    return best_h, best_c5
