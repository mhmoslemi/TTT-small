def project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):

        else:

def _corr_and_grads(h):
    """Compute all row values and the full gradient structure for soft-max."""
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    return R

def smooth_func(h, T, tau):
    """Soft-max objective and gradient for SLSQP.
    Minimizes tau * log(sum exp((R_k - T)/tau))
    where R_k = correlate(h, 1-h, 'full')[k].
    """
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    z = (R - T) / tau
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    sum_exp = np.sum(exp_z)
    value = tau * (z_max + np.log(sum_exp))
    w = exp_z / sum_exp  # softmax weights, shape (2n-1,)
    
    # Compute gradient
    grad = np.zeros(n)
    for d in range(0, n):
        # weight for lag d
        wd = w[d + n - 1]
        if wd > 1e-15:
            # First term: (1-h[j-d]) for j = d, ..., n-1
            grad[d:] += wd * (1.0 - h[:n-d])
            # Second term: -h[j+d] for j = 0, ..., n-1-d
            grad[:n-d] += wd * (-h[d:])
    for d in range(1, n):
        # weight for lag -d
        wd = w[n - 1 - d]
        if wd > 1e-15:
            # First term: (1-h[j+d]) for j = 0, ..., n-1-d
            grad[:n-d] += wd * (1.0 - h[d:])
            # Second term: -h[j-d] for j = d, ..., n-1
            grad[d:] += wd * (-h[:n-d])
    
    return value, grad

def sum_constraint(h):
    return float(np.sum(h)) - len(h) / 2.0

def _resample(h, n_new):

    if n_new == n:

def fourier_obj(params, n):

    if time.time() > _de_deadline:

def _slsqp_solve(h0, T, tau, maxiter=500):
    """Run SLSQP on the soft-max objective."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    result = minimize(
        smooth_func, h0.copy(),
        args=(T, tau),
        method='SLSQP',
        jac=True,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    if result.x is not None and np.all(np.isfinite(result.x)):
        h_sol = project(result.x, n / 2.0)
        return h_sol, c5_exact(h_sol)
    return h0.copy(), c5_exact(h0)

def _exact_direction_search(h, direction, S0):
    """Exact line search along a given direction."""
    n = len(h)
    dx = 2.0 / n
    # Ensure mean-zero direction
    direction = direction - np.mean(direction)
    # Feasible range
    pos = direction > 1e-10
    neg = direction < -1e-10
    lo = 0.0
    hi = 1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(-h[neg] / direction[neg])))
    if hi - lo < 1e-14:
        return h.copy(), c5_exact(h)
    
    T_vec = np.correlate(direction, 1.0 - h, 'full') - np.correlate(h, direction, 'full')
    C_vec = np.correlate(direction, direction, 'full')
    
    # Scan for bracket
    alphas = np.linspace(lo, hi, 50)
    vals = np.array([np.max(S0 + a * T_vec - a * a * C_vec) * dx for a in alphas])
    best_idx = int(np.argmin(vals))
    
    # Golden section in bracket
    a = alphas[max(0, best_idx - 1)]
    b = alphas[min(len(alphas) - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_vec - c1 * c1 * C_vec)) * dx
    f2 = float(np.max(S0 + c2 * T_vec - c2 * c2 * C_vec)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_vec - c1 * c1 * C_vec)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_vec - c2 * c2 * C_vec)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T_vec - opt * opt * C_vec)) * dx
    # Check endpoints
    f_lo = float(np.max(S0 + lo * T_vec - lo * lo * C_vec)) * dx
    f_hi = float(np.max(S0 + hi * T_vec - hi * hi * C_vec)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    
    h_new = h + opt * direction
    h_new = project(h_new, n / 2.0)
    return h_new, c5_exact(h_new)

def _slsqp_polish(h, deadline, best_c5):
    """Multiple rounds of SLSQP with decreasing tau."""
    n = len(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    S0 = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    
    tau_list = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    
    for tau in tau_list:
        if time.time() > deadline:
            break
        # Target slightly below current best
        T = cur_c5 * 0.99
        h_new, c5_new = _slsqp_solve(cur_h, T, tau, maxiter=300)
        if c5_new < cur_c5 - 1e-12:
            cur_h = h_new
            cur_c5 = c5_new
            S0 = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        # Gradient direction polish
        if time.time() > deadline:
            break
        # Use soft-max weights as direction
        z = (S0 - T) / tau
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        w = exp_z / np.sum(exp_z)
        # Compute weighted gradient (negative = descent direction)
        grad = np.zeros(n)
        for d in range(0, n):
            wd = w[d + n - 1]
            if wd > 1e-15:
                grad[d:] += wd * (1.0 - cur_h[:n-d])
                grad[:n-d] += wd * (-cur_h[d:])
        for d in range(1, n):
            wd = w[n - 1 - d]
            if wd > 1e-15:
                grad[:n-d] += wd * (1.0 - cur_h[d:])
                grad[d:] += wd * (-cur_h[:n-d])
        # Descent direction is -grad
        direction = -grad
        h_new, c5_new = _exact_direction_search(cur_h, direction, S0)
        if c5_new < cur_c5 - 1e-12:
            cur_h = h_new
            cur_c5 = c5_new
            S0 = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    
    return cur_h, cur_c5

    n = len(h)
    dx = 2.0 / n
    direction = direction - np.mean(direction)
    pos = direction > 1e-10
    neg = direction < -1e-10
    lo = 0.0
    hi = 1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(-h[neg] / direction[neg])))
    if hi - lo < 1e-14:
        return h.copy(), c5_exact(h)
    Tv = np.correlate(direction, 1.0 - h, 'full') - np.correlate(h, direction, 'full')
    Cv = np.correlate(direction, direction, 'full')
    alphas = np.linspace(lo, hi, 60)
    vals = np.array([float(np.max(S0 + a * Tv - a * a * Cv)) * dx for a in alphas])
    bi = int(np.argmin(vals))
    a = alphas[max(0, bi - 1)]
    b = alphas[min(len(alphas) - 1, bi + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * Tv - c1 * c1 * Cv)) * dx
    f2 = float(np.max(S0 + c2 * Tv - c2 * c2 * Cv)) * dx
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * Tv - c1 * c1 * Cv)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * Tv - c2 * c2 * Cv)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * Tv - opt * opt * Cv)) * dx
    f_lo = float(np.max(S0 + lo * Tv - lo * lo * Cv)) * dx
    f_hi = float(np.max(S0 + hi * Tv - hi * hi * Cv)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    h_new = h + opt * direction
    h_new = project(h_new, n / 2.0)
    return h_new, c5_exact(h_new)

def _slsqp_single(h0, T, tau, maxiter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        result = minimize(
            smooth_func, h0.copy(),
            args=(T, tau),
            method='SLSQP', jac=True,
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
    except Exception:
        return h0.copy(), c5_exact(h0)
    if result.x is not None and np.all(np.isfinite(result.x)):
        h_sol = project(result.x, n / 2.0)
        return h_sol, c5_exact(h_sol)
    return h0.copy(), c5_exact(h0)

def _softmax_polish(h, deadline, cur_c5):
    n = len(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(best_h, 1.0 - best_h, mode='full')
    tau_list = [0.10, 0.06, 0.035, 0.02, 0.012, 0.007, 0.004, 0.002]
    for tau in tau_list:
        if time.time() > deadline:
            break
        T = best_c5 * 0.995
        h_new, c5_new = _slsqp_single(best_h, T, tau, maxiter=350)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
            S0 = np.correlate(best_h, 1.0 - best_h, mode='full')
        if time.time() > deadline:
            break
        grad = _compute_soft_grad(best_h, T, tau)
        direction = -grad
        gn = np.linalg.norm(direction)
        if gn < 1e-14:
            break
        direction = direction / gn * 0.5
        h_new2, c5_new2 = _exact_line_search(best_h, direction, S0)
        if c5_new2 < best_c5 - 1e-13:
            best_h = h_new2
            best_c5 = c5_new2
            S0 = np.correlate(best_h, 1.0 - best_h, mode='full')
        if time.time() > deadline:
            break
        grad2 = _compute_soft_grad(best_h, T, tau * 0.5)
        direction2 = -grad2
        h_new3, c5_new3 = _exact_line_search(best_h, direction2, S0)
        if c5_new3 < best_c5 - 1e-13:
            best_h = h_new3
            best_c5 = c5_new3
            S0 = np.correlate(best_h, 1.0 - best_h, mode='full')
    return best_h, best_c5

def _perturbed_slsqp(h, deadline, cur_c5, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for trial in range(6):
        if time.time() > deadline:
            break
        h_pert = h + rng.normal(0, 0.03, n)
        h_pert = project(h_pert, n / 2.0)
        c5_pert = c5_exact(h_pert)
        start_h = h_pert if c5_pert < cur_c5 + 0.01 else h
        T = min(start_h_c5, best_c5) * 0.99 if (start_h_c5 := c5_exact(start_h)) < best_c5 else best_c5 * 0.99
        h_new, c5_new = _slsqp_single(start_h, best_c5 * 0.99, 0.04, maxiter=300)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5

_de_deadline = 0.0

def _fourier_obj(params, n):
    if time.time() > _de_deadline:
        return 1e8
    x = (np.arange(n) + 0.5) / n
    h = np.full(n, 0.5)
    for k in range(1, 4):
        ak = params[2 * (k - 1)]
        bk = params[2 * (k - 1) + 1]
        h += ak * np.cos(2.0 * np.pi * k * x)
        h += bk * np.sin(2.0 * np.pi * k * x)
    h = project(h, n / 2.0)
    return c5_exact(h)

def _fourier_de_search(n, deadline, best_c5, seed):
    global _de_deadline
    _de_deadline = deadline
    bounds = [(-0.6, 0.6)] * 6
    try:
        de_res = differential_evolution(
            _fourier_obj, bounds, args=(n,),
            seed=seed, maxiter=300, popsize=12, tol=1e-7, polish=False
        )
    except Exception:
        return None, 1e8
    if de_res.x is not None and np.all(np.isfinite(de_res.x)):
        x = (np.arange(n) + 0.5) / n
        h_fe = np.full(n, 0.5)
        for k in range(1, 4):
            ak = de_res.x[2 * (k - 1)]
            bk = de_res.x[2 * (k - 1) + 1]
            h_fe += ak * np.cos(2.0 * np.pi * k * x)
            h_fe += bk * np.sin(2.0 * np.pi * k * x)
        h_fe = project(h_fe, n / 2.0)
        return h_fe, c5_exact(h_fe)
    return None, 1e8

def _pair_polish(h, deadline, n_pairs=30):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(999)
    stall = 0
    while time.time() < deadline and stall < 100:
        scores = S0 * dx
        max_s = float(np.max(scores))
        threshold = max_s - 0.003 * max_s
        active = np.where(scores >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-scores)[:8]
        w_softmax = np.exp((scores[active] - max_s) / (0.005 * max_s + 1e-15))
        w_softmax /= np.sum(w_softmax)
        grad = np.zeros(n)
        for idx in active:
            d = idx - (n - 1)
            wt = w_softmax[np.where(active == idx)[0][0]] * dx
            m = n - abs(d)
            if m <= 0:
                continue
            if d >= 0:
                if d == 0:
                    grad[:m] += wt * (1.0 - 2.0 * h[:m])
                else:
                    grad[:m] += wt * (1.0 - h[d:d + m])
                    grad[d:d + m] += wt * (-h[:m])
            else:
                g = -d
                if g == 0:
                    grad[:] += wt * (1.0 - 2.0 * h[:])
                else:
                    grad[g:] += wt * (1.0 - h[:m])
                    grad[:m] += wt * (-h[g:g + m])
        donors = np.argsort(-grad)[:n_pairs]
        recv = np.argsort(grad)[:n_pairs]
        best_alpha = 0.0
        best_new = cur_c5
        best_pair = None
        tried = set()
        for i in donors:
            if time.time() > deadline:
                break
            for j in recv:
                if i == j:
                    continue
                key = (min(int(i), int(j)), max(int(i), int(j)))
                if key in tried:
                    continue
                tried.add(key)
                di = int(i)
                dj = int(j)
                lo_a = -min(h[dj], 1.0 - h[di])
                hi_a = min(h[di], 1.0 - h[dj])
                if hi_a - lo_a < 1e-14:
                    continue
                d_vec = np.zeros(n)
                d_vec[di] = -1.0
                d_vec[dj] = 1.0
                Tv = np.correlate(d_vec, 1.0 - h, 'full') - np.correlate(h, d_vec, 'full')
                Cv = np.correlate(d_vec, d_vec, 'full')
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo_a, hi_a
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                f1 = float(np.max(S0 + c1 * Tv - c1 * c1 * Cv)) * dx
                f2 = float(np.max(S0 + c2 * Tv - c2 * c2 * Cv)) * dx
                for _ in range(35):
                    if b - a < 1e-13:
                        break
                    if f1 < f2:
                        b, c2, f2 = c2, c1, f1
                        c1 = b - gr * (b - a)
                        f1 = float(np.max(S0 + c1 * Tv - c1 * c1 * Cv)) * dx
                    else:
                        a, c1, f1 = c1, c2, f2
                        c2 = a + gr * (b - a)
                        f2 = float(np.max(S0 + c2 * Tv - c2 * c2 * Cv)) * dx
                opt = (a + b) / 2.0
                val = float(np.max(S0 + opt * Tv - opt * opt * Cv)) * dx
                f_lo = float(np.max(S0 + lo_a * Tv - lo_a * lo_a * Cv)) * dx
                f_hi = float(np.max(S0 + hi_a * Tv - hi_a * hi_a * Cv)) * dx
                if f_lo < val:
                    opt, val = lo_a, f_lo
                if f_hi < val:
                    opt, val = hi_a, f_hi
                if abs(opt) > 1e-10 and val < best_new - 1e-14:
                    best_new = val
                    best_alpha = opt
                    best_pair = (di, dj)
        if best_pair is not None and best_new < cur_c5 - 1e-13:
            di, dj = best_pair
            h[di] -= best_alpha
            h[dj] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                idx = rng.integers(n, size=3)
                for ix in idx:
                    h[ix] += rng.uniform(-0.04, 0.04)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)
    try:
        h_parent = np.asarray(initial_h_values, dtype=float)
    except NameError:
        h_parent = _make_initial(122)
    n = len(h_parent)
    h = project(h_parent.copy(), n / 2.0)
    best_c5 = c5_exact(h)
    best_h = h.copy()

    # Phase 1: Softmax SLSQP continuation (40% time)
    d1 = t0 + budget_s * 0.40
    h_p1, c5_p1 = _softmax_polish(best_h, d1, best_c5)
    if c5_p1 < best_c5:
        best_c5 = c5_p1
        best_h = h_p1.copy()

    # Phase 2: Perturbed restarts with SLSQP (10% time)
    d2 = t0 + budget_s * 0.50
    if time.time() < d2:
        h_p2, c5_p2 = _perturbed_slsqp(best_h, d2, best_c5, rng)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()

    # Phase 3: Fourier DE for diverse starts (15% time)
    d3 = t0 + budget_s * 0.65
    if time.time() < d3:
        h_fe, c5_fe = _fourier_de_search(n, d3, best_c5, seed + 7)
        if c5_fe < best_c5:
            best_c5 = c5_fe
            best_h = h_fe.copy()
        # Also try DE at different resolutions
        for n_try in [n - 6, n + 6]:
            if time.time() > d3:
                break
            if n_try < 50 or n_try > 200 or n_try % 2 != 0:
                continue
            h_fe2, c5_fe2 = _fourier_de_search(n_try, d3, best_c5, seed + 13)
            if c5_fe2 < best_c5:
                best_c5 = c5_fe2
                best_h = h_fe2.copy()
                n = n_try

    # Phase 4: SLSQP polish from best (15% time)
    d4 = t0 + budget_s * 0.80
    if time.time() < d4:
        h_p4, c5_p4 = _softmax_polish(best_h, d4, best_c5)
        if c5_p4 < best_c5:
            best_c5 = c5_p4
            best_h = h_p4.copy()

    # Phase 5: Multi-resolution resample + polish (10% time)
    d5 = t0 + budget_s * 0.90
    if time.time() < d5:
        for dn in [-4, 4, -8, 8]:
            if time.time() > d5:
                break
            n_try = len(best_h) + dn
            if n_try < 50 or n_try > 200 or n_try % 2 != 0:
                continue
            h_r = _resample(best_h, n_try)
            c5_r = c5_exact(h_r)
            sub_dl = min(d5, time.time() + budget_s * 0.02)
            h_r, c5_r = _softmax_polish(h_r, sub_dl, c5_r)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Final pair polish (5% time)
    d6 = deadline
    h_p6, c5_p6 = _pair_polish(best_h, d6, n_pairs=40)
    if c5_p6 < best_c5:
        best_c5 = c5_p6
        best_h = h_p6.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
