        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):

        if s > target:

        else:
            hi = mid

        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += (1.0 - h[:m])
        g[:m] -= h[k:]
    else:
        ka = -k
        m = n - ka
        if m <= 0:
            return g
        g[:m] += (1.0 - h[ka:])
        g[ka:] -= h[:m]
    return g

def _top_grad(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.ones(K) / K
    else:
        w /= ws
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _grad_row(h, k)
    return grad, corr
