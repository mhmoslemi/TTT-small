def _exact_c5(h):

def _project(v, target):

    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else:

    target = n / 2.0
    h = h.copy()

    rng = np.random.default_rng(seed)
    d_vals = np.arange(2 * n - 1) - (n - 1)
    lam = 0.0
    step = 0.5

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _feasible_inner(v, A, B):
    pos = A > 1e-15
    neg = A < -1e-15
    zero = ~(pos | neg)
    if np.any(zero) and np.max(B[zero]) > v + 1e-14:
        return 0.0, 1.0, False
    hi_x = 1.0
    if np.any(pos):
        hi_x = min(hi_x, float(np.min((v - B[pos]) / A[pos])))
    lo_x = 0.0
    if np.any(neg):
        lo_x = max(lo_x, float(np.max((v - B[neg]) / A[neg])))
    return lo_x, hi_x, (lo_x <= hi_x + 1e-14)

def _solve_1d(A, B):

def _coord_descent(h, deadline, max_passes=10000, seed=42, stall_limit=200):

    target = n / 2.0
    h = _project(h.copy(), target)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    d_vals = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    
    for _ in range(max_passes):

            break
        if stall > stall_limit:
            break
        order = rng.permutation(n)
        for j in order:
            if time.time() > deadline:
                return best_h, best_c5
            j_plus = j + d_vals
            j_minus = j - d_vals
            valid_plus = (j_plus >= 0) & (j_plus < n)
            valid_minus = (j_minus >= 0) & (j_minus < n)
            
            A = np.zeros(2 * n - 1)
            A[valid_plus] += 1.0 - h[j_plus[valid_plus]]
            A[valid_minus] -= h[j_minus[valid_minus]]

            stall = 0
