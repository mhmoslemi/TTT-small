        pair_frac = 0.15 + 0.25 * frac
        rand_frac = 0.08 * (1.0 - frac)
        
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        else:
            # Log-sum-exp direction (smooth minimax subgradient)
            alpha_ls = 15.0 + 40.0 * frac
            G = _lgsumexp_dir(h, corr, alpha_ls, k_active)
            d = -G
            d -= d.mean()
            # Combine with momentum
            momentum_v = mom_beta * momentum_v + 0.3 * (d / (np.linalg.norm(d) + 1e-14))
            momentum_v -= momentum_v.mean()
            d = momentum_v
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                d = rng.standard_normal(n)
                d -= d.mean()
                nrm = float(np.linalg.norm(d))
            d = d / nrm
        
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue
        
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, L, U, dx)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        nf = _c5(hn)
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            momentum_v *= 0.95
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1
            momentum_v *= 0.5
        
        if stall > 120:
            restarts += 1
            if restarts > 40:
                h = _project(best_h + rng.normal(0, max(0.03, sigma * 3), n), target)
            else:
                h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            momentum_v = np.zeros(n)
            sigma = max(0.003, sigma * 0.85)
    
    return best_h, best_f


def _pair_heavy_optimize(h0, deadline, rng):
    """Intensive pair and triple move optimization."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy().astype(float)
    cur_f = _c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    
    while time.time() < deadline:
        r = rng.random()
        if r < 0.6:
            # Pair move with gradient-informed selection
            active = np.argsort(-corr)
            G = np.zeros(n)
            for k_idx in active[:12]:
                ell = k_idx - (n - 1)
                G += _row_grad(h, ell)
            # Increase where G is most negative, decrease where most positive
            i = int(np.argmin(G))
            j = int(np.argmax(G))
            if i == j:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        elif r < 0.8:
            # Random pair
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        else:
            # Triple move: increase i, increase j, decrease k
            idx = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx[0]), int(idx[1]), int(idx[2])
            # 2D search over (t1, t2): h[i]+=t1, h[j]+=t2, h[k]-=(t1+t2)
            # Simplify: search over t1 with t2=t1 (symmetric) or t2=0
            # For simplicity, just do a pair (i,k) or (j,k)
            if rng.random() < 0.5:
                d = np.zeros(n)
                d[i] = 1.0
                d[k] = -1.0
            else:
                d = np.zeros(n)
                d[j] = 1.0
                d[k] = -1.0
        
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-12:
            stall += 1
            continue
        
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, nf = _line_search(corr, T, Q, L, U, dx)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        nf = _c5(hn)
        if nf < best_f:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + 2e-5:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            stall += 1
            if stall > 300:
                h = _project(best_h + rng.normal(0, 0.02, n), target)
                cur_f = _c5(h)
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 6))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        base = _sine_init(n, 1, 0.0, rng)
    
    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _c5(h0)
    
    # Phase 1: Quick multi-start (8% of budget)
    d1 = t0 + budget_s * 0.08
    starts = [h0.copy()]
    freq_phase = [(1, 0.5), (2, 0.0), (3, 1.0), (4, 0.5)]
    for f_, p_ in freq_phase:
        starts.append(_sine_init(n0, f_, p_, rng))
    starts.append(_block_init(n0, rng))
    for s in starts:
        if time.time() > d1:
            break
        sub = min(d1, time.time() + max(1.5, budget_s * 0.012))
        hh, ff = _optimize(s, sub, rng)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()
    
    # Phase 2: Main deep optimization (45% of budget)
    d2 = t0 + budget_s * 0.53
    hh, ff = _optimize(best_h, d2, rng)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    
    # Phase 3: Pair-heavy refinement (17% of budget)
    d3 = t0 + budget_s * 0.70
    hh, ff = _pair_heavy_optimize(best_h, d3, rng)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    
    # Phase 4: Multi-resolution search (15% of budget)
    d4 = t0 + budget_s * 0.85
    bn = len(best_h)
    tried = set()
    dn_list = [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 14, -14, 20, -20]
    for dn in dn_list:
        if time.time() > d4:
            break
        n_try = bn + dn
        if n_try < 24 or n_try > 300:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub = min(d4, time.time() + max(2.0, budget_s * 0.014))
        hr2, fr2 = _optimize(hr, sub, rng)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try
    
    # Phase 5: Final intensive polish (15% of budget)
    hh, ff = _pair_heavy_optimize(best_h, deadline - 0.5, rng)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()
    
    # One final optimize pass on the best
    if time.time() < deadline - 1.0:
        hh, ff = _optimize(best_h, deadline - 0.3, rng)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _c5(best_h)
    return best_h, final_f, len(best_h)
