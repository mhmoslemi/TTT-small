import numpy as np
import time
from scipy.optimize import minimize


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 120
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], float(obj[i])
    return t, tv


def _row_grad(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    a = max(0, lag)
    b = min(n, n + lag)
    for idx in range(a, b):
        g[idx] += dx * (1.0 - h[idx - lag])
        g[idx - lag] -= dx * h[idx]
    return g


def _row_grad_vec(h, lags, dx):
    n = len(h)
    G = np.zeros((len(lags), n))
    for i, lag in enumerate(lags):
        a = max(0, lag)
        b = min(n, n + lag)
        ids = np.arange(a, b)
        G[i, ids] += dx * (1.0 - h[ids - lag])
        G[i, ids - lag] -= dx * h[ids]
    return G


class _EpiSolver:
    def __init__(self, n, lags, dx, target):
        self.n = n
        self.lags = lags
        self.dx = dx
        self.target = target
        self.valid_ids = []
        for lag in lags:
            a = max(0, lag)
            b = min(n, n + lag)
            self.valid_ids.append(np.arange(a, b))

    def obj(self, x):
        return x[self.n]

    def obj_jac(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def ineq(self, x):
        h = x[:self.n]
        z = x[self.n]
        m = len(self.lags)
        vals = np.empty(m)
        dx = self.dx
        for i in range(m):
            ids = self.valid_ids[i]
            lag = self.lags[i]
            vals[i] = z - dx * np.sum(h[ids] * (1.0 - h[ids - lag]))
        return vals

    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        m = len(self.lags)
        dx = self.dx
        J = np.zeros((m, n + 1))
        for i in range(m):
            ids = self.valid_ids[i]
            lag = self.lags[i]
            J[i, n] = 1.0
            np.add.at(J[i], ids, dx * (1.0 - h[ids - lag]))
            np.add.at(J[i], ids - lag, -dx * h[ids])
        return J

    def eq(self, x):
        return np.sum(x[:self.n]) - self.target

    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


def _slsqp_step(h_cur, z_init, top_k, max_iter, ftol, deadline):
    n = len(h_cur)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    idx = np.argsort(-corr)[:top_k]
    lags = (idx - (n - 1)).tolist()

    solver = _EpiSolver(n, lags, dx, target)
    x0 = np.concatenate([h_cur, [z_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, z_init + 0.02)]
    constraints = [
        {'type': 'ineq', 'fun': solver.ineq, 'jac': solver.ineq_jac},
        {'type': 'eq', 'fun': solver.eq, 'jac': solver.eq_jac}
    ]
    try:
        result = minimize(solver.obj, x0, method='SLSQP', jac=solver.obj_jac,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': ftol})
    except Exception:
        return None, None

    if result.success and result.x is not None:
        h_new = np.clip(result.x[:n], 0.0, 1.0)
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < z_init - 1e-10:
            return h_new, f_new
    return None, None


def _slsqp_phase(h_start, deadline, rng, best_f):

def _smooth_direction(h, tau, max_rows):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = dx * corr
    top_idx = np.argsort(-s)[:max_rows]
    top_s = s[top_idx]
    weights = np.exp(tau * (top_s - top_s.max()))
    weights /= weights.sum()
    lags = (top_idx - (n - 1)).tolist()
    G = _row_grad_vec(h, lags, dx)
    d = G.T @ weights
    d -= d.mean()
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        return None
    return d / nrm


def _directional_optimize(h, deadline, rng, pair_frac=0.20, rand_frac=0.08,
                          smooth_tau=30.0, smooth_rows=12, k_active=8, tau_act=0.02):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    restarts = 0
    sigma = 0.05
    uphill = 5e-4

    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            # Use smooth-max direction
            d = _smooth_direction(h, smooth_tau, smooth_rows)
            if d is None:
                d = rng.standard_normal(n)
                d -= d.mean()

        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm

        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue

        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, dx, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < 0.0 or hn.max() > 1.0:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-13:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1

        if stall > 120:
            restarts += 1
            if restarts > 40:
                h = _project(best_h + rng.normal(0, max(0.05, sigma * 3), n), target)
            else:
                h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.004, sigma * 0.9)

    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 4 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def _logistic_init(n, center, width, rng):
    x = np.arange(n) / n
    h = 1.0 / (1.0 + np.exp(-(x - center) / width))
    h = h + rng.normal(0, 0.02, n)
    return _project(h, n / 2.0)
