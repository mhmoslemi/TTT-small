import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 4.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(n):
        r = n - 1 + d
        m = n - d
        if m <= 0:
            continue
        G[r, :m] += (1.0 - h[d:d + m])
        G[r, d:] += -h[:m]
    for p in range(1, n):
        r = n - 1 - p
        m = n - p
        if m <= 0:
            continue
        G[r, p:] += (1.0 - h[:m])
        G[r, :m] += -h[p:]
    return G

def _pair_line_search(h, i, j, S, dx, radius):
    n = len(h)
    T = np.zeros(2 * n - 1)
    C2 = np.zeros(2 * n - 1)
    for d in range(n):
        r = n - 1 + d
        m = n - d
        if m <= 0:
            continue
        gi = 0.0
        if 0 <= i < m:
            gi += (1.0 - h[min(i + d, n - 1)]) if i + d < n else 0.0
        if d <= i < n:
            gi += -h[max(i - d, 0)] if i - d >= 0 else 0.0
        gj = 0.0
        if 0 <= j < m:
            gj += (1.0 - h[min(j + d, n - 1)]) if j + d < n else 0.0
        if d <= j < n:
            gj += -h[max(j - d, 0)] if j - d >= 0 else 0.0
        T[r] = gi - gj
        c2 = 0.0
        if j - i == d and d >= 0 and 0 <= i <= n - 1 - d:
            c2 += 1
        if i - j == d and d >= 0 and 0 <= j <= n - 1 - d:
            c2 += 1
        C2[r] = c2
    for p in range(1, n):
        r = n - 1 - p
        m = n - p
        if m <= 0:
            continue
        gi = 0.0
        if p <= i < n:
            gi += (1.0 - h[max(i - p, 0)])
        if 0 <= i < m:
            gi += -h[min(i + p, n - 1)] if i + p < n else 0.0
        gj = 0.0
        if p <= j < n:
            gj += (1.0 - h[max(j - p, 0)])
        if 0 <= j < m:
            gj += -h[min(j + p, n - 1)] if j + p < n else 0.0
        T[r] = gi - gj
        c2 = 0.0
        if j - i == -p and p <= i <= n - 1:
            c2 += 1
        if i - j == -p and p <= j <= n - 1:
            c2 += 1
        C2[r] = c2
    idx0 = n - 1
    c2_center = 0.0
    if i < n:
        c2_center += -1
    if j < n:
        c2_center += -1
    C2[idx0] = c2_center

    a_min = max(-h[i], h[j] - 1.0, -radius)
    a_max = min(1.0 - h[i], h[j], radius)
    if a_max - a_min < 1e-14:
        return 0.0, float(np.max(S) * dx)

    def _fval(a):
        return float(np.max(S + a * T + a * a * C2)) * dx

    cur_val = float(np.max(S) * dx)
    n_grid = 32
    grid_a = np.linspace(a_min, a_max, n_grid)
    grid_f = np.empty(n_grid)
    for k in range(n_grid):
        grid_f[k] = _fval(grid_a[k])
    best_idx = int(np.argmin(grid_f))
    if grid_f[best_idx] >= cur_val - 1e-14:
        return 0.0, cur_val

    lo_b = grid_a[max(0, best_idx - 2)]
    hi_b = grid_a[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = _fval(c1)
    f2 = _fval(c2)
    for _ in range(40):
        if hi_b - lo_b < 1e-11:
            break
        if f1 < f2:
            hi_b = c2
            c2 = c1
            f2 = f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = _fval(c1)
        else:
            lo_b = c1
            c1 = c2
            f1 = f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = _fval(c2)
    opt_a = (lo_b + hi_b) / 2.0
    opt_val = _fval(opt_a)
    candidates = [(opt_a, opt_val), (a_min, _fval(a_min)), (a_max, _fval(a_max)), (0.0, cur_val)]
    best = min(candidates, key=lambda x: x[1])
    if best[1] < cur_val - 1e-14:
        return best[0], best[1]
    return 0.0, cur_val

def _pair_descent(h, deadline, gamma=80.0, K=20, radius_init=0.3, max_iter=500000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    G = _grad_matrix(h)
    S = np.correlate(h, 1.0 - h, mode='full')

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 2000:
            break

        S_max = float(np.max(S))
        shifted = S - S_max
        w = np.exp(gamma * shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(S)] = 1.0
        else:
            w /= w_sum
        scores = w @ G
        score_diff = np.zeros((n, n))
        score_diff = scores[None, :] - scores[:, None]
        mask = np.ones((n, n), dtype=bool)
        np.fill_diagonal(mask, False)
        score_diff[~mask] = -1e9
        flat_idx = np.argsort(-score_diff.ravel())[:K * 4]
        tried = set()
        improved = False
        for fi in flat_idx:
            if time.time() > deadline:
                break
            i = int(fi // n)
            j = int(fi % n)
            if (i, j) in tried:
                continue
            tried.add((i, j))
            a, new_c5 = _pair_line_search(h, i, j, S, dx, radius)
            if a != 0.0 and new_c5 < cur_c5 - 1e-13:
                h[i] += a
                h[j] -= a
                h = np.clip(h, 0.0, 1.0)
                h = _project(h, n / 2.0)
                cur_c5 = _exact_c5(h)
                S = np.correlate(h, 1.0 - h, mode='full')
                G = _grad_matrix(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.02, 0.5)
                break
        if not improved:
            stall += 1
            if stall % 200 == 0:
                radius *= 0.8
            if stall % 500 == 0:
                radius = max(0.05, radius * 0.5)
            if radius < 0.01:
                break

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_local(h_bin, deadline, max_iter=200000, seed=123):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    T = 0.003
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T = max(T * 0.999, 0.0005)
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    # Greedy pair sweep
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    cur = best_h.copy()
    cur_c5 = best_c5
    for oi in ones:
        if time.time() > deadline:
            break
        best_oz = -1
        best_val = cur_c5
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            cv = _exact_c5(cur)
            if cv < best_val - 1e-14:
                best_val = cv
                best_oz = oz
            cur[oi] = 1.0
            cur[oz] = 0.0
        if best_oz >= 0:
            cur[oi] = 0.0
            cur[best_oz] = 1.0
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
    return best_h, best_c5

def _perturb(h, scale, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(0.5, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.5)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= scale
    perturb += scale * 0.15 * rng.standard_normal(n)
    return _project(h + perturb, n / 2.0)

def _make_periodic_candidate(n, period, seed=7):
    rng = np.random.default_rng(seed)
    base = rng.uniform(0.3, 0.7, period)
    base = (base - base.mean()) * 0.5 + 0.5
    base = np.clip(base, 0.0, 1.0)
    h = np.tile(base, n // period + 1)[:n]
    return _project(h, n / 2.0)
