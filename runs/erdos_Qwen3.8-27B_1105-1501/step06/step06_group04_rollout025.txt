class _LSEProblem:
    def __init__(self, beta, n, target):
        self.beta = beta
        self.n = n
        self.target = target
        self.dx = 2.0 / n
        self.M = 2 * n - 1  # number of correlation terms
    
    def fun(self, x):
        h = x  # already projected or within bounds
        scores = np.correlate(h, 1.0 - h, mode='full') * self.dx
        Mmax = scores.max()
        z = self.beta * (scores - Mmax)
        return (Mmax + np.log(np.exp(z).sum())) / self.beta
    
    def jac(self, x):
        h = x
        scores = np.correlate(h, 1.0 - h, mode='full') * self.dx
        Mmax = scores.max()
        z = self.beta * (scores - Mmax)
        p = np.exp(z) / np.exp(z).sum()
        G = _grad_matrix(h, np.arange(self.M))
        return self.dx * (p @ G)
    
    def cons_eq_fun(self, x):
        return np.sum(x) - self.target
    
    def cons_eq_jac(self, x):
        return np.ones(self.n)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:

def _grad_matrix(h, lag_indices):
    n = len(h)
    G = np.zeros((len(lag_indices), n))
    for idx, r in enumerate(lag_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def _lse_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    Mmax = float(scores.max())
    z = beta * (scores - Mmax)
    ez = np.exp(z)
    return float(Mmax + np.log(ez.sum()) / beta)

def _lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    Mmax = float(scores.max())
    z = beta * (scores - Mmax)
    ez = np.exp(z)
    p = ez / ez.sum()
    top_k = min(60, M)
    active = np.argsort(-p)[:top_k]
    G = _grad_matrix(h, active)
    return dx * (p[active] @ G)

class _LSEProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
        self.dx = 2.0 / n
        self.M = 2 * n - 1
    def fun(self, x):
        return _lse_objective(x, self.beta)
    def jac(self, x):
        return _lse_gradient(x, self.beta)
    def eq_fun(self, x):
        return np.sum(x) - self.n / 2.0
    def eq_jac(self, x):
        return np.ones(self.n)

def _solve_lse(h_init, beta, deadline, maxiter=200):
    n = len(h_init)
    target = n / 2.0
    prob = _LSEProb(beta, n)
    h0 = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}]
    try:
        res = minimize(prob.fun, h0, method='SLSQP', jac=prob.jac,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_new = _project(res.x, target)
            c5 = _compute_c5(h_new)
            if np.isfinite(c5):
                return h_new, c5
    except Exception:
        pass
    return h_init, _compute_c5(h_init)

    d -= d.mean()

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _binary_quantize(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    return b

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                patience += 1
            ones_list = np.where(h > 0.5)[0]
            zeros_list = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _minimax_slsqp(h_init, deadline, maxiter=100):
    n = len(h_init)
    dx = 2.0 / n
    M = 2 * n - 1
    target = n / 2.0
   
    def mm_fun(y):
        return y[0]
   
    def mm_jac(y):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
   
    def mm_ineq_fun(y):
        h = y[1:]
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        return y[0] - scores
   
    def mm_ineq_jac(y):
        h = y[1:]
        G = _grad_matrix(h, np.arange(M))
        J = np.zeros((M, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * G
        return J
   
    def mm_eq_fun(y):
        return np.sum(y[1:]) - target
   
    def mm_eq_jac(y):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
   
    h0 = h_init.copy()
    M0 = _compute_c5(h0) * 1.01 + 0.01
    y0 = np.concatenate([[M0], h0])
    bounds = [(0.0, M0 + 0.1)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': mm_ineq_fun, 'jac': mm_ineq_jac},
        {'type': 'eq', 'fun': mm_eq_fun, 'jac': mm_eq_jac}
    ]
    try:
        res = minimize(mm_fun, y0, method='SLSQP', jac=mm_jac,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_new = _project(res.x[1:], target)
            c5 = _compute_c5(h_new)
            if np.isfinite(c5) and c5 < M0:
                return h_new, c5
    except Exception:
        pass
    return h_init, _compute_c5(h_init)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s
    reserve = max(5.0, 0.02 * budget_s)
    hard_deadline = deadline - reserve

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = _sine_init(120, 3, 0.5)
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: LSE SLSQP with multiple betas (15% time)
    d1 = t0 + budget_s * 0.15
    betas = [30, 60, 120, 250, 500, 1000]
    for beta in betas:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.015)
        h_new, c5_new = _solve_lse(best_h, beta, sub_dl, maxiter=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Direct minimax SLSQP (10% time)
    d2 = t0 + budget_s * 0.25
    if time.time() < d2:
        h_mm, c5_mm = _minimax_slsqp(best_h, min(d2, time.time() + budget_s * 0.04), maxiter=80)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
        # Try again from slightly perturbed start
        h_pert = _project(best_h + rng.normal(0, 0.02, len(best_h)), len(best_h) / 2.0)
        if time.time() < d2 - 5:
            h_mm2, c5_mm2 = _minimax_slsqp(h_pert, min(d2, time.time() + budget_s * 0.03), maxiter=60)
            if c5_mm2 < best_c5:
                best_c5 = c5_mm2
                best_h = h_mm2.copy()

    # Phase 3: Line search polish (25% time)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _line_polish(best_h, d3, radius_init=0.15, max_rounds=8000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-start (15% time)
    d4 = t0 + budget_s * 0.65
    n = len(best_h)
    starts = []
    for freq in [1, 2, 3, 4, 5, 7, 11, 15]:
        for ph in [0.0, 0.7, 1.4, 2.5, 3.8]:
            starts.append(_sine_init(n, freq, ph))
    for p in [3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20]:
        for duty in [0.3, 0.4, 0.5, 0.6, 0.7]:
            starts.append(_periodic_init(n, p, duty))
    # Binary quantization of parent
    starts.append(_binary_quantize(best_h))
    # Random
    for _ in range(4):
        starts.append(_project(rng.uniform(0, 1, n), n / 2.0))
    starts.append(best_h.copy())

    for s in starts:
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.008)
        # Quick line polish from this start
        h_s, c5_s = _line_polish(s, sub_dl, radius_init=0.10, max_rounds=500)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Resolution changes (15% time)
    d5 = t0 + budget_s * 0.80
    best_n = len(best_h)
    offsets = [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, -24, 24, -30, 30, -36, 36, -44, 44]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 300:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.01)
        h_r, c5_r = _line_polish(h_r, sub_dl, radius_init=0.10, max_rounds=400)
        if c5_r < best_c5 - 1e-12:
            # Quick refine at this resolution
            h_r2, c5_r2 = _pair_refine(h_r, min(d5, time.time() + budget_s * 0.005))
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
            if c5_r < best_c5 - 1e-12 and time.time() < d5 - 10:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 6: Pair + quad refinement (10% time)
    d6 = t0 + budget_s * 0.90
    h6, c5_6 = _pair_refine(best_h, min(d6, t0 + budget_s * 0.85), max_passes=80)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6:
        h6b, c5_6b = _quad_refine(best_h, d6)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Binary SA from best (5% time)
    d7 = t0 + budget_s * 0.93
    if time.time() < d7:
        h_bin, c5_bin = _binary_sa(best_h, d7, rng, max_iters=10000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Polish the binary result
        if time.time() < d7:
            h_bp, c5_bp = _line_polish(best_h, min(d7, time.time() + budget_s * 0.01), radius_init=0.08, max_rounds=500)
            if c5_bp < best_c5:
                best_c5 = c5_bp
                best_h = h_bp.copy()

    # Phase 8: Final aggressive polish (7% time)
    h_f, c5_f = _line_polish(best_h, hard_deadline, radius_init=0.05, max_rounds=5000)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final: one more LSE polish at high beta
    if time.time() < hard_deadline - 10:
        h_final_lse, c5_final_lse = _solve_lse(best_h, 800, hard_deadline, maxiter=200)
        if c5_final_lse < best_c5:
            best_c5 = c5_final_lse
            best_h = h_final_lse.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        h_fb = _sine_init(120, 3, 0.0)
        return h_fb, _compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
