from scipy.optimize import minimize
import time

# Helper functions: _c5, _project, _grad_matrix, _lse_obj_grad, _line_search

def _lse_slsqp(h0, beta, deadline, maxiter=100):
    """Run SLSQP on LSE objective with given beta."""
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target)
    
    def fun_grad(x):
        x = np.clip(x, 0.0, 1.0)
        obj, grad = _lse_obj_grad(x, beta)
        return obj, grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(fun_grad, h, method='SLSQP', jac=True,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if result.success or result.fun < _c5(h):
        h_new = _project(result.x, target)
        if _c5(h_new) <= _c5(h):
            return h_new
    return h

def _line_refine(h, deadline, n_dirs_per_iter=10, max_iters=500):

    n = len(h)

        if np.linalg.norm(grad) > 1e-10:

    for j in range(2*n-1):
        w = p[j]
        if w < 1e-12:

            m = n - g2

            grad[g2:] += w * (-h[:m])
    grad *= dx
    return grad, corr, p

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    v = np.clip(v, -5.0, 5.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = float(corr.max())
    z = beta * (corr - max_c)
    exp_z = np.exp(z)
    s = float(exp_z.sum())
    p = exp_z / s
    obj = (max_c + np.log(s)) / beta
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-14:
            continue
        d = j - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[d:] += w * (1.0 - h[:m])
            grad[:m] += w * (-h[d:])
        else:
            g2 = -d
            m = n - g2
            grad[:m] += w * (1.0 - h[g2:])
            grad[g2:] += w * (-h[:m])
    grad *= dx
    return obj, grad, corr, p

def _line_search(h, d, dx, max_step=0.5):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = 0.0
    t_hi = max_step
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi <= t_lo + 1e-14:
        return 0.0, _c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(100):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _project_direction(h, d):
    d = d - d.mean()
    n = len(h)
    max_abs = float(np.max(np.abs(d)))
    if max_abs < 1e-15:
        return None
    return d / max_abs

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _random_init(n, rng):
    h = rng.random(n)
    return _project(h, n / 2.0)

def _quantize_binary(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[sorted_idx[:target]] = 1.0
    return hb

def _slsqp_lse(h0, beta, maxiter=80):
    """SLSQP on LSE objective using a top-level function approach."""
    n = len(h0)
    target = n / 2.0
    global _tmp_beta, _tmp_n, _tmp_target
    _tmp_beta = beta
    _tmp_n = n
    _tmp_target = target

    def _slsqp_fun(x):
        hh = np.clip(x, 0.0, 1.0)
        obj, grad, _, _ = _lse_grad(hh, _tmp_beta)
        return obj, grad

    eq_fun_vals = target
    def _eq_fun(x):
        return float(np.sum(x)) - _tmp_target
    def _eq_jac(x):
        return np.ones(_tmp_n)

    res = minimize(_slsqp_fun, h0, method='SLSQP', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None:
        h_new = _project(res.x, target)
        if _c5(h_new) <= _c5(h0):
            return h_new
    return h0

    if time.time() < d5:
        h_r, c5_r = _lse_descent(best_h, d5, beta_list=[100, 400, 1500, 5000], max_iters=1500)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Resolution changes (13%)
    d6 = t0 + budget_s * 0.80
    base_n = len(best_h)
    offsets = [-4, -2, 2, 4, -6, 6, -8, 8, -10, 10, 12, -12, 14, -14, 16, -16, 20, -20]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 300 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.02)
        h_rm, c5_rm = _lse_descent(h_r, sub_dl, beta_list=[30, 100, 400, 1500], max_iters=1000)
        if c5_rm < best_c5 - 1e-12 and time.time() < d6 - 3:
            sub_dl2 = min(d6, time.time() + budget_s * 0.008)
            h_rm2, c5_rm2 = _pair_quad_refine(h_rm, sub_dl2, max_rounds=200)
            if c5_rm2 < c5_rm:
                h_rm, c5_rm = h_rm2, c5_rm2
        if c5_rm < best_c5 - 1e-12:
            best_c5 = c5_rm
            best_h = h_rm.copy()

    # Phase 7: Intensive line search refinement (10%)
    d7 = t0 + budget_s * 0.90
    h7, c5_7 = _pair_quad_refine(best_h, d7, max_rounds=600)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7:
        h7b, c5_7b = _active_gradient_descent(best_h, d7, max_iters=2000)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Phase 8: Perturbation + re-optimize (5%)
    d8 = t0 + budget_s * 0.95
    scale = 0.02
    for _ in range(200):
        if time.time() > d8:
            break
        h_p = best_h + rng.normal(0, scale, len(best_h))
        h_p = _project(h_p, len(best_h) / 2.0)
        sub_dl = min(d8, time.time() + budget_s * 0.006)
        h_t, c5_t = _lse_descent(h_p, sub_dl, beta_list=[100, 500, 2000], max_iters=500)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.7)
        else:
            scale = min(0.10, scale * 1.3)

    # Phase 9: Final polish (5%)
    h_f, c5_f = _pair_quad_refine(best_h, deadline, max_rounds=400)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            hf = np.array(initial_h_values, dtype=float)
            hf = _project(hf, len(hf) / 2.0)
            return hf, _c5(hf), len(hf)
        except NameError:
            hf = _sine_init(122, 3, 0.0)
            return hf, _c5(hf), len(hf)
    return best_h, final_c5, len(best_h)
