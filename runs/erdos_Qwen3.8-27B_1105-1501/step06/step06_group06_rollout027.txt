            if j >= g:
                B[k] += (1.0 - h[j-g])
            # h[j] appears as second factor if j < m: contributes -h[j+g]
            if j < m:
                B[k] -= h[j+g]
    B *= dx
    
    # Now c_k + (t - h[j]) * B_k for each k
    # Minimize max_k (c_k + (t - h[j]) * B_k) over t in [max(0, h[j]-something), min(1, h[j]+something)]
    # Actually t is h[j]_new, and (t - h[j]) is the change
    # c_k_new = c_k + (t - h[j]) * B_k
    # We want to minimize max_k c_k_new over t in [0, 1]
    
    # c_k + t*B_k - h[j]*B_k = (c_k - h[j]*B_k) + t*B_k
    # Let a_k = c_k - h[j]*B_k, b_k = B_k
    # Minimize max_k (a_k + t*b_k) over t in [0, 1]
    
    a = c - h[j] * B
    # Find breakpoints
    # For each pair (i,j): a_i + t*b_i = a_j + t*b_j => t = (a_j - a_i)/(b_i - b_j)
    # Collect all breakpoints in [0, 1]
    
    # Efficient approach: find the t that minimizes the upper envelope
    # This is a linear programming problem in 1D: min t such that a_k + t*b_k <= T for all k
    # Equivalent to: T >= max_k (a_k + t*b_k), minimize T
    # The minimum T is achieved at a breakpoint or boundary
    
    # Collect candidate t values
    candidates = [0.0, 1.0]
    for i in range(len(a)):
        for j2 in range(i+1, len(a)):
            if abs(B[i] - B[j2]) > 1e-14:
                t = (a[j2] - a[i]) / (B[i] - B[j2])
                if -1e-10 < t < 1 + 1e-10:
                    candidates.append(max(0.0, min(1.0, t)))
    
    # Evaluate at all candidates
    best_t = 0.0
    best_val = float('inf')
    for t in candidates:
        val = float(np.max(a + t * B))
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

def _min_upper_envelope(a, b, lo=0.0, hi=1.0, iterations=50):
    """Minimize max_k (a[k] + t*b[k]) for t in [lo, hi]."""
    for _ in range(iterations):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(a + m1 * b)
        f2 = np.max(a + m2 * b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2
    return t, float(np.max(a + t * b))

    alpha_hi = 1.0

    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max(h[neg] / d[neg])))
    alpha_lo = max(alpha_lo, -1.0)
    alpha_hi = min(alpha_hi, 1.0)
    
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0
