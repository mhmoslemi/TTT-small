import numpy as np
import time
import cvxpy as cp

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def row_gradients(h, row_indices):
    n = len(h)
    m = len(row_indices)
    G = np.zeros((m, n))
    for k, row in enumerate(row_indices):
        s = row - (n - 1)
        mlen = n - abs(s)
        if mlen <= 0:
            continue
        if s >= 0:
            G[k, s:s + mlen] += (1.0 - h[:mlen])
            G[k, :mlen] += (-h[s:s + mlen])
        else:
            r = -s
            G[k, :mlen] += (1.0 - h[r:r + mlen])
            G[k, r:r + mlen] += (-h[:mlen])
    return G

def lp_find_direction(h, n_active=8):
    n = len(h)
    R = np.correlate(h, 1.0 - h, 'full')
    top_idx = np.argsort(-R)[:n_active]
    G = row_gradients(h, top_idx)
    a = cp.Variable(n, nonneg=True)
    b = cp.Variable(n, nonneg=True)
    d = a - b
    t = cp.Variable()
    constraints = [cp.sum(a + b) <= 1.0, cp.sum(d) == 0]
    for k in range(n_active):
        constraints.append(G[k] @ d <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None, 0.0
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, 0.0
    d_val = np.asarray(d.value).flatten()
    norm = float(np.sum(np.abs(d_val)))
    if norm < 1e-14:
        return None, 0.0
    t_val = float(t.value) if t.value is not None else 0.0
    return d_val / norm, t_val

def exact_line_search(h, d, S0, dx, max_step=0.5):
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    lo, hi = 0.0, max_step
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    lo = max(lo, -max_step)
    hi = min(hi, max_step)
    if hi - lo < 1e-12:
        return 0.0, c5_exact(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a2, b2 = lo, hi
    c1 = b2 - gr * (b2 - a2)
    c2 = a2 + gr * (b2 - a2)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):

            break
        if f1 < f2:
            b2, c2, f2 = c2, c1, f1
            c1 = b2 - gr * (b2 - a2)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a2, c1, f1 = c1, c2, f2
            c2 = a2 + gr * (b2 - a2)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a2 + b2) / 2.0

    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx

        f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
        if f_lo < val:
            opt, val = lo, f_lo
        if f_hi < val:
            opt, val = hi, f_hi

        if opt > 1e-10 and val < cur_c5 - 1e-14:
            h = h + opt * d
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def sa_pair_phase(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_sa = 0.002
    T_min = 0.00005
    patience = 0

    while time.time() < deadline:
        if patience > 8000:
            if T_sa <= T_min * 1.01:
                break
            T_sa = T_min
            patience = 0
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        alpha, new_c5 = exact_line_search(h, d, S0, dx, max_step=0.4)
        if abs(alpha) > 1e-10:
            delta = new_c5 - cur_c5
            accept = delta < 0
            if not accept and T_sa > T_min:
                accept = rng.random() < np.exp(-delta / (T_sa + 1e-15))
            if accept:
                h[i] -= alpha
                h[j] += alpha
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
        T_sa = max(T_sa * 0.99997, T_min)

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.42 * np.sin(2.0 * np.pi * 2.0 * x / n0 + 0.5)
        h0 = project(h0, n0 / 2.0)

    h0 = project(h0, len(h0) / 2.0)
    best_h = h0.copy()
    best_c5 = c5_exact(h0)
    best_n = len(h0)

    # Phase 1: Directed pair search (30%)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = directed_pair_phase(h0, d1, rng, n_pairs=25, use_both_dirs=False)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: LP-directed search (20%)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = lp_search_phase(best_h, d2, rng, n_active=8)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2

    # Phase 3: SA pairs (15%)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = sa_pair_phase(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.85
    n_base = len(best_h)
    tried_n = {n_base}
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8]:
        if time.time() > d4:
            break
        n_try = n_base + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample(best_h, n_try)
        sub_dl1 = min(d4, time.time() + budget_s * 0.025)
        h_r, c5_r = directed_pair_phase(h_r, sub_dl1, rng, n_pairs=20)
        if c5_r < best_c5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.015)
            h_r, c5_r = gradient_polish(h_r, sub_dl2)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 5: Final directed polish (15%)
    d5 = deadline
    h5, c5_5 = directed_pair_phase(best_h, d5, rng, n_pairs=30, use_both_dirs=True)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5

    # Final gradient polish with any remaining time
    if time.time() < deadline - 1.0:
        h6, c5_6 = gradient_polish(best_h, deadline)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
