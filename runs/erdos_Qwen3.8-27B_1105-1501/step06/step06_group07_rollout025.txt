def _binary_sa(h_init, deadline, rng, max_iters=50000):
    """Binary simulated annealing."""
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.008
    T_min = 3e-6
    cooling = 0.99995
    ones_set = list(np.where(h > 0.5)[0])
    zeros_set = list(np.where(h <= 0.5)[0])
    
    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones_set[rng.integers(len(ones_set))]
        j = zeros_set[rng.integers(len(zeros_set))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_set.remove(i)
            ones_set.append(j)
            zeros_set.remove(j)
            zeros_set.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            step *= 0.7
            if stall % 10 == 0:
                alpha *= 1.5
            if step < 0.005:
                break
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng, max_iters=100000):
    n = len(h_init)
    if n % 2 != 0:
        return h_init, _exact_c5(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.006
    T_min = 2e-6
    cooling = 0.99996
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])

    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _multi_start_descent(h, deadline, rng, n_starts=5, iters_each=80):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    per_time = max(2.0, (deadline - time.time()) / n_starts)
    for s in range(n_starts):
        sub_dl = time.time() + per_time
        scale = 0.08 + 0.04 * s
        h_try = h + rng.normal(0, scale, n)
        h_try = _project(h_try, target)
        h_try, c5_try = _active_descent(h_try, sub_dl, K=20, max_iters=iters_each)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def _parametric_init(n, rng, n_modes=5):
    x = (np.arange(n) + 0.5) / n
    h = np.full(n, 0.5)
    for j in range(1, n_modes + 1):
        amp = rng.uniform(0.05, 0.25) / np.sqrt(j)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * j * x + phase)
    return _project(h, n / 2.0)

def _quad_line_search(h, d, target, max_t=0.5):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C_arr = np.correlate(d, d, mode='full')
    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T_arr[None, :] - (grid[:, None] ** 2) * C_arr[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
    f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_arr - opt_t * opt_t * C_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _precise_descent(h, deadline, K=25, max_iters=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        M = top_vals[0]
        alpha = 1.0 / (0.001 * n)
        w = np.exp(alpha * (top_vals - M))
        w /= np.sum(w)
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            if k >= 0:
                m2 = n - k
                grad[k:] += dx * wi * (1.0 - h[:m2])
                grad[:m2] -= dx * wi * h[k:]
            else:
                q = -k
                m2 = n - q
                grad[:m2] += dx * wi * (1.0 - h[q:])
                grad[q:] -= dx * wi * h[:m2]
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        direction = -grad / gn
        t, f_try = _quad_line_search(h, direction, target, max_t=step)
        if t < 1e-12:
            step *= 0.7
            continue
        h_new = _project(h + t * direction, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.7 * step:
                step = min(step * 1.05, 2.0)
            else:
                step = max(step * 0.85, 0.01)
        else:
            step *= 0.7
            if step < 0.005:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _parametric_init(n0, rng, n_modes=6)

    h0 = _project(h0.copy(), n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Smooth log-sum-exp descent with momentum (18%)
    d1 = t0 + budget_s * 0.18
    h1, c5_1 = _smooth_descent(best_h, d1, eps_init=0.035, eps_min=0.002,
                               eps_decay=0.8, max_iters=500, momentum=0.25)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Precise active descent (12%)
    d2 = t0 + budget_s * 0.30
    h2, c5_2 = _precise_descent(best_h, d2, K=30, max_iters=200)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-start perturbation + active descent (15%)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _multi_start_descent(best_h, d3, rng, n_starts=6, iters_each=60)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Smooth descent again with different params (12%)
    d4 = t0 + budget_s * 0.57
    h4, c5_4 = _smooth_descent(best_h, d4, eps_init=0.015, eps_min=0.002,
                               eps_decay=0.85, max_iters=400, momentum=0.15)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    h4b, c5_4b = _precise_descent(best_h, d4, K=25, max_iters=100)
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()

    # Phase 5: Binary SA (20%)
    d5 = t0 + budget_s * 0.77
    h5, c5_5 = _binary_sa(best_h, d5, rng, max_iters=120000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    # Polish binary
    if time.time() < d5:
        h5b, c5_5b = _precise_descent(best_h, d5, K=20, max_iters=80)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution (15%)
    d6 = t0 + budget_s * 0.92
    best_n = len(best_h)
    offsets = list(range(-16, 17, 2))
    tried_n = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 256 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _precise_descent(h_r, sub_dl, K=20, max_iters=40)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Final refinement (8%)
    d7 = deadline - 1.0
    h7, c5_7 = _precise_descent(best_h, d7, K=18, max_iters=100)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h8, c5_8 = _smooth_descent(best_h, deadline - 0.5, eps_init=0.008,
                               eps_min=0.001, eps_decay=0.85, max_iters=100, momentum=0.1)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
