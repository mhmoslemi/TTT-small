import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    # Water-filling projection to sum = target, values in [0,1]
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 1, float(v.max()) + 1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    lst = 1.0 - h
    for d in range(-(n-1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                # G[r, i] = (1-h[i-d])*[i>=d] - h[i+d]*[i<=n-1-d]
                # First term: i from d to n-1, G[r,i] += (1-h[i-d])
                G[r, d:] += lst[:m]  # 1-h[0:m] added to positions d:d+m
                # Second term: i from 0 to n-1-d, G[r,i] -= h[i+d]
                G[r, :m] -= h[d:]  # -h[d:n] subtracted from positions 0:m
        else:
            g = -d
            m = n - g
            if m > 0:
                # G[r, i] = (1-h[i+g])*[i<=n-1-g] - h[i-g]*[i>=g]
                # First term: i from 0 to n-1-g, G[r,i] += (1-h[i+g])
                G[r, :m] += lst[g:]  # 1-h[g:n] added to positions 0:m
                # Second term: i from g to n-1, G[r,i] -= h[i-g]
                G[r, g:] -= h[:m]  # -h[0:m] subtracted from positions g:n
    return G

def _line_search_pair(h, corr, G, i, j, radius, dx):
    n = len(h)
    T = G[:, i] - G[:, j]
    # C[k] = 1 if |k-(n-1)| == |i-j|
    delta = abs(i - j)
    C = np.zeros(len(corr))
    if delta < len(corr):
        C[n-1-delta] = 1.0
        C[n-1+delta] = 1.0
    
    # Feasible alpha range
    lo_a = max(-h[i], h[j] - 1.0, -radius)
    hi_a = min(1.0 - h[i], h[j], radius)
    if hi_a - lo_a < 1e-14:
        return 0.0, float(np.max(corr)) * dx
    
    # Grid search
    n_grid = 40
    grid = np.linspace(lo_a, hi_a, n_grid)
    best_f = np.inf
    best_a = 0.0
    for a in grid:
        vals = corr + a * T - a * a * C
        f = float(np.max(vals)) * dx
        if f < best_f:
            best_f = f
            best_a = a
    
    # Golden section refinement around best
    idx = int(np.argmin([float(np.max(corr + a*T - a*a*C)) for a in grid]))
    lo_b = grid[max(0, idx-2)]
    hi_b = grid[min(n_grid-1, idx+2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = float(np.max(corr + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(corr + c2*T - c2*c2*C)) * dx
    for _ in range(30):
        if hi_b - lo_b < 1e-12: break
        if f1 < f2:
            hi_b, c2, f2 = c2, c1, f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = float(np.max(corr + c1*T - c1*c1*C)) * dx
        else:
            lo_b, c1, f1 = c1, c2, f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = float(np.max(corr + c2*T - c2*c2*C)) * dx
    opt_a = (lo_b + hi_b) / 2.0
    opt_f = float(np.max(corr + opt_a*T - opt_a*opt_a*C)) * dx
    
    # Compare with grid best and endpoints
    candidates = [(best_a, best_f), (lo_a, float(np.max(corr + lo_a*T - lo_a*lo_a*C))*dx),
                  (hi_a, float(np.max(corr + hi_a*T - hi_a*hi_a*C))*dx),
                  (opt_a, opt_f)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def _pair_descent(h, deadline, radius=0.15, max_iter=5000, beta=20.0, n_pairs=30):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rnd = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        scores = corr * dx
        
        # Soft-max weights
        M = float(np.max(scores))
        exps = np.exp(np.clip(beta * (scores - M), -60, 0))
        w = exps / float(np.sum(exps))
        
        # Gradient
        G = _build_G(h)
        grad = G.T @ w
        # Remove mean (projection onto sum=0 subspace)
        grad -= np.mean(grad)
        
        # Select candidate pairs: high grad -> low grad
        order = np.argsort(grad)
        # Top sources (high grad) and sinks (low grad)
        sources = order[n - n_pairs//2:]  # top n_pairs/2
        sinks = order[:n_pairs//2]  # bottom n_pairs/2
        
        best_alpha = 0.0
        best_c5_try = cur_c5
        best_pair = None
        
        for si in range(len(sources)):
            for sk in range(len(sinks)):
                i = int(sources[si])
                j = int(sinks[sk])
                if i == j: continue
                alpha, c5_try = _line_search_pair(h, corr, G, i, j, radius, dx)
                if c5_try < best_c5_try - 1e-14:
                    best_c5_try = c5_try
                    best_alpha = alpha
                    best_pair = (i, j)
        
        # Also try some random pairs
        for _ in range(5):
            i = int(rnd.integers(0, n))
            j = int(rnd.integers(0, n))
            if i == j: continue
            alpha, c5_try = _line_search_pair(h, corr, G, i, j, radius, dx)
            if c5_try < best_c5_try - 1e-14:
                best_c5_try = c5_try
                best_alpha = alpha
                best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] += best_alpha
            h[j] -= best_alpha
            h = _project(h, target)
            new_c5 = _exact_c5(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
            cur_c5 = new_c5
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.85
            if radius < 0.01:
                radius = 0.1
    
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * 2.0/n

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = v.min()-1, v.max()+1
    for _ in range(60):
        mid = (lo+hi)/2
        if np.sum(np.clip(v-mid, 0, 1)) > target: lo = mid
        else: hi = mid
    x = np.clip(v-(lo+hi)/2, 0, 1)
    r = target - x.sum()
    if abs(r) > 1e-12:
        m = x<1 if r>0 else x>0
        c = m.sum()
        if c: x[m] += r/c; x = np.clip(x,0,1)
    return x

def _grad_matrix(h):
    n = len(h); L = 2*n-1
    G = np.zeros((L, n)); lst = 1.0-h
    for d in range(-(n-1), n):
        r = n-1+d
        if d >= 0:
            m = n-d
            G[r, d:] += lst[:m]
            G[r, :m] -= h[d:]
        else:
            g = -d; m = n-g
            G[r, :m] += lst[g:]
            G[r, g:] -= h[:m]
    return G
