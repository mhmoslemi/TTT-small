import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * dx)
    return val if np.isfinite(val) else 1.0

def _scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.clip(v, 0.0, 1.0)
    # Bisection on threshold
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-14:
        if res > 0:
            mask = (x > 0) & (x < 1)
        else:
            mask = (x > 0) & (x < 1)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
        else:
            # All at boundaries; distribute to non-boundary closest
            if res > 0:
                mask2 = x < 1
            else:
                mask2 = x > 0
            cnt2 = mask2.sum()
            if cnt2 > 0:
                x[mask2] += res / cnt2
    return np.clip(x, 0.0, 1.0)

def _line_search(h, A, t_lo, t_hi, n_grid=40, n_golden=25):
    """
    Minimize max_r (S0[r] + t*T[r] + t^2*Q[r]) for t in [t_lo, t_hi].
    A must satisfy sum(A)=0 and h+t*A in [0,1]^n for t in [t_lo, t_hi].
    Returns (t_opt, objective_at_t_opt).
    """
    n = len(h)
    dx = 2.0 / n
    g0 = 1.0 - h
    
    # Precompute T and Q
    T = np.correlate(A, g0, mode='full') - np.correlate(h, A, mode='full')
    Q = -np.correlate(A, A, mode='full')
    S0 = np.correlate(h, g0, mode='full')
    
    def obj(t):
        return float(np.max(S0 + t * T + t * t * Q) * dx)
    
    # Grid search
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([obj(t) for t in ts])
    idx = np.argmin(vals)
    
    # Refine with golden section
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a = ts[max(0, idx - 1)]
    b = ts[min(n_grid - 1, idx + 1)]
    c = b - g * (b - a)
    d = a + g * (b - a)
    fc = obj(c)
    fd = obj(d)
    for _ in range(n_golden):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - g * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + g * (b - a)
            fd = obj(d)
    t_opt = (a + b) / 2.0
    return t_opt, obj(t_opt)

def _pair_t_bounds(h, i, j):
    """For A[i]=1, A[j]=-1, find t range where h+t*A in [0,1]."""
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi:
        return None
    return t_lo, t_hi

def _active_grad_dir(h, scores, top_k=8):
    """
    Compute a zero-sum direction: negative average gradient of top active constraints.
    """
    n = len(h)
    m = 2 * n - 1
    max_s = scores.max()
    # Select top active
    idx = np.argsort(-scores)[:top_k]
    
    # Build gradient rows
    G = np.zeros(top_k * n)
    for a, r in enumerate(idx):
        d = r - (n - 1)
        row = np.zeros(n)
        if d >= 0:
            m_ = n - d
            row[d:] += (1.0 - h[:m_])
            row[:m_] += -h[d:]
        else:
            g = -d
            m_ = n - g
            row[:m_] += (1.0 - h[g:])
            row[g:] += -h[:m_]
        G[a * n:(a + 1) * n] = row
    
    # Average and project to sum=0
    avg = G.reshape(top_k, n).mean(axis=0)
    avg -= avg.mean()
    norm = np.linalg.norm(avg)
    if norm < 1e-14:
        return None
    return -avg / norm  # Negative gradient direction

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_start, deadline, rng, max_iters=20000):
    """Binary simulated annealing for even n."""
    n = len(h_start)
    if n % 2 != 0:
        return None, 1.0
    target = n // 2
    dx = 2.0 / n

    h = np.zeros(n)

    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    T = 0.003
    T_min = 1e-5
    cooling = 0.9998
    patience = 0
    max_patience = max(500, max_iters // 10)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if patience > max_patience:
            break
        if not ones or not zeros:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur
        
        if delta < -1e-15 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur = new_c5
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            # Update lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c5

def _sine_init(n, freq, phase):
