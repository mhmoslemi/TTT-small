        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n]
                d -= d.mean()
                dn = np.linalg.norm(d)
                if dn < 1e-13:
                    break
                d = d / dn * 0.15
                t, new_c5 = _line_search_exact(h, d, 0.15, n_grid=25)
                if new_c5 < best_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    actual = _c5(h)
                    if actual < best_c5:
                        best_c5 = actual
                        best_h = h.copy()
                else:
                    break
            else:
                break
        except Exception:
            break
    return best_h, best_c5

def _slsqp_polish(h0, deadline, max_outer=10):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _corr(h)
        K = min(25, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        t_init = cur_c5 + 1e-5
        x0 = np.concatenate([h, [t_init]])

        def obj_fn(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x):
            hx = x[:n]
            c = _corr(hx)
            return x[n] - c[top_k]

        def ineq_jac(x):
            hx = x[:n]
            G = _grad_batch(hx, top_k)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G
            J[:, n] = 1.0
            return J

        def eq_fn(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-13})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _subset_descent(h0, deadline, max_iter=500):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    rng = np.random.default_rng(12345)
    stall = 0
    order = np.arange(n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            best_c5 = _c5(h)
            best_h = h.copy()
            stall = 0
            continue
        j = order[it % n]
        c_full = _corr(h)
        M = c_full.max()
        # Find which lags are affected by h[j]
        # h[j] appears in c_k for k such that j is in range
        # In full correlation, k corresponds to shift s = k-(n-1)
        # h[j] appears as first factor (h[n+k]) when j = n+k, i.e., k = j, valid if j <= n-1
        # h[j] appears as second factor (1-h[n]) when j = n, i.e., always
        # Actually for np.correlate(h, 1-h, 'full'):
        # result[k] = sum_n h[n + k] * (1-h[n]) for valid n
        # h[j] in first factor: n+k = j, so n = j-k, valid if 0 <= j-k <= n-1
        # h[j] in second factor: n = j, so contribution is h[j+k]*(1-h[j])... wait
        # Let me just recompute c after changing h[j]
        affected = set()
        for k in range(2 * n - 1):
            s = k - (n - 1)
            if s >= 0:
                # c_k = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
                # h[j] in first factor if j >= s and j-s <= n-1-s, i.e., j >= s
                # h[j] in second factor if j <= n-1-s
                if j >= s or j <= n - 1 - s:
                    affected.add(k)
            else:
                s2 = -s
                # c_k = sum_{i=0}^{n-1-s2} h[i]*(1-h[i+s2])
                # h[j] in first factor if j <= n-1-s2
                # h[j] in second factor if j >= s2
                if j <= n - 1 - s2 or j >= s2:
                    affected.add(k)
        # For 1D optimization: c_k(h) as function of h[j]
        # c_k = h[j]*(1-h[j+k']) + h[j+k']*(1-h[j]) + rest
        #     = h[j] - h[j]*h[j+k'] + h[j+k'] - h[j+k']*h[j] + rest
        #     = (1-2*h[j+k'])*h[j] + h[j+k'] + rest
        # So it's LINEAR in h[j]! The max of linear functions is convex PL.
        # Minimize subject to sum constraint: we need to pair j with another coordinate.
        # Instead, just try h[j] = 0 and h[j] = 1 and the current value.
        # Better: try to find the optimal h[j] given the sum constraint.
        # Since the sum must be maintained, changing h[j] requires changing another h[l].
        # For simplicity, just try small moves.
        delta = 0.05 * np.sign(rng.random() - 0.5)
        h_new = h.copy()
        h_new[j] = np.clip(h_new[j] + delta, 0.0, 1.0)
        # Maintain sum: adjust a random other coordinate
        l = int(rng.integers(n))
        if l == j:
            l = (j + 1) % n
        adj = h[j] - h_new[j]
        h_new[l] = np.clip(h_new[l] + adj, 0.0, 1.0)
        # Re-project to fix any residual
        h_new = _project(h_new, target)
        c5_new = _c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 2.0 * x / n0) + 0.2 * np.sin(2 * np.pi * 5.0 * x / n0)
    
    h0 = np.clip(h0, 0.0, 1.0)
    h_init = _project(h0, n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _c5(h_init)
    best_n = n0

    # Phase 1: Smoothed gradient descent with continuation (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _smoothed_optimize(h_init, d1, beta_start=15.0, beta_end=600.0, max_iter=200)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Active QP polish (10% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _active_qp_polish(best_h, d2, max_iter=40)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: SLSQP polish (10% time)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _slsqp_polish(best_h, d3, max_outer=8)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Smoothed descent again with higher beta (10% time)
    d4 = t0 + budget_s * 0.60
    h4, c5_4 = _smoothed_optimize(best_h, d4, beta_start=200.0, beta_end=2000.0, max_iter=150)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        best_n = len(h4)

    # Phase 5: Multi-resolution search (20% time)
    d5 = t0 + budget_s * 0.80
    resolutions = []
    for dn in [-10, -8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16, 20, 24, 28]:
        nr = best_n + dn
        if 60 <= nr <= 220 and nr % 2 == 0:
            resolutions.append(nr)
    resolutions = sorted(set(resolutions))
    tried = set()
    for nr in resolutions:
        if time.time() > d5:
            break
        if nr in tried:
            continue
        tried.add(nr)
        h_r = _resample(best_h, nr)
        sub_dl = min(d5, time.time() + budget_s * 0.025)
        h_r, c5_r = _smoothed_optimize(h_r, sub_dl, beta_start=30.0, beta_end=500.0, max_iter=60)
        if c5_r < best_c5 - 1e-12:
            # Polish the winner
            sub_dl2 = min(d5, time.time() + budget_s * 0.02)
            h_r, c5_r2 = _active_qp_polish(h_r, sub_dl2, max_iter=20)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r.copy()
                best_n = nr

    # Phase 6: Final polishing (10% time)
    d6 = t0 + budget_s * 0.90
    h6, c5_6 = _slsqp_polish(best_h, d6, max_outer=10)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)

    # Phase 7: Active QP + smoothed at high beta
    d7 = t0 + budget_s * 0.97
    h7, c5_7 = _active_qp_polish(best_h, min(d7, time.time() + budget_s * 0.03), max_iter=30)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
        best_n = len(h7)
    if time.time() < deadline - 5:
        h7b, c5_7b = _smoothed_optimize(best_h, deadline - 2, beta_start=500.0, beta_end=5000.0, max_iter=100)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()
            best_n = len(h7b)

    # Final: ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
