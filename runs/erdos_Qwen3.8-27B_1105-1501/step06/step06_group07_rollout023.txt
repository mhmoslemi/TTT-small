def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _thue_morse(n):

def _quadratic_residue(n, p):
    qr = np.zeros(n)
    x = np.arange(n)
    for i in range(n):
        if i % p < p // 2:
            r = (i % p) ** 2 % p
            qr[i] = 1.0 if r == (x[i % p]) ** 2 % p else 0.0
    return qr

def _binary_greedy_ls(h, deadline, max_passes=200):
    n = len(h)
    target = n // 2
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    cur_max = float(c.max())
    h = h.copy()
    improvements = 0

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        best_gain = 0.0
        best_i = -1
        best_j = -1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        for i in ones:
            for j in zeros:
                dc = np.zeros(2 * n - 1)
                # Type 1: h[i] as first factor, lag k = i-n for valid n
                n_lo = max(0, i - (n - 1))
                n_hi = min(n - 1, i)
                for nn in range(n_lo, n_hi + 1):
                    k = i - nn
                    dc[k + n - 1] -= (1.0 - h[nn])
                # Type 2: h[i] as second factor, n=i, lag k where i+k valid
                k_lo = max(0, -i)
                k_hi = min(n - 1, n - 1 - i)
                for k in range(k_lo, k_hi + 1):
                    dc[k + n - 1] += h[i + k]
                # Type 3: h[j] as first factor, lag k = j-n for valid n
                n_lo = max(0, j - (n - 1))
                n_hi = min(n - 1, j)
                for nn in range(n_lo, n_hi + 1):
                    k = j - nn
                    dc[k + n - 1] += (1.0 - h[nn])
                # Type 4: h[j] as second factor, n=j, lag k where j+k valid
                k_lo = max(0, -j)
                k_hi = min(n - 1, n - 1 - j)
                for k in range(k_lo, k_hi + 1):
                    dc[k + n - 1] -= h[j + k]

                new_max = float(np.max(c + dc))
                gain = cur_max - new_max
                if gain > best_gain + 1e-15:

                    best_i = i
                    best_j = j

        if best_i < 0:

        h[best_i] = 0.0
        h[best_j] = 1.0
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improvements += 1
        if time.time() > deadline:
            break

    return h, cur_max * (2.0 / n)

def _cont_polish(h_init, deadline, max_iters=300):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = 50.0
    step = 0.3

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        # Active set with softmax weights
        alpha_eff = alpha
        w = np.exp(alpha_eff * (corr - M))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum

        # Gradient of weighted max
        grad = np.zeros(n)
        for k_global in range(2 * n - 1):
            wi = w[k_global]
            if wi < 1e-10:
                continue
            k = k_global - (n - 1)
            if k >= 0:
                m = n - k
                if m <= 0:
                    continue
                grad[k:] += wi * (1.0 - h[:m])
                grad[:m] -= wi * h[k:]
            else:
                ga = -k
                m = n - ga
                if m <= 0:
                    continue
                grad[:m] += wi * (1.0 - h[ga:])
                grad[ga:] -= wi * h[:m]
        grad *= dx

        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        direction = -grad / gn
        # Line search with projection
        best_t = 0.0
        best_f = best_c5
        t = step
        for _ls in range(20):
            h_try = h + t * direction
            h_try = _project(h_try, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
            if f_try <= M + 1e-4 * t * (-gn):
                break
            t *= 0.5
            if t < 1e-10:
                break

        if best_t > 1e-12 and best_f < best_c5 - 1e-14:
            h = h + best_t * direction
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            if best_t > 0.5 * step:
                step = min(step * 1.1, 0.8)
            else:
                step = max(step * 0.7, 0.01)
        else:
            step *= 0.7
            alpha *= 1.3
            if alpha > 500:
                break

    return best_h, best_c5

                return g

            constraints.append({'type': 'ineq', 'fun': _cfun, 'jac': _cjac})

        bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.05)]
        try:
            res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.x is not None:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                else:
                    break
        except Exception:
            break

    return best_h, best_c5


def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _gen_diverse_seeds(n, rng, parent_h=None):
    seeds = []
    target = n // 2

    # 1. Parent thresholded
    if parent_h is not None and len(parent_h) == n:
        h_bin = _make_binary(n, parent_h)
        seeds.append(h_bin)
    elif parent_h is not None:
        h_r = _resample_h(parent_h, n)
        h_bin = _make_binary(n, h_r)
        seeds.append(h_bin)

    # 2. Thue-Morse
    tm = _thue_morse(n)
    # Adjust to have exactly target ones
    n_ones = int(tm.sum())
    if n_ones > target:
        excess = n_ones - target
        one_idx = np.where(tm == 1.0)[0]
        to_remove = rng.choice(one_idx, size=excess, replace=False)
        tm[to_remove] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zero_idx = np.where(tm == 0.0)[0]
        to_add = rng.choice(zero_idx, size=deficit, replace=False)
        tm[to_add] = 1.0
    seeds.append(tm.copy())

    # 3. Reversed Thue-Morse
    seeds.append(tm[::-1].copy())

    # 4. Random binary
    for _ in range(5):
        h_r = np.zeros(n)
        perm = rng.permutation(n)
        h_r[perm[:target]] = 1.0
        seeds.append(h_r)

    # 5. Block-based patterns
    # Alternating blocks of size 2
    h_blk = np.zeros(n)
    for i in range(0, n, 4):
        h_blk[i:i+2] = 1.0
    h_blk = _make_binary(n, h_blk)
    seeds.append(h_blk)

    # 6. Quadratic residue-like
    p = 29
    qr = np.zeros(n)
    for i in range(n):
        r = i % p
        qr[i] = 1.0 if (r * r) % p < p // 2 else 0.0
    qr_adj = _make_binary(n, qr)
    seeds.append(qr_adj)

    return seeds


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * x / n0)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Binary ILS at parent resolution (50% time)
    d1 = t0 + budget_s * 0.45
    seeds = _gen_diverse_seeds(n0, rng, h0)
    for idx_s, seed_h in enumerate(seeds):
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.05)
        h_try, c5_try = _binary_ilS(seed_h, sub_dl, max_rounds=100, perturb_size=3)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 2: Continuous polishing from best binary (20% time)
    d2 = t0 + budget_s * 0.65
    h_c, c5_c = _cont_polish(best_h, min(d2, time.time() + budget_s * 0.12), max_iters=400)
    if c5_c < best_c5:
        best_c5 = c5_c
        best_h = h_c.copy()

    # SLSQP refine
    if time.time() < d2:
        h_s, c5_s = _slsqp_refine(best_h, min(d2, time.time() + budget_s * 0.05), n_active=25)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Another round of continuous polish
    if time.time() < d2:
        h_c2, c5_c2 = _cont_polish(best_h, d2, max_iters=300)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2.copy()

    # Phase 3: Multi-resolution search (25% time)
    d3 = t0 + budget_s * 0.88
    best_n = len(best_h)
    tried_n = set([best_n])
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 180 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)

        # Resample and do binary ILS
        h_r = _resample_h(best_h, n_try)
        h_r_bin = _make_binary(n_try, h_r)
        sub_dl_bin = min(d3, time.time() + budget_s * 0.02)
        h_r_bin, c5_r_bin = _binary_ilS(h_r_bin, sub_dl_bin, max_rounds=50, perturb_size=3)

        # Continuous polish
        sub_dl_c = min(d3, time.time() + budget_s * 0.015)
        h_r_c, c5_r_c = _cont_polish(h_r_bin, sub_dl_c, max_iters=150)

        if c5_r_c < best_c5 - 1e-12:
            best_c5 = c5_r_c
            best_h = h_r_c.copy()
            best_n = n_try

    # Phase 4: Final polish (5% time)
    d4 = deadline
    h_f, c5_f = _cont_polish(best_h, d4, max_iters=200)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final SLSQP
    if time.time() < d4 - 5:
        h_sf, c5_sf = _slsqp_refine(best_h, d4, n_active=20)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
