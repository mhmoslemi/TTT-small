def binary_sa(h, n, max_steps, rng, T0=1.0, T1=0.01):
    """Simulated annealing for binary sequence optimization."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for step in range(max_steps):
        T = T0 * (T1/T0) ** (step / max_steps)
        # Pick random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0; h[j] = 1
        corr_new = np.correlate(h, 1-h, mode='full')
        new_max = float(corr_new.max())
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    return best_h, best_max * dx

    corr = np.correlate(h, 1-h, mode='full')

    best_h = h.copy()
    best_max = cur_max
    stall = 0
    
    for step in range(max_steps):
        # Find top worst lags
        top_k = 5
        sorted_idx = np.argsort(-corr)[:top_k]
        
        best_swap = None
        best_new_max = cur_max
        
        for lag_idx in sorted_idx:
            lag = lag_idx - (n - 1)  # Convert to actual lag
            if lag == 0:

            if swap is None:
                continue
            i, j = swap
            # Apply swap tentatively
            h[i], h[j] = h[j], h[i]
            new_corr = np.correlate(h, 1-h, mode='full')
            new_max = float(new_corr.max())
            h[i], h[j] = h[j], h[i]  # Revert
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j)

            h[i], h[j] = h[j], h[i]

                best_max = cur_max

            stall = 0
