import time
from scipy.optimize import minimize


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0

        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) < 1e-10:
            break
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x


def _lag_gradient(h, lag, dx):
    """Compute gradient of s_lag with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For shift in valid range:
    # valid shift: 0 <= shift < n and 0 <= shift + lag < n
    lo_shift = max(0, -lag)
    hi_shift = min(n, n - lag)
    # shift ranges from lo_shift to hi_shift-1
    shifts = np.arange(lo_shift, hi_shift)
    # h[shift+lag] * (1-h[shift])
    # d/dh[i]: 
    #   if i = shift+lag: (1-h[shift]) = (1-h[i-lag])
    #   if i = shift: -h[shift+lag] = -h[i+lag]
    # Term 1: i = shift + lag, for shift in [lo_shift, hi_shift)
    #   i ranges from lo_shift+lag to hi_shift+lag-1
    idx1 = shifts + lag  # these are i values where i-lag = shift is valid
    grad[idx1] += (1.0 - h[shifts])
    # Term 2: i = shift, for shift in [lo_shift, hi_shift)
    #   i ranges from lo_shift to hi_shift-1
    idx2 = shifts  # these are i values where i+lag = shift+lag is valid
    grad[idx2] += -h[idx2 + lag]
    return dx * grad


def _slsqp_solve(h, z_init, n_lags, deadline):
    """SLSQP epigraph solve."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Select top n_lags active
    top_idx = np.argsort(-corr)[:n_lags]
    top_lags = top_idx - (n - 1)
    
    def objective(x):
        return x[n]  # z
    
    def objective_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def eq_constraint(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    # Inequality constraints: z - s_lag(h) >= 0
    def make_ineq(lag):
        def fun(x):
            hx = x[:n]
            corr_val = _exact_overlap(hx, lag, dx, n)
            return x[n] - corr_val
        def jac(x):
            hx = x[:n]
            g = np.zeros(n + 1)
            g[:n] = _lag_gradient(hx, lag, dx)
            g[n] = -1.0
            return g
        return fun, jac
    
    constraints = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac}]
    for lag in top_lags:
        f, j = make_ineq(lag)
        constraints.append({'type': 'ineq', 'fun': f, 'jac': j})
    
    bounds = [(0, 1)] * n + [(0, 2)]

import numpy as np
import time
from scipy.optimize import linprog


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) < 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x


def _overlap_at_lag(h, lag, dx, n):
    lo = max(0, -lag)
    hi = min(n, n - lag)
    shifts = np.arange(lo, hi)
    return dx * float(np.sum(h[shifts + lag] * (1.0 - h[shifts])))


def _lag_grad(h, lag, dx, n):
    grad = np.zeros(n)
    lo = max(0, -lag)
    hi = min(n, n - lag)
    shifts = np.arange(lo, hi)
    idx1 = shifts + lag
    grad[idx1] += (1.0 - h[shifts])
    idx2 = shifts
    grad[idx2] += -h[idx2 + lag]
    return dx * grad


def _lp_direction(h, active_lags, dx, n):
    """Solve LP to find direction minimizing max linearized overlap increase."""
    k = len(active_lags)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((k, n + 1))
    b_ub = np.zeros(k)
    for i, lag in enumerate(active_lags):
        A_ub[i, :n] = _lag_grad(h, lag, dx, n)
        A_ub[i, n] = -1.0
        b_ub[i] = -_overlap_at_lag(h, lag, dx, n)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(-1.0, 10.0)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
        if result.success:
            d = result.x[:n].copy()
            # Center it (should already be zero-sum from equality constraint)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm > 1e-14:
                return d / nrm
    except Exception:
        pass
    return None


def _line_search(h, d, dx, n, L, U):
    """Golden section line search."""
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, _exact_c5(h + t * d)
    
    # Quadratic model for initial bracket
    corr = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # S0 + t*T - t^2*Q gives the overlap values at h + t*d
    # (this is exact for the quadratic structure of the overlap)
    
    nq = 100
    grid = np.linspace(L, U, nq)
    S0 = corr
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    
    for _ in range(30):
        if (b - a) < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = float(grid[i]), float(obj[i])
    
    return t, tv


def _compute_bounds(h, d):
    """Find valid range of t such that h + t*d stays in [0,1]."""
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    low = [0.0]
    up = [1e30]
    if np.any(pos):
        low.append(float(np.min(-h[pos] / d[pos])))
        up.append(float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        low.append(float(np.min((1.0 - h[neg]) / d[neg])))
        up.append(float(np.min(-h[neg] / d[neg])))
    return max(low), min(up)


def _optimize_lp(h, deadline, rng, k_active=15):
    """Main optimization loop using LP directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    stall = 0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(-corr[:])[:k_active]
        top_lags = top_idx - (n - 1)
        
        # Try LP direction
        d = _lp_direction(h, top_lags, dx, n)
        
        if d is None:
            # Fallback: random direction
            d = rng.standard_normal(n)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d /= nrm
        else:
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d /= nrm
        
        L, U = _compute_bounds(h, d)
        if U - L < 1e-11:
            stall += 1
            continue
        
        t, newf = _line_search(h, d, dx, n, L, U)
        
        if abs(t) < 1e-14:
            stall += 1
            continue
        
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        
        if nf < best_f - 1e-12:
            best_f = nf
            best_h = hn.copy()
        
        if nf < cur_f + 1e-4:
            h = hn
            cur_f = nf
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            stall += 1
        
        if stall > 100:
            # Restart with perturbation
            sigma = max(0.005, 0.05 * 0.9 ** (stall // 100))
            h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            stall = 0
    
    return best_h, best_f


def _sine_init(n, freq, phase, rng):
    x = np.arange(n, dtype=float)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base += rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 5))))
        if rng.random() < 0.5:
            h[pos:pos + min(bl, n - pos)] = 1.0
        pos += bl
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def _greedy_binary(n, rng):
    """Generate a binary-like initialization using greedy anti-correlation."""
    h = np.zeros(n)
    target = n // 2
    # Start with a random subset of size target
    indices = rng.permutation(n)[:target]
    h[indices] = 1.0
    # Greedily flip to reduce max overlap
    for _ in range(200):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_k = int(np.argmax(corr))
        lag = max_k - (n - 1)
        # Find a pair to flip that reduces the max
        best_improvement = 0
        best_pair = None
        lo = max(0, -lag)
        hi = min(n, n - lag)
        shifts = np.arange(lo, hi)
        # For each shift, if h[shift]=0 and h[shift+lag]=1, flipping both helps
        for shift in shifts:
            i1 = shift
            i2 = shift + lag
            if h[i1] == 0 and h[i2] == 1:
                # Flipping: h[i1] becomes 1, h[i2] becomes 0
                # This reduces overlap at lag by dx for this pair
                # But might increase overlap at other lags
                h[i1] = 1.0
                h[i2] = 0.0
                new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                improvement = float(corr.max()) - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_pair = (i1, i2)
                h[i1] = 0.0
                h[i2] = 1.0
        if best_pair is not None:
            h[best_pair[0]] = 1.0
            h[best_pair[1]] = 0.0
        else:
            break
    return _project(h, n / 2.0)


def _resample(h, n_new):

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            c1 = b - gr * (b - a)
            f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(corr + t * T - t * t * Q)) * dx
    if float(obj[i]) < tv:
        t, tv = float(grid[i]), float(obj[i])
    return t, tv


def _optimize(h, deadline, rng, k_init=12, k_max=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    stall = 0
    iters = 0
    k_active = k_init
    restarts = 0
    pair_frac = 0.12

    while time.time() < deadline:
        iters += 1
        overlaps = _all_overlaps(h)
        top_idx = np.argsort(-overlaps)[:k_active]
        top_lags = top_idx - (n - 1)

        d = None
        r = rng.random()
        if r < pair_frac:
            d = _pair_direction(h, top_lags, dx, n, rng)
        if d is None:
            d = _lp_direction(h, top_lags, dx, n)
        if d is None or float(np.linalg.norm(d)) < 1e-12:
            if r < pair_frac + 0.10:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
            else:
                d = rng.standard_normal(n)
                d -= d.mean()
                nrm = float(np.linalg.norm(d))
                if nrm < 1e-14:
                    stall += 1
                    continue
                d /= nrm

        t, nf_est = _line_search_exact(h, d, n)
        if abs(t) < 1e-14:
            stall += 1
            continue

        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        nf = _exact_c5(hn)

        improved = False
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
            improved = True
        if nf <= cur_f + 5e-5:
            h = hn
            cur_f = nf
            if improved:
                stall = 0
                if iters % 30 == 0 and k_active < k_max:
                    k_active += 2
            else:
                stall += 1
        else:
            stall += 1
            k_active = max(k_init, k_active - 1)
            if stall > 60:
                restarts += 1
                sigma = max(0.003, 0.04 * (0.93 ** restarts))
                h = _project(best_h + rng.normal(0, sigma, n), target)
                cur_f = _exact_c5(h)
                stall = 0
                k_active = k_init

    return best_h, best_f


def _init_sine(n, freq, phase, rng):
    x = np.arange(n, dtype=float)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base += rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)


def _init_block(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 4))))
        if rng.random() < 0.5:
            h[pos:pos + min(bl, n - pos)] = 1.0
        pos += bl
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def _init_greedy(n, rng):
    h = np.zeros(n)
    target = n // 2
    idx = rng.permutation(n)[:target]
    h[idx] = 1.0
    for _ in range(100):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_k = int(np.argmax(corr))
        lag = max_k - (n - 1)
        best_imp = 0.0
        best_pair = None
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for s in range(lo, hi):
            i1 = s
            i2 = s + lag
            if i1 == i2:
                continue
            if h[i1] < 0.5 and h[i2] > 0.5:
                h[i1] = 1.0
                h[i2] = 0.0
                new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                imp = float(corr.max()) - new_max
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (i1, i2)
                h[i1] = 0.0
                h[i2] = 1.0
        if best_pair is not None:
            h[best_pair[0]] = 1.0
            h[best_pair[1]] = 0.0
        else:
            break
    return _project(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
