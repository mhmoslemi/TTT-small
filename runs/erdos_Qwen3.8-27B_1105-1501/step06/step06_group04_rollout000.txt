import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_unscaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -10, 10)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

def _line_search_exact(h, d, dx, max_steps=50):
    """Exact line search along h + t*d, finding minimum of max correlation."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find bounds for t
    t_lo = -1e6
    t_hi = 1e6
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
    
    t_lo = max(t_lo, -0.5)
    t_hi = min(t_hi, 0.5)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Golden section search
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * Q)) * dx
    
    for _ in range(max_steps):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1*T - c1*c1*Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2*T - c2*c2*Q)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*Q)) * dx
    return opt_t, opt_val

def _grad_row(h, r, n):
    """Gradient of the r-th correlation w.r.t. h."""
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] += -h[d:]
    else:
        s = -d
        m = n - s
        if m > 0:
            g[:m] += (1.0 - h[s:])
            g[s:] += -h[:m]
    return g

def _subgradient_dir(h, scores, top_k=10, equalize_weight=0.5):
    """Compute subgradient direction for the max correlation.
    Returns a zero-sum direction that reduces the max."""
    n = len(h)
    max_s = float(scores.max())
    if max_s <= 1e-15:
        return None
    # Select active lags
    thr = max_s * 0.95
    active = np.where(scores >= thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    if len(active) > top_k:
        order = np.argsort(-scores[active])[:top_k]
        active = active[order]
    
    # Average gradient of active constraints
    G = np.zeros((len(active), n))
    for i, r in enumerate(active):
        G[i] = _grad_row(h, r, n)
    
    # Weight by how close to max
    weights = scores[active] / scores[active].sum()
    avg_grad = (weights[:, None] * G).sum(axis=0)
    
    # Project to zero-sum
    avg_grad -= avg_grad.mean()
    
    norm = np.linalg.norm(avg_grad)
    if norm < 1e-14:
        return None
    return -avg_grad / norm

def _fourier_perturb(h, n, freq, phase, amplitude):
    """Add a Fourier mode perturbation and project."""
    x = np.arange(n)
    pert = amplitude * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h + pert, n / 2.0)

def _block_perturb(h, n, block_size, rng):
    """Add block-wise perturbation."""
    h_new = h.copy()
    nblocks = n // block_size
    for _ in range(nblocks):
        offset = int(rng.integers(0, n))
        end = min(n, offset + block_size)
        shift = float(rng.uniform(-0.05, 0.05))
        h_new[offset:end] += shift
    return _project(h_new, n / 2.0)

def _pair_move(h, i, j, dx, target, sign=1.0):
    """Try a 2-coordinate move: increase h[i], decrease h[j]."""
    d = np.zeros(len(h))
    d[i] = sign
    d[j] = -sign
    alpha, val = _line_search_exact(h, d, dx)
    if val < _exact_c5(h) - 1e-15:
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        return h_new, val
    return None, None

            break

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, rng, max_iters=10000):
    """Binary simulated annealing."""
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.003
    T_min = 0.0001
    cooling = 0.9997
    ones_list = np.where(h > 0.5)[0].tolist()
    zeros_list = np.where(h <= 0.5)[0].tolist()
    patience = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 2000:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[int(rng.integers(len(ones_list)))]
        j = zeros_list[int(rng.integers(len(zeros_list)))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            ones_list = np.where(h > 0.5)[0].tolist()
            zeros_list = np.where(h <= 0.5)[0].tolist()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _quadruple_move(h, idx4, dx, target):

            h_new = _project(h_new, target)

    return None, None

    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)

def _periodic_init(n, period, duty):
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h = np.zeros(n)
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)
