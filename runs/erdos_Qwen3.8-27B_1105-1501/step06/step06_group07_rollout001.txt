def _corr_vec(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        for j in range(n - k):
            g[j] += (1.0 - h[j + k])
        for j in range(k, n):
            g[j] -= h[j - k]
    else:
        ga = -k
        for j in range(n - ga):
            g[j] += (1.0 - h[j + ga])
        for j in range(ga, n):
            g[j] -= h[j - ga]
    return g

def _grad_lag_fast(h, k):

def _weighted_gradient(h, K=30, alpha=50.0):
    n = len(h)
    corr = _corr_vec(h)
    M = float(np.max(corr))
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - M))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(K) / K
    else:
        w /= w_sum
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _grad_lag_fast(h, k)
    grad -= np.mean(grad)
    return grad, corr

            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_binary(h, rng, frac=0.5):
    n = len(h)
    target = n // 2
    if n % 2 == 1:
        target = n // 2
    perm = rng.permutation(n)
    b = np.zeros(n)
    b[perm[:target]] = 1.0
    b = 0.5 * b + 0.3 * h
    return _project(b, n / 2.0)

def _make_symmetric(h):
    return _project(0.5 * (h + h[::-1]), len(h) / 2.0)

def _make_fourier(n, rng):
    x = np.arange(n)
    base = 0.5
    for freq in range(1, 6):
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        base += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(base, n / 2.0)

def _combined_polish(h, deadline, rng, alpha=40.0, K=25):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0
    max_iters = 500
    for it in range(max_iters):
        if time.time() > deadline:
            break
        phase = (it % 3)
        if phase == 0:
            grad, corr = _weighted_gradient(h, K, alpha)
            gn = np.linalg.norm(grad)
            if gn < 1e-13:
                direction = None
            else:
                direction = -grad / gn
        elif phase == 1:
            d = rng.standard_normal(n)
            d -= np.mean(d)
            gn = np.linalg.norm(d)
            if gn < 1e-12:
                direction = None
            else:
                direction = d / gn
        else:
            i, j = rng.integers(0, n, 2)
            if i == j:
                direction = None
            else:
                direction = np.zeros(n)
                direction[i] = 1.0
                direction[j] = -1.0
        if direction is None:
            continue
        t, f_try = _line_search(h, direction, dx, max_t=step)
        if t < 1e-12:
            step *= 0.7
            if step < 0.01:
                break
            continue
        h_new = h + t * direction
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.8 * step:
                step = min(step * 1.05, 1.5)
            else:
                step = max(step * 0.92, 0.01)
        else:
            step *= 0.8
            if step < 0.005:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.95

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * 5 * np.arange(n0) / n0), 0, 1)

    h0 = _project(h0.copy(), n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Polish the parent (25% budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _combined_polish(h0, min(d1, time.time() + budget_s * 0.15), rng, alpha=40, K=25)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Active descent follow-up
    if time.time() < d1:
        h1b, c5_1b = _active_descent(best_h, d1, K=20, max_iters=200, alpha=50)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h1b.copy()

    # Phase 2: Multi-resolution search (30% budget)
    d2 = t0 + budget_s * 0.55
    n_base = len(best_h)
    n_offsets = [-8, -6, -4, -2, 2, 4, 6, 8, -10, 10, -12, 12, -14, 14, -16, 16]
    tried = set()
    for dn in n_offsets:
        if time.time() > d2:
            break
        n_try = n_base + dn
        if n_try < 40 or n_try > 220:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.025)
        h_r, c5_r = _combined_polish(h_r, sub_dl, rng, alpha=40, K=20)
        if time.time() < d2 - 2:
            h_r2, c5_r2 = _active_descent(h_r, min(d2, time.time() + budget_s * 0.01), K=15, max_iters=80)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_base = n_try
            # Continue with nearby sizes from new base
            n_offsets = [n_base + d for d in [-4, 4, -6, 6, -8, 8, 10, -10, 12, -12]]
            n_offsets = [int(x) for x in n_offsets if 40 <= x <= 220]

    # Phase 3: Diverse starting points (20% budget)
    d3 = t0 + budget_s * 0.75
    n_cur = len(best_h)
    seeds = []
    # Symmetric version
    hs = _make_symmetric(best_h)
    seeds.append(hs)
    # Binary version
    seeds.append(_make_binary(best_h, rng))
    # Roll variants
    for s in [1, 2, -1]:
        seeds.append(_project(np.roll(best_h, s), n_cur / 2.0))
    # Fourier seeds at current size
    for _ in range(3):
        seeds.append(_make_fourier(n_cur, rng))
    # Fourier at different sizes
    for n_f in [n_cur - 4, n_cur + 4, n_cur - 8, n_cur + 8]:
        if n_f >= 40 and n_f <= 220:
            seeds.append(_make_fourier(n_f, rng))
    
    for h_s in seeds:
        if time.time() > d3:
            break
        n_s = len(h_s)
        sub_dl = min(d3, time.time() + budget_s * 0.02)
        h_s, c5_s = _combined_polish(h_s, sub_dl, rng, alpha=35, K=20)
        if time.time() < d3 - 2:
            h_s2, c5_s2 = _active_descent(h_s, min(d3, time.time() + budget_s * 0.008), K=15, max_iters=60)
            if c5_s2 < c5_s:
                h_s, c5_s = h_s2, c5_s2
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 4: SA from best (10% budget)
    d4 = t0 + budget_s * 0.85
    if time.time() < d4:
        h_sa, c5_sa = _sa_binary(best_h, min(d4, time.time() + budget_s * 0.05), rng, T_init=0.008)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Polish SA result
        if time.time() < d4:
            h_p, c5_p = _combined_polish(best_h, d4, rng, alpha=50, K=20)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 5: Continuous SA + polish (5% budget)
    d5 = t0 + budget_s * 0.92
    if time.time() < d5:
        h_cs, c5_cs = _sa_continuous(best_h, min(d5, time.time() + budget_s * 0.03), rng, T_init=0.003)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs.copy()
        if time.time() < d5:
            h_p2, c5_p2 = _active_descent(best_h, d5, K=15, max_iters=100)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2.copy()

    # Final polish
    d_final = deadline
    if time.time() < d_final:
        h_f, c5_f = _combined_polish(best_h, d_final, rng, alpha=60, K=20)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
