Can I vectorize the evaluation? For a given h and a set of candidate swaps (i₁,j₁), (i₂,j₂), ..., (iₘ,jₘ):
- For each swap, the new correlation is: corr + Δcorr_s
- where Δcorr_s depends on the specific swap
- We need max(corr + Δcorr_s) for each s

This can be done with matrix operations if we precompute the relevant h values.

For swap (i, j): Δcorr[q] = -(1-h[i+s_q]) + (1-h[j+s_q]) + h[i-s_q] - h[j-s_q] (with validity checks, where s_q = q - (n-1))

Δcorr_s = -a_i + a_j + b_i - b_j

And the new max is max(corr + Δcorr_s) = max(corr - a_i + a_j + b_i - b_j)

For m candidates, this is an (m × (2n-1)) matrix operation. For m=2000 and 2n-1=241, that's about 500K elements. Very fast in numpy.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    v = np.asarray(v, dtype=float)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _thue_morse_binary(n):
    """Generate Thue-Morse binary sequence with exactly n/2 ones."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') & 1
    # Adjust to have exactly n/2 ones
    target = n // 2
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - current]] = 1
    elif current > target:
        ones = np.where(h == 1)[0]
        h[ones[:current - target]] = 0
    return h

def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h

def _binary_hill_climb(h, deadline, rng, max_evals=200000):
    """First-improvement hill climbing with random restarts."""
    n = len(h)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    evals = 0
    stall = 0
    
    while time.time() < deadline and evals < max_evals:
        if stall > 200:
            # Random restart perturbation
            for _ in range(10):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i], h[j] = h[j], h[i]
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            stall = 0
            continue
        
        # Try random swaps
        improved = False
        for _ in range(100):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5_new = _exact_c5(h)
            evals += 1
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                improved = True
                stall = 0
                break
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            stall += 1
        if time.time() > deadline:
            break
    
    return best_h, best_c5

def _smooth_opt(h, deadline, max_iters=50000):
    """Smooth gradient descent using log-sum-exp surrogate."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = 0.02
    lr = 0.05
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # Smooth max
        max_vals = np.max(corr)
        w = np.exp((corr - max_vals) / tau)
        w = w / np.sum(w)
        
        # Gradient of smooth max w.r.t. h
        # d/dh[j] of max_q corr[q] approx = sum_q w[q] * dcorr[q]/dh[j]
        # corr[q] = dx * sum_i h[i] * (1-h[i+q-(n-1)])
        # Let s = q - (n-1)
        # dcorr[q]/dh[j] = dx * [(1-h[j+s]) if j+s valid + (-h[j-s]) if j-s valid]
        
        grad = np.zeros(n)
        for q in range(2*n-1):
            if w[q] < 1e-8:
                continue
            s = q - (n-1)
            # j as first factor: need 0 ≤ j+s < n
            j_lo = max(0, -s)
            j_hi = min(n, n - s)  # j+s < n => j < n-s
            for j in range(j_lo, j_hi):
                grad[j] += w[q] * dx * (1.0 - h[j + s])
            # j as second factor: need 0 ≤ j-s < n, i.e., s ≤ j < s+n
            j_lo2 = max(0, s)
            j_hi2 = min(n, s + n)
            for j in range(j_lo2, j_hi2):
                grad[j] += w[q] * dx * (-h[j - s])
        
        h = h - lr * grad
        h = _project(h, target)
        c5_new = _exact_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.95
        if it % 100 == 0:
            tau *= 0.98
    
    return best_h, best_c5
