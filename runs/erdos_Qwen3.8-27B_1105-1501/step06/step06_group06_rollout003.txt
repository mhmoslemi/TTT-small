    # For shift s >= 0: C(s) = sum_{i=0}^{m-1} (h[i]+td[i])(1-h[i+s]-td[i+s])
    #   = c_s + t * sum(d[i](1-h[i+s]) - h[i]*d[i+s]) + t^2 * sum(-d[i]*d[i+s])
    # For shift s < 0 (g = -s): C(-g) = sum_{i=0}^{m-1} (h[i+g]+td[i+g])(1-h[i]-td[i])
    #   = c_{-g} + t * sum(d[i+g](1-h[i]) - h[i+g]*d[i]) + t^2 * sum(-d[i+g]*d[i])
    
    L = np.zeros(2*n-1)
    Q = np.zeros(2*n-1)
    
    for s in range(-(n-1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            L[k] = np.sum(d[:m] * omh[s:s+m]) - np.sum(h[:m] * d[s:s+m])
            Q[k] = -np.sum(d[:m] * d[s:s+m])
        else:
            g = -s
            L[k] = np.sum(d[g:g+m] * omh[:m]) - np.sum(h[g:g+m] * d[:m])
            Q[k] = -np.sum(d[g:g+m] * d[:m])
    
    # Determine feasible range of t
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = 10.0  # default upper bound
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    t_lo = -10.0
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    # We want t > 0 for descent (but allow slightly negative for safety)
    t_lo = max(t_lo, -0.01)
    t_hi = min(t_hi, 0.5)  # cap step size
    if t_max_override is not None:
        t_hi = min(t_hi, t_max_override)
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(c)) * dx
    
    # Evaluate f(t) = max_s (c_s + t*L_s + t^2*Q_s) * dx
    # Use grid + refinement
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    # Vectorized: (n_grid, 2n-1) array
    vals = c[None, :] + grid[:, None] * L[None, :] + (grid[:, None])**2 * Q[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]
    
    # Refine with golden section (or narrow the grid)
    if 1 <= best_i <= n_grid - 2:
        a = grid[best_i - 1]
        b = grid[best_i + 1]
    else:
        a = t_lo
        b = t_hi
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(c + c1 * L + c1*c1 * Q)) * dx
    f2 = float(np.max(c + c2 * L + c2*c2 * Q)) * dx
    for _ in range(60):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b = c2
            f2 = f1
            c2 = a + gr * (b - a)
            f2 = float(np.max(c + c2 * L + c2*c2 * Q)) * dx
        else:
            a = c1
            f1 = f2
            c1 = b - gr * (b - a)
            f1 = float(np.max(c + c1 * L + c1*c1 * Q)) * dx
    
    mid = (a + b) / 2.0
    mid_val = float(np.max(c + mid * L + mid*mid * Q)) * dx
    if mid_val < best_val:
        best_t = mid
        best_val = mid_val
    
    # Also check t=0
    t0_val = float(np.max(c)) * dx
    if t0_val < best_val:
        best_t = 0.0
        best_val = t0_val
    
    # Verify with actual correlation at the best point
    h_test = h + best_t * d
    if np.any(h_test < -1e-12) or np.any(h_test > 1 + 1e-12):
        # Clip
        best_t = 0.0
        best_val = _exact_c5(h)
    
    return best_t, best_val

def _generate_directions(h, active_lags, corr_vals, grads, n, radius):
    """Generate multiple candidate directions."""
    directions = []
    
    # Direction 1: LP-based (minimize max of active gradients)
    try:
        m = len(active_lags)
        G_act = grads[active_lags]  # (m, n)
        # LP: max delta s.t. G_act @ d >= delta, sum(d)=0, ||d||_inf <= radius
        # Equivalent: minimize -delta
        c_lp = np.zeros(n+1)
        c_lp[n] = -1.0
        A_ub = np.zeros((m, n+1))
        A_ub[:, :n] = G_act
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        bounds = [(-radius, radius)] * n + [(None, None)]
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                     bounds=bounds, method='highs')
        if res.success and -res.fun > 1e-10:
            d = res.x[:n].copy()
            d -= d.mean()  # ensure sum = 0 (should already be)
            d_norm = np.max(np.abs(d))
            if d_norm > 1e-12:
                d = d / d_norm * radius
                directions.append(d)
    except Exception:
        pass
    
    # Direction 2: Negative average gradient of active set
    G_act = grads[active_lags]
    avg_grad = G_act.mean(axis=0)
    avg_grad -= avg_grad.mean()  # project to sum=0
    gnorm = np.linalg.norm(avg_grad)
    if gnorm > 1e-12:
        d2 = -avg_grad / gnorm * radius
        directions.append(d2)
    
    # Direction 3: Equalization direction
    # Solve: G_act @ d = b where b[s] = -(corr_vals[active_lags[s]] - mean_active)
    # plus sum(d) = 0
    try:
        m = len(active_lags)
        c_act = corr_vals[active_lags]
        c_mean = c_act.mean()
        b_vec = -(c_act - c_mean)  # want to equalize
        # Build system: [G_act; 1^T] @ d = [b_vec; 0]
        A_sys = np.vstack([G_act, np.ones((1, n))])
        b_sys = np.concatenate([b_vec, [0.0]])
        # Minimum norm solution
        # d = A_sys^T (A_sys A_sys^T)^{-1} b_sys
        AT = A_sys.T
        ATA_inv = np.linalg.pinv(A_sys @ A_sys.T)
        d3 = AT @ (ATA_inv @ b_sys)
        d3_norm = np.max(np.abs(d3))
        if d3_norm > 1e-14:
            d3 = d3 / d3_norm * radius
            directions.append(d3)
    except Exception:
        pass
    
    # Direction 4: "Push down the max" - direction that most reduces the single worst constraint
    worst = np.argmax(corr_vals)
    d4 = -grads[worst].copy()
    d4 -= d4.mean()
    d4_norm = np.max(np.abs(d4))
    if d4_norm > 1e-12:
        d4 = d4 / d4_norm * radius
        directions.append(d4)
    
    return directions

def _eacld_optimize(h0, deadline, max_iter=2000):
    """Main optimization loop using EACLD."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.05
    stall = 0
    rng = np.random.default_rng(12345)
    
    # Active set tolerance: start broad, narrow over time
    active_tol = 0.005
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute correlations and gradients
        corr, grads = _corr_and_grads(h)
        M = float(np.max(corr))
        cur_c5 = M  # dx already included in corr
        
        # Select active set
        active = np.where(corr > M - active_tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        elif len(active) > 30:
            # Take top 30
            top_order = np.argsort(-corr[active])
            active = active[top_order[:30]]
        
        # Generate directions
        directions = _generate_directions(h, active, corr, grads, n, radius)
        
        # Add random direction
        d_rand = rng.standard_normal(n)
        d_rand -= d_rand.mean()
        d_rand_norm = np.max(np.abs(d_rand))
        if d_rand_norm > 0:
            d_rand = d_rand / d_rand_norm * radius
            directions.append(d_rand)
        
        # Try each direction, accept the best
        best_new_c5 = cur_c5
        best_new_t = 0.0
        best_new_d = None
        
        for d in directions:
            t, val = _line_search_exact(h, d, t_max_override=radius * 2)
            if val < best_new_c5 - 1e-14:
                # Verify with exact evaluation
                h_test = h + t * d
                h_test = _project(h_test, target)
                c5_test = _exact_c5(h_test)
                if c5_test < best_new_c5:
                    best_new_c5 = c5_test
                    best_new_t = t
                    best_new_d = d
        
        if best_new_t != 0.0 and best_new_d is not None:
            h = h + best_new_t * best_new_d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            # Increase radius if we made good progress
            if best_new_t > 0.5 * radius:
                radius = min(radius * 1.15, 0.15)
        else:
            stall += 1
            radius *= 0.8
            
            # Periodically try restarts
            if stall % 30 == 29:
                # Try a perturbation
                h_pert = h + rng.standard_normal(n) * 0.02
                h_pert = _project(h_pert, target)
                c5_pert = _exact_c5(h_pert)
                if c5_pert < cur_c5:
                    h = h_pert
                    cur_c5 = c5_pert
                radius = 0.05
            
            if stall % 50 == 49:
                # Narrow active set
                active_tol *= 0.7
                if active_tol < 0.0003:
                    active_tol = 0.005  # reset
            
            if stall > 100:
                # Try multi-resolution
                # (not implemented here, done in outer loop)
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
