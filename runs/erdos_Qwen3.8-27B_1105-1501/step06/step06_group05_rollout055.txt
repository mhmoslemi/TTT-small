def _pair_direction(h, rng):
    """Choose a random pair and create a direction that transfers mass between them."""
    n = len(h)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def _active_lag_direction(h, corr, n_active, rng):
    """Create a direction that reduces the top active lags."""
    n = len(h)
    # Get the top n_active lags
    top = np.argsort(-corr)[:n_active]
    d = np.zeros(n)
    for r in top:
        k = r - (n - 1)
        # For this lag, the gradient of corr_r w.r.t. h is:
        # g[a] += (1-h[b]), g[b] -= h[a]
        # where a and b are the paired indices
        if k >= 0:
            for i in range(n - k):
                d[i + k] += (1.0 - h[i])
                d[i] -= h[i + k]
        else:
            for i in range(-k, n):
                d[i] += (1.0 - h[i + k])
                d[i + k] -= h[i]
    # d is the gradient of the sum of top active lags
    # We want to descend, so negate
    d = -d
    d -= d.mean()  # maintain sum constraint
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm

def _slsqp_solve(h, top_k, max_iter=50, tol=1e-6):
    """SLSQP optimization with top_k active constraints."""
    n = len(h)
    dx = 2.0 / n
    corr = _corr_full(h)
    z0 = float(np.max(corr)) * dx + 1e-4  # initial z
    
    # Select top_k lags
    top = np.argsort(-corr)[:top_k]
    
    # Define the NLP
    # Variables: [h_0, ..., h_{n-1}, z]
    # Objective: z
    # Constraints:
    #   - sum(h) - n/2 = 0 (equality)
    #   - corr_r * dx - z <= 0 for r in top (inequality)
    # Bounds: 0 <= h_i <= 1, z free (or z >= 0)
    
    def obj(x):
        return x[n]  # z
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Precompute the index structure for each selected lag
    lag_data = []
    for r in top:
        k = r - (n - 1)
        if k >= 0:
            idx_a = np.arange(k, n)
            idx_b = np.arange(0, n - k)
        else:
            idx_a = np.arange(0, n + k)
            idx_b = np.arange(-k, n)
        lag_data.append((idx_a, idx_b))
    
    x0 = np.concatenate([h, [z0]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    constraints = []
    # Equality: sum(h) = n/2
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - n / 2.0,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    # Inequalities: corr_r * dx - z <= 0 => -(corr_r * dx - z) >= 0
    for (idx_a, idx_b) in lag_data:
        def con_fun(x, ia=idx_a, ib=idx_b):
            h_x = x[:n]
            val = np.sum(h_x[ia] * (1.0 - h_x[ib])) * dx - x[n]
            return -val  # must be >= 0
        def con_jac(x, ia=idx_a, ib=idx_b):
            h_x = x[:n]
            g = np.zeros(n + 1)
            # d/dh_a: (1 - h_b) * dx
            # d/dh_b: -h_a * dx
            np.add.at(g, ia, (1.0 - h_x[ib]) * dx)
            np.add.at(g, ib, -h_x[ia] * dx)
            g[n] = 1.0  # d/dz of -(corr*dx - z) = +1
            return g
        constraints.append({'type': 'ineq', 'fun': con_fun, 'jac': con_jac})
    
    try:
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                       constraints=constraints, method='SLSQP',
                       options={'maxiter': max_iter, 'ftol': tol})
        if res.success or res.fun < z0:
            h_new = _project(res.x[:n], n / 2.0)
            return h_new
    except Exception:
        pass
    return None

- Lag 1: $\sum_{i=0}^{n-2} h[i+1](1-h[i])$. For $i < n/2 - 1$: $h[i+1] = 1, h[i] = 1$, so term is 0. For $i = n/2 - 1$: $h[i+1] = 0, h[i] = 1$, term is 0. For $i \geq n/2$: $h[i+1] = 0$, term is 0. So lag 1 = 0.
- Lag $n/2$: $\sum_{i=0}^{n/2-1} h[i+n/2](1-h[i]) = \sum 0 \cdot 1 = 0$.
- Lag $-n/2 + 1$: $\sum_{i=0}^{n/2-1} h[i](1-h[i+n/2-1])$. For $i < n/2 - 1$: $h[i] = 1, h[i+n/2-1] = 1$ (if $i+n/2-1 < n/2$, i.e., $i < 1$), so... this gets complicated.

Actually, for the step function $h[i] = 1$ for $i \in [0, n/2)$ and $0$ otherwise:
- At lag $k > 0$: $\sum_{i=0}^{n-k-1} h[i+k](1-h[i])$
  - For $i$ where both $i < n/2$ and $i+k < n/2$: $h[i+k] = 1, h[i] = 1$, term = 0
  - For $i$ where $i < n/2$ and $i+k \geq n/2$: $h[i+k] = 0, h[i] = 1$, term = 0
  - For $i$ where $i \geq n/2$ and $i+k < n/2$: impossible since $k > 0$
  - For $i$ where $i \geq n/2$ and $i+k \geq n/2$: $h[i+k] = 0$, term = 0
  
  So lag $k > 0$ gives 0!

- At lag $k < 0$ (let $k = -q, q > 0$): $\sum_{i=0}^{n-q-1} h[i](1-h[i+q])$
  - For $i < n/2 - q$: $h[i] = 1, h[i+q] = 1$, term = 0
  - For $n/2 - q \leq i < n/2$: $h[i] = 1, h[i+q] = 0$, term = 1
  - For $i \geq n/2$: $h[i] = 0$, term = 0
  
  So the number of terms with value 1 is $q$ (for $q \leq n/2$) or $n/2$ (for $q > n/2$).
  
  The actual count is $\min(q, n/2)$ for $q \leq n/2$.

So the max overlap for the step function is at lag $-n/2 + 1$ (or $-n/2$): the sum is $n/2 - 1$ or $n/2$.

$C_5 = (n/2) \cdot dx = (n/2) \cdot (2/n) = 1$.

import numpy as np
import time
from scipy.optimize import minimize


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(20):
        if abs(res) < 1e-13:
            break
        mask = (x < 1.0 - 1e-10) if res > 0 else (x > 1e-10)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _lag_indices(k, n):
    if k >= 0:
        return np.arange(k, n), np.arange(0, n - k)
    else:
        return np.arange(0, n + k), np.arange(-k, n)


def _slsqp_solve(h, n_active, max_iter=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    z0 = float(np.max(corr)) * dx + 0.02

    top_idx = np.argsort(-corr)[:n_active]
    lag_offsets = top_idx - (n - 1)
    lag_data = [_lag_indices(int(k), n) for k in lag_offsets]

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    x0 = np.concatenate([h, [z0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    eq_fun = np.empty(0)  # placeholder

    def eq_f(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_j(x):
        g = np.empty(n + 1)
        g[:n] = 1.0
        g[n] = 0.0
        return g

    constraints = [{'type': 'eq', 'fun': eq_f, 'jac': eq_j}]

    for (ia, ib) in lag_data:
        def cfun(x, ia=ia, ib=ib):
            hx = x[:n]
            return -(np.sum(hx[ia] * (1.0 - hx[ib])) * dx - x[n])

        def cjac(x, ia=ia, ib=ib):
            hx = x[:n]
            g = np.zeros(n + 1)
            np.add.at(g, ia, (1.0 - hx[ib]) * dx)
            np.add.at(g, ib, -hx[ia] * dx)
            g[n] = 1.0
            return g

        constraints.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})

    try:
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                       constraints=constraints, method='SLSQP',
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        h_new = _project(res.x[:n], n / 2.0)
        return h_new
    except Exception:
        return None


def _pair_ls(h, i, j):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    T = -(np.correlate(h, d, mode='full') + np.correlate(d, 1.0 - h, mode='full'))
    Q = np.correlate(d, d, mode='full')
    L = -h[i]
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    for _ in range(55):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    ft = float(np.max(S0 + t * T + t * t * Q)) * dx
    for tc in [a, b, 0.0]:
        fc = float(np.max(S0 + tc * T + tc * tc * Q)) * dx
        if fc < ft:
            ft = fc
            t = tc
    return t, ft


def _general_ls(h, d, L, U):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = -(np.correlate(h, d, mode='full') + np.correlate(d, 1.0 - h, mode='full'))
    Q = np.correlate(d, d, mode='full')
    if U - L < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    for _ in range(55):

        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    ft = float(np.max(S0 + t * T + t * t * Q)) * dx

        fc = float(np.max(S0 + tc * T + tc * tc * Q)) * dx

            t = tc
    return t, ft


def _active_dir(h, corr, k_act, rng):
    n = len(h)
    top = np.argsort(-corr)[:k_act]
    d = np.zeros(n)
    for r in top:
        k = r - (n - 1)
        ia, ib = _lag_indices(k, n)
        d[ia] += (1.0 - h[ib])
        d[ib] -= h[ia]
    d = -d
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm


def _hybrid_opt(h, deadline, rng, use_slsqp=True, slsqp_every=50):
    n = len(h)
    target = n / 2.0
    h = h.astype(float).copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    stall = 0
    max_stall = 400
    iters = 0
    temp = 0.0008
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    while time.time() < deadline:
        iters += 1
        r = rng.random()
        if r < 0.55:
            # Pair move
            idx = int(rng.integers(len(pairs)))
            i, j = pairs[idx]
            t, nf = _pair_ls(h, i, j)
            if abs(t) < 1e-14:
                stall += 1
                continue
            h[i] += t
            h[j] -= t
        elif r < 0.80:
            # Active direction
            corr = np.correlate(h, 1.0 - h, mode='full')
            d = _active_dir(h, corr, 4, rng)
            if d is None:

            pos = d > 1e-15

            if np.any(neg):

            if U - L < 1e-12:

                continue
            t, nf = _general_ls(h, d, L, U)
            if abs(t) < 1e-14:

                continue
            h += t * d
        elif r < 0.92:
            # Random direction

            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                continue

            pos = d > 1e-15
            neg = d < -1e-15
            L, U = -1e10, 1e10
            if np.any(pos):

                continue

        else:

                continue

            h = result
            nf = _exact_c5(h)

                best_f = nf

            if nf <= cur_f + 1e-5:

            sig = max(0.02, 0.08 * (temp / 1e-8) * 0.1)
            h = _project(best_h + rng.normal(0, sig, n), target)
            cur_f = _exact_c5(h)
            stall = 0
            max_stall = max(80, int(max_stall * 0.75))
            temp = max(temp, 0.0003)

    return best_h, best_f


def _fourier_init(n, freq, amp, phase, rng):
    x = np.arange(n)
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    h += rng.normal(0, 0.015, n)
    return _project(h, n / 2.0)


def _resample_h(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
