        else:
            hi = mid

    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

def _waterfill_step(h, active_lags, dx):
    """Find direction that reduces all active lags by approximately the same amount."""
    n = len(h)
    G = _grad_topk(h, active_lags)  # (K, n)
    # We want to find d such that G @ d is as negative as possible in all components
    # and sum(d) = 0 (to maintain the constraint after projection)
    # This is: minimize max(G @ d) subject to ||d|| <= 1, sum(d) = 0
    # Or: minimize -min(G@d) (i.e., make all components as negative as possible)
    # Simple approach: d = -mean(G, axis=0), then project to sum=0
    d = -G.mean(axis=0)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    d /= norm
    return d

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_topk(h, lag_indices):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    K = len(lag_indices)
    G = np.zeros((K, n))
    for i, r in enumerate(lag_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[i, :m] += omh[d:]
                G[i, d:] -= h[:m]
        else:
            k = -d
            m = n - k
            if m > 0:
                G[i, k:] += omh[:m]
                G[i, :m] -= h[k:]
    return G * dx

def _lse_fista(h_init, deadline, beta_start=5.0, beta_end=400.0, n_beta=20, top_k_frac=0.99, iters_per_beta=None):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()

        iters_per_beta = max(10, int(200 / beta_start))

        t_k = 1.0

            if Z < 1e-300:

            sorted_p = np.sort(p)[::-1]
            cum = np.cumsum(sorted_p)
            K = max(2, int(np.searchsorted(cum, top_k_frac)))
            K = min(K, 2 * n - 1)

def _waterfill_opt(h_init, deadline, max_iters=200, tol=1e-10):
    """Water-filling: equalize the top active lags."""
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        max_S = S.max()
        # Active set: lags within 5% of max
        thr = max_S - 0.05 * max_S
        active = np.where(S >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(S)])
        
        G = _grad_topk(h, active)
        # Average gradient of active lags
        d = -G.mean(axis=0)
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        
        # Line search
        d = d / norm
        # Find best step
        S0 = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = np.correlate(d, d, mode='full')
        
        # Constraints on step size
        alpha_max = np.inf
        pos = d > 1e-10
        neg = d < -1e-10
        if np.any(pos):
            alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))
        alpha_max = min(alpha_max, 1.0)
        
        # Parabolic search (the max of S0 + a*T - a²*C is unimodal in a for small a)
        # Actually it's not unimodal in general. Use golden section on [0, alpha_max]
        if alpha_max < 1e-10:
            break
        g = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = 0.0, alpha_max
        c1 = b - g * (b - a)
        c2 = a + g * (b - a)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(50):
            if b - a < 1e-12:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - g * (b - a)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + g * (b - a)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        best_alpha = (a + b) / 2.0
        h = h + best_alpha * d
        h = _project(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5 - tol:
            best_c5 = c5
            best_h = h.copy()
        elif c5 >= best_c5:
            # No improvement, try smaller step
            h = best_h.copy()
            break
    return best_h, best_c5

def _binary_sa(h_init, deadline, max_iters=10000, rng=None):
    n = len(h_init)
    target = n // 2
    if rng is None:
        rng = np.random.default_rng(123)
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.003
    T_min = 0.00005
    cooling = 0.9998
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    patience = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 5000:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _pair_refine(h_init, deadline, max_rounds=100):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(55)
    stall = 0
    
    while time.time() < deadline and stall < max_rounds:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(100):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            for sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                # Line search
                S0 = np.correlate(h, 1.0 - h, mode='full')
                T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
                C = np.correlate(d, d, mode='full')
                # Max step
                am = 0.49
                if d[i] > 0:
                    am = min(am, (1.0 - h[i]) / d[i])
                else:
                    am = min(am, h[i] / (-d[i]))
                if d[j] > 0:
                    am = min(am, (1.0 - h[j]) / d[j])
                else:
                    am = min(am, h[j] / (-d[j]))
                if am < 1e-10:
                    continue
                g = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, am
                c1 = b - g * (b - a)
                c2 = a + g * (b - a)
                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
                for _ in range(40):
                    if b - a < 1e-12:
                        break
                    if f1 < f2:
                        b, c2, f2 = c2, c1, f1
                        c1 = b - g * (b - a)
                        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
                    else:
                        a, c1, f1 = c1, c2, f2
                        c2 = a + g * (b - a)
                        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
                ba = (a + b) / 2.0
                c5_try = float(np.max(S0 + ba * T - ba * ba * C)) * dx
                if c5_try < best_c5 - 1e-14:
                    h = h + ba * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 4))))
        if rng.random()
