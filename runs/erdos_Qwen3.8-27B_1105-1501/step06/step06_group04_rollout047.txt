    cons = [
        {'type': 'eq', 'fun': lambda h: np.array([np.sum(h) - target]),
         'jac': lambda h: np.array([np.ones(len(h))])}
    ]
    bounds = [(0.0, 1.0)] * len(h0)
    try:
        res = minimize(lambda h: _lse_obj_fun(h, beta, target), h0,
                       jac=lambda h: _lse_obj_grad(h, beta, target),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h_init) - 1e-12:
            return h_new
    except Exception:
        pass
    return h_init

def _line_search_exact(h, d, dx):
    n = len(h)
    if np.max(np.abs(d)) < 1e-15:
        return 0.0, compute_c5(h)
    self_sum = float(np.sum(d))
    d = d - self_sum / n
    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = np.inf
    t_lo = -np.inf
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / (-d[neg]))))
    t_lo = max(t_lo, -1.0)
    t_hi = min(t_hi, 1.0)
    if t_hi - t_lo < 1e-14:
        return 0.0, compute_c5(h)
    R0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    def f_of_t(t):
        return float(np.max(R0 + t * T - t * t * Q) * dx)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = f_of_t(c1)
    f2 = f_of_t(c2)
    for _ in range(80):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = f_of_t(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = f_of_t(c2)
    best_t = (a + b) / 2.0
    best_f = f_of_t(best_t)
    h_try = h + best_t * d
    h_try = np.clip(h_try, 0.0, 1.0)
    actual = compute_c5(h_try)
    if actual < best_f + 1e-12:
        return best_t, actual
    mid_t = (t_lo + t_hi) / 2.0
    h_mid = np.clip(h + mid_t * d, 0.0, 1.0)
    mid_f = compute_c5(h_mid)
    if mid_f < actual:
        return mid_t, mid_f
    return best_t, actual

def line_polish(h, target, deadline, max_iters=5000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.default_rng(12345)
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if stall > 200:
            h = project_feasible(h + rng.normal(0, 0.02, n), target)
            cur_c5 = compute_c5(h)
            stall = 0
            if cur_c5 > best_c5:
                h = best_h.copy()
                cur_c5 = best_c5
            continue
        if it % 50 == 0 and time.time() > deadline - 2:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        max_s = float(np.max(scores))
        thr = max_s - 0.01 * max_s
        active = np.where(scores >= thr)[0]
        if len(active) < 1:
            active = np.array([np.argmax(scores)])
        G = grad_correlation(h)
        top_k = min(len(active), 8)
        idx = rng.choice(len(active), size=min(top_k, len(active)), replace=False)
        sel = active[idx]
        grad_avg = G[sel].mean(axis=0) * dx
        grad_avg -= np.mean(grad_avg)
        norm = np.linalg.norm(grad_avg)
        if norm < 1e-14:
            stall += 1
            continue
        d = -grad_avg / norm
        self_sum = float(np.sum(d))
        d -= self_sum / n
        max_pos = np.min((1.0 - h) / np.maximum(d, 1e-15)) if np.any(d > 0) else 1.0
        max_neg = np.min(h / np.maximum(-d, 1e-15)) if np.any(d < 0) else 1.0
        scale = min(max_pos, max_neg, 0.5)
        if scale > 1e-10:
            d *= scale
        t, new_c5 = _line_search_exact(h, d, dx)
        if new_c5 < best_c5 - 1e-13:
            h = project_feasible(h + t * d, target)
            cur = compute_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
                h = best_h.copy()
        else:
            stall += 1
    return best_h, best_c5

def pair_polish(h, target, deadline, max_rounds=3000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.default_rng(777)
    stall = 0
    it = 0
    while time.time() < deadline and it < max_rounds:
        it += 1
        if stall > 150:
            h = project_feasible(h + rng.normal(0, 0.01, n), target)
            stall = 0
            continue
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        if len(interior) < 2:
            break
        i = interior[rng.integers(len(interior))]
        j = interior[rng.integers(len(interior))]
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
        t, new_c5 = _line_search_exact(h, d, dx)
        if new_c5 < best_c5 - 1e-13:
            h = project_feasible(h + t * d, target)
            cur = compute_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
                h = best_h.copy()
        else:
            stall += 1
    return best_h, best_c5

def resample_to(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = _sine_init(122, 3, 0.5)

    n0 = len(h0)
    if n0 % 2 != 0:
        h0 = resample_to(h0, n0 + 1)
        n0 = len(h0)
    target = n0 / 2.0
    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: LSE-SLSQP with multiple betas
    d1 = min(deadline, t0 + budget_s * 0.30)
    betas = [15, 40, 100, 250, 600, 1500]
    for beta in betas:
        if time.time() > d1 - 2:
            break
        h_try = solve_lse_slqp(best_h, target, beta, maxiter=300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 2: Minimax SLSQP
    d2 = min(deadline, t0 + budget_s * 0.45)
    for attempt in range(3):
        if time.time() > d2 - 3:
            break
        h_try = solve_minimax_slqp(best_h, target, maxiter=250)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
        else:
            break

    # Phase 3: Line search polish
    d3 = min(deadline, t0 + budget_s * 0.65)
    h3, c5_3 = line_polish(best_h, target, d3, max_iters=8000)
    if c5_3 < best_c5 - 1e-12:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair polish
    d4 = min(deadline, t0 + budget_s * 0.78)
    h4, c5_4 = pair_polish(best_h, target, d4, max_rounds=5000)
    if c5_4 < best_c5 - 1e-12:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution
    d5 = min(deadline, t0 + budget_s * 0.90)
    n_best = len(best_h)
    tried_n = {n_best}
    for dn in [-4, 4, -6, 6, -2, 2, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20]:
        if time.time() > d5 - 5:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 250:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample_to(best_h, n_try)
        tgt_r = n_try / 2.0
        # Quick LSE on new resolution
        h_r2 = solve_lse_slqp(h_r, tgt_r, 100, maxiter=200)
        c5_r = compute_c5(h_r2)
        if c5_r < best_c5 - 1e-12:
            # Further polish
            h_r3, c5_r3 = line_polish(h_r2, tgt_r, min(d5, time.time() + budget_s * 0.02), max_iters=3000)
            if c5_r3 < c5_r:
                h_r2, c5_r = h_r3, c5_r3
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r2.copy()
                n_best = n_try

    # Phase 6: Final LSE + minimax + line polish
    d6 = deadline
    if time.time() < d6 - 10:
        h_try = solve_lse_slqp(best_h, target, 400, maxiter=400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    if time.time() < d6 - 10:
        h_try = solve_minimax_slqp(best_h, target, maxiter=200)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    if time.time() < d6 - 3:
        h_f, c5_f = line_polish(best_h, target, d6, max_iters=5000)
        if c5_f < best_c5 - 1e-12:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        h_fb = _sine_init(122, 3, 0.0)
        return h_fb, compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
