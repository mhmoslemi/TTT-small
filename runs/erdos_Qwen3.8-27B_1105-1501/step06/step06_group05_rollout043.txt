def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    # Project v onto {x: sum(x) = target, 0 ≤ x ≤ 1}
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0, 1)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0, 1)
    # Fine-tune
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) < 1e-14:
            break
        if res > 0:
            mask = x < 1 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        adj = res / cnt
        x[mask] = np.clip(x[mask] + adj, 0, 1)
        res = target - float(np.sum(x))
    return x

def _c5_val(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def _proj(v, target):
    n = len(v)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(50):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:

        else:
            hi = mid

def _precomp_idx(n):

        idx2 = np.where((i - ks >= 0) & (i - ks < n))[0]
        info.append((idx1, idx2))
    return ks, info


def _compute_b(h, i, j, ks, info):

    b = np.zeros(2 * n - 1)

    b[i1] += om[i + ks[i1]]
    b[i2] -= h[i - ks[i2]]
    b[j1] -= om[j + ks[j1]]
    b[j2] += h[j - ks[j2]]
    return b

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)
