                g[d:] += -h[:m]
    return G

def _full_grad_matrix(h, n):
    """Compute full (2n-1, n) gradient matrix."""
    return _grad_rows(h, np.arange(2 * n - 1), n)

def _lse_optimize(h, deadline, beta_init=150, beta_max=8000, lr_init=0.02, n_phases=3):
    """Log-sum-exp smoothed optimization with Adam and Nesterov momentum."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        phase_budget = (deadline - time.time()) / (n_phases - phase)
        beta_start = beta_init * (1 + phase * 2)
        beta_end = min(beta_max, beta_init * (1 + (phase + 1) * 2) * 3)
        lr = lr_init / (1 + phase * 0.3)
        
        m = np.zeros(n)
        v = np.zeros(n)
        t = 1
        momentum = 0.9
        ph_h = h.copy()
        
        max_iters = 3000
        for it in range(max_iters):
            if time.time() > deadline:
                break
            frac = it / max(max_iters - 1, 1)
            beta = beta_start + (beta_end - beta_start) * frac
            lr_step = lr * (1.0 - 0.4 * frac)
            
            S = _corr_full(h)
            S_max = S.max()
            eS = np.exp(beta * (S - S_max))
            w = eS / eS.sum()
            
            G = _full_grad_matrix(h, n)
            grad = beta * (w @ G)
            grad = grad - grad.mean()
            
            m = momentum * m + (1 - momentum) * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            step = lr_step * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            h_new = h - step
            h_new = _project(h_new, target)
            c5 = _exact_c5(h_new)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
            t += 1
    
    return best_h, best_c5

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.05 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 25:
        combined = sorted_idx[:25]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, K=15):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    prev_dir = np.zeros(n)
    while time.time() < deadline:
        if stall > 50:
            if restart_ctr < 12:
                scale = 0.008 + 0.008 * restart_ctr * 0.4
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.06
                stall = 0
                restart_ctr += 1
                prev_dir = np.zeros(n)
                continue
            else:
                break
        S = _corr_full(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.04
            if radius < 1e-6:
                break
            continue
        # Nesterov-like: use previous direction as momentum
        p = 0.7 * p + 0.3 * prev_dir * radius
        prev_dir = p / max(np.max(np.abs(p)), 1e-12) * radius
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.05, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.92, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    cooldown = 0.99995
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        for _ in range(150):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _pair_refine(h, deadline, max_passes=150):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(888)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_full(h)
        G = _full_grad_matrix(h, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _multi_optimize(h, deadline, rng):
    """Combined optimization: alternate between LSE and LP."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # LSE first
    d_lse = min(deadline, time.time() + (deadline - time.time()) * 0.4)
    h_lse, c5_lse = _lse_optimize(h, d_lse, beta_init=100, beta_max=4000, lr_init=0.015, n_phases=3)
    if c5_lse < best_c5:
        best_c5 = c5_lse
        best_h = h_lse.copy()
    
    # LP refinement
    if time.time() < deadline:
        h_lp, c5_lp = _lp_loop(best_h, deadline, radius_init=0.1, K=15)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (8% time)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
        for ph in [0.0, 0.8, 1.6, 2.4, 3.2]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(10):
        starts.append(_block_init(n0, rng, rng.integers(3, 18)))
    # Add perturbed versions of initial
    for sc in [0.05, 0.1, 0.15]:
        starts.append(_project(h0 + rng.normal(0, sc, n0), n0 / 2.0))
    
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.005)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.08, K=12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: LSE + LP on best (25% time)
    d2 = t0 + budget_s * 0.33
    h2, c5_2 = _multi_optimize(best_h, d2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Long LP refinement (20% time)
    d3 = t0 + budget_s * 0.53
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.12, K=20)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP polish (12% time)
    d4 = t0 + budget_s * 0.65
    h4a, c5_4a = _binary_sa(best_h, t0 + budget_s * 0.58, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, radius_init=0.06, K=15)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Pair refinement (10% time)
    d5 = t0 + budget_s * 0.75
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=200)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (13% time)
    d6 = t0 + budget_s * 0.88
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20, -24, 24]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            rem = d6 - time.time()
            if rem < budget_s * 0.01:
                continue
            frac1 = 0.4
            sub_dl1 = min(d6, time.time() + rem * frac1)
            h_r, c5_r = _lp_loop(h_r, sub_dl1, radius_init=0.08, K=12)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            if time.time() < d6:
                sub_dl2 = min(d6, time.time() + rem * 0.3)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()
                    h_r = h_r2
                if time.time() < d6:
                    h_r3, c5_r3 = _lp_loop(h_r, min(d6, time.time() + rem * 0.2),
                                           radius_init=0.04, K=12)
                    if c5_r3 < best_c5:
                        best_c5 = c5_r3
                        best_h = h_r3.copy()

    # Phase 7: Aggressive perturbation + re-opt (5% time)
    d7 = t0 + budget_s * 0.93
    scale = 0.02
    for attempt in range(400):
        if time.time() > d7:
            break
        if best_c5 < 0.3800:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.04, K=12)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.1, scale * 1.15)

    # Phase 8: Final LSE + LP polish (3% time)
    d8 = t0 + budget_s * 0.965
    if time.time() < d8:
        h8, c5_8 = _lse_optimize(best_h, d8, beta_init=300, beta_max=6000, lr_init=0.008, n_phases=2)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final pair + LP polish
    if time.time() < deadline:
        h9, c5_9 = _pair_refine(best_h, min(deadline, time.time() + budget_s * 0.01), max_passes=50)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.02, K=10)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
