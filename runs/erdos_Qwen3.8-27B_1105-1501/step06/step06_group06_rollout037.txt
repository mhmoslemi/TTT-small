  where the sum is over i such that 0 ≤ i ≤ n-1 and 0 ≤ i + (p-n+1) ≤ n-1.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grads(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            G[k, :m] += (1.0 - h[s:s + m])
            G[k, s:s + m] += (-h[:m])
        else:
            g = -s
            G[k, g:g + m] += (1.0 - h[:m])
            G[k, :m] += (-h[g:g + m])
    G *= dx
    return G

def _logsumexp_gradient(h, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*c_k)) where c_k = corr_k * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
    G = _corr_grads(h)
    grad = G.T @ w
    return grad, c

def _lsqpgd_decrease(h, beta, lr, max_iter=5000, deadline=None):
    """Projected gradient descent on log-sum-exp objective"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        grad, c = _logsumexp_gradient(h, beta)
        # Lipschitz constant estimate: beta * 2 * (max row sum of |G|) * dx
        # Each |G[k,j]| <= 2, and we have 2n-1 rows, so |grad| <= 2*(2n-1)*dx
        # L ≈ beta * 2 * (2n) * dx = beta * 4
        # Actually more precisely: the gradient of f_beta has Lipschitz constant
        # at most beta * (hessian norm) which is bounded by beta * 2 * (2n-1) * dx
        lr_local = lr
        for _bt in range(50):
            h_new = h - lr_local * grad
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            c5_old = _exact_c5(h)
            if c5_new < c5_old:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            lr_local *= 0.5
        else:
            break
    return best_h, best_c5

def _slsqp_optimize(h0, deadline, max_outer=20):
    # Same as before but with improved active set selection
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        K = min(40, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]

        t_init = cur_c5 + 1e-4
        x0 = np.concatenate([h, [t_init]])

        def obj_fn(x, n=n):
            return x[n]

        def obj_grad(x, n=n):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x, top_k=top_k, n=n, dx=dx):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]

        def ineq_jac(x, top_k=top_k, n=n, dx=dx):
            hx = x[:n]
            G = _corr_grads(hx)
            J = np.zeros((len(top_k), n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J

        def eq_fn(x, n=n, target=target):
            return np.sum(x[:n]) - target

        def eq_jac(x, n=n):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]

        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 60, 'ftol': 1e-14})

        if res.success or res.status == 0:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            # Try with fewer constraints
            if K > 10:
                K = 10
                continue
            break

    return best_h, best_c5

def _coordinate_descent(h, deadline, max_sweeps=200):
    """Steepest coordinate descent: find best pair (i,j) to transfer mass"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        G = _corr_grads(h)
        
        # For each pair (i, j), compute directional derivative of max
        # Moving δ from h[i] to h[j]: direction d has d[i] = -1, d[j] = 1
        # Change in c_k: δ * (G[k,j] - G[k,i])
        # Change in max ≈ max_k (c_k + δ*(G[k,j] - G[k,i])) - max_k c_k
        # For small δ, ≈ δ * max_k (G[k,j] - G[k,i]) (if the max is unique)
        # More precisely, for the active set, ≈ δ * max_active (G[k,j] - G[k,i])
        
        # Use a simpler heuristic: minimize sum_k c_k^2 (proxy for max)
        # d/dc sum(c^2) = 2 * sum_k c_k * (G[k,j] - G[k,i])
        # = 2 * (c @ G[:,j] - c @ G[:,i])
        
        cG = c @ G  # length n: (cG)[j] = sum_k c_k * G[k,j]
        
        # Direction: transfer from high cG to low cG
        # Reducing h[i] and increasing h[j] changes objective by δ*(cG[j] - cG[i])
        # We want this to be negative: cG[j] < cG[i]
        
        i = int(np.argmax(cG))
        j = int(np.argmin(cG))
        
        if i == j:
            break
        
        # Line search on the segment from h to h + δ*(e_j - e_i)
        # Constraints: 0 ≤ h[i] - δ ≤ 1 and 0 ≤ h[j] + δ ≤ 1
        # So δ ∈ [max(-h[j], h[i]-1), min(h[i], 1-h[j])]
        delta_max = min(h[i], 1.0 - h[j])
        delta_min = max(-h[j], h[i] - 1.0)
        
        if delta_max - delta_min < 1e-14:
            break
            
        # Try several values of δ
        best_delta = 0.0
        best_val = cur_c5
        for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            d = frac * (delta_max - delta_min) + delta_min
            if d <= 0:
                continue
            h_try = h.copy()
            h_try[i] -= d
            h_try[j] += d
            if h_try[i] < -1e-10 or h_try[j] > 1 + 1e-10:
                continue
            h_try = np.clip(h_try, 0, 1)
            h_try = _project(h_try, target)
            c5_try = _exact_c5(h_try)
            if c5_try < best_val - 1e-14:
                best_val = c5_try
                best_delta = d
        
        if best_delta > 0:
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0, 1)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # Try smaller steps
            found = False
            for frac in [0.01, 0.05]:
                d = frac * (delta_max - delta_min)
                if d <= 0:
                    continue
                h_try = h.copy()
                h_try[i] -= d
                h_try[j] += d
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < cur_c5 - 1e-14:
                    h = h_try
                    cur_c5 = c5_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                # Try multiple random pairs
                rng = np.random.default_rng(sweep + 42)
                found_any = False
                for _ in range(20):
                    ii = int(rng.integers(n))
                    jj = int(rng.integers(n))
                    if ii == jj:
                        continue
                    dm = min(h[ii], 1.0 - h[jj])
                    if dm < 1e-6:
                        continue
                    for frac in [0.1, 0.3, 0.5]:
                        d = frac * dm
                        h_try = h.copy()
                        h_try[ii] -= d
                        h_try[jj] += d
                        h_try = _project(h_try, target)
                        c5_try = _exact_c5(h_try)
                        if c5_try < cur_c5 - 1e-14:
                            h = h_try
                            cur_c5 = c5_try
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            found_any = True
                            break
                    if found_any:
                        break
                if not found_any:
                    break
    
    return best_h, best_c5

def _active_descent_v2(h_init, deadline, max_iter=500):
    """Improved active descent with bundle directions"""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.05
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            # Perturbation
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.05
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            continue
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = cur_c5
        # Active set: all lags within 1e-3 of max
        active = np.where(corr > M - 5e-4)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:10]
        elif len(active) > 30:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:30]]
        
        G = _corr_grads(h)
        G_act = G[active]
        
        # Solve: minimize t subject to G_act @ d >= 0 (don't increase active constraints)
        # and ||d|| <= radius, sum(d) = 0
        # Equivalent: find d in null space of G_act (or negative half-space) with small norm
        # that decreases the "second-order" term
        
        # Simpler: use the negative of the average gradient of active constraints
        # projected to sum-zero
        avg_grad = G_act.mean(axis=0)
        avg_grad -= avg_grad.mean()
        
        # Also try null space directions of the active gradients
        # For small active set, compute null space
        if len(active) <= 20:
            # QR decomposition to find null space
            Q, R = np.linalg.qr(G_act.T, mode='complete')
            # Null space of G_act is spanned by last (n - rank) columns of Q
            rank = np.sum(np.abs(np.diag(R)) > 1e-10)
            if n - rank > 0:
                null_vecs = Q[:, rank:]
                # Project each to sum-zero
                for nv in null_vecs:
                    nv_zero = nv - nv.mean()
                    nv_norm = np.linalg.norm(nv_zero)
                    if nv_norm > 1e-10:
                        nv_zero /= nv_norm * nv_norm  # scale by 1/norm
                        # Actually let's just normalize
                        nv_zero = (nv - nv.mean()) / max(np.linalg.norm(nv - nv.mean()), 1e-10)
                        # Try this direction
                        d_try = nv_zero * radius
                        # Line search
                        _h, c5_try = _line_search_small(h, d_try, radius)
                        if c5_try < cur_c5 - 1e-14:
                            cur_c5 = c5_try
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            stall = 0
                            break
                else:
                    # Try the average gradient direction
                    grad_norm = np.linalg.norm(avg_grad)
                    if grad_norm > 1e-12:
                        d_try = -avg_grad / grad_norm * radius
                        h_try = _project(h + d_try, target)
                        c5_try = _exact_c5(h_try)
                        if c5_try < cur_c5 - 1e-14:
                            h = h_try
                            cur_c5 = c5_try
                            if cur_c5 < best_c5:
                                best_c5 = cur_c5
                                best_h = h.copy()
                            stall = 0
                    else:
                        stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        if c5_try if 'c5_try' in dir() else cur_c5 >= cur_c5 - 1e-14:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                stall += 20

    return best_h, best_c5

def _line_search_small(h, d, radius):
    """Simple line search along direction d"""
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    # C[k] = -sum(d[i]*d[i+k]) for appropriate ranges
    C = np.zeros(2 * n - 1)
    for s in range(-(n-1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            C[k] = -np.sum(d[:m] * d[s:s+m])
        else:
            g = -s
            C[k] = -np.sum(d[g:g+m] * d[:m])
    
    c5_0 = float(np.max(S0)) * dx
    # Binary search for best t in [-radius, radius]
    t_lo, t_hi = -radius, radius
    # Check box constraints
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi <= t_lo:
        return h.copy(), c5_0
    
    best_t = 0.0
    best_val = c5_0
    for frac in np.linspace(0, 1, 20):
        t = t_lo + frac * (t_hi - t_lo)
        val = float(np.max(S0 + t * T + t*t * C)) * dx
        if val < best_val:
            best_val = val
            best_t = t
    
    h_new = h + best_t * d
    h_new = _project(h_new, n/2.0)
    return h_new, best_val

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _make_init(n, rng, style=0):
    if style == 0:
        # Smooth: 0.5 + a*sin
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 3) * np.arange(n) / n + rng.uniform(0, 2*np.pi))
        h += rng.normal(0, 0.03, n)
    elif style == 1:
        # Blocky
        h = np.zeros(n)
        pos = 0
        while pos < n:
            blen = max(1, int(rng.integers(2, n//4 + 1)))
            h[pos:pos+blen] = rng.uniform(0.5, 1.0) if rng.random() < 0.5 else rng.uniform(0.0, 0.5)
            pos += blen
    else:
        # Random
        h = rng.uniform(0, 1, n)
    return _project(h, n / 2.0)
