def _adam_smooth(h, deadline, tau_init=0.01, tau_end=0.001, lr=0.005, beta1=0.9, beta2=0.999):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    m_v = np.zeros(n)  # First moment
    v_v = np.zeros(n)  # Second moment
    t = 0
    tau = tau_init
    
    while time.time() < deadline:
        t += 1
        # Anneal temperature
        tau = max(tau_end, tau_init * (tau_end/tau_init) ** (t/200))
        
        S = _corr_scaled(h)
        M = S.max()
        shifted = (S - M) / tau
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()
        
        # Compute smooth gradient (weighted)
        grad = np.zeros(n)
        for k_idx in range(len(w)):
            if w[k_idx] < 1e-12:
                continue
            k = k_idx - (n - 1)
            if k >= 0:

                if m > 0:
                    grad[k:] += w[k_idx] * (1.0 - h[:m]) * dx
                    grad[:m] += w[k_idx] * (-h[k:]) * dx
            else:
                d = -k
                m = n - d
                if m > 0:
                    grad[:m] += w[k_idx] * (1.0 - h[d:d+m]) * dx
                    grad[d:] += w[k_idx] * (-h[:m]) * dx
        
        # Remove mean to maintain sum constraint
        grad -= grad.mean()
        
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        h_new = _project(h_new, target)
        
        new_c5 = _exact_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_binary(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    return b

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.03 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _sm_grad(h, n, tau):
    """Compute smooth gradient with temperature tau."""
    S = _corr_scaled(h)
    dx = 2.0 / n
    M = S.max()
    shifted = (S - M) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for ki in range(len(w)):
        wk = w[ki]
        if wk < 1e-12:
            continue
        k = ki - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:n] += wk * (1.0 - h[:m])
                grad[:m] -= wk * h[k:n]
        else:
            d = -k
            m = n - d
            if m > 0:
                grad[:m] += wk * (1.0 - h[d:d+m])
                grad[d:n] -= wk * h[:m]
    grad *= dx
    grad -= grad.mean()
    return grad

def _smooth_loop(h, deadline, tau_init=0.008, tau_min=0.001, lr=0.003):
    """Adam-like smooth gradient descent."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    tau = tau_init
    while time.time() < deadline:
        t += 1
        tau = max(tau_min, tau_init * max(0.999, tau_min / tau_init) ** t)
        grad = _sm_grad(h, n, tau)
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad ** 2
        m_hat = m_v / (1 - 0.9 ** t)
        v_hat = v_v / (1 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = h - step
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _lp_loop(h, deadline, radius_init=0.12, K=15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > 50:
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.08
            stall = 0
            continue
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            # Try smooth direction
            p = -_sm_grad(h, n, 0.02 * cur_c5)
            p = p / (np.linalg.norm(p) + 1e-15)
            p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.05, 0.25)
            else:
                radius = max(radius * 0.92, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.05
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng, T_init=0.008, T_min=0.00001):
    """Binary SA with swap moves."""
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        n_steps = 300
        for _ in range(n_steps):
            if time.time() > deadline:
                break
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * 0.9997)
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(888)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 40)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.3)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _thue_morse_init(n, rng):
    """Create a Thue-Morse-like sequence."""
    h = np.zeros(n)
    # Thue-Morse: t_i = parity of number of 1s in binary representation of i
    tm = np.array([(i & 1) ^ ((i >> 1) & 1) ^ ((i >> 2) & 1) ^ ((i >> 3) & 1) ^
                 ((i >> 4) & 1) ^ ((i >> 5) & 1) ^ ((i >> 6) & 1) ^ ((i >> 7) & 1)
                 for i in range(256)], dtype=float)
    # Tile and crop
    h = np.tile(tm[:n], (n // n + 1, ))[:n]
    # Adjust to have exactly n/2 ones
    h = _project(h, n / 2.0)
    return h
