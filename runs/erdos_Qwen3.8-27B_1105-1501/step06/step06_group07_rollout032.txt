def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid

    if abs(res) > 1e-12:

            x = np.clip(x, 0.0, 1.0)

def _generate_kronecker(n, alpha=None):
    """Generate a binary sequence using Kronecker sequence."""
    if alpha is None:
        alpha = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    m = n // 2

    for i in range(m):
        pos = int(np.floor((i + 0.5) * alpha * n)) % n
        positions[pos] = 1.0
    # Handle possible collisions

        occupied = set(np.where(positions > 0)[0].tolist())

        chosen = rng_fill.choice(len(available), size=remaining, replace=False)
        for c in chosen:

    return positions
