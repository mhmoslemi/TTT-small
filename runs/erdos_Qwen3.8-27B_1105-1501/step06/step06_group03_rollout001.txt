    grad_smooth = w[top_idx] @ G
    grad_smooth -= grad_smooth.mean()
    if np.abs(grad_smooth).max() < 1e-12:
        return None
    return -grad_smooth

def _local_opt(h_init, deadline, radius_init=0.12, K=15, use_smooth=True):
    """Main local optimization loop."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(12345)
    tau = 0.015 * cur_c5
    
    while time.time() < deadline:
        if stall > 80:
            if restart_ctr < 15:
                scale = 0.008 + 0.008 * restart_ctr * 0.4
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                tau = 0.015 * cur_c5
                continue
            else:
                break
        
        S = _corr_scaled(h)
        
        # Try LP direction
        p = _lp_direction(h, S, n, K, radius)
        
        # Fallback to smooth direction
        if (p is None or float(np.max(np.abs(p))) < 1e-13) and use_smooth:
            p = _smooth_direction(h, S, n, tau)
            if p is not None:
                p_norm = np.max(np.abs(p))
                if p_norm > 0:
                    p = p / p_norm * radius
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if use_smooth and stall % 4 == 0:
                tau *= 0.75
            if radius < 1e-7:
                break
            continue
        
        dx = 2.0 / n
        alpha, new_c5 = _line_search(h, p, dx, radius)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.6:
                radius = min(radius * 1.08, 0.3)
            elif ratio > 0.25:
                radius = min(radius * 1.03, 0.25)
            else:
                radius = max(radius * 0.92, 0.015)
        else:
            stall += 1
            radius *= 0.72
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.04
    
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng, max_swaps=3):
    """Binary simulated annealing with multi-swap moves."""
    n = len(h_init)
    target = n // 2  # For binary, exactly n/2 ones
    dx = 2.0 / n
    
    # Round to binary
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.005
    T_min = 0.000003
    cooldown = 0.99985
    patience = 0
    max_patience = 20000
    
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])
    
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        
        # Do a batch of moves
        batch_size = min(300, max(50, int(50000 / max(1, n))))
        for _ in range(batch_size):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            
            # Choose move type: single swap or multi-swap
            m = 1
            r = rng.random()
            if r < 0.7:
                m = min(1, len(ones_list), len(zeros_list))
            elif r < 0.9:
                m = min(2, len(ones_list), len(zeros_list))
            else:
                m = min(max_swaps, len(ones_list), len(zeros_list))
            
            if m == 0:
                break
            
            # Pick m from ones and m from zeros
            ones_pick = rng.choice(len(ones_list), size=m, replace=False)
            zeros_pick = rng.choice(len(zeros_list), size=m, replace=False)
            
            swap_ones = [ones_list[i] for i in ones_pick]
            swap_zeros = [zeros_list[i] for i in zeros_pick]
            
            # Apply
            for i in swap_ones:
                h[i] = 0.0
            for j in swap_zeros:
                h[j] = 1.0
            
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = list(np.where(h > 0.5)[0])
                zeros_list = list(np.where(h <= 0.5)[0])
            else:
                # Undo
                for i in swap_ones:
                    h[i] = 1.0
                for j in swap_zeros:
                    h[j] = 0.0
        
        T = max(T_min, T * cooldown)
    
    return best_h, best_c5

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks-1, n-1), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(len(boundaries) - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b+1]] = 1.0
    return _project(h, n / 2.0)

def _pair_refine(h, deadline, max_passes=150):
    """Pair move refinement using gradient guidance."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    improved_any = False
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        
        # Compute total gradient
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        
        n_int = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
        improved_any = True
    
    return best_h, best_c5
