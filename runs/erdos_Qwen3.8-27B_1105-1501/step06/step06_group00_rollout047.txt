def _exact_c5(h):
    n = len(h)
    if n == 0:
        return 1e10
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad(h, w):
    """Compute gradient of soft-max objective. w is the weight vector for each lag."""
    n = len(h)
    L = 2 * n - 1
    # w has length L, indexed from 0 to L-1
    # Lag for index i is (i - (n-1))
    grad = np.zeros(n)
    for i in range(L):
        wi = w[i]
        if abs(wi) < 1e-15:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            # corr(lag) = sum_{r=0}^{n-1-lag} h[r]*(1-h[r+lag])
            m = n - lag
            # d/dh[k] for k in [0, m-1]: (1-h[k+lag])
            if m > 0:
                grad[:m] += wi * (1.0 - h[lag:lag+m])
            # d/dh[k] for k in [lag, n-1]: -h[k-lag]
            if lag < n:
                grad[lag:] += -wi * h[:m]
        else:
            m_ = -lag  # positive magnitude
            m = n - m_
            # corr(-m_) = sum_{r=0}^{n-1-m_} h[r+m_]*(1-h[r])
            # d/dh[k] for k in [m_, n-1]: (1-h[k-m_])
            if m > 0:
                grad[m_:] += wi * (1.0 - h[:m])
            # d/dh[k] for k in [0, m-1]: -h[k+m_]
            if m > 0:
                grad[:m] += -wi * h[m_:]
    return grad
