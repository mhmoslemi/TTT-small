import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-12: break
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0: break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def _feasible_range(h, d):
    """Find L, U such that for all t in [L,U], 0 <= h + t*d <= 1."""
    L = -1e18
    U = 1e18
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):

        U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))

        U = min(U, float(np.min((1.0 - h[neg]) / d[neg])))
    return max(L, -10.0), min(U, 10.0)

    """Direction from gradient of weighted top-m active overlaps."""
    n = len(h)

    for i, r in enumerate(idx):
        k = r - (n - 1)
        G += w[i] * _grad_shift(h, k)
    return -G  # Negative gradient (descent direction)

        g[k:] += (-h[:m])
    else:
        m = -k
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] += (-h[m:])
    return g

                j = pos_idx[rng.integers(len(pos_idx))]

            else:

        else:

        if nrm < 1e-14:

        d = d / nrm

        if U - L < 1e-12:

            nf = _exact_c5(hn)
            if nf < best_f:

                best_h = hn.copy()
            if nf <= cur_f + uphill:

                cur_f = nf
                corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            stall = 0
    
    return best_h, best_f

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-12:
            break
        mask = (x < 1.0 - 1e-12) if res > 0 else (x > 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
    return x


def _grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:])
            g[k:] += (-h[:m])
    else:
        m = -k
        if m < n:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] += (-h[m:])
    return g
