    n = len(h)

def solve_qp_direction(h, target_raw, rho, lam_z=0.85, lam_norm=0.15, maxiter=200):
    """Solve the tangent QP to find direction d."""
    n = len(h)
    C = 2 * n - 1
    R = np.correlate(h, 1.0 - h, mode='full')
    G = compute_grad_matrix(h)
    
    # Bounds for d
    lo_d = -np.minimum(h, rho)
    hi_d = np.minimum(1.0 - h, rho)
    
    # Initial z
    z_init = np.zeros(C)
    
    x0 = np.concatenate([np.zeros(n), z_init])
    
    # Bounds
    bounds = list(zip(lo_d, hi_d))
    for _ in range(C):
        bounds.append((0.0, 10.0))  # z bounds
    
    # Constraints
    # Inequality: G @ d + R - target_raw - z <= 0
    # Equality: sum(d) = 0
    
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x[:n]), 'jac': lambda x: np.concatenate([np.ones(n), np.zeros(C)])},
    ]
    
    # For SLSQP, we need to handle inequality constraints
    # G @ d + R - target - z <= 0 means -(G @ d + R - target - z) >= 0
    # SLSQP requires g(x) >= 0
    # So: target - R - G @ d + z >= 0 for each row
    
    # This is C separate inequality constraints, or we can vectorize
    # SLSQP supports vector constraint functions
    
    def ineq_fun(x):
        d = x[:n]
        z = x[n:]
        return target_raw - R - G @ d + z
    
    def ineq_jac(x):
        # Shape: (C, n+C)
        J = np.zeros((C, n + C))
        J[:, :n] = -G
        J[:, n:] = np.eye(C)
        return J

        d = x[:n]
        z = x[n:]
        return lam_z * np.mean(z) / z_scale + lam_norm * (d @ d)

        d = x[:n]
        z = x[n:]
        grad_d = 2.0 * lam_norm * d
        grad_z = np.full(C, lam_z / (C * z_scale))
        return np.concatenate([grad_d, grad_z])

Actually, I realize I should also think about what happens when the QP is infeasible (no direction can satisfy all active row constraints simultaneously). In this case, the QP will have z > 0 for some rows, meaning the target is not achievable. The direction d still has meaning: it's the direction that best reduces the weighted violations. The line search will then find the best step along this direction.
