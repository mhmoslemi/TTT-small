def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, raw, beta):
    n = len(h)
    dx = 2.0 / n
    scaled = raw * beta * dx
    max_s = float(np.max(scaled))
    exp_s = np.exp(scaled - max_s)
    w = exp_s / float(np.sum(exp_s))
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        k = n - 1 + d
        wk = w[k] * dx * beta
        if wk < 1e-14:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wk * (1.0 - h[:m])
                g[:m] -= wk * h[d:]
        else:
            ga = -d
            m = n - ga
            if m > 0:
                g[:m] += wk * (1.0 - h[ga:])
                g[ga:] -= wk * h[:m]
    g -= np.mean(g)
    return g

        h_p, c5_p = _optimize(h_p, sub_dl, radius_init=0.15, beta_init=beta_try, max_stall=80, n_dirs=5)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: Basin hopping (15%)
    d4 = soft_deadline
    scale = 0.02
    n_restarts = 0
    while time.time() < d4 and n_restarts < 200:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.004)
        h_t, c5_t = _optimize(h_pert, sub_dl, radius_init=0.10, beta_init=25.0, max_stall=40, n_dirs=4)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Phase 5: Final resolution check + polish (remaining)
    n_best = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > deadline - 5:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline - 3, time.time() + budget_s * 0.010)
        h_r, c5_r = _optimize(h_r, sub_dl, radius_init=0.10, beta_init=25.0, max_stall=40, n_dirs=4)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if deadline - time.time() > 3:
        h_fin = _project(best_h.copy(), len(best_h) / 2.0)
        h_fin, c5_fin = _optimize(h_fin, deadline, radius_init=0.08, beta_init=30.0, max_stall=50, n_dirs=4)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
