def _project(v, target):
    n = len(v)
    v = np.where(np.isfinite(v), v, target / n)
    v = np.clip(v, -10.0, 10.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _full_grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            length = n - d
            if length > 0:
                G[r, d:] += (1.0 - h[:length])
                G[r, :length] += -h[d:]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[r, :length] += (1.0 - h[g:])
                G[r, g:] += -h[:length]
    return G

def _slsqp_fun(x, n, dx):
    return x[0]

def _slsqp_fprime(x, n, dx):
    g = np.zeros(n + 1)
    g[0] = 1.0
    return g

def _slsqp_ineq_fun(x, n, dx):
    h = x[1:]
    M = x[0]
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    return M - C

def _slsqp_ineq_jac(x, n, dx):
    h = x[1:]
    G = _full_grad_matrix(h)
    m = 2 * n - 1
    J = np.zeros((m, n + 1))
    J[:, 0] = 1.0
    J[:, 1:] = -dx * G
    return J

def _slsqp_eq_fun(x, n, dx):
    return np.array([np.sum(x[1:]) - n / 2.0])

def _slsqp_eq_jac(x, n, dx):
    J = np.zeros((1, n + 1))
    J[0, 1:] = 1.0
    return J

def _run_slsqp(h_init, deadline, maxiter=150, top_k=None):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h_init.copy(), target)
    C0 = _corr_scaled(h)
    M0 = float(np.max(C0))
    if not np.isfinite(M0) or M0 > 0.99:
        return h, _compute_c5(h)
    x0 = np.zeros(n + 1)
    x0[0] = M0 + 0.05
    x0[1:] = h
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * n
    cons = [
        {'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac, 'args': (n, dx)},
        {'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac, 'args': (n, dx)},
    ]
    try:
        res = minimize(_slsqp_fun, x0, method='SLSQP', jac=_slsqp_fprime,
                       bounds=bounds, constraints=cons, args=(n, dx),
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        res = None
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    if res is not None:
        hx = res.x[1:]
        hx = _project(hx, target)
        c5x = _compute_c5(hx)
        if c5x < best_c5:
            best_c5 = c5x
            best_h = hx.copy()
        if res.success:
            hx2 = np.clip(res.x[1:], 0.0, 1.0)
            hx2 = _project(hx2, target)
            c5x2 = _compute_c5(hx2)
            if c5x2 < best_c5:
                best_c5 = c5x2
                best_h = hx2.copy()
    return best_h, best_c5

def _lse_grad(h, beta, dx):
    n = len(h)
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    Cmax = float(np.max(C))
    z = beta * (C - Cmax)
    z = np.clip(z, -500, 500)
    exp_z = np.exp(z)
    p = exp_z / float(np.sum(exp_z))
    G = _full_grad_matrix(h)
    grad = dx * (G.T @ p)
    return grad, C

def _lse_descent(h_init, deadline, betas=None):
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    if betas is None:
        betas = [8, 15, 30, 60, 120, 250, 500]
    for beta in betas:
        if time.time() > deadline:
            break
        steps = max(3, int(60 * (10.0 / beta)))
        lr = 0.8 / (beta * 0.02)
        for step in range(steps):
            if time.time() > deadline:
                break
            grad, C = _lse_grad(h, beta, 2.0 / n)
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            d = -grad / gnorm
            for alpha in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
                h_try = _project(h + alpha * d, target)
                c5_try = _compute_c5(h_try)
                if c5_try > 0.45:
                    continue
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                h = h_try
                break
    h = _project(h, target)
    c5_final = _compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * Q)) * dx
    return opt_t, opt_val

def _active_direction(h, radius, top_k=10):
    n = len(h)
    dx = 2.0 / n
    C = _corr_scaled(h)
    sorted_idx = np.argsort(-C)
    K = min(top_k, len(C))
    active = sorted_idx[:K]
    G = _full_grad_matrix(h)
    avg_grad = G[active].mean(axis=0)
    avg_grad -= avg_grad.mean()
    norm = np.linalg.norm(avg_grad)
    if norm < 1e-14:
        return None
    d = -avg_grad / norm * radius
    d = _clip_dir(h, d, radius)
    return d

def _clip_dir(h, d, radius):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        max_t = np.min((1.0 - h[pos]) / d[pos])
        d[pos] *= max_t
    if np.any(neg):
        max_t = np.min(h[neg] / (-d[neg]))
        d[neg] *= -max_t
    max_abs = np.max(np.abs(d))
    if max_abs > radius:
        d *= radius / max_abs
    return d

def _line_polish(h, deadline, radius=0.05, top_k=10, max_iters=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    rng = np.random.default_rng(123)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            break
        d = _active_direction(h, radius, top_k)
        if d is None:
            d = rng.normal(0, 1.0, n)
            d -= d.mean()
            d = _clip_dir(h, d, radius)
        if np.max(np.abs(d)) < 1e-14:
            stall += 1
            radius *= 0.9
            continue
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < best_c5 - 1e-14:
            h = _project(h + alpha * d, target)
            cur = _compute_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
                radius = min(radius * 1.1, 0.3)
            else:
                stall += 1
        else:
            stall += 1
            radius *= 0.9
            if radius < 0.005:
                radius = 0.05
    return best_h, best_c5

def _pair_polish(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    rng = np.random.default_rng(42)
    stall = 0
    while time.time() < deadline and stall < 30:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(300):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.4)
            if c5_try < best_c5 - 1e-14:
                h = _project(h + alpha * d, target)
                cur = _compute_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    interior = np.where((h > 0.01) & (h < 0.99))[0]
                    if len(interior) < 2:
                        break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_quantize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _multi_slsqp(h_init, deadline, maxiter=100):
    n = len(h_init)
    target = n / 2.0
    h = _project(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    sub = max(50, min(150, maxiter))
    h1, c5_1 = _run_slsqp(h, deadline, maxiter=sub)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    if time.time() < deadline:
        h2, c5_2 = _run_slsqp(best_h, deadline, maxiter=sub)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(5.0, 0.02 * budget_s)
    deadline = t0 + budget_s - reserve
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = _sine_init(122, 3, 0.5)
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _compute_c5(h0)

    # Phase 1: LSE descent from parent
    d1 = t0 + budget_s * 0.15
    h_lse, c5_lse = _lse_descent(best_h, d1, betas=[5, 10, 20, 50, 100, 200])
    if c5_lse < best_c5:
        best_c5 = c5_lse
        best_h = h_lse.copy()

    # Phase 2: SLSQP direct minimax
    d2 = t0 + budget_s * 0.35
    h_sl, c5_sl = _multi_slsqp(best_h, d2, maxiter=120)
    if c5_sl < best_c5:
        best_c5 = c5_sl
        best_h = h_sl.copy()

    # Phase 3: LSE from SLSQP result
    d3 = t0 + budget_s * 0.45
    h_lse2, c5_lse2 = _lse_descent(best_h, d3, betas=[30, 60, 120, 300])
    if c5_lse2 < best_c5:
        best_c5 = c5_lse2
        best_h = h_lse2.copy()

    # Phase 4: SLSQP again from LSE result
    d4 = t0 + budget_s * 0.55
    h_sl2, c5_sl2 = _multi_slsqp(best_h, d4, maxiter=100)
    if c5_sl2 < best_c5:
        best_c5 = c5_sl2
        best_h = h_sl2.copy()

    # Phase 5: Multi-start exploration
    d5 = t0 + budget_s * 0.70
    n = len(best_h)
    starts = [best_h.copy()]
    for freq in [2, 3, 4, 5, 7]:
        for ph in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0, np.pi]:
            starts.append(_sine_init(n, freq, ph))
    for p in [5, 6, 7, 9, 11, 13]:
        for duty in [0.35, 0.45, 0.55, 0.65]:
            starts.append(_periodic_init(n, p, duty))
    bq = _binary_quantize(best_h)
    starts.append(bq)
    for _ in range(3):
        starts.append(_project(rng.uniform(0, 1, n), n / 2.0))

    for s in starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.018)
        h_s, c5_s = _lse_descent(s, sub_dl, betas=[15, 40, 100])
        if c5_s < best_c5:
            h_s2, c5_s2 = _run_slsqp(h_s, min(d5, time.time() + budget_s * 0.015), maxiter=60)
            if c5_s2 < c5_s:
                h_s, c5_s = h_s2, c5_s2
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Line search polish
    d6 = t0 + budget_s * 0.82
    h_lp, c5_lp = _line_polish(best_h, d6, radius=0.06, top_k=12, max_iters=400)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 7: Pair polish
    d7 = t0 + budget_s * 0.90
    h_pp, c5_pp = _pair_polish(best_h, d7, max_passes=60)
    if c5_pp < best_c5:
        best_c5 = c5_pp
        best_h = h_pp.copy()

    # Phase 8: Try nearby resolutions
    d8 = t0 + budget_s * 0.95
    tried = {len(best_h)}
    for dn in [-2, 2, -4, 4, -6, 6, 8, -8]:
        if time.time() > d8:
            break
        n_try = len(best_h) + dn
        if n_try < 40 or n_try > 300:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d8, time.time() + budget_s * 0.012)
        h_r, c5_r = _lse_descent(h_r, sub_dl, betas=[20, 50, 150])
        if c5_r < best_c5:
            h_r2, c5_r2 = _run_slsqp(h_r, min(d8, time.time() + budget_s * 0.01), maxiter=50)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-6:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 9: Final SLSQP + line polish
    d9 = deadline
    h_f, c5_f = _run_slsqp(best_h, min(d9, time.time() + budget_s * 0.02), maxiter=80)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    if time.time() < d9:
        h_fl, c5_fl = _line_polish(best_h, d9, radius=0.03, top_k=8, max_iters=200)
        if c5_fl < best_c5:
            best_c5 = c5_fl
            best_h = h_fl.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h_fb = np.array(initial_h_values, dtype=float).copy()
            h_fb = _project(h_fb, len(h_fb) / 2.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
        except NameError:
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
