def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    u = 1.0 - h
    G = np.zeros((2 * n - 1, n))
    for idx in range(2 * n - 1):
        lag = idx - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                G[idx, lag:n] += u[:m]
                G[idx, :m] += -h[lag:n]
        else:
            m = -lag
            ov = n - m
            if ov > 0:
                G[idx, :ov] += u[m:m + ov]
                G[idx, m:n] += -h[:ov]
    return G

def _smooth_max_weights(r, dx, beta):
    scaled = beta * r / dx
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        idx = np.argmax(r)
        w = np.zeros_like(r)
        w[idx] = 1.0
    else:
        w /= s
    return w

def _line_search_pair(r, G, Q_idx, Q_val, i, j, dx, alpha_lo, alpha_hi):
    n = len(r)
    T = G[:, j].copy()
    T -= G[:, i]
    Q_vec = np.zeros(n)
    for qi, qv in zip(Q_idx, Q_val):
        Q_vec[qi] += qv
    def c5_func(alpha):
        return float(np.max(r + alpha * T + alpha * alpha * Q_vec)) * dx
    if alpha_hi - alpha_lo < 1e-14:
        return alpha_lo, c5_func(alpha_lo)
    ngrid = 33
    grid = np.linspace(alpha_lo, alpha_hi, ngrid)
    vals = np.empty(ngrid)
    for gi in range(ngrid):
        vals[gi] = c5_func(grid[gi])
    best_gi = int(np.argmin(vals))
    lo_b = grid[max(0, best_gi - 2)]
    hi_b = grid[min(ngrid - 1, best_gi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = c5_func(c1)
    f2 = c5_func(c2)
    for _ in range(40):

            c2 = c1
            f2 = f1

        else:

            c1 = c2
            f1 = f2

def _pair_descent(h, deadline, beta=20.0, rho=0.15, max_pairs=25, stall_limit=50):
