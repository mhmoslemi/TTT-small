```python
import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_max(h):
    """Return just the max correlation value (before multiplying by dx)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x
```

```python
import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_max_raw(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        idx1 = np.arange(k, n)
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, m)
        g[idx2] -= h[idx2 + k]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        idx1 = np.arange(0, m)
        g[idx1] += (1.0 - h[idx1 + ga])
        idx2 = np.arange(ga, n)
        g[idx2] -= h[idx2 - ga]
    return g * dx


def _continuous_descent(h, deadline, max_iters=500):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = 25.0
    step = 0.2
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        M = float(np.max(corr))
        K = min(30, 2 * n - 1)
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        spread = max(float(top_vals[0] - top_vals[-1]), 1e-10)
        alpha_eff = min(200.0, 30.0 / spread)
        w = np.exp(alpha_eff * (top_vals - top_vals[0]))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(K) / K
        else:
            w /= w_sum
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _grad_row(h, k)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        direction = -grad / gn
        # Line search
        best_t = 0.0
        best_f = _exact_c5(h)
        t = step
        for _ in range(20):
            h_try = _project(h + t * direction, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
            t *= 0.65
            if t < 1e-8:
                break
        if best_t > 1e-10 and best_f < best_c5 - 1e-13:
            h = _project(h + best_t * direction, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            if best_t > 0.5 * step:
                step = min(step * 1.1, 1.0)
            else:
                step = max(step * 0.7, 0.02)
        else:
            stall += 1
            step *= 0.7
            if stall > 30:
                break
    return best_h, best_c5

def _lag_targeted_search(h, deadline, rng, max_rounds=200):
    n = len(h)
    m = n // 2
    h = h.copy()
    cur_max = _corr_max_raw(h)
    best_max = cur_max
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    ones_list = list(ones)
    zeros_list = list(set(range(n)) - ones)
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = float(np.max(corr))
        # Find active lags (top few)
        top_lags = np.argsort(-corr)[:5]
        improved = False
        for lag_idx in top_lags:
            if cur_max < best_max - 0.5:
                break
            k = int(lag_idx) - (n - 1)
            # Try random swaps and check if they reduce this specific lag
            for attempt in range(20):
                if time.time() > deadline:
                    break
                i = ones_list[rng.integers(m)]
                j = zeros_list[rng.integers(m)]
                h[i] = 0.0
                h[j] = 1.0
                new_lag_val = 0.0
                if k >= 0:
                    aa = n - k
                    new_lag_val = float(np.dot(h[k:], 1.0 - h[:aa]))
                else:
                    aa = n + k
                    new_lag_val = float(np.dot(h[:aa], 1.0 - h[-k:]))
                if new_lag_val < corr[lag_idx] - 1:
                    # Accept
                    cur_max = _corr_max_raw(h)
                    ones_list.remove(i)
                    ones_list.append(j)
                    zeros_list.remove(j)
                    zeros_list.append(i)
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            continue
    return best_h, best_max

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _quad_line_search(h, d, target, max_t=0.5):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C_arr = np.correlate(d, d, mode='full')
    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_hi = max(t_hi, 0.0)
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 60
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T_arr[None, :] - (grid[:, None] ** 2) * C_arr[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
    f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_arr - opt_t * opt_t * C_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

        top_vals = corr[top_idx]
        spread = max(float(top_vals[0] - top_vals[-1]), 1e-10)
        alpha_eff = min(200.0, 50.0 / spread)
        w = np.exp(alpha_eff * (top_vals - top_vals[0]))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(K) / K
        else:
            w /= w_sum
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _grad_row(h, k)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        direction = -grad / gn
        t, f_try = _quad_line_search(h, direction, target, max_t=step)
        if t < 1e-12:
            step *= 0.7
            continue
        h_new = h + t * direction
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.8 * step:
                step = min(step * 1.05, 1.5)
            else:
                step = max(step * 0.9, 0.01)
        else:
            step *= 0.75
            if step < 0.005:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 5.0

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.25 * np.sin(2.0 * np.pi * 3.0 * x / n0) + 0.15 * np.sin(2.0 * np.pi * 7.0 * x / n0)
    n0 = len(h0)
    h0 = _project(h0.copy(), n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick continuous optimization (10% time)
    d1 = t0 + budget_s * 0.10
    h_cont, c5_cont = _continuous_descent(h0, d1, max_iters=300)
    if c5_cont < best_c5:
        best_c5 = c5_cont
        best_h = h_cont.copy()

    # Phase 2: Binary SA with restarts (55% time) - the main workhorse
    d2 = t0 + budget_s * 0.65
    h_bin = _to_binary(best_h)
    h_sa, max_sa = _binary_sa_restarts(h_bin, d2, rng, n_restarts=10)
    c5_sa = max_sa * 2.0 / len(h_sa)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Double-flip SA from best binary (10% time)
    d3 = t0 + budget_s * 0.75
    h_dbl, max_dbl = _binary_sa_double(best_h, d3, rng, T_init=0.04, max_iters=300000)
    c5_dbl = max_dbl * 2.0 / len(h_dbl)
    if c5_dbl < best_c5:
        best_c5 = c5_dbl
        best_h = h_dbl.copy()

    # Phase 4: Lag-targeted search (8% time)
    d4 = t0 + budget_s * 0.83
    h_lt, max_lt = _lag_targeted_search(best_h, d4, rng, max_rounds=500)
    c5_lt = max_lt * 2.0 / len(h_lt)
    if c5_lt < best_c5:
        best_c5 = c5_lt
        best_h = h_lt.copy()

    # Phase 5: Multi-resolution (10% time)
    d5 = t0 + budget_s * 0.93
    best_n = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, -10, 10, -14, 14, 20, -20]
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        h_rb = _to_binary(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_rb, max_rb = _binary_sa_core(h_rb, sub_dl, rng, 0.04, 1e-8, 0.99996, 100000)
        c5_rb = max_rb * 2.0 / n_try
        if c5_rb < best_c5 - 1e-12:
            best_c5 = c5_rb
            best_h = h_rb.copy()
            best_n = n_try

    # Phase 6: Final continuous polish (5% time)
    d6 = deadline
    h_pol, c5_pol = _active_descent(best_h, d6, K=20, max_iters=150)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Final binary round from polished continuous
    if time.time() < deadline - 2:
        h_final_bin = _to_binary(best_h)
        h_fb, max_fb = _binary_sa_core(h_final_bin, min(deadline - 1, time.time() + 20),
                                       rng, 0.03, 1e-8, 0.99997, 200000)
        c5_fb = max_fb * 2.0 / len(h_fb)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
```
