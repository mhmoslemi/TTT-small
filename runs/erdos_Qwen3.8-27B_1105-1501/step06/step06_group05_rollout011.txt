            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            vel = np.zeros(n)
            sigma = max(0.002, sigma * 0.80)

    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None

    if base is None or base.size < 8:
        n = 122
        x = np.arange(n)
        base = 0.5 + 0.45 * np.sin(2 * np.pi * x / n + 0.3)
        base = base + rng.normal(0, 0.02, n)

    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: Deep optimization at original resolution (35% budget)
    d1 = t0 + budget_s * 0.35
    hh, ff = _optimize(h0, d1, rng, tau0=0.02, tau_min=0.004, k_top=10,
                       pair_frac=0.22, rand_frac=0.08, momentum=0.3)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 2: Multi-resolution exploration (35% budget)
    d2 = t0 + budget_s * 0.70
    bn = len(best_h)
    n_candidates = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8]:
        nt = bn + dn
        if 80 <= nt <= 200 and nt % 2 == 0:
            n_candidates.append(nt)
    for ne in [100, 104, 108, 112, 116, 128, 132, 136, 140, 144, 148, 152, 160]:
        if ne not in n_candidates and 80 <= ne <= 200:
            n_candidates.append(ne)
    n_candidates = sorted(set(n_candidates))

    remaining = max(1.0, d2 - time.time())
    per_n = max(3.0, remaining / max(1, len(n_candidates)))
    for nt in n_candidates:
        if time.time() >= d2:
            break
        if nt == len(best_h):
            continue
        hr = _resample(best_h, nt)
        sub = min(d2, time.time() + per_n)
        hr2, fr2 = _optimize(hr, sub, rng, tau0=0.018, tau_min=0.005, k_top=10,
                             pair_frac=0.2, rand_frac=0.08, momentum=0.3)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()

    # Phase 3: Re-optimize at best resolution (20% budget)
    d3 = t0 + budget_s * 0.90
    hh, ff = _optimize(best_h, d3, rng, tau0=0.01, tau_min=0.003, k_top=8,
                       pair_frac=0.25, rand_frac=0.06, momentum=0.2)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 4: Final micro-polish (10% budget)
    hh, ff = _optimize(best_h, deadline, rng, tau0=0.005, tau_min=0.002, k_top=6,
                       pair_frac=0.40, rand_frac=0.03, momentum=0.1)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
