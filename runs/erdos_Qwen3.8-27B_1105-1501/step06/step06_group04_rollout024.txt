            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_obj_grad(h, beta, top_k=200):
    """Compute LSE smooth max and gradient, using only top-k weighted lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = len(corr)
    cmax = corr.max()
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -400, 400)
    exps = np.exp(shifted)
    exp_sum = exps.sum()
    lse = cmax + np.log(exp_sum) / beta
    weights = exps / exp_sum
    
    # Focus on top-k by weight for gradient computation
    if m > top_k:
        top_idx = np.argsort(-weights)[:top_k]
        w_sum = weights[top_idx].sum()
        if w_sum > 1e-10:
            weights = np.zeros(m)
            weights[top_idx] = weights[top_idx] / w_sum
    
    grad = np.zeros(n)
    sig = np.where(weights > 1e-10)[0]
    for k in sig:
        d = int(k - (n - 1))
        w = weights[k]
        i_start = max(0, d)
        i_end = min(n - 1, n - 1 + d)
        if i_end < i_start:
            continue
        i_vals = np.arange(i_start, i_end + 1)
        np.add.at(grad, i_vals, w * (1.0 - h[i_vals - d]))
        np.add.at(grad, i_vals - d, -w * h[i_vals])
    
    return lse, grad

def _lse_optimize(h, beta_range, target, deadline, max_iter_total=2000):
    """LSE optimization with adaptive beta and backtracking line search."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_obj, _ = _lse_obj_grad(h, beta_range[0])
    lr = 0.02
    stall = 0
    it = 0
    
    for beta in beta_range:
        if time.time() > deadline or it >= max_iter_total:
            break
        # Optimize at this beta
        for inner in range(max_iter_total // len(beta_range)):
            if time.time() > deadline or it >= max_iter_total:
                break
            it += 1
            obj, grad = _lse_obj_grad(h, beta)
            grad = grad - grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            
            # Backtracking line search
            alpha = lr
            success = False
            for _ in range(25):
                h_new = _project(h - alpha * grad, target)
                obj_new, _ = _lse_obj_grad(h_new, beta)
                if obj_new < obj - 1e-15 * max(1, abs(obj)):
                    h = h_new
                    cur_obj = obj_new
                    success = True
                    break
                alpha *= 0.4
            
            if success:
                lr = min(lr * 1.05, 0.3)
                c5 = _exact_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                lr *= 0.5
                stall += 1
                if lr < 1e-14:
                    break
            
            if stall > 50:
                break
    
    return best_h, best_c5

def _lp_direction(h, scores, radius, top_k=15):
    """LP-based descent direction (simplified from previous code)."""
    from scipy.optimize import linprog
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    
    # Build gradient matrix for active lags
    G = np.zeros((K, n))
    for idx, r in enumerate(active):
        d = int(r - (n - 1))
        if d >= 0:
            i_vals = np.arange(d, n)
            G[idx, i_vals] += (1.0 - h[i_vals - d])
            G[idx, i_vals - d] += -h[i_vals]
        else:
            i_vals = np.arange(0, n + d)
            G[idx, i_vals] += (1.0 - h[i_vals - d])
            G[idx, i_vals - d] += -h[i_vals]
    
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_c5(h, d, dx, trust_radius):
    """Golden section line search for C5."""
    n = len(h)
    target = n / 2.0
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val
