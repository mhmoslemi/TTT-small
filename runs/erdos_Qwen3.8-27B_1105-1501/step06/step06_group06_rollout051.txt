from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_and_lags(h):
    """Compute C5 and the correlation vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def _binary_local_search(h0, deadline, max_iter=50000):

    n = len(h0)

    # Start from threshold

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:

        excess = len(ones) - target
        h[ones[:excess]] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        h[zeros[:deficit]] = 1.0
    
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try random swap: pick one 1 and one 0, swap them
        best_improve = 1e-15
        best_i = -1
        best_j = -1
        n_trials = 30
        for _ in range(n_trials):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]

            if new_c5 < cur_c5 - best_improve:
                best_improve = cur_c5 - new_c5
                best_i, best_j = i, j
        
        if best_i >= 0:

            cur_c5 -= best_improve
            if cur_c5 < best_c5 - 1e-12:

            no_improve = 0

                h_new[h_new == 1.0] = 1.0

                h[i], h[j] = 0.0, 1.0
                new_c5 = _exact_c5(h)
                h[i], h[j] = 1.0, 0.0
                if new_c5 < cur_c5 - 1e-12:
                    h[i], h[j] = 0.0, 1.0
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
    
    return best_h, best_c5

def _continuous_optimize(h0, deadline, max_outer=30):
    """SLSQP-based continuous optimization with active set."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        K = min(40, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        t_init = cur_c5 + 5e-5
        
        def obj_fn(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]
        
        def ineq_jac(x):
            hx = x[:n]
            # Compute gradients of corr[k] w.r.t. h[i]
            G = np.zeros((K, n))
            for idx, k in enumerate(top_k):
                lag = k - (n - 1)
                m = n - abs(lag)
                if lag >= 0:
                    # corr[k] = sum_{i=0}^{m-1} h[i+lag] * (1-h[i])
                    # d/dh[i+lag] = (1-h[i])
                    # d/dh[i] = -h[i+lag]
                    G[idx, lag:lag+m] += (1.0 - hx[:m])
                    G[idx, :m] += (-hx[lag:lag+m])
                else:
                    g = -lag
                    # corr[k] = sum_{i=0}^{m-1} h[i] * (1-h[i+g])
                    G[idx, :m] += (1.0 - hx[g:g+m])
                    G[idx, g:g+m] += (-hx[:m])
            G *= dx
            J = np.zeros((K, n + 1))
            J[:, :n] = -G
            J[:, n] = 1.0
            return J
        
        def eq_fn(x):
            return np.sum(x[:n]) - target
        
        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        
        x0 = np.concatenate([h, [t_init]])
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-14})
        
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            break
    
    return best_h, best_c5

def _grad_descent(h_init, deadline, max_iter=2000):
    """Subgradient/active set gradient descent."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.1
    rng = np.random.default_rng(99)
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = cur_c5
        # Active set: lags near the maximum
        tol = max(1e-4, 0.02 * M)
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:20]
        
        # Compute gradient of max (using subgradient = weighted average of active gradients)
        weights = np.zeros(len(active))
        diffs = M - corr[active]
        if diffs.max() > 0:
            weights = diffs ** 2
            weights /= weights.sum()
        else:
            weights[:] = 1.0 / len(active)
        
        # Compute gradient
        G = np.zeros(n)
        for wi, k in zip(weights, active):
            lag = k - (n - 1)
            m = n - abs(lag)
            if lag >= 0:
                G[lag:lag+m] += wi * dx * (1.0 - h[:m])
                G[:m] += wi * dx * (-h[lag:lag+m])
            else:
                g = -lag
                G[:m] += wi * dx * (1.0 - h[g:g+m])
                G[g:g+m] += wi * dx * (-h[:m])
        
        # Project gradient to satisfy sum constraint
        G -= G.mean()
        
        # Line search
        step = max(step * 0.95, 0.01)
        best_t = 0.0
        best_val = cur_c5
        for alpha in [step, step*0.5, step*0.25, step*0.1]:
            h_try = _project(h - alpha * G, target)
            val = _exact_c5(h_try)
            if val < best_val:
                best_val = val
                best_t = alpha
        
        if best_t > 0:
            h = _project(h - best_t * G, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            step *= 0.8
            if stall > 50:
                # Random kick
                h = _project(h + rng.normal(0, 0.05, n), target)
                cur_c5 = _exact_c5(h)
                stall = 0
                step = 0.1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary(n, rng):

    perm = rng.permutation(n)

    return h

def _structured_binary(n, rng):
    """Generate structured 0-1 patterns."""
    h = np.zeros(n)
    # Various structures
    pattern = rng.integers(0, 5)
    if pattern == 0:
        # Alternating blocks
        blen = max(1, n // (2 * rng.integers(3, 12)))
        for i in range(n):
            h[i] = 1.0 if (i // blen) % 2 == 0 else 0.0
    elif pattern == 1:
        # Sine threshold
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h[np.sin(2 * np.pi * freq * x / n + phase) > 0] = 1.0
    elif pattern == 2:
        # Golden ratio based
        phi = (1 + np.sqrt(5)) / 2
        h[(np.arange(n) * phi) % 1 > 0.5] = 1.0
    elif pattern == 3:
        # Random blocks
        pos = 0
        while pos < n:
            blen = int(rng.integers(2, max(3, n // 5 + 1)))
            if rng.random() < 0.5:
                h[pos:pos+blen] = 1.0
            pos += blen
    else:
        # Fourier-based
        freqs = rng.integers(1, n//2 + 1, size=3)
        x = np.arange(n)
        sig = np.zeros(n)
        for f in freqs:
            sig += np.sin(2 * np.pi * f * x / n + rng.uniform(0, 2*np.pi))
        h[sig > sig.mean()] = 1.0
    
    # Fix to have exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > n // 2:
        rem = ones[rng.choice(len(ones), size=len(ones) - n//2, replace=False)]
        h[rem] = 0.0
    elif len(ones) < n // 2:
        add = zeros[rng.choice(len(zeros), size=n//2 - len(ones), replace=False)]
        h[add] = 1.0
    return h

                best_i, best_j = i, j
        
        if best_i >= 0 and best_delta < -1e-12:
            h[best_i], h[best_j] = 0.0, 1.0
            cur_c5 += best_delta
            if cur_c5 < best_c5 - 1e-12:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def _continuous_slsqp(h0, deadline, max_outer=25):
    """SLSQP minimax with active set."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _corr_vec(h)
        K = min(50, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        t_init = cur_c5 + 5e-5
        
        def obj_fn(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def ineq_fn(x):
            hx = x[:n]
            c = _corr_vec(hx)
            return x[n] - c[top_k]
        
        def ineq_jac(x):
            hx = x[:n]
            G = np.zeros((K, n))
            for idx, k in enumerate(top_k):
                lag = k - (n - 1)
                m = n - abs(lag)
                if lag >= 0:
                    G[idx, lag:lag+m] += (1.0 - hx[:m])
                    G[idx, :m] += (-hx[lag:lag+m])
                else:
                    g = -lag
                    G[idx, :m] += (1.0 - hx[g:g+m])
                    G[idx, g:g+m] += (-hx[:m])
            G *= dx
            J = np.zeros((K, n + 1))
            J[:, :n] = -G
            J[:, n] = 1.0
            return J
        
        def eq_fn(x):
            return np.sum(x[:n]) - target
        
        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        
        x0 = np.concatenate([h, [t_init]])
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-14})
        
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break
        else:
            break
    
    return best_h, best_c5

def _grad_polish(h_init, deadline, max_iter=5000):
    """Active-set gradient descent with adaptive step."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = 0.15
    rng = np.random.default_rng(31415)
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = _corr_vec(h)
        M = cur_c5
        tol = max(2e-4, 0.03 * M)
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:25]
        elif len(active) > 60:
            active = active[np.argsort(-corr[active])[:60]]
        
        # Subgradient: weighted average of active gradients
        diffs = M - corr[active]
        diffs = np.clip(diffs, 0, None)
        if diffs.sum() > 0:
            weights = diffs ** 2
            weights /= weights.sum()
        else:
            weights = np.ones(len(active)) / len(active)
        
        G = np.zeros(n)
        for wi, k in zip(weights, active):
            lag = k - (n - 1)
            m = n - abs(lag)
            if lag >= 0:
                G[lag:lag+m] += wi * dx * (1.0 - h[:m])
                G[:m] += wi * dx * (-h[lag:lag+m])
            else:
                g = -lag
                G[:m] += wi * dx * (1.0 - h[g:g+m])
                G[g:g+m] += wi * dx * (-h[:m])
        
        G -= G.mean()
        
        # Backtracking line search
        best_val = cur_c5
        best_alpha = 0.0
        for alpha in [step, step * 0.5, step * 0.25, step * 0.125, step * 0.0625]:
            h_try = _project(h - alpha * G, target)
            val = _exact_c5(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            h = _project(h - best_alpha * G, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step = min(step * 1.05, 0.4)
        else:
            stall += 1
            step *= 0.85
            if stall > 80:
                h = _project(h + rng.normal(0, 0.03, n), target)
                cur_c5 = _exact_c5(h)
                stall = 0
                step = 0.15
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_binary(n, rng):
    h = np.zeros(n)
    pattern = rng.integers(0, 6)
    x = np.arange(n)
    if pattern == 0:
        blen = max(1, n // (2 * int(rng.integers(3, 12))))
        h[(x // blen) % 2 == 0] = 1.0
    elif pattern == 1:
        freq = rng.uniform(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        h[np.sin(2 * np.pi * freq * x / n + phase) > 0] = 1.0
    elif pattern == 2:
        phi = (1 + np.sqrt(5)) / 2
        h[(x * phi) % 1 > 0.5] = 1.0
    elif pattern == 3:
        pos = 0
        while pos < n:
            blen = int(rng.integers(2, max(3, n // 4 + 1)))
            if rng.random() < 0.5:
                h[pos:pos+blen] = 1.0
            pos += blen
    elif pattern == 4:
        freqs = rng.integers(1, n // 2 + 1, size=3)
        sig = np.zeros(n)
        for f in freqs:
            sig += np.sin(2 * np.pi * f * x / n + rng.uniform(0, 2 * np.pi))
        h[sig > sig.mean()] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:n // 2]] = 1.0
    h = _fix_binary_sum(h, n // 2)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s
    
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except NameError:
        n_parent = 80
        h_parent = _structured_binary(n_parent, rng)
    
    best_h = h_parent.copy()
    best_c5 = _exact_c5(h_parent)
    best_n = n_parent
    print(f"Parent: n={best_n}, C5={best_c5:.6f}")
    
    # Phase 1: Binary hill climbing at parent resolution (20%)
    d1 = t0 + budget_s * 0.20
    h_bin = (h_parent >= 0.5).astype(float)
    h_bin = _fix_binary_sum(h_bin, n_parent // 2)
    h_b, c5_b = _binary_hill_climb(h_bin, d1, max_iter=200000, n_trial=50)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
        best_n = n_parent
        print(f"BinHill parent: n={best_n}, C5={best_c5:.6f}")
    
    # Phase 2: Multi-resolution binary search with SA (40%)
    d2 = t0 + budget_s * 0.60
    n_candidates = []
    for n_try in range(36, 162, 2):
        if n_try % 2 == 0 and n_try >= 36 and n_try <= 160:
            n_candidates.append(n_try)
    n_candidates.sort(key=lambda x: abs(x - best_n))
    
    tried_n = set()
    for n_try in n_candidates:
        if time.time() > d2:
            break
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        
        # Resample current best
        h_r = _resample(best_h, n_try)
        h_bin_r = (h_r >= 0.5).astype(float)
        h_bin_r = _fix_binary_sum(h_bin_r, n_try // 2)
        
        sub_dl = min(d2, time.time() + budget_s * 0.04)
        h_br, c5_br = _binary_hill_climb(h_bin_r, sub_dl, max_iter=80000, n_trial=30)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
            best_n = n_try
            print(f"BinMulti: n={best_n}, C5={best_c5:.6f}")
        
        # Also try structured start
        if time.time() < d2 - budget_s * 0.02:
            h_struct = _structured_binary(n_try, rng)
            sub_dl2 = min(d2, time.time() + budget_s * 0.02)
            h_sr, c5_sr = _binary_hill_climb(h_struct, sub_dl2, max_iter=40000, n_trial=25)
            if c5_sr < best_c5:
                best_c5 = c5_sr
                best_h = h_sr.copy()
                best_n = n_try
                print(f"BinStruct: n={best_n}, C5={best_c5:.6f}")
    
    # Phase 2b: SA on best binary for more escape (10%)
    d2b = t0 + budget_s * 0.72
    if best_n >= 2:
        h_sa = (best_h >= 0.5).astype(float)
        h_sa = _fix_binary_sum(h_sa, best_n // 2)
        h_s, c5_s = _binary_sa(h_sa, d2b, max_iter=150000, T0=0.015, T1=0.0005, n_eval_per_iter=15)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = best_n
            print(f"SA: n={best_n}, C5={best_c5:.6f}")
    
    # Phase 3: Continuous SLSQP from best (15%)
    d3 = t0 + budget_s * 0.87
    h_c = best_h.copy()
    h_c, c5_c = _continuous_slsqp(h_c, min(d3, time.time() + budget_s * 0.08), max_outer=25)
    if c5_c < best_c5:
        best_c5 = c5_c
        best_h = h_c.copy()
        best_n = len(h_c)
        print(f"SLSQP: n={best_n}, C5={best_c5:.6f}")
    
    # Gradient polish
    if time.time() < d3:
        h_g, c5_g = _grad_polish(best_h, d3, max_iter=4000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = len(h_g)
            print(f"GradPolish: n={best_n}, C5={best_c5:.6f}")
    
    # Phase 4: Try continuous at nearby resolutions (8%)
    d4 = t0 + budget_s * 0.95
    for dn in [-4, 4, -6, 6, -8, 8, -10, 10]:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 36 or n_try > 160 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_r, c5_r = _continuous_slsqp(h_r, sub_dl, max_outer=12)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
            print(f"ContMulti: n={best_n}, C5={best_c5:.6f}")
    
    # Final polish
    d_final = deadline
    h_f, c5_f = _continuous_slsqp(best_h, min(d_final, time.time() + budget_s * 0.03), max_outer=15)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        best_n = len(h_f)
    
    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    print(f"FINAL: n={best_n}, C5={best_c5:.6f}")
    return best_h, best_c5, best_n
