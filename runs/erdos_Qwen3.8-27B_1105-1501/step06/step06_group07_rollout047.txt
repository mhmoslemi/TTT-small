    if target >= n:

        if diff > 0:
            idx = np.argmax(1.0 - x)
        else:
            idx = np.argmin(x)
        x[idx] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, k):
    """Gradient of corr(h, 1-h)[n-1+k] * dx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        # corr[n-1+k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        # d/dh[j]: if j >= k: (1 - h[j-k]); if j < k and j+k < n: -h[j+k]
        g[k:] += (1.0 - h[:m])  # for j = k,...,n-1: deriv is (1-h[j-k])
        g[:m] -= h[k:]  # for j = 0,...,m-1: deriv is -h[j+k]
    else:
        j = -k
        m = n - j
        if m <= 0:
            return g
        # corr[n-1-k] = sum_{i=0}^{m-1} h[i]*(1-h[i+j])
        # d/dh[i]: (1-h[i+j]) for i < m; -h[i-j] for i >= j
        g[:m] += (1.0 - h[j:])  # for i = 0,...,m-1
        g[j:] -= h[:m]  # for i = j,...,n-1
    return g * dx

**Idea: "Anti-max" direction finding via small LP**

At the current point h, with active set A (top-K lags), the subgradient of f = max g_k is any convex combination of {∇g_k : k ∈ A}. The steepest descent direction in the subdifferential is:
- d = argmin_{s ∈ ∂f(h)} ||s||² = argmin_{λ ≥ 0, Σλ=1} ||Σ λ_k ∇g_k||²
- This is a quadratic program in λ, which for small K is fast.

But even better: we want the direction that most decreases the maximum. The "anti-max" direction solves:
- min_d max_{k∈A} ∇g_k(h) · d  subject to ||d|| ≤ 1
- By minimax theorem, this equals min_{||d||≤1} max_k a_k·d where a_k = ∇g_k(h)
- The optimal d is -(center of min enclosing ball of {a_k})... 

Actually, the steepest descent direction for a max function at a point where multiple constraints are active is:
d = -argmin_{s ∈ conv{∇g_k : k∈A}} ||s||

This is the negative of the closest point to the origin in the convex hull of the active gradients. For small K, this is a QP:
min_λ ||Σ λ_k a_k||²  s.t. λ ≥ 0, Σλ = 1

Actually, one more important idea: **the "equalization" approach**. At the optimum, 2-5 lags should be at the maximum. We can try to:
1. Identify the top 3-5 lags
2. Find a direction that equalizes them (makes their values as close as possible) while decreasing the maximum
3. This is a small linear program: min_d max_{k∈A} (g_k + ∇g_k · d) ≈ min_d max_{k∈A} ∇g_k · d (linearized)

The solution to min_d max_k a_k·d s.t. ||d|| ≤ 1 is d = -v where v is the center of the minimum enclosing ball of {-a_k : k ∈ A}. For small A, this can be found exactly.

But this is essentially what the steepest subgradient direction does. So let me focus on implementing that well.

Let me also think about the **Nesterov / FISTA approach** more carefully:

For f_α(h) = (1/α) log Σ exp(α g_k(h)), the gradient is:
∇f_α(h) = Σ_k w_k ∇g_k(h) where w_k = exp(α g_k) / Σ exp(α g_j)

The Lipschitz constant of ∇f_α is bounded by α * max_k ||∇g_k||² (roughly). We can estimate this empirically.

FISTA iteration:
- y_k = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
- x_{k+1} = Proj_feasible(y_k - (1/L) ∇f(y_k))
- t_{k+1} = (1 + sqrt(1 + 4 t_k²))/2

The convergence rate is O(1/k²) which is much better than the O(1/k) of plain gradient descent.

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) / 2.0 + 2.0  # tighter initial bounds
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += (1.0 - h[:m])
        g[:m] -= h[k:]
    else:
        j = -k
        m = n - j
        if m <= 0:
            return g
        g[:m] += (1.0 - h[j:])
        g[j:] -= h[:m]
    return g * dx

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _line_search(h, d, target, max_t=1.5):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    t_lo = 0.0
    
    if t_hi <= t_lo + 1e-12:
        return 0.0, _exact_c5(h)
    
    # Grid + golden section
    n_grid = 80
    grid = np.linspace(t_lo, t_hi, n_grid)
    # Compute f on grid
    vals = np.max(S[None, :] + grid[:, None] * T[None, :] - (grid[:, None]**2) * C[None, :], axis=1) * dx
    best_i = int(np.argmin(vals))
    best_t = grid[best_i]
    best_val = float(vals[best_i])
    
    def f_of_t(t):
        return float(np.max(S + t * T - t * t * C)) * dx
    
    # Golden section refinement
    a = grid[max(0, best_i - 1)]
    b = grid[min(n_grid - 1, best_i + 1)]
    gr = (5.0**0.5 - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_of_t(c1)
    f2 = f_of_t(c2)
    for _ in range(55):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f_of_t(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_of_t(c2)
    
    opt_t = (a + b) * 0.5
    opt_val = f_of_t(opt_t)
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    
    return best_t, best_val

def _steepest_dir(h, top_idx, K=15):
    """Steepest descent direction in subdifferential."""
    n = len(h)
    K = min(K, len(top_idx))
    G = np.zeros((K, n))
    for i in range(K):
        k = int(top_idx[i]) - (n - 1)
        G[i] = _grad_lag(h, k)
    # Remove mean (preserve sum constraint)
    G_mean = G.mean(axis=1, keepdims=True)
    G -= G_mean
    
    # Solve: min_lambda ||G^T lambda||^2 s.t. lambda >= 0, sum(lambda) = 1
    # => min lambda^T (G @ G^T) lambda s.t. lambda >= 0, 1^T lambda = 1
    M = G @ G.T  # K x K
    # Projected gradient method on the simplex
    lam = np.ones(K) / K
    # Adaptive step size
    step = 1.0 / float(np.max(np.abs(M)))
    for _ in range(300):
        grad_M = 2.0 * (M @ lam)
        lam = lam - step * grad_M
        # Project onto simplex {lambda >= 0, sum = 1}
        u = np.sort(lam)[::-1]
        cssv = np.cumsum(u)
        rho = int(np.max(np.where(u * np.arange(1, K+1) > (cssv - 1.0))[0])) if K > 1 else 0
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        lam = np.maximum(lam - theta, 0.0)
    
    d = -(G.T @ lam)
    return d

def _active_set_descent(h, deadline, K=15, max_iters=300):
    """Active set descent with steepest subgradient direction and exact line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_scale = 1.0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = _corr_vals(h)
        M = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        
        # Steepest subgradient direction
        d = _steepest_dir(h, top_idx, K=K)
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d = d / dn * step_scale
        
        t, f_try = _line_search(h, d, target, max_t=2.0)
        
        if t > 1e-12:
            h_new = h + t * d
            h_new = _project(h_new, target)
            f_new = _exact_c5(h_new)
            if f_new < best_c5 - 1e-15:
                best_c5 = f_new

def _binary_sa(h_init, deadline, rng, max_iters=50000):
    """Binary simulated annealing."""
    n = len(h_init)
    target = n // 2
    # Start from thresholding
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.01
    T_min = 3e-6
    cooling = 0.99993
    
    ones_set = set(int(x) for x in sorted_idx[:target])
    
    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if i in ones_set:
            # Try to move i out, j in
            if j in ones_set:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                ones_set.remove(i)
                ones_set.add(j)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            if i in ones_set:
                continue
            h[i] = 1.0
            h[j] = 0.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                ones_set.add(i)
                ones_set.remove(j)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 0.0
                h[j] = 1.0
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
