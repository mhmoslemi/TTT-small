        e = np.exp(shifted)
        w = e / np.sum(e)
    
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    
    for mi in sig:
        s = mi - (n - 1)
        wm = w[mi]
        if s >= 0:
            mv = n - s
            if mv > 0:
                grad[:mv] += wm * (1.0 - h[s:s+mv])
                grad[s:] += wm * (-h[:mv])
        else:
            g = -s
            mv = n - g
            if mv > 0:
                grad[g:] += wm * (1.0 - h[:mv])
                grad[:mv] += wm * (-h[g:g+mv])
    
    grad *= dx
    return grad, m

def _active_descent_exact(h_init, deadline, max_iter=500):
    """
    Active descent with exact line search.
    Different from previous: uses weighted active set gradient with 
    curvature-aware step sizing.
    """
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    # Precompute: for each lag s, the set of index pairs that contribute
    # c_s = dx * sum_{i in valid(s)} h[i] * (1 - h[i+s])
    # This is a quadratic form. The Hessian of c_s with respect to h is:
    # H_s[i,j] = -dx if i+s=j or j+s=i (and both valid), else 0
    # So the Hessian is a banded matrix (bandwidth 1).
    
    radius = 0.15
    stall = 0
    rng_seed = 12345
    rng = np.random.default_rng(rng_seed)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        
        # Active set: correlations within a small tolerance of max
        # Use a wider tolerance to capture near-active constraints
        tol = max(1e-6, 0.02 * (M - np.min(corr)))
        active_dense = np.where(corr > M - tol)[0]
        if len(active_dense) == 0:
            active_dense = np.where(corr == M)[0]
        # Limit to top 30
        if len(active_dense) > 30:
            order = np.argsort(-corr[active_dense])
            active_dense = active_dense[order[:30]]
        
        # Compute weighted gradient (weights proportional to violation)
        w = (corr[active_dense] - (M - tol)) ** 2
        if np.sum(w) < 1e-20:
            w = np.ones(len(active_dense))
        w = w / np.sum(w)
        
        grad = np.zeros(n)
        for ai, mi in enumerate(active_dense):
            s = mi - (n - 1)
            wi = w[ai]
            if s >= 0:
                mv = n - s
                if mv > 0:
                    grad[:mv] += wi * (1.0 - h[s:s+mv])
                    grad[s:] += wi * (-h[:mv])
            else:
                g = -s
                mv = n - g
                if mv > 0:
                    grad[g:] += wi * (1.0 - h[:mv])
                    grad[:mv] += wi * (-h[g:g+mv])
        
        grad *= dx
        
        # Project gradient to sum-zero (feasible directions)
        grad -= grad.mean()
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            stall += 10
            continue
        d = -grad / gnorm * radius
        
        # Exact line search: c_k(h + t*d) is quadratic in t
        # c_k(h + td) = c_k(h) + t * (dc_k/dh · d) + t²/2 * (d^T H_k d)
        # = S0[k] + t * T[k] + t² * Q[k]
        # where Q[k] = -dx/2 * sum_{i: valid} d[i]*d[i+k] (or similar)
        
        S0 = np.correlate(h, 1.0 - h, mode='full')  # not multiplied by dx yet
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C = -np.correlate(d, d, mode='full')  # this gives -sum d[i]*d[i+k]
        
        # Actually: c_s(h+td) = dx * sum_i (h[i]+t*d[i]) * (1-h[i+s]-t*d[i+s])
        # = dx * sum_i [h[i](1-h[i+s]) - t*h[i]*d[i+s] + t*d[i]*(1-h[i+s]) - t²*d[i]*d[i+s]]
        # = c_s(h) + t*dx*sum_i[d[i](1-h[i+s]) - h[i]*d[i+s]] - t²*dx*sum_i d[i]*d[i+s]
        # 
        # The middle term: dx*sum_i d[i](1-h[i+s]) - dx*sum_i h[i]d[i+s]
        # = dx*sum_i d[i](1-h[i+s]) - dx*sum_j h[j-s]d[j] (substituting j=i+s)
        # This is: (dx * (d conv with (1-h) at offset s)) - (dx * (h conv with d at offset s))
        
        # Let me just compute directly:
        # F(t) = max_k [S0[k]*dx + t*T_full[k]*dx + t²*C_full[k]*dx]
        # where T_full[k] and C_full[k] are the full correlation arrays
        
        # Actually I already have the full correlation arrays:
        # S0_full = correlate(h, 1-h) → this is sum h[i](1-h[i+k]) for each lag
        # T_full[k] = sum d[i](1-h[i+k]) - sum h[i]d[i+k] = correlate(d,1-h)[k] - correlate(h,d)[k]
        # C_full[k] = -sum d[i]*d[i+k] = -correlate(d,d)[k]
        
        # So F_k(t) = dx * (S0_full[k] + t*T_full[k] + t²*C_full[k])
        
        # We want min_t max_k F_k(t)
        # with the constraint that h + t*d stays in [0,1]^n
        
        # Find valid t range from box constraints
        pos = d > 1e-15
        neg = d < -1e-15
        t_max = radius
        t_min = -radius
        if np.any(pos):
            t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_min = max(t_min, float(np.max(h[neg] / (-d[neg]))))
        
        if t_max - t_min < 1e-14:
            stall += 1
            continue
        
        # Evaluate F(t) = max_k dx*(S0[k] + t*T[k] + t²*C[k]) on a grid + golden section
        # For efficiency, only evaluate on active lags (those near max)
        # But for safety, evaluate on all
        
        n_grid = 64
        grid = np.linspace(t_min, t_max, n_grid)
        # F shape: (n_grid, 2n-1)
        F = S0_full = S0[None, :]  # (1, 2n-1)
        F = F + grid[:, None] * T_full[None, :]  # Wait, T_full is T/dx
        # Let me redo with the proper arrays
        # S0_full = correlate(h, 1-h) → (2n-1,) array, each entry is sum h[i](1-h[i+k])
        # The actual correlation value is S0_full * dx
        # T_full[k] = correlate(d, 1-h)[k] - correlate(h, d)[k]
        # C_full[k] = -correlate(d, d)[k]
        
        # F_grid[t_idx, k] = S0_full[k] + t*T_full[k] + t²*C_full[k]
        # We want min over t of max_k F_grid[t_idx, k] * dx
        
        T_full_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        C_full_arr = -np.correlate(d, d, mode='full')
        
        # (n_grid, 2n-1)
        F_vals = S0[None, :] + grid[:, None] * T_full_arr[None, :] + (grid[:, None]**2) * C_full_arr[None, :]
        obj = np.max(F_vals, axis=1) * dx
        
        best_i = int(np.argmin(obj))
        best_t = grid[best_i]
        best_val = obj[best_i]
        
        # Golden section refinement
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        if 0 < best_i < n_grid - 1:
            a, b = grid[best_i-1], grid[best_i+1]
        else:
            a, b = t_min, t_max
        f1 = float(np.max(S0 + a*T_full_arr + a*a*C_full_arr)) * dx
        f2 = float(np.max(S0 + b*T_full_arr + b*b*C_full_arr)) * dx
        for _ in range(50):
            if b - a < 1e-12:
                break
            if f1 < f2:
                b, f2 = a, f1
                a = b - gr * (b - a)
                f1 = float(np.max(S0 + a*T_full_arr + a*a*C_full_arr)) * dx
            else:
                a, f1 = b, f2
                b = a + gr * (b - a)
                f2 = float(np.max(S0 + b*T_full_arr + b*b*C_full_arr)) * dx
        mid = 0.5 * (a + b)
        mid_val = float(np.max(S0 + mid*T_full_arr + mid*mid*C_full_arr)) * dx
        if mid_val < best_val:
            best_t, best_val = mid, mid_val
        
        # Apply step
        h_new = h + best_t * d
        h_new = _project(h_new, target)
        new_c5 = _exact_c5(h_new)
        
        if new_c5 < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # Increase radius if step was full
            if abs(best_t) > 0.8 * radius:
                radius = min(0.5, radius * 1.1)
        else:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
