import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

So d^T H d = -4δ_{0,lag} + 2δ_{p-q,lag} + 2δ_{p-q,-lag}

And Q_k = (d^T H_k d)/2 = -2δ_{0,lag} + δ_{p-q,lag} + δ_{p-q,-lag}

For the pair (i,j) where d = e_i - e_j:
p = i, q = j
Q_k = -2δ_{0,lag} + δ_{i-j,lag} + δ_{
==-j,lag}
   = -2δ_{0,lag} + δ_{i-j,lag} + δ_{j-i,lag}

So:
- At lag = 0: Q = -2
- At lag = i-j: Q = 1 (if i≠j)
- At lag = j-i: Q = 1 (if i≠j)
- Elsewhere: Q = 0

And the full quadratic for each lag k (with lag = k-(n-1)):
f_k(α) = S0_k + α*(G[k,i] - G[k,j]) + α²*(-2δ_{0,lag} + δ_{i-j,lag} + δ_{j-i,lag})

Note that the quadratic term is ≤ 0 at lag 0 and ≥ 0 at the other two lags. This makes sense: at lag 0, the overlap h*(1-h) is concave in each h, so moving h reduces it (the quadratic term is negative). At the cross-lags, the interaction term makes it slightly convex.

Now, for the line search, I need to minimize max_k f_k(α) over α in a feasible interval.

The feasible interval for α:
- h[i] + α ∈ [0, 1] → α ∈ [-h[i], 1-h[i]]
- h[j] - α ∈ [0, 1] → α ∈ [h[j]-1, h[j]]
So α ∈ [max(-h[i], h[j]-1, -radius), min(1-h[i], h[j], radius)]

This is a 1D non-convex optimization (max of quadratics). The previous code does grid + golden section, which is reasonable.

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    u = 1.0 - h
    L = 2 * n - 1
    G = np.zeros((L, n))
    for lag in range(-(n-1), n):
        k = lag + (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                G[k, :m] += -h[lag:]
                G[k, lag:] += u[:m]
        else:
            g = -lag
            m = n - g
            if m > 0:
                G[k, :m] += u[g:]
                G[k, g:] += -h[:m]
    return G

def _Q_vector(n, i, j):
    """Return Q_k for pair (i,j) - a vector of length 2n-1"""
    Q = np.zeros(2 * n - 1)
    Q[n-1] = -2.0  # lag 0
    lag_ij = i - j
    lag_ji = j - i
    Q[n-1 + lag_ij] = 1.0  # lag = i-j
    Q[n-1 + lag_ji] = 1.0  # lag = j-i
    # Note: if i-j = 0 (i=j), but we always have i≠j
    # Also if i-j = j-i (i.e., i=j), but i≠j always
    return Q

def _line_search_1d(S0, A, Q, alpha_lo, alpha_hi, n_grid=40):
    """Minimize max_k (S0[k] + alpha*A[k] + alpha^2*Q[k]) over alpha in [lo, hi]"""
    # Evaluate on a grid
    if alpha_hi - alpha_lo < 1e-15:
        return alpha_lo, float(np.max(S0))
    
    # Coarse grid
    t = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = np.max(S0[None, :] + t[:, None] * A[None, :] + (t[:, None])**2 * Q[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    
    # Refine around best
    lo = t[max(0, best_idx - 2)]
    hi = t[min(n_grid - 1, best_idx + 2)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = float(np.max(S0 + c1 * A + c1*c1 * Q))
    f2 = float(np.max(S0 + c2 * A + c2*c2 * Q))
    
    for _ in range(50):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = float(np.max(S0 + c1 * A + c1*c1 * Q))
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = float(np.max(S0 + c2 * A + c2*c2 * Q))
    
    opt = (lo + hi) / 2.0
    opt_val = float(np.max(S0 + opt * A + opt*opt * Q))
    
    # Also check endpoints
    f_lo = float(np.max(S0 + alpha_lo * A + alpha_lo**2 * Q))
    f_hi = float(np.max(S0 + alpha_hi * A + alpha_hi**2 * Q))
    f_grid = vals[best_idx]
    
    best = min([(opt, opt_val), (alpha_lo, f_lo), (alpha_hi, f_hi), (t[best_idx], f_grid)], key=lambda x: x[1])
    return best[0], best[1]
