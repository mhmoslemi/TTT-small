import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h, lag_indices):
    n = len(h)
    G = np.zeros((len(lag_indices), n))
    for idx, r in enumerate(lag_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def _line_search_quad(h, d, dx, trust_radius):
    """Quadratic approximation line search for max-correlation minimization."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Find valid range
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    
    # The objective is max_j (S0[j] + t*T[j] - t^2*C[j])
    # This is a max of quadratics, each concave in t (since C[j] >= 0)
    # Use golden section search
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(50):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val * dx

def _lp_direction(h, scores, radius, top_k):
    """LP to find a descent direction focusing on top-k active constraints."""
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = _grad_matrix(h, active)
    act_scores = scores[active]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _subgradient_direction(h, scores, radius):
    """Average subgradient of top active constraints."""
    n = len(h)
    m = 2 * n - 1
    max_score = float(scores.max())
    # Use top constraints within 5% of max
    thr = max_score * 0.95
    active = np.where(scores >= thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    G = _grad_matrix(h, active)
    avg_grad = G.mean(axis=0)
    avg_grad -= avg_grad.mean()  # project to sum=0
    norm = np.linalg.norm(avg_grad)
    if norm < 1e-14:
        return None
    d = -avg_grad / norm * radius
    # Clip to feasible
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        t = np.min((1.0 - h[pos]) / d[pos])
        d[pos] = np.minimum(d[pos], t * d[pos])
    if np.any(neg):
        t = np.min(h[neg] / (-d[neg]))
        d[neg] = np.maximum(d[neg], -t * (-d[neg]))
    return d

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _random_binary_init(n, rng):

    h = np.zeros(n)

    n = len(h_init)

            break
        if patience > 5000:
            break
        if not ones_list or not zeros_list:

        i = ones_list[rng.integers(len(ones_list))]

        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5

            cur_c5 = new_c5

                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:

        else:
            h[i] = 1.0
            h[j] = 0.0

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:

def _grad_matrix(h, lag_indices):
    n = len(h)
    G = np.zeros((len(lag_indices), n))
    for idx, r in enumerate(lag_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:

                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds.append((None, None))

    m = 2 * n - 1
    max_score = float(scores.max())
    if max_score <= 0:
        return None
    # Exponential weighting: emphasize constraints closer to max
    weights = np.exp(50.0 * (scores - max_score) / max_score)
    weights /= weights.sum()
    m2 = 2 * n - 1
    active = np.arange(m2)
    G = _grad_matrix(h, active)
    grad = (weights[:, None] * G).sum(axis=0)
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    d = -grad / norm * radius
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        t = np.min((1.0 - h[pos]) / d[pos])
        d[pos] = np.minimum(d[pos], t * d[pos])
    if np.any(neg):
        t = np.min(h[neg] / (-d[neg]))
        d[neg] = np.maximum(d[neg], -t * (-d[neg]))
    return d

                h_r, c5_r = _optimize_lp(h_r, min(d4, time.time() + budget_s * 0.006), radius_init=0.05, top_k_start=6, top_k_max=10)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    
    # Phase 5: Pair refinement (8%)
    d5 = t0 + budget_s * 0.71
    h5, c5_5 = _pair_refine(best_h, d5, max_rounds=200)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Perturbation exploration (12%)
    d6 = t0 + budget_s * 0.83
    scale = 0.015
    for attempt in range(400):
        if time.time() > d6:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.005)
        h_t, c5_t = _optimize_lp(h_pert, sub_dl, radius_init=0.04, top_k_start=6, top_k_max=10)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.75)
        else:
            scale = min(0.08, scale * 1.3)
    
    # Phase 7: Final intensive polish (15%)
    h7, c5_7 = _optimize_lp(best_h, deadline, radius_init=0.03, top_k_start=8, top_k_max=16)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h7p, c5_7p = _pair_refine(best_h, deadline, max_rounds=100)
    if c5_7p < best_c5:
        best_c5 = c5_7p
        best_h = h7p.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0_fb = np.array(initial_h_values, dtype=float)
            h0_fb = _project(h0_fb, len(h0_fb) / 2.0)
            return h0_fb, _exact_c5(h0_fb), len(h0_fb)
        except NameError:
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _exact_c5(h_fb), len(h_fb)
    
    return best_h, final_c5, len(best_h)
