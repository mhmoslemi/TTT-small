def _smart_pair_descent(h, deadline, max_sweeps=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        S0 = np.correlate(h, 1.0 - h, mode='full') * dx
        S_max = float(np.max(S0))
        
        # Find active lags (within 10% of max)
        active = np.where(S0 > 0.9 * S_max)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(S0))])
        
        # Compute gradient: weighted sum over active lags
        # g[i] = sum_q w[q] * dS[q]/dh[i]
        # dS[q]/dh[i] = dx * ((1-h[i+q-n+1]) - h[i-q+n-1]) (with bounds)
        w = S0[active] / float(np.sum(S0[active]))
        
        g = np.zeros(n)
        for q in active:
            k = q - (n - 1)  # lag
            # dS[q]/dh[i] = dx * ((1-h[i+k]) - h[i-k]) (with appropriate bounds)
            if k >= 0:
                # i+k in [0,n-1] means i in [0, n-1-k]
                for i in range(n - k):
                    g[i] += w[active == q] * dx * (1.0 - h[i+k])
                # i-k in [0,n-1] means i in [k, n-1]
                for i in range(k, n):
                    g[i] -= w[active == q] * dx * h[i-k]
            else:
                k_abs = -k
                # i+k = i-k_abs in [0,n-1] means i in [k_abs, n-1]
                for i in range(k_abs, n):
                    g[i] += w[active == q] * dx * (1.0 - h[i-k_abs])
                # i-k = i+k_abs in [0,n-1] means i in [0, n-1-k_abs]
                for i in range(n - k_abs):
                    g[i] -= w[active == q] * dx * h[i+k_abs]
        
        # Sort positions by gradient: high g → decrease, low g → increase
        order = np.argsort(g)
        
        improved = False
        # Try top pairs: decrease high-g, increase low-g
        tried = 0
        for ii in range(min(20, len(order))):
            if time.time() > deadline:
                break
            j = order[ii]  # low gradient → increase
            if h[j] >= 1.0 - 1e-6:
                continue
            for jj in range(min(20, len(order))):
                if tried > 200:
                    break
                i = order[len(order) - 1 - jj]  # high gradient → decrease
                if i == j:
                    continue
                if h[i] < 1e-6:
                    continue
                
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                
                alpha_lo = 0.0
                alpha_hi = min(h[i], 1.0 - h[j])
                if alpha_hi < 1e-10:
                    continue
                
                T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
                C = np.correlate(d, d, mode='full') * dx
                
                # Line search
                n_grid = 15
                grid = np.linspace(alpha_lo, alpha_hi, n_grid)
                vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
                obj = np.max(vals, axis=1)
                best_idx = int(np.argmin(obj))
                best_a = grid[best_idx]
                best_v = obj[best_idx]
                
                if best_a > 1e-8 and best_v < best_c5 - 1e-12:
                    h = h + best_a * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        S0 = np.correlate(h, 1.0 - h, mode='full') * dx
                        break
                tried += 1
        
        if not improved:
            # Try random pairs as fallback
            rng_local = np.random.default_rng(sweep)
            for _ in range(50):
                i = rng_local.integers(n)
                j = rng_local.integers(n)
                if i == j or h[i] < 1e-6 or h[j] > 1.0 - 1e-6:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha_hi = min(h[i], 1.0 - h[j])
                if alpha_hi < 1e-10:
                    continue
                T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
                C = np.correlate(d, d, mode='full') * dx
                n_grid = 15
                grid = np.linspace(0.0, alpha_hi, n_grid)
                vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
                obj = np.max(vals, axis=1)
                best_idx = int(np.argmin(obj))
                best_a = grid[best_idx]
                best_v = obj[best_idx]
                if best_a > 1e-8 and best_v < best_c5 - 1e-12:
                    h = h + best_a * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        break
            if not improved:
                break
    
    return best_h, best_c5
