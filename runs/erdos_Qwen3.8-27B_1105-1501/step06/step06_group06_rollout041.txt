def _coord_descent_step(h, i):
    """Find optimal h[i] given all other coordinates."""
    n = len(h)
    old = h[i]
    
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    
    k0 = n - 1
    C_0 = corr0[k0]
    A = corr1 - corr0
    A[k0] = 0.0
    C = corr0.copy()
    
    # Golden section search for min of max(C_0 + x - x^2, max_k(C_k + A_k*x))
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def _obj(x):
        f0 = C_0 + x - x * x
        g = np.max(C + A * x)
        return max(f0, g)
    
    fa = _obj(a)
    fb = _obj(b)
    # Initial coarse grid to find a good starting bracket
    n_coarse = 20
    xs = np.linspace(0, 1, n_coarse)
    vals = [max(C_0 + x - x*x, float(np.max(C + A*x))) for x in xs]
    best_i = int(np.argmin(vals))
    if best_i > 0: a = xs[best_i - 1]
    if best_i < n_coarse - 1: b = xs[best_i + 1]
    fa = _obj(a)
    fb = _obj(b)
    
    for _ in range(50):
        if b - a < 1e-14:
            break
        m1 = b - gr * (b - a)
        m2 = a + gr * (b - a)
        fm1 = _obj(m1)
        fm2 = _obj(m2)
        if fm1 < fm2:
            b, fb = m2, fm2
        else:
            a, fa = m1, fm1
    x_star = (a + b) / 2.0
    h[i] = x_star

def _line_search(h, d, radius, deadline=None):
    """Exact line search along direction d from h."""
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    # Quadratic terms
    C = np.zeros(2 * n - 1)
    for s in range(-(n-1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            C[k] = -np.sum(d[:m] * d[s:s+m])
        else:
            g = -s
            C[k] = -np.sum(d[g:g+m] * d[:m])
    
    # Feasible t range
    t_lo, t_hi = -radius, radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section on the exact quadratic
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def _obj(t):
        return float(np.max(S0 + t * T + t * t * C)) * dx
    
    a, b = t_lo, t_hi
    fa = _obj(a)
    fb = _obj(b)
    for _ in range(50):
        if b - a < 1e-13:
            break
        m1 = b - gr * (b - a)
        m2 = a + gr * (b - a)
        fm1 = _obj(m1)
        fm2 = _obj(m2)
        if fm1 < fm2:
            b, fb = m2, fm2
        else:
            a, fa = m1, fm1
    t_star = (a + b) / 2.0
    return t_star, _obj(t_star)

def _active_descent(h_init, deadline, max_iter=2000):
    """Active set line search method."""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.05
    stall = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            # Perturb and restart
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.05
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            continue
        
        # Get correlations
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        M = float(np.max(corr))
        
        # Active set: lags close to the maximum
        tol = max(0.001, 0.02 * M)
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:10]
        elif len(active) > 30:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:30]]
        
        # Compute gradient of active correlations
        G = _corr_grads(h)
        G_act = G[active]
        
        # Direction: negative sum of active gradients, projected to sum-zero
        d = -G_act.sum(axis=0)
        d -= d.mean()  # project to sum-zero
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            # Try LP direction
            d = _lp_direction(h, active, radius)
            if d is None or np.max(np.abs(d)) < 1e-13:
                stall += 5
                radius *= 0.7
                if radius < 1e-6:
                    break
                continue
        else:
            d = d / d_norm * radius
        
        # Line search
        t_star, new_c5 = _line_search(h, d, radius)
        
        if new_c5 < cur_c5 - 1e-14:
            h = h + t_star * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            if abs(t_star) > 0.5 * radius:
                radius = min(radius * 1.1, 0.2)
            else:
                radius = min(radius * 1.03, 0.15)
        else:
            stall += 1
            radius *= 0.8
            if radius < 1e-6:
                break
    
    return best_h, best_c5

def _lp_direction(h, active_lags, radius):
    """Find a descent direction using LP."""
    n = len(h)
    G = _corr_grads(h)
    G_act = G[active_lags]
    scores = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(active_lags), n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = -scores[active_lags]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.fun < 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_step(h0, deadline, max_outer=10):
    """SLSQP optimization step."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        K = min(20, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]
        
        t_init = cur_c5 + 1e-5
        
        def obj_fn(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]
        
        def ineq_jac(x):
            hx = x[:n]
            G = _corr_grads(hx)
            J = np.zeros((K, n + 1))
            J[:, :n] = -G[top_k, :]
            J[:, n] = 1.0
            return J
        
        def eq_fn(x):
            return np.sum(x[:n]) - target
        
        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-14})
        
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break
    
    return best_h, best_c5

At shift -1: similarly c_{-1} ≈ n/2.
At shift 2: h[n+2] = h[n], so c_2 = 0.
Max is at shift 1: C₅ = (n/2) * 2/n = 1. Bad.

For a "spread out" pattern where ones are evenly distributed: if we can make the ones and zeros as "unpredictable" as possible (like a random sequence), then for each shift k, the expected overlap is (n/2) * (1/2) = n/4, giving C₅ ≈ 1/2. Better but still above our target of 0.38.

So the continuous optimum (with fractional values) is significantly better than any binary pattern. This makes sense because the continuous relaxation allows more flexibility.
