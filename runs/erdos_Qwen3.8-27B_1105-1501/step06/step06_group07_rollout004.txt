import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection onto {0<=x<=1, sum(x)=target}
    n = len(v)
    v = v.copy()
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _fw_direction(h, g):
    """Frank-Wolfe direction: go from h toward argmin g·v on the polytope."""
    n = len(h)
    # Sort g, assign 1 to smallest, 0 to largest
    order = np.argsort(g)
    v = np.zeros(n)
    v[order[:n//2]] = 1.0
    if n % 2 == 0:
        pass
    else:
        # Handle odd n: one value is 0.5
        v[order[:n//2]] = 1.0
        v[order[n//2]] = 0.5
    # Ensure sum is exactly n/2
    v = _project(v, n / 2.0)
    d = v - h
    d -= np.mean(d)  # Ensure sum zero (should already be, but numerical safety)
    return d

def _line_search_exact(h, d, dx):
    """Exact line search along direction d (assumes sum(d)=0)."""
    n = len(h)
    # Compute S, T, Q
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Feasible interval for t >= 0
    t_max = 10.0  # large default
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
    t_max = min(t_max, 2.0)  # cap
    
    if t_max < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Grid search
    n_grid = 60
    grid = np.linspace(0.0, t_max, n_grid)
    vals = S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]
    
    # Refine around best
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_max
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(30):
        if b - a < 1e-10:
            break
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
        if f1 < f2:
            b = c2
        else:
            a = c1
    mid_t = (a + b) / 2.0

    if mid_val < best_val:

    h_try = np.clip(h_try, 0.0, 1.0)
    h_try = _project(h_try, n / 2.0)
    exact_val = _exact_c5(h_try)
    
    return best_t, min(best_val, exact_val)

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad(h, k):
    """Gradient of_corr at lag k (no dx factor)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[:m] -= h[k:]
        g[k:] += 1.0 - h[:m]
    else:
        kp = -k
        m = n - kp
        if m <= 0:
            return g
        g[kp:] += 1.0 - h[:m]
        g[:m] -= h[kp:]
    return g

def _fw_dir(h, g):
    """FW direction: h -> argmin g.v on polytope - h."""
    n = len(h)
    order = np.argsort(g)
    v = np.zeros(n)
    half = n // 2
    v[order[:half]] = 1.0
    if n % 2 == 1:
        v[order[half]] = 0.5
    d = v - h
    d -= d.mean()
    return d

def _li_search(h, d, dx):
    """Exact line search. Returns (t, c5_value)."""
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    t_max = 5.0
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
    if t_max < 1e-12:
        return 0.0, _exact_c5(h)
    
    ng = 50
    grid = np.linspace(0.0, t_max, ng)
    vals = S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    ov = np.max(vals, axis=1) * dx
    bi = int(np.argmin(ov))
    bt, bv = grid[bi], ov[bi]
    
    # Golden refine
    if bi > 0 and bi < ng - 1:
        a, b = grid[bi-1], grid[bi+1]
    else:
        a, b = 0.0, t_max
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(35):
        if b - a < 1e-11:
            break
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(S + c1 * T - c1*c1 * Q)) * dx
        f2 = float(np.max(S + c2 * T - c2*c2 * Q)) * dx
        if f1 < f2:
            b = c2
        else:
            a = c1
    mt = (a+b) * 0.5
    mv = float(np.max(S + mt*T - mt*mt*Q)) * dx
    if mv < bv:
        bt, bv = mt, mv
    
    # Check t=0
    f0 = float(np.max(S)) * dx
    if f0 < bv:
        bt, bv = 0.0, f0
    
    return bt, bv

def _active_grad(h, K=40, alpha=30.0):
    """Weighted gradient from top-K active correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = len(corr)
    K = min(K, m)
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    w /= w.sum()
    g = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        g += wi * _corr_grad(h, k)
    return g

def _binary_swap_opt(h, deadline, rng, max_iters=200000):
    """Binary local search via swaps."""
    n = len(h)
    target = n // 2
    # Make binary
    order = np.argsort(-h)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c = _exact_c5(h)
        if new_c < cur - 1e-15:
            cur = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def _continuous_opt(h, deadline, K=40, alpha=25.0, max_iters=5000):
    """Continuous optimization with FW + exact line search."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    dx = 2.0 / n
    stall = 0
    alpha_cur = alpha
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = _active_grad(h, K, alpha_cur)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = _fw_dir(h, g)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        t, f_try = _li_search(h, d, dx)
        if t > 1e-10:
            h_new = h + t * d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, n / 2.0)
            f_new = _exact_c5(h_new)

                best_c5 = f_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 50:
            alpha_cur *= 1.5
            if alpha_cur > 500:
                alpha_cur = 500.0
            stall = 0
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
