def _compute_TC(h, i, j, a):
    """Compute T and C for direction d with d[i]=a, d[j]=-a."""
    n = len(h)
    m = 2*n - 1
    b = -a
    T = np.zeros(m)
    C = np.zeros(m)
    
    # C: 3 non-zero entries
    C[n-1] = a*a + b*b
    C[j-i+n-1] = a*b  # j-i+n-1 is in [n, 2n-2] since j>i
    C[i-j+n-1] = b*a  # i-j+n-1 is in [0, n-2] since i<j
    # Wait, I assumed i<j. Let me handle both cases.
    # Actually j-i+n-1: if j>i, this is in [n, 2n-2]. If j<i, this is in [0, n-2].
    # i-j+n-1: if i>j, this is in [n, 2n-2]. If i<j, this is in [0, n-2].
    # Both are valid indices in [0, 2n-2].
    
    # T: 4 shifted array contributions
    # Term 1: a*(1-h)[i+k-(n-1)], valid for k in [n-1-i, 2n-2-i]
    lo = max(0, n-1-i)
    hi = min(m-1, 2*n-2-i)
    if hi >= lo:
        idx = np.arange(lo, hi+1)
        T[idx] += a * (1.0 - h[i:i+(hi-lo+1)])
    
    # Term 2: b*(1-h)[j+k-(n-1)], valid for k in [n-1-j, 2n-2-j]
    lo = max(0, n-1-j)
    hi = min(m-1, 2*n-2-j)
    if hi >= lo:
        T[np.arange(lo, hi+1)] += b * (1.0 - h[j:j+(hi-lo+1)])

    # h[i-k+(n-1)] for k from lo to hi: index = i-k+(n-1) goes from i-lo+(n-1) down to i-hi+(n-1)
    lo = max(0, i)
    hi = min(m-1, i+n-1)
    if hi >= lo:
        # index = i - k + (n-1), for k in [lo, hi]
        # when k=lo: idx_h = i - lo + (n-1)
        # when k=hi: idx_h = i - hi + (n-1)

        idx_h_start = i - lo + (n-1)
