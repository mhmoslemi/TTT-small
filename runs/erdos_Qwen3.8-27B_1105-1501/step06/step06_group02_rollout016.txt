import numpy as np
import time
from scipy.optimize import linprog

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_row_grads(h, rows):
    """Compute gradients for specified lag rows.
    rows: array of lag indices k in range [0, 2n-2]
    Returns: (len(rows), n) array of gradients
    """
    n = len(h)
    grads = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        d = k - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            # R_k = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
            # d/dh[i] for i in 0..m-1: -h[i+d]
            # d/dh[i] for i in d..d+m-1: (1-h[i-d])
            grads[idx, :m] += -h[d:d+m]
            grads[idx, d:d+m] += (1.0 - h[:m])
        else:
            r = -d
            # R_k = sum_{i=0}^{m-1} h[i]*(1-h[i+r])
            # d/dh[i] for i in 0..m-1: (1-h[i+r])
            # d/dh[i] for i in r..r+m-1: -h[i-r]
            grads[idx, :m] += (1.0 - h[r:r+m])
            grads[idx, r:r+m] += -h[:m]
    return grads

def _exact_line_search(h, p, S0, dx):
    """Exact line search along direction p. Returns (best_alpha, best_c5)."""
    n = len(h)
    A = np.correlate(p, 1.0 - h, 'full') - np.correlate(h, p, 'full')
    C = np.correlate(p, p, 'full')
    
    # Find valid alpha range
    lo, hi = -1.0, 1.0
    pos = p > 1e-12
    neg = p < -1e-12
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / p[neg])))
    
    if hi - lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    def f(alpha):
        return float(np.max(S0 + alpha * A - alpha * alpha * C)) * dx
    
    # Multi-start: evaluate at several points
    candidates = np.linspace(lo, hi, 15)
    vals = [f(a) for a in candidates]
    best_idx = int(np.argmin(vals))
    best_a = candidates[best_idx]
    best_v = vals[best_idx]
    
    # Refine with golden section around best candidate
    a_lo = max(lo, candidates[max(0, best_idx-1)])
    a_hi = min(hi, candidates[min(len(candidates)-1, best_idx+1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = a_lo
    b = a_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    opt = (a + b) / 2.0
    val = f(opt)
    if val < best_v:
        best_a, best_v = opt, val
    
    # Check endpoints
    f_lo = f(lo)
    f_hi = f(hi)
    if f_lo < best_v:
        best_a, best_v = lo, f_lo
    if f_hi < best_v:
        best_a, best_v = hi, f_hi
    
    return best_a, best_v

def _lp_direction(h, active_rows, max_t=1e6):
    """Find LP minimax direction for active rows.
    Returns direction p (zero-sum) or None if infeasible.
    """
    n = len(h)
    grads = _compute_row_grads(h, active_rows)
    m = len(active_rows)
    
    # LP: minimize t
    # subject to: grads[i] @ p <= t for i in range(m)
    #             sum(p) = 0
    #             -1 <= p_j <= 1
    # Variables: p[0..n-1], t
    # Total: n+1 variables
    
    # Objective: [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active row i: grads[i] @ p - t <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -1 <= p_j <= 1, t <= max_t
    bounds = [(-1.0, 1.0)] * n + [(0.0, max_t)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            return p
    except Exception:
        pass
    return None

def _lp_direction_simple(h, active_rows):
    """Simpler LP direction: minimize weighted sum of active row decreases.
    Uses the negative weighted gradient as a starting point and refines.
    """
    n = len(h)
    scores = np.correlate(h, 1.0 - h, mode='full')
    max_score = float(np.max(scores))
    weight = np.exp((scores[active_rows] - max_score) / (0.01 * max_score + 1e-15))
    weight /= float(np.sum(weight))
    
    grads = _compute_row_grads(h, active_rows)
    g = grads.T @ weight
    g -= np.mean(g)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return None
    return -g / gn

def _pair_search_v2(h, deadline, n_pairs=30):

    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.time() < deadline and stall < 100:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.01 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]
        
        grads = _compute_row_grads(h, active_idx)
        weight = np.exp((scores[active_idx] - max_score) / (0.01 * max_score + 1e-15))
        weight /= float(np.sum(weight))
        wgrad = grads.T @ weight
        
        # Find best pair: top wgrad (donor) and bottom wgrad (receiver)
        donor_pool = np.argsort(-wgrad)[:n_pairs]
        recv_pool = np.argsort(wgrad)[:n_pairs]
        
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        
        tried = set()
        for i in donor_pool:
            if time.time() > deadline:
                break
            for j in recv_pool:
                if i == j:
                    continue
                key = (min(int(i), int(j)), max(int(i), int(j)))
                if key in tried:
                    continue
                tried.add(key)
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, new_c5 = _exact_line_search(h, d, S0, dx)
                if abs(alpha) > 1e-10 and new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:

            if stall > 20 and rng.random() < 0.3:

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)

def _multi_start_optimize(h, n, deadline):
    """Try several perturbations and pick the best."""
    rng = np.random.default_rng(999)
    best = h.copy()
    best_c5 = c5_exact(h)
    
    for attempt in range(20):
        if time.time() > deadline:
            break
        h_try = h + rng.standard_normal(n) * 0.02
        h_try = project(h_try, n / 2.0)
        c5_try = c5_exact(h_try)
        if c5_try < best_c5:
            best = h_try.copy()
            best_c5 = c5_try
    
    return best, best_c5

import numpy as np

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_row_grads(h, rows):
    n = len(h)
    grads = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        d = int(k) - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            grads[idx, :m] += -h[d:d + m]
            grads[idx, d:d + m] += (1.0 - h[:m])
        else:
            r = -d
            grads[idx, :m] += (1.0 - h[r:r + m])
            grads[idx, r:r + m] += -h[:m]
    return grads

def _lp_direction(h, active_rows):
    n = len(h)
    grads = _compute_row_grads(h, active_rows)
    m = len(active_rows)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-1.0, 1.0)] * n + [(0.0, 1e6)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            p = res.x[:n]
            p -= np.mean(p)
            pn = np.linalg.norm(p)
            if pn > 1e-12:
                return p / pn
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

    for i in range(n):

        if pos_in_period < duty:

    return project(h, n / 2.0)

        sub_dl2 = min(d3, time.time() + 15)
        h_r, c5_r = _pair_search_v2(h_r, sub_dl2, n_pairs=15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Gradient polish (15% time)
    d4 = t0 + budget_s * 0.80
    h_p4, c5_p4 = _gradient_polish_v2(best_h, d4, active_frac=0.03)
    if c5_p4 < best_c5:
        best_c5 = c5_p4
        best_h = h_p4.copy()

    # Phase 5: Alternative initializations (10% time)
    d5 = t0 + budget_s * 0.90
    for attempt in range(8):
        if time.time() > d5:
            break
        x = np.arange(best_n)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.5)
        h_alt = 0.5 + amp * np.sin(2 * np.pi * freq * x / best_n + phase)
        h_alt = project(h_alt, best_n / 2.0)
        sub_dl = min(d5, time.time() + 12)
        h_alt, c5_alt = _gradient_polish_v2(h_alt, sub_dl, active_frac=0.06, use_lp=False)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    # Phase 6: Final polish (remaining time)
    h_p6, c5_p6 = _gradient_polish_v2(best_h, deadline, active_frac=0.02)
    if c5_p6 < best_c5:
        best_c5 = c5_p6
        best_h = h_p6.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
