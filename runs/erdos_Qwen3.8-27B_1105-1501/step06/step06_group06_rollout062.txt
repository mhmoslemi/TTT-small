    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    e = np.exp(beta * (corr - M))
    p = e / e.sum()
    val = M + np.log(e.sum()) / beta

    L = 2 * n - 1
    grad = np.zeros(n)
    # p[k] corresponds to lag s = k - (n-1)
    for j in range(n):
        # Term 1: sum_s p_s * (1-h[j+s]) for valid j+s
        # s ranges from max(-(n-1), -j) to min(n-1, n-1-j)
        s_lo = max(-(n-1), -j)
        s_hi = min(n-1, n-1-j)
        if s_lo <= s_hi:
            # j+s ranges from j+s_lo to j+s_hi
            u_lo, u_hi = j + s_lo, j + s_hi
            k_lo, k_hi = n-1+s_lo, n-1+s_hi
            t1 = np.dot(p[k_lo:k_hi+1], 1.0 - h[u_lo:u_hi+1])
        else:
            t1 = 0.0
        # Term 2: sum_s p_s * h[j-s] for valid j-s
        # j-s valid means 0 <= j-s < n, so s in [j-n+1, j] intersect [-(n-1), n-1]
        s_lo2 = max(-(n-1), j-(n-1))
        s_hi2 = min(n-1, j)
        if s_lo2 <= s_hi2:
            # j-s ranges from j-s_hi2 to j-s_lo2 (decreasing)
            v_lo, v_hi = j - s_hi2, j - s_lo2
            k_lo2, k_hi2 = n-1+s_lo2, n-1+s_hi2
            t2 = np.dot(p[k_lo2:k_hi2+1], h[v_hi:v_lo-1:-1])
        else:
            t2 = 0.0
        grad[j] = t1 - t2
    
    return val, grad

        v, _ = _lse_obj_grad(x, beta, target)

        _, g = _lse_obj_grad(x, beta, target)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo+hi)/2
        if np.sum(np.clip(v-mid, 0, 1)) > target: lo = mid
        else: hi = mid
    x = np.clip(v-(lo+hi)/2, 0, 1)
    r = target - x.sum()
    if abs(r) > 1e-14:
        m = (x<1) if r>0 else (x>0)
        c = m.sum()
        if c: x[m] += r/c
        x = np.clip(x, 0, 1)
    return x

def _lse_value_grad(h, beta):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full') * dx
    M = corr.max()
    e = np.exp(beta*(corr - M))
    se = e.sum()
    p = e / se
    val = M + np.log(se) / beta
    grad = np.zeros(n)
    for j in range(n):
        t1 = 0.0; t2 = 0.0
        s_lo = max(-(n-1), -j); s_hi = min(n-1, n-1-j)
        for s in range(s_lo, s_hi+1):
            t1 += p[n-1+s] * (1.0 - h[j+s])
        s_lo2 = max(-(n-1), j-(n-1)); s_hi2 = min(n-1, j)
        for s in range(s_lo2, s_hi2+1):
            t2 += p[n-1+s] * h[j-s]
        grad[j] = t1 - t2
    return val, grad

                   constraints=cons, options={'maxiter':maxiter, 'ftol':1e-15})

    """Coordinate descent with exact evaluation"""
    n = len(h)
    target = n/2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(123)
    stall = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        j = rng.integers(0, n)
        # Try to find best new value for h[j], adjusting the "mean" of others
        # Simple approach: try a range of values for h[j], project the rest
        best_local = cur_c5
        best_val = h[j]
        old_j = h[j]
        # Try values
        candidates = np.linspace(0, 1, 21)
        for val in candidates:
            h[j] = val
            # Project to maintain sum
            target_remaining = target - val
            others = np.delete(h, j)
            others_new = _project(others, target_remaining)
            h_idx = 0

                h[i] = others_new[h_idx]

            c5_try = _exact_c5(h)
            if c5_try < best_local - 1e-12:

                best_val = val
        h[j] = best_val
        if best_local < cur_c5 - 1e-13:

            # Re-project to fix sum

                h = _project(h, target)

                stall = 0
