def _project(v, target):
    """Project v onto {0 <= x <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x = np.clip(x, 0.0, 1.0)

    n = len(h)

def _project(v, target):
    n = len(v)

        return np.zeros(n)

        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

    res = target - float(np.sum(x))

def _soft_grad(h, beta):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-10:

                i1 = 2 * m // 3
                i2 = dd + 2 * m // 3
            if h[i1] >= 0.99 or h[i1] <= 0.01 or h[i2] >= 0.99 or h[i2] <= 0.01:
                continue
            d = np.zeros(n)
            if h[i1] > h[i2]:
                d[i1] = -1.0
                d[i2] = 1.0
            else:
                d[i2] = -1.0
                d[i1] = 1.0
            d -= d.mean()
            alpha, c5_try = _line_search(h, d, dx, 0.35)
            if c5_try < best_c5 - 1e-13:
                h = _project(h + alpha * d, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        if not improved:
            interior = np.where((h > 0.05) & (h < 0.95))[0]
            if len(interior) < 4:
                break
            idx = rng.choice(len(interior), size=min(4, len(interior)), replace=False)
            i, j, kk, l = interior[idx]
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = -1.0
            d[kk] = 1.0
            d[l] = 1.0
            d -= d.mean()
            alpha, c5_try = _line_search(h, d, dx, 0.3)
            if c5_try < best_c5 - 1e-13:
                h = _project(h + alpha * d, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
            if not improved:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.98

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start soft-max gradient (22% budget)
    d1 = t0 + budget_s * 0.22
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5]:
        for ph in [0.0, 1.0, 2.0, 3.0]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(6):
        starts.append(_block_init(n0, rng, rng.integers(4, 20)))
    # Also try slightly different n
    for dn in [-4, 4, -8, 8]:
        nn = n0 + dn
        if nn >= 60 and nn % 2 == 0:
            starts.append(_resample(best_h, nn))
            starts.append(_sine_init(nn, 3, 1.5))

    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.015)
        h_s, c5_s = _fista_softmax(s, sub_dl, beta_init=80.0, beta_min=15.0, lr_init=0.12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Intensive FISTA + gap-pair (22% budget)
    d2 = t0 + budget_s * 0.44
    h2, c5_2 = _fista_softmax(best_h, min(d2, time.time() + budget_s * 0.12),
                              beta_init=200.0, beta_min=20.0, lr_init=0.1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    if time.time() < d2:
        h2b, c5_2b = _gap_pair_move(best_h, d2)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: LP polish (12% budget)
    d3 = t0 + budget_s * 0.56
    h3, c5_3 = _lp_polish(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP (18% budget)
    d4 = t0 + budget_s * 0.74
    h4a, c5_4a = _binary_sa(best_h, min(d4, time.time() + budget_s * 0.10), rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_polish(best_h, d4)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Pair refinement (10% budget)
    d5 = t0 + budget_s * 0.84
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (10% budget)
    d6 = t0 + budget_s * 0.94
    base_n = len(best_h)
    tried = set()
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, -24, 24, -28, 28]:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d6, time.time() + budget_s * 0.025)
            h_r, c5_r = _fista_softmax(h_r, sub_dl, beta_init=150.0, beta_min=25.0, lr_init=0.1)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.01:
                h_r, c5_r = _lp_polish(h_r, min(d6, time.time() + budget_s * 0.015))
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 7: Final aggressive refinement (5% budget)
    d7 = t0 + budget_s * 0.975
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=150)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < deadline:
        h7b, c5_7b = _lp_polish(best_h, deadline)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    n_final = len(best_h)
    return best_h, final_c5, n_final
