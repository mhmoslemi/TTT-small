        # Shift s: g_s = sum_{j} h[j]*(1-h[j+s]) for valid j
        # dg/dh[i] = (1-h[i+s])*I(valid) - h[i-s]*I(valid)
        # First term: (1-h[i+s]) for i where 0<=i<n and 0<=i+s<n
        lo1 = max(0, -s)
        hi1 = min(n, n - s)  # exclusive
        if hi1 > lo1:
            g[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
        # Second term: -h[i-s] for i where 0<=i<n and 0<=i-s<n
        lo2 = max(0, s)
        hi2 = min(n, n + s)  # exclusive
        if hi2 > lo2:
            g[lo2:hi2] -= h[lo2-s:hi2-s]
    return g

def _line_search_quadratic(S0, T, Q, L, U):
    """Minimize max_k (S0[k] + t*T[k] - t^2*Q[k]) for t in [L,U]."""
    # Use golden section search
    if U - L < 1e-14:
        t = 0.5*(L+U)
        return t, float(np.max(S0 + t*T - t*t*Q))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    # First do a coarse grid to narrow down
    grid = np.linspace(a, b, 80)
    vals = S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*Q[None,:]
    obj = np.max(vals, axis=1)
    i = np.argmin(obj)
    a = grid[max(i-1, 0)]
    b = grid[min(i+1, len(grid)-1)]
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*Q))
    f2 = float(np.max(S0 + c2*T - c2*c2*Q))
    for _ in range(50):
        if b - a < 1e-14: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*Q))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*Q))
    t = 0.5*(a+b)
    ft = float(np.max(S0 + t*T - t*t*Q))
    # Also check the grid minimum
    if obj[i] < ft:
        t = grid[i]
        ft = float(obj[i])
    return t, ft

def _pair_direction(h, i, j, rng):
    """Direction that transfers mass from i to j (or vice versa)."""
    n = len(h)
    d = np.zeros(n)
    if h[i] > h[j]:
        d[i] = -1.0
        d[j] = 1.0
    else:
        d[i] = 1
        d[j] = -1
    d -= d.mean()
    nrm = np.linalg.norm(d)
    return d / nrm

def _subgradient_direction(h, corr, k_active, tau, rng):
    """Compute subgradient direction from active constraints."""
    n = len(h)
    # Softmax weighting
    m = float(np.max(corr))
    if tau < 1e-15:
        w = (corr == m).astype(float)
    else:
        w = np.exp((corr - m) / tau)
        w /= np.sum(w)
    # Pick top-k for efficiency
    top_k = np.argsort(-corr)[:k_active]
    G = np.zeros(n)
    for idx in top_k:
        s = idx - (n - 1)
        # Add gradient for shift s
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            G[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            G[lo2:hi2] -= h[lo2-s:hi2-s]
    d = -G
    d -= d.mean()
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = np.linalg.norm(d)
    return d / nrm

def _c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(4):
        if abs(res) < 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _subgrad_dir(h, corr, k_active, tau, rng):
    n = len(h)
    m = float(np.max(corr))
    if tau < 1e-10:
        w = (corr >= m - 1e-10).astype(float)
    else:
        w = np.exp((corr - m) / tau)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.ones_like(corr) / len(corr)
    top_k = np.argsort(-corr)[:k_active]
    G = np.zeros(n)
    for idx in top_k:
        s = int(idx - (n - 1))
        wt = w[idx] / w_sum
        # Gradient of g_s w.r.t. h
        # g_s = sum_{valid i} h[i]*(1-h[i+s])
        # d/dh[i] = (1-h[i+s])*I(i+s valid) - h[i-s]*I(i-s valid)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            G[lo1:hi1] += wt * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            G[lo2:hi2] -= wt * h[lo2 - s:hi2 - s]
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm

def _witness_pair_dir(h, corr, rng):
    """Find a direction based on witness positions of the most active shift."""
    n = len(h)
    # Find the most active shift
    idx_max = int(np.argmax(corr))
    s = idx_max - (n - 1)
    # Find witness positions: i where h[i] is high and h[i+s] is low
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if hi1 <= lo1:
        # No valid positions, return random
        d = rng.standard_normal(n)
        d -= d.mean()
        return d / max(np.linalg.norm(d), 1e-14)
    # Contribution from each position
    contrib = h[lo1:hi1] * (1.0 - h[lo1 + s:hi1 + s])
    # Pick the top contributor
    i_local = int(np.argmax(contrib))
    i = lo1 + i_local
    j = i + s
    # Direction: transfer from i to j
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    return d / max(nrm, 1e-14)

def _line_search(h, d, t_lo, t_hi, target):
    """Golden section line search minimizing _c5_eval(h + t*d)."""
    n = len(h)
    dx = 2.0 / n
    target_f = target  # for projection if needed
    
    def f_at(t):
        hn = h + t * d
        if hn.min() < -1e-12 or hn.max() > 1.0 + 1e-12:
            return 1e10
        return float(np.max(np.correlate(hn, 1.0 - hn, mode='full'))) * dx
    
    if t_hi - t_lo < 1e-13:
        t = 0.5 * (t_lo + t_hi)
        return t, f_at(t)
    
    # Coarse grid first
    n_grid = 60
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([f_at(t) for t in grid])
    i_min = int(np.argmin(vals))
    a = grid[max(i_min - 2, 0)]
    b = grid[min(i_min + 2, n_grid - 1)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_at(c1)

        if b - a < 1e-13:
            break
        if f1 < f2:

            c1 = b - gr * (b - a)
            f1 = f_at(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_at(c2)
    t = 0.5 * (a + b)
    ft = f_at(t)
    if vals[i_min] < ft:
        t = grid[i_min]
        ft = float(vals[i_min])

def _step_range(h, d):
    """Find [t_lo, t_hi] such that h + t*d stays in [0,1]."""
    pos = d > 1e-14
    neg = d < -1e-14
    t_lo = 0.0
    t_hi = 0.0
    first = True
    if np.any(pos):
        lo_p = float(np.min(-h[pos] / d[pos]))
        hi_p = float(np.min((1.0 - h[pos]) / d[pos]))
        if first:
            t_lo, t_hi = lo_p, hi_p
            first = False
        else:
            t_lo = max(t_lo, lo_p)
            t_hi = min(t_hi, hi_p)
    if np.any(neg):
        lo_n = float(np.min((1.0 - h[neg]) / d[neg]))
        hi_n = float(np.min(-h[neg] / d[neg]))
        if first:
            t_lo, t_hi = lo_n, hi_n
            first = False
        else:
            t_lo = max(t_lo, lo_n)
            t_hi = min(t_hi, hi_n)
    if first:
        return 0.0, 0.0
    return t_lo, t_hi

def _binary_local_search(h_binary, deadline, rng, max_iter=5000):
    """Local search on binary h: swap a 1 with a 0."""
    n = len(h_binary)

    cur_f = _c5_eval(h)
    best_h = h.copy()
    best_f = cur_f

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    T = 0.5  # initial temperature
    T_min = 0.01
    cool = 0.9997
    
    it = 0
    while time.time() < deadline and it < max_iter:
        # Pick random 1 and 0 to swap
        i = int(ones[int(rng.integers(len(ones)))])
        j = int(zeros[int(rng.integers(len(zeros)))])
        if i == j:
            continue
        
        # Try swap
        h[i], h[j] = h[j], h[i]
        new_f = _c5_eval(h)
        
        if new_f < best_f:
            best_f = new_f
            best_h = h.copy()
        
        # Acceptance: Metropolis criterion
        delta = new_f - cur_f
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_f = new_f
            corr = _corr_values(h)
            # Update ones/zeros
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[i], h[j] = h[j], h[i]  # undo
        
        T = max(T_min, T * cool)
        it += 1
    
    return best_h, best_f
