import numpy as np

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=target/n, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _corr_scores(h):

def _grad_for_lag(h, d):
    """Gradient of C_r = dx * correlate(h, 1-h, 'full')[r] with respect to h.
    d is the lag = r - (n-1)."""

    g = np.zeros(n)

        m = n - d
        if m > 0:

            g[:m] += (-h[d:])
    else:

        m = n - g2
        if m > 0:

def _lse_obj_and_grad(h, beta):
    """Log-sum-exp objective and its gradient."""
    scores = _corr_scores(h)
    max_s = float(scores.max())
    z = beta * (scores - max_s)
    exp_z = np.exp(z)
    sum_exp = float(exp_z.sum())
    p = exp_z / sum_exp
    obj = (1.0 / beta) * (max_s + np.log(sum_exp))
    # Gradient
    n = len(h)
    grad = np.zeros(n)
    for r in range(len(scores)):
        if p[r] > 1e-12:
            d = r - (n - 1)
            grad += p[r] * _grad_for_lag(h, d)
    return float(obj), grad

def _line_search_exact(h, d, dx, max_iterations=80):
    """Exact line search along direction d (zero-sum) for the max correlation."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')

    Q = -np.correlate(d, d, mode='full')
    
    # Feasible interval

    c2 = a + g * (b - a)
