    # Bounds

        np.maximum(-h, -rho),
        np.zeros(C)
    ])
    ub = np.concatenate([

        np.maximum(R - target_raw, 0.0) + rho + 1.0  # z upper

        g[:n] = 2 * lam_n * d / n
        g[n:] = lam_z / C

def c5_exact(h):
    n = len(h)

def project(v, target):
    v = np.asarray(v, dtype=float)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def compute_G(h):
    n = len(h)
    C = 2 * n - 1
    G = np.zeros((C, n))
    for q in range(C):
        d = q - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[q, d:] += (1.0 - h[:m])
                G[q, :m] -= h[d:]
        else:
            r = -d
            m = n - r
            if m > 0:
                G[q, :m] += (1.0 - h[r:])
                G[q, r:] -= h[:m]
    return G

def _qp_direction(h, R, G, M, rho, lam_z=5.0, lam_n=1.0):
    n = len(h)
    C = 2 * n - 1
    nvar = n + C
    zb = np.maximum(R - M, 0.0)
    x0 = np.zeros(nvar)
    x0[n:] = zb
    dl = np.maximum(-h, -rho)
    du = np.minimum(1.0 - h, rho)
    zl = np.zeros(C)
    zu = np.maximum(zb + rho * 2.0, 0.1)
    lb = np.concatenate([dl, zl])
    ub = np.concatenate([du, zu])
    z_scale = max(float(np.max(zb)), 0.01)

    def obj(x):
        d = x[:n]
        z = x[n:]
        return lam_n * float(d @ d) / (n * rho * rho) + lam_z * float(z.sum()) / (C * z_scale)

    def obj_grad(x):
        d = x[:n]
        g = np.zeros(nvar)
        g[:n] = 2.0 * lam_n * d / (n * rho * rho)
        g[n:] = lam_z / (C * z_scale)
        return g

    G_neg = -G
    J_ineq = np.zeros((C, nvar))
    J_ineq[:, :n] = G_neg
    J_ineq[:, n:] = -np.eye(C)
    J_eq = np.zeros((1, nvar))
    J_eq[0, :n] = 1.0

    def con_ineq(x):
        return M - (R + G @ x[:n]) - x[n:]

    def con_eq(x):
        return np.sum(x[:n])

    cons = [
        {'type': 'ineq', 'fun': con_ineq, 'jac': lambda x: J_ineq},
        {'type': 'eq', 'fun': con_eq, 'jac': lambda x: J_eq}
    ]
    bounds = list(zip(lb, ub))
    maxit = max(80, min(300, 40 + 3 * n))
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or not np.all(np.isfinite(res.x)):
        return None
    d = res.x[:n]
    d -= d.mean()
    if np.max(np.abs(d)) < 1e-14:
        return None
    return d

                S0 = np.correlate(h, 1.0 - h, 'full')
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
        T = max(1e-6, T * 0.9999)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.45 * np.sin(2 * np.pi * 2.0 * x / n0 + 0.5)

    h0 = project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = c5_exact(h0)
    best_n = n0

    # Phase 1: QP optimization (40% time)
    d1 = t0 + budget_s * 0.40
    h_q, c5_q = _qp_optimize(h0, d1, rng)
    if c5_q < best_c5:
        best_c5 = c5_q
        best_h = h_q.copy()

    # Phase 2: Pair search (15% time)
    d2 = t0 + budget_s * 0.55
    h_p, c5_p = _pair_search(best_h, 20, d2)
    if c5_p < best_c5:
        best_c5 = c5_p
        best_h = h_p.copy()

    # Phase 3: Multi-resolution (20% time)
    d3 = t0 + budget_s * 0.75
    tried_n = {best_n}
    for dn in [-6, 6, -4, 4, -8, 8, -10, 10, -2, 2]:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 240 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _qp_optimize(h_r, sub_dl, rng)
        sub_dl2 = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _pair_search(h_r, 15, sub_dl2)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: QP again on best (10% time)
    d4 = t0 + budget_s * 0.85
    h_q2, c5_q2 = _qp_optimize(best_h, d4, rng)
    if c5_q2 < best_c5:
        best_c5 = c5_q2
        best_h = h_q2.copy()

    # Phase 5: Random pair (5% time)
    d5 = t0 + budget_s * 0.90
    h_rp, c5_rp = _random_pair_search(best_h, d5, rng)
    if c5_rp < best_c5:
        best_c5 = c5_rp
        best_h = h_rp.copy()

    # Phase 6: Gradient polish (5% time)
    d6 = deadline
    h_g, c5_g = _gradient_polish(best_h, d6)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()

    # Final: re-evaluate and ensure validity
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
