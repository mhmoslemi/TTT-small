def _exact_c5(h):

    dx = 2.0 / n

def _project(v, target):
    # Water-filling projection to [0,1] with sum = target
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 1, float(v.max()) + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if mask.sum() > 0:
            x[mask] += res / mask.sum()
            x = np.clip(x, 0, 1)
    return x

def _smooth_grad(h, weights):

def _line_search_quad(h, d, raw, dx, radius, n_top=15):
    n = len(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Find feasible alpha range
    a_lo = 0.0
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        a_hi = min(a_hi, np.min(h[neg] / (-d[neg])))
    a_hi = max(a_hi, 1e-12)
    
    # Generate candidate alphas
    candidates = [0.0, a_hi]
    
    # Vertices of top quadratics
    top_idx = np.argsort(-raw)[:n_top]  # top n_top by current value
    for idx in top_idx:
        if Q[idx] > 1e-12:
            av = T[idx] / (2.0 * Q[idx])
            if 0 < av < a_hi:
                candidates.append(av)
    
    # Pairwise intersections of top quadratics
    for i in range(len(top_idx)):
        for j in range(i+1, len(top_idx)):
            ri, rj = top_idx[i], top_idx[j]
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-12:
                disc = B*B + 4*A*C
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    for sign in [1, -1]:
                        aa = (B + sign * sqrt_disc) / (2.0 * A)
                        if 0 < aa < a_hi:
                            candidates.append(aa)
            elif abs(B) > 1e-12:
                aa = -C / B
                if 0 < aa < a_hi:
                    candidates.append(aa)
    
    # Dense grid
    n_grid = 40
    grid = np.linspace(0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    
    # Evaluate all candidates
    best_alpha = 0.0
    best_val = float(np.max(raw) * dx)
    
    for alpha in candidates:
        if alpha < -1e-10 or alpha > a_hi + 1e-10:
            continue
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha*alpha * Q)) * dx
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Refine around best
    # Golden section on [max(0, best_alpha - step), min(a_hi, best_alpha + step)]
    step = a_hi / n_grid
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1*T - c1*c1*Q)) * dx
        f2 = float(np.max(raw + c2*T - c2*c2*Q)) * dx
        for _ in range(40):
            if hi - lo < 1e-12: break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr*(hi-lo)
                f1 = float(np.max(raw + c1*T - c1*c1*Q)) * dx

                lo, c1, f1 = c1, c2, f2

        opt = (lo+hi)/2
        val_opt = float(np.max(raw + opt*T - opt*opt*Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    
    return best_alpha, best_val

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 0.9999999
        else:
            mask = x > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    u = 1.0 - h
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-12:
            continue
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += w * u[:m]
                g[:m] -= w * h[d:]
        else:
            gp = -d
            m = n - gp
            if m > 0:
                g[:m] += w * u[gp:]
                g[gp:] -= w * h[:m]
    return g

def _line_search_quad(h, d, raw, dx, radius, n_top=18):
    n = len(h)
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    a_hi = radius
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    a_hi = max(a_hi, 1e-12)
    candidates = [0.0, a_hi]
    top_idx = np.argsort(-raw)[:n_top]
    for idx in top_idx:
        if Q[idx] > 1e-14:
            av = T[idx] / (2.0 * Q[idx])
            if 1e-10 < av < a_hi - 1e-10:
                candidates.append(av)
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            ri, rj = int(top_idx[i]), int(top_idx[j])
            A = Q[ri] - Q[rj]
            B = T[ri] - T[rj]
            C = raw[ri] - raw[rj]
            if abs(A) > 1e-14:
                disc = B * B + 4.0 * A * C
                if disc >= 0:
                    sd = np.sqrt(disc)
                    a1 = (B + sd) / (2.0 * A)
                    a2 = (B - sd) / (2.0 * A)
                    for aa in [a1, a2]:
                        if 1e-10 < aa < a_hi - 1e-10:
                            candidates.append(aa)
            elif abs(B) > 1e-14:
                aa = -C / B
                if 1e-10 < aa < a_hi - 1e-10:
                    candidates.append(aa)
    n_grid = 36
    grid = np.linspace(0.0, a_hi, n_grid)
    candidates.extend(grid.tolist())
    best_alpha = 0.0
    best_val = float(np.max(raw)) * dx
    for alpha in candidates:
        if alpha < -1e-9 or alpha > a_hi + 1e-9:
            continue
        alpha = max(0.0, min(alpha, a_hi))
        val = float(np.max(raw + alpha * T - alpha * alpha * Q)) * dx
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    step = a_hi / n_grid
    lo = max(0.0, best_alpha - step)
    hi = min(a_hi, best_alpha + step)
    if hi - lo > 1e-11:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(50):
            if hi - lo < 1e-13:
                break
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = hi - gr * (hi - lo)
                f1 = float(np.max(raw + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = lo + gr * (hi - lo)
                f2 = float(np.max(raw + c2 * T - c2 * c2 * Q)) * dx
        opt = (lo + hi) / 2.0
        val_opt = float(np.max(raw + opt * T - opt * opt * Q)) * dx
        if val_opt < best_val:
            best_val = val_opt
            best_alpha = opt
    return best_alpha, best_val

def _make_directions(h, g, n, rng, radius):
    directions = []
    d0 = -g.copy()
    d0 -= np.mean(d0)
    nm = np.linalg.norm(d0)
    if nm > 1e-12:
        d0 /= nm
        directions.append(d0)
    for bl in [4, 8, 16]:
        nb = n // bl
        if nb < 2:
            continue
        blk = np.zeros(n)
        for b in range(nb):
            sl = slice(b * bl, min((b + 1) * bl, n))
            blk[sl] = np.mean(g[sl])
        db = -blk.copy()
        db -= np.mean(db)
        nm = np.linalg.norm(db)
        if nm > 1e-12:
            directions.append(db / nm)
    idx_pos = np.argsort(g)[-3:]
    idx_neg = np.argsort(g)[:3]
    for ip in idx_pos:
        for ino in idx_neg:
            if ip == ino:
                continue
            dp = np.zeros(n)
            dp[ip] = -1.0
            dp[ino] = 1.0
            dp -= np.mean(dp)
            nm = np.linalg.norm(dp)
            if nm > 1e-12:
                directions.append(dp / nm)
    n_dirs = min(len(directions), 6)
    return directions[:n_dirs]

def _main_optimize(h, deadline, beta=20.0, radius=0.25, stall_limit=120):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    while time.time() < deadline:

        raw = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(raw))
        shifted = raw - mx
        vp = np.exp(np.clip(beta_cur * shifted, -80, 0))
        w = vp / (float(np.sum(vp)) + 1e-30)
        g = _smooth_grad(h, w)
        g -= np.mean(g)
        directions = _make_directions(h, g, n, rng, radius)
        accepted = False
        for d in directions:

            if alpha < 1e-10:

                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):

def _pair_sweep(h_bin, deadline):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.5, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.4, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_swap_cont(h6, deadline, n_pairs=1500, radius=0.12, seed=999)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
