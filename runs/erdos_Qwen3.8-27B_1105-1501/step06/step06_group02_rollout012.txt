            return logsum + sum_pen

import numpy as np

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_row(h, d_shift):
    n = len(h)
    g = np.zeros(n)
    m = n - abs(d_shift)
    if m <= 0:
        return g
    if d_shift >= 0:
        g[d_shift:d_shift + m] += (1.0 - h[:m])
        g[:m] += -h[d_shift:d_shift + m]
    else:
        r = -d_shift
        g[:m] += (1.0 - h[r:r + m])
        g[r:r + m] += -h[:m]
    return g


def _exact_1d(h, d, R0, max_golden=50):
    n = len(h)
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')

    pos = d > 1e-12
    neg = d < -1e-12
    lo_p = 0.0
    hi_p = 5.0
    if np.any(pos):
        hi_p = min(hi_p, float(np.min((1.0 - h[pos]) / d[pos])))
    lo_n = -5.0
    hi_n = 0.0
    if np.any(neg):
        lo_n = max(lo_n, float(np.max(h[neg] / d[neg])))

    def f(t):
        return float(np.max(R0 + t * D - t * t * Q))

    best_t = 0.0
    best_val = f(0.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    if hi_p > lo_p + 1e-14:
        ts = np.linspace(lo_p, hi_p, 16)
        vals = np.array([f(t) for t in ts])
        imin = int(np.argmin(vals))
        a = float(ts[max(0, imin - 1)])
        b = float(ts[min(len(ts) - 1, imin + 1)])
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1, f2 = f(c1), f(c2)
        for _ in range(max_golden):
            if b - a < 1e-14:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = f(c1)
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = f(c2)
        mid = (a + b) / 2.0
        fm = f(mid)
        if fm < best_val:
            best_val, best_t = fm, mid
        if f(lo_p) < best_val:
            best_val, best_t = f(lo_p), lo_p
        if f(hi_p) < best_val:
            best_val, best_t = f(hi_p), hi_p

    if lo_n < hi_n - 1e-14:
        ts = np.linspace(lo_n, hi_n, 8)
        vals = np.array([f(t) for t in ts])
        imin = int(np.argmin(vals))
        a = float(ts[max(0, imin - 1)])
        b = float(ts[min(len(ts) - 1, imin + 1)])
        if b > a + 1e-14:
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            f1, f2 = f(c1), f(c2)
            for _ in range(30):

def _pair_move(h, i, j, R0):
    d = np.zeros(len(h))
    d[j] = 1.0
    d[i] = -1.0
    t1, v1 = _exact_1d(h, d, R0)
    d2 = -d
    t2, v2 = _exact_1d(h, d2, R0)
    if v1 <= v2:
        return t1, v1
    return t2, v2

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
