```python
import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        L = abs(k - (n-1))
        m = n - L
        if L < n:
            G[k, L:] += (1.0 - h[:n-L])
        if m > 0:
            G[k, :m] += -h[L:L+m]
    return G

def exact_line_search(h, d, R0, dx):
    """Find best alpha in feasible range minimizing max(R0 + alpha*T - alpha^2*Q)*dx."""
    n = len(h)
    T = np.correlate(d, 1.0-h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    
    # Feasible range
    alpha_lo = 0.0
    alpha_hi = float('inf')
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max(h[neg] / (-d[neg]))))
    
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, c5_exact(h)
    
    # Cap the range
    alpha_hi = min(alpha_hi, 2.0)
    alpha_lo = max(alpha_lo, -2.0)
    
    def f(alpha):
        return float(np.max(R0 + alpha * T - alpha*alpha * Q)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = alpha_lo, alpha_hi
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = f(c2)
    
    opt = (a+b)/2.0
    val = f(opt)
    # Check endpoints
    f_lo, f_hi = f(alpha_lo), f(alpha_hi)
    if f_lo < val:
        opt, val = alpha_lo, f_lo
    if f_hi < val:
        opt, val = alpha_hi, f_hi
    
    return opt, val
```

```python
def qp_direction(h, G, R0, target_raw, rho, lam=1.0):
    n = len(h)
    
    def obj(d):
        pred = R0 + G @ d
        viol = np.maximum(0.0, pred - target_raw)
        return float(np.sum(viol**2) + lam * np.dot(d, d))
    
    def grad_fn(d):
        pred = R0 + G @ d
        viol = np.maximum(0.0, pred - target_raw)
        return 2.0 * G.T @ viol + 2.0 * lam * d
    
    cons = [{'type': 'eq', 'fun': lambda d: np.sum(d), 'jac': lambda d: np.ones(n)}]
    bounds = tuple((-h[i], 1.0-h[i]) for i in range(n))
    
    x0 = np.zeros(n)
    try:
        res = minimize(obj, x0, jac=grad_fn, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
        d = res.x.copy()
    except Exception:
        return None
    
    d -= d.mean()
    if np.max(np.abs(d)) < 1e-12:
        return None
    
    # Scale to trust region
    norm_d = np.linalg.norm(d)
    if norm_d > rho:
        d = d * (rho / norm_d)
    elif norm_d < 1e-8:
        return None
    
    return d
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def build_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        L = abs(k - (n - 1))
        m = n - L
        if L < n:
            G[k, L:] += (1.0 - h[:n - L])
        if m > 0:
            G[k, :m] += -h[L:L + m]
    return G

def exact_line_search(h, d, R0, dx):
    n = len(h)
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    alpha_lo = 0.0
    alpha_hi = 5.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max(h[neg] / (-d[neg]))))
    alpha_hi = min(alpha_hi, 5.0)
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, c5_exact(h)

    def f(a):
        return float(np.max(R0 + a * T - a * a * Q)) * dx

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = alpha_lo, alpha_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    opt = (a + b) / 2.0
    val = f(opt)
    f_lo, f_hi = f(alpha_lo), f(alpha_hi)
    if f_lo < val:
        opt, val = alpha_lo, f_lo
    if f_hi < val:
        opt, val = alpha_hi, f_hi
    return opt, val

def qp_direction(h, G, R0, target_raw, rho, lam):
    n = len(h)
    n_rows = len(R0)

    def obj(d):
        pred = R0 + G @ d
        viol = np.maximum(0.0, pred - target_raw)
        return float(np.sum(viol * viol) / n_rows + lam * np.dot(d, d))

    def grad_fn(d):
        pred = R0 + G @ d
        viol = np.maximum(0.0, pred - target_raw)
        return 2.0 * (G.T @ viol) / n_rows + 2.0 * lam * d

    cons = [{'type': 'eq', 'fun': lambda d: np.sum(d), 'jac': lambda d: np.ones(n)}]
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    x0 = np.zeros(n)
    try:
        res = minimize(obj, x0, jac=grad_fn, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 250, 'ftol': 1e-14})
        d = res.x.copy()
    except Exception:
        return None
    d -= d.mean()
    nm = np.linalg.norm(d)
    if nm < 1e-12:
        return None
    if nm > rho:
        d = d * (rho / nm)
    return d

def gradient_direction(h, G, R0, rho):
    n = len(h)
    Rmax = float(np.max(R0))
    sigma = 0.02 * Rmax + 1e-10
    weights = np.exp((R0 - Rmax) / sigma)
    active = R0 > Rmax * 0.995
    weights[~active] *= 0.1
    wsum = weights.sum()
    if wsum < 1e-15:
        return None
    weights /= wsum
    g = G.T @ weights
    g -= g.mean()
    nm = np.linalg.norm(g)
    if nm < 1e-12:
        return None
    d = -g / nm * rho
    d -= d.mean()
    return d

def golden_pair(h, i, j, dx, S0):
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(len(h))
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def pair_search(h, n_pairs, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(777)

    while time.time() < deadline and stall < 60:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.005 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]

        Rmax = float(np.max(S0))
        sigma = 0.02 * Rmax + 1e-10
        w = np.exp((S0 - Rmax) / sigma)
        w[active_idx] *= 5.0
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.ones(len(S0)) / len(S0)
        else:
            w /= wsum

        # Compute weighted gradient for donor/receiver selection
        G = build_grad_matrix(h)
        g = G.T @ w
        g -= g.mean()
        donor_pool = np.argsort(-g)[:n_pairs]
        recv_pool = np.argsort(g)[:n_pairs]

        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        tried = set()
        for i in donor_pool:
            if time.time() > deadline:
                break
            for j in recv_pool:
                if i == j:
                    continue
                key = (min(i, j), max(i, j))
                if key in tried:
                    continue
                tried.add(key)
                alpha, new_c5 = golden_pair(h, i, j, dx, S0)
                if alpha > 1e-10 and new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)

        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                idx = rng.integers(n, size=4)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.05, 0.05)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0

    return best_h, best_c5

def qp_optimize(h, deadline):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rho = 0.5
    lam = 0.01
    stall = 0
    max_stall = 40

    while time.time() < deadline and stall < max_stall:
        R0 = np.correlate(h, 1.0 - h, mode='full')
        Rmax = float(np.max(R0))
        target_raw = min(0.3808 / dx, Rmax * (1.0 - 0.0005))

        G = build_grad_matrix(h)
        d = qp_direction(h, G, R0, target_raw, rho, lam)

        if d is None:
            d = gradient_direction(h, G, R0, rho)

        if d is None:
            stall += 5
            rho *= 0.7
            continue

        alpha, new_c5 = exact_line_search(h, d, R0, dx)

        if alpha > 1e-12 and new_c5 < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = project(h_new, target_sum)
            actual_c5 = c5_exact(h_new)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h_new.copy()
            h = h_new
            cur_c5 = actual_c5
            stall = 0
            if rho < 1.0:
                rho = min(1.0, rho * 1.1)
        else:
            stall += 1
            rho *= 0.8
            if stall % 8 == 0:
                lam = max(0.001, lam * 0.5)

    return best_h, best_c5

def gradient_polish(h, deadline):

    while time.time() < deadline and stall < 25:

        max_score = float(np.max(scores))

        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:8]

        G = build_grad_matrix(h)
        G_a = G[active_idx]

        Rmax_a = float(np.max(R_a))
        sigma = 0.005 * Rmax_a + 1e-10
        w = np.exp((R_a - Rmax_a) / sigma)
        w /= w.sum()
        g = G_a.T @ w
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        d = -g / gn

        lo, hi = -1.0, 1.0
        pos = d > 1e-10

        T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
        C = np.correlate(d, d, 'full')
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a2, b2 = lo, hi
        c1 = b2 - gr * (b2 - a2)
        c2 = a2 + gr * (b2 - a2)

                break
            if f1 < f2:

                f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            else:
                a2, c1, f1 = c1, c2, f2
                c2 = a2 + gr * (b2 - a2)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt = (a2 + b2) / 2.0
        opt_val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
        f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)
```
