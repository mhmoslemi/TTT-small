def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target} using water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_corr_matrix(fixed, n):

    This is such that corr(a, fixed)[k] = sum_i a[i] * A[k,i] = (a @ A.T)[k]"""

    # So A[k, i] = fixed[i + k - (n-1)] if valid

    for k in range(2*n-1):
        offset = k - (n-1)
        if offset >= 0:
            # i + offset in [0, n-1] => i in [0, n-1-offset]
            m = n - offset
            matrix[k, :m] = fixed[offset:]  # fixed[i+offset] for i=0..m-1
            # Wait: fixed[i + offset] for i=0 is fixed[offset], for i=m-1 is fixed[n-1]
            # So matrix[k, i] = fixed[i + offset], i.e., matrix[k, :m] = fixed[offset:offset+m] = fixed[offset:]

    return matrix

def _lp_minimax(a, fixed_b, n, dx):

    Returns optimal a and tau."""

    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], float(res.x[-1])
    return None, None

def _gen_wave(n, rng, freq_range=(1,12), amp_range=(0.2,0.45)):
    """Generate a balanced wave-like starting point."""
    freq = rng.integers(freq_range[0], freq_range[1]+1)
    phase = rng.uniform(0, 2*np.pi)
    x = np.arange(n)
    h = 0.5 + rng.uniform(amp_range[0], amp_range[1]) * np.sin(2*np.pi*freq*x/n + phase)
    h += rng.normal(0, 0.08, n)
    return _project(h, n/2.0)

def _gen_binary(n, rng):
    """Generate a random binary balanced sequence."""
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    h += rng.normal(0, 0.03, n)
    return _project(h, n/2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new/2.0)

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _build_corr_matrix(fixed, n):

            m = n - off
            if m > 0:

        else:
            d = -off
            m = n - d
            if m > 0:
                A[k, d:] = fixed[0:m]

def _lp_minimax(fixed, n, dx, target):

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:

    h3, c3 = _local_lp(best_h, d3, radius=0.10)
    if c3 < best_c5:
        best_c5, best_h = c3, h3
    
    # Phase 4: Multi-resolution search (20%)
    d4 = t0 + budget_s * 0.75
    n_best = len(best_h)
    tried_n = set()
    
    offsets = list(range(-16, 17, 2))
    offsets.sort(key=lambda x: abs(x))
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 16 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        per_res = max(3.0, (d4 - time.time()) / max(1, len(tried_n)))
        sub_dl = min(d4, time.time() + per_res)
        
        h_r = _resample(best_h, n_try)
        if time.time() < d4:
            h_r, c_r = _local_lp(h_r, min(sub_dl, time.time() + per_res * 0.6), radius=0.12)
            if c_r < best_c5 and time.time() < d4:
                h_r, c_r2 = _relaxed_pair(h_r, n_try, rng, max_iters=8)
                if c_r2 < c_r:
                    c_r, h_r = c_r2, h_r
                if c_r < best_c5:
                    best_c5, best_h = c_r, h_r
                    n_best = n_try
    
    # Also try small n directly
    for n_small in [20, 24, 28, 32, 36, 40, 44, 48, 52, 56, 60, 64, 72]:
        if time.time() > d4:
            break
        if n_small in tried_n:
            continue
        tried_n.add(n_small)
        sub_dl = min(d4, time.time() + 4.0)
        h_s = _resample(best_h, n_small)
        h_s, c_s = _local_lp(h_s, sub_dl, radius=0.15)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s
    
    # Phase 5: Diverse starting points (12%)
    d5 = t0 + budget_s * 0.87
    n_div = len(best_h)
    for si in range(10):
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        for gen_fn in [_gen_wave, _gen_binary, _gen_blocks]:
            if time.time() > d5:
                break
            h_s = gen_fn(n_div, rng)
            h_s, c_s = _local_lp(h_s, min(sub_dl, time.time() + 3.0), radius=0.12)
            if c_s < best_c5:
                best_c5, best_h = c_s, h_s
    
    # Phase 6: Final relaxed pair + local polish (7%)
    d6 = deadline
    if time.time() < d6:
        h6, c6 = _relaxed_pair(best_h, len(best_h), rng, max_iters=20)
        if c6 < best_c5:
            best_c5, best_h = c6, h6
    if time.time() < d6:
        h7, c7 = _local_lp(best_h, d6, radius=0.05)
        if c7 < best_c5:
            best_c5, best_h = c7, h7
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
