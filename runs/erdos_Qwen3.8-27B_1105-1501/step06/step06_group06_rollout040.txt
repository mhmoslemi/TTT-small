        g[:n-s] += (1.0 - h[s:])
        # For j in [s, n-1]: contribution -h[j-s]
        g[s:] += (-h[:-s])
        g *= dx
    elif s < 0:
        # By symmetry: f_{-s_alt} where we swap roles
        s_abs = -s
        # f_{-s} = sum_{i=0}^{n-1-s_abs} h[i+s_abs]*(1-h[i])
        # d/dh[j]: if j >= s_abs: (1-h[j-s_abs])... 
        # Actually f_{-s_abs} = sum_{i=0}^{n-1-s_abs} h[i+s_abs]*(1-h[i])
        # For h[j] where j >= s_abs: appears as h[i+s_abs] with i=j-s_abs
        #   contribution: (1-h[j-s_abs])
        # For h[j] where j < n-s_abs: appears as (1-h[i]) with i=j
        #   contribution: -h[j+s_abs]
        g[s_abs:] += (1.0 - h[:n-s_abs])
        g[:n-s_abs] += (-h[s_abs:])
        g *= dx
    return g

def _quad_row(h, s, d):
    """Second-order term of f_s in direction d: -sum_{valid} d[i]*d[i+s]*dx"""
    n = len(h)
    dx = 2.0 / n
    if s == 0:
        return -2.0 * np.sum(d**2) * dx
    elif s > 0:
        return -np.sum(d[:n-s] * d[s:]) * dx
    elif s < 0:
        s_abs = -s
        return -np.sum(d[s_abs:] * d[:n-s_abs]) * dx

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2*np.pi*freq*x/n + phase)
    return _project(h, n/2.0)

def _block_init(n, rng):
    """Create a block structure and project."""
    h = np.zeros(n)
    # Random blocks
    pos = 0
    while pos < n:
        blen = int(rng.integers(2, max(3, n//4)))
        if rng.random() < 0.5:
            h[pos:pos+blen] = 1.0
        pos += blen
    return _project(h, n/2.0)

def _slsqp_polish(h, deadline, max_iter=30):
    """SLSQP-based polishing."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    if time.time() > deadline:
        return h, _exact_c5(h)
    
    # Use SLSQP with the top constraints
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    K = min(20, 2*n-1)
    top_k = np.argsort(-corr)[:K]
    
    def ineq_fn(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        return np.max(c) - c[top_k]  # This doesn't work well as a single constraint
    
    # Better: minimize t subject to c[k] <= t for top k
    h_current = h.copy()
    cur_c5 = _exact_c5(h_current)
    
    for outer in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_current, 1.0-h_current, mode='full') * dx
        M = cur_c5
        top_k = np.argsort(-corr)[:K]
        
        t_init = M + 1e-6
        x0 = np.append(h_current, t_init)
        
        def obj(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n+1)
            g[n] = 1.0
            return g
        
        def cons_ineq(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0-hx, mode='full') * dx
            return x[n] - c[top_k]
        
        def cons_jac(x):
            hx = x[:n]
            n_h = len(hx)
            J = np.zeros((K, n+1))
            for idx, kk in enumerate(top_k):
                s = kk - (n-1)
                J[idx, :n] = _grad_row(hx, s)
                J[idx, n] = 1.0
            return J
        
        def cons_eq(x):
            return np.sum(x[:n]) - target
        
        def cons_eq_jac(x):
            g = np.zeros(n+1)
            g[:n] = 1.0
            return g
        
        bounds = [(0.0, 1.0)]*n + [(0.0, 2.0)]
        constraints = [
            {'type': 'ineq', 'fun': cons_ineq, 'jac': cons_jac},
            {'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac},
        ]
        
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 30, 'ftol': 1e-13})
        
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h_current = h_new
                cur_c5 = c5_new
            else:
                break
        else:
            break
    
    return h_current, cur_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
        diff = target - float(np.sum(x))
        if abs(diff) > 1e-14:
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, s, dx):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g = (1.0 - 2.0 * h) * dx
    elif s > 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] += (-h[:n - s])
        g *= dx
    else:
        sa = -s
        g[sa:n] += (1.0 - h[:n - sa])
        g[:n - sa] += (-h[sa:n])
        g *= dx
    return g

def _active_descent(h_init, deadline, max_iter=800, tol_frac=1e-4, min_active=3, max_active=50):

    for it in range(max_iter):

def _slsqp_polish(h_init, deadline, max_outer=20):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    K = min(15, 2 * n - 1)

    for _ in range(max_outer):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(-corr)[:K]
        M = cur_c5
        x0 = np.append(h, M + 1e-6)

        def obj_fn(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]

        def ineq_jac(x):
            hx = x[:n]
            J = np.zeros((K, n + 1))
            for idx, kk in enumerate(top_k):
                s = kk - (n - 1)
                J[idx, :n] = _grad_row(hx, s, dx)
                J[idx, n] = 1.0
            return J

        def eq_fn(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-14})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
            else:
                break
        else:
            break

    return h, cur_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0

        if rng.random() < 0.5:

        pos += blen
    return _project(h, n / 2.0)
