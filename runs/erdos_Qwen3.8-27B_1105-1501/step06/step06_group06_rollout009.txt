```python
    # corr is the array of correlation values (2n-1 of them)
    # Return smooth max approximation
    max_c = np.max(corr)
    # Scale to avoid overflow
    shifted = corr - max_c
    return max_c + eps * np.log(np.sum(np.exp(shifted / eps)))

def smooth_max_grad(corr, eps):
```

```python
def corr_gradient(h):
    """Compute gradient of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n)."""
    n = len(h)
    grad = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        k = s + (n-1)  # index in the full correlation
        m = n - abs(s)
        if s >= 0:
            # corr = sum_{i=0}^{m-1} h[i]*(1-h[i+s])
            # d/dh[j] = (1-h[j+s]) for j in [0,m-1]

            grad[k, :m] += (1.0 - h[s:s+m])
```

```python
from scipy.optimize import minimize

def objective_and_grad(x, eps, lambda_pen, n):
    h = x
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_c = np.max(corr)
    shifted = corr / eps - max_c / eps
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Smooth max value
    f_val = max_c + eps * np.log(np.sum(np.exp(shifted)))
    
    # Penalty
    s = np.sum(h)
    target = n / 2.0
    f_val += lambda_pen * (s - target) ** 2
    
    # Gradient of smooth max: weights @ G
    G = corr_gradient(h)
    grad_sm = weights @ G
    grad_pen = 2.0 * lambda_pen * (s - target)
    grad = grad_sm + grad_pen
    
    return f_val, grad
```

```python
def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_gradient(h):
    """Compute the matrix of partial derivatives of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    one_minus_h = 1.0 - h
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            G[k, :m] += one_minus_h[s:s + m]
            if s > 0:
                G[k, s:s + m] -= h[:m]
            # for s=0: G[k, :m] = one_minus_h + (-h) = 1 - 2h. Correct.
        else:
            g = -s
            G[k, g:g + m] += one_minus_h[:m]
            G[k, :m] -= h[g:g + m]
    G *= dx
    return G

def _smooth_max_and_weighted_grad(h, eps):
    """Compute smooth max of correlations and the weighted gradient.
    Returns (smooth_max_value, gradient_vector)."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    
    max_c = np.max(corr)
    shifted = (corr - max_c) / eps
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    f_val = max_c + eps * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    
    G = _corr_gradient(h)
    grad = weights @ G
    
    return f_val, grad, corr, weights

def _smooth_lbbfgs(h0, eps_levels, max_iter_per_level, deadline):
    """Run L-BFGS-B with decreasing epsilon for smooth max."""
    n = len(h0)
    h = h0.copy()
    target = n / 2.0
    lam = 10000.0
    
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for eps in eps_levels:
        if time.time() > deadline:
            break
        
        def fun_and_grad(x):
            hx = x
            corr_raw = np.correlate(hx, 1.0 - hx, mode='full')
            dx = 2.0 / n
            corr = corr_raw * dx
            max_c = np.max(corr)
            shifted = (corr - max_c) / eps
            exp_s = np.exp(shifted)
            sum_s = np.sum(exp_s)
            f_val = max_c + eps * np.log(sum_s)
            wts = exp_s / sum_s
            G = _corr_gradient(hx)
            grad_sm = wts @ G
            s = np.sum(hx)
            grad_pen = 2.0 * lam * (s - target)
            f_val += lam * (s - target) ** 2
            grad = grad_sm + grad_pen
            return f_val, grad
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(fun_and_grad, h, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter_per_level,
                                               'ftol': 1e-16, 'gtol': 1e-14,
                                               'maxls': 50})
        h_new = _project(res.x, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
        else:
            h = h_new  # still use as starting point
            if _exact_c5(h) < best_c5:
                best_c5 = _exact_c5(h)
                best_h = h.copy()
    
    return best_h, best_c5
```
