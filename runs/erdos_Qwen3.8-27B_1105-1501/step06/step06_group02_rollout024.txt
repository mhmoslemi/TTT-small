Let me analyze the problem and the strategy carefully.

The key idea from the strategy is to use a **full-row LP (Linear Programming) epigraph descent** approach instead of just pair moves. At each step:

1. Compute the gradient matrix G of all overlap rows with respect to h
2. Solve an LP that minimizes the maximum predicted row value (using a linear approximation)
3. Use the LP solution direction for a line search with exact evaluation
4. Accept only if exact evaluation confirms improvement

This is fundamentally different from the pair-move approach because it can find directions that simultaneously reduce many active peaks.

Let me think about the implementation:

1. **Gradient matrix G**: For each row i (corresponding to lag d = i - (n-1)), the gradient of R_i with respect to h is:
   - If d >= 0, m = n - d:
     - g[0:m] += 1 - h[d:d+m]
     - g[d:d+m] += -h[0:m]
   - If d < 0, q = -d, m = n - q:
     - g[q:n] += 1 - h[0:m]
     - g[0:m] += -h[q:q+m]

2. **LP formulation**: 
   - Variables: [t, v_0, ..., v_{n-1}]
   - Minimize: t
   - Subject to: G[i,:] @ v - t <= G[i,:] @ h - R[i] for all i
   - Bounds: h_i - r <= v_i <= h_i + r (clipped to [0,1])
   - Equality: sum(v) = n/2

   The constraint G[i,:] @ v - t <= G[i,:] @ h - R[i] means:
   t >= G[i,:] @ v + R[i] - G[i,:] @ h
   
   Wait, let me re-derive. We have:
   R_i(v) ≈ R_i(h) + G[i,:] @ (v - h) = G[i,:] @ v + (R_i(h) - G[i,:] @ h)
   
   So the epigraph constraint is:
   t >= G[i,:] @ v + (R_i(h) - G[i,:] @ h)
   
   In standard form: G[i,:] @ v - t <= -(R_i(h) - G[i,:] @ h) = G[i,:] @ h - R_i(h)
   
   So b_i = G[i,:] @ h - R_i(h)

   The objective is minimize t (variable at index 0).

3. **Line search**: Once we have v from LP, direction d = v - h. Then search along h + alpha * d for alpha in feasible range.

Let me think about the LP solver. I'll use scipy.optimize.linprog with method='highs' (which is available in modern scipy).

Key considerations:
- n = 122 (from parent), so G is (243, 122), LP has 123 variables and 243+1 constraints + bounds. Very manageable.
- The radius r controls the trust region. Start with moderate r and shrink.
- Need to handle the case where LP fails (fall back to pair moves or gradient direction).

Let me also think about what's different from the parent that could help:
- The parent uses pair moves (2-coordinate changes), which are very limited in direction space
- The LP approach uses full n-dimensional directions, which can simultaneously address multiple active peaks
- This should be much more effective at breaking plateaus

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def build_gradient_matrix(h):
    """Build the gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    for i in range(L):
        d = i - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[i, :m] += 1.0 - h[d:d+m]
            G[i, d:d+m] += -h[:m]
        else:
            q = -d
            G[i, q:] += 1.0 - h[:m]
            G[i, :m] += -h[q:q+m]
    return G

def lp_descent_step(h, G, R, radius, deadline):
    """Solve LP for direction, do line search, return (new_h, new_c5, accepted)."""
    n = len(h)
    L = len(R)
    dx = 2.0 / n
    target = n / 2.0
    
    # LP: minimize t subject to G@v - t <= b, bounds
    # b_i = G[i,:] @ h - R[i]
    b = G @ h - R
    
    # A_ub: each row is [G[i,:], -1], shape (L, n+1)
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 1:] = G
    A_ub[:, 0] = -1.0
    b_ub = b.copy()
    
    # Objective: minimize t (first variable)
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    # Bounds: v_i in [max(0, h_i - r), min(1, h_i + r)]
    v_lo = np.maximum(0.0, h - radius)
    v_hi = np.minimum(1.0, h + radius)
    bounds = [(None, None)]  # t is free
    for i in range(n):
        bounds.append((v_lo[i], v_hi[i]))
    
    # Equality: sum(v) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, c5_exact(h), False
    
    if res.status != 0:
        return h, c5_exact(h), False
    
    v = res.x[1:]  # skip t
    
    # Check if direction is meaningful
    d = v - h
    # Ensure zero sum (numerical correction)
    d_sum = float(np.sum(d))
    if abs(d_sum) > 1e-10:
        d -= d_sum / n
    
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-8:
        return h, c5_exact(h), False
    
    # Line search: find best alpha in feasible range
    # Feasible range: h + alpha*d must stay in [0,1]
    pos = d > 1e-10
    neg = d < -1e-10
    lo = -1.0
    hi = 1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    if hi - lo < 1e-12:
        return h, c5_exact(h), False
    
    # Exact line evaluation: R(alpha) = R0 + alpha*T - alpha^2*C
    # where T = correlate(d, 1-h, 'full') - correlate(h, d, 'full')
    #       C = correlate(d, d, 'full')
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    R0 = R.copy()
    
    cur_c5 = float(np.max(R0) * dx)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(50):
        if b - a < 1e-10:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
    
    # Check endpoints
    f_lo = float(np.max(R0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    
    # Also check alpha=0 (current point)
    if cur_c5 < val:
        opt, val = 0.0, cur_c5
    
    if abs(opt) < 1e-10:
        return h, cur_c5, False
    
    # Apply and project
    h_new = h + opt * d
    h_new = project(h_new, target)
    
    # Exact evaluation
    new_c5 = c5_exact(h_new)
    
    if new_c5 < cur_c5 - 1e-13:
        return h_new, new_c5, True
    else:
        return h, cur_c5, False

The key insight is:

3. Use a schedule of decreasing radii
4. Fallback to pair moves if LP fails

Let me think about the radius schedule. Starting radius should be moderate - not too large (LP approximation breaks down) but not too small (no progress). Something like 0.1-0.3 for initial, then shrinking.

Actually, let me reconsider the LP. The linear approximation R_i(v) ≈ R_i(h) + G[i,:] @ (v-h) is only valid for small steps. If the radius is too large, the prediction is poor and the LP might give a direction that doesn't actually help. Actually, let me think differently. The LP minimizes the maximum of linear approximations. If we use a small radius, the linear approximation is accurate, and the LP gives a good descent direction. As we converge, we should use smaller radii.

Let me also consider: the LP solution v minimizes max_i (G[i,:] @ v + c[i]) where c[i] = R[i] - G[i,:] @ h. The optimal t* gives us the predicted minimum maximum row value. If t* < current max row value, we expect improvement.

Let me think about the gradient direction fallback. If LP fails, I can use the weighted active gradient:
- Compute weights for active rows (those near the maximum)
- Direction = -weighted sum of active row gradients
- Project to zero sum
- Line search

This is what the parent's `_gradient_polish` does, but I can make it more effective by using the full matrix.

    neg = d < -1e-10

        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-12:
        return 0.0, float(np.max(R0) * dx)

    C = np.correlate(d, d, 'full')

    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)

        else:

            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx

    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:

    if f_hi < val:
        opt, val = hi, f_hi
    
    return opt, val

def lp_step(h, G, R, radius, target, dx):

    n = len(h)
    L = len(R)
    
    b = G @ h - R
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 1:] = G
    A_ub[:, 0] = -1.0
    b_ub = b.copy()
    
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    v_lo = np.maximum(0.0, h - radius)
    v_hi = np.minimum(1.0, h + radius)
    bounds = [(None, None)]
    for i in range(n):
        bounds.append((v_lo[i], v_hi[i]))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, c5_exact(h), False

        return h, c5_exact(h), False

    d_sum = float(np.sum(d))
    if abs(d_sum) > 1e-10:
        d -= d_sum / n

        return h, c5_exact(h), False

    opt, val = line_search_exact(h, d, R, target, dx)
    
    if abs(opt) < 1e-10 or val >= cur_c5 - 1e-13:
        return h, cur_c5, False
    
    h_new = h + opt * d
    h_new = project(h_new, target)
    new_c5 = c5_exact(h_new)
    
    if new_c5 < cur_c5 - 1e-13:
        return h_new, new_c5, True
    return h, cur_c5, False

def gradient_step(h, G, R, target, dx):
    """Weighted active gradient descent step."""
    n = len(h)
    scores = R * dx
    max_score = float(np.max(scores))
    # Select active rows
    threshold = max_score - 0.01 * max_score
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-scores)[:10]
    
    # Weight by proximity to max
    w = np.exp((scores[active] - max_score) / (0.005 * max_score + 1e-15))
    w /= float(np.sum(w))
    
    # Weighted gradient (raw rows, so divide by dx for C5-scale)
    g = G[active].T @ w  # shape (n,)
    g -= np.mean(g)  # zero sum
    
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, c5_exact(h), False
    
    d = -g / gn
    cur_c5 = float(np.max(R) * dx)
    opt, val = line_search_exact(h, d, R, target, dx)
    
    if abs(opt) < 1e-10 or val >= cur_c5 - 1e-13:
        return h, cur_c5, False
    
    h_new = h + opt * d
    h_new = project(h_new, target)
    new_c5 = c5_exact(h_new)
    
    if new_c5 < cur_c5 - 1e-13:
        return h_new, new_c5, True
    return h, cur_c5, False

def pair_step(h, R, target, dx, i, j):
    """Pair move from j to i (h[i]-=alpha, h[j]+=alpha)."""
    n = len(h)
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return h, c5_exact(h), False
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    cur_c5 = float(np.max(R) * dx)
    opt, val = line_search_exact(h, d, R, target, dx)
    
    if abs(opt) < 1e-10 or val >= cur_c5 - 1e-13:
        return h, cur_c5, False
    
    h_new = h + opt * d
    h_new = project(h_new, target)
    new_c5 = c5_exact(h_new)
    
    if new_c5 < cur_c5 - 1e-13:

    return h, cur_c5, False

    h1, c5_1 = lp_polish(h, d1, max_iters=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n

    # Phase 2: Multi-weight LP on current best (15% time)
    d2 = t0 + budget_s * 0.70
    n = best_n
    target = n / 2.0
    dx = 2.0 / n
    h2 = best_h.copy()
    c5_2 = c5_exact(h2)
    stall2 = 0
    while time.time() < d2 and stall2 < 30:
        R = np.correlate(h2, 1.0 - h2, 'full')
        G = build_G(h2)
        h2, c5_2 = multi_weight_lp(h2, R, G, 0.01, target, dx, n_weights=4)
        if c5_2 < c5_exact(best_h) - 1e-14:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n
            stall2 = 0
        else:
            stall2 += 1

    # Phase 3: Multi-resolution (15% time)
    d3 = t0 + budget_s * 0.85
    h3 = best_h.copy()
    c5_3 = c5_exact(h3)
    for dn in [-4, 4, -6, 6, -2, 2, -8, 8]:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 260 or n_try % 2 != 0:
            continue
        h_r = resample(best_h, n_try)
        c5_r = c5_exact(h_r)
        if c5_r < best_c5 + 0.002:
            sub_dl = min(d3, time.time() + budget_s * 0.018)
            h_r, c5_r = lp_polish(h_r, sub_dl, max_iters=80)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 4: Final fine polish (10% time)
    d4 = deadline
    n = best_n
    target = n / 2.0
    dx = 2.0 / n
    h4 = best_h.copy()
    c5_4 = c5_exact(h4)
    radii_fine = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    for radius in radii_fine:
        if time.time() > d4:
            break
        stall4 = 0
        while time.time() < d4 and stall4 < 25:
            R = np.correlate(h4, 1.0 - h4, 'full')
            G = build_G(h4)
            d_dir = lp_descent(h4, G, R, radius, target, dx)
            if d_dir is None:
                d_dir = gradient_direction(h4, G, R, dx, frac=0.005)
            if d_dir is None:
                break
            h_new = apply_direction(h4, d_dir, target, dx)
            if h_new is not None:
                h4 = h_new
                c5_4 = c5_exact(h4)
                if c5_4 < best_c5:
                    best_c5 = c5_4
                    best_h = h4.copy()
                    best_n = n
                stall4 = 0
            else:
                stall4 += 1

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
