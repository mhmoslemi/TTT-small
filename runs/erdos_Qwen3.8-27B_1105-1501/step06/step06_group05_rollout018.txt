import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    # Project v onto {x : sum(x) = target, 0 <= x <= 1}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_row(h, ell, n):
    """Gradient of O_ell with respect to h, where ell is the lag.
    O_ell = sum_{valid i} h[i+ell]*(1-h[i])
    For ell >= 0: valid i are 0..n-1-ell
    For ell < 0: valid i are -ell..n-1
    """
    g = np.zeros(n)
    if ell >= 0:
        # Terms: h[i+ell]*(1-h[i]) for i=0..n-1-ell
        # d/dh[j] of h[i+ell]: j = i+ell, i.e., i = j-ell. Valid: 0 <= j-ell <= n-1-ell => ell <= j <= n-1
        # contribution: (1-h[j-ell])
        # d/dh[j] of (1-h[i]): j = i. Valid: 0 <= j <= n-1-ell
        # contribution: -h[j+ell]
        # Combined for j:
        # From first term: if ell <= j <= n-1, add (1-h[j-ell])
        # From second term: if 0 <= j <= n-1-ell, add (-h[j+ell])
        idx1 = np.arange(ell, n)  # j values where j >= ell and j <= n-1
        g[idx1] += (1.0 - h[idx1 - ell])
        if n - 1 - ell >= 0:
            idx2 = np.arange(0, n - ell)  # j values where 0 <= j <= n-1-ell
            g[idx2] += (-h[idx2 + ell])
    else:
        q = -ell
        # O_{-q} = sum_{i=q}^{n-1} h[i-q]*(1-h[i])
        # d/dh[j] of h[i-q]: j = i-q, i = j+q. Valid: q <= j+q <= n-1 => 0 <= j <= n-1-q
        # contribution: (1-h[j+q])
        # d/dh[j] of (1-h[i]): j = i. Valid: q <= j <= n-1
        # contribution: -h[j-q]
        if n - 1 - q >= 0:
            idx1 = np.arange(0, n - q)
            g[idx1] += (1.0 - h[idx1 + q])
        idx2 = np.arange(q, n)
        g[idx2] += (-h[idx2 - q])
    return g

For SLSQP:
- minimize f(x) subject to:
  - c_eq(x) = 0 (equality)
  - c_ineq(x) >= 0 (inequality)
