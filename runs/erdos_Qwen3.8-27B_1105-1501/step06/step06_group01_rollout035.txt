    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _jacobian_rows(h, dx):
    """Compute the Jacobian of S(h) w.r.t. h. Returns (2n-1, n) matrix."""
    n = len(h)
    m = 2 * n - 1
    J = np.zeros((m, n))
    # k = 0 (center)
    J[n-1, :] = dx * (1.0 - 2.0 * h)
    # k > 0
    for r in range(1, n):
        # S_r: index n-1+r
        # dS_r/dh_i = dx*(1-h_{i-r}) for i in [r, n-1], -dx*h_{i+r} for i in [0, n-1-r]
        J[n-1+r, r:] += dx * (1.0 - h[:-r])
        J[n-1+r, :n-r] += -dx * h[r:]
        # S_{-r}: index n-1-r
        # dS_{-r}/dh_i = dx*(1-h_{i+r}) for i in [0, n-1-r], -dx*h_{i-r} for i in [r, n-1]
        J[n-1-r, :n-r] += dx * (1.0 - h[r:])
        J[n-1-r, r:] += -dx * h[:-r]
    return J

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_direction(h, G, S, radius, n_active=10):
    """Find a direction via LP that reduces the max correlation."""
    n = len(h)
    m = 2 * n - 1
    
    # Select active lags (top n_active by correlation value)
    if n_active < m:
        active = np.argsort(-S)[:n_active]
    else:
        active = np.arange(m)
    
    # LP: minimize t
    # subject to: S[q] + G[q,:] @ d - t <= 0 for q in active
    #             sum(d) = 0
    #             bounds on d
    
    # Variables: d_0, ..., d_{n-1}, t (total n+1 variables)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for idx, q in enumerate(active):
        A_ub[idx, :n] = G[q, :]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -S[q]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = min(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

            result = minimize(obj, h, jac=grad_func, method='SLSQP',
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 400, 'ftol': 1e-9})
            if result.x is not None and np.all(np.isfinite(result.x)):
                h_new = _project(np.clip(result.x, 0, 1), target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def _proj_grad_pnorm(h0, deadline, p=8):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    lr = 0.08
    momentum = 0.85
    v = np.zeros(n)
    stall = 0

    for it in range(10000):
        if time.time() > deadline:
            break
        val, grad = _pnorm_val_grad(h, p)
        v = momentum * v + grad
        h_new = h - lr * v
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.04, 0.4)
            stall = 0
        else:
            lr *= 0.92
            stall += 1
            if stall > 60:
                h = h + 0.02 * np.random.randn(n)
                h = _project(h, target)
                lr = 0.06
                stall = 0
        if lr < 0.003:
            break
    return best_h, best_c5

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        G[n - 1 + r, r:] = dx * (1.0 - h[:-r])
        G[n - 1 + r, :n - r] += -dx * h[r:]
        G[n - 1 - r, :n - r] = dx * (1.0 - h[r:])
        G[n - 1 - r, r:] += -dx * h[:-r]
    return G

def _lp_direction(h, G, S, radius, n_active=12):
    n = len(h)
    m = 2 * n - 1
    if n_active < m:
        active = np.argsort(-S)[:n_active]
    else:
        active = np.arange(m)

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for idx, q in enumerate(active):
        A_ub[idx, :n] = G[q, :]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -S[q]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        lo = min(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_exact(h, d, radius):
    n = len(h)
    t_lo = -radius
    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])) - 1e-12)
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])) - 1e-12)
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])) - 1e-12)
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])) - 1e-12)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)

    n_grid = 30
    grid = np.linspace(t_lo, t_hi, n_grid)
    best_t = 0.0
    best_val = _exact_c5(h)
    for t in grid:
        h_try = h + t * d
        h_try = _project(h_try, n / 2.0)
        v = _exact_c5(h_try)
        if v < best_val:
            best_val = v
            best_t = t
    return best_t, best_val

def _lp_optimize(h0, deadline, start_radius=0.10):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    radius = start_radius
    stall = 0
    perturb_ctr = 0
    rng = np.random.default_rng(999)

    while time.time() < deadline:
        if stall > 60:
            if perturb_ctr < 6:
                h = h + rng.normal(0, 0.025, n)
                h = _project(h, target)
                best_c5 = max(best_c5, _exact_c5(h))
                radius = 0.08
                stall = 0
                perturb_ctr += 1
                continue
            else:
                break

        S = _corr_vals(h)
        G = _build_G(h)
        p = _lp_direction(h, G, S, radius, n_active=12)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue

        alpha, new_c5 = _line_search_exact(h, p, radius)
        if new_c5 < best_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            S_new = _corr_vals(h)
            predicted = float(np.max(S + G @ p * max(alpha, 1e-10)))
            ratio = cur / max(predicted, 1e-15)
            if ratio < 0.5:
                radius = min(radius * 1.2, 0.30)
            elif ratio > 2.0:
                radius *= 0.75
            else:
                radius = min(radius * 1.03, 0.25)
        else:
            stall += 1
            radius *= 0.7
            if stall % 30 == 0 and radius < 0.015:
                radius = 0.05

    return best_h, best_c5

def _sim_anneal(h0, deadline, n_iter=4000):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    rng = np.random.default_rng(123)
    T_init = 0.04
    T_final = 0.0005
    cur_c5 = best_c5

    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter, 1)
        T = T_init * (T_final / T_init) ** frac
        k = rng.integers(2, 6)
        idx = rng.choice(n, size=k, replace=False)
        h_new = h.copy()
        deltas = rng.uniform(-0.12, 0.12, k)
        for j in range(k):
            h_new[idx[j]] += deltas[j]
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5:
            h = h_new
            cur_c5 = c5_new
        else:
            prob = np.exp(-(c5_new - cur_c5) / T)
            if rng.random() < prob:
                h = h_new
                cur_c5 = c5_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_optimize(h0, deadline, max_passes=40):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    rng = np.random.default_rng(77)

    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        S = _corr_vals(h)
        max_idx = np.argmax(S)
        G = _build_G(h)
        grad = G[max_idx]
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(len(order), 30)):
            if time.time() > deadline:
                break
            i = order[ii]
            for jj in range(min(len(order), 30)):
                if tried > 50:
                    break
                j = order[jj]
                if i == j:
                    continue
                if h[i] <= 0.005 or h[j] >= 0.995:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search_exact(h, d, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                tried += 1
        if not improved:
            for _ in range(15):
                if time.time() > deadline:
                    break
                ii = rng.integers(len(interior))
                jj = rng.integers(len(interior))
                if ii == jj:
                    continue
                i, j = interior[ii], interior[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                for sign in [1.0, -1.0]:
                    d2 = d * sign
                    alpha, c5_try = _line_search_exact(h, d2, 0.35)
                    if c5_try < best_c5 - 1e-14:
                        h = h + alpha * d2
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved = True
            if not improved:
                break
    return best_h, best_c5

def _diverse_start(n, rng, attempt=0):
    x = np.arange(n) / n
    if attempt % 3 == 0:
        freq = rng.integers(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x + phase)
    elif attempt % 3 == 1:
        freq1 = rng.integers(1, 5)
        freq2 = rng.integers(5, 12)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * freq1 * x + p1) + 0.15 * np.sin(2 * np.pi * freq2 * x + p2)
    else:
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(2, int(rng.integers(2, n // 4 + 1)))
            if rng.random() < 0.5:
                h[pos:pos + bl] = 1.0
            pos += bl
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)
