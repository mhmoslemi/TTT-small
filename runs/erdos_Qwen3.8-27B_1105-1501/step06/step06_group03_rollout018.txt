def _corr_scaled(h):

def _project(v, target):
    # Water-filling projection onto {x : 0 <= x_i <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _correlation_gradient(h, n):
    """Compute gradient of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n)."""
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        g = G[r]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _lse_gradient(h, beta):

    w_sum = w.sum()

    grad = w @ G
    # Subtract mean to respect the constraint (projection handles it, but helps)
    grad -= grad.mean()
    return grad, float(M + np.log(w_sum) / beta)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)

def _line_search_bilinear(h, d, dx, trust_radius):
    """Line search for bilinear objective along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    # Also check endpoints
    if opt_val > f1:
        opt_t, opt_val = c1, f1
    if opt_val > f2:
        opt_t, opt_val = c2, f2
    
    return opt_t, opt_val

def _gradient_pair_move(h, deadline, max_passes=200):
    """Pair moves guided by gradient information."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        S = _corr_scaled(h)
        G = _correlation_gradient(h, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        
        n_int = min(n, 40)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        
        improved = False
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search_bilinear(h, d, dx, 0.4)

def _exact_c5(h):

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_grad(h, n):
    G = np.zeros((2 * n - 1, n))
    for r in range(2 * n - 1):
        k = r - (n - 1)
        g = G[r]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _lse_grad(h, beta):
    n = len(h)
    S = _corr_scaled(h)
    M = float(S.max())
    shifted = np.clip(beta * (S - M), -500.0, 500.0)
    w = np.exp(shifted)
    ws = float(w.sum())
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    G = _corr_grad(h, n)
    grad = w @ G
    grad -= grad.mean()
    return grad

def _resample(h, n_new):

    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new

    h3, c5_3 = _multi_lp(best_h, d3, K=15)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA (12% time)
    d4 = t0 + budget_s * 0.52
    h4, c5_4 = _binary_sa(best_h, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: LSE from binary + pair moves (10% time)
    d5 = t0 + budget_s * 0.62
    h5, c5_5 = _lse_momentum(best_h, min(d5, time.time() + (d5-d4)*0.5), beta_init=200, alpha_init=0.015)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    if time.time() < d5:
        h5b, c5_5b = _pair_moves(best_h, d5, max_passes=100)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution (18% time)
    d6 = t0 + budget_s * 0.80
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, 24, -24]
    tried = set()
    per_res = max(1.0, (d6 - time.time()) / max(1, len(offsets)))
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            t_left = d6 - time.time()
            if t_left > 2.0:
                h_r, c5_r = _lse_momentum(h_r, min(d6, time.time() + t_left*0.5), beta_init=80, alpha_init=0.02)
                if time.time() < d6 - 1.0:
                    h_r, c5_r = _multi_lp(h_r, d6, K=12)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 7: Aggressive restart (7% time)
    d7 = t0 + budget_s * 0.87
    scale = 0.02
    for _ in range(400):
        if time.time() > d7:
            break
        if best_c5 < 0.3805:
            break
        h_p = best_h + rng.normal(0, scale, len(best_h))
        h_p = _project(h_p, len(best_h) / 2.0)
        sub = min(d7, time.time() + budget_s * 0.005)
        h_t, c5_t = _lse_momentum(h_p, sub, beta_init=100, alpha_init=0.015)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.1, scale * 1.25)

    # Phase 8: Final polish (8% time)
    d8 = deadline
    h8, c5_8 = _pair_moves(best_h, min(d8, time.time() + (d8-d7)*0.4), max_passes=150)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < d8 - 2.0:
        h9, c5_9 = _lse_momentum(best_h, min(d8, time.time() + (d8-time.time())*0.5),
                                  beta_init=500, alpha_init=0.008)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
    if time.time() < d8 - 1.0:
        h10, c5_10 = _multi_lp(best_h, d8, K=10)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
