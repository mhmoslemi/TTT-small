def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        mask = (x < 0.999999) if res > 0 else (x > 1e-6)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _line_search_exact(h, d, dx, target, max_t=1.0):

    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Determine valid t range from box constraints
    t_lo = 0.0
    t_hi = max_t
    pos = d > 1e-14
    neg = d < -1e-14

    t_hi = max(0.0, min(t_hi, max_t))

            best_t = t

            c1 = a + gr * (b - a)
        if b - a < 1e-12:

    opt_t = (a + b) / 2

```python
import cvxpy as cp

def _lp_subproblem(h_cuts, a_cuts, b_cuts, n):
    """Solve the LP relaxation.
    h_cuts: list of points where cuts were generated
    a_cuts: list of gradient vectors at those points
    b_cuts: list of intercepts
    """
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for i in range(len(a_cuts)):
        constraints.append(a_cuts[i] @ h_var + b_cuts[i] <= t_var)
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK)  # or GLOP, or CBC
    if h_var.value is None:
        return None, None
    return h_var.value, t_var.value

            best_c5 = true_c5
            best_h = h_lp.copy()
            h = h_lp.copy()
        
        # Check gap
        if true_c5 - t_lp < 1e-6:
            # Close to optimal for this cut set
            break
        
        # Add cuts for newly active lags at h_lp
        corr_lp = np.correlate(h_lp, 1-h_lp, mode='full') * dx
        M = np.max(corr_lp)
        # Add cuts for lags with corr_lp > M - 0.01 * M
        active = np.where(corr_lp > M * 0.95)[0]
        for idx in active:
            k = int(idx) - (n - 1)
            g = _grad_row(h_lp, k) * dx
            a_cuts.append(g)
            b_cuts.append(corr_lp[idx] - float(g @ h_lp))
        
        # Limit cuts
        if len(a_cuts) > max_cuts:
            # Keep only the most "relevant" cuts
            # Relevance = how tight the cut is
            keep = []
            for i in range(len(a_cuts)):
                tightness = t_lp - (a_cuts[i] @ h_lp + b_cuts[i])
                keep.append((tightness, i))
            keep.sort()
            keep_idx = [i for _, i in keep[:max_cuts]]
            a_cuts = [a_cuts[i] for i in keep_idx]
            b_cuts = [b_cuts[i] for i in keep_idx]
    
    return best_h, best_c5
```

f_k(h) = sum_i h[i]*(1-h[i+k]) is bilinear. Its linearization at h0 is:
f_k(h) ≈ f_k(h0) + ∇f_k(h0)^T (h - h0)

For a bilinear function, the linearization is neither an upper nor a lower bound in general. So the LP is not necessarily a valid relaxation.

However, it can still be useful as a heuristic: solve the LP, evaluate the true objective, and iterate. This is essentially a **Gauss-Seidel / successive linearization** method.

Actually, let me reconsider. For the bilinear form f_k(h) = sum_i h[i]*(1-h[i+k]):
Let's write it as f_k(h) = sum_i h[i] - sum_i h[i]*h[i+k]

The term sum_i h[i]*h[i+k] is a quadratic form h^T A_k h where A_k is a symmetric matrix with (A_k)_{i,i+k} = (A_k)_{i+k,i} = 1/2 for valid pairs.

The Hessian of f_k is -2*A_k (the factor of 2 from the symmetric form). Whether A_k is PSD or not determines the convexity.

For k=1, A_1 has 1/2 on the super and sub diagonals (where valid). The eigenvalues of a tridiagonal Toeplitz matrix with 0 on diagonal and 1/2 on off-diagonals are (1/2)*cos(j*pi/(n+1)) for j=1,...,n. These range from close to 1/2 (positive) to close to -1/2 (negative). So A_1 is NOT PSD, meaning f_1 is NOT concave.

So the cutting plane approach won't give a valid relaxation. But it can still work as a heuristic.

Actually, for binary h, the correlation can be computed very efficiently. If h is binary with k ones, then for each lag:
overlap = (number of ones in first n-k positions) - (number of pairs of ones at distance k)

We can precompute the autocorrelation of the binary sequence using FFT! This would be O(n log n) instead of O(n^2).

Let me use this optimization:

```python
def _binary_c5_fast(h):
    """Fast C5 evaluation for binary h using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Autocorrelation of h
    # r[k] = sum_i h[i] * h[i+k] for k=0,...,n-1
    # Using FFT: r = ifft(fft(h)^2) ... need to be careful
    H = np.fft.fft(h, n=2*n)
    r = np.real(np.fft.ifft(H * np.conj(H)))[:n]
    # r[k] = sum_i h[i+k] * h[i] for i where both valid = autocorrelation
    # Actually r[k] for k in 0..n-1 gives sum_{i=0}^{n-1-k} h[i] * h[i+k]
    
    # For lag k >= 0: f_k = (sum of h in first n-k positions) - r[k]
    # sum of h in first n-k positions = sum(h) - sum(h[n-k:]) for k>0
    # For binary with sum = n/2: sum(h[:n-k]) = n/2 - sum(h[n-k:])
    
    # Actually let me just use the direct formula
    # f_k for k > 0: sum_{i=0}^{n-1-k} h[i](1-h[i+k])
    # = sum_{i=0}^{n-1-k} h[i] - r[k]
    
    # For k < 0, let k' = -k: f_{-k'} = sum_{i=k'}^{n-1} h[i](1-h[i-k'])
    # = sum_{i=k'}^{n-1} h[i] - r[k'] (by symmetry of r)
    # Wait, r[k'] = sum_{i=0}^{n-1-k'} h[i]*h[i+k'] = sum_{j=k'}^{n-1} h[j]*h[j-k']
    # So f_{-k'} = sum_{j=k'}^{n-1} h[j] - sum_{j=k'}^{n-1} h[j]*h[j-k'] = sum(h[k':]) - r[k']
    
    # Due to symmetry, f_k = f_{-k} in terms of the formula structure
    # f_k (k>0) = sum(h[:n-k]) - r[k]
    # f_{-k} = sum(h[k:]) - r[k]
    
    # So we need both sums. Let's compute prefix sums.
    hsum = np.cumsum(h)
    total = hsum[-1]  # = n/2
    
    overlaps = np.zeros(2*n-1)
    for k in range(1, n):
        overlaps[n-1+k] = hsum[n-1-k] - r[k]  # f_k for lag +k (index n-1+k in full)
        overlaps[n-1-k] = total - hsum[k-1] - r[k]  # f_{-k} for lag -k (index n-1-k in full)
    overlaps[n-1] = total - np.sum(h**2)  # lag 0
    
    return float(np.max(overlaps) * dx)
```
