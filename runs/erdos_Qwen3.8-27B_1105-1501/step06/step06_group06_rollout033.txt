**Algorithm: Curvature-Exploiting Descent (CED)**

1. Start with h0 (from initial_h_values)
2. Compute current C₅ and active set
3. Generate candidate directions:
   a. LP direction (minimize max linear increase over active set)
   b. Null-space directions (random vectors in null space of active gradient rows)
   c. Curvature-aligned: find d that minimizes max_k (a_k * t + q_k * t²) for small t
   d. Random directions with sum=0
4. For each direction, do exact 1D search (golden section on the exact max-of-quadratics)
5. Accept the best improvement
6. Repeat until convergence or time runs out

The key difference from the previous code: generate MANY directions per iteration, not just one.

Let me also think about the null-space approach more carefully. If A is the matrix of active gradient rows (|A| × n), the null space is the set of d such that A*d = 0 (first-order terms vanish for all active lags). In this subspace, c_k(h+t*d) ≈ M + t²*q_k for k in A. So we need to find d in the null space that minimizes max_{k in A} q_k(d). Since q_k(d) = -sum d[i]*d[i+k] * dx, this is a generalized eigenvalue problem or semidefinite programming problem.

Actually, for a single direction d in the null space, max_{k in A} q_k(d) is a function of d. We want to minimize this over the null space. This is equivalent to:

min_d max_{k in A} -sum_i d[i]*d[i+k] * dx
subject to A*d = 0, ||d|| = 1, and box constraints (approximately)

This is a non-convex problem, but for small active sets, we can handle it.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.2
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_grad_row(h, s, n):
    """Gradient of correlation at shift s with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    m = n - abs(s)
    if s >= 0:
        g[:m] += (1.0 - h[s:s+m]) * dx
        g[s:s+m] += (-h[:m]) * dx
    else:
        g2 = -s
        g[g2:g2+m] += (1.0 - h[:m]) * dx
        g[:m] += (-h[g2:g2+m]) * dx
    return g

def _corr_grads(h, lags):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for i, s in enumerate(lags):
        # s is the actual shift (can be negative or positive)
        shift = s - (n - 1)  # convert from correlate index to actual shift
        m = n - abs(shift)
        if shift >= 0:
            G[i, :m] += (1.0 - h[shift:shift+m]) * dx
            G[i, shift:shift+m] += (-h[:m]) * dx
        else:
            g2 = -shift
            G[i, g2:g2+m] += (1.0 - h[:m]) * dx
            G[i, :m] += (-h[g2:g2+m]) * dx
    return G

def _quadratic_terms(h, d, lags):
    """Compute q_k = -sum d[i]*d[i+k]*dx for each lag k."""
    n = len(h)
    dx = 2.0 / n
    q = np.zeros(len(lags))
    for i, s in enumerate(lags):
        shift = s - (n - 1)
        m = n - abs(shift)
        if shift >= 0:
            q[i] = -np.sum(d[:m] * d[shift:shift+m]) * dx
        else:
            g2 = -shift
            q[i] = -np.sum(d[g2:g2+m] * d[:m]) * dx
    return q

def _exact_line_max(h, d, t):
    """Compute max correlation at h + t*d."""
    n = len(h)
    dx = 2.0 / n
    # c_k(h+td) = c_k(h) + t*linear_k + t^2*quad_k
    # Let's compute directly for accuracy
    h_new = h + t * d
    c = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
    return float(np.max(c))

def _line_search_exact(h, d, t_min, t_max, n_eval=200):
    """Find t in [t_min, t_max] that minimizes max correlation of h+t*d.
    Uses the fact that each correlation is a quadratic in t."""
    n = len(h)
    dx = 2.0 / n
    # Precompute for all lags
    S0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Linear term for each lag: 
    # c_k(h+t*d) = sum_i (h[i]+t*d[i])*(1-h[i+k]-t*d[i+k])
    # = sum_i h[i]*(1-h[i+k]) + t*sum_i [d[i]*(1-h[i+k]) - h[i]*d[i+k]]
    # + t^2 * (-sum_i d[i]*d[i+k])
    
    n_lags = 2*n - 1
    linear = np.zeros(n_lags)
    quad = np.zeros(n_lags)
    for s in range(n_lags):
        shift = s - (n - 1)
        m = n - abs(shift)
        if shift >= 0:
            linear[s] = np.sum(d[:m]*(1.0-h[shift:shift+m]) - h[:m]*d[shift:shift+m])
            quad[s] = -np.sum(d[:m]*d[shift:shift+m])
        else:
            g2 = -shift
            linear[s] = np.sum(d[g2:g2+m]*(1.0-h[:m]) - h[g2:g2+m]*d[:m])
            quad[s] = -np.sum(d[g2:g2+m]*d[:m])
    
    # c_k(t) = S0[k] + t*linear[k] + t^2*quad[k], all * dx
    # We want to minimize max_k c_k(t) over t in [t_min, t_max]
    
    # The max of quadratics is convex in t (each quadratic with negative leading coeff is concave... wait)
    # quad[k] = -sum d[i]*d[i+k] which can be positive or negative
    # So the max of these is not necessarily convex
    
    # Use golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f(t):
        vals = S0 + t * linear + t*t * quad
        return float(np.max(vals)) * dx
    
    a, b = t_min, t_max
    fa, fb = f(a), f(b)
    c = b - gr * (b - a)
    d2 = a + gr * (b - a)
    fc, fd2 = f(c), f(d2)
    
    for _ in range(60):
        if b - a < 1e-14:
            break
        if fc < fd2:
            b = d2
            fd2 = None
            d2 = c
            c = b - gr * (b - a)
            fc, fd2 = f(c), f(d2)
        else:
            a = c
            c = d2
            fc = fd2
            d2 = a + gr * (b - a)
            fc, fd2 = f(c), f(d2)
    
    t_best = (a + b) / 2.0
    f_best = f(t_best)
    return t_best, f_best

def _box_limited_step(h, d, max_step):
    """Find max step size to stay in [0,1] box."""
    t_max = max_step
    t_min = -max_step
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_min = max(t_min, float(np.max(h[neg] / d[neg])))
    return t_min, t_max

def _lp_descent_direction(h, active_lags, radius):
    """Find direction d with sum(d)=0 that minimizes max increase on active lags."""
    n = len(h)
    dx = 2.0 / n
    A = _corr_grads(h, active_lags)  # (m x n) matrix
    # We want: min max_i (A[i] @ d) subject to sum(d) = 0, ||d||_inf <= radius, box
    # Reformulated as LP: min t s.t. A[i]@d <= t, sum(d)=0, -box <= d <= box, t >= -eps
    m = len(active_lags)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # A[i]@d - t <= 0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.fun < 1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _null_space_directions(A, rng, count=10):
    """Generate random directions in the null space of A (or approximately)."""
    m, n = A.shape
    # Use SVD to find null space
    U, s, Vt = np.linalg.svd(A, full_matrices=True)
    # Null space is spanned by last (n-m) rows of Vt
    null_dim = min(n - m, n)
    if null_dim <= 0:
        return []
    V_null = Vt[m:m+null_dim, :]  # (null_dim x n)
    dirs = []
    for _ in range(count):
        coeffs = rng.standard_normal(null_dim)
        # Subtract mean to ensure sum ≈ 0
        d = coeffs @ V_null
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm > 1e-10:
            d /= norm
            dirs.append(d)
    return dirs

def _curvature_direction(h, active_lags, radius):
    """Find direction that exploits curvature: minimize max q_k(d) over active lags."""
    n = len(h)
    # For direction d, q_k(d) = -sum d[i]*d[i+k]*dx
    # We want to minimize max_{k in active} q_k(d)
    # This is equivalent to maximizing min_{k in active} sum d[i]*d[i+k]*dx
    # Heuristic: use gradient of the max of q_k
    # For small d, q_k(d) ≈ -d^T B_k d where B_k is the shift matrix
    # Simple heuristic: try -gradient of sum of q_k
    dx = 2.0 / n
    # For sum of q_k over active lags:
    # sum_k q_k(d) = -sum_k sum_i d[i]*d[i+k] * dx
    # Gradient w.r.t. d[j]: -2*sum_k [d[j-k] + d[j+k]] * dx (for valid indices)
    # Let's just use a simpler approach: random directions and pick best
    return None  # Handled in main loop

def _curvature_refine(h, active_lags, radius, rng, n_random=20):
    """Try random directions and find one where curvature helps."""
    n = len(h)
    dx = 2.0 / n
    best_dir = None
    best_val = np.inf
    
    S0 = np.correlate(h, 1.0 - h, mode='full')
    
    for _ in range(n_random):
        d = rng.standard_normal(n)
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm < 1e-10:
            continue
        d /= norm
        
        # Compute max at t = radius
        h_new = h + radius * d
        if h_new.min() < -1e-8 or h_new.max() > 1 + 1e-8:
            # Scale to fit
            pos = d > 0
            neg = d < 0
            t_scale = radius
            if np.any(pos) and h[pos].max() + t_scale * d[pos].max() > 1:
                t_scale = min(t_scale, (1 - h[pos].max()) / d[pos].max())
            if np.any(neg) and h[neg].min() + t_scale * d[neg].min() < 0:
                t_scale = min(t_scale, -h[neg].min() / d[neg].min())
            if t_scale < 1e-6:
                continue
            d *= t_scale / radius
        
        h_try = h + radius * d
        if h_try.min() < -1e-8 or h_try.max() > 1 + 1e-8:
            continue
        c = np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * dx
        if c < best_val:
            best_val = c
            best_dir = d.copy()
    
    return best_dir, best_val
