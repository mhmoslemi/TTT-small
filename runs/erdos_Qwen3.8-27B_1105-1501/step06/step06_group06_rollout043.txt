def _project_simplex(v):
    n=len(v)
    if n==0: return v
    if n==1: return np.ones(1)

    upper = h > 1.0 - eps

    for _ in range(5):
        if lower.any(): d[lower] = np.maximum(d[lower], 0.0)
        if upper.any(): d[upper] = np.minimum(d[upper], 0.0)
        if free.any():
            d[free] -= d.sum() / float(free.sum())

            # find positive lower and negative upper and scale to zero sum
            pos = lower & (d > 0)
            neg = upper & (d < 0)
            P = float(d[pos].sum()) if pos.any() else 0.0

def _finish_projection(x, target):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(5):
        res = target - float(x.sum())
        if abs(res) < 1e-13:
            break
        interior = (x > 1e-8) & (x < 1.0 - 1e-8)
        if interior.any():
            x[interior] += res / float(interior.sum())
            x = np.clip(x, 0.0, 1.0)
        else:
            if res > 0:
                idx = int(np.argmax(1.0 - x))
                x[idx] = min(1.0, x[idx] + res)
            else:
                idx = int(np.argmax(x))
                x[idx] = max(0.0, x[idx] + res)
    return x


def _project(v, target):

def _exact_c5(h):

def _corr_grads_selected(h, idx):
    n = len(h)
    m = len(idx)
    G = np.zeros((m, n), dtype=float)
    for row, k in enumerate(idx):
        s = int(k) - (n - 1)
        ab = abs(s)

        if length <= 0:
            continue

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.nonzero(u > (css - 1.0) / np.arange(1.0, n + 1.0))[0]
    if rho.size == 0:
        idx = 0
    else:
        idx = int(rho[-1])
    theta = (css[idx] - 1.0) / float(idx + 1)
    return np.maximum(v - theta, 0.0)


def _simplex_qp(A):
    m = A.shape[1]
    p = np.full(m, 1.0 / max(1, m))
    if m <= 1:
        return p
    H = A.T @ A
    try:
        L = float(np.linalg.eigvalsh(H)[-1])
    except Exception:
        L = float(np.linalg.norm(H, ord="fro"))
    if L <= 1e-14:
        return p
    step = 1.0 / max(1e-12, L)
    for _ in range(200):
        Ap = A @ p
        grad = A.T @ Ap
        p_new = _project_simplex(p - grad * step)
        if float(np.linalg.norm(p_new - p)) < 1e-11:

        p = p_new
    return p


def _make_dir_feasible(h, d):
    d = np.asarray(d, dtype=float).copy()
    eps = 1e-12
    lower = h < eps
    upper = h > 1.0 - eps
    free = ~(lower | upper)
    for _ in range(5):
        if lower.any():
            d[lower] = np.maximum(d[lower], 0.0)
        if upper.any():
            d[upper] = np.minimum(d[upper], 0.0)
        if free.any():
            d[free] -= d.sum() / float(free.sum())
        else:
            pos = lower & (d > 0.0)
            neg = upper & (d < 0.0)
            P = float(d[pos].sum()) if pos.any() else 0.0
            N = float(-d[neg].sum()) if neg.any() else 0.0
            if P > 0.0 and N > 0.0:
                if P <= N:
                    d[pos] *= N / P
                else:
                    d[neg] *= P / N
            else:

        ok = True
        if lower.any() and np.any(d[lower] < -1e-12):
            ok = False
        if upper.any() and np.any(d[upper] > 1e-12):
            ok = False
        if ok:
            break
    if free.any():
        d[free] -= d.sum() / float(free.sum())
    elif abs(float(d.sum())) > 1e-12:
        d *= 0.0
    return d

    T = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
    C = -np.correlate(d, d, mode="full")
    a = np.linspace(0.0, hi, 90)
    b = np.geomspace(max(hi * 1e-6, 1e-12), hi, 30)
    grid = np.unique(np.concatenate((a, b)))
    vals = _line_values(S0, T, C, grid) * dx

def _try_step(h, d, amax, target, cur, S0=None):
    alpha, _ = _line_search(h, d, amax=amax, S0=S0)
    if alpha <= 1e-12:
        return None, None, None
    for t in (alpha, alpha * 0.5, alpha * 0.25):
        if t <= 1e-12:
            continue
        h_new = _project(h + t * d, target)
        c = _exact_c5(h_new)
        if c < cur - 1e-12:
            return t, c, h_new
    return None, None, None


def _choose_active(corr_vals, M, max_active=60):
    L = len(corr_vals)
    top_k = min(30, L)
    top = np.argsort(-corr_vals)[:top_k]

    if near.size == 0:
        near = top
    union = np.unique(np.concatenate((top, near)))
    if union.size > max_active:
        order = np.argsort(-corr_vals[union])
        union = union[order[:max_active]]
    return union

def _slsqp_polish(h0, deadline, K=18, max_outer=6, maxiter=25):

    target = float(n) / 2.0
    dx = 2.0 / n
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur

    for _ in range(max_outer):
        if time.time() > deadline - 0.05:
            break
        S0 = np.correlate(h, 1.0 - h, mode="full")
        corr = S0 * dx
        idx = np.argsort(-corr)[:K]
        idx = np.atleast_1d(idx)

        _sl["target"] = target
        _sl["dx"] = dx
        _sl["idx"] = idx
        x0 = np.empty(n + 1)
        x0[:n] = h

                _slq_obj,

                    {"type": "eq", "fun": _slq_eq, "jac": _slq_eq_jac},
                ],
                options={"maxiter": maxiter, "ftol": 1e-12},

            break

def _active_subspace_basis(h, idx, r=8):
    n = len(h)
    G = _corr_grads_selected(h, idx)
    if G.size == 0:
        return np.empty((n, 0), dtype=float)
    G = G - G.mean(axis=1, keepdims=True)
    try:
        _, S, Vt = np.linalg.svd(G, full_matrices=False)
        rank = min(r, int(np.sum(S > 1e-10)), Vt.shape[0])
        if rank <= 0:
            return np.empty((n, 0), dtype=float)
        return Vt[:rank].T.copy()
    except Exception:
        return np.empty((n, 0), dtype=float)


def _fourier_basis(n, maxfreq=5):
    i = np.arange(n)
    cols = []
    for f in range(1, maxfreq + 1):
        cols.append(np.cos(np.pi * f * (i + 0.5) / n))
        cols.append(np.sin(np.pi * f * (i + 0.5) / n))
    M = np.column_stack(cols)
    M -= M.mean(axis=0, keepdims=True)
    try:
        _, S, Vt = np.linalg.svd(M, full_matrices=False)
        rank = min(int(np.sum(S > 1e-10)), Vt.shape[0], M.shape[1])
        if rank <= 0:
            return np.empty((n, 0), dtype=float)
        return Vt[:rank].T.copy()
    except Exception:
        return np.empty((n, 0), dtype=float)

def _combined_basis(vectors, n, rmax=12):
    cols = []
    for M in vectors:
        if M is not None and getattr(M, "size", 0) > 0:
            cols.append(M)
    if not cols:
        return np.empty((n, 0), dtype=float)
    A = np.hstack(cols)
    A -= A.mean(axis=0, keepdims=True)
    norms = np.linalg.norm(A, axis=0)
    good = norms > 1e-12
    if not good.any():
        return np.empty((n, 0), dtype=float)
    A[:, good] /= norms[good]
    try:

        rank = min(rmax, int(np.sum(S > 1e-8)), A.shape[1])
        if rank <= 0:

    except Exception:
        return np.empty((n, 0), dtype=float)

    return best_h, best_f


def _de_subspace(h0, basis, deadline, radius=0.25, seed=0, maxiter=20, popsize=8):
    h = np.asarray(h0, dtype=float)
    n = len(h)
    target = float(n) / 2.0
    best_h = h.copy()
    best_f = _exact_c5(h)
    dim = 0
    if basis is not None and getattr(basis, "size", 0) > 0:
        dim = int(basis.shape[1])
    if dim == 0 or dim > 14:
        return best_h, best_f

    _cp["center"] = h
    _cp["basis"] = basis
    _cp["target"] = target
    _cp["radius"] = float(radius)
    _cp["deadline"] = float(deadline)
    _cp["best_f"] = np.inf
    _cp["best_x"] = None
    _cp["best_h"] = None

    bounds = [(-float(radius), float(radius))] * dim
    try:
        differential_evolution(
            _cp_obj,
            bounds,
            seed=int(seed),
            maxiter=int(maxiter),
            popsize=int(popsize),
            tol=1e-6,
            polish=False,
            init="latinhypercube",
            callback=_de_callback,
        )
    except Exception:
        pass
    if _cp.get("best_h") is not None:
        if _cp["best_f"] < best_f:
            best_f = float(_cp["best_f"])
            best_h = np.array(_cp["best_h"], dtype=float, copy=True)
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return _project(h.copy(), n_new / 2.0)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _default_init(n, seed):
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    h = (
        0.5
        + 0.30 * np.sin(2.0 * np.pi * 2.0 * x / n + 0.4)
        + 0.18 * np.sin(2.0 * np.pi * 5.0 * x / n + 1.1)
        + 0.08 * np.sin(2.0 * np.pi * 9.0 * x / n)
    )
    h = h + rng.normal(0.0, 0.01, n)
    return _project(h, n / 2.0)


def _alt_starts(h_best, n, rng, count=4):
    starts = []
    starts.append(_project(h_best + rng.normal(0.0, 0.03, n), n / 2.0))

    for _ in range(2):
        r = rng.random(n)
        b = np.zeros(n)
        take = int(n // 2)
        if take > 0:
            b[np.argsort(r)[-take:]] = 1.0
        starts.append(_project(b + 0.001 * r, n / 2.0))

    x = np.arange(n) / float(n)
    for f in (1, 2, 3, 5, 7):
        ph = rng.uniform(0.0, 2.0 * np.pi)
        hh = 0.5 + 0.32 * np.sin(2.0 * np.pi * f * x + ph) + 0.10 * np.sin(4.0 * np.pi * f * x + 0.7)
        starts.append(_project(hh, n / 2.0))

    hh = np.zeros(n)
    pos = 0
    while pos < n:
        hi = max(4, min(n // 4, n - pos))
        blen = int(rng.integers(2, hi + 1)) if hi > 2 else 1
        if rng.random() < 0.5:
            hh[pos:pos + blen] = 1.0
        pos += blen
    starts.append(_project(hh, n / 2.0))
    return starts[:max(0, count)]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    deadline = t0 + B
    rng = np.random.default_rng(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is None or h0.size < 10:
        h0 = _default_init(122, seed)

    n0 = int(h0.size)
    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    if time.time() < deadline - 3.0:
        t1 = min(t0 + 0.10 * B, deadline - 2.0)
        it = 25 if B > 200 else 8
        hc, cc = _slsqp_polish(h, t1, K=18, max_outer=6, maxiter=it)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    t2 = min(t0 + 0.30 * B, deadline - 4.0)
    if time.time() < t2:
        hc, cc = _active_p_descent(best_h, t2, max_iter=3000, seed=seed + 10, start_radius=0.08)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    t3 = min(t0 + 0.40 * B, deadline - 4.0)
    if time.time() < t3:
        hc, cc = _slsqp_polish(best_h, t3, K=18, max_outer=5, maxiter=18 if B > 300 else 8)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    t4 = min(t0 + 0.55 * B, deadline - 10.0)
    if time.time() < t4:
        S0 = np.correlate(best_h, 1.0 - best_h, mode="full")
        dx = 2.0 / best_n
        corr = S0 * dx
        active = _choose_active(corr, float(corr.max()), max_active=60)
        basis_act = _active_subspace_basis(best_h, active, 8)
        if basis_act.size > 0:
            sub = min(t4, time.time() + max(5.0, 0.045 * B))
            hc, cc = _powell_subspace(best_h, basis_act, sub, radius=0.12, seed=seed + 200, n_starts=2)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

        basis_f = _fourier_basis(best_n, 5)
        if basis_f.size > 0 and time.time() < t4:
            sub = min(t4, time.time() + max(5.0, 0.045 * B))
            hc, cc = _powell_subspace(best_h, basis_f, sub, radius=0.30, seed=seed + 210, n_starts=2)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

        if time.time() < t4:
            basis_rand = _random_basis(best_n, 6, seed + 220)
            basis_comb = _combined_basis([basis_act, basis_f, basis_rand], best_n, 12)
            if basis_comb.size > 0:
                sub = min(t4, time.time() + max(5.0, 0.05 * B))
                hc, cc = _de_subspace(best_h, basis_comb, sub, radius=0.30, seed=seed + 230, maxiter=20, popsize=8)
                if cc < best_c5:
                    best_c5 = cc
                    best_h = hc.copy()
                if time.time() < t4:
                    sub = min(t4, time.time() + max(5.0, 0.04 * B))
                    hc, cc = _powell_subspace(best_h, basis_comb, sub, radius=0.30, seed=seed + 240, n_starts=2)
                    if cc < best_c5:
                        best_c5 = cc
                        best_h = hc.copy()

    t5 = min(t0 + 0.80 * B, deadline - 15.0)
    offsets = [-8, 8, -16, 16, -24, 24, -32, 32, -48, 48, -64, 64]
    tried = 0
    for dn in offsets:
        if time.time() > t5 - 5.0 or tried >= 6:
            break
        nt = best_n + dn
        if nt <= 20 or nt > 240 or nt % 2 != 0:
            continue
        if nt == best_n:
            continue
        tried += 1
        hr = _resample(best_h, nt)
        sub = min(t5, time.time() + max(3.0, 0.025 * B))
        hr, cr = _active_p_descent(hr, sub, max_iter=500, seed=seed + 500 + tried, start_radius=0.06)
        if time.time() < t5 - 3.0:
            sub2 = min(t5, time.time() + max(2.0, 0.012 * B))
            hr, cr = _slsqp_polish(hr, sub2, K=14, max_outer=4, maxiter=12 if B < 500 else 18)
        if time.time() < t5 - 2.0:
            S0r = np.correlate(hr, 1.0 - hr,mode="full")
            dxr = 2.0 / nt
            corrr = S0r * dxr
            actr = _choose_active(corrr, float(corrr.max()), max_active=40)
            basr = _active_subspace_basis(hr, actr, 6)
            if basr.size > 0:
                sub3 = min(t5, time.time() + max(2.0, 0.012 * B))
                hr, cr = _powell_subspace(hr, basr, sub3, radius=0.08, seed=seed + 600 + tried, n_starts=1)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            best_n = nt

    t6 = min(t0 + 0.92 * B, deadline - 10.0)
    if time.time() < t6 and B > 60:
        starts = _alt_starts(best_h, best_n, rng, 4)
        for si, s in enumerate(starts):
            if time.time() > t6 - 3.0:
                break
            sub = min(t6, time.time() + max(3.0, 0.018 * B))
            hs, cs = _active_p_descent(s, sub, max_iter=300, seed=seed + 700 + si, start_radius=0.06)
            if time.time() < t6 - 2.0:
                sub2 = min(t6, time.time() + max(2.0, 0.008 * B))
                hs, cs = _slsqp_polish(hs, sub2, K=12, max_outer=3, maxiter=10)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()
                best_n = len(hs)

    if deadline - time.time() > 5.0:
        sub = deadline - max(2.0, 0.005 * B)
        hc, cc = _active_p_descent(best_h, sub, max_iter=3000, seed=seed + 950, start_radius=0.05)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    if time.time() < deadline - 3.0:
        sub = min(deadline - 1.0, time.time() + max(1.0, 0.005 * B))
        hc, cc = _slsqp_polish(best_h, sub, K=15, max_outer=2, maxiter=5)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _exact_c5(best_h)
    return best_h, float(best_c5), len(best_h)
