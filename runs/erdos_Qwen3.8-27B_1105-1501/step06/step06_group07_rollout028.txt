import numpy as np
from scipy.optimize import minimize
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        idx1 = np.arange(k, n)
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, m)
        g[idx2] -= h[idx2 + k]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        idx1 = np.arange(0, m)
        g[idx1] += (1.0 - h[idx1 + ga])
        idx2 = np.arange(ga, n)
        g[idx2] -= h[idx2 - ga]
    return g * dx


def _full_lse_grad(h, eps):
    """Compute full log-sum-exp gradient in h-space."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    # Stable softmax
    v_max = np.max(v)
    exp_v = np.exp((v - v_max) / eps)
    p = exp_v / np.sum(exp_v)
    # Sum gradients weighted by p
    G = np.zeros(n)
    for r in range(len(corr)):
        k = r - (n - 1)
        G += p[r] * _grad_row(h, k)
    return G, eps * (np.log(np.sum(exp_v)) + v_max / eps)


def _logit_transform(h, eps_clip=1e-10):
    """Convert h to logit space. Returns u such that sigmoid(u) ≈ h."""
    h_clipped = np.clip(h, eps_clip, 1.0 - eps_clip)
    return np.log(h_clipped / (1.0 - h_clipped))


def _solve_lambda(u, target):
    """Find lambda such that sum(sigmoid(u + lambda)) = target."""
    lo = -30.0
    hi = 30.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(1.0 / (1.0 + np.exp(-(u + mid)))))
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0


def _u_to_h(u, target):
    """Convert u-space to h-space using logit + lambda."""
    lam = _solve_lambda(u, target)
    return 1.0 / (1.0 + np.exp(-(u + lam)))


def _u_objective(u, target, eps):
    """Objective and gradient in u-space."""
    n = len(u)
    h = _u_to_h(u, target)
    G_h, F_val = _full_lse_grad(h, eps)
    # Chain rule through sigmoid + constraint
    s_val = h * (1.0 - h)
    S = float(np.sum(s_val))
    if S < 1e-14:
        return F_val, np.zeros(n)
    G_u = s_val * (G_h - float(np.dot(G_h, s_val)) / S)
    return F_val, G_u


def _lbfgs_lse(h_init, deadline, target, eps_start=0.05, eps_end=0.005,
                n_eps=5, maxiter=200):
    """L-BFGS optimization in u-space with log-sum-exp objective."""
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    h = h_init.copy()
    
    u0 = _logit_transform(h)
    
    for ei in range(n_eps):
        frac = ei / max(n_eps - 1, 1)
        eps = eps_start * (eps_end / eps_start) ** frac
        
        # Keep the last values and conjugate directions via scipy
        result = _do_lbfgs(u0, target, eps, deadline, maxiter)
        if result is not None:
            u0, h_trial = result
            c5_trial = _exact_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
    
    return best_h, best_c5


def _do_lbfgs(u0, target, eps, deadline, maxiter):
    """Run one L-BFGS call in u-space."""
    n = len(u0)
    
    def _obj_u(u):
        val, _ = _u_objective(u, target, eps)
        return val
    
    def _grad_u(u):
        _, g = _u_objective(u, target, eps)
        return g
    
    x0 = u0.copy()
    res = minimize(_obj_u, x0, method='L-BFGS-B', jac=_grad_u,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                            'iprint': -1, 'maxls': 50})
    
    if time.time() > deadline:
        return x0, _u_to_h(x0, target)
    
    h_new = _u_to_h(res.x, target)
    h_new = _project(h_new, target)
    return res.x, h_new

def _grad_row(h, k):
    n = len(h)

    g = np.zeros(n)

        if m <= 0:
            return g

        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, m)

        g[idx1] += (1.0 - h[idx1 + ga])

                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    ones_idx = np.where(h > 0.5)[0].tolist()
                    zeros_idx = np.where(h <= 0.5)[0].tolist()
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stuck = 0
                else:
                    h[start:start + block_len] = block
                    stuck += 1
            else:
                stuck += 1
        else:
            # Multi-swap: swap 2 or 3 pairs simultaneously
            n_swap = int(rng.integers(2, 4))
            if len(ones_idx) >= n_swap and len(zeros_idx) >= n_swap:
                swap_ones = rng.choice(len(ones_idx), size=n_swap, replace=False)
                swap_zeros = rng.choice(len(zeros_idx), size=n_swap, replace=False)
                old_ones = [ones_idx[si] for si in swap_ones]
                old_zeros = [zeros_idx[sz] for sz in swap_zeros]
                for oi in old_ones:
                    h[oi] = 0.0
                for oz in old_zeros:
                    h[oz] = 1.0
                new_c5 = _exact_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    ones_idx = np.where(h > 0.5)[0].tolist()
                    zeros_idx = np.where(h <= 0.5)[0].tolist()
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stuck = 0
                else:
                    for oi in old_ones:
                        h[oi] = 1.0
                    for oz in old_zeros:
                        h[oz] = 0.0
                    stuck += 1
        if stuck > 4000:
            T = max(T * 1.4, 0.003)
            stuck = 0
        T = max(T_min, T * cooling)
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _perturbation_search(h, deadline, rng, n_trials=100):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.08
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        h_try, c5_try = _active_descent(h_pert, min(deadline, time.time() + 2.0), K=15, max_iters=40)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.2, scale * 1.15)
    return best_h, best_c5


def _smooth_max_descent(h, deadline, alpha_init=20.0, alpha_mult=1.5,
                        iters_per_stage=40, max_stages=15, K=35):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    alpha = alpha_init
    step = 0.25
    stall = 0
    for stage in range(max_stages):
        for _ in range(iters_per_stage):
            if time.time() > deadline:
                return best_h, best_c5
            corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            top_idx = np.argsort(-corr)[:K]
            top_vals = corr[top_idx]
            w = np.exp(alpha * (top_vals - top_vals[0]))
            w_sum = np.sum(w)
            if w_sum < 1e-300:
                w = np.ones(K) / K
            else:
                w /= w_sum
            grad = np.zeros(n)
            for wi, idx in zip(w, top_idx):
                k = int(idx) - (n - 1)
                grad += wi * _grad_row(h, k)
            grad -= np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                return best_h, best_c5
            direction = -grad / gn
            f0 = _exact_c5(h)
            # Armijo line search
            best_t = 0.0
            best_f_ls = f0
            t = step
            for _ in range(25):
                h_try = _project(h + t * direction, target)
                f_try = _exact_c5(h_try)
                if f_try < best_f_ls:
                    best_f_ls = f_try
                    best_t = t
                if f_try <= f0 + 1e-4 * t * (-np.dot(grad, direction)):
                    break
                t *= 0.6
                if t < 1e-10:
                    break
            if best_t > 1e-12 and best_f_ls < best_c5 - 1e-13:
                h = _project(h + best_t * direction, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                stall = 0
                if best_t > 0.5 * step:
                    step = min(step * 1.1, 1.0)
                elif best_t < 0.1 * step:
                    step = max(step * 0.65, 0.01)
            else:
                stall += 1
                step *= 0.7
                if stall > 25:
                    break
        alpha *= alpha_mult
        if alpha > 1000:
            alpha = 1000.0
        if stall > 25:
            step = max(step * 0.5, 0.005)
            stall = 0
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2.0 * np.pi * 5.0 * x / n0)
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: L-BFGS with full log-sum-exp (25% of time)
    d1 = t0 + budget_s * 0.25
    eps_schedule = [0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.002]
    h1, c5_1 = _lbfgs_lse(h, d1, n0 / 2.0, eps_schedule)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2 Smooth max gradient descent (10%)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _smooth_max_descent(best_h, d2, alpha_init=15.0, alpha_mult=1.4,
                                   iters_per_stage=40, max_stages=20, K=30)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA with improvements (25%)
    d3 = t0 + budget_s * 0.60
    # First pass: standard binary SA
    h3a, c5_3a = _binary_sa(best_h, min(d3, time.time() + budget_s * 0.08), rng,
                            max_iters=80000, T_init=0.014)
    if c5_3a < best_c5:
        best_c5 = c5_3a
        best_h = h3a.copy()
    # Second pass: binary SA with block moves
    h3b, c5_3b = _binary_sa_blocks(best_h, min(d3, time.time() + budget_s * 0.08), rng,
                                   max_iters=50000)
    if c5_3b < best_c5:
        best_c5 = c5_3b
        best_h = h3b.copy()
    # Polish binary with active descent
    if time.time() < d3:
        h3c, c5_3c = _active_descent(best_h, d3, K=15, max_iters=80)
        if c5_3c < best_c5:
            best_c5 = c5_3c
            best_h = h3c.copy()

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.80
    best_n = len(best_h)
    n_offsets = [-6, -4, -2, 2, 4, 6, -10, 10, -8, 8, 12, -12, 14, -14, 16, -16, 20, -20]
    tried_n = set()
    for dn in n_offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        # Quick L-BFGS
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        eps_short = [0.03, 0.015, 0.008]
        h_r, c5_r = _lbfgs_lse(h_r, sub_dl, n_try / 2.0, eps_short)
        # Binary round + SA
        if time.time() < d4:
            sub_dl2 = min(d4, time.time() + budget_s * 0.012)
            h_rb, c5_rb = _binary_sa(h_r, sub_dl2, rng, max_iters=40000, T_init=0.012)
            if c5_rb < c5_r:
                h_r, c5_r = h_rb, c5_rb
        # Polynomial active descent
        if time.time() < d4:
            sub_dl3 = min(d4, time.time() + budget_s * 0.008)
            h_r2, c5_r2 = _active_descent(h_r, sub_dl3, K=15, max_iters=60)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Perturbation and restarts (10%)
    d5 = t0 + budget_s * 0.90
    h5, c5_5 = _perturbation_search(best_h, d5, rng, n_trials=80)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final polish (10%)
    d6 = deadline - 2.0
    if time.time() < d6:
        h6, c5_6 = _active_descent(best_h, d6, K=12, max_iters=100)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
        # One more binary SA from the polished solution
        if time.time() < d6:
            h6b, c5_6b = _binary_sa(best_h, d6, rng, max_iters=30000, T_init=0.008)
            if c5_6b < best_c5:
                best_c5 = c5_6b
                best_h = h6b.copy()

    # Final: ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
