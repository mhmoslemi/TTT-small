        return np.ones(n)

        for T in [0.05, 0.02]:
            if time.time() > sub_dl1:
                break
            h_r, c5_r = _smoothed_slsqp(h_r, T, sub_dl1, maxiter=25)
        if time.time() > deadline:
            break
        sub_dl2 = min(deadline, time.time() + time_per * 0.6)
        h_r, c5_r = _active_refine(h_r, sub_dl2, max_iter=200, init_tol=0.003)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            h_best = h_r.copy()
            best_n = n_try

    return h_best, best_c5, best_n

def _symmetry_restart(h, n, rng):
    h2 = 1.0 - h
    h2 = _project(h2, n / 2.0)
    if rng.random() < 0.5:
        h2 = np.roll(h2, rng.integers(1, n))
    h2 = _project(h2, n / 2.0)
    return h2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = np.full(n0, 0.5)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0

    h = _project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Smoothed SLSQP with decreasing temperature (20% time)
    d1 = t0 + budget_s * 0.20
    for T in [0.1, 0.05, 0.03, 0.015, 0.008]:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.04)
        h1, c5_1 = _smoothed_slsqp(h, T, sub_dl, maxiter=35)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = len(h1)
        h = h1

    # Phase 2: Active refinement (25% time)
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _active_refine(best_h, d2, max_iter=400, init_tol=0.004)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: Exact SLSQP (10% time)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _exact_slsqp(best_h, d3, maxiter=30)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Multi-resolution (30% time)
    d4 = t0 + budget_s * 0.85
    best_h, best_c5, best_n = _multi_resolution(
        best_h, best_c5, best_n, d4, time_per=budget_s * 0.03)

    # Phase 5: Symmetry restarts + refinement (8% time)
    d5 = t0 + budget_s * 0.93
    for _ in range(3):
        if time.time() > d5:
            break
        h_s = _symmetry_restart(best_h, best_n, rng)
        c5_s = _exact_c5(h_s)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_s, c5_s = _active_refine(h_s, sub_dl, max_iter=150, init_tol=0.003)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 6: Final polish (7% time)
    d6 = deadline
    h6, c5_6 = _active_refine(best_h, d6, max_iter=300, init_tol=0.002)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)

    # Final validity check
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
