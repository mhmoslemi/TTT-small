import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Project v onto {x : sum(x) = target, 0 <= x <= 1}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_full(h):
    """Compute full cross-correlation h * (1-h) and return as array."""
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_grads_full(h):
    """Compute gradient of each correlation with respect to h.
    Returns array of shape (2n-1, n) where G[k, i] = d corr[k] / d h[i]
    """
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m_ = n - abs(s)
        if s >= 0:
            # corr[k] = sum_{i=0}^{m_-1} h[i] * (1-h[i+s])
            # d/dh[i] = (1-h[i+s]) for i in 0..m_-1
            # d/dh[j] = -sum of h[i] where i+s=j, i.e., i=j-s, for j in s..s+m_-1
            G[k, :m_] += (1.0 - h[s:s + m_])
            G[k, s:s + m_] += (-h[:m_])
        else:
            g = -s
            # corr[k] = sum_{i=0}^{m_-1} h[i+g] * (1-h[i])
            # d/dh[j] for j in g..g+m_-1: (1-h[j-g])
            # d/dh[j] for j in 0..m_-1: -h[j+g]
            G[k, g:g + m_] += (1.0 - h[:m_])
            G[k, :m_] += (-h[g:g + m_])
    return G

def _smooth_max_optimize(h0, deadline, n_mu_steps=20):
    """Minimize smooth max (log-sum-exp) of correlations using L-BFGS-B."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    mus = np.geomspace(0.1, 1e-4, n_mu_steps)

        if time.time() > deadline:
            break
        
        def obj_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            # Smooth max with penalty for constraint violation
            m = float(np.max(c))
            val = mu * np.log(np.sum(np.exp((c - m) / mu))) + m
            # Penalty for sum constraint
            s = float(np.sum(hx))
            val += 100.0 * (s - target) ** 2
            # Penalty for bounds (shouldn't trigger with L-BFGS-B bounds)
            return val
        
        def obj_grad(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            m = float(np.max(c))
            w = np.exp((c - m) / mu)
            w /= w.sum()
            # Gradient of smooth max
            G = _corr_grads_full(hx) * dx
            grad = (w[:, None] * G).sum(axis=0)
            # Gradient of penalty
            grad += 200.0 * (float(np.sum(hx)) - target) * np.ones(n)
            return grad
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj_fn, h, method='L-BFGS-B', jac=obj_grad,

            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    
    return best_h, best_c5

def _coordinate_descent(h0, deadline, max_rounds=500):
    """Coordinate descent on individual values."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(123)
    
    for round_ in range(max_rounds):
        if time.time() > deadline:
            break
        
        improved = False
        # Random order
        order = rng.permutation(n)
        
        for i in order:
            if time.time() > deadline:
                break
            old_val = h[i]
            # Try adjusting h[i] while maintaining sum constraint
            # If we change h[i] by delta, we need to adjust another element
            # Simpler: try a small change and re-project
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_test = h.copy()
                h_test[i] = old_val + delta
                h_test = _project(h_test, target)
                c5_test = _exact_c5(h_test)
                if c5_test < best_c5 - 1e-14:
                    h = h_test
                    best_c5 = c5_test
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            # Perturb randomly
            h = h + rng.normal(0, 0.01, n)
            h = _project(h, target)
            best_c5 = min(best_c5, _exact_c5(h))
            if _exact_c5(h) > best_c5 + 0.01:
                h = best_h.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
