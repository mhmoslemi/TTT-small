import numpy as np
from scipy.optimize import minimize, linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grads_batch(h, lags):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for i, s in enumerate(lags):
        m = n - abs(s)
        if s >= 0:
            G[i, :m] += (1.0 - h[s:s+m])
            G[i, s:s+m] += (-h[:m])
        else:
            g = -s
            G[i, g:g+m] += (1.0 - h[:m])
            G[i, :m] += (-h[g:g+m])
    G *= dx
    return G

def _line_search(h, d, t_max):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    t_lo, t_hi = -t_max, t_max
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -t_max)
    t_hi = min(t_hi, t_max)
    if t_hi - t_lo < 1e-15:
        return 0.0, _exact_c5(h)
    ng = 80
    grid = np.linspace(t_lo, t_hi, ng)
    vals = S0[None, :] + grid[:, None] * T[None, :] + grid[:, None]**2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    bi = int(np.argmin(obj))
    best_t, best_val = grid[bi], obj[bi]
    if 0 < bi < ng - 1:
        a, b = grid[bi-1], grid[bi+1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    fa = float(np.max(S0 + a*T + a*a*Q)) * dx
    fb = float(np.max(S0 + b*T + b*b*Q)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if fa < fb:
            b, fb = a, fa
            a = b - gr * (b - a)
            if a < t_lo: a = t_lo
            fa = float(np.max(S0 + a*T + a*a*Q)) * dx
        else:
            a, fa = b, fb
            b = a + gr * (b - a)
            if b > t_hi: b = t_hi
            fb = float(np.max(S0 + b*T + b*b*Q)) * dx
    mid = (a + b) / 2.0
    mv = float(np.max(S0 + mid*T + mid*mid*Q)) * dx
    if mv < best_val:
        best_t, best_val = mid, mv
    return best_t, best_val

def _qp_descent(h0, deadline, max_iter=200, tol=2e-5):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur = _exact_c5(h)
    bh, bc = h.copy(), cur
    radius = 0.06
    stall = 0
    min_r = 0.003
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 80:
            if radius > min_r:
                radius = max(min_r, radius * 0.5)
            else:
                break
        cv = _corr_vec(h)
        M = cur
        act = np.where(cv > M - tol)[0]
        if len(act) == 0:
            act = np.argsort(-cv)[:5]
        elif len(act) > 60:
            oo = np.argsort(-cv[act])
            act = act[oo[:60]]
        lags = act - (n - 1)
        G = _grads_batch(h, lags)
        cvec = np.zeros(n + 1)
        cvec[-1] = 1.0
        A_ub = np.zeros((len(act), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -cv[act]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(-radius, radius)] * n + [(None, None)]
        found_dir = False
        try:
            res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                         bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n].copy()
                d -= d.mean()
                found_dir = True
        except Exception:
            pass
        if not found_dir:
            gs = G.sum(axis=0)
            gs -= gs.mean()
            pn = np.linalg.norm(gs)
            if pn < 1e-13:
                stall += 15
                radius *= 0.6
                continue
            d = -gs / pn * radius
            d -= d.mean()
        alpha, new_c5 = _line_search(h, d, radius)
        h_new = h + alpha * d
        h_new = _project(h_new, target)
        nc = _exact_c5(h_new)
        if nc < cur - 1e-14:
            h, cur = h_new, nc
            if cur < bc:
                bc, bh = cur, h.copy()
            stall = 0
            if abs(alpha) > 0.4 * radius:
                radius = min(radius * 1.12, 0.30)
            else:
                radius = min(radius * 1.02, 0.25)
        else:
            stall += 1
            radius *= 0.78
            if radius < min_r:
                break
    return bh, bc

def _grad_refine(h0, deadline, n_steps=500):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur = _exact_c5(h)
    bh, bc = h.copy(), cur
    step = 0.002
    for i in range(n_steps):
        if time.time() > deadline:
            break
        cv = _corr_vec(h)
        M = cur
        act = np.where(cv > M - 5e-5)[0]
        if len(act) == 0:
            act = np.argsort(-cv)[:5]
        elif len(act) > 40:
            oo = np.argsort(-cv[act])
            act = act[oo[:40]]
        lags = act - (n - 1)
        G = _grads_batch(h, lags)
        gs = G.sum(axis=0)
        gs -= gs.mean()
        pn = np.linalg.norm(gs)
        if pn < 1e-14:
            break
        d = -gs / pn * step
        alpha, nc = _line_search(h, d, step)
        h_new = _project(h + alpha * d, target)
        nc2 = _exact_c5(h_new)
        if nc2 < cur - 1e-15:
            h, cur = h_new, nc2
            if cur < bc:
                bc, bh = cur, h.copy()
            step = min(step * 1.01, 0.008)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return bh, bc

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    hn = np.interp(xn, xo, h)
    return _project(hn, n_new / 2.0)

def _fourier_opt(h0, n_modes, deadline):
    n = len(h0)
    target = n / 2.0
    x = (np.arange(n) + 0.5) / n
    basis = [np.ones(n)]
    for k in range(1, n_modes + 1):
        basis.append(np.cos(2 * np.pi * k * x))
        basis.append(np.sin(2 * np.pi * k * x))
    B = np.column_stack(basis)
    db = B.shape[1]
    try:
        c0 = np.linalg.lstsq(B, h0, rcond=None)[0]
    except Exception:
        c0 = np.zeros(db)
        c0[0] = 0.5

    def objf(c):
        hh = _project(B @ c, target)
        return _exact_c5(hh)

    res = minimize(objf, c0, method='Nelder-Mead',
                   options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
    c_best = res.x if res.fun < objf(c0) else c0
    h_final = _project(B @ c_best, target)
    return h_final, _exact_c5(h_final)

def _random_start(n, rng):
    types = rng.integers(0, 4)
    if types == 0:
        h = 0.5 + rng.normal(0, 0.15, n)
    elif types == 1:
        freq = rng.integers(1, 6)
        ph = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x + ph) + rng.normal(0, 0.05, n)
    elif types == 2:
        nb = rng.integers(3, 12)
        h = np.zeros(n)
        edges = np.sort(rng.uniform(0, 1, nb))
        for i in range(nb):
            s, e = int(edges[i] * n), int(edges[i+1] * n) if i+1 < nb else n
            if rng.random() < 0.5:
                h[s:e] = 1.0
        h = h + rng.normal(0, 0.05, n)
    else:
        h = np.zeros(n)
        h[::2] = 1.0
        h += rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)
