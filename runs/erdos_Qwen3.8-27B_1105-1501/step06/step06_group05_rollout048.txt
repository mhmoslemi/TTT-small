        if f_new < f + 1e-5:
            h = h_new
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                h = _project(best_h + rng.normal(0, sigma, n), n / 2.0)
                f = _exact_c5(h)
                stall = 0
                sigma = max(0.005, sigma * 0.85)
    
    return best_h, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
