    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _min_convex_max_lines(c, d, t_init=0.5):
    """Find t in [0,1] minimizing f(t) = max_k (c[k] + d[k]*t).
    f is convex (max of linear functions).
    """
    # Check endpoints
    f0 = float(np.max(c))
    f1 = float(np.max(c + d))
    
    # Ternary/golden section search for the minimum
    # Since f is convex on [0,1], we can use golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    
    # Quick check: if f is increasing from 0, min is at 0
    # f'_right(0) = max of d[k] where c[k] = max(c)
    active_0 = np.isclose(c, f0, atol=1e-14)
    if np.min(d[active_0]) > 1e-14:
        return 0.0
    # If f is decreasing at 1, min is at 1
    active_1 = np.isclose(c + d, f1, atol=1e-14)
    if np.max(d[active_1]) < -1e-14:
        return 1.0
    
    # Golden section search
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f_c1 = float(np.max(c + d * c1))
    f_c2 = float(np.max(c + d * c2))
    
    for _ in range(60):
        if b - a < 1e-14:
            break
        if f_c1 < f_c2:
            b = c2
            c2 = c1
            f_c2 = f_c1
            c1 = b - gr * (b - a)
            f_c1 = float(np.max(c + d * c1))
        else:
            a = c1
            c1 = c2
            f_c1 = f_c2
            c2 = a + gr * (b - a)
            f_c2 = float(np.max(c + d * c2))
    
    t = (a + b) / 2.0
    f_t = float(np.max(c + d * t))
    
    # Compare with endpoints
    best_t = t
    best_f = f_t
    if f0 < best_f:
        best_t = 0.0
        best_f = f0
    if f1 < best_f:
        best_t = 1.0
        best_f = f1
    return best_t

def _coordinate_descent(h, max_sweeps=50, tol=1e-14, deadline=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        max_change = 0.0
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            s0 = _compute_scores(h)
            h[i] = 1.0
            s1 = _compute_scores(h)
            h[i] = old
            
            # Linear interpolation: score[t] = s0 + t*(s1-s0)
            t_opt = _min_convex_max_lines(s0, s1 - s0, old)
            h[i] = t_opt
            max_change = max(max_change, abs(t_opt - old))
        
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        if max_change < tol:
            break
    
    best_h = _project(best_h, target)
    best_c5 = _exact_c5(best_h)
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _scores(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _min_max_lines(c, d):
    n = len(c)
    f0 = float(np.max(c))
    f1 = float(np.max(c + d))
    act0 = c >= f0 - 1e-14
    if np.min(d[act0]) > 1e-14:
        return 0.0
    act1 = (c + d) >= f1 - 1e-14
    if np.max(d[act1]) < -1e-14:
        return 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    fc1 = float(np.max(c + d * c1))
    fc2 = float(np.max(c + d * c2))
    for _ in range(50):
        if b - a < 1e-15:
            break
        if fc1 < fc2:
            b = c2; c2 = c1; fc2 = fc1
            c1 = b - gr * (b - a)
            fc1 = float(np.max(c + d * c1))
        else:
            a = c1; c1 = c2; fc1 = fc2
            c2 = a + gr * (b - a)
            fc2 = float(np.max(c + d * c2))

def _lp_direction(h, scores, radius):
    n = len(h)
    nn = 2 * n - 1
    G = np.zeros((nn, n))
    for d in range(-(n - 1), n):
        r = n - 1 + d
        if d >= 0:
            m = n - d
            if m > 0:
                G[r, d:] += (1.0 - h[:m])
                G[r, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[r, :m] += (1.0 - h[g:])
                G[r, g:] += -h[:m]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((nn, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.10, stall_limit=50):

    target = n / 2.0

    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = _scores(h)
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-14:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        h_new = h + p
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur_c5 - 1e-13:

            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius = min(radius * 1.05, 0.3)
        else:
            stall += 1
            radius *= 0.7
    return best_h, best_c5

    # Phase 3: Binary optimization (15%)
    d3 = t0 + budget_s * 0.40
    h_bin = _to_binary(h_best, n0)
    sub_d = min(d3, time.time() + budget_s * 0.06)
    h_bin, c5_bin = _binary_sa(h_bin, sub_d, max_iter=40000, T_init=0.0025)
    sub_d2 = min(d3, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _greedy_binary_sweep(h_bin, sub_d2)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    # CD on binary-projected
    if time.time() < d3:
        h_poly = _project(h_best.copy(), n0 / 2.0)
        h_poly, c5_poly = _coord_descent(h_poly, max_sweeps=150, deadline=d3)
        if c5_poly < c5_best:
            c5_best = c5_poly
            h_best = h_poly.copy()

    # Phase 4: Multi-n exploration (25%)
    d4 = t0 + budget_s * 0.65
    n_best = len(h_best)
    tried_ns = set()
    dn_list = list(range(-40, 41, 4)) + list(range(-30, 31, 2))
    dn_list = sorted(set(dn_list), key=abs)
    for dn in dn_list:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 300 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(h_best, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.015)
        h_r, c5_r = _coord_descent(h_r, max_sweeps=80, deadline=sub_dl)
        # Try binary at this n
        if time.time() < d4 - 10:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d4, time.time() + budget_s * 0.010)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=20000, T_init=0.002, seed=seed + n_try)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
            h_p = _project(h_r.copy(), n_try / 2.0)
            sub_dl3 = min(d4, time.time() + budget_s * 0.008)
            h_p, c5_p = _coord_descent(h_p, max_sweeps=60, deadline=sub_dl3)
            if c5_p < c5_r:
                c5_r = c5_p
                h_r = h_p
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r.copy()
            n_best = n_try

    # Phase 5: Basin hopping (15%)
    d5 = t0 + budget_s * 0.80
    scale = 0.04
    for it in range(300):
        if time.time() > d5:
            break
        h_pert = _smooth_perturb(h_best, scale, rng)
        sub_dl = min(d5, time.time() + budget_s * 0.006)
        h_t, c5_t = _coord_descent(h_pert, max_sweeps=40, deadline=sub_dl)
        if c5_t < c5_best:
            c5_best = c5_t
            h_best = h_t.copy()
            scale = max(0.003, scale * 0.80)
        else:
            scale = min(0.12, scale * 1.08)

    # Phase 6: Try random binary starts and SA (10%)
    d6 = t0 + budget_s * 0.90
    n_cur = len(h_best)
    for trial in range(5):
        if time.time() > d6:
            break
        h_rand = _random_binary(n_cur, rng)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_rand, c5_rand = _binary_sa(h_rand, sub_dl, max_iter=30000, T_init=0.003, seed=seed + 1000 + trial)
        if c5_rand < c5_best:
            c5_best = c5_rand
            h_best = h_rand.copy()
        # CD polish
        if c5_rand < c5_best + 0.005 and time.time() < d6:
            h_pol = _project(h_rand.copy(), n_cur / 2.0)
            h_pol, c5_pol = _coord_descent(h_pol, max_sweeps=60, deadline=d6)
            if c5_pol < c5_best:
                c5_best = c5_pol
                h_best = h_pol.copy()

    # Phase 7: Final intensive CD (10%)
    h_best, c5_best = _coord_descent(h_best, max_sweeps=500, deadline=deadline, recompute_every=20)

    h_best = _project(h_best, len(h_best) / 2.0)
    final_c5 = _exact_c5(h_best)
    if final_c5 < c5_best:
        c5_best = final_c5
    return h_best, c5_best, len(h_best)
