def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_G(h):
    """Compute gradient matrix G where G[l, r] = dS(l)/dh[r]"""
    n = len(h)
    u = 1.0 - h
    G = np.zeros((2*n-1, n))
    for d in range(n):
        # Positive lag d: S(d) = sum_{r=0}^{n-1-d} h[r]*(1-h[r+d])
        row = n - 1 + d
        m = n - d
        if m > 0:
            G[row, :m] += u[d:d+m]  # d/dh[r] of h[r]*u[r+d] = u[r+d]
            if d > 0:
                G[row, d:] += -h[:m]  # d/dh[r+d] of h[r]*u[r+d] = -h[r]

        # Negative lag -d: S(-d) = sum_{r=0}^{n-1-d} h[r+d]*(1-h[r])
