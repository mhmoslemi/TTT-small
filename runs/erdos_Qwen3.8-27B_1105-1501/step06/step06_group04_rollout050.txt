def _line_search(h, d, max_radius=0.5):

    dx = 2.0 / n

    pos = d > 1e-14

        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = min(t_lo, float(np.max(-h[pos] / d[pos])))
    if np.any(neg):
        t_lo = min(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_lo = max(t_lo, -max_radius)
    t_hi = min(t_hi, max_radius)
    if t_hi - t_lo < 1e-13:
        return 0.0, _compute_c5(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    def f_at(t):
        return float(np.max(S0 + t * T - t * t * C) * dx)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1, f2 = f_at(c1), f_at(c2)

            b, c2, f2 = c2, c1, f1

            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = f_at(c2)
    best_t, best_f = (a + b) / 2.0, f_at((a + b) / 2.0)
    for tc in [t_lo, t_hi]:
        fc = f_at(tc)
        if fc < best_f:
            best_f, best_t = fc, tc

def _lp_direction(h, scores, radius, top_k=20):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = np.zeros((K, n))
    for idx, r in enumerate(active):
        d = r - (n - 1)
        if d > 0:
            mo = n - d
            G[idx, d:] += (1.0 - h[:mo])
            G[idx, :mo] += -h[d:]
        elif d < 0:
            g2 = -d
            mo = n - g2
            G[idx, :mo] += (1.0 - h[g2:])
            G[idx, g2:] += -h[:mo]
        else:
            G[idx, :] = 1.0 - 2.0 * h
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G * (2.0 / n)
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-10:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

    # Phase 3: SLSQP minimax (10%)
    d3 = t0 + budget_s * 0.50
    if len(best_h) <= 200:
        h_mm, c5_mm = _slsqp_minimax(best_h, min(d3, t0 + budget_s * 0.08))
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Phase 4: Intensive line polish (20%)
    d4 = t0 + budget_s * 0.70
    h_lp, c5_lp = _line_polish(best_h, d4, best_c5, n_dirs_per_iter=10)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 5: Pair and quad refine (12%)
    d5 = t0 + budget_s * 0.82
    h_pr, c5_pr = _pair_refine(best_h, min(d5, t0 + budget_s * 0.60))
    if c5_pr < best_c5:
        best_c5 = c5_pr
        best_h = h_pr.copy()
    if time.time() < d5:
        h_qr, c5_qr = _quad_refine(best_h, d5)
        if c5_qr < best_c5:
            best_c5 = c5_qr
            best_h = h_qr.copy()

    # Phase 6: Resolution changes (10%)
    d6 = t0 + budget_s * 0.92
    best_n = len(best_h)
    offsets = [-6, 6, -10, 10, -14, 14, -20, 20, -30, 30, -40, 40]
    tried = set()
    for dn in offsets:
        if time.time() > d6 - 10:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 300 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.02)
        h_r2, c5_r2 = _line_polish(h_r, sub_dl, best_c5, n_dirs_per_iter=6)
        if c5_r2 < best_c5 - 1e-11:
            best_c5 = c5_r2
            best_h = h_r2.copy()
            best_n = n_try

    # Phase 7: Final polish (8%)
    d7 = hard_dl
    h_final, c5_final = _line_polish(best_h, d7, best_c5, n_dirs_per_iter=6)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 > 0.5:
        try:
            h_fb = np.array(initial_h_values, dtype=float)
            h_fb = _project(h_fb, len(h_fb) / 2.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
        except NameError:
            h_fb = _sine_init(120, 3, 0.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
