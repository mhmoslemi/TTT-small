def _slsqp_solve(h, dx, z_init, active_lags, max_constr=40, timeout=30):
    """Solve SLSQP subproblem with given active lags."""
    n = len(h)
    target = n / 2.0
    
    # Select top active_lags constraints (up to max_constr)
    overlaps = _overlaps(h)
    # active_lags is an array of indices into overlaps
    # Sort by overlap value descending, take top max_constr
    sorted_idx = np.argsort(-overlaps)[active_lags][:max_constr]
    
    def objective(x):
        return x[-1]  # minimize z
    
    def objective_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    constraints = []
    for r in sorted_idx:
        k = r - (n - 1)
        def _mk_cons(k=k, r=r):
            def cons(x):
                h_cand = x[:n]
                # Compute overlap at lag k
                if k == 0:
                    val = np.sum(h_cand * (1 - h_cand)) * dx
                elif k > 0:
                    val = np.sum(h_cand[k:] * (1 - h_cand[:-k])) * dx
                else:
                    q = -k
                    val = np.sum(h_cand[:-q] * (1 - h_cand[q:])) * dx
                return x[-1] - val  # z - O_k >= 0
            
            def cons_jac(x):
                h_cand = x[:n]
                g_h = _row_grad(h_cand, k, dx)
                j = np.zeros(len(x))
                j[:n] = -g_h  # derivative of (z - O_k) w.r.t. h is -grad(O_k)
                j[-1] = 1.0   # derivative w.r.t. z is 1
                return j
            
            return {'type': 'ineq', 'fun': cons, 'jac': cons_jac}
        constraints.append(_mk_cons())

    def eq_con(x):
        return np.sum(x[:n]) - target
    def eq_jac(x):
        j = np.zeros(len(x))
        j[:n] = 1.0
        return j
    constraints.append({'type': 'eq', 'fun': eq_con, 'jac': eq_jac})
    
    bounds = [(0, 1)] * n + [(best_f * 0.999, 1.0)]  # z bounds

        f_new = _exact_c5(h_new)
        return h_new, f_new
