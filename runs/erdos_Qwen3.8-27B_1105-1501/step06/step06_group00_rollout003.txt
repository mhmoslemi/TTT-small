```python
def _line_search(h, d, dx, radius):

            av = T[i] / (2.0 * C[i])
            if t_lo < av < t_hi:
                candidates.append(av)
    
    # Add pairwise intersections of top quadratics
    for i in range(len(idx)):
        for j in range(i+1, len(idx)):

            a2 = C[jj] - C[ii]
            a1 = T[ii] - T[jj]
            a0 = S0[jj] - S0[ii]
            if abs(a2) < 1e-14:
                if abs(a1) > 1e-14:
                    ai = -a0 / a1
                    if t_lo < ai < t_hi:
                        candidates.append(ai)
            else:
                disc = a1*a1 - 4.0*a2*a0
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for sign in [1, -1]:
                        ai = (-a1 + sign*sq) / (2.0*a2)
                        if t_lo < ai < t_hi:
                            candidates.append(ai)
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= t_lo) & (candidates <= t_hi)]
    if len(candidates) == 0:
        return 0.0, float(np.max(S0) * dx)

    best_val = float('inf')
    best_alpha = 0.0
    for a in candidates:
        val = float(np.max(S0 + a*T - a*a*C)) * dx
        if val < best_val:
            best_val = val
            best_alpha = float(a)
```

One important detail: the line search. I want it to be both accurate and fast. The key is to generate good candidate alphas.

For the quadratic form f(a) = max_k (S0[k] + a*T[k] - a^2*C[k]):

- Vertices: a = T[k] / (2*C[k]) for each k (where the quadratic for k is maximized)
- Intersections: where S0[i] + a*T[i] - a^2*C[i] = S0[j] + a*T[j] - a^2*C[j]

```python
def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible range
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi <= t_lo + 1e-13:
        return 0.0, float(np.max(S0) * dx)
    
    # Generate candidates
    cands = [t_lo, t_hi]
    # Grid
    cands.extend(np.linspace(t_lo, t_hi, 25).tolist())
    # Vertices of top quadratics
    top_k = min(8, len(S0))
    idx = np.argsort(S0)[-top_k:]
    for i in idx:
        if C[i] > 1e-12:
            av = T[i] / (2.0 * C[i])
            if t_lo < av < t_hi:
                cands.append(float(av))
    # Intersections of top 4
    top4 = idx[:4]
    for ii in range(len(top4)):
        for jj in range(ii+1, len(top4)):
            i, j = top4[ii], top4[jj]
            a2 = C[j] - C[i]
            a1 = T[i] - T[j]
            a0 = S0[j] - S0[i]
            if abs(a2) < 1e-12:
                if abs(a1) > 1e-12:
                    ai = -a0 / a1
                    if t_lo < ai < t_hi:
                        cands.append(float(ai))
            else:
                disc = a1*a1 - 4.0*a2*a0
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for s in [1.0, -1.0]:
                        ai = (-a1 + s*sq) / (2.0*a2)
                        if t_lo < ai < t_hi:
                            cands.append(float(ai))
    
    cands = np.array(cands)
    cands = cands[(cands >= t_lo) & (cands <= t_hi)]
    if len(cands) == 0:
        return 0.0, float(np.max(S0) * dx)
    
    # Vectorized evaluation
    # val[a] = max_k (S0[k] + a*T[k] - a^2*C[k])
    A = cands[:, None]  # (m, 1)
    vals = S0[None, :] + A * T[None, :] - (A*A) * C[None, :]
    max_vals = vals.max(axis=1) * dx
    
    best_idx = int(np.argmin(max_vals))
    return float(cands[best_idx]), float(max_vals[best_idx])
```
