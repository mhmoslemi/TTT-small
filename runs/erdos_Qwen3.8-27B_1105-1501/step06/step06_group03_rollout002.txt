    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    tau = tau_init
    iters = 0
    while time.time() < deadline:
        iters += 1
        if stall > 80:
            if restart_ctr < max_restarts:
                scale = 0.015 + 0.01 * restart_ctr * 0.4
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.10
                stall = 0
                restart_ctr += 1
                tau = max(0.005, tau * 0.7)
                continue
            else:
                break
        S = _corr_scaled(h)
        p = None
        if use_ls:
            p = _lp_direction(h, S, n, K, radius)
        if (p is None or float(np.max(np.abs(p))) < 1e-13) and use_smooth:
            p = _logsumexp_gradient(h, S, n, tau)
            if p is not None:
                p = p * radius
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if use_smooth:
                tau = max(0.003, tau * 0.85)
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.08, 0.30)
            elif ratio > 0.3:
                radius = min(radius * 1.03, 0.26)
            else:
                radius = max(radius * 0.92, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.06
    return best_h, best_c5

def _targeted_binary_moves(h, deadline, max_rounds=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    while time.time() < deadline and stall < 30:
        S = _corr_scaled(h)
        M = float(S.max())
        tol = 0.02 * M
        active = np.where(S >= M - tol)[0]
        ones_mask = h > 0.5
        zeros_mask = h <= 0.5
        if not np.any(ones_mask) or not np.any(zeros_mask):
            break
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        improved = False
        n_try = min(30, len(active))
        for kk in range(n_try):
            if time.time() > deadline:
                break
            r = active[kk]
            k = r - (n - 1)
            best_delta = 0.0
            best_i, best_j = -1, -1
            for i in ones_idx:
                if k >= 0:
                    if i + k < n:
                        val = 1.0 - h[i+k]
                        if val > best_delta:
                            best_delta = val
                            best_i, best_j = i, i + k
                else:
                    d = -k
                    if i - d >= 0:
                        val = 1.0 - h[i-d]
                        if val > best_delta:
                            best_delta = val
                            best_i, best_j = i, i - d
            if best_delta <= 0 or best_i < 0:
                continue
            d = np.zeros(n)
            d[best_i] = -1.0
            d[best_j] = 1.0
            d[d < 0] = -1.0
            d[d > 0] = 1.0
            alpha, c5_try = _line_search(h, d, dx, 0.4, n_grid=40)
            if c5_try < best_c5 - 1e-13:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    ones_mask = h > 0.5
                    zeros_mask = h <= 0.5
                    ones_idx = np.where(ones_mask)[0]
                    zeros_idx = np.where(zeros_mask)[0]
        if not improved:
            stall += 1
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=60):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        G = _grad_rows(h, np.arange(2*n-1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 40)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.3, n_grid=32)
                if c5_try < best_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.99
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick multi-start (8%)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 5]:
        for ph in [0.0, 1.0, 2.0]:
            starts.append(_sine_init(n0, freq, ph))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.007)
        h_s, c5_s = _opt_loop(s, sub_dl, radius_init=0.08, K=12, tau_init=0.02)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main LP optimization (35%)
    d2 = t0 + budget_s * 0.43
    h2, c5_2 = _opt_loop(best_h, d2, radius_init=0.14, K=20, tau_init=0.012,
                         use_smooth=True, use_ls=True, max_restarts=20)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Targeted binary moves (12%)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _targeted_binary_moves(best_h, d3, max_rounds=300)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3:
        h3b, c5_3b = _opt_loop(best_h, d3, radius_init=0.06, K=15, tau_init=0.008,
                               use_smooth=True, use_ls=True, max_restarts=8)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.75
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18, 22, -22, 26]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d4, time.time() + budget_s * 0.022)
            h_r, c5_r = _opt_loop(h_r, sub_dl, radius_init=0.10, K=14, tau_init=0.015)
            if c5_r < best_c5 and time.time() < d4 - budget_s * 0.02:
                h_r, c5_r = _targeted_binary_moves(h_r, min(d4, time.time() + budget_s * 0.012))
                if c5_r < best_c5:
                    h_r, c5_r = _opt_loop(h_r, min(d4, time.time() + budget_s * 0.01),
                                          radius_init=0.04, K=10, tau_init=0.008)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Pair refinement (10%)
    d5 = t0 + budget_s * 0.85
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=150)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Perturbation + re-opt (8%)
    d6 = t0 + budget_s * 0.94
    scale = 0.025
    for attempt in range(400):
        if time.time() > d6:
            break
        if best_c5 < 0.3806:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.004)
        h_t, c5_t = _opt_loop(h_pert, sub_dl, radius_init=0.05, K=12, tau_init=0.01)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 7: Final polish (6%)
    d7 = deadline
    h7, c5_7 = _opt_loop(best_h, d7, radius_init=0.03, K=15, tau_init=0.006,
                         use_smooth=True, use_ls=True, max_restarts=5)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < deadline:
        h7b, c5_7b = _pair_refine(best_h, deadline, max_passes=80)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
