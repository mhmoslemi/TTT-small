    ah = 1.0 - h
    corr = np.correlate(h, ah, mode='full') * dx
    M = float(np.max(corr))
    active = np.where(corr > M - tol)[0]
    if len(active) == 0:
        active = np.argsort(-corr)[:10]
    elif len(active) > 35:
        active = active[np.argsort(-corr[active])[:35]]
    grad = np.zeros(n)
    cnt = float(len(active))
    for idx in active:
        k = int(idx)
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            if m > 0:
                grad[:m] += ah[s:s + m]
                grad[s:s + m] += -h[:m]
        else:
            gg = -s
            m = n - gg
            if m > 0:
                grad[gg:gg + m] += ah[:m]
                grad[:m] += -h[gg:gg + m]
    grad *= dx / cnt
    return grad


def transport_polish(h, deadline, max_iters=15000, rng_seed=456):
    n = len(h)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(rng_seed)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        grad = avg_active_grad(h, tol=0.004)
        pos_idx = np.where(grad > 0.005)[0]
        neg_idx = np.where(grad < -0.005)[0]
        if len(pos_idx) == 0:
            pos_idx = np.argsort(grad)[-max(1, n // 6):]
        if len(neg_idx) == 0:
            neg_idx = np.argsort(grad)[:max(1, n // 6)]
        i = int(pos_idx[int(rng.integers(len(pos_idx)))])
        j = int(neg_idx[int(rng.integers(len(neg_idx)))])
        if i == j:
            stall += 1
            continue
        d = np.zeros(n)
        d[j] = 1.0
        d[i] = -1.0
        alpha, val = exact_line_search(h, d, max_alpha=0.4)
        if alpha > 1e-10 and val < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = project(h_new, n / 2.0)
            new_c5 = exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5 - 1e-15:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def pairwise_sweep(h, deadline, tol=0.003):
    """Try all promising pairs based on gradient, not just random ones."""
    n = len(h)
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    grad = avg_active_grad(h, tol=tol)
    pos_idx = np.where(grad > 0.003)[0]
    neg_idx = np.where(grad < -0.003)[0]
    if len(pos_idx) == 0 or len(neg_idx) == 0:
        return best_h, best_c5
    # Sort by magnitude to try most promising first
    pos_idx = pos_idx[np.argsort(-grad[pos_idx])][:min(15, len(pos_idx))]
    neg_idx = neg_idx[np.argsort(grad[neg_idx])][:min(15, len(neg_idx))]
    n_improvements = 0
    for i in pos_idx:
        if time.time() > deadline:
            break
        for j in neg_idx:
            if time.time() > deadline:
                break
            if i == j:
                continue
            d = np.zeros(n)
            d[j] = 1.0
            d[i] = -1.0
            alpha, val = exact_line_search(h, d, max_alpha=0.3)
            if alpha > 1e-10 and val < cur_c5 - 1e-14:
                h_new = h + alpha * d
                h_new = project(h_new, n / 2.0)
                new_c5 = exact_c5(h_new)
                if new_c5 < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = new_c5
                    n_improvements += 1
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    return best_h, best_c5


def resample_to_n(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)


def binary_perturb(h, rng, frac=0.3):
    """Push a fraction of values toward 0 or 1."""
    n = len(h)
    h2 = h.copy()
    n_perturb = max(1, int(frac * n))
    idx = rng.choice(n, size=n_perturb, replace=False)
    for i in idx:
        if h2[i] > 0.5:
            h2[i] = 1.0
        else:
            h2[i] = 0.0
    return project(h2, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except NameError:
        n_best = 122
        h_best = np.zeros(n_best)
        for i in range(n_best):
            h_best[i] = 1.0 if (i % 4 < 2) else 0.0
        h_best = project(h_best, n_best / 2.0)

    h_best = project(h_best, n_best / 2.0)
    best_c5 = exact_c5(h_best)

    # Phase 1: Soft-max descent (45% time)
    d1 = t0 + budget_s * 0.45
    h1, c5_1 = soft_max_descent(h_best, d1, total_iters=600, rng_seed=111)
    if c5_1 < best_c5:
        best_c5 = c5_1
        h_best = h1
        n_best = len(h_best)

    # Phase 2: Pairwise sweep for targeted improvements (10% time)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = pairwise_sweep(h_best, d2, tol=0.003)
    if c5_2 < best_c5:
        best_c5 = c5_2
        h_best = h2
        n_best = len(h2)

    # Phase 3: Random transport polish (15% time)
    d3 = t0 + budget_s * 0.70
    h3, c5_3 = transport_polish(h_best, d3, max_iters=12000, rng_seed=789)
    if c5_3 < best_c5:
        best_c5 = c5_3
        h_best = h3
        n_best = len(h3)

    # Phase 4: Binary perturbation + re-optimize (10% time)
    d4 = t0 + budget_s * 0.80
    for attempt in range(6):
        if time.time() > d4:
            break
        h_bp = binary_perturb(h_best, rng, frac=0.25)
        sub_dl = min(d4, time.time() + budget_s * 0.012)
        h_try, c5_try = soft_max_descent(h_bp, sub_dl, total_iters=60, rng_seed=200 + attempt)
        if c5_try < best_c5 - 1e-12:
            sub_dl2 = min(d4, time.time() + budget_s * 0.008)
            h_try, c5_try = transport_polish(h_try, sub_dl2, max_iters=3000, rng_seed=300 + attempt)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                h_best = h_try
                n_best = len(h_best)

    # Phase 5: Try nearby n values (10% time)
    d5 = t0 + budget_s * 0.90
    for dn in [-6, 6, -4, 4, -8, 8, -10, 10]:
        if time.time() > d5:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try % 2 != 0:
            continue
        h_r = resample_to_n(h_best, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = soft_max_descent(h_r, sub_dl, total_iters=80, rng_seed=500 + dn)
        if c5_r < best_c5 - 1e-12:
            sub_dl2 = min(d5, time.time() + budget_s * 0.01)
            h_r, c5_r = transport_polish(h_r, sub_dl2, max_iters=4000, rng_seed=600 + dn)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                h_best = h_r
                n_best = n_try

    # Phase 6: Final high-gamma polish (10% time)
    d6 = t0 + budget_s * 0.97
    h6, c5_6 = soft_max_descent(h_best, d6, total_iters=200, rng_seed=999)
    if c5_6 < best_c5:
        best_c5 = c5_6
        h_best = h6
        n_best = len(h6)

    # Final pairwise sweep
    if time.time() < deadline - 5:
        h7, c5_7 = pairwise_sweep(h_best, deadline - 2, tol=0.002)
        if c5_7 < best_c5:
            best_c5 = c5_7
            h_best = h7
            n_best = len(h_best)

    # Final validation
    h_best = project(h_best, len(h_best) / 2.0)
    final_c5 = exact_c5(h_best)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return h_best, best_c5, len(h_best)
