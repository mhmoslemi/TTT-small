        second = -h[1:]
        for s in range(n):
            m = n - 1 - s
            wseg = wt - (wk[m:].sum() if m < n - 1 else 0.0)
            if m > 0 and wseg > 1e-15:
                g_val[s] = wseg * first[s] + wt * second[s]
    pos_Ds = np.arange(1, n)
    D_s = pos_Ds
    rows2 = n - 1 + D_s
    wk2 = w[rows2]
    wt2 = wk2.sum()
    if wt2 > 1e-15:
        first = (1.0 - h[1:])
        second = -h[:-1]
        for s in range(n):
            m = n - 1 - s
            wseg = wt2 - (wk2[m:].sum() if m < n - 1 else 0.0)
            if m > 0 and wseg > 1e-15:
                g_val[s] += wseg * first[s] + wt2 * second[s]
    agg = w[n - 1]
    if agg > 1e-15:
        g_val += agg * (1.0 - 2.0 * h)
    g_val *= dx
    return S, g_val, float(np.max(vs))

def _compute_T(h, i, j):
    """Compute T[r] = G[r,i] - G[r,j] for all shift rows."""
    n = len(h)
    T = np.zeros(2 * n - 1)
    for D in range(1, n):
        ri = 0.0
        rj = 0.0
        if i < n - D:
            ri += 1.0 - h[i + D]
        if i >= D:
            ri -= h[i - D]
        if j < n - D:
            rj += 1.0 - h[j + D]
        if j >= D:
            rj -= h[j - D]
        T[n - 1 + D] = ri - rj
        T[n - 1 - D] = rj - ri
        if D == 0:
            pass
    T[n - 1] = (1.0 - 2.0 * h[i]) - (1.0 - 2.0 * h[j])
    return T

def _compute_Qc(h, i, j):
    """Compute Qc[r] (quadratic coefficient) for direction e_i - e_j."""
    n = len(h)
    Qc = np.zeros(2 * n - 1)
    d = i - j
    if d > 0:
        Qc[n - 1] = -2.0
        if j < n - d:
            Qc[n - 1 + d] = -1.0
        if j >= d:
            Qc[n - 1 - d] = -1.0
    else:
        dd = -d
        Qc[n - 1] = -2.0
        if i < n - dd:
            Qc[n - 1 + dd] = 1.0
        if i >= dd:
            Qc[n - 1 - dd] = 1.0
    return Qc

def _pair_optimize(h, deadline, gamma, K, radius, max_iters=50000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        S, g, gmax = _gradient_and_weights(h, cur_c5, gamma)
        score = -(g)
        candidates = []
        for _ in range(200):
            i = int(np.random.randint(0, n))
            j = int(np.random.randint(0, n))
            if i != j:
                adj = (score[i] - score[j]) * (1.0 - abs(h[i] - h[j]))
                if h[i] < 1.0 - 1e-9 and h[j] > 1e-9:
                    candidates.append((adj, i, j))
        candidates.sort(key=lambda x: x[0], reverse=True)
        K_actual = min(K, len(candidates))
        if K_actual == 0:
            best_i, best_j = 0, 1
            if h[best_i] < 1e-9 or h[best_j] < 1e-9:
                stall += 10
                continue
        else:
            idx = int(np.random.choice(K_actual, p=np.array([max(c[0],0.001) for c in candidates[:K_actual]]) / sum(max(c[0],0.001) for c in candidates[:K_actual])))
            best_i, best_j = candidates[idx][1], candidates[idx][2]
        i, j = best_i, best_j
        a_lo = max(-radius, -h[i], h[j] - 1.0)
        a_hi = min(radius, 1.0 - h[i], h[j])
        if a_hi - a_lo < 1e-12:
            stall += 5
            continue
        T = _compute_T(h, i, j)
        Qc = _compute_Qc(h, i, j)
        a_star, f_star = _pair_line_search(h, i, j, S, T, Qc, dx, a_lo, a_hi)
        if f_star < cur_c5 - 1e-12:
            h[i] += a_star
            h[j] -= a_star
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5, best_h = cur_c5, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _binary_pair_optimize(h_bin, deadline, max_iters=100000):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    stall = 0
    rng = np.random.default_rng(999)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 500:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        c5_new = _exact_c5(cur)
        if c5_new < cur_c5 - 1e-14:
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5, best_h = c5_new, cur.copy()
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            stall = 0
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    freqs = rng.uniform(0.5, 8.0, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.4, 1.0, size=4)
    p = np.zeros(n)
    for f, ph, a in zip(freqs, phases, amps):
        p += a * np.sin(2.0 * np.pi * f * x / n + ph)
    p *= amp
    p += amp * 0.3 * rng.normal(0, 1, n)
    return _project(h + p, n / 2.0)

def _binary_sa(h_bin, deadline, T_init=0.003, max_iter=50000, seed=42):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    rng = np.random.default_rng(seed)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 800:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        c5_new = _exact_c5(cur)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, cur.copy()
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            stall = 0
            T *= 0.999
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Pair optimization on parent (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _pair_optimize(h_init, d1, gamma=80.0, K=8, radius=0.3, max_iters=200000)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Binary exploration (15%)
    d2 = t0 + budget_s * 0.40
    h_bin = _to_binary(best_h)
    h_bin, c5_bin = _binary_sa(h_bin, min(d2, time.time() + budget_s * 0.05), T_init=0.003, seed=77)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin
    sub_dl = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _binary_pair_optimize(h_bin, sub_dl)
    if c5_bin < best_c5:
        best_c5, best_h = c5_bin, h_bin
    # LP-style polish from binary
    if time.time() < d2:
        h_pol = _project(h_bin.copy(), n0 / 2.0)
        sub_dl2 = min(d2, time.time() + budget_s * 0.03)
        h_pol, c5_pol = _pair_optimize(h_pol, sub_dl2, gamma=100.0, K=10, radius=0.2, max_iters=80000)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.65
    n_curr = len(best_h)
    tried_ns = set([n_curr])
    offset_list = list(range(-30, 31, 2))
    for dn in sorted(offset_list, key=abs):
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 240 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _pair_optimize(h_r, sub_dl, gamma=80.0, K=6, radius=0.25, max_iters=60000)
        if time.time() < d3 - 3:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_pair_optimize(h_b, sub_dl2, max_iter=20000)
            if c5_b < c5_r:
                c5_r, h_r = c5_b, h_b
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r
            n_curr = n_try

    # Phase 4: Intensive refinement (20%)
    d4 = t0 + budget_s * 0.85
    n_curr = len(best_h)
    # Long pair optimization
    sub_dl = min(d4, time.time() + budget_s * 0.08)
    h4 = _project(best_h.copy(), n_curr / 2.0)
    h4, c5_4 = _pair_optimize(h4, sub_dl, gamma=120.0, K=12, radius=0.4, max_iters=200000)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4
    # Binary round
    if time.time() < d4:
        h_b = _to_binary(best_h)
        sub_dl = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl, T_init=0.002, seed=314)
        sub_dl2 = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _binary_pair_optimize(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b
    # Final pair polish
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_curr / 2.0)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _pair_optimize(h4, sub_dl, gamma=150.0, K=15, radius=0.15, max_iters=100000)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Phase 5: Basin hopping + final polish (10%)
    d5 = deadline
    scale = 0.03
    n_restarts = 0
    while time.time() < d5 - 5 and n_restarts < 60:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.006)
        h_t, c5_t = _pair_optimize(h_pert, sub_dl, gamma=100.0, K=8, radius=0.2, max_iters=40000)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t
            scale = max(0.005, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.08)

    # Final: try a few nearby n values
    n_best = len(best_h)
    for dn in [-2, 2, -4, 4, -6, 6]:
        if time.time() > deadline - 5:
            break
        n_try = n_best + dn
        if n_try < 50 or n_try > 260:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline - 5, time.time() + budget_s * 0.006)
        h_r, c5_r = _pair_optimize(h_r, sub_dl, gamma=100.0, K=8, radius=0.2, max_iters=30000)
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _pair_optimize(h6, deadline, gamma=200.0, K=15, radius=0.1, max_iters=100000)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
