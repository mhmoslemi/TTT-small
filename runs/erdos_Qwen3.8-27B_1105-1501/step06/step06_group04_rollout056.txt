def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target} using bisection on threshold."""
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=0.0, posinf=2.0, neginf=-2.0), -10.0, 10.0)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_optimize(h_init, beta, target, maxiter=200):
    n = len(h_init)
    
    def fun_and_grad(x):
        # x is h values
        C = _corr_scaled(x)
        maxC = C.max()
        log_sum = maxC + np.log(np.sum(np.exp(beta * (C - maxC))))
        obj = log_sum / beta
        p = np.exp(beta * (C - maxC))
        p /= p.sum()
        G = _full_grad_matrix(x)
        grad = p @ G  # weighted sum of gradient rows
        return obj, grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 
                    'jac': lambda x: np.ones(len(x))}]
    bounds = [(0, 1)] * n
    
    res = minimize(fun_and_grad, h_init, jac=True, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-14})

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _full_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d == 0:
            G[r] = (1.0 - 2.0 * h) * dx
        elif d > 0:
            mm = n - 1 - d
            G[r, :mm + 1] += (1.0 - h[d:mm + 1 + d]) * dx
            G[r, d:] += -h[:mm + 1] * dx
        else:
            gg = -d
            G[r, gg:] += (1.0 - h[:n - gg]) * dx
            G[r, :n - gg + 1] += -h[gg:] * dx
    return G

def _project(v, target):
    n = len(v)
    v = np.nan_to_num(v, nan=0.5, posinf=2.0, neginf=-2.0)
    v = np.clip(v, -5.0, 5.0)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt

            ia, ja = interior[i], interior[j]
            for sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[ia] = -sign
                d[ja] = sign
                t, c5_try = _golden_line(h, d, 0.5)
                if c5_try < best_c5 - 1e-13:
                    h_new = _project(h + t * d, target)
                    c5_actual = _exact_c5(h_new)
                    if c5_actual < best_c5:
                        h = h_new
                        best_c5 = c5_actual
                        best_h = h.copy()
                        improved = True
                        interior = np.where((h > 0.01) & (h < 0.99))[0]
                        if len(interior) < 2:
                            break
                        break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return _project(h, n / 2.0)

def _random_init(n, rng):
    h = rng.random(n)
    return _project(h, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 4))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def _multi_start_optimize(n, deadline, best_h, best_c5, rng):
    starts = []
    for freq in [1, 2, 3, 4, 5, 7, 11, 13]:
        for ph in [0.0, 0.7, 1.5, 2.5, 3.8]:
            starts.append(_sine_init(n, freq, ph))
    for p in [3, 4, 5, 6, 7, 8, 10, 12, 15, 20]:
        for duty in [0.3, 0.35, 0.4, 0.5, 0.6, 0.65, 0.7]:
            starts.append(_periodic_init(n, p, duty))
    for _ in range(6):
        starts.append(_random_init(n, rng))
    for _ in range(6):
        starts.append(_block_init(n, rng))
    starts.append(best_h.copy())

    for s in starts:
        if time.time() > deadline:
            break
        h_try = s
        c5_try = _exact_c5(h_try)
        if c5_try >= best_c5 + 0.005:
            continue
        sub_dl = min(deadline, time.time() + 8.0)
        for beta in [30, 80, 200]:
            if time.time() > sub_dl:
                break
            h_try, c5_try = _slsqp_lse(h_try, beta, n / 2.0, maxiter=80)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def _subgrad_loop(h, target, deadline, base_step=0.08, cool=0.999):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = base_step
    stale = 0
    while time.time() < deadline:
        h_new, c5_new = _neg_subgrad_line(h, target, radius=step)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stale = 0
            step = min(step * 1.03, 0.3)
        else:
            stale += 1
            step *= cool
        if stale > 60:
            break
    return best_h, best_c5

def _nesterov_subgrad(h_init, target, deadline, lr_init=0.05):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    mu = 0.5
    stale = 0
    while time.time() < deadline:
        y = h + mu * v
        C = _corr_scaled(y)
        maxC = C.max()
        tol = max(1e-5, 0.01 * maxC)
        active = C >= maxC - tol
        G = _full_grad_matrix(y)
        grad = G[active].mean(axis=0)
        v = mu * v - lr * grad
        h_new = _project(h + v, target)
        h_new -= h_new.mean() - (target / n)
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stale = 0
        else:
            stale += 1
        h = h_new
        mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        lr *= 0.998
        if stale > 80:
            break
    return best_h, best_c5

def _coord_descent(h, target, deadline):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(77)
    order = np.arange(n)
    stalls = 0
    while time.time() < deadline and stalls < 30:
        np.random.default_rng(stalls).shuffle(order)
        improved = False
        for idx in order:
            if time.time() > deadline:
                return best_h, best_c5
            if h[idx] <= 1e-8 or h[idx] >= 1.0 - 1e-8:
                continue
            for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
                h_try = h.copy()
                h_try[idx] += delta
                if h_try[idx] < 0 or h_try[idx] > 1:
                    continue
                h_try = _project(h_try, target)
                c5_try = _exact_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                    break
        if improved:
            stalls = 0
        else:
            stalls += 1
            break
    return best_h, best_c5

def _multi_resolution(best_h, best_c5, deadline):
    base_n = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, -24, 24, -30, 30, -40, 40]
    tried = set()
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = base_n + dn
        if n_try < 30 or n_try > 280 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(deadline, time.time() + 12.0)
        for beta in [50, 150]:
            if time.time() > sub_dl:
                break
            h_r, c5_r = _slsqp_lse(h_r, beta, n_try / 2.0, maxiter=100)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def _perturb_reopt(best_h, best_c5, deadline, rng):
    n = len(best_h)
    scale = 0.04
    for _ in range(200):
        if time.time() > deadline:
            break
        h_p = _project(best_h + scale * rng.standard_normal(n), n / 2.0)
        sub_dl = min(deadline, time.time() + 5.0)
        h_try, c5_try = _slsqp_lse(h_p, 100, n / 2.0, maxiter=60)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.01, scale * 0.7)
        else:
            scale = min(0.15, scale * 1.3)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s - 5.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = _sine_init(120, 3, 0.5)
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)

    # Phase 1: LSE SLSQP from parent with increasing beta
    d1 = t0 + budget_s * 0.15
    h = best_h.copy()
    for beta in [10, 20, 50, 100, 200, 400]:
        if time.time() > d1:
            break
        h, c5 = _slsqp_lse(h, beta, n0 / 2.0, maxiter=120)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: LSE descent continuation
    d2 = t0 + budget_s * 0.28
    h2, c5_2 = _lse_descent(best_h, n0 / 2.0, [300, 600, 1200], 300, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Subgradient with line search
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _subgrad_loop(best_h, n0 / 2.0, d3, base_step=0.1, cool=0.998)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Nesterov accelerated subgradient
    d4 = t0 + budget_s * 0.50
    h4, c5_4 = _nesterov_subgrad(best_h, n0 / 2.0, d4, lr_init=0.04)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-start
    d5 = t0 + budget_s * 0.62
    n = len(best_h)
    best_h, best_c5 = _multi_start_optimize(n, d5, best_h, best_c5, rng)

    # Phase 6: Multi-resolution
    d6 = t0 + budget_s * 0.74
    best_h, best_c5 = _multi_resolution(best_h, best_c5, d6)

    # Phase 7: Perturbation + re-optimize
    d7 = t0 + budget_s * 0.82
    best_h, best_c5 = _perturb_reopt(best_h, best_c5, d7, rng)

    # Phase 8: Pair refinement
    d8 = t0 + budget_s * 0.90
    n = len(best_h)
    h8, c5_8 = _pair_refine(best_h, n / 2.0, d8, max_rounds=300)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Phase 9: Coordinate descent
    d9 = t0 + budget_s * 0.95
    h9, c5_9 = _coord_descent(best_h, len(best_h) / 2.0, d9)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    # Phase 10: Final LSE polish
    h10, c5_10 = _slsqp_lse(best_h, 500, len(best_h) / 2.0, maxiter=200)
    if c5_10 < best_c5:
        best_c5 = c5_10
        best_h = h10.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h_fb = np.array(initial_h_values, dtype=float)
            h_fb = _project(h_fb, len(h_fb) / 2.0)
            return h_fb, _exact_c5(h_fb), len(h_fb)
        except NameError:
            h_fb = _sine_init(120, 3, 0.0)
            return h_fb, _exact_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
