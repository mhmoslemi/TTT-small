def _row_grad(h, k, dx):

def _line_search_1d(h, d, corr, dx, n):
    """Line search: minimize max_ell (corr[ell] + t*dT[ell] - t^2*dQ[ell]) * dx over t in [L, U]."""
    # T[ell] = linear coefficient, Q[ell] = quadratic coefficient
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Bounds on t
    pos = d > 1e-15
    neg = d < -1e-15
    L = 0.0
    U = 1e10
    if np.any(pos):
        L = max(L, -float(np.min(h[pos] / d[pos])))
        U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        L = max(L, float(np.min((1.0 - h[neg]) / d[neg])))
        U = min(U, -float(np.min(h[neg] / d[neg])))
    
    if U - L < 1e-12:
        return None
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    
    def f_of_t(t):
        return float(np.max(corr + t * T - t * t * Q)) * dx
    
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_of_t(c1)
    f2 = f_of_t(c2)
    
    for _ in range(60):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f_of_t(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_of_t(c2)
    
    t = 0.5 * (a + b)
    return t
