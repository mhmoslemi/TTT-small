def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 3.0
    hi = float(v.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[q, k:] += dx * (1.0 - h[:m])
                G[q, :m] += -dx * h[k:]
        else:
            d_abs = -k
            m = n - d_abs
            if m > 0:
                G[q, :m] += dx * (1.0 - h[d_abs:])
                G[q, d_abs:] += -dx * h[:m]
    return G

def _line_search_quad(h, d, dx, radius):

    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx

        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))

        return 0.0, float(np.max(S0))
    n_grid = 60

    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1)
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]

def _lp_direction(h, G, S, radius):
    n = len(h)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = -S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _adam_descent(h, deadline, tau_init=0.005, lr_init=0.01):

    target = n / 2.0

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    tau = tau_init
    lr = lr_init
    stall = 0
    for it in range(5000):
        if time.time() > deadline:
            break
        S = _corr_vals(h)
        S_max = float(np.max(S))
        w = np.exp((S - S_max) / tau)
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(S))] = 1.0
        else:
            w /= w_sum
        G = _build_G(h, dx)
        grad = G.T @ w
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = _project(h_new, target)
        cur_c5 = _exact_c5(h_new)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 0.05)
        else:
            h = h_new
            stall += 1
            lr *= 0.95
        if it % 100 == 0:
            tau *= 0.96
        if stall > 80:
            h = best_h + np.random.default_rng(it).normal(0, 0.03, n)
            h = _project(h, target)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init
            stall = 0
    return best_h, best_c5

            signs = rng.choice([-1.0, 1.0], size=3)
            d = np.zeros(n)
            d[k1] = signs[0]
            d[k2] = signs[1]
            d[k3] = signs[2]
            total = d.sum()
            if abs(total) < 1e-10:
                continue
            d = d - total / n
            h_new = _project(h + 0.3 * d, target)
        c5_new = _exact_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_new.copy()
    return best_h, best_c5

def _pair_optimize(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(314)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_vals(h)
        G = _build_G(h, dx)
        max_idx = np.argmax(S)
        grad = G[max_idx]
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(len(order), 60)):
            if time.time() > deadline:
                break
            i = order[ii]
            for jj in range(min(len(order), 60)):
                if tried > 120:
                    break
                j = order[jj]
                if i == j:
                    continue
                if h[i] <= 0.005 or h[j] >= 0.995:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search_quad(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-15:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                tried += 1
        if not improved:
            for _ in range(30):
                if time.time() > deadline:
                    break
                if len(interior) < 2:
                    break
                ii = rng.integers(len(interior))
                jj = rng.integers(len(interior))
                if ii == jj:
                    continue
                i, j = interior[ii], interior[jj]
                for sign in [1, -1]:
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    alpha, c5_try = _line_search_quad(h, d, dx, 0.4)
                    if c5_try < best_c5 - 1e-15:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved = True
            if not improved:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_init(n, seed):
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * rng.integers(2, 7) * x / n + rng.uniform(0, 6.28))
    h = h + 0.3 * np.sin(2 * np.pi * rng.integers(1, 5) * x / n + rng.uniform(0, 6.28))
    h = h + rng.normal(0, 0.08, n)
    return _project(h, n / 2.0)

def _basin_hop(h, deadline, n_hops=10, rng=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    if rng is None:
        rng = np.random.default_rng(7777)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_h = h.copy()
    for hop in range(n_hops):
        if time.time() > deadline:
            break
        noise_scale = 0.03 + 0.02 * (1 - hop / max(n_hops, 1))
        perturbed = cur_h + rng.normal(0, noise_scale, n)
        perturbed = _project(perturbed, target)
        sub_dl = min(deadline, time.time() + max(2.0, (deadline - time.time()) / max(n_hops - hop, 1)))
        h_local, c5_local = _lp_optimize(perturbed, sub_dl, start_radius=0.08)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_local.copy()
            cur_h = h_local.copy()
        elif c5_local < _exact_c5(cur_h):
            cur_h = h_local.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x / n0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: LP optimization from parent (35% budget)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _lp_optimize(best_h, d1, start_radius=0.15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Adam smooth descent (15% budget)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _adam_descent(best_h, d2, tau_init=0.004, lr_init=0.015)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Diverse starts with LP (12% budget)
    d3 = t0 + budget_s * 0.62
    n_starts = 4
    for si in range(n_starts):
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + max(5.0, (d3 - time.time()) / max(n_starts - si, 1)))
        h_s = _diverse_init(n0, seed + 100 + si * 7)
        h_s, c5_s = _lp_optimize(h_s, sub_dl, start_radius=0.12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 4: Basin hopping (15% budget)
    d4 = t0 + budget_s * 0.77
    h4, c5_4 = _basin_hop(best_h, d4, n_hops=12, rng=rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Simulated annealing with block moves (8% budget)
    d5 = t0 + budget_s * 0.85
    h5, c5_5 = _block_sa(best_h, d5, t_init=0.0015, t_min=5e-5, rng=rng)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (8% budget)
    d6 = deadline - budget_s * 0.05
    n_best = len(best_h)
    offsets = [-6, 6, -4, 4, -8, 8, -2, 2, -10, 10, -12, 12, -16, 16, -20, 20]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        sub_dl = min(d6, time.time() + max(3.0, budget_s * 0.01))
        h_r = _resample(best_h, n_try)
        h_r, c5_r = _lp_optimize(h_r, sub_dl, start_radius=0.10)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    # Phase 7: Final LP polish (4% budget)
    d7 = deadline
    h7, c5_7 = _lp_optimize(best_h, d7, start_radius=0.06)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Final pair polish if time remains
    if time.time() < deadline - 2.0:
        h8, c5_8 = _pair_optimize(best_h, deadline, max_passes=40)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
