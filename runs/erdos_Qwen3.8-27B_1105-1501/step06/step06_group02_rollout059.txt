def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):

        m = n - abs(d)

            G[k, :m] += -h[d:d + m]
        else:
            g = -d
            G[k, :m] += (1.0 - h[g:g + m])
            G[k, g:g + m] += -h[:m]

            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def _try_direction(h, d, S0, dx, target):
    n = len(h)
    d = d - np.mean(d)
    pos = d > 1e-10
    neg = d < -1e-10
    lo, hi = 0.0, 0.5
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, -float(np.min(h[neg] / (-d[neg]))))
    if hi - lo < 1e-13:
        return None, None
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    opt, val = _line_search(S0, T, C, dx, lo, hi)
    if opt < 1e-12:
        return None, None
    h_new = np.clip(h + opt * d, 0.0, 1.0)
    h_new = project(h_new, target)
    new_c5 = c5_exact(h_new)
    return h_new, new_c5

def _grad_optimize(h, deadline, temp_init=0.05, temp_min=0.003, adapt_temp=True):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, 'full')
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    temp = temp_init
    stall = 0
    improvement_count = 0

    while time.time() < deadline and stall < 80:
        G = _grad_matrix(h)
        max_raw = float(np.max(S0))
        temp_eff = temp * max_raw + 1e-10
        weights = np.exp((S0 - max_raw) / temp_eff)
        wsum = float(np.sum(weights))
        if wsum < 1e-30:
            break
        weights /= wsum
        g = G.T @ weights
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        result = _try_direction(h, d, S0, dx, target)
        if result[0] is not None:
            h_new, new_c5 = result
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                S0 = np.correlate(h, 1.0 - h, 'full')
                stall = 0
                improvement_count += 1
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                if adapt_temp:
                    temp = max(temp_min, temp * 0.97)
                continue
        stall += 1
        if stall % 15 == 0:
            temp = min(temp_init, temp * 1.5)
            if stall > 40:
                break

    return best_h, best_c5

def _pair_optimize(h, deadline, n_pairs=12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, 'full')
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(999)

    while time.time() < deadline and stall < 100:
        max_raw = float(np.max(S0))
        threshold = max_raw - 0.008 * max_raw
        active = np.where(S0 >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-S0)[:5]
        G = _grad_matrix(h)
        active_scores = S0[active]
        temp = (float(np.max(active_scores)) - float(np.min(active_scores))) * 0.5 + 0.01 * max_raw + 1e-8
        w = np.exp((active_scores - float(np.max(active_scores))) / temp)
        w /= float(np.sum(w))
        g = G[active].T @ w
        g -= np.mean(g)
        donors = np.argsort(-g)[:n_pairs]
        receivers = np.argsort(g)[:n_pairs]
        best_alpha = 0.0
        best_val = cur_c5
        best_pair = None
        for i in donors:
            if time.time() > deadline:
                break
            for j in receivers:
                if i == j:
                    continue
                lo_a = -min(h[j], 1.0 - h[i])
                hi_a = min(h[i], 1.0 - h[j])
                if hi_a - lo_a < 1e-14:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                C = np.correlate(d, d, 'full')
                opt, val = _line_search(S0, T, C, dx, lo_a, hi_a)
                if val < best_val - 1e-14:
                    best_val = val
                    best_alpha = opt
                    best_pair = (i, j)
        if best_pair is not None and best_alpha > 1e-10:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, 'full')
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1
            if stall > 25:
                idx = rng.integers(n, size=3)
                for k in idx:
                    h[k] += rng.uniform(-0.04, 0.04)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, 'full')
                stall = 0

    return best_h, best_c5

def _multi_direction_optimize(h, deadline):
    """Try the gradient direction plus several perturbed directions."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, 'full')
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(555)
    max_iters = 0

    while time.time() < deadline and stall < 60 and max_iters < 500:
        max_iters += 1
        G = _grad_matrix(h)
        max_raw = float(np.max(S0))
        temp = 0.01 * max_raw + 1e-8
        weights = np.exp((S0 - max_raw) / temp)
        weights /= float(np.sum(weights))
        g = G.T @ weights
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d_base = -g / gn
        best_result = None
        for trial in range(5):
            if trial == 0:
                d = d_base.copy()
            else:
                d = d_base + 0.1 * rng.standard_normal(n)
                d -= np.mean(d)
                d /= np.linalg.norm(d)
            result = _try_direction(h, d, S0, dx, target)
            if result[0] is not None:
                h_new, new_c5 = result
                if best_result is None or new_c5 < best_result[1]:
                    best_result = (h_new, new_c5)
        if best_result is not None and best_result[1] < cur_c5 - 1e-14:
            h = best_result[0]
            cur_c5 = best_result[1]
            S0 = np.correlate(h, 1.0 - h, 'full')
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stall += 1

    return best_h, best_c5

def _binary_optimize(h, deadline, rng):
    n = len(h)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    cur_c5 = c5_exact(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin <= 0.5)[0]
    patience = 0
    stall = 0

    while time.time() < deadline and patience < 6000:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = c5_exact(h_bin)
        if new_c5 < cur_c5 - 1e-15:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
                patience = 0
                stall = 0
            else:
                patience += 1
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin <= 0.5)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            patience += 1
            stall += 1
        if stall > 200:
            stall = 0

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _sine_init(n, freq=2, phase=0.5):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)

def _qp_global_step(h, deadline):
    """Try a QP-motivated global direction using the full gradient structure."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, 'full')
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    G = _grad_matrix(h)
    max_raw = float(np.max(S0))

    # Try several target levels
    for frac in [0.999, 0.998, 0.996, 0.993]:
        target_raw = max_raw * frac - 0.3
        # Find direction d that minimizes max_k(S0[k] + G[k]@d) subject to ||d||<=rho
        # Approximate with: d = -G_active.T @ w / ||G_active.T @ w|| where w = active weights
        # But use ALL rows weighted by violation
        viol = np.maximum(S0 - target_raw, 0.0)
        total_viol = float(np.sum(viol))
        if total_viol < 1e-10:
            continue
        w = viol / total_viol
        g = G.T @ w
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            continue
        d = -g / gn
        # Cap the step
        rho = 0.3
        d = np.clip(d, -rho, rho)
        d -= np.mean(d)
        result = _try_direction(h, d, S0, dx, target)
        if result[0] is not None:
            h_new, new_c5 = result
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h_new.copy()

    return best_h, best_c5

def _adaptive_temp_optimize(h, deadline):
    """Optimize with geometrically decreasing temperature schedule."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, 'full')
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    temps = [0.08, 0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003, 0.0015]
    stall_total = 0

    for temp in temps:
        if time.time() > deadline:
            break
        if stall_total > 200:
            break
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.12)
        h, c5 = _grad_optimize(h, sub_dl, temp_init=temp, temp_min=temp * 0.5, adapt_temp=False)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall_total = 0
        else:
            stall_total += 1

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0)

    h = project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)

    # Phase 1: Adaptive temp gradient optimization (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _adaptive_temp_optimize(h, d1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Pair optimization (15% time)
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _pair_optimize(best_h, d2)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Multi-direction optimization (10% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _multi_direction_optimize(best_h, d3)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()

    # Phase 4: QP global step (5% time)
    d4 = t0 + budget_s * 0.50
    h4, c5_4 = _qp_global_step(best_h, d4)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Multi-resolution search (30% time)
    d5 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    tried = {n_curr}
    offsets = list(range(-20, 21, 2))
    # Sort by absolute value to try nearby resolutions first
    offsets.sort(key=lambda x: abs(x))
    time_per_n = max(5.0, budget_s * 0.02)
    for dn in offsets:
        if time.time() > d5 - 10:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 240 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + time_per_n)
        h_r, c5_r = _grad_optimize(h_r, sub_dl, temp_init=0.03, temp_min=0.005)
        if time.time() < d5 - 5:
            sub_dl2 = min(d5, time.time() + time_per_n * 0.5)
            h_r, c5_r2 = _pair_optimize(h_r, sub_dl2, n_pairs=8)
            if c5_r2 < c5_r:
                c5_r = c5_r2
        if c5_r < best_c5:
            best_c5, best_h = c5_r, h_r.copy()
            n_curr = n_try
            tried.add(n_curr)

    # Phase 6: Binary optimization (8% time)
    d6 = t0 + budget_s * 0.88
    h6, c5_6 = _binary_optimize(best_h, d6, rng)
    if c5_6 < best_c5:
        best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final gradient polish (6% time)
    d7 = deadline
    h7, c5_7 = _grad_optimize(best_h, d7, temp_init=0.01, temp_min=0
