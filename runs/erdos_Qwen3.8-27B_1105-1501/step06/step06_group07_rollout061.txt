        else:

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    residual = target - float(np.sum(x))
    if abs(residual) > 1e-13:
        if residual > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_S(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _line_search(h, d, dx, S, max_t=0.5):
    n = len(h)
    g = 1.0 - h
    B = np.correlate(d, g, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    t_hi = max_t
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if t_hi < 1e-14:
        return 0.0, float(np.max(S)) * dx
    
    def _f(t):
        return float(np.max(S + t * B - t * t * C)) * dx
    
    n_g = 40
    grid = np.linspace(0.0, t_hi, n_g)
    fvals = np.array([_f(t) for t in grid])

def _grad_active(h, S, K, alpha):
    n = len(h)
    K = min(K, n - 1)
    top = np.argsort(-S)[:K]
    top_v = S[top]
    w = np.exp(alpha * (top_v - top_v[0]))
    w /= w.sum()
    
    grad = np.zeros(n)
