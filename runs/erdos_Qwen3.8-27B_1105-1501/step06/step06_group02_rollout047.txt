import numpy as np
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_matrix(h):
    """Compute gradient matrix for all lags. Shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        centered = q - (n - 1)
        m = n - abs(centered)
        if m <= 0: continue
        if centered >= 0:
            # f_q = sum_{i=0}^{m-1} h[i+centered] * (1 - h[i])
            d_h_high = 1.0 - h[:m]  # d/d h[i+centered]
            d_h_low = -h[centered:centered+m]  # d/d h[i]
            G[q, centered:centered+m] += d_h_high
            G[q, :m] += d_h_low
        else:
            r = -centered
            # f_q = sum_{i=0}^{m-1} h[i] * (1 - h[i+r])
            d_h_low = 1.0 - h[r:r+m]  # d/d h[i]
            d_h_high = -h[:m]  # d/d h[i+r]
            G[q, :m] += d_h_low
            G[q, r:r+m] += d_h_high
    return G

def softmax_weights(scores, temperature):
    """Compute softmax weights from scores."""
    shifted = scores - np.max(scores)
    w = np.exp(shifted / temperature)
    w /= np.sum(w)
    return w

def _line_search(h, direction, S0, dx, max_iter=50):
    """Exact line search along direction. Returns (best_alpha, best_c5)."""
    n = len(h)
    target = n / 2.0
    
    # Compute bounds
    pos = direction > 1e-10
    neg = direction < -1e-10
    hi = 1.0
    lo = -1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / direction[neg])))
    
    if hi - lo < 1e-13:
        return 0.0, c5_exact(h)
    
    # Precompute T and C for quadratic formula
    T = np.correlate(direction, 1.0 - h, 'full') - np.correlate(h, direction, 'full')
    C = np.correlate(direction, direction, 'full')
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(max_iter):
        if b - a < 1e-13: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val: opt, val = lo, f_lo
    if f_hi < val: opt, val = hi, f_hi
    return opt, val

def _gradient_descent_step(h, G, S0, dx, temp, momentum_vec=None, momentum_alpha=0.3):
    """One step of gradient descent with softmax weighting.
    Returns (new_h, new_c5, momentum_vec)."""
    n = len(h)
    target = n / 2.0
    
    scores = S0 * dx
    w = softmax_weights(scores, temp)
    g = G.T @ w  # weighted gradient
    g -= np.mean(g)  # zero sum
    
    if momentum_vec is not None:
        g = momentum_alpha * momentum_vec + (1 - momentum_alpha) * g
    
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h, c5_exact(h), g
    
    direction = -g / gn
    
    # Line search
    opt, val = _line_search(h, direction, S0, dx)
    
    if opt > 1e-10 and val < c5_exact(h) - 1e-14:
        new_h = project(h + opt * direction, target)
        new_c5 = c5_exact(new_h)
        return new_h, new_c5, g
    
    # Try reverse direction
    opt_r, val_r = _line_search(h, -direction, S0, dx)
    if opt_r > 1e-10 and val_r < c5_exact(h) - 1e-14:
        new_h = project(h - opt_r * direction, target)
        new_c5 = c5_exact(new_h)
        return new_h, new_c5, g
    
    return h, c5_exact(h), g

def _pair_search(h, S0, dx, n_candidates, rng):

    if len(active) == 0:

    g = G_sub.T @ w

    donors = np.argsort(-g)[:n_candidates]
    receivers = np.argsort(g)[:n_candidates]
    
    best_alpha = 0.0
    best_val = cur_c5
    best_pair = None
    
    for i in donors:
        for j in receivers:
            if i == j: continue
            # Golden search for pair (i donating to j)
            d = np.zeros(n)

            d[j] = 1.0

            C = np.correlate(d, d, 'full')
            lo = -min(h[j], 1.0-h[i])

            flo = np.max(S0+lo*T-lo*lo*C)*dx

def _binary_optimize(h, deadline, rng):
    """Optimize a binary relaxation."""
    n = len(h)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
    cur_c5 = c5_exact(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    patience = 0
    
    while time.time() < deadline and patience < 5000:
        ones = np.where(h_b > 0.5)[0]
        zeros = np.where(h_b <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0: break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_c5 = c5_exact(h_b)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_b.copy()
                patience = 0
            else:
                patience += 1
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
            patience += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

import numpy as np
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        c = q - (n - 1)
        m = n - abs(c)
        if c >= 0:
            G[q, c:c+m] += (1.0 - h[:m])
            G[q, :m] += (-h[c:c+m])
        else:
            r = -c
            G[q, :m] += (1.0 - h[r:r+m])
            G[q, r:r+m] += (-h[:m])
    return G

def _line_search(h, direction, S0, dx):
    n = len(h)
    pos = direction > 1e-10
    neg = direction < -1e-10
    hi = 2.0
    lo = -2.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / direction[neg])))
    if hi - lo < 1e-13:
        return 0.0, c5_exact(h)
    T = np.correlate(direction, 1.0-h, 'full') - np.correlate(h, direction, 'full')
    C = np.correlate(direction, direction, 'full')
    gr = (np.sqrt(5.0)-1.0)/2.0
    a, b = lo, hi
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    for _ in range(45):
        if b-a < 1e-13: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T - c1*c1*C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T - c2*c2*C)) * dx
    opt = (a+b)/2.0
    val = float(np.max(S0 + opt*T - opt*opt*C)) * dx
    flo = float(np.max(S0 + lo*T - lo*lo*C)) * dx
    fhi = float(np.max(S0 + hi*T - hi*hi*C)) * dx
    if flo < val: opt, val = lo, flo
    if fhi < val: opt, val = hi, fhi
    return opt, val

def _grad_step(h, S0, dx, k_active=20, use_full=True, temp_scale=0.05):
    """Gradient step using top-k or all rows."""
    n = len(h)
    target = n/2.0
    scores = S0 * dx
    G = compute_grad_matrix(h)
    
    if use_full:
        # Softmax weighting over all rows
        scores_shifted = scores - np.max(scores)
        w = np.exp(scores_shifted / (temp_scale * np.max(scores) + 1e-15))
        w /= np.sum(w)
        g = G.T @ w
    else:
        # Top-k uniform
        top_k = np.argsort(-scores)[:k_active]
        g = G[top_k].mean(axis=0)
    
    g -= np.mean(g)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h, c5_exact(h)
    
    direction = -g / gn
    opt, val = _line_search(h, direction, S0, dx)
    cur = c5_exact(h)
    
    if opt > 1e-10 and val < cur - 1e-13:
        new_h = project(h + opt*direction, target)
        return new_h, c5_exact(new_h)
    
    # Try reverse
    opt_r, val_r = _line_search(h, -direction, S0, dx)
    if opt_r > 1e-10 and val_r < cur - 1e-13:
        new_h = project(h - opt_r*direction, target)
        return new_h, c5_exact(new_h)
    
    return h, cur

def _pair_optimize(h, S0, dx, n_cand, rng, max_rounds=100):

    n = len(h)
    target = n/2.0
    h = h.copy()

        if len(active) == 0: active = np.argsort(-scores)[:10]

        w = np.ones(len(active)) / len(active)
        g = G_sub.T @ w
        g -= np.mean(g)

                C = np.correlate(d, d, 'full')
                lo = -min(h[j], 1-h[i])
                hi = min(h[i], 1-h[j])
                if hi-lo < 1e-13: continue
                gr = (np.sqrt(5.0)-1.0)/2.0
                a2,b2 = lo,hi
                c1 = b2-gr*(b2-a2); c2 = a2+gr*(b2-a2)
                f1 = np.max(S0+c1*T-c1*c1*C)*dx
                f2 = np.max(S0+c2*T-c2*c2*C)*dx
                for _ in range(30):
                    if b2-a2 < 1e-13: break
                    if f1 < f2:
                        b2,c2,f2 = c2,c1,f1
                        c1 = b2-gr*(b2-a2)
                        f1 = np.max(S0+c1*T-c1*c1*C)*dx
                    else:

                opt=(a2+b2)/2

                if flo<val: opt,val=lo,flo

            h[i] -= best_a; h[j] += best_a
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1-h, 'full')
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
    return h, c5_exact(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n)+0.5)/n
    x_new = (np.arange(n_new)+0.5)/n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new/2.0)

def _optimize_at_resolution(h, deadline, rng, n_iter=50):

                cur_c5 = c5_exact(h)
                stall = 0
        
        S0 = np.correlate(h, 1-h, 'full')
    
    return best_h, best_c5
