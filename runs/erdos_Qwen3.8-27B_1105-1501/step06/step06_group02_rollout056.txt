def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads_full(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k_idx in range(2*n-1):
        ell = k_idx - (n - 1)
        if ell >= 0:
            m = n - ell
            if m <= 0:
                continue
            G[k_idx, ell:ell+m] += (1.0 - h[0:m])
            G[k_idx, 0:m] += -h[ell:ell+m]
        else:
            r = -ell
            m = n - r
            if m <= 0:
                continue
            G[k_idx, 0:m] += (1.0 - h[r:r+m])
            G[k_idx, r:r+m] += -h[0:m]
    return G

def solve_qp(h, target_raw, rho, lambda_z, lambda_norm):
    """Solve the target QP. Returns direction d (zero-sum) or None."""
    n = len(h)
    R_raw = np.correlate(h, 1.0 - h, mode='full')

    z_init = np.maximum(0.0, R_raw - target_raw)
    x0 = np.zeros(n + C)
    x0[n:] = z_init
    
    # Objective and gradient
    scale_z = max(np.mean(z_init), 1e-6)
    norm_scale = max(n * rho * rho, 1e-12)
    
    def obj(x):
        d = x[:n]
        z = x[n:]
        return lambda_z * np.sum(z) / scale_z + lambda_norm * np.sum(d*d) / norm_scale
    
    def grad(x):
        d = x[:n]
        g = np.zeros(n + C)
        g[:n] = 2.0 * lambda_norm * d / norm_scale
        g[n:] = lambda_z / scale_z
        return g
    
    # Equality constraint: sum(d) = 0
    def eq_fun(x):
        return np.sum(x[:n])
    
    def eq_jac(x):
        j = np.zeros(1 + n + C)
        j[:n] = 1.0
        return j
    
    # Inequality constraints: z_q - G[q,:] @ d - (R_raw[q] - target) >= 0
    # i.e., z_q + target - R_raw[q] - G[q,:] @ d >= 0
    def ineq_fun(x):
        d = x[:n]
        z = x[n:]
        return z - G @ d - (R_raw - target_raw)
    
    def ineq_jac(x):
        # Shape: (C, n+C)
        J = np.zeros((C, n + C))
        J[:, :n] = -G
        J[:, n:] = np.eye(C)
        return J
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    bounds = list(zip(d_lo, d_hi)) + [(0.0, 2.0 * (R_raw.max() - target_raw + 1.0)) for _ in range(C)]
    # Actually, let's cap z at something reasonable

    # Ensure zero sum
    d -= np.mean(d)

    return d, R_raw, G

def qp_direction_lbfgs(h, target_raw, rho, mu, T_smooth=10.0):
    n = len(h)
    R_raw = np.correlate(h, 1.0 - h, mode='full')
    G = compute_grads_full(h)
    
    d_lo = -np.minimum(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    
    def obj_grad(v):
        d = v - np.mean(v)  # ensure zero sum
        s = T_smooth * (G @ d + R_raw - target_raw)
        # Softplus: log(1+exp(s)), but handle large s
        sp = np.where(s > 50, s - np.log(2), np.log1p(np.exp(s)))
        f = mu * np.sum(d*d) / (rho*rho*n) + np.mean(sp) / T_smooth
        # Gradient
        sig = 1.0 / (1.0 + np.exp(-s))  # sigmoid
        g = 2*mu*d / (rho*rho*n) + G.T @ sig / len(s)
        g -= np.mean(g)  # zero-sum
        # Handle gradient at kinks (where s is very negative or positive)
        # sigmoid is already 0 or 1 there, so it's fine
        return f, g
    
    # Initial point
    v0 = np.zeros(n)
    
    from scipy.optimize import minimize
    result = minimize(obj_grad, v0, jac=True, method='L-BFGS-B',
                      bounds=list(zip(d_lo, d_hi)),
                      options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-8})
    
    d = result.x - np.mean(result.x)
    if np.max(np.abs(d)) < 1e-12:
        return None, R_raw
    return d, R_raw
