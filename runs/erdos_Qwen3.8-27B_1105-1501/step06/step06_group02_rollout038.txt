def compute_row_grad(h, lag):
    """Gradient of row with centered lag 'lag' w.r.t. h.
    lag is in range [-(n-1), n-1]. The row index in correlate output is lag + (n-1)."""
    n = len(h)
    g = np.zeros(n)
    m = n - abs(lag)
    if m <= 0:
        return g
    if lag >= 0:
        # Row = sum_{i=0}^{m-1} h[i+lag] * (1 - h[i])
        # d/dh[j]: if j appears as h[i+lag] (i = j-lag, valid if 0<=j-lag<m): contrib = (1-h[j-lag])
        #          if j appears as h[i] (i = j, valid if 0<=j<m): contrib = -h[j+lag]
        g[lag:lag+m] += (1.0 - h[:m])       # j = i+lag, i in [0,m)
        g[:m] += -h[lag:lag+m]               # j = i, i in [0,m)
    else:
        r = -lag
        # Row = sum_{i=0}^{m-1} h[i] * (1 - h[i+r])
        g[r:r+m] += (1.0 - h[:m])           # j = i+r, i in [0,m)
        g[:m] += -h[r:r+m]                   # j = i, i in [0,m)
    return g

def gradient_direction(h, R0, dx, n_active=10):
    """Find a descent direction using weighted gradient of active rows."""
    n = len(h)
    scores = R0 * dx
    max_score = float(np.max(scores))
    # Select active rows
    threshold = max_score - 0.02 * max_score
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-scores)[:n_active]
    active = active[:n_active]
    
    # Compute weighted gradient
    w = np.exp((scores[active] - max_score) / (0.01 * max_score + 1e-15))
    w /= w.sum()
    
    g = np.zeros(n)
    for idx, r in enumerate(active):
        lag = r - (n - 1)
        g += w[idx] * compute_row_grad(h, lag)
    
    # Project to zero-sum
    g -= g.mean()
    
    # Scale
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return None, None
    d = -g / gn
    return d, None

    t_start = time.time()
    scores = R0 * dx
    max_score = float(np.max(scores))
    
    # Select rows to reduce
    threshold = max_score - 0.03 * max_score
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-scores)[:n_lp_rows]
    active = active[:n_lp_rows]
    
    # Compute gradients for active rows
    grads = np.zeros((len(active), n))
    for idx, r in enumerate(active):
        lag = r - (n - 1)
        grads[idx] = compute_row_grad(h, lag)
    
    # Scale gradients so that |grad| ~ 1
    gscales = np.linalg.norm(grads, axis=1, keepdims=True)
    gscales[gscales < 1e-15] = 1.0
    grads /= gscales

    # s.t. grads @ d <= -t (reduce each active row)
    #      sum(d) = 0

def pair_line_search(h, i, j, R0, dx):
    """Golden section line search for direction e_j - e_i."""
    n = len(h)
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Use vertex-based line search
    cands = [lo, hi, (lo+hi)/2.0]
    # C has at most 4 non-zero entries, so just check all
    for k in range(len(C)):
        if abs(C[k]) > 1e-14:
            v = T[k] / (2.0 * C[k])
            if lo < v < hi:
                cands.append(v)
    
    best_val = float('inf')
    best_a = cands[0]
    for a in cands:
        val = float(np.max(R0 + a*T - a*a*C)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    
    # Short golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g, b_g = lo, hi
    c1 = b_g - gr*(b_g - a_g)
    c2 = a_g + gr*(b_g - a_g)
    f1 = float(np.max(R0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(R0 + c2*T - c2*c2*C)) * dx
    for _ in range(30):
        if b_g - a_g < 1e-13:
            break
        if f1 < f2:
            b_g, c2, f2 = c2, c1, f1
            c1 = b_g - gr*(b_g - a_g)
            f1 = float(np.max(R0 + c1*T - c1*c1*C)) * dx
        else:
            a_g, c1, f1 = c1, c2, f2
            c2 = a_g + gr*(b_g - a_g)
            f2 = float(np.max(R0 + c2*T - c2*c2*C)) * dx
    opt = (a_g + b_g) / 2.0
    val = float(np.max(R0 + opt*T - opt*opt*C)) * dx
    if val < best_val:
        best_val = val
        best_a = opt
    
    return best_a, best_val

    b_g = min(hi, best_a + window)
    if b_g - a_g > 1e-12:
        c1 = b_g - gr*(b_g - a_g)
        c2 = a_g + gr*(b_g - a_g)
        f1 = float(np.max(R0 + c1*T - c1*c1*C)) * dx
        f2 = float(np.max(R0 + c2*T - c2*c2*C)) * dx
        for _ in range(35):
            if b_g - a_g < 1e-13:
                break
            if f1 < f2:
                b_g, c2, f2 = c2, c1, f1
                c1 = b_g - gr*(b_g - a_g)
                f1 = float(np.max(R0 + c1*T - c1*c1*C)) * dx
            else:
                a_g, c1, f1 = c1, c2, f2
                c2 = a_g + gr*(b_g - a_g)
                f2 = float(np.max(R0 + c2*T - c2*c2*C)) * dx
        opt = (a_g + b_g) / 2.0
        val = float(np.max(R0 + opt*T - opt*opt*C)) * dx
        if val < best_val:
            best_val = val
            best_a = opt
    for a_chk in [lo, hi]:
        val = float(np.max(R0 + a_chk*T - a_chk*a_chk*C)) * dx
        if val < best_val:
            best_val = val
            best_a = a_chk
    return best_a, best_val

def _pair_ls(h, i, j, R0, dx):
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(len(h))
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    cands = [lo, hi, (lo+hi)/2.0]
    nz = np.where(np.abs(C) > 1e-14)[0]
    for k in nz:
        v = T[k] / (2.0 * C[k])
        if lo < v < hi:
            cands.append(v)
    best_val = float('inf')
    best_a = cands[0]
    for a in cands:
        val = float(np.max(R0 + a*T - a*a*C)) * dx
        if val < best_val:
            best_val = val
            best_a = a
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g, b_g = lo, hi
    c1 = b_g - gr*(b_g - a_g)
    c2 = a_g + gr*(b_g - a_g)
    f1 = float(np.max(R0 + c1*T - c1*c1*C)) * dx
    f2 = float(np.max(R0 + c2*T - c2*c2*C)) * dx
    for _ in range(30):
        if b_g - a_g < 1e-13:
            break
        if f1 < f2:
            b_g, c2, f2 = c2, c1, f1
            c1 = b_g - gr*(b_g - a_g)
            f1 = float(np.max(R0 + c1*T - c1*c1*C)) * dx
        else:
            a_g, c1, f1 = c1, c2, f2
            c2 = a_g + gr*(b_g - a_g)
            f2 = float(np.max(R0 + c2*T - c2*c2*C)) * dx
    opt = (a_g + b_g) / 2.0
    val = float(np.max(R0 + opt*T - opt*opt*C)) * dx
    if val < best_val:
        best_val = val
        best_a = opt
    return best_a, best_val

def _lp_direction(h, R0, dx, n_rows=8, t_max=0.05):
    n = len(h)
    scores = R0 * dx
    max_score = float(np.max(scores))
    threshold = max_score - 0.05 * max_score
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-scores)[:n_rows]
    active = active[:n_rows]
    if len(active) < 2:
        return None
    grads = np.zeros((len(active), n))
    for idx, r in enumerate(active):
        lag = r - (n - 1)
        grads[idx] = _row_grad(h, lag)
    gscales = np.linalg.norm(grads, axis=1, keepdims=True)
    gscales[gscales < 1e-15] = 1.0
    grads = grads / gscales
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ineq = np.zeros((len(active) + 1, n + 1))
    A_ineq[:len(active), :n] = grads
    A_ineq[:len(active), n] = -1.0
    b_ineq = np.zeros(len(active) + 1)
    A_ineq[len(active), n] = -1.0
    b_ineq[len(active)] = t_max
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-1.0, 1.0)] * n + [(-t_max, t_max)]
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 1.5})
        if res.success and res.x is not None:
            d = res.x[:n]
            t_val = res.x[n]
            if t_val < -1e-6 and np.linalg.norm(d) > 1e-10:
                return d
    except Exception:
        pass
    return None

def _directed_search(h, deadline, n_pairs=25, use_lp=True, lp_every=8):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    R0 = compute_corr(h)
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(int(time.time()*1e6) % 2**31)
    it = 0
    while time.time() < deadline and stall < 120:
        it += 1
        scores = R0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.008 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:12]
        w = np.exp((scores[active_idx] - max_score) / (0.004 * max_score + 1e-15))
        w /= w.sum()
        g = np.zeros(n)
        for idx, r in enumerate(active_idx):
            lag = r - (n - 1)
            g += w[idx] * _row_grad(h, lag)
        g -= g.mean()
        gn = np.linalg.norm(g)
        cur_best_new = cur_c5
        cur_best_action = None
        if gn > 1e-14:
            d = -g / gn
            alpha, new_c5 = _line_search(h, d, R0, dx)
            if alpha > 1e-10 and new_c5 < cur_best_new - 1e-14:
                cur_best_new = new_c5
                cur_best_action = ('dir', d, alpha)
        if use_lp and it % lp_every == 0 and time.time() < deadline - 2.0:
            d_lp = _lp_direction(h, R0, dx, n_rows=min(10, len(active_idx)))
            if d_lp is not None:
                d_ln = np.linalg.norm(d_lp)
                d_lp = d_lp / d_ln
                alpha, new_c5 = _line_search(h, d_lp, R0, dx)
                if alpha > 1e-10 and new_c5 < cur_best_new - 1e-14:
                    cur_best_new = new_c5
                    cur_best_action = ('dir', d_lp, alpha)
        donors = np.argsort(-h)[:n_pairs]
        receivers = np.argsort(h)[:n_pairs]
        tried = set()
        for i in donors:
            for j in receivers:
                if i == j:
                    continue
                key = (min(i,j), max(i,j))
                if key in tried:
                    continue
                tried.add(key)
                alpha, new_c5 = _pair_ls(h, i, j, R0, dx)
                if alpha > 1e-10 and new_c5 < cur_best_new - 1e-14:
                    cur_best_new = new_c5
                    cur_best_action = ('pair', i, j, alpha)
        if cur_best_action is not None and cur_best_new < cur_c5 - 1e-13:
            if cur_best_action[0] == 'dir':
                _, d, alpha = cur_best_action
                h = h + alpha * d
            else:
                _, i, j, alpha = cur_best_action
                h[i] -= alpha
                h[j] += alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R0 = compute_corr(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 25:
                n_pert = rng.integers(3, 8)
                idx = rng.integers(n, size=n_pert)
                for k in idx:
                    h[k] += rng.uniform(-0.08, 0.08)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R0 = compute_corr(h)
                stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _multicoord_search(h, deadline, rng):
    """Multi-coordinate moves: random subsets of 4-8 points with optimized signs."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    R0 = compute_corr(h)
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline and stall < 40:
        k = rng.integers(4, 9)
        idx = rng.integers(n, size=k)
        best_alpha = 0.0
        best_val = cur_c5
        best_signs = None
        for trial in range(4):
            signs = np.ones(k)
            n_neg = rng.integers(1, k)
            neg_idx = rng.choice(k, size=n_neg, replace=False)
            signs[neg_idx] = -1.0
            d = np.zeros(n)
            for ii in range(k):
                d[idx[ii]] = signs[ii]
            result = _line_search(h, d, R0, dx)
            alpha, val = result
            if alpha > 1e-10 and val < best_val - 1e-14:
                best_val = val
                best_alpha = alpha
                best_signs = (idx.copy(), signs.copy())
        if best_signs is not None:
            idx_arr, signs_arr = best_signs
            h[idx_arr] += best_alpha * signs_arr
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R0 = compute_corr(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.97
    rng = np.random.default_rng(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.4 * np.sin(2 * np.pi * 2.0 * x / n0 + 0.5)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0
    # Phase 1: Intensive directed search (35%)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _directed_search(h, d1, n_pairs=25, use_lp=True, lp_every=6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    # Phase 2: Multi-coordinate exploration (15%)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _multicoord_search(best_h, d2, rng)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Phase 3: Multi-resolution (25%)
    d3 = t0 + budget_s * 0.75
    n_candidates = []
    for dn in range(-24, 26, 2):
        n_try = best_n + dn
        if 60 <= n_try <= 240 and n_try % 2 == 0:
            n_candidates.append(n_try)
    rng2 = np.random.default_rng(seed + 1)
    rng2.shuffle(n_candidates)
    for n_try in n_candidates:
        if time.time() > d3:
            break
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _directed_search(h_r, sub_dl, n_pairs=20, use_lp=True, lp_every=5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    # Phase 4: Back to directed search with LP (15%)
    d4 = t0 + budget_s * 0.90
    h4, c5_4 = _directed_search(best_h, d4, n_pairs=30, use_lp=True, lp_every=3)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    # Phase 5: Final polish
    if time.time() < deadline:
        h5, c5_5 = _directed_search(best_h, deadline, n_pairs=40, use_lp=False, lp_every=999)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
