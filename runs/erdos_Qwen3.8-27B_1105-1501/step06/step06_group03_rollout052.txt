import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _grad_max_smooth(h, tau):
    n = len(h)
    S = _corr_scaled(h)
    M = float(S.max()) - M
    shifted = (S - S.max()) / tau
    w = np.exp(shifted)
    w /= w.sum()
    top_k = min(30, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad = w[top_idx] @ G
    grad -= grad.mean()
    return grad, S

def _line_search_quad(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 40
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.04 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _nesterov_smooth(h, deadline, tau_init, tau_final, n_steps=None):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h_cur)
    t = 1.0
    tau = tau_init
    n_total = n_steps if n_steps else 2000
    tau_decay = (tau_final / tau_init) ** (1.0 / max(n_total, 1))
    for step in range(n_total):
        if time.time() > deadline:
            break
        tau *= tau_decay
        y = h_cur + ((t - 1.0) / (t + 2.0)) * (h_cur - h_prev)
        grad, S = _grad_max_smooth(y, tau)
        step_size = tau / (1.0 + 4.0 * tau / dx)
        step_size = min(step_size, 0.05)
        z = y - step_size * grad
        h_new = _project(z, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_prev = h_cur
        h_cur = h_new
        t += 1.0
    return best_h, best_c5

def _grad_subgradient(y, tau):
    n = len(y)
    S = _corr_scaled(y)
    M = S.max()
    tone = (S - M + 0.05) / tau
    w = np.exp(tone)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(S))
        w[np.argmax(S)] = 1.0
    else:
        w /= w_sum
    top_k = min(25, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(y, top_idx, n)
    grad = w[top_idx] @ G
    grad -= grad.mean()
    return grad

def _continuous_sa(h_init, deadline, rng):

            else:

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick multi-start (8% time)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        for ph in [0.0, 0.7, 1.5, 2.3, 3.1]:
            starts.append(_sine_init(n0, freq, ph))
    for nb in [3, 4, 5, 6, 8, 10, 12]:
        starts.append(_block_init(n0, rng, nb))
    rng.shuffle(starts)
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _lp_refine(s, sub_dl, 0.10, 12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Alternating smooth + LP (25% time)
    d2 = t0 + budget_s * 0.33
    h2, c5_2 = _alternating_smooth_lp(best_h, d2, 0.015, 0.002)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Direct LP refinement (15% time)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _lp_refine(best_h, d3, 0.12, 20)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP polish (12% time)
    d4a = t0 + budget_s * 0.58
    h4a, c5_4a = _binary_sa(best_h, d4a, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    d4 = t0 + budget_s * 0.60
    if time.time() < d4:
        h4b, c5_4b = _lp_refine(best_h, d4, 0.06, 15)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Continuous SA + LP (10% time)
    d5a = t0 + budget_s * 0.68
    h5a, c5_5a = _continuous_sa(best_h, d5a, rng)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    d5 = t0 + budget_s * 0.70
    if time.time() < d5:
        h5b, c5_5b = _lp_refine(best_h, d5, 0.08, 15)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Pair refinement (10% time)
    d6 = t0 + budget_s * 0.80
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Multi-resolution (12% time)
    d7 = t0 + budget_s * 0.92
    base_n = len(best_h)
    offsets = [0, 2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 14, -14, 18, -18]
    tried = set()
    for dn in offsets:
        if time.time() > d7:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d7, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_refine(h_r, sub_dl, 0.10, 12)
            if c5_r < best_c5:
                sub_dl2 = min(d7, time.time() + budget_s * 0.01)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d7 - 0.05:
                    h_r, c5_r = _lp_refine(h_r, min(d7, time.time() + budget_s * 0.008),
                                           0.05, 12)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 8: Aggressive perturbation (5% time)
    d8 = t0 + budget_s * 0.965
    scale = 0.025
    for attempt in range(500):
        if time.time() > d8:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d8, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_refine(h_pert, sub_dl, 0.05, 10)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.15)

    # Phase 9: Final polish
    d9 = deadline
    h9, c5_9 = _pair_refine(best_h, d9, max_passes=80)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()
    if time.time() < deadline - 0.1:
        h10, c5_10 = _lp_refine(best_h, deadline, 0.03, 10)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
