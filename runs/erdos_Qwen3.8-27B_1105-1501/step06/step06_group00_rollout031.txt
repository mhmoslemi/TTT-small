def _ccd_sweep(h, deadline, grid_size=128):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    improved = False
    
    for j in range(n):
        if time.time() > deadline:
            break
        
        old_x = h[j]
        alpha, const = _compute_alpha_const(h, j, corr)
        
        # Grid search for optimal x
        x_grid = np.linspace(0, 1, grid_size)
        f_vals = np.max(alpha[None, :] * x_grid[:, None] + const[None, :], axis=1) * dx
        best_idx = np.argmin(f_vals)
        x_new = x_grid[best_idx]
        
        # Refine around best
        if 0 < best_idx < grid_size - 1:
            lo = x_grid[best_idx - 1]
            hi = x_grid[best_idx + 1]
            x_fine = np.linspace(lo, hi, 64)
            f_fine = np.max(alpha[None, :] * x_fine[:, None] + const[None, :], axis=1) * dx
            x_new = x_fine[np.argmin(f_fine)]

            h[j] = x_new
            # Update corr
            delta = x_new - old_x
            corr += delta * alpha
            improved = True
    
    return h, improved

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection: find theta such that sum(clip(v-theta, 0, 1)) = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _ccd_one_sweep(h, corr, dx, deadline, grid_size=256):

        if time.time() > deadline:
            break

            if j + d < n:
                a -= h[j + d]
            alpha[n - 1 + d] = a

        for m in range(1, n):
            a = 0.0
            if j + m < n:
                a += (1.0 - h[j + m])
            if j - m >= 0:
                a -= h[j - m]
            alpha[n - 1 - m] = a
        # Compute const
        const = corr - alpha * old_x
        # Grid search
        x_grid = np.linspace(0.0, 1.0, grid_size)
        # f(x) = dx * max(alpha * x + const)
        vals = alpha[None, :] * x_grid[:, None] + const[None, :]
        f_vals = np.max(vals, axis=1) * dx
        best_idx = int(np.argmin(f_vals))
        x_new = x_grid[best_idx]
        # Refine

            lo = x_grid[best_idx - 1]
            hi = x_grid[best_idx + 1]
            x_fine = np.linspace(lo, hi, 64)

            h[j] = x_new

import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _ccd_sweep(h, corr, dx, deadline, grid_size=256):
    n = len(h)
    n_improved = 0
    for j in range(n):
        if time.time() > deadline:
            break
        old_x = h[j]
        # Compute alpha
        alpha = np.zeros(2 * n - 1)
        for d in range(n):
            a = 0.0
            if d <= j:
                a += 1.0 - h[j - d]
            if j + d < n:
                a -= h[j + d]
            alpha[n - 1 + d] = a
        for m in range(1, n):
            a = 0.0
            if j + m < n:
                a += 1.0 - h[j + m]
            if j - m >= 0:
                a -= h[j - m]
            alpha[n - 1 - m] = a
        # const for linear terms
        const = corr - alpha * old_x
        # Fix d=0 (index n-1)
        const_0 = corr[n - 1] - old_x + old_x * old_x
        # Grid evaluation
        x_grid = np.linspace(0.0, 1.0, grid_size)
        M = alpha[None, :] * x_grid[:, None] + const[None, :]
        M[:, n - 1] = const_0 + x_grid - x_grid * x_grid
        f_vals = np.max(M, axis=1) * dx
        best_idx = int(np.argmin(f_vals))
        x_new = x_grid[best_idx]
        # Refine

            lo = x_grid[best_idx - 1]
            hi = x_grid[best_idx + 1]

            Mf[:, n - 1] = const_0 + xf - xf * xf

            h[j] = x_new

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
