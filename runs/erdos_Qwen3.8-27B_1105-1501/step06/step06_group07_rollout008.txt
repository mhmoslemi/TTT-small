                pass  # handled separately

                s -= h[p]  # = h[i-k]

            is_quad[j] = True

    """Find the optimal value of h[i] given all other coordinates fixed."""
    n = len(h)
    dx = 2.0 / n

        if k == 0:
            continue
        sv = 0.0
        # First factor: h[i] appears if 0≤i≤n-1 and 0≤i+k≤n-1
        if 0 <= i + k <= n-1:

        # Second factor: h[i] = h[p+k] where p = i-k, valid if 0≤p≤n-1 and 0≤p+k≤n-1
        p = i - k
        if 0 <= p <= n-1 and 0 <= p + k <= n-1:
            sv -= h[p]
        s[j] = sv

    # -t² + (1-s[j])*t + (F0 - old_t + old_t² - corr[j] + s[j]*old_t) = 0
    # t² - (1-s[j])*t - (F0 - old_t + old_t² - corr[j] + s[j]*old_t) = 0

        if j == n-1:
            continue
        a = 1.0
        b = -(1.0 - s[j])

        disc = b*b - 4*a*c

            t2 = (-b - np.sqrt(disc)) / (2*a)
            if 0.0 <= t1 <= 1.0:
                candidates.append(t1)
            if 0.0 <= t2 <= 1.0:
                candidates.append(t2)
    
    # Line-line intersections: l_j(t) = l_m(t)
    # s[j]*t + (corr[j] - s[j]*old_t) = s[m]*t + (corr[m] - s[m]*old_t)
    # (s[j]-s[m])*t = corr[m] - corr[j] + (s[j]-s[m])*old_t
    # t = (corr[m] - corr[j] + (s[j]-s[m])*old_t) / (s[j]-s[m])

    j_indices = [j for j in range(2*n-1) if j != n-1]
    for ai in range(len(j_indices)):

        for bi in range(ai+1, len(j_indices)):

            ds = s[ja] - s[jb]
            if abs(ds) < 1e-12:

            t_val = (corr[jb] - corr[ja] + ds * old_t) / ds
            if 0.0 <= t_val <= 1.0:
                candidates.append(t_val)
    
    # Evaluate C5 at all candidates
    best_t = old_t
    best_val = np.inf
    for t in candidates:
        t = max(0.0, min(1.0, t))
        # Compute C5(t)
        new_corr = corr + (t - old_t) * s
        # Fix k=0 entry: should be F0 + (t-old_t) - (t²-old_t²)

        new_corr[n-1] = F0 + (t - old_t) - (t*t - old_t*old_t)
        val = np.max(new_corr) * dx
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

  This is separable in (t_i, t_j)! No interaction term.

For most pairs (i,j) and most lags k, there's no interaction. The interaction only happens for the specific lag k = j-i (or i-j).
