def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Water-filling projection onto [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, k):
    """Gradient of correlation at lag k."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        # corr_k = sum_{i=0}^{m-1} h[i+k]*(1-h[i])
        # d/dh[i+k] = (1-h[i]) for i in [0, m)
        # d/dh[i] = -h[i+k] for i in [0, m)
        g[k:] += (1.0 - h[:m])  # derivative w.r.t. h[i+k]
        g[:m] -= h[k:]  # derivative w.r.t. h[i]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        # corr_k = sum_{i=ga}^{n-1} h[i]*(1-h[i-ga])
        # d/dh[i] = (1-h[i-ga]) for i in [ga, n)
        # d/dh[i-ga] = -h[i] for i in [ga, n)
        g[ga:] += (1.0 - h[:m])
        g[:m] -= h[ga:]
    return g

            opt_t = (a + b) / 2.0
            opt_obj = float(np.max(S + opt_t * T - opt_t**2 * Q)) * dx

Actually, for a max of concave quadratics, the function is "piecewise concave" - between any two points where the same quadratic is active, it's concave. The transitions happen at intersection points of pairs of quadratics.

A more precise approach:
1. Find all intersection points of pairs of the K active quadratics
2. Sort them
3. Between consecutive intersection points, one quadratic is the max
4. Find the minimum of each active quadratic on its interval
5. Take the overall minimum

This is exact for a fixed set of active quadratics! Let me implement this.

For K active quadratics a_r(t) = S_r + t*T_r - t^2*Q_r:
- Intersection of a_i and a_j: solve S_i + t*T_i - t^2*Q_i = S_j + t*T_j - t^2*Q_j
  => (Q_j - Q_i)*t^2 + (T_i - T_j)*t + (S_j - S_i) = 0
  This is a quadratic in t, giving at most 2 roots.

For K quadratics, there are O(K^2) intersection points, each giving at most 2 roots. So O(K^2) critical points total.

For each interval between consecutive critical points (in the valid range [0, t_hi]), the active quadratic is fixed. Within that interval, the minimum of a concave quadratic is at one of the endpoints or at the vertex (if the vertex is in the interval).

This gives an EXACT 1D minimization in O(K^2) time! Much better than grid search.

def _exact_1d_min(h, S, T, Q, t_hi):
    """Exact minimization of max_r (S_r + t*T_r - t^2*Q_r) on [0, t_hi].
    S, T, Q are arrays of length 2n-1 (all lags)."""
    n_lags = len(S)
    
    # Collect all critical points
    crit = [0.0, t_hi]
    for i in range(n_lags):
        for j in range(i+1, n_lags):
            # (Q_j - Q_i)*t^2 + (T_i - T_j)*t + (S_j - S_i) = 0
            a = Q[j] - Q[i]
            b = T[i] - T[j]
            c = S[j] - S[i]
            if abs(a) < 1e-15:
                if abs(b) < 1e-15:
                    continue
                t = -c / b
                if 0 < t < t_hi:
                    crit.append(t)
            else:
                disc = b*b - 4*a*c
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-b + sqrt_disc) / (2*a)
                    t2 = (-b - sqrt_disc) / (2*a)
                    if 0 < t1 < t_hi:
                        crit.append(t1)
                    if 0 < t2 < t_hi:
                        crit.append(t2)
    
    crit = sorted(set(crit))
    
    # Add a small epsilon around each critical point
    # Actually, just evaluate at each critical point and at the vertex of each active quadratic
    
    best_t = 0.0
    best_val = float(np.max(S))  # at t=0
    
    # For each interval [crit[i], crit[i+1]], find which lags could be active
    # and evaluate at the midpoint and at the vertex of each candidate quadratic
    
    for i in range(len(crit) - 1):
        t_mid = (crit[i] + crit[i+1]) / 2.0
        # Find which quadratic is max at t_mid
        vals = S + t_mid * T - t_mid**2 * Q
        active = int(np.argmax(vals))
        val_mid = vals[active]
        
        if val_mid < best_val:
            best_val = val_mid
            best_t = t_mid
        
        # Also check the vertex of the active quadratic
        # a_r(t) = S_r + t*T_r - t^2*Q_r
        # vertex at t = T_r / (2*Q_r) if Q_r > 0
        if Q[active] > 1e-15:
            t_v = T[active] / (2 * Q[active])
            if crit[i] <= t_v <= crit[i+1]:
                val_v = S[active] + t_v * T[active] - t_v**2 * Q[active]
                if val_v < best_val:
                    best_val = val_v
                    best_t = t_v
    
    # Also check the endpoints
    for t_check in [0.0, t_hi]:
        val = float(np.max(S + t_check * T - t_check**2 * Q))
        if val < best_val:
            best_val = val
            best_t = t_check
    
    return best_t, best_val

    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0

    for _ in range(max_iters):
        if time.time() > deadline or stall > 40:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))

        # Find coordinates most "helpful" to move
        # Gradient tells us which direction to push each coordinate
        K = 20
        top_idx = np.argsort(-corr)[:K]
        grad = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            grad += _grad_lag(h, k) / K
        grad -= grad.mean()

        # Pick i with most negative grad (should increase) and j with most positive (should decrease)
        pos_candidates = np.where(h < 0.8)[0]
        neg_candidates = np.where(h > 0.2)[0]
        if len(pos_candidates) < 2 or len(neg_candidates) < 2:
            stall += 10
            continue

        # Weighted random selection biased by gradient
        p_pos = np.exp(-20 * grad[pos_candidates])
        p_pos /= p_pos.sum()
        p_neg = np.exp(20 * grad[neg_candidates])
        p_neg /= p_neg.sum()

        i = pos_candidates[rng.choice(len(pos_candidates), p=p_pos)]
        j = neg_candidates[rng.choice(len(neg_candidates), p=p_neg)]

        # Try moving mass from j to i
        D = np.zeros(n)
        D[i] = 1.0
        D[j] = -1.0

        T_arr = np.correlate(D, 1.0 - h, mode='full') - np.correlate(h, D, mode='full')
        Q_arr = np.correlate(D, D, mode='full')
        corr_raw = np.correlate(h, 1.0 - h, mode='full')

        t_opt, obj_opt = _line_search(h, D, corr_raw, T_arr, Q_arr, top_idx[:30], t_max=1.0)
        if t_opt > 1e-10 and obj_opt < best_c5 - 1e-14:
            h_new = h + t_opt * D
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _binary_polish(h, deadline, rng, max_flips=200):
    """Try binarizing and then doing local binary flips."""
    n = len(h)
    m = n // 2
    if m * 2 != n:
        return h, _exact_c5(h)

    # Binarize
    order = np.argsort(-h)
    hb = np.zeros(n)
    hb[order[:m]] = 1.0
    cur_c5 = _exact_c5(hb)
    best_hb = hb.copy()
    best_c5b = cur_c5

    if cur_c5 >= _exact_c5(h) - 1e-12:
        return h, _exact_c5(h)

    # Local binary search: flip pairs
    for _ in range(max_flips):
        if time.time() > deadline:
            break
        # Try random pair flip
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        ia, ja = ones[i], zeros[j]
        hb[ia] = 0
        hb[ja] = 1
        new_c5 = _exact_c5(hb)
        if new_c5 < best_c5b - 1e-14:
            best_c5b = new_c5
            best_hb = hb.copy()
        else:
            hb[ia] = 1
            hb[ja] = 0

    # Compare binary with continuous
    c5_cont = _exact_c5(h)
    if best_c5b < c5_cont:
        return best_hb, best_c5b
    return h, c5_cont


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.15 * np.sin(2 * np.pi * 3 * x / n0) + 0.08 * np.sin(2 * np.pi * 7 * x / n0 + 0.5)

    n0 = len(h0)
    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Intensive active-set descent (35% time)
    d1 = t0 + budget_s * 0.35
    for K in [12, 18, 25, 35]:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.07)
        h1, c5_1 = _ascent_descent(best_h, sub_dl, K=K, max_iters=300, alpha_base=30.0)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Vertex-directed search (15% time)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _vertex_search(best_h, min(d2, time.time() + budget_s * 0.14), rng, n_trials=60)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        # Follow up with active descent
        h2b, c5_2b = _ascent_descent(best_h, min(d2, time.time() + budget_s * 0.03), K=20, max_iters=150)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Pair swap (15% time)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = _pair_swap(best_h, min(d3, time.time() + budget_s * 0.14), rng, max_iters=400)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-n exploration (20% time)
    d4 = t0 + budget_s * 0.85
    best_n = len(best_h)
    offsets = [-4, 4, -6, 6, -8, 8, -2, 2, -10, 10]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = _ascent_descent(h_r, sub_dl, K=20, max_iters=100)
        h_r, c5_r = _pair_swap(h_r, min(d4, time.time() + budget_s * 0.008), rng, max_iters=150)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Binary polish (5% time)
    d5 = t0 + budget_s * 0.90
    h5, c5_5 = _binary_polish(best_h, d5, rng, max_flips=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final intensive polish (remaining time)
    d6 = deadline
    h6, c5_6 = _ascent_descent(best_h, d6, K=15, max_iters=200, alpha_base=40.0)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    h6b, c5_6b = _pair_swap(best_h, d6, rng, max_iters=200)
    if c5_6b < best_c5:
        best_c5 = c5_6b
        best_h = h6b.copy()

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
