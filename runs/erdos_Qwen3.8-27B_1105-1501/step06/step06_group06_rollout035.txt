import numpy as np
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_at_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    s = k - (n - 1)
    m = n - abs(s)
    g = np.zeros(n)
    if s >= 0:
        g[:m] += (1.0 - h[s:s + m])
        g[s:s + m] += (-h[:m])
    else:
        g2 = -s
        g[g2:g2 + m] += (1.0 - h[:m])
        g[:m] += (-h[g2:g2 + m])
    return g * dx

def _smooth_grad(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = float(corr.max())
    w = np.exp(mu * (corr - cm))
    Z = float(w.sum())
    w = w / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * _grad_at_lag(h, k)
    return grad, corr

def _smooth_optimize(h0, deadline, mu_start=50.0, mu_end=500.0, max_iter=200):

    target = n / 2.0
    h = h0.copy()

    best_h = h.copy()
    best_c5 = cur

    prev_grad = None
    step_count = 0

    for it in range(max_iter):

            break

        grad, corr = _smooth_grad(h, mu)

        # Barzilai-Borwein step size
        if it > 0 and prev_grad is not None:
            s = h - prev_h
            y = grad - prev_grad
            sy = float(s @ y)
            yy = float(y @ y)
            ss = float(s @ s)
            if yy > 1e-20 and ss > 1e-20:
                if (it % 2 == 0):
                    step = ss / max(sy, 1e-15)
                else:
                    step = sy / max(yy, 1e-15)
                step = max(0.1, min(step, 10.0))
            else:
                step = 1.0
        else:
            step = 1.0

        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(30):
            h_try = h - alpha * grad
            h_try = _project(h_try, target)
            c_try = _exact_c5(h_try)
            if c_try < cur - 1e-15:
                improved = True
                break
            alpha *= 0.6

        if improved:
            h = h_try
            cur = c_try

                best_c5 = cur
                best_h = h.copy()
            step_count = 0

            if step_count > 30:
                # Perturbation
                rng_p = np.random.default_rng(it * 1000 + 7)
                h = h + rng_p.normal(0, 0.02, n)
                h = _project(h, target)
                cur = _exact_c5(h)
                step_count = 0

        prev_h = h.copy()
        prev_grad = grad.copy()

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _structured_init(n, rng):
    h = np.zeros(n)
    if n % 4 == 0:
        pat = np.array([1.0, 1.0, 0.0, 0.0])
        h = np.tile(pat, n // 4)
    elif n % 2 == 0:
        pat = np.array([1.0, 0.0])
        h = np.tile(pat, n // 2)
    h = h + rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / n + phase)
    h = h + rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)

def _ts_init(n, rng):
    # Thue-Morse inspired balanced sequence
    h = np.zeros(n)
    period = 4
    pat = np.array([1.0, 1.0, 0.0, 0.0])
    reps = n // period
    rem = n % period
    h[:reps * period] = np.tile(pat, reps)
    if rem > 0:
        h[reps * period:n] = pat[:rem]
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def _slsqp_polish(h0, deadline, max_outer=20):

    dx = 2.0 / n
    target = n / 2.0

    best_h = h.copy()

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _all_corr(h)
        K = min(40, 2 * n - 1)
        top_k = np.argsort(-corr)[:K]

        def _obj_fn(x, n=n):
            return x[n]

        def _obj_grad(x, n=n):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def _ineq_fn(x, n=n, top_k=top_k, dx=dx):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * dx
            return x[n] - c[top_k]

        def _ineq_jac(x, n=n, top_k=top_k, dx=dx):
            hx = x[:n]
            J = np.zeros((len(top_k), n + 1))
            for ii, kk in enumerate(top_k):
                J[ii, :n] = -_grad_at_lag(hx, kk)
                J[ii, n] = 1.0
            return J

        def _eq_fn(x, n=n):
            return np.sum(x[:n]) - target

        def _eq_jac(x, n=n):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        t_init = cur_c5 + 1e-4
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _ineq_fn, 'jac': _ineq_jac},
            {'type': 'eq', 'fun': _eq_fn, 'jac': _eq_jac},
        ]
        res = minimize(_obj_fn, x0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 60, 'ftol': 1e-14})

        if res.success or res.x is not None:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break

    return best_h, best_c5
