Let me think about the LP formulation more carefully:

Given current state `h`, raw overlap rows `R0[i]`, and gradient matrix `G` where `G[i,j] = dR_i/dh_j`.

The first-order approximation of row `i` at point `v` is:
`R_i(v) ≈ R0[i] + G[i,:] @ (v - h)`

We want to minimize `max_i R_i(v)` approximately, subject to:
- `sum(v) = n/2`
- `|v_j - h_j| <= radius` (trust region)
- `0 <= v_j <= 1`

This is an LP:
- Variables: `t, v_0, ..., v_{n-1}`
- Minimize: `t`
- Subject to: `R0[i] + G[i,:] @ (v - h) <= t` for all `i`
  i.e., `G[i,0]*v_0 + ... + G[i,n-1]*v_{n-1} - t <= G[i,:] @ h - R0[i]`
- `sum(v_j) = n/2`
- `h_j - radius <= v_j <= h_j + radius` (intersected with [0,1])

After solving, we get `v`. The direction is `d = v - h` (should have sum ≈ 0).

Then we do exact line search along `d`:
- `R0 = correlate(h, 1-h, 'full')`
- `T = correlate(d, 1-h, 'full') - correlate(h, d, 'full')`
- `C = correlate(d, d, 'full')`
- Find alpha that minimizes `max(R0 + alpha*T - alpha^2*C) * dx`

def _solve_lp(h, G, R0, radius):
    """Solve the LP trust region problem.
    Minimize t subject to:
      G[i,:] @ v + (R0[i] - G[i,:] @ h) <= t  for all i
      sum(v) = n/2
      h[j] - radius <= v[j] <= h[j] + radius
      0 <= v[j] <= 1
    """
    n = len(h)
    L = G.shape[0]
    
    # Variables: [t, v_0, ..., v_{n-1}]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    # Inequality constraints: G[i,:] @ v - t <= G[i,:] @ h - R0[i]
    # i.e., -t + G[i,:] @ v <= G[i,:] @ h - R0[i]
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = G @ h - R0
    
    # Equality constraints: sum(v) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: [t has no bound, v_j has bounds]
    bounds = [(None, None)]  # t is free
    for j in range(n):
        lo = max(0.0, h[j] - radius)
        hi = min(1.0, h[j] + radius)
        bounds.append((lo, hi))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        v = res.x[1:]
        return v
    return None
```

def _gradient_direction(h, G, R0, n_top=30):
    """Compute a descent direction by weighting active row gradients."""
    n = len(h)
    # Find active rows (those close to the max)
    max_R = np.max(R0)
    threshold = max_R - 0.01 * max_R
    active = np.where(R0 >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(-R0)[:n_top]
    
    # Weight by how close to max
    weights = np.exp((R0[active] - max_R) / (0.005 * max_R + 1e-15))
    weights /= np.sum(weights)
    
    # Weighted gradient (negated for descent)
    g = G[active].T @ weights
    # Make it zero-sum
    g -= np.mean(g)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return None
    return -g / gn
```

def compute_G(h):

    G = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        d = i - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            G[i, :m] += (1.0 - h[d:d + m])
            G[i, d:d + m] += (-h[:m])
        else:
            g = -d
            G[i, g:] += (1.0 - h[:m])
            G[i, :m] += (-h[g:g + m])
    return G

def _golden_line_search(R0, T, C, lo, hi, dx):
    if hi - lo < 1e-14:
        a = (lo + hi) / 2.0
        return a, float(np.max(R0 + a * T - a * a * C)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(R0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def _feasible_interval(h, d):
    lo_b = -1e10
    hi_b = 1e10
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        lo_b = max(lo_b, float(np.max(-h[pos] / d[pos])))
        hi_b = min(hi_b, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo_b = max(lo_b, float(np.max((1.0 - h[neg]) / d[neg])))
        hi_b = min(hi_b, float(np.min(-h[neg] / d[neg])))
    if hi_b - lo_b < 1e-12:
        return None
    return lo_b, hi_b

def _solve_lp(h, G, R0, radius):
    n = len(h)
    L = G.shape[0]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = G @ h - R0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(None, None)]
    for j in range(n):
        lo = max(0.0, h[j] - radius)
        hi = min(1.0, h[j] + radius)
        bounds.append((lo, hi))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[1:]
    except Exception:
        pass
    return None

def _lp_descent(h, deadline, max_stall=80):
    n = len(h)
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius = 0.15
    
    while time.time() < deadline and stall < max_stall:
        R0 = np.correlate(h, 1.0 - h, 'full')
        G = compute_G(h)
        
        v = _solve_lp(h, G, R0, radius)
        
        if v is not None:
            d = v - h
            d -= np.mean(d)  # ensure zero sum
            dn = np.linalg.norm(d)
            if dn > 1e-10:
                interval = _feasible_interval(h, d)
                if interval is not None:
                    lo, hi = interval
                    lo = max(lo, -1.0)
                    hi = min(hi, 1.0)
                    if hi > lo:
                        T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                        C = np.correlate(d, d, 'full')
                        alpha, val = _golden_line_search(R0, T, C, lo, hi, 2.0/n)
                        if abs(alpha) > 1e-10:
                            h_new = h + alpha * d
                            h_new = project(h_new, target)
                            new_c5 = c5_exact(h_new)
                            if new_c5 < cur_c5 - 1e-12:
                                h = h_new
                                cur_c5 = new_c5
                                if cur_c5 < best_c5:
                                    best_c5 = cur_c5
                                    best_h = h.copy()
                                stall = 0
                                # Slightly increase radius if we hit boundary
                                if abs(alpha) > 0.9 * min(abs(lo), abs(hi)) + 0.1:
                                    radius = min(0.3, radius * 1.3)
                                continue
                        # No improvement from this radius
                        radius *= 0.6
                        stall += 1
                    else:
                        radius *= 0.5
                        stall += 1
                else:
                    radius *= 0.5
                    stall += 1
            else:
                radius *= 0.5
                stall += 1
        else:
            radius *= 0.5
            stall += 1
        
        if radius < 1e-5:
            radius = 0.15  # reset
            stall += 2
    
    return best_h, best_c5

def _gradient_polish(h, deadline):
    """Fallback gradient-based polish."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    
    while time.time() < deadline and stall < 50:
        R0 = np.correlate(h, 1.0 - h, 'full')
        G = compute_G(h)
        max_R = float(np.max(R0))
        threshold = max_R - 0.005 * max_R
        active = np.where(R0 >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-R0)[:20]
        weights = np.exp((R0[active] - max_R) / (0.005 * max_R + 1e-15))
        weights /= np.sum(weights)
        g = G[active].T @ weights
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        d = -g / gn
        interval = _feasible_interval(h, d)
        if interval is None:
            break
        lo, hi = interval
        lo = max(lo, -0.5)
        hi = min(hi, 0.5)
        if hi <= lo:
            break
        T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
        C = np.correlate(d, d, 'full')
        alpha, val = _golden_line_search(R0, T, C, lo, hi, dx)
        if abs(alpha) > 1e-10 and val < cur_c5 - 1e-12:
            h_new = h + alpha * d
            h_new = project(h_new, target)
            new_c5 = c5_exact(h_new)
            if new_c5 < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _pair_search(h, deadline, rng, n_pairs=15):
    """Simple pair search as fallback."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    
    while time.time() < deadline and stall < 50:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.008 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]
        
        # Compute weighted gradient
        G = compute_G(h)
        weights = np.exp((S0[active_idx] - max_score) / (0.01 * max_score + 1e-15))
        weights /= np.sum(weights)
        wgrad = G[active_idx].T @ weights
        
        donor_pool = np.argsort(-wgrad)[:n_pairs]
        recv_pool = np.argsort(wgrad)[:n_pairs]
        
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        
        for i in donor_pool:
            for j in recv_pool:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                interval = _feasible_interval(h, d)
                if interval is None:
                    continue
                lo, hi = interval
                T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                C = np.correlate(d, d, 'full')
                alpha, new_c5 = _golden_line_search(S0, T, C, lo, hi, dx)
                if abs(alpha) > 1e-10 and new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.3:
                idx = rng.integers(n, size=4)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.03, 0.03)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    
    return best_h, best_c5

    dx = 2.0 / n

def _feasible_interval(h, d):
    lo_b = -1e10
    hi_b = 1e10
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        lo_b = max(lo_b, float(np.max(-h[pos] / d[pos])))
        hi_b = min(hi_b, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo_b = max(lo_b, float(np.max((1.0 - h[neg]) / d[neg])))
        hi_b = min(hi_b, float(np.min(-h[neg] / d[neg])))
    if hi_b - lo_b < 1e-12:

    return lo_b, hi_b

def _solve_lp(h, G, R0, radius, top_frac=0.0):

            idx = np.argsort(-R0)[:max(5, int(0.1 * L))]
            mask[idx] = True
        G_sel = G[mask]
        R0_sel = R0[mask]
        L = G_sel.shape[0]
    else:
        G_sel = G
        R0_sel = R0
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G_sel
    b_ub = G_sel @ h - R0_sel
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(None, None)]
    lo_v = np.maximum(0.0, h - radius)
    hi_v = np.minimum(1.0, h + radius)
    for j in range(n):
        bounds.append((lo_v[j], hi_v[j]))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
