import numpy as np
from scipy.optimize import linprog
import time

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _exact_c5(h):
    return float(_corr_values(h).max())

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _smooth_val(h, tau):
    scores = _corr_values(h)
    ms = float(scores.max())
    return ms + float(np.log(np.sum(np.exp(tau * (scores - ms))))) / tau

def _grad_smooth(h, tau):
    n = len(h)
    dx = 2.0 / n
    scores = _corr_values(h)
    ms = float(scores.max())
    exp_s = np.exp(tau * (scores - ms))
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    m = 2 * n - 1
    for r in range(m):
        pr = p[r]
        if pr < 1e-14:
            continue
        d = r - (n - 1)
        w = pr * dx
        if d >= 0:
            mlen = n - d
            if mlen > 0:
                grad[d:] += w * (1.0 - h[:mlen])
                grad[:mlen] -= w * h[d:]
        else:
            g = -d
            mlen = n - g
            if mlen > 0:
                grad[:mlen] += w * (1.0 - h[g:])
                grad[g:] -= w * h[:mlen]
    return grad

def _nesterov_phase(h0, deadline, tau_list, steps_per_tau):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for tau in tau_list:
        if time.time() > deadline:
            break
        t = 1.0
        x_prev = h.copy()
        alpha = min(1.0, 2.0 * n / (tau * 4.0 + 1.0))

        for step in range(steps_per_tau):
            if time.time() > deadline:
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            beta = (t - 1.0) / t_new
            y = h + beta * (h - x_prev)
            y = _project(y, target)

            g = _grad_smooth(y, tau)
            gn2 = float(np.dot(g, g))
            if gn2 < 1e-18:
                break

            a = alpha
            fy = _smooth_val(y, tau)
            for _bt in range(40):
                x_new = _project(y - a * g, target)
                fx = _smooth_val(x_new, tau)
                if fx <= fy - 0.4 * a * gn2:
                    break
                a *= 0.5
            else:
                x_new = _project(y - a * g, target)

            x_prev = h
            h = x_new
            t = t_new

            if step % 20 == 0:
                c5 = _exact_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def _grad_matrix_subset(h, lag_indices):
    n = len(h)
    G = np.zeros((len(lag_indices), n))
    for idx, r in enumerate(lag_indices):
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _lp_direction(h, scores, radius, top_k):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = _grad_matrix_subset(h, active)
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h0, deadline, radius_init=0.12, top_k=15):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(777)
    it = 0
    while time.time() < deadline:
        it += 1
        if stall > 100:
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.1
            stall = 0
        scores = _corr_values(h)
        d = _lp_direction(h, scores, radius, top_k)
        if d is not None and np.max(np.abs(d)) > 1e-13:
            alpha, new_c5 = _line_search(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                radius = min(radius * 1.04, 0.3)
                continue
        # Fallback: negative average gradient of top constraints
        max_s = float(scores.max())
        thr = max_s - 0.05 * max_s
        active = np.where(scores >= thr)[0]
        if len(active) >= 2:
            G = _grad_matrix_subset(h, active)
            avg_g = G.mean(axis=0)
            avg_g -= avg_g.mean()
            ng = np.linalg.norm(avg_g)
            if ng > 1e-14:
                d2 = -avg_g / ng * radius
                pos = d2 > 0
                neg = d2 < 0
                if np.any(pos):
                    d2[pos] = np.minimum(d2[pos], (1.0 - h[pos]) + 1e-15)
                if np.any(neg):
                    d2[neg] = np.maximum(d2[neg], -(h[neg]) + 1e-15)
                ma = np.max(np.abs(d2))
                if ma > radius:
                    d2 *= radius / ma
                if np.max(np.abs(d2)) > 1e-12:
                    alpha, new_c5 = _line_search(h, d2, dx, radius * 1.5)
                    if new_c5 < cur_c5 - 1e-14:
                        h = h + alpha * d2
                        h = _project(h, target)
                        cur_c5 = _exact_c5(h)
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                            stall = 0
                        else:
                            stall += 1
                        continue
        stall += 1
        radius *= 0.88
        if radius < 0.004:
            radius = 0.08
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    fill = max(1, int(duty * period))
    for i in range(n):
        if (i % period) < fill:
            h[i] = 1.0
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_init, deadline, rng, max_iters=6000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.0001
    cooling = 0.9997
    ones_list = np.where(h > 0.5)[0].tolist()
    zeros_list = np.where(h <= 0.5)[0].tolist()
    patience = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 4000:
            break
        if not ones_list or not zeros_list:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            ones_list = np.where(h > 0.5)[0].tolist()
            zeros_list = np.where(h <= 0.5)[0].tolist()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5
