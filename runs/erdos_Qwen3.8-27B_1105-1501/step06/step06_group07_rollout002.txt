def _weighted_gradient(h, K, alpha):
    """Weighted subgradient from top-K active constraints."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]

        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = 0.0, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(S + c1*T_arr - c1*c1*Q_arr)) * dx
    f2 = float(np.max(S + c2*T_arr - c2*c2*Q_arr)) * dx
    for _ in range(40):
        if b - a < 1e-12: break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S + c1*T_arr - c1*c1*Q_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S + c2*T_arr - c2*c2*Q_arr)) * dx
    opt_t = (a+b)/2.0
    opt_val = float(np.max(S + opt_t*T_arr - opt_t*opt_t*Q_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _gradient_descent(h, deadline, K=25, alpha=40.0, max_iters=200):
    """Exact gradient descent with line search."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0
    
    for it in range(max_iters):
        if time.time() > deadline: break
        g, corr = _weighted_gradient(h, K, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14: break
        d = -g / gn
        t, f_try = _line_search_exact(h, d, target, max_t=step)
        if t < 1e-12:
            step *= 0.7
            if step < 0.005: break
            continue
        h_new = _project(h + t*d, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.8 * step: step = min(step * 1.1, 1.5)
            else: step = max(step * 0.85, 0.01)
        else:
            step *= 0.7
            if step < 0.005: break
    return best_h, best_c5

def _to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _binary_sa(h_init, deadline, rng, label=""):
    """Intensive binary simulated annealing."""
    n = len(h_init)
    target = n // 2
    h = _to_binary(h_init)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T = 0.008
    T_min = 1e-8
    cooling = 0.99998
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    
    best_iter = 0
    last_improve = 0
    it = 0
    while time.time() < deadline:
        if T < T_min: break
        it += 1
        i_idx = rng.integers(len(ones))
        j_idx = rng.integers(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        
        if delta < 0 or (delta < T * 5 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                last_improve = it
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T = max(T_min, T * cooling)
        # Reheat
        if it - last_improve > 50000:
            T = max(T, 0.003)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n: return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hopping(h, deadline, rng, n_rounds=20):
    """Basin hopping: perturb and polish."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.1
    
    for r in range(n_rounds):
        if time.time() > deadline: break
        # Perturb
        h_p = h + rng.normal(0, scale, n)
        h_p = _project(h_p, n / 2.0)
        # Polish
        sub_dl = min(deadline, time.time() + (deadline - time.time()) / (n_rounds - r + 1))
        h_new, c5_new = _gradient_descent(h_p, sub_dl, K=20, alpha=50, max_iters=50)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            scale = max(0.03, scale * 0.8)
        else:
            scale = min(0.3, scale * 1.2)
    
    return best_h, best_c5
