import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_grad(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)
        w = weights[k_idx] * dx
        if w < 1e-14:
            continue
        if lag >= 0:
            m = n - lag
            if m <= 0:
                continue
            grad[:m] += w * (1.0 - h[lag:])
            grad[lag:n] -= w * h[:m]
        else:
            s = -lag
            m = n - s
            if m <= 0:
                continue
            grad[s:n] += w * (1.0 - h[:m])
            grad[:m] -= w * h[s:]
    return grad

def _line_search(h, d, dx, radius, top_k=12):
    n = len(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    t_lo = 0.0
    t_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi - t_lo < 1e-13:
        return 0.0, float(np.max(S) * dx)
    active = S - S.max()
    top_ids = np.argsort(-active)[:max(top_k, 6)]
    top_ids = np.unique(top_ids)
    candidates = [t_lo, t_hi]
    for idx in top_ids:
        qv = Q[idx]
        tv = T[idx]
        if qv > 1e-14:
            av = tv / (2.0 * qv)
            if t_lo < av < t_hi:
                candidates.append(av)
    for i in range(len(top_ids)):
        for j in range(i + 1, len(top_ids)):
            a1, b1, c1 = top_ids[i], top_ids[j], top_ids[i]
            ac = Q[c1] - Q[b1]
            bc = T[a1] - T[b1]
            cc = S[a1] - S[b1]
            if abs(ac) < 1e-16:
                if abs(bc) > 1e-16:
                    rr = -cc / bc
                    if t_lo < rr < t_hi:
                        candidates.append(rr)
            else:
                disc = bc * bc - 4.0 * ac * cc
                if disc >= 0:
                    sq = np.sqrt(disc)
                    r1 = (-bc + sq) / (2.0 * ac)
                    r2 = (-bc - sq) / (2.0 * ac)
                    if t_lo < r1 < t_hi:
                        candidates.append(r1)
                    if t_lo < r2 < t_hi:
                        candidates.append(r2)
    n_grid = 30
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    for gt in grid_t:
        candidates.append(float(gt))
    best_alpha = t_lo
    best_val = float(np.max(S + t_lo * T - t_lo * t_lo * Q)) * dx
    for ca in candidates:
        ca = float(ca)
        if ca < t_lo - 1e-12 or ca > t_hi + 1e-12:
            continue
        ca = max(t_lo, min(t_hi, ca))
        val = float(np.max(S + ca * T - ca * ca * Q)) * dx
        if val < best_val:
            best_val = val
            best_alpha = ca
    lo_b = max(t_lo, best_alpha - 0.15 * (t_hi - t_lo))
    hi_b = min(t_hi, best_alpha + 0.15 * (t_hi - t_lo))
    if hi_b - lo_b > 1e-10:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(60):
            if hi_b - lo_b < 1e-13:
                break
            if f1 < f2:
                hi_b, c2, f2 = c2, c1, f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
            else:
                lo_b, c1, f1 = c1, c2, f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S + opt_t * T - opt_t * opt_t * Q)) * dx
        if opt_val < best_val:
            best_val = opt_val
            best_alpha = opt_t
    return best_alpha, best_val

def _smooth_grad_direction(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    scaled = raw * dx
    mx = float(np.max(scaled))
    shifted = beta * (scaled - mx)
    shifted = np.clip(shifted, -80.0, 0.0)
    exp_v = np.exp(shifted)
    w = exp_v / (float(np.sum(exp_v)) + 1e-300)
    grad = _compute_grad(h, w, dx)
    grad -= np.mean(grad)
    nrm = np.linalg.norm(grad)
    if nrm < 1e-14:
        return None
    return grad / nrm

def _block_direction(h, g, blk_size):
    n = len(h)
    d = np.zeros(n)
    for start in range(0, n, blk_size):
        end = min(start + blk_size, n)
        avg = np.mean(g[start:end])
        d[start:end] = -avg
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        return None
    return d / nrm

def _pair_direction(h, g, n_pairs=8):
    n = len(h)
    idx = np.argsort(g)
    lows = idx[:n_pairs]
    highs = idx[-n_pairs:]
    d = np.zeros(n)
    for i in range(min(len(lows), len(highs))):
        d[lows[i]] = 1.0
        d[highs[i]] = -1.0
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        return None
    return d / nrm

def _lowfreq_direction(h, rng, n_modes=3):
    n = len(h)
    x = np.arange(n, dtype=float)
    d = np.zeros(n)
    for _ in range(n_modes):
        f = rng.uniform(0.5, 4.0)
        p = rng.uniform(0, 2 * np.pi)
        d += np.sin(2.0 * np.pi * f * x / n + p)
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        return None
    return d / nrm

def _gradient_loop(h, deadline, radius_init=0.15, beta_init=20.0, max_iter=50000):

    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    beta = beta_init
    stall = 0
    rng = np.random.default_rng(999)
    blk_sizes = [4, 8, 16, 32]
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        d_grad = _smooth_grad_direction(h, beta)
        if d_grad is None:
            stall += 5
            continue
        g = -d_grad
        directions = [d_grad]
        for bs in blk_sizes:
            bd = _block_direction(h, g, bs)
            if bd is not None:
                directions.append(bd)
        pd = _pair_direction(h, g, n_pairs=6)
        if pd is not None:
            directions.append(pd)
        lf = _lowfreq_direction(h, rng, n_modes=2)
        if lf is not None:
            directions.append(0.5 * d_grad + 0.5 * lf)
            tmp = 0.5 * d_grad + 0.5 * lf
            tmp -= np.mean(tmp)
            tn = np.linalg.norm(tmp)
            if tn > 1e-14:
                directions[-1] = tmp / tn
        best_alpha = 0.0
        best_new_c5 = cur_c5
        for d in directions:
            if time.time() > deadline:
                break
            d -= np.mean(d)
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-14:
                continue
            d = d / d_norm
            alpha, new_c5 = _line_search(h, d, 2.0 / n, radius, top_k=10)
            if new_c5 < best_new_c5 - 1e-14:
                best_new_c5 = new_c5
                best_alpha = alpha
                best_d = d
        if best_alpha != 0.0 and best_new_c5 < cur_c5 - 1e-13:
            h_new = h + best_alpha * best_d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project(h_new, target)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < best_c5 - 1e-14:
                h = h_new
                cur_c5 = actual_c5
                best_c5 = actual_c5
                best_h = h.copy()
                stall = 0
                radius = min(radius * 1.05, 0.35)
                beta = min(beta * 1.02, 60.0)
            else:
                stall += 1
                radius *= 0.85
        else:
            stall += 1
            radius = max(0.03, radius * 0.75)
            if stall % 30 == 0:
                beta = max(10.0, beta * 0.9)
    return best_h, best_c5

    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    T = T_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi = ones[i]
        oz = zeros[j]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
            T *= 0.999
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    for oi in ones:
        if time.time() > deadline:
            break
        improved = False
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _exact_c5(cur)
            if c_try < cur_c5 - 1e-14:
                cur_c5 = c_try
                improved = True
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                ones = list(np.where(cur > 0.5)[0])
                zeros = list(np.where(cur <= 0.5)[0])
                break
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
        if not improved:
            continue
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amplitude, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    perturb = np.zeros(n)
    for _ in range(3):
        f = rng.uniform(0.3, 5.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.5, 1.0)
        perturb += a * np.sin(2.0 * np.pi * f * x / n + p)
    perturb *= amplitude
    perturb += amplitude * 0.15 * rng.normal(0, 1, n)
    return _project(h + perturb, n / 2.0)

def _lp_direction(h, scores, radius):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k_idx in range(m):
        lag = k_idx - (n - 1)
        if lag >= 0:
            mm = n - lag
            if mm > 0:
                G[k_idx, :mm] += (1.0 - h[lag:])
                G[k_idx, lag:n] += -h[:mm]
        else:
            s = -lag
            mm = n - s
            if mm > 0:
                G[k_idx, s:n] += (1.0 - h[:mm])
                G[k_idx, :mm] += -h[s:]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.10, stall_limit=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    while time.time() < deadline:
        if stall > stall_limit:
            break
        scores = np.correlate(h, 1.0 - h, mode='full')
        p = _lp_direction(h, scores, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0 / n, radius, top_k=10)
        if new_c5 < cur_c5 - 1e-13:
            h_new = h + alpha * p
            h_new = _project(h_new, target)
            actual = _exact_c5(h_new)
            if actual < best_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                best_c5 = actual
                best_h = h.copy()
                stall = 0
                radius = min(radius * 1.04, 0.25)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.72
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.04
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Gradient-based optimization (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _gradient_loop(h_init, d1, radius_init=0.12, beta_init=18.0, max_iter=30000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # LP polish
    if time.time() < d1:
        h1 = _project(best_h.copy(), n0 / 2.0)
        h1, c5_1 = _lp_loop(h1, min(d1, time.time() + budget_s * 0.05), radius_init=0.06, stall_limit=30)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Binary optimization (20%)
    d2 = t0 + budget_s * 0.45
    h_bin = _to_binary(best_h, n0)
    sub_dl = min(d2, time.time() + budget_s * 0.08)
    h_bin, c5_bin = _binary_sa(h_bin, sub_dl, max_iter=40000, T_init=0.0025)
    sub_dl2 = min(d2, time.time() + budget_s * 0.04)
    h_bin, c5_bin = _pair_sweep(h_bin, sub_dl2)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    sub_dl3 = min(d2, time.time() + budget_s * 0.04)
    h_pol = _project(h_bin.copy(), n0 / 2.0)
    h_pol, c5_pol = _gradient_loop(h_pol, sub_dl3, radius_init=0.08, beta_init=25.0, max_iter=8000)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()

    # Phase 3: Multi-n exploration (25%)
    d3 = t0 + budget_s * 0.70
    n_curr = n0
    offsets = list(range(-20, 21, 2))
    rng.shuffle(offsets)
    offsets.sort(key=lambda x: abs(x))
    tried_ns = set([n0])
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n0 + dn
        if n_try < 60 or n_try > 240 or n_try % 2 != 0 or n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _gradient_loop(h_r, sub_dl, radius_init=0.10, beta_init=20.0, max_iter=5000)
        if time.time() < d3 - 5:
            h_b = _to_binary(h_r, n_try)
            sub_dl2 = min(d3, time.time() + budget_s * 0.008)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=10000, T_init=0.002)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5 + 0.002 and time.time() < d3 - 3:
            sub_dl3 = min(d3, time.time() + budget_s * 0.008)
            h_r2 = _project(h_r.copy(), n_try / 2.0)
            h_r2, c5_r2 = _gradient_loop(h_r2, sub_dl3, radius_init=0.06, beta_init=25.0, max_iter=3000)
            if c5_r2 < c5_r:
                c5_r = c5_r2
                h_r = h_r2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 4: Intensive local optimization (20%)
    d4 = t0 + budget_s * 0.90
    n_best = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h4 = _project(best_h.copy(), n_best / 2.0)
    h4, c5_4 = _gradient_loop(h4, sub_dl, radius_init=0.06, beta_init=30.0, max_iter=15000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4:
        h_b = _to_binary(best_h, len(best_h))
        sub_dl2 = min(d4, time.time() + budget_s * 0.04)
        h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=25000, T_init=0.0015)
        sub_dl3 = min(d4, time.time() + budget_s * 0.02)
        h_b, c5_b = _pair_sweep(h_b, sub_dl3)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), len(best_h) / 2.0)
        sub_dl4 = min(d4, time.time() + budget_s * 0.03)
        h4, c5_4 = _lp_loop(h4, sub_dl4, radius_init=0.04, stall_limit=40)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping (10%)
    d5 = deadline
    scale = 0.02
    n_restarts = 0
    while time.time() < d5 - 5 and n_restarts < 80:
        n_restarts += 1
        h_pert = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.006)
        h_t, c5_t = _gradient_loop(h_pert, sub_dl, radius_init=0.05, beta_init=25.0, max_iter=3000)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.06, scale * 1.08)

    # Final polish
    if deadline - time.time() > 5:
        h6 = _project(best_h.copy(), len(best_h) / 2.0)
        h6, c5_6 = _gradient_loop(h6, deadline, radius_init=0.02, beta_init=35.0, max_iter=5000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
