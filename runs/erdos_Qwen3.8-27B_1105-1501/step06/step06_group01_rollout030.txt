def _solve_lp_for_a(b, n):
    """Solve min_a max_q dx*corr(a,b)_q subject to feasibility."""
    dx = 2.0 / n
    # Build matrix M where M[q,i] = dx * (coeff of a_i in corr(a,b)_q)
    # corr(a,b)_q = sum over valid i of a[i] * b[i + (q-(n-1))]
    # So M[q,i] = dx * b[i + (q-(n-1))] if valid, 0 otherwise
    
    M = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        k = q - (n-1)
        if k >= 0:
            # corr = sum_{i=0}^{n-1-k} a[i]*b[i+k]
            for i in range(n-k):
                M[q, i] = dx * b[i+k]
        else:
            # k < 0, let d = -k
            d = -k
            # corr = sum_{i=d}^{n-1} a[i]*b[i-d]
            for i in range(d, n):
                M[q, i] = dx * b[i-d]
    
    # LP: min t s.t. M @ a - t <= 0, sum(a) = n/2, 0<=a<=1, t>=0
    # Using linprog
    c = np.zeros(n+1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    bounds = [(0,1)]*n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def _exact_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _build_M(b_vec):
    """Build matrix M where M[q, i] is the coefficient of a[i] in dx*corr(a, b_vec)[q]."""
    n = len(b_vec)
    dx = 2.0 / n
    M = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                M[q, :m] = dx * b_vec[k:]
        else:
            d_abs = -k
            m = n - d_abs
            if m > 0:
                M[q, d_abs:] = dx * b_vec[:m]
    return M

def _lp_solve(M, target):
    """Solve min t s.t. M@x <= t, sum(x)=target, 0<=x<=1, t>=0."""
    n = M.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((M.shape[0], 1))])
    b_ub = np.zeros(M.shape[0])

    b_eq = np.array([float(target)])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.clip(res.x[:n], 0.0, 1.0)
    except Exception:
        pass
    return None

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b_pts = t_lo, t_hi
    c1 = b_pts - gr * (b_pts - a)
    c2 = a + gr * (b_pts - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(40):
        if b_pts - a < 1e-13:
            break
        if f1 < f2:
            b_pts, c2, f2 = c2, c1, f1
            c1 = b_pts - gr * (b_pts - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b_pts - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b_pts) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))

Actually, I realize there's an issue. The LP `min_a max_q dx*corr(a,b)_q` gives us the best `a` for a fixed `b`. But `corr(a,b)` is not the same as `corr(a, 1-a)`. The objective we want to minimize is `max_q dx*corr(a, 1-a)_q`, not `max_q dx*corr(a, b)_q` for an arbitrary `b`.

The relaxed approach works as follows:
- We want to minimize f(h) = max_q dx*corr(h, 1-h)_q
- We relax by separating h and (1-h) into independent variables a and b
- The relaxed objective is: max_q dx*corr(a, b)_q (which is ≤ f at the true solution where b = 1-a)
- By solving the LP alternately, we find a "min-max" pair (a*, b*)
- The true solution h* should satisfy h* = a* and 1-h* = b*, i.e., a* + b* = 1
- If a* + b* ≠ 1, we're at a non-trivial relaxed solution

The key insight: even when a* + b* ≠ 1, the candidates a*, 1-b*, and interpolations are valid feasible points, and they might be better than the current h.

This is similar to the "dual decomposition" approach in optimization.

Let me also think about whether we can get a lower bound from this. The LP gives us:
LP_value(a*, b*) = max_q dx * corr(a*, b*)_q

And for any feasible h: f(h) ≥ LP_value(h, 1-h) (trivially, since h and 1-h are feasible pairs)

But also: max_q dx*corr(a*, b*)_q ≤ max_q dx*corr(h, 1-h)_q for any h that is "better" in some sense...

Actually, the LP value is a lower bound on the minimum of f over the feasible set defined by the two separate feasibility constraints. But since the feasible sets for a and b are the same (both in [0,1]^n with sum n/2), the LP relaxation is exact in the sense that the feasible region for (a,b) is the product of the feasible region for h with itself. The original problem has the additional constraint b = 1-a.

So the LP value is a valid lower bound: min_h f(h) ≥ LP_value.

If we can get LP_value close to the current best f(h), we know we're near optimal.

import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _build_M(b_vec):
    n = len(b_vec)
    dx = 2.0 / n
    M = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                M[q, :m] = dx * b_vec[k:]
        else:
            d_abs = -k
            m = n - d_abs
            if m > 0:
                M[q, d_abs:] = dx * b_vec[:m]
    return M

def _lp_solve(M, target):
    n = M.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((M.shape[0], 1))])
    b_ub = np.zeros(M.shape[0])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([float(target)])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.clip(res.x[:n], 0.0, 1.0)
    except Exception:
        pass
    return None

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt, b_pt = t_lo, t_hi
    c1 = b_pt - gr * (b_pt - a_pt)
    c2 = a_pt + gr * (b_pt - a_pt)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))

            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))

            candidates.append(a_proj)
            if b_star is not None:
                b_proj = _project(b_star, target)
                u_R = _project(1.0 - b_proj, target)
                candidates.append(u_R)
                for alpha in [0.3, 0.5, 0.7]:
                    interp = (1.0 - alpha) * u_R + alpha * a_proj
                    candidates.append(_project(interp, target))
        elif b_star is not None:
            b_proj = _project(b_star, target)
            candidates.append(_project(1.0 - b_proj, target))

        improved = False
        for cand in candidates:
            if time.time() > deadline:
                break
            d = cand - h
            if np.max(np.abs(d)) < 1e-10:
                continue
            alpha, c5_try = _line_search(h, d, dx, 1.0)
            if c5_try < best_c5 - 1e-12:
                h_new = h + alpha * d
                h_new = _project(h_new, target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    improved = True
        if not improved:
            if a_star is not None:
                h_new = _project(a_star, target)
                c5_new = _exact_c5(h_new)
                if c5_new <= best_c5 + 1e-10:
                    h = h_new
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new.copy()
            elif b_star is not None:
                h_new = _project(1.0 - b_star, target)
                c5_new = _exact_c5(h_new)
                if c5_new <= best_c5 + 1e-10:
                    h = h_new
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new.copy()
        if improved:
            stall = 0
        else:
            stall += 1
        h_sig = tuple(np.round(h[::10], 4))
        history.append(h_sig)
        if len(history) > 30:
            history.pop(0)
        if stall > 80 and len(history) >= 20:
            if history[-1] == history[-10]:
                break
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            stall = 0
        if stall > 300:
            break
    return best_h, best_c5

def _diverse_start(n, rng):
    mode = rng.integers(0, 5)
    x = np.arange(n)
    if mode == 0:
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 1:
        h = np.zeros(n)
        split = n // 2
        perm = rng.permutation(n)
        h[perm[:split]] = 1.0
        h = h + rng.normal(0, 0.1, n)
    elif mode == 2:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x / n) + 0.2 * np.sin(2 * np.pi * 5 * x / n)
        h = h + rng.normal(0, 0.08, n)
    elif mode == 3:
        num_blocks = rng.integers(3, 10)
        block_sizes = rng.integers(3, n // num_blocks + 2, size=num_blocks)
        total = block_sizes.sum()
        h = np.zeros(n)
        pos = 0
        for bs in block_sizes:
            actual = min(bs, n - pos)
            if actual <= 0:
                break
            if rng.random() < 0.5:
                h[pos:pos + actual] = 1.0
            pos += actual + rng.integers(0, 3)
        h = h + rng.normal(0, 0.05, n)
    else:
        h = rng.uniform(0.1, 0.9, n)
    h = _project(h, n / 2.0)
    return h

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _lp_direction_optimize(h, deadline, start_radius=0.10):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = start_radius
    stall = 0
    rng = np.random.default_rng(999)
    while time.time() < deadline:
        if stall > 60:
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            radius = 0.08
            stall = 0
            if stall > 200:
                break
        S = _corr_vals(h)
        max_q = int(np.argmax(S))
        k = max_q - (n - 1)
        d = np.zeros(n)
        if k >= 0:
            m = n - k
            d[:m] += (1.0 - h[k:k + m])
            d[k:n] -= h[:m]
        else:
            da = -k
            m = n - da
            d[da:n] += (1.0 - h[:m])
            d[:m] -= h[da:da + m]
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-10:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        d = d / dnorm * radius
        alpha, c5_try = _line_search(h, d, dx, radius)
        if c5_try < best_c5 - 1e-13:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            stall = 0
            radius = min(radius * 1.05, 0.30)
        else:
            stall += 1
            radius *= 0.85
            if stall % 30 == 0 and radius < 0.01:
                radius = 0.05
        if stall > 400:
            break
    return best_h, best_c5

def _pair_edges(h, deadline, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(314)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_vals(h)
        max_q = int(np.argmax(S))
        k = max_q - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            m = n - k
            grad[:m] += (1.0 - h[k:k + m])
            grad[k:n] -= h[:m]
        else:
            da = -k
            m = n - da
            grad[da:n] += (1.0 - h[:m])
            grad[:m] -= h[da:da + m]
        order = np.argsort(-np.abs(grad))
        tried = 0
        for ii in range(min(30, len(order))):
            if time.time() > deadline:
                break
            i = order[ii]
            for jj in range(min(30, len(order))):
                if tried > 60:
                    break
                j = order[jj]
                if i == j:
                    continue
                for sign in [1, -1]:
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    alpha, c5_try = _line_search(h, d, dx, 0.5)
                    if c5_try < best_c5 - 1e-13:
                        h_new = h + alpha * d
                        h_new = _project(h_new, target)
                        c5_new = _exact_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new.copy()
                            improved = True
                    tried += 1
        if not improved:
            for _ in range(10):
                if time.time() > deadline:
                    break
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                for sign in [1, -1]:
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    alpha, c5_try = _line_search(h, d, dx, 0.4)
                    if c5_try < best_c5 - 1e-13:
                        h_new = h + alpha * d
                        h_new = _project(h_new, target)
                        c5_new = _exact_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new.copy()
                            improved = True
            if not improved:
                break
    return best_h, best_c5

def _make_fallback(n):
    x = np.arange(n)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / n) + 0.15 * np.cos(2 * np.pi * 7 * x / n)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        n0 = 122
        h0 = _make_fallback(n0)
    h0 = _project(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Relaxed LP from parent (25% budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _relaxed_optimize(best_h, d1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP direction refinement (15% budget)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _lp_direction_optimize(best_h, d2, start_radius=0.10)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Diverse starts with relaxed LP (20% budget)
    d3 = t0 + budget_s * 0.60
    n_starts = 6
    for si in range(n_starts):
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + max(5.0, budget_s * 0.025))
        h_div = _diverse_start(n0, rng)
        h_div, c5_div = _relaxed_optimize(h_div, sub_dl, rng)
        if c5_div < best_c5:
            best_c5 = c5_div
            best_h = h_div.copy()

    # Phase 4: Pair/edge optimization (10% budget)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _pair_edges(best_h, d4, max_passes=200)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (12% budget)
    d5 = t0 + budget_s * 0.82
    n_cur = len(best_h)
    tried_n = set()
    offsets = [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, 2, -2, 16, -16]
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_cur + dn
        if n_try < 20 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _relaxed_optimize(h_r, sub_dl, rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try

    # Phase 6: Smooth gradient polish (5% budget)
    d6 = t0 + budget_s * 0.88
    h6, c5_6 = _smooth_descent(best_h, d6)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final relaxed LP + smooth (7% budget)
    d7 = deadline
    h7, c5_7 = _relaxed_optimize(best_h, d7, rng)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    h7b, c5_7b = _smooth_descent(best_h, deadline)
    if c5_7b < best_c5:
        best_c5 = c5_7b
        best_h = h7b.copy()

    # Final repair
    n = len(best_h)
    best_h = _project(best_h, n / 2.0)
    final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
