import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_full_grads(h):
    n = len(h)
    C = 2 * n - 1
    G = np.zeros((C, n))
    for ell in range(0, n):
        m = n - ell
        r1 = n - 1 + ell
        if ell == 0:
            G[r1, :] = 1.0 - 2.0 * h[:]
        else:
            G[r1, ell:] += 1.0 - h[:m]
            G[r1, :m] += -h[ell:]
        if ell > 0:
            r2 = n - 1 - ell
            G[r2, :m] += 1.0 - h[ell:]
            G[r2, ell:] += -h[:m]
    return G

def _line_search(h, d, S0, dx):
    """Exact line search along direction d from base h."""
    n = len(h)
    # Compute bounds
    pos = d > 1e-12
    neg = d < -1e-12
    hi = 1.0
    lo = -1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    
    def f(tau):
        return float(np.max(S0 + tau * D - tau * tau * Q)) * dx
    
    # Grid search
    n_grid = 50
    taus = np.linspace(lo, hi, n_grid)
    vals = np.array([f(t) for t in taus])
    best_idx = np.argmin(vals)
    
    # Refine around best
    if best_idx == 0:
        a, b = lo, taus[1]
    elif best_idx == n_grid - 1:
        a, b = taus[-2], hi
    else:
        a, b = taus[best_idx - 1], taus[best_idx + 1]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    f1, f2 = f(x1), f(x2)
    for _ in range(50):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, x2, f2 = x2, x1, f1
            x1 = b - gr * (b - a)
            f1 = f(x1)
        else:
            a, x1, f1 = x1, x2, f2
            x2 = a + gr * (b - a)
            f2 = f(x2)
    
    opt = (a + b) / 2.0
    best_val = min(f(opt), f(lo), f(hi))
    if f(opt) < best_val:
        opt_val = f(opt)
    else:
        opt_val = best_val
    
    # Check endpoints
    if f(lo) < opt_val:
        opt, opt_val = lo, f(lo)
    if f(hi) < opt_val:
        opt, opt_val = hi, f(hi)
    
    return opt, opt_val

def _qp_step(h, target_raw, rho, S0, G, dx):
    """Solve QP to find direction reducing all rows below target."""
    n = len(h)
    C = 2 * n - 1
    
    # Variable bounds for d
    d_lo = -np.minimum(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    
    # z upper bound
    z_max = float(np.max(np.abs(S0))) + 2.0 * target_raw + 2.0
    
    # Initial point
    d0 = np.zeros(n)
    z0 = np.maximum(S0 - target_raw, 0.0)
    x0 = np.concatenate([d0, z0])
    
    # Bounds
    bounds = list(zip(d_lo, d_hi)) + [(0.0, z_max)] * C
    
    # Objective and gradient
    lam_z = 0.7
    lam_norm = 0.3
    scale_z = max(float(np.mean(z0)), 1e-6)
    
    def obj(x):
        d = x[:n]
        z = x[n:]
        return lam_z * (np.sum(z) / (C * scale_z)) + lam_norm * (np.sum(d * d) / (rho * rho))
    
    def obj_grad(x):
        d = x[:n]
        z = x[n:]
        g = np.zeros(len(x))
        g[:n] = 2.0 * lam_norm * d / (rho * rho)
        g[n:] = lam_z / (C * scale_z)
        return g
    
    # Constraints
    # Equality: sum(d) = 0
    # Inequality: S0[k] + G[k]@d - target - z[k] <= 0
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]), 'jac': lambda x: np.concatenate([np.ones(n), np.zeros(C)])}]
    
    for k in range(C):
        Gk = G[k, :].copy()
        cons.append({
            'type': 'ineq',
            'fun': lambda x, k=k: target_raw + x[k+n] - S0[k] - np.dot(Gk, x[:n]),
            'jac': lambda x, k=k: np.concatenate([-Gk, np.zeros(k), [1.0], np.zeros(C-k-1)])
        })
    
    result = minimize(obj, x0, jac=obj_grad, bounds=bounds, constraints=cons,
                      method='SLSQP', options={'maxiter': 200, 'ftol': 1e-10})
    
    if result.success or (result.fun < obj(x0)):
        d = result.x[:n].copy()
        d -= np.mean(d)  # enforce zero sum
        if np.max(np.abs(d)) > 1e-12:
            return d
    return None

        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    def f_eval(tau):
        return float(np.max(S0 + tau * D - tau * tau * Q)) * dx
    n_grid = 60
    taus = np.linspace(lo, hi, n_grid)
    vals = np.empty(n_grid)
    for idx in range(n_grid):
        vals[idx] = f_eval(taus[idx])
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        a, b = lo, taus[1]
    elif best_idx == n_grid - 1:
        a, b = taus[-2], hi
    else:
        a, b = taus[best_idx - 1], taus[best_idx + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    f1, f2 = f_eval(x1), f_eval(x2)
    for _ in range(55):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, x2, f2 = x2, x1, f1
            x1 = b - gr * (b - a)
            f1 = f_eval(x1)
        else:
            a, x1, f1 = x1, x2, f2
            x2 = a + gr * (b - a)
            f2 = f_eval(x2)
    opt = (a + b) / 2.0
    opt_val = f_eval(opt)
    f_lo = f_eval(lo)
    f_hi = f_eval(hi)
    if f_lo < opt_val:
        opt, opt_val = lo, f_lo
    if f_hi < opt_val:
        opt, opt_val = hi, f_hi
    return opt, opt_val

def _smoothed_gradient_step(h, S0, G, beta, step_size, dx):
    n = len(h)
    C = 2 * n - 1
    max_s0 = float(np.max(S0))
    weights = np.exp(beta * (S0 - max_s0))
    wsum = float(np.sum(weights))
    if wsum < 1e-30:
        return None
    g = (G.T @ (weights / wsum))
    g -= np.mean(g)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return None
    d = -g / gn
    return d

def _pair_golden(h, i, j, S0, dx):
    n = len(h)
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C2 = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C2)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C2)) * dx
    for _ in range(35):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C2)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C2)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C2)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C2)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C2)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _directed_pair_search(h, deadline, n_pairs=20):
    n = len(h)
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(12345)

    while time.time() < deadline and stall < 60:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.008 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:8]

        G = _compute_full_grads(h)
        grads = G[active_idx, :]
        weight = np.exp((scores[active_idx] - max_score) / (0.02 * max_score + 1e-15))
        weight /= float(np.sum(weight))
        wgrad = grads.T @ weight

        donor_pool = np.argsort(-wgrad)[:n_pairs]
        recv_pool = np.argsort(wgrad)[:n_pairs]

        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        tried = set()

        for i in donor_pool:
            if time.time() > deadline:
                break
            for j in recv_pool:
                if i == j:
                    continue
                key = (min(int(i), int(j)), max(int(i), int(j)))
                if key in tried:
                    continue
                tried.add(key)
                alpha, new_c5 = _pair_golden(h, int(i), int(j), S0, dx)
                if alpha > 1e-10 and new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (int(i), int(j))

        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, n / 2.0)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.4:
                idx = rng.integers(n, size=3)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.03, 0.03)
                h = project(h, n / 2.0)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h = None

    if h is None or len(h) < 20:
        n_init = 122
        x = np.arange(n_init)
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * 2.0 * x / n_init + 0.3)
        h = project(h, n_init / 2.0)

    n = len(h)
    h = project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n

    # Phase 1: QP-based full-direction optimization
    rhos = [0.25, 0.18, 0.12, 0.08, 0.05, 0.035, 0.02, 0.012]
    lam_norms = [0.003, 0.008, 0.02, 0.05]
    stall_count = 0
    max_stall = 40

    while time.time() < deadline and stall_count < max_stall:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        S0 = np.correlate(best_h, 1.0 - best_h, mode='full')
        current_max_raw = float(np.max(S0))
        G = _compute_full_grads(best_h)
        improved = False

        for rho in rhos:
            if time.time() > deadline:
                break
            for lam in lam_norms:
                if time.time() > deadline:
                    break
                target_raw = current_max_raw - 0.005 * current_max_raw
                d = _qp_direction(best_h, target_raw, rho, G, S0, lam_norm=lam)
                if d is None:
                    continue
                tau, tau_c5 = _line_search_dir(best_h, d, S0, dx_cur)
                if tau > 1e-10 and tau_c5 < best_c5 - 1e-12:
                    h_new = best_h + tau * d
                    h_new = project(h_new, n_cur / 2.0)
                    actual_c5 = c5_exact(h_new)
                    if actual_c5 < best_c5 - 1e-12:
                        best_h = h_new
                        best_c5 = actual_c5
                        improved = True
                        break
            if improved:
                break

        if improved:
            stall_count = 0
        else:
            stall_count += 1
            # Try smoothed gradient step
            if stall_count % 3 == 0 and time.time() < deadline:
                S0_r = np.correlate(best_h, 1.0 - best_h, mode='full')
                G_r = _compute_full_grads(best_h)
                for beta in [20.0, 50.0, 100.0]:
                    d_sm = _smoothed_gradient_step(best_h, S0_r, G_r, beta, 0.01, dx_cur)
                    if d_sm is not None:
                        tau, tau_c5 = _line_search_dir(best_h, d_sm, S0_r, dx_cur)
                        if tau > 1e-10 and tau_c5 < best_c5 - 1e-12:
                            h_new = best_h + tau * d_sm
                            h_new = project(h_new, n_cur / 2.0)
                            actual_c5 = c5_exact(h_new)
                            if actual_c5 < best_c5 - 1e-12:
                                best_h = h_new
                                best_c5 = actual_c5
                                stall_count = 0
                                break

    # Phase 2: Directed pair search to polish
    d2 = min(deadline, time.time() + budget_s * 0.25)
    h_p2, c5_p2 = _directed_pair_search(best_h, d2, n_pairs=25)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()
        best_n = len(best_h)

    # Phase 3: Multi-resolution search
    if time.time() < deadline:
        n_cur = len(best_h)
        tried_ns = {n_cur}
        for dn in [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20]:
            if time.time() > deadline:
                break
            n_try = n_cur + dn
            if n_try < 30 or n_try > 250 or n_try % 2 != 0:
                continue
            if n_try in tried_ns:
                continue
            tried_ns.add(n_try)
            h_r = _resample(best_h, n_try)
            c5_r = c5_exact(h_r)
            if c5_r < best_c5:
                sub_dl = min(deadline, time.time() + budget_s * 0.03)
                h_r, c5_r2 = _directed_pair_search(h_r, sub_dl, n_pairs=20)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r.copy()
                    best_n = n_try

    # Phase 4: Final QP polish on best
    if time.time() < deadline * 0.85:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        S0 = np.correlate(best_h, 1.0 - best_h, mode='full')
        G = _compute_full_grads(best_h)
        for rho in [0.01, 0.007, 0.004, 0.002]:
            if time.time() > deadline:
                break
            current_max_raw = float(np.max(S0))
            target_raw = current_max_raw - 0.003 * current_max_raw
            d = _qp_direction(best_h, target_raw, rho, G, S0, lam_norm=0.01)
            if d is not None:
                tau, tau_c5 = _line_search_dir(best_h, d, S0, dx_cur)
                if tau > 1e-10 and tau_c5 < best_c5 - 1e-12:
                    h_new = best_h + tau * d
                    h_new = project(h_new, n_cur / 2.0)
                    actual_c5 = c5_exact(h_new)
                    if actual_c5 < best_c5 - 1e-12:
                        best_h = h_new
                        best_c5 = actual_c5

    # Phase 5: Final short pair polish with remaining time
    remaining = deadline - time.time()
    if remaining > 5:
        d5 = time.time() + max(3, remaining * 0.7)
        h_p5, c5_p5 = _directed_pair_search(best_h, d5, n_pairs=30)
        if c5_p5 < best_c5:
            best_c5 = c5_p5
            best_h = h_p5.copy()

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)

    return best_h, final_c5, len(best_h)
