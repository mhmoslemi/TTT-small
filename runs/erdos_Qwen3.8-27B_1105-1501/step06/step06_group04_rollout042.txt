        cur_c5 = float(C.max())
        max_val = cur_c5
        
        # Find active set (lags within 1% of max)
        active = C >= max_val * 0.999
        n_active = active.sum()
        if n_active == 0:
            active[np.argmax(C)] = True
        
        # Subgradient: average gradient of active constraints
        # For lag d (shift = d - (n-1)):
        # ∂C[d]/∂h[p] = dx * [(1-h[p+shift]) * valid1 - h[p-shift] * valid2]
        # where valid1 = (p+shift in [0,n) and p in valid_i_range)
        #       valid2 = (p-shift in [0,n) and p-shift in valid_i_range)
        
        g = np.zeros(n)
        for d_idx in np.where(active)[0]:
            shift = d_idx - (n - 1)
            if shift >= 0:
                # C[d] = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift]) * dx
                # ∂/∂h[p]:
                #   from h[i] term (i=p): (1-h[p+shift]) if p <= n-1-shift
                #   from h[i+shift] term (i+shift=p, i=p-shift): -h[p-shift] if shift <= p <= n-1
                valid_i = np.arange(n-shift)
                # Contribution to h[p] from first term: (1-h[p+shift]) for p in [0, n-1-shift]
                # Contribution to h[p] from second term: -h[p-shift] for p in [shift, n-1]
                mask1 = np.arange(n) <= (n-1-shift)
                mask2 = np.arange(n) >= shift
                g += dx * ((1-h[np.clip(np.arange(n)+shift, 0, n-1)]) * mask1 - 
                           h[np.clip(np.arange(n)-shift, 0, n-1)] * mask2)
            else:
                m = -shift
                # C[d] = sum_{i=m}^{n-1} h[i]*(1-h[i-m]) * dx
                # ∂/∂h[p]:
                #   from h[i] term (i=p): (1-h[p-m]) if p >= m
                #   from h[i-m] term (i-m=p, i=p+m): -h[p+m] if p <= n-1-m
                mask1 = np.arange(n) >= m
                mask2 = np.arange(n) <= (n-1-m)
                g += dx * ((1-h[np.clip(np.arange(n)-m, 0, n-1)]) * mask1 - 
                           h[np.clip(np.arange(n)+m, 0, n-1)] * mask2)

                break

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    stall = 0
    rng = np.random.default_rng(42)
    while time.time() < deadline and stall < 40:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(300):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            for sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                alpha, c5_try = _line_search_c5(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                    break
            if improved:
                break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _lp_direction(h, scores, radius, top_k=12):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = np.zeros((K, n))
    for idx, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            mm = n - d
            if mm > 0:
                G[idx, d:] += (1.0 - h[:mm])
                G[idx, :mm] += -h[d:]
        else:
            gg = -d
            mm = n - gg
            if mm > 0:
                G[idx, :mm] += (1.0 - h[gg:])
                G[idx, gg:] += -h[:mm]
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.1, top_k=12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline and stall < 60:
        scores = _corr_values(h)
        d = _lp_direction(h, scores, radius, top_k=top_k)
        if d is not None and np.max(np.abs(d)) > 1e-13:
            alpha, new_c5 = _line_search_c5(h, d, dx, radius)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                radius = min(radius * 1.05, 0.3)
                continue
        # Subgradient step
        max_val = float(scores.max())
        thr = max_val - 0.01 * max_val
        active = np.where(scores >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        G = np.zeros((len(active), n))
        idx = np.arange(n)
        for ai, r in enumerate(active):
            d_shift = r - (n - 1)
            if d_shift >= 0:
                mm = n - d_shift
                if mm > 0:
                    G[ai, d_shift:] += (1.0 - h[:mm])
                    G[ai, :mm] += -h[d_shift:]
            else:
                gg = -d_shift
                mm = n - gg
                if mm > 0:
                    G[ai, :mm] += (1.0 - h[gg:])
                    G[ai, gg:] += -h[:mm]
        avg_g = G.mean(axis=0)
        avg_g -= avg_g.mean()
        gn = np.linalg.norm(avg_g)
        if gn > 1e-12:
            d2 = -avg_g / gn * radius
            alpha, new_c5 = _line_search_c5(h, d2, dx, radius * 2)
            if new_c5 < cur_c5 - 1e-14:
                h = h + alpha * d2
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                continue
        stall += 1
        radius *= 0.85
        if radius < 0.004:
            radius = 0.08
        if stall > 20:
            h = h + rng.normal(0, 0.02, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.1
            stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = _sine_init(122, 3, 0.5)
    n0 = len(h0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)

    # Phase 1: Binary local search (50% time)
    d1 = t0 + budget_s * 0.50
    h_bin = (h0 > 0.5).astype(float)
    if n0 % 2 != 0:
        target_n = n0 + 1 if n0 + 1 <= n0 + 2 else n0 - 1
        h_bin = _resample(h0, target_n)
        h_bin = (h_bin > 0.5).astype(float)
    hb, c5b = _binary_local_search(h_bin, min(d1, deadline - 5), rng,
                                   sa_T0=0.005, sa_Tmin=0.00002)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = hb.copy()

    # Phase 1b: More binary search from perturbations
    if time.time() < d1 - 10:
        for _ in range(3):
            if time.time() > d1 - 5:
                break
            h_pert = best_h + rng.normal(0, 0.3, len(best_h))
            h_pert_bin = (h_pert > 0.5).astype(float)
            if len(h_pert_bin) % 2 == 0:
                hb2, c5b2 = _binary_local_search(h_pert_bin, min(d1, time.time() + budget_s * 0.05), rng,
                                                 sa_T0=0.004, sa_Tmin=0.00005)
                if c5b2 < best_c5:
                    best_c5 = c5b2
                    best_h = hb2.copy()

    # Phase 2: Continuous LP refinement (20% time)
    d2 = t0 + budget_s * 0.70
    h_lp, c5_lp = _lp_loop(best_h, d2, radius_init=0.08, top_k=12)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()

    # Phase 3: Subgradient refinement
    if time.time() < d2 + budget_s * 0.05:
        d3 = t0 + budget_s * 0.75
        h_sg, c5_sg = _subgradient_refine(best_h, d3, lr_init=0.04)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 4: Pair refinement
    if time.time() < deadline - 20:
        d4 = t0 + budget_s * 0.82
        h_pr, c5_pr = _pair_refine(best_h, d4, max_passes=50)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr.copy()

    # Phase 5: Multi-resolution (15% time)
    d5 = t0 + budget_s * 0.95
    best_n = len(best_h)
    offsets = [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -18, 18, -22, 22,
               -26, 26, -30, 30, -36, 36, -44, 44, -50, 50]
    tried_n = set()
    for dn in offsets:
        if time.time() > d5 - 5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 300 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.02)
        h_r_bin = (h_r > 0.5).astype(float)
        hb_r, c5b_r = _binary_local_search(h_r_bin, sub_dl, rng, sa_T0=0.003, sa_Tmin=0.0001)
        if c5b_r < best_c5:
            best_c5 = c5b_r
            best_h = hb_r.copy()
            best_n = n_try
        if time.time() < d5 - 10:
            h_lp_r, c5_lp_r = _lp_loop(best_h, min(d5, time.time() + budget_s * 0.01),
                                        radius_init=0.05, top_k=8)
            if c5_lp_r < best_c5:
                best_c5 = c5_lp_r
                best_h = h_lp_r.copy()
                best_n = len(h_lp_r)

    # Phase 6: Final polish
    if time.time() < deadline - 2:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.03, top_k=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        h_fb = _sine_init(122, 3, 0.0)
        return h_fb, _exact_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
