def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def exact_line_search(h, d, S0, dx):
    """Exact 1D line search along direction d (zero-sum). Returns (best_alpha, best_c5)."""
    n = len(h)
    d = d - np.mean(d)  # ensure zero sum
    if np.linalg.norm(d) < 1e-15:
        return 0.0, float(np.max(S0) * dx)
    
    # Compute bounds
    pos = d > 1e-12
    neg = d < -1e-12
    hi = 1e10
    lo = -1e10
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    if hi <= lo + 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Precompute T and Q
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    
    def f(alpha):
        val = S0 + alpha * T - alpha * alpha * Q
        return float(np.max(val) * dx)
    
    # Include alpha=0
    best_alpha = 0.0
    best_val = f(0.0)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    # Also try some grid points
    n_grid = 20
    grid_als = np.linspace(lo, hi, n_grid)
    for ga in grid_als:
        fv = f(ga)
        if fv < best_val:
            best_val = fv
            best_alpha = ga
    
    # Golden section for refinement
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    mid = (a + b) / 2.0
    fm = f(mid)
    if fm < best_val:
        best_val = fm
        best_alpha = mid
    # Check endpoints
    for ep in [lo, hi]:
        fe = f(ep)
        if fe < best_val:
            best_val = fe
            best_alpha = ep
    
    return best_alpha, best_val
