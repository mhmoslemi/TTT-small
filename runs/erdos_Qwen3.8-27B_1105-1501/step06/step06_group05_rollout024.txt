            f1 = _eval(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _eval(c2)
    
    t_gs = 0.5 * (a + b)
    f_gs = _eval(t_gs)
    if f_gs < f_best:
        t_best, f_best = t_gs, f_gs
    
    # Also check t=0
    f0 = float(np.max(S)) * dx
    if f0 < f_best:

def _slp_direction(h, corr, n_lags=243):
    # SLP: solve LP to minimize max of linearized lags
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    
    # Build gradient matrix for all lags
    # For lag k (0-indexed in output, corresponding to shift k - (n-1)):
    # f_k = sum_i h[i] * (1-h[i+k]) for appropriate ranges
    # Gradient wrt h[j]:
    #   contributes -h[j+k] where h[j] is the first factor
    #   contributes (1-h[j-k]) where h[j] is the second factor (in 1-h)
    
    # More carefully: output index r corresponds to shift k = r - (n-1)
    # For k >= 0: f_k = sum_{i=0}^{n-k-1} h[i+k] * (1-h[i])
    #   df_k/dh[j] = (1-h[j-k]) if j in [k, n-1] (from first factor, i = j-k)
    #                + (-h[j+k]) if j in [0, n-k-1] (from second factor, i = j)
    # For k < 0: let q = -k > 0, f_k = sum_{i=0}^{n-q-1} h[i] * (1-h[i+q])
    #   df_k/dh[j] = (1-h[j+q]) if j in [0, n-q-1] (from first factor)
    #                + (-h[j-q]) if j in [q, n-1] (from second factor, i = j-q)
    
    # Actually, let me use the formula from the previous code which was verified:
    # For output r, k = r - (n-1):
    #   if k >= 0: nn = arange(0, n-k); a = nn+k; b = nn
    #     grad[b] += -h[a]; grad[a] += (1-h[b])
    #   if k < 0: nn = arange(-k, n); a = nn+k; b = nn
    #     grad[b] += -h[a]; grad[a] += (1-h[b])
    
    # This is O(M * n) which is about 243 * 122 ≈ 30000 operations. Fine.
    
    # Build A_ub matrix (M rows x (n+1) cols) for LP
    # Variables: [x_0, ..., x_{n-1}, tau]
    # Constraint: sum_i G[r,i] * x_i - tau <= sum_i G[r,i] * h_i - corr[r]
    # i.e., G[r,:] @ x - tau <= G[r,:] @ h - corr[r]
    
    # For efficiency, build G as a matrix
    G = np.zeros((M, n))
    for r in range(M):
        k = r - (n - 1)
        row = G[r, :]
        if k >= 0:
            nn = np.arange(0, n - k)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
        else:
            nn = np.arange(-k, n)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
    
    # A_ub: M x (n+1), each row has G[r,:] and -1 in last col
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = G @ h - corr  # = G[r,:] @ h - corr[r]
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize tau
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    
    # Bounds: 0 <= x[i] <= 1, tau >= 0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            x_opt = res.x[:n]
            d = x_opt - h
            d -= d.mean()  # Ensure zero mean (should already be, but safety)
            nrm = float(np.linalg.norm(d))
            if nrm > 1e-12:
                return d / nrm
    except Exception:
        pass
    return None

def _active_gradient_direction(h, corr, k_active, tau_temp, rng):
    n = len(h)
    M = 2 * n - 1
    scores = corr
    m = float(np.max(scores))
    w = np.exp((scores - m) / tau_temp)
    w /= w.sum()
    
    idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = r - (n - 1)
        row = np.zeros(n)
        if k >= 0:
            nn = np.arange(0, n - k)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
        else:
            nn = np.arange(-k, n)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
        G += w[r] * row
    
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm

def _pair_move_optimize(h, i, j):
    # Optimize: h[i] + t, h[j] - t
    n = len(h)
    dx = 2.0 / n
    # Feasible range for t
    L = max(-h[i], -(1.0 - h[j]))  # h[j] - t >= 0 → t <= h[j]; h[i]+t <= 1 → t <= 1-h[i]
    U = min(1.0 - h[i], h[j])       # h[i]+t >= 0 → t >= -h[i]; h[j]-t >= 0 → t <= h[j]
    
    if U - L < 1e-14:
        return 0.0, float(np.max(np.correlate(h, 1.0-h, mode='full'))) * dx
    
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    # Minimize max_k (S_k + t*T_k - t^2*Q_k) over t in [L, U]
    # Use grid + golden
    n_grid = 60
    grid = np.linspace(L, U, n_grid)
    vals = S[None, :] + grid[:, None] * T[None, :] - (grid[:, None]**2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    idx = int(np.argmin(obj))
    
    best_t = float(grid[idx])
    best_f = float(obj[idx])
    
    # Golden refine
    a = grid[max(idx-1, 0)]
    b = grid[min(idx+1, n_grid-1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    
    def _eval_pt(t):
        return float(np.max(S + t * T - t*t * Q)) * dx
    
    f1 = _eval_pt(c1)
    f2 = _eval_pt(c2)
    for _ in range(30):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _eval_pt(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _eval_pt(c2)
    
    t_gs = 0.5 * (a + b)
    f_gs = _eval_pt(t_gs)
    if f_gs < best_f:
        best_t, best_f = t_gs, f_gs
    
    # Check t=0
    f0 = float(np.max(S)) * dx
    if f0 < best_f:
        best_t, best_f = 0.0, f0
    
    return best_t, best_f

def _optimize(h, deadline, rng, use_slp=False, slp_frac=0.3, pair_frac=0.25, rand_frac=0.10,
              k_active=8, tau_temp=0.02, min_time_per_iter=0.05):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    restarts = 0
    sigma = 0.04
    uphill = 3e-4
    iter_count = 0
    
    while time.time() < deadline:
        iter_count += 1
        r = rng.random()
        d = None
        
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            # Pair move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                stall += 1
                continue
            t_move, f_pair = _pair_move_optimize(h, i, j)
            if abs(t_move) > 1e-10:
                hn = h.copy()
                hn[i] += t_move
                hn[j] -= t_move
                if hn.min() >= -1e-12 and hn.max() <= 1.0 + 1e-12:
                    hn = np.clip(hn, 0.0, 1.0)
                    hn = _project(hn, target)
                    nf = _exact_c5(hn)
                    if nf < best_f:
                        best_f = nf
                        best_h = hn.copy()
                        corr = np.correlate(hn, 1.0 - hn, mode='full')
                    if nf < cur_f + uphill:
                        h = hn
                        cur_f = nf
                    else:
                        h = best_h.copy()
                        cur_f = best_f
                        corr = np.correlate(h, 1.0 - h, mode='full')
                    stall = 0
                    continue
            stall += 1
            continue
        elif use_slp and r < rand_frac + pair_frac + slp_frac:
            d = _slp_direction(h, corr)
            if d is None:
                d = rng.standard_normal(n)
                d -= d.mean()
        else:
            d = _active_gradient_direction(h, corr, k_active, tau_temp, rng)
        
        if d is not None:
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                stall += 1
                continue
            d = d / nrm
            L, U = _feasible_bounds(h, d)
            if U - L < 1e-11:
                # Try opposite direction
                d = -d
                L, U = _feasible_bounds(h, d)
                if U - L < 1e-11:
                    stall += 1
                    continue
            
            t_opt, obj_val = _line_search(h, d, L, U)
            if abs(t_opt) < 1e-14:
                stall += 1
                continue
            
            hn = h + t_opt * d
            if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
                hn = _project(hn, target)
            nf = _exact_c5(hn)
            
            if nf < best_f:
                best_f = nf
                best_h = hn.copy()
            if nf < cur_f + uphill:
                h = hn
                cur_f = nf
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
            else:
                h = best_h.copy()
                cur_f = best_f
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall += 1
        
        if stall > 100:
            restarts += 1
            if restarts > 40:
                h = _project(best_h + rng.normal(0, max(0.03, sigma * 3), n), target)
            else:
                h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.003, sigma * 0.9)
    
    return best_h, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 6))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _binary_random_init(n, rng):
    h = rng.integers(0, 2, n).astype(float)
    h += rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)

def _kronecker_init(n, alpha, rng):
    # Kronecker sequence: h[i] = 1 if {i*alpha} < 0.5 else 0
    x = (np.arange(n) * alpha) % 1.0
    h = (x < 0.5).astype(float)
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

import numpy as np
import time
from scipy.optimize import linprog


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) < 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = res / cnt
        x[mask] = np.clip(x[mask] + step, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _feasible_bounds(h, d):
    L = 0.0
    U = 0.0
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    if np.any(pos_mask):
        U = float(np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    if np.any(neg_mask):
        L = float(np.min(-h[neg_mask] / d[neg_mask]))
    if L > U:
        return 0.0, 0.0
    return L, U


def _line_search(h, d, L, U):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    if U - L < 1e-14:
        return 0.0, float(np.max(S)) * dx

    n_grid = 120
    grid = np.linspace(L, U, n_grid)
    vals = S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    t_best = float(grid[i])
    f_best = float(obj[i])

    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, n_grid - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)

    def _ev(t):
        return float(np.max(S + t * T - t * t * Q)) * dx

    f1 = _ev(c1)
    f2 = _ev(c2)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _ev(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _ev(c2)

    t_gs = 0.5 * (a + b)
    f_gs = _ev(t_gs)
    if f_gs < f_best:
        t_best, f_best = t_gs, f_gs

    f0 = float(np.max(S)) * dx
    if f0 < f_best:
        t_best, f_best = 0.0, f0
    return t_best, f_best


def _compute_gradients(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for r in range(M):
        k = r - (n - 1)
        row = G[r, :]
        if k >= 0:
            nn = np.arange(0, n - k)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
        else:
            nn = np.arange(-k, n)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
    return G


def _slp_direction(h, corr, top_k=None):
    n = len(h)
    M = 2 * n - 1
    G = _compute_gradients(h)

    if top_k is not None and top_k < M:
        idx = np.argsort(-corr)[:top_k]
        G = G[idx, :]
        corr_s = corr[idx]
    else:
        corr_s = corr

    A_ub = np.hstack([G, -np.ones((len(corr_s), 1))])
    b_ub = G @ h - corr_s
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            x_opt = res.x[:n]
            d = x_opt - h
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm > 1e-10:
                return d / nrm
    except Exception:
        pass
    return None


def _active_gradient_direction(h, corr, k_active, tau_temp, rng):
    n = len(h)
    M = 2 * n - 1
    m = float(np.max(corr))
    w = np.exp((corr - m) / tau_temp)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        w = np.ones(M) / M
    else:
        w /= w_sum

    idx = np.argsort(-corr)[:k_active]
    G_sum = np.zeros(n)
    for r in idx:
        k = r - (n - 1)
        row = np.zeros(n)
        if k >= 0:
            nn = np.arange(0, n - k)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
        else:
            nn = np.arange(-k, n)
            a = nn + k
            b = nn
            row[b] += -h[a]
            row[a] += (1.0 - h[b])
        G_sum += w[r] * row

    d = -G_sum
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm


def _pair_move(h, i, j):
    n = len(h)
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-14:
        return 0.0, 0.0

    dx = 2.0 / n
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    S = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    n_grid = 50
    grid = np.linspace(L, U, n_grid)
    vals = S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    idx = int(np.argmin(obj))
    best_t = float(grid[idx])
    best_f = float(obj[idx])

    f0 = float(np.max(S)) * dx
    if f0 <= best_f:
        return 0.0, f0

    a = grid[max(idx - 1, 0)]
    b = grid[min(idx + 1, n_grid - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)

    def _ev(t):
        return float(np.max(S + t * T - t * t * Q)) * dx

    f1 = _ev(c1)
    f2 = _ev(c2)
    for _ in range(30):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _ev(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _ev(c2)

    t_gs = 0.5 * (a + b)
    f_gs = _ev(t_gs)
    if f_gs < best_f:
        best_t, best_f = t_gs, f_gs

    if best_f >= f0:
        return 0.0, f0
    return best_t, best_f


def _optimize(h, deadline, rng, use_slp=False, slp_frac=0.25, pair_frac=0.30,
              rand_frac=0.10, k_active=8, tau_temp=0.02, slp_top_k=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    restarts = 0
    sigma = 0.04
    uphill = 2e-4

    while time.time() < deadline:
        r = rng.random()
        made_move = False

        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm > 1e-14:
                d /= nrm
                L, U = _feasible_bounds(h, d)
                if U - L > 1e-11:
                    t_opt, obj_val = _line_search(h, d, L, U)
                    if abs(t_opt) > 1e-13:
                        hn = h + t_opt * d
                        hn = _project(hn, target)
                        nf = _exact_c5(hn)
                        if nf < best_f:
                            best_f = nf
                            best_h = hn.copy()
                        if nf < cur_f + uphill:
                            h = hn
                            cur_f = nf
                            corr = np.correlate(h, 1.0 - h, mode='full')
                            stall = 0
                        else:
                            h = best_h.copy()
                            cur_f = best_f
                            corr = np.correlate(h, 1.0 - h, mode='full')
                            stall += 1
                        made_move = True

        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                t_m, f_m = _pair_move(h, i, j)
                if abs(t_m) > 1e-10 and f_m < cur_f - 1e-12:
                    hn = h.copy()
                    hn[i] += t_m
                    hn[j] -= t_m
                    hn = np.clip(hn, 0.0, 1.0)
                    hn = _project(hn, target)
                    nf = _exact_c5(hn)
                    if nf < best_f:
                        best_f = nf
                        best_h = hn.copy()
                    if nf < cur_f + uphill:
                        h = hn
                        cur_f = nf
                        corr = np.correlate(h, 1.0 - h, mode='full')
                        stall = 0
                    else:
                        h = best_h.copy()
                        cur_f = best_f
                        corr = np.correlate(h, 1.0 - h, mode='full')
                        stall += 1
                    made_move = True

        elif use_slp and r < rand_frac + pair_frac + slp_frac:
            d = _slp_direction(h, corr, top_k=slp_top_k)
            if d is not None:
                d -= d.mean()
                nrm = float(np.linalg.norm(d))
                if nrm > 1e-14:
                    d /= nrm
                    L, U = _feasible_bounds(h, d)
                    if U - L > 1e-11:
                        t_opt, obj_val = _line_search(h, d, L, U)
                        if abs(t_opt) > 1e-13:
                            hn = h + t_opt * d
                            hn = _project(hn, target)
                            nf = _exact_c5(hn)
                            if nf < best_f:
                                best_f = nf
                                best_h = hn.copy()
                            if nf < cur_f + uphill:
                                h = hn
                                cur_f = nf
                                corr = np.correlate(h, 1.0 - h, mode='full')
                                stall = 0
                            else:
                                h = best_h.copy()
                                cur_f = best_f
                                corr = np.correlate(h, 1.0 - h, mode='full')
                                stall += 1
                            made_move = True

        else:
            d = _active_gradient_direction(h, corr, k_active, tau_temp, rng)
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm > 1e-14:
                d /= nrm
                L, U = _feasible_bounds(h, d)
                if U - L > 1e-11:
                    t_opt, obj_val = _line_search(h, d, L, U)
                    if abs(t_opt) > 1e-13:
                        hn = h + t_opt * d
                        hn = _project(hn, target)
                        nf = _exact_c5(hn)
                        if nf < best_f:
                            best_f = nf
                            best_h = hn.copy()
                        if nf < cur_f + uphill:
                            h = hn
                            cur_f = nf
                            corr = np.correlate(h, 1.0 - h, mode='full')
                            stall = 0
                        else:
                            h = best_h.copy()
                            cur_f = best_f
                            corr = np.correlate(h, 1.0 - h, mode='full')
                            stall += 1
                        made_move = True

        if not made_move:
            stall += 1

        if stall > 80:
            restarts += 1
            amp = max(0.003, sigma * (1.0 if restarts < 20 else 2.5))
            h = _project(best_h + rng.normal(0, amp, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.002, sigma * 0.9)

    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _kronecker_init(n, alpha, rng):
    x = (np.arange(n) * alpha) % 1.0
    h = (x < 0.5).astype(float)
    h += rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 8))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h += rng.normal(0, 0.03, n)
    return _project(h, n / 2.0)


def _random_binary_init(n, rng):
    h = rng.integers(0, 2, n).astype(float)
    h += rng.normal(0, 0.08, n)
    return _project(h, n / 2.0)

    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: Multi-start exploration (18% budget)
    d1 = t0 + budget_s * 0.18
    starts = [h0.copy()]

    golden_ratio = 0.6180339887498949
    freqs = [(1, 0.0), (1, 1.2), (2, 0.3), (2, 1.5), (3, 0.7), (5, 0.0)]
    for f_, p_ in freqs:
        starts.append(_sine_init(n0, f_, p_, rng))
    starts.append(_kronecker_init(n0, golden_ratio, rng))
    starts.append(_kronecker_init(n0, 1.0 - golden_ratio, rng))
    starts.append(_kronecker_init(n0, 0.5 + 0.1 * (np.sqrt(5) - 1), rng))
    starts.append(_random_binary_init(n0, rng))
    starts.append(_block_init(n0, rng))

    # Try different n values
    for dn in [-4, -2, 2, 4]:
        nc = n0 + dn
        if nc >= 24 and nc % 2 == 0:
            starts.append(_resample(h0, nc))

    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + max(2.0, budget_s * 0.022))
        hh, ff = _optimize(s, sub_dl, rng, k_active=6, tau_temp=0.03)
        if ff < best_f:
            best_f = ff
            best_h = hh.copy()

    # Phase 2: Deep optimization with SLP (52% budget)
    d2 = t0 + budget_s * 0.70
    hh, ff = _optimize(best_h, d2, rng, use_slp=True, slp_frac=0.30,
                       pair_frac=0.30, rand_frac=0.08,
                       k_active=10, tau_temp=0.015, slp_top_k=80)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: Multi-resolution refinement (12% budget)
    d3 = t0 + budget_s * 0.82
    bn = len(best_h)
    tried_n = {bn}
    for dn in [-2, 2, -4, 4, -6, 6, -10, 10, -8, 8]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 20 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        hr = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + max(2.0, budget_s * 0.015))
        hr2, fr2 = _optimize(hr, sub_dl, rng, use_slp=True, slp_frac=0.2,

                             k_active=8, tau_temp=0.02, slp_top_k=60)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 4: Final intensive polish (18% budget)
    hh, ff = _optimize(best_h, deadline, rng, use_slp=True,
                       slp_frac=0.35, pair_frac=0.40, rand_frac=0.05,
                       k_active=15, tau_temp=0.008, slp_top_k=100)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
