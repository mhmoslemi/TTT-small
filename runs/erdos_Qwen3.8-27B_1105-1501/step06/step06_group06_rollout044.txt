f_0 = Σ_i h[i]·(1-h[i])·dx = (Σ_i h[i] - Σ_i h[i]²)·dx

∂f_s/∂h[a] = dx · (Σ_{i: i+s=a} (1-h[i]) - Σ_{i: i=a} h[i+s] - δ_{a≥s}·h[a-s]·(wait this is getting complicated))

f_s = dx · Σ_{i=0}^{n-1-s} h[i+s] - dx · Σ_{i=0}^{n-1-s} h[i+s]·h[i]

∂f_s/∂h[a]:
- From first term: dx if a ∈ [s, n-1] (i.e., a = i+s for i = a-s, valid when 0 ≤ a-s ≤ n-1-s, i.e., s ≤ a ≤ n-1)
- From second term: -dx · [h[a-s] if s ≤ a ≤ n-1 (from h[i+s]·h[i] with i+s=a) + h[a+s] if 0 ≤ a ≤ n-1-s (from h[i+s]·h[i] with i=a, so i+s=a+s)]

Let me simplify: instead of a full QP, I'll use a **projected gradient method with Barzilai-Borwein step size** and **momentum**. This is simpler and can be very effective for this type of problem.

**Barzilai-Borwein step size**:
α_k = (s_k^T y_k) / (y_k^T y_k) or α_k = (s_k^T s_k) / (s_k^T y_k)
where s_k = h_k - h_{k-1} and y_k = g_k - g_{k-1}

**Momentum**:
v_{k+1} = β·v_k - α_k·g_k
h_{k+1} = h_k + v_{k+1} (projected)

This combination is known to work well for nonconvex problems.

import numpy as np
from scipy.optimize import minimize
import time

_slsqp_state = {}

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_corr(h, s):
    n = len(h)
    dx = 2.0 / n
    m = n - abs(s)
    g = np.zeros(n)
    if s >= 0:
        g[:m] -= h[s:s + m]
        g[s:s + m] += (1.0 - h[:m])
    else:
        gs = -s
        g[gs:gs + m] -= h[:m]
        g[:m] += (1.0 - h[gs:gs + m])
    return g * dx

def _slsqp_obj(x):
    return x[_slsqp_state['n']]

def _slsqp_obj_grad(x):
    g = np.zeros(_slsqp_state['n'] + 1)
    g[_slsqp_state['n']] = 1.0
    return g

def _slsqp_con(x):
    n = _slsqp_state['n']
    dx = _slsqp_state['dx']
    act = _slsqp_state['active']
    hx = x[:n]
    c = np.correlate(hx, 1.0 - hx, mode='full') * dx
    K = len(act)
    out = np.zeros(K + 1)
    for j in range(K):
        out[j] = x[n] - c[act[j]]
    out[-1] = np.sum(hx) - n / 2.0
    return out

def _slsqp_con_jac(x):
    n = _slsqp_state['n']
    dx = _slsqp_state['dx']
    act = _slsqp_state['active']
    hx = x[:n]
    K = len(act)
    J = np.zeros((K + 1, n + 1))
    for j in range(K):
        s = int(act[j]) - (n - 1)
        J[j, :n] = -_grad_corr(hx, s)
        J[j, n] = 1.0
    J[-1, :n] = 1.0
    return J

def _slsqp_optimize(h0, deadline, max_outer=20, active_frac=0.35):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _corr_vec(h)
        M = cur
        sorted_c = np.sort(corr)[::-1]
        frac_idx = min(len(corr) - 1, int(n * active_frac))
        tol = max(1e-6, M - sorted_c[frac_idx])
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:12]
        if len(active) > 50:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:50]]

        _slsqp_state['n'] = n
        _slsqp_state['dx'] = dx
        _slsqp_state['active'] = active

        x0 = np.concatenate([h, [M + 5e-5]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 3.0)]
        cons = [{'type': 'ineq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]

        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 60, 'ftol': 1e-14})

        h_new = _project(res.x[:n], target)
        c5_new = _exact_c5(h_new)
        if c5_new < cur - 1e-14:
            h = h_new
            cur = c5_new
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
        else:
            break

    return best_h, best_c5

def _subgrad_bb_optimize(h0, deadline, max_iter=2000, momentum=0.7):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur
    v = np.zeros(n)
    alpha_prev = 0.1 / np.linalg.norm(h) if np.linalg.norm(h) > 0 else 0.1
    prev_h = h.copy()
    prev_grad = None
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = _corr_vec(h)
        M = cur
        tol = max(1e-5, M - np.sort(corr)[::-1][min(len(corr)-1, n)])
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:15]
        if len(active) > 45:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:45]]

        grad = np.zeros(n)
        for idx in active:
            s = int(idx) - (n - 1)
            grad += _grad_corr(h, s)
        grad /= len(active)
        grad -= grad.mean()

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-13:
            stall += 5
        else:
            if prev_grad is not None:
                s_vec = h - prev_h
                y_vec = grad - prev_grad
                sy = float(np.dot(s_vec, y_vec))
                yy = float(np.dot(y_vec, y_vec))
                ss = float(np.dot(s_vec, s_vec))
                if sy > 1e-14 and yy > 1e-14:
                    alpha_bb1 = sy / yy
                    alpha_bb2 = ss / sy
                    alpha_bb = 0.5 * (alpha_bb1 + alpha_bb2)
                    if alpha_bb > 1e-15 and alpha_bb < 1.0:
                        alpha_prev = alpha_bb
            alpha = alpha_prev / max(grad_norm, 1e-10)
            alpha = min(alpha, 0.5)

            v = momentum * v - alpha * grad
            v -= v.mean()
            h_trial = _project(h + v, target)
            cur_trial = _exact_c5(h_trial)

            if cur_trial < cur - 1e-15:
                prev_h = h.copy()
                prev_grad = grad.copy()
                h = h_trial
                cur = cur_trial
                stall = 0
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
            else:
                alpha_prev *= 0.6
                v = 0.8 * v
                stall += 1
                if stall > 50:
                    h = _project(h + 0.01 * np.random.randn(n), target)
                    cur = _exact_c5(h)
                    v = np.zeros(n)
                    stall = 0
                    prev_grad = None

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_perturb(h, scale, rng):
    n = len(h)
    return _project(h + scale * rng.standard_normal(n), n / 2.0)
