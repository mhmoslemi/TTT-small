    # h[i] goes 1->0, h[j] goes 0->1
    # delta_c[k] = h[i+k] + h[i-k] - h[j+k] - h[j-k]
    # where k ranges from -(n-1) to n-1
    # h[m] = 0 for m < 0 or m >= n
    
    delta = np.zeros(2*n-1)
    # k = i - m for the h[i-k] term: m = i-k, so k = i-m
    # h[i+k] is nonzero for 0 <= i+k < n, i.e., -i <= k < n-i
    # h[i-k] is nonzero for 0 <= i-k < n, i.e., i-n+1 < k <= i
    # Similarly for j
    
    # For h[i+k]: k in [-i, n-i-1]
    lo1 = max(-i, -(n-1))
    hi1 = min(n-i-1, n-1)
    if lo1 <= hi1:
        delta[lo1+(n-1):hi1+(n-1)+1] += h[lo1+i:hi1+i+1]
    
    # For h[i-k]: k in [i-(n-1), i]
    lo2 = max(i-(n-1), -(n-1))
    hi2 = min(i, n-1)
    if lo2 <= hi2:
        delta[lo2+(n-1):hi2+(n-1)+1] += h[i-hi2:i-lo2+1]  # h[i-k] for k=lo2..hi2
    
    # For -h[j+k]: k in [-j, n-j-1]
    lo3 = max(-j, -(n-1))
    hi3 = min(n-j-1, n-1)
    if lo3 <= hi3:
        delta[lo3+(n-1):hi3+(n-1)+1] -= h[lo3+j:hi3+j+1]
    
    # For -h[j-k]: k in [j-(n-1), j]

    hi4 = min(j, n-1)
    if lo4 <= hi4:
        delta[lo4+(n-1):hi4+(n-1)+1] -= h[j-hi4:j-lo4+1]
    
    return delta

    n = len(h)
    # Pad h with zeros on both sides
    hp = np.zeros(2*n+1)
    hp[n:n+n] = h  # h[0] at position n, h[n-1] at position 2n-1
    # k ranges from -(n-1) to n-1
    # h[i+k] corresponds to hp[n+i+k] for k in range
    # h[i-k] corresponds to hp[n+i-k]
    # h[j+k] corresponds to hp[n+j+k]
    # h[j-k] corresponds to hp[n+j-k]
    
    ks = np.arange(-(n-1), n)  # 2n-1 values
    delta = hp[n+i+ks] + hp[n+i-ks] - hp[n+j+ks] - hp[n+j-ks]
    return delta

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _c5_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = x < 1.0 if res > 0 else x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _make_pad(h):
    n = len(h)
    return np.concatenate([np.zeros(n), h, np.zeros(n)])

def _compute_delta(pad, i, j, n):
    ks = np.arange(-(n - 1), n, dtype=int)
    return pad[n + i + ks] + pad[n + i - ks] - pad[n + j + ks] - pad[n + j - ks]

def _greedy_descent(h, max_rounds=20000):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    c_arr = _c5_arr(cur)
    cur_max = float(np.max(c_arr))
    best_h = cur.copy()
    best_c5 = cur_max * dx
    pad = _make_pad(cur)
    ksg = np.arange(-(n - 1), n, dtype=int)
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        k_star = int(np.argmax(c_arr))
        k_val = k_star - (n - 1)
        ones_idx = np.where(cur > 0.5)[0]
        zeros_idx = np.where(cur <= 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        # Score: for shift k_val, delta at that shift
        # delta_c[k_val] = h[i+k_val] + h[i-k_val] - h[j+k_val] - h[j-k_val]
        # Want most negative => minimize h[i+k_val]+h[i-k_val], maximize h[j+k_val]+h[j-k_val]
        scores_i = np.zeros(len(ones_idx))
        for a in range(len(ones_idx)):
            i = ones_idx[a]
            mi = i + k_val
            mm = i - k_val
            si = 0.0
            if 0 <= mi < n:
                si += cur[mi]
            if 0 <= mm < n:
                si += cur[mm]
            scores_i[a] = si
        scores_j = np.zeros(len(zeros_idx))
        for a in range(len(zeros_idx)):
            j = zeros_idx[a]
            mj = j + k_val
            mm2 = j - k_val
            sj = 0.0
            if 0 <= mj < n:
                sj += cur[mj]
            if 0 <= mm2 < n:
                sj += cur[mm2]
            scores_j[a] = sj
        # Sort: want low score_i (minimize) and high score_j (maximize)
        i_order = np.argsort(scores_i)[:15]
        j_order = np.argsort(scores_j)[::-1][:15]
        best_delta_val = 0.0
        best_i = -1
        best_j = -1
        best_full_delta = None
        for ai in i_order:
            for aj in j_order:
                i = int(ones_idx[ai])
                j = int(zeros_idx[aj])
                d = _compute_delta(pad, i, j, n)
                new_max = float(np.max(c_arr + d))
                if new_max < best_delta_val - 1e-14:
                    best_delta_val = new_max
                    best_i = i
                    best_j = j
                    best_full_delta = d
        if best_i >= 0 and best_delta_val < cur_max - 1e-12:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            c_arr = c_arr + best_full_delta
            cur_max = best_delta_val
            pad = _make_pad(cur)
            improved = True
            if cur_max * dx < best_c5 - 1e-14:
                best_c5 = cur_max * dx
                best_h = cur.copy()
    return best_h, best_c5

def _basin_hop(h, deadline, n_trials=60, seed=100):

        #Perturb: flip k random 1->0 and k random 0->1

            break

def _lp_polish(h, deadline, radius=0.08, stall_limit=80):
    """LP-based local optimization on continuous h."""
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius_cur = radius
    while time.time() < deadline and stall < stall_limit:
        scores = _c5_arr(h) * dx
        # Subgradient direction: -G^T * sign(scores - max)
        # Simplified: use the LP direction
        # Build gradient matrix for only active constraints (top few scores)
        top_k = int(np.argsort(scores)[::-1][:min(20, len(scores))])
        # Direction: minimize max of (scores + d*G)
        # Use subgradient: d = -sum over active of G_row
        Gactive = np.zeros(n)
        for r in top_k:
            d_shift = r - (n - 1)
            if d_shift >= 0:
                m = n - d_shift
                if m > 0:
                    Gactive[:m] += (1.0 - h[d_shift:])
                    Gactive[d_shift:] += -h[:m]
            else:
                g = -d_shift
                m = n - g
                if m > 0:
                    Gactive[:m] += (1.0 - h[g:])
                    Gactive[g:] += -h[:m]
        # Line search along -Gactive
        d = -Gactive
        # Ensure sum(d) = 0 approximately
        d -= np.mean(d)
        # Clip to bounds
        pos = d > 1e-10
        neg = d < -1e-10
        t_max = radius_cur
        if np.any(pos):
            t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_max = min(t_max, float(np.min(h[neg] / (-d[neg]))))
        if t_max < 1e-10:
            stall += 1
            radius_cur *= 0.8
            continue
        # Line search
        best_t = 0.0
        best_f = cur_c5
        for t_try in np.linspace(0, t_max, 30):
            h_try = _project(h + t_try * d, target)
            f_try = _exact_c5(h_try)

                best_f = f_try
                best_t = t_try
        if best_t > 1e-10 and best_f < cur_c5 - 1e-13:
            h = _project(h + best_t * d, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius_cur = min(radius, radius_cur * 1.05)
        else:
            stall += 1
            radius_cur *= 0.85
            if radius_cur < 0.005:
                break
    return best_h, best_c5
