def _slp_direction(h, c, G, n):
    """Solve the SLP to get a direction."""
    target = n / 2.0
    # Variables: [h_1, ..., h_n, t]
    # Objective: minimize t
    cobj = np.zeros(n + 1)
    cobj[-1] = 1.0
    
    # Inequality constraints: G @ (h' - h) + c <= t
    # => G @ h' - t <= G @ h - c
    # A_ub @ [h', t] <= b_ub
    A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
    b_ub = G @ h - c
    
    # Equality: sum(h') = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= h_i <= 1, t >= 0 (or -inf for t, but we know it's bounded)
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(cobj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        h_lp = res.x[:n]
        d = h_lp - h
        return d
    return None

Actually, for pair moves, the change in correlation at lag k can be computed quickly. If we change h[i] by +δ and h[j] by -δ (maintaining the sum), the change in c_k is:
Δc_k = δ * (contribution of i at lag k) - δ * (contribution of j at lag k)

The "contribution of position i at lag k" is the derivative G[k, i] (from the gradient matrix).

So Δc_k = δ * (G[k, i] - G[k, j])

The new max is max_k (c_k + δ*(G[k,i] - G[k,j])). We want to minimize this over δ (and the choice of i, j).

For a given pair (i,j), the optimal δ is found by the same line search (but restricted to the feasible range for that pair).

This gives a coordinate descent that's O(n²) per full pass but very effective for polishing.

import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-14:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _build_grad_matrix(h):
    n = len(h)

def _line_search(c, T, Q, dx, L, U):

        return t, float(np.max(c + t * T - t * t * Q)) * dx

    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(c + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(c + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(45):
        if (b - a) < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(c + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)

    t = 0.5 * (a + b)
    tv = float(np.max(c + t * T - t * t * Q)) * dx
    if obj[i] < tv:

    return t, tv


def _slp_solve(h, c, G, n):
    target = n / 2.0
    M = 2 * n - 1
    cobj = np.zeros(n + 1)
    cobj[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = G @ h - c
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(cobj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n] - h
    except Exception:
        pass
    return None


def _weighted_grad_dir(h, c, G, n, k_top=12, tau=0.015):
    M = 2 * n - 1
    m = float(np.max(c))
    w = np.exp((c - m) / tau)
    w = w / (w.sum() + 1e-30)
    top = np.argsort(-c)[:k_top]
    w_top = w[top]
    w_top = w_top / (w_top.sum() + 1e-30)
    d = -G[top, :].T @ w_top
    d -= d.mean()
    return d


def _step_move(h, d, c, dx):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    L, U = 0.0, 1e30
    if np.any(pos):
        U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        L = max(L, float(np.min(-h[neg] / d[neg])))
        U = min(U, 1e30)
    if U - L < 1e-12:
        return None, None, None
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    t, tv = _line_search(c, T, Q, dx, L, U)
    return t, tv, (L, U)

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)

        sub_deadline = min(d3, time.time() + max(2.0, budget_s * 0.012))
        hr2, fr2 = _slp_optimize(hr, sub_deadline, rng, slp_interval=3, pull_frac=0.15)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = nt

    # Phase 4: final polish (7% budget)
    hh, ff = _pair_optimize(best_h, deadline, rng, n_scan=40)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
