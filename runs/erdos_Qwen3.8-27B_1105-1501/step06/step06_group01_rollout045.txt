def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_partner_matrix(partner, dx):
    """
    Build matrix A where A[k, i] = dx * partner[i+k-(n-1)] if valid.
    Then A @ h = dx * corr(h, partner, 'full').
    """
    n = len(partner)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        offset = k - (n - 1)
        # c[k] = sum_i h[i] * partner[i + offset] for valid i
        # i + offset must be in [0, n-1], so i in [-offset, n-1-offset]
        i_start = max(0, -offset)
        i_end = min(n - 1, n - 1 - offset)
        if i_start <= i_end:
            A[k, i_start:i_start + (i_end - i_start + 1)] = dx * partner[i_start + offset:i_end + offset + 1]
    return A

def _solve_partner_lp(partner, target):
    """
    Solve: min tau s.t. A @ h <= tau*1, sum(h) = target, 0 <= h <= 1
    where A = _build_partner_matrix(partner)
    Returns optimal h or None.
    """
    n = len(partner)
    dx = 2.0 / n
    m = 2 * n - 1
    A = _build_partner_matrix(partner, dx)
    
    # Variables: h[0..n-1], tau
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    # Inequality: A @ h - tau <= 0 → [A, -1] @ [h; tau] <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            return h_opt
    except Exception:
        pass
    return None
