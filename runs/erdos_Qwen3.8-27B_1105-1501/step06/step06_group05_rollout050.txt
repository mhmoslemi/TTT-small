            d = d / nrm
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue
        t, newf = _line_search_dir(h, d, corr, dx, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < 0.0 or hn.max() > 1.0:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-13:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = _corr_all(h)
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = _corr_all(h)
            stall += 1
        if stall > 120:
            restarts += 1
            if restarts > 25:
                h = _project(best_h + rng.normal(0, max(0.05, sigma * 3), n), target)
            else:
                h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = _corr_all(h)
            stall = 0
            sigma = max(0.003, sigma * 0.87)
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(2, max(3, n // 5))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        base = _sine_init(n, 1, 0.0, rng)

    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: SLSQP-based optimization (35% of budget)
    d1 = t0 + budget_s * 0.35
    sub_deadline = min(d1, time.time() + max(5.0, budget_s * 0.08))
    h_slq, f_slq = _slsqp_phase(h0, sub_deadline, rng, k_max=20, max_rounds=20)
    if f_slq < best_f:
        best_f = f_slq
        best_h = h_slq.copy()

    # Step with larger K
    if time.time() < d1:
        sub_deadline2 = min(d1, time.time() + max(10.0, budget_s * 0.12))
        h_slq2, f_slq2 = _slsqp_phase(best_h, sub_deadline2, rng, k_max=35, max_rounds=15)
        if f_slq2 < best_f:
            best_f = f_slq2
            best_h = h_slq2.copy()

    # Phase 2: Line search optimization (30% of budget)
    d2 = t0 + budget_s * 0.65
    h_ls, f_ls = _ls_optimize(best_h, d2, rng, pair_frac=0.22, rand_frac=0.08, k_active=10, tau=0.018)
    if f_ls < best_f:
        best_f = f_ls
        best_h = h_ls.copy()

    # Phase 3: SLSQP again on the improved point (10%)
    d3 = t0 + budget_s * 0.75
    if time.time() < d3:
        h_slq3, f_slq3 = _slsqp_phase(best_h, d3, rng, k_max=30, max_rounds=10)
        if f_slq3 < best_f:
            best_f = f_slq3
            best_h = h_slq3.copy()

    # Phase 4: Multi-resolution exploration (10%)
    d4 = t0 + budget_s * 0.85
    bn = len(best_h)
    tried = set()
    for dn in [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12]:
        if time.time() > d4:
            break
        n_try = bn + dn
        if n_try < 30 or n_try > 200:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + max(3.0, budget_s * 0.014))
        hr2, fr2 = _ls_optimize(hr, sub_dl, rng, pair_frac=0.25, rand_frac=0.06, k_active=8, tau=0.02)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 5: Final combined polish (15%)
    # Alternate SLSQP and line search
    remaining = deadline - time.time()
    if remaining > 5.0:
        half = time.time() + remaining * 0.5
        h_f1, f_f1 = _ls_optimize(best_h, half, rng, pair_frac=0.35, rand_frac=0.05, k_active=12, tau=0.012)
        if f_f1 < best_f:
            best_f = f_f1
            best_h = h_f1.copy()
        if time.time() < deadline:
            h_f2, f_f2 = _slsqp_phase(best_h, deadline, rng, k_max=25, max_rounds=8)
            if f_f2 < best_f:
                best_f = f_f2
                best_h = h_f2.copy()

    # Final check with complement
    if time.time() < deadline - 2.0:
        h_comp = _project(1.0 - best_h, len(best_h) / 2.0)
        f_comp = _exact_c5(h_comp)
        if f_comp < best_f:
            best_f = f_comp
            best_h = h_comp.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
