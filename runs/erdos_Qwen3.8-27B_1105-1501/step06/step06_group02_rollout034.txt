    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0

    res = target - float(np.sum(x))

def compute_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        d = i - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            G[i, :m] += -h[d:]
            G[i, d:] += (1.0 - h[:m])
        else:
            q = -d
            m = n - q
            if m <= 0:
                continue
            G[i, :m] += (1.0 - h[q:])
            G[i, q:] += -h[:m]
    return G

def _lp_solve(h, G, R, radius):
    n = len(h)
    target = n / 2.0
    c = R - G @ h
    
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = c.copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target])
    
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = [(None, None)] + list(zip(lo.tolist(), hi.tolist()))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    v = res.x[1:]
    if not np.all(np.isfinite(v)):
        return None
    
    d = v - h
    d -= np.mean(d)
    if np.max(np.abs(d)) < 1e-14:
        return None
    return d

def _line_search(h, d, R0, dx, max_iter=50):
    n = len(h)
    
    lo = 0.0
    hi = 1.0
    for i in range(n):
        if d[i] > 1e-15:
            hi = min(hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            lo = max(lo, (1.0 - h[i]) / d[i])
            hi = min(hi, h[i] / (-d[i]))
    
    lo = min(lo, 0.0)
    if hi <= 0.0:
        hi = 1e-10
    hi = max(hi, 1e-10)
    
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    
    for _ in range(max_iter):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
    
    f0 = float(np.max(R0)) * dx
    if f0 < val:
        opt, val = 0.0, f0
    
    f_lo = float(np.max(R0 + lo * T - lo * lo * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    
    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_hi < val:
        opt, val = hi, f_hi
    
    return opt, val

def _lp_descent(h, deadline, radii):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, mode='full')
    G = compute_G(h)
    stall_total = 0
    
    for radius in radii:
        while time.time() < deadline:
            if stall_total > 50:
                return best_h, best_c5
            
            d = _lp_solve(h, G, R0, radius)
            
            if d is None:
                stall_total += 1
                break
            
            alpha, new_c5 = _line_search(h, d, R0, dx)
            
            if alpha > 1e-10 and new_c5 < cur_c5 - 1e-12:
                h = h + alpha * d
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R0 = np.correlate(h, 1.0 - h, mode='full')
                G = compute_G(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall_total = max(0, stall_total - 2)
            else:
                stall_total += 1
                break
    
    return best_h, best_c5

def _pair_search(h, deadline, n_try=15):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline and stall < 40:
        # Find active rows
        scores = R0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.004 * max_score
        active = np.where(scores >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(-scores)[:5]
        
        # Compute gradient of active rows (weighted)
        grad = np.zeros(n)
        for r in active[:min(10, len(active))]:
            d = r - (n - 1)
            m = n - abs(d)
            if m <= 0:
                continue
            if d >= 0:
                grad[:m] += -h[d:d+m]
                grad[d:d+m] += (1.0 - h[:m])
            else:
                q = -d
                grad[:m] += (1.0 - h[q:q+m])
                grad[q:q+m] += -h[:m]
        
        donors = np.argsort(-grad)[:n_try]
        recvs = np.argsort(grad)[:n_try]
        
        best_alpha = 0.0
        best_val = cur_c5
        best_pair = None
        
        for i in donors:
            if time.time() > deadline:
                break
            for j in recvs:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, new_c5 = _line_search(h, d, R0, dx, max_iter=30)
                if alpha > 1e-10 and new_c5 < best_val - 1e-14:
                    best_val = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        
        if best_pair is not None and best_val < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10 and rng.random() < 0.4:
                idx = rng.integers(n, size=3)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.08, 0.08)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)
