            v *= 0.5
            # Occasionally increase tau for exploration
            if lr < 0.01:
                lr = 0.05
                tau = min(tau * 1.2, 0.05)
    
    return best_h, best_c5

def _compute_flip_effects(h, n):
    """Compute effect[i, s] = (1-h[i+s])*I(0≤i+s<n) - h[i-s]*I(0≤i-s<n)
    for all i in [0,n) and s in [-(n-1), n-1].
    Returns array of shape (n, 2n-1) where index s corresponds to r = s + (n-1)."""
    effects = np.zeros((n, 2*n-1))
    for i in range(n):
        for s in range(-(n-1), n):
            idx = s + (n-1)
            val = 0.0
            if 0 <= i+s < n:
                val += (1.0 - h[i+s])
            if 0 <= i-s < n:
                val -= h[i-s]
            effects[i, idx] = val
    return effects
```

        # For a pair (i,j): new max = max_s (S_s - effects[i,s] + effects[j,s])
        # We want this to be < M.
        
        # Strategy: for each i with h[i]=1, compute score_i = max_s effects[i,s]
        # (we want this to be negative, meaning flipping i reduces all S_s)
        # Then for each j with h[j]=0, compute penalty_j = -min_s effects[j,s]
        # (we want this to be small, meaning adding h[j] doesn't increase S too much)
        
        # Actually, let me just try all pairs where the individual effect is good.
        
        # Compute "quality" of each flip
        # For i in ones: flip_quality_i = max_s (S_s - effects[i,s])
        #   If this is < M, flipping i alone (and compensating elsewhere) helps.
        
        # Simpler: just try random pairs and see what works.
        rng = np.random.default_rng()
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_trials = min(200 * n_ones * n_zeros // 100, 3000)
        
        for _ in range(n_trials):
            i = ones[rng.integers(n_ones)]
            j = zeros[rng.integers(n_zeros)]
            new_S = S - effects[i] + effects[j]
            new_M = float(np.max(new_S))
            if new_M < best_c5 - 1e-13:
                # Apply the swap
                h[i] = 0.0
                h[j] = 1.0
                S = new_S
                best_c5 = new_M
                best_h = h.copy()
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                found = True
                break
        
        if found:
            stall = 0
        else:
            stall += 1
        
        # If stuck, try a random restart or perturbation
        if stall > 20:
            # Random perturbation: flip a few random pairs
            n_pert = rng.integers(2, 8)
            for _ in range(n_pert):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
            # Recompute
            S = np.correlate(h, 1.0-h, mode='full') * dx
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            stall = 0
            # Don't update best_h here (we might have made it worse)
    
    return best_h, best_c5

def _compute_flip_effects(h, n):
    effects = np.zeros((n, 2*n-1))
    one_minus_h = 1.0 - h
    for s_idx, s in enumerate(range(-(n-1), n)):
        # For all i: (1-h[i+s]) where 0≤i+s<n, minus h[i-s] where 0≤i-s<n
        # First term: i+s in [0,n-1] => i in [-s, n-1-s] intersect [0,n-1]
        lo1 = max(0, -s)
        hi1 = min(n-1, n-1-s)
        if lo1 <= hi1:
            effects[lo1:hi1+1, s_idx] += one_minus_h[lo1+s:hi1+s+1]
        # Second term: i-s in [0,n-1] => i in [s, n-1+s] intersect [0,n-1]
        lo2 = max(0, s)
        hi2 = min(n-1, n-1+s)
        if lo2 <= hi2:
            effects[lo2:hi2+1, s_idx] -= h[lo2-s:hi2-s+1]
    return effects
```
