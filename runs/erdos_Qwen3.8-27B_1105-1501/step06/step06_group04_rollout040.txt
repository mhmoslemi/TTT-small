def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    n = len(h)

                grad[j1] += w[p] * dx * (1.0 - h[j1 - k])

def _soft_max_value(c, tau):
    c_shifted = c - c.max()
    return c.max() + np.log(np.sum(np.exp(tau * c_shifted))) / tau

def _momentum_descent(h, deadline, tau_start=30, tau_end=200, 
                      lr_start=0.01, momentum=0.9, max_iters=50000):
    """Soft-max gradient descent with momentum and adaptive tau."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    
    n_iters = max_iters
    stall = 0
    stall_count = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if stall_count > 2000:
            break
        
        progress = it / max(1, n_iters - 1)
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_start / (1.0 + 0.01 * progress * n)
        
        grad, cur_c5 = _softmax_gradient(h, tau)
        
        # Update with momentum
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = _project(h_new, target)
        
        new_c5 = _exact_c5(h_new)
        if new_c5 < cur_c5 - 1e-12:
            h = h_new
            cur_c5 = new_c5
            stall = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall_count = 0
            else:
                stall += 1
        else:
            # Backtrack
            v *= 0.5
            h_new = h + v * 0.5
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5:
                h = h_new
                cur_c5 = new_c5
            stall += 1
            stall_count += 1
            if stall > 50:
                v *= 0.1
                stall = 0
    
    return best_h, best_c5

def _sa_binary(h, deadline, rng, max_iters=20000):
    """Simulated annealing on binary patterns with better scheduling."""
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    
    ones = set(np.where(h_bin > 0.5)[0].tolist())
    zeros = set(np.where(h_bin <= 0.5)[0].tolist())
    
    T = 0.003
    T_min = 0.00001
    patience = 0
    max_patience = 5000
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if patience > max_patience:
            break
        if not ones or not zeros:
            break
        
        ones_list = list(ones)
        zeros_list = list(zeros)
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = _exact_c5(h_bin)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            cur_c5 = new_c5
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
                patience = 0
            else:
                patience += 1
        else:
            h_bin[i] = 1
            h_bin[j] = 0
        
        T = max(T_min, T * 0.9997)
    
    return best_h, best_c5

def _relax_and_optimize(h_bin, deadline):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=100):

            if i == j:

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_shift = c - c.max()
    exp_c = np.exp(tau * c_shift)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-14:
            continue
        k = p - (n - 1)
        if k >= 0:
            if k < n:
                j1 = np.arange(k, n)
                grad[j1] += w[p] * dx * (1.0 - h[j1 - k])
                j2 = np.arange(0, n - k)
                grad[j2] += w[p] * dx * (-h[j2 + k])
        else:
            k2 = -k
            j1 = np.arange(0, n - k2)
            grad[j1] += w[p] * dx * (1.0 - h[j1 + k2])
            j2 = np.arange(k2, n)
            grad[j2] += w[p] * dx * (-h[j2 - k2])
    return grad, float(np.max(c))

def _grad_descent_continuation(h, deadline, tau_start=15, tau_end=400,
                                lr_base=0.008, momentum=0.88, max_outer=50,
                                max_inner=5000, tau_mult=1.4):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_start
    outer = 0
    stall_outer = 0
    while time.time() < deadline and outer < max_outer:
        outer += 1
        v = np.zeros(n)
        cur_c5 = _exact_c5(h)
        inner_stall = 0
        for inner in range(max_inner):
            if time.time() > deadline:
                break
            if inner_stall > 2000:
                break
            grad, cur_c5 = _softmax_grad(h, tau)
            lr = lr_base / (1.0 + 0.005 * inner)
            v = momentum * v - lr * grad
            h_new = _project(h + v, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                inner_stall = 0
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                v *= 0.3
                h_new = _project(h + v, target)
                new_c5 = _exact_c5(h_new)
                if new_c5 < cur_c5:
                    h = h_new
                    cur_c5 = new_c5
                inner_stall += 1
        if _exact_c5(h) < best_c5:
            stall_outer = 0
        else:
            stall_outer += 1
            h = best_h.copy()
        tau *= tau_mult
        if tau > tau_end:
            break
    return best_h, best_c5

def _binary_bi(h, deadline, max_rounds=500):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    ones = set(np.where(h_b > 0.5)[0].tolist())
    zeros = set(np.where(h_b <= 0.5)[0].tolist())
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline or no_improve > 15:
            break
        best_delta = 0.0
        best_i, best_j = None, None
        ones_list = list(ones)
        zeros_list = list(zeros)
        n_to_try = min(3000, len(ones_list) * len(zeros_list))
        rng_local = np.random.default_rng(rnd + 42)
        for _ in range(n_to_try):
            i = ones_list[rng_local.integers(len(ones_list))]
            j = zeros_list[rng_local.integers(len(zeros_list))]
            h_b[i] = 0
            h_b[j] = 1
            nc = _exact_c5(h_b)
            h_b[i] = 1
            h_b[j] = 0
            d = nc - cur_c5
            if d < best_delta:
                best_delta = d
                best_i, best_j = i, j
        if best_i is not None:
            h_b[best_i] = 0
            h_b[best_j] = 1
            cur_c5 += best_delta
            ones.discard(best_i)
            ones.add(best_j)
            zeros.discard(best_j)
            zeros.add(best_i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5

def _sa_binary(h, deadline, rng, max_iters=50000):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    ones = set(np.where(h_b > 0.5)[0].tolist())
    zeros = set(np.where(h_b <= 0.5)[0].tolist())
    T = 0.004
    T_min = 5e-5
    patience = 0
    for it in range(max_iters):
        if time.time() > deadline or patience > 8000:
            break
        if not ones or not zeros:
            break
        ol = list(ones)
        zl = list(zeros)
        i = ol[rng.integers(len(ol))]
        j = zl[rng.integers(len(zl))]
        h_b[i] = 0
        h_b[j] = 1
        nc = _exact_c5(h_b)
        d = nc - cur_c5
        if d < 0 or rng.random() < np.exp(-d / max(T, T_min)):
            cur_c5 = nc
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_b.copy()
                patience = 0
            else:
                patience += 1
        else:
            h_b[i] = 1
            h_b[j] = 0
        T = max(T_min, T * 0.9998)
    return best_h, best_c5

def _lp_polish(h, deadline, top_k=20, max_iters=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        scores = _corr_values(h)
        top = np.argsort(-scores)[:top_k]
        G = np.zeros((top_k, n))
        for idx, r in enumerate(top):
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                G[idx, d:] += (1.0 - h[:m])
                G[idx, :m] += -h[d:]
            else:
                g = -d
                m = n - g
                G[idx, :m] += (1.0 - h[g:])
                G[idx, g:] += -h[:m]
        c = np.zeros(n)
        A_ub = -G
        b_ub = np.zeros(top_k)
        A_eq = np.ones((1, n))
        bounds = [(0.0, 1.0)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([target]),
                          bounds=bounds, method='highs')
            if res.success:
                h_new = res.x
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    continue
        except Exception:
            pass
        break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_refine(h, deadline, max_passes=80):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(7777)
    stall = 0
    while time.time() < deadline and stall < 25:
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(min(200, len(interior) * 3)):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            for s in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = -s
                d[j] = s
                best_t = 0.0
                best_v = _exact_c5(h)
                for t in [0.02, 0.05, 0.1, 0.2, 0.4]:
                    ht = _project(h + t * s * d, target)
                    v = _exact_c5(ht)
                    if v < best_v:
                        best_v = v
                        best_t = t * s
                if best_v < best_c5 - 1e-13:
                    h = _project(h + best_t * d, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                        interior = np.where((h > 0.01) & (h < 0.99))[0]
                        break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pat = np.zeros(period)
    fill = max(1, int(duty * period))
    pat[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pat, reps)
    h[reps * period:] = pat[:n - reps * period]
    return _project(h, n / 2.0)
