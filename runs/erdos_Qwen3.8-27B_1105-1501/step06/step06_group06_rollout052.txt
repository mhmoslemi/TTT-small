```python
def _lp_minimax_step(h, t_target, n_lags=None):
    """Find direction d that decreases max correlation below t_target"""
    n = len(h)
    dx = 2.0 / n
    # Linearize all (or top) correlations at h
    # R(s) ≈ R(s;h) + grad_s · d
    # Want: R(s;h) + grad_s · d <= t_target for all s
    #      sum(d) = 0
    #      |d| <= radius (or |d|_inf small)
    # Minimize: 0 (feasibility) or sum of violations
    
    # Use top K lags for efficiency
    corr = _corr_all(h)
    K = min(50, 2*n-1)
    top_k = np.argsort(-corr)[:K]
    
    # Build constraint matrix
    G = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        s = k - (n-1)
        if s >= 0:
            m = n - s
            G[idx, :m] += (1.0 - h[s:s+m]) * dx
            G[idx, s:s+m] += (-h[:m]) * dx
        else:
            g = -s
            m = n - g
            G[idx, g:g+m] += (1.0 - h[:m]) * dx
            G[idx, :m] += (-h[g:g+m]) * dx
    
    # LP: find d minimizing max of (corr[top_k] + G·d)
    # Equivalent to: minimize t_s.t. G·d + corr[top_k] <= t, sum(d)=0
    # Variables: d[0..n-1], t
    # c = [0,...,0, 1] (n+1 variables)
    c_vec = np.zeros(n+1)
    c_vec[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((K,1))])
    b_ub = -corr[top_k]
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    bounds = [(-0.3, 0.3)] * n + [(0, None)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0],
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except:
        pass
    return None, None
```

def _exact_c5(h):

    dx = 2.0 / n

def _corr_all(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_sum(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_corr(h, k, n):
    """Gradient of corr[k] w.r.t. h"""
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return g * dx
        g[:m] += (1.0 - h[s:s + m]) * dx
        g[s:s + m] += (-h[:m]) * dx
    else:
        gg = -s
        m = n - gg
        if m <= 0:
            return g * dx
        g[gg:gg + m] += (1.0 - h[:m]) * dx
        g[:m] += (-h[gg:gg + m]) * dx
    return g
