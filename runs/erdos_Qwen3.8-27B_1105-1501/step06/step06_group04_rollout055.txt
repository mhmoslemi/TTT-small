def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

def _direction_topk(h, k, mode='avg'):
    """Direction from top-k active constraints."""
    n = len(h)
    C = _corr_scaled(h)
    Cmax = C.max()
    if mode == 'top_avg':
        # Average gradient of top-k constraints
        idx = np.argsort(-C)[:k]
        G = _grad_matrix(h, idx)
        d = -G.mean(axis=0)
    elif mode == 'single':
        idx = np.argmax(C)
        d = -_grad_row_unscaled(h, idx - (n - 1))
    elif mode == 'diff':
        # Difference between top and second gradient
        idx = np.argsort(-C)[:2]
        if len(idx) < 2:
            return None
        g1 = _grad_row_unscaled(h, idx[0] - (n - 1))
        g2 = _grad_row_unscaled(h, idx[1] - (n - 1))
        d = -(g1 - g2)
    else:
        return None
    d = d - d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm

def _direction_random(h, rng, sparsity=0.3):
    """Random balanced direction with some sparsity."""
    n = len(h)
    d = rng.normal(0, 1, n)
    # Make sparse
    mask = rng.random(n) > sparsity
    d[mask] = 0
    d = d - d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm

def _direction_pair(h, rng):
    """Random pair direction."""
    n = len(h)
    interior = np.where((h > 0.05) & (h < 0.95))[0]
    if len(interior) < 2:
        return None
    i = interior[rng.integers(len(interior))]
    j = interior[rng.integers(len(interior))]
    if i == j:
        return None
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def _quantize_parent(h, rng):
    """Create binary version from parent ordering."""
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def _fanout_lse(h, deadline, betas=None):

                            best_h = h.copy()
                            stall = 0
                        improved = True
                        break
        elif r < 0.5:
            d = _direction_topk(h, 3, 'single')
            if d is not None:
                d = d * radius
                t, new_c5 = _line_search_exact(h, d, dx, radius * 2)
                if t != 0 and new_c5 < cur_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    cur_c5 = _compute_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                    improved = True
        elif r < 0.65:
            d = _direction_topk(h, 3, 'diff')
            if d is not None:
                d = d * radius
                t, new_c5 = _line_search_exact(h, d, dx, radius * 2)
                if t != 0 and new_c5 < cur_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    cur_c5 = _compute_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                    improved = True
        elif r < 0.8:
            d = _direction_pair(h, rng)
            if d is not None:
                t, new_c5 = _line_search_exact(h, d, dx, 0.45)
                if t != 0 and new_c5 < cur_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    cur_c5 = _compute_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                    improved = True
        elif r < 0.9:
            d = _direction_quad(h, rng)
            if d is not None:
                t, new_c5 = _line_search_exact(h, d, dx, 0.35)
                if t != 0 and new_c5 < cur_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    cur_c5 = _compute_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                    improved = True
        else:
            d = _direction_random(h, rng, sparsity=0.4)
            if d is not None:
                d = d * radius * 0.7
                t, new_c5 = _line_search_exact(h, d, dx, radius * 1.5)
                if t != 0 and new_c5 < cur_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    cur_c5 = _compute_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        stall = 0
                    improved = True

        if improved:
            radius = min(radius * 1.01, 0.35)
        else:
            stall += 1
            radius *= 0.97
            if radius < 0.008:
                radius = 0.08

    return best_h, best_c5

def _subgradient_descent(h, deadline, lr=0.01, max_iters=2000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 300:
            break
        C = _corr_scaled(h)
        Cmax = C.max()
        active = np.where(C >= Cmax - 1e-4)[0]
        G = _full_grad_matrix(h)
        avg_grad = G[active].mean(axis=0)
        avg_grad = avg_grad - avg_grad.mean()
        norm = np.linalg.norm(avg_grad)
        if norm < 1e-14:
            break
        d = -avg_grad / norm
        h_new = h + lr * d
        h_new = _project(h_new, target)
        c5_new = _compute_c5(h_new)
        if c5_new < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if lr < 1e-4:
                lr = 0.005
    return best_h, best_c5

def _fanout_solvers(h, deadline, betas=None):
    if betas is None:
        betas = [30, 80, 200, 500, 1000]
    best_h = h.copy()
    best_c5 = _compute_c5(h)

    remaining = deadline - time.time()
    if remaining < 10:
        return best_h, best_c5

    time_per = remaining / (len(betas) + 3)

    for beta in betas:
        if time.time() > deadline - 15:
            break
        sub_dl = min(deadline - 10, time.time() + time_per * 0.7)
        h_try, c5_try = _slsqp_lse(best_h, beta, sub_dl, maxiter=60)
        if h_try is not None and c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    if time.time() < deadline - 20:
        sub_dl = min(deadline - 5, time.time() + (deadline - time.time()) * 0.25)
        h_try, c5_try = _slsqp_minimax(best_h, sub_dl, maxiter=40)
        if h_try is not None and c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: Quick line polish (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _polish_line(best_h, d1, max_iters=2000, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LSE SLSQP (18%)
    d2 = t0 + budget_s * 0.26
    h2, c5_2 = _fanout_solvers(best_h, d2, betas=[30, 80, 200, 500])
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Line polish after solver (12%)
    d3 = t0 + budget_s * 0.38
    h3, c5_3 = _polish_line(best_h, d3, max_iters=3000, seed=2)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Subgradient descent (8%)
    d4 = t0 + budget_s * 0.46
    h4, c5_4 = _subgradient_descent(best_h, d4, lr=0.008, max_iters=3000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-start exploration (14%)
    d5 = t0 + budget_s * 0.60
    n = len(best_h)
    starts = []
    for freq in [2, 3, 5, 7, 11]:
        for ph in [0.0, 1.2, 2.5]:
            starts.append(_sine_init(n, freq, ph))
    for p in [5, 7, 9, 11, 13, 15]:
        for duty in [0.35, 0.5, 0.65]:
            starts.append(_periodic_init(n, p, duty))
    starts.append(_quantize_parent(best_h))
    starts.append(best_h.copy())

    for s in starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_s, c5_s = _polish_line(s, sub_dl, max_iters=800, seed=3)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Second solver round (14%)
    d6 = t0 + budget_s * 0.74
    h6, c5_6 = _fanout_solvers(best_h, d6, betas=[100, 300, 800, 1500, 2500])
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Line + subgradient polish (10%)
    d7 = t0 + budget_s * 0.84
    h7, c5_7 = _polish_line(best_h, d7, max_iters=4000, seed=4)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7:
        h7b, c5_7b = _subgradient_descent(best_h, d7, lr=0.004, max_iters=2000)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Phase 8: Resolution exploration (8%)
    d8 = t0 + budget_s * 0.92
    best_n = len(best_h)
    tried_res = set()
    for dn in [-6, 6, -12, 12, -18, 18, -24, 24, -36, 36, -48, 48]:
        if time.time() > d8:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 350:
            continue
        if n_try in tried_res:
            continue
        tried_res.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d8, time.time() + budget_s * 0.008)
        h_r, c5_r = _polish_line(h_r, sub_dl, max_iters=400, seed=5)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 9: Final intensive polish (8%)
    h_f, c5_f = _polish_line(best_h, deadline, max_iters=6000, seed=6)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)

    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0_fb = np.array(initial_h_values, dtype=float)
            h0_fb = _project(h0_fb, len(h0_fb) / 2.0)
            return h0_fb, _compute_c5(h0_fb), len(h0_fb)
        except NameError:
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)

    return best_h, final_c5, len(best_h)
