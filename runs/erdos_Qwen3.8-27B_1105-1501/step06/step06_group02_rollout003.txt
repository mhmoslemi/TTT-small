def lp_trust_step(h, radius, max_rows=None):
    """
    Solve LP trust-region step. Returns (direction, lp_t) or (None, None) on failure.
    direction is v - h where v is the LP solution.
    """
    n = len(h)
    R0 = np.correlate(h, 1.0 - h, 'full')
    G = build_grad_matrix(h)  # shape (2n-1, n)
    
    # Select rows to include
    if max_rows is not None and max_rows < 2*n-1:
        idx = np.argsort(-R0)[:max_rows]
        G_sel = G[idx]
        R_sel = R0[idx]
    else:
        G_sel = G
        R_sel = R0
    
    L = G_sel.shape[0]
    
    # LP: minimize t
    # s.t. G[i,:]@v + (R[i] - G[i,:]@h) <= t  =>  G[i,:]@v - t <= G[i,:]@h - R[i]
    # 0 <= v_i <= 1
    # sum(v) = n/2
    # h_i - radius <= v_i <= h_i + radius
    
    # Variables: [t, v_0, ..., v_{n-1}]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize t
    
    # Inequality constraints: G_sel @ v - t <= G_sel @ h - R_sel
    # In linprog form: A_ub @ x <= b_ub
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0  # coefficient of t
    A_ub[:, 1:] = G_sel
    b_ub = G_sel @ h - R_sel
    
    # Equality constraint: sum(v) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    low = np.zeros(n + 1)
    high = np.zeros(n + 1)
    high[0] = np.inf  # t is free (unbounded above, but the objective minimizes it)
    # Actually t should be bounded below by something, let's use -inf
    low[0] = -1e10
    for i in range(n):
        low[i+1] = max(0.0, h[i] - radius)
        high[i+1] = min(1.0, h[i] + radius)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(low, high)), method='highs')
    
    if not res.success:
        return None, None
    
    v = res.x[1:]
    d = v - h
    # Check if direction is meaningful
    if np.linalg.norm(d) < 1e-12:
        return None, None
    
    return d, float(res.x[0])

    n = len(h)

                G[i, :m] = 1.0 - h[d:d + m]
                G[i, d:d + m] = -h[:m]
        else:
            g = -d
            G[i, g:] = 1.0 - h[:m]
            G[i, :m] = -h[g:g + m]
    return G


def line_search(h, d, S0, dx):
    n = len(h)
    pos = d > 1e-12
    neg = d < -1e-12
    hi = 2.0
    lo = -2.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(55):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val


def lp_step(h, radius, row_mode='full', n_rows=None):
    n = len(h)
    R0 = np.correlate(h, 1.0 - h, 'full')
    G = build_grad_matrix(h)
    if row_mode == 'active' and n_rows is not None:
        n_rows = min(n_rows, 2 * n - 1)
        idx = np.argsort(-R0)[:n_rows]
        G_sel = G[idx]
        R_sel = R0[idx]
    elif row_mode == 'topk' and n_rows is not None:
        n_rows = min(n_rows, 2 * n - 1)
        idx = np.argsort(-R0)[:n_rows]
        G_sel = G[idx]
        R_sel = R0[idx]
    else:
        G_sel = G
        R_sel = R0
    L = G_sel.shape[0]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    A_ub = np.empty((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G_sel
    b_ub = G_sel @ h - R_sel
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    low = np.empty(n + 1)
    high = np.empty(n + 1)
    low[0] = -1e8
    high[0] = 1e8
    for i in range(n):
        low[i + 1] = max(0.0, h[i] - radius)
        high[i + 1] = min(1.0, h[i] + radius)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(low, high)), method='highs')
    except Exception:
        return None, None
    if not res.success:
        return None, None
    v = res.x[1:]
    d = v - h
    if np.linalg.norm(d) < 1e-11:
        return None, None
    return d, float(res.x[0])


def lp_descent(h, deadline, rng, radii=None, row_mode='full', n_rows=None, max_stall=200):
    n = len(h)
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, 'full')
    stall = 0
    if radii is None:
        radii = [0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    ridx = 0
    perturbed = 0
    while time.time() < deadline and stall < max_stall:
        if ridx >= len(radii):
            ridx = len(radii) // 2
        radius = radii[ridx]
        d, lp_t = lp_step(h, radius, row_mode=row_mode, n_rows=n_rows)
        if d is None:
            stall += 1
            ridx = min(ridx + 1, len(radii) - 1)
            continue
        alpha, new_c5 = line_search(h, d, S0, 2.0 / n)
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-12:
            h = h + alpha * d
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, 'full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            perturbed = 0
        else:
            stall += 1
            if stall >= 5:
                ridx = min(ridx + 1, len(radii) - 1)
                stall = 0
                if ridx >= len(radii) - 1 and perturbed < 15:
                    idx = rng.integers(n, size=8)
                    for k in idx:
                        h[k] += rng.uniform(-0.04, 0.04)
                    h = project(h, target)
                    cur_c5 = c5_exact(h)
                    S0 = np.correlate(h, 1.0 - h, 'full')
                    ridx = 1
                    perturbed += 1
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)


def _sine_init(n, freq=2, phase=0.5):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)


def _pair_polish(h, deadline, n_pairs=12):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    rng = np.random.default_rng(999)
    stall = 0
    while time.time() < deadline and stall < 50:
        scores = S0 * dx
        max_s = float(np.max(scores))
        thr = max_s - 0.006 * max_s
        act = np.where(scores >= thr)[0]
        if len(act) == 0:
            act = np.argsort(-scores)[:8]
        G = build_grad_matrix(h)
        w = np.exp((scores[act] - max_s) / (0.01 * max_s + 1e-15))
        w /= float(np.sum(w))
        wg = G[act].T @ w
        donors = np.argsort(-wg)[:n_pairs]
        recvs = np.argsort(wg)[:n_pairs]
        best_a, best_nc, best_p = 0.0, cur_c5, None
        for i in donors:
            if time.time() > deadline:
                break
            for j in recvs:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                a, nc = line_search(h, d, S0, dx)
                if a > 1e-10 and nc < best_nc - 1e-14:
                    best_nc, best_a, best_p = nc, a, (i, j)
        if best_p is not None and best_nc < cur_c5 - 1e-13:
            i, j = best_p
            h[i] -= best_a
            h[j] += best_a
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.4:
                idx = rng.integers(n, size=5)
                for k in idx:
                    h[k] += rng.uniform(-0.03, 0.03)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    return best_h, best_c5


def _gradient_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, 'full')
    stall = 0
    while time.time() < deadline and stall < 40:
        scores = S0 * dx
        max_s = float(np.max(scores))
        thr = max_s - 0.004 * max_s
        act = np.where(scores >= thr)[0]
        if len(act) == 0:
            act = np.argsort(-scores)[:6]
        G = build_grad_matrix(h)
        w = np.exp((scores[act] - max_s) / (0.008 * max_s + 1e-15))
        w /= float(np.sum(w))
        g = G[act].T @ w
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        alpha, new_c5 = line_search(h, d, S0, dx)
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-13:
            h = h + alpha * d
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, 'full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0

    # Phase 1: Main LP descent (45%)
    d1 = t0 + budget_s * 0.45
    radii1 = [0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    h1, c51 = lp_descent(h, d1, rng, radii=radii1, row_mode='full', max_stall=120)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: LP with active rows (15%)
    d2 = t0 + budget_s * 0.60
    radii2 = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    h2, c52 = lp_descent(best_h, d2, rng, radii=radii2, row_mode='active', n_rows=n0, max_stall=100)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2.copy()

    # Phase 3: Multi-resolution (15%)
    d3 = t0 + budget_s * 0.75
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8]
    tried_n = {best_n}
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = lp_descent(h_r, sub_dl, rng, radii=[0.06, 0.03, 0.015, 0.008], max_stall=60)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Gradient polish + pair search (12%)
    d4 = t0 + budget_s * 0.87
    h4, c54 = _gradient_polish(best_h, min(d4, time.time() + budget_s * 0.05))
    if c54 < best_c5:
        best_c5 = c54
        best_h = h4.copy()
    h4b, c54b = _pair_polish(best_h, d4, n_pairs=10)
    if c54b < best_c5:
        best_c5 = c54b
        best_h = h4b.copy()

    # Phase 5: Final LP polish with tiny radii (5%)
    d5 = deadline
    radii5 = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    h5, c55 = lp_descent(best_h, d5, rng, radii=radii5, row_mode='full', max_stall=80)
    if c55 < best_c5:
        best_c5 = c55
        best_h = h5.copy()

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)

    return best_h, final_c5, len(best_h)
