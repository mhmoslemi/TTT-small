def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _to_binary(h, rng):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)

def _grad_at_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n - kk])
        g[:n - kk] -= h[kk:]
    return g


def _lp_direction(h, corr, K=10):
    n = len(h)
    active_idx = np.argsort(-corr)[:K]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, ar in enumerate(active_idx):
        k = ar - (n - 1)
        g = _grad_at_k(h, k)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[ar]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.linalg.norm(d) > 1e-12:
                return d
    except Exception:
        pass
    return None


def _weighted_grad_dir(h, corr, K=10, tau=0.015):
    n = len(h)
    active_idx = np.argsort(-corr)[:K]
    scores = corr[active_idx]
    m = float(scores.max())
    w = np.exp((scores - m) / tau)
    w = w / w.sum()
    d = np.zeros(n)
    for wi, ar in zip(w, active_idx):
        k = ar - (n - 1)
        d -= wi * _grad_at_k(h, k)
    d -= d.mean()
    nrm = np.linalg.norm(d)
    if nrm < 1e-14:
        return None
    return d / nrm

        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 20 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub = min(d3, time.time() + max(3.0, budget_s * 0.02))
        hr2, fr2 = _continuous_opt(hr, sub, rng, use_lp=True, K=8)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try
        h_bin = _to_binary(hr2, rng)
        hr3, fr3 = _binary_local_search(h_bin, min(d3, time.time() + 5), rng)
        if fr3 < best_f:
            best_f = fr3
            best_h = hr3.copy()
            bn = n_try

    # Phase 4: Final polish (10% budget)
    d4 = deadline - 10
    hc, fc = _continuous_opt(best_h, d4, rng, use_lp=True, K=12, pair_frac=0.15, tau=0.008)
    if fc < best_f:
        best_f = fc
        best_h = hc.copy()

    # Phase 5: Final binary snap + refine
    if time.time() < deadline - 5:
        h_bin = _to_binary(best_h, rng)
        hb, fb2 = _binary_local_search(h_bin, min(deadline - 5, time.time() + 20), rng)
        if fb2 < best_f:
            best_f = fb2
            best_h = hb.copy()
        if time.time() < deadline - 2:
            hc2, fc2 = _continuous_opt(best_h, deadline - 1, rng, use_lp=True, K=12, pair_frac=0.2)
            if fc2 < best_f:
                best_f = fc2
                best_h = hc2.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
