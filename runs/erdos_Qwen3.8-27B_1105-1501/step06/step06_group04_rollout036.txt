import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _lse_gradient(h, beta):
    n = len(h)
    S = _corr_raw(h)
    S_max = S.max()
    exp_S = np.exp(beta * (S - S_max))
    sum_exp = exp_S.sum()
    p = exp_S / sum_exp
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-14:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            m = n - lag
            if m <= 0:
                continue
            grad[:m] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:m])
        else:
            g = -lag
            m = n - g
            if m <= 0:
                continue
            grad[:m] += w * (1.0 - h[g:])
            grad[g:] += w * (-h[:m])
    return grad

def _lse_optimize(h0, deadline, beta_start=20.0, beta_end=200.0, lr_init=0.02, n_inner=200):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    v = np.zeros(n)
    momentum = 0.8
    lr = lr_init
    patience = 0
    
    phase = 0
    total_phases = 5
    for phase in range(total_phases):
        if time.time() > deadline:
            break
        beta = beta_start * (beta_end / beta_start) ** (phase / max(1, total_phases - 1))
        for it in range(n_inner):
            if time.time() > deadline:
                break
            grad = _lse_gradient(h, beta)
            grad -= grad.mean()  # project to sum=0
            v = momentum * v - lr * grad
            h_new = _project(h + v, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
            h = h_new
            if patience > 50:
                lr *= 0.5
                v = np.zeros(n)
                patience = 0
    return best_h, best_c5
