def compute_grads(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            m = n - s  # number of terms
            # First contribution: (1-h[j-s]) for j in [s, n-1]
            for j in range(s, n):
                G[k, j] += (1.0 - h[j-s])
            # Second contribution: -h[j+s] for j in [0, n-1-s]
            for j in range(0, m):
                G[k, j] += -h[j+s]
        else:
            r = -s
            m = n - r
            # First contribution: (1-h[j+r]) for j in [0, n-1-r]
            for j in range(0, m):
                G[k, j] += (1.0 - h[j+r])
            # Second contribution: -h[j-r] for j in [r, n-1]
            for j in range(r, n):
                G[k, j] += -h[j-r]
    return G

def solve_qp_step(h, target_raw, rho, mu):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    G = compute_grads(h)
    
    d = cp.Variable(n)
    z = cp.Variable(2*n-1)
    
    obj = cp.Minimize(cp.sum_squares(d) / (2*rho**2) + mu * cp.sum(z))
    constraints = [

        cp.sum(d) == 0,
        d >= -h,
        d <= 1 - h,
        z >= 0

    if d.value is None:
        return None
    d_val = np.asarray(d.value).flatten()

    d_val -= np.mean(d_val)  # enforce sum = 0

import numpy as np

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            for j in range(s, n):
                G[k, j] += (1.0 - h[j - s])
            for j in range(0, m):
                G[k, j] += -h[j + s]
        else:
            r = -s
            m = n - r
            for j in range(0, m):
                G[k, j] += (1.0 - h[j + r])
            for j in range(r, n):
                G[k, j] += -h[j - r]
    return G

def line_search(h, d, S0=None):
    n = len(h)
    dx = 2.0 / n
    if S0 is None:
        S0 = np.correlate(h, 1.0 - h, mode='full')

    C = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    hi_a = 1e18
    if np.any(pos):

    vals = np.array([float(np.max(S0 + a * T - a * a * C)) * dx for a in alphas])
    i0 = int(np.argmin(vals))
    lo_b = max(alphas[max(0, i0 - 2)], lo_a)
    hi_b = min(alphas[min(N - 1, i0 + 2)], hi_a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo_b, hi_b
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)

def qp_step(h, rho, mu, target_raw):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    G = compute_grads(h)
    d = cp.Variable(n)
    z = cp.Variable(2 * n - 1, nonneg=True)
    t = cp.Variable()
    obj = cp.Minimize(cp.sum_squares(d) / (2.0 * rho * rho) + mu * cp.sum(z) + t)
    constraints = [
        G @ d - z <= target_raw - R,
        cp.sum(d) == 0,
        d >= -h,
        d <= 1.0 - h
    ]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-8, verbose=False)
    except Exception:
        return None
    if d.value is None:
        return None
    d_val = np.asarray(d.value).flatten().astype(float)
    if not np.all(np.isfinite(d_val)):
        return None
    d_val -= np.mean(d_val)
    d_val = np.clip(d_val, -h, 1.0 - h)
    d_val -= np.mean(d_val)
    return d_val

def gradient_direction(h, active_frac=0.05, n_active=15):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    scores = R
    max_s = float(np.max(scores))
    threshold = max_s - active_frac * max_s
    active = np.where(scores >= threshold)[0]
    if len(active) > n_active:
        top_k = np.argsort(-scores[active])[:n_active]
        active = active[top_k]
    G = compute_grads(h)
    g = G[active, :].T @ (scores[active] / (np.sum(scores[active]) + 1e-15))
    g -= np.mean(g)
    norm = np.linalg.norm(g)
    if norm < 1e-15:
        return None
    return -g / norm

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def sigmoid_init(n, freq, amp, phase):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)
