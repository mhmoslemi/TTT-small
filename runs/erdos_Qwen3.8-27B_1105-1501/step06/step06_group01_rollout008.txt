import numpy as np

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_jacobian(h, dx):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for q in range(m):
        k = q - (n - 1)
        if k >= 0:
            mc = n - k
            if mc > 0:
                G[q, :mc] += -dx * h[:mc]
                G[q, k:] += dx * (1.0 - h[:mc])
        else:
            r = -k
            mc = n - r
            if mc > 0:
                G[q, :mc] += dx * (1.0 - h[r:r+mc])
                G[q, r:r+mc] += -dx * h[:mc]
    return G

    dx = 2.0 / n
    G = _build_jacobian(x, dx)

    if U < 1e-30:
        return 0.0, np.zeros(n)
    U_factor = U ** (1.0 / p - 1.0)
    w = S_clipped ** (p - 1.0)
    grad = U_factor * (G.T @ w)
    return phi, grad
