import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _smooth_gradient(h, raw_scores, beta):
    """Compute the negative smooth-max gradient with zero mean."""
    n = len(h)
    dx = 2.0 / n
    mx = float(np.max(raw_scores))
    # Softmax weights
    v = beta * (raw_scores - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros(len(raw_scores))
        w[int(np.argmax(raw_scores))] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient slice by slice
    g = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            if m <= 0:
                continue
            # d/dh[j] for j in [tau, n-1]: (1-h[j-tau])
            g[tau:] += w[r] * (1.0 - h[:m])
            # d/dh[j] for j in [0, m-1]: -h[j+tau]
            g[:m] -= w[r] * h[tau:tau+m]
        else:
            gg = -tau
            m = n - gg
            if m <= 0:
                continue
            # d/dh[j] for j in [0, m-1]: (1-h[j+gg])
            g[:m] += w[r] * (1.0 - h[gg:])
            # d/dh[j] for j in [gg, n-1]: -h[j-gg]
            g[gg:] -= w[r] * h[:m]
    
    # Zero out mean to maintain feasibility
    g -= float(np.mean(g))
    return g

def _line_search(h, d, radius):
    """Exact line search along direction d. Returns (alpha, c5_value)."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    # Compute feasible range
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    
    if a_hi - a_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Precompute T and Q
    u = 1.0 - h
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    S0 = np.correlate(h, u, mode='full')
    
    def f(alpha):
        return float(np.max(S0 + alpha * T - alpha * alpha * Q)) * dx
    
    # Dense grid
    n_grid = 64
    grid_t = np.linspace(a_lo, a_hi, n_grid)
    grid_f = np.empty(n_grid)
    for idx in range(n_grid):
        grid_f[idx] = f(grid_t[idx])
    
    best_idx = int(np.argmin(grid_f))
    
    # Golden section refinement around best
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_b - gr * (hi_b - lo_b)
    c2 = lo_b + gr * (hi_b - lo_b)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(50):
        if hi_b - lo_b < 1e-12:
            break
        if f1 < f2:
            hi_b = c2
            c2 = c1
            f2 = f1
            c1 = hi_b - gr * (hi_b - lo_b)
            f1 = f(c1)
        else:
            lo_b = c1
            c1 = c2
            f1 = f2
            c2 = lo_b + gr * (hi_b - lo_b)
            f2 = f(c2)
    
    opt_t = (lo_b + hi_b) / 2.0
    opt_val = f(opt_t)
    
    # Also check endpoints and grid best
    candidates = [
        (opt_t, opt_val),
        (a_lo, f(a_lo)),
        (a_hi, f(a_hi)),
        (grid_t[best_idx], float(grid_f[best_idx])),
        (0.0, float(np.max(S0)) * dx),
    ]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_single(h, r):
    n = len(h)
    g = np.zeros(n)
    tau = r - (n - 1)
    if tau >= 0:
        m = n - tau
        if m > 0:
            g[tau:] += (1.0 - h[:m])
            g[:m] -= h[tau:tau + m]
    else:
        gg = -tau
        m = n - gg
        if m > 0:
            g[:m] += (1.0 - h[gg:])
            g[gg:] -= h[:m]
    g -= float(np.mean(g))
    return g

def _grad_smooth(h, raw, beta):
    n = len(h)
    mx = float(np.max(raw))
    v = beta * (raw - mx)
    v = np.clip(v, -50.0, 0.0)
    w = np.exp(v)
    wsum = float(np.sum(w))
    if wsum < 1e-30:
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    w /= wsum
    active = w > 1e-7
    if not np.any(active):
        idx = int(np.argmax(raw))
        return _grad_single(h, idx)
    g = np.zeros(n)
    for r in np.where(active)[0]:
        wr = w[r]
        tau = r - (n - 1)
        if tau >= 0:
            m = n - tau
            if m > 0:
                g[tau:] += wr * (1.0 - h[:m])
                g[:m] -= wr * h[tau:tau + m]
        else:
            gg = -tau
            m = n - gg
            if m > 0:
                g[:m] += wr * (1.0 - h[gg:])
                g[gg:] -= wr * h[:m]
    g -= float(np.mean(g))
    return g

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    pos = d > 1e-15
    neg = d < -1e-15
    a_lo = -radius
    a_hi = radius
    if np.any(pos):
        a_lo = max(a_lo, float(np.min(-h[pos] / d[pos])))
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        a_hi = min(a_hi, float(np.min(h[neg] / (-d[neg]))))
    if a_hi <= a_lo + 1e-14:
        return 0.0, _exact_c5(h)
    u = 1.0 - h
    S0 = np.correlate(h, u, mode='full')
    T = np.correlate(d, u, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    def f(a):
        return float(np.max(S0 + a * T - a * a * Q)) * dx

    ng = 80
    gt = np.linspace(a_lo, a_hi, ng)
    gf = np.empty(ng)
    for i in range(ng):
        gf[i] = f(gt[i])
    bi = int(np.argmin(gf))
    lo = gt[max(0, bi - 2)]
    hi = gt[min(ng - 1, bi + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    for _ in range(45):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    opt_t = (lo + hi) / 2.0
    opt_v = f(opt_t)
    top_idx = np.argsort(S0)[-8:]
    for r in top_idx:
        if Q[r] > 1e-15:
            av = T[r] / (2.0 * Q[r])
            if a_lo <= av <= a_hi:
                fv = f(av)
                if fv < opt_v:
                    opt_t, opt_v = av, fv
    for t in [a_lo, a_hi, 0.0]:
        ft = f(t)
        if ft < opt_v:
            opt_t, opt_v = t, ft
    return opt_t, opt_v

def _make_dirs(h, g, raw, rng, n_max=8):
    n = len(h)
    dirs = []
    gn = np.linalg.norm(g)
    if gn > 1e-12:
        d = g / gn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    r_max = int(np.argmax(raw))
    gs = _grad_single(h, r_max)
    gsn = np.linalg.norm(gs)
    if gsn > 1e-12:
        d = gs / gsn
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for blk in [4, 8, 16]:
        if n % blk != 0:
            continue
        nblk = n // blk
        gb = g.reshape(nblk, blk).mean(axis=1)
        d = np.tile(gb, (1, blk)).reshape(n).copy()
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    for freq in [1, 2]:
        x = np.arange(n, dtype=float)
        d = np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-12:
            dirs.append(d / dn)
    idx_s = np.argsort(g)
    sources = idx_s[:4]
    sinks = idx_s[-4:]
    for s in sources[:2]:
        for k in sinks[:2]:
            if s == k:
                continue
            d = np.zeros(n)
            d[s] = 1.0
            d[k] = -1.0
            d -= np.mean(d)
            dirs.append(d)
            break
    return dirs[:n_max]

def _opt_smooth(h, deadline, beta_lo=5.0, beta_hi=60.0, radius_init=0.18, max_stall=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    total_iters = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > max_stall:
            break
        total_iters += 1
        frac = float(total_iters) / float(total_iters + stall + 1)
        beta = beta_lo + (beta_hi - beta_lo) * min(1.0, frac * 2.0)
        u = 1.0 - h
        raw = np.correlate(h, u, mode='full')
        g = _grad_smooth(h, raw, beta)
        dirs = _make_dirs(h, g, raw, rng, 8)
        best_alpha = 0.0
        best_val = cur_c5
        best_d = None
        for d in dirs:
            if time.time() > deadline:
                break
            alpha, val = _line_search(h, d, radius)
            if val < best_val - 1e-14:
                best_val, best_alpha, best_d = val, alpha, d
        if best_d is not None and best_alpha != 0.0:
            h_new = h + best_alpha * best_d
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                radius = min(radius * 1.06, 0.45)
            else:
                stall += 1
                radius = max(0.04, radius * 0.75)
        else:
            stall += 1
            radius = max(0.04, radius * 0.78)
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=50000, T_init=0.003):
    n = len(h_bin)
    dx = 2.0 / n
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    T = T_init
    stall = 0
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 400 or len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[int(rng.integers(0, len(ones)))]
        oz = zeros[int(rng.integers(0, len(zeros)))]
        cur[oi] = 0.0
        cur[oz] = 1.0
        new_c5 = _exact_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(cur > 0.5)[0].tolist()
            zeros = np.where(cur <= 0.5)[0].tolist()
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9995
        else:
            cur[oi] = 1.0
            cur[oz] = 0.0
            stall += 1
    return best_h, best_c5

def _pair_sweep(h_bin, deadline):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur > 0.5)[0].tolist()
    zeros = np.where(cur <= 0.5)[0].tolist()
    for oi in ones:
        if time.time() > deadline:
            break
        for oz in zeros:
            cur[oi] = 0.0
            cur[oz] = 1.0
            c_try = _exact_c5(cur)
            if c_try < cur_c5 - 1e-14:
                cur_c5 = c_try
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = cur.copy()
                ones = np.where(cur > 0.5)[0].tolist()
                zeros = np.where(cur <= 0.5)[0].tolist()
                break
            else:
                cur[oi] = 1.0
                cur[oz] = 0.0
    return best_h, best_c5

def _resample(h, n_new):
    if n_new == len(h):
        return h.copy()
    x_old = (np.arange(len(h)) + 0.5) / len(h)
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _smooth_perturb(h, amp, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(3):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        pert += np.sin(2.0 * np.pi * freq * x / n + phase)
    pert *= amp / 3.0
    pert += amp * 0.3 * rng.normal(0, 1, n)
    return _project(h + pert, n / 2.0)

def _pair_continuous(h, deadline, n_iter=4000, radius=0.20, seed=333):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    stall = 0
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        alpha, c_try = _line_search(h, d, radius)
        if c_try < cur_c5 - 1e-13 and alpha != 0.0:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            nc = _exact_c5(h_new)
            if nc < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)

    # Phase 1: Smooth-max gradient descent (25%)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _opt_smooth(h_init, d1, beta_lo=4.0, beta_hi=40.0, radius_init=0.15, max_stall=80)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-n exploration (20%)
    d2 = t0 + budget_s * 0.45
    n_curr = len(best_h)
    for dn in range(-16, 17, 4):
        if time.time() > d2:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 240:
            continue
        if n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.015)
        h_r, c5_r = _opt_smooth(h_r, sub_dl, beta_lo=8.0, beta_hi=30.0, radius_init=0.12, max_stall=40)
        if time.time() < d2 - 2:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d2, time.time() + budget_s * 0.008)
            h_b, c5_b = _binary_sa(h_b, sub_dl2, max_iter=12000, T_init=0.002)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try

    # Phase 3: Binary optimization (20%)
    d3 = t0 + budget_s * 0.65
    n_b = len(best_h)
    h_b = _to_binary(best_h)
    sub_dl = min(d3, time.time() + budget_s * 0.08)
    h_b, c5_b = _binary_sa(h_b, sub_dl, max_iter=60000, T_init=0.004)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    if time.time() < d3:
        sub_dl2 = min(d3, time.time() + budget_s * 0.04)
        h_b, c5_b = _pair_sweep(h_b, sub_dl2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    if time.time() < d3:
        h_pol = _project(h_b.copy(), n_b / 2.0)
        sub_dl3 = min(d3, time.time() + budget_s * 0.04)
        h_pol, c5_pol = _opt_smooth(h_pol, sub_dl3, beta_lo=10.0, beta_hi=50.0, radius_init=0.10, max_stall=40)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Intensive continuous optimization (20%)
    d4 = t0 + budget_s * 0.85
    n_c = len(best_h)
    sub_dl = min(d4, time.time() + budget_s * 0.06)
    h4 = _project(best_h.copy(), n_c / 2.0)
    h4, c5_4 = _opt_smooth(h4, sub_dl, beta_lo=15.0, beta_hi=80.0, radius_init=0.10, max_stall=80)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl2 = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _pair_continuous(h4, sub_dl2, n_iter=5000, radius=0.18, seed=444)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    if time.time() < d4:
        h4 = _project(best_h.copy(), n_c / 2.0)
        sub_dl3 = min(d4, time.time() + budget_s * 0.05)
        h4, c5_4 = _opt_smooth(h4, sub_dl3, beta_lo=25.0, beta_hi=100.0, radius_init=0.06, max_stall=50)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Basin hopping + final polish (15%)
    d5 = deadline
    scale = 0.02
    n_restarts = 0
    while time.time() < d5 - 10 and n_restarts < 80:
        n_restarts += 1
        h_p = _smooth_perturb(best_h, scale, rng)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.008)
        h_t, c5_t = _opt_smooth(h_p, sub_dl, beta_lo=10.0, beta_hi=50.0, radius_init=0.08, max_stall=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.06, scale * 1.08)

    # Final: try a few nearby n values
    n_final = len(best_h)
    for dn in [-4, 4, -8, 8]:
        if time.time() > d5 - 15:
            break
        n_try = n_final + dn
        if n_try < 60 or n_try > 240 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5 - 5, time.time() + budget_s * 0.015)
        h_r, c5_r = _opt_smooth(h_r, sub_dl, beta_lo=15.0, beta_hi=60.0, radius_init=0.08, max_stall=35)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if deadline - time.time() > 5:
        h_f = _project(best_h.copy(), len(best_h) / 2.0)
        h_f, c5_f = _opt_smooth(h_f, deadline, beta_lo=20.0, beta_hi=80.0, radius_init=0.05, max_stall=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
