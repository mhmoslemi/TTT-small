    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _bp_mat(b, dx):
    """Matrix for LP fixing b: M[k,i] = dx*corr(e_i,b,'full')[k]"""
    n = len(b)
    M = np.zeros((2*n-1, n))
    for i in range(n):
        M[n-1-i : n-1-i+n, i] = dx * b
    return M

def _bn_mat(a, dx):
    """Matrix for LP fixing a: N[k,m] = dx*corr(a,e_m,'full')[k]"""
    n = len(a)
    N = np.zeros((2*n-1, n))
    rev = a[::-1].copy()
    for m in range(n):
        N[m : m+n, m] = dx * rev
    return N

def _lp_solve(M, dx, target):
    """min tau s.t. M@x <= tau, sum(x)=target, 0<=x<=1"""
    n = M.shape[1]
    m = M.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        x = np.clip(res.x[:n], 0.0, 1.0)
        return _proj(x, target)
    return None

def _bridge_search(a, b2, target, n_alpha=33):
    """Search along h(alpha) = (1-alpha)*a + alpha*(1-b2), both sum to target"""
    best_h = a.copy()
    best_c5 = _exact_c5(a)
    comp = 1.0 - b2
    comp = _proj(comp, target)
    a = _proj(a, target)
    for alpha in np.linspace(0.0, 1.0, n_alpha):
        h = (1.0 - alpha) * a + alpha * comp
        h = _proj(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _lp_alternate(h, deadline, max_iter=200):
    """Alternating LP method"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = _proj(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Step 1: Fix b = 1-h, solve LP for a
        b = 1.0 - h
        b = _proj(b, target)
        M_a = _bp_mat(b, dx)
        a_opt = _lp_solve(M_a, dx, target)
        if a_opt is None:
            break
        # Step 2: Fix a_opt, solve LP for b
        M_b = _bn_mat(a_opt, dx)
        b_opt = _lp_solve(M_b, dx, target)
        if b_opt is None:
            break
        # Step 3: Bridge search
        h_new, c5_new = _bridge_search(a_opt, b_opt, target)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Try just the endpoints
            c5_a = _exact_c5(a_opt)
            if c5_a < best_c5 - 1e-12:
                best_c5 = c5_a
                best_h = a_opt.copy()
                h = a_opt.copy()
            else:
                comp = _proj(1.0 - b_opt, target)
                c5_c = _exact_c5(comp)
                if c5_c < best_c5 - 1e-12:
                    best_c5 = c5_c
                    best_h = comp.copy()
                    h = comp.copy()
                else:
                    break  # No improvement, stop
    return best_h, best_c5

def _local_lp(h, deadline, radius=0.15):
    """LP-based local direction search (like previous but refined)"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.time() < deadline:
        if stall > 60:
            if stall < 100:
                h = _proj(h + rng.normal(0, 0.02, n), target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            elif stall < 150:
                # Binary restart with noise
                idx = np.argsort(-h)
                h_bin = np.zeros(n)
                h_bin[idx[:int(target)]] = 1.0
                h = _proj(h_bin + rng.normal(0, 0.05, n), target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                break
        
        # Build gradient matrix for active lags
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        # Linearized direction: find d that minimizes max_k (S_k + G_k @ d)
        # G_k @ d = dx * (corr(d, 1-h, 'full')[k] - corr(h, d, 'full')[k])
        # Build G matrix
        G = np.zeros((2*n-1, n))
        for i in range(n):
            # corr(e_i, 1-h, 'full') * dx
            ki = n - 1 - i
            G[ki:ki+n, i] += dx * (1.0 - h)
            # -corr(h, e_i, 'full') * dx
            # corr(h, e_i, 'full')[k] = h[k - (n-1) + (n-1-i)] = h[k-i] if valid
            # Actually: corr(h, e_i, 'full')[k] = sum_j h[j] * e_i[j+k-(n-1)]
            # = h[n-1-i-k] if 0 <= n-1-i-k <= n-1
            # So for column i, rows k where n-1-i-k in [0,n-1]: k in [n-1-i-(n-1), n-1-i] = [-i, n-1-i]
            # Clamped: k in [0, n-1-i]
            length = n - i
            G[0:length, i] += -dx * h[:length][::-1]
        
        # LP: min s.t. G@d + S <= s, sum(d)=0, |d| <= radius
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([G, -np.ones((2*n-1, 1))])
        b_ub = -S
        A_eq = np.ones((1, n + 1))
        A_eq[0, -1] = 0.0
        b_eq = np.zeros(1)
        bounds = [(-radius, radius)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            # Line search
            S0 = S
            T = (np.correlate(d, 1.0-h, mode='full') - np.correlate(h, d, mode='full')) * dx
            C = np.correlate(d, d, mode='full') * dx
            # Binary search for best step
            lo, hi = -1.0, 1.0
            pos = d > 1e-15
            neg = d < -1e-15
            if np.any(pos):
                lo = max(lo, float(np.min(-h[pos] / d[pos])))
                hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
            if np.any(neg):
                lo = max(lo, float(np.min((1.0 - h[neg]) / d[neg])))
                hi = min(hi, float(np.min(-h[neg] / d[neg])))
            if hi - lo < 1e-12:
                stall += 1
                continue
            grid = np.linspace(lo, hi, 50)
            vals = np.max(S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:], axis=1)
            best_i = int(np.argmin(vals))
            best_t = grid[best_i]
            best_val = float(vals[best_i])
            # Golden section
            gi = (np.sqrt(5.0)-1.0)/2.0
            a_g, b_g = (lo, hi) if best_i <= 0 else (hi, lo) if best_i >= 49 else (grid[best_i-1], grid[best_i+1])
            if best_i > 0 and best_i < 49:
                a_g, b_g = grid[best_i-1], grid[best_i+1]
            c1 = b_g - gi*(b_g-a_g)
            c2 = a_g + gi*(b_g-a_g)
            f1 = float(np.max(S0 + c1*T - c1*c1*C))
            f2 = float(np.max(S0 + c2*T - c2*c2*C))
            for _ in range(30):
                if b_g - a_g < 1e-12:
                    break
                if f1 < f2:
                    b_g, c2, f2 = c2, c1, f1
                    c1 = b_g - gi*(b_g-a_g)
                    f1 = float(np.max(S0 + c1*T - c1*c1*C))
                else:
                    a_g, c1, f1 = c1, c2, f2
                    c2 = a_g + gi*(b_g-a_g)
                    f2 = float(np.max(S0 + c2*T - c2*c2*C))
            opt_t = (a_g+b_g)/2.0
            opt_val = float(np.max(S0 + opt_t*T - opt_t*opt_t*C))
            if opt_val < best_val:
                best_t, best_val = opt_t, opt_val
            
            if best_val < cur_c5 - 1e-13:
                h = h + best_t * d
                h = _proj(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _pair_opt(h, deadline, max_passes=100):
    """Pair coordinate descent"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(S))
        # Gradient at the active lag
        # dS_k/dh[i] = dx*(1-h[i+k-n+1]) - dx*h[k-i] (roughly)
        # Just try pairs
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        # Greedy: try moving from high to low
        # Sort by something related to the gradient
        # For lag k_star: contribution of h[i] is h[i]*(1-h[i+k_star-n+1])
        # Moving mass from i to j: delta ≈ h[i-dx_i]*(1-h[j+lag]) - (1-h[j])*(h[i+lag])... complex
        # Just try random pairs
        n_try = min(30, len(interior))
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i1 = int(rng.integers(len(interior)))
            i2 = int(rng.integers(len(interior)))
            if i1 == i2:
                continue
            i, j = interior[i1], interior[i2]
            for sign in [1.0, -1.0]:
                d = np.zeros(n)
                d[i] = -sign
                d[j] = sign
                # Quick line search
                S0 = np.correlate(h, 1.0-h, mode='full') * dx
                T = (np.correlate(d, 1.0-h, mode='full') - np.correlate(h, d, mode='full')) * dx
                C = np.correlate(d, d, mode='full') * dx
                # Best step in [-1, 1] clipped by bounds
                pos = d > 1e-15
                neg = d < -1e-15
                lo = -1.0
                hi = 1.0
                if np.any(pos):
                    lo = max(lo, float(np.min(-h[pos] / d[pos])))
                    hi = min(hi, float(np.min((1.0-h[pos]) / d[pos])))
                if np.any(neg):
                    lo = max(lo, float(np.min((1.0-h[neg]) / d[neg])))
                    hi = min(hi, float(np.min(-h[neg] / d[neg])))
                if hi - lo < 1e-12:
                    continue
                grid = np.linspace(lo, hi, 20)
                vals = np.max(S0[None,:] + grid[:,None]*T[None,:] - (grid[:,None]**2)*C[None,:], axis=1)
                best_t = grid[int(np.argmin(vals))]
                best_val = float(np.min(vals))
                if best_val < best_c5 - 1e-13:
                    h = _proj(h + best_t * d, target)
                    c5 = _exact_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _proj(h_new, n_new / 2.0)

def _diverse_start(n, rng):
    """Generate diverse initial conditions"""
    strategy = rng.integers(5)
    if strategy == 0:
        # Sine wave
        freq = int(rng.integers(1, 10))
        phase = rng.uniform(0, 2*np.pi)
        x = np.arange(n)
        h = 0.5 + 0.45*np.sin(2*np.pi*freq*x/n + phase)
    elif strategy == 1:
        # Binary blocks
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, int(rng.integers(1, n//4 + 1)))
            if rng.random() < 0.5:
                h[pos:pos+bl] = 1.0
            pos += bl
    elif strategy == 2:
        # Multiple frequencies
        h = np.zeros(n)
        x = np.arange(n)
        for _ in range(3):
            f = int(rng.integers(1, 8))
            p = rng.uniform(0, 2*np.pi)
            h += 0.15 * np.sin(2*np.pi*f*x/n + p)
        h = 0.5 + h
    elif strategy == 3:
        # Random smooth
        h = np.zeros(n)
        for _ in range(5):
            center = rng.uniform(0, n)
            width = rng.uniform(n//10, n//3)
            h += np.exp(-0.5*((np.arange(n)-center)/width)**2)
        h = h / max(1.0, h.max()) * 0.8 + 0.1
    else:
        # Block pattern
        block_size = int(rng.integers(3, 15))
        pattern = (rng.random(block_size) < 0.5 + 0.2*rng.standard_normal(block_size)).astype(float)
        h = np.tile(pattern, n // len(pattern) + 1)[:n]
    return _proj(h, n / 2.0)

def _exact_c5(h):

    dx = 2.0 / n

def _proj(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:

def _bp_mat(b, dx):
