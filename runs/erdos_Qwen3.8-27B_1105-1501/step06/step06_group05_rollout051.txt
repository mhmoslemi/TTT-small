    t = 0.5 * (a + b)
    return t, _ls_obj(t, corr, T, Q, dx)


def _active_direction(h, corr, k_active, tau, n):
    scores = corr
    m = float(np.max(scores))
    top_idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    for r in top_idx:
        ell = r - (n - 1)
        w = np.exp((scores[r] - m) / tau)
        if ell >= 0:
            i = np.arange(0, n - ell)
        else:
            i = np.arange(-ell, n)
        j = i + ell
        G[i] += w * (1.0 - h[j])
        G[j] += w * (-h[i])
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new * 0.5)


def _pair_polish(h, n, dx, corr, n_pairs=20, rng=None):
    best_h = h.copy()
    best_f = _exact_c5(h)
    for _ in range(n_pairs):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        L = max(-h[i], h[j] - 1.0)
        U = min(1.0 - h[i], h[j])
        if U - L < 1e-12:
            continue
        corr = np.correlate(h, 1.0 - h, mode='full')
        t, f_new = _line_search(h, d, n, dx, corr, L, U)
        if abs(t) > 1e-12:
            h_new = h + t * d
            h_new = _project(h_new, n * 0.5)
            f_new = _exact_c5(h_new)
            if f_new < best_f - 1e-12:
                best_f = f_new
                best_h = h_new.copy()
                h = h_new
    return best_h, best_f


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None

    if base is None or base.size < 10:
        n0 = 122
        base = _sine_init(n0, 1, 0.0, rng)
    else:
        n0 = len(base)

    h0 = _project(base, n0 * 0.5)
    best_h = h0.copy()
    best_f = _exact_c5(h0)
    n = n0
    dx = 2.0 / n
    lag_idx = _precompute_lag_idx(n)

    # Phase 1: Aggressive L-BFGS-B with growing active set (45% budget)
    d1 = t0 + budget_s * 0.45
    stall = 0
    mu = 8.0
    nu = n * 3.0

    while time.time() < d1:
        elapsed_frac = (time.time() - t0) / max(0.1, d1 - t0)
        M = min(2 * n - 1, 10 + int(45 * elapsed_frac))
        sub_dl = min(deadline, time.time() + 25)
        h_new = _lbfgs_step(h0, n, dx, lag_idx, M, mu, nu, sub_dl)
        f_new = _exact_c5(h_new)
        if f_new < best_f - 1e-11:
            best_f = f_new
            best_h = h_new.copy()
            h0 = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 4:
                mu = min(mu * 1.4, 150.0)
                h0 = _project(h0 + rng.normal(0, 0.015, n), n * 0.5)
                stall = 0

    h0 = best_h.copy()

    # Phase 2: Hybrid L-BFGS-B + line search + pair polish (30% budget)
    d2 = t0 + budget_s * 0.75
    iter_ct = 0
    sigma = 0.03
    while time.time() < d2:
        sub_dl = min(d2, time.time() + 12)
        r = rng.random()
        if r < 0.50:
            M_cur = min(2 * n - 1, 20 + int(30 * rng.random()))
            h_new = _lbfgs_step(h0, n, dx, lag_idx, M_cur, 12.0, n * 3.0, sub_dl)
            f_new = _exact_c5(h_new)
        elif r < 0.70:
            corr = np.correlate(h0, 1.0 - h0, mode='full')
            d = _active_direction(h0, corr, 12, 0.015, n)
            if d is None:
                continue
            pos = d > 1e-15
            neg = d < -1e-15
            L = 0.0
            U = 1e10
            if np.any(pos):
                U = min(U, float(np.min((1.0 - h0[pos]) / d[pos])))
                L = max(L, float(np.min(-h0[pos] / d[pos])))
            if np.any(neg):
                L = max(L, float(np.min((1.0 - h0[neg]) / d[neg])))
                U = min(U, float(np.min(-h0[neg] / d[neg])))
            if U - L < 1e-12:
                continue
            t, f_new = _line_search(h0, d, n, dx, corr, L, U)
            h_new = _project(h0 + t * d, n * 0.5)
        else:
            h_new, f_new = _pair_polish(h0, n, dx, None, n_pairs=15, rng=rng)

        iter_ct += 1
        f_cur = _exact_c5(h0)
        if f_new < best_f - 1e-11:
            best_f = f_new
            best_h = h_new.copy()
            h0 = h_new.copy()
        elif f_new < f_cur + 5e-4:
            h0 = h_new.copy()
        else:
            h0 = best_h.copy()
            if iter_ct % 20 == 0:
                h0 = _project(best_h + rng.normal(0, sigma, n), n * 0.5)
                sigma = max(0.005, sigma * 0.9)

    h0 = best_h.copy()

    # Phase 3: Multi-resolution (15% budget)
    d3 = t0 + budget_s * 0.90
    bn = len(best_h)
    tried = set()
    deltas = [-4, 4, -8, 8, -12, 12, -16, 16, -6, 6, -10, 10, -20, 20, -2, 2]
    for dn in deltas:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 30 or n_try > 250:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        lr = _precompute_lag_idx(n_try)
        dxr = 2.0 / n_try
        sub = min(d3, time.time() + max(4.0, budget_s * 0.018))
        for M_try in [15, 30, 50]:
            if time.time() > sub:
                break
            hr2 = _lbfgs_step(hr, n_try, dxr, lr, M_try, 10.0, n_try * 2.5, sub)
            fr2 = _exact_c5(hr2)
            if fr2 < best_f - 1e-11:
                best_f = fr2
                best_h = hr2.copy()
                bn = n_try
                hr = hr2.copy()
        # Also try pair polish at this resolution
        if time.time() < d3:
            hr3, fr3 = _pair_polish(best_h, bn, 2.0 / bn, None, n_pairs=30, rng=rng)
            if fr3 < best_f - 1e-11:
                best_f = fr3
                best_h = hr3.copy()

    # Phase 4: Final deep polish (10% budget)
    d4 = deadline
    bn = len(best_h)
    n = bn
    dx = 2.0 / n
    lag_idx = _precompute_lag_idx(n)
    h0 = best_h.copy()
    mu4 = 15.0

    while time.time() < d4:
        M_cur = min(2 * n - 1, 30 + int(20 * rng.random()))
        h_new = _lbfgs_step(h0, n, dx, lag_idx, M_cur, mu4, n * 4.0, min(d4, time.time() + 10))
        f_new = _exact_c5(h_new)
        if f_new < best_f - 1e-11:
            best_f = f_new
            best_h = h_new.copy()
            h0 = h_new.copy()
        else:
            # Try pair polish as fallback
            h3, f3 = _pair_polish(h0, n, dx, None, n_pairs=10, rng=rng)
            if f3 < best_f - 1e-11:
                best_f = f3
                best_h = h3.copy()
                h0 = h3.copy()
            elif f3 < _exact_c5(h0):
                h0 = h3.copy()

    best_h = _project(best_h, len(best_h) * 0.5)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
