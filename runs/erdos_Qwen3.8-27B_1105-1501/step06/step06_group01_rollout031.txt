                    order = np.argsort(-np.abs(grad_active))

            radius *= 0.8

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    # For k = 0, 1, ..., n-1 (positive lags, q = n-1+k):
    # G[n-1+k, k:] = dx*(1-h[:n-k])  [positive part]
    # G[n-1+k, :n-k] -= dx*h[k:]  [negative part]
    for k in range(n):
        m = n - k
        G[n-1+k, k:] += dx * (1.0 - h[:m])
        G[n-1+k, :m] -= dx * h[k:k+m]
    # For k = 1, ..., n-1 (negative lags, q = n-1-k):
    # G[n-1-k, :n-k] = -dx*h[k:]  [h[i] part]
    # G[n-1-k, k:] = dx*(1-h[:n-k])  [h[i+k] part]
    for k in range(1, n):
        m = n - k
        G[n-1-k, :m] += -dx * h[k:k+m]
        G[n-1-k, k:] += dx * (1.0 - h[:m])
    return G
