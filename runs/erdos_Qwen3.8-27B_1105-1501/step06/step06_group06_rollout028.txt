def active_gradient(h, active_idx):
    """Gradient of correlations at given lags, with respect to h."""
    n = len(h)
    # For lag s (where s = k - (n-1)):
    # d/dh_i [sum_j h_j*(1-h_{j+s})] = (1-h_{i+s}) - h_{i-s} (for valid indices)
    cols = np.zeros((len(active_idx), n))
    for row, k in enumerate(active_idx):
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            cols[row, :m] = 1.0 - h[s:s+m]
            cols[row, s:] = -h[:m]
        else:
            g = -s
            m = n - g
            cols[row, g:] = 1.0 - h[:m]
            cols[row, :m] = -h[g:]
    return cols

    """Find the best weighting of active gradients to maximize descent."""
    G = active_gradient(h, active_idx[:k_max])
    m = G.shape[0]
    # Solve: min_w ||G^T w||^2 s.t. sum(w)=1, w>=0
    # Gradient of objective: 2 * G * G^T * w
    # This is a QP: min w^T (G G^T) w s.t. sum(w)=1, w>=0
    # Use scipy
    from scipy.optimize import minimize as sc_min
    w0 = np.ones(m) / m
    def obj(w):
        gv = G.T @ w
        return gv @ gv
    def obj_grad(w):
        gv = G.T @ w
        return 2.0 * G @ gv
    cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0, 'jac': lambda w: np.ones(m)}]
    bounds = [(0, 1)] * m
    res = sc_min(obj, w0, jac=obj_grad, method='SLSQP', bounds=bounds, constraints=cons)
    if res.success:
        w = res.x
        d = G.T @ w
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm > 1e-12:
            return -d / norm

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def active_gradient(h, active_idx):
    n = len(h)
    m = len(active_idx)
    cols = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        s = k - (n - 1)
        if s >= 0:
            msh = n - s
            cols[row, :msh] += (1.0 - h[s:s + msh])
            cols[row, s:] += (-h[:msh])
        else:
            g = -s
            msh = n - g
            cols[row, g:] += (1.0 - h[:msh])
            cols[row, :msh] += (-h[g:])
    return cols

def weighted_gradient_direction(h, active_idx, k_max=12):
    G = active_gradient(h, active_idx[:k_max])
    m = G.shape[0]
    if m == 0:
        return None
    w0 = np.ones(m) / m
    def wobj(w):
        gv = G.T @ w
        return float(gv @ gv)
    def wgrad(w):
        gv = G.T @ w
        return 2.0 * G @ gv
    cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0, 'jac': lambda w: np.ones(m)}]
    bounds = [(0.0, 1.0)] * m
    res = minimize(wobj, w0, jac=wgrad, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 30, 'ftol': 1e-10})
    if res.success:
        d = G.T @ res.x
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm > 1e-12:
            return -d / norm
    return None

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
