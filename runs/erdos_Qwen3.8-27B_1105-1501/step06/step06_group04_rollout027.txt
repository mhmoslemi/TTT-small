import numpy as np

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_row(h, r):
    n = len(h)
    d = r - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:]
    else:
        s = -d
        m = n - s
        if m > 0:
            g[:m] += (1.0 - h[s:])
            g[s:] -= h[:m]
    return g

def _line_search_quad(S0, T, Q, dx, h, d, t_max):
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo, t_hi = 0.0, t_max
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    if t_hi <= t_lo + 1e-15:
        return 0.0, float(np.max(S0) * dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    t_opt = (a + b) / 2.0
    f_opt = float(np.max(S0 + t_opt * T + t_opt * t_opt * Q)) * dx
    return t_opt, f_opt

def _compute_scores(h, scores_vals, n_active):
    n = len(h)
    m = 2 * n - 1
    max_s = float(scores_vals.max())
    if max_s < 1e-15:
        return np.zeros(n)
    thr = max_s - 0.05 * max_s
    active = np.where(scores_vals >= thr)[0]
    if len(active) > n_active:
        top = np.argsort(-scores_vals[active])[:n_active]
        active = active[top]
    w = scores_vals[active] - (max_s - thr)
    w = w / w.sum()
    score = np.zeros(n)
    for idx in range(len(active)):
        score += w[idx] * _grad_row(h, active[idx])
    return score

def _pair_move(h, i, j, dx, target):
    """Try to improve by transferring mass between i and j. Returns modified h."""
    n = len(h)
    d = np.zeros(n)
    if h[j] >= h[i]:
        d[i] = 1.0
        d[j] = -1.0
    else:
        d[i] = -1.0
        d[j] = 1.0
    h_new, c5_try = _try_direction(h, d, dx, target)
    return h_new, c5_try

def _try_direction(h, d, dx, target, t_max=0.5):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    t_opt, f_opt = _line_search_quad(S0, T, Q, dx, h, d, t_max)
    if t_opt < 1e-14:
        return h.copy(), float(np.max(S0) * dx)
    h_new = h + t_opt * d
    h_new = _project(h_new, target)
    c5_new = _c5(h_new)
    return h_new, c5_new

def _lp_direction(h, scores_vals, radius, top_k=15):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores_vals)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = np.zeros((K, n))
    for idx, r in enumerate(active):
        G[idx] = _grad_row(h, r)
    act_scores = scores_vals[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = []
    for j in range(n):
        bounds.append((max(-radius, -h[j]), min(radius, 1.0 - h[j])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_sa(h_start, deadline, rng, max_iters=20000):
