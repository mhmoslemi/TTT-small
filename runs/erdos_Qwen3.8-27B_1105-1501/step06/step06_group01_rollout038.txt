        return np.ones(n)

def _binary_local_search(h_bin, deadline, max_iters=5000):

    n = len(h_bin)

def _sa_binary(h_init, deadline, T_start=0.02, T_end=0.001, max_iters=10000):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)

def _binary_snap(h):

def _pair_polish(h, deadline, max_passes=30):
    """Polish using pair moves with line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        if len(interior) < 2:
            break
        # Try gradient-informed pairs
        S = _corr_vals(h)
        max_q = int(np.argmax(S))
        # Get gradient of the active constraint
        k = max_q - (n - 1)
        if k >= 0:
            m = n - k
            grad = np.zeros(n)
            grad[k:] += (1.0 - h[:m])
            grad[:m] += -h[k:]
        else:
            d_abs = -k
            m = n - d_abs
            grad = np.zeros(n)
            grad[:m] += (1.0 - h[d_abs:])
            grad[d_abs:] += -h[:m]
        
        # Move mass from high grad to low grad
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(20, len(order))):
            if time.time() > deadline:
                break
            i = order[ii]
            if h[i] <= 0.05:
                continue
            for jj in range(min(20, len(order))):
                if tried > 60:
                    break
                j = order[jj]
                if i == j or h[j] >= 0.95:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:

                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                tried += 1
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 3.0, float(v.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    n_grid = 60
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1)
    best_i = int(np.argmin(obj_grid))
    best_t = float(grid[best_i])
    best_val = float(obj_grid[best_i])
    if best_i > 0 and best_i < n_grid - 1:
        a, b = float(grid[best_i - 1]), float(grid[best_i + 1])
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[q, k:] += dx * (1.0 - h[:m])
                G[q, :m] += -dx * h[k:]
        else:
            d_abs = -k
            m = n - d_abs
            if m > 0:
                G[q, :m] += dx * (1.0 - h[d_abs:])
                G[q, d_abs:] += -dx * h[:m]
    return G

def _lp_solve(h, deadline, start_radius=0.12, max_perturb=12):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = start_radius
    stall = 0
    perturb_ctr = 0
    rng = np.random.default_rng(12345 + n)
    while time.time() < deadline:
        if stall > 70:
            if perturb_ctr < max_perturb:
                if perturb_ctr % 3 == 0:
                    h = h + rng.normal(0, 0.04, n)
                elif perturb_ctr % 3 == 1:
                    idx = rng.choice(n, size=n // 4, replace=False)
                    h[idx] += rng.normal(0, 0.08, len(idx))
                else:
                    h_bin = _binary_snap(h)
                    h = h_bin + rng.normal(0, 0.02, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.10
                stall = 0
                perturb_ctr += 1
                continue
            else:
                break
        S = _corr_vals(h)
        G = _build_G(h, dx)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
        b_ub = -float(S.max())
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for i in range(n):
            lo = max(-radius, -h[i] + 1e-12)
            hi = min(radius, 1.0 - h[i] - 1e-12)
            bounds.append((lo, hi) if lo < hi else (0.0, 0.0))
        bounds.append((None, None))
        p = None
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
        except Exception:
            pass
        if p is None or float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.72
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            predicted = float(np.max(S + G @ p))
            ratio = cur_c5 / max(predicted, 1e-15)
            if ratio < 0.6:
                radius = min(radius * 1.25, 0.4)
            elif ratio > 2.0:
                radius *= 0.7
            else:
                radius = min(radius * 1.04, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 20 == 0 and radius < 0.015:
                radius = 0.08
    return best_h, best_c5

def _subgrad_descent(h, deadline, tau_init=0.006):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    tau = tau_init
    lr = 0.004
    rng = np.random.default_rng(77)
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        S = _corr_vals(h)
        S_max = float(S.max())
        w = np.exp((S - S_max) / tau)
        wsum = float(w.sum())
        if wsum < 1e-30:
            w = np.zeros_like(S)
            w[int(np.argmax(S))] = 1.0
        else:
            w /= wsum
        G = _build_G(h, dx)
        grad = G.T @ w
        h_new = h - lr * grad
        if rng.random() < 0.1:
            h_new += rng.normal(0, 0.01, n)
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.015)
        else:
            if c5_new < _exact_c5(h) + 0.001:
                h = h_new
            lr *= 0.92
        if it % 40 == 0:
            tau *= 0.97
            lr = max(lr, 0.001)
    return best_h, best_c5

def _binary_snap(h):
    n = len(h)
    target = int(n / 2)
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def _sa_binary(h_bin, deadline, T_start=0.012, T_end=0.001, max_iters=8000):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(555 + n)
    ones = list(np.where(h >= 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    n_tot = max(500, int(2000 * (T_start / T_end) ** 0.4))
    for it in range(min(n_tot, max_iters)):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / n_tot)
        i = int(rng.integers(len(ones)))
        j = int(rng.integers(len(zeros)))
        ii, jj = ones[i], zeros[j]
        h[ii] = 0.0
        h[jj] = 1.0
        ones[i] = jj
        zeros[j] = ii
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[ii] = 1.0
            h[jj] = 0.0
            ones[i] = ii
            zeros[j] = jj
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_polish(h, deadline, max_passes=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_vals(h)
        max_q = int(np.argmax(S))
        k = max_q - (n - 1)
        if k >= 0:
            m = n - k
            grad = np.zeros(n)
            grad[k:] += (1.0 - h[:m])
            grad[:m] += -h[k:]
        else:
            d_abs = -k
            m = n - d_abs
            grad = np.zeros(n)
            grad[:m] += (1.0 - h[d_abs:])
            grad[d_abs:] += -h[:m]
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(30, len(order))):
            if time.time() > deadline:
                break
            i = int(order[ii])
            if h[i] <= 0.03:
                continue
            for jj in range(min(30, len(order))):
                if tried > 80:
                    break
                j = int(order[jj])
                if i == j or h[j] >= 0.97:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                tried += 1
        if not improved:
            interior = np.where((h > 0.05) & (h < 0.95))[0]
            if len(interior) < 2:
                break
            for _ in range(15):
                if time.time() > deadline:
                    break
                i = int(interior[rng.integers(len(interior))])
                j = int(interior[rng.integers(len(interior))])
                if i == j:
                    continue
                for sign in (1, -1):
                    d = np.zeros(n)
                    d[i] = -float(sign)
                    d[j] = float(sign)
                    alpha, c5_try = _line_search(h, d, dx, 0.4)
                    if c5_try < best_c5 - 1e-14:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved = True
            if not improved:
                break
    return best_h, best_c5
