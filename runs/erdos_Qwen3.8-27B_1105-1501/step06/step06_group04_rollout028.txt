def _lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation values
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    vals = beta * corr * dx
    vals_max = vals.max()
    exp_vals = np.exp(vals - vals_max)
    w = exp_vals / exp_vals.sum()
    # Gradient
    grad = np.zeros(n)
    # Zero lag contribution
    grad += w[n-1] * (1.0 - 2.0 * h)
    # Non-zero lag contributions
    for d in range(1, n):

        # -h[j+d] * [j <= n-1-d]
        for j in range(0, n-d):
            grad[j] -= A * h[j+d]
    return dx * grad

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))

        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

            x[mask] += res / cnt

    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * corr * dx
    m = vals.max()
    exp_vals = np.exp(vals - m)
    w = exp_vals / exp_vals.sum()
    
    # Compute beta * dx * sum_k w_k * dc_k/dh
    grad = np.zeros(n)
    # Zero lag
    grad += w[n-1] * (1.0 - 2.0 * h)
    # Non-zero lags
    for d in range(1, n):
        A = w[n-1+d] + w[n-1-d]
        if A < 1e-15:
            continue
        grad[d:] += A * (1.0 - h[:n-d])
        grad[:n-d] -= A * h[d:]
    return beta * dx * grad

def _proj_grad_lse(h, beta, target, max_iters=500, lr_init=0.01):
    """Projected gradient method for LSE objective."""
    n = len(h)
    h = h.copy()
    h = _project(h, target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = lr_init
    
    for it in range(max_iters):
        grad = _lse_grad(h, beta)
        # Backtracking line search
        alpha = lr
        cur_obj = _lse_val(h, beta)
        for _ in range(30):
            h_try = _project(h - alpha * grad, target)
            new_obj = _lse_val(h_try, beta)
            if new_obj < cur_obj - 1e-10 * alpha * np.linalg.norm(grad):
                h = h_try
                break
            alpha *= 0.5
        else:
            # No improvement, reduce learning rate
            lr *= 0.5
            continue
        
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        lr = min(lr * 1.02, 0.05)
    
    return best_h, best_c5

def _animate_beta(h, target, deadline, beta_start=3.0, beta_end=80.0, n_phases=8):
    """Run projected gradient with annealed beta."""
    betas = np.geomspace(beta_start, beta_end, n_phases)
    h = h.copy()
    for beta in betas:
        if time.time() > deadline:
            break
        it_budget = int((deadline - time.time()) * 300 / max(1, len(betas) - (betas <= beta).sum() + 1))
        it_budget = max(50, min(it_budget, 300))
        h, _ = _proj_grad_lse(h, beta, target, max_iters=it_budget, lr_init=0.005)
    return h, _exact_c5(h)

def _lp_polish(h, deadline, target):
    """LP-based direction finding polish."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.05
    stall = 0
    
    while time.time() < deadline and stall < 100:
        scores = _corr_values(h)
        max_score = float(scores.max())
        # Focus on top lags
        thr = max_score - 0.01 * max_score
        active = np.where(scores >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        
        # Compute gradient of max (approximated by average of active)
        grad = np.zeros(n)
        for idx in active:
            r = idx
            d = r - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grad[d:] += (1.0 - h[:m])
                    grad[:m] += -h[d:]
            else:
                g = -d
                m = n - g
                if m > 0:
                    grad[:m] += (1.0 - h[g:])
                    grad[g:] += -h[:m]
        
        grad /= len(active)
        # Remove mean (to maintain sum constraint approximately)
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            stall += 1
            continue
        
        direction = -grad / norm * radius
        # Clip to feasible box
        for i in range(n):
            if direction[i] > 0 and h[i] + direction[i] > 1:
                direction[i] = 1.0 - h[i]
            elif direction[i] < 0 and h[i] + direction[i] < 0:
                direction[i] = -h[i]
        
        # Project and line search
        best_alpha = 0.0
        best_val = cur_c5
        for ft in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_try = _project(h + ft * direction, target)
            val = _exact_c5(h_try)
            if val < best_val:
                best_val = val
                best_alpha = ft
        
        if best_val < cur_c5 - 1e-14:
            h = _project(h + best_alpha * direction, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            radius *= 0.7
    
    return best_h, best_c5

def _pair_refine(h, deadline, target):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.time() < deadline and stall < 30:
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(300):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            for ft in [0.1, 0.05, 0.02, 0.01]:

                    best_c5 = val
                    best_h = h.copy()
                    improved = True
                    interior = np.where((h > 0.05) & (h < 0.95))[0]
                    break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
