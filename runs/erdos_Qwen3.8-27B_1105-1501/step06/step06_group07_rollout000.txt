import numpy as np
import time


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_raw(h, other):
    return np.correlate(h, other, mode='full')


def _grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += (1.0 - h[:m])
        g[:m] -= h[k:]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        g[:m] += (1.0 - h[ga:])
        g[ga:] -= h[:m]
    return g


def _weighted_gradient(h, K, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)
    K_eff = min(K, len(corr))
    top_idx = idx[:K_eff]
    top_vals = corr[top_idx]
    spread = max(float(top_vals[0] - top_vals[-1]), 1e-12)
    alpha_eff = alpha / spread
    alpha_eff = np.clip(alpha_eff, 5.0, 500.0)
    w = np.exp(alpha_eff * (top_vals - top_vals[0]))
    w /= np.sum(w)
    grad = np.zeros(n)
    for wi, idx_val in zip(w, top_idx):
        k = int(idx_val) - (n - 1)
        grad += wi * _grad_for_lag(h, k)
    grad -= np.mean(grad)
    return grad, corr * dx


def _scores_to_vertex(scores, n, target_int):
    idx = np.argsort(-scores)
    v = np.zeros(n)
    v[idx[:target_int]] = 1.0
    return v


def _line_search_exact(h, D, max_t):
    n = len(h)
    dx = 2.0 / n
    S = _corr_raw(h, 1.0 - h)
    T = _corr_raw(D, 1.0 - h) - _corr_raw(h, D)
    Q = _corr_raw(D, D)

    pos = D > 1e-14
    neg = D < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / D[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-D[neg]))))
    if t_hi < 1e-12:
        return 0.0, _exact_c5(h)
    t_hi = min(t_hi, max_t)

    n_grid = 60
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S + c2 * T - c2 * c2 * Q)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S + opt_t * T - opt_t * opt_t * Q)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    if best_t > t_hi:
        best_t = t_hi
    return best_t, best_val


def _vertex_direction_polish(h, deadline, K, max_iters=300, alpha_base=30.0):
    n = len(h)
    target_int = n // 2
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step_scale = 1.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, corr_vals = _weighted_gradient(h, K, alpha_base)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        n_dirs = 4
        improved = False
        for dsign in range(n_dirs):
            if time.time() > deadline:
                break
            if dsign == 0:
                scores = grad
            elif dsign == 1:
                scores = -grad
            elif dsign == 2:
                scores = grad + 0.1 * np.sin(2.0 * np.pi * 3.0 * np.arange(n) / n)
            else:
                scores = -grad + 0.1 * np.cos(2.0 * np.pi * 2.0 * np.arange(n) / n)

            v = _scores_to_vertex(scores, n, target_int)
            D = v - h
            Dn = np.linalg.norm(D)
            if Dn < 1e-14:
                continue
            D = D / Dn * step_scale

            t, f_pred = _line_search_exact(h, D, step_scale * Dn / max(Dn, 1e-14))
            if t < 1e-12:
                continue
            h_new = h + t * D
            h_new = _project(h_new, target)
            f_new = _exact_c5(h_new)
            if f_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = f_new
                best_h = h.copy()
                improved = True
                step_scale = min(step_scale * 1.05, 2.0)

        if not improved:
            step_scale *= 0.75
            if step_scale < 0.05:
                break
    return best_h, best_c5


def _pair_swap_polish(h, deadline, max_rounds=50):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        grad, _ = _weighted_gradient(h, 20, 50.0)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_i, best_j = -1, -1
        n_trial = min(len(ones) * len(zeros), 500)
        idx_i = rng_pair_indices(len(ones), len(zeros), n_trial)
        for ii, jj in idx_i:
            i = ones[ii]
            j = zeros[jj]
            D = np.zeros(n)
            D[i] = -1.0
            D[j] = 1.0
            S = _corr_raw(h, 1.0 - h)
            T = _corr_raw(D, 1.0 - h) - _corr_raw(h, D)
            Q = _corr_raw(D, D)
            f0 = float(np.max(S)) * dx
            # For binary swap, t is either 0 or 1 (or intermediate for continuous)
            t_hi = 1.0
            if h[i] < 1.0 and h[j] > 0.0:
                t_hi = min((1.0 - h[j]), h[i])
            if t_hi < 1e-14:
                continue
            # Check t=1 (or t_hi)
            for t_try in [t_hi, t_hi * 0.5]:
                f_try = float(np.max(S + t_try * T - t_try * t_try * Q)) * dx
                if f_try < f0 + best_delta - 1e-14:
                    best_delta = f_try - f0
                    best_i, best_j = i, j
                    best_t = t_try
        if best_i >= 0 and best_delta < -1e-14:
            h[best_i] -= best_t
            h[best_j] += best_t
            h = _project(h, target)
            f_new = _exact_c5(h)
            if f_new < best_c5 - 1e-14:
                best_c5 = f_new
                best_h = h.copy()
        else:
            break
    return best_h, best_c5


def rng_pair_indices(n1, n2, n_total):
    import numpy as np
    rng_local = np.random.default_rng(12345)
    pairs = []
    for _ in range(n_total):
        pairs.append((rng_local.integers(n1), rng_local.integers(n2)))
    return pairs


def _steepest_descent(h, deadline, K=25, max_iters=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, _ = _weighted_gradient(h, K, 40.0)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        D = -grad / gn
        t, f_pred = _line_search_exact(h, D, step)
        if t < 1e-13:
            step *= 0.7
            continue
        h_new = h + t * D
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.8 * step:
                step = min(step * 1.1, 2.0)
            else:
                step = max(step * 0.85, 0.05)
        else:
            step *= 0.7
            if step < 0.02:
                break
    return best_h, best_c5


def _basin_hop(h, deadline, rng, max_hops=100):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    eps = 0.15

    for hop in range(max_hops):
        if time.time() > deadline:
            break
        noise = rng.normal(0, eps, n)
        noise -= np.mean(noise)
        h_pert = h + noise
        h_pert = _project(h_pert, target)
        h_pol, c5_pol = _steepest_descent(h_pert, time.time() + min(5.0, (deadline - time.time()) * 0.3), K=20, max_iters=30)
        if c5_pol < best_c5 - 1e-13:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            h = h_pol.copy()
            eps = max(0.03, eps * 0.85)
        else:
            eps = min(0.4, eps * 1.15)
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / (2.0 * n)
    x_new = (np.arange(n_new) + 0.5) / (2.0 * n_new)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _multi_seed_init(h0, n0, rng):
    seeds = [h0.copy()]
    # Symmetrized
    h_sym = 0.5 * h0 + 0.5 * h0[::-1]
    seeds.append(_project(h_sym, n0 / 2.0))
    # Shifted
    for s in [1, 2, 3]:
        h_shift = np.roll(h0, s)
        seeds.append(_project(h_shift, n0 / 2.0))
    # Binary thresholding
    scores = h0.copy()
    target_int = n0 // 2
    v = _scores_to_vertex(scores, n0, target_int)
    seeds.append(v.copy())
    # Inverse binary
    v2 = 1.0 - v
    seeds.append(v2.copy())
    # Sine perturbation
    x = np.arange(n0)
    h_sine = h0 + 0.1 * np.sin(2.0 * np.pi * 5.0 * x / n0)
    seeds.append(_project(h_sine, n0 / 2.0))
    return seeds

def budgetInner(budget_s, idx, n_seeds):
