import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -10.0, 10.0)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_all_lags(h, dx):
    """Compute gradient of all C_r with respect to h. Returns array of shape (2n-1, n)."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            # C_r = dx * sum_{i=0}^{n-d-1} h[i] * (1-h[i+d])
            m_len = n - d
            G[r, :m_len] += (1.0 - h[d:d+m_len])
            G[r, d:n] += -h[:m_len]
        else:
            g = -d  # positive
            m_len = n - g
            # C_r = dx * sum_{i=0}^{n-g-1} h[i+g] * (1-h[i])
            G[r, g:n] += (1.0 - h[:m_len])
            G[r, :m_len] += -h[g:g+m_len]
    return G * dx  # scaled by dx

def _lse_objective_and_grad(h, beta):
    """Compute LSE objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    scores = _compute_all_scores(h)
    m = len(scores)
    max_s = float(scores.max())
    z = beta * (scores - max_s)
    # Clip for numerical stability
    z = np.clip(z, -700, 700)
    exp_z = np.exp(z)
    sum_exp = float(exp_z.sum())
    p = exp_z / sum_exp
    obj = max_s + np.log(sum_exp) / beta
    # Gradient: sum_r p_r * grad_C_r * dx (but scores already include dx)
    # Actually scores = dx * correlate(...), so grad of scores w.r.t. h is dx * grad_unscaled
    # Let me recompute: C_r(h) = dx * R_r(h) where R_r = correlate(h, 1-h, 'full')[r]
    # dC_r/dh[j] = dx * dR_r/dh[j]
    # So grad of Q_beta = sum_r p_r * dC_r/dh = dx * sum_r p_r * dR_r/dh
    # I need the unscaled gradient of R_r
    G = _grad_all_lags(h, 1.0)  # unscaled (dx=1)
    # Weighted sum
    grad = (p @ G)  # shape (n,)
    return float(obj), grad

def _line_search(h, d, dx):
    """Exact line search along h + t*d, returning best t and value."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo, t_hi = -np.inf, np.inf
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    
    # Golden section search on max of quadratics
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    
    for _ in range(80):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * Q)) * dx
    return opt_t, opt_val

def _clip_direction(h, d, max_step):
    """Clip direction to stay feasible within max_step."""
    d = d - d.mean()  # ensure zero sum
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        scale = np.min((1.0 - h[pos]) / d[pos])
        if scale < max_step:
            d[pos] *= scale
    if np.any(neg):
        scale = np.min(h[neg] / (-d[neg]))
        if scale < max_step:
            d[neg] *= -scale
    # Final scale to max_step if needed
    max_abs = np.max(np.abs(d))
    if max_abs > max_step:
        d *= max_step / max_abs
    return d

def _lse_slsqp(h, beta, maxiter=200, ftol=1e-12, max_time=30):
    """Run SLSQP on LSE objective."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    t_start = time.time()
    
    def fun(x):
        if time.time() - t_start > max_time:
            return float('inf'), np.zeros(n)
        hj = np.clip(x, 0.0, 1.0)
        # Project to maintain sum constraint (SLSQP handles it but helps)
        hj = _project(hj, target)
        obj, grad = _lse_objective_and_grad(hj, beta)
        return obj, grad
    
    def eq_fun(x):
        return np.array([np.sum(x) - target])
    
    def eq_jac(x):
        return np.array([np.ones(n)])
    
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(fun, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    except Exception:
        pass
    
    return best_h, best_c5

def _direct_minimax_slsqp(h, maxiter=100, max_time=30):
    """Direct minimax: minimize M subject to M >= C_r(h) for all r."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    t_start = time.time()
    
    m = 2 * n - 1
    
    def fun(y):
        if time.time() - t_start > max_time:
            return float('inf')
        return float(y[0])
    
    def fun_grad(y):
        if time.time() - t_start > max_time:
            return float('inf'), np.zeros(n + 1)
        g = np.zeros(n + 1)
        g[0] = 1.0
        return float(y[0]), g
    
    def ineq_fun(y):
        # M - C_r(h) >= 0
        hj = y[1:]
        scores = _compute_all_scores(hj)
        return y[0] - scores
    
    def ineq_jac(y):
        hj = y[1:]
        G = _grad_all_lags(hj, dx)  # shape (m, n), already scaled
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -G
        return J
    
    def eq_fun(y):
        return np.array([np.sum(y[1:]) - target])
    
    def eq_jac(y):
        row = np.zeros(n + 1)
        row[1:] = 1.0
        return np.array([row])
    
    M0 = _compute_c5(h) + 0.01
    x0 = np.zeros(n + 1)
    x0[0] = M0
    x0[1:] = h
    
    bounds = [(0.0, M0 + 0.1)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    try:
        res = minimize(fun_grad, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = _project(res.x[1:], target)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    except Exception:
        pass
    
    return best_h, best_c5

def _line_polish(h, deadline, n_iter=2000):
    """Exact line search polish with diverse directions."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    rng = np.random.default_rng(123)
    stall = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        scores = _compute_all_scores(h)
        max_s = scores.max()
        
        # Direction 1: negative gradient of top active constraints
        thr = max_s - 0.05 * max_s
        active = np.where(scores >= thr)[0]
        
        if len(active) > 0:
            # Weighted gradient of active constraints
            w = np.exp((scores[active] - max_s) * 50) / np.sum(np.exp((scores[active] - max_s) * 50))
            G = _grad_all_lags(h, 1.0)
            d = -(w @ G[active])
            d = _clip_direction(h, d, 0.3)
        else:
            d = rng.normal(0, 1, n)
            d = _clip_direction(h, d, 0.3)
        
        if np.max(np.abs(d)) < 1e-12:
            # Try random direction
            d = rng.normal(0, 0.1, n)
            d = _clip_direction(h, d, 0.3)
        
        t, c5_try = _line_search(h, d, dx)
        if c5_try < best_c5 - 1e-14:
            h = h + t * d
            h = _project(h, target)
            cur = _compute_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Try random pair directions
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                d2 = np.zeros(n)
                d2[i] = -1.0
                d2[j] = 1.0
                t2, c5_2 = _line_search(h, d2, dx)
                if c5_2 < best_c5 - 1e-14:
                    h = h + t2 * d2
                    h = _project(h, target)
                    cur = _compute_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        stall = 0
                    break
            else:
                stall += 1
    
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng, max_iters=10000):
    """Binary simulated annealing."""
    n = len(h_init)
    target = n // 2

def _periodic_init(n, period, duty):

    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_all_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.nan_to_num(v, nan=0.5, posinf=5.0, neginf=-5.0)
    v = np.clip(v, -10.0, 10.0)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_all_lags(h, dx):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d == 0:
            G[r, :] = 1.0 - 2.0 * h
        elif d > 0:
            m_len = n - d
            G[r, d:] += (1.0 - h[:m_len])
            G[r, :m_len] += -h[d:]
        else:
            g = -d
            m_len = n - g
            G[r, :m_len] += (1.0 - h[g:])
            G[r, g:] += -h[:m_len]
    return G * dx

def _lse_objective_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = _compute_all_scores(h)
    max_s = float(scores.max())
    z = beta * (scores - max_s)
    z = np.clip(z, -700, 700)
    exp_z = np.exp(z)
    sum_exp = float(exp_z.sum())
    p = exp_z / sum_exp
    obj = max_s + np.log(sum_exp) / beta
    G = _grad_all_lags(h, dx)
    grad = (p @ G)
    return float(obj), grad

def _line_search(h, d, dx):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    t_lo, t_hi = -np.inf, np.inf
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    for _ in range(70):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * Q)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * Q)) * dx
    return opt_t, opt_val

def _clip_direction(h, d, max_step):
    d = d - d.mean()
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        scale = float(np.min((1.0 - h[pos]) / d[pos]))
        if scale < max_step:
            d[pos] *= scale
    if np.any(neg):
        scale = float(np.min(h[neg] / (-d[neg])))
        if scale < max_step:
            d[neg] *= -scale
    max_abs = float(np.max(np.abs(d)))
    if max_abs > max_step and max_abs > 0:
        d *= max_step / max_abs
    return d

def _lse_slsqp(h, beta, maxiter=200, max_time=30.0):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    t_start = time.time()

    def fun(x):
        if time.time() - t_start > max_time:
            return float('inf'), np.zeros(n)
        hj = _project(x, target)
        obj, grad = _lse_objective_and_grad(hj, beta)
        return obj, grad

    def eq_fun(x):
        return np.array([np.sum(x) - target])

    def eq_jac(x):
        return np.array([np.ones(n)])

    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(fun, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    except Exception:
        pass
    return best_h, best_c5

def _direct_minimax_slsqp(h, maxiter=80, max_time=45.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    t_start = time.time()
    m = 2 * n - 1

    def fun(y):
        if time.time() - t_start > max_time:
            return float('inf'), np.zeros(n + 1)
        g = np.zeros(n + 1)
        g[0] = 1.0
        return float(y[0]), g

    def ineq_fun(y):
        hj = y[1:]
        scores = _compute_all_scores(hj)
        return y[0] - scores

    def ineq_jac(y):
        hj = y[1:]
        G = _grad_all_lags(hj, dx)
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -G
        return J

    def eq_fun(y):
        return np.array([np.sum(y[1:]) - target])

    def eq_jac(y):
        row = np.zeros(n + 1)
        row[1:] = 1.0
        return np.array([row])

    M0 = _compute_c5(h) + 0.005
    x0 = np.zeros(n + 1)
    x0[0] = M0
    x0[1:] = h
    bounds = [(0.0, M0 + 0.05)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(fun, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None:
            h_new = _project(res.x[1:], target)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    except Exception:
        pass
    return best_h, best_c5

def _line_polish(h, deadline, n_iter=3000):

def _binary_sa(h_init, deadline, rng, max_iters=20000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.00003
    cooling = 0.9997
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    patience = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 6000:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones_list = np.where(h > 0.5)[0]
            zeros_list = np.where(h <= 0.5)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 3 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h_fb = np.array(initial_h_values, dtype=float).copy()
            h_fb = _project(h_fb, len(h_fb) / 2.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
        except NameError:
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
