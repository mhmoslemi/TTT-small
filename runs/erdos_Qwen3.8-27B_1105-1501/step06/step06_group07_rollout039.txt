import numpy as np
from scipy.optimize import minimize
import time


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _to_binary(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:m]] = 1.0
    return b


def _binary_greedy(h_bin, deadline, max_sweeps=20000, max_no_improve=200):
    n = len(h_bin)
    h = h_bin.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    corr = _corr_array(h)
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if no_improve > max_no_improve:
            break
        best_delta = 0.0
        best_swap = None
        best_corr = None
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                nc = _corr_array(h)
                nm = np.max(nc)
                if nm < cur_max - 1e-15:
                    delta = cur_max - nm
                    if delta > best_delta:
                        best_delta = delta
                        best_swap = (i, j)
                        best_corr = nc.clone()
                        best_max_new = nm
                h[j] = 0.0
            h[i] = 1.0

        if best_swap is None:
            no_improve += 1
            continue

        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
        corr = best_corr
        cur_max = best_max_new
        no_improve = 0
        if cur_max < float(np.max(best_h)) * 0 + _exact_c5(best_h) - 1e-14:
            pass
        c5_val = float(np.max(corr) * 2.0 / n)
        if c5_val < _exact_c5(best_h) - 1e-14:
            best_h = h.copy()

    return best_h, float(np.max(_corr_array(best_h)) * 2.0 / n)


def _binary_sa(h_bin, deadline, rng, T_init=0.8, T_min=0.005, cooling=0.9999, max_iters=300000):
    n = len(h_bin)
    h = h_bin.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    corr = _corr_array(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * 2.0 / n
    T = T_init
    ones_list = list(ones)
    zeros_list = list(zeros)

    for it in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        nc = _corr_array(h)
        nm = float(np.max(nc))
        delta = nm - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_max = nm
            corr = nc
            ones_list[ones_list.index(i)] = j
            zeros_list[zeros_list.index(j)] = i
            ones.add(j)
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            c5_val = nm * 2.0 / n
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling

    return best_h, best_c5


def _grad_correlation(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        idx1 = np.arange(k, n)
        g[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, m)
        g[idx2] -= h[idx2 + k]
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        idx1 = np.arange(0, m)
        g[idx1] += (1.0 - h[idx1 + ga])
        idx2 = np.arange(ga, n)
        g[idx2] -= h[idx2 - ga]
    return g


def _smooth_max_gradient(h, alpha, K=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(-corr)[:K]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.ones(K) / K
    else:
        w /= ws
    grad = np.zeros(n)
    for wi, idx in zip(w, top_idx):
        k = int(idx) - (n - 1)
        grad += wi * _grad_correlation(h, k)
    return grad, corr


def _line_search_dir(h, d, target, f0, max_t=1.0, n_grid=40):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_arr = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C_arr = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_hi = max(t_hi, 1e-12)
    t_hi = min(t_hi, max_t)
    if t_hi < 1e-12:
        return 0.0, f0
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T_arr[None, :] - (grid[:, None] ** 2) * C_arr[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
    f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    for _ in range(40):
        if b - a < 1e-11:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_arr - opt_t * opt_t * C_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _continuous_refine(h, deadline, max_iters=200, K=25, alpha=30.0):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.5

    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, corr = _smooth_max_gradient(h, alpha, K)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        direction = -grad / gn
        f0 = _exact_c5(h)
        t, f_try = _line_search_dir(h, direction, target, f0, step)
        if t < 1e-12:
            step *= 0.7
            continue
        h_new = _project(h + t * direction, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.8 * step:
                step = min(step * 1.1, 1.5)
            else:
                step *= 0.9
        else:
            step *= 0.75
            if step < 0.005:
                break
        alpha = min(alpha * 1.05, 200.0)

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
