import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += (1.0 - h[:m])
        g[:m] -= h[k:]
    else:
        a = -k
        m = n - a
        if m <= 0:
            return g
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g * dx

def _line_eval(h, d, S, T, Q, t):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(S + t * T - t * t * Q)) * dx

def _line_search_exact(h, d, max_t=2.0):
    n = len(h)
    dx = 2.0 / n
    S = _corr_raw(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    t_lo = 0.0
    t_hi = max_t
    pos = d > 1e-12
    neg = d < -1e-12
    if np.any(pos):
        t_hi = min(t_hi, float((1.0 - h[pos]).min() / d[pos].max()))
    if np.any(neg):
        t_hi = min(t_hi, float(h[neg].min() / (-d[neg].max())))
    t_hi = min(t_hi, max_t)
    if t_hi < 1e-12:
        return 0.0, _exact_c5(h), h.copy()

    n_grid = 80
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = np.max(S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :], axis=1) * dx
    best_i = int(np.argmin(vals))
    best_t = float(grid[best_i])
    best_val = float(vals[best_i])

    if 0 < best_i < n_grid - 1:
        a, b = float(grid[best_i - 1]), float(grid[best_i + 1])
    else:
        a, b = 0.0, float(t_hi)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = _line_eval(h, d, S, T, Q, c1)
    f2 = _line_eval(h, d, S, T, Q, c2)
    for _ in range(80):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _line_eval(h, d, S, T, Q, c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _line_eval(h, d, S, T, Q, c2)
    opt_t = (a + b) / 2.0
    opt_val = _line_eval(h, d, S, T, Q, opt_t)
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val

    f0 = _exact_c5(h)
    if f0 < best_val:
        best_t, best_val = 0.0, f0

    best_h = h + best_t * d
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, n / 2.0)
    return best_t, best_val, best_h

def _active_descent(h, deadline, K=45, max_iters=800, K_min=12):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0
    stall = 0
    it_count = 0

    while it_count < max_iters:
        if time.time() > deadline:
            break
        it_count += 1

        corr = _corr_raw(best_h) * (2.0 / n)
        M = float(np.max(corr))
        m_total = 2 * n - 1
        k_act = max(K_min, min(K, m_total))
        top_idx = np.argsort(-corr)[:k_act]
        top_vals = corr[top_idx]

        # Adaptive alpha for smooth max
        spread = max(top_vals[0] - top_vals[-1], 1e-12)
        alpha_adapt = min(k_act / spread, 300.0)
        w = np.exp(-alpha_adapt * (top_vals - top_vals[0]))
        w /= w.sum()

        grad = np.zeros(n)
        for idx in range(k_act):
            wi = w[idx]
            if wi < 1e-10:
                continue
            lag = int(top_idx[idx]) - (n - 1)
            grad += wi * _grad_for_lag(best_h, lag)

        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        directions = [grad / gn * step]

        # Add perturbed directions
        for p in range(3):
            rng_p = np.random.default_rng(it_count * 31 + p * 97 + n)
            d_pert = grad + 0.15 * gn * rng_p.standard_normal(n)
            d_pert -= np.mean(d_pert)
            pn = np.linalg.norm(d_pert)
            if pn > 1e-14:
                directions.append(d_pert / pn * step)

        improved = False
        for d in directions:
            t, f_new, h_new = _line_search_exact(best_h, d, max_t=step)
            if f_new < best_c5 - 1e-13:
                best_c5 = f_new
                best_h = h_new
                improved = True
                if t > 0.6 * step:
                    step = min(step * 1.25, 4.0)
                elif t < 0.08 * step:
                    step = max(step * 0.65, 0.005)
                break

        if improved:
            stall = 0
        else:
            stall += 1
            step *= 0.65
            if step < 0.002:
                break
            if stall > 40:
                break

    return best_h, best_c5

def _pair_optimization(h, deadline, max_rounds=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        order = np.argsort(best_h)
        for idx in range(n // 2):
            if time.time() > deadline:
                break
            i = int(order[idx])
            j = int(order[n - 1 - idx])
            if best_h[i] < 1e-6 or best_h[j] > 1.0 - 1e-6:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            t, f_new, h_new = _line_search_exact(best_h, d, max_t=0.6)
            if f_new < best_c5 - 1e-12:
                best_c5 = f_new
                best_h = h_new
                improved = True
                if idx > n // 4:
                    order = np.argsort(best_h)
        if not improved:
            break

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    if n_new < 2:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _basin_hopping(h, deadline, rng, n_trials=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.12

    for trial in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = best_h + rng.standard_normal(n) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)

            best_h = h_opt
            scale = max(0.02, scale * 0.82)

def _equalize_top(h, deadline, n_iter=100):
    """Try to equalize the top correlations by adjusting active set weights."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    m = 2 * n - 1

    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = _corr_raw(best_h) * (2.0 / n)
        top_idx = np.argsort(-corr)[:30]
        top_vals = corr[top_idx]
        M = float(top_vals[0])

        # Try to find direction that reduces the spread of top correlations
        # by moving mass where it helps the most active ones
        spread = M - float(top_vals[-1])
        if spread < 1e-8:
            break

        # Weight toward the most active
        w = np.exp(50.0 * (top_vals - M))
        w /= w.sum()

        grad = np.zeros(n)
        for idx in range(len(top_idx)):
            if w[idx] < 1e-8:
                continue
            lag = int(top_idx[idx]) - (n - 1)
            grad += w[idx] * _grad_for_lag(best_h, lag)

        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:

        d = grad / gn
        t, f_new, h_new = _line_search_exact(best_h, d, max_t=0.5)
        if f_new < best_c5 - 1e-13:
            best_c5 = f_new
            best_h = h_new

    return best_h, best_c5
