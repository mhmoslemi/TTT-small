def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        else:
            hi = mid
    theta = (lo + hi) / 2.0

        idx = j + k_vals
        delta[p_lo4:p_hi4+1] -= h[idx]

    best_c5 = cur_c5

    zeros_list = np.where(h <= 0.5)[0].tolist()

**Alternative formulation: minimize the L∞ norm of the correlation**

**Alternative: use the fact that the correlation is a convolution**

The correlation at lag k is:
C[k] = sum_i h[i] * (1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]

The first term is just sum(h) = n/2 (constant). So:
C[k] = n/2 - sum_i h[i]*h[i+k]

Minimizing max_k C[k] is equivalent to maximizing min_k sum_i h[i]*h[i+k].

This is an interesting reformulation! We want to maximize the minimum autocorrelation.

For binary h, sum_i h[i]*h[i+k] counts the number of positions where both h[i]=1 and h[i+k]=1. We want to maximize the minimum of these counts over all k.

This is related to the problem of finding binary sequences with good autocorrelation properties (like Barker sequences or m-sequences).

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _precompute_swap_info(n):
    info = []
    for pos in range(n):
        lo1 = pos
        hi1 = min(pos + n - 1, 2 * n - 2)
        lo2 = max(n - 1 - pos, 0)
        hi2 = min(2 * n - 2 - pos, 2 * n - 2)
        if lo1 <= hi1:
            p1 = np.arange(lo1, hi1 + 1)
            idx1 = pos + n - 1 - p1
        else:
            p1 = np.empty(0, dtype=int)
            idx1 = np.empty(0, dtype=int)
        if lo2 <= hi2:
            p2 = np.arange(lo2, hi2 + 1)
            idx2 = p2 - (n - 1) + pos
        else:
            p2 = np.empty(0, dtype=int)
            idx2 = np.empty(0, dtype=int)
        info.append((p1, idx1, p2, idx2))
    return info

def _swap_delta(h, i, j, info):
    m = 2 * len(info) - 1
    delta = np.zeros(m)
    p1, idx1, p2, idx2 = info[i]
    if len(p1) > 0:
        delta[p1] -= (1.0 - h[idx1])
    if len(p2) > 0:
        delta[p2] += h[idx2]
    p1, idx1, p2, idx2 = info[j]
    if len(p1) > 0:
        delta[p1] += (1.0 - h[idx1])
    if len(p2) > 0:
        delta[p2] -= h[idx2]
    return delta

def _fast_binary_sa(h_init, deadline, rng, max_patience=50000):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    corr = _corr_raw(h)
    cur_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    info = _precompute_swap_info(n)
    ones_arr = sorted_idx[:target].copy()
    zeros_arr = sorted_idx[target:].copy()
    T = 0.012
    T_min = 0.000003
    cooldown = 0.99996
    patience = 0
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 5:
                break
            T = max(T_min, T * 0.8)
            patience = 0
        batch = 400
        for _ in range(batch):
            if time.time() > deadline:
                break
            ii = rng.integers(len(ones_arr))
            jj = rng.integers(len(zeros_arr))
            i = ones_arr[ii]
            j = zeros_arr[jj]
            d = _swap_delta(h, i, j, info)
            new_cand = corr + d
            new_c5 = float(np.max(new_cand)) * dx
            delta_obj = new_c5 - cur_c5
            if delta_obj < 0 or (delta_obj < 2.0 * T and rng.random() < np.exp(-delta_obj / T)):
                h[i] = 0.0
                h[j] = 1.0
                corr = new_cand
                cur_c5 = new_c5
                ones_arr[ii] = j
                zeros_arr[jj] = i
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for r, idx in enumerate(indices):
        k = idx - (n - 1)
        g = G[r]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 50
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    tol = max(0.08 * M, 1e-5)
    near_idx = np.where(S >= M - tol)[0]
    if len(near_idx) < K:
        sorted_idx = np.argsort(-S)
        near_idx = np.unique(np.concatenate([near_idx, sorted_idx[:K]]))
    if len(near_idx) > 35:
        sorted_idx = np.argsort(-S)
        near_idx = sorted_idx[:35]
    G = _grad_rows(h, near_idx, n)
    n_act = len(near_idx)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[near_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, K=20, max_iter=5000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        if stall > 50:
            break
        S = _corr_raw(h) * dx
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.05, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.93, 0.015)
        else:
            stall += 1
            radius *= 0.72
            if stall % 15 == 0 and radius < 0.01:
                radius = 0.06
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks <= 1:
        h[:] = 1.0
        return _project(h, n / 2.0)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _pair_refine(h, deadline, max_passes=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        S = _corr_raw(h) * dx
        G = _grad_rows(h, np.arange(2 * n - 1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_sel = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_sel]
        improved = False
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.35)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _cooling_sa_phase(h_init, deadline, rng, n_starts=3):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = _exact_c5(h_init)
    sub_budget = (deadline - time.time()) / max(n_starts, 1)
    for s in range(n_starts):
        sub_dl = time.time() + sub_budget
        if time.time() > deadline:
            break
        if s == 0:
            h_try = h_init.copy()
        else:
            noise_scale = 0.05 + 0.03 * s
            h_try = h_init + rng.normal(0, noise_scale, n)
            h_try = _project(h_try, n / 2.0)
        h_sa, c5_sa = _fast_binary_sa(h_try, sub_dl, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick LP to refine initial (5%)
    d1 = t0 + budget_s * 0.05
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.12, K=18)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Multi-start fast binary SA (30%)
    d2 = t0 + budget_s * 0.35
    n_starts = 5
    sub_dl = (d2 - time.time()) / n_starts
    starts = [best_h.copy()]
    for freq in [1, 3, 5]:
        starts.append(_sine_init(n0, freq, rng.random() * 6.28))
    for _ in range(2):
        starts.append(_block_init(n0, rng, rng.integers(3, 12)))
    for s_h in starts:
        if time.time() > d2:
            break
        h_sa, c5_sa = _fast_binary_sa(s_h, min(d2, time.time() + sub_dl), rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: LP refinement of best (15%)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.15, K=25)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Long fast SA from refined solution (15%)
    d4 = t0 + budget_s * 0.65
    h4, c5_4 = _fast_binary_sa(best_h, d4, rng, max_patience=80000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: LP + SA iteration (15%)
    d5 = t0 + budget_s * 0.80
    for _round in range(3):
        if time.time() > d5:
            break
        sub_lp = min(d5, time.time() + budget_s * 0.04)
        h_r1, c5_r1 = _lp_loop(best_h, sub_lp, radius_init=0.10, K=20)
        if c5_r1 < best_c5:
            best_c5 = c5_r1
            best_h = h_r1.copy()
        sub_sa = min(d5, time.time() + budget_s * 0.04)
        h_r2, c5_r2 = _fast_binary_sa(best_h, sub_sa, rng)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()

    # Phase 6: Multi-resolution (12%)
    d6 = t0 + budget_s * 0.92
    base_n = len(best_h)
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18]:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 60 or n_try > 250 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d6, time.time() + budget_s * 0.025)
            h_r, c5_r = _fast_binary_sa(h_r, sub_dl, rng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            if time.time() < d6 - budget_s * 0.01:
                sub_dl2 = min(d6, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = _lp_loop(best_h, sub_dl2, radius_init=0.08, K=15)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase 7: Pair refinement + final LP (8%)
    d7 = t0 + budget_s * 0.965
    h7, c5_7 = _pair_refine(best_h, min(d7, time.time() + budget_s * 0.03), max_passes=200)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.04, K=20)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
