    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):

        else:

def _exact_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project to [0,1]^n with sum = target (water-filling)."""
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x

def _coord_opt_h(h, i):
    """Find optimal h[i] given others. Returns new value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    kags = np.arange(2*n-1) - (n-1)
    idx_l = i - kags
    idx_r = i + kags
    L = np.where((idx_l >= 0) & (idx_l < n), h[np.clip(idx_l, 0, n-1)], 0.0)
    R = np.where((idx_r >= 0) & (idx_r < n), h[np.clip(idx_r, 0, n-1)], 0.0)
    L[(idx_l < 0) | (idx_l >= n)] = 0.0
    R[(idx_r < 0) | (idx_r >= n)] = 0.0
    slopes = 1.0 - L - R
    intercepts = corr - h[i] * slopes

    # Check boundaries

    f1 = float(np.max(intercepts + slopes))

    n = len(h)
    target = n / 2.0
    h = h.copy()

    best_h = h.copy()

def _binary_sa(h, deadline, max_iters=40000, seed=0):
    """Binary simulated annealing."""
    n = len(h)
    target = n // 2  # For binary, n must be even for exact target
    if target * 2 != n:
        target = n / 2.0
        # Fall through to non-binary
        return h, _exact_c5(h)
    
    # Initialize binary from h
    sorted_idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    
    rng = np.random.default_rng(seed)
    T = 0.01
    T_min = 1e-6
    cooling = 0.99995
    ones = list(np.where(h_b > 0.5)[0])
    zeros = list(np.where(h_b <= 0.5)[0])
    
    for it in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_c5 = _exact_c5(h_b)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
        T *= cooling
    
    return best_h, best_c5

def _resample(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    grid_t = np.linspace(0.0, 1.0, 21)
    grid_f = np.max(intercepts[None, :] + grid_t[:, None] * slopes[None, :], axis=1)
    gi = int(np.argmin(grid_f))
    if grid_f[gi] < best_f:
        best_f = grid_f[gi]
        best_t = float(grid_t[gi])
    return best_t


def _coord_descent(h, deadline, max_sweeps=500, order='random', seed=0):
    n = len(h)
    target = n / 2.0
    h = _project(h.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    prev_best = best_c5
    stagnant = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order_arr = np.arange(n)
        if order == 'random':
            rng.shuffle(order_arr)
        elif order == 'reverse':
            order_arr = order_arr[::-1]
        changed = False
        for i in order_arr:
            nv = _coord_opt_val(h, i)
            if abs(h[i] - nv) > 1e-13:
                h[i] = nv
                changed = True
        h = _project(h, target)
        c5 = _exact_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if not changed:
            break
        if stagnant > 30 and sweep > 50:
            break
    return best_h, best_c5


def _binary_sa(h_init, deadline, max_iters=50000, seed=0):
    n = len(h_init)
    if n % 2 != 0:
        return h_init.copy(), _exact_c5(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h_b = np.zeros(n)
    h_b[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(seed)
    T = 0.012
    T_min = 1e-6
    cooling = 0.99993
    ones = list(np.where(h_b > 0.5)[0])
    zeros = list(np.where(h_b <= 0.5)[0])
    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_c5 = _exact_c5(h_b)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
        T *= cooling
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _make_diverse_start(n, rng, variant):
    h = np.zeros(n)
    if variant == 0:
        order = rng.permutation(n)
        h[order[:n // 2]] = 1.0
    elif variant == 1:
        freq = int(rng.integers(2, 6))
        phase = rng.random()
        x = np.arange(n) / n
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif variant == 2:
        h[:n // 2] = 1.0
    elif variant == 3:
        h[::2] = 1.0
        h = _project(h, n / 2.0)
        h = np.clip(h, 0, 1)
        h = (h > 0.5).astype(float)
    elif variant == 4:
        h[rng.random(n) < 0.5] = 1.0
    elif variant == 5:
        block = n // 4
        for s in range(0, n, 2 * block):
            h[s:s + block] = 1.0
    elif variant == 6:
        freq = int(rng.integers(1, 4))
        x = np.arange(n) / n
        h = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x) + 0.2 * np.sin(4 * np.pi * x)
        h = np.clip(h, 0, 1)
    elif variant == 7:
        h = 0.5 + 0.35 * np.sin(np.pi * np.arange(n) / n * 2)
        h = np.clip(h, 0, 1)
    else:
        order = rng.permutation(n)
        h[order[:n // 2]] = 1.0
    return _project(h, n / 2.0)


def _multi_coord_descent(h, deadline, n_rounds=3):
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        orders = ['forward', 'random', 'reverse']
        for o in orders:
            if time.time() > deadline:
                break
            sub_d = min(deadline, time.time() + (deadline - time.time()) / 6)
            h_r, c5_r = _coord_descent(best_h, sub_d, max_sweeps=400,
                                       order=o, seed=42 + r * 10 + hash(o) % 100)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        raw = np.array(initial_h_values, dtype=float)
        n0 = len(raw)
    except NameError:
        n0 = 120
        x = np.arange(n0) / n0
        raw = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x)

    h0 = _project(raw.copy(), n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Deep coordinate descent from parent (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _multi_coord_descent(best_h, d1, n_rounds=2)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Binary SA + coordinate polish (30% time)
    d2 = t0 + budget_s * 0.50
    n_sa = 40
    sa_budget_each = (d2 - time.time()) / max(1, n_sa)
    for trial in range(n_sa):
        if time.time() > d2:
            break
        sub_d = min(d2, time.time() + sa_budget_each * 0.5)
        h_sa, c5_sa = _binary_sa(best_h, sub_d, max_iters=20000, seed=seed + 200 + trial)
        if time.time() < d2 - 1:
            polish_d = min(d2, time.time() + sa_budget_each * 0.4)
            h_pol, c5_pol = _coord_descent(h_sa, polish_d, max_sweeps=300,
                                           order='random', seed=seed + 800 + trial)
            if c5_pol < c5_sa:
                h_sa, c5_sa = h_pol, c5_pol
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 3: Multi-resolution (20% time)
    d3 = t0 + budget_s * 0.70
    base_n = len(best_h)
    tried = set()
    for dn in [-6, 6, -8, 8, -12, 12, -4, 4, -16, 16, -20, 20, -24, 24, -10, 10]:
        if time.time() > d3:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_d = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _coord_descent(h_r, sub_d, max_sweeps=300,
                                   order='random', seed=seed + 2000 + abs(dn))
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r
            base_n = n_try

    # Phase 4: Diverse starts + coordinate descent (20% time)
    d4 = t0 + budget_s * 0.90
    n_div = 20
    div_budget_each = max(1.0, (d4 - time.time()) / max(1, n_div))
    for v in range(n_div):
        if time.time() > d4:
            break
        h_div = _make_diverse_start(base_n, rng, v % 8)
        sub_d = min(d4, time.time() + div_budget_each * 0.7)
        h_d, c5_d = _coord_descent(h_div, sub_d, max_sweeps=300,
                                   order='random', seed=seed + 4000 + v)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d

    # Also try rotations of current best
    for shift in [1, -1, 2, -2, 3, -3]:
        if time.time() > d4:
            break
        h_rot = np.roll(best_h, shift)
        h_rot = _project(h_rot, len(h_rot) / 2.0)
        sub_d = min(d4, time.time() + budget_s * 0.008)
        h_d, c5_d = _coord_descent(h_rot, sub_d, max_sweeps=200,
                                   order='random', seed=seed + 5000 + shift)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d

    # Phase 5: Final deep polish (remaining time)
    d5 = deadline - 2.0
    if time.time() < d5:
        h_f, c5_f = _multi_coord_descent(best_h, d5, n_rounds=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # One more SA round from best if time allows
    if time.time() < deadline - 5:
        sa_d = min(deadline - 2, time.time() + 10)
        h_sa2, c5_sa2 = _binary_sa(best_h, sa_d, max_iters=30000, seed=seed + 9999)
        if time.time() < deadline - 2:
            h_sa2, c5_sa2 = _coord_descent(h_sa2, deadline - 2, max_sweeps=200,
                                           order='random', seed=seed + 9998)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
