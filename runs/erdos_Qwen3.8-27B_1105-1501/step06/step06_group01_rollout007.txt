        else:

        else:

        G[n-1-r, :m] = dx * (1.0 - h[r:r+m])

def _lp_direction(h, G, S, radius):
    n = len(h)
    num_lags = 2 * n - 1
    # Variables: p (n entries) + t (1 entry)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality: G @ p - t * 1 <= -S
    # This is [G | -1] @ [p; t] <= -S
    A_ub = np.zeros((num_lags, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -S
    
    # Equality: sum(p) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -radius <= p_i <= radius, t is free (but bounded by problem)
    bounds = [( -radius, radius)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
