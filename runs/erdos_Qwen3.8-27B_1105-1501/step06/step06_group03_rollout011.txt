Let me be careful about the Nesterov acceleration. The standard formulation:

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_all(h, indices):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[:m] += (1.0 - h[k:k+m])
                g[k:] += -h[:m]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    G *= dx
    return G

def _ray_line_search(h, p, dx, trust_r):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(p, 1.0 - h, mode='full') - np.correlate(h, p, mode='full')
    C = -np.correlate(p, p, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = p > 1e-14
    neg = p < -1e-14
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / p[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / p[neg]))
    t_lo = max(t_lo, -trust_r)
    t_hi = min(t_hi, trust_r)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    n_grid = 36
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] + (grid[:, None]**2) * C[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i-1], grid[best_i+1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1*T + c1*c1*C)) * dx
    f2 = float(np.max(S0 + c2*T + c2*c2*C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(S0 + c1*T + c1*c1*C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(S0 + c2*T + c2*c2*C)) * dx
    opt_t = (a+b)/2.0
    opt_val = float(np.max(S0 + opt_t*T + opt_t*opt_t*C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _soft_direction(h, S, tau, top_k):
    n = len(h)
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:top_k]
    w = np.exp((S[active] - M) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    G = _grad_all(h, active)
    g = w @ G
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm

def _soft_descend(h0, deadline, top_k=20, tau_init=None, alpha_init=0.06):
    n = len(h0)
    target = n / 2.0
    h = _project(h0.copy(), target)
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    alpha = alpha_init
    tau = tau_init if tau_init is not None else 0.004 * max(cur_c5, 0.01)
    tau_min = 0.0003
    stall = 0
    vel = np.zeros(n)
    beta_mom = 0.4

    while time.time() < deadline:
        if stall > 40:
            break
        S = _corr_scaled(h)
        d = _soft_direction(h, S, tau, top_k)
        if d is None:
            stall += 5
            continue
        step_dir = beta_mom * vel + (1.0 - beta_mom) * d
        step_dir -= step_dir.mean()
        sn = np.linalg.norm(step_dir)
        if sn < 1e-14:
            step_dir = d
        else:
            step_dir = step_dir / sn
        t, new_c5 = _ray_line_search(h, step_dir, dx, alpha)
        if new_c5 < cur_c5 - 1e-14:
            vel = step_dir
            h_new = h + t * step_dir
            h_new = _project(h_new, target)
            cur_c5 = _exact_c5(h_new)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_new.copy()
            h = h_new
            stall = 0
            if abs(t) > 0.6 * alpha:
                alpha = min(alpha * 1.08, 0.25)
            elif abs(t) < 0.2 * alpha:
                alpha = max(alpha * 0.75, 0.004)
            tau = max(tau_min, tau * 0.88)
        else:
            vel = 0.0
            stall += 1
            alpha *= 0.82
            if stall % 8 == 0:
                tau = max(tau_min, tau * 0.75)
    return best_h, best_c5

def _surgical_reduce(h0, deadline, max_rounds=200):
    """Focus on reducing the single highest correlation via targeted coordinate moves."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h0.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0

    for _round in range(max_rounds):
        if time.time() > deadline or stall > 30:
            break
        S = _corr_scaled(h)
        k_top = int(np.argmax(S))
        k_val = S[k_top]
        g = _grad_all(h, [k_top])[0]
        g -= g.mean()
        # Find best single coordinate to reduce this correlation
        # Moving h[i] up reduces C_k if g[i] > 0 (gradient of C_k w.r.t. h[i])
        # We want to move in direction -g (to reduce C_k)
        # Best candidate: coordinate with most negative -g[i] = most positive g[i]
        # subject to h[i] < 1 (can increase)
        # Or coordinate with most positive -g[i] = most negative g[i]
        # subject to h[i] > 0 (can decrease)
        candidates_up = np.where(g < -1e-8)[0]
        candidates_up = candidates_up[h[candidates_up] < 0.999]
        candidates_down = np.where(g > 1e-8)[0]
        candidates_down = candidates_down[h[candidates_down] > 0.001]
        
        best_move_c5 = best_c5
        best_move_t = 0
        best_move_dir = None
        
        tried = []
        for cands, sign in [(candidates_up, -1), (candidates_down, 1)]:
            if len(cands) == 0:
                continue
            # Sort by |g| descending
            order = np.argsort(-np.abs(g[cands]))[:8]
            for ci in order:
                i = cands[ci]
                d = np.zeros(n)
                d[i] = sign
                # Find partner to maintain sum=0
                if sign == -1:
                    # h[i] decreases, need h[j] to increase
                    partners = np.where(g > 1e-8)[0]
                    partners = partners[h[partners] < 0.999]
                else:
                    partners = np.where(g < -1e-8)[0]
                    partners = partners[h[partners] > 0.001]
                if len(partners) == 0:
                    continue
                # Try best partner
                p_order = np.argsort(-np.abs(g[partners]))[:4]
                for pi in p_order:
                    j = partners[pi]
                    if j == i:
                        continue
                    d2 = np.zeros(n)
                    d2[i] = sign
                    d2[j] = -sign
                    t, c5_try = _ray_line_search(h, d2, dx, 0.3)
                    if c5_try < best_move_c5 - 1e-14:
                        best_move_c5 = c5_try
                        best_move_t = t
                        best_move_dir = d2.copy()
        
        if best_move_dir is not None:
            h = h + best_move_t * best_move_dir
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
        else:
            # Try soft direction as fallback
            S2 = _corr_scaled(h)
            d = _soft_direction(h, S2, 0.003, 10)
            if d is not None:
                t, c5_try = _ray_line_search(h, d, dx, 0.08)
                if c5_try < best_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def _pair_polish(h0, deadline, max_passes=300):
    """Systematic pair coordinate descent with exact line search."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    stall = 0

    for _pass in range(max_passes):
        if time.time() > deadline or stall > 50:
            break
        improved = False
        S = _corr_scaled(h)
        G_all = _grad_all(h, np.arange(2*n-1))
        total_grad = S @ G_all
        total_grad -= total_grad.mean()
        # Select top coordinates by |gradient|
        n_cand = min(n, 40)
        cands = np.argsort(-np.abs(total_grad))[:n_cand]
        # Try pairs with opposite sign gradients
        for ii in range(len(cands)):
            if time.time() > deadline:
                break
            i = cands[ii]
            for jj in range(ii+1, len(cands)):
                j = cands[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                t, c5_try = _ray_line_search(h, d, dx, 0.35)
                if c5_try < best_c5 - 1e-14:
                    h = h + t * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            stall += 1
            # Try random perturbation
            if stall % 10 == 0:
                h_pert = h + rng.normal(0, 0.02, n)
                h_pert = _project(h_pert, target)
                c5_pert = _exact_c5(h_pert)
                if c5_pert < best_c5:
                    h = h_pert
                    best_h = h.copy()
                    best_c5 = c5_pert
                    stall = 0
                else:
                    h = _project(h + rng.normal(0, 0.05, n), target)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks-1, replace=False).tolist())
    boundaries = [0] + boundaries + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b+1]] = 1.0
    return _project(h, n / 2.0)

def _de_swarm(h_best, deadline, pop_size=12):
    """Differential evolution swarm on feasible set."""
    n = len(h_best)
    target = n / 2.0
    rng = np.random.default_rng(777)
    dx = 2.0 / n
    best_c5 = _exact_c5(h_best)
    best_h = h_best.copy()
    pop = [best_h.copy()]
    # Initialize population with diverse starting points
    pop.append(h_best + rng.normal(0, 0.05, n))
    pop.append(_sine_init(n, rng.integers(1, 8), rng.random() * 6.28))
    pop.append(_block_init(n, rng, rng.integers(4, 12)))
    while len(pop) < pop_size:
        base = pop[rng.integers(len(pop))]
        pop.append(_project(base + rng.normal(0, 0.08, n), target))
    pop = [_project(p, target) for p in pop]
    pop_c5 = [_exact_c5(p) for p in pop]
    
    F = 0.5
    CR = 0.7
    gen = 0
    while time.time() < deadline and gen < 500:
        gen += 1
        i = rng.integers(len(pop))
        a, b, c = rng.choice([x for x in range(len(pop)) if x != i], size=3, replace=False)
        trial = pop[a] + F * (pop[b] - pop[c])
        # Binary crossover
        mask = rng.random(n) < CR
        trial[mask] = h_best[mask] if rng.random() < 0.3 else trial[mask]
        trial = _project(trial, target)
        trial_c5 = _exact_c5(trial)
        if trial_c5 < pop_c5[i] + 1e-12:
            pop[i] = trial
            pop_c5[i] = trial_c5
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = trial.copy()
        # Occasional mutation with large step
        if gen % 20 == 0:
            k = rng.integers(len(pop))
            pop[k] = _project(pop[k] + rng.normal(0, 0.15, n), target)
            pop_c5[k] = _exact_c5(pop[k])
    return best_h, best_c5
