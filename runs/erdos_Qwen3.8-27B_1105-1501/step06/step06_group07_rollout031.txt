def _exact_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection onto {0 <= x <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_line_search(h, direction, target, dx, max_t=2.0):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_binary_seed(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    if n % 2 != 0:

    m = n // 2
    idx = rng.permutation(n)[:m]

def _threshold_vertex(h, scores, n):
    """Create binary vertex from scores: top n/2 get 1."""

def _pair_swaps(h, deadline, rng, max_swaps=500):

        s = h[i] + h[j]

        if hi - lo < 1e-14:

def _basin_hopping(h, deadline, rng, n_pert=50, polish_time=0.3):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_for_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    else:
        a = -k
        if a == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - a] += (1.0 - h[a:])
            g[a:] -= h[:n - a]
    return g


def _compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _exact_line_search(h, direction, max_t=1.5):
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    T = (np.correlate(direction, 1.0 - h, mode='full')
         - np.correlate(h, direction, mode='full'))
    Q = np.correlate(direction, direction, mode='full')

    pos = direction > 1e-14
    neg = direction < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / direction[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-direction[neg]))))
    t_hi = max(t_hi, 0.0)
    if t_hi < 1e-14:
        return 0.0, float(np.max(S * dx))

    n_grid = 50
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))

    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S + c1 * T - c1 * c1 * Q) * dx)
    f2 = float(np.max(S + c2 * T - c2 * c2 * Q) * dx)
    for _ in range(45):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S + c1 * T - c1 * c1 * Q) * dx)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S + c2 * T - c2 * c2 * Q) * dx)

    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S + opt_t * T - opt_t * opt_t * Q) * dx)
    val_0 = float(np.max(S) * dx)
    if val_0 < opt_val:
        opt_t, opt_val = 0.0, val_0
    return opt_t, opt_val


def _lp_minimax(h_old, K=15, eps=0.01):
    n = len(h_old)
    dx = 2.0 / n
    corr = np.correlate(h_old, 1.0 - h_old, mode='full') * dx
    top_idx = np.argsort(-corr)[:K]

    A_rows = []
    for idx in top_idx:
        k = int(idx) - (n - 1)
        c = np.zeros(n)
        if k >= 0:
            c[k:] = 1.0 - h_old[:n - k]
        else:
            a = -k
            c[:n - a] = 1.0 - h_old[a:]
        A_rows.append(c)
    A = np.array(A_rows)

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        A @ h_var <= t_var
    ]
    obj = cp.Minimize(t_var + eps * cp.sum_squares(h_var - h_old))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None, None
    if h_var.value is None:
        return None, None
    return np.array(h_var.value), float(t_var.value)


def _alternating_lp(h_init, deadline, max_iters=40, eps_init=0.05):
    h = h_init.copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    eps = eps_init

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        h_new, _ = _lp_minimax(h, K=15, eps=eps)
        if h_new is None:
            break
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        eps *= 0.88

    return best_h, best_c5


def _momentum_descent(h, deadline, rng, K=5, max_rounds=3000, mu=0.35):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    momentum = np.zeros(n)
    step_scale = 1.0
    no_improve = 0

    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        corr = _compute_corrs(h)
        cur_c5 = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        gap = max(top_vals[0] - top_vals[-1], 1e-15)
        alpha = min(25.0 / gap, 250.0)
        w = np.exp(alpha * (top_vals - top_vals[0]))
        w /= np.sum(w)

        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _grad_for_lag(h, k)
        grad -= np.mean(grad)
        momentum = mu * momentum + (1.0 - mu) * grad
        gn = np.linalg.norm(momentum)
        if gn < 1e-14:
            break
        direction = -momentum / gn * step_scale

        t, f_try = _exact_line_search(h, direction, max_t=1.2)
        if t < 1e-12:
            step_scale *= 0.7
            no_improve += 1
            if no_improve > 25:
                h = _project(h + rng.normal(0, 0.06, n), target)
                c5_p = _exact_c5(h)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h.copy()
                no_improve = 0
                step_scale = 0.8
                momentum *= 0.3
            continue

        h_new = h + t * direction
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < cur_c5 - 1e-13:
            h = h_new
            if f_new < best_c5:
                best_c5 = f_new
                best_h = h.copy()
            no_improve = 0
            step_scale = min(step_scale * 1.05, 3.0) if t > 0.4 else max(step_scale * 0.92, 0.08)
        else:
            no_improve += 1
            step_scale *= 0.8
            if step_scale < 0.05:
                h = _project(h + rng.normal(0, 0.07, n), target)
                c5_p = _exact_c5(h)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h.copy()
                step_scale = 0.5
                no_improve = 0

    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _basin_hopping(h, deadline, rng, n_pert=60, polish_s=0.2):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    eps = 0.12

    for _ in range(n_pert):
        if time.time() > deadline:
            break
        h_pert = _project(h + rng.normal(0, eps, n), target)
        sub_dl = min(deadline, time.time() + polish_s)
        h_opt, c5_opt = _momentum_descent(h_pert, sub_dl, rng, K=4, max_rounds=150, mu=0.3)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
            eps = max(0.03, eps * 0.85)
        else:
            eps = min(0.35, eps * 1.12)

    return best_h, best_c5


def _pair_optimize(h, deadline, rng, max_rounds=4000):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-14:
            continue
        n_grid = 15
        grid = np.linspace(lo, hi, n_grid)
        best_ti = h[i]
        best_val = _exact_c5(h)
        save_i, save_j = h[i], h[j]
        for ti in grid:
            h[i] = ti
            h[j] = s - ti
            val = _exact_c5(h)
            if val < best_val:
                best_val = val
                best_ti = ti
        h[i] = best_ti
        h[j] = s - best_ti
        c5_now = _exact_c5(h)
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
        else:
            h[i] = save_i
            h[j] = save_j

    return best_h, best_c5


def _symmetry_variant(h, variant):
    n = len(h)
    if variant == 0:
        return _project(1.0 - h, n / 2.0)
    elif variant == 1:
        return _project(h[::-1].copy(), n / 2.0)
    elif variant == 2:
        return _project(1.0 - h[::-1].copy(), n / 2.0)
    elif variant == 3:
        return _project(0.5 * h + 0.5 * h[::-1], n / 2.0)
    else:
        shift = variant % max(n // 4, 1)
        return _project(np.roll(h, shift), n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.96

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = (np.arange(n0) + 0.5) / n0
        h0 = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * x) + 0.12 * np.cos(2 * np.pi * 7 * x + 0.7)
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: LP alternating (20% time)
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = _alternating_lp(h, d1, max_iters=35, eps_init=0.04)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Momentum descent with various K (25% time)
    d2 = t0 + budget_s * 0.45
    for K in [4, 6, 8]:
        if time.time() > d2:
            break
        sub_dl = min(d2, time.time() + budget_s * 0.05)
        h_k, c5_k = _momentum_descent(best_h, sub_dl, rng, K=K, max_rounds=1500, mu=0.35)
        if c5_k < best_c5:
            best_c5 = c5_k
            best_h = h_k.copy()

    # Phase 3: LP again from improved point (10% time)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _alternating_lp(best_h, d3, max_iters=20, eps_init=0.02)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Follow with momentum
    sub_dl = min(d3, time.time() + budget_s * 0.03)
    h3b, c5_3b = _momentum_descent(best_h, sub_dl, rng, K=4, max_rounds=500, mu=0.3)
    if c5_3b < best_c5:
        best_c5 = c5_3b
        best_h = h3b.copy()

    # Phase 4: Basin hopping (25% time)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = _basin_hopping(best_h, d4, rng, n_pert=70, polish_s=0.18)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Pair optimization (5% time)
    d5 = t0 + budget_s * 0.85
    h5, c5_5 = _pair_optimize(best_h, d5, rng, max_rounds=3000)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (8% time)
    d6 = deadline
    best_n = len(best_h)
    tried_n = set([best_n])
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, 10, -10, 12, -12, 14, -14]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_r, c5_r = _momentum_descent(h_r, sub_dl, rng, K=4, max_rounds=200, mu=0.3)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Symmetry-based restarts (2% time)
    if time.time() < deadline - 10:
        for var in range(4):
            if time.time() > deadline - 5:
                break
            h_s = _symmetry_variant(best_h, var)
            c5_s = _exact_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            sub_dl = min(deadline - 3, time.time() + budget_s * 0.008)
            h_sp, c5_sp = _momentum_descent(h_s, sub_dl, rng, K=4, max_rounds=200, mu=0.3)
            if c5_sp < best_c5:
                best_c5 = c5_sp
                best_h = h_sp.copy()

    # Final polish
    if time.time() < deadline - 2:
        h_f, c5_f = _momentum_descent(best_h, deadline, rng, K=4, max_rounds=400, mu=0.25)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = _project(best_h, len(best_h) / 2.0)
        final_c5 = _exact_c5(best_h)

    return best_h, final_c5, len(best_h)
