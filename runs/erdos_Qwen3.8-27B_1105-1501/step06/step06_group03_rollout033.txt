    where S_r = dx * c[r] are the scaled correlation values.
    Returns the gradient vector of length n.

    S_max = float(np.max(S))
    exps = np.exp(beta * (S - S_max))
    w = exps / exps.sum()

    # ∂S_r/∂h_i = dx * [(1-h[i+d]) * 1_{i<=n-1-d} - h[i-d] * 1_{i>=d}]
    # where d = |r - (n-1)|

        d = abs(r - (n - 1))
        if d == 0:
            # ∂S_r/∂h_i = dx * [(1-h[i]) - h[i]] = dx * (1 - 2*h[i]) for all i
            grad += w[r] * dx * (1.0 - 2.0 * h)
        else:
            # ∂S_r/∂h_i = dx * [(1-h[i+d]) * 1_{i<=n-1-d} - h[i-d] * 1_{i>=d}]
            # First term: for i in [0, n-1-d], add (1-h[i+d])
            grad[:n-d] += w[r] * dx * (1.0 - h[d:])
            # Second term: for i in [d, n-1], subtract h[i-d]
            # i.e., for j in [0, n-1-d], grad[j+d] -= h[j]
            grad[d:] -= w[r] * dx * h[:n-d]
    
    return grad

    active = sorted_idx[:K]
    tol = max(0.03 * M, 1e-5)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = np.zeros((len(combined), n))
    for idx, r in enumerate(combined):
        k = r - (n - 1)
        d = abs(k)
        if d == 0:
            G[idx] = (1.0 - 2.0 * h)
        else:
            G[idx, :n - d] += (1.0 - h[d:])
            G[idx, d:] -= h[:n - d]
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((len(combined), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h_init, deadline, radius_init=0.12, K=18):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = _project_simplex(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(12345)
    while time.time() < deadline:
        if stall > 50:
            if restart_ctr < 15:
                scale = 0.005 + 0.008 * restart_ctr * 0.4
                h = _project_simplex(h + rng.normal(0, scale, n), target)
                best_local = _exact_c5(h)
                if best_local > best_c5 + 0.005:
                    h = best_h.copy()
                    best_local = best_c5
                radius = 0.06 + 0.02 * restart_ctr
                stall = 0
                restart_ctr += 1
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.7
            if radius < 1e-5:
                break
            continue
        alpha, new_c5 = _line_search_exact(h, p, dx, radius)
        if new_c5 < best_c5 - 1e-14:
            h = _project_simplex(h + alpha * p, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
                ratio = abs(alpha) / max(radius, 1e-12)
                if ratio > 0.6:
                    radius = min(radius * 1.08, 0.30)
                else:
                    radius = max(radius * 0.92, 0.02)
            else:
                stall += 1
                radius *= 0.85
        else:
            stall += 1
            radius *= 0.78
            if stall % 10 == 0 and radius < 0.02:
                radius = 0.05
    return best_h, best_c5

def _pair_refine(h_init, deadline, max_passes=300):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = _project_simplex(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    stall = 0
    for _pass in range(max_passes):
        if time.time() > deadline or stall > 60:
            break
        improved = False
        S = _corr_scaled(h)
        G = np.zeros((2 * n - 1, n))
        for r in range(2 * n - 1):
            d = abs(r - (n - 1))
            if d == 0:
                G[r] = (1.0 - 2.0 * h)
            else:
                G[r, :n - d] = (1.0 - h[d:])
                G[r, d:] = -h[:n - d]
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_cand = min(n, 50)
        candidates = np.argsort(-np.abs(total_grad))[:n_cand]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, min(ii + 8, len(candidates))):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search_exact(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-13:
                    h = _project_simplex(h + alpha * d, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                                     
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_simplex(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project_simplex(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.98
    rng = np.random.default_rng(seed)

    try:

        n0 = len(h0)

    best_c5 = _exact_c5(h0)

    # Phase 1: FISTA smooth max (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _fista_smooth(best_h, best_h.size / 2.0, d1, max_beta=400, init_beta=12)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Active descent (15% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _active_descent(best_h, best_h.size / 2.0, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP loop (20% time)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.14, K=20)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-resolution (20% time)
    d4 = t0 + budget_s * 0.80
    base_n = len(best_h)
    tried = set()
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14, -16, 16, -20, 20, -24, 24]
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d4, time.time() + budget_s * 0.025)
            h_r, c5_r = _fista_smooth(h_r, n_try / 2.0, sub_dl, max_beta=300, init_beta=15)
            if c5_r < best_c5:
                sub_dl2 = min(d4, time.time() + budget_s * 0.015)
                h_r, c5_r = _lp_loop(h_r, sub_dl2, radius_init=0.08, K=15)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 5: Pair refinement (12% time)
    d5 = t0 + budget_s * 0.92
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=200)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Pertur
b + re-optimize (5% time)
    d6 = t0 + budget_s * 0.97
    scale = 0.02
    for attempt in range(200):
        if time.time() > d6:
            break
        h_pert = _project_simplex(best_h + rng.normal(0, scale, len(best_h)), len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.006)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=15)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.15)

    # Final polish
    if time.time() < deadline:
        h_f, c5_f = _active_descent(best_h, len(best_h) / 2.0, deadline)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project_simplex(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
