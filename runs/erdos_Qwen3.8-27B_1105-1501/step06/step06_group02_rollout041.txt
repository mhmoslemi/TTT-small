import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads(h, indices):
    n = len(h)
    C = 2 * n - 1
    G = np.zeros((len(indices), n))
    for idx, m in enumerate(indices):
        lag = m - (n - 1)
        if lag >= 0:
            k = n - lag
            G[idx, lag:n] += 1.0 - h[0:k]
            G[idx, 0:k] += -h[lag:n]
        else:
            r = -lag
            k = n - r
            G[idx, 0:k] += 1.0 - h[r:n]
            G[idx, r:n] += -h[0:k]
    return G

def _qp_obj(x, args):
    n, z_scale, norm_scale, lam_z, lam_norm = args
    d = x[:n]
    z = x[n:]
    return lam_z * float(np.sum(z)) / z_scale + lam_norm * float(d @ d) / norm_scale

def _qp_grad(x, args):
    n, z_scale, norm_scale, lam_z, lam_norm = args
    K = len(x) - n
    d = x[:n]
    g_d = 2.0 * lam_norm * d / norm_scale
    g_z = np.full(K, lam_z / z_scale)
    return np.concatenate([g_d, g_z])

def _qp_ineq(x, args):
    G_sub, R_sub, target_raw, n = args
    d = x[:n]
    z = x[n:]
    return target_raw + z - G_sub @ d - R_sub

def _qp_ineq_jac(x, args):
    G_sub, n = args
    K = G_sub.shape[0]
    total = n + K
    J = np.zeros((K, total))
    J[:, :n] = -G_sub
    J[:, n:] = np.eye(K)
    return J

def _qp_eq(x, args):
    n = args[0]
    return float(np.sum(x[:n]))

def _qp_eq_jac(x, args):
    n = args[0]
    total = len(x)
    J = np.zeros(total)
    J[:n] = 1.0
    return J

def qp_step(h, R, target_raw, rho, top_k=25, lam_z=0.8, lam_norm=0.2, maxiter=120):
    n = len(h)
    max_R = float(np.max(R))
    if target_raw >= max_R - 1e-10:
        return None
    active = np.argsort(-R)[:top_k]
    G_sub = compute_grads(h, active.tolist())
    R_sub = R[active]
    K = len(active)
    d_lo = np.maximum(-h, -rho)
    d_hi = np.minimum(1.0 - h, rho)
    z_cap = float(np.max(np.maximum(R_sub - target_raw, 0.0))) + 1.0
    z_lo = np.zeros(K)
    z_hi = np.full(K, z_cap)
    bounds_lo = np.concatenate([d_lo, z_lo])
    bounds_hi = np.concatenate([d_hi, z_hi])
    d0 = np.zeros(n)
    z0 = np.maximum(R_sub - target_raw, 0.0) * 0.5
    z0 = np.minimum(z0, z_hi)
    x0 = np.concatenate([d0, z0])
    x0 = np.clip(x0, bounds_lo + 1e-10, bounds_hi - 1e-10)
    z_scale = max(float(np.sum(z0)) / K, 1e-6)
    norm_scale = n * rho * rho
    obj_args = (n, z_scale, norm_scale, lam_z, lam_norm)
    ineq_args = (G_sub, R_sub, target_raw, n)
    eq_args = (n,)
    constraints = [
        {'type': 'ineq', 'fun': _qp_ineq, 'jac': _qp_ineq_jac, 'args': ineq_args},
        {'type': 'eq', 'fun': _qp_eq, 'jac': _qp_eq_jac, 'args': eq_args}
    ]
    bounds = list(zip(bounds_lo, bounds_hi))
    try:
        res = minimize(_qp_obj, x0, args=obj_args, jac=_qp_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        d = res.x[:n].copy()
        d -= np.mean(d)
        if not np.all(np.isfinite(d)):
            return None
        if np.max(np.abs(d)) < 1e-13:
            return None
        d = np.clip(d, d_lo - 1e-8, d_hi + 1e-8)
        d -= np.mean(d)
        return d
    except Exception:
        return None

def line_search(h, d, R, dx):
    n = len(h)
    if np.max(np.abs(d)) < 1e-14:
        return 0.0, float(np.max(R)) * dx
    pos = d > 1e-10
    neg = d < -1e-10
    hi = 5.0
    lo = -5.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, float(np.max(R)) * dx
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    def f(a):
        return float(np.max(R + a * D - a * a * Q)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(45):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    best_a = (a + b) / 2.0
    best_v = f(best_a)
    for ep in [lo, hi, 0.0]:
        v = f(ep)
        if v < best_v:
            best_a, best_v = ep, v
    return best_a, best_v

def active_gradient_dir(h, R, top_k=15):
    n = len(h)
    max_R = float(np.max(R))
    threshold = max_R - 0.005 * max_R
    active = np.where(R >= threshold)[0]
    if len(active) < 3:
        active = np.argsort(-R)[:top_k]
    weights = np.exp((R[active] - max_R) / (0.01 * max_R + 1e-15))
    weights /= float(np.sum(weights))
    G_sub = compute_grads(h, active.tolist())
    d = G_sub.T @ weights
    d = -d
    d -= np.mean(d)
    return d

def pair_move(h, i, j, R, dx):
    d = np.zeros(len(h))
    d[i] = -1.0
    d[j] = 1.0
    return line_search(h, d, R, dx)

        n_best = len(best_h)
        h_f = best_h.copy()
        c5_f = c5_exact(h_f)
        R_f = np.correlate(h_f, 1.0 - h_f, 'full')
        dx_f = 2.0 / n_best
        rho_f = 0.06
        stall_f = 0
        sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.6)
        while time.time() < sub_deadline and stall_f < 50:
            max_R_f = float(np.max(R_f))
            d_g = active_gradient_dir(h_f, R_f, 10)
            a, v = line_search(h_f, d_g, R_f, dx_f)
            if a > 1e-10 and v < c5_f - 1e-14:
                h_f = project(h_f + a * d_g, n_best / 2.0)
                c5_f = c5_exact(h_f)
                R_f = np.correlate(h_f, 1.0 - h_f, 'full')
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
                stall_f = 0
            else:
                i = rng.integers(n_best)
                j = rng.integers(n_best)
                if i != j:
                    d_p = np.zeros(n_best)
                    d_p[i], d_p[j] = -1.0, 1.0
                    a, v = line_search(h_f, d_p, R_f, dx_f)
                    if a > 1e-10 and v < c5_f - 1e-14:
                        h_f = project(h_f + a * d_p, n_best / 2.0)
                        c5_f = c5_exact(h_f)
                        R_f = np.correlate(h_f, 1.0 - h_f, 'full')
                        if c5_f < best_c5:
                            best_c5 = c5_f
                            best_h = h_f.copy()
                        stall_f = 0
                    else:
                        stall_f += 1
                else:
                    stall_f += 1
            rho_f *= 0.98
    
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
