Now for the subgradient method:

The subgradient of f(h) = max_k corr_k(h) at h is the average of the gradients of the active constraints (those with corr_k close to the max).

For the correlation at lag d (index k = d + n - 1):
corr_d = sum_i h[i] * (1-h[i+d]) (with bounds)

The gradient wrt h[j] is:
- If d >= 0: (1-h[j-d]) for valid j-d, plus (-h[j+d]) for valid j+d
- If d < 0: (1-h[j+|d|]) for valid j+|d|, plus (-h[j-|d|]) for valid j-|d|

Let me implement this efficiently:

```python
def _subgradient(h, scores, top_k=20):
    """Compute average subgradient over top-k active constraints."""
    n = len(h)
    max_score = float(scores.max())
    threshold = max_score - 0.05 * max_score
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    if len(active) > top_k:
        # Take top_k
        idx = np.argsort(-scores[active])[:top_k]
        active = active[idx]
    
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += (1.0 - h[:m])
                grad[:m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += (1.0 - h[g:])
                grad[g:] += -h[:m]
    
    # Project to sum=0
    grad -= grad.mean()
    return grad
```

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

    t_hi = min(t_hi, max_radius)
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    best_t, best_val = 0.0, float(np.max(S0) * dx)
    n_grid = min(30, max(8, int(15 * (t_hi - t_lo))))
    ts = np.linspace(t_lo, t_hi, n_grid)
    vals = np.array([float(np.max(S0 + t * T + t * t * Q) * dx) for t in ts])
    bi = int(np.argmin(vals))
    best_t, best_val = ts[bi], vals[bi]
    if 0 < bi < n_grid - 1:
        a, b = ts[bi-1], ts[bi+1]
    else:
        a, b = t_lo, t_hi
    g = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - g * (b - a)
    d2 = a + g * (b - a)
    fc = float(np.max(S0 + c * T + c * c * Q) * dx)
    fd = float(np.max(S0 + d2 * T + d2 * d2 * Q) * dx)
    for _ in range(45):
        if b - a < 1e-13:
            break
        if fc < fd:
            b, d2, fd = d2, c, fc
            c = b - g * (b - a)
            fc = float(np.max(S0 + c * T + c * c * Q) * dx)
        else:
            a, c, fc = c, d2, fd
            d2 = a + g * (b - a)
            fd = float(np.max(S0 + d2 * T + d2 * d2 * Q) * dx)
    mid = (a + b) / 2.0
    vm = float(np.max(S0 + mid * T + mid * mid * Q) * dx)
    if vm < best_val:
        best_t, best_val = mid, vm
    return best_t, best_val

def _subgradient(h, scores, top_k=20):
    n = len(h)
    max_s = float(scores.max())
    thr = max_s - 0.04 * max_s
    active = np.where(scores >= thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    if len(active) > top_k:
        idx = np.argsort(-scores[active])[:top_k]
        active = active[idx]
    grad = np.zeros(n)
    for r in active:
        d = r - (n - 1)
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:] += (1.0 - h[:m])
                grad[:m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                grad[:m] += (1.0 - h[g:])
                grad[g:] += -h[:m]
    grad -= grad.mean()
    return grad

def _sg_optimize(h, deadline, rng, top_k0=25, top_k_min=5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    beta = 0.75
    vel = np.zeros(n)
    step = 0.015
    min_step, max_step = 0.0005, 0.08
    stall = 0
    max_stall = 60
    it = 0
    while time.time() < deadline - 1:
        it += 1
        if it % 200 == 0 and time.time() > deadline - 2:
            break
        scores = _corr_values(h)
        tk = max(top_k_min, int(top_k0 * max(0.3, 1.0 - it / 20000.0)))
        sg = _subgradient(h, scores, top_k=tk)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        vel = beta * vel - step * sg
        y = h + vel
        y = _project(y, target)
        new_c5 = _exact_c5(y)
        if new_c5 < cur_c5 - 1e-14:
            h = y
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            step = min(step * 1.03, max_step)
            if it % 30 == 0 and step > 0.005:
                d = -sg / sg_norm
                d -= d.mean()
                alpha, cv = _line_search(h, d, 0.08)
                if cv < best_c5 - 1e-13:
                    h = _project(h + alpha * d, target)
                    cur_c5 = _exact_c5(h)
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
        else:
            stall += 1
            if stall > max_stall:
                h = best_h + rng.normal(0, 0.012, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                vel = np.zeros(n)
                step = 0.01
                stall = 0
                max_stall = 40
            step = max(step * 0.97, min_step)
    return best_h, best_c5

def _pair_swap_refine(h, deadline, rng, max_passes=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for _ in range(max_passes):
        if time.time() > deadline - 1:
            break
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        improved = False
        for _ in range(300):
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            alpha, cv = _line_search(h, d, 0.4)
            if cv < best_c5 - 1e-13:
                h = _project(h + alpha * d, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    return best_h, best_c5

def _binary_sa(h_cont, deadline, rng):
    n = len(h_cont)
    if n % 2 != 0:
        return h_cont, _exact_c5(h_cont)
    target = n // 2
    sorted_idx = np.argsort(-h_cont)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 0.003
    T_min = 0.00005
    patience = 0
    max_pat = 8000
    while time.time() < deadline - 1:
        if patience > max_pat:
            n_flips = int(rng.integers(1, max(2, n // 15)))
            for _ in range(n_flips):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i], h[j] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_c5 = _exact_c5(h)
            if cur_c5 > best_c5 + 0.01:
                h = best_h.copy()
                cur_c5 = best_c5
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            patience = 0
            T = 0.003
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < -1e-14:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        elif T > T_min and rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            patience += 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = 1.0, 0.0
            patience += 1
        T = max(T_min, T * 0.9997)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _quick_lp(h, scores, radius, top_k=12):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = np.zeros((K, n))
    for idx, r in enumerate(active):
        d = r - (n - 1)
        if d >= 0:
            mm = n - d
            if mm > 0:
                G[idx, d:] += (1.0 - h[:mm])
                G[idx, :mm] += -h[d:]
        else:
            g = -d
            mm = n - g
            if mm > 0:
                G[idx, :mm] += (1.0 - h[g:])
                G[idx, g:] += -h[:mm]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None
