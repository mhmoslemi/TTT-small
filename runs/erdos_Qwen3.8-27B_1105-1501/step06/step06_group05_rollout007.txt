def _lp_direction(h, f_vals, n_active=None):
    """Solve LP to get descent direction."""
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(n - 1, -n, -1):
        idx = k + (n - 1)
        if k >= 0:
            nn = np.arange(0, n - k)
        else:
            nn = np.arange(-k, n)
        a = nn + k
        b = nn
        G[idx, b] += -h[a]
        G[idx, a] += (1.0 - h[b])
    
    if n_active is not None and n_active < M:
        order = np.argsort(-f_vals)[:n_active]
    else:
        order = np.arange(M)
    Ma = len(order)
    
    # LP: minimize tau
    # s.t. G_k . h - tau <= f_k - G_k . h0  for k in order
    #      sum(h) = n/2
    #      0 <= h <= 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A = np.zeros((Ma, n + 1))
    b_ub = np.zeros(Ma)
    for i, k in enumerate(order):
        A[i, :n] = G[k]
        A[i, n] = -1.0
        b_ub[i] = f_vals[k] - G[k] @ h
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n] - h
            nrm = float(np.linalg.norm(d))
            if nrm > 1e-14:
                return d / nrm
    except Exception:
        pass
    return None

def _grad_subdir(h, f_vals, k_active, rng):
    """Subgradient direction from top-k constraints."""
    n = len(h)
    M = 2 * n - 1
    G = np.zeros(n)
    for kk in range(k_active):
        idx = kk  # top-k by f_vals after sorting
        k = idx - (n - 1)
        if k >= 0:
            nn = np.arange(0, n - k)
        else:
            nn = np.arange(-k, n)
        a = nn + k
        b = nn
        G[b] += -h[a]
        G[a] += (1.0 - h[b])
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm

def _feasible_range(h, d):
    """Find [L, U] such that h + t*d is in [0,1]^n for t in [L, U]."""
    L = -np.inf
    U = np.inf
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        L = max(L, float(np.min(-h[neg] / d[neg])))
    return L, U

def _optimize_lp(h, deadline, rng, use_lp=True, n_active=None, pair_frac=0.2, rand_frac=0.1):
    """Main optimization loop using LP-guided directions with pair moves."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    stall = 0
    
    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            # Random direction
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            # Pair move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        else:
            # LP or subgradient direction
            corr = np.correlate(h, 1.0 - h, mode='full')
            if use_lp:
                d = _lp_direction(h, corr, n_active)
                if d is None:
                    # Fallback to subgradient
                    order = np.argsort(-corr)
                    h_sorted_corr = corr[order]
                    d = _grad_subdir(h, h_sorted_corr, min(10, 2*n-1), rng)
            else:
                order = np.argsort(-corr)
                h_sorted_corr = corr[order]
                d = _grad_subdir(h, h_sorted_corr, 10, rng)
        
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        
        L, U = _feasible_range(h, d)
        if U - L < 1e-12:
            stall += 1
            continue
        # Ensure 0 is in range
        L = min(L, 0.0)
        U = max(U, 0.0)
        
        S, Q = _compute_ST(h, d)
        corr = np.correlate(h, 1.0 - h, mode='full')
        t, newf = _line_search(corr, S, Q, dx, L, U)
        
        if abs(t) < 1e-14:
            stall += 1
            continue
        
        hn = h + t * d
        hn = _project(hn, target)
        nf = _exact_c5(hn)
        
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + 5e-5:
            h = hn
            cur_f = nf
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            stall += 1
        
        if stall > 200:
            h = _project(best_h + rng.normal(0, 0.03, n), target)
            cur_f = _exact_c5(h)
            stall = 0
    
    return best_h, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.05, n)
    return _project(base, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 4 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def _two_phase_init(n, rng):
    """Initialize with a two-phase structure."""
    h = np.zeros(n)
    split = n // 2
    h[:split] = 1.0
    # Add noise and some mixing
    h = h + rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)

        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _compute_f(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_ST(h, d):
    n = len(h)
    M = 2 * n - 1
    S = np.zeros(M)
    Q = np.zeros(M)
    for k in range(n - 1, -n, -1):
        idx = k + (n - 1)
        if k >= 0:
            nn = np.arange(0, n - k)
        else:
            nn = np.arange(-k, n)
        a = nn + k
        b = nn
        S[idx] = np.sum(d[a] * (1.0 - h[b])) - np.sum(h[a] * d[b])
        Q[idx] = np.sum(d[a] * d[b])
    return S, Q
