def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    """Project v onto [0,1]^n with sum = target."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection for threshold
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        interior = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            x[interior] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads_full(h):
    """Compute the full (2n-1, n) gradient matrix for rows of correlate(h, 1-h, 'full')."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        ell = k - (n-1)
        if ell >= 0:
            m = n - ell
            # row = sum_{t=0}^{m-1} h[t+ell] * (1-h[t])
            # d/dh[j]: for j >= ell: (1-h[j-ell]) from h[j] term
            #           for j < m:    -h[j+ell] from (1-h[j]) term
            if ell == 0:
                G[k, :] = 1.0 - 2.0 * h[:]
            else:
                G[k, ell:] += (1.0 - h[:m])       # for j=ell..n-1: (1-h[j-ell])
                G[k, :m] += -h[ell:ell+m]          # for j=0..m-1: -h[j+ell]
        else:
            r = -ell
            m = n - r
            # row = sum_{t=0}^{m-1} h[t] * (1-h[t+r])
            # d/dh[j]: for j < m:    (1-h[j+r]) from h[j] term

            G[k, :m] += (1.0 - h[r:r+m])           # for j=0..m-1: (1-h[j+r])
            G[k, r:] += -h[:m]                      # for j=r..n-1: -h[j-r]
    return G

def exact_line_search(h, d, R0, dx, n_max_iter=50):
    """Exact 1D line search for f(alpha) = max_k (R0[k] + alpha*D[k] - alpha^2*Q[k]) * dx"""
    n = len(h)
    D = np.correlate(d, 1.0-h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    
    # Feasible interval
    pos = d > 1e-12
    neg = d < -1e-12
    alpha_max = 1e10
    alpha_min = -1e10
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_min = max(alpha_min, float(np.max(h[neg] / d[neg])))
    
    if alpha_max - alpha_min < 1e-14:
        return 0.0, c5_exact(h)
    
    def f(alpha):
        return float(np.max(R0 + alpha * D - alpha*alpha * Q)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = alpha_min, alpha_max
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(n_max_iter):
        if b - a < 1e-12 * max(1.0, abs(a)):
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    
    opt = (a + b) / 2.0
    val = f(opt)
    # Check endpoints
    for pt in [alpha_min, alpha_max, 0.0]:
        v = f(pt)
        if v < val:
            val, opt = v, pt
    return opt, val

def solve_qp_direction(h, R, M, rho, lambda_z=0.85, lambda_norm=0.15, max_iter=200):
    """Solve the QP tangent subproblem to get a displacement direction d."""
    n = len(h)
    C = 2*n - 1
    G = compute_grads_full(h)
    
    # Cap for z
    z_cap = float(np.max(R)) - M + 2.0
    
    try:
        d_var = cp.Variable(n)
        z_var = cp.Variable(C)
        
        obj = lambda_z * cp.sum(z_var) + lambda_norm * cp.sum_squares(d_var) / (rho*rho)
        
        constraints = [
            cp.sum(d_var) == 0,
            R + G @ d_var - M <= z_var,
            d_var >= -np.minimum(h, rho),
            d_var <= np.minimum(1.0 - h, rho),
            z_var >= 0,
            z_var <= z_cap
        ]
        
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        # Try solvers in order
        solved = False
        for solver in [cp.ECOS, cp.CVXOPT]:
            try:
                prob.solve(solver=solver, max_iters=max_iter)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved or d_var.value is None:
            return None
        
        d = np.array(d_var.value).flatten()
        if not np.all(np.isfinite(d)):
            return None
        d -= np.mean(d)  # Ensure zero sum
        return d
    
    except Exception:
        return None

def solve_qp_slsqp(h, R, M, rho, lambda_z=0.85, lambda_norm=0.15):
    """Fallback QP solver using scipy SLSQP."""
    n = len(h)
    C = 2*n - 1
    G = compute_grads_full(h)
    z_cap = float(np.max(R)) - M + 2.0
    
    x0 = np.zeros(n + C)  # d=0, z=0
    
    def obj(x):
        d = x[:n]
        z = x[n:]
        return lambda_z * np.mean(z) + lambda_norm * np.sum(d*d) / (rho*rho)
    
    def grad_obj(x):
        d = x[:n]
        g = np.zeros(n + C)
        g[:n] = 2.0 * lambda_norm * d / (rho*rho)
        g[n:] = lambda_z / C
        return g
    
    # Equality: sum(d) = 0
    def eq_con(x):
        return np.sum(x[:n])
    
    def eq_jac(x):
        J = np.zeros(n + C)
        J[:n] = 1.0
        return J
    
    # Inequality: R + G@d - M - z <= 0 (i.e., -(R + G@d - M - z) >= 0)
    # SLSQP expects constraint >= 0, so we use: M + z - R - G@d >= 0
    def ineq_cons(x):
        d = x[:n]
        z = x[n:]
        return M + z - R - G @ d
    
    # Bounds
    lb_d = -np.minimum(h, rho)
    ub_d = np.minimum(1.0 - h, rho)
    lb = np.concatenate([lb_d, np.zeros(C)])
    ub = np.concatenate([ub_d, np.full(C, z_cap)])
    
    try:
        res = minimize(obj, x0, jac=grad_obj, bounds=list(zip(lb, ub)),
                      constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac},
                                  {'type': 'ineq', 'fun': ineq_cons}],
                      method='SLSQP', options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or res.x is not None:
            d = res.x[:n]
            if np.all(np.isfinite(d)):
                d -= np.mean(d)
                return d
    except Exception:
        pass
    return None

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        interior = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            x[interior] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads_full(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        if ell >= 0:
            m = n - ell
            if ell == 0:
                G[k, :] = 1.0 - 2.0 * h[:]
            else:
                G[k, ell:] += (1.0 - h[:m])
                G[k, :m] += -h[ell:ell + m]
        else:
            r = -ell
            m = n - r
            G[k, :m] += (1.0 - h[r:r + m])
            G[k, r:] += -h[:m]
    return G

def exact_line_search(h, d, R0, dx):
    n = len(h)
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    pos = d > 1e-12
    neg = d < -1e-12
    a_max = 1e10
    a_min = -1e10
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_min = max(a_min, float(np.max(h[neg] / d[neg])))
    if a_max - a_min < 1e-14:
        return 0.0, c5_exact(h)
    def f(alpha):
        return float(np.max(R0 + alpha * D - alpha * alpha * Q)) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = a_min, a_max
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if b - a < 1e-12 * max(1.0, abs(a)):
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    opt = (a + b) / 2.0
    val = f(opt)
    for pt in [a_min, a_max, 0.0]:
        v = f(pt)
        if v < val:
            val, opt = v, pt
    return opt, val

def solve_qp_cvxpy(h, R, M, rho, lam_z=0.85, lam_n=0.15):
    n = len(h)
    C = 2 * n - 1
    G = compute_grads_full(h)
    z_cap = max(float(np.max(R)) - M + 2.0, 1.0)
    try:
        d_var = cp.Variable(n)
        z_var = cp.Variable(C)
        obj = lam_z * cp.sum(z_var) + lam_n * cp.sum_squares(d_var) / (rho * rho)
        constraints = [
            cp.sum(d_var) == 0,
            R + G @ d_var - M <= z_var,
            d_var >= -np.minimum(h, rho),
            d_var <= np.minimum(1.0 - h, rho),
            z_var >= 0,
            z_var <= z_cap
        ]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        solved = False
        for solver in [cp.ECOS, cp.CVXOPT]:
            try:
                prob.solve(solver=solver, max_iters=300)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        if not solved or d_var.value is None:
            return None
        d = np.array(d_var.value).flatten()
        if not np.all(np.isfinite(d)):
            return None
        d -= np.mean(d)
        return d
    except Exception:
        return None

def solve_qp_slsqp(h, R, M, rho, lam_z=0.85, lam_n=0.15):
    n = len(h)
    C = 2 * n - 1
    G = compute_grads_full(h)
    z_cap = max(float(np.max(R)) - M + 2.0, 1.0)
    x0 = np.zeros(n + C)
    def obj_fn(x):
        d = x[:n]
        z = x[n:]
        return lam_z * np.mean(z) + lam_n * np.dot(d, d) / (rho * rho)
    def grad_fn(x):
        d = x[:n]
        g = np.zeros(n + C)
        g[:n] = 2.0 * lam_n * d / (rho * rho)
        g[n:] = lam_z / C
        return g
    def ineq_fn(x):
        d = x[:n]
        z = x[n:]
        return M + z - R - G @ d
    lb = np.concatenate([-np.minimum(h, rho), np.zeros(C)])
    ub = np.concatenate([np.minimum(1.0 - h, rho), np.full(C, z_cap)])
    try:
        res = sp_minimize(obj_fn, x0, jac=grad_fn,
                          bounds=list(zip(lb, ub)),
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x[:n])},
                                      {'type': 'ineq', 'fun': ineq_fn}],
                          method='SLSQP', options={'maxiter': 150, 'ftol': 1e-10})
        if res.x is not None:
            d = res.x[:n]
            if np.all(np.isfinite(d)) and np.max(np.abs(d)) > 1e-12:
                d -= np.mean(d)
                return d
    except Exception:
        pass
    return None

def golden_pair(h, i, j, dx, S0):
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(len(h))
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C2 = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C2)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C2)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C2)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C2)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C2)) * dx
    for pt in [lo, hi]:
        v = float(np.max(S0 + pt * T - pt * pt * C2)) * dx
        if v < val:
            val, opt = v, pt
    return opt, val

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _qp_search(h_r, sub_dl, 15)
        sub_dl2 = min(d3, time.time() + budget_s * 0.02)
        h_r, c5_r = _gradient_polish(h_r, sub_dl2)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: More QP rounds with varied parameters (10% time)
    d4 = t0 + budget_s * 0.85
    h_p4, c5_p4 = _qp_search(best_h, d4, 30)
    if c5_p4 < best_c5:
        best_c5 = c5_p4
        best_h = h_p4.copy()

    # Phase 5: Gradient polish (5% time)
    d5 = t0 + budget_s * 0.90
    h_p5, c5_p5 = _gradient_polish(best_h, d5)
    if c5_p5 < best_c5:
        best_c5 = c5_p5
        best_h = h_p5.copy()

    # Phase 6: Final pair polish (4% time)
    d6 = deadline
    h_p6, c5_p6 = _pair_search(best_h, d6, 30)
    if c5_p6 < best_c5:
        best_c5 = c5_p6
        best_h = h_p6.copy()

    # Final: project and verify
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
