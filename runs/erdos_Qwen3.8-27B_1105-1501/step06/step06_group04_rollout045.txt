def _lse_solve(h, beta, n_iter=300, tol=1e-10):
    """SLSQP with log-sum-exp objective."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    m = 2 * n - 1
    G = _all_grads(h)

        R = np.correlate(hh, 1.0 - hh, mode='full') * dx
        mx = float(np.max(R))
        z = beta * R - beta * mx
        lse = mx + np.log(np.exp(z).sum()) / beta
        return lse

    def grad(x):
        hh = np.clip(x, 0.0, 1.0)
        R = np.correlate(hh, 1.0 - hh, mode='full') * dx
        mx = float(np.max(R))
        z = beta * R - beta * mx
        w = np.exp(z)
        w /= w.sum()
        g = G.T @ (w * dx)
        return g

    sw = scores[top_idx]
    sw = sw - sw.min() + 0.1
    G = _all_grads(h)
    g = G[top_idx].T @ (sw * dx)
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm

def _dir_from_single(h, scores):
    """Direction: negative gradient of single max constraint."""
    n = len(h)
    dx = 2.0 / n
    r = int(np.argmax(scores))
    g = _all_grads(h)[r] * dx
    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm

def _dir_random(h, rng, support=10):
    """Random zero-sum direction with limited support."""
    n = len(h)
    idx = rng.choice(n, size=min(2 * support, n), replace=False)
    d = np.zeros(n)
    d[idx] = rng.normal(0, 1, size=len(idx))
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm

def _dir_pair(h, rng):
    """Random pair direction."""
    n = len(h)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    return d

def _line_polish(h, deadline, max_rounds=200):
    """Iterative line search polish with diverse directions."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    rng = np.random.default_rng(12345)
    stall = 0

    while time.time() < deadline and stall < 60:
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        improved = False
        directions = []
        d1 = _dir_from_top(h, scores, k=6)
        if d1 is not None:
            directions.append(d1)
        d2 = _dir_from_top(h, scores, k=15)
        if d2 is not None:
            directions.append(d2)
        d3 = _dir_from_single(h, scores)
        if d3 is not None:
            directions.append(d3)
        for _ in range(3):
            dr = _dir_random(h, rng, support=8)
            if dr is not None:
                directions.append(dr)
        for _ in range(4):
            dp = _dir_pair(h, rng)
            if dp is not None:
                directions.append(dp)

        for d in directions:
            t, c5_try = _line_search(h, d, trust_radius=0.3, max_iter=60)
            if abs(t) > 1e-12 and c5_try < best_c5 - 1e-14:
                h = h + t * d
                h = _project(h, target)
                cur = _compute_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    stall = 0
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps * period:] = pattern[:n - reps * period]
    return _project(h, n / 2.0)

def _binary_quantize(h):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[sorted_idx[:target]] = 1.0
    return hb

def _full_optimize(h, deadline, use_slsqp=True, use_minimax=True):
    """Full optimization pipeline on a single h."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)

    # SLSQP log-sum-exp with multiple betas
    if use_slsqp and time.time() < deadline - 10:
        for beta in [20, 50, 120, 300, 600, 1200]:
            if time.time() > deadline - 15:
                break
            sl_dl = min(deadline - 5, time.time() + (deadline - time.time()) * 0.15)
            h_try, c5_try = _lse_solve(h, beta, n_iter=200)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()

    # Direct minimax
    if use_minimax and time.time() < deadline - 10:
        sl_dl = min(deadline - 5, time.time() + 30)
        h_try, c5_try = _minimax_solve(best_h, maxiter=150)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Line search polish
    if time.time() < deadline - 5:
        lp_dl = min(deadline - 2, time.time() + (deadline - time.time()) * 0.4)
        h_try, c5_try = _line_polish(best_h, lp_dl, max_rounds=300)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.default_rng(seed)

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 120
        h0 = _sine_init(n0, 3, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: Direct SLSQP optimization on parent (35% time)
    d1 = t0 + budget_s * 0.35
    h1, c5_1 = _full_optimize(best_h, d1, use_slsqp=True, use_minimax=True)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Binary quantize + optimize (10% time)
    d2 = t0 + budget_s * 0.45
    hb = _binary_quantize(best_h)
    c5_b = _compute_c5(hb)
    if time.time() < d2 - 5:
        h2, c5_2 = _full_optimize(hb, d2, use_slsqp=True, use_minimax=False)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Multi-start with diverse initializations (20% time)
    d3 = t0 + budget_s * 0.65
    n = len(best_h)
    starts = []
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        for ph in [0.0, 0.7, 1.5, 2.5]:
            starts.append(_sine_init(n, freq, ph))
    for p in [4, 5, 6, 7, 8, 10, 12, 15]:
        for duty in [0.3, 0.4, 0.5, 0.6, 0.7]:
            starts.append(_periodic_init(n, p, duty))
    for _ in range(6):
        starts.append(_project(best_h + rng.normal(0, 0.05, n), n / 2.0))

    for s in starts:
        if time.time() > d3 - 5:
            break
        sub_dl = min(d3 - 2, time.time() + budget_s * 0.015)
        h_s, c5_s = _full_optimize(s, sub_dl, use_slsqp=True, use_minimax=False)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 4: Multi-resolution (20% time)
    d4 = t0 + budget_s * 0.85
    best_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d4 - 10:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4 - 5, time.time() + budget_s * 0.025)
        h_r, c5_r = _full_optimize(h_r, sub_dl, use_slsqp=True, use_minimax=True)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Intensive line polish (10% time)
    d5 = deadline
    h5, c5_5 = _line_polish(best_h, d5, max_rounds=500)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final: one more SLSQP pass if time allows
    if time.time() < deadline - 10:
        hf, c5_f = _lse_solve(best_h, 800, n_iter=300)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0_f = np.array(initial_h_values, dtype=float)
            h0_f = _project(h0_f, len(h0_f) / 2.0)
            return h0_f, _compute_c5(h0_f), len(h0_f)
        except NameError:
            h_fb = _sine_init(120, 3, 0.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
