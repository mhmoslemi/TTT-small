def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def overlap_matrix(h):
    """Return all overlap values (the full correlation)."""
    return np.correlate(h, 1.0 - h, mode='full')

def project(v, target):
    """Project v onto {0<=x<=1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grads(h, active_shifts):
    """Compute gradient matrix for given shifts. Returns (len(active_shifts), n)."""
    n = len(h)
    G = np.zeros((len(active_shifts), n))
    for k, d in enumerate(active_shifts):
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            # O(d) = sum_{i=0}^{m-1} h[i+d]*(1-h[i])
            # dO/dd_h[j] = (1-h[j-d]) for j>=d, plus -h[j+d] for j+d<n
            G[k, d:] += (1.0 - h[:m])       # (1-h[j-d]) for j=d..n-1
            G[k, :m] += -h[d:d+m]           # -h[j+d] for j=0..m-1
        else:
            # O(-d) = sum_{i=0}^{m-1} h[i]*(1-h[i+|d|])
            # dO/dh[j] = (1-h[j+|d|]) for j+|d|<n, plus -h[j-|d|]) for j>=|d|
            r = -d
            G[k, :m] += (1.0 - h[r:r+m])    # (1-h[j+r]) for j=0..m-1
            G[k, r:] += -h[:m]              # -h[j-r] for j=r..n-1

def subgradient_descent(h, deadline, n_iter, eps_active=0.01):
    """Subgradient method with Polyak step size."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur
    # Best upper bound for step size (Polyak)
    C_best = cur  # target is 0 (we want to minimize)

            break
        
        S = overlap_matrix(h)
        scores = S * dx
        max_score = np.max(scores)
        
        # Active set: within eps of max
        active = np.where(scores >= max_score - eps_active)[0]
        active_shifts = active - (n - 1)
        
        # Weighted gradient
        weights = np.exp((scores[active] - max_score) / (0.005 + 1e-15))
        weights /= np.sum(weights)
        
        G = compute_grads(h, active_shifts)
        g = G.T @ weights  # gradient of weighted active overlaps
        
        # Normalize
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        # Line search: try several step sizes
        best_step_val = cur
        best_alpha = 0.0
        # Try decreasing step sizes
        alphas = np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001] * (n / 100))
        for alpha in alphas:
            h_new = h - alpha * g
            h_new = project(h_new, target)
            val = c5_exact(h_new)
            if val < best_step_val - 1e-14:
                best_step_val = val
                best_alpha = alpha
                h_new_best = h_new.copy()
        
        if best_alpha > 0:
            h = h_new_best
            cur = best_step_val
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
    
    return best_h, best_c5
