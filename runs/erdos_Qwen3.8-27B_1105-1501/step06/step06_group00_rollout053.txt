def _compute_gradient(h, scores, beta=20.0):
    """Compute soft-max gradient of C5 w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    S = scores  # scores * dx already? No, let's use raw scores
    
    # Soft-max weights
    S_scaled = S * dx  # the actual C5-contributing values
    M = np.max(S_scaled)
    # Clip for numerical stability
    exp_args = np.clip(beta * (S_scaled - M), -60.0, 0.0)
    w = np.exp(exp_args)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        # All weights are zero, use uniform on argmax
        w = np.zeros(2*n-1)
        idx = np.where(S_scaled == M)[0]
        w[idx] = 1.0 / len(idx)
    else:
        w = w / w_sum
    
    # Compute gradient
    # g[i] = sum_m w[m] * dS[m]/dh[i] * dx (the dx factor)
    # Actually, since we're only using g for relative comparisons, we can drop dx.
    
    g = np.zeros(n)
    
    # Contribution from lag 0 (m = n-1)
    w0 = w[n-1]
    g += w0 * (1.0 - 2.0*h)
    
    # Contributions from positive lags (m = n-1+k for k = 1, ..., n-1)
    for k in range(1, n):
        wk = w[n-1+k]
        if wk < 1e-15:
            continue
        # (1-h[i-k]) for i >= k
        if k < n:
            g[k:] += wk * (1.0 - h[:n-k])
        # -h[i+k] for i <= n-1-k
        if k < n:
            g[:n-k] += -wk * h[k:]
    
    # Contributions from negative lags (m = n-1-k for k = 1, ..., n-1)
    for k in range(1, n):
        wk = w[n-1-k]
        if wk < 1e-15:
            continue
        # (1-h[i+k]) for i <= n-1-k
        g[:n-k] += wk * (1.0 - h[k:])
        # -h[i-k] for i >= k
        k = k  # already k
        g[k:] += -wk * h[:n-k]
    
    return g

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    radius_cur = radius
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        
        scores = np.correlate(h, 1.0 - h, mode='full')
        g = _compute_gradient(h, scores, beta)
        
        # Select sources (high gradient) and sinks (low gradient)
        # We want to decrease h at high-grad points and increase at low-grad points
        sources = np.argsort(g)[-n_source:]  # highest gradient
        sinks = np.argsort(g)[:n_sink]      # lowest gradient
        
        # Try all source-sink pairs
        best_improvement = None
        best_t = 0
        best_pair = None
        
        for a in sources:
            for b in sinks:
                if a == b:
                    continue
                # d[b] = +1, d[a] = -1
                # Compute T for all lags
                # (expensive, but only K² pairs)
                T, C = _pair_TC(h, a, b, n)
                # Line search
                t_lo = 0.0
                t_hi = min(1.0 - h[b], h[a], radius_cur)
                if t_hi <= 1e-12:
                    continue
                alpha, val = _line_search_pair(scores, T, C, dx, t_lo, t_hi)
                if val < cur_c5 - 1e-13:
                    if best_improvement is None or val < best_improvement:
                        best_improvement = val
                        best_t = alpha
                        best_pair = (a, b)
        
        if best_pair is not None:
            a, b = best_pair
            h[b] += best_t
            h[a] -= best_t
            # Project to ensure sum constraint (should already be satisfied)
            h = np.clip(h, 0.0, 1.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            radius_cur = min(radius_cur * 1.05, 0.30)
        else:
            stall += 1
            radius_cur *= 0.8
            if radius_cur < 1e-4:
                break
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_gradient(h, scores, beta=20.0):
    n = len(h)
    dx = 2.0 / n
    S = scores * dx
    M = float(np.max(S))
    exp_a = np.clip(beta * (S - M), -60.0, 0.0)
    w = np.exp(exp_a)
    ws = float(np.sum(w))
    if ws < 1e-30:
        w = np.zeros(2*n-1)
        idx = np.where(S >= M - 1e-12)[0]
        w[idx] = 1.0 / len(idx)
    else:
        w = w / ws
    
    center = n - 1
    g = np.zeros(n)
    # Lag 0
    g += w[center] * (1.0 - 2.0 * h)
    # Positive lags
    for k in range(1, n):
        wk = w[center + k]
        if wk < 1e-15:
            continue
        m = n - k
        if m > 0:
            g[k:] += wk * (1.0 - h[:m])
            g[:m] += -wk * h[k:]
    # Negative lags
    for k in range(1, n):
        wk = w[center - k]
        if wk < 1e-15:
            continue
        m = n - k
        if m > 0:
            g[:m] += wk * (1.0 - h[k:])
            g[k:] += -wk * h[:m]
    return g
