def _line_search_quad(h, d, corr, L, U, dx, max_iter=40):
    """Quadratic model line search (fast)."""
    n = len(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    S0 = corr.copy()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)

            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    ft = float(np.max(S0 + t * T - t * t * Q)) * dx

def _feasible_bounds(h, d):

    pos = d > 1e-15

    up = [1e30]

        up.append(float(np.min(-h[neg] / d[neg])))
    L = max(low)
    U = min(up)
    return L, U

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)

def _binary_init(n, rng, duty=0.5):
    """Generate a binary-like pattern with 50% duty cycle."""
    h = np.zeros(n)
    # Create blocks
    cursor = 0
    while cursor < n:
        blen = max(2, int(n * rng.uniform(0.05, 0.25)))
        blen = min(blen, n - cursor)
        if rng.random() < duty:
            h[cursor:cursor+blen] = 1.0
        cursor += blen
    h += rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)

import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-14:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _feasible_bounds(h, d):
    pos = d > 1e-15
    neg = d < -1e-15
    low = [0.0]
    up = [1e30]
    if np.any(pos):
        low.append(float(np.min(-h[pos] / d[pos])))
        up.append(float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        low.append(float(np.min((1.0 - h[neg]) / d[neg])))
        up.append(float(np.min(-h[neg] / d[neg])))
    return max(low), min(up)


def _line_search(h, d, L, U, dx, corr, max_iter=35):
    n = len(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    S0 = corr
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    # Verify with exact eval and try a few nearby points
    t_eval = t
    best_tv = _exact_c5(h + t * d)
    for frac in [0.5, 0.7, 1.3, 1.5]:
        tc = t * frac
        if tc < L - 1e-14 or tc > U + 1e-14:
            continue
        tc = max(L, min(U, tc))
        hv = h + tc * d
        if hv.min() < -1e-9 or hv.max() > 1.0 + 1e-9:
            continue
        tv = _exact_c5(hv)
        if tv < best_tv:
            best_tv = tv
            t_eval = tc
    return t_eval, best_tv


def _subgradient(h, corr, k_active, tau):
    n = len(h)
    mx = float(np.max(corr))
    w = np.exp((corr - mx) / tau)
    idx = np.argsort(-corr)[:k_active]
    G = np.zeros(n)
    for ri in idx:
        k = ri - (n - 1)
        if k >= 0:
            nn = n - k
            a = np.arange(nn)
            d1 = np.zeros(n)
            d1[a] = 1.0 - h[a + k]
            b = np.arange(k, n)
            d2 = np.zeros(n)
            d2[b] = -h[b - k]
            G += w[ri] * (d1 + d2)
        else:
            k2 = -k
            nn = n - k2
            a = np.arange(nn)
            d1 = np.zeros(n)
            d1[a] = 1.0 - h[a + k2]
            b = np.arange(k2, n)
            d2 = np.zeros(n)
            d2[b] = -h[b - k2]
            G += w[ri] * (d1 + d2)
    return G


def _targeted_pair(h, corr, n, rng):
    i_act = int(np.argmax(corr))
    k = i_act - (n - 1)
    d = np.zeros(n)
    if k >= 0:
        nn = n - k
        if nn <= 0:
            return None
        i = int(rng.integers(nn))
        if rng.random() < 0.5:
            j = int(rng.integers(n))
            while j == i:
                j = int(rng.integers(n))
            d[i] = -1.0
            d[j] = 1.0
        else:
            j = int(rng.integers(n))
            while j == i + k:
                j = int(rng.integers(n))
            d[i + k] = 1.0
            d[j] = -1.0
    else:
        k2 = -k
        nn = n - k2
        if nn <= 0:
            return None
        i = int(rng.integers(nn))
        if rng.random() < 0.5:
            j = int(rng.integers(n))
            while j == i:
                j = int(rng.integers(n))
            d[i] = -1.0
            d[j] = 1.0
        else:
            j = int(rng.integers(n))
            while j == i + k2:
                j = int(rng.integers(n))
            d[i + k2] = 1.0
            d[j] = -1.0
    return d


def _optimize(h, deadline, rng, k_active=8, tau0=0.025, tau_min=0.006,
              tau_decay=0.9995, pair_frac=0.18, rand_frac=0.07,
              uphill_tol=2e-4, stall_limit=80, use_momentum=True,
              momentum_beta=0.5):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    sigma = 0.04
    tau = tau0
    mom = np.zeros(n)
    steps = 0

    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
        elif r < rand_frac + pair_frac:
            d = _targeted_pair(h, corr, n, rng)
            if d is None:
                d = rng.standard_normal(n)
        else:
            G = _subgradient(h, corr, k_active, tau)
            d = -G
            if use_momentum:
                mom = momentum_beta * mom + d
                d = mom

        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm

        L, U = _feasible_bounds(h, d)
        if U - L < 1e-12:
            stall += 1
            continue

        t, newf = _line_search(h, d, L, U, 2.0 / n, corr)
        if abs(t) < 1e-14:
            stall += 1
            continue

        hn = h + t * d
        if hn.min() < -1e-9 or hn.max() > 1.0 + 1e-9:
            hn = _project(hn, target)

        nf = _exact_c5(hn)
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill_tol:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                h = _project(best_h + rng.normal(0, sigma, n), target)
                cur_f = _exact_c5(h)
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
                sigma = max(0.003, sigma * 0.82)
                mom = np.zeros(n)

        tau = max(tau_min, tau * tau_decay)
        steps += 1

    return best_h, best_f


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)


def _binary_init(n, rng):
    h = np.zeros(n)
    cursor = 0
    while cursor < n:
        blen = max(2, int(n * rng.uniform(0.04, 0.20)))
        blen = min(blen, n - cursor)
        if rng.random() < 0.5:
            h[cursor:cursor + blen] = 1.0
        cursor += blen
    h += rng.normal(0, 0.08, n)
    return _project(h, n / 2.0)


def _fourier_init(n, rng):
    h = np.full(n, 0.5)
    for k in range(1, 8):
        amp = rng.uniform(0.01, 0.12) / (k + 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2.0 * np.pi * k * np.arange(n) / n + phase)
    return _project(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
