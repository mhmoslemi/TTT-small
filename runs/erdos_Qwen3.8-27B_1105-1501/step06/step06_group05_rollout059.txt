import numpy as np
import time
import functools
from scipy.optimize import minimize


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x = x.copy()
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _overlap_grad(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    q = abs(k)
    if q >= n:
        return g
    idx1 = np.arange(q, n)
    g[idx1] += dx * (1.0 - h[idx1 - q])
    idx2 = np.arange(n - q)
    g[idx2] += -dx * h[idx2 + q]
    return g


def _overlap_val(h, k, dx):
    n = len(h)
    q = abs(k)
    if q >= n:
        return 0.0
    r0 = (n - 1) + k
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr[r0]) * dx


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _epi_fun(x, n):
    return x[n]


def _epi_fun_grad(x, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _epi_eq_fun(x, n):
    return np.sum(x[:n]) - n / 2.0


def _epi_eq_grad(x, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def _epi_ineq_fun(x, n, lags, dx):
    h = x[:n]
    z = x[n]
    m = len(lags)
    vals = np.zeros(m)
    for i in range(m):
        k = int(lags[i])
        q = abs(k)
        if q >= n:
            vals[i] = -z
        else:
            if k >= 0:
                idx = np.arange(n - q)
                v = dx * float(np.sum(h[idx + q] * (1.0 - h[idx])))
            else:
                idx = np.arange(n - q)
                v = dx * float(np.sum(h[idx + q] * (1.0 - h[idx])))
            vals[i] = z - v
    return vals


def _epi_ineq_grad(x, n, lags, dx):
    h = x[:n]
    m = len(lags)
    J = np.zeros((m, n + 1))
    for i in range(m):
        k = int(lags[i])
        J[i, :n] = _overlap_grad(h, k, dx)
        J[i, n] = 1.0
    return J


def _slsqp_solve(h0, n, maxiter=150, top_m=40, time_budget=30.0):
    t_start = time.time()
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    m_total = len(corr)
    top_m = min(top_m, m_total)
    active_idx = np.argsort(-corr)[:top_m]
    active_lags = active_idx - (n - 1)

    bounds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
    z0 = float(np.max(corr)) * dx + 0.02
    x0 = np.concatenate([h0, [z0]])

    p_fun = functools.partial(_epi_fun, n=n)
    p_fun_grad = functools.partial(_epi_fun_grad, n=n)
    p_eq = functools.partial(_epi_eq_fun, n=n)
    p_eq_grad = functools.partial(_epi_eq_grad, n=n)
    p_ineq = functools.partial(_epi_ineq_fun, n=n, lags=active_lags, dx=dx)
    p_ineq_grad = functools.partial(_epi_ineq_grad, n=n, lags=active_lags, dx=dx)

    cons = [
        {'type': 'eq', 'fun': p_eq, 'jac': p_eq_grad},
        {'type': 'ineq', 'fun': p_ineq, 'jac': p_ineq_grad}
    ]

    try:
        result = minimize(p_fun, x0, method='SLSQP', jac=p_fun_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-10})
    except Exception:
        return h0, _exact_c5(h0)

    if time.time() - t_start > time_budget:
        return h0, _exact_c5(h0)

    h_new = _project(result.x[:n], n / 2.0)
    f_new = _exact_c5(h_new)
    f_old = _exact_c5(h0)
    if f_new < f_old:
        return h_new, f_new
    return h0, f_old

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 5))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv


def _polish(h, deadline, rng, pair_frac=0.20, rand_frac=0.08, k_active=10, tau=0.02):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    sigma = 0.04
    uphill = 5e-4
    restarts = 0

    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            scores = corr
            m = float(np.max(scores))
            w = np.exp((scores - m) / tau)
            idx = np.argsort(-scores)[:k_active]
            G = np.zeros(n)
            for ri in idx:
                k = int(ri - (n - 1))
                q = abs(k)
                if q < n:
                    idx1 = np.arange(q, n)
                    G[idx1] += w[ri] * dx * (1.0 - h[idx1 - q])
                    idx2 = np.arange(n - q)
                    G[idx2] += w[ri] * (-dx * h[idx2 + q])
            d = -G
            d -= d.mean()
            nrm = float(np.linalg.norm(d))
            if nrm < 1e-14:
                d = rng.standard_normal(n)
                d -= d.mean()
                nrm = float(np.linalg.norm(d))
            d = d / nrm

        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm

        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-11:
            stall += 1
            continue

        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, dx, L, U)
        if abs(t) < 1e-14:
            stall += 1
            continue

        hn = h + t * d
        if hn.min() < 0.0 or hn.max() > 1.0:
            hn = _project(hn, target)
        nf = _exact_c5(hn)

        if nf < best_f - 1e-13:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall += 1

        if stall > 100:
            restarts += 1
            noise_scale = max(0.01, sigma)
            h = _project(best_h + rng.normal(0, noise_scale, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma *= 0.85

    return best_h, best_f


def _coordinated_descent(h, deadline, rng, top_m=15, max_rounds=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_f = _exact_c5(h)
    h = h.copy()

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        m = float(np.max(corr))
        idx = np.argsort(-corr)[:top_m]

        for ri in idx:
            k = int(ri - (n - 1))
            q = abs(k)
            if q < n:
                idx1 = np.arange(q, n)

                idx2 = np.arange(n - q)
                G[idx2] += -dx * h[idx2 + q]
        d = -G
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            continue

        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-12:
            continue

        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')

        if nf < best_f:
            best_f = nf
            best_h = hn.copy()

    return best_h, best_f


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None

    best_h = None
    best_f = float('inf')

    if base is not None and base.size >= 8:
        n0 = base.size
        best_h = _project(base, n0 / 2.0)
        best_f = _exact_c5(best_h)
    else:
        n0 = 100
        best_h = _sine_init(n0, 1, 0.0, rng)
        best_f = _exact_c5(best_h)

    # Phase 1: Multi-n SLSQP with top-m constraints (25% budget)
    d1 = t0 + budget_s * 0.25
    n_candidates = []
    bn = len(best_h)
    for dn in [0, -4, 4, -8, 8, -12, 12, -16, 16]:
        nt = bn + dn
        if 20 <= nt <= 220:
            n_candidates.append(nt)
    for extra in [40, 50, 60, 80, 100, 140, 160]:
        if extra not in n_candidates:
            n_candidates.append(extra)
    n_candidates = sorted(set(n_candidates))

    for nc in n_candidates:
        if time.time() > d1 - 2:
            break
        if best_h is not None:
            h_start = _resample(best_h, nc)
        else:
            h_start = _sine_init(nc, 1, 0.0, rng)
        time_per = max(3.0, (d1 - time.time()) / max(1, len(n_candidates)))
        h_out, f_out = _slsqp_solve(h_start, nc, maxiter=80, top_m=30,
                                     time_budget=min(time_per, 25.0))
        if f_out < best_f:
            best_f = f_out
            best_h = h_out.copy()

    # Phase 2: Diversified starts + SLSQP (25% budget)
    d2 = t0 + budget_s * 0.50
    bn2 = len(best_h)
    starts = []
    freq_phase = [(1, 0.0), (1, 1.0), (2, 0.5), (2, 1.5), (3, 0.0),
                  (3, 1.0), (4, 0.5), (5, 0.0)]
    for f_, p_ in freq_phase:
        starts.append(_sine_init(bn2, f_, p_, rng))
    for _ in range(3):
        starts.append(_block_init(bn2, rng))
    # Complement
    starts.append(_project(1.0 - best_h, bn2 / 2.0))
    # Reverse
    starts.append(_project(best_h[::-1].copy(), bn2 / 2.0))
    # Add noise to best
    starts.append(_project(best_h + rng.normal(0, 0.05, bn2), bn2 / 2.0))
    starts.append(_project(best_h + rng.normal(0, 0.03, bn2), bn2 / 2.0))

    for s in starts:
        if time.time() > d2 - 2:
            break
        time_per = max(2.0, (d2 - time.time()) / max(1, len(starts)))
        h_out, f_out = _slsqp_solve(s, bn2, maxiter=60, top_m=25,
                                     time_budget=min(time_per, 20.0))
        if f_out < best_f:
            best_f = f_out
            best_h = h_out.copy()

    # Phase 3: Expanding SLSQP (all constraints) on best (20% budget)
    d3 = t0 + budget_s * 0.70
    bn3 = len(best_h)
    h_out, f_out = _slsqp_solve_expand(best_h, bn3, maxiter=80,
                                        time_budget=min(budget_s * 0.12, 50.0))
    if f_out < best_f:
        best_f = f_out
        best_h = h_out.copy()

    # Second expand with perturbation
    if time.time() < d3 - 5:
        h_pert = _project(best_h + rng.normal(0, 0.02, bn3), bn3 / 2.0)
        h_out2, f_out2 = _slsqp_solve_expand(h_pert, bn3, maxiter=60,
                                              time_budget=min(d3 - time.time() - 1, 30.0))
        if f_out2 < best_f:
            best_f = f_out2
            best_h = h_out2.copy()

    # Phase 4: Coordinated descent + polish (20% budget)
    d4 = t0 + budget_s * 0.90
    h_out, f_out = _coordinated_descent(best_h, min(d4, time.time() + budget_s * 0.08),
                                         rng, top_m=20, max_rounds=300)
    if f_out < best_f:
        best_f = f_out
        best_h = h_out.copy()

    h_out2, f_out2 = _polish(best_h, d4, rng, pair_frac=0.15, rand_frac=0.05,
                             k_active=8, tau=0.02)
    if f_out2 < best_f:
        best_f = f_out2
        best_h = h_out2.copy()

    # Phase 5: Final resampling + polish (10% budget)
    bn5 = len(best_h)
    for dn in [-2, 2, -4, 4]:
        if time.time() > deadline - 10:
            break
        nt = bn5 + dn
        if nt < 20 or nt > 250 or nt % 2 != 0:
            continue
        hr = _resample(best_h, nt)
        time_left = deadline - time.time() - 5
        if time_left < 3:
            break
        hr2, fr2 = _polish(hr, min(deadline - 2, time.time() + time_left * 0.4),
                           rng, pair_frac=0.25, rand_frac=0.05, k_active=8, tau=0.015)
        if fr2 < best_f:
            best_f = fr2
            best_h = hr2.copy()
            bn5 = nt

    # Final SLSQP polish on best
    if time.time() < deadline - 10:
        bnf = len(best_h)
        time_left = deadline - time.time() - 2
        h_fin, f_fin = _slsqp_solve(best_h, bnf, maxiter=50, top_m=35,
                                     time_budget=min(time_left, 20.0))
        if f_fin < best_f:
            best_f = f_fin
            best_h = h_fin.copy()

    # Final projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
