Let me try a different approach: **subgradient method with averaging**.

The subgradient of f(h) = max_k S_k(h) is the average of gradients of active constraints.
If multiple S_k are equal to the max, the subgradient is the average of their individual gradients.

Algorithm:
1. Start with h0
2. At each step, compute S = corr(h), find active set A = {k : S[k] ≈ max}
3. Compute subgradient g = mean of gradients of constraints in A
4. Take step: h_new = h - α * g
5. Project onto constraint set
6. Decrease α
7. Average iterates (for convergence)

            if np.any(pos):
                t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
            if np.any(neg):
                t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
            t_lo = max(t_lo, 0.0)
            t_hi = min(t_hi, 0.5)
            if t_hi - t_lo < 1e-14:
                continue
            g = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = t_lo, t_hi
            c1 = b - g * (b - a)
            c2 = a + g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
            for _ in range(40):
                if b - a < 1e-13:
                    break
                if f1 < f2:
                    b, c2, f2 = c2, c1, f1
                    c1 = b - g * (b - a)
                    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
                else:
                    a, c1, f1 = c1, c2, f2
                    c2 = a + g * (b - a)
                    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
            opt_t = (a + b) / 2.0
            h_new = h + opt_t * d
            h_new = _project(h_new, target)
            c5_try = _exact_c5(h_new)
            if c5_try < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 3, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(best_h)

    # Phase 1: Fast PGD from initial (15%)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _pgd_run(best_h, d1, beta0=60.0, lr0=0.06)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Subgradient with averaging (10%)
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _subgrad_run(best_h, d2, alpha0=0.06)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary annealing (12%)
    d3 = t0 + budget_s * 0.37
    h3, c5_3 = _binary_anneal(best_h, d3, max_iters=25000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: PGD from binary solution (10%)
    d4 = t0 + budget_s * 0.47
    h4, c5_4 = _pgd_run(best_h, d4, beta0=100.0, lr0=0.05)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-start with PGD (20%)
    d5 = t0 + budget_s * 0.67
    n = len(best_h)
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        for ph in [0.0, 0.7, 1.5, 2.3]:
            starts.append(_sine_init(n, freq, ph))
    for p in [4, 5, 6, 7, 8, 10, 12, 15]:
        for duty in [0.35, 0.45, 0.5, 0.55, 0.65]:
            xv = np.arange(n)
            hv = np.where((xv % p) < int(duty * p), 1.0, 0.0)
            starts.append(_project(hv.astype(float), n / 2.0))
    for _ in range(6):
        hv = (rng.random(n) > 0.5).astype(float)
        starts.append(_project(hv, n / 2.0))

    for s in starts:
        if time.time() > d5:
            break
        remaining = d5 - time.time()
        if remaining < 2:
            break
        sub_dl = min(d5, time.time() + max(1.0, remaining * 0.08))
        h_s, c5_s = _pgd_run(s, sub_dl, beta0=80.0, lr0=0.06)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Multi-resolution (15%)
    d6 = t0 + budget_s * 0.82
    best_n = len(best_h)
    offsets = [-4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18, -24, 24, -30, 30, 36, -36]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + max(1.0, budget_s * 0.012))
        h_r, c5_r = _pgd_run(h_r, sub_dl, beta0=80.0, lr0=0.06)
        if c5_r < best_c5 - 1e-13 and time.time() < d6 - 5:
            h_r2, c5_r2 = _pair_refine(h_r, min(d6, time.time() + budget_s * 0.006), max_passes=30)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 7: Pair refinement (7%)
    d7 = t0 + budget_s * 0.89
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=120)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final PGD polish (6%)
    d8 = t0 + budget_s * 0.95
    h8, c5_8 = _pgd_run(best_h, d8, beta0=200.0, lr0=0.03)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Phase 9: Final subgradient (5%)
    h9, c5_9 = _subgrad_run(best_h, deadline, alpha0=0.03)
    if c5_9 < best_c5:
        best_c5 = c5_9
        best_h = h9.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h0f = np.array(initial_h_values, dtype=float)
            h0f = _project(h0f, len(h0f) / 2.0)
            return h0f, _exact_c5(h0f), len(h0f)
        except NameError:
            h_fb = _sine_init(122, 3, 0.0)
            return h_fb, _exact_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
