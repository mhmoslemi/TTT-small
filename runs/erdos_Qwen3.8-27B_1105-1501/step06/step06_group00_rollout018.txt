    # g[x] second part = -sum_d w[d+n-1] * h[x-d] * I(0<=x-d<n)
    # = -sum_y h[y] * w[(x-y)+(n-1)] * I(0<=y<n)  [substituting y=x-d, d=x-y]
    # = -sum_y h[y] * w[x-y+n-1]
    
    # So g[x] = sum_y (1-h[y]) * w[y-x+n-1] - sum_y h[y] * w[x-y+n-1]
    
    # Note: w[y-x+n-1] and w[x-y+n-1] are related by: w[n-1+k] and w[n-1-k]
    # Let me define k = y-x, then y-x+n-1 = k+n-1 and x-y+n-1 = -k+n-1
    # These are symmetric: w[n-1+k] vs w[n-1-k]
    
    # Let me just compute this with a loop or matmul.
    # Actually, I can use np.correlate:
    # sum_y (1-h[y]) * w[y-x+n-1] = cross-correlation of (1-h) and w, evaluated at shift (n-1-x)
    # np.correlate(a, b, 'full')[k] = sum_j a[j] * b[j + k - (len(b)-1)]
    # Hmm, not quite the right convention.
    
    # Let me just do it with a simple approach: build a circulant-like matrix.
    # Actually for n up to ~200, just do a double loop or use einsum.
    
    # Simple approach: for each shift d, add w[d] * grad_shift to g
    # grad_shift[x] for shift d = (1-h[x+d])*I(0<=x+d<n) - h[x-d]*I(0<=x-d<n)
    
    g = np.zeros(n)
    for d_idx in range(2 * n - 1):
        d = d_idx - (n - 1)
        if abs(weights[d_idx]) < 1e-15:
            continue
        wd = weights[d_idx]
        # First part: (1-h[x+d]) for x where 0<=x+d<n, i.e., -d<=x<n-d (clamped to [0,n-1])
        if d >= 0:
            # x from 0 to n-1-d
            x_start = 0
            x_end = n - d  # exclusive
            g[x_start:x_end] += wd * (1.0 - h[d:d+n-d])
        else:
            # x from -d to n-1
            p = -d
            x_start = p
            x_end = n
            g[x_start:x_end] += wd * (1.0 - h[d+x_start:d+x_end])
            # h[d+x] for x in [p, n-1] = h[d+p : d+n] = h[0 : n-p] = h[0:n+d]
            # Wait, d is negative. d+p = d-d = 0. d+n = n+d = n-p.
            # So h[0:n-p], length n-p = n+d. Correct.
        
        # Second part: -h[x-d] for x where 0<=x-d<n, i.e., d<=x<n+d (clamped)
        if d >= 0:
            # x from d to n-1
            x_start = d
            x_end = n
            g[x_start:x_end] -= wd * h[0:n-d]
            # h[x-d] for x in [d, n-1] = h[0:n-d]
        else:
            # x from 0 to n-1+d
            p = -d
            x_start = 0
            x_end = n - p  # n + d
            g[x_start:x_end] -= wd * h[p:n]
            # h[x-d] for x in [0, n-p-1] = h[x+p] for x in [0, n-p-1] = h[p:n]
    
    return g

Actually, I realize I can compute the gradient more efficiently. Let me define:
- a[x] = 1-h[x]
- b[x] = h[x]

Then g[x] = sum_d w[d] * (a[x+d]*I(0<=x+d<n) - b[x-d]*I(0<=x-d<n))

The first term is a "shifted" version of a weighted by w. This is a convolution!

First term at position x: sum_d w[d] * a[x+d] * I(0<=x+d<n)

If we define the sum over ALL d (not just valid ones), and a[y] = 0 for y outside [0,n-1], then:
First term = sum_d w[d] * a[x+d] = (w * a)(x) where * is cross-correlation.

Specifically, if we pad a to length 2n-1 (with the valid part in the middle), then:
first_term = np.correlate(w, a_padded, 'full') evaluated at appropriate positions.

Actually, let me just be pragmatic. For n ≤ 200, a simple O(n²) loop is fast enough (200*200 = 40000 operations, which is nothing for numpy).

Let me restructure:

```python
def _compute_grad_smooth(h, gamma=50.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = gamma * corr
    scaled_max = scaled.max()
    exp_vals = np.exp(scaled - scaled_max)
    w = exp_vals / exp_vals.sum()
    
    g = np.zeros(n)
    # For each shift d, add w[d_idx] * grad_shift_d
    # grad_shift_d[x] = (1-h[x+d]) if 0<=x+d<n, else 0
    #                  + (-h[x-d]) if 0<=x-d<n, else 0
    
    for d_idx in range(2 * n - 1):
        d = d_idx - (n - 1)
        wd = w[d_idx]
        if wd < 1e-15:
            continue
        # Part 1: (1-h[x+d]) for valid x
        if d >= 0:
            m = n - d
            g[:m] += wd * (1.0 - h[d:d+m])
        else:
            p = -d
            m = n - p
            g[p:p+m] += wd * (1.0 - h[:m])
            # h[x+d] for x in [p, n-1]: x+d = x-p, ranges from 0 to m-1
            # So h[0:m]
            # Wait: x ranges from p to n-1, x+d = x-p, ranges from 0 to n-1-p = m-1
            # h[x+d] = h[x-p], and x-p ranges from 0 to m-1
            # g[p:p+m] += wd * (1 - h[0:m])  -- this is what I have above. ✓
        
        # Part 2: -h[x-d] for valid x
        if d >= 0:
            m = n - d
            g[d:d+m] -= wd * h[:m]
            # x ranges from d to n-1, x-d ranges from 0 to n-1-d = m-1
            # h[x-d] = h[0:m]
            # g[d:d+m] -= wd * h[0:m] ✓
        else:
            p = -d
            m = n - p
            g[:m] -= wd * h[p:p+m]
            # x ranges from 0 to n-1-p = m-1, x-d = x+p ranges from p to n-1
            # h[x-d] = h[x+p], and x+p ranges from p to n-1
            # h[p:p+m] where m = n-p ✓
    
    return g
```
