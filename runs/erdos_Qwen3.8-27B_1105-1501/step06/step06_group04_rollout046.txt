    R = np.correlate(h, 1.0 - h, mode='full')

def _corr_values(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.clip(np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0), 0, 1)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:

    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    """Compute gradient of all C_r with respect to h. Returns (2n-1, n) matrix."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        if d >= 0:
            m = n - d
            if m > 0:
                G[k, d:] += (1.0 - h[:m])
                G[k, :m] += -h[d:]
        else:
            g = -d
            m = n - g
            if m > 0:
                G[k, :m] += (1.0 - h[g:])
                G[k, g:] += -h[:m]
    return G  # unscaled; multiply by dx for scaled gradient

def _lse_obj_grad(x, beta, dx):
    """Compute LSE objective and gradient."""
    n = len(x)
    R = np.correlate(x, 1.0-x, mode='full') * dx
    max_R = float(np.max(R))
    z = beta * (R - max_R)
    exp_z = np.exp(z)
    sum_exp = float(np.sum(exp_z))
    Q = (max_R + np.log(sum_exp)) / beta
    p = exp_z / sum_exp

    grad = dx * (p @ G)

    S0 = np.correlate(h, 1.0-h, mode='full')
    T = np.correlate(d, 1.0-h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasibility bounds
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -t_radius)
    t_hi = min(t_hi, t_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    
    # Golden section search on [t_lo, t_hi]
    g_ratio = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g_ratio * (b - a)
    c2 = a + g_ratio * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    for _ in range(80):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g_ratio * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g_ratio * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C)) * dx
    return opt_t, opt_val

def _dg_lse_solve(h_init, beta, maxiter=300):
    """SLSQP optimization of LSE objective."""
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    
    def fun(x):
        x_clip = np.clip(x, 0.0, 1.0)
        Q, grad = _lse_obj_grad(x_clip, beta, dx)
        return Q, grad
    
    def eq_fun(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    x0 = h_init.copy()
    try:
        res = minimize(fun, x0, method='SLSQP', jac=True, bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = _project(res.x, target)
        return h_new
    except Exception:
        return h_init.copy()

def _dg_minimax_solve(h_init, maxiter=200):
    """Direct minimax SLSQP: minimize M s.t. M >= C_r(h) for all r."""
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    m = 2 * n - 1
    M0 = _compute_c5(h_init) + 0.01
    
    def obj(y):
        return y[0], np.array([1.0] + [0.0]*n)
    
    def ineq(y):
        h = np.clip(y[1:], 0.0, 1.0)
        M = y[0]
        R = np.correlate(h, 1.0-h, mode='full') * dx
        return M - R  # want >= 0
    
    def ineq_jac(y):
        h = np.clip(y[1:], 0.0, 1.0)
        G = _grad_matrix(h)  # unscaled, (m, n)
        J = np.zeros((m, n+1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * G
        return J
    
    def eq_fun(y):
        return np.sum(y[1:]) - target
    
    def eq_jac(y):
        return np.array([0.0] + [1.0]*n)
    
    bounds = [(0.0, M0 * 2)] + [(0.0, 1.0)] * n
    cons = [{'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    y0 = np.zeros(n+1)
    y0[0] = M0
    y0[1:] = h_init
    try:
        res = minimize(obj, y0, method='SLSQP', jac=True, bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = _project(res.x[1:], target)
        return h_new
    except Exception:
        return h_init.copy()

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init(n, period, duty):
    h = np.zeros(n)
    pattern = np.zeros(period)
    fill = max(1, int(duty * period))
    pattern[:fill] = 1.0
    reps = n // period
    h[:reps * period] = np.tile(pattern, reps)
    h[reps*period:] = pattern[:n - reps*period]
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _line_polish(h, deadline, max_iters=5000):
    """Exact line search polish with diverse directions."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    rng = np.random.default_rng(12345)
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        scores = _corr_values(h)
        max_score = float(scores.max())
        
        # Direction 1: negative gradient of top active constraints
        thr_frac = 0.05
        active = np.where(scores >= max_score * (1 - thr_frac))[0]
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        
        G = _grad_matrix(h)
        # Weighted average gradient of active constraints
        w = scores[active] - scores[active].min() + 1e-10
        w = w / w.sum()
        avg_grad = (w @ G[active]) * dx
        avg_grad -= avg_grad.mean()
        
        d = -avg_grad
        # Scale to unit
        norm = np.linalg.norm(d)
        if norm > 1e-14:
            d /= norm
        
        # Feasibility scaling
        pos = d > 1e-15
        neg = d < -1e-15
        t_max = 1.0
        if np.any(pos):
            t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_max = min(t_max, float(np.max(h[neg] / (-d[neg]))))
        
        if t_max > 1e-10:
            d_scaled = d * t_max * 0.5
            alpha, new_c5 = _line_search_exact(h, d_scaled, dx, t_max)
            if new_c5 < best_c5 - 1e-14:
                h = h + alpha * d_scaled
                h = _project(h, target)
                best_c5 = _compute_c5(h)
                best_h = h.copy()
                stall = 0
                continue
        
        # Direction 2: Random balanced direction
        n_rand = rng.integers(4, min(20, n//3))
        idx = rng.choice(n, size=n_rand, replace=False)
        d = np.zeros(n)
        signs = rng.choice([-1.0, 1.0], size=n_rand)
        vals = rng.uniform(0.1, 1.0, n_rand)
        d[idx] = signs * vals
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm > 1e-14:
            d /= norm
            pos = d > 1e-15
            neg = d < -1e-15
            t_max = 1.0
            if np.any(pos):
                t_max = min(t_max, float(np.min((1.0 - h[pos]) / d[pos])))
            if np.any(neg):
                t_max = min(t_max, float(np.max(h[neg] / (-d[neg]))))
            if t_max > 1e-10:
                d_scaled = d * t_max * 0.3
                alpha, new_c5 = _line_search_exact(h, d_scaled, dx, t_max)
                if new_c5 < best_c5 - 1e-14:
                    h = h + alpha * d_scaled
                    h = _project(h, target)
                    best_c5 = _compute_c5(h)
                    best_h = h.copy()
                    stall = 0
                    continue
        
        # Direction 3: Pair move
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) >= 2:
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i != j:
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, new_c5 = _line_search_exact(h, d, dx, 0.5)
                if new_c5 < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    best_c5 = _compute_c5(h)
                    best_h = h.copy()
                    stall = 0
                    continue
        
        stall += 1
    
    return best_h, best_c5

def _active_constraint_direction(h, scores, radius, top_k=10):
    """LP-free direction: project negative weighted gradient of top-k active constraints."""
    n = len(h)
    sorted_idx = np.argsort(-scores)[:top_k]
    G = _grad_matrix(h)
    # Use softmax weights over top-k
    w = scores[sorted_idx] - scores[sorted_idx].min() + 1e-10
    w = np.exp(w * 5) if w.max() > w.min() + 1e-10 else np.ones_like(w)
    w /= w.sum()
    d = -(w @ G[sorted_idx])
    d -= d.mean()  # ensure zero sum
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    d = d / norm * radius
    # Feasibility clip
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        max_step = np.min(1.0 - h[pos]) / d[pos]
        d[pos] *= max(0.0, min(1.0, max_step))
    if np.any(neg):
        max_step = np.min(h[neg]) / (-d[neg])
        d[neg] *= max(0.0, min(1.0, max_step))
    return d
