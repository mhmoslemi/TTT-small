import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):

    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.03 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 40:
        combined = sorted_idx[:40]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _fourier_restart(h, n, rng, freq_min=1, freq_max=6):

def _resample(h, n_new):

    if n_new == n:

def _block_init(n, rng, n_blocks):

def _targeted_descent(h, deadline, max_iter=200):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        S = _corr_scaled(h)
        M = S.max()
        worst_idx = np.where(S >= M - 0.01 * M)[0]
        top_k = min(20, 2 * n - 1)
        G = _grad_rows(h, worst_idx, n)
        # Find direction that reduces worst correlations
        # Minimize sum of S[k] for k in worst_idx, subject to sum(d)=0, -h<=d<=1-h
        c_vec = G.sum(axis=0)
        c_vec -= c_vec.mean()
        # Try moving in -c_vec direction
        d = -c_vec
        d -= d.mean()
        if np.abs(d).max() < 1e-10:
            stall += 1
            continue
        # Scale to fit within bounds
        pos = d > 1e-10
        neg = d < -1e-10
        scale = 1.0
        if np.any(pos):
            scale = min(scale, 0.5 * np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            scale = min(scale, 0.5 * np.max(h[neg] / (-d[neg])))
        if scale < 1e-10:
            stall += 1
            continue
        d = d * min(scale, 0.15)
        alpha, new_c5 = _line_search(h, d, dx, 0.2)
        if new_c5 < best_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=150):

        improved = False
        for ii in range(len(candidates)):

                break

            for jj in range(ii + 1, min(ii + 8, len(candidates))):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0

                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:

                        best_h = h.copy()
                        improved = True

        starts.append(_block_init(n0, rng, nb))
    time_per_start = max(0.05, (deadline - time.time()) / max(len(starts), 1))
    for s in starts:
        if time.time() > deadline:
            break
        sub_dl = min(deadline, time.time() + time_per_start)
        h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.12, K=15, max_stall=20)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (8%)
    d1 = t0 + budget_s * 0.08
    if time.time() < d1:
        h1, c5_1 = _multi_start_lp(best_h, n0, d1, rng)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Aggressive main LP (25%)
    d2 = t0 + budget_s * 0.33
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.18, K=25, use_smooth=False, max_stall=60)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Smooth LP (12%)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.10, K=30, use_smooth=True, max_stall=50, tau_init=0.01 * best_c5)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Targeted descent (8%)
    d4 = t0 + budget_s * 0.53
    h4, c5_4 = _targeted_descent(best_h, d4, max_iter=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Binary SA + LP polish (15%)
    d5 = t0 + budget_s * 0.68
    h5a, c5_5a = _binary_sa_fast(best_h, t0 + budget_s * 0.60, rng, T_init=0.004, T_min=0.000002)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5:
        h5b, c5_5b = _lp_loop(best_h, d5, radius_init=0.08, K=20, max_stall=30)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Multi-resolution (17%)
    d6 = t0 + budget_s * 0.85
    base_n = len(best_h)
    offsets = list(range(-24, 25, 2))
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 280 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d6, time.time() + budget_s * 0.018)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.12, K=20, max_stall=25)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.015:
                sub_dl2 = min(d6, time.time() + budget_s * 0.012)
                h_r2, c5_r2 = _binary_sa_fast(h_r, sub_dl2, rng, T_init=0.003, T_min=0.000003)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d6:
                    h_r, c5_r = _lp_loop(h_r, min(d6, time.time() + budget_s * 0.008),
                                         radius_init=0.06, K=15, max_stall=20)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Fourier restarts + LP (6%)
    d7 = t0 + budget_s * 0.91
    for attempt in range(50):
        if time.time() > d7:
            break
        if best_c5 < 0.3800:
            break
        h_pert = _fourier_restart(best_h, len(best_h), rng, freq_min=1, freq_max=8)
        sub_dl = min(d7, time.time() + budget_s * 0.01)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.10, K=20, max_stall=20)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 8: Pair refine + final LP (5%)
    d8 = t0 + budget_s * 0.96
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=80)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final LP polish
    if time.time() < deadline:
        h_f, c5_f = _lp_loop(best_h, deadline, radius_init=0.05, K=20, max_stall=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
