            h_new = project(h + t_opt * d, target)
            new_c5 = c5_exact(h_new)
            if new_c5 < best_c5 - 1e-14:
                h = h_new
                R = compute_all_rows(h)
                Rmax = float(np.max(R))
                thresh = Rmax - 0.02 * Rmax
                active = np.where(R >= thresh)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(R)])
                G = compute_gradients(h)
                G_active = G[active]
                cur_c5 = new_c5
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            scale *= 0.7
            if scale < 0.02:
                scale = 0.3
                stall = max(stall - 10, 0)
    return best_h, best_c5

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _binary_optimize(h, deadline, rng, max_iters=20000):
    n = len(h)
    target = n // 2
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    cur_c5 = c5_exact(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    patience = 0
    T = 0.002
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if patience > 8000:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if h_bin[i] == h_bin[j]:
            continue
        if h_bin[i] == 1.0:
            old_i, new_i = 1.0, 0.0
            old_j, new_j = 0.0, 1.0
        else:
            old_i, new_i = 0.0, 1.0
            old_j, new_j = 1.0, 0.0
        h_bin[i] = new_i
        h_bin[j] = new_j
        new_c5 = c5_exact(h_bin)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-5 and rng.random() < np.exp(-delta / (T + 1e-15))):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
                patience = 0
            else:
                patience += 1
        else:
            h_bin[i] = old_i
            h_bin[j] = old_j
            patience += 1
        T = max(1e-5, T * 0.9999)
    return best_h, best_c5

def _perturbation_restart(h, n_pert, rng, deadline, base_eps=0.01):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    for _ in range(n_pert):
        if time.time() > deadline:
            break
        h_pert = h.copy()
        n_move = rng.integers(3, min(10, n // 4) + 1)
        idx = rng.integers(n, size=n_move)
        for k in idx:
            h_pert[k] += rng.uniform(-0.3, 0.3)
        h_pert = project(h_pert, n / 2.0)
        c5_pert = c5_exact(h_pert)
        dl = time.time() + (deadline - time.time()) / (n_pert - _)
        h_res, c5_res = _active_set_descent(h_pert, min(dl, deadline), base_eps, max_iters=200)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    return best_h, best_c5

def _fractional_optimize(h, deadline, n_frac=8, max_iters=1000):
    n = len(h)
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R = compute_all_rows(h)
    Rmax = float(np.max(R))
    G = compute_gradients(h)
    stall = 0
    rng = np.random.default_rng(999)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        thresh = Rmax - 0.015 * Rmax
        active = np.where(R >= thresh)[0]
        if len(active) < 2:
            active = np.where(R >= Rmax - 0.05 * Rmax)[0]
        if len(active) < 2:
            break
        p = np.exp((R[active] - Rmax) / (0.003 * Rmax + 1e-15))
        p /= p.sum()
        g = G[active].T @ p
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        pos_idx = np.argsort(-g)
        neg_idx = np.argsort(g)
        n_p = min(n_frac, n // 3)
        pos_sel = pos_idx[:n_p]
        neg_sel = neg_idx[:n_p]
        alpha_pos = np.maximum(g[pos_sel], 0.0)
        alpha_neg = np.maximum(-g[neg_sel], 0.0)
        sp = alpha_pos.sum()
        sn = alpha_neg.sum()
        if sp < 1e-10 or sn < 1e-10:
            stall += 1
            continue
        d = np.zeros(n)
        d[pos_sel] = alpha_pos / sp
        d[neg_sel] = -alpha_neg / sn
        d_scaled = d / np.linalg.norm(d)
        ms = _max_step(h, d_scaled)
        if ms < 1e-13:
            stall += 1
            continue
        t_opt = _line_search(h, d_scaled, R, min(ms, 0.5))
        if t_opt > 1e-14:
            h_new = project(h + t_opt * d_scaled, target)
            new_c5 = c5_exact(h_new)
            if new_c5 < best_c5 - 1e-14:
                h = h_new
                R = compute_all_rows(h)
                Rmax = float(np.max(R))
                G = compute_gradients(h)
                cur_c5 = new_c5
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 50:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.4 * np.sin(2 * np.pi * 2.0 * x / n0) + 0.1 * np.sin(2 * np.pi * 5.0 * x / n0 + 1.0)
        h0 = project(h0, n0 / 2.0)

    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0

    d1 = t0 + budget_s * 0.25
    h, c5 = _active_set_descent(h, d1, eps_frac=0.01, max_iters=2000)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n0

    d2 = t0 + budget_s * 0.38
    h, c5 = _soft_max_descent(h, d2, iters=400)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n0

    d3 = t0 + budget_s * 0.50
    h, c5 = _multi_group_descent(h, d3, n_groups=4, max_iters=600)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n0

    d4 = t0 + budget_s * 0.62
    h, c5 = _fractional_optimize(h, d4, n_frac=6, max_iters=800)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n0

    # Multi-resolution
    d5 = t0 + budget_s * 0.78
    tried_n = {best_n}
    for dn in [-6, 6, -4, 4, -8, 8, -2, 2, -10, 10]:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 30 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample_h(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.018)
        h_r, c5_r = _active_set_descent(h_r, sub_dl, eps_frac=0.01, max_iters=300)
        sub_dl2 = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _soft_max_descent(h_r, sub_dl2, iters=100)
        if c5_r < best_c5:
            best_c5, best_h, best_n = c5_r, h_r.copy(), n_try

    # Perturbation restarts
    d6 = t0 + budget_s * 0.87
    h, c5 = _perturbation_restart(best_h, 6, rng, d6, base_eps=0.01)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), best_n

    # Final polish
    d7 = deadline
    h, c5 = _active_set_descent(best_h, d7, eps_frac=0.005, max_iters=1000)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    h, c5 = _soft_max_descent(best_h, deadline, iters=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Binary check
    if time.time() < deadline - 30:
        h_bin, c5_bin = _binary_optimize(best_h, deadline, rng, max_iters=15000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = len(h_bin)

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_h = project(best_h, len(best_h) / 2.0)
        final_c5 = c5_exact(best_h)

    return best_h, float(final_c5), int(len(best_h))
