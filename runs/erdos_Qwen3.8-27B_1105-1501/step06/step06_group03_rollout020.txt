def _lp_direction(h, S, n, K, radius):
    """LP-based direction finding (improved from previous)."""
    dx = 2.0 / n
    M = float(S.max())
    sorted_idx = np.argsort(-S)

    tol = max(0.03 * M, 1e-5)

        combined = sorted_idx[:30]

    G = np.zeros((len(combined), n))
    for idx, r in enumerate(combined):
        k = r - (n - 1)
        if k >= 0:
            m = n - k
            if m > 0:
                G[idx, :m] += (1.0 - h[k:k+m]) * dx
                G[idx, k:] += -h[:m] * dx
        else:
            d_ = -k
            m = n - d_
            if m > 0:
                G[idx, d_:] += (1.0 - h[:m]) * dx
                G[idx, :m] += -h[d_:d_+m] * dx
    
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = (M - S[combined]) + 1e-10
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-9)
        hi_b = min(radius, 1.0 - h[j] - 1e-9)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _line_search_exact(h, d, dx, trust_radius):
    """Exact line search using quadratic model of C5."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    def f_of_t(t):
        vals = S0 + t * T - t * t * C
        return float(np.max(vals) * dx)
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_of_t(c1)
    f2 = f_of_t(c2)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f_of_t(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_of_t(c2)
    
    opt_t = (a + b) / 2.0
    opt_val = f_of_t(opt_t)
    for tt in [t_lo, t_hi]:
        vv = f_of_t(tt)
        if vv < opt_val:
            opt_t, opt_val = tt, vv
    return opt_t, opt_val

    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -trust_radius
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    def f_of_t(t):
        vals = S0 + t * T - t * t * C
        return float(np.max(vals) * dx)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_of_t(c1)
    f2 = f_of_t(c2)
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f_of_t(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_of_t(c2)
    
    opt_t = (a + b) / 2.0
    opt_val = f_of_t(opt_t)
    for tt in [t_lo, t_hi]:
        vv = f_of_t(tt)
        if vv < opt_val:
            opt_t, opt_val = tt, vv
    return opt_t, opt_val

def _lp_direction(h, G, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.02 * M, 1e-5)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G[combined]
    A_ub[:, n] = 1.0
    b_ub = (M - S[combined]) + 1e-10
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-9)
        hi_b = min(radius, 1.0 - h[j] - 1e-9)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, K=18):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(999)
    
    while time.time() < deadline:
        if stall > 100:
            h = h + rng.normal(0, 0.015, n)
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            radius = 0.08
            stall = 0
            continue
        
        S = _corr_scaled(h)
        G = _grad_matrix(h, n)
        p = _lp_direction(h, G, S, n, K, radius)
        
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if radius < 1e-5:
                break
            continue
        
        alpha, new_c5 = _line_search_exact(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.06, 0.25)
            else:
                radius = max(radius * 0.95, 0.02)
        else:
            stall += 1
            radius *= 0.75
    
    return best_h, best_c5
