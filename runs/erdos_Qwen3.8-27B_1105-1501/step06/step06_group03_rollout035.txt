A KKT-like condition would be:

- The gradient of sum_r λ_r Q_r(h) is orthogonal to the feasible directions (i.e., proportional to the constraint gradient, which is just the all-ones vector for the sum constraint)
- So: sum_r λ_r ∇Q_r(h) = μ * 1 (for some μ)

2. Find weights λ_r such that sum_r λ_r ∇Q_r is as close to constant as possible
3. The direction is then -sum_r λ_r ∇Q_r + mean(sum_r λ_r ∇Q_r) * 1 (to make it sum-zero)
4. This is like a weighted subgradient descent

def _line_search(h, p, dx, trust_radius):

def _optimal_subgrad_direction(h, S, n, active_indices):
    G = _grad_rows(h, active_indices, n)
    G_c = G - G.mean(axis=1, keepdims=True)
    m = len(active_indices)
    Q = G_c.T @ G_c
    x0 = np.ones(m) / m

    def obj_fn(lam):
        return float(lam @ Q @ lam)

    def grad_fn(lam):
        return 2.0 * Q @ lam

    try:
        cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac}]
        res = scipy_minimize(obj_fn, x0, jac=grad_fn, method='SLSQP',
                             bounds=[(0.0, 1.0)] * m, constraints=cons,
                             options={'maxiter': 50, 'ftol': 1e-12})
        lam = np.maximum(res.x, 0.0)
        lam_sum = lam.sum()
        if lam_sum < 1e-15:
            lam = np.ones(m) / m
        else:
            lam = lam / lam_sum
    except Exception:
        lam = np.ones(m) / m
    g = G_c.T @ lam
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm

def _sum_jac(x):

    w = w / w_sum
    top_k = min(30, 2 * n - 1)
    top_idx = np.argsort(-S)[:top_k]
    G = _grad_rows(h, top_idx, n)
    grad_smooth = w[top_idx] @ G
    grad_smooth -= grad_smooth.mean()
    norm = np.linalg.norm(grad_smooth)
    if norm < 1e-12:
        return None
    return -grad_smooth / norm


def _random_direction(n, rng):
    d = rng.standard_normal(n)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm


def _local_optimize(h0, deadline, radius_init=0.12, K=25, use_osg=True, use_smooth=True):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restarts = 0
    rng = np.random.default_rng(12345)
    tau = 0.015 * max(cur_c5, 0.1)

    while time.time() < deadline:
        if stall > 50:
            if restarts < 8:
                scale = 0.008 + 0.005 * restarts
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.06
                stall = 0
                restarts += 1
                continue
            break

        S = _exact_scores(h)
        M = float(S.max())
        directions = []

        p_lp = _lp_direction(h, S, n, K, radius)
        if p_lp is not None and np.max(np.abs(p_lp)) > 1e-12:
            directions.append(p_lp)

        tol = max(0.02 * M, 1e-9)
        active_set = np.where(S >= M - tol)[0]
        if len(active_set) > 35:
            active_set = np.argsort(-S)[:35]
        if use_osg and len(active_set) >= 3:
            p_osg = _optimal_subgrad_direction(h, S, n, active_set)
            if p_osg is not None:
                directions.append(p_osg)

        if use_smooth:
            p_sm = _soft_direction(h, S, n, tau)
            if p_sm is not None:
                directions.append(p_sm)

        directions.append(_random_direction(n, rng))

        best_step = None
        best_new_c5 = cur_c5
        for p in directions:
            if p is None:
                continue
            p = p.copy()
            p -= p.mean()
            p_norm = np.linalg.norm(p)
            if p_norm < 1e-14:
                continue
            p = p / p_norm * radius
            alpha, new_c5 = _line_search(h, p, dx, radius * 1.5)
            if new_c5 < best_new_c5 - 1e-14:
                best_new_c5 = new_c5
                best_step = (alpha, p)

        if best_step is not None:
            alpha, p = best_step
            h_new = h + alpha * p
            h_new = _project(h_new, target)
            new_c5 = _exact_c5(h_new)
            if new_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                if alpha > 0.7 * radius:
                    radius = min(radius * 1.08, 0.30)
                elif alpha > 0.3 * radius:
                    radius = min(radius * 1.02, 0.25)
                else:
                    radius = max(radius * 0.95, 0.015)
                tau *= 0.85
                tau = max(tau, 1e-6)
            else:
                stall += 1
                radius *= 0.8
        else:
            stall += 1
            radius *= 0.75
            if stall % 10 == 0 and radius < 0.01:
                radius = 0.04
            if radius < 1e-5:
                break

    return best_h, best_c5


def _de_population(h_best, n, deadline, rng):
    target = n / 2.0
    pop_size = 18
    F = 0.6
    CR = 0.7
    pop = [h_best.copy()]
    for i in range(1, pop_size):
        h_new = h_best + rng.normal(0, 0.03 + 0.01 * i / pop_size, n)
        h_new = _project(h_new, target)
        pop.append(h_new)
    scores = [_exact_c5(h) for h in pop]
    global_best_h = h_best.copy()
    global_best_c5 = _exact_c5(h_best)

    gen = 0
    while time.time() < deadline:
        gen += 1
        if gen % 10 == 0:
            pop[0] = global_best_h.copy()
            scores[0] = global_best_c5
        for i in range(pop_size):
            if time.time() > deadline:
                break
            candidates = list(range(pop_size))
            candidates.remove(i)
            a, b, c = rng.choice(candidates, size=3, replace=False)
            v = pop[a].copy() + F * (pop[b] - pop[c])
            v = _project(v, target)
            mask = rng.random(n) < CR
            trial = pop[i].copy()
            trial[mask] = v[mask]
            trial = _project(trial, target)
            trial_c5 = _exact_c5(trial)
            if trial_c5 <= scores[i] + 1e-12:
                pop[i] = trial
                scores[i] = trial_c5
                if trial_c5 < global_best_c5 - 1e-13:
                    global_best_c5 = trial_c5
                    global_best_h = trial.copy()
    return global_best_h, global_best_c5


def _binary_sa(h_init, deadline, rng, max_iter=50000):
    n = len(h_init)
    target = n // 2
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n, dtype=float)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.00001
    cooldown = 0.99995
    patience = 0
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h <= 0.5)[0])
    count = 0
    while time.time() < deadline and count < max_iter:
        count += 1
        if patience > 8000:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.9)
            patience = 0
        if T < T_min:
            break
        batch = min(500, max_iter - count)
        for _ in range(batch):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            accept = delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T))
            if accept:
                cur_c5 = new_c5
                ones_list = list(np.where(h > 0.5)[0])
                zeros_list = list(np.where(h <= 0.5)[0])
                if cur_c5 < best_c5 - 1e-13:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                patience += 1
        T = max(T_min, T * cooldown)
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _pair_refine(h_init, deadline, max_passes=150):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    improved = True
    passes = 0
    while improved and passes < max_passes and time.time() < deadline:
        improved = False
        passes += 1
        S = _exact_scores(h)
        total_grad = np.zeros(n)
        for r in range(2 * n - 1):
            total_grad += S[r] * _grad_rows(h, np.array([r]), n)[0]
        total_grad -= total_grad.mean()
        n_cand = min(n, 40)
        candidates = np.argsort(-np.abs(total_grad))[:n_cand]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                return best_h, best_c5
            i = candidates[ii]
            if abs(h[i] - 0.5) < 0.3:
                continue
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                if abs(h[i] - 0.5) < 0.3 or abs(h[j] - 0.5) < 0.3:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.35)
                if c5_try < best_c5 - 1e-13:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
    return best_h, best_c5


def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)


def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks < 2:
        h[:] = 0.5
        return _project(h, n / 2.0)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_deadline = t0 + budget_s * 0.99

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)
    best_n = n0

    # Phase 1: Multi-start quick local opt (8%)
    d1 = t0 + budget_s * 0.08
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5]:
        for ph in [0.0, 1.2, 2.5]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(6):
        starts.append(_block_init(n0, rng, int(rng.integers(3, 12))))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.006)
        h_s, c5_s = _local_optimize(s, sub_dl, radius_init=0.08, K=20)
        if c5_s < best_c5 - 1e-13:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n0

    # Phase 2: Main local optimization (30%)
    d2 = t0 + budget_s * 0.38
    h2, c5_2 = _local_optimize(best_h, d2, radius_init=0.15, K=30, use_osg=True, use_smooth=True)
    if c5_2 < best_c5 - 1e-13:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: DE population (22%)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _de_population(best_h, best_n, d3, rng)
    if c5_3 < best_c5 - 1e-13:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + LP polish (15%)
    d4 = t0 + budget_s * 0.75
    rng_sa = np.random.default_rng(seed + 77)
    h4a, c5_4a = _binary_sa(best_h, min(d4, t0 + budget_s * 0.70), rng_sa, max_iter=30000)
    if c5_4a < best_c5 - 1e-13:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _local_optimize(best_h, d4, radius_init=0.06, K=25)
        if c5_4b < best_c5 - 1e-13:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Resolution sweep (15%)
    d5 = t0 + budget_s * 0.90
    base_n = best_n
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = base_n + dn
        if n_try < 44 or n_try > 240 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d5, time.time() + budget_s * 0.025)
            h_r, c5_r = _local_optimize(h_r, sub_dl, radius_init=0.10, K=20)
            if c5_r < best_c5 - 1e-13:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try
    # Try much smaller n
    for n_small in [62, 66, 70, 74, 78, 82, 86, 90]:
        if time.time() > d5:
            break
        if n_small % 2 != 0:
            continue
        h_r = _resample(best_h, n_small)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.01:
            sub_dl = min(d5, time.time() + budget_s * 0.015)
            h_r, c5_r = _local_optimize(h_r, sub_dl, radius_init=0.12, K=15)
            if c5_r < best_c5 - 1e-13:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_small

    # Phase 6: Pair refinement (8%)
    d6 = t0 + budget_s * 0.96
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=200)
    if c5_6 < best_c5 - 1e-13:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final polish (4%)
    if time.time() < hard_deadline:
        h7, c5_7 = _local_optimize(best_h, hard_deadline, radius_init=0.04, K=20)
        if c5_7 < best_c5 - 1e-13:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
