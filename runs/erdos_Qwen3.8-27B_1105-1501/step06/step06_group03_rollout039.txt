        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d + m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, target, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.03 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 40:
        combined = sorted_idx[:40]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.max(np.abs(d)) > 1e-14:
                return d
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, K=25, use_lse=False, lse_mu=None):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(12345)
    mu = lse_mu if lse_mu is not None else 0.02 * cur_c5
    while time.time() < deadline:
        if stall > 80:
            if stall < 200:
                scale = 0.008 + 0.005 * min(stall - 80, 20) * 0.3
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.06
                stall = 0
                mu = 0.02 * cur_c5
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if (p is None or np.max(np.abs(p)) < 1e-13) and use_lse:
            g = _lse_grad(h, mu)
            if np.linalg.norm(g) > 1e-14:
                p = -g / np.linalg.norm(g) * radius
        if p is None or np.max(np.abs(p)) < 1e-13:
            if use_lse:
                h_try, f_try = _lse_backtrack(h, _lse_grad(h, mu), mu, target)
                if h_try is not None and f_try < cur_c5 - 1e-12:
                    h = h_try
                    cur_c5 = f_try
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    stall = 0
                    mu *= 0.95
                    continue
            stall += 1
            radius *= 0.75
            if use_lse:
                mu *= 0.9
            if radius < 1e-7:
                break
            continue
        alpha, new_c5 = _line_search(h, p, dx, target, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.6:
                radius = min(radius * 1.05, 0.3)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.9, 0.02)
        else:
            stall += 1
            radius *= 0.78
    return best_h, best_c5

def _block_cd(h_init, deadline, block_size=5, n_rounds=500):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    round_count = 0
    while time.time() < deadline and round_count < n_rounds:
        round_count += 1
        start = rng.integers(0, max(1, n - block_size + 1))
        end = min(start + block_size, n)
        h_block = h[start:end].copy()
        h_before_start = h[:start].copy()
        h_after_end = h[end:].copy()
        target_block = float(np.sum(h_block))

        def obj_block(x):
            x = np.clip(x, 0.0, 1.0)
            x = _project(x, target_block)
            h_full = np.concatenate([h_before_start, x, h_after_end])
            return _exact_c5(h_full)

        res = minimize(obj_block, h_block, method='Nelder-Mead',
                       options={'maxiter': 100, 'xatol': 1e-7, 'fatol': 1e-9,
                                'adaptive': True})
        if res.fun < best_c5 - 1e-12:
            h_new_block = np.clip(res.x, 0.0, 1.0)
            h_new_block = _project(h_new_block, target_block)
            h[start:end] = h_new_block
            h = _project(h, target)
            c5_new = _exact_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def _binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005
    T_min = 0.000003
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        for _ in range(300):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < -1e-14 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * 0.99995)
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _subspace_optimize(h, deadline, n_sub=12, n_iters=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(7777)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        idx = rng.choice(n, size=n_sub, replace=False)
        h_sub = h[idx].copy()
        h_masked = h.copy()
        target_sub = float(np.sum(h_sub))

        def obj_sub(x):
            x = np.clip(x, 0.0, 1.0)
            x = _project(x, target_sub)
            h_masked[idx] = x
            return _exact_c5(h_masked)

        res = minimize(obj_sub, h_sub, method='Nelder-Mead',
                       options={'maxiter': 80, 'xatol': 1e-7, 'fatol': 1e-9})
        if res.fun < best_c5 - 1e-12:
            h_sub_new = np.clip(res.x, 0.0, 1.0)
            h_sub_new = _project(h_sub_new, target_sub)
            h[idx] = h_sub_new
            h = _project(h, target)
            c5_new = _exact_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 40)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, min(ii + 15, len(candidates))):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, target, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(n_blocks - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.99

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: LP with larger K (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.15, K=30, use_lse=False)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: LP with LSE fallback (15% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.10, K=40, use_lse=True,
                        lse_mu=0.015)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Block coordinate descent (15% time)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _block_cd(best_h, d3, block_size=5, n_rounds=800)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < d3:
        h3b, c5_3b = _block_cd(best_h, d3, block_size=8, n_rounds=400)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Binary SA + LP polish (12% time)
    d4 = t0 + budget_s * 0.67
    h4a, c5_4a = _binary_sa(best_h, t0 + budget_s * 0.62, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = _lp_loop(best_h, d4, radius_init=0.08, K=30, use_lse=False)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()

    # Phase 5: Multi-resolution (15% time)
    d5 = t0 + budget_s * 0.82
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -12, 12, -16, 16, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d5, time.time() + budget_s * 0.025)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=25, use_lse=False)
            if c5_r < best_c5 and time.time() < d5 - budget_s * 0.015:
                sub_dl2 = min(d5, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d5:
                    h_r, c5_r = _lp_loop(h_r, min(d5, time.time() + budget_s * 0.012),
                                         radius_init=0.05, K=20)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Subspace optimization (7% time)
    d6 = t0 + budget_s * 0.89
    h6, c5_6 = _subspace_optimize(best_h, d6, n_sub=14, n_iters=300)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Pair refinement + final LP (7% time)
    d7 = t0 + budget_s * 0.96
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=60)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < deadline:
        hf, c5_f = _lp_loop(best_h, deadline, radius_init=0.04, K=25, use_lse=True,
                            lse_mu=0.005)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
