def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-14:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _row_grad(h, corr_idx, n):
    g = np.zeros(n)
    dx = 2.0 / n
    k = corr_idx - (n - 1)
    if k >= 0:
        j = np.arange(0, n - k)
        if len(j) > 0:
            g[j + k] += dx * (1.0 - h[j])
            g[j] -= dx * h[j + k]
    else:
        q = -k
        j = np.arange(0, n - q)
        if len(j) > 0:
            g[j] += dx * (1.0 - h[j + q])
            g[j + q] -= dx * h[j]
    return g


def _overlap_single(h, corr_idx, n):
    """Compute a single overlap value (for SLSQP)."""
    dx = 2.0 / n
    k = corr_idx - (n - 1)
    if k >= 0:
        return dx * float(np.sum(h[k:] * (1.0 - h[:n - k])))
    else:
        q = -k
        return dx * float(np.sum(h[:n - q] * (1.0 - h[q:])))

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    total_lags = 2 * n - 1

    max_ov = float(np.max(overlaps))

    k_idx = np.argsort(-overlaps)[:top_k]
    
    # Initial point
    x0 = np.concatenate([h0.copy(), [max_ov]])
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    # Objective
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Constraints
    constraints = []
    
    # Equality: sum(h) = target
    def eq_fun(x):
        return float(np.sum(x[:n])) - target
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    # Inequality: z - overlap_k >= 0 for each top-k lag

        def ineq_fun(x, idx=idx):
            h = x[:n]
            z = x[-1]
            return z - _overlap_single(h, idx, n)
        def ineq_jac(x, idx=idx):
            h = x[:n]
            j = np.zeros(n + 1)
            j[:n] = -_row_grad(h, idx, n)
            j[-1] = 1.0
            return j
        constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
    
    try:
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                       constraints=constraints, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.success or (res.x is not None and res.x[-1] < max_ov):
            h_new = _project(res.x[:n], target)
            f_new = _exact_c5(h_new)
            if f_new < max_ov:
                return h_new, f_new
    except Exception:
        pass
    
    return h0.copy(), max_ov

import numpy as np
import time
from scipy.optimize import minimize

# Global state for SLSQP constraint callbacks
_slq_n = 0
_slq_target = 0.0
_slq_k_idx = None


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(12):
        if abs(res) <= 1e-14:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _row_grad(h, corr_idx, n):
    g = np.zeros(n)
    dx = 2.0 / n
    k = corr_idx - (n - 1)
    if k >= 0:
        j = np.arange(0, n - k)
        if len(j) > 0:
            g[j + k] += dx * (1.0 - h[j])
            g[j] -= dx * h[j + k]
    else:
        q = -k
        j = np.arange(0, n - q)
        if len(j) > 0:
            g[j] += dx * (1.0 - h[j + q])
            g[j + q] -= dx * h[j]
    return g


def _overlap_single(h, corr_idx, n):
    dx = 2.0 / n
    k = corr_idx - (n - 1)
    if k >= 0:
        return dx * float(np.sum(h[k:] * (1.0 - h[:n - k])))
    else:
        q = -k
        return dx * float(np.sum(h[:n - q] * (1.0 - h[q:])))


def _slq_obj(x):
    return x[-1]


def _slq_obj_grad(x):
    g = np.zeros(_slq_n + 1)
    g[-1] = 1.0
    return g


def _slq_eq_fun(x):
    return float(np.sum(x[:_slq_n])) - _slq_target


def _slq_eq_jac(x):
    j = np.zeros(_slq_n + 1)
    j[:_slq_n] = 1.0
    return j


def _slq_ineq_fun(x):
    n = _slq_n
    h = x[:n]
    z = x[-1]
    out = np.empty(len(_slq_k_idx))
    for i, idx in enumerate(_slq_k_idx):
        out[i] = z - _overlap_single(h, idx, n)
    return out


def _slq_ineq_jac(x):
    n = _slq_n
    h = x[:n]
    J = np.zeros((len(_slq_k_idx), n + 1))
    for i, idx in enumerate(_slq_k_idx):
        J[i, :n] = -_row_grad(h, idx, n)
        J[i, -1] = 1.0
    return J


def _slsqp_solve(h0, top_k=35, maxiter=25):
    global _slq_n, _slq_target, _slq_k_idx
    n = len(h0)
    _slq_n = n
    _slq_target = n / 2.0
    overlaps = _all_overlaps(h0)
    _slq_k_idx = np.argsort(-overlaps)[:top_k]
    max_ov = float(np.max(overlaps))

    x0 = np.concatenate([h0, [max_ov]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': _slq_eq_fun, 'jac': _slq_eq_jac},
        {'type': 'ineq', 'fun': _slq_ineq_fun, 'jac': _slq_ineq_jac},
    ]
    try:
        res = minimize(_slq_obj, x0, jac=_slq_obj_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = _project(res.x[:n], n / 2.0)
            f_new = _exact_c5(h_new)
            if f_new < max_ov - 1e-12:
                return h_new, f_new
    except Exception:
        pass
    return h0.copy(), max_ov


def _line_search(h, d, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')

    pos = d > 1e-15
    neg = d < -1e-15
    L = 0.0
    U = 1e10
    if np.any(pos):
        U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        U = min(U, float(np.min(-h[neg] / d[neg])))
    if U - L < 1e-14:
        return h.copy(), _exact_c5(h)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = L, U

    def _at(t):
        return float(np.max(corr + t * T - t * t * Q)) * dx

    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = _at(c1), _at(c2)
    for _ in range(55):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = _at(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = _at(c2)

    t = 0.5 * (a + b)
    h_new = h + t * d
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project(h_new, n / 2.0)
    return h_new, _exact_c5(h_new)


def _soft_direction(h, overlaps, n, tau):
    m = float(np.max(overlaps))
    w = np.exp((overlaps - m) / tau)
    sw = float(np.sum(w))
    if sw < 1e-30:
        return None
    w = w / sw
    d = np.zeros(n)
    for idx in range(len(overlaps)):
        if w[idx] > 1e-10:
            d -= w[idx] * _row_grad(h, idx, n)
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm


def _pair_dir(n, rng):
    d = np.zeros(n)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    d[i] = -1.0
    d[j] = 1.0
    return d


def _rand_dir(n, rng):
    d = rng.standard_normal(n)
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm


def _optimize(h, deadline, rng, slsqp_every=50, slsqp_topk=30, slsqp_iter=20):

    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)

        elif r < 0.15:
            d = _pair_dir(n, rng)
        else:
            d = _soft_direction(h, overlaps, n, tau)
        if d is None:
            stall += 1
            continue

        h_new, f_new = _line_search(h, d, n)

        if f_new < best_f - 1e-15:
            best_f = f_new
            best_h = h_new.copy()

        if f_new < cur_f + uphill:
            h = h_new
            cur_f = f_new
            overlaps = _all_overlaps(h)
            stall = 0
            tau = max(0.002, tau * 0.995)
        else:
            h = best_h.copy()
            cur_f = best_f
            overlaps = _all_overlaps(h)
            stall += 1

        if slsqp_every > 0 and stall > 0 and stall % slsqp_every == 0 and time.time() < deadline - 15:
            h_s, f_s = _slsqp_solve(h, top_k=slsqp_topk, maxiter=slsqp_iter)
            if f_s < best_f - 1e-12:

                best_h = h_s.copy()
                h = h_s.copy()
                cur_f = f_s
                overlaps = _all_overlaps(h)
                stall = 0
                sigma = max(0.004, sigma * 0.75)

            cur_f = _exact_c5(h)
            overlaps = _all_overlaps(h)
            stall = 0
            sigma = max(0.002, sigma * 0.82)
            tau = min(0.04, tau * 1.15)

    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    base += rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)
