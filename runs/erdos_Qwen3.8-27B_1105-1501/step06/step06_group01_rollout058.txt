def _build_G_fast(h, dx):
    n = len(h)
    m = 2*n - 1
    G = np.zeros((m, n))
    for k in range(-(n-1), n):
        q = k + n - 1  # index in G
        if k >= 0:
            # S_q = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # dS_q/dh[j] for j in 0..n-1-k: dx*(1-h[j+k])
            # dS_q/dh[j] for j in k..n-1: -dx*h[j-k]
            m_pts = n - k
            G[q, :m_pts] += dx * (1.0 - h[k:])  # first term
            G[q, k:] += -dx * h[:m_pts]           # second term
        else:

            k2 = -k
            m_pts = n - k2
            # S_q = dx * sum_{i=k2}^{n-1} h[i]*(1-h[i-k2])
            G[q, k2:] += dx * (1.0 - h[:m_pts])   # first term
            G[q, :m_pts] += -dx * h[k2:]           # second term

            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

def _binary_from_h(h):
    n = len(h)
    target = int(n // 2)
    sorted_idx = np.argsort(-h)

    h_bin[sorted_idx[:target]] = 1.0
    return h_bin

def _build_G(h, dx):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(-(n - 1), n):
        q = k + n - 1
        if k >= 0:
            m_pts = n - k
            if m_pts > 0:
                G[q, :m_pts] += dx * (1.0 - h[k:])
                G[q, k:] += -dx * h[:m_pts]
        else:
            k2 = -k
            m_pts = n - k2
            if m_pts > 0:
                G[q, k2:] += dx * (1.0 - h[:m_pts])
                G[q, :m_pts] += -dx * h[k2:]
    return G

                break
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i = int(ones[rng.integers(len(ones))])
            j = int(zeros[rng.integers(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta <= 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        t *= decay_per_iter ** 2000
        if t < t_floor:
            t = t_floor
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _random_harmonic_start(n, rng):
    freq = int(rng.integers(2, 12))
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
    h = h + rng.normal(0, 0.08, n)
    h = _project(h, n / 2.0)
    return h

def _random_block_start(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(2, max(3, n // 8))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    return _project(h, n / 2.0)

def _pair_optimize(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 2:
            break
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        G = _build_G(h, dx)
        max_idx = int(np.argmax(S))
        grad = G[max_idx]
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(len(order), 20)):
            if time.time() > deadline:
                break
            i = int(order[ii])
            for jj in range(min(len(order), 20)):
                if tried > 60:
                    break
                j = int(order[jj])
                if i == j:
                    continue
                if h[i] <= 0.01 or h[j] >= 0.99:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                tried += 1
        if not improved:
            for _ in range(15):
                if time.time() > deadline:
                    break
                if len(interior) < 2:
                    break
                ii = int(rng.integers(len(interior)))
                jj = int(rng.integers(len(interior)))
                if ii == jj:
                    continue
                i, j = interior[ii], interior[jj]
                for sign in [1, -1]:
                    d = np.zeros(n)
                    d[i] = -float(sign)
                    d[j] = float(sign)
                    alpha, c5_try = _line_search(h, d, dx, 0.4)
                    if c5_try < best_c5 - 1e-14:
                        h = h + alpha * d
                        h = _project(h, target)
                        cur = _exact_c5(h)
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                            improved = True
            if not improved:
                break
    return best_h, best_c5

def _alternating_optimize(h, deadline, rng):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    sub_dl = deadline - (deadline - time.time()) * 0.5
    while time.time() < sub_dl:
        h_fp, c5_fp = _lp_optimize(h, min(time.time() + 8.0, sub_dl), start_radius=0.10)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()
            h = h_fp.copy()
        h_bin = _binary_from_h(h)
        h_sa, c5_sa = _sa_binary(h_bin, min(5.0, sub_dl - time.time()), rng, t_init=0.02, t_floor=0.002)
        h_soft = h_sa + rng.normal(0, 0.01, n)
        h_soft = _project(h_soft, target)
        c5_soft = _exact_c5(h_soft)
        if c5_soft < best_c5:
            best_c5 = c5_soft
            best_h = h_soft.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / n0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Binary SA (30% budget)
    d1 = t0 + budget_s * 0.30
    h_bin0 = _binary_from_h(h0)
    h_sa, c5_sa = _sa_binary(h_bin0, d1 - time.time(), rng, t_init=0.04, t_floor=0.0005)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 2: LP refinement of best (20% budget)
    d2 = t0 + budget_s * 0.50
    h2 = best_h.astype(float)
    h2, c5_2 = _lp_optimize(h2, d2, start_radius=0.15)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Multi-resolution exploration (25% budget)
    d3 = t0 + budget_s * 0.75
    n_best = len(best_h)
    n_candidates = [n_best - 12, n_best - 8, n_best - 4, n_best, n_best + 4, n_best + 8, n_best + 12, n_best + 16]
    n_candidates = [n for n in n_candidates if n >= 40 and n <= 200 and n % 2 == 0]
    n_candidates = sorted(set(n_candidates))
    per_n = max(3.0, (d3 - time.time()) / max(len(n_candidates), 1))
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try == n_best:
            continue
        h_r = _resample(best_h, n_try)
        sub_sa_dl = min(d3, time.time() + per_n * 0.4)
        h_r_bin = _binary_from_h(h_r)
        h_r_sa, c5_r_sa = _sa_binary(h_r_bin, sub_sa_dl - time.time(), rng, t_init=0.03, t_floor=0.002)
        if c5_r_sa < best_c5:
            best_c5 = c5_r_sa
            best_h = h_r_sa.copy()
            n_best = n_try
        sub_lp_dl = min(d3, time.time() + per_n * 0.5)
        h_r_soft = h_r_bin.astype(float) + rng.normal(0, 0.01, n_try)
        h_r_soft = _project(h_r_soft, n_try / 2.0)
        h_r_lp, c5_r_lp = _lp_optimize(h_r_soft, sub_lp_dl, start_radius=0.12)
        if c5_r_lp < best_c5:
            best_c5 = c5_r_lp
            best_h = h_r_lp.copy()
            n_best = n_try

    # Phase 4: Diverse starts with LP (10% budget)
    d4 = t0 + budget_s * 0.85
    n_starts = 4
    per_start = max(2.0, (d4 - time.time()) / n_starts)
    for si in range(n_starts):
        if time.time() > d4:
            break
        if si % 2 == 0:
            h_s = _random_harmonic_start(n_best, rng)
        else:
            h_s = _random_block_start(n_best, rng)
        sub_dl = min(d4, time.time() + per_start)
        h_s, c5_s = _lp_optimize(h_s, sub_dl, start_radius=0.12)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Final refinement (15% budget)
    d5 = deadline
    h5, c5_5 = _lp_optimize(best_h, d5, start_radius=0.08)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Pair optimization
    if time.time() < deadline:
        sub_dl = min(deadline, time.time() + (deadline - time.time()) * 0.3)
        h6, c5_6 = _pair_optimize(best_h, sub_dl, max_passes=40)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final LP polish
    if time.time() < deadline:
        h7, c5_7 = _lp_optimize(best_h, deadline, start_radius=0.05)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
