def _line_search(S0, T, Q, dx, L, U):

        return t, float(np.max(S0 + t * T - t * t * Q)) * dx

def _row_grad(h, k):

    g = np.zeros(n)
    if k >= 0:
        nn = np.arange(0, n - k)
    else:
        nn = np.arange(-k, n)
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g


def _active_direction(h, corr, k_active, tau, rng):
    n = len(h)
    scores = corr
    mval = float(np.max(scores))
    w = np.exp((scores - mval) / tau)
    idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = r - (n - 1)
        G += w[r] * _row_grad(h, k)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm

    below = h < 0.5
    above = h > 0.5
    d[below] = -h[below]
    d[above] = 1.0 - h[above]
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return np.zeros(n)
    return d / nrm


def _optimize(h, deadline, rng, pair_frac, rand_frac, k_active, tau,
              use_slsqp=False, slsqp_every=30):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    restarts = 0
    sigma = 0.04
    uphill = 5e-5
    it_count = 0
    while time.time() < deadline:
        it_count += 1
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        elif r < rand_frac + pair_frac + 0.05:
            d = _binary_pressure(h)
        else:
            d = _active_direction(h, corr, k_active, tau, rng)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        pos = d > 1e-15
        neg = d < -1e-15
        low = [0.0]
        up = [1e30]
        if np.any(pos):
            low.append(float(np.min(-h[pos] / d[pos])))
            up.append(float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            low.append(float(np.min((1.0 - h[neg]) / d[neg])))
            up.append(float(np.min(-h[neg] / d[neg])))
        L = max(low)
        U = min(up)
        if U - L < 1e-12:
            stall += 1
            continue
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, dx, L, U)
        if abs(t) < 1e-15:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        else:
            hn = np.clip(hn, 0.0, 1.0)
            ssum = float(np.sum(hn))
            if abs(ssum - target) > 1e-10:
                hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_f - 1e-14:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + uphill:
            h = hn
            cur_f = nf
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            stall += 1

        if use_slsqp and it_count % slsqp_every == 0 and stall > 10:
            sl_h, sl_f = _slsqp_solve(h, 100, top_k=60)
            if sl_f < best_f - 1e-14:
                best_f = sl_f
                best_h = sl_h.copy()
                h = sl_h.copy()
                cur_f = sl_f
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
                it_count = 0

        if stall > 120:
            restarts += 1
            if restarts > 40:
                h = _project(best_h + rng.normal(0, max(0.04, sigma * 3), n), target)
            else:
                h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            sigma = max(0.003, sigma * 0.85)
    return best_h, best_f


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)


def _run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None
    if base is None or base.size < 8:
        n = 122
        base = _sine_init(n, 1, 0.0, rng)

    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: SLSQP from initial (15% budget)
    d1 = t0 + budget_s * 0.15
    sl_h, sl_f = _slsqp_solve(h0, 300, top_k=None)
    if sl_f < best_f:
        best_f = sl_f
        best_h = sl_h.copy()

    # Try SLSQP from perturbed starts
    for _ in range(3):
        if time.time() > d1:
            break
            hp = _project(best_h + rng.normal(0, 0.06, n0), n0 / 2.0)
            sl_h, sl_f = _slsqp_solve(hp, 200, top_k=80)
        if sl_f < best_f:
            best_f = sl_f
            best_h = sl_h.copy()

    # Phase 2: Line-search optimization with SLSQP injection (50% budget)
    d2 = t0 + budget_s * 0.65
    hh, ff = _optimize(best_h, d2, rng, pair_frac=0.15, rand_frac=0.08,
                       k_active=10, tau=0.02, use_slsqp=True, slsqp_every=40)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: Multi-n exploration (15% budget)
    d3 = t0 + budget_s * 0.80
    bn = len(best_h)
    tried_ns = set()
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -18, 18]:
        if time.time() > d3:
            break
        n_try = bn + dn
        if n_try < 30 or n_try > 260:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        hr = _resample(best_h, n_try)
        # Quick SLSQP
        sl_h, sl_f = _slsqp_solve(hr, 150, top_k=50)
        if sl_f < best_f:
            best_f = sl_f
            best_h = sl_h.copy()
            bn = n_try
        # Short line-search
        sub_deadline = min(d3, time.time() + 15.0)
        hh2, ff2 = _optimize(best_h, sub_deadline, rng, pair_frac=0.2,
                             rand_frac=0.05, k_active=8, tau=0.02)
        if ff2 < best_f:
            best_f = ff2
            best_h = hh2.copy()

    # Phase 4: Final deep polish (15% budget)
    d4 = t0 + budget_s * 0.95
    hh, ff = _optimize(best_h, d4, rng, pair_frac=0.35, rand_frac=0.04,
                       k_active=14, tau=0.012, use_slsqp=True, slsqp_every=25)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Final SLSQP polish
    if time.time() < deadline - 5:
        sl_h, sl_f = _slsqp_solve(best_h, 200, top_k=None)
        if sl_f < best_f:
            best_f = sl_f
            best_h = sl_h.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
