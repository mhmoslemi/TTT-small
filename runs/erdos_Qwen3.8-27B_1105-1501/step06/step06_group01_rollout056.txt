def _solve_lp_fixed_b(b, n, target):
    """Solve: min tau s.t. corr(a, b)[k]*dx <= tau for all k, sum(a)=target, 0<=a<=1"""
    dx = 2.0 / n
    # Build coefficient matrix: M[k, i] such that sum_i M[k,i]*a[i] = corr(a,b)[k]*dx
    # M[k, i] = dx * b[i - (k - (n-1))] if valid
    M = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        l = k - (n-1)  # lag
        for m in range(n):
            idx = m - l
            if 0 <= idx < n:
                M[k, m] = dx * b[idx]
    
    # LP: min tau
    # s.t. M @ a - tau <= 0 (i.e., M @ a + (-1)*tau <= 0)
    #      sum(a) = target
    #      0 <= a_i <= 1
    #      tau >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _build_M(b, n):
    """Build (2n-1) x n matrix M where M @ a = corr(a, b, 'full') * dx"""
    dx = 2.0 / n
    M = np.zeros((2*n-1, n))
    for l in range(-(n-1), n):
        k = n - 1 + l
        if l >= 0:
            M[k, l:] = dx * b[:n-l]
        else:
            d = -l
            M[k, :n-d] = dx * b[d:]
    return M

def _solve_minimax_lp(fixed, n, target):
    """Solve min tau s.t. M(fixed) @ x <= tau*ones, sum(x)=target, 0<=x<=1.
    Returns (x_opt, tau_opt) or (None, None)."""
    M = _build_M(fixed, n)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _global_lp_iter(h, deadline, max_iters=200):

    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        b = 1.0 - h
        a, tau = _solve_minimax_lp(b, n, target)
        if a is None:
            break
        a = _project(a, target)
        c5_a = _exact_c5(a)
        if c5_a < best_c5 - 1e-12:
            best_c5 = c5_a
            best_h = a.copy()
            h = a.copy()
        else:
            # Try blends
            found = False
            for alpha in [0.5, 0.3, 0.7, 0.2, 0.8]:
                h_blend = _project(alpha * a + (1-alpha) * h, target)
                c5_blend = _exact_c5(h_blend)
                if c5_blend < best_c5 - 1e-12:
                    best_c5 = c5_blend
                    best_h = h_blend.copy()
                    h = h_blend.copy()
                    found = True
                    break
            if not found:

                if b_new is not None:
                    b_new = _project(b_new, target)

                    if c5_b < best_c5 - 1e-12:

                        best_h = h_from_b.copy()
                        h = h_from_b.copy()
                    else:

                        h = _project(h + 0.05 * np.random.default_rng(it+1).normal(0, 1, n), target)
                else:

    return best_h, best_c5

def _alternating_lp(h, deadline, max_rounds=100):
    """Two-sided alternating LP with random restarts."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        # Pick a starting point for (a, b)
        if rnd % 3 == 0:
            a = h.copy()
            b = 1.0 - h
        elif rnd % 3 == 1:
            a = h + rng.normal(0, 0.1, n)
            a = _project(a, target)
            b = 1.0 - a
        else:

            b = _project(rng.standard_normal(n) * 0.5 + 0.5, target)
            a = h.copy()
        
        # Alternate a few times
        for it in range(5):
            if time.time() > deadline:
                break
            # Fix b, solve for a
            a_new, _ = _solve_minimax_lp(b, n, target)

            a = a_new
            b = b_new
    
    return best_h, best_c5

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = t_lo
    b_g = t_hi
    c1 = b_g - gr * (b_g - a_g)
    c2 = a_g + gr * (b_g - a_g)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(40):
        if b_g - a_g < 1e-12:
            break
        if f1 < f2:
            b_g, c2, f2 = c2, c1, f1
            c1 = b_g - gr * (b_g - a_g)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a_g, c1, f1 = c1, c2, f2
            c2 = a_g + gr * (b_g - a_g)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a_g + b_g) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _polish(h, deadline, max_moves=500):
    """Local polish with random coordinate moves and line search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(999)

            break

                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
