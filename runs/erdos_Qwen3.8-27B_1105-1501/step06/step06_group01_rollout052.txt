def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:

        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lp_pair_explore(h, deadline, n_trials=30, seed=42):

    Use LP pair exploration:

    3. Generate candidates from a*, 1-a*, midpoints
    4. Refine with line search
    """
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(seed)
    
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        
        # Choose partner
        mode = trial % 4
        if mode == 0:
            partner = 1.0 - h
        elif mode == 1:
            # Random feasible partner
            partner = _project(rng.random(n), target)
        elif mode == 2:
            # Smoothed complement
            from scipy.ndimage import uniform_filter1d
            partner = _project(uniform_filter1d(1.0 - h, size=5), target)
        else:
            # Binary threshold of complement
            partner = _project(1.0 - h + rng.normal(0, 0.1, n), target)

            continue
        
        # Generate candidates
        candidates = [a_star, _project(1.0 - a_star, target)]
        # Midpoints
        for alpha in [0.25, 0.5, 0.75]:
            cand = _project((1.0 - alpha) * best_h + alpha * a_star, target)
            candidates.append(cand)
        
        for cand in candidates:
            c = _exact_c5(cand)
            if c < best_c5 - 1e-12:
                best_c5 = c
                best_h = cand.copy()

        if best_h is not a_star:
