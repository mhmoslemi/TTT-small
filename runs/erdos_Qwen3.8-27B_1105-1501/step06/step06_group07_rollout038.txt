        if c > 0:

def _grad_at_shift(h, s):
    """Gradient of O(s) = sum h[n+s]*(1-h[n]) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[s:] += (1.0 - h[:m])
        g[:m] -= h[s:]
    else:
        a = -s
        m = n - a
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g

def _smooth_grad(h, alpha=50.0, K=30):

import numpy as np

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def _project(v, target):
    n = len(v)
    if n == 0:
        return v
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        m = (x < 1.0 - 1e-15) if res > 0 else (x > 1e-15)
        c = int(np.sum(m))
        if c > 0:
            x[m] += res / c
            x = np.clip(x, 0.0, 1.0)
    return x


def _grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[s:] += (1.0 - h[:m])
        g[:m] -= h[s:]
    else:
        a = -s
        m = n - a
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g


def _smooth_grad(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(-corr)[:K]
    vals = corr[idx]
    w = np.exp(alpha * (vals - vals[0]))
    w /= np.sum(w)
    grad = np.zeros(n)
    for wi, r in zip(w, idx):
        s = (n - 1) - int(r)
        grad += wi * _grad_shift(h, s)
    grad *= dx
    return grad, corr


def _line_min(h, d, target, max_t=1.5):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    T2 = -np.correlate(d, d, mode='full')
    t_hi = max_t
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
    t_hi = max(t_hi, 0.0)
    if t_hi < 1e-14:
        return 0.0, h.copy(), _c5(h)
    n_g = 100
    grid = np.linspace(0.0, t_hi, n_g)
    vals = S0[None, :] + grid[:, None] * T1[None, :] + (grid[:, None] ** 2) * T2[None, :]
    obj = np.max(vals, axis=1) * dx
    bi = int(np.argmin(obj))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if 0 < bi < n_g - 1:
        a, b = grid[bi - 1], grid[bi + 1]
    elif bi == 0:
        a, b = 0.0, grid[1]
    else:
        a, b = grid[-2], t_hi
    dS0, dT1, dT2 = S0.copy(), T1.copy(), T2.copy()

    def f(t):
        return float(np.max(dS0 + t * dT1 + t * t * dT2) * dx)

    fa, fb = f(a), f(b)
    for _ in range(100):
        if b - a < 1e-15:
            break
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        if f(c1) < f(c2):
            b, fb = c1, f(c1)
        else:

    t_opt = (a + b) / 2.0

def _slsqp_step(h, deadline, n_active=25, max_calls=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _c5(h)
    for _ in range(max_calls):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        t0 = float(np.max(corr)) + 0.002
        x0 = np.concatenate([h, [t0]])
        top_idx = np.argsort(-corr)[:n_active]
        h_ref = h.copy()
        lam = 0.01

        def obj_fn(x):
            return x[n] + lam * np.dot(x[:n] - h_ref, x[:n] - h_ref)

        def obj_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 2.0 * lam * (x[:n] - h_ref)
            g[n] = 1.0
            return g

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target,
                 'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
        for r in top_idx:
            k = int(r) - (n - 1)
            cons.append({'type': 'ineq',
                         'fun': lambda x, k=k: x[n] - _shift_overlap(x[:n], k) * dx,
                         'jac': lambda x, k=k: np.concatenate([-_shift_grad(x[:n], k) * dx, [1.0]])})
        bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.1)]
        try:
            res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac,
                           constraints=cons, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.x is not None:
                h_new = _project(res.x[:n], target)
                c5_new = _c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = c5_new
                    h = h_new.copy()
                else:
                    break
        except Exception:
            break
    return best_h, best_c5


def _shift_overlap(h, s):

    if s >= 0:
        m = n - s
        if m <= 0:
            return 0.0
        return float(np.dot(h[s:], 1.0 - h[:m]))
    else:
        a = -s
        m = n - a
        if m <= 0:
            return 0.0
        return float(np.dot(h[:m], 1.0 - h[a:]))


def _shift_grad(h, s):
    return _grad_shift(h, s)

    # Phase 3: Pair descent (20% budget)
    d3 = t0 + budget_s * 0.60
    h_pd, c5_pd = _pair_descent(best_h, min(d3, time.time() + budget_s * 0.08),
                                max_passes=8, pair_sample=3000)
    if c5_pd < best_c5:
        best_c5 = c5_pd
        best_h = h_pd.copy()
    h_pd2, c5_pd2 = _block_pair_descent(best_h, min(d3, time.time() + budget_s * 0.06),
                                        block_size=2, max_passes=10)
    if c5_pd2 < best_c5:
        best_c5 = c5_pd2
        best_h = h_pd2.copy()

    # Phase 4: Multi-resolution (15% budget)
    d4 = t0 + budget_s * 0.75
    tried = set()
    for dn in [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, 14, -16, 16]:
        if time.time() > d4:
            break
        n_try = len(best_h) + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.018)
        h_r, c5_r = _descend_precise(h_r, sub_dl, [25, 50], [80, 200], 50)
        if time.time() < d4 - 2:
            h_r2, c5_r2 = _pair_descent(h_r, min(d4, time.time() + 5), max_passes=3,
                                        pair_sample=1500)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Perturbation + reoptimize (15% budget)
    d5 = t0 + budget_s * 0.90
    h_pt, c5_pt = _perturb_reopt(best_h, rng, min(d5, time.time() + budget_s * 0.08),
                                 n_trials=60, scale0=0.03)
    if c5_pt < best_c5:
        best_c5 = c5_pt
        best_h = h_pt.copy()

    # Phase 6: Final intensive polish (10% budget)
    d6 = deadline
    h_f, c5_f = _descend_precise(best_h, min(d6, time.time() + budget_s * 0.03),
                                 [20, 40, 80], [100, 300, 600], 100)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    h_f2, c5_f2 = _pair_descent(best_h, min(d6, time.time() + budget_s * 0.02),
                                max_passes=5, pair_sample=5000)
    if c5_f2 < best_c5:
        best_c5 = c5_f2
        best_h = h_f2.copy()
    h_f3, c5_f3 = _descend_precise(best_h, deadline, [15, 30, 60], [200, 500], 100)
    if c5_f3 < best_c5:
        best_c5 = c5_f3
        best_h = h_f3.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = _c5(best_h)
    return best_h, best_c5, len(best_h)
