def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2*n - 1, n))
    # Positive lags (ell = 0, 1, ..., n-1)
    for ell in range(n):

def solve_qp(h, R0, G, target, rho, lambda_z, lambda_norm):
    n = len(h)
    C = 2 * n - 1
    total_dim = n + C
    
    # Bounds
    d_lo = -np.minimum(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    z_lo = np.zeros(C)
    z_hi = np.full(C, 1e6)  # large upper bound for z
    
    bounds = list(zip(d_lo, d_hi)) + list(zip(z_lo, z_hi))
    
    # Initial point
    d0 = np.zeros(n)
    z0 = np.maximum(G @ d0 + R0 - target, 0)
    z0 = np.minimum(z0, 1e6)
    x0 = np.concatenate([d0, z0])
    
    # Scale for z
    scale_z = max(np.mean(z0), 1e-9)
    
    # Norm scale
    norm_scale = n * rho * rho
    
    # Objective
    def obj(x):
        d = x[:n]
        z = x[n:]
        return lambda_z * np.sum(z) / scale_z + lambda_norm * np.sum(d**2) / norm_scale
    
    def obj_grad(x):
        d = x[:n]
        z = x[n:]
        gd = np.zeros(total_dim)
        gd[:n] = 2 * lambda_norm * d / norm_scale
        gd[n:] = lambda_z / scale_z
        return gd
    
    # Inequality constraints: G@d + R0 - target - z ≤ 0
    # For SLSQP, constraint is f(x) ≤ 0
    def ineq(x):
        d = x[:n]
        z = x[n:]
        return G @ d + R0 - target - z
    
    def ineq_jac(x):
        J = np.zeros((C, total_dim))
        J[:, :n] = G
        J[:, n:] = -np.eye(C)
        return J
    
    # Equality constraint: sum(d) = 0
    def eq(x):
        return np.sum(x[:n])
    
    def eq_jac(x):
        J = np.zeros(total_dim)
        J[:n] = 1.0
        return J
    
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    result = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 300, 'ftol': 1e-10})
    
    if result.success or (hasattr(result, 'x') and np.all(np.isfinite(result.x))):
        d = result.x[:n].copy()
        # Ensure sum is close to zero
        d -= np.mean(d)
        return d
    return None

def line_search(h, d, R0):
    n = len(h)
    dx = 2.0 / n
    # Compute D and Q for exact quadratic formula
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    
    # Bounds on alpha
    pos = d > 1e-12
    neg = d < -1e-12
    hi = 1e10
    lo = -1e10
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    # Try alpha = 0 (current) and bounds
    def f(alpha):
        return float(np.max(R0 + alpha * D - alpha * alpha * Q)) * dx
    
    best_alpha = 0.0
    best_val = f(0)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    if b - a < 1e-14:
        return 0.0, best_val
    
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = f(c1), f(c2)
    
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    
    opt = (a + b) / 2.0
    val = f(opt)
    if f(lo) < val:
        opt, val = lo, f(lo)
    if f(hi) < val:
        opt, val = hi, f(hi)
    
    return opt, val

import numpy as np

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for ell in range(n):
        k = n - 1 + ell
        m = n - ell
        G[k, ell:] += (1.0 - h[:m])
        G[k, :m] += -h[ell:]
    for r in range(1, n):
        k = n - 1 - r
        m = n - r
        G[k, :m] += (1.0 - h[r:])
        G[k, r:] += -h[:m]
    return G

def line_search(h, d, R0):
    n = len(h)
    dx = 2.0 / n
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    pos = d > 1e-12
    neg = d < -1e-12
    hi = 1e10
    lo = -1e10
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, float(np.max(R0)) * dx
    def f(alpha):
        return float(np.max(R0 + alpha * D - alpha * alpha * Q)) * dx
    best_alpha = 0.0
    best_val = f(0.0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = max(lo, -3.0), min(hi, 3.0)
    if b - a < 1e-14:
        for al in [lo, hi]:
            v = f(al)
            if v < best_val:
                best_val, best_alpha = v, al
        return best_alpha, best_val
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = f(c1), f(c2)
    for _ in range(55):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    opt = (a + b) / 2.0
    val = f(opt)
    for al in [a, b, lo, hi]:
        v = f(al)
        if v < val:
            val, opt = v, al
    if val < best_val:
        best_val, best_alpha = val, opt
    return best_alpha, best_val

def solve_qp(h, R0, G, target, rho, lam_z, lam_norm, maxiter=250):
    n = len(h)
    C = 2 * n - 1
    dim = n + C
    d_lo = -np.minimum(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    z_lo = np.zeros(C)
    z_hi = np.full(C, rho * 5 + 1.0)
    bounds = list(zip(d_lo, d_hi)) + list(zip(z_lo, z_hi))
    d0 = np.zeros(n)
    pred = R0 - target
    z0 = np.clip(pred, 0.0, z_hi)
    x0 = np.concatenate([d0, z0])
    scale_z = max(np.mean(z0), rho * 0.1, 1e-8)
    norm_scale = n * rho * rho

    def obj(x):
        d = x[:n]
        z = x[n:]
        return lam_z * np.sum(z) / scale_z + lam_norm * np.sum(d * d) / norm_scale

    def obj_grad(x):
        d = x[:n]
        g = np.zeros(dim)
        g[:n] = 2.0 * lam_norm * d / norm_scale
        g[n:] = lam_z / scale_z
        return g

    def ineq_fun(x):
        d = x[:n]
        z = x[n:]
        return G @ d + R0 - target - z

    J_ineq = np.zeros((C, dim))
    J_ineq[:, :n] = G
    J_ineq[:, n:] = -np.eye(C)

    def ineq_jac(x):
        return J_ineq

    def eq_fun(x):
        return np.sum(x[:n])

    J_eq = np.zeros(dim)
    J_eq[:n] = 1.0

    def eq_jac(x):
        return J_eq

    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if np.all(np.isfinite(res.x)):
            d = res.x[:n].copy()
            d -= np.mean(d)
            if np.max(np.abs(d)) > 1e-12:
                return d
    except Exception:
        pass
    return None

def qp_local_search(h, deadline, target_c5):

def golden_pair(h, i, j, dx, S0):

    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])

            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    for al in [a, b, lo, hi]:
        v = float(np.max(S0 + al * T - al * al * C)) * dx

            else:
                g = -d_lag

                w[:m] += weight[k] * (-h[g:g + m])
        donor_pool = np.argsort(-w)[:n_pairs]

def gradient_polish(h, deadline):

        weight /= float(np.sum(weight))
        g = np.zeros(n)
        for k, r in enumerate(active_idx):
            d_lag = r - (n - 1)
            m = n - abs(d_lag)

                continue
            if d_lag >= 0:
                if d_lag == 0:
                    g[:m] += weight[k] * (1.0 - 2.0 * h[:m])
                else:
                    g[:m] += weight[k] * (1.0 - h[d_lag:d_lag + m])
                    g[d_lag:d_lag + m] += weight[k] * (-h[:m])
            else:
                gg = -d_lag
                g[gg:] += weight[k] * (1.0 - h[:m])
                g[:m] += weight[k] * (-h[gg:gg + m])
        g -= np.mean(g)
        gn = np.linalg.norm(g)

        d_dir = -g / gn
        pos = d_dir > 1e-10

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)
