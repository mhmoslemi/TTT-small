import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _build_A1(b, dx):
    """Build matrix A1 where (A1 @ a)_q = dx * sum a[n+k] * b[n], k=q-(n-1)"""
    n = len(b)
    A = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        k = q - (n-1)
        if k >= 0:
            A[q, k:] = dx * b[:n-k]
        else:
            d = -k
            A[q, :n-d] = dx * b[d:]
    return A

def _build_A2(a, dx):
    """Build matrix A2 where (A2 @ b)_q = dx * sum a[n+k] * b[n], k=q-(n-1)"""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for q in range(2*n-1):
        k = q - (n-1)
        if k >= 0:
            A[q, :n-k] = dx * a[k:]
        else:
            d = -k
            A[q, d:] = dx * a[:n-d]
    return A

def _solve_minimax_lp(A, n):
    """Solve: min tau s.t. A @ x <= tau, sum(x) = n/2, 0 <= x <= 1"""
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _segment_line_search(u_L, u_R, n, dx):
    """Find best alpha in [0,1] minimizing C((1-alpha)*u_R + alpha*u_L)"""
    target = n / 2.0
    best_alpha = 0.0
    best_val = _exact_c5((1.0 - 0.0) * u_R + 0.0 * u_L)  # u_R
    # Also check u_L
    val_L = _exact_c5(u_L)
    if val_L < best_val:
        best_val = val_L
        best_alpha = 1.0
    # Grid search
    alphas = np.linspace(0.0, 1.0, 50)
    for a in alphas:
        h = (1.0 - a) * u_R + a * u_L
        h = _project(h, target)
        val = _exact_c5(h)
        if val < best_val:
            best_val = val
            best_alpha = a
    # Refine with golden section
    if 0.0 < best_alpha < 1.0:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = best_alpha - 0.05, best_alpha + 0.05
        a = max(a, 0.0)
        b = min(b, 1.0)
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = _exact_c5(_project((1-c1)*u_R + c1*u_L, target))
        f2 = _exact_c5(_project((1-c2)*u_R + c2*u_L, target))
        for _ in range(40):
            if b - a < 1e-10:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = _exact_c5(_project((1-c1)*u_R + c1*u_L, target))
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = _exact_c5(_project((1-c2)*u_R + c2*u_L, target))
        opt_a = (a + b) / 2.0
        opt_f = _exact_c5(_project((1-opt_a)*u_R + opt_a*u_L, target))
        if opt_f < best_val:
            best_val = opt_f
            best_alpha = opt_a
    h_best = _project((1.0 - best_alpha) * u_R + best_alpha * u_L, target)
    return h_best, best_val

def _segment_search_fast(u_L, u_R, n, dx):
    """Fast segment search using precomputed coefficients."""
    target = n / 2.0
    # Precompute cross-correlations
    corr_RR = np.correlate(u_R, u_R, 'full')  # length 2n-1
    corr_LL = np.correlate(u_L, u_L, 'full')
    corr_RL = np.correlate(u_R, u_L, 'full')
    corr_LR = np.correlate(u_L, u_R, 'full')
    
    # Compute SR and SL for each lag
    SR = np.zeros(2*n-1)
    SL = np.zeros(2*n-1)
    for q in range(2*n-1):
        k = q - (n-1)
        if k >= 0:
            SR[q] = np.sum(u_R[k:])
            SL[q] = np.sum(u_L[k:])
        else:
            d = -k
            SR[q] = np.sum(u_R[:n-d])
            SL[q] = np.sum(u_L[:n-d])
    
    pRR = corr_RR
    pRL = corr_RL
    pLR = corr_LR
    pLL = corr_LL
    
    A = dx * (-pRR + pRL + pLR - pLL)
    B = dx * (-SR + SL + 2*pRR - pRL - pLR)
    C = dx * (SR - pRR)
    
    # Grid search
    alphas = np.linspace(0.0, 1.0, 100)
    S = A * alphas[:, None]**2 + B[None, :] * alphas[:, None] + C[None, :]
    obj = np.max(S, axis=1)
    best_i = int(np.argmin(obj))
    best_alpha = alphas[best_i]
    best_val = float(obj[best_i])
    
    # Golden section refinement
    if 0 < best_i < len(alphas)-1:
        a = alphas[best_i-1]
        b = alphas[best_i+1]
    else:
        a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr*(b-a)
    c2 = a + gr*(b-a)
    f1 = float(np.max(A*c1**2 + B*c1 + C))
    f2 = float(np.max(A*c2**2 + B*c2 + C))
    for _ in range(50):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr*(b-a)
            f1 = float(np.max(A*c1**2 + B*c1 + C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr*(b-a)
            f2 = float(np.max(A*c2**2 + B*c2 + C))
    opt_a = (a+b)/2
    opt_f = float(np.max(A*opt_a**2 + B*opt_a + C))
    if opt_f < best_val:
        best_alpha, best_val = opt_a, opt_f
    
    h_best = _project((1-best_alpha)*u_R + best_alpha*u_L, target)
    # Verify with exact
    exact_val = _exact_c5(h_best)
    return h_best, exact_val

import numpy as np
from scipy.optimize import linprog
import time


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _build_A1(b, dx):
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        if k >= 0:
            A[q, k:] = dx * b[:n - k]
        else:
            d = -k
            A[q, :n - d] = dx * b[d:]
    return A


def _build_A2(a, dx):

def _solve_minimax_lp(A, n):
    m = A.shape[0]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

            best_h = h_seg.copy()
            h = h_seg.copy()
            improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                h = best_h + rng.normal(0, 0.1, n)
                h = _project(h, target)
                stall = 0
                if stall > 60:
                    break
    return best_h, best_c5


def _diverse_start(n, rng):
    mode = int(rng.integers(0, 6))
    if mode == 0:
        freq = int(rng.integers(1, 9))
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / n + ph1) + 0.2 * np.cos(2 * np.pi * (freq + 2) * x / n + ph2)
    elif mode == 1:
        h = np.zeros(n)
        n_blocks = int(rng.integers(3, 10))
        bs = max(1, n // n_blocks)
        for i in range(n_blocks):
            if rng.random() < 0.5:
                h[i * bs:(i + 1) * bs] = 1.0
    elif mode == 2:
        h = rng.random(n)
        h = np.convolve(h, np.ones(5) / 5.0, mode='same')
    elif mode == 3:
        h = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
        h += rng.normal(0, 0.15, n)
    elif mode == 4:
        period = int(rng.integers(n // 4, n // 2))
        pattern = (rng.random(period) > 0.5).astype(float)
        reps = int(np.ceil(n / period))
        h = np.tile(pattern, reps)[:n]
        h += rng.normal(0, 0.08, n)
    else:
        k = int(rng.integers(2, 6))
        h = 0.5 * np.ones(n)
        for _ in range(k):
            center = int(rng.integers(0, n))
            width = int(rng.integers(n // 8, n // 3))
            x = np.arange(n)
            h += 0.3 * np.exp(-0.5 * ((x - center) / width) ** 2) * (1 if rng.random() < 0.5 else -1)
    return _project(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _gradient_polish(h, deadline, max_iters=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = 0.01
    tau = 0.01
    for it in range(max_iters):
        if time.time() > deadline:
            break
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        S_max = float(np.max(S))
        w = np.exp((S - S_max) / tau)
        w_sum = float(np.sum(w))
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for q in range(2 * n - 1):
            k = q - (n - 1)
            if k >= 0:
                m = n - k
                grad[:m] += w[q] * dx * (-(1.0 - h[:m]))
                grad[k:] += w[q] * dx * (1.0 - h[k:]) * 0
                grad[k:] += w[q] * dx * h[:m]
            else:
                d = -k
                m = n - d
                grad[d:] += w[q] * dx * (1.0 - h[:m])
                grad[:m] += w[q] * dx * h[d:]
        h = h - lr * grad
        h = _project(h, target)
        cur = _exact_c5(h)
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.85
        if it % 30 == 29:
            tau *= 0.95
            lr = max(lr, 0.003)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.default_rng(seed)
    try:
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    except (NameError, Exception):
        n = 122
        h = _diverse_start(n, np.random.default_rng(seed))
    h = np.clip(h, 0.0, 1.0)
    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    d1 = t0 + budget_s * 0.38
    h1, c5_1 = _relaxed_pair_optimize(best_h, d1, max_iters=800, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    d2 = t0 + budget_s * 0.42
    h2, c5_2 = _gradient_polish(best_h, d2, max_iters=600)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    d3 = t0 + budget_s * 0.68
    n_starts = 7
    for si in range(n_starts):
        if time.time() > d3:
            break
        remaining = d3 - time.time()
        if remaining < 10:
            break
        sub_dl = time.time() + remaining / (n_starts - si) * 0.85
        h_s = _diverse_start(n, rng)
        h_s, c5_s = _relaxed_pair_optimize(h_s, sub_dl, max_iters=150, seed=100 + si)
        if c5_s < best_c5 - 1e-10:
            best_c5 = c5_s
            best_h = h_s.copy()
            h_s, c5_s2 = _gradient_polish(best_h, min(sub_dl + 5, d3), max_iters=100)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s.copy()

    d4 = t0 + budget_s * 0.88
    n_best = len(best_h)
    tried = set([n_best])
    offsets = []
    for dn in range(-24, 25, 2):
        if dn == 0:
            continue
        nt = n_best + dn
        if 30 <= nt <= 260 and nt % 2 == 0:
            offsets.append(dn)
    rng.shuffle(offsets)
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = n_best + dn
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.025)
        h_r, c5_r = _relaxed_pair_optimize(h_r, sub_dl, max_iters=120, seed=200 + n_try)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try

    d5 = deadline
    h5, c5_5 = _relaxed_pair_optimize(best_h, d5, max_iters=400, seed=999)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    remaining = deadline - time.time()
    if remaining > 10:
        h6, c5_6 = _gradient_polish(best_h, deadline, max_iters=300)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)

    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
