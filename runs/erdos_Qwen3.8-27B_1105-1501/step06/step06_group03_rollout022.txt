def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x in [0,1]^n : sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_row(h, k, n):
    """Gradient of the correlation at lag k with respect to h."""
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += (1.0 - h[:m])
            g[:m] += -h[k:]
    else:
        d = -k
        m = n - d
        if m > 0:
            g[:m] += (1.0 - h[d:d+m])
            g[d:] += -h[:m]
    return g

def _line_search(h, d, dx, trust_radius):
    """Line search along direction d from h, constrained to [0,1]^n."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = trust_radius
    t_lo = -trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:

    # Golden section search

    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _subgradient_direction(h, S, n, K):
    """Subgradient-based direction: negative average of top K active gradients."""
    M = float(S.max())
    # Find active or near-active lags
    tol = max(0.02 * M, 1e-8)
    active = np.where(S >= M - tol)[0]
    if len(active) > K:
        # Weight by how close to max
        weights = S[active] - (M - tol)
        weights = weights ** 2 + 1e-10
        top_idx = active[np.argsort(-weights)[:K]]
    else:
        top_idx = active
    
    if len(top_idx) == 0:
        top_idx = np.argsort(-S)[:min(K, 2*n-1)]
    
    # Compute weighted sum of gradients
    d = np.zeros(n)
    for r in top_idx:
        k = r - (n - 1)
        w = (S[r] - (M - tol)) + 1e-6
        d += w * _grad_row(h, k, n)
    
    d -= d.mean()  # Ensure zero-sum (preserve integral constraint)
    norm = np.linalg.norm(d)
    if norm < 1e-12:
        return None
    return -d / norm

    A_ub[:, n] = 1.0

    return M + tau * np.log(np.sum(np.exp(shifted)))

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _mseq_init(n, rng):
    """Initialize with a pattern inspired by m-sequences / low autocorrelation."""
    # Use a sum of sines with carefully chosen frequencies to get low peak correlation
    x = np.arange(n)
    h = np.zeros(n)
    n_coeffs = min(8, n // 4)
    for k in range(1, n_coeffs + 1):
        h += np.sin(2 * np.pi * k * x / n + rng.uniform(0, 2*np.pi)) * (1.0 / k)
    h -= h.min()
    h /= (h.max() - h.min() + 1e-10)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks-1, n-2), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(len(boundaries) - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b+1]] = 1.0
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    # Phase 2: Main LP optimization (18% time)
    d2 = t0 + budget_s * 0.26
    h2, c5_2 = _optimize(best_h, d2, K=18, radius_init=0.14)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Directed SA (15% time)
    d3 = t0 + budget_s * 0.41
    h3, c5_3 = _directed_sa(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    if time.time() < deadline:
        sub_dl = min(d3 + budget_s * 0.03, deadline)
        h3b, c5_3b = _optimize(best_h, sub_dl, K=12, radius_init=0.06)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: Multi-resolution (18% time)
    d4 = t0 + budget_s * 0.59
    base_n = len(best_h)
    tried = set()
    offsets = [0, -2, 2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12,
               14, -14, 16, -16, 18, -18, 20, -20, 24, -24, 28, -28, 32, -32]
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = base_n + dn
        if n_try < 80 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d4, time.time() + budget_s * 0.012)
            h_r, c5_r = _optimize(h_r, sub_dl, K=14, radius_init=0.09)
            if c5_r < best_c5 and time.time() < d4 - budget_s * 0.008:
                sub_dl2 = min(d4, time.time() + budget_s * 0.008)
                h_r2, c5_r2 = _directed_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d4 - budget_s * 0.003:
                    h_r, c5_r = _optimize(h_r, min(d4, time.time() + budget_s * 0.003),
                                          K=10, radius_init=0.04)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Pair refinement (10% time)
    d5 = t0 + budget_s * 0.69
    h5, c5_5 = _pair_refine(best_h, d5, max_passes=300)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Aggressive perturbation (12% time)
    d6 = t0 + budget_s * 0.81
    scale = 0.015
    for attempt in range(600):
        if time.time() > d6:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.003)
        h_t, c5_t = _optimize(h_pert, sub_dl, K=10, radius_init=0.05)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.82)
        else:
            scale = min(0.08, scale * 1.12)

    # Phase 7: Second SA round with different parameters (7% time)
    d7 = t0 + budget_s * 0.88
    h7, c5_7 = _directed_sa(best_h, d7, rng)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7 + budget_s * 0.02:
        h7b, c5_7b = _optimize(best_h, min(d7 + budget_s * 0.02, deadline), K=10, radius_init=0.04)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()

    # Phase 8: Final intensive polish (7% time)
    d8 = deadline
    h8, c5_8 = _pair_refine(best_h, d8, max_passes=150)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline:
        h9, c5_9 = _optimize(best_h, deadline, K=8, radius_init=0.025)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
