The key is to make the function "incoherent" at small lags while maintaining the k=0 value low. Since k=0 gives ∫h - ∫h², and we need ∫h = 1, we want to maximize ∫h². import numpy as np
import time
from scipy.optimize import minimize

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -10.0, 10.0)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(12):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = res / cnt
        x[mask] = np.clip(x[mask] + step, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _overlap_grad(h, k, n, dx):
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] += (-h[k:])
    else:
        q = -k
        g[:n - q] += (1.0 - h[q:])
        g[q:] += (-h[:n - q])
    return g * dx

def _slsqp_obj(x, n, alags, dx, target):
    return x[n]

def _slsqp_obj_jac(x, n, alags, dx, target):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _slsqp_eq(x, n, alags, dx, target):
    return float(x[:n].sum()) - target

def _slsqp_eq_jac(x, n, alags, dx, target):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def _slsqp_ineq(x, n, alags, dx, target):
    h = x[:n]
    z = x[n]
    c = _corr_full(h) * dx
    idx = np.array([int(lag) + (n - 1) for lag in alags])
    return z - c[idx]

def _slsqp_ineq_jac(x, n, alags, dx, target):
    h = x[:n]
    m = len(alags)
    J = np.zeros((m, n + 1))
    for i, lag in enumerate(alags):
        J[i, :n] = _overlap_grad(h, int(lag), n, dx)
        J[i, n] = -1.0
    return J

def _slsqp_solve(h, n, top_M, z_init, rng, maxiter=250):
    dx = 2.0 / n
    target = n / 2.0
    c = _corr_full(h) * dx
    sorted_idx = np.argsort(-c)
    top_idx = sorted_idx[:top_M]
    if top_M < len(c):
        n_extra = min(3, len(c) - top_M)
        extra = rng.choice(len(c) - top_M, size=n_extra, replace=False) + top_M
        top_idx = np.concatenate([top_idx, extra])
    alags = [int(l) - (n - 1) for l in top_idx]
    x0 = np.concatenate([h, [z_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
    constraints = [
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac},
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac}
    ]
    args = (n, alags, dx, target)
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_jac,
                       bounds=bounds, constraints=constraints, args=args,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
    except Exception:
        return None, None
    if res.x is not None and res.x.shape[0] == n + 1:
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        return h_new, f_new
    return None, None

def _line_search_step(h, d, corr, dx):
    n = len(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-15
    neg = d < -1e-15
    L, U = -1e30, 1e30
    if np.any(pos):
        L = max(L, float(np.min(-h[pos] / d[pos])))
        U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        L = max(L, float(np.min((1.0 - h[neg]) / d[neg])))
        U = min(U, float(np.min(-h[neg] / d[neg])))
    if U - L < 1e-12:
        return 0.0, _exact_c5(h)
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = corr[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(corr + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv

def _active_direction(h, corr, dx, n, k_active, tau, rng):
    scores = corr * dx
    m = float(np.max(scores))
    w = np.exp((scores - m) / tau)
    idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = int(r) - (n - 1)
        G += w[r] * _overlap_grad(h, k, n, dx)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _run_slsqp_loop(h, best_f, deadline, rng, M_min=10, M_max=25):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_f = _exact_c5(h)
    best_local = cur_f
    best_h = h.copy()
    no_improve = 0
    while time.time() < deadline:
        M = int(rng.integers(M_min, M_max + 1))
        z_init = cur_f + 0.002
        h_new, f_new = _slsqp_solve(h, n, M, z_init, rng)
        if h_new is not None and f_new < best_local:
            best_local = f_new
            best_h = h_new.copy()
            h = h_new.copy()
            cur_f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 8:
                h = _project(best_h + rng.normal(0, 0.015, n), target)
                cur_f = _exact_c5(h)
                no_improve = 0
            elif no_improve > 20:
                h = _project(best_h + rng.normal(0, 0.04, n), target)
                cur_f = _exact_c5(h)
                no_improve = 0
    return best_h, best_local

def _run_linesearch_loop(h, best_f, deadline, rng, pair_frac=0.20, rand_frac=0.10, k_active=10, tau=0.02):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_local = cur_f
    best_h = h.copy()
    corr = _corr_full(h)
    stall = 0
    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        else:
            d = _active_direction(h, corr, dx, n, k_active, tau, rng)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        t, newf = _line_search_step(h, d, corr, dx)
        if abs(t) < 1e-14:
            stall += 1
            continue
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        if nf < best_local:
            best_local = nf
            best_h = hn.copy()
        if nf < cur_f + 5e-4:
            h = hn
            cur_f = nf
            corr = _corr_full(h)
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_local
            corr = _corr_full(h)
            stall += 1
        if stall > 120:
            sigma = max(0.005, 0.05 * (0.9 ** (stall // 120)))
            h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = _corr_full(h)
            stall = 0
    return best_h, best_local
