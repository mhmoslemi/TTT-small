Let me analyze the problem and the strategy carefully.

The key idea from the strategy is to use a **full-row trust-region LP (linear programming) epigraph descent** instead of the pair-based local search used in the parent. The core insight is:

1. At each state `h`, compute the raw overlap rows `R_k = correlate(h, 1-h, mode="full")`
2. Compute the gradient matrix `G` where `G[k, i]` is the partial derivative of row `k` with respect to `h[i]`
3. Solve an LP: minimize `t` subject to `G[i,:] @ v + c[i] <= t` for all rows, where `c[i] = R[i] - G[i,:] @ h`
4. This gives a direction `d = v - h` that tries to reduce the maximum overlap row uniformly
5. Do a line search along `d` and accept if the exact objective improves

This is fundamentally different from the parent's pair-based search because it uses a full-dimensional direction from an LP solver that considers all rows simultaneously.

Let me reconsider the line search. The direction is `d = v - h`. We want to find `alpha in [0, 1]` that minimizes `c5(h + alpha*d)`. Since both `h` and `v = h + d` are feasible, any convex combination is also feasible (the feasible region is convex). So `alpha in [0, 1]` is always feasible.

The exact formula: `c5(h + alpha*d) = max(R0 + alpha*T - alpha^2*C) * dx`

where:
- `R0 = correlate(h, 1-h, 'full')`
- `T = correlate(d, 1-h, 'full') - correlate(h, d, 'full')`
- `C = correlate(d, d, 'full')`

`= R0 + alpha*(correlate(d, 1-h, 'full') - correlate(h, d, 'full')) - alpha^2*correlate(d, d, 'full')`

Yes, that's correct.

Now, the golden section search over `alpha in [0, 1]` with this formula is fast since we just need to compute `max` of vectors of length `2n-1`.

import numpy as np
import time
from scipy.optimize import linprog

def _c5_exact(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _c5_and_rows(h):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(R) * (2.0 / n)), R

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    """Build full gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            # dR_k/dh[i] = (1-h[i+d]) for i < m, minus h[i-d] for i >= d
            G[k, :m] += (1.0 - h[d:d+m])
            if d > 0:
                G[k, d:d+m] += -h[:m]
        else:
            q = -d
            m = n - q
            if m <= 0:
                continue
            # dR_k/dh[i] = (1-h[i-q]) for i >= q, minus h[i+q] for i < m
            G[k, q:] += (1.0 - h[:m])
            G[k, :m] += -h[q:q+m]
    return G

def _line_search(h, d, R0, dx, alpha_lo=0.0, alpha_hi=1.0):
    """Golden section line search along direction d from h."""
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    def f(a):
        return float(np.max(R0 + a * T - a * a * C)) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = alpha_lo, alpha_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = f(c1), f(c2)
    for _ in range(50):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    opt = (a + b) / 2.0
    val = f(opt)
    # Check endpoints
    f_lo, f_hi = f(alpha_lo), f(alpha_hi)
    if f_lo < val:
        opt, val = alpha_lo, f_lo
    if f_hi < val:
        opt, val = alpha_hi, f_hi
    return opt, val

def _lp_trust_step(h, R, G, radius):
    """Solve the LP trust region step. Returns candidate v or None."""
    n = len(h)
    L = 2 * n - 1
    
    # Build LP: min t s.t. G @ v - t <= G @ h - R, sum(v) = n/2, h-r <= v <= h+r
    # Variables: [t, v0, v1, ..., v_{n-1}]
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = G @ h - R
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    v_lo = np.maximum(0.0, h - radius)
    v_hi = np.minimum(1.0, h + radius)
    bounds = [(None, None)] + list(zip(v_lo, v_hi))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    v = res.x[1:]
    if not np.all(np.isfinite(v)):
        return None
    
    # Project to ensure feasibility
    v = _project(v, n / 2.0)
    return v

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _pair_search(h, deadline, n_pairs=15):
    """Fallback pair-based local search."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = _c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(123)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    while time.time() < deadline and stall < 100:
        scores = R0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.005 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]
        
        # Compute weighted gradient
        grads = np.zeros(n)
        weights = np.exp((scores[active_idx] - max_score) / (0.01 * max_score + 1e-15))
        weights /= np.sum(weights)
        
        for k, w in zip(active_idx, weights):
            d = k - (n - 1)
            if d >= 0:
                m = n - d
                if m > 0:
                    grads[:m] += w * (1.0 - h[d:d+m])
                    if d > 0:
                        grads[d:d+m] += -w * h[:m]
            else:
                q = -d
                m = n - q
                if m > 0:
                    grads[q:] += w * (1.0 - h[:m])
                    grads[:m] += -w * h[q:q+m]
        
        donor_pool = np.argsort(-grads)[:n_pairs]
        recv_pool = np.argsort(grads)[:n_pairs]
        
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        
        for i in donor_pool:
            for j in recv_pool:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                lo = -min(h[j], 1.0 - h[i])
                hi = min(h[i], 1.0 - h[j])
                if hi - lo < 1e-14:
                    continue
                T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                C = np.correlate(d, d, 'full')
                a, b = lo, hi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                f1 = float(np.max(R0 + c1 * T - c1*c1 * C)) * dx
                f2 = float(np.max(R0 + c2 * T - c2*c2 * C)) * dx
                for _ in range(35):
                    if b - a < 1e-13:
                        break
                    if f1 < f2:
                        b, c2, f2 = c2, c1, f1
                        c1 = b - gr * (b - a)
                        f1 = float(np.max(R0 + c1 * T - c1*c1 * C)) * dx
                    else:
                        a, c1, f1 = c1, c2, f2
                        c2 = a + gr * (b - a)
                        f2 = float(np.max(R0 + c2 * T - c2*c2 * C)) * dx
                opt = (a + b) / 2.0
                val = float(np.max(R0 + opt * T - opt*opt * C)) * dx
                if val < best_new_c5 - 1e-14:
                    best_new_c5 = val
                    best_alpha = opt
                    best_pair = (i, j)
        
        if best_pair is not None and best_alpha > 1e-10:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = _project(h, target)
            cur_c5 = _c5_exact(h)
            R0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 30:
                idx = rng.integers(n, size=3)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.03, 0.03)
                h = _project(h, target)
                cur_c5 = _c5_exact(h)
                R0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    
    return best_h, best_c5

    h_r, c5_r = _lp_descent(h_r, sub_dl, radii=[0.1, 0.05, 0.02, 0.01, 0.005])
    if c5_r < best_c5:
        best_c5 = c5_r
        best_h = h_r.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96

    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        h = _sine_init(122, 2, 0.5)

    n = len(h)
    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _c5_exact(h)
    best_n = n

    # Phase 1: LP trust region (55% time)
    d1 = t0 + budget_s * 0.55
    h1, c51 = _lp_descent(h, d1)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Second LP pass with different radii (15% time)
    d2 = t0 + budget_s * 0.70
    radii2 = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
    h2, c52 = _lp_descent(best_h, d2, radii=radii2)
    if c52 < best_c5:
        best_c5 = c52
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: Multi-resolution (12% time)
    d3 = t0 + budget_s * 0.82
    tried_n = {best_n}
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r, c5_r = _lp_resample_optimize(best_h, n_try, d3)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: Pair search polish (8% time)
    d4 = t0 + budget_s * 0.90
    h4, c54 = _pair_search(best_h, d4, 20)
    if c54 < best_c5:
        best_c5 = c54
        best_h = h4.copy()

    # Phase 5: Final LP polish (6% time)
    d5 = deadline
    radii5 = [0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5]
    h5, c55 = _lp_descent(best_h, d5, radii=radii5)
    if c55 < best_c5:
        best_c5 = c55
        best_h = h5.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5_exact(best_h)
    return best_h, final_c5, len(best_h)
