            break
        if stall > 80:
            break
        scores = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(scores))
        # Soft-max weights
        exps = np.exp(np.clip(beta * (scores - M), -50, 0))
        w = exps / (float(np.sum(exps)) + 1e-30)
        # Gradient of smooth max
        # grad_k = dx * (sum over valid i of (1-h[i+k]) for active k)
        # Use the weight-weighted gradient
        G = np.zeros(n)
        nlag = len(scores)
        center = n - 1
        for lag_idx in range(nlag):
            k = lag_idx - center
            if k >= 0:
                m = n - k
                if m > 0:
                    G[0:m] += w[lag_idx] * dx * (1.0 - h[k:k+m])
                    G[k:n] += w[lag_idx] * dx * (-h[0:m])
            else:

                m = n - g
                if m > 0:
                    G[g:n] += w[lag_idx] * dx * (1.0 - h[0:m])
                    G[0:m] += w[lag_idx] * dx * (-h[g:n])
        # Remove mean to preserve sum constraint
        G -= np.mean(G)
        # Step

        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
