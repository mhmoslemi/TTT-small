def _exact_c5(h):

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def _project(v, target):

    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

    corr = corr_raw * dx
