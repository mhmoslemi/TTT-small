For k ≠ 0, C_k = dx * |{n: h[n+k]=1 and h[n]=0}| (for k>0, with appropriate bounds).

So for binary solutions, we want to minimize the maximum number of "0→1" transitions at any lag.

            p_rand = rng.normal(0, 1, n)
            p_rand -= p_rand.mean()
            p_rand = p_rand / np.linalg.norm(p_rand) * radius
            directions.append(p_rand)
        best_alpha = 0.0
        best_new_c5 = cur_c5
        for p in directions:
            alpha, c5_try = line_search(h, p, dx, radius)
            if c5_try < best_new_c5:
                best_new_c5 = c5_try
                best_alpha = alpha
                best_dir = p
        if best_new_c5 < cur_c5 - 1e-15:
            h = h + best_alpha * best_dir
            h = project(h, target)
            cur_c5 = exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                beta = beta * 1.05
                if abs(best_alpha) / max(radius, 1e-12) > 0.7:
                    radius = min(radius * 1.08, 0.3)
                elif abs(best_alpha) / max(radius, 1e-12) > 0.3:
                    radius = min(radius * 1.02, 0.25)
                else:
                    radius = max(radius * 0.95, 0.02)
                stall = 0
            else:
                radius = max(radius * 0.9, 0.02)
                stall += 1
        else:
            stall += 1
            radius *= 0.8
            beta *= 1.1
            if stall % 10 == 0 and radius < 0.01:
                h = h + rng.normal(0, 0.02, n)
                h = project(h, target)
                cur_c5 = exact_c5(h)
                radius = 0.06
            if radius < 1e-6:
                break
    return best_h, best_c5

def binary_sa(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.008
    T_min = 2e-6
    cooldown = 0.99985
    patience = 0
    max_patience = 20000
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.85)
            patience = 0
        if T < T_min:
            break
        for _ in range(300):
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.integers(len(ones_list))]
            j = zeros_list[rng.integers(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def binary_best_improve(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    h = np.zeros(n)
    sorted_idx = np.argsort(-h_init)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones_list = np.where(h > 0.5)[0]
        zeros_list = np.where(h <= 0.5)[0]
        for i in ones_list:
            if time.time() > deadline:
                break
            for j in zeros_list:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = exact_c5(h)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    cur_c5 = new_c5
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)

def block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks < 2:
        n_blocks = 2
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks - 1, n - 1), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(len(boundaries) - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return project(h, n / 2.0)

def pair_refine(h0, deadline, max_passes=300):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(555)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = corr_scaled(h)
        all_idx = np.arange(2 * n - 1)
        G = grad_rows(h, all_idx, n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 80)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = project(h, target)
                    cur = exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def multi_pair_refine(h0, deadline):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(444)
    stall = 0
    while time.time() < deadline and stall < 50:
        interior = np.where((h > 0.02) & (h < 0.98))[0]
        if len(interior) < 4:
            break
        idx = rng.choice(len(interior), size=min(4, len(interior)), replace=False)
        i, j, k, l = interior[idx]
        found = False
        for signs in [(-1, -1, 1, 1), (1, 1, -1, -1)]:
            d = np.zeros(n)
            d[i] = signs[0]
            d[j] = signs[1]
            d[k] = signs[2]
            d[l] = signs[3]
            alpha, c5_try = line_search(h, d, dx, 0.3)
            if c5_try < best_c5 - 1e-14:
                h = h + alpha * d
                h = project(h, target)
                cur = exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    stall = 0
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = sine_init(n0, 2, 0.5)

    h0 = project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = exact_c5(h0)

    # Phase 1: Multi-start (10% time)
    d1 = t0 + budget_s * 0.10
    starts = [best_h.copy()]
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        for ph in [0.0, 0.5, 1.0, 1.5, 2.0, 2.5]:
            starts.append(sine_init(n0, freq, ph))
    for _ in range(12):
        starts.append(block_init(n0, rng, rng.integers(3, 16)))
    for s in starts:
        if time.time() > d1:
            break
        sub_dl = min(d1, time.time() + budget_s * 0.007)
        h_s, c5_s = local_opt(s, sub_dl, radius_init=0.10, K=12, beta_init=50.0)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Main local optimization (35% time)
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = local_opt(best_h, d2, radius_init=0.14, K=18, beta_init=40.0)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Second pass with different params (10% time)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = local_opt(best_h, d3, radius_init=0.06, K=22, beta_init=80.0)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Binary SA + best-improve (15% time)
    d4 = t0 + budget_s * 0.70
    h4a, c5_4a = binary_sa(best_h, t0 + budget_s * 0.62, rng)
    if c5_4a < best_c5:
        best_c5 = c5_4a
        best_h = h4a.copy()
    if time.time() < d4:
        h4b, c5_4b = binary_best_improve(best_h, min(d4, time.time() + budget_s * 0.04), rng)
        if c5_4b < best_c5:
            best_c5 = c5_4b
            best_h = h4b.copy()
    if time.time() < d4:
        h4c, c5_4c = local_opt(best_h, d4, radius_init=0.05, K=15, beta_init=60.0)
        if c5_4c < best_c5:
            best_c5 = c5_4c
            best_h = h4c.copy()

    # Phase 5: Pair refinement (8% time)
    d5 = t0 + budget_s * 0.78
    h5, c5_5 = pair_refine(best_h, d5, max_passes=400)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (10% time)
    d6 = t0 + budget_s * 0.88
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16, -20, 20, 24, -24, 28, -28]
    tried = set()
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample(best_h, n_try)
        c5_r = exact_c5(h_r)
        if c5_r < best_c5 + 0.003:
            sub_dl = min(d6, time.time() + budget_s * 0.018)
            h_r, c5_r = local_opt(h_r, sub_dl, radius_init=0.10, K=14, beta_init=50.0)
            if c5_r < best_c5 and time.time() < d6 - budget_s * 0.01:
                sub_dl2 = min(d6, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d6:
                    h_r, c5_r = local_opt(h_r, min(d6, time.time() + budget_s * 0.008),
                                         radius_init=0.04, K=14, beta_init=60.0)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Aggressive perturbation (6% time)
    d7 = t0 + budget_s * 0.94
    scale = 0.04
    for attempt in range(400):
        if time.time() > d7:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d7, time.time() + budget_s * 0.004)
        h_t, c5_t = local_opt(h_pert, sub_dl, radius_init=0.05, K=12, beta_init=60.0)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.75)
        else:
            scale = min(0.12, scale * 1.3)

    # Phase 8: Final polish (4% time)
    d8 = deadline
    h8, c5_8 = pair_refine(best_h, d8, max_passes=100)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()
    if time.time() < deadline:
        h9, c5_9 = multi_pair_refine(best_h, deadline)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
    if time.time() < deadline:
        h_f, c5_f = local_opt(best_h, deadline, radius_init=0.03, K=10, beta_init=80.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    n_final = len(best_h)
    if final_c5 > best_c5 + 1e-10:
        final_c5 = best_c5
    return best_h, final_c5, n_final
