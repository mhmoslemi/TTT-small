import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    # Project v to {x: 0<=x<=1, sum(x)=target}
    n = len(v)
    v = v.copy()
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) < 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _build_grad(h):
    # Returns (2n-1) x n gradient matrix
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        row = 0
        if k == 0:
            row = 0  # corr[0] corresponds to lag n-1... 
            # Wait, I need to be careful about indexing.
            # Output index r corresponds to lag k = r - (n-1)
            pass
        r = k + n - 1
        if k >= 0:
            # corr[r] = sum_{i=0}^{n-k-1} h[i+k]*(1-h[i])

                a = i + k

import numpy as np
import time
from scipy.optimize import linprog


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _build_grad(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    one_minus_h = 1.0 - h
    for k in range(-(n - 1), n):
        r = k + n - 1
        if k >= 0:
            m = n - k
            a_idx = np.arange(k, n)
            b_idx = np.arange(m)
            G[r, a_idx] += one_minus_h[b_idx]
            G[r, b_idx] += -h[a_idx]
        else:
            q = -k
            m = n - q
            a_idx = np.arange(m)
            b_idx = np.arange(q, n)
            G[r, a_idx] += one_minus_h[b_idx]
            G[r, b_idx] += -h[a_idx]
    return G


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(40):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv


def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, max(2, n // 6))))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _slp_step(h, rng, max_slp=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = _build_grad(h)
    h_curr = h.copy()
    f_curr = float(np.max(corr)) * dx
    improved = False
    for _ in range(max_slp):
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.empty((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n:] = -1.0
        b_ub = G @ h_curr - corr
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        tau_ub = float(np.max(corr)) + n
        bounds = [(0.0, 1.0)] * n + [(0.0, tau_ub)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success or res.x is None:
            break
        h_new = res.x[:n]
        d = h_new - h_curr
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-10:
            break
        d = d / d_norm * min(d_norm, 0.3)
        T = G @ d
        Q = np.correlate(d, d, mode='full')
        t, _ = _line_search(corr, T, Q, dx, 0.0, 1.0)
        if t < 1e-12:
            break
        h_try = h_curr + t * d
        if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:
            h_try = _project(h_try, n / 2.0)
        else:
            s = float(np.sum(h_try))
            if abs(s - n / 2.0) > 1e-10:
                h_try = _project(h_try, n / 2.0)
        f_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full'))) * dx
        if f_try < f_curr - 1e-13:
            f_curr = f_try
            h_curr = h_try
            corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
            G = _build_grad(h_curr)
            improved = True
        else:
            break
    return h_curr, f_curr, improved


def _heuristic_step(h, rng):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f_curr = float(np.max(corr)) * dx
    h_curr = h.copy()
    r = rng.random()
    if r < 0.3:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            return h_curr, f_curr, False
        d = np.zeros(n)
        d[i] = -1.0
        d[j] = 1.0
    elif r < 0.6:
        k_active = 10
        m = float(np.max(corr))
        scores = corr
        w = np.exp((scores - m) / 0.02)
        idx = np.argsort(-scores)[:k_active]
        G = np.zeros(n)
        for ri in idx:
            k = ri - (n - 1)
            row = np.zeros(n)
            if k >= 0:
                a_idx = np.arange(k, n)
                b_idx = np.arange(n - k)
                row[a_idx] += (1.0 - h_curr[b_idx])
                row[b_idx] += -h_curr[a_idx]
            else:
                q = -k
                a_idx = np.arange(n - q)
                b_idx = np.arange(q, n)
                row[a_idx] += (1.0 - h_curr[b_idx])
                row[b_idx] += -h_curr[a_idx]
            G += w[ri] * row
        d = -G
        d -= d.mean()
    else:
        d = rng.standard_normal(n)
        d -= d.mean()
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return h_curr, f_curr, False
    d = d / nrm
    pos = d > 1e-15
    neg = d < -1e-15
    low = [0.0]
    up = [1e30]
    if np.any(pos):
        low.append(float(np.min(-h_curr[pos] / d[pos])))
        up.append(float(np.min((1.0 - h_curr[pos]) / d[pos])))
    if np.any(neg):
        low.append(float(np.min((1.0 - h_curr[neg]) / d[neg])))
        up.append(float(np.min(-h_curr[neg] / d[neg])))
    L = max(low)
    U = min(up)
    if U - L < 1e-11:
        return h_curr, f_curr, False
    T = np.correlate(d, 1.0 - h_curr, mode='full') - np.correlate(h_curr, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    t, _ = _line_search(corr, T, Q, dx, L, U)
    if abs(t) < 1e-12:
        return h_curr, f_curr, False
    h_try = h_curr + t * d
    if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:
        h_try = _project(h_try, n / 2.0)
    f_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full'))) * dx
    if f_try < f_curr - 1e-13:
        return h_try, f_try, True
    return h_curr, f_curr, False


def _optimize_mixed(h, deadline, rng, slp_frac=0.65):
    n = len(h)
    best_h = h.copy()
    best_f = _exact_c5(h)
    h_curr = h.copy()
    f_curr = best_f
    stall = 0
    sigma = 0.04
    iters = 0
    while time.time() < deadline:
        r = rng.random()
        if r < slp_frac:
            h_new, f_new, improved = _slp_step(h_curr, rng, max_slp=6)
            h_curr = h_new
            f_curr = f_new
        else:
            h_new, f_new, improved = _heuristic_step(h_curr, rng)
            if f_new < f_curr:
                h_curr = h_new
                f_curr = f_new
            else:
                stall += 1
        if f_curr < best_f - 1e-13:
            best_f = f_curr
            best_h = h_curr.copy()
            stall = 0
        iters += 1
        if not (f_curr < best_f - 1e-13) and iters % 5 == 0:
            stall += 1
        if stall > 30:
            sigma = max(0.008, sigma * 0.85)
            h_curr = _project(best_h + rng.normal(0, sigma, n), n / 2.0)
            f_curr = _exact_c5(h_curr)
            stall = 0
            sigma = max(0.008, sigma * 0.9)
    return best_h, best_f

    t0 = time.time()
    deadline = t0 + budget_s
