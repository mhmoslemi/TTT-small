# Optimizer 1: Smooth minimax with projected gradient + momentum
def _smooth_opt(h0, deadline, mu_start=20, mu_end=300, n_mu=8, iters_per_mu=150):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    mus = np.geomspace(mu_start, mu_end, n_mu)
    for mu in mus:
        if time.time() > deadline:
            break
        L = 4.0  # initial step size parameter
        t_mom = 1.0
        h_prev = h.copy()
        for it in range(iters_per_mu):
            if time.time() > deadline:
                break
            corr = _full_corrs(h)
            # Softmax weights
            mc = mu * corr
            mc -= mc.max()
            p = np.exp(mc)
            p /= p.sum()
            
            # Gradient
            G = _corr_grads(h)  # (2n-1, n)
            grad = G.T @ p  # n-vector
            
            # Backtracking
            step = 1.0 / L
            h_new = None
            for _ in range(20):
                h_trial = _project(h - step * grad, target)
                c5_t = _exact_c5(h_trial)
                if c5_t < best_c5 + 1e-10:
                    h_new = h_trial
                    break
                step *= 0.5
            if h_new is None:
                L *= 2.0
                continue
            
            # Momentum (Nesterov)
            t_new = 0.5 * (1 + np.sqrt(1 + 4 * t_mom**2))
            h_mom = h_new + (t_mom - 1) / t_new * (h_new - h_prev)
            h_mom = _project(h_mom, target)
            
            h_prev = h
            h = h_mom
            t_mom = t_new
            
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

def _corr_grads(h):

            m = n - g

        for ph_frac in [0, 0.33, 0.66]:
            ph = ph_frac * 2 * np.pi
            s = 0.5 + 0.25 * np.sin(2 * np.pi * freq * (x / 2.0) + ph)
            s = s + rng.normal(0, 0.04, best_n)
            starts.append(_project(s, best_n / 2.0))
    for _ in range(3):
        s = np.zeros(best_n)
        block_sizes = rng.integers(3, max(4, best_n // 8), size=best_n // 3 + 2)
        pos = 0
        for bs in block_sizes:
            if pos + bs > best_n:
                bs = best_n - pos
            if bs <= 0:
                break
            s[pos:pos + bs] = 1.0 if rng.random() < 0.5 else 0.0
            pos += bs
        starts.append(_project(s, best_n / 2.0))

    for s in starts:
        if time.time() > d6:
            break
        sub_dl = min(d6, time.time() + budget_s * 0.012)
        h_s, c5_s = _active_opt(s, sub_dl, max_iter=100, initial_radius=0.1)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 7: Final polish (15% time)
    d7 = deadline
    h7, c5_7 = _active_opt(best_h, min(d7, time.time() + budget_s * 0.06), max_iter=300, initial_radius=0.08)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
        best_n = len(h7)
    if time.time() < deadline - 10:
        h7b, c5_7b = _smooth_opt(best_h, deadline, mu_start=100, mu_end=600, n_mu=4, iters_per_mu=100)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()
            best_n = len(h7b)

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
