```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x : 0 ≤ x_i ≤ 1, sum(x) = target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            x[mask] += res / np.sum(mask)
    return np.clip(x, 0.0, 1.0)

def _compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _corr_grad(h, active_lags):
    """Compute gradient of correlation at active lags w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(active_lags), n))
    for i, k in enumerate(active_lags):
        s = k - (n - 1)  # actual lag
        m = n - abs(s)
        if s >= 0:
            G[i, s:s+m] += (1.0 - h[:m])
            G[i, :m] += (-h[s:s+m])
        else:
            g = -s
            G[i, g:g+m] += (1.0 - h[:m])
            G[i, :m] += (-h[g:g+m])
    G *= dx
    return G
```

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _correct_grad(h, k):
    """Correct gradient of out[k] = np.correlate(h, 1-h)[k] w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    g = np.zeros(n)

    return g * dx

    """Gradients for multiple lags"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(active_lags), n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        G[i, js] = (1.0 - 2.0 * h[k - js])
    G *= dx
    return G

def _project(v, target):
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)
