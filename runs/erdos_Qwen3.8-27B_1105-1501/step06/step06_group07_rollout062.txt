def _min_max_lines(slopes, intercepts):
    """Find v in [0,1] minimizing max_k (slopes[k]*v + intercepts[k])."""
    # Binary search on optimal value t
    # Lower bound: min of the function
    lo_v = 0.0
    hi_v = 1.0
    
    # Check if min is at boundary
    val_0 = np.max(intercepts)
    val_1 = np.max(slopes + intercepts)
    
    # If all slopes >= 0, min at v=0
    if np.min(slopes) >= -1e-15:
        return 0.0, val_0
    # If all slopes <= 0, min at v=1
    if np.max(slopes) <= 1e-15:
        return 1.0, val_1
    
    # Interior: binary search on t
    # For given t, feasible v must satisfy:
    # For s_k > 0: v <= (t - c_k) / s_k
    # For s_k < 0: v >= (t - c_k) / s_k
    # Feasible iff: max_{s_k<0} (t-c_k)/s_k <= min_{s_k>0} (t-c_k)/s_k and within [0,1]
    
    t_lo = min(val_0, val_1) - 1.0  # safe lower bound
    t_hi = max(val_0, val_1) + 1.0  # safe upper bound
    
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2
        # Check feasibility at t_mid
        pos_mask = slopes > 1e-15
        neg_mask = slopes < -1e-15
        
        v_lo = 0.0
        v_hi = 1.0
        if np.any(neg_mask):
            v_lo = max(v_lo, np.max((t_mid - intercepts[neg_mask]) / slopes[neg_mask]))
        if np.any(pos_mask):
            v_hi = min(v_hi, np.min((t_mid - intercepts[pos_mask]) / slopes[pos_mask]))
        
        if v_lo <= v_hi:
            t_hi = t_mid  # t_mid is feasible, try smaller
        else:
            t_lo = t_mid  # t_mid not feasible, need larger
    
    t_opt = t_hi
    # Find v at t_opt

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: sum(x)=target, 0<=x<=1}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _min_max_affine(slopes, intercepts):
    """Find v in [0,1] minimizing max_k (slopes[k]*v + intercepts[k]).
    Returns (v_opt, opt_value)."""
    # Check boundary cases
    val_0 = float(np.max(intercepts))
    val_1 = float(np.max(slopes + intercepts))
    
    min_slope = float(np.min(slopes))
    max_slope = float(np.max(slopes))
    
    if min_slope >= -1e-14:
        return 0.0, val_0
    if max_slope <= 1e-14:
        return 1.0, val_1
    
    # Interior minimum: binary search on t
    # For given t, feasible v must satisfy:
    #   For slopes[k] > 0: v <= (t - intercepts[k]) / slopes[k]
    #   For slopes[k] < 0: v >= (t - intercepts[k]) / slopes[k]
    #   And 0 <= v <= 1
    
    pos = slopes > 1e-14
    neg = slopes < -1e-14
    
    t_lo = min(val_0, val_1) - 1.0
    t_hi = max(val_0, val_1) + 1.0
    
    for _ in range(80):
        t_mid = (t_lo + t_hi) / 2.0
        v_lo_bound = 0.0
        v_hi_bound = 1.0
        if np.any(neg):
            v_lo_bound = max(v_lo_bound, float(np.max((t_mid - intercepts[neg]) / slopes[neg])))
        if np.any(pos):
            v_hi_bound = min(v_hi_bound, float(np.min((t_mid - intercepts[pos]) / slopes[pos])))
        if v_lo_bound <= v_hi_bound + 1e-15:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    t_opt = t_hi
    v_lo_bound = 0.0
    v_hi_bound = 1.0
    if np.any(neg):
        v_lo_bound = max(v_lo_bound, float(np.max((t_opt - intercepts[neg]) / slopes[neg])))
    if np.any(pos):
        v_hi_bound = min(v_hi_bound, float(np.min((t_opt - intercepts[pos]) / slopes[pos])))
    v_opt = (v_lo_bound + v_hi_bound) / 2.0
    v_opt = float(np.clip(v_opt, 0.0, 1.0))
    return v_opt, t_opt

def _coord_descent(h_init, deadline, max_sweeps=200, tol=1e-8):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    # Precompute lag for each correlation index
    # corr[r] has lag = r - (n-1)
    lags = np.arange(2*n - 1) - (n - 1)
    
    lam = 0.0  # Lagrange multiplier
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Compute current correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # For each coordinate j, compute slopes and solve for optimal v
        h_new = h.copy()
        for j in range(n):
            # slope_r(j) = (1-h[j-lag_r])*I(j-lag_r in S_r) - h[j+lag_r]*I(j in S_r)
            # S_r = [max(0, -lag_r), min(n-1, n-1-lag_r)]
            
            slopes = np.zeros(2*n - 1)
            for r in range(2*n - 1):
                lag = lags[r]
                # Case 1: j - lag in S_r (i.e., h[j] appears as h[i+lag] with i=j-lag)
                i1 = j - lag
                if 0 <= i1 < n and 0 <= i1 + lag < n:
                    slopes[r] += (1.0 - h[i1])  # which is (1-h[j-lag])
                # Case 2: j in S_r (i.e., h[j] appears in (1-h[i]) with i=j)
                if 0 <= j < n and 0 <= j + lag < n:
                    slopes[r] -= h[j + lag]
            
            # Intercept: corr[r] - slope_r * h[j]
            intercepts = corr - slopes * h[j]
            
            # Solve: min_v max_r (slopes[r]*v + intercepts[r]) + lam*v
            # = min_v max_r ((slopes[r]+lam)*v + intercepts[r])
            v_opt, _ = _min_max_affine(slopes + lam, intercepts)
            h_new[j] = v_opt
        
        # Update Lagrange multiplier to satisfy sum constraint
        s = float(np.sum(h_new))
        if abs(s - target) < tol:
            h = h_new
        else:
            # Bisection on lambda
            if s > target:
                # Need to decrease sum, increase lambda
                lam_lo = lam
                lam_hi = lam + 1.0
            else:
                lam_hi = lam
                lam_lo = lam - 1.0
            
            for _ in range(30):
                lam_mid = (lam_lo + lam_hi) / 2.0
                # Recompute h_new with lam_mid
                h_try = h.copy()
                s_try = 0.0
                for j in range(n):
                    slopes = np.zeros(2*n - 1)
                    for r in range(2*n - 1):
                        lag = lags[r]
                        i1 = j - lag
                        if 0 <= i1 < n and 0 <= i1 + lag < n:
                            slopes[r] += (1.0 - h[i1])
                        if 0 <= j < n and 0 <= j + lag < n:
                            slopes[r] -= h[j + lag]
                    intercepts = corr - slopes * h[j]
                    v_opt, _ = _min_max_affine(slopes + lam_mid, intercepts)
                    h_try[j] = v_opt
                    s_try += v_opt
                
                if s_try > target:
                    lam_lo = lam_mid
                else:
                    lam_hi = lam_mid
                
                if abs(s_try - target) < tol:
                    break
            
            lam = (lam_lo + lam_hi) / 2.0
            # Use h_try from last iteration
            h = _project(h_try, target)
        
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check convergence
        if float(np.max(np.abs(h - best_h))) < 1e-10:
            break
    
    return best_h, best_c5

    #           which is 0<=j-lag_r<n and 0<=j<n (second is trivial)

    #   => 0 <= j-lag_r < n (and 0<=j<n is automatic since j in [0,n-1])

    # For valid2: 0 <= j+lag_r < n

    # Need to handle boundary: h[i]=0 for i outside [0,n-1]
    h_ext = np.concatenate([np.zeros(1), h, np.zeros(1)])  # 1-indexed? No...

    h_pad = np.zeros(2*n + 1)
    h_pad[n:n+n] = h  # h_pad[i+n] = h[i] for i in [0,n-1]
    
    # For valid1: h[j-lag_r] = h_pad[j-lag_r+n]
    # For valid2: h[j+lag_r] = h_pad[j+lag_r+n]
    
    j_idx = np.arange(n)
    # j-lag_r for each (j, r)
    jl = j_idx[None,:] - lags[:,None]  # shape (2n-1, n)
    jp = j_idx[None,:] + lags[:,None]
    
    slopes = np.zeros((2*n-1, n))
    mask1 = (jl >= 0) & (jl < n)
    mask2 = (jp >= 0) & (jp < n)
    
    # (1 - h[j-lag_r]) for valid entries
    vals1 = np.zeros_like(slopes)
    vals1[mask1] = 1.0 - h[jl[mask1]]
    vals2 = np.zeros_like(slopes)
    vals2[mask2] = h[jp[mask2]]
    
    slopes = vals1 - vals2
    return slopes, lags

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:

            x = np.clip(x, 0.0, 1.0)

def _min_max_affine(slopes, intercepts):
    val_0 = float(np.max(intercepts))
    val_1 = float(np.max(slopes + intercepts))
    if float(np.min(slopes)) >= -1e-14:
        return 0.0, val_0
    if float(np.max(slopes)) <= 1e-14:
        return 1.0, val_1
    pos = slopes > 1e-14
    neg = slopes < -1e-14
    t_lo = min(val_0, val_1) - 1.0
    t_hi = max(val_0, val_1) + 1.0
    for _ in range(70):

        if v_lo <= v_hi + 1e-15:

def _coord_desc(h_init, deadline, max_sweeps=80, random_order=False, rng=None):

    target = n / 2.0
    lags = np.arange(2 * n - 1) - (n - 1)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        h_new = np.zeros(n)
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

            jp = j + lags
            m1 = (jl >= 0) & (jl < n)

def _grad_row(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[k:] += (1.0 - h[:-k] if k > 0 else 1.0 - h)
        if k > 0:
            g[:-k] -= h[k:]
        else:
            g -= h
    else:
        ga = -k
        m = n - ga
        if m <= 0:
            return g
        g[:m] += (1.0 - h[ga:])
        g[ga:] -= h[:m]
    return g * dx


def _quad_line_search(h, d, target, max_t=0.8):

        return 0.0, _exact_c5(h)
    n_grid = 40

def _active_descent(h_init, deadline, K=20, max_iters=150):

        alpha = 20.0
        w = np.exp(alpha * (top_vals - M))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(K) / K
        else:
            w /= w_sum
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _grad_row(h, k)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        direction = -grad / gn
        t, f_try = _quad_line_search(h, direction, target, step)
        if t < 1e-12:
            step *= 0.7
            stall += 1
            if stall > 20:
                break
            continue
        h_new = h + t * direction
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            stall = 0
            if t > 0.8 * step:
                step = min(step * 1.05, 1.5)
            else:
                step = max(step * 0.9, 0.01)
        else:
            step *= 0.75
            stall += 1
            if step < 0.005 or stall > 30:
                break
    return best_h, best_c5


def _binary_sa(h_init, deadline, rng, max_iters=50000):
    n = len(h_init)
    target = n // 2
    if target * 2 != n:
        target = int(round(n / 2.0))
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.015
    T_min = 3e-6
    cooling = 0.99990
    ones_set = set(int(x) for x in np.where(h > 0.5)[0])
    zeros_list = [int(x) for x in np.where(h <= 0.5)[0]]
    for _ in range(max_iters):
        if time.time() > deadline or T < T_min:
            break
        if len(ones_set) == 0 or len(zeros_list) == 0:
            break
        i = list(ones_set)[int(rng.integers(len(ones_set)))]
        j = zeros_list[int(rng.integers(len(zeros_list)))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _exact_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_set.remove(i)
            ones_set.add(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * cooling)
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _slsqp_epigraph(h, deadline, lam_reg=0.005, n_active=25, max_calls=3):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    for call in range(max_calls):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        t0 = float(np.max(corr)) + 0.001
        x0 = np.concatenate([h, [t0]])
        top_idx = np.argsort(-corr)[:n_active]
        h_base = h.copy()

        def _obj(x):
            return x[n] + lam_reg * np.dot(x[:n] - h_base, x[:n] - h_base)

        def _obj_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 2.0 * lam_reg * (x[:n] - h_base)
            g[n] = 1.0
            return g

        constraints = [{'type': 'eq',
                        'fun': lambda x, t=target: np.sum(x[:n]) - t,
                        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
        for r in top_idx:
            k = int(r) - (n - 1)

            def _cfun(x, kk=k):
                hv = x[:n]
                if kk >= 0:
                    m = n - kk
                    val = np.dot(hv[kk:], 1.0 - hv[:m])
                else:
                    ga = -kk
                    m = n - ga
                    val = np.dot(hv[:m], 1.0 - hv[ga:])
                return x[n] - val * dx

            def _cjac(x, kk=k):
                g = _grad_row(x[:n], kk)
                j = -g.copy()
                j[n] = 1.0
                return j

            constraints.append({'type': 'ineq', 'fun': _cfun, 'jac': _cjac})

        bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.05)]
        try:
            res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.x is not None:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    break
        except Exception:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.95

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.3 * np.sin(2.0 * np.pi * 5.0 * x / n0)
        h0 = _project(h0, n0 / 2.0)

    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Quick active descent polish (10% time)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _active_descent(h, d1, K=20, max_iters=80)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Coordinate descent (25% time) - the new main method
    d2 = t0 + budget_s * 0.35
    h2, c5_2 = _coord_desc(best_h, d2, max_sweeps=60, random_order=False, rng=None)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    # Second pass with random order
    if time.time() < d2 - 5:
        d2b = min(d2, time.time() + budget_s * 0.08)
        h2b, c5_2b = _coord_desc(best_h, d2b, max_sweeps=40, random_order=True, rng=rng)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Pair coordinate descent (10% time)
    d3 = t0 + budget_s * 0.45
    h3, c5_3 = _pair_coord_desc(best_h, d3, n_pairs=3000, rng=rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Polish with active descent
    if time.time() < d3 + budget_s * 0.02:
        h3b, c5_3b = _active_descent(best_h, min(d3 + budget_s * 0.02, time.time() + budget_s * 0.02),
                                     K=15, max_iters=50)
