import numpy as np
import time
from scipy.optimize import linprog

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-14:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _compute_gradients(h):
    n = len(h)
    m = 2 * n - 1
    grads = np.zeros((m, n))
    for j in range(m):
        k = j - (n - 1)
        if k >= 0:
            idx = np.arange(n - k)
            if len(idx) > 0:
                grads[j, idx + k] += (1.0 - h[idx])
                grads[j, idx] += (-h[idx + k])
        else:
            q = -k
            idx = np.arange(n - q)
            if len(idx) > 0:
                grads[j, idx] += (1.0 - h[idx + q])
                grads[j, idx + q] += (-h[idx])
    return grads

def _slp_step(h, tau_max=10.0):
    """One SLP iteration. Returns (h_new, success)."""
    n = len(h)
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    grads = _compute_gradients(h)
    
    # LP: min tau s.t. grads[j] @ h_new - tau <= corr[j] - grads[j] @ h
    #     sum(h_new) = n/2, 0 <= h_new <= 1, 0 <= tau <= tau_max
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    rhs = corr - grads @ h
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, tau_max)]
    
    res = linprog(c, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10.0})
    if not res.success:
        return h.copy(), False
    return res.x[:n], True

def _line_search_1d(h0, h1, f0, n_evals=20):
    """Line search along h_t = (1-t)*h0 + t*h1 for t in [0,1]."""
    n = len(h0)
    target = n / 2.0
    best_t = 0.0
    best_f = f0
    for i in range(1, n_evals + 1):
        t = i / n_evals
        ht = (1 - t) * h0 + t * h1
        ht = np.clip(ht, 0.0, 1.0)
        ht = _project(ht, target)
        ft = _exact_c5(ht)
        if ft < best_f:
            best_f = ft
            best_t = t
    return best_t, best_f, _project((1-best_t)*h0 + best_t*h1, target) if best_t > 0 else h0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
