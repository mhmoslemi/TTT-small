def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

            lo = mid
        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

        else:

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_mat(fixed, dx, vary_a):
    """Build coefficient matrix.
    If vary_a=True: A[q,i] = coeff of a[i] in dx*corr(a,fixed)[q]
    If vary_a=False: A[q,j] = coeff of b[j] in dx*corr(fixed,b)[q]
    """
    n = len(fixed)
    A = np.zeros((2*n - 1, n))
    for q in range(2*n - 1):
        m = q - (n - 1)
        if vary_a:
            if m >= 0:
                A[q, m:] = fixed[:n-m]
            else:
                d = -m
                A[q, :n-d] = fixed[d:]
        else:
            if m >= 0:
                A[q, :n-m] = fixed[m:]
            else:
                d = -m
                A[q, d:] = fixed[:n-d]
    A *= dx
    return A

def _solve_lp(A, target):
    n = A.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = np.zeros(A.shape[0])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _alt_step(h, b, dx, target):
    """One alternating step: solve LP for a given b, then for b given a.
    Returns list of candidate (candidate, c5) pairs."""
    n = len(h)
    A_a = _build_mat(b, dx, True)
    a_star = _solve_lp(A_a, target)
    if a_star is None:
        return []
    
    candidates = [(a_star, None)]
    
    A_b = _build_mat(a_star, dx, False)
    b_star = _solve_lp(A_b, target)
    if b_star is not None:
        candidates.append((1.0 - b_star, None))
        for alpha in (0.3, 0.5, 0.7):
            candidates.append((alpha * a_star + (1.0 - alpha) * (1.0 - b_star), None))
    
    return candidates

    h = h.copy()

        b = _make_partner(h, mode, rng, target)

                h = cand.copy()
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
