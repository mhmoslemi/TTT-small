import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection to [0,1]^n with sum = target
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if np.any(mask):
            cnt = int(np.sum(mask))
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(b, dx):
    # Build coefficient matrix for min-max LP
    # A[q, i] = dx * b[i + q - n + 1] if 0 <= i + q - n + 1 < n, else 0
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        row_start = n - 1 - i
        A[row_start:row_start + n, i] = dx * b
    return A

def _solve_minmax_lp(A, dx, target):
    # min tau s.t. A@a - tau*1 <= 0, sum(a)=target, 0<=a<=1, tau>=0
    n = A.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = np.zeros(A.shape[0])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def _bridge_search(hL, hR, n):
    # Search for best point on segment between hL and hR (both have sum n/2)
    target = n / 2.0
    best_c5 = min(_exact_c5(hL), _exact_c5(hR))
    best_h = hL.copy() if best_c5 == _exact_c5(hL) else hR.copy()
    # Golden section search
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    hc = _project((1 - c) * hR + c * hL, target)
    hd = _project((1 - d) * hR + d * hL, target)
    fc = _exact_c5(hc)
    fd = _exact_c5(hd)
    for _ in range(40):
        if b - a < 1e-5:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            hc = _project((1 - c) * hR + c * hL, target)
            fc = _exact_c5(hc)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            hd = _project((1 - d) * hR + d * hL, target)
            fd = _exact_c5(hd)
    # Also check endpoints and midpoint
    for alpha in [0.0, 0.5, 1.0]:
        h_mid = _project((1 - alpha) * hR + alpha * hL, target)
        c5_mid = _exact_c5(h_mid)
        if c5_mid < best_c5:
            best_c5 = c5_mid
            best_h = h_mid
    return best_h, best_c5

def _relax_pair_optimize(h, deadline, mode='complement', rng=None):
    # Relaxed pair LP optimization
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    if rng is None:
        rng = np.random.default_rng(123)
    
    stall = 0
    for iteration in range(50):
        if time.time() > deadline:
            break
        if stall > 15:
            break
        
        # Choose partner b
        if mode == 'complement':
            b = 1.0 - h
            b = _project(b, target)
        elif mode == 'random':
            b = rng.uniform(0, 1, n)
            b = _project(b, target)
        elif mode == 'noisy_complement':
            b = 1.0 - h + rng.normal(0, 0.1, n)
            b = _project(b, target)
        
        # Build matrix and solve for a given b
        A = _build_A(b, dx)
        a_opt = _solve_minmax_lp(A, dx, target)
        if a_opt is None:
            stall += 1
            continue
        a_opt = _project(a_opt, target)
        
        # Evaluate a_opt
        c5_a = _exact_c5(a_opt)
        
        # Now solve for b* given a_opt
        A2 = _build_A(a_opt, dx)
        b_opt = _solve_minmax_lp(A2, dx, target)
        if b_opt is None:
            # Just use a_opt as candidate
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a_opt.copy()
                stall = 0
            else:
                stall += 1
            continue
        b_opt = _project(b_opt, target)
        
        # Build bridge
        uL = a_opt
        uR = _project(1.0 - b_opt, target)
        
        # Search on bridge
        h_bridge, c5_bridge = _bridge_search(uL, uR, n)
        
        # Also try a_opt and uR directly
        c5_uR = _exact_c5(uR)
        
        # Pick best candidate
        candidates = [
            (a_opt, c5_a),
            (uR, c5_uR),
            (h_bridge, c5_bridge),
        ]
        best_candidate = min(candidates, key=lambda x: x[1])
        
        if best_candidate[1] < best_c5 - 1e-12:
            best_c5 = best_candidate[1]
            best_h = best_candidate[0].copy()
            h = best_candidate[0].copy()
            stall = 0
        else:
            # Try updating h to a_opt anyway (might help next iteration)
            h = a_opt.copy()
            stall += 1
    
    return best_h, best_c5

- Move: h[i] -= δ, h[j] += δ (maintains sum)
- Find the best δ that minimizes the objective
- This is a 1D optimization (the objective is a max of linear functions of δ, so it's piecewise linear)

For the 1D optimization, I can use the fact that each f_q(h + δ*d) is linear in δ (where d has -1 at i, +1 at j). So the max of these linear functions is convex in δ, and I can find the minimum by checking the breakpoints.

But this might be slow for large n. Let me just use a simple golden section search on δ.

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(b, dx):
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        row_start = n - 1 - i
        A[row_start:row_start + n, i] = dx * b
    return A

def _solve_minmax(A, target):
    n = A.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = np.zeros(A.shape[0])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def _bridge_search(hL, hR, n, max_iter=35):
    target = n / 2.0
    cL = _exact_c5(hL)
    cR = _exact_c5(hR)
    best_c5 = min(cL, cR)
    best_h = hL.copy() if cL <= cR else hR.copy()
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    hc1 = _project((1.0 - c1) * hR + c1 * hL, target)
    hc2 = _project((1.0 - c2) * hR + c2 * hL, target)
    fc1 = _exact_c5(hc1)
    fc2 = _exact_c5(hc2)
    for _ in range(max_iter):
        if b - a < 1e-6:
            break
        if fc1 < fc2:
            b, c2, fc2 = c2, c1, fc1
            c1 = b - gr * (b - a)
            hc1 = _project((1.0 - c1) * hR + c1 * hL, target)
            fc1 = _exact_c5(hc1)
        else:
            a, c1, fc1 = c1, c2, fc2
            c2 = a + gr * (b - a)
            hc2 = _project((1.0 - c2) * hR + c2 * hL, target)
            fc2 = _exact_c5(hc2)
    for alpha in [0.5, (a + b) / 2.0]:
        hm = _project((1.0 - alpha) * hR + alpha * hL, target)
        cm = _exact_c5(hm)
        if cm < best_c5:
            best_c5 = cm
            best_h = hm
    return best_h, best_c5

def _relax_pair_opt(h, deadline, rng, mode='complement', max_iter=80):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 20:
            break
        if mode == 'complement':
            b = _project(1.0 - h, target)
        elif mode == 'random':
            b = _project(rng.uniform(0, 1, n), target)
        elif mode == 'noisy':
            b = _project(1.0 - h + rng.normal(0, 0.15, n), target)
        else:
            b = _project(h + rng.normal(0, 0.1, n), target)
        A = _build_A(b, dx)
        a_opt = _solve_minmax(A, target)
        if a_opt is None:
            stall += 1
            continue
        a_opt = _project(a_opt, target)
        c5_a = _exact_c5(a_opt)
        A2 = _build_A(a_opt, dx)
        b_opt = _solve_minmax(A2, target)
        if b_opt is None:
            if c5_a < best_c5 - 1e-12:
                best_c5 = c5_a
                best_h = a_opt.copy()
                h = a_opt.copy()
                stall = 0
            else:
                stall += 1
            continue
        b_opt = _project(b_opt, target)
        uL = a_opt
        uR = _project(1.0 - b_opt, target)
        h_bridge, c5_bridge = _bridge_search(uL, uR, n)
        c5_R = _exact_c5(uR)
        best_cand_c5 = min(c5_a, c5_R, c5_bridge)
        if c5_a <= c5_R and c5_a <= c5_bridge:
            best_cand = a_opt
        elif c5_R <= c5_a and c5_R <= c5_bridge:
            best_cand = uR
        else:
            best_cand = h_bridge
        if best_cand_c5 < best_c5 - 1e-12:
            best_c5 = best_cand_c5
            best_h = best_cand.copy()
            h = best_cand.copy()
            stall = 0
        else:
            h = a_opt.copy()
            stall += 1
    return best_h, best_c5

def _coord_polish(h, deadline, rng, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        if stall > 30:
            break
        improved = False
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        n_try = min(30, len(interior) * 3)
        for _ in range(n_try):
            if time.time() > deadline:
                break
            i = interior[rng.integers(len(interior))]
            j = interior[rng.integers(len(interior))]
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            t_lo, t_hi = 0.0, 0.0
            if d[i] < 0:
                t_lo = max(t_lo, -h[i] / d[i])
                t_hi = min(t_hi, (1.0 - h[i]) / d[i])
            else:
                t_lo = max(t_lo, -h[i] / d[i] if d[i] != 0 else 0)
                t_hi = min(t_hi, (1.0 - h[i]) / d[i] if d[i] != 0 else 0.9)
            if d[j] > 0:
                t_lo = max(t_lo, -h[j] / d[j])
                t_hi = min(t_hi, (1.0 - h[j]) / d[j])
            else:
                t_lo = max(t_lo, -h[j] / d[j] if d[j] != 0 else 0)
                t_hi = min(t_hi, (1.0 - h[j]) / d[j] if d[j] != 0 else 0.9)
            if t_hi <= t_lo:
                continue
            S0 = np.correlate(h, 1.0 - h, mode='full') * dx
            T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
            C = np.correlate(d, d, mode='full') * dx
            obj = np.max(S0 + t_hi * T - t_hi * t_hi * C)
            n_grid = 25
            grid = np.linspace(t_lo, t_hi, n_grid)
            vals = np.max(S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :], axis=1)
            best_i = int(np.argmin(vals))
            best_t = grid[best_i]
            best_val = vals[best_i]
            if best_i > 0 and best_i < n_grid - 1:
                a2, b2 = grid[best_i - 1], grid[best_i + 1]
            else:
                a2, b2 = t_lo, t_hi
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1s = b2 - gr * (b2 - a2)
            c2s = a2 + gr * (b2 - a2)
            f1 = float(np.max(S0 + c1s * T - c1s * c1s * C))
            f2 = float(np.max(S0 + c2s * T - c2s * c2s * C))
            for _ in range(25):
                if b2 - a2 < 1e-12:
                    break
                if f1 < f2:
                    b2, c2s, f2 = c2s, c1s, f1
                    c1s = b2 - gr * (b2 - a2)
                    f1 = float(np.max(S0 + c1s * T - c1s * c1s * C))
                else:
                    a2, c1s, f1 = c1s, c2s, f2
                    c2s = a2 + gr * (b2 - a2)
                    f2 = float(np.max(S0 + c2s * T - c2s * c2s * C))
            opt_t = (a2 + b2) / 2.0
            opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
            if opt_val < best_val:
                best_t, best_val = opt_t, opt_val
            if best_val < best_c5 - 1e-12:
                h = h + best_t * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                    stall = 0
        if not improved:
            stall += 1
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _diverse_start(n, rng, mode=0):
    if mode == 0:
        freq = rng.integers(1, 7)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
        h += rng.normal(0, 0.08, n)
    elif mode == 1:
        h = rng.uniform(0, 1, n)
    elif mode == 2:
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, int(rng.integers(1, n // 6 + 1)))
            if rng.random() < 0.5:
                h[pos:pos + bl] = 1.0
            pos += bl
    else:
        x = np.arange(n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.integers(2, 9) * x / n) + 0.15 * np.sin(2 * np.pi * rng.integers(1, 5) * x / n)
        h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)
