def _exact_line_search(h, direction, S0, dx, lo=None, hi=None):
    """Exact line search along direction. direction must have zero mean."""
    n = len(h)
    d = direction - np.mean(direction)

    neg = d < -1e-12

        hi = 1.0

    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)

    C = np.correlate(d, d, 'full')

        a, b = lo, candidates[1]
    elif best_idx == n_eval - 1:
        a, b = candidates[-2], hi
    else:

        b = candidates[min(n_eval - 1, best_idx + 2)]

    c2 = a + gr * (b - a)

def c5_exact(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def row_grads_weighted(h, weights):
    """Compute sum_k weights[k] * dR_k/dh for all lags. Returns (n,) array."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        d = k - (n - 1)
        ad = abs(d)
        m = n - ad
        if m <= 0:
            continue
        if d == 0:
            grad[:m] += wk * (1.0 - 2.0 * h[:m])
        elif d > 0:
            grad[:m] += wk * (1.0 - h[d:d+m])
            grad[d:d+m] += -wk * h[:m]
        else:
            g = -d
            grad[g:] += wk * (1.0 - h[:m])
            grad[:m] += -wk * h[g:g+m]
    return grad

            h_up = _resample(h_f, n)
            c5_up = c5_exact(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()

    # Phase 3: Smooth SLSQP with annealing (38% budget)
    d3 = min(deadline, t0 + budget_s * 0.58)
    target = best_c5 - 0.004 * best_c5
    for tau_frac in [1.0, 0.5, 0.2, 0.08, 0.03]:
        if time.time() > d3:
            break
        tau = 0.035 * best_c5 * tau_frac
        sub_dl = min(d3, time.time() + budget_s * 0.045)
        h_sm, c5_sm = _slsqp_smooth(best_h, target, tau, sub_dl, maxiter=350)
        if c5_sm < best_c5 - 1e-9:
            best_c5 = c5_sm
            best_h = h_sm.copy()
            target = best_c5 - 0.004 * best_c5

    # Phase 4: Gradient direction polish (12% budget)
    d4 = min(deadline, t0 + budget_s * 0.70)
    h_g, c5_g = _gradient_polish(best_h, d4, max_iters=70)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()

    # Phase 5: Pair search (18% budget)
    d5 = min(deadline, t0 + budget_s * 0.88)
    h_ps, c5_ps = _pair_search(best_h, d5, rng, n_pairs=20)
    if c5_ps < best_c5:
        best_c5 = c5_ps
        best_h = h_ps.copy()

    # Phase 6: Binary round + local (7% budget)
    d6 = min(deadline, t0 + budget_s * 0.95)
    h_bi, c5_bi = _binary_local(best_h, d6, rng)
    if c5_bi < best_c5:
        best_c5 = c5_bi
        best_h = h_bi.copy()

    # Phase 7: Final gradient polish
    h_f, c5_f = _gradient_polish(best_h, deadline, max_iters=30)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    # Final SLSQP squeeze with tiny tau
    if time.time() < deadline - 5:
        target_final = best_c5 - 0.002 * best_c5
        h_sf, c5_sf = _slsqp_smooth(best_h, target_final, 0.005 * best_c5, deadline - 2, maxiter=200)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
