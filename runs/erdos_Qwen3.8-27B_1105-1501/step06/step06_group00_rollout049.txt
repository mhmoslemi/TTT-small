def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -radius
    t_hi = radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
    if np.any(neg):
        t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        t_lo = max(t_lo, float(np.min(-h[neg] / (-d[neg]))))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Coarse grid
    n_grid = 50
    grid_t = np.linspace(t_lo, t_hi, n_grid)
    best_idx = 0
    best_val = float('inf')
    for idx in range(n_grid):
        t = grid_t[idx]
        val = float(np.max(S0 + t * T - t * t * C)) * dx
        if val < best_val:
            best_val = val
            best_idx = idx
    
    # Golden section
    lo_b = grid_t[max(0, best_idx - 2)]
    hi_b = grid_t[min(n_grid - 1, best_idx + 2)]

        opt_t = grid_t[best_idx]
        opt_val = best_val
    else:

        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        for _ in range(35):

                c1 = hi_b - gr * (hi_b - lo_b)

            else:
                lo_b = c1; c1 = c2; f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
        opt_t = (lo_b + hi_b) / 2.0
        opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    
    # Check endpoints and grid best

             (t_lo, float(np.max(S0) * dx)),
             (grid_t[best_idx], best_val)]
    best = min(cands, key=lambda x: x[1])
    return best[0], best[1]

def _gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(scores))
    exps = np.exp(np.clip(beta * (scores - M), -60.0, 0.0))
    w = exps / float(np.sum(exps))
    
    # Build G and compute g = G.T @ w
    g = np.zeros(n)
    for d in range(-(n-1), n):
        r = n - 1 + d
        wr = w[r]
        if abs(wr) < 1e-15:
            continue
        if d >= 0:
            m = n - d
            if m > 0:
                g[d:] += wr * (1.0 - h[:m])
                g[:m] += -wr * h[d:]
        else:
            gg = -d
            m = n - gg
            if m > 0:
                g[:m] += wr * (1.0 - h[gg:])
                g[gg:] += -wr * h[:m]
    return g, w, scores
