import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.nan_to_num(v, nan=target / n, posinf=1.0, neginf=0.0)
    v = np.clip(v, -2.0, 2.0)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_raw(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = n - d
        if m > 0:
            g[d:] += (1.0 - h[:m])
            g[:m] -= h[d:]
    else:
        gp = -d
        m = n - gp
        if m > 0:
            g[:m] += (1.0 - h[gp:])
            g[gp:] -= h[:m]
    return g

def _periodic_to_h(pattern, p, n):
    h = np.zeros(n)
    reps = n // p
    if reps > 0:
        h[:reps * p] = np.tile(pattern, reps)
    rem = n - reps * p
    if rem > 0:
        h[reps * p:] = pattern[:rem]
    return h

def _periodic_c5(pattern, p, n):
    h = _periodic_to_h(pattern, p, n)
    return _c5(h)

def _periodic_grad_fd(pattern, p, n):
    eps = 1e-6
    grad = np.zeros(p)
    for i in range(p):
        x_p = pattern.copy()
        x_m = pattern.copy()
        x_p[i] = min(1.0, x_p[i] + eps)
        x_m[i] = max(0.0, x_m[i] - eps)
        grad[i] = (_periodic_c5(x_p, p, n) - _periodic_c5(x_m, p, n)) / (2.0 * eps)
    return grad

def _periodic_lse_obj(x, beta, p, n):
    h = _periodic_to_h(x, p, n)
    C = _corr_scaled(h)
    Cmax = np.max(C)
    z = beta * (C - Cmax)
    z = np.clip(z, -400, 400)
    ez = np.exp(z)
    return Cmax + np.log(np.sum(ez)) / beta

def _periodic_lse_grad(x, beta, p, n):
    h = _periodic_to_h(x, p, n)
    C = _corr_scaled(h)
    Cmax = np.max(C)
    z = beta * (C - Cmax)
    z = np.clip(z, -400, 400)
    ez = np.exp(z)
    s = np.sum(ez)
    weights = ez / s
    nh = len(h)
    dx = 2.0 / nh
    # Gradient of LSE w.r.t. h
    gh = np.zeros(nh)
    for r in range(2 * nh - 1):
        if weights[r] > 1e-7:
            d = r - (nh - 1)
            gh += weights[r] * _grad_raw(h, d)
    gh *= dx
    # Chain rule: dh[rep*p + j]/dpattern[j] = 1
    # So gradient w.r.t. pattern[j] = sum over all positions j + rep*p
    gp = np.zeros(p)
    for j in range(p):
        positions = np.arange(j, nh, p)
        gp[j] = np.sum(gh[positions])
    return gp

def _solve_periodic_slsqp(pattern_init, p, n, beta, maxiter):
    target_p = p / 2.0
    x0 = _project(pattern_init.copy(), target_p)
    bounds = [(0.0, 1.0)] * p
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_p,
             'jac': lambda x: np.ones(p)}]

    def obj(x):
        return _periodic_lse_obj(x, beta, p, n)

    def jac(x):
        return _periodic_lse_grad(x, beta, p, n)

    try:
        res = minimize(obj, x0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        x_opt = _project(res.x, target_p)
        return x_opt, _periodic_c5(x_opt, p, n)
    except Exception:
        return x0, _periodic_c5(x0, p, n)

def _solve_full_lse(h_init, beta, maxiter):
    n = len(h_init)
    target = n / 2.0
    x0 = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        C = _corr_scaled(x)
        Cmax = np.max(C)
        z = beta * (C - Cmax)
        z = np.clip(z, -400, 400)
        ez = np.exp(z)
        return Cmax + np.log(np.sum(ez)) / beta

    def jac(x):
        C = _corr_scaled(x)
        Cmax = np.max(C)
        z = beta * (C - Cmax)
        z = np.clip(z, -400, 400)
        ez = np.exp(z)
        s = np.sum(ez)
        w = ez / s
        dx = 2.0 / n
        g = np.zeros(n)
        for r in range(2 * n - 1):
            if w[r] > 1e-7:
                d = r - (n - 1)
                g += w[r] * _grad_raw(x, d)
        g *= dx
        return g

    try:
        res = minimize(obj, x0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        x_opt = _project(res.x, target)
        return x_opt, _c5(x_opt)
    except Exception:
        return x0, _c5(x0)

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Cmat = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _c5(h)
    gphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gphi * (b - a)
    c2 = a + gphi * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Cmat)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Cmat)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gphi * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Cmat)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gphi * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Cmat)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * Cmat)) * dx
    return opt_t, opt_val

def _active_direction(h, n_active, radius):
    n = len(h)
    dx = 2.0 / n
    scores = _corr_scaled(h)
    max_s = float(np.max(scores))
    idx = np.argsort(-scores)[:n_active]
    g = np.zeros(n)
    for r in idx:
        d = r - (n - 1)
        g += _grad_raw(h, d)
    g *= dx / len(idx)
    g -= np.mean(g)
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    d = -g / norm * radius
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        sc = np.min((1.0 - h[pos]) / d[pos])
        d[pos] *= min(1.0, sc)
    if np.any(neg):
        sc = np.min(h[neg] / (-d[neg]))
        d[neg] *= min(1.0, sc)
    d -= np.mean(d)
    return d

def _random_dir(n, rng, frac=0.25):
    d = np.zeros(n)
    k = max(2, int(n * frac))
    idx = rng.choice(n, size=k, replace=False)
    d[idx] = rng.normal(0, 1, k)
    d -= np.mean(d)
    return d

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _periodic_init_full(n, p, duty):
    h = np.zeros(n)
    pattern = np.zeros(p)
    fill = max(1, int(duty * p))
    pattern[:fill] = 1.0
    reps = n // p
    h[:reps * p] = np.tile(pattern, reps)
    h[reps * p:] = pattern[:n - reps * p]
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _optimize_all_periods(n, deadline, max_betas=3):
