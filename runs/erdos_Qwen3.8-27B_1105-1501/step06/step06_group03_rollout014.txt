import numpy as np
from scipy.optimize import linprog
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def exact_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_rows_batch(h, indices, n):
    dx = 2.0 / n
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[:m] += dx * (1.0 - h[k:])
                g[k:] += -dx * h[:m]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += -dx * h[d:]
                g[d:] += dx * (1.0 - h[:m])
    return G

def line_search(h, p, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = -np.correlate(h, p, mode='full') + np.correlate(p, 1.0 - h, mode='full')
    C = -np.correlate(p, p, mode='full')
    pos = p > 1e-14
    neg = p < -1e-14
    t_lo = 0.0
    t_hi = trust_radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / p[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0)) * dx
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] + (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a = grid[best_i - 1]
        b = grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * C)) * dx
    for _ in range(80):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def lp_direction(h, S, n, active_idx, radius):
    G = grad_rows_batch(h, active_idx, n)
    n_act = len(active_idx)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = 0.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            p = res.x[:n]
            if float(np.max(np.abs(p))) > 1e-12:
                return p
    except Exception:
        pass
    return None

def soft_max_gradient(h, S, n, theta, top_k):
    M = float(S.max())
    shifted = (S - M) / theta
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return None
    w = w / w_sum
    top_idx = np.argsort(-S)[:top_k]
    G = grad_rows_batch(h, top_idx, n)
    g_smooth = w[top_idx] @ G
    g_smooth -= g_smooth.mean()
    if np.abs(g_smooth).max() < 1e-14:
        return None
    norm = np.linalg.norm(g_smooth)
    if norm < 1e-14:
        return None
    return -g_smooth / norm

def freq_direction(h, n, freq, step_max):
    i = np.arange(n)
    d = np.sin(2.0 * np.pi * freq * i / n)
    d -= d.mean()
    d_max = float(np.max(np.abs(d)))
    if d_max < 1e-14:
        return None
    d = d / d_max * step_max
    return d

def pair_optimize(h, i, j, dx):
    """Optimize pair (i,j) keeping sum constant. Returns best (alpha, c5)."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    p = np.zeros(n)
    p[i] = 1.0
    p[j] = -1.0
    T = -np.correlate(h, p, mode='full') + np.correlate(p, 1.0 - h, mode='full')
    C = -np.correlate(p, p, mode='full')
    t_lo = 0.0
    t_hi = 0.5
    if h[i] > h[j]:
        t_hi = min(t_hi, (h[i] - h[j]) / 2.0)
        t_lo = -min(h[j], 1.0 - h[i])
        t_hi = min(t_hi, (1.0 - h[i]) / 2.0 + h[i] - 0.5)
    t_lo = max(t_lo, -h[j])
    t_hi = min(t_hi, 1.0 - h[i])
    t_lo = max(t_lo, h[i] - 1.0)
    t_hi = min(t_hi, -h[j])
    # Simpler: just use ray search
    if h[i] >= h[j]:
        t_max = (h[i] - h[j]) / 2.0 + 0.25
        t_max = min(t_max, 0.4)
    else:
        t_max = 0.4
    t_lo2 = -0.4
    t_hi2 = t_max
    grid = np.linspace(t_lo2, t_hi2, 40)
    best_t = 0.0
    best_v = float(np.max(S0))
    for t in grid:
        new_h = h.copy()
        new_h[i] += t
        new_h[j] -= t
        if new_h[i] < -0.01 or new_h[i] > 1.01 or new_h[j] < -0.01 or new_h[j] > 1.01:
            continue
        v = S0 + t * T + t * t * C
        mv = float(np.max(v))
        if mv < best_v:
            best_v = mv
            best_t = t
    return best_t, best_v * dx

def local_optimize(h0, deadline, radius_init=0.10, K=15, use_soft=True):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    rng = np.random.default_rng(12345)
    theta = 0.005 * max(cur_c5, 0.1)
    top_k = min(25, max(12, n // 5))
    while time.time() < deadline:
        if stall > 80:
            h = h + rng.normal(0, 0.02 + 0.005 * stall, n)
            h = project(h, target)
            cur_c5 = exact_c5(h)
            radius = 0.08
            stall = 0
            if stall > 200:
                break
        S = exact_scores(h)
        M = float(S.max())
        gap = M - float(S.min())
        tol = max(0.1 * gap, 1e-6)
        near_idx = np.where(S >= M - tol)[0]
        if len(near_idx) > K:
            near_idx = np.argsort(-S)[:K]
        elif len(near_idx) < 5:
            near_idx = np.argsort(-S)[:min(10, 2 * n - 1)]
        directions = []
        p_lp = lp_direction(h, S, n, near_idx, radius)
        if p_lp is not None:
            directions.append(p_lp)
        if use_soft:
            p_soft = soft_max_gradient(h, S, n, theta, top_k)
            if p_soft is not None:
                directions.append(p_soft * radius)
        top5 = np.argsort(-S)[:5]
        for r in top5[:3]:
            G_r = grad_rows_batch(h, [r], n)[0]
            G_r -= G_r.mean()
            if np.abs(G_r).max() > 1e-14:
                p_r = -G_r / np.abs(G_r).max() * radius
                directions.append(p_r)
        if rng.random() < 0.3:
            d_rand = rng.normal(0, 1, n)
            d_rand -= d_rand.mean()
            d_rand = d_rand / np.abs(d_rand).max() * radius
            directions.append(d_rand)
        best_step = None
        best_new_c5 = cur_c5
        for p in directions:
            if time.time() > deadline:
                break
            p = p - p.mean()
            if np.abs(p).max() < 1e-14:
                continue
            alpha, new_c5 = line_search(h, p, 2.0 / n, radius * 1.5)
            if new_c5 < best_new_c5 - 1e-14:
                best_new_c5 = new_c5
                best_step = (alpha, p)
        if best_step is not None:
            alpha, p = best_step

                cur_c5 = actual_c5

                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                ratio = abs(alpha) / max(radius * 1.5, 1e-12)
                if ratio > 0.7:
                    radius = min(radius * 1.1, 0.3)
                elif ratio > 0.3:
                    radius = min(radius * 1.03, 0.25)
                else:
                    radius = max(radius * 0.92, 0.02)
                theta *= 0.8
                theta = max(theta, 5e-5)
            else:
                stall += 1
                radius *= 0.78
                if stall % 8 == 0:
                    radius = max(radius, 0.04)
        else:
            stall += 1
            radius *= 0.75
            if stall % 15 == 0 and radius < 0.02:

    return best_h, best_c5

def pair_refine(h0, deadline, max_passes=300):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(999)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        S = exact_scores(h)
        M = float(S.max())
        improved = False
        n_try = min(n, 50)
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        if len(interior) >= 4:
            candidates = rng.choice(len(interior), size=min(n_try, len(interior)), replace=False)
            cand_idx = interior[candidates]
        else:
            cand_idx = np.argsort(-np.abs(h - 0.5))[:n_try]
        for ii in range(len(cand_idx)):
            if time.time() > deadline:
                break
            i = cand_idx[ii]
            for jj in range(ii + 1, min(ii + 12, len(cand_idx))):
                j = cand_idx[jj]
                p = np.zeros(n)
                if h[i] > h[j]:
                    p[i] = -1.0
                    p[j] = 1.0
                else:
                    p[i] = 1.0
                    p[j] = -1.0
                alpha, c5_try = line_search(h, p, dx, 0.4)
                if c5_try < best_c5 - 1e-13:
                    h_new = h + alpha * p
                    h_new = project(h_new, target)
                    actual = exact_c5(h_new)
                    if actual < best_c5 - 1e-13:
                        h = h_new
                        best_c5 = actual
                        best_h = h.copy()
                        improved = True
        if not improved and _pass > 20:
            break
    return best_h, best_c5

def freq_sweep(h0, deadline):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for freq in range(1, n // 2 + 1):
        if time.time() > deadline:
            break
        for amp in [0.01, 0.03, 0.05, 0.08, 0.12]:
            d = freq_direction(h, n, freq, amp)
            if d is None:
                continue
            alpha, c5_try = line_search(h, d, dx, amp * 2)
            if c5_try < best_c5 - 1e-13:
                h_new = h + alpha * d
                h_new = project(h_new, target)
                actual = exact_c5(h_new)
                if actual < best_c5 - 1e-13:
                    h = h_new
                    best_c5 = actual
                    best_h = h.copy()
    return best_h, best_c5

def binarize_push(h0, deadline):
    """Try to push values toward binary while maintaining sum."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    rng = np.random.default_rng(777)
    for _ in range(500):
        if time.time() > deadline:
            break
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        if len(interior) < 2:
            break
        i = interior[rng.integers(len(interior))]
        target_val = 1.0 if h[i] > 0.5 else 0.0
        delta = target_val - h[i]
        if abs(delta) < 0.01:
            continue
        others = np.where(np.arange(n) != i)[0]
        if delta > 0:
            mask = h[others] > 0.05
        else:
            mask = h[others] < 0.95
        if not np.any(mask):
            continue
        j_candidates = others[mask]
        j = j_candidates[rng.integers(len(j_candidates))]
        p = np.zeros(n)
        p[i] = 1.0
        p[j] = -1.0
        alpha_needed = delta
        alpha, c5_try = line_search(h, p, dx, max(0.3, abs(alpha_needed) + 0.1))
        if c5_try < best_c5 - 1e-13:
            h_new = h + alpha * p
            h_new = project(h_new, target)
            actual = exact_c5(h_new)
            if actual < best_c5 - 1e-13:
                h = h_new
                best_c5 = actual
                best_h = h.copy()
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def multi_res_sweep(best_h, best_c5, deadline, rng):
    base_n = len(best_h)
    tried = set()
    offsets = [0]
    for d in range(2, 30, 2):
        offsets.append(-d)
        offsets.append(d)
    for dn in offsets:
        if time.time() > deadline:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample(best_h, n_try)
        c5_r = exact_c5(h_r)
        if c5_r < best_c5 + 0.01:
            sub_dl = min(deadline, time.time() + 15.0)
            h_r, c5_r = local_optimize(h_r, sub_dl, radius_init=0.08, K=12, use_soft=True)
            if c5_r < best_c5 + 0.003 and time.time() < deadline - 10:
                sub_dl2 = min(deadline, time.time() + 10.0)
                h_r, c5_r = pair_refine(h_r, sub_dl2, max_passes=100)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()
    return best_h, best_c5

def sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project(h, n / 2.0)
