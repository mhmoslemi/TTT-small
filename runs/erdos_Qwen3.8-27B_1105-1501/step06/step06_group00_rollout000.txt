def _project(v, target):
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0

        Q[n-1+abs_d] = 1.0
        Q[n-1-abs_d] = 1.0
    
    # Now minimize max_k (S_k + alpha*T_k - alpha^2*Q_k) over alpha in [a_lo, a_hi]
    # This is max of (2n-3) affine functions and 2 concave quadratics
    # Strategy: sample densely, then refine
    
    # Coarse grid
    n_grid = 40
    alphas = np.linspace(a_lo, a_hi, n_grid)
    vals = np.empty(n_grid)
    for idx_a in range(n_grid):
        a = alphas[idx_a]
        vals[idx_a] = float(np.max(S + a * T - a * a * Q))
    
    best_idx = int(np.argmin(vals))
    
    # Refine around best
    lo_b = alphas[max(0, best_idx-2)]
    hi_b = alphas[min(n_grid-1, best_idx+2)]
    if hi_b - lo_b < 1e-10:
        opt_a = alphas[best_idx]
    else:
        # Golden section on the refined interval
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi_b - gr * (hi_b - lo_b)
        c2 = lo_b + gr * (hi_b - lo_b)
        f1 = float(np.max(S + c1 * T - c1 * c1 * Q))
        f2 = float(np.max(S + c2 * T - c2 * c2 * Q))
        for _ in range(30):
            if hi_b - lo_b < 1e-12:
                break
            if f1 < f2:
                hi_b = c2
                c2 = c1
                f2 = f1
                c1 = hi_b - gr * (hi_b - lo_b)
                f1 = float(np.max(S + c1 * T - c1 * c1 * Q))
            else:
                lo_b = c1
                c1 = c2
                f1 = f2
                c2 = lo_b + gr * (hi_b - lo_b)
                f2 = float(np.max(S + c2 * T - c2 * c2 * Q))
        opt_a = (lo_b + hi_b) / 2.0
    
    # Also check endpoints
    candidates = [opt_a, a_lo, a_hi]
    best_a = min(candidates, key=lambda a: float(np.max(S + a*T - a*a*Q)))
    
    new_c5 = float(np.max(S + best_a * T - best_a * best_a * Q)) * dx
    return best_a, new_c5

def _pair_sweep(h, deadline, max_pairs=200, beta=20.0, radius=0.12, stall_limit=80):
    """Iterative pair-based local optimization."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    S = _scores(h)
    cur_c5 = float(S.max() * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    rng = np.random.default_rng(777)
    
    while time.time() < deadline:
        if stall > stall_limit:
            break
        
        # Compute smooth gradient
        grad, S, w, cur_c5 = _smooth_grad(h, beta)
        
        # Candidate pairs: high grad (source) → low grad (sink)
        # We want to transfer from low h to high h? No:
        # grad[i] > 0 means increasing h[i] decreases the objective
        # So we want to increase h[i] (source of mass to i) and decrease h[j] (source from j)
        # Transfer: h[i] += alpha, h[j] -= alpha
        # This is good if grad[i] > grad[j]
        
        # Select top sources and sinks
        n_sel = min(16, n // 6)
        top_idx = np.argsort(grad)[-n_sel:]  # highest grad
        bot_idx = np.argsort(grad)[:n_sel]   # lowest grad
        
        # Try all pairs from top (as i, receiver) and bot (as j, donor)
        best_alpha = 0.0
        best_c5_trial = cur_c5
        found = False
        
        for i in top_idx:
            for j in bot_idx:
                if i == j:
                    continue
                alpha, c5_try = _pair_line_search(h, int(i), int(j), S, dx, radius)
                if c5_try < best_c5_trial:
                    best_c5_trial = c5_try
                    best_alpha = alpha
                    best_pair = (int(i), int(j))
                    found = True
        
        # Also try a few random pairs
        for _ in range(min(8, n // 20)):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            alpha, c5_try = _pair_line_search(h, i, j, S, dx, radius)
            if c5_try < best_c5_trial:
                best_c5_trial = c5_try
                best_alpha = alpha
                best_pair = (i, j)
                found = True
        
        if found and best_c5_trial < cur_c5 - 1e-13:
            h[best_pair[0]] += best_alpha
            h[best_pair[1]] -= best_alpha
            h = np.clip(h, 0.0, 1.0)
            h = _project(h, target)
            S = _scores(h)
            cur_c5 = float(S.max() * dx)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            radius *= 0.85
            if radius < 0.005:
                radius = 0.08  # reset
            beta = max(8.0, beta * 0.95)
    
    return best_h, best_c5

def _binary_sa(h_bin, deadline, max_iter=20000, T_init=0.003):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(31337)
    T = T_init
    stall = 0
    ones = list(np.where(cur > 0.5)[0])
    zeros = list(np.where(cur <= 0.5)[0])
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.9997
        else:
            stall += 1
    return best_h, best_c5

def _to_binary(h, n):
    idx = np.argsort(h)[-n // 2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
