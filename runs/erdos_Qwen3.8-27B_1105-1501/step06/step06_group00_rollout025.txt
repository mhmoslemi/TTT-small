One more thought: the C vector for the line search. I established:
- C[n-1] = 2 (for D=0)
- C[n-1+(i-j)] = -1 (for D=i-j, if i≠j)
- C[n-1+(j-i)] = -1 (for D=j-i, if i≠j)

But wait, I need to double-check the sign convention. The formula is:
c_D(h') = c_D(h) + α·T[D] - α²·C[D]

For D=0: the quadratic term is -2α² (as computed above). So -α²·C[0] = -2α², meaning C[0] = 2. ✓

For D = j-i > 0: the quadratic term is +α². So -α²·C = +α², meaning C = -1. ✓

So in the line search, the function to minimize is:
f(α) = dx * max_D (S[D] + α·T[D] - α²·C[D])

where C has positive entries (at D=0) and negative entries (at D=±|i-j|).

The terms with C>0 (concave in α) and C<0 (convex in α) can have complex interaction. The golden section search should handle this fine since we're just evaluating the function.

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    # Water-filling projection
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = float(np.sum(np.clip(v - mid, 0, 1)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1) if res > 0 else (x > 0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _compute_scores(h, S, gamma):
    """Compute gradient-like score vector using soft-max."""
    n = len(h)
    dx = 2.0 / n
    v = S * dx
    v_max = float(np.max(v))
    log_w = gamma * (v - v_max)
    # Clip for numerical stability
    log_w = np.clip(log_w, -50, 0)
    w = np.exp(log_w)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros(2*n-1)
        w[np.argmax(S)] = 1.0
    else:
        w /= w_sum
    
    # Compute g = dx * G.T @ w
    # G[D,a] for D>=0: (1-h[a+D])*1[a<=n-1-D] - h[a-D]*1[a>=D]
    # G[D,a] for D<0 (p=-D): (1-h[a-p])*1[a>=p] - h[a+p]*1[a<=n-1-p]
    
    g = np.zeros(n)
    # Positive D terms
    for d in range(0, n):
        m = n - d
        r = n - 1 + d
        wd = w[r]
        if wd > 1e-15:
            g[:m] += wd * (1.0 - h[d:])
            g[d:] += -wd * h[:m]
    # Negative D terms
    for p in range(1, n):
        m = n - p
        r = n - 1 - p
        wd = w[r]
        if wd > 1e-15:
            g[p:] += wd * (1.0 - h[:m])
            g[:m] += -wd * h[p:]
    
    g *= dx
    return g

def _pair_direction_T(h, i, j, G):
    """Compute T = G[:,i] - G[:,j]"""
    return G[:, i] - G[:, j]

def _line_search(h, S, G, i, j, radius):
    """Exact line search for direction e_i - e_j"""
    n = len(h)
    dx = 2.0 / n
    
    a_min = max(-h[i], h[j] - 1.0, -radius)
    a_max = min(1.0 - h[i], h[j], radius)
    
    if a_min >= a_max - 1e-14:
        return 0.0, float(np.max(S) * dx)
    
    T = G[:, i] - G[:, j]
    
    # C vector (at most 3 nonzeros)
    # Instead of building full C, use the fact that:
    # S + a*T - a^2*C where C[n-1]=2, C[n-1±di]=-1
    di = i - j
    
    def f_of_alpha(a):
        vec = S + a * T
        vec[n-1] -= 2.0 * a * a
        if di != 0:
            vec[n-1+di] += a * a
            vec[n-1-di] += a * a
        return float(np.max(vec)) * dx
    
    # Grid search
    n_grid = 33
    alphas = np.linspace(a_min, a_max, n_grid)
    vals = np.array([f_of_alpha(a) for a in alphas])
    best_idx = int(np.argmin(vals))
    
    # Golden section refinement
    lo = alphas[max(0, best_idx - 2)]
    hi = alphas[min(n_grid-1, best_idx + 2)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = f_of_alpha(c1)
    f2 = f_of_alpha(c2)
    
    for _ in range(30):
        if hi - lo < 1e-10:
            break
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = f_of_alpha(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = f_of_alpha(c2)
    
    opt_a = (lo + hi) / 2.0
    opt_val = f_of_alpha(opt_a)
    
    # Check boundaries and 0
    candidates = [(0.0, float(np.max(S) * dx)), (a_min, f_of_alpha(a_min)), (a_max, f_of_alpha(a_max)), (opt_a, opt_val)]
    return min(candidates, key=lambda x: x[1])

def _build_G(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for d in range(0, n):
        m = n - d
        G[n-1+d, :m] += (1.0 - h[d:])
        G[n-1+d, d:] += (-h[:m])
    for p in range(1, n):
        m = n - p
        G[n-1-p, p:] += (1.0 - h[:m])
        G[n-1-p, :m] += (-h[p:])
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _to_binary(h):
    n = len(h)
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin
