def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    # x = [d_0, ..., d_{n-1}, t]

    d -= np.mean(d)

        return None
    
    return d

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)

def compute_grad_matrix(h):

    offsets = [-4, 4, -6, 6, -8, 8, -2, 2, -10, 10]
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.025)
        h_r, c5_r = _trust_region_optimize(h_r, sub_dl, rho_init=0.08)
        if c5_r > best_c5:
            continue
        sub_dl2 = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r2 = _pair_search(h_r, 12, sub_dl2)
        if c5_r2 < best_c5 - 1e-10:
            best_c5 = c5_r2
            best_h = h_r.copy()
            best_n = n_try

    # Phase 4: SA pair search
    d4 = t0 + budget_s * 0.82
    h4, c5_4 = _sa_pair_search(best_h, d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Gradient polish
    d5 = t0 + budget_s * 0.88
    h5, c5_5 = _gradient_polish(best_h, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final trust region with smaller rho
    d6 = t0 + budget_s * 0.94
    h6, c5_6 = _trust_region_optimize(best_h, d6, rho_init=0.04)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final polish
    d7 = deadline
    h7, c5_7 = _gradient_polish(best_h, d7)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
