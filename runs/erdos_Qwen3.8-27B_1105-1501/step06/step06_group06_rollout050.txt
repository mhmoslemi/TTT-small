def _project(v, target):
    # Project onto {x: 0 <= x[i] <= 1, sum(x) = target}
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target: lo = mid
        else: hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0: mask = x < 1.0
        else: mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

            stall += 5
            if stall > 50:
                radius *= 0.6
                if radius < 1e-5:
                    break
            continue
        d = d / d_norm * radius
        alpha, new_c5 = _exact_line_search(h, d, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            if abs(alpha) > 0.7 * radius:
                radius = min(radius * 1.05, 0.25)
            else:
                radius = min(radius * 1.01, 0.20)
        else:
            stall += 1
            radius *= 0.7
            if stall > 60:
                h = _project(h + 0.015 * rng.standard_normal(n), target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_polish(h0, deadline, max_outer=12):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    for outer in range(max_outer):
        if time.time() > deadline:
            break
        corr = _all_corr(h)
        top_k = np.argsort(-corr)[:25]

        def obj_fn(x):
            return x[n]

        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        def ineq_fn(x):
            hx = x[:n]
            c = np.correlate(hx, 1.0 - hx, mode='full') * (2.0 / n)
            return x[n] - c[top_k]

        def ineq_jac(x):
            hx = x[:n]
            J = np.zeros((len(top_k), n + 1))
            for i, k in enumerate(top_k):
                s = k - (n - 1)
                J[i, :n] = -_grad_lag(hx, s)
                J[i, n] = 1.0
            return J

        def eq_fn(x):
            return np.sum(x[:n]) - target

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

        t_init = cur_c5 + 1e-5
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        ]
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-13})
        if res.success:
            h_new = _project(res.x[:n], target)
            c5_new = _exact_c5(h_new)
            if c5_new < cur_c5 - 1e-13:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break
        else:
            break
    return best_h, best_c5

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
    h = h + rng.normal(0, 0.04, n)
    return _project(h, n / 2.0)

def _random_init(n, rng):
    h = 0.5 + 0.3 * rng.standard_normal(n)
    return _project(h, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 4 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0_raw = np.array(initial_h_values, dtype=float)
    except NameError:
        h0_raw = _block_init(122, rng)

    n0 = len(h0_raw)
    h = _project(h0_raw, n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n0

    # Phase 1: Smooth descent on initial (25% time)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _smooth_descent(h, d1, rng, beta0=40.0)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: Active polish (15% time)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _active_polish(best_h, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: SLSQP polish (10% time)
    d3 = t0 + budget_s * 0.50
    h3, c5_3 = _slsqp_polish(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Multi-resolution exploration (25% time)
    d4 = t0 + budget_s * 0.75
    offsets = list(range(-20, 21, 4))
    offsets = [o for o in offsets if (best_n + o) >= 80 and (best_n + o) <= 180]
    rng2 = np.random.default_rng(seed + 1)
    rng2.shuffle(offsets)
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 60 or n_try > 200 or n_try % 2 != 0:
            continue
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.03)
        h_r, c5_r = _smooth_descent(h_r, sub_dl, rng2, beta0=50.0, max_iter=5000)
        if time.time() < d4 - 5:
            sub_dl2 = min(d4, time.time() + budget_s * 0.02)
            h_r, c5_r = _active_polish(h_r, sub_dl2, max_iter=500)
        if c5_r < best_c5 - 1e-13:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 5: Multi-start (15% time)
    d5 = t0 + budget_s * 0.90
    rng3 = np.random.default_rng(seed + 2)
    starts = []
    for freq in [1, 2, 3, 5]:
        for ph in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0]:
            starts.append(_sine_init(best_n, freq, ph, rng3))
    for _ in range(6):
        starts.append(_random_init(best_n, rng3))
    for _ in range(3):
        starts.append(_block_init(best_n, rng3))
    rng3.shuffle(starts)
    for s in starts:
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_s, c5_s = _smooth_descent(s, sub_dl, rng3, beta0=50.0, max_iter=3000)
        if c5_s < best_c5 - 1e-13:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 6: Final combined polish (10% time)
    d6 = t0 + budget_s * 0.97
    h6, c5_6 = _active_polish(best_h, min(d6, time.time() + budget_s * 0.04), max_iter=1000)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
        best_n = len(h6)
    if time.time() < d6:
        h6b, c5_6b = _slsqp_polish(best_h, d6, max_outer=8)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
            best_n = len(h6b)

    # Final line-search polish
    if time.time() < deadline - 5:
        corr = _all_corr(best_h)
        M = float(corr.max())
        active = np.where(corr > M - 1e-3)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:15]
        elif len(active) > 20:
            active = active[np.argsort(-corr[active])[:20]]
        w = corr[active] - (M - 1e-3)
        w = np.maximum(w, 0)
        if w.sum() > 0:
            w = w / w.sum()
        d_final = np.zeros(best_n)
        for i, k in enumerate(active):
            s_val = k - (best_n - 1)
            d_final -= w[i] * _grad_lag(best_h, s_val)
        d_final = d_final - np.mean(d_final)
        if np.linalg.norm(d_final) > 1e-12:
            d_final = d_final / np.linalg.norm(d_final)
            alpha_f, c5_f = _exact_line_search(best_h, d_final, 0.15)
            if c5_f < best_c5:
                h_f = best_h + alpha_f * d_final
                best_h = _project(h_f, len(h_f) / 2.0)
                best_c5 = _exact_c5(best_h)

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
