def _project(v, target):
    # Project v onto {0 <= x <= 1, sum(x) = target}
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(x)
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-10
        else:
            mask = x > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _mw_direction(h, S, K=30, iters=20):
    """Compute Chebyshev descent direction using multiplicative weights."""
    n = len(h)
    m = 2 * n - 1
    
    # Identify active set
    sorted_idx = np.argsort(-S)[:K]
    active_S = S[sorted_idx]
    spread = max(active_S[0] - active_S[-1], 1e-10)
    
    weights = np.ones(K) / K
    best_dir = None
    best_obj = np.inf
    
    for _ in range(iters):
        # Compute weighted gradient
        g = np.zeros(n)
        for wi, idx in zip(weights, sorted_idx):
            k = int(idx) - (n - 1)
            g += wi * _grad_for_lag(h, k)
        
        # Project to sum-zero
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        
        # Evaluate objective in direction d
        obj = 0.0
        for idx in sorted_idx:
            k = int(idx) - (n - 1)
            gk = _grad_for_lag(h, k)
            obj = max(obj, float(np.dot(gk, d)))
        
        if obj < best_obj:
            best_obj = obj
            best_dir = d
        
        # Update weights (increase weight for violated constraints)
        violations = np.zeros(K)
        for i, idx in enumerate(sorted_idx):
            k = int(idx) - (n - 1)
            gk = _grad_for_lag(h, k)
            violations[i] = float(np.dot(gk, d))
        
        eta = 2.0  # learning rate
        weights *= np.exp(eta * violations)
        weights /= np.sum(weights)
    
    return best_dir

        if m <= 0:
            return g
        idx1 = np.arange(0, m)
        g[idx1] += (1.0 - h[idx1 + ga])
        idx2 = np.arange(ga, n)
        g[idx2] -= h[idx2 - ga]
    return g


def _line_min(S, T, Q, t_lo, t_hi):
    if t_hi - t_lo < 1e-14:
        return t_lo, float(np.max(S))
    n_grid = 25
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = np.max(S[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :], axis=1)
    best_i = int(np.argmin(vals))
    best_t = grid[best_i]
    best_v = float(vals[best_i])
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S + c1 * T - c1 * c1 * Q))
    f2 = float(np.max(S + c2 * T - c2 * c2 * Q))
    for _ in range(35):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S + c1 * T - c1 * c1 * Q))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S + c2 * T - c2 * c2 * Q))
    ot = (a + b) / 2.0
    ov = float(np.max(S + ot * T - ot * ot * Q))
    if ov < best_v:
        best_t, best_v = ot, ov
    return best_t, best_v


def _mw_direction(h, S, K, n_iters):
    n = len(h)
    sorted_idx = np.argsort(-S)[:K]
    grads = np.zeros((K, n))
    for i, idx in enumerate(sorted_idx):
        k = int(idx) - (n - 1)
        grads[i] = _grad_lag(h, k)
    grads -= grads.mean(axis=1, keepdims=True)
    weights = np.ones(K) / K
    best_dir = None
    best_obj = np.inf
    for _ in range(n_iters):
        g = weights @ grads
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        objs = grads @ d
        obj = float(np.max(objs))
        if obj < best_obj:
            best_obj = obj
            best_dir = d
        eta = 1.8
        weights *= np.exp(eta * objs)
        ws = np.sum(weights)
        if ws < 1e-300:
            weights = np.ones(K) / K
        else:
            weights /= ws
    if best_dir is None:
        best_dir = -grads[0] / (np.linalg.norm(grads[0]) + 1e-15)
    return best_dir


def _gradient_descent(h, deadline, K=35, max_iters=500):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    S = np.correlate(h, 1.0 - h, mode='full')
    step = 1.0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        d = _mw_direction(h, S, K, n_iters=12)
        pos = d > 1e-14
        neg = d < -1e-14
        t_hi = step
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            t_hi = min(t_hi, float(np.min(h[neg] / (-d[neg]))))
        t_hi = max(0.0, min(t_hi, 2.0))
        if t_hi < 1e-12:
            step *= 0.5
            stall += 1
            if stall > 20:
                break
            continue
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t_opt, v_opt = _line_min(S, T, Q, 0.0, t_hi)
        if t_opt < 1e-12:
            step *= 0.6
            stall += 1
            if stall > 20:
                break
            continue
        h_new = h + t_opt * d
        h_new = _project(h_new, target)
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            S = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            if t_opt > 0.7 * step:
                step = min(step * 1.1, 2.0)
            else:
                step = max(step * 0.8, 0.05)
        else:
            step *= 0.65
            stall += 1
            if stall > 30:
                break
    return best_h, best_c5


def _pair_local_search(h, deadline, max_passes=50):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    S = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(S)) * dx
    best_h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        active_mask = S > (np.max(S) - 0.02 * np.max(S) + 0.001)
        active_idx = np.where(active_mask)[0]
        pair_score = np.zeros(n)
        for idx in active_idx[:min(len(active_idx), 50)]:
            k = int(idx) - (n - 1)
            g = _grad_lag(h, k)
            pair_score += np.abs(g)
        order = np.argsort(-pair_score)
        n_try = min(n, 60)
        candidates = order[:n_try]
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                for sign in (1, -1):
                    d = np.zeros(n)
                    d[j] = float(sign)
                    d[i] = -float(sign)
                    if sign > 0:
                        t_max = min(h[i], 1.0 - h[j])
                    else:
                        t_max = min(h[j], 1.0 - h[i])
                    if t_max < 1e-10:
                        continue
                    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
                    Q = np.correlate(d, d, mode='full')
                    t_opt, v_opt = _line_min(S, T, Q, 0.0, t_max)
                    if v_opt * dx < best_c5 - 1e-13:
                        h = h + t_opt * d
                        h = _project(h, target)
                        S = np.correlate(h, 1.0 - h, mode='full')
                        cur = float(np.max(S)) * dx
                        if cur < best_c5:
                            best_c5 = cur
                            best_h = h.copy()
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _basin_hopping(h, deadline, rng, n_trials=300):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = 0.08
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_pert = h + rng.normal(0, scale, n)
        h_pert = _project(h_pert, target)
        h_try = h_pert.copy()
        for _inner in range(10):
            if time.time() > deadline:
                break
            S_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            d_try = _mw_direction(h_try, S_try, K=25, n_iters=8)
            pos = d_try > 1e-14
            neg = d_try < -1e-14
            t_hi = 0.5
            if np.any(pos):
                t_hi = min(t_hi, float(np.min((1.0 - h_try[pos]) / d_try[pos])))
            if np.any(neg):
                t_hi = min(t_hi, float(np.min(h_try[neg] / (-d_try[neg]))))
            if t_hi < 1e-12:
                break
            T = np.correlate(d_try, 1.0 - h_try, mode='full') - np.correlate(h_try, d_try, mode='full')
            Q = np.correlate(d_try, d_try, mode='full')
            t_opt, _ = _line_min(S_try, T, Q, 0.0, t_hi)
            if t_opt < 1e-12:
                break
            h_try = h_try + t_opt * d_try
            h_try = _project(h_try, target)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
            scale = max(0.02, scale * 0.85)
        else:
            scale = min(0.25, scale * 1.15)
    return best_h, best_c5


def _fourier_init(n, modes=8):
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.zeros(n)
    h += 0.5
    amplitudes = np.array([0.25, 0.12, 0.06, 0.03, 0.02, 0.01, 0.008, 0.005])
    phases = np.array([0, 1.2, 2.5, 0.7, 3.1, 1.8, 4.2, 0.3])
    for m in range(min(modes, n // 2)):
        h += amplitudes[m] * np.cos(2 * np.pi * (m + 1) * x / 2.0 + phases[m])
    h = _project(h, n / 2.0)
    return h


def _binary_sa_init(n, rng, max_iters=50000):
    k_ones = n // 2
    h = np.zeros(n)
    h[:k_ones] = 1.0
    rng.shuffle(h)
    S = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(S)) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.05
    T_min = 0.0001
    ones_set = set(np.where(h > 0.5)[0].tolist())
    zeros_set = set(np.where(h <= 0.5)[0].tolist())
    for _ in range(max_iters):
        if T < T_min:
            break
        i = rng.choice(sorted(ones_set)) if ones_set else 0
        j = rng.choice(sorted(zeros_set)) if zeros_set else n - 1
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        S = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(S)) * (2.0 / n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones_set.discard(i)
            ones_set.add(j)
            zeros_set.discard(j)
            zeros_set.add(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T_min, T * 0.9998)
    return best_h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _fourier_init(n0)

    h = _project(h0.copy(), n0 / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)

    # Phase 1: Gradient descent (25% budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _gradient_descent(h, d1, K=35, max_iters=400)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair local search (25% budget)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _pair_local_search(best_h, d2, max_passes=40)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Gradient descent again (10% budget)
    d3 = t0 + budget_s * 0.60
    h3, c5_3 = _gradient_descent(best_h, d3, K=30, max_iters=200)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Pair local search again (15% budget)
    d4 = t0 + budget_s * 0.75
    h4, c5_4 = _pair_local_search(best_h, d4, max_passes=30)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (15% budget)
    d5 = t0 + budget_s * 0.90
    n_cur = len(best_h)
    offsets = [-6, 6, -10, 10, -14, 14, -18, 18, -22, 22, -26, 26]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_cur + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = _gradient_descent(h_r, sub_dl, K=25, max_iters=80)
        sub_dl2 = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _pair_local_search(h_r, sub_dl2, max_passes=15)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_cur = n_try

    # Phase 6: Basin hopping (5% budget)
    d6 = min(deadline, t0 + budget_s * 0.96)
    h6, c5_6 = _basin_hopping(best_h, d6, rng, n_trials=200)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Final polish
    final_dl = deadline - 0.5
    if time.time() < final_dl:
        h_f, c5_f = _pair_local_search(best_h, final_dl, max_passes=20)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
