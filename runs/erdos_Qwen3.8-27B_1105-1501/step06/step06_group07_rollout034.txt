    out = np.empty_like(z)
    pos = z >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    out[~pos] = ez / (1.0 + ez)
    return out

def _solve_lambda(u, target):
    lo = -50.0
    hi = 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(_sigmoid(u + mid))) > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def _u_to_h(u, target):
    return _sigmoid(u + _solve_lambda(u, target))

def _logit(h):
    return np.log(np.clip(h, 1e-10, None) / np.clip(1.0 - h, 1e-10, None))

def _lag_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[:m] -= h[k:k + m]
        g[k:] += (1.0 - h[:m])
    else:
        q = -k
        m = n - q
        if m <= 0:
            return g
        g[:m] += (1.0 - h[q:q + m])
        g[q:] -= h[:m]
    return g

def _lsexp_obj(u):
    target = _G['target']
    eps = _G['eps']
    h = _u_to_h(u, target)
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    v_max = float(np.max(v))
    shifted = (v - v_max) / eps
    return float(eps * (np.log(np.sum(np.exp(shifted))) + v_max / eps))

def _lsexp_grad(u):
    target = _G['target']
    eps = _G['eps']
    n = len(u)
    dx = 2.0 / n
    h = _u_to_h(u, target)
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    v_max = float(np.max(v))
    shifted = (v - v_max) / eps
    p = np.exp(shifted)
    p /= float(np.sum(p))
    G_h = np.zeros(n)
    for r in range(len(v)):
        if p[r] < 1e-14:
            continue
        k = r - (n - 1)
        G_h += p[r] * _lag_gradient(h, k)
    G_h *= dx
    s = h * (1.0 - h)
    s_sum = float(np.sum(s))
    if s_sum < 1e-30:
        return np.zeros(n)
    return s * (G_h - float(np.dot(G_h, s)) / s_sum)

def _lsmbfgs_phase(u, eps_seq, deadline, maxiter_per_eps=200):
    """Run L-BFGS with decreasing eps sequence."""
    n = len(u)
    target = _G['target']
    best_h = _u_to_h(u, target)
    best_c5 = _exact_c5(best_h)
    
    for eps in eps_seq:
        if time.time() > deadline:
            break
        _G['eps'] = eps
        try:
            res = minimize(_lsexp_obj, u, jac=_lsexp_grad, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_eps, 'ftol': 1e-15, 'gtol': 1e-10})
            u = res.x.copy()
        except Exception:
            pass
        h = _u_to_h(u, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, u

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _active_set_refine(h, deadline, K=30, max_iters=500, alpha_init=50.0):
    """Active-set gradient descent with adaptive alpha."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.3
    alpha = alpha_init
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        w = np.exp(alpha * (top_vals - M))
        w /= float(np.sum(w))
        
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _lag_gradient(h, k)
        grad *= dx
        grad -= float(np.mean(grad))
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        direction = -grad / gn
        # Simple line search
        best_t = 0.0
        best_f = best_c5 if _exact_c5(h) > best_c5 else _exact_c5(h)
        f0 = _exact_c5(h)
        t = step
        for _ in range(20):
            h_try = _project(h + t * direction, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
            if f_try <= f0 - 1e-7 * t * gn:
                break
            t *= 0.6
            if t < 1e-8:
                break
        
        if best_t > 1e-9 and best_f < best_c5 - 1e-14:
            h = _project(h + best_t * direction, target)
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            if best_t > 0.5 * step:
                step = min(step * 1.1, 1.0)
            else:
                step = max(step * 0.7, 0.005)
        else:
            step *= 0.7
            if step < 0.003:
                break
    
    return best_h, best_c5

def _perturb_and_optimize(h, deadline, rng, n_perturb, scale_start, max_iters_each):
    """Try random perturbations and optimize each briefly."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = scale_start
    
    for _ in range(n_perturb):
        if time.time() > deadline:
            break
        h_p = _project(h + rng.normal(0, scale, n), target)
        # Quick L-BFGS from this point
        _G['target'] = target
        u0 = _logit(h_p)
        _G['eps'] = 0.02

                          options={'maxiter': max_iters_each, 'ftol': 1e-13, 'gtol': 1e-8})
            h_try = _u_to_h(res.x, target)
        except Exception:
            h_try = h_p
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.002, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.1)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_G = {}


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def _sigmoid(z):
    out = np.empty_like(z)
    pos = z >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    out[~pos] = ez / (1.0 + ez)
    return out


def _solve_lambda(u, target):
    lo = -50.0
    hi = 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(_sigmoid(u + mid))) > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0


def _u_to_h(u, target):
    return _sigmoid(u + _solve_lambda(u, target))


def _logit(h):
    return np.log(np.clip(h, 1e-10, None) / np.clip(1.0 - h, 1e-10, None))


def _lag_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return g
        g[:m] -= h[k:k + m]
        g[k:] += (1.0 - h[:m])
    else:
        q = -k
        m = n - q
        if m <= 0:
            return g
        g[:m] += (1.0 - h[q:q + m])
        g[q:] -= h[:m]
    return g


def _lsexp_obj(u):
    target = _G['target']
    eps = _G['eps']
    h = _u_to_h(u, target)
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    v_max = float(np.max(v))
    shifted = (v - v_max) / eps
    return float(eps * (np.log(np.sum(np.exp(shifted))) + v_max / eps))


def _lsexp_grad(u):
    target = _G['target']
    eps = _G['eps']
    n = len(u)
    dx = 2.0 / n
    h = _u_to_h(u, target)
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    v_max = float(np.max(v))
    shifted = (v - v_max) / eps
    p = np.exp(shifted)
    p /= float(np.sum(p))
    G_h = np.zeros(n)
    for r in range(len(v)):
        if p[r] < 1e-14:
            continue
        k = r - (n - 1)
        G_h += p[r] * _lag_gradient(h, k)
    G_h *= dx
    s = h * (1.0 - h)
    s_sum = float(np.sum(s))
    if s_sum < 1e-30:
        return np.zeros(n)
    return s * (G_h - float(np.dot(G_h, s)) / s_sum)


def _lsmbfgs_phase(u, eps_seq, deadline, max_iter_per_eps=300):
    n = len(u)
    target = _G['target']
    best_h = _u_to_h(u, target)
    best_c5 = _exact_c5(best_h)
    for eps in eps_seq:
        if time.time() > deadline:
            break
        _G['eps'] = eps
        try:
            res = minimize(_lsexp_obj, u, jac=_lsexp_grad, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_eps, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'maxls': 30})
            u = res.x.copy()
        except Exception:
            pass
        h = _u_to_h(u, target)
        c5 = _exact_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5, u


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _active_set_refine(h, deadline, K=30, max_iters=600, alpha_init=50.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 0.25
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        alpha = alpha_init * (1.0 + it / 100.0)
        w = np.exp(alpha * (top_vals - M))
        w /= float(np.sum(w))

        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _lag_gradient(h, k)
        grad *= dx
        grad -= float(np.mean(grad))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        direction = -grad / gn
        f0 = _exact_c5(h)
        best_t = 0.0
        best_f = f0
        t = step
        for _ in range(25):
            h_try = _project(h + t * direction, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
            if f_try <= f0 - 1e-7 * t * gn:
                break
            t *= 0.55
            if t < 1e-9:
                break

        if best_t > 1e-9 and best_f < best_c5 - 1e-14:
            h = _project(h + best_t * direction, target)
            c5_now = _exact_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            stall = 0
            if best_t > 0.5 * step:
                step = min(step * 1.08, 0.8)
            elif best_t < 0.1 * step:
                step = max(step * 0.65, 0.003)
        else:
            stall += 1
            step = max(step * 0.7, 0.002)
            if stall > 50:
                break

    return best_h, best_c5


def _quad_line_search(h, d, dx, target, max_t=0.5):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T_arr = (np.correlate(d, 1.0 - h, mode='full') -
             np.correlate(h, d, mode='full'))
    C_arr = np.correlate(d, d, mode='full')

    pos = d > 1e-14
    neg = d < -1e-14
    t_hi = max_t
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if t_hi < 1e-14:
        return 0.0, _exact_c5(h)

    n_grid = 60
    grid = np.linspace(0.0, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T_arr[None, :] - (grid[:, None] ** 2) * C_arr[None, :]
    obj_vals = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_vals))
    best_t = grid[best_i]
    best_val = obj_vals[best_i]

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = 0.0, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
    f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T_arr - c1 * c1 * C_arr)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T_arr - c2 * c2 * C_arr)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T_arr - opt_t * opt_t * C_arr)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val


def _precise_descent(h, deadline, K=25, max_iters=300):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    step = 1.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        alpha = 1.0 / (n * 0.001)
        w = np.exp(alpha * (top_vals - M))
        w /= float(np.sum(w))

        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _lag_gradient(h, k)
        grad *= dx
        grad -= float(np.mean(grad))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break

        direction = -grad / gn
        t, f_try = _quad_line_search(h, direction, dx, target, max_t=step)
        if t < 1e-12:
            step *= 0.6
            if step < 0.003:
                break
            continue
        h_new = _project(h + t * direction, target)
        f_new = _exact_c5(h_new)
        if f_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = f_new
            best_h = h.copy()
            if t > 0.7 * step:
                step = min(step * 1.05, 1.5)
            else:
                step = max(step * 0.85, 0.005)
        else:
            step *= 0.7
            if step < 0.003:
                break

    return best_h, best_c5


def _perturb_search(h, deadline, rng, n_trials, scale_start):
    n = len(h)
    target = n / 2.0
    _G['target'] = target
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    scale = scale_start

    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_p = _project(h + rng.normal(0, scale, n), target)
        u0 = _logit(h_p)
        _G['eps'] = 0.015
        try:
            res = minimize(_lsexp_obj, u0, jac=_lsexp_grad, method='L-BFGS-B',
                           options={'maxiter': 80, 'ftol': 1e-13, 'gtol': 1e-8})
            h_try = _u_to_h(res.x, target)
        except Exception:
            h_try = h_p
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
            scale = max(0.003, scale * 0.82)
        else:
            scale = min(0.12, scale * 1.12)

    return best_h, best_c5


def _alternating_optimize(h, deadline, n_rounds=6):
    """Alternate between L-BFGS (broad) and active-set (sharp)."""
    n = len(h)
    target = n / 2.0
    _G['target'] = target
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    u = _logit(h)

    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        # Broad: L-BFGS with moderate eps
        eps_broad = max(0.008, 0.03 * (0.7 ** rnd))
        _G['eps'] = eps_broad
        try:
            res = minimize(_lsexp_obj, u, jac=_lsexp_grad, method='L-BFGS-B',
                           options={'maxiter': 150, 'ftol': 1e-14, 'gtol': 1e-9})
            u = res.x.copy()
        except Exception:
            pass
        h_broad = _u_to_h(u, target)
        c5_broad = _exact_c5(h_broad)
        if c5_broad < best_c5:
            best_c5 = c5_broad
            best_h = h_broad.copy()

        # Sharp: active-set with small steps
        h_sharp, c5_sharp = _precise_descent(h_broad, min(deadline, time.time() + 15),
                                            K=20, max_iters=100)
        if c5_sharp < best_c5:
            best_c5 = c5_sharp
            best_h = h_sharp.copy()
            u = _logit(h_sharp)

    return best_h, best_c5
