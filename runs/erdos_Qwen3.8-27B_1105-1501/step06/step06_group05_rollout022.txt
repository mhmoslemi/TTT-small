    except Exception:

def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-13:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 140
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    # Gold section refinement
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(40):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv

def _active_direction(h, corr, k_active, tau, rng):
    n = len(h)
    scores = corr
    m = float(np.max(scores))
    # Use softmax weights for active constraints
    w = np.exp((scores - m) / tau)
    idx = np.argsort(-scores)[:k_active]
    G = np.zeros(n)
    dx = 2.0 / n
    for r in idx:
        k = r - (n - 1)
        G += w[r] * _row_grad(h, k)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm

def _row_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nn = np.arange(0, n - k)
    else:
        nn = np.arange(-k, n)
    if len(nn) == 0:
        return g
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g

def _lsq_direction(h, corr, K, n):
    """Least-squares direction to reduce top-K constraints simultaneously."""
    dx = 2.0 / n
    top_idx = np.argsort(-corr)[:K]
    lags = top_idx - (n - 1)
    # Build gradient matrix G (K x n)
    G = np.zeros((K, n))
    for i, lag in enumerate(lags):
        G[i] = _row_grad(h, lag) * dx
    # Find d that minimizes ||Gd||^2 subject to 1^T d = 0
    # Using null space of 1^T: d = N * y where N is null space basis
    # Simpler: minimize ||Gd + c||^2 s.t. 1^T d = 0
    # Use Lagrange: G^T G d + G^T c - lambda * 1 = 0, 1^T d = 0
    # Let's use: d = -G^T (G G^T + eps I)^{-1} ones, then project to sum=0
    K_mat = G @ G.T + 1e-6 * np.eye(K)
    try:
        d = -G.T @ np.linalg.solve(K_mat, np.ones(K))
    except np.linalg.LinAlgError:
        d = rng.standard_normal(n)  # fallback
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm

def _optimize_line_search(h, deadline, rng, K_dir=10, use_lsq=0.3, 
                          pair_frac=0.15, rand_frac=0.08, sigma_init=0.05):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = _corr_values(h)
    stall = 0
    sigma = sigma_init
    
    while time.time() < deadline:
        r = rng.random()
        if r < rand_frac:
            d = rng.standard_normal(n)
            d -= d.mean()
        elif r < rand_frac + pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        elif r < rand_frac + pair_frac + use_lsq:
            d = _lsq_direction(h, corr, K_dir, n)
            if d is None:
                continue
        else:
            d = _active_direction(h, corr, K_dir, 0.02, rng)
        
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            stall += 1
            continue
        d = d / nrm
        
        # Find bounds
        pos = d > 1e-15
        neg = d < -1e-15
        L = 0.0
        U = 1e30
        if np.any(pos):
            L = max(L, float(np.min(-h[pos] / d[pos])))
            U = min(U, float(np.min((1.0 - h[pos]) / d[pos])))
        if np.any(neg):
            L = max(L, float(np.min((1.0 - h[neg]) / d[neg])))
            U = min(U, float(np.min(-h[neg] / d[neg])))
        
        if U - L < 1e-11:
            stall += 1
            continue
        
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        t, newf = _line_search(corr, T, Q, dx, L, U)
        
        if abs(t) < 1e-14:
            stall += 1
            continue
        
        hn = h + t * d
        if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
            hn = _project(hn, target)
        nf = _exact_c5(hn)
        
        if nf < best_f - 1e-13:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + 5e-4:
            h = hn
            cur_f = nf
            corr = _corr_values(h)
            stall = 0
        else:
            h = best_h.copy()
            cur_f = best_f
            corr = _corr_values(h)
            stall += 1
        
        if stall > 100:
            h = _project(best_h + rng.normal(0, sigma, n), target)
            cur_f = _exact_c5(h)
            corr = _corr_values(h)
            stall = 0
            sigma = max(0.003, sigma * 0.85)
    
    return best_h, best_f

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base = base + rng.normal(0, 0.03, n)
    return _project(base, n / 2.0)

def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 4 + 1)))
        if rng.random() < 0.5:
            h[pos:pos+bl] = 1.0
        pos += bl
    h = h + rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-12:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x

def _slsqp_obj(x, n):
    return x[-1]

def _slsqp_obj_grad(x, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _slsqp_eq_fun(x, n):
    return float(np.sum(x[:n])) - n / 2.0

def _slsqp_eq_jac(x, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _slsqp_ineq_fun(x, lag, n, dx):
    h_x = x[:n]
    z = x[-1]
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return 1.0
        s = dx * float(np.sum(h_x[lag:lag + m] * (1.0 - h_x[:m])))
    else:
        q = -lag
        m = n - q
        if m <= 0:
            return 1.0
        s = dx * float(np.sum(h_x[:m] * (1.0 - h_x[q:q + m])))
    return s - z

def _slsqp_ineq_jac(x, lag, n, dx):
    h_x = x[:n]
    g = np.zeros(n + 1)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += dx * (1.0 - h_x[:m])
            g[:m] -= dx * h_x[lag:]
    else:
        q = -lag
        m = n - q
        if m > 0:
            g[:m] += dx * (1.0 - h_x[q:q + m])
            g[q:] -= dx * h_x[:m]
    g[-1] = -1.0
    return g

def _slsqp_optimize(h, K, maxiter=200, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    old_f = _exact_c5(h)
    corr = _corr_values(h)
    top_idx = np.argsort(-corr)[:K]
    lags = list(top_idx - (n - 1))
    
    constraints = [{'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    for lag in lags:
        constraints.append({'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac,
                           'args': (lag, n, dx)})
    
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[-1] = float(np.max(corr)) + 0.005
    
    bounds = [(0.0, 1.0)] * n + [(-1.0, 2.0)]

    except Exception:

def _row_grad(h, k):

        m = n - q
        if m <= 0:

        g[:m] += (1.0 - h[q:q+m])

def _lsq_direction(h, corr, K, rng):
    n = len(h)
    dx = 2.0 / n
    top_idx = np.argsort(-corr)[:K]
    lags = top_idx - (n - 1)
    G = np.zeros((K, n))
    for i, lag in enumerate(lags):
        G[i] = _row_grad(h, lag) * dx
    # Solve: minimize ||Gd - (-1)||^2 s.t. 1^T d = 0
    # Using Lagrange multipliers: 2G^T(Gd + 1) - lambda*1 = 0, 1^T d = 0
    # G^T G d + G^T (-1) ... let me use a simpler approach
    # d = -G^T (G G^T + eps I)^{-1} 1, then project to sum=0
    K_mat = G @ G.T + 1e-8 * np.eye(K)
    try:
        d = -G.T @ np.linalg.solve(K_mat, np.ones(K))
    except np.linalg.LinAlgError:
        d = rng.standard_normal(n)
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        return None
    return d / nrm

        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm
