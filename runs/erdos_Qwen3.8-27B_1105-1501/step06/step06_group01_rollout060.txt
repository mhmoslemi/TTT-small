import numpy as np
from scipy.optimize import linprog
import time

def _dx(n):
    return 2.0 / n

def _score(h):
    """Compute C5 = max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=target}"""
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A1(b, dx):
    """Build coefficient matrix for LP: min_a max_k dx*correlate(a,b)[k]
    A1[k,j] = dx * b[j-k+(n-1)] for valid range"""
    n = len(b)
    m = n - 1
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - m)
        j_hi = min(n-1, k)
        if j_lo <= j_hi:
            # A[k, j_lo..j_hi] = dx * b[j_lo..j_hi - k + m]
            idx = np.arange(j_lo, j_hi+1) - k + m
            A[k, j_lo:j_hi+1] = dx * b[idx]
    return A

def _build_A2(a, dx):
    """Build coefficient matrix for LP: min_b max_k dx*correlate(a,b)[k]
    A2[k,m] = dx * a[m+k-n+1] for valid range"""
    n = len(a)
    m = n - 1
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

        m_hi = min(n-1, m + k)  # n-1+k

            idx = np.arange(m_lo, m_hi+1) + k - n + 1
            A[k, m_lo:m_hi+1] = dx * a[idx]
    return A

def _solve_relaxed_lp(A, n, target):
    """Solve: min tau s.t. A@x <= tau, sum(x)=target, 0<=x<=1, tau>=0"""
    n_p = n + 1
    c = np.zeros(n_p)
    c[n] = 1.0  # minimize tau
    # Constraints: A@x - tau <= 0  =>  [A | -1] @ [x; tau] <= 0
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = np.zeros(A.shape[0])
    A_eq = np.zeros((1, n_p))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def _line_search_score(h, d, dx, t_radius):
    """Line search along h + t*d, return best t and score"""
    n = len(h)
    # S0 = corr(h, 1-h) * dx
    # We want to minimize max of (S0 + t*T - t^2*C)
    # where T = corr(d, 1-h) - corr(h, d) (linear part)
    # and C = corr(d, d) (quadratic part)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    
    # Find valid range for t
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -t_radius
    t_hi = t_radius
    if np.any(pos):
        t_lo = max(t_lo, np.min(-h[pos] / d[pos]))
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = max(t_lo, np.min((1.0 - h[neg]) / d[neg]))
        t_hi = min(t_hi, np.min(-h[neg] / d[neg]))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1*c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2*c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t*opt_t * C))
    return opt_t, opt_val

def _diverse_start(n, rng):
    """Generate diverse random initial conditions."""
    method = rng.integers(0, 5)
    x = np.arange(n, dtype=float)
    if method == 0:
        # Harmonic
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.35 * np.sin(2*np.pi*freq*x/n + phase)
    elif method == 1:
        # Two harmonics
        f1, f2 = rng.integers(1, 6), rng.integers(3, 12)
        p1, p2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.25*np.sin(2*np.pi*f1*x/n + p1) + 0.15*np.sin(2*np.pi*f2*x/n + p2)
    elif method == 2:
        # Binary with noise
        h = np.zeros(n)
        # Random blocks
        pos = 0
        while pos < n:
            bl = max(2, rng.integers(2, n//4 + 1))
            if rng.random() < 0.5:
                h[pos:pos+bl] = 1.0
            pos += bl
        h = h + rng.normal(0, 0.1, n)
    elif method == 3:
        # Sawtooth
        period = rng.integers(n//6, n//3)
        phase = rng.uniform(0, period)
        t = (x + phase) % period
        h = 0.5 * (t < period/2).astype(float)
        h = h + rng.normal(0, 0.15, n)
    else:
        # Smooth noise
        freqs = rng.integers(1, 8, size=3)
        amps = rng.uniform(0.1, 0.3, size=3)
        phases = rng.uniform(0, 2*np.pi, size=3)
        h = np.zeros(n)
        for f, a, p in zip(freqs, amps, phases):
            h += a * np.sin(2*np.pi*f*x/n + p)
        h += 0.5
    h = _project(h, n / 2.0)
    return h

def _resample(h, n_new):
    """Resample h to n_new points."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _relaxed_pair_cycle(h, deadline, rng, max_iters=40):
    """Run the relaxed pair LP cycle."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _score(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Choose partner b
        if it < 3:
            b = 1.0 - h  # complement (classic)
        elif it < 6:
            # Perturbed complement
            b = 1.0 - h + rng.normal(0, 0.1, n)
            b = _project(b, target)
        else:
            # Random balanced partner
            b = _diverse_start(n, rng)
        
        # Solve LP for a (fix b)
        A1 = _build_A1(b, dx)
        a_star, tau_a = _solve_relaxed_lp(A1, n, target)
        
        # Evaluate a_star as candidate
        if a_star is not None:
            a_cand = _project(a_star, target)
            c5_a = _score(a_cand)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a_cand.copy()
                h = a_cand.copy()
        
        # Solve LP for b (fix a_star or h)
        if a_star is not None:
            A2 = _build_A2(a_star, dx)
        else:
            A2 = _build_A2(h, dx)
        b_star, tau_b = _solve_relaxed_lp(A2, n, target)
        
        if b_star is not None:
            # 1 - b_star is a valid candidate
            c_star = 1.0 - _project(b_star, target)
            c5_c = _score(c_star)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = c_star.copy()
                h = c_star.copy()
            
            # Try interpolation between a_star and 1-b_star
            if a_star is not None:
                u1 = _project(a_star, target)
                u2 = 1.0 - _project(b_star, target)
                # Check both endpoints
                # Try a few interpolation points
                for alpha in [0.25, 0.5, 0.75]:
                    if time.time() > deadline:
                        break
                    h_interp = (1.0 - alpha) * u2 + alpha * u1
                    h_interp = _project(h_interp, target)
                    c5_i = _score(h_interp)
                    if c5_i < best_c5:

def _local_lp_optimize(h, deadline, start_radius=0.1):
    """Local LP-based optimization (similar to previous but tuned)."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()

    stall = 0

        if stall > 60:
            break

        S = np.correlate(h, 1.0 - h, mode='full') * dx
        G = _build_A1(1.0 - h, dx)  # This gives corr(a, 1-h) = a * (1-h) part

        # S(h + d) ≈ S(h) + [corr(d, 1-h) - corr(h, d)] * dx
        # The direction minimizing max(S + linearization) is an LP
        # min tau s.t. S[q] + (T[q] - C[q]*0) ... 
        # Actually it's: min_d max_q (S[q] + corr(d,1-h)[q]*dx - corr(h,d)[q]*dx)
        # = min_d max_q (S[q] + G_col_q @ d)
        # where G_col_q is the q-th row of the combined matrix
        
        # The combined matrix for the linear part:
        # corr(d, 1-h) * dx - corr(h, d) * dx
        # = dx * (corr(d, 1-h) - corr(h, d))
        # For the first term: coefficient of d[j] in corr(d, 1-h)[q] is (1-h)[j-q+m] (from A1 formula with b=1-h)
        # For the second term: coefficient of d[j] in corr(h, d)[q] is ... 
        # correlate(h, d)[q] = sum_j h[j] * d[n-1-q+j]
        # Coefficient of d[m'] in output[q] is h[m' - n + 1 + q] ... 
        # This is A2 formula with a=h, but we want coefficient of d (the second arg)
        # From A2: A2[k, m] = dx * a[m+k-n+1]
        # So the matrix for corr(h, d) has column m = dx * h[m+k-n+1]
        # i.e., _build_A2(h, dx) gives exactly this
        
        # So combined direction matrix:
        # L = _build_A1(1-h, dx) - _build_A2(h, dx)
        # min_d max_q (S[q] + L[q,:] @ d)

        L = _build_A1(1.0 - h, dx) - _build_A2(h, dx)

        n_p = n + 1

        c[n] = 1.0

        A_eq = np.zeros((1, n_p))

def _score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - np.sum(x)
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A1(b, dx):
    n = len(b)
    m = n - 1
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - m)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1) - k + m
            A[k, j_lo:j_hi+1] = dx * b[idx]
    return A

def _build_A2(a, dx):
    n = len(a)
    m = n - 1
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_lo = max(0, m - k)
        m_hi = min(n - 1, m + k)
        if m_lo <= m_hi:
            idx = np.arange(m_lo, m_hi + 1) + k - n + 1
            A[k, m_lo:m_hi+1] = dx * a[idx]
    return A

def _solve_minimax_lp(A, n, target):
    n_p = n + 1
    c = np.zeros(n_p)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = np.zeros(A.shape[0])
    A_eq = np.zeros((1, n_p))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def _line_search(h, d, dx, t_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -t_radius
    t_hi = t_radius
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b_ = t_lo, t_hi
    c1 = b_ - gr * (b_ - a)
    c2 = a + gr * (b_ - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(45):
        if b_ - a < 1e-13:
            break
        if f1 < f2:
            b_, c2, f2 = c2, c1, f1
            c1 = b_ - gr * (b_ - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b_ - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b_) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _diverse_start(n, rng):
    method = rng.integers(0, 6)
    x = np.arange(n, dtype=float)
    if method == 0:
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + phase)
    elif method == 1:
        f1, f2 = int(rng.integers(1, 5)), int(rng.integers(4, 14))
        p1, p2 = rng.uniform(0, 2 * np.pi), rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.22 * np.sin(2 * np.pi * f1 * x / n + p1) + 0.18 * np.sin(2 * np.pi * f2 * x / n + p2)
    elif method == 2:
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(2, int(rng.integers(2, max(3, n // 4))))
            if rng.random() < 0.5:
                h[pos:pos+bl] = 1.0
            pos += bl
        h = h + rng.normal(0, 0.12, n)
    elif method == 3:
        period = int(rng.integers(max(3, n // 8), max(4, n // 3)))
        phase = rng.uniform(0, period)
        t = (x + phase) % period
        h = 0.5 * (t < period / 2).astype(float) + rng.normal(0, 0.15, n)
    elif method == 4:
        h = np.zeros(n)
        for _ in range(3):
            f = int(rng.integers(1, 8))
            a = rng.uniform(0.1, 0.3)
            p = rng.uniform(0, 2 * np.pi)
            h += a * np.sin(2 * np.pi * f * x / n + p)
        h += 0.5
    else:
        h = 0.5 + rng.normal(0, 0.3, n)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _relaxed_pair_cycle(h, deadline, rng, max_iters=60):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _score(h)
    history = [h.copy()]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Choose partner
        if it < 4:
            b = 1.0 - h
        elif it < 8:
            b = 1.0 - h + rng.normal(0, 0.08, n)
            b = _project(b, target)
        elif len(history) > 2 and it % 3 == 0:
            idx = int(rng.integers(0, len(history)))
            b = 1.0 - history[idx] + rng.normal(0, 0.05, n)
            b = _project(b, target)
        else:
            b = _diverse_start(n, rng)
        
        # Solve LP for a (fix b)
        A1 = _build_A1(b, dx)
        a_star, tau_a = _solve_minimax_lp(A1, n, target)
        
        if a_star is not None:
            a_cand = _project(a_star, target)
            c5_a = _score(a_cand)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a_cand.copy()
                h = a_cand.copy()
                if len(history) < 10:
                    history.append(a_cand.copy())
        
        # Solve LP for b (fix a_star or h)
        a_fix = a_star if a_star is not None else h
        A2 = _build_A2(a_fix, dx)
        b_star, tau_b = _solve_minimax_lp(A2, n, target)
        
        if b_star is not None:
            c_star = 1.0 - _project(b_star, target)
            c5_c = _score(c_star)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = c_star.copy()
                h = c_star.copy()
                if len(history) < 10:
                    history.append(c_star.copy())
            
            # Interpolation between a_star and 1-b_star
            if a_star is not None:
                u1 = _project(a_star, target)
                u2 = 1.0 - _project(b_star, target)
                for alpha in [0.3, 0.5, 0.7]:
                    if time.time() > deadline:
                        break
                    h_int = (1.0 - alpha) * u2 + alpha * u1
                    h_int = _project(h_int, target)
                    c5_i = _score(h_int)
                    if c5_i < best_c5:
                        best_c5 = c5_i
                        best_h = h_int.copy()
                        h = h_int.copy()
        
        # Periodic line search toward a_star
        if a_star is not None and it % 4 == 2:
            d = a_star - h
            d = d - np.mean(d)
            if np.max(np.abs(d)) > 1e-10:
                alpha_ls, c5_ls = _line_search(h, d, dx, 0.6)
                if c5_ls < best_c5:
                    h_new = h + alpha_ls * d
                    h_new = _project(h_new, target)
                    c5_new = _score(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
    
    return best_h, best_c5

def _local_lp(h, deadline, start_radius=0.1):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _score(h)
    radius = start_radius
    stall = 0
    
    while time.time() < deadline:
        if stall > 70:
            break
        
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        L = _build_A1(1.0 - h, dx) - _build_A2(h, dx)
        
        n_p = n + 1
        c = np.zeros(n_p)
        c[n] = 1.0
        A_ub = np.hstack([L, -np.ones((2 * n - 1, 1))])
        b_ub = -S
        A_eq = np.zeros((1, n_p))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-radius, radius)] * n + [(None, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and np.all(np.isfinite(res.x)):
                d = res.x[:n]
                if np.max(np.abs(d)) < 1e-12:
                    stall += 1
                    radius *= 0.7
                    if radius < 1e-5:
                        break
                    continue
                alpha, new_c5 = _line_search(h, d, dx, radius)
                if new_c5 < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _score(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                    stall = 0
                    pred = float(np.max(S + L @ d))
                    ratio = cur / max(pred, 1e-15)
                    if ratio < 0.65:
                        radius = min(radius * 1.2, 0.4)
                    elif ratio > 1.6:
                        radius *= 0.72
                    else:
                        radius = min(radius * 1.01, 0.35)
                else:
                    stall += 1
                    radius *= 0.73
                    if stall % 25 == 0 and radius < 0.015:
                        radius = 0.06
            else:
                stall += 1
                radius *= 0.8
        except Exception:
            stall += 1
            radius *= 0.8
    
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _score(h)
    rng = np.random.default_rng(9999)
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        max_idx = int(np.argmax(S))
        L = _build_A1(1.0 - h, dx) - _build_A2(h, dx)
        grad = L[max_idx]
        
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(len(order), 20)):
            if time.time() > deadline:
                break
            i = order[ii]
            if h[i] <= 0.005:
                continue
            for jj in range(min(len(order), 20)):
                if tried > 40:
                    break
                j = order[jj]
                if i == j or h[j] >= 0.995:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.5)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _score(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
                tried += 1
        
        if not improved:
            for _ in range(20):
                if time.time() > deadline:
                    break
                ii = int(rng.integers(len(interior)))
                jj = int(rng.integers(len(interior)))
                if ii == jj:
                    continue
                i, j = interior[ii], interior[jj]
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _score(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
            if not improved:
                break
    
    return best_h, best_c5

def _smooth_gradient_step(h, deadline, tau=0.01, max_iter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _score(h)
    lr = 0.01
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        S = np.correlate(h, 1.0 - h, mode='full') * dx
        S_max = float(np.max(S))
        w = np.exp((S - S_max) / tau)
        w_sum = np.sum(w)
        if w_sum < 1e-15:
            break
        w = w / w_sum
        L = _build_A1(1.0 - h, dx) - _build_A2(h, dx)
        grad = L.T @ w
        h = h - lr * grad
        h = _project(h, target)
        cur = _score(h)
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.88
        if it % 80 == 0:
            tau *= 0.96
            lr = max(lr, 0.003)
    
    return best_h, best_c5

def _noise_escape(h, n_samples, rng):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = _score(h)
    for _ in range(n_samples):
        noise_scale = rng.uniform(0.05, 0.2)
        h_pert = h + rng.normal(0, noise_scale, n)
        h_pert = _project(h_pert, target)
        c5_p = _score(h_pert)
        if c5_p < best_c5:
            best_c5 = c5_p
            best = h_pert.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0, dtype=float)
        h0 = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / n0)
    
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _score(h0)
    
    # Phase 1: Relaxed pair LP from parent (32% budget)
    d1 = t0 + budget_s * 0.32
    h1, c5_1 = _relaxed_pair_cycle(best_h, d1, rng, max_iters=60)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Local LP polish (12% budget)
    d2 = t0 + budget_s * 0.44
    h2, c5_2 = _local_lp(best_h, d2, start_radius=0.1)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: Diverse starts with relaxed pair + polish (22% budget)
    d3 = t0 + budget_s * 0.66
    n_diverse = 5
    per_div = (d3 - time.time()) / max(1, n_diverse)
    for di in range(n_diverse):
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + max(per_div, 8.0))
        h_d = _diverse_start(n0, rng)
        h_d, c5_d = _relaxed_pair_cycle(h_d, sub_dl, rng, max_iters=35)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d.copy()
            if time.time() < d3:
                h_d2, c5_d2 = _local_lp(h_d, min(d3, time.time() + 12.0), 0.08)
                if c5_d2 < best_c5:
                    best_c5 = c5_d2
                    best_h = h_d2.copy()
    
    # Phase 4: Smooth gradient + pair refine (12% budget)
    d4 = t0 + budget_s * 0.78
    h4, c5_4 = _smooth_gradient_step(best_h, min(d4, time.time() + budget_s * 0.06), tau=0.008, max_iter=300)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    h4b, c5_4b = _pair_refine(best_h, d4, max_passes=40)
    if c5_4b < best_c5:
        best_c5 = c5_4b
        best_h = h4b.copy()
    
    # Phase 5: Multi-resolution (10% budget)
    d5 = t0 + budget_s * 0.88
    n_best = len(best_h)
    tried_n = set()
    offsets = list(range(-14, 15, 2))
    rng_shuffled = offsets[:]
    rng.shuffle(rng_shuffled)
    for dn in rng_shuffled:
        if time.time() > d5:
            break
        n_try = n_best + dn
        if n_try < 20 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.013)
        h_r, c5_r = _relaxed_pair_cycle(h_r, sub_dl, rng, max_iters=25)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try
            if time.time() < d5:
                h_r2, c5_r2 = _local_lp(h_r, min(d5, time.time() + 6.0), 0.06)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()
    
    # Phase 6: Final polish (remaining)
    d6 = deadline
    h6, c5_6 = _local_lp(best_h, d6, start_radius=0.04)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    if time.time() < deadline:
        h7, c5_7 = _pair_refine(best_h, deadline, max_passes=25)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _score(best_h)
    
    return best_h, final_c5, len(best_h)
