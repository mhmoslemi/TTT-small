def exact_line_search(h, d, S0, dx):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    hi = float('inf')
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        hi = min(hi, float(np.min(-h[neg] / d[neg])))
    if hi <= 1e-14:
        return 0.0, c5_exact(h)
    
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Grid search first
    n_grid = 40
    alphas = np.linspace(0, hi, n_grid)
    vals = np.array([float(np.max(S0 + a * T - a * a * C)) * dx for a in alphas])
    best_idx = np.argmin(vals)
    
    # Refine around best
    lo_a = alphas[max(0, best_idx - 2)]
    hi_a = alphas[min(n_grid - 1, best_idx + 2)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo_a, hi_a
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    
    # Compare with grid best and endpoints
    if vals[best_idx] < val:
        opt, val = alphas[best_idx], vals[best_idx]
    if vals[0] < val:
        opt, val = 0.0, vals[0]
    if vals[-1] < val:

    return opt, val

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_row_gradients(h, active_rows):
    n = len(h)
    grads = np.zeros((len(active_rows), n))
    for k, r in enumerate(active_rows):
        d = r - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            if d == 0:
                grads[k, :m] = 1.0 - 2.0 * h[:m]
            else:
                grads[k, :m] += (1.0 - h[d:d + m])
                grads[k, d:d + m] += -h[:m]
        else:
            g = -d
            grads[k, g:] += (1.0 - h[:m])
            grads[k, :m] += -h[g:g + m]
    return grads

    hi_a = alphas[min(n_grid - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo_a, hi_a
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx

    opt = (a + b) / 2.0

def lp_direction(h, S0, dx, n_active=20):

    threshold = max_score * 0.97

    if len(active_idx) == 0:

    elif len(active_idx) > n_active:
        sorted_active = np.argsort(-scores[active_idx])[:n_active]
        active_idx = active_idx[sorted_active]
    G = compute_row_gradients(h, active_idx)

    p = cp.Variable(n)

    a, b = lo, hi

    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):

def directed_pair_search(h, n_pairs, deadline):

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np

import cvxpy as cp

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_row_gradients(h, active_rows):
    n = len(h)
    grads = np.zeros((len(active_rows), n))
    for k, r in enumerate(active_rows):
        d = r - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            if d == 0:
                grads[k, :m] = 1.0 - 2.0 * h[:m]
            else:
                grads[k, :m] += (1.0 - h[d:d + m])
                grads[k, d:d + m] += -h[:m]
        else:
            g = -d
            grads[k, g:] += (1.0 - h[:m])
            grads[k, :m] += -h[g:g + m]
    return grads

def exact_line_search(h, d, S0, dx):

    pos = d > 1e-15
    neg = d < -1e-15
    hi = float('inf')
    if np.any(pos):

    if np.any(neg):

    if hi <= 1e-14:

    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')

    alphas = np.linspace(0, hi, n_grid)

    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(40):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def directed_pair_search(h, n_pairs, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline and stall < 60:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.005 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]
        G = compute_row_gradients(h, active_idx)
        weight = np.exp((scores[active_idx] - max_score) / (0.01 * max_score + 1e-15))
        weight /= float(np.sum(weight))
        wgrad = G.T @ weight
        donor_pool = np.argsort(-wgrad)[:n_pairs]
        recv_pool = np.argsort(wgrad)[:n_pairs]
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        tried = set()
        for i in donor_pool:
            if time.time() > deadline:
                break
            for j in recv_pool:
                if i == j:
                    continue
                key = (min(i, j), max(i, j))
                if key in tried:
                    continue
                tried.add(key)
                alpha, new_c5 = golden_pair(h, i, j, dx, S0)
                if alpha > 1e-10 and new_c5 < best_new_c5 - 1e-14:
                    best_new_c5 = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 12 and rng.random() < 0.4:
                idx = rng.integers(n, size=3)
                for idx_k in idx:
                    h[idx_k] += rng.uniform(-0.04, 0.04)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    return best_h, best_c5

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def gradient_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    while time.time() < deadline and stall < 25:
        scores = S0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.003 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:8]
        G = compute_row_gradients(h, active_idx)
        weight = np.exp((scores[active_idx] - max_score) / (0.005 * max_score + 1e-15))
        weight /= float(np.sum(weight))
        g = G.T @ weight
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        d = -g / gn
        opt, val = exact_line_search(h, d, S0, dx)
        if opt > 1e-10 and val < cur_c5 - 1e-14:
            h = h + opt * d
            h = project(h, target)
            cur_c5 = c5_exact(h)
            S0 = np.correlate(h, 1.0 - h, mode='full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def random_pair_sa(h, deadline, rng):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = 0.002
    T_min = 5e-5
    patience = 0
    while time.time() < deadline:
        if patience > 4000:
            if T <= T_min:
                break
            T = T_min
            patience = 0
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        alpha, new_c5 = golden_pair(h, i, j, dx, S0)
        if abs(alpha) > 1e-10:
            delta = new_c5 - cur_c5
            if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / (T + 1e-15))):
                h[i] -= alpha
                h[j] += alpha
                h = project(h, target)
                cur_c5 = c5_exact(h)
                S0 = np.correlate(h, 1.0 - h, mode='full')
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
        T = max(T_min, T * 0.9999)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.97

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.45 * np.sin(2.0 * np.pi * 2.0 * x / n0 + 0.5)

    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0

    # Phase 1: LP multi-row optimization (30% time)
    d1 = t0 + budget_s * 0.30
    h_p1, c5_p1 = lp_optimize(h, d1, n_active=15)
    if c5_p1 < best_c5:
        best_c5 = c5_p1
        best_h = h_p1.copy()

    # Phase 2: Single-row LP (10% time)
    d2 = t0 + budget_s * 0.40
    h_p2, c5_p2 = single_row_lp_optimize(best_h, d2)
    if c5_p2 < best_c5:
        best_c5 = c5_p2
        best_h = h_p2.copy()

    # Phase 3: Directed pair search (15% time)
    d3 = t0 + budget_s * 0.55
    h_p3, c5_p3 = directed_pair_search(best_h, 15, d3)
    if c5_p3 < best_c5:
        best_c5 = c5_p3
        best_h = h_p3.copy()

    # Phase 4: LP with more rows (10% time)
    d4 = t0 + budget_s * 0.65
    h_p4, c5_p4 = lp_optimize(best_h, d4, n_active=30)
    if c5_p4 < best_c5:
        best_c5 = c5_p4
        best_h = h_p4.copy()

    # Phase 5: Gradient polish (8% time)
    d5 = t0 + budget_s * 0.73
    h_p5, c5_p5 = gradient_polish(best_h, d5)
    if c5_p5 < best_c5:
        best_c5 = c5_p5
        best_h = h_p5.copy()

    # Phase 6: Random pair SA (10% time)
    d6 = t0 + budget_s * 0.83
    h_p6, c5_p6 = random_pair_sa(best_h, d6, rng)
    if c5_p6 < best_c5:
        best_c5 = c5_p6
        best_h = h_p6.copy()

    # Phase 7: Multi-resolution (10% time)
    d7 = t0 + budget_s * 0.93
    tried_n = {best_n}
    for dn in [-4, 4, -6, 6, -2, 2, -8, 8]:
        if time.time() > d7:
            break
        n_try = best_n + dn
        if n_try < 20 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample(best_h, n_try)
        sub_dl = min(d7, time.time() + budget_s * 0.012)
        h_r, c5_r = lp_optimize(h_r, sub_dl, n_active=12)
        if c5_r < best_c5:
            sub_dl2 = min(d7, time.time() + budget_s * 0.008)
            h_r, c5_r = gradient_polish(h_r, sub_dl2)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 8: Final LP + pair polish (7% time)
    d8 = deadline
    h_pf, c5_pf = lp_optimize(best_h, min(deadline, t0 + budget_s * 0.955), n_active=10)
    if c5_pf < best_c5:
        best_c5 = c5_pf
        best_h = h_pf.copy()
    remaining = deadline - time.time()
    if remaining > 3:
        h_pf2, c5_pf2 = directed_pair_search(best_h, 20, deadline)
        if c5_pf2 < best_c5:
            best_c5 = c5_pf2
            best_h = h_pf2.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
