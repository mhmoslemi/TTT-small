import numpy as np
from scipy.optimize import linprog
import time

def _score(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {0≤x≤1, sum(x)=target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(p, dx):
    """Build matrix A where A[k,i] = dx * corr(e_i, p, 'full')[k].
    This is a (2n-1) x n matrix."""
    n = len(p)
    A = np.zeros((2 * n - 1, n))
    # For lag m (where k = n-1+m), A[n-1+m, i] = dx * p[i+m] if valid
    for m in range(-(n-1), n):
        k = n - 1 + m
        if m >= 0:
            # Valid: 0 <= i <= n-1-m, A[k,i] = dx * p[i+m]
            cnt = n - m
            if cnt > 0:
                A[k, :cnt] = dx * p[m:m+cnt]
        else:
            # m < 0, let d = -m. Valid: d <= i <= n-1, A[k,i] = dx * p[i-d]
            d = -m
            cnt = n - d
            if cnt > 0:
                A[k, d:] = dx * p[:cnt]
    return A

def _lp_minimax(p, target, dx):
    """Solve min_a max_k dx*corr(a,p)[k] s.t. 0<=a<=1, sum(a)=target.
    Returns optimal a and optimal tau."""
    n = len(p)
    A = _build_A(p, dx)
    # min tau s.t. A @ a - tau <= 0, sum(a) = target, 0<=a<=1, tau>=0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], float(res.x[-1])
    return None, None

def _pair_line_search(h, i, j, sign, target):

        # h[j] + t*(1) >= 0 → t >= -h[j]
        # h[j] + t*(1) <= 1 → t <= 1 - h[j]

        t_hi = min(h[i], 1.0 - h[j])
    else:

    grid = np.linspace(t_lo, t_hi, n_grid)

def _build_A(p, dx, n):
