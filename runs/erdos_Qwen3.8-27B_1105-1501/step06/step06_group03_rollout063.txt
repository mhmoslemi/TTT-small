def _smooth_objective(h, tau, mu, n, target):
    # log-sum-exp of f_k/tau, divided by tau
    S = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    # log-sum-exp
    m = S.max()
    log_sum = m/tau + np.log(np.sum(np.exp((S - m)/tau)))
    obj = log_sum
    # barrier
    interior = np.clip(h, 1e-15, 1-1e-15)
    barrier = -mu * (np.sum(np.log(interior)) + np.sum(np.log(1 - interior)))
    return obj + barrier

def _smooth_gradient(h, tau, mu, n, target):
    S = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    m = S.max()
    w = np.exp((S - m)/tau)
    w = w / w.sum()
    # gradient of log-sum-exp is weighted average of gradients of S_k
    G = _grad_rows(h, np.arange(len(S)), n)
    grad_S = w @ G
    # gradient of barrier
    interior = np.clip(h, 1e-15, 1-1e-15)
    grad_barrier = -mu * (1.0/interior - 1.0/(1-interior))
    return grad_S + grad_barrier

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 48
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0 + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.05 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 25:
        combined = sorted_idx[:25]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ud=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lp_loop(h, deadline, radius_init=0.12, K=15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    while time.time() < deadline:
        if stall > 60:
            if restart_ctr < 10:
                scale = 0.01 + 0.01 * restart_ctr * 0.3
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                continue
            else:
                break
        S = _corr_scaled(h)
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            stall += 1
            radius *= 0.72
            if radius < 1e-6:
                break
            continue
        alpha, new_c5 = _line_search(h, p, 2.0/n, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.06, 0.28)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.95, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 12 == 0 and radius < 0.015:
                radius = 0.05
    return best_h, best_c5

def _pair_sweep(h_init, deadline, max_sweeps=200, rng_seed=123):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    rng = np.random.default_rng(rng_seed)
    
    S = np.correlate(h, 1.0 - h, mode='full')
    all_idx = np.arange(2 * n - 1)
    G = _grad_rows(h, all_idx, n)
    
    pair_list = []
    for i in range(n):
        for j in range(i + 1, n):
            pair_list.append((i, j))
    pair_list = np.array(pair_list)
    
    sweep = 0
    no_improve_sweeps = 0
    while time.time() < deadline and sweep < max_sweeps:
        rng.shuffle(pair_list)
        improved = False
        for pi in range(len(pair_list)):
            if time.time() > deadline:
                break
            i, j = pair_list[pi]
            if h[i] < 1e-10 and h[j] > 1.0 - 1e-10:
                continue
            if h[i] > 1.0 - 1e-10 and h[j] < 1e-10:
                continue
            delta_lo = max(-h[i], h[j] - 1.0)
            delta_hi = min(1.0 - h[i], h[j])
            if delta_hi - delta_lo < 1e-14:
                continue
            linear = G[i, :] - G[j, :]
            gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = delta_lo, delta_hi
            c1 = b - gr_phi * (b - a)
            c2 = a + gr_phi * (b - a)
            f1 = float(np.max(S + c1 * linear))
            f2 = float(np.max(S + c2 * linear))
            for _ in range(25):
                if b - a < 1e-13:
                    break
                if f1 < f2:
                    b, c2, f2 = c2, c1, f1
                    c1 = b - gr_phi * (b - a)
                    f1 = float(np.max(S + c1 * linear))
                else:
                    a, c1, f1 = c1, c2, f2
                    c2 = a + gr_phi * (b - a)
                    f2 = float(np.max(S + c2 * linear))
            delta_opt = (a + b) / 2.0
            pred_max = float(np.max(S + delta_opt * linear))
            if pred_max < float(np.max(S)) - 1e-13:
                h[i] = np.clip(h[i] + delta_opt, 0.0, 1.0)
                h[j] = np.clip(h[j] - delta_opt, 0.0, 1.0)
                S = np.correlate(h, 1.0 - h, mode='full')
                G[i, :] = _grad_rows(h, [i], n)[0, :]
                G[j, :] = _grad_rows(h, [j], n)[0, :]
                cur = float(np.max(S)) * dx
                if cur < best_c5 - 1e-14:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
        sweep += 1
        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
    best_c5 = _exact_c5(best_h)
    return best_h, best_c5

def _smoothed_grad_descent(h_init, deadline, tau_init=0.015, tau_end=0.002):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    h_old = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    momentum = 0.85
    tau = tau_init
    step = 0.008
    n_iter = 0
    stall = 0
    while time.time() < deadline:
        h_prop = h + momentum * (h - h_old)
        h_prop = _project(h_prop, target)
        S = np.correlate(h_prop, 1.0 - h_prop, mode='full')
        m = S.max()
        w = np.exp(tau * (S - m))
        w = w / w.sum()
        all_idx = np.arange(2 * n - 1)
        G = _grad_rows(h_prop, all_idx, n)
        grad = w @ G
        grad -= grad.mean()
        h_old = h.copy()
        h = h_prop - step * grad
        h = _project(h, target)
        n_iter += 1
        if n_iter % 5 == 0:
            tau = max(tau * 0.997, tau_end)
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step = max(0.001, step * 0.8)
                if step < 0.001:
                    break
    return best_h, best_c5

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * 0.99995)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Quick LP (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.12, K=15)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair sweep (25%) - main new contribution
    d2 = t0 + budget_s * 0.33
    h2, c5_2 = _pair_sweep(best_h, d2, max_sweeps=300, rng_seed=42)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP refinement after pair sweep (15%)
    d3 = t0 + budget_s * 0.48
    h3, c5_3 = _lp_loop(best_h, d3, radius_init=0.10, K=15)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: More pair sweeps (18%) - with different random orderings
    d4 = t0 + budget_s * 0.66
    h4, c5_4 = _pair_sweep(best_h, d4, max_sweeps=200, rng_seed=999)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Smoothed gradient descent (12%)
    d5 = t0 + budget_s * 0.78
    h5, c5_5 = _smoothed_grad_descent(best_h, d5, tau_init=0.012, tau_end=0.003)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Binary SA + LP polish (10%)
    d6 = t0 + budget_s * 0.88
    h6a, c5_6a = _binary_sa(best_h, t0 + budget_s * 0.84, rng)
    if c5_6a < best_c5:
        best_c5 = c5_6a
        best_h = h6a.copy()
    if time.time() < d6:
        h6b, c5_6b = _lp_loop(best_h, d6, radius_init=0.08, K=15)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Multi-resolution (8%)
    d7 = t0 + budget_s * 0.96
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12]
    tried = set()
    for dn in offsets:
        if time.time() > d7:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.003:
            sub_dl = min(d7, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=12)
            if c5_r < best_c5 and time.time() < d7 - budget_s * 0.01:
                sub_dl2 = min(d7, time.time() + budget_s * 0.012)
                h_r, c5_r = _pair_sweep(h_r, sub_dl2, max_sweeps=30, rng_seed=77)
            if c5_r < best_c5 - 1e-12:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 8: Final polish (4%)
    d8 = deadline
    h8, c5_8 = _pair_sweep(best_h, d8, max_sweeps=50, rng_seed=2024)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
