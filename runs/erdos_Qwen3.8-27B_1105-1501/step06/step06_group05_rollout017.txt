    term1 = (1.0 - h[ml_clipped]) * valid_l
    term2 = -h[mp_clipped] * valid_p
    G = term1 + term2
    return G

def _line_search_corr(corr, g, L, U):
    # Minimize max(corr + t*g) * dx for t in [L, U]
    # corr, g are 1D arrays of length 2n-1
    # Use ternary search
    if U - L < 1e-15:
        t = 0.5*(L+U)
        return t, float(np.max(corr + t*g))
    a, b = L, U
    for _ in range(60):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        f1 = np.max(corr + m1*g)
        f2 = np.max(corr + m2*g)
        if f1 < f2:
            b = m2
        else:
            a = m1
        if b - a < 1e-15:
            break
    t = 0.5*(a+b)
    return t, float(np.max(corr + t*g))

import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(8):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _compute_G(h):
    n = len(h)
    lags = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(n)[:, None]
    lag_idx = lags[None, :]
    ml = m_idx - lag_idx
    mp = m_idx + lag_idx
    valid_l = (ml >= 0) & (ml < n)
    valid_p = (mp >= 0) & (mp < n)
    ml_c = np.clip(ml, 0, n - 1)
    mp_c = np.clip(mp, 0, n - 1)
    term1 = (1.0 - h[ml_c]) * valid_l
    term2 = -h[mp_c] * valid_p
    return term1 + term2


def _line_search_pl(corr, g, L, U):
    if U - L < 1e-15:
        t = 0.5 * (L + U)
        return t, float(np.max(corr + t * g))
    a, b = L, U
    for _ in range(55):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + m1 * g)
        f2 = np.max(corr + m2 * g)
        if f1 < f2:
            b = m2
        else:
            a = m1
        if b - a < 1e-15:
            break
    t = 0.5 * (a + b)
    return t, float(np.max(corr + t * g))


def _resample(h, n_new):
    n = len(h)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _optimize_pairs(h, deadline, rng, max_cycles=5000):

    dx = 2.0 / n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_f = _exact_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    G = _compute_G(h)
    tau = 0.015
    stall = 0
    max_stall = 200
    cycles = 0

    while time.time() < deadline - 2 and cycles < max_cycles:
        m = float(np.max(corr))
        w = np.exp((corr - m) / tau)
        w /= w.sum()

            if time.time() > deadline - 2:

                if new_f < best_f - 1e-14:

                corr = new_corr

def _optimize_gradient(h, deadline, rng):
