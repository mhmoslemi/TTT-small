        return 0.0, float(np.max(S0))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * C))
    for _ in range(40):

        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * C))
    return opt_t, opt_val

def _smooth_grad(h, tau):
    """Compute gradient of the smooth max approximation."""
    n = len(h)
    dx = 2.0 / n
    # Compute all lag values
    shifts = np.arange(-(n-1), n)  # -n+1, ..., n-1
    lag_vals = np.zeros(len(shifts))
    for idx, s in enumerate(shifts):
        if s >= 0:
            m = n - s
            lag_vals[idx] = dx * np.sum(h[s:] * (1.0 - h[:m]))
        else:
            m = n + s  # s is negative, so n+s = n-|s|
            lag_vals[idx] = dx * np.sum(h[:m] * (1.0 - h[-s:]))
    
    # Smooth max gradient: weighted average of individual gradients
    w = np.exp(lag_vals / tau)
    w /= np.sum(w)
    
    # Compute gradient
    grad = np.zeros(n)
    for idx, s in enumerate(shifts):
        if w[idx] < 1e-15:
            continue
        if s >= 0:
            m = n - s
            # ∂g_s/∂h[j] = (1-h[j+s]) for j in [0, m-1], plus (-h[j-s]) for j in [s, n-1]
            grad[:m] += w[idx] * dx * (1.0 - h[s:s+m])  # (1-h[j+s]) for j=0..m-1
            grad[s:] += w[idx] * (-dx) * h[:m]  # (-h[j-s]) for j=s..n-1
        else:
            m = n + s  # positive
            # ∂g_{-m}/∂h[j] = (1-h[j-m]) for j in [m, n-1], plus (-h[j+m]) for j in [0, m-1]
            grad[m:] += w[idx] * dx * (1.0 - h[:m])  # (1-h[j-m]) for j=m..n-1
            grad[:m] += w[idx] * (-dx) * h[m:]  # (-h[j+m]) for j=0..m-1
    
    return grad
