def _slsqp_epigraph(h, active_lags, dx, max_iter=200):
    n = len(h)
    z0 = _exact_c5(h) + 0.01  # start with some margin
    
    x0 = np.concatenate([h, [z0]])
    
    def objective(x):
        return x[n]  # z
    
    # Inequality constraints: z - s_ℓ(h) >= 0
    def make_ineq(ℓ):
        def f(x):
            h_x = x[:n]
            z_x = x[n]
            s = _overlap_value(h_x, ℓ, dx)
            return z_x - s
        def grad(x):
            h_x = x[:n]
            g = np.zeros(n+1)
            g[:n] = -_overlap_grad(h_x, ℓ, dx)
            g[n] = 1.0
            return g
        return {'type': 'ineq', 'fun': f, 'jac': grad}
    
    # Equality constraint: sum(h) = n/2
    def eq_fun(x):
        return np.sum(x[:n]) - n/2.0
    def eq_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for ℓ in active_lags:
        constraints.append(make_ineq(ℓ))
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = minimize(objective, x0, method='SLSQP', jac=lambda x: np.concatenate([np.zeros(n), [1.0]]),
                   bounds=bounds, constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-10})
    
    if res.success:
        h_new = res.x[:n]
        z_new = res.x[n]
        return h_new, z_new
    return None, None

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(6):

            break
        if res > 0:

        else:

def _line_search(S0, T, Q, dx, L, U):

        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 200
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx

    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]

def _active_direction(h, corr, k_active, tau, rng):

        nrm = float(np.linalg.norm(d))
