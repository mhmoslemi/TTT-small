    # The function f(t) = max_k (corr[k] + t*T[k] - t^2*Q[k]) * dx
    # This is the max of concave quadratic functions, so it's not concave in general
    # But each individual function is concave, so f(t) is the max of concave = not necessarily concave
    # We use golden section search on the interval [L, U]
    if U - L < 1e-14:

def _slp_direction(h, n_active=15):
    """Solve SLP to get a direction."""
    n = len(h)
    M = 2 * n - 1
    raw = np.correlate(h, 1.0 - h, mode='full')
    G = _all_grads(h)
    
    # Select active set: top n_active by raw value
    top_idx = np.argsort(-raw)[:n_active]
    # Also always include the global max
    max_idx = np.argmax(raw)
    if max_idx not in top_idx:
        top_idx = np.append(top_idx, max_idx)
    
    # Build LP:
    # min tau
    # s.t. G[J, x] - tau <= G[J, h] - raw[J]  for J in active set
    #      sum(x) = n/2
    #      0 <= x <= 1
    
    m_act = len(top_idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    A_ub = np.zeros((m_act, n + 1))
    b_ub = np.zeros(m_act)
    for i, k_idx in enumerate(top_idx):
        A_ub[i, :n] = G[k_idx, :]
        A_ub[i, n] = -1.0
        b_ub[i] = G[k_idx, :] @ h - raw[k_idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None
    
    x_star = res.x[:n]
    tau_star = res.x[n]
    
    # Direction is x_star - h
    direction = x_star - h
    return direction, tau_star

def _fw_direction(h, n_active=10, tau_smooth=0.02):
    """Frank-Wolfe style direction using softened subgradient."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    G = _all_grads(h)
    M = len(raw)
    
    # Soften: weight by softmax of raw/tau
    m = float(np.max(raw))
    w = np.exp((raw - m) / tau_smooth)
    w = w / w.sum()
    
    # Weighted gradient
    Gw = w @ G  # shape (n,)
    
    # Frank-Wolfe: minimize Gw · v over {v: 0<=v<=1, sum(v)=n/2}
    # Solution: set v[i]=1 for the n/2 indices with smallest Gw[i]
    idx = np.argsort(Gw)[:n // 2]
    s = np.zeros(n)
    s[idx] = 1.0
    
    return s - h

def _pair_moves(h, deadline, rng, n_trials=None):
    """Try all pairs of indices and find the best improvement."""
    n = len(h)
    target = n / 2.0
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    
    if n_trials is None:
        n_trials = max(5000, n * n * 2)
    
    trials = 0
    while time.time() < deadline and trials < n_trials:
        # Try a pair
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            trials += 1
            continue
        # Move: increase h[i] by delta, decrease h[j] by delta
        # or vice versa
        # h[i] can increase by at most (1 - h[i])
        # h[j] can decrease by at most h[j]
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        
        # Line search
        corr = np.correlate(h, 1.0 - h, mode='full')
        T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
        Q = np.correlate(d, d, mode='full')
        dx = 2.0 / n
        
        L = 0.0
        U = min(1.0 - h[i], h[j])
        
        if U <= L + 1e-12:
            trials += 1
            continue
        
        t, newf = _line_search(corr, T, Q, dx, L, U)
        if t < 1e-12:
            trials += 1
            continue
        
        hn = h + t * d
        hn = np.clip(hn, 0, 1)
        # Fix sum
        hn = _project(hn, target)
        nf = _exact_c5(hn)
        
        if nf < best_f - 1e-12:
            best_f = nf
            best_h = hn.copy()
        if nf < cur_f + 1e-6:
            h = hn
            cur_f = nf
        
        trials += 1
    
    return best_h, best_f

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.array(initial_h_values, dtype=float)
    except Exception:
        base = None

    if base is None or base.size < 8:
        n = 122
        x = np.arange(n)
        base = 0.5 + 0.5 * np.sin(2 * np.pi * x / n)
        base = base + rng.normal(0, 0.03, n)
        base = _project(base, n / 2.0)

    n0 = len(base)
    h0 = _project(base, n0 / 2.0)
    best_h = h0.copy()
    best_f = _exact_c5(h0)

    # Phase 1: SLP optimization with full active set (40% budget)
    d1 = t0 + budget_s * 0.35
    hh, ff = _optimize_slp(h0, d1, rng, n_active=20, tau_s=0.03, use_fw_frac=0.15)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 2: SLP with smaller active set + more FW (25% budget)
    d2 = t0 + budget_s * 0.60
    hh, ff = _optimize_slp(best_h, d2, rng, n_active=10, tau_s=0.02, use_fw_frac=0.3)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 3: Pair optimization (20% budget)
    d3 = t0 + budget_s * 0.80
    hh, ff = _pair_optimize(best_h, d3, rng, n_rounds=300)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    # Phase 4: Multi-resolution (10% budget)
    d4 = t0 + budget_s * 0.93
    bn = len(best_h)
    tried = set([bn])
    for dn in [-4, 4, -6, 6, -8, 8, -2, 2, -10, 10, 12, -12]:
        if time.time() > d4:
            break
        n_try = bn + dn
        if n_try < 30 or n_try > 250:
            continue
        if n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sub_deadline = min(d4, time.time() + max(3.0, budget_s * 0.03))
        hr2, fr2 = _optimize_slp(hr, sub_deadline, rng, n_active=12, tau_s=0.025, use_fw_frac=0.2)
        if fr2 < best_f - 1e-6:
            best_f = fr2
            best_h = hr2.copy()
            bn = n_try

    # Phase 5: Final SLP polish (7% budget)
    hh, ff = _optimize_slp(best_h, deadline, rng, n_active=25, tau_s=0.015, use_fw_frac=0.1)
    if ff < best_f:
        best_f = ff
        best_h = hh.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_f = _exact_c5(best_h)
    return best_h, final_f, len(best_h)
