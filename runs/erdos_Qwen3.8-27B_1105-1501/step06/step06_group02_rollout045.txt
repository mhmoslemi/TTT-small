```python
def qp_solve(h, target_raw, rho, lam_z, lam_norm):
    """Solve the target QP. Returns direction d or None."""
    n = len(h)
    C = 2 * n - 1
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, 'full')
    G = compute_grad_matrix(h)
    
    # Variable: x = [d (length n), z (length C)]
    # Objective: lam_z * mean(z) / scale_z + lam_norm * sum(d^2) / norm_scale
    # Constraints:
    #   G @ d + R - target_raw - z <= 0  (C inequalities)
    #   sum(d) = 0  (1 equality)
    #   diag_i <= d_i <= -diag_i  (2n bounds)
    #   0 <= z_i  (C bounds)
    
    diag = np.minimum(np.minimum(h, 1.0 - h), rho)
    d_lo = -diag
    d_hi = diag
    
    # Scale for z
    z_scale = max(float(np.mean(np.maximum(R - target_raw, 0))), 1e-9)
    norm_scale = max(float(np.sum(diag**2)), rho**2)
    
    # Initial point
    d0 = np.zeros(n)
    z0 = np.maximum(R - target_raw, 0.0)
    x0 = np.concatenate([d0, z0])
    
    # Bounds
    bounds = list(zip(d_lo, d_hi)) + [(0.0, z_scale * C + 1.0)] * C
    
    # Objective
    def obj(x):
        d = x[:n]
        z = x[n:]
        return lam_z * float(np.sum(z)) / (z_scale * C) + lam_norm * float(np.sum(d**2)) / norm_scale
    
    def obj_grad(x):
        d = x[:n]
        z = x[n:]
        gd = 2.0 * lam_norm * d / norm_scale
        gz = lam_z / (z_scale * C) * np.ones(C)
        return np.concatenate([gd, gz])
    
    # Inequality constraints: g_i(x) >= 0 means -g_i <= 0 in SLSQP's convention
    # SLSQP: constraint['fun'](x) >= 0
    # We want: R_i + G_i @ d - target - z_i <= 0
    # i.e.: -(R_i + G_i @ d - target - z_i) >= 0
    # i.e.: target - R_i - G_i @ d + z_i >= 0
    
    def ineq(x):
        d = x[:n]
        z = x[n:]
        return target_raw - R - G @ d + z
    
    def ineq_jac(x):
        # Shape: (C, n+C)
        J = np.zeros((C, n + C))
        J[:, :n] = -G
        J[:, n:] = np.eye(C)
        return J
    
    # Equality constraint: sum(d) = 0
    def eq(x):
        return np.sum(x[:n])
    
    eq_mat = np.zeros(1 + C)
    eq_mat[:n] = 1.0
    
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': lambda x: eq_mat}
    ]
    
    maxit = 200
    result = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                      constraints=constraints, method='SLSQP',
                      options={'maxiter': maxit, 'ftol': 1e-12})
    
    if result.success or (hasattr(result, 'x') and np.all(np.isfinite(result.x))):
        d = result.x[:n]
        d -= np.mean(d)  # Ensure zero sum
        if np.max(np.abs(d)) > 1e-12:
            return d
    return None
```

def qp_direction(h, target_raw, rho, lam_z=0.85, lam_norm=0.15):
    """Solve QP to find direction that pushes all rows below target_raw.
    Returns direction vector or None."""
    n = len(h)
    C = 2 * n - 1
    R = np.correlate(h, 1.0 - h, 'full')
    G = compute_grad_matrix(h)
    
    # Bounds for d
    diag = np.minimum(np.minimum(h, 1.0 - h), rho)
    d_lo = -diag
    d_hi = diag
    
    # Scales
    viol = np.maximum(R - target_raw, 0.0)
    z_scale = max(float(np.mean(viol)), 1e-8)
    norm_scale = max(float(np.sum(diag**2)), 1e-8)
    
    # Initial point
    d0 = np.zeros(n)
    z0 = viol.copy()
    x0 = np.concatenate([d0, z0])
    
    # Bounds
    bounds = list(zip(d_lo, d_hi)) + [(0.0, 2.0 * z_scale + 1.0)] * C
    
    def obj(x):
        d = x[:n]
        z = x[n:]
        return lam_z * float(np.sum(z)) / (z_scale * C) + lam_norm * float(np.dot(d, d)) / norm_scale
    
    def obj_grad(x):
        d = x[:n]
        return np.concatenate([2.0 * lam_norm * d / norm_scale,
                               np.full(C, lam_z / (z_scale * C))])
    
    # Constraints
    # Ineq: target - R - G@d + z >= 0
    # Eq: sum(d) = 0
    
    ineq_jac_mat = np.zeros((C, n + C))
    ineq_jac_mat[:, :n] = -G
    ineq_jac_mat[:, n:] = np.eye(C)
    
    eq_vec = np.zeros(n + C)
    eq_vec[:n] = 1.0
    
    cons = [
        {'type': 'ineq', 'fun': lambda x: target_raw - R - G @ x[:n] + x[n:],
         'jac': lambda x: ineq_jac_mat},
        {'type': 'eq', 'fun': lambda x: np.sum(x[:n]),
         'jac': lambda x: eq_vec}
    ]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': 300, 'ftol': 1e-14})

            d -= np.mean(d)

    except Exception:
        pass
