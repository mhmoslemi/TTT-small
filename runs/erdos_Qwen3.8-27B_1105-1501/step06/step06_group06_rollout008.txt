    t_lo, t_hi = 0.0, radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    fa = float(np.max(S0 + a*T + a*a*C)) * dx
    fb = float(np.max(S0 + b*T + b*b*C)) * dx
    for _ in range(50):
        if b - a < 1e-12:

        c = b - gr*(b-a)
        dp2 = a + gr*(b-a)

        fd = float(np.max(S0 + dp2*T + dp2*dp2*C)) * dx

        else:
            a, fa = c, fc

    f0 = float(np.max(S0)) * dx
    if f0 < fmid:

def _lp_direction(h, active_lags, radius, dx):
    n = len(h)
    G = _grads_at_lags(h, active_lags)
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((len(active_lags), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -scores[active_lags]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]+1e-9), min(radius, 1.0-h[j]-1e-9)) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _gauss_newton_dir(h, active_lags, dx):
    """Gauss-Newton direction: minimize ||G d + c||^2 s.t. sum(d)=0, ||d||<=1"""
    n = len(h)
    G = _grads_at_lags(h, active_lags)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_act = c[active_lags]
    # Add sum constraint
    # Solve (G^T G + λ * 11^T) d = -G^T c_act
    A = G.T @ G
    b = -G.T @ c_act
    # Add zero-sum constraint via Lagrange multiplier
    # [A  1] [d]   [-b]
    # [1^T 0] [λ] = [0 ]
    n_con = n + 1
    M = np.zeros((n_con, n_con))
    M[:n,:n] = A
    M[:n,n] = 1.0
    M[n,:n] = 1.0
    rhs = np.zeros(n_con)
    rhs[:n] = b
    try:
        sol = np.linalg.solve(M, rhs)
        d = sol[:n]
    except np.linalg.LinAlgError:
        d = -b
        d -= d.mean()
    norm = np.linalg.norm(d)
    if norm > 1e-12:
        d = d / norm
    return d

def _fourier_dir(n, rng):
    """Random Fourier-based direction with zero mean."""
    k1 = rng.integers(1, max(2, n//4))
    k2 = rng.integers(1, max(2, n//4))
    p1 = rng.uniform(0, 2*np.pi)
    p2 = rng.uniform(0, 2*np.pi)
    x = np.arange(n)
    d = np.sin(2*np.pi*k1*x/n + p1) + 0.5*np.cos(2*np.pi*k2*x/n + p2)
    d -= d.mean()
    d /= (np.linalg.norm(d) + 1e-15)
    return d

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    return _project(h, n/2.0)
