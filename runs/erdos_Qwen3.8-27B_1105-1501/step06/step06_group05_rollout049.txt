        return dx * np.dot(h[:n-q], 1.0 - h[q:])

def _row_grad_vec(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        idx1 = np.arange(k, n)
        g[idx1] += dx * (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        g[idx2] += -dx * h[idx2 + k]
    else:
        q = -k
        idx1 = np.arange(0, n - q)
        g[idx1] += dx * (1.0 - h[idx1 + q])
        idx2 = np.arange(q, n)
        g[idx2] += -dx * h[idx2 - q]
    return g

def _line_search(S0, T, Q, dx, L, U):

        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 80
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(30):
        if (b - a) < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv

def _get_active_lags(h, top_k=20, frac=0.95):
    """Get indices of the top_k most active lags (those with highest overlap)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    vals = corr * dx
    m = float(np.max(vals))
    threshold = m * frac
    active_idx = np.where(vals >= threshold)[0]
    # Also include top_k by value
    top_idx = np.argsort(-vals)[:top_k]
    combined = np.unique(np.concatenate([active_idx, top_idx]))
    return combined, vals

def _slsqp_epoch(h, deadline, top_k=20, frac=0.95, max_iter=30):
    """One SLSQP epoch: solve epigraph subproblem and return improved h."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_f = _exact_c5(h)
    
    active_idx, vals = _get_active_lags(h, top_k, frac)
    active_lags = active_idx - (n - 1)  # convert to lag values
    
    z0 = cur_f
    
    # Precompute gradients for active lags
    grads = np.zeros((len(active_lags), n))
    for i, lag in enumerate(active_lags):
        grads[i] = _row_grad_vec(h, lag, dx)
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Build constraint functions
    constraints = []
    # Inequality constraints: z - s_lag(x[:n]) >= 0
    for i, lag in enumerate(active_lags):
        def make_ineq(lag_):
            def fun(x):
                hh = x[:n]
                return x[n] - _row_val(hh, lag_)
            def jac(x):
                j = np.zeros(n + 1)
                j[n] = 1.0
                j[:n] = -_row_grad_vec(x[:n], lag_, dx)
                return j
            return {'type': 'ineq', 'fun': fun, 'jac': jac}
        constraints.append(make_ineq(lag))
    
    # Equality constraint: sum(h) - n/2 = 0
    def eq_fun(x):
        return np.sum(x[:n]) - target
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(z0 * 0.5, z0 * 1.5 + 0.1)]
    
    x0 = np.concatenate([h, [z0]])
    
    try:
        res = minimize(obj, x0, jac=obj_grad, constraints=constraints,
                       bounds=bounds, method='SLSQP', options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.success or res.fun < z0:
            h_new = res.x[:n]
            # Project to feasibility
            h_new = _project(h_new, target)
            f_new = _exact_c5(h_new)
            if f_new < cur_f - 1e-12:
                return h_new, f_new
    except Exception:
        pass
    
    return h, cur_f

def _active_direction(h, vals, k_active, tau, rng):
    """Compute a weighted descent direction from active lags."""
    n = len(h)
    dx = 2.0 / n
    m = float(np.max(vals))
    w = np.exp((vals - m) / tau)
    w = w / w.sum()
    # Take top k
    idx = np.argsort(-vals)[:k_active]
    G = np.zeros(n)
    for r in idx:
        k = r - (n - 1)
        G += w[r] * _row_grad_vec(h, k, dx)
    d = -G
    d -= d.mean()
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-14:
        d = rng.standard_normal(n)
        d -= d.mean()
        nrm = float(np.linalg.norm(d))
    return d / nrm

def _step_along_direction(h, d, corr, dx, cur_f):
    """Take a step along direction d using exact line search."""
    n = len(h)
    # Ensure sum-zero
    d = d - d.mean()
    # Compute bounds
    pos = d > 1e-15
    neg = d < -1e-15
    low = [0.0]
    up = [1e10]

        low.append(float(np.min(-h[pos] / d[pos])))
        up.append(float(np.min((1.0 - h[pos]) / d[pos])))

        up.append(float(np.min(-h[neg] / d[neg])))
    L = max(low)
    U = min(up)
    if U - L < 1e-11:
        return h, cur_f

    if hn.min() < -1e-12 or hn.max() > 1.0 + 1e-12:
        hn = _project(hn, n / 2.0)
    nf = _exact_c5(hn)
    return hn, nf

def _pair_swap(h, rng):
    """Random pair swap that preserves sum."""
    n = len(h)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return h
    # Transfer delta from i to j (or vice versa)
    hi_, hj_ = h[i], h[j]
    # Max we can transfer
    max_d = min(hi_, 1.0 - hj_)  # positive: i->j
    min_d = min(1.0 - hi_, hj_)  # negative: j->i
    delta = delta_target = rng.uniform(-min_d, max_d)
    # Actually let's just do a random amount
    delta = rng.uniform(0, max_d) if rng.random() < 0.5 else -rng.uniform(0, min_d)
    hn = h.copy()
    hn[i] += delta
    hn[j] -= delta
    return hn

        up.append(float(np.min((1.0 - h[pos]) / d[pos])))

        low.append(float(np.min((1.0 - h[neg]) / d[neg])))
        up.append(float(np.min(-h[neg] / d[neg])))
    L = max(low)
    U = min(up)
    if U - L < 1e-11:
        return h, _exact_c5(h)
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    t, newf = _line_search(corr, T, Q, dx, L, U)
    if abs(t) < 1e-14:
        return h, _exact_c5(h)
    hn = h + t * d
    if hn.min() < -1e-12 or hn.max() > 1.0 + 1e-12:
        hn = _project(hn, n / 2.0)
    return hn, _exact_c5(hn)


class _EpiProblem:
    def __init__(self, active_lags, n, dx, target):
        self.active_lags = active_lags
        self.n = n
        self.dx = dx
        self.target = target

    def obj(self, x):
        return x[self.n]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def ineq_fun(self, x):
        n = self.n
        hh = x[:n]
        result = np.zeros(len(self.active_lags))
        for i, lag in enumerate(self.active_lags):
            result[i] = x[n] - _row_val(hh, lag)
        return result

    def ineq_jac(self, x):
        n = self.n
        hh = x[:n]
        jac = np.zeros((len(self.active_lags), n + 1))
        for i, lag in enumerate(self.active_lags):
            jac[i, n] = 1.0
            jac[i, :n] = -_row_grad_vec(hh, lag, self.dx)
        return jac

    def eq_fun(self, x):
        return np.sum(x[:self.n]) - self.target

    def eq_jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j


def _slsqp_epoch(h, top_k=20, frac=0.95, max_iter=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_f = _exact_c5(h)

    active_idx, vals, m = _get_active_lags(h, top_k, frac)
    active_lags = (active_idx - (n - 1)).tolist()

    z0 = cur_f

    prob = _EpiProblem(active_lags, n, dx, target)
    bounds = [(0.0, 1.0)] * n + [(z0 * 0.3, z0 * 2.0 + 0.1)]
    x0 = np.concatenate([h, [z0]])

    constraints = [
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}
    ]

    try:
        res = minimize(prob.obj, x0, jac=prob.obj_grad,
                       constraints=constraints, bounds=bounds,
                       method='SLSQP', options={'maxiter': max_iter, 'ftol': 1e-10})
        h_new = res.x[:n]
        h_new = _project(h_new, target)
        f_new = _exact_c5(h_new)
        if f_new < cur_f - 1e-12:
            return h_new, f_new
    except Exception:
        pass
    return h, cur_f


def _optimize_hybrid(h, deadline, rng, slsqp_frac=0.35, k_active=12, tau=0.02, top_k=25):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    cur_f = _exact_c5(h)
    best_h = h.copy()
    best_f = cur_f
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    sigma = 0.05
    uphill = 5e-4

    while time.time() < deadline:
        r = rng.random()

        if r < slsqp_frac:
            h_new, f_new = _slsqp_epoch(h, top_k=top_k, frac=0.93)
            if f_new < best_f - 1e-13:
                best_f = f_new
                best_h = h_new.copy()
            if f_new < cur_f + uphill:
                h = h_new
                cur_f = f_new
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
            else:
                stall += 1
        elif r < slsqp_frac + 0.25:
            vals = corr * dx
            d = _subgradient_direction(h, vals, k_active, tau)
            if d is None:
                d = rng.standard_normal(n)
                d -= d.mean()
                d /= np.linalg.norm(d)
            h_new, f_new = _step_direction(h, d, corr, dx)
            if f_new < best_f - 1e-13:
                best_f = f_new
                best_h = h_new.copy()
            if f_new < cur_f + uphill:
                h = h_new
                cur_f = f_new
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
            else:
                stall += 1
        elif r < slsqp_frac + 0.45:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                stall += 1
                continue

            d[j] = 1.0
            d[i] = -1.0
            h_new, f_new = _step_direction(h, d, corr, dx)
            if f_new < best_f - 1e-13:
                best_f = f_new
                best_h = h_new.copy()
            if
