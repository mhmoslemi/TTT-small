import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_G(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        G[n - 1 + k, k:] += dx * (1.0 - h[:n - k])
        G[n - 1 + k, :n - k] -= dx * h[k:]
        G[n - 1 - k, :n - k] += dx * (1.0 - h[k:])
        G[n - 1 - k, k:] -= dx * h[:n - k]
    return G

def _lse_obj_grad(h, tau):
    n = len(h)
    S = _corr_vals(h)
    S_max = float(np.max(S))
    log_w = (S - S_max) / tau
    log_w -= np.max(log_w)  # Further stabilization
    w = np.exp(log_w)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(S))
        w[np.argmax(S)] = 1.0
    else:
        w /= w_sum
    G = _build_G(h)
    obj = tau * (S_max / tau + np.log(w_sum))  # = S_max + tau * log(w_sum)
    grad = G.T @ w
    return obj, grad

def _slsqp_optimize(h, tau, deadline, max_iter=200, ftol=1e-8):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        x = np.clip(x, 0, 1)
        o, _ = _lse_obj_grad(x, tau)
        return o
    
    def jac(x):
        x = np.clip(x, 0, 1)
        _, g = _lse_obj_grad(x, tau)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or res.fun < obj(h):
            h_new = _project(res.x, target)
            return h_new
    except Exception:
        pass
    return h

def _slsqp_anneal(h, deadline, tau_start=0.05, tau_end=0.005, n_rounds=6):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    taus = np.geomspace(tau_start, tau_end, n_rounds)
    t0 = time.time()
    
    for tau in taus:
        if time.time() > deadline:
            break
        # Multiple SLSQP runs per tau
        sub_deadline = min(deadline, time.time() + (deadline - t0) / (n_rounds + 1))
        for _ in range(3):
            if time.time() > sub_deadline:
                break
            h = _slsqp_optimize(h, tau, sub_deadline, max_iter=100)
            h = _project(h, target)
            c5 = _exact_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def _gradient_descent(h, deadline, lr=0.01, tau=0.01):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    while time.time() < deadline:
