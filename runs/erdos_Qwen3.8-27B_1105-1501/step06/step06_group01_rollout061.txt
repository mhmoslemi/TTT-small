import numpy as np
from scipy.optimize import linprog
import time

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5(h):
    """Compute C5 = max_k dx * corr(h, 1-h, full)[k]"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _build_M(b, n):
    """Build coefficient matrix M where (M@a)[p] = corr(a, b, 'full')[p]"""
    M = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        p = n - 1 + k
        if k >= 0:
            # M[p, j] = b[j-k] for j in [k, n-1]
            j_start, j_end = k, n
            M[p, j_start:j_end] = b[0:n-k]
        else:
            # M[p, j] = b[j-k] = b[j+|k|] for j in [0, n-1+k]
            d = -k
            j_start, j_end = 0, n-d
            M[p, j_start:j_end] = b[d:n]
    return M

def _solve_lp(a_fixed, n):
    """Solve: min tau s.t. dx*M@a - tau <= 0, sum(a)=n/2, 0<=a<=1"""
    dx = 2.0 / n
    M = _build_M(a_fixed, n) * dx
    # Variables: a_0..a_{n-1}, tau
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize tau
    
    # Inequality: M@a - tau <= 0 => [M, -1] @ [a, tau] <= 0
    A_ub = np.hstack([M, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(a) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def _bilinear_step(h, n):
    """One step of bilinear LP: fix b=1-h, solve for a, return a"""
    b = 1.0 - h
    a = _solve_lp(b, n)
    if a is not None:
        a = np.clip(a, 0, 1)
        a = _project(a, n/2.0)
    return a

def _bilinear_iterate(h, deadline, max_iter=200):
    """Iterate bilinear LP: h_{k+1} = argmin max corr(a, 1-h_k)"""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):

        b = 1.0 - h

        if a is not None:

            c5_a = _c5(a)
            if c5_a < best_c5 - 1e-14:

                best_h = a.copy()
                h = a
            else:

                a_noisy = _project(a_noisy, n/2.0)

                    h = a_noisy
                else:

                            h = h_try

                    else:
                        # Stagnation - perturb
                        h = h + rng.normal(0, 0.1, n)
                        h = _project(h, n/2.0)
    
    return best_h, best_c5

def _lp_local_dir(h, dx, radius):
    """Find LP direction to reduce max correlation (local method)"""
    n = len(h)
    S = np.correlate(h, 1.0-h, mode='full') * dx
    M = _build_M(1.0-h, n) * dx  # derivative of corr(h,1-h) w.r.t. h at current point
    
    # Actually, d/dh [corr(h, 1-h)] is not simply corr(., 1-h) because
    # h appears in both arguments. The gradient w.r.t. h[i] of the full objective
    # is more complex. Let me use the local linearization approach from the parent.
    
    # For the local LP: we want to find direction d such that
    # S + G@d < S (componentwise at the max)

import numpy as np

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))

        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])) - 1e-12)
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])) - 1e-12)
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])) - 1e-12)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    ngrid = 50
    grid = np.linspace(t_lo, t_hi, ngrid)
    vals = S0[None, :] + grid[:, None] * T[None, :] + (grid[:, None] ** 2) * C[None, :]
    obj = np.max(vals, axis=1)
    best_i = int(np.argmin(obj))
    best_t = grid[best_i]
    best_val = obj[best_i]
    if best_i > 0 and best_i < ngrid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T + c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T + c2 * c2 * C))
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T + c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T + c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T + opt_t * opt_t * C))
    if opt_val < best_val:
        best_t, best_val = opt_t, opt_val
    return best_t, best_val

def _bilinear_phase(h, deadline, max_iter=500):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    rng = np.random.default_rng(999)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 60:
            h = h + rng.normal(0, 0.08, n)
            h = _project(h, n / 2.0)
            stall = 0
            if stall > 200:
                break
        b = 1.0 - h
        a = _solve_bilinear_lp(b, n)
        if a is None:
            stall += 1
            continue
        a = _project(a, n / 2.0)
        c5_a = _c5(a)
        if c5_a < best_c5 - 1e-14:
            best_c5 = c5_a
            best_h = a.copy()
            h = a
            stall = 0
        else:
            improved = False
            for alpha in [0.2, 0.4, 0.6, 0.8]:
                h_try = (1.0 - alpha) * h + alpha * a
                h_try = _project(h_try, n / 2.0)
                c5_try = _c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
            if not improved:
                d = a - h
                if float(np.max(np.abs(d))) > 1e-8:
                    t, c5_ls = _line_search(h, d, 1.0)
                    if c5_ls < best_c5 - 1e-14 and abs(t) > 1e-10:
                        h_new = h + t * d
                        h_new = _project(h_new, n / 2.0)
                        c5_new = _c5(h_new)
                        if c5_new < best_c5 - 1e-14:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new
                            improved = True
            if not improved:
                stall += 1
    return best_h, best_c5

def _local_phase(h, deadline, max_iters=3000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    radius = 0.15
    stall = 0
    rng = np.random.default_rng(777)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 100:
            if stall < 200:
                h = h + rng.normal(0, 0.04, n)
                h = _project(h, n / 2.0)
                radius = 0.12
                stall = 0
            else:
                break
        d = _local_lp_dir(h, radius)
        if d is None or float(np.max(np.abs(d))) < 1e-12:
            stall += 1
            radius *= 0.8
            if radius < 1e-5:
                break
            continue
        t, new_c5 = _line_search(h, d, radius)
        if new_c5 < best_c5 - 1e-14 and abs(t) > 1e-10:
            h = h + t * d
            h = _project(h, n / 2.0)
            cur = _c5(h)
            if cur < best_c5 - 1e-14:
                best_c5 = cur
                best_h = h.copy()
            stall = 0
            S = _corr_full(h)
            S_max = float(np.max(S))
            predicted = float(np.max(S + np.dot(_jacobian(h, n), d)))
            ratio = cur / max(predicted, 1e-15)
            if ratio < 0.5:
                radius = min(radius * 1.1, 0.4)
            elif ratio > 2.0:
                radius *= 0.75
            else:
                radius = min(radius * 1.01, 0.35)
        else:
            stall += 1
            radius *= 0.7
            if stall % 30 == 0 and radius < 0.05:
                radius = 0.08
    return best_h, best_c5

def _diverse_start(n, rng):
    mode = rng.integers(0, 4)
    if mode == 0:
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n, dtype=float)
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x / n + phase)
        h += rng.normal(0, 0.08, n)
    elif mode == 1:
        h = np.zeros(n)
        pos = 0
        while pos < n:
            bl = max(1, int(rng.integers(1, n // 4 + 1)))
            if rng.random() < 0.5:
                h[pos:pos + bl] = 1.0
            pos += bl
        h = h + rng.normal(0, 0.15, n)
    elif mode == 2:
        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:n // 2]] = 1.0
        h = h + rng.normal(0, 0.2, n)
    else:
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 15)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        x = np.arange(n, dtype=float)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f1 * x / n + p1) + 0.2 * np.sin(2 * np.pi * f2 * x / n + p2)
        h += rng.normal(0, 0.1, n)
    return _project(h, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _pair_moves(h, deadline, max_passes=40):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    rng = np.random.default_rng(314)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        S = _corr_full(h)
        max_idx = int(np.argmax(S))
        J = _jacobian(h, n)
        grad = J[max_idx, :]
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        if len(interior) < 2:
            break
        order = np.argsort(-grad)
        tried = 0
        for ii in range(min(len(order), 30)):
            if time.time() > deadline:
                break
            i = order[ii]
            if h[i] <= 0.05:
                continue
            for jj in range(min(len(order), 30)):
                if tried > 60:
                    break
                j = order[jj]
                if i == j or h[j] >= 0.95:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                t, c5_try = _line_search(h, d, 0.5)
                if c5_try < best_c5 - 1e-14 and abs(t) > 1e-10:
                    h_new = h + t * d
                    h_new = _project(h_new, n / 2.0)
                    c5_new = _c5(h_new)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new
                        improved = True
                tried += 1
        if not improved:
            for _ in range(15):
                if time.time() > deadline or len(interior) < 2:
                    break
                ii = int(rng.integers(len(interior)))
                jj = int(rng.integers(len(interior)))
                if ii == jj:
                    continue
                i, j = interior[ii], interior[jj]
                for sign in [1, -1]:
                    d = np.zeros(n)
                    d[i] = -sign
                    d[j] = sign
                    t, c5_try = _line_search(h, d, 0.4)
                    if c5_try < best_c5 - 1e-14 and abs(t) > 1e-10:
                        h_new = h + t * d
                        h_new = _project(h_new, n / 2.0)
                        c5_new = _c5(h_new)
                        if c5_new < best_c5 - 1e-14:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new
                            improved = True
            if not improved:
                break
    return best_h, best_c5

def _smooth_gradient_descent(h, deadline, max_iters=3000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    tau = 0.01
    lr = 0.008
    for it in range(max_iters):
        if time.time() > deadline:
            break
        S = _corr_full(h)
        S_max = float(np.max(S))
        w = np.exp((S - S_max) / tau)
        w_sum = float(np.sum(w))
        if w_sum > 0:
            w /= w_sum
        J = _jacobian(h, n)
        grad = J.T @ w
        h = h - lr * grad
        h = _project(h, n / 2.0)
        cur = _c5(h)
        if cur < best_c5 - 1e-14:
            best_c5 = cur
            best_h = h.copy()
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.92
        if it % 100 == 0:
            tau *= 0.95
            lr = max(lr, 0.003)
    return best_h, best_c5

def _combined_search(h, deadline):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    mode = 0
    stall = 0
    rng = np.random.default_rng(5555)
    while time.time() < deadline:
        if stall > 80:
            if mode < 3:
                mode = (mode + 1) % 4
                h = h + rng.normal(0, 0.06, n)
                h = _project(h, n / 2.0)
                stall = 0
            else:
                break
        if mode == 0:
            b = 1.0 - h
            a = _solve_bilinear_lp(b, n)
            if a is not None:
                a = _project(a, n / 2.0)
                c5_a = _c5(a)
                if c5_a < best_c5 - 1e-14:
                    best_c5 = c5_a
                    best_h = a.copy()
                    h = a
                    stall = 0
                    continue
                for alpha in [0.3, 0.5, 0.7]:
                    h_try = (1 - alpha) * h + alpha * a
                    h_try = _project(h_try, n / 2.0)
                    c5_try = _c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try
                        stall = 0
                        break
                else:
                    stall += 1
            else:
                stall += 1
        elif mode == 1:
            radius = 0.12
            d = _local_lp_dir(h, radius)
            if d is not None and float(np.max(np.abs(d))) > 1e-12:
                t, c5_try = _line_search(h, d, radius)
                if c5_try < best_c5 - 1e-14 and abs(t) > 1e-10:
                    h_new = h + t * d
                    h_new = _project(h_new, n / 2.0)
                    c5_new = _c5(h_new)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new
                        stall = 0
                        continue
            stall += 1
        elif mode == 2:
            S = _corr_full(h)
            w = np.exp((S - float(np.max(S))) / 0.01)
            w /= float(np.sum(w))
            J = _jacobian(h, n)
            grad = J.T @ w
            h = h - 0.006 * grad
            h = _project(h, n / 2.0)
            cur = _c5(h)
            if cur < best_c5 - 1e-14:
                best_c5 = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            d = rng.normal(0, 1.0, n)
            d = d - float(np.mean(d))
            d /= max(float(np.max(np.abs(d))), 1e-10)
            t, c5_try = _line_search(h, d, 0.15)
            if c5_try < best_c5 - 1e-14 and abs(t) > 1e-10:
                h_new = h + t * d
                h_new = _project(h_new, n / 2.0)
                c5_new = _c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    stall = 0
            else:
                stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        x = np.arange(n0, dtype=float)
        h0 = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x / n0)
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _c5(h0)

    # Phase 1: Bilinear LP from parent (25% budget)
    d1 = t0 + budget_s * 0.25
    h1, c5_1 = _bilinear_phase(best_h, d1, max_iter=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Local LP refinement (15% budget)
    d2 = t0 + budget_s * 0.40
    h2, c5_2 = _local_phase(best_h, d2, max_iters=2000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Combined search (15% budget)
    d3 = t0 + budget_s * 0.55
    h3, c5_3 = _combined_search(best_h, d3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Diverse starts with bilinear LP (15% budget)
    d4 = t0 + budget_s * 0.70
    n_starts = 6
    per_start = (d4 - time.time()) / max(n_starts, 1)
    for si in range(n_starts):
        if time.time() > d4 or per_start < 3.0:
            break
        sub_dl = min(d4, time.time() + max(per_start, 5.0))
        h_s = _diverse_start(n0, rng)
        h_s, c5_s = _bilinear_phase(h_s, sub_dl, max_iter=100)
        if c5_s < best_c5:
            # Polish
            if time.time() < d4:
                h_s, c5_s = _local_phase(h_s, min(d4, time.time() + budget_s * 0.03), max_iters=500)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 5: Pair moves (10% budget)
    d5 = t0 + budget_s * 0.80
    h5, c5_5 = _pair_moves(best_h, d5, max_passes=60)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Multi-resolution (8% budget)
    d6 = t0 + budget_s * 0.88
    n_best = len(best_h)
    tried_n = set()
    offsets = [-4, 4, -6, 6, -8, 8, -10, 10, -12, 12, 16, -16, 20, -20]
    for dn in offsets:
        if time.time() > d6:
            break
        n_try = n_best + dn
        if n_try < 20 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d6, time.time() + budget_s * 0.015)
        h_r, c5_r = _bilinear_phase(h_r, sub_dl, max_iter=80)
        if c5_r < best_c5 - 1e-10:
            if time.time() < d6:
                h_r, c5_r = _local_phase(h_r, min(d6, time.time() + budget_s * 0.02), max_iters=300)
            if c5_r < best_c5 - 1e-10:
                best_c5 = c5_r
                best_h = h_r.copy()
                n_best = n_try

    # Phase 7: Final local polish (7% budget)
    d7 = deadline
    h7, c5_7 = _local_phase(best_h, d7, max_iters=3000)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # One more combined push if time remains
    if time.time() < deadline - 5:
        h8, c5_8 = _combined_search(best_h, min(deadline, time.time() + budget_s * 0.04))
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation and repair
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _c5(best_h)

    return best_h, final_c5, len(best_h)
