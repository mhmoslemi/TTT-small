import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    """Project v onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_A(b, n):
    """Build the constraint matrix A where A @ h gives the correlation values.
    For fixed b, corr_k = sum_i A[k,i] * h[i]"""
    A = np.zeros((2*n-1, n))
    # Lag 0: p = n-1
    A[n-1] = b
    # Positive lags k=1,...,n-1: p = n-1+k
    for k in range(1, n):
        # corr = sum_{i=k}^{n-1} h[i] * b[i-k]
        # So A[n-1+k, i] = b[i-k] for i >= k
        A[n-1+k, k:] = b[:n-k]
    # Negative lags k=1,...,n-1: p = n-1-k
    for k in range(1, n):
        # corr = sum_{i=0}^{n-1-k} h[i] * b[i+k]
        # So A[n-1-k, i] = b[i+k] for i <= n-1-k
        A[n-1-k, :n-k] = b[k:n]
    return A

def _solve_lp(A, target, dx):
    """Solve: min tau s.t. A@h <= tau, sum(h)=target, 0<=h<=1, tau>=0"""
    n = len(A[0])
    # Variables: [h(1..n), tau], total n+1
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize tau
    
    # Inequality: A@h - tau <= 0  →  [A, -1] @ [h; tau] <= 0
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = np.zeros(A.shape[0])
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.all(np.isfinite(res.x)):
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def _relax_pair_descent(h, deadline, max_iters=200):
    """Coordinate descent with LP sub-problems."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Step 1: Fix b = 1-h, solve for h
        b = 1.0 - h
        A = _build_A(b, n)
        h_new, tau1 = _solve_lp(A, target, dx)
        
        if h_new is not None:
            h_new = _project(h_new, target)
            c5_new = _exact_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            # Use h_new or 1-b_new as the next iterate
            # Actually, h_new is the LP solution, which may differ from original h
            # Let's try both h_new and the "other side"
