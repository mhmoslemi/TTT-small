def _grad_for_k(h, k):
    """Gradient of C[h, 1-h] at lag k w.r.t. h."""

    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:

        if m <= 0: return g
        g[k:] += (1.0 - h[:m])
        g[:m] -= h[k:]
    else:
        a = -k
        m = n - a
        if m <= 0: return g
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g * dx

def _active_grad(h, alpha, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top = np.argsort(-corr)[:K]
    tv = corr[top]
    spread = max(tv[0] - tv[-1], 1e-15)
    w = np.exp(-alpha * (tv[0] - tv) / spread)
    w /= w.sum()
    g = np.zeros(n)
    for wi, idx in zip(w, top):
        k = int(idx) - (n - 1)
        g += wi * _grad_for_k(h, k)
    g -= g.mean()
    return g, corr
