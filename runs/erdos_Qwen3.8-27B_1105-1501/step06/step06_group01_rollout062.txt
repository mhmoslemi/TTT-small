import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _gradient_active(h, n_active=5, tol_frac=0.02):
    """Compute gradient weighted by active lags."""
    n = len(h)
    dx = 2.0 / n
    S = _corr_vals(h)
    S_max = float(np.max(S))
    # Select active lags: those within tol of max
    active = np.where(S >= S_max * (1.0 - tol_frac) - 1e-10)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(S))])
    # Or take top n_active
    if len(active) > n_active:
        top = np.argsort(-S[active])[:n_active]
        active = active[top]
    
    # Compute gradient
    grad = np.zeros(n)
    for q in active:
        k = q - (n - 1)
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if k >= 0:
            term1[:n-k] = 1.0 - h[k:]
            term2[k:] = h[:n-k]
        else:
            term1[-k:] = 1.0 - h[:n+k]
            term2[:n+k] = h[-k:]
        grad += dx * (term1 - term2)
    
    # Scale to unit norm
    gnorm = float(np.linalg.norm(grad))
    if gnorm > 1e-15:
        grad /= gnorm
    return grad, S_max, active

def _line_search_active(h, d, dx, max_step=1.0):
    """Line search: minimize max_k (S0 + t*T - t^2*C)_k with box constraints."""
    n = len(h)
    # S0 = current correlation values
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    # T[j] = derivative of corr at h in direction d
    # T = corr(d, 1-h) - corr(h, d) all times dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    # C[j] = corr(d, d) * dx (quadratic term)
    C = np.correlate(d, d, mode='full') * dx
    
    # Box constraint: h + t*d in [0,1]
    pos = d > 1e-15
    neg = d < -1e-15
    t_lo = -max_step
    t_hi = max_step
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(40):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _lp_relax_pair(h, partner, target):
    """Solve LP: min tau s.t. dx*corr(a, partner) <= tau, sum(a)=target, 0<=a<=1."""
    n = len(h)
    dx = 2.0 / n
    
    # Build matrix A where A[m, q] = dx * partner[m - k] if valid
    # where k = q - (n-1)
    # corr(a, partner, 'full')[q] = sum_n a[n+k] * partner[n] where k = q-(n-1)
    # = sum_m a[m] * partner[m-k] (substituting m = n+k, n = m-k)
    # So A[m, q] = dx * partner[m-k] if 0 <= m-k < n
    
    A = np.zeros((n, 2*n - 1))
    for m in range(n):
        for q in range(2*n - 1):
            k = q - (n - 1)
            idx = m - k
            if 0 <= idx < n:
                A[m, q] = dx * partner[idx]
    
    # LP: min tau
    # s.t. A @ a - tau <= 0  (i.e., A @ a <= tau*1)
    #      1^T a = target
    #      0 <= a_i <= 1
    #      tau >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize tau
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _anneal(h, deadline, rng, T_init=0.01, T_min=0.001, n_steps=None):
    """Simulated annealing with 2-coordinate moves."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    T = T_init
    it = 0
    max_it = 50000 if n_steps is None else n_steps
    
    while time.time() < deadline and it < max_it:
        it += 1
        # Choose 2 random positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Try move: increase one, decrease other
        for sign in [1, -1]:
            d = np.zeros(n)
            d[i] = -sign
            d[j] = sign
            
            alpha, c5_try = _line_search_active(h, d, dx, max_step=0.5)
            if c5_try < best_c5:
                h = h + alpha * d
                h = _project(h, target)
                cur = _exact_c5(h)
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
                break
        else:
            # Simulated annealing: try a random perturbation
            if rng.random() < 0.1:  # Occasionally do larger moves
                delta = 0.05 * rng.standard_normal(2)
                d = np.zeros(n)
                d[i] = -delta[0]
                d[j] = delta[1]
                # Project to maintain sum
                alpha, c5_try = _line_search_active(h, d, dx, max_step=0.3)
                new_c5 = c5_try
                if new_c5 < best_c5 + T:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
        
        # Cool down
        T *= 0.9995
        T = max(T, T_min)
        
        # Every 1000 steps, check progress
        if it % 2000 == 0:
            # Reset h to best if stuck
            if _exact_c5(h) > best_c5 + 0.005:
                h = best_h.copy()
    
    return best_h, best_c5

def _lp_explore(h, deadline, rng, n_partners=20):
    """Explore using LP relaxation with diverse partners."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for p in range(n_partners):
        if time.time() > deadline:
            break
        
        # Generate diverse partners
        mode = p % 4
        if mode == 0:
            partner = 1.0 - best_h
        elif mode == 1:
            partner = best_h + 0.1 * rng.standard_normal(n)
        elif mode == 2:
            # Fourier-based
            freq = rng.integers(1, 6)
            phase = rng.uniform(0, 2*np.pi)
            x = np.arange(n)
            partner = 0.5 + 0.4 * np.sin(2*np.pi*freq*x/n + phase)
        else:
            partner = 0.5 + 0.3 * rng.standard_normal(n)
        
        partner = _project(np.clip(partner, 0, 1), target)
        
        # Solve LP for a
        a = _lp_relax_pair(best_h, partner, target)
        if a is not None:
            a = _project(a, target)
            c5_a = _exact_c5(a)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = a.copy()
            
            # Also try 1-partner as candidate
            u_r = 1.0 - partner
            u_r = _project(u_r, target)
            c5_r = _exact_c5(u_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = u_r.copy()
            
            # Try interpolation
            for alpha in np.linspace(0, 1, 11)[1:10]:
                u = (1-alpha) * u_r + alpha * a
                u = _project(u, target)
                c5_u = _exact_c5(u)
                if c5_u < best_c5:
                    best_c5 = c5_u
                    best_h = u.copy()
    
    return best_h, best_c5

def _gradient_descent(h, deadline, lr_init=0.01):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
