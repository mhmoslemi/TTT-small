import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    v = np.clip(v, -10.0, 10.0)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h):
    """Compute the full gradient matrix G where G[r, j] = d corr[r] / d h[j]."""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            m_len = n - d
            if m_len > 0:
                G[r, d:] += (1.0 - h[:m_len])
                G[r, :m_len] += -h[d:]
        else:
            g = -d
            m_len = n - g
            if m_len > 0:
                G[r, :m_len] += (1.0 - h[g:])
                G[r, g:] += -h[:m_len]
    return G

def _lse_objective_and_grad(h, beta):
    """Compute LSE objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = dx * corr
    # Numerically stable log-sum-exp
    max_val = float(scaled.max())
    z = beta * (scaled - max_val)
    exp_z = np.exp(z)
    sum_exp = float(exp_z.sum())
    p = exp_z / sum_exp
    obj = max_val + np.log(sum_exp) / beta
    # Gradient: dx * p @ G
    G = _grad_matrix(h)
    grad = dx * (p @ G)
    return float(obj), grad

def _lse_solve(h, beta, deadline, maxiter=200):
    """Solve LSE minimization using SLSQP."""
    n = len(h)
    target = n / 2.0
    
    def fun(x):
        val, grad = _lse_objective_and_grad(x, beta)
        return val, grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(fun, h, method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = _project(res.x, target)
            return h_new
    except Exception:
        pass
    return h

def _line_search(h, d, dx, radius):
    """Exact line search along direction d."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / (-d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    
    # Golden section search
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(80):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _direction_from_weights(h, p, dx, top_k=None):
    """Compute descent direction from constraint weights p."""
    n = len(h)
    if top_k is not None and top_k < len(p):
        idx = np.argsort(-p)
        kept = np.zeros_like(p)
        kept[idx[:top_k]] = p[idx[:top_k]]
        p = kept
    p = p / p.sum()
    G = _grad_matrix(h)
    grad = dx * (p @ G)
    grad -= grad.mean()  # project to sum=0
    norm = np.linalg.norm(grad)

The gradient of the LSE objective at position j is:
∇Q[j] = dx * sum_r p_r * G[r, j]

Let me compute this directly without forming the full matrix:

For d=0 (r = n-1): contribution is p[n-1] * (1 - 2*h[j])
For d>0 (r = n-1+d): 
  - If j >= d: contribution includes p[n-1+d] * (1-h[j-d])
  - If j < n-d: contribution includes -p[n-1+d] * h[j+d]
For d<0 (r = n-1-d, d>0):
  - If j < n-d: contribution includes p[n-1-d] * (1-h[j+d])
  - If j >= d: contribution includes -p[n-1-d] * h[j-d]

def _lse_grad_direct(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    # d = 0
    grad += p[n-1] * (1.0 - 2.0 * h)
    # d > 0
    for d in range(1, n):
        m = n - d
        wd = p[n-1+d]
        if wd > 1e-15:
            grad[d:] += wd * (1.0 - h[:m])
            grad[:m] -= wd * h[d:]
        wn = p[n-1-d]
        if wn > 1e-15:
            grad[:m] += wn * (1.0 - h[d:])
            grad[d:] -= wn * h[:m]
    return dx * grad

def _project(v, target):

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

    if abs(res) > 1e-10:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0

def _lse_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = dx * corr
    max_val = float(scaled.max())

    exp_z = np.exp(z)
