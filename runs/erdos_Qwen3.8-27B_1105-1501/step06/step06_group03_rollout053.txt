import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]
        else:
            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 64
    grid = np.linspace(t_lo, t_hi, n_grid)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * C[None, :]
    obj_grid = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj_grid))
    best_t = grid[best_i]
    best_val = obj_grid[best_i]
    if best_i > 0 and best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)

    opt_t = (a + b) / 2.0

    return best_t, best_val

def _lp_direction(h, S, n, K, radius):
    M = float(S.max())
    sorted_idx = np.argsort(-S)
    active = sorted_idx[:K]
    tol = max(0.04 * M, 1e-6)
    near_idx = np.where(S >= M - tol)[0]
    combined = np.unique(np.concatenate([active, near_idx]))
    if len(combined) > 30:
        combined = sorted_idx[:30]
    G = _grad_rows(h, combined, n)
    n_act = len(combined)
    c_vec = np.zeros(n + 1)
    c_vec[n] = -1.0
    A_ub = np.zeros((n_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = M - S[combined]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo_b = max(-radius, -h[j] + 1e-10)
        hi_b = min(radius, 1.0 - h[j] - 1e-10)
        if lo_b >= hi_b:
            lo_b, hi_b = 0.0, 0.0
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def _lse_gradient(h, tau, n, n_sig=None):
    S = _corr_scaled(h)
    S_tau = S * tau
    S_shifted = S_tau - S_tau.max()
    exp_S = np.exp(S_shifted)
    w_sum = exp_S.sum()
    if w_sum < 1e-300:
        return None, S
    weights = exp_S / w_sum
    if n_sig is None:
        n_sig = min(len(S), 40)
    top_idx = np.argsort(-weights)[:n_sig]
    G = _grad_rows(h, top_idx, n)
    grad = weights[top_idx] @ G
    grad -= grad.mean()
    return grad, S

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-n exploration (12% time)
    d1 = t0 + budget_s * 0.12
    n_candidates = []
    base_n = n0
    for dn in range(-30, 31, 2):
        n_try = base_n + dn
        if 50 <= n_try <= 200 and n_try % 2 == 0:
            n_candidates.append(n_try)
    # Also include some specific values
    for nc in [60, 80, 100, 110, 130, 140, 150, 160]:
        if nc not in n_candidates:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    # Start with the original n
    starts = {}
    starts[base_n] = best_h.copy()
    for nc in n_candidates:
        if nc != base_n:
            starts[nc] = _resample(best_h, nc)

    for nc in n_candidates:
        if time.time() > d1:
            break
        if nc not in starts:
            continue
        h_s = starts[nc]
        sub_dl = min(d1, time.time() + budget_s * 0.012)
        h_opt, c5_opt = _quick_opt(h_s, sub_dl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Also try a few structured inits at the best n
    best_n = len(best_h)
    if time.time() < d1:
        for freq in [1, 2, 3, 4]:
            for ph in [0.0, 1.0, 2.0]:
                if time.time() > d1:
                    break
                h_s = _sine_init(best_n, freq, ph)
                sub_dl = min(d1, time.time() + budget_s * 0.005)
                h_opt, c5_opt = _quick_opt(h_s, sub_dl)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 2: LSE + FW optimization (15% time)
    d2 = t0 + budget_s * 0.27
    h_lse, c5_lse = _lse_fw_optimize(best_h, min(d2, time.time() + budget_s * 0.07),
                                      tau_start=50, tau_end=5)
    if c5_lse < best_c5:
        best_c5 = c5_lse
        best_h = h_lse.copy()
    if time.time() < d2:
        h_gd, c5_gd = _grad_descent_opt(best_h, min(d2, time.time() + budget_s * 0.06),
                                         step_init=0.03, tau_start=40)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()

    # Phase 3: LP optimization (20% time)
    d3 = t0 + budget_s * 0.47
    h_lp, c5_lp = _lp_loop(best_h, min(d3, time.time() + budget_s * 0.10),
                           radius_init=0.14, K=18, use_smooth=False)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp.copy()
    if time.time() < d3:
        h_lp2, c5_lp2 = _lp_loop(best_h, min(d3, time.time() + budget_s * 0.08),
                                  radius_init=0.08, K=25, use_smooth=True)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2.copy()

    # Phase 4: More LSE passes with different params (10% time)
    d4 = t0 + budget_s * 0.57
    h_lse2, c5_lse2 = _lse_fw_optimize(best_h, min(d4, time.time() + budget_s * 0.05),
                                        tau_start=30, tau_end=3)
    if c5_lse2 < best_c5:
        best_c5 = c5_lse2
        best_h = h_lse2.copy()
    if time.time() < d4:
        h_gd2, c5_gd2 = _grad_descent_opt(best_h, min(d4, time.time() + budget_s * 0.03),
                                           step_init=0.02, tau_start=60, tau_end=4)
        if c5_gd2 < best_c5:
            best_c5 = c5_gd2
            best_h = h_gd2.copy()

    # Phase 5: Binary SA (15% time)
    d5 = t0 + budget_s * 0.72
    h_sa, c5_sa = _binary_sa(best_h, min(d5, time.time() + budget_s * 0.09), rng)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    if time.time() < d5:
        h_lp3, c5_lp3 = _lp_loop(best_h, min(d5, time.time() + budget_s * 0.04),
                                  radius_init=0.06, K=15)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3.copy()

    # Phase 6: Pair + multi-pair refinement (10% time)
    d6 = t0 + budget_s * 0.82
    h_pr, c5_pr = _pair_refine(best_h, min(d6, time.time() + budget_s * 0.05), max_passes=100)
    if c5_pr < best_c5:
        best_c5 = c5_pr
        best_h = h_pr.copy()
    if time.time() < d6:
        h_mp, c5_mp = _multi_pair_refine(best_h, d6)
        if c5_mp < best_c5:
            best_c5 = c5_mp
            best_h = h_mp.copy()

    # Phase 7: Multi-resolution exploration (10% time)
    d7 = t0 + budget_s * 0.92
    base_n2 = len(best_h)
    tried_n = set()
    for dn in [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -16, 16]:
        if time.time() > d7:
            break
        n_try = base_n2 + dn
        if n_try < 50 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d7, time.time() + budget_s * 0.015)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=14)
            if c5_r < best_c5 and time.time() < d7 - budget_s * 0.01:
                sub_dl2 = min(d7, time.time() + budget_s * 0.01)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d7:
                    h_r, c5_r = _lp_loop(h_r, min(d7, time.time() + budget_s * 0.008),
                                          radius_init=0.05, K=14)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Final aggressive perturbation + re-opt (5% time)
    d8 = t0 + budget_s * 0.97
    scale = 0.025
    for attempt in range(500):
        if time.time() > d8:
            break
        if best_c5 < 0.3805:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d8, time.time() + budget_s * 0.004)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.06, K=14)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.85)
        else:
            scale = min(0.10, scale * 1.15)

    # Phase 9: Final polish
    d9 = deadline
    h_final, c5_final = _lp_loop(best_h, d9, radius_init=0.03, K=12)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
    if time.time() < d9:
        h_pf, c5_pf = _pair_refine(best_h, d9, max_passes=30)
        if c5_pf < best_c5:
            best_c5 = c5_pf
            best_h = h_pf.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
