        return np.ones(n)

def _build_jacobian(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        offset = q - (n - 1)
        r = abs(offset)
        m = n - r  # number of overlapping terms
        if m <= 0:
            continue
        if offset >= 0:
            G[q, r:r+m] += dx * (1.0 - h[:m])
            G[q, :m] -= dx * h[r:r+m]
        else:
            G[q, :m] += dx * (1.0 - h[r:r+m])
            G[q, r:r+m] -= dx * h[:m]
    return G

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_jacobian(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for q in range(2 * n - 1):
        k = q - (n - 1)
        r = abs(k)
        m = n - r
        if m <= 0:
            continue
        if k >= 0:
            G[q, k:k+m] += dx * (1.0 - h[:m])
            G[q, :m] -= dx * h[k:k+m]
        else:
            G[q, :m] += dx * (1.0 - h[r:r+m])
            G[q, r:r+m] -= dx * h[:m]
    return G

def _pnorm_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    S = _corr_vals(h)
    S_max = max(float(np.max(S)), 1e-15)
    r = S / S_max
    r_p = r ** p
    U = float(np.sum(r_p))
    if U < 1e-300:
        return np.zeros(n)
    weights = r ** (p - 1)
    G = _build_jacobian(h, dx)
    grad = G.T @ weights
    scale = U ** (1.0 / p - 1.0)
    return grad * scale

def _pnorm_optimize(h, deadline, p_list=None, max_iter_per_p=3000):
    n = len(h)
    target = n / 2.0
    h = h.copy()

    best_c5 = _exact_c5(h)

    for p in p_list:

        lr = 0.004
        momentum = 0.8
        v_mom = np.zeros(n)
        stalled = 0
        best_at_p_start = best_c5
        no_improve_count = 0
        for it in range(max_iter_per_p):
            if time.time() > deadline:
                break
            grad = _pnorm_grad(h, p)
            v_mom = momentum * v_mom + (1.0 - momentum) * grad
            h_new = h - lr * v_mom
            h_new = _project(h_new, target)

            if c5_new < best_c5 - 1e-15:

                best_h = h.copy()
                stalled = 0

def _line_search_polish(h, deadline, max_iter=400):

def _resample_h(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
