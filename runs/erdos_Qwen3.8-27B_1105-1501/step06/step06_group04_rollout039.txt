def _project(v, target):
    n = len(v)
    v = np.nan_to_num(v, nan=target / n, posinf=1.0, neginf=0.0)
    v = np.clip(v, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0

        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _corr_vec(h):

def _grad_lse(h, beta, dx):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    C = R * dx
    M = float(np.max(C))
    z = beta * (C - M)
    ez = np.exp(z)
    se = float(np.sum(ez))
    p = ez / se
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        pr = p[r]
        if pr < 1e-14:
            continue
        d = r - (n - 1)
        if d == 0:
            grad += pr * (1.0 - 2.0 * h) * dx
        elif d > 0:
            m = n - d
            grad[:m] += pr * (1.0 - h[d:]) * dx
            grad[d:] += pr * (-h[:m]) * dx
        else:
            e = -d
            m = n - e
            grad[e:] += pr * (1.0 - h[:m]) * dx
            grad[:m] += pr * (-h[e:]) * dx
    return grad

    z = beta * (C - M)
    ez = np.exp(z)

    val = _lse_val(x, beta, dx)
    grad = _grad_lse(x, beta, dx)
    return val, grad

def _sum_eq(x, target):
    return float(np.sum(x)) - target

    return np.ones(len(x))

def _slsqp_lse(h0, beta, target, dx, maxiter=200):
    n = len(h0)
    try:
        res = minimize(
            fun=lambda x: _slsqp_lse_func(x, beta, dx)[0],
            jac=lambda x: _slsqp_lse_func(x, beta, dx)[1],
            x0=h0.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}],
            args=(target,),
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        h_new = _project(res.x, target)
        return h_new
    except Exception:
        return h0.copy()

def _line_search(h, d, dx, trust_radius):

def _binary_hill_climb(h_start, deadline, max_passes=500):

                    if 0 <= j - d < n:
                        val -= h[j - d]

                    best_i = i
                    best_j = j

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

    d5 = t0 + budget_s * 0.87
    n_best = len(best_h)
    offsets = [-4, 4, -2, 2, -6, 6, -8, 8, -12, 12, -16, 16, 20, -20, 24, -24]
    tried_ns = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = n_best + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r0 = _compute_c5(h_r)
        if c5_r0 < best_c5:
            best_c5 = c5_r0
            best_h = h_r.copy()
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _run_lse_annealing(h_r, sub_dl,
                                        beta_schedule=[50, 200, 1000, 3000])
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_best = n_try
        if time.time() < d5 - 10:
            h_r2, c5_r2 = _run_line_polish(h_r, min(d5, time.time() + budget_s * 0.008),
                                            n_iterations=2000)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                n_best = n_try

    # Phase 6: Final LSE + line polish (7% of budget)
    d6 = min(hard_deadline, t0 + budget_s * 0.94)
    if time.time() < d6:
        n = len(best_h)
        h6, c5_6 = _run_lse_annealing(best_h, d6,
                                       beta_schedule=[500, 2000, 8000, 20000])
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    if time.time() < hard_deadline:
        h7, c5_7 = _run_line_polish(best_h, hard_deadline, n_iterations=5000)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        h_fb = _sine_init(120, 3, 0.0)
        return h_fb, _compute_c5(h_fb), len(h_fb)
    return best_h, final_c5, len(best_h)
