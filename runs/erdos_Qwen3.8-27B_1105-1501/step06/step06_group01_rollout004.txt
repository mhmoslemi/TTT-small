```python
def _solve_lp_factor(b, dx, target):
    """Solve min_a max_k correlate(a, b, 'full')[k]*dx subject to constraints"""
    n = len(b)
    m = 2 * n - 1
    M = _build_corr_matrix(b, dx)
    
    # Variables: a_0, ..., a_{n-1}, t
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: M @ a - t <= 0 for each row
    # i.e., M @ a + (-1) * t <= 0
    A_ub = np.hstack([M, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(a) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= a_i <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None
```

```python
def _relax_pair_iterate(h, deadline, max_iters=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Step 1: Fix b = 1 - h, find optimal a
        b = 1.0 - h
        a_opt = _solve_lp_factor(b, dx, target)
        if a_opt is None:
            break
        
        # Project to ensure feasibility
        a_opt = _project(a_opt, target)
        
        # Step 2: Fix a = a_opt, find optimal b
        b_opt = _solve_lp_factor(a_opt, dx, target)
        if b_opt is None:
            break
        b_opt = _project(b_opt, target)
        
        # Step 3: Two feasible endpoints
        u_L = a_opt  # feasible (sum = n/2, in [0,1])
        u_R = 1.0 - b_opt  # feasible (sum = n - n/2 = n/2, in [0,1])
        
        # Evaluate both endpoints
        c5_L = _exact_c5(u_L)
        c5_R = _exact_c5(u_R)
        
        if c5_L < best_c5:
            best_c5 = c5_L
            best_h = u_L.copy()
        if c5_R < best_c5:
            best_c5 = c5_R
            best_h = u_R.copy()
        
        # Step 4: Line search between u_L and u_R
        # h(alpha) = (1-alpha)*u_R + alpha*u_L
        n_grid = 20
        for alpha in np.linspace(0, 1, n_grid):
            h_test = (1 - alpha) * u_R + alpha * u_L
            # Check bounds
            if h_test.min() >= -1e-10 and h_test.max() <= 1 + 1e-10:
                h_test = np.clip(h_test, 0, 1)
                h_test = _project(h_test, target)
                c5_test = _exact_c5(h_test)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
        
        # Check convergence
        if np.max(np.abs(a_opt - h)) < 1e-6:
            break
        
        # Update h for next iteration
        # Use the best endpoint as new center
        if c5_L <= c5_R:
            h = u_L.copy()
        else:
            h = u_R.copy()
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import linprog
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.array(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _build_corr_matrix(b, dx):
    n = len(b)
    m = 2 * n - 1
    M = np.zeros((m, n))
    for i in range(n):
        q_lo = i
        q_hi = min(m - 1, i + n - 1)
        b_idx = i + n - 1 - np.arange(q_lo, q_hi + 1)
        M[q_lo:q_hi + 1, i] = dx * b[b_idx]
    return M

def _solve_lp_factor(b_fixed, dx, target):
    n = len(b_fixed)
    m = 2 * n - 1
    M = _build_corr_matrix(b_fixed, dx)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([M, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _relax_pair(h, deadline, max_iters=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = np.array(h, dtype=float)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Solve for a given b = 1-h
        b = 1.0 - h
        a_opt = _solve_lp_factor(b, dx, target)
        if a_opt is None:
            break
        a_opt = _project(a_opt, target)
        
        # Solve for b given a = a_opt
        b_opt = _solve_lp_factor(a_opt, dx, target)
        if b_opt is None:
            break
        b_opt = _project(b_opt, target)
        
        # Endpoints
        u_L = a_opt
        u_R = 1.0 - b_opt
        u_R = _project(u_R, target)
        
        c5_L = _exact_c5(u_L)
        c5_R = _exact_c5(u_R)
        
        if c5_L < best_c5:
            best_c5 = c5_L
            best_h = u_L.copy()
        if c5_R < best_c5:
            best_c5 = c5_R
            best_h = u_R.copy()
        
        # Line search between endpoints
        for alpha in np.linspace(0.0, 1.0, 15):
            h_test = (1.0 - alpha) * u_R + alpha * u_L
            h_test = _project(h_test, target)
            c5_test = _exact_c5(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
        
        # Update h
        if c5_L <= c5_R:
            h = u_L.copy()
        else:
            h = u_R.copy()
        
        # Convergence check
        if np.max(np.abs(h - best_h)) < 1e-8:
            break
    
    return best_h, best_c5
