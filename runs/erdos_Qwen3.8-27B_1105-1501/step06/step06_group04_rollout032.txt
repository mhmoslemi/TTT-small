    best_h = h.copy()
    best_obj = _lse_obj_v2(h, beta)
    v = np.zeros(n)  # momentum velocity

def _lse_obj_and_grad(h, beta):
    """Return (M_beta, gradient) of the log-sum-exp smoothed max."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    T = np.correlate(h, 1.0 - h, mode='full')
    S = T * dx
    max_S = float(S.max())
    shifted = beta * (S - max_S)
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    M = max_S + log_sum / beta
    weights = exp_shifted / exp_shifted.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for r in range(m):
        w = weights[r]
        if w < 1e-14:
            continue
        d = r - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            mm = n - d
            grad[d:] += w * (1.0 - h[:mm])
            grad[:mm] += -w * h[d:]
        else:
            g = -d
            mm = n - g
            grad[:mm] += w * (1.0 - h[g:])
            grad[g:] += -w * h[:mm]
    grad *= dx
    return M, grad

    best_M = _lse_obj_and_grad(h, beta)[0]

def _exact_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def _proj(v, target):

        return np.zeros(n)

        return np.ones(n)

            lo = mid
        else:

def _lso_obj(h, beta):
    n = len(h)

    max_S = float(S.max())

def _lso_grad(h, beta):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:

    return _proj(np.interp(xn, xo, h), n_new / 2.0)

            cur = best.copy()
    return best, best_c5
