def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):

    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    # Feasible alpha range
    t_lo, t_hi = -trust_radius, trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    # Golden section search on max of quadratics
    gr = (np.sqrt(5.0) - 1.0) / 2.0

def _minmax_direction(h, S, G, active_indices, radius):
    """Find direction that minimizes max over active lags."""
    n = len(h)
    K = len(active_indices)
    if K == 0:
        return None
    # LP: min t s.t. G[k]@d <= t - S[k] for k in active, |d|_inf <= radius, sum(d) = 0
    # Variables: d (n) + t (1) = n+1
    # Constraints: K rows of (G[active] - I) @ [d; t] <= -S[active]
    #             box constraints
    #             sum(d) = 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([G[active_indices], -np.eye(K)])
    b_ub = -S[active_indices]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-12)
        hi = min(radius, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def _simulated_anneal_pairs(h, deadline, target_c5=None):
    """Simulated annealing on pair mass transfers."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    T = 0.01  # temperature
    T_min = 1e-6
    cooling = 0.995
    rng = np.random.default_rng(123)
    
    interior = np.where((h > 0.01) & (h < 0.99))[0]
    
    while time.time() < deadline and T > T_min:
        if len(interior) < 2:
            break
        i = interior[rng.integers(len(interior))]
        j = interior[rng.integers(len(interior))]
        if i == j:
            continue
        # Try transferring mass from i to j (or vice versa)
        for sign in [1, -1]:
            d = np.zeros(n)
            d[i] = -sign
            d[j] = sign
            alpha, new_c5 = _line_search(h, d, dx, 0.5)
            if alpha == 0:
                continue
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                h = h + alpha * d
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                break
        T *= cooling
        if len(interior) > 10:
            interior = np.where((h > 0.01) & (h < 0.99))[0]
    
    return best_h, best_c5

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _line_search(h, d, dx, trust_radius):
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full') * dx
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo, t_hi = -trust_radius, trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1

            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _build_G(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        # Positive lag: S[mid+r] = sum_{i=0}^{m-1} h[i+r]*(1-h[i])
        # dS/dh[j] for j < r: dx*(1-h[j]) ... no
        # dS/dh[j]: if j >= r and j < n, contributes from i=j-r: dx*(1-h[j-r])
        #           if j < r and j + r < n, contributes from i=j: dx*(-h[j+r])
        # Wait let me redo: S[mid+r] = sum_{i=0}^{m-1} h[i+r]*(1-h[i])
        # dS/dh[j]:
        #   If j is in [r, n-1]: j = i+r for i=j-r, term is (1-h[i]) = (1-h[j-r]), contributes dx*(1-h[j-r])
        #   If j is in [0, m-1]: j = i, term is -h[j+r]*... contributes -dx*h[j+r]
        for j in range(r, n):
            G[mid + r, j] += dx * (1.0 - h[j - r])
        for j in range(0, m):
            G[mid + r, j] += -dx * h[j + r]
        # Negative lag: S[mid-r] = sum_{i=r}^{n-1} h[i]*(1-h[i-r])
        # dS/dh[j]:
        #   If j >= r: j contributes as h[j]*(1-h[j-r]), d/dh[j] = dx*(1-h[j-r])
        #   If j < r: j = i-r for i=j+r, term is -h[i], d/dh[j] = -dx*h[j+r]
        for j in range(r, n):
            G[mid - r, j] += dx * (1.0 - h[j - r])
        for j in range(0, m):
            G[mid - r, j] += -dx * h[j + r]
    return G

def _minmax_lp_dir(h, S, G, active_idx, radius):
    n = len(h)
    K = len(active_idx)
    if K == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([G[active_idx], -np.eye(K)])
    b_ub = -S[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-radius, -h[i] + 1e-11)
        hi = min(radius, 1.0 - h[i] - 1e-11)
        if lo >= hi:
            bounds.append((0.0, 0.0))
        else:
            bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

    h6, c5_6 = _final_polish(best_h, d6)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    
    # Extra: one more min-max round if time
    if time.time() < deadline - 5:
        h7, c5_7 = _minmax_optimize(best_h, deadline, start_K=5)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    
    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_c5(best_h)
    
    return best_h, final_c5, len(best_h)
