import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(R) * dx)

def _project(v, target):
    n = len(v)
    v = np.clip(v, -10, 10)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h, lag_indices=None):
    n = len(h)
    m = 2 * n - 1
    if lag_indices is None:
        lag_indices = np.arange(m)
    G = np.zeros((len(lag_indices), n))
    for idx, r in enumerate(lag_indices):
        d = int(r) - (n - 1)
        if d >= 0:
            mm = n - d
            if mm > 0:
                G[idx, d:] += (1.0 - h[:mm])
                G[idx, :mm] += -h[d:]
        else:
            g = -d
            mm = n - g
            if mm > 0:
                G[idx, :mm] += (1.0 - h[g:])
                G[idx, g:] += -h[:mm]
    return G

def _lse_objective_and_grad(h, beta, dx):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    C = dx * R
    maxC = C.max()
    z = beta * (C - maxC)
    exp_z = np.exp(z)
    logsum = maxC + np.log(exp_z.sum()) / beta
    p = exp_z / exp_z.sum()
    G = _grad_matrix(h)
    grad = dx * (G * p[:, None]).sum(axis=0)
    return logsum, grad

def _solsqp_lse(h_init, beta, maxiter=200):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    
    def fun(y):
        h = y
        val, grad = _lse_objective_and_grad(h, beta, dx)
        return val, grad
    
    constraints = [
        {'type': 'eq', 'fun': lambda y: np.sum(y) - target,
         'jac': lambda y: np.ones(n)},
    ]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(fun, h_init, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
    if result.success or result.fun < _compute_c5(h_init):
        h_new = _project(result.x, target)
        return h_new
    return h_init

def _line_search_exact(h, d, dx, radius):
    n = len(h)
    # R(t) = R0 + t*T - t^2*C  (unscaled)
    R0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    # Feasible interval
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    t_lo = max(t_lo, -radius)
    t_hi = min(t_hi, radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(R0) * dx)
    
    # Golden section search on f(t) = max_r dx*(R0[t] + t*T[t] - t^2*C[t])
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C) * dx)
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C) * dx)
    for _ in range(80):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C) * dx)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C) * dx)
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(R0 + opt_t * T - opt_t * opt_t * C) * dx)
    return opt_t, opt_val

def _solsqp_minimax(h_init, deadline, maxiter=100):
    """Direct minimax SLSQP: minimize M subject to M >= C_r(h) for all r."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    m = 2 * n - 1
    
    h = h_init.copy()
    cur_c5 = _compute_c5(h)
    
    def fun(y):
        return y[0]
    
    def fun_grad(y):
        grad = np.zeros(n + 1)
        grad[0] = 1.0
        return y[0], grad
    
    def ineq_constraints(y):
        h_y = y[1:]
        R = np.correlate(h_y, 1.0 - h_y, mode='full')
        return y[0] - dx * R
    
    def ineq_constraints_jac(y):
        h_y = y[1:]
        G = _grad_matrix(h_y)
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * G
        return J
    
    def eq_fun(y):
        return np.sum(y[1:]) - target
    
    def eq_jac(y):
        j = np.zeros(n + 1)
        j[1:] = 1.0
        return j
    
    constraints = [
        {'type': 'ineq', 'fun': ineq_constraints, 'jac': ineq_constraints_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]
    
    # Initial M
    M0 = cur_c5 + 0.05
    
    y0 = np.concatenate([[M0], h])
    bounds = [(0.0, M0 + 0.5)] + [(0.0, 1.0)] * n
    
    result = minimize(fun_grad, y0, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    
    best_h = h.copy()
    best_c5 = cur_c5
    
    if result.x is not None:
        h_candidate = _project(result.x[1:], target)
        c5_candidate = _compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    return best_h, best_c5

            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix(h, lag_indices=None):
    n = len(h)
    m = 2 * n - 1
    if lag_indices is None:
        lag_indices = np.arange(m)
    G = np.zeros((len(lag_indices), n))
    for idx, r in enumerate(lag_indices):
        d = int(r) - (n - 1)
        if d >= 0:
            mm = n - d
            if mm > 0:
                G[idx, d:] += (1.0 - h[:mm])
                G[idx, :mm] += -h[d:]
        else:
            g = -d
            mm = n - g
            if mm > 0:
                G[idx, :mm] += (1.0 - h[g:])
                G[idx, g:] += -h[:mm]
    return G

def _line_search(h, d, radius):
    n = len(h)
    dx = 2.0 / n
    R0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = radius
    t_lo = -radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(R0) * dx)
    
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C) * dx)
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C) * dx)
    for _ in range(80):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C) * dx)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C) * dx)
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(R0 + opt_t * T - opt_t * opt_t * C) * dx)
    return opt_t, opt_val

def _solsqp_lse(h_init, beta, deadline, maxiter=300):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    
    def objective(y):
        val, grad = _lse_obj_grad(y, beta, dx)
        return val, grad
    
    def _lse_obj_grad(h, beta, dx):
        R = np.correlate(h, 1.0 - h, mode='full')
        C = dx * R
        maxC = C.max()
        z = beta * (C - maxC)
        exp_z = np.exp(z)
        logsum = maxC + np.log(exp_z.sum()) / beta
        p = exp_z / exp_z.sum()
        G = _grad_matrix(h)
        grad = dx * (G * p[:, None]).sum(axis=0)
        return logsum, grad
    
    constraints = [{'type': 'eq', 'fun': lambda y: np.sum(y) - target,
                   'jac': lambda y: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(objective, h_init, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    h_new = _project(result.x, target)
    return h_new

            break

        active = sorted_idx[:top_k]
        
        def fun(y):
            return y[0]

            grad = np.zeros(n + 1)
            grad[0] = 1.0
            return y[0], grad

            R_y = np.correlate(h_y, 1.0 - h_y, mode='full')
            return y[0] - dx * R_y[active]
        
        def ineq_jac(y):
            h_y = y[1:]
            G = _grad_matrix(h_y, active)
            J = np.zeros((top_k, n + 1))
            J[:, 0] = 1.0
            J[:, 1:] = -dx * G
            return J
        
        def eq_fun(y):
            return np.sum(y[1:]) - target
        
        def eq_jac(y):
            j = np.zeros(n + 1)
            j[1:] = 1.0
            return j
        
        constraints = [
            {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        ]
        
        M0 = float(C.max()) + 0.01
        y0 = np.concatenate([[M0], h])
        bounds = [(0.0, M0 + 0.3)] + [(0.0, 1.0)] * n
        
        result = minimize(fun_grad, y0, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 80, 'ftol': 1e-14})
        
        if result.x is not None:
            h_candidate = _project(result.x[1:], target)
            c5_candidate = _compute_c5(h_candidate)

                best_c5 = c5_candidate
                h = h_candidate.copy()
