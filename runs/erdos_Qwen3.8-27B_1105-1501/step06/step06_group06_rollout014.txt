Now for the main optimization loop, let me think about what directions to try:

1. **Active gradient direction**: Average the gradients of the top-K active constraints, project to sum-zero subspace. This gives a direction that (approximately) decreases all active constraints simultaneously.

2. **SVD-based null space**: After computing the active constraint gradient matrix G_active (K x n), find directions in the null space. Then for each null space direction, check if it decreases the max. This is more sophisticated.

3. **LP direction**: Solve the LP: minimize t subject to G_active * d <= -epsilon (decrease each active constraint by at least epsilon), sum(d) = 0, ||d||_inf <= radius. This is what the previous code did.

4. **Random perturbations in null space**: Generate random vectors, project to sum-zero, and do line search.

5. **Structured perturbations**: Add small sine waves of various frequencies (preserving sum=0).

Let me focus on making the line search very efficient and the direction generation smart.

Key insight: The objective is max of quadratics in t. For a given direction d, I can find the exact optimal t. The question is finding good directions.

- The gradient of the max objective at the current point is approximately the average of the active gradients (for the max function, the subdifferential is the convex hull of the active gradients)
- We want a direction d such that G_active * d < 0 (decreasing all active constraints) and sum(d) = 0
- The steepest such direction (in the sense of minimizing the maximum increase) is found by solving a small LP

import numpy as np
from scipy.optimize import minimize, linprog
import time

_slsqp_state = {}

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-14:
        if res > 0:
            mask = (x > 0.0) & (x < 1.0)
            if not np.any(mask):
                mask = x < 1.0
        else:
            mask = (x > 0.0) & (x < 1.0)
            if not np.any(mask):
                mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def corr_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            G[k, :m] += (1.0 - h[s:s + m])
            G[k, s:s + m] += (-h[:m])
        else:
            g = -s
            G[k, g:g + m] += (1.0 - h[:m])
            G[k, :m] += (-h[g:g + m])
    return G

def line_search(h, d, max_alpha=0.3):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Linear term: L[k] = d/dt of corr_k at t=0
    # corr_k = sum d[i]*(1-h[i+s]) - sum h[i]*d[i+s] (for s>=0)
    L = np.zeros(2 * n - 1)
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            L[k] = np.sum(d[:m] * (1.0 - h[s:s+m])) - np.sum(h[:m] * d[s:s+m])
        else:
            g = -s
            L[k] = np.sum(d[g:g+m] * (1.0 - h[:m])) - np.sum(h[g:g+m] * d[:m])
    
    # Quadratic term: Q[k] = -sum d[i]*d[i+s]
    Q = np.zeros(2 * n - 1)
    for s in range(-(n - 1), n):
        k = n - 1 + s
        m = n - abs(s)
        if s >= 0:
            Q[k] = -np.sum(d[:m] * d[s:s+m])
        else:
            g = -s
            Q[k] = -np.sum(d[g:g+m] * d[:m])
    
    # Feasible range
    pos = d > 1e-12
    neg = d < -1e-12
    a_min = -max_alpha
    a_max = max_alpha
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])) * 0.999)
    if np.any(neg):
        a_min = max(a_min, float(np.max(h[neg] / d[neg])) * 0.999)
    
    if a_max - a_min < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Grid search + golden section
    n_grid = 50
    grid = np.linspace(a_min, a_max, n_grid)
    vals = S0[None, :] + grid[:, None] * L[None, :] + grid[:, None]**2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_a = grid[best_i]
    best_v = obj[best_i]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    if 0 < best_i < n_grid - 1:
        a, b = grid[best_i - 1], grid[best_i + 1]
    else:
        a, b = a_min, a_max
    fa = float(np.max(S0 + a * L + a * a * Q)) * dx
    fb = float(np.max(S0 + b * L + b * b * Q)) * dx
    for _ in range(40):
        if b - a < 1e-15:
            break
        if fa < fb:
            b, fb = a, fa
            a = b - gr * (b - a)
            fa = float(np.max(S0 + a * L + a * a * Q)) * dx
        else:
            a, fa = b, fb
            b = a + gr * (b - a)
            fb = float(np.max(S0 + b * L + b * b * Q)) * dx
    mid = (a + b) / 2.0
    mv = float(np.max(S0 + mid * L + mid * mid * Q)) * dx
    if mv < best_v:
        best_a, best_v = mid, mv
    
    return best_a, best_v

def lp_direction(h, active_idx, radius):
    """Find direction minimizing max active gradient dot d, with sum(d)=0, ||d||_inf<=radius."""
    n = len(h)
    dx = 2.0 / n
    G = corr_grads(h)
    G_act = G[active_idx, :]
    K = len(active_idx)
    
    # Variables: d (n) + t (1)
    # Minimize t
    # Subject to: G_act @ d <= t  =>  G_act @ d - t <= 0
    #             sum(d) = 0
    #             -radius <= d_i <= radius
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def active_descent(h_init, deadline, max_iter=1000):
    """Main optimization loop."""
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = 0.08
    stall = 0
    stall_best = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = cur_c5
        
        # Find active constraints
        tol = 0.001
        active = np.where(corr > M - tol)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:10]
        elif len(active) > 30:
            top_order = np.argsort(-corr[active])
            active = active[top_order[:30]]
        
        # Generate directions
        directions = []
        
        # Direction 1: LP based
        d_lp = lp_direction(h, active, radius)
        if d_lp is not None and np.max(np.abs(d_lp)) > 1e-12:
            directions.append(d_lp)
        
        # Direction 2: Average active gradient (negative), projected to sum=0
        G = corr_grads(h)
        grad_avg = G[active, :].mean(axis=0)
        grad_avg -= grad_avg.mean()  # sum to zero
        gn = np.linalg.norm(grad_avg)
        if gn > 1e-12:
            directions.append(-grad_avg / gn * radius)
        
        # Direction 3: SVD null space of active gradients
        if len(active) < n:
            try:
                G_act = G[active, :]
                # Center rows
                U, S_vals, Vt = np.linalg.svd(G_act, full_matrices=False)
                # Null space directions are the last rows of Vt
                null_dim = min(3, Vt.shape[0] - len(active))

                        d_null = Vt[i, :].copy()
                        d_null -= d_null.mean()
                        dn = np.linalg.norm(d_null)
                        if dn > 1e-10:
                            d_null = d_null / dn * radius
                            # Check if it decreases the max
                            directions.append(d_null)
                            directions.append(-d_null)
            except Exception:
                pass
        
        # Direction 4: Random in null space (project random vector to sum=0)
        for _ in range(2):
            d_rand = rng.standard_normal(n)
            d_rand -= d_rand.mean()
            d_rand /= np.max(np.abs(d_rand))
            d_rand *= radius * 0.5

            actual_c5 = exact_c5(h_trial)
            
            if actual_c5 < cur_c5 - 1e-14:
                h = h_trial
                cur_c5 = actual_c5
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                stall_best = 0
                if abs(alpha) > 0.5 * radius:
                    radius = min(radius * 1.1, 0.3)
                break  # restart with new active set
        
        if not improved:

                radius = 0.08

            else:

    return best_h, best_c5

    x_old = (np.arange(n) + 0.5) / n

    top_k = _slsqp_state['top_k']

    c = np.correlate(hx, 1.0 - hx, mode='full') * (2.0 / n)
    return x[n] - c[top_k]
