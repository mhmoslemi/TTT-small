```python
    n = len(h)
    n_rows = 2*n - 1
    total_dim = n + n_rows
    
    # Box bounds for d
    d_lo = np.maximum(-h, -rho)  # actually -min(h, rho)
    d_hi = np.minimum(1.0 - h, rho)
    
    # z bounds
    z_lo = np.zeros(n_rows)
    # z upper: predicted violation could be at most max(R) + rho*max|G| - M + something
    # Let's use a generous upper bound
    z_hi = np.full(n_rows, 10.0)
    
    bounds = list(zip(d_lo, d_hi)) + list(zip(z_lo, z_hi))
    
    # Initial point: d=0, z = max(R - M, 0)
    d0 = np.zeros(n)
    z0 = np.maximum(R - M, 0.0)
    x0 = np.concatenate([d0, z0])
    
    # Objective
    def obj(x):
        d = x[:n]
        z = x[n:]
        return lambda_z * np.sum(z) / n_rows + lambda_n * np.sum(d*d) / n
    
    def grad_obj(x):
        d = x[:n]
        z = x[n:]
        g = np.zeros(total_dim)
        g[:n] = 2 * lambda_n * d / n
        g[n:] = lambda_z / n_rows
        return g

    def eq_con(x):
        return np.sum(x[:n])

        j = np.zeros(total_dim)
        j[:n] = 1.0
        return j

    def ineq_con_jac(x):
        return J_ineq
    
    constraints = [
        {'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac},
        {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_con_jac}
    ]
    
    result = minimize(obj, x0, jac=grad_obj, bounds=bounds,
                      constraints=constraints, method='SLSQP',
                      options={'maxiter': max_iter, 'ftol': 1e-10})
    
    d = result.x[:n].copy()
    d -= np.mean(d)  # enforce sum=0 more precisely
    return d
```

Wait, there's an issue. The ineq constraint as I've written it:
(M - R_k) - (G @ d)_k + z_k >= 0

This means: z_k >= (G @ d)_k + R_k - M

Which means: predicted row k = R_k + (G@d)_k <= M + z_k

So z_k is the amount by which the predicted row k exceeds M. This is correct.

But wait, if M is below the current max(R), then for the rows currently above M, we need z_k > 0 at the initial point (d=0, z_k = R_k - M). This is feasible because z has upper bound.

The QP will try to find d that reduces z (by making predicted rows lower) while keeping d small.

This looks correct. Let me now think about the line search.

After getting direction d from the QP, I do:
1. Compute T = G@d (= correlate(d, 1-h) - correlate(h, d))
2. Compute Q = correlate(d, d, 'full')
3. For alpha in [0, alpha_max]: C(alpha) = max(R + alpha*T - alpha*Q) * dx
4. Find alpha that minimizes C(alpha)

The bounds on alpha:
- alpha_max = min over i where d[i] > 0 of (1-h[i])/d[i], and max over i where d[i] < 0 of h[i]/(-d[i])
- Actually both give the same bound since we need h[i] + alpha*d[i] in [0,1]

Let me also try the negative direction (alpha < 0) since the QP might give a direction where going the other way is better.

```python
import numpy as np
import time
from scipy.optimize import minimize

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        k = d + (n - 1)
        m = n - abs(d)
        if d >= 0:
            G[k, :m] += (1.0 - h[d:d + m])
            G[k, d:] += (-h[:m])
        else:
            r = -d
            G[k, r:] += (1.0 - h[:m])
            G[k, :m] += (-h[r:r + m])
    return G

def make_qp_context(h, R, G, M, rho, lambda_z, lambda_n):
    n = len(h)
    n_rows = 2 * n - 1
    total_dim = n + n_rows
    d_lo = np.maximum(-h, -rho)
    d_hi = np.ones(n) - h
    d_hi = np.minimum(d_hi, rho)
    z_hi = np.full(n_rows, 20.0)
    bounds = list(zip(d_lo, d_hi)) + [(0.0, zh) for zh in z_hi]
    ineq_vals = M - R
    J_ineq = np.zeros((n_rows, total_dim))
    J_ineq[:, :n] = -G
    J_ineq[:, n:] = np.eye(n_rows)
    J_eq = np.zeros(total_dim)
    J_eq[:n] = 1.0
    return n, n_rows, total_dim, bounds, ineq_vals, J_ineq, J_eq

def qp_obj(x, ctx):
    n, n_rows, total_dim, bounds, ineq_vals, J_ineq, J_eq = ctx
    lam_z, lam_n = ctx['lz'], ctx['ln']
    d = x[:n]
    z = x[n:]
    return lam_z * np.sum(z) / n_rows + lam_n * np.sum(d * d) / n

def qp_grad(x, ctx):
    n, n_rows, total_dim, bounds, ineq_vals, J_ineq, J_eq = ctx
    lam_z, lam_n = ctx['lz'], ctx['ln']
    d = x[:n]
    g = np.zeros(total_dim)
    g[:n] = 2.0 * lam_n * d / n
    g[n:] = lam_z / n_rows
    return g

def qp_ineq(x, ctx):
    n, n_rows, total_dim, bounds, ineq_vals, J_ineq, J_eq = ctx
    d = x[:n]
    z = x[n:]
    return ineq_vals - J_ineq[:, :n] @ d + z

def qp_eq(x, ctx):
    n, n_rows, total_dim, bounds, ineq_vals, J_ineq, J_eq = ctx
    return np.sum(x[:n])

def solve_direction_qp(h, R, G, M, rho, lambda_z, lambda_n, max_iter=200):
    n = len(h)
    n_rows = 2 * n - 1
    base_ctx = make_qp_context(h, R, G, M, rho, lambda_z, lambda_n)
    ctx = dict(base_ctx)
    ctx['lz'] = lambda_z
    ctx['ln'] = lambda_n
    total_dim = base_ctx[2]
    z0 = np.maximum(R - M, 0.0)
    x0 = np.concatenate([np.zeros(n), z0])
    constraints = [
        {'type': 'eq', 'fun': lambda x: qp_eq(x, ctx), 'jac': lambda x: base_ctx[6]},
        {'type': 'ineq', 'fun': lambda x: qp_ineq(x, ctx), 'jac': lambda x: base_ctx[5]},
    ]
    try:
        result = minimize(qp_obj, x0, jac=qp_grad, args=(ctx,),
                          bounds=ctx[3], constraints=constraints,
                          method='SLSQP', options={'maxiter': max_iter, 'ftol': 1e-12})
        d = result.x[:n].copy()
    except Exception:
        return None
    if not np.all(np.isfinite(d)):
        return None
    d -= np.mean(d)
    if np.max(np.abs(d)) < 1e-14:
        return None
    return d

def exact_line_search(h, d, R=None, max_samples=80):
    n = len(h)
    dx = 2.0 / n
    if R is None:
        R = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    alpha_max = 5.0
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min((-h[neg]) / d[neg])))
    if alpha_max < 1e-10:
        return 0.0, c5_exact(h)
    best_alpha = 0.0
    best_val = c5_exact(h)
    alphas = np.linspace(0, alpha_max, max_samples)
    vals = np.array([float(np.max(R + a * T - a * a * Q)) * dx for a in alphas])
    idx = int(np.argmin(vals))
    best_alpha = float(alphas[idx])
    best_val = float(vals[idx])
    if 0 < idx < max_samples - 1:
        lo_a = float(alphas[max(0, idx - 3)])
        hi_a = float(alphas[min(max_samples - 1, idx + 3)])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = lo_a, hi_a
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(R + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(R + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(50):
            if b - a < 1e-14:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = float(np.max(R + c1 * T - c1 * c1 * Q)) * dx
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(R + c2 * T - c2 * c2 * Q)) * dx
        opt = (a + b) / 2.0
        opt_val = float(np.max(R + opt * T - opt * opt * Q)) * dx
        if opt_val < best_val:
            best_alpha, best_val = opt, opt_val
    end_val = float(np.max(R + alpha_max * T - alpha_max * alpha_max * Q)) * dx
    if end_val < best_val:
        best_alpha, best_val = alpha_max, end_val
    return best_alpha, best_val

def pair_golden(h, i, j, S0):
    n = len(h)
    dx = 2.0 / n
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(45):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def _qp_phase(h, deadline, rng, target_c5=0.3808):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    G = compute_grad_matrix(h)
    cur_c5 = float(np.max(R)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    rhos = [0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005]
    lam_zs = [0.5, 0.3, 0.2, 0.1]
    lam_ns = [0.5, 1.0, 2.0]
    stall = 0
    stall_total = 0
    while time.time() < deadline and stall_total < len(rhos) * len(lam_zs) * len(lam_ns):
        improved_this_round = False
        for rho in rhos:
            if time.time() > deadline:
                break
            for lam_z in lam_zs:
                if time.time() > deadline:
                    break
                M_raw = target_c5 / dx
                d = solve_direction_qp(h, R, G, M_raw, rho, lam_z, 1.0, max_iter=150)
                if d is None:
                    continue
                alpha, new_c5 = exact_line_search(h, d, R)
                if alpha > 1e-10 and new_c5 < best_c5 - 1e-13:
                    h_new = h + alpha * d
                    h_new = project(h_new, target)
                    actual_c5 = c5_exact(h_new)
                    if actual_c5 < best_c5 - 1e-13:
                        h = h_new
                        R = np.correlate(h, 1.0 - h, mode='full')
                        G = compute_grad_matrix(h)
                        cur_c5 = actual_c5
                        best_c5 = actual_c5
                        best_h = h.copy()
                        stall = 0
                        stall_total = 0
                        improved_this_round = True
                        break
                    else:
                        h = h_new
                        R = np.correlate(h, 1.0 - h, mode='full')
                        G = compute_grad_matrix(h)
                alpha_neg, new_c5_neg = exact_line_search(h, -d, R)
                if alpha_neg > 1e-10 and new_c5_neg < best_c5 - 1e-13:
                    h_new = h - alpha_neg * d
                    h_new = project(h_new, target)
                    actual_c5 = c5_exact(h_new)
                    if actual_c5 < best_c5 - 1e-13:
                        h = h_new
                        R = np.correlate(h, 1.0 - h, mode='full')
                        G = compute_grad_matrix(h)
                        cur_c5 = actual_c5
                        best_c5 = actual_c5
                        best_h = h.copy()
                        stall = 0
                        stall_total = 0
                        improved_this_round = True
                        break
            if improved_this_round:
                break
        if not improved_this_round:
            stall += 1
            stall_total += 1
            if stall >= 3:
                break
        if cur_c5 < target_c5 - 0.001:
            target_c5 = cur_c5 - 0.0005
    return best_h, best_c5

def _pair_phase(h, deadline, rng, n_pairs=20):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    G = compute_grad_matrix(h)
    cur_c5 = float(np.max(R)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < deadline and stall < 60:
        scores = R * dx
        max_s = float(np.max(scores))
        thresh = max_s - 0.004 * max_s
        active = np.where(scores >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(-scores)[:10]
        weights = np.exp((R[active] - R[active].max()) / (0.01 * max_s + 1e-15))
        weights /= np.sum(weights)
        g = G[active, :] @ weights
        g -= np.mean(g)
        donors = np.argsort(-g)[:n_pairs]
        receivers = np.argsort(g)[:n_pairs]
        best_alpha = 0.0
        best_new = cur_c5
        best_pair = None
        tried = set()
        for i in donors:
            if time.time() > deadline:
                break
            for j in receivers:
                if i == j:
                    continue
                key = (min(i, j), max(i, j))
                if key in tried:
                    continue
                tried.add(key)
                alpha, new_c5 = pair_golden(h, i, j, R)
                if alpha > 1e-10 and new_c5 < best_new - 1e-14:
                    best_new = new_c5
                    best_alpha = alpha
                    best_pair = (i, j)
        if best_pair is not None and best_new < cur_c5 - 1e-13:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R = np.correlate(h, 1.0 - h, mode='full')
            G = compute_grad_matrix(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.4:
                idx = rng.integers(n, size=6)
                for ix in idx:
                    h[ix] += rng.uniform(-0.03, 0.03)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R = np.correlate(h, 1.0 - h, mode='full')
                G = compute_grad_matrix(h)
                stall = 0
    return best_h, best_c5

def _random_pair_phase(h, deadline, rng):
    n = len(h)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def _binary_phase(h, deadline, rng):
    n = len(h)
    target = n // 2
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    cur_c5 = c5_exact(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    R = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    ones = list(np.where(h_bin > 0.5)[0])
    zeros = list(np.where(h_bin <= 0.5)[0])
    patience = 0
    while time.time() < deadline and patience < 6000 and len(ones) > 0 and len(zeros) > 0:
        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = c5_exact(h_bin)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            R = np.correlate(h_bin, 1.0 - h_bin, mode='full')
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            patience = 0
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            patience += 1
    return best_h, best_c5

def _gradient_polish(h, deadline):

    target = n / 2.0

        active = np.where(R >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(-R)[:8]
        weights = np.exp((R[active] - R[active].max()) / (0.005 * max_r + 1e-15))
        weights /= np.sum(weights)
        g = G[active, :] @ weights
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:

        d = -g / gn
        pos = d > 1e-10
        neg = d < -1e-10

        if np.any(pos):

        if hi - lo < 1e-12:
            break
        alpha, new_c5 = exact_line_search(h, d, R)
        if alpha > 1e-10 and new_c5 < cur_c5 - 1e-14:
            h = h + alpha * d
            h = project(h, target)

            R = np.correlate(h, 1
0 - h, mode='full')
            G = compute_grad_matrix(h)
            if cur_c5 < best_c5:
```

    return float(np.max(corr) * dx)


def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        k = d + (n - 1)
        m = n - abs(d)
        if d >= 0:
            G[k, :m] += (1.0 - h[d:d + m])
            G[k, d:] += (-h[:m])
        else:
            r = -d
            G[k, r:] += (1.0 - h[:m])
            G[k, :m] += (-h[r:r + m])
    return G


def solve_direction_qp(h, R, G, M, rho, lam_z, lam_n, max_iter=150):
    n = len(h)
    n_rows = 2 * n - 1
    total_dim = n + n_rows
    d_lo = np.maximum(-h, -rho)
    d_hi = np.minimum(1.0 - h, rho)
    bounds = list(zip(d_lo, d_hi))
    for _ in range(n_rows):
        bounds.append((0.0, 20.0))
    ineq_vals = M - R
    J_ineq = np.zeros((n_rows, total_dim))
    J_ineq[:, :n] = -G
    J_ineq[:, n:] = np.eye(n_rows)
    J_eq = np.zeros(total_dim)
    J_eq[:n] = 1.0

    def _obj(x):
        d = x[:n]
        z = x[n:]
        return lam_z * np.sum(z) / n_rows + lam_n * np.dot(d, d) / n

    def _grad(x):
        g = np.zeros(total_dim)
        g[:n] = 2.0 * lam_n * x[:n] / n
        g[n:] = lam_z / n_rows
        return g

    def _eq(x):
        return float(np.sum(x[:n]))

    def _eq_jac(x):
        return J_eq

    def _ineq(x):
        return ineq_vals - J_ineq[:, :n] @ x[:n] + x[n:]

    def _ineq_jac(x):
        return J_ineq

    z0 = np.maximum(R - M, 0.0)
    x0 = np.concatenate([np.zeros(n), z0])
    constraints = [
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
    ]
    try:
        result = minimize(_obj, x0, jac=_grad, bounds=bounds,
                          constraints=constraints, method='SLSQP',
                          options={'maxiter': max_iter, 'ftol': 1e-12})
        d = result.x[:n].copy()
    except Exception:
        return None
    if not np.all(np.isfinite(d)):
        return None
    d -= np.mean(d)
    if np.max(np.abs(d)) < 1e-14:
        return None
    return d


def exact_line_search(h, d, R, n_samples=80):
    n = len(h)
    dx = 2.0 / n
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    pos = d > 1e-12
    neg = d < -1e-12
    alpha_max = 5.0
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_max = min(alpha_max, float(np.min((-h[neg]) / d[neg])))
    if alpha_max < 1e-10:
        return 0.0, c5_exact(h)
    alphas = np.linspace(0, alpha_max, n_samples)
    vals = np.array([float(np.max(R + a * T - a * a * Q)) * dx for a in alphas])
    idx = int(np.argmin(vals))
    best_alpha = float(alphas[idx])
    best_val = float(vals[idx])
    if 0 < idx < n_samples - 1:
        lo_a = float(alphas[max(0, idx - 3)])
        hi_a = float(alphas[min(n_samples - 1, idx + 3)])
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = lo_a, hi_a
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        f1 = float(np.max(R + c1 * T - c1 * c1 * Q)) * dx
        f2 = float(np.max(R + c2 * T - c2 * c2 * Q)) * dx
        for _ in range(50):
            if b - a < 1e-14:
                break
            if f1 < f2:
                b, c2, f2 = c2, c1, f1
                c1 = b - gr * (b - a)
                f1 = float(np.max(R + c1 * T - c1 * c1 * Q)) * dx
            else:
                a, c1, f1 = c1, c2, f2
                c2 = a + gr * (b - a)
                f2 = float(np.max(R + c2 * T - c2 * c2 * Q)) * dx
        opt = (a + b) / 2.0
        opt_val = float(np.max(R + opt * T - opt * opt * Q)) * dx
        if opt_val < best_val:
            best_alpha, best_val = opt, opt_val
    end_val = float(np.max(R + alpha_max * T - alpha_max * alpha_max * Q)) * dx
    if end_val < best_val:
        best_alpha, best_val = alpha_max, end_val
    return best_alpha, best_val


def pair_golden(h, i, j, R):
    n = len(h)
    dx = 2.0 / n
    lo = -min(h[j], 1.0 - h[i])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R + c2 * T - c2 * c2 * C)) * dx
    for _ in range(45):
        if b - a < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(R + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(R + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(R + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val


def _qp_phase(h, deadline, rng, target_c5=0.3808):
    n = len(h)
    target = n / 2.0
    R = np.correlate(h, 1.0 - h, mode='full')
    G = compute_grad_matrix(h)
    cur_c5 = float(np.max(R)) * (2.0 / n)
    best_h = h.copy()
    best_c5 = cur_c5
    rhos = [0.2, 0.15, 0.1, 0.07, 0.05, 0.035, 0.025, 0.018, 0.012, 0.008, 0.005, 0.003]
    lam_zs = [0.5, 0.3, 0.15, 0.08, 0.04]
    lam_ns = [0.5, 1.0, 2.0, 4.0]
    stall = 0
    while time.time() < deadline and stall < 8:
        improved = False
        for rho in rhos:
            if time.time() > deadline:
                break
            for lz in lam_zs:
                if time.time() > deadline:
                    break
                for ln in lam_ns:
                    if time.time() > deadline:
                        break
                    M_raw = min(target_c5, cur_c5 * 0.997) / (2.0 / n)
                    d = solve_direction_qp(h, R, G, M_raw, rho, lz, ln, max_iter=120)
                    if d is None:
                        continue
                    found = False
                    for sign in [1, -1]:
                        sd = sign * d
                        alpha, new_c5 = exact_line_search(h, sd, R)
                        if alpha > 1e-10 and new_c5 < best_c5 - 1e-13:
                            h_new = project(h + alpha * sd, target)
                            actual = c5_exact(h_new)
                            if actual < best_c5 - 1e-14:
                                h = h_new
                                R = np.correlate(h, 1.0 - h, mode='full')
                                G = compute_grad_matrix(h)
                                cur_c5 = actual
                                best_c5 = actual
                                best_h = h.copy()
                                stall = 0
                                improved = True
                                found = True
                                break
                    if found:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if cur_c5 > target_c5:
                target_c5 = cur_c5 - 0.0001 * stall
        if cur_c5 < target_c5:
            target_c5 = cur_c5 - 0.0002
    return best_h, best_c5
