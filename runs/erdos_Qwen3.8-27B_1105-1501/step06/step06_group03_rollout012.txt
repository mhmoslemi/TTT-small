def _line_search(h, d, dx, radius):
    """Line search along direction d, respecting bounds."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    
    pos = d > 1e-15
    neg = d < -1e-15
    t_hi = radius
    t_lo = -radius
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max(h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(60):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    return opt_t, float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx

def _lp_direction(h, S, n, K=15, radius=0.1):
    """Find LP direction that decreases the maximum correlation."""
    M = float(S.max())
    tol = max(0.03 * M, 1e-6)

    c_vec = np.zeros(n + 1)

    A_ub[:, n] = 1.0

def _swap_sa(h_init, deadline, rng, T_init=0.003, T_min=1e-7, cooldown=0.99995):

    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    patience = 0
    max_patience = 20000
    
    while time.time() < deadline:
        if patience > max_patience:
            if T <= T_min * 1.5:
                break
            T = max(T_min, T * 0.95)
            patience = 0
        
        for _ in range(1000):
            # Pick two random positions and swap their values
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Swap
            h[i], h[j] = h[j], h[i]
            new_c5 = _exact_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):

                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
            else:
                h[i], h[j] = h[j], h[i]
        T = max(T_min, T * cooldown)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks-1, n-1), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(len(boundaries) - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)
