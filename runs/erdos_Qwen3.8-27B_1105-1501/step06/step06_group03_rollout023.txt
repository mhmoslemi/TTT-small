def _project(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(v)

        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))

            lo = mid
        else:

def _exact_c5(h):

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
                ones_list = np.where(h > 0.5)[0]
                zeros_list = np.where(h <= 0.5)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _focus_optimize(h, deadline, K=18, radius_init=0.12):
    global dx_val
    n = len(h)
    dx_val = 2.0 / n
    best_h, best_c5 = _lp_loop(h, deadline, radius_init=radius_init, K=K, use_smooth=False)
    rem = deadline - (time.time() + 1)
    if rem > 15:
        h2, c5_2 = _smooth_grad_loop(best_h, time.time() + rem * 0.4)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        rem2 = deadline - (time.time() + 1)
        if rem2 > 10:
            h3, c5_3 = _lp_loop(best_h, time.time() + rem2, radius_init=0.06, K=K, use_smooth=True)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global dx_val
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.985

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    dx_val = 2.0 / n0
    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Aggressive LP optimization from initial (22% time)
    d1 = t0 + budget_s * 0.22
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.14, K=18, use_smooth=False)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Smoothed gradient to escape (10% time)
    d2 = t0 + budget_s * 0.32
    h2, c5_2 = _smooth_grad_loop(best_h, d2, mu_init=0.01 * best_c5)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: LP with smooth backup on same n (20% time)
    d3 = t0 + budget_s * 0.52
    h3, c5_3 = _focus_optimize(best_h, d3, K=20, radius_init=0.10)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: Multi-n exploration (25% time)
    d4 = t0 + budget_s * 0.77
    base_n = len(best_h)
    n_candidates = []
    for dn in [0, -2, 2, -4, 4, -6, 6, -10, 10, -14, 14, -20, 20, -30, 30, -40, 40]:
        nt = base_n + dn
        if 60 <= nt <= 300 and nt % 2 == 0:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))
    time_per_n = max(5.0, (d4 - time.time()) / max(len(n_candidates), 1))
    for nt in n_candidates:
        if time.time() > d4 - 10:
            break
        if nt == base_n:
            continue
        h_r = _resample(best_h, nt)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.008:
            sub_dl = min(d4 - 15, time.time() + time_per_n)
            h_r, c5_r = _focus_optimize(h_r, sub_dl, K=14, radius_init=0.10)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 5: Binary SA on best + LP polish (12% time)
    d5 = t0 + budget_s * 0.89
    h5a, c5_5a = _binary_sa(best_h, t0 + budget_s * 0.84, rng)
    if c5_5a < best_c5:
        best_c5 = c5_5a
        best_h = h5a.copy()
    if time.time() < d5 - 5:
        h5b, c5_5b = _lp_loop(best_h, d5, radius_init=0.06, K=16)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            best_h = h5b.copy()

    # Phase 6: Pair refinement (7% time)
    d6 = t0 + budget_s * 0.96
    h6, c5_6 = _pair_refine(best_h, d6, max_passes=150)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Phase 7: Final multi-pair + LP polish
    if time.time() < deadline - 3:
        h7, c5_7 = _multi_pair_refine(best_h, deadline - 10)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    if time.time() < deadline - 2:
        h8, c5_8 = _lp_loop(best_h, deadline, radius_init=0.03, K=12)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final validation and projection
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
