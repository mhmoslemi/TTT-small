import numpy as np
import time

def _c5(h):
    """Compute C5 objective."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v, target):
    """Project v to [0,1]^n with sum = target using water-filling."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    x = np.clip(v - theta, 0, 1)
    res = target - x.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0, 1)
    return x

def _coord_descent(h, deadline, max_sweeps=1000, grid_size=32):
    """Coordinate descent with exact line search per coordinate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    cur = _c5(h)
    best_h = h.copy()
    best_c = cur
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Try optimal value for coordinate i
            h_save = h[i]
            # Grid search
            grid = np.linspace(0, 1, grid_size)
            best_val = cur
            best_v = h_save
            for v in grid:
                h[i] = v
                c = _c5(h)
                if c < best_val - 1e-14:
                    best_val = c
                    best_v = v
            h[i] = best_v
            if best_v != h_save:
                improved = True
                cur = best_val
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
        if not improved:
            break
    # Final projection to maintain constraint
    best_h = _project(best_h, target)
    final_c = _c5(best_h)
    return best_h, final_c
