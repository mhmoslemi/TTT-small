import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _corr_matrix(h):
    """Compute full correlation array."""
    return np.correlate(h, 1.0 - h, mode='full')

def _soft_direction(h, gamma):
    """Compute soft-max weighted gradient direction (negative = descent)."""
    n = len(h)
    dx = 2.0 / n
    corr = _corr_matrix(h) * dx
    M = np.max(corr)
    # Soft-max weights
    w = np.exp(gamma * (corr - M))
    p = w / np.sum(w)
    
    # Gradient of each correlation w.r.t. h
    # For lag s (offset from center n-1):
    # d(c_s)/d(h[j]) = (1-h[j+s]) if j+s in range - h[j-s] if j-s in range
    L = len(corr)  # 2n-1
    grad = np.zeros(n)
    
    for k in range(L):
        s = k - (n - 1)  # lag offset
        weight = p[k] * dx
        if abs(weight) < 1e-15:
            continue
        # For this lag s, the gradient contribution to h[j]:
        # h[j] appears as first factor when i=j: contributes (1-h[j+s]) if 0<=j+s<n
        # h[j] appears as second factor when i=j-s: contributes -h[j-s] if 0<=j-s<n
        if s >= 0:
            # i ranges from 0 to n-1-s
            # h[j] as first factor: j in [0, n-1-s], contributes (1-h[j+s])
            # h[j] as second factor: j-s in [0, n-1-s], i.e., j in [s, n-1], contributes -h[j-s]
            # Combined: for j in [0, n-1-s]: +weight*(1-h[j+s])
            #           for j in [s, n-1]: -weight*h[j-s]
            m = n - s
            grad[:m] += weight * (1.0 - h[s:s+m])
            grad[s:] += -weight * h[:m]
        else:
            g = -s
            # Similar logic with g instead of s
            m = n - g
            grad[g:] += weight * (1.0 - h[:m])
            grad[:m] += -weight * h[g:g+m]
    
    # Center to preserve sum
    grad -= np.mean(grad)
    return -grad  # Negative gradient = descent direction

def _exact_line_search(h, d, n_grid=64):
    """Minimize exact C5 along h + alpha*d for feasible alpha."""
    n = len(h)
    dx = 2.0 / n
    
    # Quadratic form coefficients
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    
    # Feasible range
    pos = d > 1e-14
    neg = d < -1e-14
    alpha_hi = np.inf
    alpha_lo = -np.inf
    if np.any(pos):
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max(h[neg] / (-d[neg]))))
    alpha_lo = max(alpha_lo, -1.0)
    alpha_hi = min(alpha_hi, 1.0)
    
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, _exact_c5(h)
    
    # Evaluate on grid
    alphas = np.linspace(alpha_lo, alpha_hi, n_grid)
    # Vectorized evaluation
    # c(alpha) = c0 + alpha*L + alpha^2*Q
    # f(alpha) = max_k(c0[k] + alpha*L[k] + alpha^2*Q[k]) * dx
    vals = c0[None, :] + alphas[:, None] * L[None, :] + alphas[:, None]**2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    
    best_i = int(np.argmin(obj))
    best_alpha = alphas[best_i]
    best_val = obj[best_i]
    
    # Golden section refinement
    if 0 < best_i < n_grid - 1:
        a = alphas[best_i - 1]
        b = alphas[best_i + 1]
    elif best_i == 0:
        a = alpha_lo
        b = alphas[1]
    else:
        a = alphas[-2]
        b = alpha_hi
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    # Evaluate current
    def f_alpha(a_val):
        c = c0 + a_val * L + a_val**2 * Q
        return float(np.max(c) * dx)
    
    fa = f_alpha(a)
    fb = f_alpha(b)
    for _ in range(50):
        if b - a < 1e-12:
            break
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        fc1 = f_alpha(c1)
        fc2 = f_alpha(c2)
        if fc1 < fc2:
            b, fb = c1, fc1
        else:
            a, fa = c2, fc2
    
    mid = (a + b) / 2.0
    fmid = f_alpha(mid)
    if fmid < best_val:
        best_alpha, best_val = mid, fmid
    elif fa < best_val:
        best_alpha, best_val = a, fa
    elif fb < best_val:
        best_alpha, best_val = b, fb
    
    return best_alpha, best_val

def _perturb(h, rng, scale=0.01):
    """Add random perturbation and reproject."""
    n = len(h)
    d = rng.normal(0, 1, n)
    d -= np.mean(d)  # Preserve sum
    h_new = h + scale * d
    return _project(h_new, n / 2.0)

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _soft_descent(h, deadline, max_iter=500, gamma_start=5, gamma_end=500):
    """Main soft-gradient descent loop."""
    n = len(h)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    cur_c5 = best_c5
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Adaptive gamma
        frac = it / max(1, max_iter)
        gamma = gamma_start * (gamma_end / gamma_start) ** frac
        
        d = _soft_direction(h, gamma)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            stall += 10
            continue
        
        d = d / d_norm * 0.5  # Scale direction
        
        alpha, new_val = _exact_line_search(h, d)
        
        if alpha != 0 and new_val < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, n / 2.0)
            actual_c5 = _exact_c5(h_new)
            if actual_c5 < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Perturbation for escape

            cur_c5 = _exact_c5(h)
            stall = 0
    
    return best_h, best_c5

import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target=None):
    n = len(v)
    if target is None:
        target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _soft_grad(h, gamma):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(corr))
    w = np.exp(gamma * (corr - M))
    sw = float(np.sum(w))
    if sw < 1e-300:
        p = np.full(len(corr), 1.0 / len(corr))
    else:
        p = w / sw
    grad = np.zeros(n)
    for k in range(len(corr)):
        s = k - (n - 1)
        wt = p[k] * dx
        if wt < 1e-15:
            continue
        if s >= 0:
            m = n - s
            grad[:m] += wt * (1.0 - h[s:s + m])
            grad[s:] += -wt * h[:m]
        else:
            g = -s
            m = n - g
            grad[g:] += wt * (1.0 - h[:m])
            grad[:m] += -wt * h[g:g + m]
    grad -= np.mean(grad)
    return -grad

def _line_search(h, d):
    n = len(h)
    dx = 2.0 / n
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    pos = d > 1e-14
    neg = d < -1e-14
    alpha_lo = -1.0
    alpha_hi = 1.0
    if np.any(pos):
        alpha_lo = max(alpha_lo, float(np.max(-h[pos] / d[pos])))
        alpha_hi = min(alpha_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        alpha_lo = max(alpha_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        alpha_hi = min(alpha_hi, float(np.max(-h[neg] / d[neg])))
    if alpha_hi - alpha_lo < 1e-14:
        return 0.0, _exact_c5(h)
    n_grid = 80
    alphas = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = c0[None, :] + alphas[:, None] * L[None, :] + alphas[:, None] ** 2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    best_i = int(np.argmin(obj))
    best_alpha = float(alphas[best_i])
    best_val = float(obj[best_i])
    if 0 < best_i < n_grid - 1:
        a, b = float(alphas[best_i - 1]), float(alphas[best_i + 1])
    elif best_i == 0:
        a, b = float(alpha_lo), float(alphas[1])
    else:
        a, b = float(alphas[-2]), float(alpha_hi)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    def _f(a_val):
        c = c0 + a_val * L + a_val * a_val * Q
        return float(np.max(c) * dx)
    fa, fb = _f(a), _f(b)
    for _ in range(60):
        if b - a < 1e-13:
            break
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        fc1, fc2 = _f(c1), _f(c2)
        if fc1 < fc2:
            b, fb = c1, fc1
        else:
            a, fa = c2, fc2
    mid = (a + b) / 2.0
    fmid = _f(mid)
    if fmid < best_val:
        best_alpha, best_val = mid, fmid
    elif fa < best_val:
        best_alpha, best_val = a, fa
    elif fb < best_val:
        best_alpha, best_val = b, fb
    return best_alpha, best_val

def _descent(h, deadline, rng, max_iter=3000, gamma_lo=3.0, gamma_hi=5000.0):
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, max_iter))
        gamma = gamma_lo * (gamma_hi / gamma_lo) ** frac
        d = _soft_grad(h, gamma)
        dn = float(np.linalg.norm(d))
        if dn < 1e-15:
            stall += 5
            continue
        d = d / dn * 0.3
        d -= np.mean(d)
        alpha, predicted = _line_search(h, d)
        if abs(alpha) > 1e-14 and predicted < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, n / 2.0)
            actual = _exact_c5(h_new)
            if actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 15:
            scale = 0.003 + 0.004 * min(stall / 30.0, 1.0)
            pert = rng.normal(0, 1, n)
            pert -= np.mean(pert)
            h = _project(h + scale * pert, n / 2.0)

    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_relax(h, rng):
    n = len(h)
    target = n // 2
    order = np.argsort(-h)

    h_b[order[:target]] = 1.0
