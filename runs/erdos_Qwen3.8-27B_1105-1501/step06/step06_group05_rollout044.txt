    n = len(h)
    x0 = np.concatenate([h, [z0]])
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    # Inequality constraints: z - slide_j(h) >= 0 for j in active_idx
    def ineq_fun(x):
        h_x = x[:n]
        z_x = x[n]
        sl = _slides(h_x)
        return z_x - sl[active_idx]
    
    def ineq_jac(x):
        h_x = x[:n]
        # Returns shape (len(active_idx), n+1)
        m = len(active_idx)
        J = np.zeros((m, n+1))
        for i, j in enumerate(active_idx):
            J[i, :n] = -_slide_grad(h_x, j)  # d(z - slide_j)/dh = -d(slide_j)/dh

import numpy as np
import time
from scipy.optimize import minimize


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) < 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _slides(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _slide_grad_one(h, j):
    n = len(h)
    g = np.zeros(n)
    lag = j - (n - 1)
    if lag >= 0:
        m1 = np.arange(lag, n)
        g[m1] += (1.0 - h[m1 - lag])
        m2 = np.arange(0, n - lag)
        g[m2] -= h[m2 + lag]
    else:
        q = -lag
        m1 = np.arange(q, n)
        g[m1] += (1.0 - h[m1 - q])
        m2 = np.arange(0, n - q)
        g[m2] -= h[m2 + q]
    return g


def _slide_grad_all(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for j in range(2 * n - 1):
        G[j] = _slide_grad_one(h, j)
    return G


def _slsqp_solve(h, z0, active_idx, max_iter=40):
    n = len(h)
    x0 = np.concatenate([h, [z0]])

    def obj(x):
        return x[n]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:n]
        zx = x[n]
        sl = _slides(hx)
        return zx - sl[active_idx]

    def ineq_jac(x):
        hx = x[:n]
        m = len(active_idx)
        J = np.zeros((m, n + 1))
        for i, j in enumerate(active_idx):
            J[i, :n] = -_slide_grad_one(hx, j)
            J[i, n] = 1.0
        return J

    def eq_fun(x):
        return float(np.sum(x[:n])) - n / 2.0

    def eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J

    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success or res.fun < z0:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return h, z0


def _slsqp_active_set(h, max_rounds=5, top_m=40, time_limit=5.0):
    n = len(h)
    deadline = time.time() + time_limit
    cur = h.copy()
    sl = _slides(cur)
    z0 = float(np.max(sl)) + 1e-6
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        idx = np.argsort(-sl)[:top_m]
        idx = list(idx)
        h_new, z_new = _slsqp_solve(cur, z0, idx, max_iter=30)
        h_new = _project(h_new, n / 2.0)
        f_new = _exact_c5(h_new)
        f_cur = _exact_c5(cur)
        if f_new < f_cur - 1e-14:
            cur = h_new
            sl = _slides(cur)
            z0 = float(np.max(sl)) + 1e-6
        else:
            break
    return cur


def _line_search_step(h, d, corr, dx):
    n = len(h)
    target = n / 2.0
    pos = d > 1e-14
    neg = d < -1e-14
    low = [0.0]
    up = [1e30]
    if np.any(pos):
        low.append(float(np.min(-h[pos] / d[pos])))
        up.append(float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        low.append(float(np.min((1.0 - h[neg]) / d[neg])))
        up.append(float(np.min(-h[neg] / d[neg])))
    L = max(low)
    U = min(up)
    if U - L < 1e-12:
        return h, None
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = corr[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = 0.6180339887
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(35):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(corr + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(corr + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    if obj[i] < float(np.max(corr + t * T - t * t * Q)) * dx:
        t = grid[i]
    if abs(t) < 1e-14:
        return h, None
    hn = h + t * d
    if hn.min() < -1e-10 or hn.max() > 1.0 + 1e-10:
        hn = _project(hn, target)
    return hn, t


def _pair_swap(h, i, j):
    n = len(h)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    corr = _slides(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    hn, t = _line_search_step(h, d, corr_raw, dx)
    if hn is not None:
        return hn
    return h


def _optimize_hybrid(h, deadline, rng, n_slsqp=3, n_ls=50, pair_frac=0.30, rand_frac=0.10):

    target = n / 2.0
    cur = h.copy()
    best = h.copy()

    stall = 0
    iters = 0
    while time.time() < deadline:

            f = _exact_c5(cur)

                best_f = f
                best = cur.copy()
                stall = 0
                continue
            else:
                cur = best.copy()
                stall += 1
                continue

        r = rng.random()
        if r < pair_frac:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                i = (i + 1) % n
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0

            d -= d.mean()
        else:
            sl = _slides(cur)

            d = -G

        nrm = float(np.linalg.norm(d))

            continue
        d /= nrm

            stall += 1
        if stall > 100:

def _sine_init(n, freq, phase, rng):
    x = np.arange(n)
    base = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    base += rng.normal(0, 0.02, n)
    return _project(base, n / 2.0)


def _block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bl = max(1, int(rng.integers(1, n // 4 + 1)))
        if rng.random() < 0.5:
            h[pos:pos + bl] = 1.0
        pos += bl
    h += rng.normal(0, 0.05, n)
    return _project(h, n / 2.0)


def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
