import numpy as np

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _row_grad(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[s:s+m] += (1.0 - h[:m])
            g[:m] += -h[s:s+m]
    else:
        r = -s
        m = n - r
        if m > 0:
            g[:m] += (1.0 - h[r:r+m])
            g[r:r+m] += -h[:m]
    return g

def _line_search(h, d, S0, dx):
    n = len(h)
    lo = -0.5
    hi = 0.5
    pos = d > 1e-10
    neg = d < -1e-10
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-13:
        return 0.0, c5_exact(h)
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx

        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(S0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(S0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(S0 + hi * T - hi * hi * C)) * dx

        opt, val = lo, f_lo
    if f_hi < val:

def _lp_direction(h, S0, top_m, l1_cap, dx):
    n = len(h)
    scores = S0 * dx
    max_score = float(np.max(scores))
    if max_score <= 0:
        return None
    top_idx = np.argsort(-scores)[:top_m]
    G = np.array([_row_grad(h, int(k)) for k in top_idx])
    w = np.exp((scores[top_idx] - max_score) / (0.008 * max_score + 1e-15))
    w /= float(np.sum(w))
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(d_var) == 0,
        d_var >= -h + 1e-8,
        d_var <= 1.0 - h - 1e-8,
        cp.norm(d_var, 1) <= l1_cap,
    ]
    for k in range(top_m):
        constraints.append(w[k] * G[k] @ d_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var - 0.01 * l1_cap * cp.norm(d_var, 1)), constraints)
    solver = None
    try:
        prob.solve(solver=cp.ECOS, max_time=3.0)
        solver = 'ECOS'
    except Exception:
        pass
    if solver is None:
        try:
            prob.solve(solver=cp.CVXOPT, max_time=3.0)
        except Exception:
            try:
                prob.solve(solver=cp.PDLP, max_time=3.0)
            except Exception:
                return None
    if d_var.value is None:
        return None
    d = np.array(d_var.value).flatten()
    d -= np.mean(d)
    dnorm = float(np.linalg.norm(d, 1))
    if dnorm < 1e-10:
        return None
    return d

def _golden_pair(h, i, j, dx, S0):

    lo = -min(h[j], 1.0 - h[i])

        else:

    opt = (a + b) / 2.0

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

        tried_n.add(n_try)
        h_r = _resample(best_h, n_try)
        if c5_exact(h_r) < best_c5 + 0.002:
            sub_dl = min(d3, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_descent(h_r, sub_dl, top_m=12, l1_cap_start=1.5)
            if c5_r >= best_c5:
                sub_dl2 = min(d3, time.time() + budget_s * 0.015)
                h_r, c5_r = _directed_pair_search(h_r, 12, sub_dl2)
            if c5_r < best_c5:
                sub_dl3 = min(d3, time.time() + budget_s * 0.01)
                h_r, c5_r = _gradient_polish(h_r, sub_dl3)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = n_try

    # Phase 4: Final LP + pair polish (10%)
    d4 = t0 + budget_s * 0.90
    h4, c5_4 = _lp_descent(best_h, d4, top_m=20, l1_cap_start=1.0)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Final directed pair (5%)
    d5 = t0 + budget_s * 0.95
    h5, c5_5 = _directed_pair_search(best_h, 20, d5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Final gradient polish with remaining time
    h_f, c5_f = _gradient_polish(best_h, deadline)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
