def _project(v, target):
    # Bisection to find theta such that sum(clip(v-theta, 0, 1)) = target
    n = len(v)
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        mask = (x < 1.0) if res > 0 else (x > 0.0)
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    if s >= 0:
        m = n - s
        # h[j] * (1-h[j+s]) for j=0..m-1

        grad[:m] += (-h[g:g+m])
