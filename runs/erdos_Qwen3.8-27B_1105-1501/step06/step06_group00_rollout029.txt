import numpy as np
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _compute_g(h, S, gamma=20.0):
    n = len(h)
    maxS = float(np.max(S))
    shifted = gamma * (S - maxS)
    z = np.exp(shifted)
    sum_z = float(np.sum(z))
    if sum_z < 1e-300:
        z = np.zeros(len(S))
        z[np.argmax(S)] = 1.0
        sum_z = 1.0
    w = z / sum_z
    g = np.zeros(n)
    for k in range(n):
        d1_lo = max(-(n - 1), -k)
        d1_hi = min(n - 1, n - 1 - k)
        if d1_lo <= d1_hi:
            D1 = np.arange(d1_lo, d1_hi + 1)
            g[k] += np.dot(w[D1 + n - 1], 1.0 - h[k + D1])
        d2_lo = max(-(n - 1), k - (n - 1))
        d2_hi = min(n - 1, k)
        if d2_lo <= d2_hi:
            D2 = np.arange(d2_lo, d2_hi + 1)
            g[k] -= np.dot(w[D2 + n - 1], h[k - D2])
    return g

def _pair_search(h, i, j, S, alpha_max=0.5):
    n = len(h)
    a_min = max(-h[i], h[j] - 1.0, -alpha_max)
    a_max = min(1.0 - h[i], h[j], alpha_max)
    if a_max <= a_min + 1e-14:
        return 0.0, float(np.max(S))
    T = np.zeros(2 * n - 1)
    diff = j - i
    for D in range(-(n - 1), n):
        idx = D + (n - 1)
        k_lo = max(0, -D)
        k_hi = min(n - 1, n - 1 - D)
        t = 0.0
        if k_lo <= i <= k_hi:
            t += (1.0 - h[i + D])
        if k_lo <= j <= k_hi:
            t -= (1.0 - h[j + D])
        k1 = i - D
        if k_lo <= k1 <= k_hi:
            t -= h[k1]
        k2 = j - D
        if k_lo <= k2 <= k_hi:
            t += h[k2]
        T[idx] = t
    C_d0 = 2.0
    Cd = 0
    Ci = 0
    if diff != 0:

    Ci = (n - 1) - diff
    best_a = 0.0
    best_v = float(np.max(S))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = a_min, a_max
    if hi - lo < 1e-12:

    f1 = f(c1)

            c1 = c2

    opt_a = (lo + hi) / 2.0

    best_a, best_v = min(candidates, key=lambda x: x[1])
    return best_a, best_v

def _optimize_pairs(h, deadline, gamma=20.0, K=30, alpha_max=0.4, stall_limit=40, min_improve=1e-12):

    h = h.copy()
    S = _exact_c5_internal(h)
    cur_max = float(np.max(S))
    best_h = h.copy()
    best_val = cur_max
    stall = 0
    radius = alpha_max
    while time.time() < deadline:
        if stall > stall_limit:
            break
        g = _compute_g(h, S, gamma)
        scores = np.zeros((n, n))
        for i in range(n):
            scores[i, :] = g - g[i]
        idx_sorted = np.argsort(-scores, axis=None)
        total_checks = 0
        improved = False
        for flat_idx in idx_sorted:
            total_checks += 1
            if total_checks > K:
                break
            ii = int(flat_idx // n)
            jj = int(flat_idx % n)
            if ii == jj:
                continue
            if g[jj] - g[ii] < -1e-10:
                continue
            a, v = _pair_search(h, ii, jj, S, radius)
            if v < cur_max - min_improve:
                h[ii] += a
                h[jj] -= a
                h = np.clip(h, 0.0, 1.0)
                S = _exact_c5_internal(h)
                cur_max = float(np.max(S))
                if cur_max < best_val:
                    best_val = cur_max
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        if not improved:
            stall += 1
            radius *= 0.85
            if radius < 0.02:
                radius = 0.15
            gamma_eff = gamma * 1.1
            if gamma_eff > 80:
                gamma_eff = 80
        else:
            radius = min(radius * 1.02, 0.6)
    return best_h, best_val
