def project(v, target):

        best_c5 = c5_1
        best_h = h1.copy()
        best_n = len(h1)

    # Phase 2: SLSQP quick polish (10% time)
    d2 = t0 + budget_s * 0.55
    h2, c5_2 = slsqp_quick(best_h, d2, max_outer=10)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: More soft descent with high gamma (10% time)
    d3 = t0 + budget_s * 0.65
    h3, c5_3 = soft_descent(best_h, d3, max_iters=20000)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
        best_n = len(h3)

    # Phase 4: Pair transport polish (15% time)
    d4 = t0 + budget_s * 0.80
    h4, c5_4 = pair_polish(best_h, d4, n_pairs=80000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
        best_n = len(h4)

    # Phase 5: Multi-resolution (8% time)
    d5 = t0 + budget_s * 0.88
    offsets = [-4, 4, -8, 8, -6, 6, -10, 10, 12, -12, 16, -16, 20, -20, 24, -24]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 50 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = resample(best_h, n_try)
        c5_r = exact_c5(h_r)
        sub_dl = min(d5, time.time() + budget_s * 0.015)
        h_r, c5_r = soft_descent(h_r, sub_dl, max_iters=2000)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Multi-start diversity (5% time)
    d6 = t0 + budget_s * 0.93
    starts = []
    starts.append(threshold_parent(best_h))
    starts.append(binary_init(best_n, rng))
    x_grid = np.arange(best_n)
    for freq in [2, 3, 5]:
        pat = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x_grid / best_n)
        starts.append(project(pat, best_n / 2.0))
    for s in starts:
        if time.time() > d6:
            break
        sub_dl = min(d6, time.time() + budget_s * 0.01)
        h_s, c5_s = soft_descent(s, sub_dl, max_iters=3000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 7: Final SLSQP + soft polish (7% time)
    d7 = t0 + budget_s * 0.97
    h7, c5_7 = slsqp_quick(best_h, d7, max_outer=10)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
        best_n = len(h7)
    if time.time() < deadline:
        h7b, c5_7b = soft_descent(best_h, deadline, max_iters=5000)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()
            best_n = len(h7b)

    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
