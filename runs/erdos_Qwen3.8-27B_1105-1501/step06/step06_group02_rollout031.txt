            if m <= 0:

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    G = compute_gradients(h)
    
    # Trust region bounds
    low = np.maximum(0.0, h - radius)
    high = np.minimum(1.0, h + radius)
    
    # LP: minimize t
    # subject to: G[i,:]·v - t <= R0[i] - G[i,:]·h  for all i
    #             sum(v) = n/2
    #             low[j] <= v[j] <= high[j]
    
    L = 2 * n - 1
    # c = [1, 0, 0, ..., 0] (minimize t)
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    # A_ub: G rows with -1 in t column
    # A_ub[i, 0] = -1 (for t)
    # A_ub[i, 1:] = G[i, :]
    # b_ub[i] = R0[i] - G[i, :] @ h
    A_ub = np.zeros((L, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = G
    b_ub = R0 - G @ h
    
    # Equality: sum(v) = target, t is free
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target])
    
    # Bounds: t is free, v in [low, high]
    bounds = [(None, None)] + list(zip(low, high))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, False
    
    v = res.x[1:]
    # Check feasibility
    if np.any(v < -1e-9) or np.any(v > 1.0 + 1e-9):
        return None, False
    v = np.clip(v, 0.0, 1.0)
    
    # Direction
    d = v - h
    # Ensure zero sum (should be approximately zero from equality constraint)
    d -= np.mean(d)
    
    # Check if direction is meaningful
    if np.abs(d).max() < 1e-12:
        return None, False
    
    # Compute line search bounds
    pos = d > 1e-10
    neg = d < -1e-10
    hi = 1.0
    lo = -1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    
    if hi - lo < 1e-12:
        return None, False
    
    # Ensure we include alpha=0
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    
    # Compute T and C
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Line search
    alpha, val = line_search(R0, T, C, lo, hi, dx)
    
    if alpha <= 1e-12:
        return None, False
    
    new_h = h + alpha * d
    new_h = project(new_h, target)
    new_c5 = c5_exact(new_h)
    
    return new_h, new_c5

    if rng is None:
        rng = np.random.default_rng(12345)
    cur_c5 = float(np.max(S0)) * dx
    scores = S0 * dx
    max_score = float(np.max(scores))
    threshold = max_score - 0.006 * max_score
    active_idx = np.where(scores >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.argsort(-scores)[:8]
    G = compute_gradients(h)
    grads = G[active_idx]
    w = np.exp((scores[active_idx] - max_score) / (0.01 * max_score + 1e-15))
    w /= float(np.sum(w))
    wgrad = grads.T @ w
    donor_pool = np.argsort(-wgrad)[:n_pairs]
    recv_pool = np.argsort(wgrad)[:n_pairs]
    best_alpha = 0.0
    best_new_c5 = cur_c5
    best_pair = None
    tried = set()
    for i in donor_pool:
        for j in recv_pool:
            if i == j:
                continue
            key = (min(i, j), max(i, j))
            if key in tried:
                continue
            tried.add(key)
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
            T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
            C = np.correlate(d, d, 'full')
            lo_a = max(min(h[i], 1.0 - h[j]), -0.5)
            hi_a = min(max(1.0 - h[j], -h[i]), 0.5)
            hi_a = min(hi_a, max(h[i], 0.0))
            lo_a = max(lo_a, -min(h[j], 1.0 - h[i]))
            if hi_a - lo_a < 1e-13:
                continue
            alpha, val = line_search(S0, T, C, lo_a, hi_a, dx)
            if alpha > 1e-12 and val < best_new_c5 - 1e-14:
                best_new_c5 = val
                best_alpha = alpha
                best_pair = (i, j)
    if best_pair is not None and best_alpha > 1e-12:
        i, j = best_pair
        h[i] -= best_alpha
        h[j] += best_alpha
        h = project(h, target)
        new_c5 = c5_exact(h)
        if new_c5 < cur_c5 - 1e-13:
            S0_new = np.correlate(h, 1.0 - h, mode='full')
            return h, new_c5, S0_new
    return None, None, None

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def optimize_main(h, deadline, use_lp=True):
    n = len(h)
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    S0 = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    radii = [0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.0015, 0.0007, 0.0003]
    radii_idx = 0
    rng = np.random.default_rng(42)
    while time.time() < deadline:
        improved = False
        if use_lp and radii_idx < len(radii):
            new_h, new_c5 = lp_trust_step(h, radii[radii_idx])
            if new_h is not None and new_c5 is not None and new_c5 < cur_c5 - 1e-12:
                h = new_h
                cur_c5 = new_c5
                S0 = np.correlate(h, 1.0 - h, mode='full')
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
        if not improved:
            new_h, new_c5 = gradient_direction_step(h)
            if new_h is not None and new_c5 is not None and new_c5 < cur_c5 - 1e-12:
                h = new_h
                cur_c5 = new_c5
                S0 = np.correlate(h, 1.0 - h, mode='full')
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
        if not improved:
            rhood = max(5, min(20, n // 6))
            res = pair_search_step(h, S0, rhood, rng)
            if res[0] is not None:
                h, cur_c5, S0 = res
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
                improved = True
        if not improved:
            if use_lp and radii_idx < len(radii):
                radii_idx += 1
            stall += 1
            if stall > 80:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except NameError:
        n0 = 122
        x = np.arange(n0)
        h0 = 0.5 + 0.38 * np.sin(2.0 * np.pi * 2.0 * x / n0 + 0.3) + 0.12 * np.sin(2.0 * np.pi * 5.0 * x / n0)
        h0 = project(h0, n0 / 2.0)
    h = project(h0, n0 / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n0
    # Phase 1: Main LP optimization (45% time)
    d1 = t0 + budget_s * 0.40
    h1, c51 = optimize_main(h, d1, use_lp=True)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()
    # Phase 2: Multi-resolution (25% time)
    d2 = t0 + budget_s * 0.65
    offsets = [4, -4, 6, -6, 8, -8, 2, -2, 10, -10, 12, -12]
    tried_n = {best_n}
    for dn in offsets:
        if time.time() > d2:
            break
        n_try = best_n + dn
        if n_try < 24 or n_try > 300 or n_try % 2 != 0:
            continue
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        h_r = resample(best_h, n_try)
        sub_dl = min(d2, time.time() + budget_s * 0.025)
        h_r, c5_r = optimize_main(h_r, sub_dl, use_lp=True)
        if c5_r < best_c5 - 1e-10:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try
    # Phase 3: Second LP pass with different radii (15% time)
    d3 = t0 + budget_s * 0.80
    if time.time() < d3:
        h3, c53 = optimize_main(best_h, d3, use_lp=True)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()
    # Phase 4: Final pair search polish (10% time)
    d4 = deadline
    if time.time() < d4:
        n_h = len(best_h)
        cur_c5 = c5_exact(best_h)
        S0 = np.correlate(best_h, 1.0 - h1 if False else best_h, mode='full') if False else np.correlate(best_h, 1.0 - best_h, mode='full')
        rng = np.random.default_rng(seed + 100)
        stall = 0
        while time.time() < d4 and stall < 200:
            n_pairs = max(8, min(n_h // 5, 25))
            res = pair_search_step(best_h, S0, n_pairs, rng)
            if res[0] is not None:
                best_h, cur_c5, S0 = res
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    new_h, new_c5 = gradient_direction_step(best_h)
                    if new_h is not None and new_c5 is not None and new_c5 < cur_c5 - 1e-12:
                        best_h = new_h
                        cur_c5 = new_c5
                        S0 = np.correlate(best_h, 1.0 - best_h, mode='full')
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                        stall = 0
    # Phase 5: Final gradient polish
    if time.time() < deadline:
        h5, c55 = optimize_main(best_h, deadline, use_lp=False)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5.copy()
    # Final validation
    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
