    A_eq = np.zeros((1, n + 1))

    return None

def _lp_loop(h, deadline, radius_init=0.12, K=15, use_smooth=False):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    radius = radius_init
    stall = 0
    restart_ctr = 0
    rng = np.random.default_rng(777)
    tau = 0.02 * cur_c5
    while time.time() < deadline:
        if stall > 60:
            if restart_ctr < 12:
                scale = 0.008 + 0.008 * restart_ctr * 0.3
                h = h + rng.normal(0, scale, n)
                h = _project(h, target)
                cur_c5 = _exact_c5(h)
                radius = 0.08
                stall = 0
                restart_ctr += 1
                tau = 0.02 * cur_c5
                continue
            else:
                break
        S = _corr_raw(h) * dx
        p = _lp_direction(h, S, n, K, radius)
        if p is None or float(np.max(np.abs(p))) < 1e-13:
            if use_smooth:
                M = float(S.max())
                shifted = (S - M) / tau
                shifted -= shifted.max()
                w = np.exp(shifted)
                w_sum = w.sum()
                if w_sum > 1e-300:
                    w = w / w_sum
                    top_k = min(20, 2 * n - 1)
                    top_idx = np.argsort(-S)[:top_k]
                    G = _grad_rows(h, top_idx, n)
                    grad_s = w[top_idx] @ G
                    grad_s -= grad_s.mean()
                    norm = np.linalg.norm(grad_s)
                    if norm > 1e-12:
                        p = -grad_s / norm * radius
            if p is None or float(np.max(np.abs(p))) < 1e-13:
                stall += 1
                radius *= 0.72
                if use_smooth and stall % 5 == 0:
                    tau *= 0.8
                if radius < 1e-6:
                    break
                continue
        alpha, new_c5 = _line_search(h, p, dx, radius)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = abs(alpha) / max(radius, 1e-12)
            if ratio > 0.7:
                radius = min(radius * 1.06, 0.28)
            elif ratio > 0.3:
                radius = min(radius * 1.02, 0.25)
            else:
                radius = max(radius * 0.95, 0.02)
        else:
            stall += 1
            radius *= 0.75
            if stall % 12 == 0 and radius < 0.015:
                radius = 0.05
    return best_h, best_c5

def _binary_sa_fast(h_init, deadline, rng):
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    C = _corr_raw(h)
    cur_c5 = float(np.max(C)) * dx
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 5e-7
    cooldown = 0.99997
    patience = 0
    max_patience = 30000
    shift = np.arange(-(n - 1), n)
    while time.time() < deadline:
        if patience > max_patience:
            if T < T_min * 3:
                break
            T = max(T_min, T * 0.8)
            patience = 0
        if T < T_min:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(300):
            if time.time() > deadline:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            valid_a_i = (i + shift >= 0) & (i + shift < n)
            valid_b_i = (i - shift >= 0) & (i - shift < n)
            valid_a_j = (j + shift >= 0) & (j + shift < n)
            valid_b_j = (j - shift >= 0) & (j - shift < n)
            delta = np.zeros(2 * n - 1)
            bi = i + shift[valid_a_i]
            delta[valid_a_i] -= (1.0 - h[bi])
            ai = i - shift[valid_b_i]
            delta[valid_b_i] += h[ai]
            bj = j + shift[valid_a_j]
            delta[valid_a_j] += (1.0 - h[bj])
            aj = j - shift[valid_b_j]
            delta[valid_b_j] -= h[aj]
            new_C = C + delta
            new_max = float(np.max(new_C)) * dx
            delta_c5 = new_max - cur_c5
            if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
                h[i] = 0.0
                h[j] = 1.0
                C = new_C
                cur_c5 = new_max
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
        T = max(T_min, T * cooldown)
    return best_h, best_c5

def _fista_lse(h_init, deadline, beta_start=20.0, beta_end=500.0, momentum_decay=0.0):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    beta = beta_start
    alpha = 1.0 / (float(n) * dx)
    t = 1.0
    stale = 0
    while time.time() < deadline:
        corr = _corr_raw(h) * dx
        corr_max = float(np.max(corr))
        exp_arg = beta * (corr - corr_max)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum
        grad = _grad_weighted(w, h, n)
        grad -= grad.mean()
        x_new = _project(y - alpha * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - h)
        h = x_new
        t = t_new
        cur_c5 = _exact_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 200:
            alpha *= 0.7
            stale = 0
            y = h.copy()
            t = 1.0
        if stale > 2000:
            beta = min(beta * 1.3, beta_end)
            alpha = min(alpha, 2.0 / (beta * float(n) * dx * dx + 1e-10))
            stale = 0
            y = h.copy()
            t = 1.0
    return best_h, best_c5

def _pair_refine(h, deadline, max_passes=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    stall = 0
    while time.time() < deadline and stall < 60:
        S = _corr_raw(h) * dx
        G = _grad_rows(h, np.arange(2 * n - 1), n)
        total_grad = S @ G
        total_grad -= total_grad.mean()
        n_int = min(n, 40)
        candidates = np.argsort(-np.abs(total_grad))[:n_int]
        improved = False
        for ii in range(len(candidates)):
            if time.time() > deadline:
                break
            i = candidates[ii]
            for jj in range(ii + 1, len(candidates)):
                j = candidates[jj]
                if total_grad[i] * total_grad[j] >= 0:
                    continue
                d = np.zeros(n)
                if total_grad[i] < 0:
                    d[i] = 1.0
                    d[j] = -1.0
                else:
                    d[j] = 1.0
                    d[i] = -1.0
                alpha, c5_try = _line_search(h, d, dx, 0.4)
                if c5_try < best_c5 - 1e-14:
                    h = h + alpha * d
                    h = _project(h, target)
                    cur = _exact_c5(h)
                    if cur < best_c5:
                        best_c5 = cur
                        best_h = h.copy()
                        improved = True
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.99

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except NameError:
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    best_h = _project(h0, n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: Quick LP polish (8%)
    d1 = t0 + budget_s * 0.08
    h1, c5_1 = _lp_loop(best_h, d1, radius_init=0.12, K=18, use_smooth=True)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Fast binary SA (25%)
    d2 = t0 + budget_s * 0.33
    h2, c5_2 = _binary_sa_fast(best_h, d2, rng)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2
    if time.time() < d2:
        h2b, c5_2b = _lp_loop(best_h, min(d2, time.time() + budget_s * 0.04),
                              radius_init=0.08, K=18, use_smooth=True)
        if c5_2b < best_c5:
            best_h, best_c5 = h2b, c5_2b

    # Phase 3: FISTA log-sum-exp (18%)
    d3 = t0 + budget_s * 0.51
    h3, c5_3 = _fista_lse(best_h, d3, beta_start=25.0, beta_end=400.0)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3
    if time.time() < d3:
        h3b, c5_3b = _lp_loop(best_h, min(d3, time.time() + budget_s * 0.03),
                              radius_init=0.06, K=18)
        if c5_3b < best_c5:
            best_h, best_c5 = h3b, c5_3b

    # Phase 4: n-sweeping (22%)
    d4 = t0 + budget_s * 0.73
    base_n = len(best_h)
    tried = set()
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12,
               -14, 14, -16, 16, -18, 18, -20, 20, -24, 24, -28, 28]
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = base_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d4, time.time() + budget_s * 0.018)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=15, use_smooth=True)
            if c5_r < best_c5 and time.time() < d4 - budget_s * 0.03:
                sub_dl2 = min(d4, time.time() + budget_s * 0.015)
                h_r2, c5_r2 = _binary_sa_fast(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d4 - budget_s * 0.01:
                    h_r, c5_r = _lp_loop(h_r, min(d4, time.time() + budget_s * 0.008),
                                         radius_init=0.05, K=15)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: Pair refinement + perturbation (15%)
    d5 = t0 + budget_s * 0.88
    h5, c5_5 = _pair_refine(best_h, min(d5, time.time() + budget_s * 0.06))
    if c5_5 < best_c5:
        best_h, best_c5 = h5, c5_5

    if time.time() < d5:
        scale = 0.02
        for attempt in range(200):
            if time.time() > d5:
                break
            if best_c5 < 0.3805:
                break
            h_pert = best_h + rng.normal(0, scale, len(best_h))
            h_pert = _project(h_pert, len(best_h) / 2.0)
            sub_dl = min(d5, time.time() + budget_s * 0.006)
            h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=15, use_smooth=True)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_t.copy()
                scale = max(0.002, scale * 0.75)
            else:
                scale = min(0.10, scale * 1.3)

    # Phase 6: Final polish (12%)
    d6 = deadline
    h6, c5_6 = _pair_refine(best_h, min(d6, time.time() + budget_s * 0.04))
    if c5_6 < best_c5:
        best_h, best_c5 = h6, c5_6
    if time.time() < deadline:
        h6b, c5_6b = _lp_loop(best_h, deadline, radius_init=0.03, K=20, use_smooth=True)
        if c5_6b < best_c5:
            best_h, best_c5 = h6b, c5_6b

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
