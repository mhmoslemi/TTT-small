def _project(v, target):

        return np.zeros(n)

        else:
            hi = mid

    x = np.clip(v - theta, 0.0, 1.0)

        else:
            mask = x > 0.0

            x[mask] += res / cnt

    # Phase 1: Quick LP-free optimization on parent (10%)
    d1 = t0 + budget_s * 0.10
    h1, c5_1 = _opt_continuous(h_init, d1, K=35, radius_init=0.12, stall_limit=50, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Main intensive optimization (40%)
    d2 = t0 + budget_s * 0.50
    n_best = len(best_h)
    h2 = _project(best_h.copy(), n_best / 2.0)
    h2, c5_2 = _opt_continuous(h2, d2, K=50, radius_init=0.18, stall_limit=200, seed=2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 2b: Binary optimization (10%)
    d2b = t0 + budget_s * 0.60
    h_bin = _to_binary(best_h)
    h_bin, c5_bin = _binary_sa(h_bin, min(d2b, time.time() + budget_s * 0.04), T_init=0.002, max_iter=20000)
    h_bin, c5_bin = _binary_pair_sweep(h_bin, d2b)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Polish binary result
    h_pol = _project(h_bin.copy(), len(h_bin) / 2.0)
    h_pol, c5_pol = _opt_continuous(h_pol, d2b, K=30, radius_init=0.08, stall_limit=40, seed=3)
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol.copy()
    
    # Phase 3: Multi-resolution exploration (20%)
    d3 = t0 + budget_s * 0.80
    n_curr = len(best_h)
    offsets = [-8, -6, -4, -2, 2, 4, 6, 8, -12, 12, -16, 16]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_curr + dn
        if n_try < 60 or n_try > 280 or n_try % 2 != 0 or n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d3, time.time() + budget_s * 0.012)
        h_r, c5_r = _opt_continuous(h_r, sub_dl, K=30, radius_init=0.12, stall_limit=40, seed=10 + dn)
        if c5_r < best_c5 + 0.005 and time.time() < d3 - 2:
            h_b = _to_binary(h_r)
            sub_dl2 = min(d3, time.time() + budget_s * 0.006)
            h_b, c5_b = _binary_pair_sweep(h_b, sub_dl2, max_passes=3)
            if c5_b < c5_r:
                c5_r = c5_b
                h_r = h_b
            h_pol2 = _project(h_r.copy(), n_try / 2.0)
            sub_dl3 = min(d3, time.time() + budget_s * 0.006)
            h_pol2, c5_p2 = _opt_continuous(h_pol2, sub_dl3, K=25, radius_init=0.06, stall_limit=30, seed=20 + dn)
            if c5_p2 < c5_r:
                c5_r = c5_p2
                h_r = h_pol2
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            n_curr = n_try
    
    # Phase 4: Basin hopping (5%)
    d4 = t0 + budget_s * 0.85
    scale = 0.02
    n_restarts = 0
    while time.time() < d4 and n_restarts < 80:
        n_restarts += 1
        h_p = _perturb(best_h, scale, rng)
        sub_dl = min(d4, time.time() + budget_s * 0.004)
        h_t, c5_t = _opt_continuous(h_p, sub_dl, K=25, radius_init=0.08, stall_limit=20, seed=100 + n_restarts)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.1)
    
    # Phase 5: Final intensive polish (15%)
    d5 = t0 + budget_s * 0.99
    n_final = len(best_h)
    h5 = _project(best_h.copy(), n_final / 2.0)
    h5, c5_5 = _opt_continuous(h5, d5, K=60, radius_init=0.10, stall_limit=150, seed=999)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Try one more binary pass if time remains
    if deadline - time.time() > 10:
        h_b = _to_binary(best_h)
        h_b, c5_b = _binary_pair_sweep(h_b, deadline, max_passes=4)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_pol = _project(best_h.copy(), n_final / 2.0)
            h_pol, c5_p = _opt_continuous(h_pol, deadline, K=40, radius_init=0.05, stall_limit=30, seed=4242)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pol.copy()
    
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, final_c5, len(best_h)
