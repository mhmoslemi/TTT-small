    # Returns array of shape (2n-1, n)
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Shift to avoid overflow
    max_c = np.max(corr)
    return (1.0/lam) * (lam * max_c + np.log(np.sum(np.exp(lam * (corr - max_c)))))

    weights /= np.sum(weights)
    G = _corr_grad(h)

        return np.zeros(n)

        return np.ones(n)

    for _ in range(70):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.maximum(lam * (corr - max_c), -700.0)
    return max_c + np.log(np.sum(np.exp(shifted))) / lam

def _ls_exp_grad(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.maximum(lam * (corr - max_c), -700.0)
    e = np.exp(shifted)
    w = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(-(n - 1), n):
        idx = k + n - 1
        wk = w[idx] * dx

            j = -k
            m = n - j

        return _ls_exp_obj(x, lam)

        return _ls_exp_grad(x, lam)

        return np.sum(x) - target

        return np.ones(n)

            break
        n_modes = rng.integers(2, 6)
        coefs = rng.normal(0, 0.04, 2 * n_modes)
        pert = np.zeros(n)
        for m in range(n_modes):
            pert += coefs[2 * m] * np.sin(np.pi * (m + 1) * x)
            pert += coefs[2 * m + 1] * np.cos(np.pi * (m + 1) * x)
        h_try = h + pert
        h_try = _project(h_try, target)
        sub_dl = min(deadline, time.time() + t_per)
        h_try = _quick_optimize(h_try, sub_dl, smooth_iters=1, active_iters=200)
        c5_try = _exact_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def _structured_init(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = max(1, int(rng.integers(2, max(3, n // 4 + 1))))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return _project(h, n / 2.0)

def _sine_init(n, freq, phase, amp, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    h = h + rng.normal(0, 0.02, n)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        h = _structured_init(122, rng)
    n = len(h)
    h = _project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    best_n = n

    # Phase 1: Smooth log-sum-exp optimization
    d1 = t0 + budget_s * 0.18
    h1 = _smooth_phase(h, d1)
    c5_1 = _exact_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n

    # Phase 2: Active set minimax (main optimization)
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = _active_set_phase(best_h, d2, max_iter=3000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
        best_n = len(h2)

    # Phase 3: Smooth + active on the improved solution
    d3 = t0 + budget_s * 0.58
    if time.time() < d3:
        h3 = _smooth_phase(best_h, min(d3, time.time() + 20.0))
        h3, c5_3 = _active_set_phase(h3, d3, max_iter=1000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = len(h3)

    # Phase 4: Multi-resolution search
    d4 = t0 + budget_s * 0.82
    best_h, best_c5, best_n = _multi_res_phase(best_h, best_c5, best_n, d4, rng)

    # Phase 5: Fourier escape
    d5 = t0 + budget_s * 0.92
    if time.time() < d5:
        h5, c5_5 = _fourier_escape(best_h, best_n, rng, d5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            best_n = len(h5)

    # Phase 6: Final polish
    d6 = deadline
    if time.time() < d6 - 5:
        h6, c5_6 = _active_set_phase(best_h, min(d6, time.time() + 40.0), max_iter=2000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
            best_n = len(h6)
    # Final SLSQP on the best
    if time.time() < d6 - 2:
        h6b = _slsqp_smooth(best_h, 200.0, max_iter=50)
        h6b = _project(h6b, len(h6b) / 2.0)
        c5_6b = _exact_c5(h6b)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
            best_n = len(h6b)

    # Ensure validity
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
