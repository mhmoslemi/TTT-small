def _active_descent(h, deadline, K=20, max_iters=150):

        top_idx = np.argsort(-corr)[:K]

        for r in top_idx:
            k_val = int(r) - (n - 1)
            constraints.append(_make_constraint(k_val, n, dx))

        bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.1)]
        try:
            res = minimize(_obj_fn, x0, jac=_obj_jac, method='SLSQP',
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-12})
            if time.time() > deadline:
                break
            if res.x is not None and res.success:
                h_new = _project(res.x[:n], target)
                c5_new = _exact_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    lam_r *= 3.0
                else:
                    break
        except Exception:
            break
    return best_h, best_c5


def _make_constraint(k, n, dx):
    def _cfun(x):
        hv = x[:n]
        if k >= 0:
            m = n - k
            val = float(np.dot(hv[k:], 1.0 - hv[:m]))
        else:
            q = -k
            m = n - q
            val = float(np.dot(hv[:m], 1.0 - hv[q:]))
        return x[n] - val * dx

    def _cjac(x):
        hv = x[:n]
        g = np.zeros(n + 1)
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:n] = 1.0 - hv[0:m]
                g[0:m] -= hv[k:n]
        else:
            q = -k
            m = n - q
            if m > 0:
                g[0:m] = 1.0 - hv[q:n]
                g[q:n] -= hv[0:m]
        g *= dx
        g[n] = 1.0
        return -g

    return {'type': 'ineq', 'fun': _cfun, 'jac': _cjac}


def _structured_perturb(h, rng, scale=0.08):
    n = len(h)
    x = np.arange(n)
    modes = [2, 3, 4, 5, 6, 7, 8]
    freq = rng.choice(modes)
    phase = rng.uniform(0, 2 * np.pi)
    pert = scale * np.sin(2 * np.pi * freq * x / n + phase)
    return h + pert


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(122, 0.5)

    n0 = len(h0)
    best_h = _project(h0.copy(), n0 / 2.0)
    best_c5 = _exact_c5(best_h)

    # Phase 1: L-BFGS in u-space with eps schedule
    d1 = t0 + budget_s * 0.30
    eps_schedule = [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]
    for eps_val in eps_schedule:
        if time.time() > d1:
            break
        remaining = d1 - time.time()
        if remaining < 5:
            break
        mid_dl = min(d1, time.time() + remaining * 0.5)
        h_new, c5_new = _lbfgs_run(best_h, eps_val, mid_dl, maxiter=250)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Multi-start with structured perturbations
    d2 = t0 + budget_s * 0.55
    n_starts = 15
    for i in range(n_starts):
        if time.time() > d2:
            break
        remaining = d2 - time.time()
        if remaining < 10:
            break
        per_start = remaining / (n_starts - i)
        h_pert = _structured_perturb(best_h, rng, scale=0.06 + 0.02 * (i % 3))
        h_pert = _project(h_pert, len(h_pert) / 2.0)
        sub_dl = min(d2, time.time() + per_start * 0.6)
        h_try, c5_try = _lbfgs_run(h_pert, 0.01, sub_dl, maxiter=100)
        if time.time() < d2 - 2:
            h_try2, c5_try2 = _active_descent(h_try, min(d2, time.time() + per_start * 0.3), K=15, max_iters=80)
            if c5_try2 < c5_try:
                h_try, c5_try = h_try2, c5_try2
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: SLSQP refinement
    d3 = t0 + budget_s * 0.70
    h_s, c5_s = _slsqp_refine(best_h, d3, n_active=30, max_calls=5)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s
    # Active descent follow-up
    if time.time() < d3:
        h_a, c5_a = _active_descent(best_h, d3, K=15, max_iters=100)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a

    # Phase 4: Multi-resolution
    d4 = t0 + budget_s * 0.85
    best_n = len(best_h)
    offsets = [-6, 6, -10, 10, -14, 14, -8, 8, -20, 20, -4, 4]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 260 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d4, time.time() + budget_s * 0.02)
        h_r, c5_r = _lbfgs_run(h_r, 0.01, sub_dl, maxiter=100)
        if time.time() < d4 - 3:
            h_r2, c5_r2 = _active_descent(h_r, min(d4, time.time() + budget_s * 0.015), K=15, max_iters=50)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r
            best_n = n_try

    # Phase 5: Binary search
    d5 = t0 + budget_s * 0.93
    if len(best_h) % 2 == 0:
        h_b, c5_b = _binary_search(best_h, d5, rng, max_iters=30000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
        # Polish binary with active descent
        if time.time() < d5:
            h_bp = _project(best_h.astype(float), len(best_h) / 2.0)
            h_bp2, c5_bp2 = _active_descent(h_bp, d5, K=15, max_iters=60)
            if c5_bp2 < best_c5:
                best_c5 = c5_bp2
                best_h = h_bp2

    # Phase 6: Final polish with L-BFGS at very small eps
    if time.time() < deadline - 5:
        h_f, c5_f = _lbfgs_run(best_h, 0.0003, deadline - 2, maxiter=200)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        h_f2, c5_f2 = _active_descent(best_h, deadline - 1, K=12, max_iters=50)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    # Final feasibility fix
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
