def line_search_exact(h, d, overlaps=None, max_iter=50):
    """Exact line search along direction d (which has sum=0).
    Returns (best_alpha, best_c5)."""
    n = len(h)
    dx = 2.0 / n
    if overlaps is None:
        overlaps = compute_overlaps(h)

    pos = d > 1e-12
    neg = d < -1e-12
    alpha_max = 1.0
    if np.any(pos):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos]) / d[pos])))
    alpha_min = 0.0
    if np.any(neg):
        alpha_min = max(alpha_min, float(np.max(h[neg] / (-d[neg]))))

        return 0.0, c5_exact(h)
    
    # Precompute D and Q for the quadratic formula
    D = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    Q = np.correlate(d, d, 'full')
    
    def f_of_alpha(a):
        return float(np.max(overlaps + a * D - a * a * Q)) * dx
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = alpha_min, alpha_max
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f_of_alpha(c1)
    f2 = f_of_alpha(c2)
    
    for _ in range(max_iter):
        if b - a < 1e-12:

        if f1 < f2:
            b, c2, f2 = c2, c1, f1

            f1 = f_of_alpha(c1)
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = f_of_alpha(c2)

    for endpoint in [alpha_min, alpha_max]:
        f_e = f_of_alpha(endpoint)
        if f_e < best_f:
            best_f = f_e
            opt = endpoint
