import numpy as np

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(np.min(v)) - 2.0, float(np.max(v)) + 2.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
    return np.clip(x, 0.0, 1.0)

def _grad_matrix_all(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for r in range(m):
        d = r - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[r, d:] += (1.0 - h[:length])
                G[r, :length] += -h[d:]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[r, :length] += (1.0 - h[g:])
                G[r, g:] += -h[:length]
    return G

def _grad_subset(h, indices):
    n = len(h)
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        d = r - (n - 1)
        if d >= 0:
            length = n - d
            if length > 0:
                G[idx, d:] += (1.0 - h[:length])
                G[idx, :length] += -h[d:]
        else:
            g = -d
            length = n - g
            if length > 0:
                G[idx, :length] += (1.0 - h[g:])
                G[idx, g:] += -h[:length]
    return G

def _lse_value(h, beta):
    C = _corr_scaled(h)
    Cmax = float(C.max())
    z = beta * (C - Cmax)
    return Cmax + np.log(np.sum(np.exp(z))) / beta

def _lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    C = _corr_scaled(h)
    Cmax = float(C.max())
    z = beta * (C - Cmax)
    p = np.exp(z)
    p /= p.sum()
    G = _grad_matrix_all(h)
    return dx * (p @ G)

def _solve_lse_slsqp(h_init, beta, maxiter=400, ftol=1e-14):
    n = len(h_init)
    target = n / 2.0
    h0 = h_init.copy()

    def obj(h):
        return _lse_value(h, beta)

    def obj_grad(h):
        return _lse_grad(h, beta)

    def eq_fun(h):
        return float(np.sum(h) - target)

    def eq_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    best_h = h_init.copy()
    best_exact = _compute_c5(h_init)

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_sol = _project(res.x, target)
            c5_sol = _compute_c5(h_sol)
            if c5_sol < best_exact:
                best_exact = c5_sol
                best_h = h_sol
    except Exception:
        pass

    # Also try the endpoint of the optimization path
    try:
        h_end = _project(res.x, target)
        c5_end = _compute_c5(h_end)
        if c5_end < best_exact:
            best_exact = c5_end
            best_h = h_end
    except Exception:
        pass

    return best_h, best_exact

def _line_search(h, d, trust_radius):
    n = len(h)
    dx = 2.0 / n
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = np.correlate(d, d, mode='full')
    t_lo, t_hi = 0.0, 0.0
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_hi = float(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = float(np.max(h[neg] / d[neg]))
    t_lo = max(t_lo, -trust_radius)
    t_hi = min(t_hi, trust_radius)
    if t_hi - t_lo < 1e-14:
        return 0.0, _compute_c5(h)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - g * (b - a)
    c2 = a + g * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(50):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - g * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + g * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C)) * dx
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C)) * dx
    return opt_t, opt_val

def _clip_direction(h, d, radius):
    d = d.copy()
    pos = d > 0
    neg = d < 0
    if np.any(pos):
        max_pos = float(np.min(1.0 - h[pos])) / float(np.max(d[pos]))
        if max_pos < 1.0:
            d[pos] *= max_pos
    if np.any(neg):
        max_neg = float(np.min(h[neg])) / float(np.max(-d[neg]))
        if max_neg < 1.0:
            d[neg] *= -max_neg
    max_abs = float(np.max(np.abs(d)))
    if max_abs > radius:
        d *= radius / max_abs
    return d

def _active_direction(h, scores, radius, frac=0.05):
    n = len(h)
    m = 2 * n - 1
    max_s = float(scores.max())
    thr = max_s - frac * max_s
    active = np.where(scores >= thr)[0]
    if len(active) < 2:
        thr = max_s - 0.15 * max_s
        active = np.where(scores >= thr)[0]
    if len(active) < 2:
        return None
    G = _grad_subset(h, active)
    avg_grad = G.mean(axis=0)
    avg_grad -= avg_grad.mean()
    norm = np.linalg.norm(avg_grad)
    if norm < 1e-14:
        return None
    d = -avg_grad / norm
    d = _clip_direction(h, d, radius)
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    return d

def _single_active_direction(h, scores, radius, k=0):
    n = len(h)
    active = np.where(scores >= float(scores.max()) - 0.02 * float(scores.max()))[0]
    if k >= len(active):
        k = len(active) - 1
    r = active[k]
    G = _grad_subset(h, np.array([r]))[0]
    G -= G.mean()
    norm = np.linalg.norm(G)
    if norm < 1e-14:
        return None
    d = -G / norm
    d = _clip_direction(h, d, radius)

    return d

def _random_direction(h, radius, rng):
    n = len(h)
    d = rng.normal(0, 1, n)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    d = d / norm * radius
    d = _clip_direction(h, d, radius)
    if float(np.max(np.abs(d))) < 1e-12:
        return None
    return d

def _lp_direction(h, scores, radius, top_k=20):
    n = len(h)
    m = 2 * n - 1
    sorted_idx = np.argsort(-scores)
    K = min(top_k, m)
    active = sorted_idx[:K]
    G = _grad_subset(h, active)
    act_scores = scores[active]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -act_scores
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[-1] < -1e-12:
            return res.x[:n]
    except Exception:
        pass
    return None

def _periodic_init(n, period, duty):

        h2c, c5_2c = _line_polish(best_h, d2, max_iters=500)
        if c5_2c < best_c5:
            best_c5 = c5_2c
            best_h = h2c.copy()

    # Phase 3: Multi-start (15%)
    d3 = t0 + budget_s * 0.55
    n = len(best_h)
    starts = []
    for freq in [1, 2, 3, 4, 5, 7]:
        for ph in [0.0, 0.7, 1.5, 2.5]:
            starts.append(_sine_init(n, freq, ph))
    for p in [5, 6, 7, 8, 10, 12]:
        for duty in [0.35, 0.45, 0.55, 0.65]:
            starts.append(_periodic_init(n, p, duty))
    # Binary starts from parent
    b_start = _binary_from_rank(best_h)
    starts.append(b_start)
    # Random balanced starts
    for _ in range(3):
        r = rng.uniform(0, 1, n)
        starts.append(_project(r, n / 2.0))
    starts.append(best_h.copy())

    for s in starts:
        if time.time() > d3:
            break
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_s, c5_s = _lse_anneal(s, sub_dl, betas=[30.0, 100.0, 300.0])
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 4: Line search intensive (15%)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _line_polish(best_h, d4, max_iters=5000)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Multi-resolution (12%)
    d5 = t0 + budget_s * 0.82
    best_n = len(best_h)
    offsets = [-2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -14, 14, -20, 20]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = best_n + dn
        if n_try < 40 or n_try > 250 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        sub_dl = min(d5, time.time() + budget_s * 0.012)
        h_r, c5_r = _lse_anneal(h_r, sub_dl, betas=[30.0, 100.0, 300.0])
        if c5_r < c5_r:
            pass
        if c5_r < best_c5 - 1e-12 and time.time() < d5 - 5:
            h_r2, c5_r2 = _line_polish(h_r, min(d5, time.time() + budget_s * 0.006), max_iters=300)
            if c5_r2 < c5_r:
                h_r, c5_r = h_r2, c5_r2
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_try

    # Phase 6: Pair + quad refinement (8%)
    d6 = t0 + budget_s * 0.90
    h6, c5_6 = _pair_refine(best_h, min(d6, t0 + budget_s * 0.87))
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < d6:
        h6b, c5_6b = _quad_refine(best_h, d6)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()

    # Phase 7: Final LSE + minimax + line polish (7%)
    d7 = t0 + budget_s * 0.97
    h7, c5_7 = _lse_anneal(best_h, min(d7, t0 + budget_s * 0.94), betas=[200.0, 500.0, 1000.0])
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()
    if time.time() < d7:
        h7b, c5_7b = _minimax_solve(best_h, d7)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7b.copy()
    if time.time() < d7:
        h7c, c5_7c = _line_polish(best_h, d7, max_iters=2000)
        if c5_7c < best_c5:
            best_c5 = c5_7c
            best_h = h7c.copy()

    # Phase 8: Aggressive perturbation (3%)
    d8 = deadline
    scale = 0.02
    for _ in range(100):
        if time.time() > d8:
            break
        h_p = best_h + rng.normal(0, scale, len(best_h))
        h_p = _project(h_p, len(best_h) / 2.0)
        h_t, c5_t = _line_polish(h_p, min(d8, time.time() + budget_s * 0.004), max_iters=200)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.10, scale * 1.3)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h)

    if not np.isfinite(final_c5) or final_c5 >= 0.5:
        try:
            h_fb = np.array(initial_h_values, dtype=float).copy()
            h_fb = _project(h_fb, len(h_fb) / 2.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)
        except NameError:
            h_fb = _sine_init(120, 3, 0.0)
            return h_fb, _compute_c5(h_fb), len(h_fb)

    return best_h, final_c5, len(best_h)
