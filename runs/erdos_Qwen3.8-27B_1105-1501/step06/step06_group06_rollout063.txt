            break
        
        def obj(x):
            v, _ = _lsq_obj_grad(x, alpha)
            return v
        
        def grad(x):
            _, g = _lsq_obj_grad(x, alpha)
            return g
        
        def eq_f(x):
            return np.sum(x) - target
        
        def eq_j(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_f, 'jac': eq_j}]
        
        x0 = h.copy()
        res = minimize(obj, x0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 500, 'ftol': 1e-16})
        
        if res.x is not None:
            h_new = _project(res.x, target)
            c5_new = _exact_c5(h_new)
            obj_new = obj(h_new)
            obj_old = obj(h)
            if obj_new < obj_old + 1e-12:
                h = h_new
                c5_exact = _exact_c5(h)
                if c5_exact < best_c5:
                    best_c5 = c5_exact
                    best_h = h.copy()
    
    return best_h, best_c5

def _subgradient_momentum(h0, deadline, max_iter=2000):
    """Subgradient method with momentum for the exact max."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    lr = 0.01
    beta = 0.9
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        cv = _corr_values(h)
        M = np.max(cv)
        # Near-active set
        tol = max(1e-6, 0.01 * (M - np.min(cv)))
        active = cv > M - tol
        
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if not active[k_idx]:
                continue
            s = k_idx - (n - 1)
            if s >= 0:
                m = n - s
                grad[:m] += (1.0 - h[s:s+m])
                grad[s:s+m] += (-h[:m])
            else:
                g = -s
                m = n - g
                grad[g:g+m] += (1.0 - h[:m])
                grad[:m] += (-h[g:g+m])
        grad *= dx
        grad /= np.sum(active)  # average over active set
        
        # Momentum step
        v = beta * v - lr * grad
        h_new = h + v
        h_new = _project(h_new, target)
        
        c5_new = _exact_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_try = _project(h - 0.1 * grad, target)
            c5_try = _exact_c5(h_try)
            if c5_try < _exact_c5(h) - 1e-14:
                h = h_try
                v *= 0.5
            else:
                lr *= 0.95
                v *= 0.5
        
        # Decay learning rate
        lr = max(lr * 0.999, 0.001)
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

            c = _corr_values(x)

            return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_corr_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += w * (1.0 - h[s:s+m])
            grad[s:s+m] -= w * h[:m]
        else:
            g = -s
            m = n - g
            grad[g:g+m] += w * (1.0 - h[:m])
            grad[:m] -= w * h[g:g+m]
    grad *= dx
    return grad

def _lsq_value_and_grad(h, alpha):
    n = len(h)
    cv = _corr_values(h)
    mc = np.max(cv)
    exps = np.exp(alpha * (cv - mc))
    sw = np.sum(exps)
    w = exps / sw
    val = mc + np.log(sw) / alpha
    grad = _weighted_corr_grad(h, w)
    return val, grad

def _optimize_lsq_slsqp(h0, deadline, alphas):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return _lsq_value_and_grad(x, alpha)[0]
        def grd(x):
            return _lsq_value_and_grad(x, alpha)[1]
        def eqf(x):
            return np.sum(x) - target
        def eqj(x):
            return np.ones(n)
        
        res = minimize(obj, h, method='SLSQP', jac=grd,
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': eqf, 'jac': eqj}],
                      options={'maxiter': 1000, 'ftol': 1e-18})
        
        if res.x is not None:
            h_new = _project(res.x, target)
            c5 = _exact_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
    
    return best_h, best_c5

    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _weighted_corr_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += w * (1.0 - h[s:s + m])
            grad[s:s + m] -= w * h[:m]
        else:
            g = -s
            m = n - g
            grad[g:g + m] += w * (1.0 - h[:m])
            grad[:m] -= w * h[g:g + m]
    grad *= dx
    return grad

def _lsq_value_and_grad(h, alpha):
    cv = _corr_values(h)
    mc = np.max(cv)
    exps = np.exp(alpha * (cv - mc))
    sw = np.sum(exps)
    w = exps / sw
    val = mc + np.log(sw) / alpha
    grad = _weighted_corr_grad(h, w)
    return val, grad

def _lsq_obj_fn(x, alpha):
    return _lsq_value_and_grad(x, alpha)[0]

def _lsq_grad_fn(x, alpha):

def _eq_j(x):

def _optimize_lsq_slsqp(h0, deadline, alphas):

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()

    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)
