def _curvature_direction(h, active, n_constraints):
    """Find a direction in the null space of active gradients that has negative curvature."""
    n = len(h)
    dx = 2.0 / n
    G = _corr_grads(h)
    A = G[active]  # K × n
    # Add sum constraint
    A_full = np.vstack([A / np.linalg.norm(A, axis=1)[:, None], np.ones(n) / np.sqrt(n)])
    
    # Find null space of A_full
    U, S, Vt = np.linalg.svd(A_full)
    null_space = Vt[A_full.shape[0]:]  # (n - K - 1) × n
    
    # For each null space vector, compute the quadratic form
    best_d = None
    best_val = np.inf
    for i in range(null_space.shape[0]):

        d = d / np.linalg.norm(d)
        quad = _hessian_diag(h, d)
        val = np.max(quad[active])  # We want this to be negative
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d, best_val

- If both (s ≤ j < n-s): both terms involve h[j], so the total contribution is h[j](1-h[j+s]) + h[j-s](1-h[j]) = h[j] - h[j]h[j+s] + h[j-s] - h[j-s]h[j]

This is NOT linear in h[j]! The terms h[j]*h[j+s] and h[j-s]*h[j] make it... wait, those are products of h[j] with FIXED values h[j+s] and h[j-s]. So:

h[j](1-h[j+s]) + h[j-s](1-h[j]) = h[j] - h[j]*h[j+s] + h[j-s] - h[j-s]*h[j]
= h[j](1 - h[j+s] - h[j-s]) + h[j-s]

This IS linear in h[j]! Because h[j+s] and h[j-s] are fixed (we're only changing h[j]).

So each correlation term is linear in any single coordinate h[j]. The max of linear functions is piecewise-linear convex, and we can find its minimum exactly by checking the breakpoints.

This gives us an EXACT coordinate descent method! At each step, pick coordinate j, and find the value of h[j] that minimizes max_k corr_k (as a function of h[j] with all other coordinates fixed).

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(60):

            lo = mid
        else:
            hi = mid

def _coord_descent_fast(h0, deadline, max_sweeps=10000):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h0.copy(), target)
    corr_vec = np.correlate(h, 1.0 - h, mode='full') * dx
    cur = float(np.max(corr_vec))
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(42)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for j in order:
            others_sum = float(np.sum(h)) - h[j]
            lo = max(0.0, target - others_sum)
            hi = min(1.0, target - others_sum)
            if hi - lo < 1e-14:
                continue
            
            # Compute G_col for coordinate j
            G_col = np.zeros(2 * n - 1)
            for s in range(-(n - 1), n):
                k = n - 1 + s
                val = 0.0
                if s >= 0:
                    if j < n - s:
                        val += (1.0 - h[j + s])
                    if s <= j:
                        val += (-h[j - s])
                else:
                    g = -s
                    if g <= j:
                        val += (1.0 - h[j - g])
                    if j < n - g:
                        val += (-h[j + g])
                G_col[k] = val * dx
            
            cur_val = h[j]
            # Minimize max_k (corr_vec[k] + G_col[k] * (t - cur_val)) over t in [lo, hi]
            a, b = lo, hi
            fa = float(np.max(corr_vec + G_col * (a - cur_val)))
            fb = float(np.max(corr_vec + G_col * (b - cur_val)))
            for _ in range(30):
                if b - a < 1e-13:
                    break
                m1 = a + (1.0 - gr) * (b - a)
                m2 = a + gr * (b - a)
                fm1 = float(np.max(corr_vec + G_col * (m1 - cur_val)))
                fm2 = float(np.max(corr_vec + G_col * (m2 - cur_val)))
                if fm1 < fm2:
                    b, fb = m2, fm2
                else:
                    a, fa = m1, fm1
            t_opt = (a + b) / 2.0
            f_opt = float(np.max(corr_vec + G_col * (t_opt - cur_val)))
            
            if f_opt < cur - 1e-13:
                delta = t_opt - cur_val
                h[j] = t_opt
                corr_vec += G_col * delta  # Update incrementally
                cur = f_opt
                improved = True
                if cur < best_c - 1e-14:
                    best_c = cur
                    best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c
