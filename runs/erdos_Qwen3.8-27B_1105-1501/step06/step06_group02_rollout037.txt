def line_search(h, direction, R0, dx, max_iter=50):
    """Exact line search along h + alpha * direction for alpha in [lo, hi]."""
    n = len(h)
    d = direction
    pos = d > 1e-10
    neg = d < -1e-10
    lo, hi = -1.0, 1.0
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, c5_exact(h)
    
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(max_iter):
        if b - a < 1e-13:

        if f1 < f2:

            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx

            c2 = a + gr * (b - a)

                              n_steps_per_eps=30):

    target = n / 2.0

    best_c5 = cur_c5

    factor = (eps_end / eps_start) ** (1.0 / max(n_steps_per_eps * 20, 1))

import numpy as np
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def line_search(h, direction, R0, dx, max_iter=50):
    n = len(h)
    d = direction
    pos = d > 1e-10
    neg = d < -1e-10
    lo, hi = -0.5, 0.5
    if np.any(pos):
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max(h[neg] / d[neg])))
    if hi - lo < 1e-14:
        return 0.0, float(np.max(R0)) * dx
    T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    C = np.correlate(d, d, 'full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
    f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
    opt = (a + b) / 2.0
    val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
    f_lo = float(np.max(R0 + lo * T - lo * lo * C)) * dx
    f_hi = float(np.max(R0 + hi * T - hi * hi * C)) * dx
    if f_lo < val:
        opt, val = lo, f_lo
    if f_hi < val:
        opt, val = hi, f_hi
    return opt, val

def weighted_gradient(h, weights, n):
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        r = abs(k - (n - 1))
        if r == 0:
            g += w * (1.0 - 2.0 * h)
        else:
            m = n - r
            if k < n - 1:
                g[:m] += w * (1.0 - h[r:r + m])
                g[r:r + m] += w * (-h[:m])
            else:
                g[r:r + m] += w * (1.0 - h[:m])
                g[:m] += w * (-h[r:r + m])
    return g

def smooth_opt(h, max_time, eps_start, eps_end, steps_per_eps, min_improve=1e-15):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, 'full')
    eps = eps_start
    n_total = max(int(steps_per_eps * 10), 10)
    factor = (eps_end / eps_start) ** (1.0 / n_total)
    stall_count = 0
    while time.time() < max_time and eps > eps_end * 0.5:
        improved = False
        for step in range(steps_per_eps):
            if time.time() > max_time:
                break
            scores = R0 * dx
            max_s = float(np.max(scores))
            denom = eps * max_s + 1e-15
            weights = np.exp((scores - max_s) / denom)
            sw = np.sum(weights)
            if sw < 1e-30:
                break
            weights /= sw
            g = weighted_gradient(h, weights, n)
            g -= np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            d = -g / gn
            alpha, val = line_search(h, d, R0, dx)
            if alpha > 1e-10 and val < cur_c5 - min_improve:
                h = h + alpha * d
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R0 = np.correlate(h, 1.0 - h, 'full')
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
            else:
                break
        if not improved:
            stall_count += 1
            if stall_count > 5:
                break
        else:
            stall_count = 0
        eps *= factor
    return best_h, best_c5

def pair_refine(h, max_time, n_pairs=15):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    cur_c5 = c5_exact(h)
    best_h = h.copy()
    best_c5 = cur_c5
    R0 = np.correlate(h, 1.0 - h, 'full')
    stall = 0
    rng = np.random.default_rng(12345)
    while time.time() < max_time and stall < 60:
        scores = R0 * dx
        max_score = float(np.max(scores))
        threshold = max_score - 0.003 * max_score
        active_idx = np.where(scores >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.argsort(-scores)[:10]
        g = weighted_gradient(h, np.ones(2*n-1)/len(active_idx), n)
        for k in active_idx:
            g += weighted_gradient(h, np.zeros(2*n-1), n)
        # Use active-weighted gradient
        w = np.zeros(2*n-1)
        for k in active_idx:
            w[k] = np.exp((scores[k] - max_score) / (0.01 * max_score + 1e-15))
        w /= np.sum(w)
        g = weighted_gradient(h, w, n)
        g -= np.mean(g)
        donor_pool = np.argsort(-g)[:n_pairs]
        recv_pool = np.argsort(g)[:n_pairs]
        best_alpha = 0.0
        best_new_c5 = cur_c5
        best_pair = None
        tried = set()
        for i in donor_pool:
            if time.time() > max_time:
                break
            for j in recv_pool:
                if i == j:
                    continue
                key = (min(i, j), max(i, j))
                if key in tried:
                    continue
                tried.add(key)
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                lo = -min(h[j], 1.0 - h[i])
                hi = min(h[i], 1.0 - h[j])
                if hi - lo < 1e-14:
                    continue
                T = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
                C = np.correlate(d, d, 'full')
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
                f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
                for _ in range(35):
                    if b - a < 1e-13:
                        break
                    if f1 < f2:
                        b, c2, f2 = c2, c1, f1
                        c1 = b - gr * (b - a)
                        f1 = float(np.max(R0 + c1 * T - c1 * c1 * C)) * dx
                    else:
                        a, c1, f1 = c1, c2, f2
                        c2 = a + gr * (b - a)
                        f2 = float(np.max(R0 + c2 * T - c2 * c2 * C)) * dx
                opt = (a + b) / 2.0
                val = float(np.max(R0 + opt * T - opt * opt * C)) * dx
                if val < best_new_c5 - 1e-14:
                    best_new_c5 = val
                    best_alpha = opt
                    best_pair = (i, j)
        if best_pair is not None and best_new_c5 < cur_c5 - 1e-14:
            i, j = best_pair
            h[i] -= best_alpha
            h[j] += best_alpha
            h = project(h, target)
            cur_c5 = c5_exact(h)
            R0 = np.correlate(h, 1.0 - h, 'full')
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15 and rng.random() < 0.3:
                idx = rng.integers(n, size=3)
                for ik in idx:
                    h[ik] += rng.uniform(-0.03, 0.03)
                h = project(h, target)
                cur_c5 = c5_exact(h)
                R0 = np.correlate(h, 1.0 - h, 'full')
                stall = 0
    return best_h, best_c5

def make_sine_init(n, freq):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / n) + 0.1 * np.sin(4.0 * np.pi * x / n)
    return project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.95

    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        h = make_sine_init(122, 2)
    n = len(h)
    h = project(h, n / 2.0)
    best_h = h.copy()
    best_c5 = c5_exact(h)
    best_n = n

    # Phase 1: Quick gradient polish at initial n (12% time)
    d1 = t0 + budget_s * 0.12
    h1, c51 = smooth_opt(h, d1, eps_start=0.04, eps_end=0.002, steps_per_eps=40)
    if c51 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c51, len(h1)

    # Phase 2: Pair refine on best (8% time)
    d2 = t0 + budget_s * 0.20
    h2, c52 = pair_refine(best_h, d2, n_pairs=12)
    if c52 < best_c5:
        best_h, best_c5 = h2.copy(), c52

    # Phase 3: Multi-resolution exploration (50% time)
    d3 = t0 + budget_s * 0.70
    n_candidates = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10]:
        nt = best_n + dn
        if nt >= 100 and nt <= 150 and nt % 2 == 0:
            n_candidates.append(nt)
    if not n_candidates:
        n_candidates = [best_n]
    time_per_n = (d3 - time.time()) / max(len(n_candidates), 1)
    for n_try in n_candidates:
        if time.time() > d3:
            break
        if n_try == best_n:
            continue
        h_r = resample(best_h, n_try)
        sub_dl = min(d3, time.time() + time_per_n * 0.7)
        h_r, c5_r = smooth_opt(h_r, sub_dl, eps_start=0.03, eps_end=0.001, steps_per_eps=35)
        sub_dl2 = min(d3, time.time() + time_per_n * 0.3)
        h_r, c5_r2 = pair_refine(h_r, sub_dl2, n_pairs=10)
        c5_r = min(c5_r, c5_r2)
        if c5_r < best_c5:
            best_h, best_c5, best_n = h_r.copy(), c5_r, n_try

    # Phase 4: Re-optimization at best_n with different eps schedule (12% time)
    d4 = t0 + budget_s * 0.82
    h4 = best_h.copy()
    h4, c54 = smooth_opt(h4, d4, eps_start=0.02, eps_end=0.0005, steps_per_eps=50)
    if c54 < best_c5:
        best_h, best_c5 = h4.copy(), c54

    # Phase 5: Perturbation + re-optimize cycles (13% time)
    d5 = deadline
    for round_i in range(8):
        if time.time() > d5:
            break
        sub_dl = min(d5, time.time() + budget_s * 0.025)
        freq = 1.0 + round_i * 0.7
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(best_n, dtype=float)
        pert = 0.015 * np.sin(2.0 * np.pi * freq * x / best_n + phase)
        pert -= np.mean(pert)
        h_p = project(best_h + pert, best_n / 2.0)
        h_p, c5_p = smooth_opt(h_p, sub_dl, eps_start=0.02, eps_end=0.001, steps_per_eps=30)
        if c5_p < best_c5:
            best_h, best_c5 = h_p.copy(), c5_p
            # Also try pair refine on the improved solution
            sub_dl2 = min(d5, time.time() + budget_s * 0.015)
            h_p2, c5_p2 = pair_refine(h_p, sub_dl2, n_pairs=10)
            if c5_p2 < best_c5:
                best_h, best_c5 = h_p2.copy(), c5_p2

    # Final check: try one more gradient sweep if time remains
    if time.time() < deadline - 5:
        h_f, c5_f = smooth_opt(best_h, deadline, eps_start=0.01, eps_end=0.0003, steps_per_eps=40)
        if c5_f < best_c5:
            best_h, best_c5 = h_f.copy(), c5_f

    best_h = project(best_h, len(best_h) / 2.0)
    final_c5 = c5_exact(best_h)
    return best_h, final_c5, len(best_h)
