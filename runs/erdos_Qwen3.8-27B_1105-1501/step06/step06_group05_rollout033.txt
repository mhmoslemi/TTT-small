import numpy as np
import time


def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    for _ in range(10):
        if abs(res) <= 1e-13:
            break
        if res > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        x[mask] += res / cnt
        x = np.clip(x, 0.0, 1.0)
        res = target - float(np.sum(x))
    return x


def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _row_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        nn = np.arange(0, n - k)
    else:
        nn = np.arange(-k, n)
    if len(nn) == 0:
        return g
    a = nn + k
    b = nn
    g[b] += -h[a]
    g[a] += (1.0 - h[b])
    return g


def _line_search(S0, T, Q, dx, L, U):
    if U - L < 1e-14:
        t = 0.5 * (L + U)
        return t, float(np.max(S0 + t * T - t * t * Q)) * dx
    nq = 100
    grid = np.linspace(L, U, nq)
    vals = S0[None, :] + grid[:, None] * T[None, :] - (grid[:, None] ** 2) * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    i = int(np.argmin(obj))
    a = grid[max(i - 1, 0)]
    b = grid[min(i + 1, nq - 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    for _ in range(50):
        if (b - a) < 1e-14:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * Q)) * dx
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * Q)) * dx
    t = 0.5 * (a + b)
    tv = float(np.max(S0 + t * T - t * t * Q)) * dx
    if obj[i] < tv:
        t, tv = grid[i], obj[i]
    return t, tv


def _direction_active(h, corr, n_active, tau, rng):

    m = float(np.max(scores))

    wsum = float(np.sum(w))
    if wsum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / wsum
    G = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx] - (n - 1)
        G += w[idx] * _row_grad(h, k)
    return -G


def _step_bounds(h, d):
    pos = d > 1e-15
    neg = d < -1e-15
    low = [0.0]
    up = [1e30]
    if np.any(pos):

    if np.any(neg):

        up.append(float(np.min(-h[neg] / d[neg])))
    L = max(low)
    U = min(up)
    return L, U


def _try_step(h, d, corr, dx, target, uphill):

    if nrm < 1e-14:
        return None

    L, U = _step_bounds(h, d)
    if U - L < 1e-12:
        return None
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    t, newf = _line_search(corr, T, Q, dx, L, U)
    if abs(t) < 1e-14:
        return None
    hn = h + t * d
    hn = _project(hn, target)
    nf = _exact_c5(hn)
    return hn, nf

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)


def _init_random(n, rng):
    h = rng.random(n)
    return _project(h, n / 2.0)


def _init_sine(n, freq, phase, rng):
