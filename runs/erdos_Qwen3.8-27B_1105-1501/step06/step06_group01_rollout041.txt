def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _lag_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            grad[:m] -= h[k:]
            grad[k:] += (1.0 - h[:m])
    else:
        d = -k
        m = n - d
        if m > 0:
            grad[d:] += (1.0 - h[:m])
            grad[:m] -= h[d:]
    return grad

def _line_search(h, d, S0, dx, trust_radius, n_gs=25):
    T = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    C = np.correlate(d, d, mode='full') * dx
    t_lo = -trust_radius
    t_hi = trust_radius
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t_lo = max(t_lo, float(np.min(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.min((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    if t_hi - t_lo < 1e-14:
        return 0.0, float(np.max(S0))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
    f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    for _ in range(n_gs):
        if b - a < 1e-12:
            break
        if f1 < f2:
            b, c2, f2 = c2, c1, f1
            c1 = b - gr * (b - a)
            f1 = float(np.max(S0 + c1 * T - c1 * c1 * C))
        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            f2 = float(np.max(S0 + c2 * T - c2 * c2 * C))
    opt_t = (a + b) / 2.0
    opt_val = float(np.max(S0 + opt_t * T - opt_t * opt_t * C))
    return opt_t, opt_val

def _lp_refine(h, deadline, start_radius=0.10):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = _exact_c5(h)
    radius = start_radius
    stall = 0

    while time.time() < deadline:
        if stall > 60:
            break
        S = _corr_vals(h)
        G = np.zeros((2 * n - 1, n))
        for q in range(2 * n - 1):
            k = q - (n - 1)
            if k >= 0:
                m = n - k
                if m > 0:
                    G[q, k:] += dx * (1.0 - h[:m])
                    G[q, :m] += -dx * h[k:]
            else:
                d_abs = -k
                m = n - d_abs
                if m > 0:
                    G[q, :m] += dx * (1.0 - h[d_abs:])
                    G[q, d_abs:] += -dx * h[:m]

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
        b_ub = -S
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for i in range(n):
            lo_i = max(-radius, -h[i] + 1e-12)
            hi_i = min(radius, 1.0 - h[i] - 1e-12)
            if lo_i >= hi_i:
                bounds.append((0.0, 0.0))
            else:
                bounds.append((lo_i, hi_i))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                p = res.x[:n]
            else:
                stall += 1
                radius *= 0.75
                if radius < 1e-5:
                    break
                continue
        except Exception:
            stall += 1
            radius *= 0.75
            if radius < 1e-5:
                break
            continue

        if float(np.max(np.abs(p))) < 1e-12:
            stall += 1
            radius *= 0.75
            if radius < 1e-5:
                break
            continue

        S0 = _corr_vals(h)
        alpha, new_c5 = _line_search(h, p, S0, dx, radius)
        if new_c5 < best_c5 - 1e-14:
            h = h + alpha * p
            h = _project(h, target)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            ratio = cur_c5 / max(new_c5, 1e-15)
            if ratio < 0.7:
                radius = min(radius * 1.2, 0.35)
            elif ratio > 1.5:
                radius *= 0.8
            else:
                radius = min(radius * 1.03, 0.30)
        else:
            stall += 1
            radius *= 0.72

    return best_h, best_c5

    # Phase 4: Diverse starts with subgradient (20% budget)
    d4 = t0 + budget_s * 0.80
    n_starts = 6
    for si in range(n_starts):
        if time.time() > d4:
            break
        sub_dl = min(d4, time.time() + budget_s * 0.025)
        h_s = _diverse_start(n0, rng)
        h_s, c5_s = _smooth_descent(h_s, sub_dl, tau_init=0.006, patience=150)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Binary local search on best (8% budget)
    d5 = t0 + budget_s * 0.88
    h5, c5_5 = _binary_ls(best_h, d5, max_passes=100)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: LP polish from binary result + subgradient (remaining up to deadline)
    d6 = deadline
    h6, c5_6 = _lp_refine(best_h, d6, start_radius=0.05)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()

    # Multi-resolution: try nearby n values
    if time.time() < deadline - 5:
        n_best = len(best_h)
        tried = set()
        for dn in [-4, 4, -6, 6, -8, 8, -12, 12]:
            if time.time() > deadline - 3:
                break
            n_try = n_best + dn
            if n_try < 40 or n_try > 200 or n_try % 2 != 0:
                continue
            if n_try in tried:
                continue
            tried.add(n_try)
            h_r = _resample(best_h, n_try)
            sub_dl = min(deadline - 2, time.time() + budget_s * 0.01)
            h_r, c5_r = _lp_refine(h_r, sub_dl, start_radius=0.08)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    return best_h, final_c5, len(best_h)
