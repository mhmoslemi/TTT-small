    return u

        if m <= 0: return a
        a[:m] -= h[k:]
        a[k:] += (1.0 - h[:m])
    else:
        q = -k
        m = n - q
        if m <= 0: return a
        a[:m] += (1.0 - h[q:])
        a[q:] -= h[:m]

    G_u = s * (G_h - Gh_dot_s)
    
    # Objective value

    target = n / 2.0

    best_h = h.copy()

        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        M = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]

        alpha = 100.0 / n
        w = np.exp(alpha * (top_vals - M))
        w /= np.sum(w)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _lag_grad(h, k, dx)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import expit, logit
import time

_global_eps = 0.05

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _solve_lambda(u, target):
    n = len(u)
    lo = float(u.min()) - 50.0
    hi = float(u.max()) + 50.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(expit(u + mid)))
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def _h_from_u(u):
    n = len(u)
    lam = _solve_lambda(u, n / 2.0)
    return expit(u + lam)

def _lag_grad_vec(h, k, dx):
    n = len(h)
    a = np.zeros(n)
    if k >= 0:
        mk = n - k
        if mk <= 0:
            return a
        a[:mk] -= h[k:]
        a[k:] += (1.0 - h[:mk])
    else:
        q = -k
        mq = n - q
        if mq <= 0:
            return a
        a[:mq] += (1.0 - h[q:])
        a[q:] -= h[:mq]
    return a * dx

def _smooth_obj_and_grad(u, eps):
    n = len(u)
    h = _h_from_u(u)
    s = h * (1.0 - h)
    S = float(np.sum(s))
    if S < 1e-15:
        return 0.0, np.zeros(n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    vm = float(np.max(v))
    exponents = (v - vm) / eps
    exponents -= np.max(exponents)
    p = np.exp(exponents)
    psum = float(np.sum(p))
    if psum < 1e-300:
        p = np.ones(len(v)) / len(v)
    else:
        p /= psum
    G_h = np.zeros(n)
    threshold = 1e-10
    for r in range(len(v)):
        if p[r] < threshold:
            continue
        k = r - (n - 1)
        pr = p[r] * dx
        if k >= 0:
            mk = n - k
            G_h[:mk] -= pr * h[k:]
            G_h[k:] += pr * (1.0 - h[:mk])
        else:
            q = -k
            mq = n - q
            G_h[:mq] += pr * (1.0 - h[q:])
            G_h[q:] -= pr * h[:mq]
    Gh_dot_s = float(np.dot(G_h, s)) / S
    G_u = s * (G_h - Gh_dot_s)
    obj = float(vm + eps * np.log(psum))
    return obj, G_u

def _lbfgs_obj(u):
    return _smooth_obj_and_grad(u, _global_eps)[0]

def _lbfgs_jac(u):
    return _smooth_obj_and_grad(u, _global_eps)[1]

def _lbfgs_phase(u, eps_list, deadline, maxiter=300):
    global _global_eps
    best_u = u.copy()
    h_curr = _h_from_u(u)
    best_exact = _exact_c5(h_curr)
    for eps in eps_list:
        if time.time() > deadline:
            break
        _global_eps = eps
        for attempt in range(2):
            if time.time() > deadline:
                break
            try:
                res = minimize(_lbfgs_obj, u, jac=_lbfgs_jac, method='L-BFGS-B',
                              options={'maxiter': maxiter, 'maxls': 50,
                                       'ftol': 1e-14, 'gtol': 1e-10,
                                       'maxfun': maxiter * 3})
                if res.x is not None:
                    h_try = _h_from_u(res.x)
                    c5_try = _exact_c5(h_try)
                    if c5_try < best_exact - 1e-10:
                        best_exact = c5_try
                        best_u = res.x.copy()
                        u = res.x.copy()
                    elif attempt == 1:
                        break
            except Exception:
                break
    return best_u, best_exact

def _active_polish(h, deadline, K=20, max_iters=200, step_init=1.0):

        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        M = float(np.max(corr))
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        alpha = 80.0
        w = np.exp(alpha * (top_vals - M))
        w /= np.sum(w)
        grad = np.zeros(n)
        dx = 2.0 / n
        for wi, idx in zip(w, top_idx):
            k = int(idx) - (n - 1)
            grad += wi * _lag_grad_vec(h, k, dx)
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        direction = -grad / gn
        best_t = 0.0
        best_f = best_c5
        t = step
        for _ in range(25):
            h_try = _project(h + t * direction, target)
            f_try = _exact_c5(h_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t
            t *= 0.7
            if t < 1e-9:
                break
        if best_t > 1e-10 and best_f < best_c5 - 1e-13:
            h = _project(h + best_t * direction, target)
            cur = _exact_c5(h)
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            if best_t > 0.8 * step:
                step = min(step * 1.1, 2.0)
            else:
                step = max(step * 0.8, 0.01)
        else:
            step *= 0.7
            if step < 0.005:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)
