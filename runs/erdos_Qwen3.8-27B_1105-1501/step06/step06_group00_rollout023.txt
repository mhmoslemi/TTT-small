The exact quadratic structure:
- Let S = corr(h, 1-h) (full mode, length 2n-1)
- For a direction d (zero-sum), the new S(α) = S + α*T - α²*Q
  where T = corr(d, 1-h) - corr(h, d) and Q = corr(d, d)
- We want to minimize max_k S(α,k) * dx

For the line search, the objective f(α) = max_k (S_k + α*T_k - α²*Q_k) is the max of quadratic functions. Its minimum can be found by:
- Checking vertices of each quadratic: α_v,k = T_k / (2*Q_k)
- Checking intersections of pairs of quadratics
- Grid search + golden section refinement

def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=float)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-12:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _feasible_interval(h, d, radius):
    """Find the interval of alpha such that h + alpha*d stays in [0,1]."""
    n = len(h)
    a_lo = -radius
    a_hi = radius
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.min(h[neg] / (-d[neg]))))
    return a_lo, a_hi

def _line_search_exact(h, d, dx, radius):
    """Exact line search using the quadratic structure."""
    n = len(h)
    S0 = np.correlate(h, 1.0 - h, mode='full')
    T = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = np.correlate(d, d, mode='full')
    
    a_lo, a_hi = _feasible_interval(h, d, radius)
    if a_hi - a_lo < 1e-14:
        return 0.0, float(np.max(S0) * dx)
    
    # Collect candidate alphas
    candidates = [a_lo, a_hi, 0.0]
    
    # Vertices of individual quadratics
    for k in range(len(S0)):
        if Q[k] > 1e-14:
            av = T[k] / (2.0 * Q[k])
            if a_lo <= av <= a_hi:
                candidates.append(av)
    
    # Grid
    n_grid = 48
    grid = np.linspace(a_lo, a_hi, n_grid)
    candidates.extend(grid.tolist())
    
    # Pairwise intersections of top quadratics (by current value)
    top_k = min(16, len(S0))
    sorted_idx = np.argsort(-S0)[:top_k]
    for i in range(top_k):
        for j in range(i+1, top_k):
            ki, kj = int(sorted_idx[i]), int(sorted_idx[j])
            # Solve: S_i + a*T_i - a^2*Q_i = S_j + a*T_j - a^2*Q_j
            # (Q_j - Q_i)*a^2 + (T_i - T_j)*a + (S_i - S_j) = 0
            A = Q[j] - Q[i]
            B = T[i] - T[j]
            C = S[i] - S[j]
            if abs(A) < 1e-14:
                # Linear: B*a + C = 0
                if abs(B) > 1e-14:
                    ar = -C / B
                    if a_lo <= ar <= a_hi:
                        candidates.append(ar)
            else:
                disc = B*B - 4*A*C
                if disc >= 0:
                    sq = math.sqrt(disc)
                    for ar in [(-B + sq) / (2*A), (-B - sq) / (2*A)]:
                        if a_lo <= ar <= a_hi:
                            candidates.append(ar)
    
    # Evaluate all candidates
    best_val = float('inf')
    best_alpha = 0.0
    for a in candidates:
        if a < a_lo or a > a_hi:
            continue
        val = float(np.max(S0 + a * T - a * a * Q)) * dx
        if val < best_val:
            best_val = val
            best_alpha = a
    
    # Local refinement around best
    if best_val < float('inf'):
        lo_r = max(a_lo, best_alpha - 0.02 * max(a_hi - a_lo, 1e-4))
        hi_r = min(a_hi, best_alpha + 0.02 * max(a_hi - a_lo, 1e-4))
        if hi_r - lo_r > 1e-10:
            n_ref = 32
            for a in np.linspace(lo_r, hi_r, n_ref):
                val = float(np.max(S0 + a * T - a * a * Q)) * dx
                if val < best_val:
                    best_val = val
                    best_alpha = a
    
    return best_alpha, best_val

def _smooth_gradient(h, beta):
    """Compute smooth-max gradient of the C5 objective."""
    n = len(h)
    dx = 2.0 / n
    S = np.correlate(h, 1.0 - h, mode='full')
    s_scaled = S * dx
    mx = float(np.max(s_scaled))
    v = np.clip(beta * (s_scaled - mx), -80.0, 0.0)
    w = np.exp(v)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros_like(S)
        w[np.argmax(S)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: for each lag l, d/dh_i of sum_i' h(i')*(1-h(i'+l))
    # Using the row formula
    g = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[n-1+l] * dx
        if abs(wl) < 1e-15:
            continue
        if l >= 0:
            m = n - l
            if m > 0:
                # d/dh_i of sum_{p=0}^{m-1} h[p]*(1-h[p+l])
                # For coordinate i in [0, m-1]: contributes (1 - h[i+l])
                # For coordinate i in [l, n-1]: contributes -h[i-l]
                g[:m] += wl * (1.0 - h[l:l+m])
                g[l:] += -wl * h[:m]
        else:
            s = -l
            m = n - s
            if m > 0:
                # lag = -s, sum_{p=0}^{m-1} h[p+s]*(1-h[p])
                # For coordinate i in [s, n-1]: contributes (1-h[i-s])
                # For coordinate i in [0, m-1]: contributes -h[i+s]
                g[s:] += wl * (1.0 - h[:m])
                g[:m] += -wl * h[s:s+m]
    
    # Remove mean to maintain zero-sum constraint
    g -= np.mean(g)
    return -g  # negative gradient (descent direction)

def _block_direction(g, block_size):
    """Create a block-averaged direction from gradient."""
    n = len(g)
    nb = n // block_size
    if nb < 1:
        return np.zeros(n)
    d = np.zeros(n)
    for b in range(nb):
        s = b * block_size
        e = s + block_size
        if e <= n:
            d[s:e] = np.mean(g[s:e])
    # Handle remainder
    if nb * block_size < n:
        d[nb*block_size:] = np.mean(g[nb*block_size:])
    d -= np.mean(d)
    return d

def _low_freq_direction(n, rng):
    """Generate a low-frequency direction."""
    x = np.arange(n, dtype=float)
    n_harm = int(rng.integers(2, 5))
    d = np.zeros(n)
    for _ in range(n_harm):
        freq = rng.uniform(1.0, 8.0)
        phase = rng.uniform(0, 2*np.pi)
        d += np.sin(2*np.pi*freq*x/n + phase) * rng.uniform(0.5, 1.5)
    d -= np.mean(d)
    nrm = np.linalg.norm(d)
    if nrm > 1e-12:
        d /= nrm
    return d

def _make_directions(h, rng, max_dirs=6):

    n = len(h)
    dirs = []
    
    # 1. Smooth gradient (multiple betas)
    for beta in [15.0, 30.0]:
        g = _smooth_gradient(h, beta)
        nrm = np.linalg.norm(g)
        if nrm > 1e-10:
            dirs.append(g / nrm)
    
    # 2. Block-averaged gradient
    g15 = _smooth_gradient(h, 15.0)
    for bs in [8, 16]:
        d = _block_direction(g15, bs)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    
    # 3. Low-frequency random
    for _ in range(2):

            if nrm_g > 1e-10:
                d = 0.7 * d + 0.3 * (g / nrm_g)
                d -= np.mean(d)
                nrm = np.linalg.norm(d)
                if nrm > 1e-10:
                    dirs.append(d / nrm)
            else:
                dirs.append(d)
    
    # 4. Pair directions from gradient (top positive - top negative)
    g = _smooth_gradient(h, 20.0)
    if len(dirs) < max_dirs:
        idx_pos = np.argsort(g)[-3:]
        idx_neg = np.argsort(g)[:3]
        for ip in idx_pos:
            for ineg in idx_neg:
                d = np.zeros(n)
                d[ineg] = 1.0
                d[ip] = -1.0
                d -= np.mean(d)
                nrm = np.linalg.norm(d)
                if nrm > 1e-10:
                    dirs.append(d / nrm)
                if len(dirs) >= max_dirs:
                    break
            if len(dirs) >= max_dirs:
                break
    
    return dirs[:max_dirs]

def _pair_swap_optimize(h, deadline, max_iters=30000):
    """Pair swap optimization for fine-tuning."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(777)
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        hi, hj = h[i], h[j]
        if abs(hi - hj) < 1e-4:
            stall += 1
            continue
        
        # Try swapping: h[i] += delta, h[j] -= delta (or vice versa)
        # Use line search on the 2D direction
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        # Ensure the swap keeps values in [0,1]
        # h[i] + a*1 and h[j] - a*1 must be in [0,1]
        # a in [-h[j], 1-h[i]] but also we want a > 0 for i to increase
        # Actually let's just use the general line search
        alpha, c5_try = _line_search_exact(h, d, dx, 0.5)
        
        if c5_try < cur_c5 - 1e-14:
            h_new = h + alpha * d
            h_new = _project(h_new, target)
            c5_actual = _exact_c5(h_new)
            if c5_actual < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_actual
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_transfer(h, n):
    """Create a binary version by selecting top n/2 values."""
    idx = np.argsort(h)[-n//2:]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def _binary_sa(h_bin, deadline, max_iter=20000, T_init=0.003):
    """Binary simulated annealing."""
    n = len(h_bin)
    ones = list(np.where(h_bin > 0.5)[0])
    zeros = list(np.where(h_bin <= 0.5)[0])
    cur = h_bin.copy()
    cur_c5 = _exact_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(4242)
    T = T_init
    stall = 0
    
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        if stall > 150:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        h_new = cur.copy()
        h_new[oi] = 0.0
        h_new[oz] = 1.0
        new_c5 = _exact_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = h_new
            cur_c5 = new_c5
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = cur.copy()
                stall = 0
            T *= 0.999
        else:
            stall += 1
    
    return best_h, best_c5

        for _ in range(3):
            f = rng.uniform(1.0, 8.0)
            p = rng.uniform(0, 2 * np.pi)
            d += rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * f * x / n + p)
        d -= np.mean(d)
        nrm = np.linalg.norm(d)
        if nrm > 1e-10:
            dirs.append(d / nrm)
    return dirs[:nd]

def _local_opt(h, deadline, r0=0.20, stall_lim=70, nd=5):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(9999)
    rad = r0
    stall = 0
    while time.time() < deadline:
        if stall > stall_lim:
            break
        if deadline - time.time() < 0.3:
            break
        dirs = _gen_dirs(h, rng, nd)
        best_new = cur
        best_a = 0.0
        best_d = None
        for d in dirs:
            d = d - np.mean(d)
            nrm = np.linalg.norm(d)
            if nrm < 1e-12:
                continue
            a, c_try = _line_search(h, d, dx, rad)
            if c_try < best_new - 1e-14:
                best_new = c_try
                best_a = a
                best_d = d
        if best_d is not None and best_new < cur - 1e-14:
            h = h + best_a * best_d
            h = _project(h, target)
            cur = _exact_c5(h)
            if cur < best_c - 1e-14:
                best_c = cur
                best_h = h.copy()
            stall = 0
            rad = min(rad * 1.04, 0.35)
        else:
            stall += 1
            rad *= 0.75
            if rad < 0.008:
                pert = rng.normal(0, 0.05, n)
                pert -= np.mean(pert)
                hp = _project(h + pert, target)
                cp = _exact_c5(hp)
                if cp < cur - 1e-12:
                    h = hp
                    cur = cp
                    stall = 0
                    rad = 0.15
                elif stall % 30 == 0:
                    rad = 0.08
    return best_h, best_c

def _pair_opt(h, deadline, max_it=50000):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    cur = _exact_c5(h)
    best_h = h.copy()
    best_c = cur
    rng = np.random.default_rng(31337)
    stall = 0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        if stall > 250:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-5:
            stall += 1
            continue
        d = np.zeros(n)
        d[i] = 1.0
        d[j] = -1.0
        a, c5t = _line_search(h, d, dx, 1.0)
        if c5t < cur - 1e-14:
            h_new = h + a * d
            h_new = _project(h_new, target)
            c5a = _exact_c5(h_new)
            if c5a < cur - 1e-14:
                h = h_new
                cur = c5a
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    xo = (np.arange(n) + 0.5) / n
    xn = (np.arange(n_new) + 0.5) / n_new
    return _project(np.interp(xn, xo, h), n_new / 2.0)

def _to_bin(h, n):
    idx = np.argsort(h)[-n // 2:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _bin_sa(hb, deadline, mx=20000, Ti=0.002):
    n = len(hb)
    ones = list(np.where(hb > 0.5)[0])
    zeros = list(np.where(hb <= 0.5)[0])
    cur = hb.copy()
    cc = _exact_c5(cur)
    bh = cur.copy()
    bc = cc
    rng = np.random.default_rng(42)
    T = Ti
    stall = 0
    for _ in range(mx):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        if not ones or not zeros:
            break
        i = int(rng.integers(0, len(ones)))
        j = int(rng.integers(0, len(zeros)))
        oi, oz = ones[i], zeros[j]
        hn = cur.copy()
        hn[oi] = 0.0
        hn[oz] = 1.0
        nc = _exact_c5(hn)
        dl = nc - cc
        if dl < 0 or rng.random() < np.exp(-dl / max(T, 1e-12)):
            cur = hn
            cc = nc
            ones = list(np.where(cur > 0.5)[0])
            zeros = list(np.where(cur <= 0.5)[0])
            if cc < bc - 1e-14:
                bc = cc
                bh = cur.copy()
                stall = 0
            T *= 0.999
        else:
            stall += 1
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h_init = _project(h0.copy(), n0 / 2.0)
    best_h = h_init.copy()
    best_c = _exact_c5(h_init)

    # Phase 1: Quick local opt (12%)
    d1 = t0 + budget_s * 0.12
    h1, c1 = _local_opt(h_init, d1, r0=0.18, stall_lim=50, nd=5)
    if c1 < best_c:
        best_c = c1
        best_h = h1.copy()

    # Phase 2: Pair swap (10%)
    d2 = t0 + budget_s * 0.22
    h2 = _project(best_h.copy(), len(best_h) / 2.0)
    h2, c2 = _pair_opt(h2, d2, max_it=30000)
    if c2 < best_c:
        best_c = c2
        best_h = h2.copy()

    # Phase 3: Binary exploration (13%)
    d3 = t0 + budget_s * 0.35
    nb = len(best_h)
    hbin = _to_bin(best_h, nb)
    hbin, cb = _bin_sa(hbin, d3, mx=15000, Ti=0.002)
    if cb < best_c:
        best_c = cb
        best_h = hbin.copy()
    if time.time() < d3:
        hp = _project(hbin.copy(), nb / 2.0)
        sd = min(d3, time.time() + budget_s * 0.025)
        hp, cp = _local_opt(hp, sd, r0=0.10, stall_lim=35, nd=4)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Multi-resolution (20%)
    d4 = t0 + budget_s * 0.55
    nc = len(best_h)
    offsets = [-8, -4, 4, 8, -12, 12, -16, 16, -6, 6]
    tried = set()
    for dn in offsets:
        if time.time() > d4:
            break
        n_try = nc + dn
        if n_try < 60 or n_try > 260 or n_try in tried:
            continue
        tried.add(n_try)
        hr = _resample(best_h, n_try)
        sd = min(d4, time.time() + budget_s * 0.018)
        hr, cr = _local_opt(hr, sd, r0=0.15, stall_lim=35, nd=4)
        if cr < best_c + 0.004 and time.time() < d4 - 3:
            hb = _to_bin(hr, n_try)
            sd2 = min(d4, time.time() + budget_s * 0.008)
            hb, cb2 = _bin_sa(hb, sd2, mx=6000, Ti=0.001)
            if cb2 < cr:
                cr = cb2
                hr = hb
        if cr < best_c + 0.002 and time.time() < d4 - 2:
            sd3 = min(d4, time.time() + budget_s * 0.01)
            hr, cr = _local_opt(_project(hr.copy(), n_try / 2.0), sd3, r0=0.06, stall_lim=25, nd=3)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            nc = n_try

    # Phase 5: Intensive optimization (25%)
    d5 = t0 + budget_s * 0.80
    nc = len(best_h)
    for r_init in [0.12, 0.06, 0.03]:
        if time.time() > d5 - 3:
            break
        h5 = _project(best_h.copy(), nc / 2.0)
        sd = min(d5, time.time() + budget_s * 0.06)
        h5, c5 = _local_opt(h5, sd, r0=r_init, stall_lim=60, nd=5)
        if c5 < best_c:
            best_c = c5
            best_h = h5.copy()
        if time.time() < d5 - 5:
            h5 = _project(best_h.copy(), nc / 2.0)
            sd = min(d5, time.time() + budget_s * 0.03)
            h5, c5 = _pair_opt(h5, sd, max_it=15000)
            if c5 < best_c:
                best_c = c5
                best_h = h5.copy()

    # Phase 6: Basin hopping (10%)
    d6 = deadline
    scale = 0.025
    n_re = 0
    while time.time() < d6 and n_re < 60:
        n_re += 1
        nc = len(best_h)
        x = np.arange(nc, dtype=float)
        pert = np.zeros(nc)
        for _ in range(3):
            f = rng.uniform(0.5, 12.0)
            p = rng.uniform(0, 2 * np.pi)
            pert += rng.uniform(0.5, 1.2) * np.sin(2 * np.pi * f * x / nc + p)
        pert += 0.3 * scale * rng.normal(0, 1, nc)
        mxp = np.max(np.abs(pert))
        if mxp > 1e-10:
            pert *= scale / mxp
        hp = _project(best_h + pert, nc / 2.0)
        sd = min(d6, time.time() + budget_s * 0.005)
        ht, ct = _local_opt(hp, sd, r0=0.06, stall_lim=20, nd=3)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            scale = max(0.003, scale * 0.8)
        else:
            scale = min(0.08, scale * 1.1)

    # Final polish
    if deadline - time.time() > 3:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _local_opt(hf, deadline, r0=0.02, stall_lim=30, nd=3)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()
    if deadline - time.time() > 2:
        hf = _project(best_h.copy(), len(best_h) / 2.0)
        hf, cf = _pair_opt(hf, deadline, max_it=5000)
        if cf < best_c:
            best_c = cf
            best_h = hf.copy()

    best_h = _project(best_h, len(best_h) / 2.0)
    final_c = _exact_c5(best_h)
    return best_h, final_c, len(best_h)
