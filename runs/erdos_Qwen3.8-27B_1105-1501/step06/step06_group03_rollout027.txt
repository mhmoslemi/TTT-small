def _exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = float(v.min()) - 2.0, float(v.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _grad_rows(h, indices, n):
    G = np.zeros((len(indices), n))
    for idx, r in enumerate(indices):
        k = r - (n - 1)
        g = G[idx]
        if k >= 0:
            m = n - k
            if m > 0:
                g[k:] += (1.0 - h[:m])
                g[:m] += -h[k:]

            d = -k
            m = n - d
            if m > 0:
                g[:m] += (1.0 - h[d:d+m])
                g[d:] += -h[:m]
    return G

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

                            if cur < best_c5:

                                best_h = h.copy()
                                improved = True
                                break
            if not improved:
                break
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _sine_init(n, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _block_init(n, rng, n_blocks):
    h = np.zeros(n)
    if n_blocks < 2:
        return _project(h, n / 2.0)
    boundaries = sorted(rng.choice(np.arange(1, n), size=min(n_blocks - 1, n - 1), replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for b in range(len(boundaries) - 1):
        if rng.random() < 0.5:
            h[boundaries[b]:boundaries[b + 1]] = 1.0
    return _project(h, n / 2.0)

def _multi_start(h_parent, n0, deadline, rng, best_c5_ref):
    """Try diverse starting points with short optimization."""
    best_h = h_parent.copy()
    best_c = _exact_c5(h_parent)
    starts = [h_parent.copy()]
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        for ph in [0.0, 0.8, 1.6, 2.5, 3.8]:
            starts.append(_sine_init(n0, freq, ph))
    for _ in range(12):
        starts.append(_block_init(n0, rng, rng.integers(3, n0 // 5 + 1)))
    for s in starts:
        if time.time() > deadline:
            break
        c5_s = _exact_c5(s)
        if c5_s < best_c + 0.01:
            sub_dl = min(deadline, time.time() + 2.0)
            h_s, c5_s = _lp_loop(s, sub_dl, radius_init=0.10, K=14, use_smooth=False)
            if c5_s < best_c:
                best_c = c5_s
                best_h = h_s.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.99

    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 122
        h0 = _sine_init(n0, 2, 0.5)

    h0 = _project(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = _exact_c5(h0)

    # Phase 1: Multi-start (7%)
    d1 = t0 + budget_s * 0.07
    h1, c5_1 = _multi_start(best_h, n0, d1, rng, best_c5)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Main LP optimization (18%)
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _lp_loop(best_h, d2, radius_init=0.14, K=20, use_smooth=False)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # Phase 3: Binary SA (15%)
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _binary_sa(best_h, d3, rng)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()

    # Phase 4: LP after binary (12%)
    d4 = t0 + budget_s * 0.52
    h4, c5_4 = _lp_loop(best_h, d4, radius_init=0.08, K=22, use_smooth=True)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Extended resolution sweep (22%)
    d5 = t0 + budget_s * 0.74
    base_n = len(best_h)
    offsets = [0, -2, 2, -4, 4, -6, 6, -8, 8, -10, 10, -12, 12, -14, 14,
               -16, 16, -18, 18, -20, 20, -24, 24, -28, 28, -32, 32, -36, 36]
    tried = set()
    for dn in offsets:
        if time.time() > d5:
            break
        n_try = base_n + dn
        if n_try < 50 or n_try > 220 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(best_h, n_try)
        c5_r = _exact_c5(h_r)
        if c5_r < best_c5 + 0.004:
            sub_dl = min(d5, time.time() + budget_s * 0.02)
            h_r, c5_r = _lp_loop(h_r, sub_dl, radius_init=0.10, K=16, use_smooth=False)
            if c5_r < best_c5 - 1e-12 and time.time() < d5 - budget_s * 0.02:
                sub_dl2 = min(d5, time.time() + budget_s * 0.012)
                h_r2, c5_r2 = _binary_sa(h_r, sub_dl2, rng)
                if c5_r2 < c5_r:
                    h_r, c5_r = h_r2, c5_r2
                if time.time() < d5:
                    sub_dl3 = min(d5, time.time() + budget_s * 0.01)
                    h_r, c5_r = _lp_loop(h_r, sub_dl3, radius_init=0.05, K=16)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Aggressive perturbation (7%)
    d6 = t0 + budget_s * 0.81
    scale = 0.018
    for attempt in range(600):
        if time.time() > d6:
            break
        if best_c5 < 0.3803:
            break
        h_pert = best_h + rng.normal(0, scale, len(best_h))
        h_pert = _project(h_pert, len(best_h) / 2.0)
        sub_dl = min(d6, time.time() + budget_s * 0.003)
        h_t, c5_t = _lp_loop(h_pert, sub_dl, radius_init=0.05, K=14, use_smooth=True)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_t.copy()
            scale = max(0.002, scale * 0.82)
        else:
            scale = min(0.10, scale * 1.18)

    # Phase 7: Pair refinement (8%)
    d7 = t0 + budget_s * 0.89
    h7, c5_7 = _pair_refine(best_h, d7, max_passes=400)
    if c5_7 < best_c5:
        best_c5 = c5_7
        best_h = h7.copy()

    # Phase 8: Final LP polish (8%)
    d8 = deadline
    h8, c5_8 = _lp_loop(best_h, d8, radius_init=0.04, K=20, use_smooth=True)
    if c5_8 < best_c5:
        best_c5 = c5_8
        best_h = h8.copy()

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _exact_c5(best_h)
    n_final = len(best_h)

    # Safety: never return worse than parent
    parent_c5 = _exact_c5(h0)
    if final_c5 > parent_c5:
        best_h = h0.copy()
        final_c5 = parent_c5
        n_final = n0

    return best_h, final_c5, n_final
