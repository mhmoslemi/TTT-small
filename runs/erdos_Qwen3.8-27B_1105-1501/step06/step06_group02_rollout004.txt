```python
def project(v, target):

    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(80):

        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)

        else:
```

```python
def soft_gradient(h, tau):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    # Stable softmax
    shifted = (R - np.max(R)) / tau
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        m = n - abs(d)
        if m <= 0:
            continue
        if d >= 0:
            g[:m] += p[k] * (1.0 - h[d:d+m])
            g[d:d+m] += -p[k] * h[:m]
        else:
            r = -d
            g[r:] += p[k] * (1.0 - h[:m])
            g[:m] += -p[k] * h[r:r+m]
    return g
```
