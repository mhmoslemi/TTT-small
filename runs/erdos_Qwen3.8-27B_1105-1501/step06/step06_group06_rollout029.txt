def _project(v, target):
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

        else:

            x[mask] += res / cnt

def _soft_dir(h, gamma):
    """Compute soft-max weighted descent direction (sum-zero)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = corr.max()
    # Numerically stable soft-max
    shifted = gamma * (corr - M)
    # Clip to avoid underflow
    shifted = np.clip(shifted, -50.0, 0.0)
    exps = np.exp(shifted)
    sw = exps.sum()
    if sw < 1e-30:
        return None
    w = exps / sw
    
    d = np.zeros(n)
    for k in range(2 * n - 1):
        w_k = w[k]
        if w_k < 1e-12:
            continue
        s = k - (n - 1)
        m = n - abs(s)
        if s >= 0:
            d[:m] -= dx * w_k * (1.0 - h[s:s+m])
            d[s:s+m] += dx * w_k * h[:m]
        else:
            g = -s
            d[g:g+m] -= dx * w_k * (1.0 - h[:m])
            d[:m] += dx * w_k * h[g:g+m]
    
    d -= d.mean()
    return d

def _line_search(h, direction, n_max_grid=48):

        f_c2 = float(np.max(c0 + c2*L + c2*c2*Q)) * dx

        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(v - theta, 0.0, 1.0)
    res = target - float(np.sum(x))
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            x = x.copy()
            x[mask] += res / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _correlate_quad(h, d):
    """Return c0, L, Q such that c(alpha) = c0 + alpha*L + alpha^2*Q for each lag."""
    c0 = np.correlate(h, 1.0 - h, mode='full')
    L = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    Q = -np.correlate(d, d, mode='full')
    return c0, L, Q

def _line_search(h, d, n_grid=40):
    """Minimize exact C5 along h + alpha*d. Returns (alpha, best_val)."""
    n = len(h)
    dx = 2.0 / n
    c0, L, Q = _correlate_quad(h, d)
    
    pos = d > 1e-14
    neg = d < -1e-14
    a_hi = 0.4
    a_lo = -0.4
    if np.any(pos):
        a_hi = min(a_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        a_lo = max(a_lo, float(np.max(h[neg] / d[neg])))
    
    if a_hi - a_lo < 1e-13:
        a_mid = (a_lo + a_hi) / 2.0
        vals = c0 + a_mid * L + a_mid * a_mid * Q
        return a_mid, float(np.max(vals) * dx)
    
    # Grid
    grid = np.linspace(a_lo, a_hi, n_grid)
    grid = np.unique(np.concatenate([grid, [0.0]]))
    grid = grid[(grid >= a_lo - 1e-12) & (grid <= a_hi + 1e-12)]
    
    vals = c0[None, :] + grid[:, None] * L[None, :] + grid[:, None] ** 2 * Q[None, :]
    obj = np.max(vals, axis=1) * dx
    bi = int(np.argmin(obj))
    best_a = grid[bi]
    best_val = obj[bi]
    
    # Golden section refinement
    if 0 < bi < len(grid) - 1:
        a, b = grid[bi - 1], grid[bi + 1]
    else:
        a, b = a_lo, a_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    fa = float(np.max(c0 + a * L + a * a * Q)) * dx
    fb = float(np.max(c0 + b * L + b * b * Q)) * dx
    for _ in range(45):
        if b - a < 1e-14:
            break
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        fc1 = float(np.max(c0 + c1 * L + c1 * c1 * Q)) * dx
        fc2 = float(np.max(c0 + c2 * L + c2 * c2 * Q)) * dx
        if fc1 < fc2:
            b, fb = c1, fc1
        else:
            a, fa = c2, fc2
    mid = (a + b) / 2.0
    fm = float(np.max(c0 + mid * L + mid * mid * Q)) * dx
    if fm < best_val:
        best_a, best_val = mid, fm
    
    return float(best_a), float(best_val)

def _soft_dir(h, gamma):
    """Soft-max weighted descent direction (sum-zero)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = corr.max()
    shifted = np.clip(gamma * (corr - M), -40.0, 0.0)
    exps = np.exp(shifted)
    sw = exps.sum()
    if sw < 1e-30:
        return None
    w = exps / sw
    
    d = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-11:
            continue
        s = k - (n - 1)
        m = n - abs(s)
        if s >= 0:
            d[:m] -= dx * wk * (1.0 - h[s:s + m])
            d[s:s + m] += dx * wk * h[:m]
        else:
            g = -s
            d[g:g + m] -= dx * wk * (1.0 - h[:m])
            d[:m] += dx * wk * h[g:g + m]
    
    d -= d.mean()
    return d

def _soft_descent(h, deadline, max_iters=200):
    """Soft-max gradient descent with exact line search."""
    n = len(h)
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    gamma = 15.0
    stall = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        direction = _soft_dir(h, gamma)
        if direction is None or np.max(np.abs(direction)) < 1e-14:
            stall += 5
            gamma *= 1.5
            continue
        
        # Normalize direction for reasonable step size
        dnorm = np.sum(np.abs(direction))
        if dnorm < 1e-12:
            stall += 5
            continue
        direction = direction / dnorm * 0.3
        
        alpha, new_c5 = _line_search(h, direction)
        if new_c5 < cur_c5 - 1e-14:
            h = h + alpha * direction
            h = _project(h, n / 2.0)
            cur_c5 = _exact_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            stall = 0
            gamma = max(8.0, gamma * 0.97)
        else:
            stall += 1
            gamma *= 1.3
            if gamma > 50000:
                gamma = 50000
            if stall > 30:
                # Perturb to escape
                h = h + rng.normal(0, 0.01, n)
                h = _project(h, n / 2.0)
                cur_c5 = _exact_c5(h)
                gamma = 15.0
                stall = 0
            elif stall > 60:
                break
    
    return best_h, best_c5

def _pair_polish(h, deadline, max_iters=3000):
    """Pair transport optimization for fine adjustments."""
    n = len(h)
    dx = 2.0 / n
    cur_c5 = _exact_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(999)
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 200:
            break
        
        # Choose a pair based on gradient
        d_grad = _soft_dir(h, 40.0)
        if d_grad is not None:
            neg_idx = np.argsort(d_grad)[:8]
            pos_idx = np.argsort(-d_grad)[:8]
        else:
            neg_idx = rng.integers(0, n, 8)
            pos_idx = rng.integers(0, n, 8)
        
        improved = False
        # Try some pairs
        for ii in range(min(4, len(neg_idx))):
            if time.time() > deadline:
                break
            i = int(neg_idx[ii % len(neg_idx)])
            j = int(pos_idx[ii % len(pos_idx)])
            if i == j:
                continue
            # Also try random pair every few iterations
            if it % 3 == 0:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
            
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            
            alpha, new_c5 = _line_search(h, d, n_grid=24)
            if new_c5 < cur_c5 - 1e-15:
                h = h + alpha * d
                h = _project(h, n / 2.0)
                cur_c5 = _exact_c5(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1
    
    return best_h, best_c5

def _resample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_threshold(h):
    n = len(h)
    target = n // 2
    # Sort by value, take top 'target' as 1s
    order = np.argsort(-h)
    bits = np.zeros(n)
    bits[order[:target]] = 1.0
    return bits

def _random_balanced(rng, n):
    target = n // 2
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    return h

def _wave_init(n, freq, phase, amp):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return _project(h, n / 2.0)

def _struct_init(n, rng):
    h = np.zeros(n)
    target = n // 2
    # Create blocks of roughly equal size
    block_size = max(2, n // (target + 1))
    placed = 0
    pos = 0
    while placed < target and pos < n:
        b = min(block_size, n - pos, target - placed)
        h[pos:pos + b] = 1.0
        placed += b
        pos += block_size + max(1, int(rng.integers(0, block_size)))
    return _project(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    # Get initial construction
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_best = _struct_init(122, rng)
    
    n_best = len(h_best)
    h_best = _project(h_best, n_best / 2.0)
    best_c5 = _exact_c5(h_best)
    
    # Phase 1: Soft-max descent on initial (30% time)
    d1 = t0 + budget_s * 0.30
    h1, c5_1 = _soft_descent(h_best.copy(), d1, max_iters=300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        h_best = h1.copy()
        n_best = len(h1)
    
    # Phase 2: Pair polish (15% time)
    d2 = t0 + budget_s * 0.45
    h2, c5_2 = _pair_polish(h_best.copy(), d2, max_iters=2000)
    if c5_2 < best_c5:
        best_c5 = c5_2
        h_best = h2.copy()
        n_best = len(h2)
    
    # Phase 3: Multi-resolution search (30% time)
    d3 = t0 + budget_s * 0.75
    offsets = [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12, 14, -14, 16, -16]
    tried = set()
    for dn in offsets:
        if time.time() > d3:
            break
        n_try = n_best + dn
        if n_try < 60 or n_try > 200 or n_try % 2 != 0:
            continue
        if n_try in tried:
            continue
        tried.add(n_try)
        h_r = _resample(h_best, n_try)
        # Soft descent on resampled
        sub_dl = min(d3, time.time() + budget_s * 0.015)
        h_r, c5_r = _soft_descent(h_r, sub_dl, max_iters=100)
        if c5_r < d3:
            pass
        # Pair polish
        if time.time() < d3 - 20:
            sub_dl2 = min(d3, time.time() + budget_s * 0.012)
            h_r, c5_r = _pair_polish(h_r, sub_dl2, max_iters=800)
        if c5_r < best_c5 - 1e-12:
            best_c5 = c5_r
            h_best = h_r.copy()
            n_best = n_try
    
    # Phase 4: Diverse starts (10% time)
    d4 = t0 + budget_s * 0.87
    starts = []
    # Binary threshold of current best
    starts.append(_binary_threshold(h_best))
    # Wave starts
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.35, 0.45]:
            starts.append(_wave_init(n_best, freq, 0.0, amp))
            starts.append(_wave_init(n_best, freq, np.pi / 2, amp))
    # Random balanced
    for _ in range(3):
        starts.append(_random_balanced(rng, n_best))
    # Structured
    for _ in range(2):
        starts.append(_struct_init(n_best, rng))
    
    for s in starts:
        if time.time() > d4:
            break
        s = _project(s, n_best / 2.0)
        s_c5 = _exact_c5(s)
        if s_c5 > best_c5 + 0.1:
            continue  # Skip obviously bad starts
        sub_dl = min(d4, time.time() + budget_s * 0.01)
        h_s, c5_s = _soft_descent(s, sub_dl, max_iters=80)
        if c5_s < best_c5:
            best_c5 = c5_s
            h_best = h_s.copy()
            n_best = len(h_s)
    
    # Phase 5: Final polish (13% time)
    d5 = deadline
    # Soft descent
    sub_dl = min(d5, time.time() + budget_s * 0.05)
    h5, c5_5 = _soft_descent(h_best.copy(), sub_dl, max_iters=200)
    if c5_5 < best_c5:
        best_c5 = c5_5
        h_best = h5.copy()
        n_best = len(h5)
    
    # Pair polish
    if time.time() < d5 - 5:
        h5b, c5_5b = _pair_polish(h_best.copy(), d5, max_iters=2000)
        if c5_5b < best_c5:
            best_c5 = c5_5b
            h_best = h5b.copy()
            n_best = len(h5b)
    
    # Final validation
    h_best = _project(h_best, n_best // 2 * 2 / 2.0)
    h_best = _project(h_best, len(h_best) / 2.0)
    final_c5 = _exact_c5(h_best)
    best_c5 = min(best_c5, final_c5)
    
    return h_best, best_c5, len(h_best)
