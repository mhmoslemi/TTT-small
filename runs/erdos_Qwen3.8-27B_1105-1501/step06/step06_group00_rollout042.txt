import numpy as np

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(corr):
    return float(np.max(corr))

def _project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = h.copy()
    ones = np.sum(h > 0.5)
    target = n // 2
    if ones > target:
        idx = np.where(h > 0.5)[0]
        np.random.default_rng(0).shuffle(idx)
        h[idx[:ones-target]] = 0.0
    elif ones < target:
        idx = np.where(h <= 0.5)[0]
        np.random.default_rng(0).shuffle(idx)
        h[idx[:target-ones]] = 1.0
    return h

def _swap_delta(corr, h, i, j):
    """Compute delta to corr when swapping h[i]=1 to 0 and h[j]=0 to 1.
    Returns the delta array (same shape as corr)."""
    n = len(h)
    delta = np.zeros(2*n - 1)
    # For position p with change dp:
    # For k=1..n-1:
    #   delta[n-1+k] += dp*(1-h[p-k])*[k<=p] - dp*h[p+k]*[k<=n-1-p]

def _local_search_binary(h, deadline, max_steps=200000, seed=42):

            break
        i = ones[int(rng.integers(0, len(ones)))]
        j = zeros[int(rng.integers(0, len(zeros)))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_corr = _corr(h)
        new_max = _c5_val(new_corr)
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_max = new_max
            corr = new_corr
            ones = list(np.where(h > 0.5)[0])
            zeros = list(np.where(h <= 0.5)[0])
            if cur_max * dx < best_c5:
                best_c5 = cur_max * dx
                best_h = h.copy()
                stall = 0
            T *= T_decay
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        if step % 1000 == 0 and stall > 3000:
            T = max(T * 2.0, 0.002)  # reheat
            stall = 0
        step = getattr(_sa_binary, '_step', 0) + 1
        _sa_binary._step = step
    return best_h, best_c5

def _resample_binary(h, n_new):
    """Resample binary h to new length n_new, maintaining n_new/2 ones."""
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_cont = np.interp(x_new, x_old, h)
    # Threshold to binary with exactly n_new/2 ones
    target = n_new // 2
    idx = np.argsort(h_cont)[-target:]
    h_new = np.zeros(n_new)
    h_new[idx] = 1.0
    return h_new

def _targeted_search(h, deadline, max_rounds=2000, seed=77):
    """Targeted search: focus on reducing the max lags."""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    h = h.copy()
    corr = _corr(h)
    cur_max = _c5_val(corr)
    best_c5 = cur_max * dx
    best_h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if stall > 200:
            break
        # Find top lags
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n - 1)
        # Find positions that contribute to this lag
        if max_lag > 0:
            bad = [i for i in range(n - max_lag) if h[i+max_lag] > 0.5 and h[i] <= 0.5]
            # These are positions where shifted h has 1 but h has 0
            # To reduce: move a 1 to one of these positions from elsewhere
            good_sources = [i for i in ones if i not in range(n - max_lag) or h[i+max_lag] > 0.5]
        else:
            k = -max_lag
            bad = [i for i in range(k, n) if h[i] > 0.5 and h[i-k] <= 0.5]
            good_sources = list(ones)
        
        improved = False
        # Try moving ones from good_sources to bad positions
        n_try = min(len(bad) * max(1, len(good_sources)), 100)
        for _ in range(n_try):
            if not bad or not good_sources:
                break
            bi = bad[int(rng.integers(0, len(bad)))]
            si = good_sources[int(rng.integers(0, len(good_sources)))]
            if h[bi] > 0.5 or h[si] <= 0.5:
                continue
            # Swap si (1) with bi (0)
            h[si] = 0.0
            h[bi] = 1.0
            new_corr = _corr(h)
            new_max = _c5_val(new_corr)
            if new_max < cur_max - 1e-14:
                cur_max = new_max
                corr = new_corr
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                if cur_max * dx < best_c5:
                    best_c5 = cur_max * dx
                    best_h = h.copy()
                improved = True
                bad = []  # recompute next round
                good_sources = []
                break
            else:
                h[si] = 1.0
                h[bi] = 0.0
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def _multi_n_search(h_start, n_start, deadline, rng):
    """Search over multiple n values."""
    best_h = h_start.copy()
    best_c5 = _c5_val(_corr(h_start)) * (2.0 / len(h_start))
    
    # Try nearby n values
    n_candidates = list(range(max(80, n_start - 20), min(252, n_start + 20) + 1, 2))
    rng.shuffle(n_candidates)
    
    for n_try in n_candidates:
        if time.time() > deadline - 30:
            break
        if n_try == n_start or n_try < 60 or n_try % 2 != 0:
            continue
        h_r = _resample_binary(h_start, n_try)
        sub_dl = min(deadline, time.time() + 25)
        h_r, c5_r = _local_search_binary(h_r, sub_dl, max_steps=50000, seed=int(rng.integers(0, 10000)))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    return best_h, best_c5

def _intensive_polish(h, deadline, seed=999):
    """Intensive local search with multiple rounds."""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.default_rng(seed)
    h = h.copy()
    corr = _corr(h)
    cur_max = _c5_val(corr)
    best_c5 = cur_max * dx
    best_h = h.copy()
    
    for round_num in range(10):
        if time.time() > deadline - 5:
            break
        sub_dl = min(deadline - 5, time.time() + (deadline - time.time()) / (10 - round_num))
        h, cur_max2 = _local_search_binary(h, sub_dl, max_steps=100000, seed=seed + round_num)
        new_c5 = _c5_val(_corr(h)) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            cur_max = new_c5 / dx
        else:
            # Try SA to escape
            if time.time() < deadline - 10:
                h_sa, c5_sa = _sa_binary(h, min(deadline - 10, time.time() + 20), T_init=0.003, seed=seed + 100 + round_num)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    
    # Start from initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    # Make binary
    h_bin = np.zeros(n0)
    target_ones = n0 // 2
    idx_top = np.argsort(h0)[-target_ones:]
    h_bin[idx_top] = 1.0
    
    best_h = h_bin.copy()
    best_c5 = _c5_val(_corr(h_bin)) * (2.0 / n0)
    
    # Phase 1: Initial local search (15% of budget)
    d1 = t0 + budget_s * 0.15
    h1, c5_1 = _local_search_binary(best_h, d1, max_steps=80000, seed=1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Phase 2: Targeted search (10%)
    d2 = t0 + budget_s * 0.25
    h2, c5_2 = _targeted_search(best_h, d2, max_rounds=3000, seed=2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # Phase 3: SA (15%)
    d3 = t0 + budget_s * 0.40
    h3, c5_3 = _sa_binary(best_h, d3, T_init=0.004, T_decay=0.9997, seed=3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Polish after SA
    if time.time() < d3:
        h3b, c5_3b = _local_search_binary(best_h, d3, max_steps=30000, seed=4)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()
    
    # Phase 4: Multi-n exploration (30%)
    d4 = t0 + budget_s * 0.70
    h4, c5_4 = _multi_n_search(best_h, len(best_h), d4, rng)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()
    
    # Phase 5: Intensive polish on best (15%)
    d5 = t0 + budget_s * 0.85
    h5, c5_5 = _intensive_polish(best_h, d5, seed=5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()
    
    # Phase 6: Final targeted + local search (10%)
    d6 = deadline - 2
    h6, c5_6 = _targeted_search(best_h, d6, max_rounds=1000, seed=6)
    if c5_6 < best_c5:
        best_c5 = c5_6
        best_h = h6.copy()
    if time.time() < deadline - 3:
        h6b, c5_6b = _local_search_binary(best_h, deadline - 2, max_steps=50000, seed=7)
        if c5_6b < best_c5:
            best_c5 = c5_6b
            best_h = h6b.copy()
    
    # Final verification
    n_final = len(best_h)
    final_c5 = _c5_val(_corr(best_h)) * (2.0 / n_final)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(best_h > 0.5))
    if n_ones != n_final // 2:
        best_h = _project_binary(best_h, n_final)
        final_c5 = _c5_val(_corr(best_h)) * (2.0 / n_final)
    
    return best_h, final_c5, n_final
