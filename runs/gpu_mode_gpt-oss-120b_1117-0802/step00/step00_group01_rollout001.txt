    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Linear projections

    hidden_stride = left.stride(3)

"""
TriMul forward (outgoing) implementation with a Triton kernel.
The heavy N³ reduction over the sequence dimension is performed by a
custom Triton matmul kernel that computes, for each batch and hidden
channel, the matrix product:
    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
All other operations (LayerNorm, Linear projections, gating, final
projection) are done in PyTorch. The kernel uses a 2‑D tiling (BLOCK_M×BLOCK_N)
and loops over K in blocks (BLOCK_K). The batch and hidden dimensions are
flattened into the third program ID.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul over the sequence dimension (N³).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_h,
    out_stride_b, out_stride_i, out_stride_j, out_stride_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)                # tile over i
    pid_n = tl.program_id(1)                # tile over j
    pid_z = tl.program_id(2)                # combined batch * hidden index

    # derive batch index b and hidden index h from pid_z
    b = pid_z // H
    h = pid_z % H

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_b  + h * stride_h
    right_base = right_ptr + b * stride_b  + h * stride_h
    out_base   = out_ptr   + b * out_stride_b + h * out_stride_h

    # offsets inside the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for block_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = block_k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)  -> left[i, k]
        left_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right block: shape (BLOCK_K, BLOCK_N)  -> right[k, j] = right[j, k]
        right_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # fused multiply‑add
        acc += tl.dot(left, right)

    # store the result tile
    out_ptrs = out_base + offs_i[:, None] * out_stride_i + offs_j[None, :] * out_stride_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version) with Triton acceleration.

    Args:
        data: Tuple[input_tensor, mask, weights_dict, config_dict]
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask         : torch.Tensor of shape [B, N, N] (float or bool)
            - weights      : dict of weight tensors
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight         = weights["norm.weight"]
    norm_bias           = weights["norm.bias"]
    left_proj_weight    = weights["left_proj.weight"]
    right_proj_weight   = weights["right_proj.weight"]
    left_gate_weight    = weights["left_gate.weight"]
    right_gate_weight   = weights["right_gate.weight"]
    out_gate_weight     = weights["out_gate.weight"]
    to_out_norm_weight  = weights["to_out_norm.weight"]
    to_out_norm_bias    = weights["to_out_norm.bias"]
    to_out_weight       = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is None:
        mask_ = torch.ones_like(left, dtype=dtype)
    else:
        mask_ = mask.unsqueeze(-1).to(dtype)      # [B, N, N, 1]
    left  = left  * mask_
    right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Triton kernel: compute the N³ reduction (matrix multiplication)
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # strides (number of elements, already in contiguous layout)
    stride_b = left.stride(0)   # B stride = N*N*H
    stride_i = left.stride(1)   # i stride = N*H
    stride_k = left.stride(2)   # k stride = H
    stride_h = left.stride(3)   # channel stride = 1

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_h = out.stride(3)

    # Tiling parameters (tuned for typical GPU SMs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left, right, out,
        stride_b, stride_i, stride_k, stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # back to dim channels

    return output
