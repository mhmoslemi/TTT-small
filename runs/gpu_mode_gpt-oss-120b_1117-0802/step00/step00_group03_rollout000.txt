def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']
    # weights dict includes tensors for each param, as given.

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])

    # Apply mask if provided (nomask maybe). mask shape B,N,N.
    mask = mask.unsqueeze(-1)  # B,N,N,1
    left = left * mask
    right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute core contraction via Triton kernel
    # left and right are B,N,N,hidden_dim (float32)
    # Convert to contiguous if needed
    left = left.contiguous()
    right = right.contiguous()
    # allocate output
    out = torch.empty_like(left)   # shape B,N,N,hidden_dim
    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(left.shape[1], BLOCK_M),
            triton.cdiv(left.shape[2], BLOCK_N),
            left.shape[0] * hidden_dim)  # B*D

    # compute strides
    stride_lb0 = left.stride(0)
    stride_lb1 = left.stride(1)
    stride_lb2 = left.stride(2)
    stride_lb3 = left.stride(3)
    stride_rb0 = right.stride(0)
    stride_rb1 = right.stride(1)
    stride_rb2 = right.stride(2)
    stride_rb3 = right.stride(3)
    stride_ob0 = out.stride(0)
    stride_ob1 = out.stride(1)
    stride_ob2 = out.stride(2)
    stride_ob3 = out.stride(3)

    tri_mul_kernel[grid](
        left, right, out,
        left.shape[0], left.shape[1], hidden_dim,
        stride_lb0, stride_lb1, stride_lb2, stride_lb3,
        stride_rb0, stride_rb1, stride_rb2, stride_rb3,
        stride_ob0, stride_ob1, stride_ob2, stride_ob3,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # After kernel compute out (still shape B,N,N,hidden_dim)
    # Now out_norm and final projection
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

    out = out * out_gate  # broadcast out_gate shape B,N,N,hidden_dim

    # Final linear projection
    out = torch.nn.functional.linear(out, weights['to_out.weight'])

    return out

"""
TriMul forward pass (outgoing version) with a Triton kernel.

The heavy part of TriMul is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication
    left_d[b] @ right_d[b].T  for each hidden channel d.
The Triton kernel below performs this per‑batch‑channel GEMM using
tiling on the sequence dimension (N), while the surrounding
normalizations, linear projections and gating are done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for each (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float32  (B, N, N, D)  left tensor
    right_ptr,         # *float32  (B, N, N, D)  right tensor
    out_ptr,           # *float32  (B, N, N, D)  output tensor
    batch_size: tl.int32,
    seqlen: tl.int32,
    hidden_dim: tl.int32,
    # strides for left
    stride_lb0: tl.int32, stride_lb1: tl.int32,
    stride_lb2: tl.int32, stride_lb3: tl.int32,
    # strides for right
    stride_rb0: tl.int32, stride_rb1: tl.int32,
    stride_rb2: tl.int32, stride_rb3: tl.int32,
    # strides for out
    stride_ob0: tl.int32, stride_ob1: tl.int32,
    stride_ob2: tl.int32, stride_ob3: tl.int32,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_bd = tl.program_id(2)  # combined batch & hidden index

    # decode batch and hidden indices
    b = pid_bd // hidden_dim
    d = pid_bd % hidden_dim

    # coordinate offsets for the M and N tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for M and N (handle edge blocks)
    mask_m = offs_m < seqlen
    mask_n = offs_n < seqlen

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the summed index)
    num_blocks_k = tl.cdiv(seqlen, BLOCK_K)
    for k in range(0, num_blocks_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < seqlen

        #--- left tile (M x K) -------------------------------------------------
        # address = b*stride_lb0 + d*stride_lb3 + i*stride_lb1 + k*stride_lb2
        left_ptrs = left_ptr + \
            b * stride_lb0 + d * stride_lb3 + \
            offs_m[:, None] * stride_lb1 + offs_k[None, :] * stride_lb2
        mask_a = mask_m[:, None] & mask_k[None, :]
        a = tl.load(left_ptrs, mask=mask_a, other=0.0).to(tl.float32)

        #--- right tile (K x N) ------------------------------------------------
        # we need right transposed: right[b, j, k, d] -> (k, j)
        # address = b*stride_rb0 + d*stride_rb3 + k*stride_rb2 + j*stride_rb1
        right_ptrs = right_ptr + \
            b * stride_rb0 + d * stride_rb3 + \
            offs_k[:, None] * stride_rb2 + offs_n[None, :] * stride_rb1
        mask_b = mask_k[:, None] & mask_n[None, :]
        b_tile = tl.load(right_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b_tile)

    # store the result
    out_ptrs = out_ptr + \
        b * stride_ob0 + d * stride_ob3 + \
        offs_m[:, None] * stride_ob1 + offs_n[None, :] * stride_ob2
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul module (outgoing version).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] float32
            - mask:        [B, N, N] (bool or float) – will be broadcasted
            - weights: dict with all linear / layer‑norm parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # D

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim = C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,D]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if any). mask shape [B, N, N] -> [B,N,N,1]
    # ------------------------------------------------------------------
    mask_exp = mask.to(left.dtype).unsqueeze(-1)   # broadcast over D
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,D]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,D]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton: out = left @ right^T over K dimension
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # [B,N,N,D]

    # tile sizes – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # strides (int64 -> Python int)
    stride_lb0 = left.stride(0)
    stride_lb1 = left.stride(1)
    stride_lb2 = left.stride(2)
    stride_lb3 = left.stride(3)

    stride_rb0 = right.stride(0)
    stride_rb1 = right.stride(1)
    stride_rb2 = right.stride(2)
    stride_rb3 = right.stride(3)

    stride_ob0 = out.stride(0)
    stride_ob1 = out.stride(1)
    stride_ob2 = out.stride(2)
    stride_ob3 = out.stride(3)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_lb0, stride_lb1, stride_lb2, stride_lb3,
        stride_rb0, stride_rb1, stride_rb2, stride_rb3,
        stride_ob0, stride_ob1, stride_ob2, stride_ob3,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: normalization, out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
