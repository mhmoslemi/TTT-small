    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Get weights
    norm_weight, norm_bias = weights['norm.weight'], weights['norm.bias']
    left_proj_weight = weights['left_proj.weight'] # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']
    # We'll work in float32, but optionally cast to fp16 for compute
    # 1. LayerNorm
    # Compute mean and var per [bs,N,N] across dim dim
    # Use torch.nn.functional.layer_norm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # 2. Linear projections (no bias)
    # Use torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape output to (bs, N, N, hidden_dim) by inverse permute
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Continue with to_out_norm, gating and final linear
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    out = out.to(torch.float32)
    return out

"""
TriMul (outgoing) forward pass optimized with a Triton matrix‑multiply kernel.

Algorithm:
1. Layer‑norm the input tensor over the feature dimension.
2. Linear projections → left/right tensors (shape [B,N,N,H]).
3. Apply optional mask (broadcast on the hidden dimension).
4. Compute sigmoid gates (left/right/out) and apply them to left/right.
5. Contract:   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply of shape (B*H, N, N) @ (B*H, N, N)^T.
   Implemented as a custom Triton kernel for speed.
6. Layer‑norm on the hidden dimension, multiply by out‑gate, and a final linear
   projection back to the original feature dimension.

All operations except the contraction are done with PyTorch; the heavy N³
contraction is accelerated by Triton. The kernel uses float32 accumulation for
numerical stability and works for arbitrary batch size (≤2) and hidden size.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_az, stride_am, stride_ak,          # batch, row, col strides for A
    stride_bz, stride_bk, stride_bn,          # batch, row(k), col(n) strides for B (B is accessed transposed)
    stride_cz, stride_cm, stride_cn,          # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # batch (combined B * hidden)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = A_ptr + pid_z * stride_az
    b_batch_ptr = B_ptr + pid_z * stride_bz
    c_batch_ptr = C_ptr + pid_z * stride_cz

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K x N) – B is accessed as transposed (k,n)
        b = tl.load(
            b_batch_ptr
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # promote to FP32 for the dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    tl.store(
        c_batch_ptr
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Contraction via Triton (batched matmul of left @ right^T)
    # ------------------------------------------------------------------
    # Prepare tensors for batched matmul: reshape hidden dim into batch
    left = left.contiguous()
    right = right.contiguous()

    # Bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)      # [BH, N, N]
    right_mat = right_perm.view(batch_hidden, N, N)    # [BH, N, N]
    out_mat = torch.empty_like(left_mat)               # result [BH, N, N]

    # Triton kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
        batch_hidden,                  # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                 # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),     # A batch, row, col
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1), # B batch, row(k), col (j) – transposed access
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),       # C batch, row, col
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.float()
```
