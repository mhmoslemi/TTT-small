@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride, a_row_stride, a_col_stride,
    b_row_stride, b_col_stride,
    c_row_stride, c_col_stride,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)  # batch dimension
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col

    # Determine offsets
    # Compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute pointer base for this batch
    a_ptrs = a_ptr + pid_b * batch_stride + offs_m[:, None] * a_row_stride + tl.arange(0, BLOCK_K)[None, :] * a_col_stride
    b_ptrs = b_ptr + pid_b * batch_stride + offs_n[None, :] * b_col_stride + tl.arange(0, BLOCK_K)[:, None] * b_row_stride
    # Actually need careful: B shape is (batch, K, N). Since we want B (K,N). The b_ptr input should be B matrix where first dimension is K rows, second is N columns (since we want A: MxK and B: KxN). Our B_T after permuting right shape (batch, K, N). So b_ptr stride for row K (first dimension) = b_row_stride, column N = b_col_stride.

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # maybe keep in fp32

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A block (M x BLOCK_K)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        # Load B block (BLOCK_K x N)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :], other=0.0)

        # compute dot: (M x BLOCK_K) @ (BLOCK_K x N) => (M x N)
        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # Write back
    c_ptrs = c_ptr + pid_b * batch_stride + offs_m[:, None] * c_row_stride + offs_n[None, :] * c_col_stride
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,

):

    pid_m = tl.program_id(0)  # row tile
    pid_n = tl.program_id(1)  # col tile

    # Compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k

        # Use offsets for loads

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a Triton kernel for the
core O(N³) matrix multiplication.
The kernel computes, for each batch and hidden channel:

    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]

which is equivalent to a batched GEMM: left @ rightᵀ.
All other operations (LayerNorm, linear projections, gating) are done
with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication C = A @ B where:
        A: (batch, M, K)
        B: (batch, K, N)
        C: (batch, M, N)
    All tensors are stored in row‑major order.
    """
    pid_b = tl.program_id(2)        # batch * hidden dimension
    pid_m = tl.program_id(0)        # tile row
    pid_n = tl.program_id(1)        # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile (k = 0)
    a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + cur_k
        valid_k = k_range < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & valid_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=valid_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        # Advance pointers to next K‑block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write result
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (TriMul) “outgoing” operation.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor – [B, N, N, dim] torch.Tensor
            mask         – [B, N, N] torch.Tensor (float or bool)
            weights      – dict of model parameters (all torch.Tensors)
            config       – dict with keys "dim" and "hidden_dim"
    Returns:
        output_tensor – [B, N, N, dim] torch.Tensor (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 0. Grab dimensions and weights
    # ------------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Ensure everything is on the right device and dtype
    def to_tensor(t):
        return t.to(device=device, dtype=dtype)

    norm_weight = to_tensor(weights["norm.weight"])
    norm_bias = to_tensor(weights["norm.bias"])
    left_proj_weight = to_tensor(weights["left_proj.weight"])
    right_proj_weight = to_tensor(weights["right_proj.weight"])
    left_gate_weight = to_tensor(weights["left_gate.weight"])
    right_gate_weight = to_tensor(weights["right_gate.weight"])
    out_gate_weight = to_tensor(weights["out_gate.weight"])
    to_out_norm_weight = to_tensor(weights["to_out_norm.weight"])
    to_out_norm_bias = to_tensor(weights["to_out_norm.bias"])
    to_out_weight = to_tensor(weights["to_out.weight"])

    # Cast inputs
    x = input_tensor.to(dtype)
    mask = mask.to(dtype)  # 0/1 (or bool -> float)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and masking
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Batched GEMM: out = left @ rightᵀ   (performed in Triton)
    # ------------------------------------------------------------------
    # rightᵀ for the Einstein summation
    right_T = right.permute(0, 2, 1, 3).contiguous()      # [B,N,N,hidden] -> (k,j)

    # Bring hidden dimension next to batch and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # [B,hidden,N,N]
    right_perm = right_T.permute(0, 3, 1, 2).contiguous() # [B,hidden,N,N]

    # Merge batch & hidden into a single batch dimension
    batch_hidden = B * hidden_dim
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)
    stride_b_n = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Choose a reasonable block size (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_hidden,
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=1e-5
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    # Ensure final dtype is float32 as required
    out = out.to(torch.float32)
    return out
