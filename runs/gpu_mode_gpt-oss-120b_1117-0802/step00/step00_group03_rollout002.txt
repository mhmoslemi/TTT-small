Define kernel:

```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)  # combined b,d index
    pid_y = tl.program_id(1)  # j tile
    pid_x = tl.program_id(0)  # i tile

    # compute b,d from pid_z
    d = pid_z % D
    b = pid_z // D

    # compute tile offsets
    i_offset = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    # create mask for out-of-range
    i_mask = i_offset < N
    j_mask = j_offset < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # compute pointer offsets
        # left b,i,k,d
        left_ptrs = left_ptr + b * stride_left_b + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k + d * stride_left_d
        # right b,j,k,d (will treat as B[k,j] due to dot expects k as first dim)
        right_ptrs = right_ptr + b * stride_right_b + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j + d * stride_right_d

        # load tiles (masked)
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_ptr + b * stride_out_b + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j + d * stride_out_d
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])
```

"""
TriMul forward kernel (outgoing) using Triton.

Algorithm:
1. Layer‑normalize the input tensor over the feature dimension.
2. Linear projections for left/right tensors and three sigmoid gates.
3. Apply mask (if provided) and element‑wise gating.
4. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]   (tri‑multiplicative update)
   Implemented as many independent GEMMs (one per hidden dimension) in a
   fused Triton kernel (tiling over i, j and looping over k).
5. Layer‑normalize the intermediate result, multiply by the output gate,
   and apply a final linear projection back to the original feature dimension.
All operations are performed in float32 (the kernel loads float32 and
accumulates in float32) and the final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_i = tl.program_id(0)          # tile along i (seq_len)
    pid_j = tl.program_id(1)          # tile along j (seq_len)
    pid_z = tl.program_id(2)          # combined (batch, hidden)

    # decode batch and hidden index from pid_z
    d = pid_z % D
    b = pid_z // D

    # tile offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # pointers for left[b,i,k,d]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_k
            + d * stride_left_d
        )
        # pointers for right[b,j,k,d] accessed as (k, j) for dot
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_offset[:, None] * stride_right_k
            + j_offset[None, :] * stride_right_j
            + d * stride_right_d
        )

        left_tile = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # accumulate product over k
        acc += tl.dot(left_tile, right_tile)

    # write out
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
        + d * stride_out_d
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] torch.Tensor (float32)
            - mask_tensor:  [B, N, N] torch.Tensor (float32 or bool)
            - weights_dict: mapping of weight names to torch.Tensor
            - config_dict: {"dim": int, "hidden_dim": int, ...}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Gating mechanisms
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. TriMul contraction via Triton (out_tmp shape: [B, N, N, hidden_dim])
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out_tmp = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_tmp.stride()

    # tuning parameters (adjusted for typical B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim                    # combined batch‑hidden dimension
    )

    tri_mul_kernel[grid](
        left, right, out_tmp,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out_tmp,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
