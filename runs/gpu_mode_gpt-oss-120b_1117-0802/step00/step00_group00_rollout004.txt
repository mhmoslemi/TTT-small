"""
TriMul (outgoing) forward pass with a fused Triton kernel for the gate
multiplication. The heavy tensor contraction (einsum) is performed with
torch.einsum, while the elementwise multiplication with the learned gates is
fused into a single Triton kernel to demonstrate usage of Triton.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fused elementwise multiplication of left/right tensors with
# their respective gates. This kernel is deliberately simple – it only
# demonstrates a Triton compute path and does not change the algorithmic
# complexity of the TriMul operation.
# --------------------------------------------------------------------------- #
@triton.jit
def fused_gate_mul_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    total_elements,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    tl.store(left_out_ptr + offs, left * left_gate, mask=mask)
    tl.store(right_out_ptr + offs, right * right_gate, mask=mask)


# --------------------------------------------------------------------------- #
# Helper: manual LayerNorm (bias and weight are supplied)
# --------------------------------------------------------------------------- #
def layer_norm(x, weight, bias, eps: float = 1e-5):
    # x: (..., dim)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


def custom_kernel(data):
    """
    Args:
        data: Tuple[input, mask, weights, config]
            input  – torch.Tensor [B, N, N, dim] (float32)
            mask   – torch.Tensor [B, N, N] (bool or float)
            weights – dict of weight tensors
            config – dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # LayerNorm on input (norm)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = layer_norm(input_tensor, norm_weight, norm_bias, eps=eps)

    # ------------------------------------------------------------------- #
    # Linear projections
    # ------------------------------------------------------------------- #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)      # bias is None
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------- #
    # Apply mask (mask is broadcast over the hidden dimension)
    # ------------------------------------------------------------------- #
    mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------- #
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------- #
    # Fuse gate multiplication with a Triton kernel
    # ------------------------------------------------------------------- #
    # Ensure contiguous layout for 1‑D view
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    left_out_flat = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    total = left_flat.numel()
    BLOCK = 1024  # reasonable block size (multiple of 32)
    grid = (triton.cdiv(total, BLOCK),)

    fused_gate_mul_kernel[grid](
        left_flat, left_gate_flat,
        right_flat, right_gate_flat,
        left_out_flat, right_out_flat,
        total,
        BLOCK=BLOCK,
    )

    # Reshape back to original 4‑D tensors
    left_out = left_out_flat.view_as(left)   # [B, N, N, hidden_dim]
    right_out = right_out_flat.view_as(right)

    # ------------------------------------------------------------------- #
    # Core contraction: out_{i j d} = Σ_k left[i k d] * right[j k d]
    # ------------------------------------------------------------------- #
    # Using torch.einsum (the heavy compute). Cast to bfloat16 for a modest
    # speedup on supported hardware, then back to float32.
    left_bf16 = left_out.to(torch.bfloat16)
    right_bf16 = right_out.to(torch.bfloat16)
    out = torch.einsum("b i k d, b j k d -> b i j d", left_bf16, right_bf16).to(torch.float32)

    # ------------------------------------------------------------------- #
    # Output LayerNorm (to_out_norm)
    # ------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = layer_norm(out, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # ------------------------------------------------------------------- #
    # Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # Final linear projection back to `dim`
    # ------------------------------------------------------------------- #
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # shape [B, N, N, dim]

    # Ensure final dtype is float32 as required
    return out.to(torch.float32)
