def tri_mul_kernel(L_ptr, R_ptr, Out_ptr,
                   batch, N, hidden_dim,
                   stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                   stride_r_b, stride_r_j, stride_r_k, stride_r_h,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    b = pid_z // hidden_dim
    h = pid_z % hidden_dim

    # Compute offsets for i and j
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Create range vectors
    range_i = i_offset + tl.arange(0, BLOCK_M)
    range_j = j_offset + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_i = range_i < N
    mask_j = range_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        range_k = k_offset + tl.arange(0, BLOCK_K)
        # Load L block (i,k)
        L_ptrs = (L_ptr
                  + b * stride_l_b
                  + range_i[:, None] * stride_l_i
                  + range_k[None, :] * stride_l_k
                  + h * stride_l_h)
        # Note: we can't broadcast range_i across row dimension? Actually range_i[:, None] is shape (BLOCK_M,1), range_k[None,:] shape (1,BLOCK_K). This yields shape (BLOCK_M,BLOCK_K) as needed.
        L = tl.load(L_ptrs, mask=mask_i[:, None] & (range_k < N), other=0.0)

        # Load R block transposed (k,j). Need pointer with k first then j.
        R_ptrs = (R_ptr
                  + b * stride_r_b
                  + range_k[:, None] * stride_r_k
                  + range_j[None, :] * stride_r_j
                  + h * stride_r_h)
        R = tl.load(R_ptrs, mask=(range_k < N)[:, None] & mask_j[None, :], other=0.0)

        # Compute dot
        acc += tl.dot(L, R)  # shape (BLOCK_M, BLOCK_N)

    # Write back
    # Output pointer offsets
    O_ptrs = (Out_ptr

"""
TriMul "outgoing" implementation with Triton fusion.

The heavy part of TriMul is the contraction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a batched matrix multiplication of left and the
transpose of right for each hidden channel d. This kernel tiles the i and
j dimensions (BLOCK_M × BLOCK_N) and accumulates over k in blocks
(BLOCK_K). One kernel instance is launched per (batch, hidden_channel)
pair, so the third grid dimension is batch * hidden_dim.
All other operations (LayerNorm, linear projections, gating, masking) are
performed in PyTorch; only the O(N³) contraction is accelerated.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------- #
# Triton kernel for the contraction (left @ right.T) per hidden channel
# ------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    L_ptr, R_ptr, O_ptr,                    # pointers
    B, N, H,                                 # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)   # block index for i
    pid_j = tl.program_id(1)   # block index for j
    pid_z = tl.program_id(2)   # combined batch * hidden

    # decode batch and hidden channel from pid_z
    b = pid_z // H
    h = pid_z % H

    # starting offsets for this block
    i_off = pid_i * BLOCK_M
    j_off = pid_j * BLOCK_N

    # ranges for i, j
    range_i = i_off + tl.arange(0, BLOCK_M)
    range_j = j_off + tl.arange(0, BLOCK_N)

    # bounds masks for i and j
    mask_i = range_i < N
    mask_j = range_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_off = kk * BLOCK_K
        range_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = range_k < N

        # load left block (i, k)
        L_ptrs = (L_ptr +
                  b * stride_l_b +
                  range_i[:, None] * stride_l_i +
                  range_k[None, :] * stride_l_k +
                  h * stride_l_h)
        mask_L = mask_i[:, None] & mask_k[None, :]
        L = tl.load(L_ptrs, mask=mask_L, other=0.0)

        # load right block transposed (k, j)
        R_ptrs = (R_ptr +
                  b * stride_r_b +
                  range_k[:, None] * stride_r_k +
                  range_j[None, :] * stride_r_j +
                  h * stride_r_h)
        mask_R = mask_k[:, None] & mask_j[None, :]
        R = tl.load(R_ptrs, mask=mask_R, other=0.0)

        # accumulate dot product over k
        acc += tl.dot(L, R)   # (BLOCK_M, BLOCK_N)

    # store the result
    O_ptrs = (O_ptr +
              b * stride_o_b +
              range_i[:, None] * stride_o_i +
              range_j[None, :] * stride_o_j +
              h * stride_o_h)
    mask_O = mask_i[:, None] & mask_j[None, :]
    tl.store(O_ptrs, acc, mask=mask_O)


# ------------------------------------------------------------------- #
# Entry point
# ------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [bs, seqlen, seqlen, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # Optional mask (mask broadcasted to last dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Triton contraction: out = left @ right^T per hidden channel
    # -----------------------------------------------------------------
    out = torch.empty_like(left)  # same shape & dtype (float32)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_i = triton.cdiv(seq_len, BLOCK_M)
    grid_j = triton.cdiv(seq_len, BLOCK_N)
    grid_z = bs * hidden_dim
    grid = (grid_i, grid_j, grid_z)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Output LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
