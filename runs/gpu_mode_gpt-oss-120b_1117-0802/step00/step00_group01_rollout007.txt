    Y_ptr,
    weight_ptr,
    bias_ptr,
    epsilon,
    M, C,
    stride_xm, stride_xc,
    stride_ym, stride_yc,
    stride_w, stride_b,
    BLOCK_C: tl.constexpr

    row = pid  # each program computes the row "m"

    # Compute pointer offsets for the row base
    x_ptr_row = X_ptr + row * stride_xm
    y_ptr_row = Y_ptr + row * stride_ym

    sum_val = tl.zeros((1,), dtype=tl.float32)

        mask = cur_offs_c < C

"""
TriMul (outgoing) forward pass with Triton‑accelerated LayerNorm.
The heavy pairwise reduction is performed with a batched GEMM (torch.bmm)
while both LayerNorms (input and output) are fused into a custom
Triton kernel. All arithmetic is carried out in float16 and the final
output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension.
# Input  : (M, C)   where M = batch * seq_len * seq_len
# Output : (M, C)   normalized and scaled with weight & bias.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    in_ptr, out_ptr,          # pointers
    weight_ptr, bias_ptr,     # LayerNorm parameters
    M, C,                     # logical sizes
    stride_in_m, stride_in_c,  # strides for the input tensor
    stride_out_m, stride_out_c,  # strides for the output tensor
    stride_w, stride_b,        # strides for weight & bias (usually 1)
    epsilon,                  # small constant (float)
    BLOCK_C: tl.constexpr,    # tile size for the C dimension
):
    pid = tl.program_id(0)
    if pid >= M:
        return

    # Base pointers for this row
    row_in = in_ptr + pid * stride_in_m
    row_out = out_ptr + pid * stride_out_m

    # ---------- mean & variance (float32 accumulator) ----------
    sum_val = tl.zeros((1,), dtype=tl.float32)
    sum_sq = tl.zeros((1,), dtype=tl.float32)

    for c_start in range(0, C, BLOCK_C):
        c_off = c_start + tl.arange(0, BLOCK_C)
        mask = c_off < C
        x = tl.load(row_in + c_off * stride_in_c, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ---------- normalize, scale & shift ----------
    for c_start in range(0, C, BLOCK_C):
        c_off = c_start + tl.arange(0, BLOCK_C)
        mask = c_off < C
        x = tl.load(row_in + c_off * stride_in_c, mask=mask, other=0.0).to(tl.float32)
        norm = (x - mean) * inv_std
        w = tl.load(weight_ptr + c_off * stride_w, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + c_off * stride_b, mask=mask, other=0.0).to(tl.float32)
        y = norm * w + b
        tl.store(row_out + c_off * stride_out_c, y.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Helper: Triton‑based LayerNorm (float16 input/output)
# ----------------------------------------------------------------------
def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    x:    (B, N, N, C)  float16
    weight, bias: (C,)    float16
    returns tensor of same shape and dtype as x.
    """
    B, N, _, C = x.shape
    M = B * N * N
    x_flat = x.view(M, C)
    y_flat = torch.empty_like(x_flat)

    # strides for the flattened view (row‑major)
    stride_xm, stride_xc = x_flat.stride()
    stride_ym, stride_yc = y_flat.stride()
    stride_w = weight.stride(0)
    stride_b = bias.stride(0)

    BLOCK_C = 64  # good for C up to 384

    grid = (triton.cdiv(M, 1),)  # one program per output row
    layernorm_kernel[grid](
        x_flat, y_flat,
        weight, bias,
        M, C,
        stride_xm, stride_xc,
        stride_ym, stride_yc,
        stride_w, stride_b,
        eps,
        BLOCK_C=BLOCK_C,
    )
    return y_flat.view(B, N, N, C)


# ----------------------------------------------------------------------
# Main entry point expected by the framework
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) implementation.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of parameter tensors (float32)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        output: torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Cast everything to float16 for the heavy path
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, device=device)
    norm_bias = weights["norm.bias"].to(torch.float16, device=device)
    x_norm = triton_layernorm(x, norm_weight, norm_bias, eps)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)

    left = F.linear(x_norm, left_proj_w)   # shape [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Apply mask (if present)
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask_f = torch.ones_like(mask, dtype=torch.float16, device=device)
    else:
        mask_f = mask.to(torch.float16, device=device)
    mask_f = mask_f.unsqueeze(-1)  # [B, N, N, 1]

    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gating values
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate = F.linear(x_norm, left_gate_w).sigmoid()
    right_gate = F.linear(x_norm, right_gate_w).sigmoid()
    out_gate = F.linear(x_norm, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction: batched GEMM over the sequence dimension
    #    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) and use bmm
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # BMM with transpose on the second operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)
    # bring back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (Triton)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)

    out_norm = triton_layernorm(out, to_out_norm_w, to_out_norm_b, eps)

    # ------------------------------------------------------------------
    # 8) Apply output gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate          # gate broadcasted over hidden dim

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    final_out = F.linear(out_norm, to_out_w)   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 9) Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return final_out.to(torch.float32)
