def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  N,
                  stride_left_batch, stride_left_i, stride_left_k,
                  stride_right_batch, stride_right_j, stride_right_k,
                  stride_out_batch, stride_out_i, stride_out_j,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    i_off = pid_i * BLOCK_M
    j_off = pid_j * BLOCK_N
    bh = pid_bh

    # offsets
    # compute end indices for masks
    i_idx = i_off + tl.arange(0, BLOCK_M)
    j_idx = j_off + tl.arange(0, BLOCK_N)

    # mask for i/j out-of-range
    i_mask = i_idx < N
    j_mask = j_idx < N

    # base pointers for each bh
    left_base = left_ptr + bh * stride_left_batch
    right_base = right_ptr + bh * stride_right_batch
    out_base = out_ptr + bh * stride_out_batch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k_start in range(0, N, BLOCK_K):
        # get k indices
        k_idx = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # load left tile (M x K)
        left_ptrs = left_base + i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # load right tile transposed (K x N)
        # right_base hold original layout (j,k). We need k as row, j as col.
        right_ptrs = right_base + k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_j
        right_T = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # dot = left (M,K) * right_T (K,N) -> (M,N)
        # use tl.dot
        acc += tl.dot(left, right_T)

    # store to output
    out_ptrs = out_base + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    bh = pid_bh

    # Index ranges
    i = i_offset + tl.arange(0, BLOCK_M)
    j = j_offset + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # Masks for boundaries
    i_mask = i < N
    j_mask = j < N

    # Base pointers for this bm
    left_base = left_ptr + bh * stride_left_batch
    right_base = right_ptr + bh * stride_right_batch
    out_base = out_ptr + bh * stride_out_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start + k
        k_mask = k_offset < N

        # Load left tile (M x K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0, dtype=tl.float16)
        left = tl.cast(left, tl.float32)

        # Load right tile transposed (K x N)
        # We need to load right[j,k] but transposed: (k,j)
        right_ptrs = right_base + k_offset[:, None] * stride_right_k + j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0, dtype=tl.float16)
        right = tl.cast(right, tl.float32)

        # Compute partial result
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm weight
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Apply layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for kernel
    # Cast to fp16 for efficiency
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N) and make contiguous
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape  # now left_perm shape [B, H, N, N]
    # Flatten batch and hidden
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)  # shape [BH,N,N] float32

    # Compute strides
    stride_left_batch = left_mat.stride(0)
    stride_left_i = left_mat.stride(1)
    stride_left_k = left_mat.stride(2)
    stride_right_batch = right_mat.stride(0)
    stride_right_j = right_mat.stride(1)
    stride_right_k = right_mat.stride(2)
    stride_out_batch = out_mat.stride(0)
    stride_out_i = out_mat.stride(1)
    stride_out_j = out_mat.stride(2)

    # Triton launch config
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_j, stride_right_k,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, H, N, N)
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,H

    # to_out_norm (layernorm on hidden_dim)
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Multiply by out_gate (shape [B,N,N,hidden_dim])
    out_hidden = out_hidden * out_gate

    # Final linear
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out

"""
Triangular Multiplicative Update (outgoing) kernel.

The heavy O(N³·H) pairwise reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented in Triton as a batched GEMM:
for each (b,h) we compute  out_h = left_h @ right_hᵀ
where left_h and right_h are N×N matrices.
All other operations (LayerNorm, Linear, gating, masking) are done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,            # const __fp16*
    right_ptr,           # const __fp16*
    out_ptr,             # float*
    N,                   # sequence length (int)
    # strides for left: (batch, i, k)
    stride_left_batch, stride_left_i, stride_left_k,
    # strides for right (batch, j, k) – we load transposed (k, j)
    stride_right_batch, stride_right_j, stride_right_k,
    # strides for output (batch, i, j)
    stride_out_batch, stride_out_i, stride_out_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)          # flattened batch * hidden index

    # start offsets of this tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    bh = pid_bh

    # index vectors
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # bounds masks
    i_mask = i < N
    j_mask = j < N

    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr + bh * stride_left_batch
    right_base = right_ptr + bh * stride_right_batch
    out_base   = out_ptr + bh * stride_out_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        k0 = k_start + k
        k_mask = k0 < N

        # left tile: shape (BLOCK_M, cur_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k0[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0,
                       dtype=tl.float16)
        left = tl.cast(left, tl.float32)

        # right tile transposed: we need right[k, j] (original is right[j, k])
        right_ptrs = right_base + k0[:, None] * stride_right_k + j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0,
                        dtype=tl.float16)
        right = tl.cast(right, tl.float32)

        # GEMM on this tile
        acc += tl.dot(left, right)

    # write back the block
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel
    for the O(N³·H) reduction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ---------- Optional mask ----------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(dtype)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for Triton ----------
    # use half‑precision for the heavy multiplication
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # shape -> (B, H, N, N) and make contiguous
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_h.shape
    BH = B * H

    # view as a batch of matrices: (BH, N, N)
    left_mat = left_h.view(BH, N, N)
    right_mat = right_h.view(BH, N, N)

    # output buffer (float32)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    s_l_batch, s_l_i, s_l_k = left_mat.stride()
    s_r_batch, s_r_j, s_r_k = right_mat.stride()
    s_o_batch, s_o_i, s_o_j = out_mat.stride()

    # Triton launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )
    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        s_l_batch,
        s_l_i,
        s_l_k,
        s_r_batch,
        s_r_j,
        s_r_k,
        s_o_batch,
        s_o_i,
        s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output normalisation ----------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # Gating after normalisation
    out_hidden = out_hidden * out_gate

    # ---------- Final linear ----------
    out = F.linear(
        out_hidden,
        weights["to_out.weight"].to(dtype),
    )  # shape [B, N, N, dim]

    return out
