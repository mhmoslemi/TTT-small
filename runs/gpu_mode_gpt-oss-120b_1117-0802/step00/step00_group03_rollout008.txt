def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype # Could be float32
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Normalize input
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    eps = 1e-5
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Apply mask if needed
    if not nomask:
        # mask shape [B,N,N]; unsqueeze to [B,N,N,1] then cast to same dtype
        mask_ = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Core contraction: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # Implement via Triton kernel.
    B, N, _, _ = left.shape
    # Allocate out
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]
    # Get strides
    # left shape [B,N,N,D]; compute strides
    left_stride_b = left.stride(0)
    left_stride_i = left.stride(1)
    left_stride_k = left.stride(2)
    left_stride_d = left.stride(3)
    # right shape [B,N,N,D]; same strides
    right_stride_b = right.stride(0)
    right_stride_i = right.stride(1)
    right_stride_k = right.stride(2)
    right_stride_d = right.stride(3)
    # out shape same
    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_d = out.stride(3)
    # Determine block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # Ensure BLOCK_M and BLOCK_N not exceed N
    # Launch grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*hidden_dim)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    # Launch kernel
    tri_mul_gemm_kernel[grid](
        left, right, out,
        B, hidden_dim, N, N, N,
        left_stride_b, left_stride_d, left_stride_i, left_stride_k,
        right_stride_b, right_stride_d, right_stride_i, right_stride_k,
        out_stride_b, out_stride_d, out_stride_i, out_stride_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # After kernel out shape: same as left
    # Apply output normalization and gating
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

   sequence dimension for each head (d). Implemented in Triton:
   - Each program processes a tile of size BLOCK_M×BLOCK_N of the
     (i,j) plane for a specific (batch,head) pair.
   - K‑dimension (the inner sum) is tiled with BLOCK_K.
   - Loads are masked for out‑of‑bounds entries.
5. Layer‑norm the result over the hidden dimension, multiply by out_gate,
   and project back to the original dimension.

All heavy arithmetic (step 4) is performed in the Triton kernel; the
remaining operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Dict, Tuple

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # const float*
    right_ptr,         # const float*
    out_ptr,           # float*
    B, D, M, N, K,     # int64 scalars
    stride_left_batch, stride_left_head, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_head, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_head, stride_out_row, stride_out_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # tile row index (i)
    pid_n = tl.program_id(1)   # tile col index (j)
    pid_h = tl.program_id(2)   # combined batch*head index

    # Resolve batch and head from the combined index
    batch_idx = pid_h // D
    head_idx = pid_h % D

    # Compute absolute positions for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (left_ptr
                  + batch_idx * stride_left_batch
                  + head_idx * stride_left_head
                  + offs_i[:, None] * stride_left_row
                  + offs_k[None, :] * stride_left_col)
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # Load a tile from right (transposed view): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (right_ptr
                  + batch_idx * stride_right_batch
                  + head_idx * stride_right_head
                  + offs_k[:, None] * stride_right_row
                  + offs_j[None, :] * stride_right_col)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # Convert to float32 (if needed) and accumulate
        acc += tl.dot(a, b)

    # Store the computed tile to the output tensor
    c_ptrs = (out_ptr
              + batch_idx * stride_out_batch
              + head_idx * stride_out_head
              + offs_i[:, None] * stride_out_row
              + offs_j[None, :] * stride_out_col)
    tl.store(c_ptrs,
             acc,
             mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    O(N³) contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ----- LayerNorm on the input -----
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ----- Linear projections -----
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ----- Optional mask -----
    if not nomask:
        mask_ = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ----- Gating -----
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, D = left.shape  # D should equal hidden_dim

    # ----- Triton contraction (batched GEMM) -----
    out = torch.empty_like(left)  # shape [B, N, N, D]

    # Strides for the tensors (in elements, not bytes)
    l_str0, l_str1, l_str2, l_str3 = left.stride()
    r_str0, r_str1, r_str2, r_str3 = right.stride()
    o_str0, o_str1, o_str2, o_str3 = out.stride()

    # Parameters for the kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_gemm_kernel[grid](
        left, right, out,
        B, hidden_dim, N, N, N,
        l_str0, l_str3, l_str1, l_str2,
        r_str0, r_str3, r_str2, r_str1,
        o_str0, o_str3, o_str1, o_str2,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing -----
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    out_norm = out_norm * out_gate

    to_out_weight = weights["to_out.weight"].to(device)
    final_out = F.linear(out_norm, to_out_weight)  # [B, N, N, dim]

    return final_out
