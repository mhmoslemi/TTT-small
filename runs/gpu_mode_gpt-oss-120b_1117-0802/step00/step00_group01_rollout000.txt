def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # Linear projections (use torch.nn.functional.linear)
    left_proj_weight = weights["left_proj.weight"]   # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # weight for to_out_norm and to_out
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # left, right
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    
    # mask
    if mask is None:
        # assume all ones? but mask is provided as Tensor; if shape matches then use.
        mask_exp = torch.ones_like(left[..., :1])  # broadcast.
    else:
        mask_exp = mask.unsqueeze(-1)   # [B,N,N,1]
    
    # Apply mask and gates

    right = right * mask_exp * right_gate
    
    # Cast to half for kernel

"""
TriMul forward (outgoing) implementation.

The module consists of:
1. LayerNorm on the input (dim = C).
2. Two linear projections (left/right) and three gating linear layers.
3. Mask and gate application.
4. A batch of matrix multiplications:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is computed as a batched GEMM of A @ Bᵀ for each hidden dimension d.
   This reduction is performed by a custom Triton kernel using float16
   operands and float32 accumulation.
5. Final LayerNorm, gating, and linear projection back to dim.

Only the computationally intensive reduction (step 4) lives in Triton,
the rest uses efficient PyTorch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _matmul_T_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bj,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many batches.
    A : [BATCH, M, K]   (float16)
    B : [BATCH, N, K]   (float16)   interpreted as transposed inside the kernel
    C : [BATCH, M, N]   (float32)
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_b = tl.program_id(2)          # batch index

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_batch_off = pid_b * stride_Ab
    b_batch_off = pid_b * stride_Bb
    c_batch_off = pid_b * stride_Cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K  # shape (BLOCK_K,)

        # Load A tile (M x K)
        a_ptrs = (
            A_ptr
            + a_batch_off
            + offs_m[:, None] * stride_Am
            + offs_k[None, :] * stride_Ak
        )
        a = tl.load(a_ptrs, mask=k_mask[None, :], other=tl.float16(0.0))

        # Load B tile transposed (K x N) -> original B is (N, K)
        b_ptrs = (
            B_ptr
            + b_batch_off
            + offs_k[:, None] * stride_Bk   # k dimension
            + offs_n[None, :] * stride_Bj   # n (original j) dimension
        )
        b = tl.load(b_ptrs, mask=k_mask[:, None], other=tl.float16(0.0))

        # Convert to FP32 for the dot product
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back with boundary masks
    c_ptrs = (
        C_ptr
        + c_batch_off
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)

    # Gating values
    # ----------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()    # (B,N,N,hidden_dim)

    # ----------------------------------------------------------------------
    # Optional mask handling
    # ----------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.to(input_tensor.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Cast to half-precision for the reduction kernel
    # ----------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # Bring hidden dimension to the second axis and fuse with batch
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_total = B * hidden_dim

    left_flat = left.reshape(batch_total, N, N)    # (B*H, N, N)
    right_flat = right.reshape(batch_total, N, N)

    # Output buffer (float32 accumulator)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ----------------------------------------------------------------------
    # Stride extraction (element strides, not bytes)
    # ----------------------------------------------------------------------
    stride_A_batch, stride_A_m, stride_A_k = left_flat.stride()
    stride_B_batch, stride_B_j, stride_B_k = right_flat.stride()
    stride_C_batch, stride_C_m, stride_C_n = out_flat.stride()

    # ----------------------------------------------------------------------
    # Triton launch configuration
    # ----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),     # rows
        triton.cdiv(N, BLOCK_N),     # cols
        batch_total,                 # batch dimension
    )

    _matmul_T_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_total,
        N,
        N,
        N,
        stride_A_batch,
        stride_A_m,
        stride_A_k,
        stride_B_batch,
        stride_B_k,   # k‑stride (original column stride)
        stride_B_j,   # j‑stride (original row stride)
        stride_C_batch,
        stride_C_m,
        stride_C_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim)
    # ----------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # Final LayerNorm, gating, and output projection
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate                                   # (B,N,N,hidden_dim)
    to_out_weight = weights["to_out.weight"]               # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)                     # (B,N,N,dim)

    return out
```
